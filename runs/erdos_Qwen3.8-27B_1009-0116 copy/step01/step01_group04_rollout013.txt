def corr_gradient(h, k_offset):
    """Gradient of c[k_offset] w.r.t. h, where k_offset is the index in the full correlation output."""
    n = len(h)
    k = k_offset - (n - 1)  # actual lag
    grad = np.zeros(n)
    # Term 1: (1-h[n-k]) for valid n where 0 <= n < n and 0 <= n+k < n
    # h[n+k] contributes (1-h[n])
    # For i = n+k (the index in h from the first factor):
    #   valid when 0 <= i < n and 0 <= i-k < n
    lo = max(0, k)
    hi = min(n, n + k)
    if hi > lo:
        grad[lo:hi] += 1.0 - h[lo-k:hi-k]
    # Term 2: -h[i+k] for valid n where n=i
    # (1-h[n]) contributes -h[n+k]
    # For i = n (the index from the second factor):
    #   valid when 0 <= i < n and 0 <= i+k < n
    lo2 = max(0, -k)
    hi2 = min(n, n - k)
    if hi2 > lo2:
        grad[lo2:hi2] -= h[lo2+k:hi2+k]
    return grad

def compute_c5(h, dx):

    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if abs(s - tgt) > 1e-6:
        # Binary search for shift to fix sum
        lo, hi = -2.0, 2.0
        for _ in range(50):
            mid = (lo + hi) / 2
            y = np.clip(x - mid, 0.0, 1.0)
            if np.sum(y) > tgt:
                lo = mid
            else:
                hi = mid
        x = np.clip(x - (lo + hi) / 2, 0.0, 1.0)
    return x

def corr_and_weights(h, tau):
    """Compute correlation and softmax weights."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    shifted = (corr - corr.max()) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    w /= w_sum
    return corr, w

def compute_grad(h, w):
    """Analytical gradient of logsumexp(corr/tau) w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    for k_idx in range(len(w)):
        wi = w[k_idx]
        if wi < 1e-12:
            continue
        k = k_idx - (n - 1)
        # First term: (1-h[i-k]) for max(0,k) <= i < min(n, n+k)
        lo = max(0, k)
        hi = min(n, n + k)
        if hi > lo:
            grad[lo:hi] += wi * (1.0 - h[lo-k:hi-k])
        # Second term: -h[i+k] for max(0,-k) <= i < min(n, n-k)
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if hi2 > lo2:
            grad[lo2:hi2] -= wi * h[lo2+k:hi2+k]
    return grad

def lse_objective(x, tau, dx):
    """Smooth objective: dx * tau * logsumexp(corr/tau)"""
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    logsum = m + tau * np.log(np.sum(np.exp((corr - m) / tau)))
    return dx * logsum

def lse_gradient(x, tau, dx):
    """Gradient of the smooth objective."""
    corr, w = corr_and_weights(x, tau)
    return dx * compute_grad(x, w)

        return lse_objective(x, tau, dx)
    
    def jac(x):
        return lse_gradient(x, tau, dx)
    
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    
    def constraint_jac(x):
        return np.ones(n)
    
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    try:
