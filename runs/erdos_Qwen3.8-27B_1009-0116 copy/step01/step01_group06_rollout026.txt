def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-12:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-12:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-12:
            if d2 > 0:
                cands = np.where(h < 1.0)[0]
                if len(cands) > 0:
                    idx = cands[np.argmax(1.0 - h[cands])]
                    h[idx] = min(1.0, h[idx] + d2)
            else:
                cands = np.where(h > 0.0)[0]
                if len(cands) > 0:
                    idx = cands[np.argmin(h[cands])]
                    h[idx] = max(0.0, h[idx] + d2)
    return h

def lsm_obj_grad(x, dx, beta):
    n = len(x)
    h = np.clip(x, 0.0, 1.0)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    bc = beta * corr
    mx = np.max(bc)
    eb = np.exp(bc - mx)
    se = eb.sum()
    obj = (mx + np.log(se)) / beta
    w = eb / se
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        lo = max(0, -lag)
        hi = min(n, n - lag)
        for j in range(lo, hi):
            grad[j] += wk * (1.0 - h[j + lag])
            grad[j] -= wk * h[j]
    grad *= dx * beta
    return obj, grad

    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    res = minimize(lsm_obj_grad, h0.copy(), args=(dx, beta), jac=True,
                   method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq',
                                 'fun': lambda x: x.sum() - target,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    ho = project_feasible(res.x, n)
    return ho, eval_c5(ho, dx)
