def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub

                break
    return h

import numpy as np
import time


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    return h


def _c5(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / len(h)))


def _opt_i(h, i):
    old = h[i]
    h[i] = 0.0
    c0 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = 1.0
    c1 = np.correlate(h, 1.0 - h, mode='full')
    h[i] = old
    A = c1 - c0
    B = c0
    msk = np.where(np.abs(A) > 1e-12)[0]
    if len(msk) == 0:
        return old
    am = A[msk]
    bm = B[msk]
    cands = [0.0, 1.0]
    kp = len(am)
    for p in range(kp):
        for q in range(p + 1, kp):
            d = am[p] - am[q]
            if abs(d) > 1e-12:
                x = (bm[q] - bm[p]) / d
                if 0.0 < x < 1.0:
                    cands.append(x)
    bx = old
    bf = np.inf
    for x in cands:
        x = float(np.clip(x, 0.0, 1.0))
        f = np.max(am * x + bm)
        if f < bf:
            bf = f
            bx = x
    return bx


def _cd(h, passes):
    n = len(h)
    for _ in range(passes):
        chg = False
        for i in range(n):
            nv = _opt_i(h, i)
            if abs(nv - h[i]) > 1e-10:
                h[i] = nv
                chg = True
        if not chg:
            break


def _cd_rep(h, rounds):
    n = len(h)
    for _ in range(rounds):
        _cd(h, 5)
        h = _repair(h, n)
    _cd(h, 10)
    return _repair(h, n)


def _bin_init(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h


def _sin_init(n, rng):
    freq = float(rng.choice([1.5, 2.0, 2.5, 3.0, 4.0, 5.0, 6.0, 7.0]))
    ph = rng.uniform(0.0, 6.2832)
    t = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + 0.45 * np.sin(np.pi * freq * t + ph) + 0.05 * rng.randn(n)
    return _repair(h, n)
