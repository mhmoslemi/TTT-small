def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
            # Second pass
            diff2 = target - h.sum()
            if abs(diff2) > 1e-12:
                if diff2 > 0:
                    mask2 = h < 1.0
                else:
                    mask2 = h > 0.0
                if mask2.sum() > 0:
                    h[mask2] += diff2 / mask2.sum()
                    h = np.clip(h, 0.0, 1.0)
    return h

def lse_objective(h, t):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    return (cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t) * (2.0 / len(h))

            grad[mm:] -= w[k] * h[:n-mm]
    return grad * dx

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n * 0.5
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coordinate_descent(h, n, max_sweeps=30, tol=1e-10):
    h = h.copy()
    best = compute_c5(h)
    dx = 2.0 / n
    
    for sweep in range(max_sweeps):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            # Compute g_k = ∂C[k]/∂h[i]
            c = np.correlate(h, 1.0 - h, mode='full')
            g = np.zeros(2*n-1)
            m_neg = i  # for m>=0: i-m in [0,n-1] → m in [0,i], loop m=0..i, h[i-m]
            # Actually let me compute g directly
            for m in range(-(n-1), n):
                k = m + n - 1
                # ∂C[k]/∂h[i] = (1-h[i-m])*[valid1] - h[i+m]*[valid2]
                val = 0.0
                if 0 <= i-m < n and (m >= 0 and i >= m or m < 0 and i < n+m):
                    # Simplified: 0 <= i-m < n
                    val += (1.0 - h[i-m])
                if 0 <= i+m < n:
                    val -= h[i+m]
                g[k] = val
            
            def f1d(delta):
                h_test = h.copy()
                h_test[i] = np.clip(h[i] + delta, 0.0, 1.0)
                c_test = np.correlate(h_test, 1.0 - h_test, mode='full')
                return float(np.max(c_test) * dx)
            
            lo = -h[i]
            hi = 1.0 - h[i]
            res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded', 
                                 options={'xatol': 1e-7})
            new_val = h[i] + res.x
            new_val = np.clip(new_val, 0.0, 1.0)
            old_val = h[i]
            h[i] = new_val
            
            if abs(new_val - old_val) > 1e-8:
                # Repair sum
                h = project_feasible(h, n)
                val = compute_c5(h)
                if val < best - tol:
                    best = val

Hmm wait, the coordinate descent as I wrote it is very slow because for each coordinate I'm doing a full correlate and a full minimize_scalar with many evaluations of correlate. For n=376, each correlate is O(n²) = 140k operations, and minimize_scalar does ~30 evaluations, so per coordinate it's ~4M operations, times 376 coordinates = 1.5B per sweep. Way too slow.

I need to be smarter. Let me precompute the correlation and update it incrementally when one coordinate changes.

When h[i] changes by δ:
C[k] changes by δ * g_k where g_k = ∂C[k]/∂h[i]

So I can:
1. Precompute C = correlate(h, 1-h)
2. For coordinate i, compute g = ∂C/∂h[i] (this is O(n))
3. The new C if h[i] changes to h[i]+δ is: C + δ*g (element-wise)
4. The new max is max(C + δ*g)
5. Find δ that minimizes max(C + δ*g) over [0, 1-h[i]] ∪ [0-h[i], 0]... actually over [-h[i], 1-h[i]]
6. Update C in-place

This makes each coordinate update O(n) instead of O(n²)!

ΔC[k] = δ * [(1-h[i-m])*[valid1] - h[i+m]*[valid2]]

This is exactly δ * g_i[k] where g_i is computed with the current h (before the change). So the update is exact!
