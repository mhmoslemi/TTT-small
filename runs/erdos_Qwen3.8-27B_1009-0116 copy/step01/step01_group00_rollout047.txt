    n = len(h)

        grad[:n-s] += (1.0 - h[s:])

        # i in [s, n)
        grad[s:] -= h[:n-s]
    else:
        # i in [0, n+s)
        grad[:n+s] -= h[-s:]
    return grad

from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _grad_smooth_max(h, alpha):
    """Gradient of logsumexp(alpha * corr) / alpha w.r.t. h"""
    n = len(h)
    corr = _corr(h)
    # Softmax weights
    shifted = alpha * (corr - corr.max())
    weights = np.exp(shifted)
    weights /= weights.sum()
    # Compute weighted gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-10:
            continue
        s = k - (n - 1)
        w = weights[k]
        # Term 1: (1-h[i+s]) for i where 0<=i+s<n
        if s >= 0:
            grad[:n-s] += w * (1.0 - h[s:n])
        else:
            grad[-s:] += w * (1.0 - h[0:n+s])
        # Term 2: -h[i-s] for i where 0<=i-s<n
        if s >= 0:
            grad[s:] -= w * h[0:n-s]
        else:
            grad[:n+s] -= w * h[-s:]
    return grad

import numpy as np
from scipy.optimize import minimize
import time


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)


def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)


def _grad_topk(h, top_k):
    n = len(h)
    corr = _corr(h)
    idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in idx:
        s = k - (n - 1)
        if s >= 0:
            grad[:n - s] += (1.0 - h[s:n])
            grad[s:] -= h[:n - s]
        else:
            grad[-s:] += (1.0 - h[0:n + s])
            grad[:n + s] -= h[-s:]
    return grad / top_k


def _grad_smooth(h, alpha):
    n = len(h)
    corr = _corr(h)
    m = corr.max()
    log_w = alpha * (corr - m)
    log_w = np.clip(log_w, -400, 400)
    exp_w = np.exp(log_w - log_w.max())
    w = exp_w / exp_w.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        wk = w[k]
        if s >= 0:
            grad[:n - s] += wk * (1.0 - h[s:n])
            grad[s:] -= wk * h[:n - s]
        else:
            grad[-s:] += wk * (1.0 - h[0:n + s])
            grad[:n + s] -= wk * h[-s:]
    return grad


def _slsqp_smooth(x, alpha):
    corr = _corr(x)
    m = corr.max()
    s = np.clip(alpha * (corr - m), -300, 300)
    return m + np.log(np.sum(np.exp(s)) - np.exp(s.max()) + 1) / alpha + 0.5 * s.max() / alpha


def _subgrad_loop(h, n, dx, target, t_start, budget, rng, n_iter=3000):
    h = h.copy()
    best = _c5(h, dx)
    step = 0.008
    for it in range(n_iter):
        if time.time() - t_start > budget - 200:
            break
        progress = it / n_iter
        top_k = 3 if progress < 0.3 else (5 if progress < 0.7 else 8)
        grad = _grad_topk(h, top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        h_trial = h - step * grad
        h_trial = _project(h_trial, target)
        c5_t = _c5(h_trial, dx)
        if c5_t < best - 1e-12:
            h = h_trial
            best = c5_t
            if it % 100 == 0:
                step = min(step * 1.03, 0.03)
        else:
            step *= 0.85
            if step < 1e-6:
                step = 0.001
    return h, best


def _slsqp_polish(h, n, target, alpha, maxiter, t_start, budget):
    if time.time() - t_start > budget - 30:
        return h, _c5(h, 2.0 / n)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def eq_con(x):
        return np.sum(x) - target

    cons = [{'type': 'eq', 'fun': eq_con}]
    try:
        res = minimize(
            lambda x: _slsqp_smooth(x, alpha), h,
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        hc = _project(res.x, target)
        return hc, _c5(hc, dx)
    except Exception:
        return h, _c5(h, dx)


def _local_search(h, n, dx, target, t_start, budget, rng):
    h = h.copy()
    best = _c5(h, dx)
    deltas = np.concatenate([np.array([0.001, 0.002, 0.005, 0.01, 0.02, 0.04]),
                             -np.array([0.001, 0.002, 0.005, 0.01, 0.02, 0.04])])
    stall = 0
    while time.time() - t_start < budget - 20:
        improved_any = False
        for _ in range(n):
            i = rng.randint(n)
            js = rng.choice(n, size=min(n, 20), replace=False)
            for j in js:
                if i == j:
                    continue
                hi, hj = h[i], h[j]
                for d in deltas:
                    ni = hi + d
                    nj = hj - d
                    if ni < -1e-12 or ni > 1.0 + 1e-12:
                        continue
                    if nj < -1e-12 or nj > 1.0 + 1e-12:
                        continue
                    h[i] = np.clip(ni, 0.0, 1.0)
                    h[j] = np.clip(nj, 0.0, 1.0)
                    c5_t = _c5(h, dx)
                    if c5_t < best - 1e-11:
                        best = c5_t
                        improved_any = True
                        stall = 0
                    else:
                        h[i] = hi
                        h[j] = hj
            if stall > 200:
                break
        if not improved_any:
            stall += 1
        if stall > 500:
            break
    return h, best
