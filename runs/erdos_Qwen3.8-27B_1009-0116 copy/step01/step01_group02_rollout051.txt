import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(compute_corr(h, n))) * dx

def smooth_max_val(corr, beta):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def all_corr_grads(h, N):
    grads = np.zeros((2 * N - 1, N))
    for p in range(2 * N - 1):
        lo = max(0, p - N + 1)
        hi = min(N - 1, p)
        m = np.arange(lo, hi + 1)
        j2 = p - m
        np.add.at(grads[p], m, 1.0 - h[j2])
        np.add.at(grads[p], j2, -h[m])
    return grads

def repair(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s < 1e-15:
        return np.full(n, 0.5)
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if diff > 0:
        idx = np.argsort(h)[:int(np.ceil(diff + 1))]
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
    elif diff < 0:
        idx = np.argsort(-h)[:int(np.ceil(-diff + 1))]
        for i in idx:
            rem = min(-diff, h[i])
            h[i] -= rem
            diff += rem
    return h

            break
        dx = 2.0 / n
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n

                    return sm * dx + pen
