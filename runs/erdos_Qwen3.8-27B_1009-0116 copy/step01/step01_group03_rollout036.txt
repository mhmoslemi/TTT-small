- Adds: h[j+s*] + h[j-s*]

To DECREASE result[k*], we want Δ > 0, i.e., h[i+s*] + h[i-s*] > h[j+s*] + h[j-s*].

So we want to remove 1's that have many 1-neighbors at distance |s*|, and add 0's that have few 1-neighbors at distance |s*|.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    shifted = (corr - m) * beta
    shifted -= shifted.max()  # for numerical stability (shifted.max() should be 0)
    exps = np.exp(shifted)
    exp_sum = exps.sum()
    f = (m + np.log(exp_sum) / beta) * dx
    w = exps / exp_sum
    
    # Gradient: grad[j] = -dx * sum_k w[k] * (h[j+s_k] + h[j-s_k])
    # s_k = k - (n-1)
    grad = np.zeros(n)
    for k in range(2*n-1):
        s = k - (n-1)
        wk = w[k]
        # h[j+s] for valid j
        if s >= 0:
            grad[:n-s] -= wk * h[s:]
        else:
            grad[-s:] -= wk * h[:n+s]
        # h[j-s] for valid j
        if s >= 0:
            grad[s:] -= wk * h[:n-s]
        else:
            grad[:n+s] -= wk * h[-s:]
    grad *= dx
    return f, grad

    def obj_and_grad(x):

        return smooth_obj_grad(x, beta)
