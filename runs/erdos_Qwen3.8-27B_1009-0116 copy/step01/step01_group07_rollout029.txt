import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _norm(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s > 1e-12 and s != target:
        h = h * (target / s)
    return np.clip(h, 0.0, 1.0)

def _smooth_obj(h, alpha, dx, target):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    sh = np.clip(corr - m, -10.0, 0.0)
    return m + np.log(np.sum(np.exp(alpha * sh)) / len(sh) + 1e-30) / alpha + 500.0 * (np.sum(h) - target) ** 2

def _fast_cd(h, max_rounds=300, verbose=False):
    n = len(h)
    dx = 2.0 / n
    lags = np.arange(-(n - 1), n)
    corr = np.correlate(h.astype(np.float64), 1.0 - h, mode='full').copy()
    cmax = float(np.max(corr))
    best_cmax = cmax
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for ss in step_sizes:
        for _ in range(max_rounds):
            improved = False
            for i in range(n):
                h_i = h[i]
                for d in [ss, -ss]:
                    new_vi = h_i + d
                    if new_vi < 0.0 or new_vi > 1.0:
                        continue
                    idx1i = i + lags
                    idx2i = i - lags
                    m1i = (idx1i >= 0) & (idx1i < n)
                    m2i = (idx2i >= 0) & (idx2i < n)
                    base_i = d * np.where(m1i, 1.0 - h[idx1i], 0.0) - d * np.where(m2i, h[idx2i], 0.0)
                    # Try local neighbors first, then wider
                    for radius in range(1, n):
                        j = (i + radius) % n
                        if j == (i - radius + n) % n and radius > 1:
                            pass
                        j2 = (i - radius + n) % n
                        for jj in ([j] if radius == 1 else [j, j2]):
                            if jj == i:
                                continue
                            new_vj = h[jj] - d
                            if new_vj < 0.0 or new_vj > 1.0:
                                continue
                            idx1j = jj + lags
                            idx2j = jj - lags
                            m1j = (idx1j >= 0) & (idx1j < n)
                            m2j = (idx2j >= 0) & (idx2j < n)
                            base_j = -d * np.where(m1j, 1.0 - h[idx1j], 0.0) + d * np.where(m2j, h[idx2j], 0.0)
                            new_max = float(np.max(corr + base_i + base_j))
                            if new_max < cmax - 1e-13:
                                corr = corr + base_i + base_j
                                h[i] = new_vi
                                h[jj] = new_vj
                                cmax = new_max
                                if cmax < best_cmax:
                                    best_cmax = cmax
                                improved = True
                                break
                        if improved:
                            break
                    if improved:
                        break
                if improved:
                    break
            if not improved:
                break
    return h, best_cmax * dx

def _slsqp_run(h0, alpha, dx, target, maxiter=80):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return _smooth_obj(x, alpha, dx, target)

    cons = []
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return _norm(res.x, target)

def _lbfgs_run(h0, dx, target, maxiter=300):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n

    def obj_p(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(corr)
        sh = np.clip(corr - m, -5.0, 0.0)
        sm = m + np.log(np.sum(np.exp(2000.0 * sh)) / len(sh) + 1e-30) / 2000.0
        return sm + 2000.0 * (np.sum(x) - target) ** 2

    res = minimize(obj_p, h0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return _norm(res.x, target)

def _perturb(h0, seed, n):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    if len(h0) == n:
        h = h0.astype(float).copy()
        h += rng.uniform(-0.1, 0.1, n)
    else:
        h = np.full(n, 0.5)
        for k in range(rng.randint(1, 6)):
            f = rng.randint(1, max(2, n // 3))
            p = rng.uniform(0, 2.0 * np.pi)
            a = rng.uniform(0.05, 0.4)
            h += a * np.sin(2.0 * np.pi * f * np.arange(n) / n + p)
    return _norm(h, target)

def run(seed=42, budget_s=1100, **kwargs):

    if 'initial_h_values' in globals():

    best_h = parent.copy()

    best_n = len(best_h)

    # Phase 1: SLSQP with increasing alpha at parent resolution
    pn = len(parent)

    pdx = 2.0 / pn

        except Exception:
            pass

    # Phase 2: L-BFGS-B

        try:
            hn = _lbfgs_run(parent, pdx, ptarget, maxiter=200)
            cn = _c5(hn)
            if cn < best_c5:
                best_c5, best_h, best_n = cn, hn.copy(), pn
                parent = hn.copy()
        except Exception:
            pass

    # Phase 3: Fast coordinate descent (multiple rounds)
    if elapsed() < budget_s * 0.6:
        for _ in range(3):
            if elapsed() > budget_s * 0.65:
                break
            hc, cg = _fast_cd(parent.copy(), max_rounds=200)
            if cg < best_c5:
                best_c5, best_h, best_n = cg, hc.copy(), pn
                parent = hc.copy()

    # Phase 4: Try other resolutions
    for n in [80, 100, 120, 90, 60]:
        if elapsed() > budget_s * 0.85 or n == pn:
            continue
        dx = 2.0 / n
        target = n / 2.0
        # Upsample from best_h
        if len(best_h) > 1:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, period=2.0)
        else:
            h_interp = np.full(n, 0.5)
        h_interp = _norm(h_interp, target)

        # SLSQP
        for al in [200, 2000, 10000]:
            if elapsed() > budget_s * 0.88:
                break
            try:
                hn = _slsqp_run(h_interp, al, dx, target, maxiter=60)
                cn = _c5(hn)
                if cn < best_c5:
                    best_c5, best_h, best_n = cn, hn.copy(), n
                    h_interp = hn.copy()
            except Exception:
                pass

        # CD
        if elapsed() < budget_s * 0.92:
            hc, cg = _fast_cd(h_interp, max_rounds=150)
            if cg < best_c5:
                best_c5, best_h, best_n = cg, hc.copy(), n

        if cg < best_c5:
            best_c5, best_h = cg, hc.copy()

    # Final constraint fix
    tgt = best_n / 2.0
    diff = tgt - np.sum(best_h)
    if abs(diff) > 1e-12:
        m = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        cnt = int(np.sum(m))
        if cnt > 0:
            best_h[m] += diff / cnt
        best_h = np.clip(best_h, 0.0, 1.0)
        diff2 = tgt - np.sum(best_h)
        if abs(diff2) > 1e-13:
            best_h -= diff2 / best_n
            best_h = np.clip(best_h, 0.0, 1.0)

    c5f = _c5(best_h)
    print(f"Final: n={best_n}, C5={c5f:.8f}")
    return best_h, c5f, best_n
