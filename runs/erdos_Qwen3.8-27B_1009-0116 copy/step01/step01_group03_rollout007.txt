    diff = ts - h.sum()
    if abs(diff) < 1e-12:

import numpy as np
from scipy.optimize import minimize
import time

_budget = 1100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-10
    else:
        mask = h > 1e-10
    cnt = np.count_nonzero(mask)
    if cnt == 0:
        h += diff / len(h)
        return np.clip(h, 0, 1)
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def obj_lse(h, beta, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = corr.max()
    s = (corr - m) * beta
    smax = s.max()
    lse = (m + smax + np.log(np.sum(np.exp(s - smax)))) / beta
    pen = mu * (h.sum() - n / 2.0) ** 2
    return dx * lse + pen

def lbfgs_phase(h, n, ts, betas, mu, maxiter_each):
    global _budget
    cur = h.copy()

        if time.time() > _budget - 30:

                       options={'maxiter': maxiter_each, 'ftol': 1e-15, 'gtol': 1e-9, 'maxls': 50})
        cur = project_sum(res.x, ts)
    return cur
