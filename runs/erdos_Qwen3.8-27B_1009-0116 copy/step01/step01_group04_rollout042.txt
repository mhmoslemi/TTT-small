    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / _tau, -100.0, 0.0)
    return (m + _tau * np.log(np.sum(np.exp(sh)))) * _dx

    return np.sum(x) - _N / 2.0

    return np.ones(_N)

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    # Fine adjustment
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-10:
        # distribute diff uniformly
        x += diff / n
        x = np.clip(x, 0.0, 1.0)
    return x

    _dx = dx
    _tau = tau

            best = x.copy()
    rem = budget_s - (time.time() - t0)
    if rem > 5:
        x2, v2 = coord_descend(best, n, dx, rem, rng)
        if v2 < bval:
            bval = v2
            best = x2
    return best, bval

def project(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    f = interp1d(xs, x_src, kind='linear', bounds_error=False, fill_value=(x_src[0], x_src[-1]))
    return repair(f(xd), n_dst)

def perturb(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t_start = time.time()
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(95, 0.5)
    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = repair(h0, n0)
    best_h, best_n, best_val = h0.copy(), n0, compute_c5(h0, dx0)

    # Phase 1: Multiple restarts at n0
    p1_time = budget_s * 0.38
    n_rst = 5
    per_r = p1_time / n_rst
    inits = [h0.copy()]
    for i in range(1, n_rst):
        inits.append(perturb(h0, n0, 0.04 + 0.03 * i, rng))
    for xs in inits:
        el = time.time() - t_start
        if el > budget_s * 0.42:
            break
        alloc = min(per_r, budget_s * 0.42 - el)
        if alloc < 12:
            break
        xb, vb = anneal_slsqp(xs, n0, dx0, alloc, rng)
        if vb < best_val:
            best_val, best_h, best_n = vb, xb.copy(), n0

    # Phase 2: Finer grid (2*n0+1)
    el = time.time() - t_start
    if el < t_start + budget_s * 0.55:
        n1 = 2 * n0 + 1
        dx1 = 2.0 / n1
        xp = project(best_h, n0, n1)
        p2b = min(budget_s * 0.24, budget_s * 0.78 - el)
        if p2b > 30:
            xb2, vb2 = anneal_slsqp(xp, n1, dx1, p2b * 0.55, rng)
            if vb2 < best_val:
                best_val, best_h, best_n = vb2, xb2.copy(), n1
            if time.time() - t_start < t_start + budget_s * 0.77:
                xp2 = perturb(xp, n1, 0.06, rng)
                rem2 = min(p2b * 0.45, budget_s * 0.18 - (time.time() - t_start))
                if rem2 > 15:
                    xb3, vb3 = anneal_slsqp(xp2, n1, dx1, rem2, rng)
                    if vb3 < best_val:
                        best_val, best_h, best_n = vb3, xb3.copy(), n1

    # Phase 3: Try another nearby grid size for diversity
    el = time.time() - t_start
    if el < t_start + budget_s * 0.78:
        n3 = n0 + 16 if n0 + 16 != n1 else n0 + 12
        dx3 = 2.0 / n3
        xp3 = project(best_h, best_n, n3)
        p3b = min(budget_s * 0.12, budget_s * 0.82 - el)
        if p3b > 15:
            xb4, vb4 = anneal_slsqp(xp3, n3, dx3, p3b, rng)
            if vb4 < best_val:
                best_val, best_h, best_n = vb4, xb4.copy(), n3

    # Phase 4: Final CD polish
    rem = budget_s - (time.time() - t_start) - 3
    if rem > 8:
        dxf = 2.0 / best_n
        hp, vp = coord_descend(best_h, best_n, dxf, rem, rng)
        if vp < best_val:
            best_val, best_h = vp, hp

    dxf = 2.0 / best_n
    best_h = repair(best_h, best_n)
    return (best_h, compute_c5(best_h, dxf), best_n)
