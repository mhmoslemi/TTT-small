Let me think about another angle: what if I use **CVXPY** for a convex relaxation?

The problem: min C5 = min max_k corr_k(h)
subject to 0 ≤ h ≤ 1, sum(h) = n/2

If corr_k were linear in h, this would be a linear program. But corr_k is bilinear. However, for a given h, corr_k is fixed. So we can do:

minimize t
subject to corr_k(h) ≤ t for all k
0 ≤ h ≤ 1
sum(h) = n/2

This is a non-convex problem (bilinear constraints). But we can linearize around the current h:
corr_k(h + δ) ≈ corr_k(h) + ∑_j (∂corr_k/∂h_j) * δ_j

Then solve the LP, take a step, repeat. This is a **linearization / cutting plane** approach.

Actually, since corr_k is bilinear, the linearization at a point h₀ is:
corr_k(h) ≈ corr_k(h₀) + ∑_j (∂corr_k/∂h_j)|_{h₀} * (h_j - h_{0,j})

This is a first-order approximation. Solving the LP gives a direction. Then we take a damped step.

This is essentially a **sequential linear programming (SLP)** approach, similar to what SLSQP does internally. But we can implement it more carefully with CVXPY.

def _slp_step(h, n, dx, max_iters=200):

    best_h = h.copy()
    best_c5 = _c5(h, dx)
    
    for it in range(max_iters):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        M = np.max(corr)
        
        # Compute gradients for all active constraints (top few correlations)
        h_var = cp.Variable(n)
        t = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]
        
        # Add linearized correlation constraints for top correlations
        top_k = np.argsort(corr)[-5:]  # worst 5 shifts
        for k_idx in top_k:
            s = k_idx - (n - 1)
            # grad of corr[k_idx] at best_h
            g = np.zeros(n)
            for j in range(n):
                jm = j - s
                jp = j + s
                if 0 <= jm < n:
                    g[j] += (1 - best_h[jm])
                if 0 <= jp < n:
                    g[j] += (-best_h[jp])
            # Linearized: corr[k] ≈ corr[k_idx] + g @ (h - best_h)
            constraints.append(corr[k_idx] + g @ (h_var - best_h) <= t)

        constraints.append(t >= M * 0.99)  # ensure we don't decrease too fast
        
        obj = cp.Minimize(t)
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS)  # or another solver
        
        if h_var.value is not None:

                best_c5 = new_c5
        
        # Check convergence

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    s = h.sum()
    if s < 1e-15:
        return np.full(n, target / n)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = np.maximum(1.0 - h, 1e-15)
            h = np.clip(h + diff / room.sum() * room, 0, 1)
        else:
            mass = np.maximum(h, 1e-15)
            h = np.clip(h + diff / mass.sum() * mass, 0, 1)
    h = np.clip(h, 0, 1)
    diff = target - h.sum()
    if diff > 1e-10:
        for i in np.argsort(1.0 - h):
            a = min(diff, 1.0 - h[i])
            h[i] += a
            diff -= a
            if diff < 1e-12:
                break
    elif diff < -1e-10:
        for i in np.argsort(h):
            s = min(-diff, h[i])
            h[i] -= s
            diff += s
            if -diff < 1e-12:
                break
    return h


def _c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def _grad_corr_k(h, k_idx, n):
    s = k_idx - (n - 1)
    g = np.zeros(n)
    for j in range(n):
        jm = j - s
        jp = j + s
        if 0 <= jm < n:
            g[j] += (1.0 - h[jm])
        if 0 <= jp < n:
            g[j] -= h[jp]
    return g


def _project_nu(h, n):
    """Project onto {0<=h<=1, sum=h target} using bisection."""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        hp = np.clip(h - mid, 0, 1)
        s = hp.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix remaining
    diff = target - hp.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            for i in np.argsort(1.0 - hp):
                a = min(diff, 1.0 - hp[i])
                hp[i] += a
                diff -= a
                if diff < 1e-12:
                    break
        else:
            for i in np.argsort(hp):
                s = min(-diff, hp[i])
                hp[i] -= s
                diff += s
                if -diff < 1e-12:
                    break
    return hp

    rng = np.random.RandomState(seed)
    n = len(initial_h_values)
    dx = 2.0 / n

    best_h = h.copy()
    best_c5 = _c5(h, dx)

    def elapsed():
        return time.time() - t0

    # Phase 1: Subgradient with momentum
    alpha = 0.02
    momentum = np.zeros(n)
    mu = 0.9
    for it in range(5000):
        if elapsed() > budget_s - 800:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_peak = np.argmax(corr)
        g = _grad_corr_k(h, k_peak, n)
        momentum = mu * momentum - alpha * g
        h_new = _project_nu(h + momentum, n)
        # Actually, subgradient: we want to DECREASE the max, 
        # so we should move AGAINST the gradient: h -= alpha * g
        # Wait, g is gradient of corr[k_peak], and we want to minimize it.
        # So h_new = h - alpha * g (then project)
        h_new = _project_nu(h - alpha * g, n)
        c_new = _c5(h_new, dx)
        if c_new < _c5(h, dx) * 1.001:  # accept if not too much worse
            h = h_new
        alpha *= 0.999
