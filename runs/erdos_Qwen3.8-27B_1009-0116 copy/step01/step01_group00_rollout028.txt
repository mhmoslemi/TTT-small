```python
        return _smooth_obj(x, alpha)
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
```

```python
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(np.sum(h))
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_alpha = 100.0
_G_n = 70

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = float(np.max(corr))
    s = _G_alpha * (corr - m)
    s = np.clip(s, -500.0, 500.0)
    return m + np.log(np.sum(np.exp(s))) / _G_alpha

def _eqcon(x):
    return float(np.sum(x)) - _G_n / 2.0

def _compute_c5(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = float(np.sum(h))
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(np.sum(h))
    if abs(diff) > 1e-12:
        order = np.argsort(1.0 - h) if diff > 0 else np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _slsqp_solve(h, n, alpha, maxiter=500, ftol=1e-14):
    global _G_alpha, _G_n
    _G_alpha = float(alpha)
    _G_n = int(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eqcon}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return _repair(res.x, n)
    except Exception:
        return h

def _coord_descent(h, n, dx, steps, max_time_s, rng):
    best_c5 = _compute_c5(h, dx)
    t0 = time.time()
    stagnation = 0
    while time.time() - t0 < max_time_s:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() - t0 > max_time_s:
                    return h, best_c5
                hi = h[i]
                hj = h[j]
                for s in steps:
                    for sign in (1.0, -1.0):
                        move = sign * s
                        ni = hi + move
                        nj = hj - move
                        if ni < -1e-10 or ni > 1.0 + 1e-10:
                            continue
                        if nj < -1e-10 or nj > 1.0 + 1e-10:
                            continue
                        ni = max(0.0, min(1.0, ni))
                        nj = max(0.0, min(1.0, nj))
                        h[i] = ni
                        h[j] = nj
                        c5 = _compute_c5(h, dx)
                        if c5 < best_c5 - 1e-13:
                            best_c5 = c5
                            improved = True
                            hi = ni
                            hj = nj
                        else:
                            h[i] = hi
                            h[j] = hj
        if not improved:
            stagnation += 1
            if stagnation >= 3:
                break
        else:
            stagnation = 0
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    n = len(initial_h_values)

    _G_n = n
```

```python
    best_c5 = _compute_c5(best_h, dx)

    # Phase 1: SLSQP warm-up with increasing alpha
    for a in [30, 100, 300, 600]:

        c5 = _compute_c5(h_new, dx)
        if c5 < best_c5:

            best_h = h_new

    # Phase 2: Iterative coordinate descent + perturbation loop
    steps_coarse = np.array([0.01, 0.02, 0.05, 0.1])
    steps_fine = np.array([0.002, 0.005, 0.01, 0.02])
    steps_vfine = np.array([0.0005, 0.001, 0.002, 0.005])
```
