```python
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    """Return the correlation array (not just max)"""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def to_binary(h, n, rng):
    h = np.round(h).astype(float)
    cur = int(h.sum())
    target = n // 2
    if cur > target:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, cur - target, replace=False)] = 0
    elif cur < target:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, target - cur, replace=False)] = 1
    return h

def best_improve_local_search(h, n, max_sweeps=100, time_budget=60):
    """Best-improvement local search: in each sweep, find the best swap and apply it"""
    t0 = time.time()
    current_c5 = compute_c5(h)
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_budget:
            break
        best_delta = -1e-10  # minimum improvement to accept
        best_a, best_b = -1, -1
        
        for a in zeros:
            for b in ones:
                h[a], h[b] = 1.0, 0.0
                c5 = compute_c5(h)
                delta = current_c5 - c5
                if delta > best_delta:
                    best_delta = delta
                    best_a, best_b = a, b
                h[a], h[b] = 0.0, 1.0

            current_c5 -= best_delta
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
        else:
            break
    
    return h, current_c5
```

```python
def swap_delta(h, a, b, n):
    """Compute change in c[m] for all m when h[a]:0→1, h[b]:1→0"""
    # For each m from -(n-1) to n-1:
    # delta_c[m] = -h[a+m] + (1-h[a-m]) + h[b+m] - (1-h[b-m])
    # where indices must be in [0, n-1]
    
    # This affects c at lags m where any of a+m, a-m, b+m, b-m is in [0,n-1]
    # a+m valid: -a ≤ m ≤ n-1-a
    # a-m valid: -(n-1-a) ≤ m ≤ a
    # b+m valid: -b ≤ m ≤ n-1-b  
    # b-m valid: -(n-1-b) ≤ m ≤ b
    
    # Total affected range: m from -(n-1) to n-1 (all of them potentially)
    # But we can compute just the delta for all m in O(n)
    
    delta = np.zeros(2*n-1)
    # Index in delta: m + (n-1)
    off = n - 1
    
    # Contribution from h[a] going 0→1:
    for m in range(-a, n-a):  # a+m in [0, n-1]
        if 0 <= m + off < 2*n-1:
            delta[m + off] += -h[a+m]
    for m in range(-(n-1-a), a+1):  # a-m in [0, n-1]
        if 0 <= m + off < 2*n-1:
            delta[m + off] += (1 - h[a-m])
    
    # Contribution from h[b] going 1→0:
    for m in range(-b, n-b):  # b+m in [0, n-1]
        if 0 <= m + off < 2*n-1:
            delta[m + off] += h[b+m]
    for m in range(-(n-1-b), b+1):  # b-m in [0, n-1]
        if 0 <= m + off < 2*n-1:
            delta[m + off] -= (1 - h[b-m])
    
    return delta
```

```python
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_c5_corr(h):
    """Return (c5_value, corr_array)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr

def to_binary(h, n, rng):
    h = np.round(h).astype(float)
    cur = int(h.sum())
    target = n // 2
    if cur > target:
        ones = np.where(h == 1)[0]
        h[rng.choice(ones, cur - target, replace=False)] = 0
    elif cur < target:
        zeros = np.where(h == 0)[0]
        h[rng.choice(zeros, target - cur, replace=False)] = 1
    return h

def greedy_local_search(h, n, max_sweeps, time_limit):
    """Best-improvement: try all swaps, apply the best one, repeat"""
    t0 = time.time()
    c5 = compute_c5(h)
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    for sweep in range(max_sweeps):
        if time.time() - t0 > time_limit:
            break
        best_delta = 0
        best_a, best_b = -1, -1
        for a in zeros:
            for b in ones:
                h[a], h[b] = 1.0, 0.0
                c5_new = compute_c5(h)
                delta = c5 - c5_new
                if delta > best_delta + 1e-15:
                    best_delta = delta
                    best_a, best_b = a, b
                h[a], h[b] = 0.0, 1.0
        if best_a >= 0:
            h[best_a], h[best_b] = 1.0, 0.0
            c5 -= best_delta
            zeros = np.where(h == 0)[0]
            ones = np.where(h == 1)[0]
        else:
            break
    return h, c5

def sa_binary(h_init, budget, seed):
    """SA on binary with pair swaps"""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    h = to_binary(h_init.copy(), n, rng)
    c5_cur = compute_c5(h)
    h_best = h.copy()
    c5_best = c5_cur
    
    t0 = time.time()
    T = 0.003
    T_min = 1e-7
    stuck = 0
    stuck_thresh = 10000
    
    while time.time() - t0 < budget:
        frac = (time.time() - t0) / budget
        T = max(T_min, T * (1 - 0.7 * frac))
        
        if stuck > stuck_thresh:
            T = max(T, 0.005)
            stuck = 0
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        a = zeros[rng.randint(len(zeros))]
        b = ones[rng.randint(len(ones))]
        
        h[a], h[b] = 1.0, 0.0
        c5_new = compute_c5(h)
        delta = c5_new - c5_cur
        
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            c5_cur = c5_new
            if c5_new < c5_best:
                h_best = h.copy()
                c5_best = c5_new
                stuck = 0
            else:
                stuck += 1
        else:
            h[a], h[b] = 0.0, 1.0
            stuck += 1
    
    return h_best, c5_best

def slsqp_refine(h, n, beta, maxiter=400):
    ts = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        m = corr.max()
        shifted = (corr - m) * beta
        return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        x = np.clip(res.x, 0, 1)
        x += (ts - x.sum()) / n
        return np.clip(x, 0, 1)
    return h
```
