    return np.correlate(h, 1.0 - h, mode="full")

    n = len(h)
    return float(np.max(compute_corr(h)) * 2.0 / n)

def project_feasible(h, ts):
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    return h

    def __init__(self, beta):
        self.beta = beta
    def __call__(self, h):
        corr = compute_corr(h)
        m = corr.max()
        n = len(h)
        return (m + np.log(np.sum(np.exp((corr - m) * self.beta)) / len(corr))) / self.beta * (2.0 / n)

        self.ts = ts

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP warm-up with annealed beta
    for beta in [8.0, 25.0, 80.0, 250.0, 600.0]:
        if budget_s - (time.time() - t_start) < 10:
            break
        h = slsqp_refine(best_h.copy(), ts, beta, 400)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Phase 2: Binary pair-swap exhaustive search (main effort)
    if budget_s - (time.time() - t_start) > 180:
        h_bin = make_binary(best_h, ts)
        t_stop2 = t_start + budget_s * 0.28
        h_bin, c5_bin = binary_swap_search(h_bin, ts, t_stop2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 3: Simulated annealing to escape local optima
    if budget_s - (time.time() - t_start) > 150:
        t_stop3 = t_start + budget_s * 0.50
        h_sa, c5_sa = simulated_annealing(best_h, ts, t_stop3, seed, n_iter=100000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 4: Second round of exhaustive swap search
    if budget_s - (time.time() - t_start) > 100:
        t_stop4 = t_start + budget_s * 0.72
        h_sw, c5_sw = binary_swap_search(best_h, ts, t_stop4)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw.copy()

    # Phase 5: Continuous SLSQP polish
    if budget_s - (time.time() - t_start) > 30:
        for beta in [200.0, 500.0, 1200.0]:
            if budget_s - (time.time() - t_start) < 10:
                break
            h = slsqp_refine(best_h.copy(), ts, beta, 300)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 6: Try alternative resolutions
    if budget_s - (time.time() - t_start) > 50:
        for n2 in [100, 200, 280]:
            if budget_s - (time.time() - t_start) < 35:
                break
            xo = np.linspace(0, 2, len(best_h))
            xn = np.linspace(0, 2, n2)
            h2 = np.interp(xn, xo, best_h)
            h2 = project_feasible(h2, n2 / 2.0)
            ts2 = n2 / 2.0
            h2 = slsqp_refine(h2, ts2, 50.0, 200)
            h2 = slsqp_refine(h2, ts2, 200.0, 200)
            h2b = make_binary(h2, ts2)
            t_lim = min(t_start + 30, t_start + (budget_s - (time.time() - t_start)) * 0.6)
            h2b, c5_2b = binary_swap_search(h2b, ts2, t_lim)
            c5_2 = compute_c5(h2b)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2b.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
