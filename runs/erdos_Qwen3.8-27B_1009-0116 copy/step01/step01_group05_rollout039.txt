```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5(h, dx):
    return np.correlate(h, 1.0 - h, mode="full").max() * dx

def _proj(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _sobj_cv(x, dx, n, eps):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _sgrad_cv(x, dx, n, eps):
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _slsqp_refine(h, dx, n, target, t_end, max_rounds=5):
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    h = h.copy()
    for eps in [0.05, 0.01, 0.005, 0.001, 0.0005]:
        if time.time() > t_end - 10:
            break
        def obj(x):
            return _sobj_cv(x, dx, n, eps)
        def jac(x):
            return _sgrad_cv(x, dx, n, eps)
        def efun(x):
            return x.sum() - target
        def ejac(x):
            return np.ones(n)
        try:
            res = minimize(obj, h, jac=jac,
                           bounds=[(0,1)]*n,
                           constraints=[{'type':'eq','fun':efun,'jac':ejac}],
                           method='SLSQP',
                           options={'maxiter':300, 'ftol':1e-15})
            h = _proj(res.x, target)
        except:
            break
        c = _c5(h, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    return best_h, best_c5

def _coord_descent(h, dx, n, target, n_passes=100, t_end=None, rng=None):
    best_c5 = _c5(h, dx)
    h = h.copy()
    for p in range(n_passes):
        if t_end and time.time() > t_end:
            break
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
```

Let me also think about what makes this problem hard and what the optimal structure looks like.

The function h: [0,2] → [0,1] with ∫h = 1 means the "average" is 0.5. We want to minimize max_k ∫ h(x)(1-h(x+k)) dx.

```python
_G = {}

def _obj(x):
    dx, n, eps = _G['dx'], _G['n'], _G['eps']
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _grad(x):
    dx, n, eps = _G['dx'], _G['n'], _G['eps']
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)

        vm = (jm >= 0) & (jm < n)

        jp = j + s
        vp = (jp >= 0) & (jp < n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def _obj(x):
    cv = np.correlate(x, 1.0 - x, mode="full") * _G['dx']
    m = cv.max()
    return m + _G['eps'] * np.log(np.sum(np.exp((cv - m) / _G['eps'])))

def _grad(x):
    n, dx, eps = _G['n'], _G['dx'], _G['eps']
    cv = np.correlate(x, 1.0 - x, mode="full") * dx
    m = cv.max()
    w = np.exp((cv - m) / eps)
    w /= w.sum()
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        ji = j + s
        vi = (ji >= 0) & (ji < n)
        if vi.any():
            g[vi] += wt * (1.0 - x[ji[vi]])
        js = j - s
        vs = (js >= 0) & (js < n)
        if vs.any():
            g[vs] -= wt * x[js[vs]]
    return g

def _cfun(x):
    return x.sum() - _G['tgt']

def _cjac(x):
    return np.ones(_G['n'])

def _c5(h, dx):
    return np.correlate(h, 1.0 - h, mode="full").max() * dx

def _proj(h, tgt):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _slsqp(h, dx, n, tgt, te, eps_lst):
    _G.update(dx=dx, n=n, tgt=tgt)
    bh, bc = h.copy(), _c5(h, dx)
    h = h.copy()
    for eps in eps_lst:
        if time.time() > te - 8:
            break
        _G['eps'] = eps
        try:
            r = minimize(_obj, h, jac=_grad,
                         bounds=[(0.0, 1.0)] * n,
                         constraints=[{'type': 'eq', 'fun': _cfun, 'jac': _cjac}],
                         method='SLSQP', options={'maxiter': 250, 'ftol': 1e-16})
            h = _proj(r.x, tgt)
        except Exception:
            break
        c = _c5(h, dx)
        if c < bc:
            bc, bh = c, h.copy()
    return bh, bc

def _coord(h, dx, n, tgt, passes, te, rng, gn):
    bc = _c5(h, dx)
    h = h.copy()
    for p in range(passes):
        if te is not None and time.time() > te - 4:
            break
        ord_ = np.arange(n)
        if rng is not None:
            rng.shuffle(ord_)
        imp = False
        for i in ord_:
            if te is not None and time.time() > te - 4:
                return h, bc
            sv = h.copy()
            bct, bht = bc, None
            for t in np.linspace(0.0, 1.0, gn):
                sv[i] = t
                ht = _proj(sv.copy(), tgt)
                c = _c5(ht, dx)
                if c < bct - 1e-13:
                    bct, bht = c, ht.copy()
            if bht is not None:
                h, bc, imp = bht, bct, True
        if not imp and p > 12:
            break
    return h, bc

def _interp(h_src, n_dst):
    if len(h_src) == n_dst:
        return h_src.copy()
    return np.interp(np.linspace(0, 1, n_dst),
                     np.linspace(0, 1, len(h_src)), h_src)
```
