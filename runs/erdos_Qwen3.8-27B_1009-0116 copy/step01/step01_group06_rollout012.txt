        return eval_c5(x, dx)

        return [x.sum() - target]

        return np.ones((1, n))

def project_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-13:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-13:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-13:
            if d2 > 0:
                idx = np.where(h < 1.0 - 1e-12)[0]
            else:
                idx = np.where(h > 1e-12)[0]
            for j in idx:
                if abs(d2) < 1e-13:
                    break
                step = np.sign(d2) * min(abs(d2), abs(1.0 - h[j]) if d2 > 0 else h[j])
                h[j] += step
                d2 -= step
    return h

        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def subgrad_opt(h, n, dx, maxiter=50000, lr_init=0.008):
    h = h.copy()
    best_h = h.copy()
    best_c5 = eval_c5(h, dx)
    lr = lr_init
    for it in range(maxiter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        tol = 1e-9 * max_val + 1e-12
        active = np.where(np.abs(corr - max_val) < tol)[0]
        grad = np.zeros(n)
        for k_idx in active:
            lag = int(k_idx) - (n - 1)
            for i in range(n):
                j = i + lag
                if 0 <= j < n:
                    grad[i] += (1.0 - h[j])
                    grad[j] -= h[i]
        grad *= dx / len(active)
        h_new = project_feasible(h - lr * grad, n)
        c5_new = eval_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.05)
        else:
            lr *= 0.96
            if lr < 1e-11:
                lr = lr_init
    return best_h, best_c5

def refine_with_lp_and_slsqp(h_start, n, dx, budget_per=30):
    t0 = time.time()
    best_h = project_feasible(h_start, n)
    best_c5 = eval_c5(best_h, dx)
    if time.time() - t0 < budget_per - 5:
        h, c = lp_optimize(best_h, n, dx, max_iter=150, blend=0.5)
        if c < best_c5:
            best_h, best_c5 = h, c
    if time.time() - t0 < budget_per - 2:
        h, c = slsqp_opt(best_h, n, dx, maxiter=300, ftol=1e-15)
        if c < best_c5:
            best_h, best_c5 = h, c
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    n = 70
    dx = 2.0 / n
    h0 = project_feasible(np.asarray(initial_h_values, dtype=float), n)
    best_h = h0.copy()
    best_c5 = eval_c5(best_h, dx)

    # Phase 1: LP linearization from initial
    if time.time() - t0 < budget_s - 200:
        h, c = lp_optimize(best_h, n, dx, max_iter=400, blend=0.5)
        if c < best_c5:
            best_h, best_c5 = h, c
        h, c = lp_optimize(best_h, n, dx, max_iter=200, blend=0.3)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 2: SLSQP polish
    if time.time() - t0 < budget_s - 120:
        h, c = slsqp_opt(best_h, n, dx, maxiter=1000, ftol=1e-15)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 3: Subgradient refinement
    if time.time() - t0 < budget_s - 60:
        h, c = subgrad_opt(best_h, n, dx, maxiter=40000, lr_init=0.005)
        if c < best_c5:
            best_h, best_c5 = h, c
        if time.time() - t0 < budget_s - 30:
            h2, c2 = slsqp_opt(best_h, n, dx, maxiter=500, ftol=1e-15)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2

    # Phase 4: Try different n values via interpolation
    for n_try in [60, 80, 90, 100]:
        if time.time() - t0 > budget_s - 40:
            break
        dx_try = 2.0 / n_try
        x_old = np.linspace(0, 2, n + 1)
        x_new = np.linspace(0, 2, n_try + 1)
        h_interp = np.interp(x_new[:n_try], x_old, best_h, left=best_h[0], right=best_h[-1])
        h_interp = project_feasible(h_interp, n_try)
        h_try, c_try = refine_with_lp_and_slsqp(h_interp, n_try, dx_try, budget_per=25)
        if c_try < best_c5:
            # Convert back to n=70 grid
            h_back = np.interp(x_old[:n], x_new, h_try, left=h_try[0], right=h_try[-1])
            h_back = project_feasible(h_back, n)
            c_back = eval_c5(h_back, dx)
            if c_back < best_c5:
                best_h, best_c5 = h_back, c_back
            else:
                best_h = h_interp  # keep the n_try solution for reporting
                best_c5 = c_try
                # Actually we must return n consistent with h
                # Let's just record and convert at end
                pass

    # Phase 5: Multi-start with LP + SLSQP
    attempt = 0
    while time.time() - t0 < budget_s - 3:
        attempt += 1
        if attempt <= 2:
            h_start = best_h + 0.12 * np.random.randn(n)
        elif attempt <= 4:
            h_start = best_h + 0.04 * np.random.randn(n)
        elif attempt <= 6:
            h_start = np.clip(0.5 + 0.35 * np.sin(2 * np.pi * np.arange(n) / n + np.random.uniform(0, 2*np.pi)), 0, 1)
        elif attempt <= 8:
            h_start = np.clip(0.5 + 0.25 * np.cos(np.pi * np.arange(n) / (n/2.0) * (1 + 0.5*np.random.rand())), 0, 1)
        elif attempt % 5 == 0:
            h_start = np.random.rand(n)
        else:
            h_start = best_h.copy()
            idx = np.random.choice(n, size=n//3, replace=False)
            h_start[idx] = np.clip(h_start[idx] + 0.3 * np.random.randn(len(idx)), 0, 1)
        h_start = project_feasible(h_start, n)
        h_try, c_try = refine_with_lp_and_slsqp(h_start, n, dx, budget_per=15)
        if c_try < best_c5:
            best_h, best_c5 = h_try, c_try

    # Final SLSQP polish if time permits
    if time.time() - t0 < budget_s - 1:
        h, c = slsqp_opt(best_h, n, dx, maxiter=800, ftol=1e-15)
        if c < best_c5:
            best_h, best_c5 = h, c

    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)
    return (best_h, best_c5, n)
