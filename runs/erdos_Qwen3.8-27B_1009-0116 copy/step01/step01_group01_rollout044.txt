def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

def compute_all_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_obj(h, n, dx, t):
    c = compute_all_corr(h)
    cmax = c.max()
    return cmax + np.log(np.sum(np.exp(t * (c - cmax)))) / t) * dx

def gradient_smooth_max(h, n, dx, t):
    c = compute_all_corr(h)
    cmax = c.max()
    w = np.exp(t * (c - cmax))
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-8:
            continue
        m = k - (n - 1)
        # First term: (1-h[j+m]) for valid j+m
        js = max(0, -m)
        je = min(n, n - m)
        if je > js:
            g[js:je] += w[k] * (1.0 - h[js+m:je+m])
        # Second term: -h[j-m] for valid j-m
        js2 = max(0, m)
        je2 = min(n, n + m)
        if je2 > js2:
            g[js2:je2] -= w[k] * h[js2-m:je2-m]
    return dx * g

def projected_subgradient(h, n, steps, t_init, t_final, alpha_init, seed):

    best_h = h.copy()
    best_c5 = compute_c5(h)

        t = t_init + (t_final - t_init) * frac
        alpha = alpha_init / (1 + 0.01 * step)
        
        g = gradient_smooth_max(h, n, dx, t)
        h_new = h - alpha * g
        h_new = np.clip(h_new, 0.0, 1.0)

        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
