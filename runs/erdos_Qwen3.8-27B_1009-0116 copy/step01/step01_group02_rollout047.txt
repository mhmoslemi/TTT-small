import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

def optimize_from_start(h0, n, rng, t0, budget_s):
    """Multi-phase optimization from a starting point."""
    h_cur = repair(h0.copy(), n)
    best_c = compute_c5(h_cur, n)
    target_sum = n / 2.0

    def eq_con(x):
        return np.sum(x) - target_sum
    cons = [{'type': 'eq', 'fun': eq_con}]
    
    # Phase A: SLSQP with log-sum-exp, increasing beta
    for beta in [30, 80, 200, 500]:
        if time.time() - t0 > budget_s - 10:
            break
        try:
            def obj(x):
                return lse_obj(x, n, beta)
            r = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                        constraints=cons, options={'maxiter': 400, 'ftol': 1e-14})
            hc = repair(r.x, n)
            cc = compute_c5(hc, n)
            if cc < best_c:
                h_cur, best_c = hc, cc
        except Exception:
            pass
    
    # Phase B: SLSQP on exact max
    if time.time() - t0 > budget_s - 8:
        return h_cur, best_c
    try:
        def obj2(x):
            return compute_c5(x, n)
        r = minimize(obj2, h_cur, method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': 600, 'ftol': 1e-15})
        hc = repair(r.x, n)
        cc = compute_c5(hc, n)
        if cc < best_c:
            h_cur, best_c = hc, cc
    except Exception:
        pass
    
    # Phase C: Neighborhood search (push-pull with annealing)
    if time.time() - t0 > budget_s - 5:
        return h_cur, best_c

        i = int(rng.randint(n))

            hp[i] += eps

                if cp < best_c:
                    best_c = cp
        temp *= 0.995
        if temp < 0.003:
            temp = 0.003
