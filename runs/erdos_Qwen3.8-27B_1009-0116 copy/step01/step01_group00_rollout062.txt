        return np.full(n, 0.5)
    h *= t / s

For np.correlate(h, 1-h, mode='full'), the result has 2n-1 elements. Element m corresponds to the correlation at lag m-(n-1).

h[i] appears in the correlation at lag k if i and i+k are both valid indices (0 ≤ i < n, 0 ≤ i+k < n).

So for a given i, the affected lags are k = 0, ±1, ±2, ..., ±(n-1-i) and k = 0, -1, -2, ..., -(i-(n-1-i))... Actually, it's simpler than that. h[i] appears in corr[m] for all m where i is in the range of valid indices for that lag.

For np.correlate(h, 1-h, mode='full'):

import time

_gn = 0
_ga = 100.0

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return float(np.max(_corr(h)) * dx)

def _smx(c, a):
    m = float(np.max(c))
    s = np.clip(a * (c - m), -500, 500)
    return m + float(np.log(np.sum(np.exp(s))) / a)

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    t = 0.5 * n
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (t / s)
    h = np.clip(h, 0.0, 1.0)
    d = t - h.sum()
    if abs(d) > 1e-12:
        idx = np.argsort(1.0 - h) if d > 0 else np.argsort(h)
        for i in idx:
            if d > 0:
                a = min(d, 1.0 - h[i])
                h[i] += a
                d -= a
            else:
                a = min(-d, h[i])
                h[i] -= a
                d += a
            if abs(d) < 1e-12:
                break
    return h

def _obj(x):
    return _smx(_corr(x), _ga)

def _con(x):
    return x.sum() - 0.5 * _gn

def _slsqp(h0, n, alpha, maxiter):
    global _gn, _ga
    _gn, _ga = n, alpha
    try:
        r = minimize(_obj, h0, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        return r
    except Exception:
        return None

def _rp_cd(h, n, dx, rng, ncap, tcap, t0):
    """Random pair coordinate descent."""

    ds = [0.001, 0.003, 0.007, 0.015, 0.03, 0.06, 0.12, 0.2]

                    ni, nj = hi + dd, hj - dd

                        bc, hi, hj = c, ni, nj
                        imp = True
                    else:
                        h[i], h[j] = hi, hj
        if not imp:

    return h, bc
