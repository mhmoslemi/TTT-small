def compute_c5_full(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5(h, n):
    return float(np.max(compute_c5_full(h, n)))

def repair(h, n, target=None):
    if target is None:
        target = n / 2.0
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s < 1e-15:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute diff among non-boundary elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            adj = diff / cnt
            h[mask] += adj
            h = np.clip(h, 0.0, 1.0)
        # If still off, adjust any element
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            idx = int(np.argmax(h)) if diff2 < 0 else int(np.argmin(h))
            h[idx] = np.clip(h[idx] + diff2, 0, 1)
    return h

def project_onto_simplex_box(h, target):
    """Project h onto {0<=h_i<=1, sum(h)=target}"""
    h = np.array(h, dtype=float)
    # Binary search on lambda for projection
    # h_i = clip(h_i - lambda, 0, 1), find lambda such that sum = target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        hp = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(hp)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2, 0.0, 1.0)

def subgradient_step(h, n, alpha, c5_vals):
    """One subgradient step using the active lag(s)"""
    n_corr = 2 * n - 1
    # Find top few active lags
    top_k = np.argsort(c5_vals)[-3:]
    g = np.zeros(n)
    for k in top_k:
        lag = k - (n - 1)  # Convert to actual lag
        g_lag = np.zeros(n)
        if lag == 0:
            g_lag = 1.0 - 2.0 * h
        elif lag > 0:
            # c[lag] = sum_{i=0}^{n-1-lag} h[i](1-h[i+lag])
            # dc/dh[j] = (1-h[j+lag]) for j in [0, n-1-lag]  (first factor)
            #            -h[j-lag] for j in [lag, n-1]       (second factor)
            g_lag[:n-lag] += (1.0 - h[lag:])
            g_lag[lag:] += (-h[:n-lag])
        else:
            lag2 = -lag
            # c[-lag2] = sum_{i=lag2}^{n-1} h[i](1-h[i-lag2])
            # dc/dh[j] = (1-h[j-lag2]) for j in [lag2, n-1]  (first factor)
            #            -h[j+lag2] for j in [0, n-1-lag2]    (second factor)
            g_lag[lag2:] += (1.0 - h[:n-lag2])
            g_lag[:n-lag2] += (-h[lag2:])
        g += g_lag
    g /= len(top_k)
    h_new = h - alpha * g
    h_new = project_onto_simplex_box(h_new, n / 2.0)
    return h_new

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_full(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5(h, n):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def project_box_sum(h, target):
    h = np.array(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h - mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    return project_box_sum(h, target)

def subgradient(h, n, top_m=5):
    c5v = compute_c5_full(h, n)
    n_corr = 2 * n - 1
    top_idx = np.argsort(-c5v)[:top_m]
    g = np.zeros(n)
    cnt = 0
    for k in top_idx:
        lag = int(k) - (n - 1)
        gl = np.zeros(n)
        if lag == 0:
            gl = 1.0 - 2.0 * h
        elif lag > 0:
            gl[:n - lag] += (1.0 - h[lag:])
            gl[lag:] += (-h[:n - lag])
        else:
            lag2 = -lag
            gl[lag2:] += (1.0 - h[:n - lag2])
            gl[:n - lag2] += (-h[lag2:])
        g += gl
        cnt += 1
    g /= max(cnt, 1)
    return g

def logsumexp_obj(h, n, beta):
    c5v = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
    m = np.max(c5v)
    return m + np.log(np.sum(np.exp(beta * (c5v - m)))) / beta

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    init = globals().get('initial_h_values', None)
    if init is not None:
        n = len(init)
        h_best = np.array(init, dtype=float).copy()
    else:
        n = 70
        h_best = np.full(n, 0.5)
    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    def time_left():
        return budget_s - (time.time() - t0)

    def eq_con(x):
        return np.sum(x) - target
    cons = [{'type': 'eq', 'fun': eq_con}]

    def exact_obj(x):
        return compute_c5(x, n)

    def lse_obj(x, beta=50.0):
        return logsumexp_obj(x, n, beta)

    # Phase 1: Subgradient descent with Polyak step
    best_c0 = 0.5
    alpha = 0.08
    stall = 0
    max_stall = 300

    while time_left() > 350 and stall < max_stall:
        g = subgradient(h_best, n, top_m=5)
        step = alpha * (best_c5 - best_c0) / (np.dot(g, g) + 1e-12)
        step = np.clip(step, 0.001, 0.5)
        h_new = project_box_sum(h_best - step * g, target)
        c_new = compute_c5(h_new, n)
        if c_new < best_c5 - 1e-13:
            best_c5 = c_new
            h_best = h_new
            stall = 0
            alpha = min(alpha * 1.05, 0.15)
        else:
            stall += 1
            alpha *= 0.97
            if alpha < 0.01:
                alpha = 0.08
                h_pert = project_box_sum(h_best + rng.normal(0, 0.06, n), target)
                c_p = compute_c5(h_pert, n)
                if c_p < best_c5:
                    h_best, best_c5 = h_pert, c_p
                    stall = 0

    # Phase 2: Alternating smooth SLSQP + exact subgradient
    stall2 = 0
    for it in range(60):
        if time_left() < 200:
            break
        # Smooth SLSQP with increasing beta
        beta = 30.0 + 10.0 * (it % 6)
        h_start = h_best if it == 0 else project_box_sum(
            h_best + rng.normal(0, 0.02 + 0.005 * it, n), target)
        try:
            r = minimize(lse_obj, h_start, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         args=(beta,),
                         options={'maxiter': 200, 'ftol': 1e-13})
            hc = repair(r.x, n)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-14:
                h_best, best_c5 = hc, c
                stall2 = 0
            else:
                stall2 += 1
        except Exception:
            stall2 += 1

        # Exact subgradient burst
        for _ in range(10):
            if time_left() < 210:
                break
            g = subgradient(h_best, n, top_m=3)
            h_new = project_box_sum(h_best - 0.03 * g, target)
            c_new = compute_c5(h_new, n)
            if c_new < best_c5 - 1e-14:
                h_best, best_c5 = h_new, c_new
                stall2 = 0

        if stall2 > 15:
            # Larger kick
            h_pert = project_box_sum(h_best + rng.normal(0, 0.1, n), target)
            c_p = compute_c5(h_pert, n)
            if c_p < best_c5:
                h_best, best_c5 = h_pert, c_p
                stall2 = 0
            else:
                h_pert2 = project_box_sum(1.0 - h_best + rng.normal(0, 0.03, n), target)
                c_p2 = compute_c5(h_pert2, n)
                if c_p2 < best_c5:
                    h_best, best_c5 = h_pert2, c_p2
                    stall2 = 0

    # Phase 3: Exact SLSQP polish (multiple tries)
    for attempt in range(8):
        if time_left() < 80:
            break
        if attempt < 3:
            h_start = h_best.copy()
        else:
            h_start = project_box_sum(h_best + rng.normal(0, 0.02 * (attempt - 2), n), target)
        try:
            r = minimize(exact_obj, h_start, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 400, 'ftol': 1e-15})
            hc = repair(r.x, n)
            c = compute_c5(hc, n)
            if c < best_c5 - 1e-14:
                h_best, best_c5 = hc, c
        except Exception:
            pass

    # Phase 4: Coordinate descent (mass transfer between pairs)
    if time_left() > 60:
        improved = True
        while improved and time_left() > 40:
            improved = False
            for i in range(n):
                if time_left() < 45:
                    break
                # Try transferring mass to neighbors and random partners
                partners = [(i + 1) % n, (i - 1) % n, int(rng.randint(n))]
                for j in partners:
                    if j == i:
                        continue
                    for delta in (0.01, 0.03, 0.05, -0.01, -0.03, -0.05):
                        hp = h_best.copy()
                        hp[i] += delta
                        hp[j] -= delta
                        if hp[i] < -1e-9 or hp[i] > 1 + 1e-9 or hp[j] < -1e-9 or hp[j] > 1 + 1e-9:
                            continue
                        hp = np.clip(hp, 0, 1)
                        diff = target - np.sum(hp)
                        if abs(diff) > 1e-10:
                            continue
                        c = compute_c5(hp, n)
                        if c < best_c5 - 1e-14:
                            h_best, best_c5 = hp, c
                            improved = True
                            break

    # Phase 5: Final SLSQP with very tight tolerance
    if time_left() > 15:
        try:
            r = minimize(exact_obj, h_best, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 600, 'ftol': 1e-16})
            hf = repair(r.x, n)
            cf = compute_c5(hf, n)
            if cf < best_c5:
                h_best, best_c5 = hf, cf
        except Exception:
            pass

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
