    # As second factor (i+L=idx, i=idx-L): term h[idx-L]*(1-h[idx]) if 0<=idx-L<n

def lgbfgs_obj(x, n, dx, penalty):
    corr = np.correlate(x, 1.0 - x, mode='full')
    obj = float(np.max(corr) * dx)
    s = x.sum() - n / 2.0

def lgbfgs_grad(x, n, dx, penalty):
    # Subgradient of max(corr) * dx
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    active = np.where(np.abs(corr - max_c) < 1e-10)[0]
    grad = np.zeros(len(x))
    for k_idx in active:
        L = int(k_idx) - (len(x) - 1)
        for p in range(len(x)):
            if 0 <= p + L < len(x):
                grad[p] += (1.0 - x[p + L])
            if 0 <= p - L < len(x):
                grad[p] += (-x[p - L])
    grad /= len(active)
    grad *= dx
    # Gradient of penalty term
    s = x.sum() - n / 2.0
    grad += 2.0 * penalty * s * np.ones(len(x))
    return grad

def smooth_obj(x, n, dx, beta, penalty):
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    log_s = np.log(np.sum(np.exp(beta * (corr - max_c))))
    obj = (max_c + log_s / beta) * dx
    s = x.sum() - n / 2.0
    obj += penalty * s * s

def smooth_grad(x, n, dx, beta, penalty):
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    w = exp_c / exp_c.sum()
    
    m = len(x)
    grad = np.zeros(m)
    for k_idx in range(len(corr)):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        L = k_idx - (m - 1)
        for p in range(m):
            if 0 <= p + L < m:
                grad[p] += wk * (1.0 - x[p + L])
            if 0 <= p - L < m:
                grad[p] += wk * (-x[p - L])
    grad *= dx
    s = x.sum() - n / 2.0
    grad += 2.0 * penalty * s
    return grad
