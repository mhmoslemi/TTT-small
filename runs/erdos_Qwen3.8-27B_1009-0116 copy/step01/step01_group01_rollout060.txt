from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx), c

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-12:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def binary_swap_search(h, n, max_iter, t0, t_limit, rng):
    """Local search via random swaps maintaining sum constraint."""
    h = h.copy()
    dx = 2.0 / n
    c, _ = np.correlate(h, 1.0 - h, mode='full'), None
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = c.max()
    best_h = h.copy()
    best_val = current_max
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        a = rng.integers(len(ones))
        b = rng.integers(len(zeros))
        a = ones[a]
        b = zeros[b]
        k = b - a  # the relative shift
        
        # Compute delta for all lags
        # c[m] where m = n-1+lag, lag ranges from -(n-1) to (n-1)
        # delta_c[lag] = h[a+lag] + h[a-lag] - h[b+lag] - h[b-lag] (with 0 for OOB)
        lags = np.arange(-(n-1), n)  # all lags
        # h[a+lag] for valid a+lag in [0,n-1]
        a_plus = a + lags
        a_plus_valid = (a_plus >= 0) & (a_plus < n)
        a_minus = a - lags
        a_minus_valid = (a_minus >= 0) & (a_minus < n)
        b_plus = b + lags

        b_minus = b - lags
        b_minus_valid = (b_minus >= 0) & (b_minus < n)
        
        delta = np.zeros(2*n-1)
        delta[a_plus_valid] += h[a_plus[a_plus_valid]]
        delta[a_minus_valid] += h[a_minus[a_minus_valid]]
        delta[b_plus_valid] -= h[b_plus[b_plus_valid]]
        delta[b_minus_valid] -= h[b_minus[b_minus_valid]]

            ones = np.where(h > 0.5)[0]
            zeros = np.where(h <= 0.5)[0]
            c = new_c
            current_max = new_max
            if current_max < best_val:

                best_h = h.copy()
    
    return best_h, best_val * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def local_search_binary(h, t_limit, rng, verbose=False):
    n = len(h)
    dx = 2.0 / n
    h = np.round(h)  # make binary
    h = repair(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    best_cmax = c.max()
    best_h = h.copy()
    
    ones_idx = list(np.where(h == 1.0)[0])
    zeros_idx = list(np.where(h == 0.0)[0])
    
    no_improve = 0
    while time.time() < t_limit and no_improve < 50000:
        a = ones_idx[rng.integers(len(ones_idx))]
        b = zeros_idx[rng.integers(len(zeros_idx))]
        
        # Compute new correlate
        h[a], h[b] = h[b], h[a]  # swap
        c_new = np.correlate(h, 1.0 - h, mode='full')
        new_max = c_new.max()
        
        if new_max < best_cmax - 1e-15:
            c = c_new
            best_cmax = new_max
            best_h = h.copy()
            no_improve = 0
            # Update indices
            # a was 1, now 0; b was 0, now 1
            ones_idx.remove(a)
            zeros_idx.append(a)
            zeros_idx.remove(b)
            ones_idx.append(b)
        else:
            h[a], h[b] = h[b], h[a]  # undo
            no_improve += 1
    
    return best_h, best_cmax * dx, c

def subgradient_step(h, c, n, lr, rng):
    """One subgradient step on h, projected to feasible set."""
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)
    # Gradient of c[k_star] w.r.t h
    grad = np.zeros(n)
    for j in range(n):
        # Contribution of h[j] to c[k_star]:
        # As "first" factor: h[j]*(1-h[j+lag]) if j+lag in range → grad gets (1-h[j+lag])
        # As "second" factor: h[j-lag]*(1-h[j]) if j-lag in range → grad gets -h[j-lag]
        if 0 <= j + lag < n:
            grad[j] += (1.0 - h[j + lag])
        if 0 <= j - lag < n:
            grad[j] -= h[j - lag]
    h_new = h - lr * grad
    h_new = repair(h_new, n)
    return h_new

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c)) * dx

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    diff2 = target - h.sum()
    if abs(diff2) > 5e-13:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.02
    for it in range(20000):
        if time.time() > t_limit:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        cur_max = c.max()
        k_star = int(np.argmax(c))
        w_lags = []
        for d in range(-n + 1, n):
            k2 = (n - 1) + d
            if 0 <= k2 < 2 * n - 1 and c[k2] > cur_max - 0.02 * dx:
                w_lags.append(k2)
        grad = np.zeros(n)
        for k2 in w_lags:
            grad += grad_c_at_k(h, k2)
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            h = h - lr * grad / (grad_norm / 0.5)
            h = repair(h, n)
        lr *= 0.9995
        v = compute_c5(h)
        if v < best_c5 - 1e-12:
            best_c5 = v
            best_h = h.copy()
    return best_h, best_c5

def try_periodic(n, p, t_limit, rng):
    if n % p != 0 or p < 4:
        return None, None
    from itertools import combinations
    best_h, best_c5 = None, None
    if p <= 12:
        for ones in combinations(range(p), p // 2):
            period = np.zeros(p)
            for i in ones:
                period[i] = 1.0
            h = np.tile(period, n // p)
            c5 = compute_c5(h)
            if best_c5 is None or c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    if best_h is not None and best_c5 < 0.45:
        h, c5 = local_search(best_h, min(t_limit, time.time() + 10), rng, 0.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    h0 = np.array(initial_h_values, dtype=float)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Binary local search from initial
    t1 = t0 + 0.28 * budget_s
    h, c5 = local_search(h0, t1, rng, sa_temp=0.015)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 2: Periodic + alternative starts
    t2 = t0 + 0.50 * budget_s
    for n_try in [n0, 256, 192, 128, 96]:
        if time.time() > t2:
            break
        for p in [4, 6, 8, 10, 12] if n_try % 2 == 0 else [4, 8]:
            if n_try % p != 0:
                continue
            hp, cp = try_periodic(n_try, p, min(t2, time.time() + 8), rng)
            if cp is not None and cp < best_c5:
                best_c5, best_h = cp, hp.copy()
    # Golden ratio start
    if time.time() < t2:
        phi = (1.0 + 5.0 ** 0.5) / 2.0
        for n_try in [n0, 200, 160]:
            if time.time() > t2:
                break
            x = np.arange(n_try)
            hg = ((x * (phi - 1.0)) % 1.0 < 0.5).astype(float)
            hg = repair(hg, n_try)
            hl, cl = local_search(hg, min(time.time() + 12, t2), rng, 0.01)
            if cl < best_c5:
                best_c5, best_h = cl, hl.copy()

    # Phase 3: Subgradient refinement
    t3 = t0 + 0.78 * budget_s
    h, c5 = subgradient_refine(best_h, t3, rng)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    # Phase 4: SLSQP polish
    t4 = t0 + 0.95 * budget_s
    n = len(best_h)
    dx = 2.0 / n
    def obj_smooth(x):
        xx = np.clip(x, 0.0, 1.0)
        c = np.correlate(xx, 1.0 - xx, mode='full')
        cmax = c.max()
        tp = 400.0
        return (cmax + np.log(np.sum(np.exp((c - cmax) * tp))) / tp) * dx
    def obj_jac(x):
        xx = np.clip(x, 0.0, 1.0)
        c = np.correlate(xx, 1.0 - xx, mode='full')
        cmax = c.max()
        tp = 400.0
        w = np.exp((c - cmax) * tp)
        w /= w.sum()
        wc = np.zeros(2 * n - 1)
        for k_i in range(2 * n - 1):
            if w[k_i] < 1e-8:
                continue
            wc += w[k_i] * grad_c_at_k(xx, k_i)
        return wc * dx
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n * 0.5}]
    try:
        res = minimize(obj_smooth, best_h.copy(), jac=obj_jac, method='SLSQP',
                       bounds=bnds, constraints=cons, options={'maxiter': 3000, 'ftol': 1e-14})
        hr = repair(res.x, n)
        vr = compute_c5(hr)
        if vr < best_c5:
            best_c5, best_h = vr, hr.copy()
    except Exception:
        pass

    # Phase 5: Final binary local search from polished solution
    t5 = t0 + 0.99 * budget_s
    if time.time() < t5:
        h, c5 = local_search(best_h, t5, rng, sa_temp=0.005)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    nf = len(best_h)
    best_h = repair(best_h, nf)
    c5_final = compute_c5(best_h)
    if c5_final < best_c5:
        best_c5 = c5_final
    return (best_h, best_c5, nf)
