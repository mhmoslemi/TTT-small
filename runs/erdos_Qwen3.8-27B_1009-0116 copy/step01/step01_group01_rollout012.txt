        v3 = compute_c5(h3)
        if v3 < best_c5:
            best_c5, best_h = v3, h3.copy()

    # Phase 4: Binary projection + CD
    if elapsed() < budget_s * 0.90:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = repair(h_bin, n)
        v_bin = compute_c5(h_bin)
        if v_bin < best_c5:
            best_c5, best_h = v_bin, h_bin.copy()
        h_f, v_f = coordinate_descent(h_bin, n, max_sweeps=40)
        h_f = repair(h_f, n)
        v_f = compute_c5(h_f)
        if v_f < best_c5:
            best_c5, best_h = v_f, h_f.copy()

    # Phase 5: Final SLSQP polish
    if elapsed() < budget_s * 0.95:
        T_GLOBAL = 800.0
        t_rem = budget_s * 0.95 - elapsed()
        maxiter = max(100, int(t_rem / 0.3))
        res = minimize(obj_wrap, best_h.copy(), method='SLSQP', jac=grad_wrap,
                       bounds=bnds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'iprint': 0})
        h2 = repair(res.x, n)
        v2 = compute_c5(h2)
        if v2 < best_c5:
            best_c5, best_h = v2, h2.copy()

    # Final CD polish
    if elapsed() < budget_s * 0.98:
        h, v = coordinate_descent(best_h, n, max_sweeps=20)
        h = repair(h, n)
        v = compute_c5(h)
        if v < best_c5:
            best_c5, best_h = v, h.copy()

    best_h = repair(best_h, n)
    c5_final = compute_c5(best_h)
    return (best_h, c5_final, n)
