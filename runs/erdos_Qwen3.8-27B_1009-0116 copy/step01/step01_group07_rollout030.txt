def _grad_smooth(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    amax = np.max(corr)
    e = np.exp(alpha * (corr - amax))
    w = e / np.sum(e)
    
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-15:
            continue
        k = p - (n - 1)  # actual lag
        if k >= 0:
            # (1-h[j+k]) for j in [0, n-k-1]
            # -h[j-k] for j in [k, n-1]
            grad[:n-k] += w[p] * (1.0 - h[k:])
            grad[k:] -= w[p] * h[:n-k]
        else:
            l = -k
            # (1-h[j-l]) for j in [l, n-1]  → (1-h[0:n-l])
            # -h[j+l] for j in [0, n-l-1]   → -h[l:n]
            grad[l:] += w[p] * (1.0 - h[:n-l])
            grad[:n-l] -= w[p] * h[l:]
    grad *= dx
    return grad

def _project(h, target_sum):
    # Project onto {0 <= x <= 1, sum(x) = target_sum}
    lo, hi = min(h.min() - 1.0, -1.0), max(h.max() + 1.0, 1.0)
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _proj(h, tgt):

def _pgd(h, alpha, lr, iters, tgt):
    best_c = _c5(h)
    cur_h = h.copy()
    for it in range(iters):
        g = _grad(cur_h, alpha)
        cur_h = _proj(cur_h - lr*g, tgt)
        c = _c5(cur_h)
        if c < best_c:
            best_c = c
            h = cur_h
        else:
            lr *= 0.5
            cur_h = h.copy()
        if lr < 1e-10: break
    return h, best_c

def _local_search(h, tgt, passes=10):
    n = len(h)
    best_c = _c5(h)
    for _ in range(passes):
        moved = False

                            break
                if moved and (i+j) % 3 == 0:  # occasional early exit

            if not moved:
                break
        if not moved:
            break
    return h, best_c
