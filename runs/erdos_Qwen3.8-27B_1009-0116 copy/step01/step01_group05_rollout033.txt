from itertools import combinations
from scipy.optimize import minimize
import time

def _c5_discrete(h, dx):
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()

def _exhaustive_small(n, target):
    """Enumerate all binary sequences of length n with exactly target ones."""
    best_c5 = float('inf')
    best_h = None
    dx = 2.0 / n
    ranges = range(len(range(n)))
    for combo in combinations(ranges, target):
        h = np.zeros(n)
        h[list(combo)] = 1.0
        c5 = _c5_discrete(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

from scipy.optimize import minimize
from itertools import combinations
import time

_G_N = 70
_G_DX = 2.0/70
_G_TARGET = 35.0
_G_EPS = 0.01

def _g_obj(x):
    cv = np.correlate(x, 1.0-x, mode="full") * _G_DX
    m = cv.max()
    e = _G_EPS
    return m + e * np.log(np.sum(np.exp((cv-m)/e)))

def _g_grad(x):
    cv = np.correlate(x, 1.0-x, mode="full") * _G_DX
    m = cv.max()
    e = _G_EPS
    w = np.exp((cv-m)/e)
    w /= w.sum()
    n = _G_N
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2*n-1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx] * _G_DX
        s = idx - (n-1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

    return x.sum() - _G_TARGET

    return np.ones(len(x))

_G_DX = 2.0/70
_G_TGT = 35.0
_G_EPS = 0.01

def _corr(h, n):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def _c5(h, n):
    return float(_corr(h, n).max())

def _proj(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _g_obj(x):
    cv = np.correlate(x, 1.0 - x, mode="full") * _G_DX
    m = cv.max()
    e = _G_EPS
    return m + e * np.log(np.sum(np.exp((cv - m) / e)))

def _g_grad(x):
    cv = np.correlate(x, 1.0 - x, mode="full") * _G_DX
    m = cv.max()
    e = _G_EPS
    w = np.exp((cv - m) / e)
    w /= w.sum()
    n = _G_N
    g = np.zeros(n)
    j = np.arange(n)

            continue

        s = idx - (n - 1)

        if vm.any():

        jp = j + s
        vp = (jp >= 0) & (jp < n)

    return g

    return x.sum() - _G_TGT

    global _G_N, _G_DX, _G_TGT, _G_EPS
    _G_N, _G_DX, _G_TGT = n, 2.0 / n, n / 2.0
    h, bh, bc = h0.copy(), h0.copy(), _c5(h0, n)
    cons = [{'type': 'eq', 'fun': _g_efun, 'jac': _g_ejac}]
    bnds = [(0.0, 1.0)] * n
    for eps in eps_list:
        if time.time() > t_end - 3:
            break
        _G_EPS = eps
        try:
            r = minimize(_g_obj, h, jac=_g_grad, bounds=bnds,
                        constraints=cons, method='SLSQP',
                        options={'maxiter': maxiter, 'ftol': 1e-18})
            h = _proj(r.x, n / 2.0)
            c = _c5(h, n)
            if c < bc:
                bc, bh = c, h.copy()
        except Exception:
            break
    return bh, bc
