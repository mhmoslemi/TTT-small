def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def compute_grad_lse(h, dx, beta):
    """Compute gradient of log-sum-exp of f_k values."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    f_vals = corr * dx  # shape (2n-1,)
    
    # Softmax weights
    f_shifted = f_vals - np.max(f_vals)
    p = np.exp(beta * f_shifted)
    p_sum = p.sum()
    p /= p_sum
    
    # Gradient: sum over k of p[k] * grad_f_k
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        lag = k - (n - 1)
        # grad[m] += p[k] * dx * [(1-h[m+lag])*I(0<=m+lag<n) - h[m-lag]*I(0<=m-lag<n)]
        if lag >= 0:
            valid_m1 = range(0, n - lag)  # m where m+lag < n
            valid_m2 = range(lag, n)  # m where m-lag >= 0
            for m in valid_m1:
                grad[m] += p[k] * dx * (1.0 - h[m + lag])
            for m in valid_m2:
                grad[m] += p[k] * dx * (-h[m - lag])

            best_h = h_new.copy()

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)

        lag = int(k) - (n - 1)
        if lag >= 0:
            grad[:n-lag] += wk * (1.0 - h[lag:])
            grad[lag:] += wk * (-h[:n-lag])
        else:
            pl = -lag
            grad[pl:] += wk * (1.0 - h[:n-pl])
            grad[:n-pl] += wk * (-h[pl:])
    return grad

    v_v = np.zeros(n)

            best_h = h.copy()
