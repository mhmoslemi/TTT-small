            if s < 1e-14:

    n = len(h)
    target = float(n) / 2.0
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    corr = corr_raw * dx
    max_c = np.max(corr)
    exp_sh = np.exp(beta * (corr - max_c))
    f = max_c + np.log(np.sum(exp_sh)) / beta
    p = exp_sh / np.sum(exp_sh)
    s = h.sum() - target
    f += penalty * s * s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            upper = n - 1 - lag
            idx1 = np.arange(0, upper + 1)
            grad[idx1] += pk * dx * (1.0 - h[idx1 + lag])
            idx2 = np.arange(lag, n)
            grad[idx2] += pk * dx * (-h[idx2 - lag])
        else:
            L = -lag
            idx1 = np.arange(L, n)
            grad[idx1] += pk * dx * (1.0 - h[idx1 - L])
            idx2 = np.arange(0, n - L)
            grad[idx2] += pk * dx * (-h[idx2 + L])
    if penalty > 0:
        grad += 2.0 * penalty * s
    return f, grad

def optimize_lbfgsb_smooth(h, n, dx, betas, maxiter=150, penalty_base=50.0):
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = eval_c5(h, dx)
    for i, beta in enumerate(betas):
        penalty = penalty_base * (1.0 + 5.0 * i / max(len(betas)-1, 1))
        def obj(x):
            return smooth_obj_grad(x, dx, beta, penalty)[0]
        def jac(x):
            return smooth_obj_grad(x, dx, beta, penalty)[1]
        try:
            res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8})
            h_new = project_feasible(res.x, n)
            c5_new = eval_c5(h_new, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_c5

def optimize_slsqp_smooth(h, n, dx, beta=60, maxiter=300):
    h = project_feasible(h, n)
    target = float(n) / 2.0
    def obj(x):
        return smooth_obj_grad(x, dx, beta, 0.0)[0]
    def jac(x):
        return smooth_obj_grad(x, dx, beta, 0.0)[1]
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                   'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        return h_new, eval_c5(h_new, dx)
    except Exception:
        return project_feasible(h, n), eval_c5(project_feasible(h, n), dx)

def optimize_slsqp_exact(h, n, dx, maxiter=200):
    h = project_feasible(h, n)
    target = float(n) / 2.0
    def obj(x):
        return eval_c5(x, dx)
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                   'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c = eval_c5(h_new, dx)
        return h_new, c
    except Exception:
        return project_feasible(h, n), eval_c5(project_feasible(h, n), dx)

def make_structural_init(n, rng, style):
    if style == 0:
        h = rng.random(n)
    elif style == 1:
        freq = rng.uniform(1.0, 4.0)
        phase = rng.uniform(0, 2 * np.pi)
        h = 0.5 + 0.45 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    elif style == 2:
        h = 0.5 + 0.4 * np.cos(4 * np.pi * np.arange(n) / n + rng.uniform(0, 3))
    elif style == 3:
        blk = int(rng.integers(max(3, n // 10), max(4, n // 3)))
        nv = (n + blk - 1) // blk
        vals = 0.2 + 0.6 * rng.random(nv)
        h = np.repeat(vals, blk)[:n]
    else:
        h = np.zeros(n)
        m = n // 2
        for i in range(m):
            j = (i * 3 + rng.integers(0, 2)) % n
            h[j] = 1.0
            h[i] = 0.0
    return project_feasible(np.clip(h, 0, 1), n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    # Phase 0: Initialize from parent
    n0 = len(initial_h_values)
    dx0 = 2.0 / n0
    best_h = project_feasible(np.array(initial_h_values, dtype=float), n0)
    best_c5 = eval_c5(best_h, dx0)
    
    # Quick polish
    if time.time() - t0 < budget_s - 120:
        h, c = optimize_slsqp_smooth(best_h, n0, dx0, beta=50, maxiter=150)
        if c < best_c5:
            best_h, best_c5 = h, c
        h, c = optimize_slsqp_exact(best_h, n0, dx0, maxiter=100)
        if c < best_c5:
            best_h, best_c5 = h, c
    
    # Phase 1: Multi-n search
    n_candidates = sorted(set([n0, 60, 70, 80, 90, 100, 120]))
    n_candidates = [n for n in n_candidates if n <= 150 and n >= 40]
    
    for n in n_candidates:
        if time.time() - t0 > budget_s - 40:
            break
        dx = 2.0 / n
        remaining = budget_s - 40 - (time.time() - t0)
        n_trials = max(2, min(12, int(remaining / 25)))
        
        # Build starting points
        if n == len(best_h):
            h_interp = best_h.copy()
        else:
            x_old = np.linspace(0, 2, len(best_h))
            x_new = np.linspace(0, 2, n)
            h_interp = project_feasible(np.interp(x_new, x_old, best_h), n)
        
        for trial in range(n_trials):
            if time.time() - t0 > budget_s - 15:
                break
            if trial == 0:
                h_start = h_interp
            elif trial == 1:
                h_start = project_feasible(h_interp + 0.15 * rng.randn(n), n)
            elif trial == 2:
                h_start = project_feasible(h_interp + 0.05 * rng.randn(n), n)
            elif trial == 3:
                h_start = make_structural_init(n, rng, 1)
            elif trial == 4:
                h_start = make_structural_init(n, rng, 3)
            else:
                style = (trial - 5) % 5
                h_start = make_structural_init(n, rng, style)
            
            # Smooth L-BFGS-B with increasing beta
            betas = [8.0, 20.0, 50.0, 100.0]
            h_opt, c_opt = optimize_lbfgsb_smooth(h_start, n, dx, betas, maxiter=120)
            
            # SLSQP polish (smooth)
            if time.time() - t0 < budget_s - 10:
                h_opt, c2 = optimize_slsqp_smooth(h_opt, n, dx, beta=80, maxiter=200)
                if c2 < c_opt:
                    c_opt = c2
            
            # Exact SLSQP
            if time.time() - t0 < budget_s - 8:
                h_fin, c_fin = optimize_slsqp_exact(h_opt, n, dx, maxiter=150)
                if c_fin < c_opt:
                    h_opt, c_opt = h_fin, c_fin
            
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
    
    # Phase 2: Intensive local refinement on best
    if time.time() - t0 < budget_s - 10:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        for _ in range(5):
            if time.time() - t0 > budget_s - 5:
                break
            for beta in [80, 120, 150]:
                if time.time() - t0 > budget_s - 5:
                    break
                h, c = optimize_slsqp_smooth(best_h, n_b, dx_b, beta=beta, maxiter=200)
                if c < best_c5:
                    best_c5 = c
                    best_h = h
            h, c = optimize_slsqp_exact(best_h, n_b, dx_b, maxiter=200)
            if c < best_c5:
                best_c5 = c
                best_h = h
    
    # Phase 3: Tiny perturbations + re-optimize
    if time.time() - t0 < budget_s - 8:
        n_b = len(best_h)
        dx_b = 2.0 / n_b
        for _ in range(20):
            if time.time() - t0 > budget_s - 5:
                break
            h_p = project_feasible(best_h + 0.02 * rng.randn(n_b), n_b)
            h_f, c_f = optimize_slsqp_smooth(h_p, n_b, dx_b, beta=100, maxiter=150)
            if c_f < best_c5:
                h_f2, c_f2 = optimize_slsqp_exact(h_f, n_b, dx_b, maxiter=100)
                if c_f2 < c_f:
                    h_f, c_f = h_f2, c_f2
            if c_f < best_c5:
                best_c5 = c_f
                best_h = h_f
    
    # Final
    n_ret = len(best_h)
    dx_ret = 2.0 / n_ret
    best_h = project_feasible(best_h, n_ret)
    best_c5 = eval_c5(best_h, dx_ret)
    return (best_h, best_c5, n_ret)
