```python
        # d corr[k] / d h[i] = -h[i+m_k] + (1-h[i-m_k]) where m_k = k-(n-1)
        # I.e., for position i, the gradient vector is:
        grad = np.zeros((n, n_corr))
        for i in range(n):
            m_start = (n-1) - i  # smallest k such that i + m_k >= 0, m_k = k-(n-1) >= -i
            # k ranges from (n-1)-i to 2*(n-1)-i
            # For the term -h[i+m_k]: valid when 0 <= i+m_k < n, i.e., -i <= m_k < n-i
            #   m_k = k-(n-1), so: (n-1)-i <= k < (n-1)+(n-1)-i = 2(n-1)-i
            #   And m_k >= -i means k >= (n-1)-i
            #   And m_k < n-i means k < 2(n-1)-i
            # For the term +(1-h[i-m_k]): valid when 0 <= i-m_k < n, i.e., i-n < m_k <= i
            #   m_k = k-(n-1), so: i-(n-1) < k <= (n-1)+i
            #   i.e., k > i-(n-1) and k <= (n-1)+i
            
        # This is getting complex. Let me just use the direct formula.
        # corr[k] = sum_{m: 0<=m<n, 0<=m+k-(n-1)<n} (1-h[m]) * h[m+k-(n-1)]
        # Let p = k-(n-1). corr[k] = sum_m (1-h[m]) * h[m+p] for valid m.
        # d corr[k]/d h[i]:
        #   = -h[i+p] (if 0<=i+p<n, the "1-h" part) + (1-h[i-p]) (if 0<=i-p<n, the "h" part)
        
        for i in range(n):
            g_i = np.zeros(n_corr)
            for k in range(n_corr):
                p = k - (n-1)
                g = 0.0
                if 0 <= i + p < n:
                    g += -h[i + p]
                if 0 <= i - p < n:
                    g += (1.0 - h[i - p])
                g_i[k] = g
            grad[i] = g_i
```

This precomputation is O(n * n_corr) = O(n²) which is fine.

Then for a move Δ from j to i (h[i] += Δ, h[j] -= Δ):
  new_corr[k] = corr[k] + Δ * (grad[i][k] - grad[j][k])
  
And max(new_corr) can be found in O(n_corr).

But wait, this is only valid for small Δ (linearization). For large changes, the second-order terms matter. Actually, since corr[k] is bilinear in h (sum of products h[a]*h[b]), changing two coordinates makes it quadratic. The linear approximation is exact only if the two changed coordinates never appear in the same product term.

For exactness, I should just recompute the affected corr values. When h[i] changes by Δ and h[j] changes by -Δ:
- corr[k] changes only if the term involves position i or position j.
- For position i at offset p: the terms are (1-h[i])*h[i+p] and (1-h[i-p])*h[i].
- After change: (1-h[i]+(-Δ))*h[i+p] + (1-h[i-p])*(h[i]+Δ) = old + Δ*(h[i-p] - h[i+p]) [if both valid]
  Wait: (1-h[i]-Δ)*h[i+p] - (1-h[i])*h[i+p] = -Δ*h[i+p]
  (1-h[i-p])*(h[i]+Δ) - (1-h[i-p])*h[i] = Δ*(1-h[i-p])
  Total from i: Δ*(1-h[i-p]) - Δ*h[i+p]

Similarly for j: -Δ*(1-h[j-p]) + Δ*h[j+p] (since h[j] changes by -Δ)

But if i+p = j or j+p = i, there's cross terms (the product h[i]*h[j] appears). Let me handle this:

If i+p = j: the term (1-h[i])*h[i+p] = (1-h[i])*h[j] becomes (1-h[i]-Δ)*(h[j]-Δ) = (1-h[i])*h[j] - Δ*(1-h[i]) - Δ*h[j] + Δ²
Change: -Δ*(1-h[i]) - Δ*h[j] + Δ² (using old values)

Similarly if j+p = i: (1-h[j])*h[j+p] = (1-h[j])*h[i] becomes (1-h[j]+Δ)*(h[i]+Δ) = (1-h[j])*h[i] + Δ*(1-h[j]) + Δ*h[i] + Δ²
Change: Δ*(1-h[j]) + Δ*h[i] + Δ²

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_raw(h):
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')))

def _c5_full(h):
    return _c5_raw(h) * (2.0 / len(h))

def _make_feasible(h, target):
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(h)
            for i in range(len(idx)):
                a = int(idx[i])
                room = 1.0 - h[a]
                add = min(room, diff)
                h[a] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            idx = np.argsort(h)[::-1]
            for i in range(len(idx)):
                a = int(idx[i])
                rem = min(h[a], -diff)
                h[a] -= rem
                diff += rem
                if abs(diff) < 1e-13:
                    break
    return np.clip(h, 0.0, 1.0)

def _pairwise_cd(h, max_passes=200, rng=None):
    n = len(h)
    h = h.copy()
    bc = _c5_raw(h)
    fns = (0.0, 0.25, 0.5, 0.75, 1.0, 0.15, 0.35, 0.65, 0.85)
    for p in range(max_passes):
        improved = False
        for i in range(n):
            hi_val = h[i]
            for j in range(i + 1, n):
                s = h[i] + h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-10:
                    continue
                span = hi - lo
                best_c = bc
                best_a = h[i]
                h[i] = hi_val
                for f in fns:
                    a = lo + f * span
                    if a < lo - 1e-12 or a > hi + 1e-12:
                        continue
                    a = min(max(a, lo), hi)
                    if abs(a - best_a) < 1e-14:
                        continue
                    h[i] = a
                    h[j] = s - a
                    c = _c5_raw(h)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_a = a
                h[i] = best_a
                h[j] = s - best_a
                if best_c < bc - 1e-15:
                    bc = best_c
                    improved = True
        if not improved:
            break
    return h, bc

    n = len(h)
    if rng is None:
        rng = np.random.RandomState(0)

        j = rng.randint(n)

        s = h[i] + h[j]
        lo = max(0.0, s - 1.0)
        hi = min(1.0, s)

            continue
        span = hi - lo
        f = fns[rng.randint(len(fns))]
        a = lo + f * span
        h[i] = a
        h[j] = s - a
        c = _c5_raw(h)
        if c < bc - 1e-15:
            bc = c
    return h, bc

def _smooth_obj(h, alpha, dx, target):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -2.0 / alpha, 2.0 / alpha)
    sm_val = (m + np.log(np.sum(np.exp(alpha * sh))) / alpha) * dx
    pen = 10000.0 * (np.sum(h) - target) ** 2
    return sm_val + pen

def _slsqp_opt(h, n, target, dx, alpha, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return _smooth_obj(x, alpha, dx, target)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return _make_feasible(res.x, target)
    except Exception:
        return h
```
