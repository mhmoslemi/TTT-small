    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        clipped = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(clipped)
        if s > tgt:
            lo = mid
        else:
            hi = mid

from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
    x = np.clip(x, 0.0, 1.0)
    return x

def symmetrize(h, n):
    hs = (h + 1.0 - h[::-1]) / 2.0
    return repair(hs, n)

def build_full(free, n):
    m = (n - 1) // 2
    h = np.zeros(n)
    h[:m] = free[:m]
    h[m] = 0.5
    h[m + 1:] = 1.0 - free[:m][::-1]
    return h

def extract_free(h, n):
    m = (n - 1) // 2
    return h[:m].copy()

def smooth_obj(free, n, dx, tau):
    h = build_full(free, n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    sh = np.clip((corr - mx) / tau, -50.0, 0.0)
    return dx * (mx + tau * np.log(np.sum(np.exp(sh))))

def optimize_reduced(free0, n, dx, time_budget, rng, method='L-BFGS-B'):

    best_free = free0.copy()
    h0 = build_full(free0, n)
    best_val = compute_c5(h0, dx)
    t0 = time.time()
    taus = [0.5, 0.25, 0.12, 0.06, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]

        if time.time() - t0 > time_budget - 2:
            break
        def obj(f):
            return smooth_obj(f, n, dx, tau)
        bounds = [(0.0, 1.0)] * m

                r = minimize(obj, free0, method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-8, 'factr': 1e8})
            else:

                             options={'maxiter': 1000, 'ftol': 1e-16})
            free0 = np.clip(r.x, 0.0, 1.0)
            h_new = build_full(free0, n)
            val = compute_c5(h_new, dx)
            if val < best_val:
                best_val = val
                best_free = free0.copy()
        except Exception:
            pass

import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_gn = 0
_gdx = 1.0
_gtau = 1.0

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
    return np.clip(x, 0.0, 1.0)

def symmetrize(h, n):
    return repair((h + 1.0 - h[::-1]) / 2.0, n)

def build_full(free, n):
    m = (n - 1) // 2
    h = np.empty(n)
    h[:m] = free[:m]
    h[m] = 0.5
    h[m+1:] = 1.0 - free[:m][::-1]
    return h

def extract_free(h, n):
    return h[: (n - 1) // 2].copy()

def _sobj(f):
    h = build_full(f, _gn)
    corr = np.correlate(h, 1.0 - h, mode="full")
    mx = np.max(corr)
    sh = np.clip((corr - mx) / _gtau, -50.0, 0.0)
    return _gdx * (mx + _gtau * np.log(np.sum(np.exp(sh))))

def _fobj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    mx = np.max(corr)
    sh = np.clip((corr - mx) / _gtau, -50.0, 0.0)
    return _gdx * (mx + _gtau * np.log(np.sum(np.exp(sh))))

def _ceq(x):
    return np.sum(x) - _gn / 2.0

def _ceq_jac(x):
    return np.ones(_gn)

def opt_reduced(free0, n, dx, tbud):

    _gn = n
    _gdx = dx
    m = (n - 1) // 2

            break

        bd = [(0.0, 1.0)] * m

            free0 = np.clip(r.x, 0.0, 1.0)
            improved = True
        except Exception:
            pass
        if not improved:
            try:
                r = minimize(_sobj, free0, method='SLSQP', bounds=bd,
                             options={'maxiter': 600, 'ftol': 1e-16})
                free0 = np.clip(r.x, 0.0, 1.0)
            except Exception:
                pass

        if v < best_v:
            best_v = v
            best_f = free0.copy()
    return build_full(best_f, n), best_v
