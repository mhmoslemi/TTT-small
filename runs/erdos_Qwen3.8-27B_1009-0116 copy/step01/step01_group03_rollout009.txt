import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full").astype(np.float64)

def _make_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n, dtype=np.float64)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def _to_binary(h, n):
    idx = np.argsort(-h)[:n // 2]
    out = np.zeros(n, dtype=np.float64)
    out[idx] = 1.0
    return out

def _swap_corr_change(n, h, i, j):
    """Compute correlation delta for swapping h[i] and h[j] (h[i]!=h[j])."""
    d_i = 1.0 - 2.0 * h[i]
    d_j = 1.0 - 2.0 * h[j]
    ms = min(i, j)
    me = max(2 * n - 2 - i, 2 * n - 2 - j)
    s = np.arange(ms, me + 1) - (n - 1)
    dc = np.zeros(me - ms + 1)
    js = i - s
    m1 = (js >= 0) & (js < n)
    js2 = i + s
    m2 = (js2 >= 0) & (js2 < n)
    if np.any(m1):
        dc[m1] += d_i * (1.0 - h[js[m1]])
    if np.any(m2):
        dc[m2] -= d_i * h[js2[m2]]
    js = j - s
    m1 = (js >= 0) & (js < n)
    js2 = j + s
    m2 = (js2 >= 0) & (js2 < n)
    if np.any(m1):
        dc[m1] += d_j * (1.0 - h[js[m1]])
    if np.any(m2):
        dc[m2] -= d_j * h[js2[m2]]
    return ms, me, dc

def _sa_swap(n, h_init, seed, t_budget, T0=2.0, Tend=0.003):

        h_parent = None

    n_main = len(h_parent) if h_parent is not None else 140
    best_h = None
    best_c5 = 1.0

    if h_parent is not None:
        h_init = _to_binary(h_parent, n_main)
    else:
        h_init = _make_binary(n_main, seed)

    # SA round 1: from parent / initial
    if _rem() > 80:
        ta = min(350, _rem() * 0.5)
        h1, c1 = _sa_swap(n_main, h_init, seed, ta)
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    # SA round 2: random start
    if _rem() > 80:
        h_r = _make_binary(n_main, seed + 7)
        ta = min(200, _rem() * 0.3)
        h2, c2 = _sa_swap(n_main, h_r, seed + 200, ta, T0=3.0)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # SA round 3: perturbed parent
    if best_h is not None and _rem() > 60:
        rng_p = np.random.RandomState(seed + 55)
        h_pert = best_h.copy()
        for _ in range(8):
            a = rng_p.randint(n_main)
            b = rng_p.randint(n_main)
            if a != b and h_pert[a] != h_pert[b]:
                h_pert[a], h_pert[b] = h_pert[b], h_pert[a]
        ta = min(150, _rem() * 0.25)
        h3, c3 = _sa_swap(n_main, h_pert, seed + 400, ta, T0=2.5)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Hill climb on best
    if best_h is not None and _rem() > 30:
        ta = min(200, _rem() * 0.3)
        h4, c4 = _hill_climb_fast(n_main, best_h, ta)
        if c4 < best_c5:
            best_c5, best_h = c4, h4

    # Try other n values
    n_alts = [n_main - 20, n_main + 10, n_main + 20]
    for n_a in n_alts:
        if n_a < 60 or _rem() < 50:
            continue
        if best_h is not None:
            xo = np.linspace(0, 2, len(best_h))
            xn = np.linspace(0, 2, n_a)
            h_interp = np.interp(xn, xo, best_h)
            h_ia = _to_binary(h_interp, n_a)
        else:
            h_ia = _make_binary(n_a, seed + n_a)
        ta = min(80, _rem() * 0.3)
        ha, ca = _sa_swap(n_a, h_ia, seed + n_a * 3, ta)
        if _rem() > 20:
            ha, ca = _hill_climb_fast(n_a, ha, min(20, _rem() * 0.15))
        if ca < best_c5:
            best_c5, best_h = ca, ha

    # Final SA polish
    if best_h is not None and _rem() > 25:
        n_f = len(best_h)
        rng_f = np.random.RandomState(seed + 999)
        h_pf = best_h.copy()
        for _ in range(4):
            a = rng_f.randint(n_f)
            b = rng_f.randint(n_f)
            if a != b and h_pf[a] != h_pf[b]:
                h_pf[a], h_pf[b] = h_pf[b], h_pf[a]
        ta = min(50, _rem() * 0.5)
        hf, cf = _sa_swap(n_f, h_pf, seed + 1234, ta, T0=1.0, Tend=0.001)
        if _rem() > 10:
            hf, cf = _hill_climb_fast(n_f, hf, min(20, _rem() * 0.3))
        if cf < best_c5:
            best_c5, best_h = cf, hf

    # Verify and finalize
    if best_h is None:
        best_h = _make_binary(140, seed)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
