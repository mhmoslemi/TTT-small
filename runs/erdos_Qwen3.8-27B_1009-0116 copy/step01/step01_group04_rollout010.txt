import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x.astype(float), 0.0, 1.0)
    s = x.sum()
    tgt = n * 0.5
    if s > 1e-14:
        x *= tgt / s
    x = np.clip(x, 0.0, 1.0)
    diff = tgt - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-9) & (x < 1.0 - 1e-9)
        c = free.sum()
        if c > 0:
            x[free] += diff / c
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_obj(x, dx, tau):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = corr.max()
    if tau < 1e-15:
        return m * dx
    sh = np.clip((corr - m) / tau, -100.0, 0.0)
    return (m + tau * np.log(np.sum(np.exp(sh)))) * dx

def slsqp(x0, n, dx, tau, mi):
    bd = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n * 0.5,
             'jac': lambda x: np.ones(n)}]
    try:
        r = minimize(lambda x: smooth_obj(x, dx, tau), x0,
                     method='SLSQP', bounds=bd, constraints=cons,
                     options={'maxiter': mi, 'ftol': 1e-18})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def corr_gradient(x, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        nv = n - lag
        grad[:nv] += (1.0 - x[lag:])
        grad[lag:] -= x[:nv]
    else:
        l2 = -lag
        nv = n - l2
        grad[:nv] += (1.0 - x[l2:])
        grad[l2:] -= x[:nv]
    return grad

def subgrad_polish(h, n, dx, n_steps, alpha_init):
    x = h.copy()
    alpha = alpha_init
    best_v = compute_c5(x, dx)
    for _ in range(n_steps):
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argmax(corr)
        lag = idx - (n - 1)
        g = corr_gradient(x, n, lag)
        x_new = repair(x - alpha * g, n)
        v_new = compute_c5(x_new, dx)
        if v_new < best_v:
            x = x_new
            best_v = v_new
            alpha = min(alpha * 1.03, 0.08)
        else:
            alpha *= 0.85
            if alpha < 1e-5:
                break
