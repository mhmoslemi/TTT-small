def _subgrad(h, dx, top_k=3):
    """Compute subgradient of max_k O(k) w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top_k shifts
    idx = np.argpartition(corr, -top_k)[-top_k:]
    top_corr = corr[idx]
    max_corr = np.max(top_corr)
    # Soft weighting
    tau = 0.01
    w = np.exp((top_corr - max_corr) / (tau * dx))
    w /= w.sum()
    
    g = np.zeros(n)
    for wi, ei in zip(w, idx):
        s = ei - (n - 1)  # shift
        if s == 0:
            g += wi * (1.0 - 2.0 * h)
        elif s > 0:

            g[:n-s] -= wi * h[s:]
        else:
            p = -s
            g[:n-p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n-p]
    return g * dx

    v = np.zeros(n)

import numpy as np
import time

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _project(h, target):
    h = np.clip(h, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c > 0:
            h[m] += d / c
    return np.clip(h, 0.0, 1.0)

def _subgrad_h(h, dx, top_k=5, tau_frac=0.05):
    n = len(h)
    corr = _corr(h)
    idx = np.argpartition(corr, -top_k)[-top_k:]
    vals = corr[idx]
    mx = vals.max()
    tau = max(tau_frac * dx, 1e-4)
    w = np.exp((vals - mx) / tau)
    w /= w.sum()
    g = np.zeros(n)
    for wi, ei in zip(w, idx):
        s = ei - (n - 1)
        if s == 0:
            g += wi * (1.0 - 2.0 * h)
        elif s > 0:
            g[s:] += wi * (1.0 - h[:-s])
            g[:n - s] -= wi * h[s:]
        else:
            p = -s
            g[:n - p] += wi * (1.0 - h[p:])
            g[p:] -= wi * h[:n - p]
    return g * dx

def _nesterov(h0, target, dx, n_iter=800, top_k=5):
    n = len(h0)
    x = h0.copy()
    y = h0.copy()
    best = _c5(h0)
    best_x = h0.copy()
    for t in range(1, n_iter + 1):
        g = _subgrad_h(y, dx, top_k)
        alpha = 2.0 / (t + 2.0)
        y_new = _project(y - alpha * g, target)
        beta = (1.0 + np.sqrt(1.0 + 4.0 * 4.0 / (n_iter + 1.0) ** 2)) / 2.0
        x_new = y_new + (beta - 1.0) / (4.0 / (n_iter + 1.0) + 1.0) * (y_new - x)
        x_new = _project(x_new, target)
        x = x_new
        y = y_new
        c = _c5(x)
        if c < best:
            best = c
            best_x = x.copy()
    return best_x, best

def _local(h, target, max_sweeps=200, time_limit=None, t0=None):
    n = len(h)
    dx = 2.0 / n
    c = _c5(h)
    for sweep in range(max_sweeps):
        if time_limit and time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            for di in (0.02, -0.02, 0.01, -0.01):

                    if j == i:
                        continue
                    h2 = h.copy()
                    h2[i] += di
                    h2[j] -= di
                    if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
                        c2 = _c5(h2)
                        if c2 < c - 1e-12:
                            h = h2
                            c = c2
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, c

                best_c5 = c_opt
                best_h = h_opt.copy()
                best_n = n
                print(f"  nesterov n={n} r={ci}: C5={c_opt:.6f}")

            if remaining() < 10:
                break
            h_loc, c_loc = _local_search(h_opt, target, max_sweeps=200,
                                         t_limit=max(0, remaining() - 5), t0=t0)
            if c_loc < best_c5:
                best_c5 = c_loc
                best_h = h_loc.copy()
                best_n = n
                print(f"  local n={n} r={ci}: C5={c_loc:.6f}")

            if remaining() < 8 or n > 100:
                continue
            h_bin, c_bin_final = _binarize_and_optimize(h_loc, target,
                                                        remaining() - 3, t0)
            if c_bin_final < best_c5:
                best_c5 = c_bin_final
                best_h = h_bin.copy()
                best_n = n
                print(f"  binary n={n} r={ci}: C5={c_bin_final:.6f}")

    tgt = best_n / 2.0
    diff = tgt - best_h.sum()
    if abs(diff) > 1e-13:
        m = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        cnt = int(m.sum())
        if cnt > 0:
            best_h[m] += diff / cnt
    best_h = np.clip(best_h, 0.0, 1.0)
    diff2 = tgt - best_h.sum()
    if abs(diff2) > 1e-14:
        best_h += diff2 / best_n
        best_h = np.clip(best_h, 0.0, 1.0)

    if remaining() > 15:
        h_final, c_final = _local_search(best_h, tgt, max_sweeps=100,
                                         t_limit=remaining() - 5, t0=t0)
        if c_final < best_c5:
            best_h = h_final
            best_c5 = c_final

    final_c5 = _c5(best_h)
    print(f"Final: n={best_n}, C5={final_c5:.6f}")
    return best_h, final_c5, best_n
