To minimize this, maximize ∑_k p_k ∑_i h[i]h[i+k] = h^T (∑_k p_k Q_k) h

where Q_k is the matrix with (Q_k)_{i,j} = 1 if j = i+k (mod appropriate).

So the inner problem is: maximize h^T P h subject to 0 ≤ h ≤ 1, ∑h = n/2, where P = ∑ p_k Q_k.

This is a quadratic maximization over a polytope - non-convex in general (P is not necessarily negative semidefinite).

But we can use Frank-Wolfe (conditional gradient) or projected gradient for this.

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _cd_pass(h, rng):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    order = rng.permutation(n)
    for m in order:
        h[m] = 0.0
        c_lo = np.correlate(h, 1.0 - h, mode='full')
        h[m] = 1.0
        c_hi = np.correlate(h, 1.0 - h, mode='full')
        coeff = c_hi - c_lo
        # Minimize max(c_lo + t*coeff) for t in [0,1]
        ts = np.linspace(0.0, 1.0, 40)
        vals = np.max(c_lo + ts[:, None] * coeff[None, :], axis=1)
        ib = np.argmin(vals)
        if ib == 0:
            lo_t, hi_t = 0.0, 1.0/39
        elif ib == 39:
            lo_t, hi_t = 38.0/39, 1.0
        else:
            lo_t, hi_t = ts[ib-1], ts[ib+1]
        for _ in range(25):
            m1 = lo_t + (hi_t - lo_t) * 0.382
            m2 = hi_t - (hi_t - lo_t) * 0.382
            if np.max(c_lo + m1 * coeff) <= np.max(c_lo + m2 * coeff):
                hi_t = m2
            else:
                lo_t = m1
        h[m] = (lo_t + hi_t) * 0.5
    # Renormalize
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        free = (h > 0.005) & (h < 0.995)
        if np.any(free):
            h[free] += diff / np.sum(free)
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp(h):
    n = len(h)
    dx = 2.0 / n
    tgt = n / 2.0
    alpha = 300.0
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        s = np.clip(c - m, -50.0/alpha, 50.0/alpha)
        return (m + np.log(np.sum(np.exp(alpha * s))) / alpha) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        r = minimize(obj, h, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': 80, 'ftol': 1e-14})
        return r.x
    except Exception:
        return h

def _fix_sum(h):
    target = len(h) / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)
    
    best_c5 = _c5(parent)
    best_h = parent.copy()
    best_n = len(parent)
    
    n_list = [50, 60, 70, 80, 90, 100, 120]
    
    for ni, n in enumerate(n_list):
        rem = budget_s - (time.time() - t0)
        if rem < 50:
            break
        
        # Starting points
        starts = []
        if n == best_n:
            starts.append(best_h.copy())
        starts.append(parent.copy() if n == len(parent) else
                     np.interp(np.linspace(0,2,n,endpoint=False),
                               np.linspace(0,2,len(parent),endpoint=False),
                               np.concatenate([parent, [parent[0]]])))
        # Random start
        hr = rng.uniform(0.15, 0.85, n)
        hr = _fix_sum(np.clip(hr, 0, 1))
        starts.append(hr)
        
        for si, h in enumerate(starts):
            rem = budget_s - (time.time() - t0)
            if rem < 40:
                break
            
            # Coordinate descent
            max_passes = min(400, int((rem - 30) / 0.02))
            for p in range(max_passes):
                rem = budget_s - (time.time() - t0)
                if rem < 30:
                    break
                h = _cd_pass(h, rng)
                c = _c5(h)
                if c < best_c5:
                    best_c5 = c
                    best_h = h.copy()
                    best_n = n
            
            # SLSQP polish
            rem = budget_s - (time.time() - t0)
            if rem > 35:
                h2 = _slsqp(_fix_sum(h))
                h2 = _fix_sum(np.clip(h2, 0, 1))
                c2 = _c5(h2)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h2.copy()

                    h = h2
                
                # Another CD round after SLSQP
                for p in range(100):

                    if rem < 30:
                        break
                    h = _cd_pass(h, rng)
                c3 = _c5(_fix_sum(h))
                if c3 < best_c5:
                    best_c5 = c3
                    best_h = _fix_sum(h).copy()
                    best_n = n
    
    best_h = _fix_sum(best_h)
    final_c5 = _c5(best_h)
    print(f"Final: n={best_n}, C5={final_c5:.6f}")
    return best_h, final_c5, best_n
