def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def lbgfsb_smooth(h, n, dx, beta, penalty, maxiter=200, gtol=1e-10):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        vals = corr * dx
        mv = np.max(vals)
        sh = np.clip(beta * (vals - mv), -50, 50)
        v = mv + np.log(np.exp(sh).sum()) / beta
        t = float(n) / 2.0
        v += penalty * (x.sum() - t) ** 2
        return v
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        vals = corr * dx
        mv = np.max(vals)
        sh = np.clip(beta * (vals - mv), -50, 50)
        exps = np.exp(sh)
        es = exps.sum()
        g = np.zeros(n)
        for ki in range(len(corr)):
            w = exps[ki] / es
            if w < 1e-14:
                continue
            lag = ki - (n - 1)
            wdx = w * dx
            if lag >= 0:
                g[:n - lag] += wdx * (1.0 - x[lag:])
                g[lag:] += wdx * (-x[:n - lag])
            else:

                g[:n + lag] += wdx * (-x[-lag:])
        t = float(n) / 2.0

        return g
    bounds = [(0.0, 1.0)] * n

        return eval_c5(x, dx)

        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': ceq, 'jac': cjac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho
    except Exception:
        return h

                    h = h_new
                    best_c5 = c
                    improved = True
                    break

    def f(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        mv = c.max()
        e = np.exp(np.clip(beta * (c - mv), -50, 50))
        return mv + np.log(e.sum()) / beta + pen * (x.sum() - t) ** 2
    def g(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        mv = c.max()
        e = np.exp(np.clip(beta * (c - mv), -50, 50))
        w = e / e.sum()
        gr = np.zeros(n)
        for k in range(len(c)):
            wk = w[k]
            if wk < 1e-14:
                continue
            lag = k - (n - 1)
            wdx = wk * dx
            if lag >= 0:
                gr[:n - lag] += wdx * (1.0 - x[lag:])
                gr[lag:] -= wdx * x[:n - lag]
            else:
                gr[-lag:] += wdx * (1.0 - x[:n + lag])
                gr[:n + lag] -= wdx * x[-lag:]
        gr += 2.0 * pen * (x.sum() - t)
        return gr

    try:
        r = minimize(f, h, method='L-BFGS-B', jac=g, bounds=bnd,
                     options={'maxiter': maxit, 'gtol': gtol})
        return r.x
    except Exception:
        return h

                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxit, 'ftol': ftol})

    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    tgt = float(n) / 2.0
    d = tgt - h.sum()
    if abs(d) > 1e-14:
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s > 1e-14:
                h += d * r / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += d * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = tgt - h.sum()
        if abs(d2) > 1e-14:
            i = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[i] = float(np.clip(h[i] + d2, 0.0, 1.0))
    return h

class OptCtx:
    def __init__(self, n, dx, beta, pen):
        self.n = n
        self.dx = float(dx)
        self.beta = float(beta)
        self.pen = float(pen)
        self.tgt = float(n) / 2.0
    def f(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * self.dx
        mv = c.max()
        e = np.exp(np.clip(self.beta * (c - mv), -50.0, 50.0))
        val = mv + np.log(e.sum()) / self.beta
        if self.pen > 0:
            val += self.pen * (x.sum() - self.tgt) ** 2
        return val
    def g(self, x):
        c = np.correlate(x, 1.0 - x, mode='full') * self.dx
        mv = c.max()
        e = np.exp(np.clip(self.beta * (c - mv), -50.0, 50.0))
        w = e / e.sum()
        n = self.n
        gr = np.zeros(n)
        for k in range(len(c)):
            wk = w[k]
            if wk < 1e-14:
                continue
            lag = k - (n - 1)
            wdx = wk * self.dx
            if lag >= 0:
                gr[:n - lag] += wdx * (1.0 - x[lag:])
                gr[lag:] -= wdx * x[:n - lag]
            else:
                gr[-lag:] += wdx * (1.0 - x[:n + lag])
                gr[:n + lag] -= wdx * x[-lag:]
        if self.pen > 0:
            gr += 2.0 * self.pen * (x.sum() - self.tgt)
        return gr
    def eq(self, x):
        return x.sum() - self.tgt
    def eqj(self, x):
        return np.ones(self.n)

def do_lbfgsb(h, n, dx, beta, pen, maxit, gtol):
    ctx = OptCtx(n, dx, beta, pen)
    bnd = [(0.0, 1.0)] * n
    try:
        res = minimize(ctx.f, h, method='L-BFGS-B', jac=ctx.g, bounds=bnd,
                       options={'maxiter': maxit, 'gtol': gtol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def do_slsqp(h, n, dx, beta, maxit, ftol):
    ctx = OptCtx(n, dx, beta, 0.0)
    bnd = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': ctx.eq, 'jac': ctx.eqj}]
    try:
        res = minimize(ctx.f, h, method='SLSQP', jac=ctx.g, bounds=bnd,
                       constraints=cons, options={'maxiter': maxit, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return project_feasible(h, n)

def push_binary(h, n, dx, thresh):
    hp = h.copy()
    hp[hp < thresh] = 0.0
    hp[hp > 1.0 - thresh] = 1.0
    return project_feasible(hp, n)
