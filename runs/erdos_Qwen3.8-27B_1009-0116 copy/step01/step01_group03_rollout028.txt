    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:

            inits.append(project_feasible(np.interp(xn, xo, best_h), ts))
        # Random binary
        hr = np.zeros(n)
        hr[rng.choice(n, n // 2, replace=False)] = 1.0
        inits.append(hr)
        # Structured
        if rem() > 100:
            inits.append(make_structured(n, rng))

        for h0 in inits:
            if rem() < 25:
                break
            h = h0.copy()
            is_parent = (n == n_init and h0 is inits[0])
            n_slsqp = 5 if is_parent else 3
            betas = [5.0, 20.0, 60.0, 200.0, 500.0][:n_slsqp]
            for beta in betas:
                if rem() < 10:
                    break
                h = slsqp_run(h, n, beta, maxiter=250)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

            if rem() > 60:
                src = best_h if len(best_h) == n else h
                sa_time = min(120 if is_parent else 50, rem() * 0.4)
                ht, ct = sa_continuous(src, n, ts, rng, sa_time)
                if ct < best_c5:
                    best_c5 = ct
                    best_h = ht.copy()

            if rem() > 20:
                src = best_h if len(best_h) == n else h
                sweeps = 15 if is_parent else 8
                hc, cc = coord_descent(src, n, ts, rng, max_sweeps=sweeps)
                if cc < best_c5:
                    best_c5 = cc
                    best_h = hc.copy()

    # Final polish
    if rem() > 15:
        nb = len(best_h)
        hf = slsqp_run(best_h, nb, 800.0, maxiter=600)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_h = hf
            best_c5 = cf
        if rem() > 10:
            hf2 = slsqp_run(best_h, nb, 1500.0, maxiter=400)
            cf2 = compute_c5(hf2)
            if cf2 < best_c5:
                best_h = hf2
                best_c5 = cf2

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
