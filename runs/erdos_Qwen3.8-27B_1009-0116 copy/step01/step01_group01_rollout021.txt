            order = rng.permutation(n0)
            for i in order:
                old = h[i]
                best_t = old
                best_v = best_c5
                deltas = np.array([-0.04, -0.025, -0.015, -0.01, -0.005,
                                   0.005, 0.01, 0.015, 0.025, 0.04])
                for dlt in deltas:
                    t = old + dlt
                    if t < -0.01 or t > 1.01:
                        continue
                    t = np.clip(t, 0.0, 1.0)
                    if abs(t - old) < 1e-9:
                        continue
                    h[i] = t
                    d = n0 * 0.5 - h.sum()
                    if abs(d) > 1e-14:
                        if d > 0:
                            slack = 1.0 - h
                            slack[i] = 0.0
                            ts = slack.sum()
                            if ts > 0:
                                h += (d / ts) * slack
                        else:
                            slack = h.copy()
                            slack[i] = 0.0
                            ts = slack.sum()
                            if ts > 0:
                                h += (d / ts) * slack
                        h = np.clip(h, 0.0, 1.0)
                        if abs(n0 * 0.5 - h.sum()) > 1e-14:
                            h = repair(h, n0)
                    v = compute_c5(h)
                    if v < best_v - 1e-14:
                        best_v = v
                        best_t = t
                h[i] = best_t
                if best_t != old:
                    h = repair(h, n0)
                    best_c5 = best_v
                    improved = True
            if not improved:
                break
        if compute_c5(h) < best_c5:
            best_c5, best_h = compute_c5(h), h.copy()

    # Phase 3: SLSQP polish
    if time_ok(0.45):
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n0 * 0.5,
                 'jac': lambda x: np.ones(n0)}]
        bnds = [(0.0, 1.0)] * n0
        res = minimize(obj_c5, best_h.copy(), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-15})
        hk = repair(res.x, n0)
        vk = compute_c5(hk)
        if vk < best_c5:
            best_c5, best_h = vk, hk.copy()

    # Phase 4: Resolution increase to 2*n0
    if time_ok(0.5):
        n1 = n0 * 2
        xo = np.linspace(0.0, 2.0, len(best_h))
        xn = np.linspace(0.0, 2.0, n1)
        h1 = np.interp(xn, xo, best_h)
        h1 = repair(h1, n1)
        cands_mid = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
        h1, c5_1 = coord_descent(h1, n1, 150, cands_mid, rng, time_ok)
        if c5_1 < best_c5:
            best_c5, best_h = c5_1, h1.copy()

        # Fine CD at n1
        if time_ok(0.65) and c5_1 < best_c5:
            for sweep in range(80):
                if not time_ok(0.65):
                    break
                improved = False
                for i in range(n1):
                    old = h1[i]
                    best_t = old
                    best_v = c5_1
                    deltas = np.array([-0.03, -0.015, -0.008, 0.008, 0.015, 0.03])
                    for dlt in deltas:
                        t = np.clip(old + dlt, 0.0, 1.0)
                        if abs(t - old) < 1e-9:
                            continue
                        h1[i] = t
                        d = n1 * 0.5 - h1.sum()
                        if abs(d) > 1e-14:
                            if d > 0:
                                slack = 1.0 - h1
                                slack[i] = 0.0
                                ts = slack.sum()
                                if ts > 0:
                                    h1 += (d / ts) * slack
                            else:
                                slack = h1.copy()
                                slack[i] = 0.0
                                ts = slack.sum()
                                if ts > 0:
                                    h1 += (d / ts) * slack
                            h1 = np.clip(h1, 0.0, 1.0)
                            if abs(n1 * 0.5 - h1.sum()) > 1e-14:
                                h1 = repair(h1, n1)
                        v = compute_c5(h1)
                        if v < best_v - 1e-14:
                            best_v = v
                            best_t = t
                    h1[i] = best_t
                    if best_t != old:
                        h1 = repair(h1, n1)
                        c5_1 = best_v
                        improved = True
                if not improved:
                    break
            if c5_1 < best_c5:
                best_c5, best_h = c5_1, h1.copy()

    # Phase 5: Try resolution 3*n0 if time permits
    if time_ok(0.7) and len(best_h) <= 800:
        n2 = n0 * 3
        if n2 <= 1500:
            xo = np.linspace(0.0, 2.0, len(best_h))
            xn = np.linspace(0.0, 2.0, n2)
            h2 = np.interp(xn, xo, best_h)
            h2 = repair(h2, n2)
            cands_s = np.array([0.0, 0.5, 1.0])
            h2, c5_2 = coord_descent(h2, n2, 80, cands_s, rng, time_ok)
            if c5_2 < best_c5:
                best_c5, best_h = c5_2, h2.copy()

    # Phase 6: Final SLSQP on best
    if time_ok(0.85):
        nf = len(best_h)
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - nf * 0.5,
                 'jac': lambda x: np.ones(nf)}]
        bnds = [(0.0, 1.0)] * nf
        res = minimize(obj_c5, best_h.copy(), method='SLSQP', bounds=bnds,
                       constraints=cons, options={'maxiter': 200, 'ftol': 1e-15})
        hk = repair(res.x, nf)
        vk = compute_c5(hk)
        if vk < best_c5:
            best_c5, best_h = vk, hk.copy()

    # Phase 7: One more fine CD pass on final best
    if time_ok(0.92):
        nf = len(best_h)
        cands_final = np.linspace(0.0, 1.0, 15)
        h_f, c5_f = coord_descent(best_h, nf, 100, cands_final, rng, time_ok)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    best_h = repair(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
