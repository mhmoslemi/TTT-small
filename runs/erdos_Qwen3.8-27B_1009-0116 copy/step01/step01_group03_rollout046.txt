def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    np.clip(h, 0.0, 1.0, out=h)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-10)[0]
    else:
        idx = np.where(h > 1e-10)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    np.clip(h, 0.0, 1.0, out=h)
    diff2 = target_sum - h.sum()
    if abs(diff2) > 1e-8:
        free = np.where((h > 1e-8) & (h < 1.0 - 1e-8))[0]
        tgt = free[0] if len(free) > 0 else idx[0]
        h[tgt] += diff2
        np.clip(h, 0.0, 1.0, out=h)
    return h
