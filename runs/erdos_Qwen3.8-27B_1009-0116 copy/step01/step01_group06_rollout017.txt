import time
import math

_G_N = 70

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def _con_eq(x):
    return x.sum() - _G_N / 2.0

def _con_jac(x):
    return np.ones(len(x))

def lse_val_grad_base(x, dx, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    v_max = np.max(vals)
    exp_vals = np.exp(beta * (vals - v_max))
    w_sum = np.sum(exp_vals)
    obj = v_max + np.log(w_sum) / beta
    weights = exp_vals / w_sum
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = weights[k_idx]
        if w < 1e-14:
            continue
        lag = k_idx - (n - 1)
        for m in range(n):
            if 0 <= m + lag < n:
                grad[m] += w * (1.0 - x[m + lag])
            if 0 <= m - lag < n:
                grad[m] -= w * x[m - lag]
    grad *= dx
    return obj, grad

def obj_lse_pen(x, params):
    dx, beta, mu, n = params
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    v_max = np.max(vals)
    base = v_max + np.log(np.sum(np.exp(beta * (vals - v_max)))) / beta
    pen = mu * (x.sum() - n / 2.0) ** 2
    return base + pen

def jac_lse_pen(x, params):
    dx, beta, mu, n = params
    obj, grad = lse_val_grad_base(x, dx, beta)
    grad += 2.0 * mu * (x.sum() - n / 2.0)
    return grad

def slsqp_obj(x):
    return eval_c5(x, 2.0 / _G_N)

def slsqp_jac(x):
    # Use subgradient of max
    dx = 2.0 / _G_N
    n = _G_N
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-10)[0]
    grad = np.zeros(n)
    for k_idx in active:
        lag = k_idx - (n - 1)
        for m in range(n):
            if 0 <= m + lag < n:
                grad[m] += 1.0 - x[m + lag]
            if 0 <= m - lag < n:
                grad[m] -= x[m - lag]
    grad /= float(len(active))
    return dx * grad

def lse_anneal_lbfgs(h, n, dx, beta_list, mu=500.0, maxiter_per_beta=300):
    bounds = [(0.0, 1.0)] * n

        res = minimize(obj_lse_pen, h, jac=jac_lse_pen, args=params,

        h = project_feasible(res.x, n)

def slsqp_refine(h, n, dx, maxiter=2000):
    global _G_N
    _G_N = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]

import numpy as np

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def _lse_core(x, dx, beta):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    v_max = np.max(vals)
    exp_v = np.exp(beta * (vals - v_max))
    obj = v_max + np.log(np.sum(exp_v)) / beta
    weights = exp_v / np.sum(exp_v)
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        w = weights[ki]
        if w < 1e-15:
            continue
        lag = ki - (n - 1)
        for m in range(n):
            if 0 <= m + lag < n:
                grad[m] += w * (1.0 - x[m + lag])
            if 0 <= m - lag < n:
                grad[m] -= w * x[m - lag]
    grad *= dx
    return obj, grad
