    """Adjust h to have sum exactly target while staying in [0,1]."""
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    n = len(h)

    else:

    nf = np.sum(free)
    if nf == 0:
        return h
    step = diff / nf
    h[free] += step
    h = np.clip(h, 0.0, 1.0)
    return h

def _binary_search(h, target, rng, max_sweeps=100):

    n = len(h)

        rng.shuffle(idx)

                if cs < bc:
                    bc, bh, bn, best_par = cs, hs.copy(), n, hs
                hub = hs if cs < cb else hub
            if rem > 15:
                hp, cp = _perturb_regreedy(hub, tgt, rng, n_pert=rng.randint(2, 5))
                if cp < bc:
                    bc, bh, bn, best_par = cp, hp.copy(), n, hp
                    hub = hp
            if rem > 10:
                hc, cc = _coord_nb(hub, tgt, rng, its=35)
                if cc < bc:
                    bc, bh, bn, best_par = cc, hc.copy(), n, hc
            if rem > 12:
                hr, cr = _refine_smooth(hc if cc < (cs if rem > 18 else cb) else hub, tgt, dx)
                if cr < bc:
                    bc, bh, bn, best_par = cr, hr.copy(), n, hr
    tgt = bn / 2.0
    bh = _fix_sum(bh, tgt)
    d = tgt - bh.sum()
    if abs(d) > 1e-14:
        m = (bh > 1e-8) & (bh < 1.0 - 1e-8)
        if m.sum() > 0:
            bh[m] += d / m.sum()
            bh = np.clip(bh, 0.0, 1.0)
    c5f = _c5(bh)
    print(f"Final: n={bn}, C5={c5f:.6f}")
    return bh, c5f, bn
