```python
    """Coordinate descent with exact 1D minimization"""
    n = len(h)
```

```python
def _delta_c(h, i, di, j, dj, N):
    """Compute delta in correlation array when h[i] changes by di and h[j] by dj."""
    # c[m] indexed as c[m + N - 1] for m in [-(N-1), N-1]
    dc = np.zeros(2*N - 1)
    for m in range(-(N-1), N):
        # Contribution from i:
        term_i = 0.0
        if 0 <= i + m <= N - 1:
            term_i += (1.0 - h[i + m])
        if 0 <= i - m <= N - 1:
            term_i -= h[i - m]
        # Contribution from j:
        term_j = 0.0
        if 0 <= j + m <= N - 1:
            term_j += (1.0 - h[j + m])
        if 0 <= j - m <= N - 1:
            term_j -= h[j - m]
        dc[m + N - 1] = di * term_i + dj * term_j
    return dc
```

```python
def _delta_c_vec(h, i, di, j, dj, N):
    m = np.arange(-(N-1), N)  # shape (2N-1,)
    # For index i:
    valid_ip = (i + m >= 0) & (i + m <= N - 1)
    valid_im = (i - m >= 0) & (i - m <= N - 1)
    term_i = np.zeros(2*N-1)
    term_i[valid_ip] += 1.0 - h[i + m[valid_ip]]
    term_i[valid_im] -= h[i - m[valid_im]]
    # For index j:
    valid_jp = (j + m >= 0) & (j + m <= N - 1)
    valid_jm = (j - m >= 0) & (j - m <= N - 1)
    term_j = np.zeros(2*N-1)
    term_j[valid_jp] += 1.0 - h[j + m[valid_jp]]
    term_j[valid_jm] -= h[j - m[valid_jm]]
    return di * term_i + dj * term_j
```
