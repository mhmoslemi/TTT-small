def nesterov_opt(h, n, dx, T, maxiter=50000, lr=0.01, beta=0.9):
    target = float(n) / 2.0
    h = h.copy()
    v = np.zeros(n)
    best_c5 = eval_c5(h, dx)
    best_h = h.copy()
    
    for it in range(maxiter):
        # Nesterov lookahead
        y = h + beta * v
        y = np.clip(y, 0.0, 1.0)
        # Project to feasible set for gradient eval
        # (sum constraint)
        g = soft_max_grad(y, dx, T, n)
        v = beta * v - lr * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        
        c5 = eval_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def perturb_block(h, n, rng):
    h0 = h.copy()
    blen = rng.randint(5, min(15, n))
    start = rng.randint(0, n - blen)
    h0[start:start + blen] = 1.0 - h0[start:start + blen]
    return project_feasible(h0, n)

def perturb_5period(h, n, rng):
    h0 = h.copy()
    for p in range(0, n, 5):
        shift = rng.uniform(-0.15, 0.15)
        end = min(p + 5, n)
        h0[p:end] += shift
    return project_feasible(h0, n)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    n = 70
    dx = 2.0 / n

    best_h = np.array(initial_h_values, dtype=float).copy()
    best_h = project_feasible(best_h, n)
    best_c5 = eval_c5(best_h, dx)

    # Phase 1: Quick raw SLSQP
    if time.time() - t0 < budget_s - 250:
        h, c = slsqp_raw(best_h, n, dx, 800, 1e-14)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 2: Soft-max SLSQP annealing
    for T in [3, 5, 10, 20, 40, 80]:
        if time.time() - t0 > budget_s - 200:
            break
        h, c = slsqp_smo(best_h, n, dx, T, 400, 1e-13)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 3: Nesterov accelerated gradient on soft-max
    if time.time() - t0 < budget_s - 120:
        h, c = nesterov_opt(best_h, n, dx, 40, 40000, 0.004, 0.9)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 4: Raw SLSQP polish
    if time.time() - t0 < budget_s - 50:
        h, c = slsqp_raw(best_h, n, dx, 1500, 1e-15)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 5: Second soft-max round from improved point
    for T in [15, 30, 60]:
        if time.time() - t0 > budget_s - 60:
            break
        h, c = slsqp_smo(best_h, n, dx, T, 300, 1e-14)
        if c < best_c5:
            best_h, best_c5 = h, c
        h, c = slsqp_raw(best_h, n, dx, 500, 1e-14)
        if c < best_c5:
            best_h, best_c5 = h, c

    # Phase 6: Multi-start with diverse perturbations
    attempt = 0
    while time.time() - t0 < budget_s - 5:
        attempt += 1
        m = attempt % 7
        if m == 0:
            h0 = best_h + 0.12 * rng.randn(n)
        elif m == 1:
            h0 = best_h + 0.05 * rng.randn(n)
        elif m == 2:
            h0 = best_h + 0.02 * rng.randn(n)
        elif m == 3:
            h0 = perturb_block(best_h, n, rng)
        elif m == 4:
            h0 = perturb_5period(best_h, n, rng)
        elif m == 5:
            h0 = 1.0 - best_h  # complement (same sum constraint)
        else:
            h0 = rng.rand(n)
        h0 = project_feasible(h0, n)
        h, c = slsqp_raw(h0, n, dx, 200, 1e-13)
        if c < best_c5:
            best_h, best_c5 = h, c
        if c < best_c5 + 0.002 and time.time() - t0 < budget_s - 10:
            h2, c2 = slsqp_smo(h, n, dx, 25, 200, 1e-13)
            if c2 < best_c5:
                best_h, best_c5 = h2, c2
            h3, c3 = slsqp_raw(h2, n, dx, 300, 1e-14)
            if c3 < best_c5:
                best_h, best_c5 = h3, c3

    # Phase 7: Try n=80 if time permits
    if time.time() - t0 < budget_s - 60:
        n2 = 80
        dx2 = 2.0 / n2
        # Interpolate best_h to n2
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0 / n2
        h2_init = np.interp(x_new, x_old, best_h, period=2.0)
        h2_init = project_feasible(h2_init, n2)
        c2 = eval_c5(h2_init, dx2)
        if time.time() - t0 < budget_s - 30:
            h2_opt, c2_opt = slsqp_raw(h2_init, n2, dx2, 600, 1e-13)
            if c2_opt < best_c5:
                best_h, best_c5, _ = h2_opt, c2_opt, n2
                n = n2

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    dx_final = 2.0 / len(best_h)
    best_c5 = eval_c5(best_h, dx_final)
    n_final = len(best_h)
    return (best_h, best_c5, n_final)
