        top_k = np.argsort(corr)[-min(10, len(corr)):][::-1]
        # Minimize the max of top lags via SLSQP
        def top_lag_obj(x):
            cc = get_corr(x, n)
            return float(np.max(cc[top_k])) * dx

        def eqf(x):
            return np.sum(x) - n / 2.0
        cons = [{'type': 'eq', 'fun': eqf}]
        for trial in range(5):
            if tleft() < 15:
                break
            h_try = h_best + (rng.normal(0, 0.01, n) if trial > 0 else 0)
            h_try = repair(h_try, n)
            try:
                r = minimize(top_lag_obj, h_try, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 400, 'ftol': 1e-15})
                hc = repair(r.x, n)
                c = compute_c5(hc, n)
                if c < best_c5:
                    h_best, best_c5 = hc, c
            except Exception:
                pass

    # Phase 3: Final fine-grained polish
    if tleft() > 10:
        def eqf2(x):
            return np.sum(x) - n / 2.0
        cons2 = [{'type': 'eq', 'fun': eqf2}]
        for trial in range(4):
            if tleft() < 8:
                break
            sigma = 0.003 * (trial + 1)
            h_try = repair(h_best + rng.normal(0, sigma, n), n)
            try:
                r = minimize(lambda x: compute_c5(x, n), h_try, method='SLSQP',
                            bounds=bounds, constraints=cons2,
                            options={'maxiter': 500, 'ftol': 1e-16})
                hc = repair(r.x, n)
                c = compute_c5(hc, n)
                if c < best_c5:
                    h_best, best_c5 = hc, c
            except Exception:
                pass

        # Tiny coordinate refinement
        for _ in range(1000):
            if tleft() < 5:
                break
            i, j = int(rng.randint(n)), int(rng.randint(n))
            if i == j:
                continue
            for eps in [0.001, 0.002, 0.005]:
                for sign in [1, -1]:
                    hi = h_best[i] + sign * eps
                    hj = h_best[j] - sign * eps
                    if 0 <= hi <= 1 and 0 <= hj <= 1:
                        hp = h_best.copy()
                        hp[i], hp[j] = hi, hj
                        c = compute_c5(hp, n)
                        if c < best_c5 - 1e-15:
                            h_best, best_c5 = hp, c
                            break

    h_best = repair(h_best, n)
    best_c5 = compute_c5(h_best, n)
    return (h_best, best_c5, n)
