import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

def smooth_max_obj(h, t_param):
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(c)
    # Avoid overflow
    shifted = (c - cmax) * t_param
    shifted = np.clip(shifted, -500, 0)
    return (cmax + np.log(np.sum(np.exp(shifted))) / t_param) * (2.0 / len(h))

def exact_1d_min(c, n, i, h_i):
    """
    For coordinate i with current value h_i, find optimal t in [0,1]
    that minimizes max_l c_l (with sum constraint handled externally).
    Returns optimal t.
    """

    # lag l corresponds to offset l-(n-1) in the original indexing

    # c_full[m] = sum_j h[j] * (1-h)[j + m - (n-1)]
    # h[i] appears when j=i or j+m-(n-1)=i
    
    a = c.copy()  # Start with current c values
    b = np.zeros(2*n-1)
    
    # h[i] appears in c_full[m] when:
    # j = i: h[i]*(1-h[i+m-n+1]) → contributes h[i]*(1-h[i+m-n+1])

    # For each m, check if j=i is valid (i in [0,n-1], always true)
    # and if i+m-n+1 is in [0,n-1]
    # and if j=i+m-n+1 is in [0,n-1]

    # i+m-n+1 ranges i-n+1 to i+n-1

    # h[i+l] where l = m-n+1, so l ranges (m1_lo-n+1) to (m1_hi-n+1)
    # = (n-1-i-n+1) to (2n-2-i-n+1) = -i to n-1-i
    # h indices: i + l where l in [-i, n-1-i], so h indices in [0, n-1] ✓
    
    for m in range(m1_lo, m1_hi + 1):

        h_il = h[i + l]
        a[m] -= hi * (1 - h_il)  # remove old contribution
        b[m] += (1 - h_il)  # new coefficient of t

def sobj(h, tp):

def sgrad(h, tp):
