    m = corr.max()
    sh = np.clip(corr - m, -500.0 / alpha, 0.0)
    exp_sh = np.exp(alpha * sh)
    p = exp_sh / exp_sh.sum()
    dx = 2.0 / len(h)
    grad = np.correlate(p, 1.0 - 2.0 * h, mode='valid') * dx
    return grad

def _fista(h, S, alphas, iters_per_alpha, lr_base):
    """FISTA-style optimization with alpha schedule."""
    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    t = 1.0
    for alpha in alphas:
        for _ in range(iters_per_alpha):
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h_cur + ((t - 1.0) / t_new) * (h_cur - h_prev)
            y = np.clip(y, 0.0, 1.0)
            grad = _compute_grad(y, alpha)
            lr = lr_base / (alpha * n)
            h_new = _project_box_sum(y - lr * grad, S)
            h_prev = h_cur
            h_cur = h_new
            t = t_new
    return h_cur

def _greedy_pair_swap(h, S, n_rounds=20):
    """Swing-based local search: try small transfers between pairs."""
    n = len(h)
    bc = _c5(h)
    rng = np.random.RandomState(int(bc * 1e6))
    for _ in range(n_rounds):
        improved = False
        for _ in range(n * 3):
            i = rng.randint(n)
            j = (i + rng.randint(1, n)) % n
            if i == j:
                continue
            for d in (0.02, -0.02, 0.05, -0.05, 0.1, -0.1):
                h2 = h.copy()
                h2[i] += d
                h2[j] -= d
                if 0 <= h2[i] <= 1 and 0 <= h2[j] <= 1:
                    c2 = _c5(h2)
                    if c2 < bc - 1e-12:
                        h = h2
                        bc = c2
                        improved = True
                        break
        if not improved:
            break
    return h, bc
