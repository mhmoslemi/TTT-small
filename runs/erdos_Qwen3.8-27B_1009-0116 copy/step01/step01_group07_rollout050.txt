        if s > t:

    p = exps / exps.sum()

def c5_val(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def sm_obj(h, alpha):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    s = float(np.exp(alpha * (c - m)).sum())
    return (m + np.log(s) / alpha) * (2.0 / n)

def sm_grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    exps = np.exp(alpha * (c - m))
    p = exps / float(exps.sum())
    F = (m + np.log(float(exps.sum())) / alpha) * dx
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            v1 = idx[lag:]
            grad[v1] += pk * (1.0 - h[v1 - lag])
            v2 = idx[:n - lag]
            grad[v2] -= pk * h[v2 + lag]
        else:
            ll = -lag
            v1 = idx[:n - ll]
            grad[v1] += pk * (1.0 - h[v1 + ll])
            v2 = idx[ll:]
            grad[v2] -= pk * h[v2 - ll]
    grad *= dx
    return F, grad

def pgd(h0, target, seed, n_steps=300, budget_fn=None):
    n = len(h0)
    rng = np.random.RandomState(seed)
    alphas = [30, 60, 120, 250, 500, 1000, 2000, 4000, 8000]
    steps_per_alpha = max(20, n_steps // len(alphas))
    lr = 0.05
    h = h0.copy()
    
    for ai, alpha in enumerate(alphas):
        for s in range(steps_per_alpha):
            if budget_fn is not None and budget_fn() < 15:
                return h
            F, g = sm_grad(h, alpha)
            # Backtracking line search
            for ls in range(12):
                h_new = project_box_sum(h - lr * g, target)
                F_new = sm_obj(h_new, alpha)
                if F_new < F - 1e-14:
                    h = h_new
                    break
                lr *= 0.5
            else:
                pass
            lr = max(lr, 0.005) * 1.05  # slowly increase lr
            # Periodic perturbation
            if s % 100 == 99:
                for _ in range(5):
                    i = rng.randint(n)
                    j = rng.randint(n)
                    d = rng.uniform(-0.15, 0.15)
                    h2 = h.copy()
                    h2[i] += d
                    h2[j] -= d
                    h2 = project_box_sum(h2, target)
                    if sm_obj(h2, alpha) < F + 0.005:
                        h = h2
                        F = sm_obj(h, alpha)
    return h
