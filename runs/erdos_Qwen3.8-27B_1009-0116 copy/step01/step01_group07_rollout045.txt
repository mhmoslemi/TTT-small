Wait, I need to be more careful with the negative shift case.

For s < 0, let ss = -s:
C(-ss) = sum_{i=ss}^{n-1} h[i]*(1-h[i-ss])

d/dh[j]:
- j as "first" index: j in [ss, n-1], contrib = (1-h[j-ss])
- j as "second" index: i-ss = j, i = j+ss, i in [ss, n-1] → j in [0, n-1-ss], contrib = -h[j+ss]

So:
```python
ss = -s
g[ss:] += (1.0 - h[:-ss])  # h[j-ss] for j in [ss, n-1] → h[0:n-ss]
g[:n-ss] -= h[ss:]         # h[j+ss] for j in [0, n-ss-1] → h[ss:n]
```

Let me verify: n=4, ss=2 (s=-2):
C(-2) = h[2]*(1-h[0]) + h[3]*(1-h[1])

d/dh[0] = -h[2] (from second term of first product)
d/dh[1] = -h[3] (from second term of second product)
d/dh[2] = (1-h[0]) (from first term of first product)
d/dh[3] = (1-h[1]) (from first term of second product)

Using formula:
g[2:] += (1-h[:-2]) → g[2] += 1-h[0], g[3] += 1-h[1] ✓
g[:2] -= h[2:] → g[0] -= h[2], g[1] -= h[3] ✓

```python
import numpy as np
from scipy.optimize import minimize
import time

_g_target = 35.0
_g_alpha = 200.0

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _obj(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    sh = np.clip(corr - m, -1.0, 1.0)
    dx = 2.0 / len(h)
    return (m + np.log(np.sum(np.exp(_g_alpha * sh))) / _g_alpha) * dx + 1e4 * (np.sum(h) - _g_target) ** 2

def _eq(h):
    return np.sum(h) - _g_target

def _eq_jac(h):
    return np.ones_like(h)

def _subgrad_descent(h, target, n_iter=5000, t_end=99999.0):
    n = len(h)
    h = h.copy()
    best_c5 = _c5(h)
    for it in range(n_iter):
        if time.time() > t_end:
            break
        lr = 0.03 * (0.9998 ** it)
        corr = np.correlate(h, 1.0 - h, mode='full')
        s = int(np.argmax(corr)) - (n - 1)
        g = np.zeros(n)
        if s >= 0:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
        else:
            ss = -s
            g[ss:] += (1.0 - h[:-ss])
            g[:n - ss] -= h[ss:]
        h = h - lr * g
        h = np.clip(h, 0.0, 1.0)
        sc = target / max(np.sum(h), 1e-12)
        h = np.clip(h * sc, 0.0, 1.0)
        c = _c5(h)
        if c < best_c5:
            best_c5 = c
    return h, best_c5

def _local_search(h, target, t_end, seed=0):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    deltas = [0.003, 0.005, 0.01, 0.02, 0.05, 0.1, 0.15, 0.008, 0.03, 0.07]
    stall = 0
    stall_limit = 500
    
    while time.time() < t_end - 2:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice(deltas)
        if rng.rand() < 0.5:
            d = -d
        hi, hj = h[i] + d, h[j] - d
        if hi < -0.01 or hi > 1.01 or hj < -0.01 or hj > 1.01:
            continue
        h[i] = min(max(hi, 0.0), 1.0)
        h[j] = min(max(hj, 0.0), 1.0)
        new_c5 = _c5(h)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = h.copy()
            stall = 0
        else:
            h[i] -= d
            h[j] += d
            h[i] = min(max(h[i], 0.0), 1.0)
            h[j] = min(max(h[j], 0.0), 1.0)
            stall += 1
        
        if stall > stall_limit:
            # Random perturbation to escape
            for _ in range(rng.randint(3, 15)):
                ii = rng.randint(n)
                jj = rng.randint(n)
                if ii == jj: continue
                dd = rng.uniform(-0.15, 0.15)
                if 0 <= h[ii] + dd <= 1 and 0 <= h[jj] - dd <= 1:
                    h[ii] += dd
                    h[jj] -= dd
            sc = target / max(np.sum(h), 1e-12)
            h = np.clip(h * sc, 0.0, 1.0)
            stall = 0
            stall_limit = int(stall_limit * 1.4)
    
    return best_h, best_c5

def _sys_local_search(h, target, t_end, seed=0):
    """Systematic local search: scan all positions with all deltas."""
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5(h)
    deltas = [0.01, 0.02, 0.05, 0.1, 0.005, 0.03, 0.08, 0.15]
    adj = [1, -1, 2, -2, 3, -3, 4, -4]  # adjustment indices
    
    improved = True
    while improved and time.time() < t_end - 2:
        improved = False
        rng = np.random.RandomState(seed)
        order = rng.permutation(n)
        for i in order:
            if time.time() > t_end - 2:
                break
            for j in adj:
                jj = (i + j) % n
                if jj == i: continue
                for d in deltas:
                    hi, hj = h[i] + d, h[jj] - d
                    if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                        continue
                    h[i], h[jj] = hi, hj
                    new_c5 = _c5(h)
                    if new_c5 < best_c5 - 1e-13:
                        best_c5 = new_c5
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i], h[jj] = hi - d, hj + d
    
    return best_h, best_c5

def _interpolate(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)
    
    pn = len(parent)
    parent = np.clip(parent, 0.0, 1.0)
    target_p = pn / 2.0
    sc = target_p / max(np.sum(parent), 1e-12)
    parent = np.clip(parent * sc, 0.0, 1.0)
    
    best_h = parent.copy()
    best_c5 = _c5(parent)
    best_n = pn
    print(f"Initial: n={pn}, C5={best_c5:.6f}")
    
    # Phase 1: SLSQP + subgradient at various n
    n_list = [pn, 84, 100, 112, 126, 140]
    n_list = sorted(set(n for n in n_list if n <= 200))
    
    for n in n_list:
        if time.time() > t_end - 100:
            break
        target = n / 2.0
        _g_target = target
        
        if n == pn:
            h_init = parent.copy()
        else:
            h_init = _interpolate(parent, n)
            sc = target / max(np.sum(h_init), 1e-12)
            h_init = np.clip(h_init * sc, 0.0, 1.0)
        
        h_best_n = h_init.copy()
        c5_best_n = _c5(h_init)
        
        # SLSQP with increasing alpha
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _eq, 'jac': _eq_jac}]
        
        for alpha in [50, 100, 200, 500]:
            if time.time() > t_end - 80:
                break
            _g_alpha = alpha
            try:
                mi = 80 if n <= 100 else 50
                res = minimize(_obj, h_best_n, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': mi, 'ftol': 1e-10})
                hn = np.clip(res.x, 0.0, 1.0)
                sc = target / max(np.sum(hn), 1e-12)
                hn = np.clip(hn * sc, 0.0, 1.0)
                cn = _c5(hn)
                if cn < c5_best_n:
                    c5_best_n = cn
                    h_best_n = hn.copy()
            except Exception:
                pass
        
        # Subgradient descent
        if time.time() < t_end - 60:
            hs, cs = _subgrad_descent(h_best_n, target, n_iter=3000, t_end=t_end - 50)
            if cs < c5_best_n:
                c5_best_n = cs
                h_best_n = hs.copy()
        
        # Update global best
        if c5_best_n < best_c5:
            best_c5 = c5_best_n
            best_h = h_best_n.copy()
            best_n = n
            print(f"  n={n}: C5={c5_best_n:.6f}")
    
    # Phase 2: Local search on best solution
    if time.time() < t_end - 30:
        target_b = best_n / 2.0
        # Convert back to best_n if needed
        if len(best_h) > best_n:
            best_h = best_h[:best_n]
            sc = target_b / max(np.sum(best_h), 1e-12)
            best_h = np.clip(best_h * sc, 0.0, 1.0)
        
        # Random local search (most of the time)
        t_ls = t_end - 30
        h_ls, c5_ls = _local_search(best_h, target_b, t_ls, seed=seed)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            print(f"  Local search: C5={c5_ls:.6f}")
    
    # Phase 3: Systematic local search if time permits
    if time.time() < t_end - 10:
        target_b = best_n
