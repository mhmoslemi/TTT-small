from scipy.optimize import minimize
import time

def _c5_of(h):
    return np.correlate(h, 1.0 - h, mode="full").max()

def _c5_of_dx(h, dx):
    return np.correlate(h, 1.0 - h, mode="full").max() * dx

def _to_binary(h, target):
    """Convert to binary with exactly 'target' ones."""
    n = len(h)
    # Threshold at 0.5
    b = (h >= 0.5).astype(float)
    # Adjust count
    while b.sum() > target:
        # Find the 1 with smallest h value
        idxs = np.where(b == 1)[0]
        i = idxs[np.argmin(h[idxs])]
        b[i] = 0
    while b.sum() < target:
        # Find the 0 with largest h value
        idxs = np.where(b == 0)[0]
        i = idxs[np.argmax(h[idxs])]
        b[i] = 1
    return b

def _greedy_binary_swap(h, target, max_rounds=100):
    """Greedy: try all swaps, accept if improves max C(k)."""
    n = len(h)
    h = h.copy()
    improved = True

    while improved and rounds < max_rounds:
        improved = False
        rounds += 1
        c_max = _c5_of(h)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_gain = 1e-10

        order = idxs[np.argsort(h[idxs])]

import numpy as np

_GDX, _GN, _GEPS, _GTGT = 0.02, 70, 0.01, 35.0

def _corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _maxc(h):
    return np.max(_corr(h))

def _c5(h, dx):
    return _maxc(h) * dx

def _to_bin(h, tgt):
    b = (h >= 0.5).astype(float)
    cnt = int(b.sum())
    if cnt > tgt:
        idx = np.where(b == 1)[0]
        for i in idx[np.argsort(h[idx])][:cnt - tgt]:
            b[i] = 0.0
    elif cnt < tgt:
        idx = np.where(b == 0)[0]
        for i in idx[np.argsort(-h[idx])][:tgt - cnt]:
            b[i] = 1.0
    return b

def _proj(h, tgt):
    lo, hi = -3.0, 4.0
    for _ in range(80):
        m = 0.5 * (lo + hi)
        if np.clip(h - m, 0, 1).sum() > tgt:
            lo = m
        else:
            hi = m
    return np.clip(h - 0.5 * (lo + hi), 0, 1)

def _greedy(h, tgt, rounds=200):
    h = h.copy()
    for _ in range(rounds):
        cm = _maxc(h)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        bg, bi, bj = 1e-10, -1, -1
        for i in ones:
            h[i] = 0.0
            for j in zeros:
                h[j] = 1.0
                nm = _maxc(h)
                h[j] = 0.0
                if cm - nm > bg:
                    bg, bi, bj = cm - nm, i, j
            h[i] = 1.0
        if bi < 0:
            break
        h[bi], h[bj] = 0.0, 1.0
    return h

def _anneal(h, tgt, rng, t_budget, t0=2.0):

        h[i], h[j] = 0.0, 1.0
        nm = _maxc(h)
        if nm < cm or rng.random() < np.exp(-(nm - cm) / t):
            cm = nm
            stuck = 0

                    break
                a, b = int(rng.choice(o)), int(rng.choice(z))
                h[a], h[b] = 0.0, 1.0
            cm = _maxc(h)
            t = max(t * 3.0, 1.0)
            stuck = 0
        t *= 0.9997
    return bh, bm

def _sobj(x):
    cv = _corr(x)
    mx = cv.max()
    e = _GEPS
    return mx + e * np.log(np.sum(np.exp((cv - mx) / e)))

def _sgrad(x):
    cv = _corr(x)
    mx = cv.max()
    e = _GEPS
    w = np.exp((cv - mx) / e)
    w /= w.sum()
    n = _GN
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-14:
            continue
        wt = w[idx] * _GDX
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - x[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * x[jp[vp]]
    return g

def _efun(x):
    return x.sum() - _GTGT

def _ejac(x):
    return np.ones(len(x))

    global _GDX, _GN, _GEPS, _GTGT

                   bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': _efun, 'jac': _ejac}],
                   options={'maxiter': maxit, 'ftol': 1e-14})
    return _proj(res.x, tgt)
