    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:

def smooth_max_obj(h, n, beta):
    corr = get_corr(h)
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqp(h, n, ts, beta, maxiter=500):
    def obj(x):
        return smooth_max_obj(x, n, beta)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})

        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        delta = new_c5 - best_c5
        
        if delta < 0 or rng.random() < np.exp(-delta / T):

                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
        
        T *= cool

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-13:
        return h
    if diff > 0:
        idx = np.where(h < 1.0 - 1e-12)[0]
    else:
        idx = np.where(h > 1e-12)[0]
    if len(idx) == 0:
        return h
    h[idx] += diff / len(idx)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10 and len(idx) > 0:
        h[idx[0]] = np.clip(h[idx[0]] + diff2, 0, 1)
    return h

def smooth_obj(h, n, beta):
    corr = get_corr(h)
    m = corr.max()
    shifted = (corr - m) * beta
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * (2.0 / n)

def slsqp_opt(h, n, ts, beta, maxiter=400):
    def obj(x):
        return smooth_obj(x, n, beta)
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - ts}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones"""
    h_b = (h >= 0.5).astype(float)
    target = n // 2
    n_ones = int(h_b.sum())
    if n_ones < target:
        zeros = np.where(h_b == 0)[0]
        flip = zeros[np.argsort(-h[zeros])[:target - n_ones]]
        h_b[flip] = 1.0
    elif n_ones > target:
        ones = np.where(h_b == 1)[0]
        flip = ones[np.argsort(h[ones])[:n_ones - target]]
        h_b[flip] = 0.0
    return h_b

def anneal(h, n, rng, time_limit, t_init=0.06, t_min=0.0005, cool=0.9985, max_restarts=3):
    """Simulated annealing with swap moves, with restarts"""
    t_end = time.time() + time_limit
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_time = time.time()
    
    for restart in range(max_restarts):
        if time.time() > t_end:
            break
        h_cur = h.copy() if restart == 0 else best_h.copy()
        if restart > 0:
            # Perturb
            n_pert = max(3, n // 20)
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            for _ in range(n_pert):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        
        T = t_init
        no_improve = 0
        while time.time() < t_end and T > t_min:
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
            c5_new = compute_c5(h_cur)
            delta = c5_new - best_c5
            if delta < 0:
                best_c5 = c5_new
                best_h = h_cur.copy()
                best_time = time.time()
                no_improve = 0
            elif rng.random() < np.exp(-delta / T):
                no_improve += 1
            else:
                h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
                no_improve += 1
            T *= cool
            if no_improve > 5000:
                break
        h = best_h
    
    return best_h, best_c5

def guided_search(h, n, rng, time_limit):
    """Local search targeting worst lag"""
    t_end = time.time() + time_limit
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    while time.time() < t_end:
        corr = get_corr(h_cur)
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        
        if lag > 0:
            bad = [(i + lag, i) for i in range(n - lag) if h_cur[i + lag] == 1 and h_cur[i] == 0]
        elif lag < 0:
            L = -lag
            bad = [(i, i + L) for i in range(n - L) if h_cur[i] == 1 and h_cur[i + L] == 0]
        else:
            bad = []
        
        if bad:
            i, j = bad[rng.randint(len(bad))]
        else:
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        c5_new = compute_c5(h_cur)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(140)
        h_init[rng.choice(140, 70, replace=False)] = 1.0
    
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    def remaining():
        return budget_s - (time.time() - t_start)
    
    # Phase 1: SLSQP on parent
    ts = n_init / 2.0
    for beta in [10, 50, 200]:
        if remaining() < 20:
            break
        h_s = slsqp_opt(h_init.copy(), n_init, ts, beta, 300)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()
    
    # Phase 2: Binary local search on parent
    h_b = make_binary(best_h, n_init)
    t_alloc = min(remaining() * 0.3, 200)
    if t_alloc > 15:
        h_a, c5_a = anneal(h_b, n_init, rng, t_alloc)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()
    
    # Phase 3: Try other n values
    n_list = sorted(set([100, 120, n_init, 140, 160, 180, 200, 240, 280]))
    n_list = [n for n in n_list if n % 2 == 0]
    
    for n in n_list:
        if remaining() < 50:
            break
        if n == n_init:
            continue  # already handled
        
        # Interpolate
        xo = np.linspace(0, 2, len(best_h))
        xn = np.linspace(0, 2, n)
        h_n = np.interp(xn, xo, best_h)
        h_n = project_feasible(h_n, n / 2.0)
        
        # SLSQP
        if remaining() > 40:
            h_n = slsqp_opt(h_n, n, n / 2.0, 100, 200)
            h_n = slsqp_opt(h_n, n, n / 2.0, 400, 200)
            c5_s = compute_c5(h_n)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_n.copy()
        
        # Binary + anneal
        h_bn = make_binary(h_n, n)
        t_alloc = min(remaining() * 0.35, 120)
        if t_alloc > 15:
            h_an, c5_an = anneal(h_bn, n, rng, t_alloc)
            if c5_an < best_c5:
                best_c5 = c5_an
                best_h = h_an.copy()
            
            # Guided search
            if remaining() > 20:
                t_g = min(remaining() * 0.15, 40)
                h_gn, c5_gn = guided_search(h_an, n, rng, t_g)
                if c5_gn < best_c5:
                    best_c5 = c5_gn
                    best_h = h_gn.copy()

    if remaining() > 30:

        h_fin = slsqp_opt(best_h.copy(), n_best, n_best / 2.0, 500, 300)
        c5_f = compute_c5(h_fin)
        if c5_f < best_c5:
            best_c5 = c5_f

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            r = rng.random()
            if r < 0.75:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
                c5_new = compute_c5(h_cur)
                delta = c5_new - best_c5
                if delta < -1e-15:
                    best_c5 = c5_new
                    best_h = h_cur.copy()
                    no_imp = 0
                elif rng.random() < np.exp(-delta / T):
                    no_imp += 1
                else:
                    h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
                    no_imp += 1
            else:
                if len(ones) >= 2 and len(zeros) >= 2:
                    i1 = ones[rng.randint(len(ones))]
                    i2 = ones[rng.randint(len(ones))]
                    j1 = zeros[rng.randint(len(zeros))]
                    j2 = zeros[rng.randint(len(zeros))]
                    if i1 != i2 and j1 != j2 and i1 != j1 and i2 != j2:
                        h_cur[i1], h_cur[j1] = h_cur[j1], h_cur[i1]
                        h_cur[i2], h_cur[j2] = h_cur[j2], h_cur[i2]
                        c5_new = compute_c5(h_cur)
                        delta = c5_new - best_c5
                        if delta < -1e-15:
                            best_c5 = c5_new
                            best_h = h_cur.copy()
                            no_imp = 0
                        elif rng.random() < np.exp(-delta / T):
                            no_imp += 1
                        else:
                            h_cur[j1], h_cur[i1] = h_cur[i1], h_cur[j1]
                            h_cur[j2], h_cur[i2] = h_cur[i2], h_cur[j2]
                            no_imp += 1
            T *= cool
            if no_imp > 8000:
                break
    return best_h, best_c5

def guided_search(h, n, rng, time_limit):
    t_end = time.time() + time_limit
    h_cur = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    while time.time() < t_end:
        corr = get_corr(h_cur)
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        bad = []
        if lag > 0:
            for i in range(n - lag):
                if h_cur[i + lag] == 1 and h_cur[i] == 0:
                    bad.append((i + lag, i))
        elif lag < 0:
            L = -lag
            for i in range(n - L):
                if h_cur[i] == 1 and h_cur[i + L] == 0:
                    bad.append((i, i + L))
        if bad:
            i, j = bad[rng.randint(len(bad))]
        else:
            ones = np.where(h_cur == 1)[0]
            zeros = np.where(h_cur == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
        c5_new = compute_c5(h_cur)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_cur.copy()
        else:
            h_cur[i], h_cur[j] = h_cur[j], h_cur[i]
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float)
    else:
        h_init = np.zeros(140)
        h_init[rng.choice(140, 70, replace=False)] = 1.0

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t_start)

    # Phase 1: SLSQP on parent
    ts_init = n_init / 2.0
    for beta in [20, 100, 400]:
        if remaining() < 15:
            break
        h_s = slsqp_opt(h_init.copy(), n_init, ts_init, beta, 300)
        c5_s = compute_c5(h_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: Binary + anneal on parent
    h_b = make_binary(best_h, n_init)
    t_alloc = min(remaining() * 0.30, 250)
    if t_alloc > 20:
        h_a, c5_a = anneal(h_b, n_init, rng, t_alloc)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()
        # Guided
        if remaining() > 30:
            t_g = min(remaining() * 0.12, 60)
            h_g, c5_g = guided_search(h_a, n_init, rng, t_g)
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()

    # Phase 3: Other resolutions
    n_candidates = [100, 120, 140, 160, 180, 200, 240, 280, 320]
    n_candidates = [n for n in n_candidates if n % 2 == 0 and n != n_init]
    n_candidates.sort()

    for n in n_candidates:
        if remaining() < 60:
            break

        xo = np.linspace(0, 2, len(best_h))
        xn = np.linspace(0, 2, n)
        h_n = np.interp(xn, xo, best_h)
        h_n = project_feasible(h_n, n / 2.0)

        # SLSQP
        if remaining() > 50:
            h_n = slsqp_opt(h_n, n, n / 2.0, 80, 200)
            h_n = slsqp_opt(h_n, n, n / 2.0, 300, 200)
            c5_s = compute_c5(h_n)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_n.copy()

        # Binary + anneal
        h_bn = make_binary(h_n, n)
        t_an = min(remaining() * 0.30, 150)
        if t_an > 15:
            h_an, c5_an = anneal(h_bn, n, rng, t_an)
            if c5_an < best_c5:
                best_c5 = c5_an
                best_h = h_an.copy()
            # Guided
            if remaining() > 25:
                t_gn = min(remaining() * 0.10, 40)
                h_gn, c5_gn = guided_search(h_an, n, rng, t_gn)
                if c5_gn < best_c5:
                    best_c5 = c5_gn
                    best_h = h_gn.copy()
            # SLSQP polish from binary result
            if remaining() > 25:
                h_pn = slsqp_opt(h_an, n, n / 2.0, 500, 200)
                c5_pn = compute_c5(h_pn)
                if c5_pn < best_c5:
                    best_c5 = c5_pn
                    best_h = h_pn.copy()

    # Phase 4: Final refinement on best solution
    n_best = len(best_h)
    if remaining() > 35:
        h_fin = slsqp_opt(best_h.copy(), n_best, n_best / 2.0, 600, 400)
        c5_f = compute_c5(h_fin)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_fin
        h_fb = make_binary(best_h, n_best)
        if remaining() > 25:
            h_fa, c5_fa = anneal(h_fb, n_best, rng, min(remaining() * 0.4, 60))
            if c5_fa < best_c5:
                best_c5 = c5_fa
                best_h = h_fa
        if remaining() > 15:
            h_fg, c5_fg = guided_search(best_h, n_best, rng, min(remaining() * 0.3, 30))
            if c5_fg < best_c5:
                best_c5 = c5_fg
                best_h = h_fg

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
