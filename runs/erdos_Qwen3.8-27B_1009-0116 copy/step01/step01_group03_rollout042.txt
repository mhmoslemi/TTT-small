import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        if diff2 > 0:
            idx = np.argsort(h)[:min(int(abs(diff2)) + 1, len(h))]
        else:
            idx = np.argsort(-h)[:min(int(abs(diff2)) + 1, len(h))]
        for i in idx:
            h[i] = np.clip(h[i] + diff2 / len(idx), 0, 1)
        diff3 = ts - h.sum()
        if abs(diff3) > 1e-12:
            free = np.where(h < 1 - 1e-9)[0] if diff3 > 0 else np.where(h > 1e-9)[0]
            if len(free) > 0:
                h[free[0]] = np.clip(h[free[0]] + diff3, 0, 1)
    return h

def smooth_obj(h, n, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / n
    m = corr.max()
    shifted = np.clip((corr - m) * beta, -50, 50)
    return (m + np.log(np.sum(np.exp(shifted))) / beta) * dx

def optimize_lbfgs(h, n, ts, beta, maxiter=400):
    def obj(x):
        return smooth_obj(x, n, beta)
    res = minimize(obj, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'gtol': 1e-9})
    if res.x is not None and np.all(np.isfinite(res.x)):
        return project_feasible(res.x, ts)
    return project_feasible(h, ts)

def targeted_local_search(h, n, ts, max_time=60, rng=None):
    if rng is None:
        rng = np.random.RandomState(123)
    best_c5 = compute_c5(h)
    t0 = time.time()
    stall = 0
    while time.time() - t0 < max_time:
        improved = False
        order = rng.permutation(n)
        # Single bit flips in random order
        for j in order:
            orig = h[j]
            target = 1.0 if orig < 0.5 else 0.0
            if abs(target - orig) < 0.01:
                candidates = [v for v in (0.0, 1.0) if abs(v - orig) > 0.01]
            else:
                candidates = [target]
            for tv in candidates:
                h[j] = tv
                h = project_feasible(h, ts)
                c5t = compute_c5(h)
                if c5t < best_c5 - 1e-14:
                    best_c5 = c5t
                    improved = True
                else:
                    h[j] = orig
                    h = project_feasible(h, ts)
        # Targeted worst-lag moves
        corr = np.correlate(h, 1.0 - h, mode="full")
        wl = np.argmax(corr) - (n - 1)
        for _ in range(300):
            i = rng.randint(max(1, n - abs(wl) - 1))
            j = i + wl
            if j < 0 or j >= n:
                continue
            if h[j] > 0.5:
                oj = h[j]; h[j] = 0.0; h = project_feasible(h, ts)
                c5t = compute_c5(h)
                if c5t < best_c5 - 1e-14:
                    best_c5 = c5t; improved = True
                else:
                    h[j] = oj; h = project_feasible(h, ts)
            elif h[i] < 0.5:
                oi = h[i]; h[i] = 1.0; h = project_feasible(h, ts)
                c5t = compute_c5(h)
                if c5t < best_c5 - 1e-14:
                    best_c5 = c5t; improved = True
                else:
                    h[i] = oi; h = project_feasible(h, ts)
        # Block moves (preserves sum)
        for _ in range(200):
            blen = rng.randint(2, max(3, n // 10))
            a = rng.randint(0, n - blen)
            c = rng.randint(0, n - blen)

                continue
            block = h[a:a+blen].copy()

            h[a:a+blen] = dest
            h[c:c+blen] = block
            c5t = compute_c5(h)
            if c5t < best_c5 - 1e-14:
                best_c5 = c5t; improved = True
            else:
                h[a:a+blen] = block

        # Pair swaps

                continue
            hi, hj = h[i], h[j]
            h[i], h[j] = hj, hi
            h = project_feasible(h, ts)
            c5t = compute_c5(h)
            if c5t < best_c5 - 1e-14:
                best_c5 = c5t; improved = True

                h[i], h[j] = hi, hj

                    h[k] = 1.0 - h[k]
                h = project_feasible(h, ts)
                stall = 0
        else:
            stall = 0
    return h, best_c5
