        best_h, best_c5 = _coord_descent(best_h, ft, n_sweeps=30, n_ts=301)

    # Phase 6: SLSQP polish
    if time.time() - t0 < budget_s - 5:
        ft = best_n / 2.0
        ph = _slsqp_round(best_h, ft, [5000, 20000], maxiter=40)
        ph, pc = _refine_coord(ph, ft, n_ts=301, n_rounds=15)
        if pc < best_c5:
            best_c5 = pc
            best_h = ph.copy()

    # Final sum fix
    ft = best_n / 2.0
    diff = ft - np.sum(best_h)
    if abs(diff) > 1e-12:
        mask = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            best_h[mask] += diff / cnt
        best_h = np.clip(best_h, 0.0, 1.0)
        diff2 = ft - np.sum(best_h)
        if abs(diff2) > 1e-13:
            best_h = best_h - diff2 / best_n
            best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _c5(best_h)
    print(f"Final: n={best_n}, C5={final_c5:.6f}")
    return best_h, final_c5, best_n
