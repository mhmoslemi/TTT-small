    """One subgradient step on the max correlation."""
    corr = compute_corr(h, n)
    max_c = np.max(corr)
    if num_active is None:
        # Use lags within 5% of max
        threshold = max_c * 0.95
    else:
        sorted_idx = np.argsort(corr)[::-1]
        threshold_idx = sorted_idx[min(num_active, len(sorted_idx)-1)]

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def repair(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = max(1, int(np.sum(mask)))
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_obj(h, n, beta):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = beta * corr
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / beta * dx

def subgradient_direction(h, n, top_k=5):
    corr = compute_corr(h, n)

    # Weighted subgradient

    weights = np.maximum(weights, 0)
    if np.sum(weights) < 1e-15:
        weights = np.ones(len(active))
    weights = weights / np.sum(weights)

            if 0 <= jp < n:

            g[j] += w * term
    g -= np.mean(g)
