    """Aggressive coordinate descent with multiple step sizes and pair moves."""
    n = len(h)
    bc = _c5(h)

                for j_off in [1, -1, 2, -2]:

                        c = _c5(h2)
                        if c < bc - 1e-12:
                            h = h2
                            bc = c
                            improved = True
        if not improved:
            break
    return h, bc

import numpy as np
from scipy.optimize import minimize
import time

_glob_half = 35
_glob_dx = 2.0 / 70
_glob_alpha = 200.0

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _fix_int(h, target):
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    res = h - diff / len(h)
    res = np.clip(res, 0.0, 1.0)
    diff2 = target - np.sum(res)
    mask = (res > 1e-10) & (res < 1.0 - 1e-10)
    cnt = int(np.sum(mask))
    if cnt > 0:
        res[mask] += diff2 / cnt
        res = np.clip(res, 0.0, 1.0)
    return res

def _dual_grad(h, top_k, alpha):
    n = len(h)
    half = n // 2
    corr = _corr_arr(h)
    m = np.max(corr)
    top_idx = np.argsort(corr)[-top_k:]
    sh = (corr[top_idx] - m) * alpha
    w = np.exp(sh)
    s = np.sum(w)
    if s < 1e-30:
        return np.zeros(half)
    w = w / s
    grad_h = np.zeros(n)
    for i in range(len(top_idx)):
        wi = w[i]
        if wi < 1e-10:
            continue
        d = int(top_idx[i]) - (n - 1)
        for j in range(n):
            jp = j + d
            jm = j - d
            if 0 <= jp < n:
                grad_h[j] += wi * (1.0 - h[jp])
            if 0 <= jm < n:
                grad_h[j] += wi * (-h[jm])
    return grad_h[:half] - grad_h[half:]

def _pd_dual(h, max_iter, top_k):
    n = len(h)
    half = n // 2
    h = h.copy()
    h[:half] = (h[:half] + 1.0 - h[half:]) / 2.0
    h[half:] = 1.0 - h[:half]
    best_c = _c5(h)
    step = 0.008
    for it in range(max_iter):
        alpha_sm = 15.0 + 150.0 * it / max(1, max_iter // 3)
        tik = max(1, min(top_k, 1 + it // 250))
        gv = _dual_grad(h, tik, alpha_sm)
        gnorm = np.linalg.norm(gv)
        if gnorm < 1e-14:
            break
        v = h[:half]
        found = False
        a = step
        for _ in range(25):
            v_try = np.clip(v - a * gv, 0.0, 1.0)
            h_try = np.empty(n)
            h_try[:half] = v_try
            h_try[half:] = 1.0 - v_try
            c_try = _c5(h_try)
            if c_try <= best_c + 1e-13:
                h = h_try
                best_c = min(best_c, c_try)
                found = True
                break
            a *= 0.5
        if not found:
            step *= 0.8
            if step < 1e-11:
                break
    return h, best_c

def _greedy_dual(h, rounds):
    n = len(h)
    half = n // 2
    h = h.copy()
    h[:half] = (h[:half] + 1.0 - h[half:]) / 2.0
    h[half:] = 1.0 - h[:half]
    best_c = _c5(h)
    steps = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    for _ in range(rounds):
        changed = False
        for ss in steps:
            for i in range(half):
                for sign in [1, -1]:
                    v2 = h[:half].copy()
                    v2[i] += sign * ss
                    if v2[i] < 0.0 or v2[i] > 1.0:
                        continue
                    h2 = np.empty(n)
                    h2[:half] = v2
                    h2[half:] = 1.0 - v2
                    c2 = _c5(h2)
                    if c2 < best_c - 1e-13:
                        h = h2
                        best_c = c2
                        changed = True
                        break
                if changed:
                    break
            if changed:
                break
        if not changed:
            break
    return h, best_c

def _anneal_dual(h, n_iter, sd):
    n = len(h)
    half = n // 2
    h = h.copy()

    best_c = bc

        h2[half:] = 1.0 - v
        c2 = _c5(h2)
        delta = c2 - bc
        if delta < 0 or rng.random() < np.exp(-max(0.0, delta) / max(t * 0.005, 1e-12)):
            h = h2
            bc = c2
            if bc < best_c:
                best_c = bc
                best_h = h2.copy()
    return best_h, best_c

def _slsqp_obj_dual(v):
    n2 = _glob_half
    dx = _glob_dx
    al = _glob_alpha
    h = np.empty(2 * n2)
    h[:n2] = v
    h[n2:] = 1.0 - v
    corr = _corr_arr(h)
    m = float(np.max(corr))
    sh = np.clip(corr - m, -0.5, 0.5)
    return (m + np.log(np.sum(np.exp(al * sh))) / al) * dx

def _make_dual(h, half):
    h = h.copy()
    h[:half] = (h[:half] + 1.0 - h[half:]) / 2.0
    h[half:] = 1.0 - h[:half]
    return h

    x_dst = np.linspace(0.0, 2.0, n_dst, endpoint=False) + 1.0 / n_dst
    return np.interp(x_dst, x_src, h_src, period=2.0)
