import time

    return float(np.max(corr) * dx)

def _smooth_obj(x, alpha):
    corr = np.correlate(x, 1.0 - x, mode='full')
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 500)
    return m + np.log(np.sum(np.exp(s))) / alpha

def _smooth_grad(x, alpha, dx):
    n = len(x)
    eps = 1e-6
    grad = np.zeros(n)
    for i in range(n):
        xp = x.copy(); xp[i] += eps
        xm = x.copy(); xm[i] -= eps
        grad[i] = (_smooth_obj(xp, alpha) - _smooth_obj(xm, alpha)) / (2*eps)
    return grad

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, target / n)
    # Proportional scaling first
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                a = min(diff, 1.0 - h[idx])
                h[idx] += a
                diff -= a
            else:
                b = min(-diff, h[idx])
                h[idx] -= b
                diff += b
            if abs(diff) < 1e-12:
                break
    return h
