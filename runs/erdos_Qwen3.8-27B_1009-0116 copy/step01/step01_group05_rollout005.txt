import numpy as np
from scipy.optimize import minimize
import time

def _project(h, target):
    lo, hi = -4.0, 5.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def _c5(h, dx):
    return np.correlate(h, 1.0 - h, mode="full").max() * dx

def _smooth_obj(h, dx, eps):
    ac = np.correlate(h, h, mode="full")
    n = len(h)
    m = ac.min()
    return (n / 2.0 + eps * np.log(np.sum(np.exp(-(ac - m) / eps)))) * dx

def _smooth_grad(h, dx, eps):
    ac = np.correlate(h, h, mode="full")
    n = len(h)
    m = ac.min()
    w = np.exp(-(ac - m) / eps)
    w /= w.sum()
    j = np.arange(n)
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-13:
            continue
        s = idx - (n - 1)
        wt = -dx * w[idx]
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] += wt * h[jp[vp]]
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * h[jm[vm]]
    return g

def _slsqp_opt(h, dx, n, target, eps, maxiter=300):
    def f_obj(x):
        return _smooth_obj(x, dx, eps)
    def f_grad(x):
        return _smooth_grad(x, dx, eps)
    def c_eq(x):
        return x.sum() - target
    def c_eq_jac(x):
        return np.ones(len(x))
    res = minimize(f_obj, h, jac=f_grad,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}],
                   method='SLSQP',
                   options={'maxiter': maxiter, 'ftol': 1e-18})
    return _project(res.x, target)

def _proj_grad(h, dx, n, target, rng, t_end, n_iter=5000):
    best = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    for it in range(n_iter):
        if time.time() > t_end - 5:
            break
        eps = max(0.0001, 0.1 * (0.97 ** (it // 100)))
        g = _smooth_grad(h, dx, eps)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        # Backtracking line search
        alpha = min(1.0, 5.0 / (gn + 1e-8))
        c_cur = _c5(h, dx)
        for _ in range(20):
            h_new = _project(h - alpha * g, target)
            c_new = _c5(h_new, dx)
            if c_new < c_cur + 1e-15:
                h = h_new
                break
            alpha *= 0.5
        else:
            break
        c_new = _c5(h, dx)
        if c_new < best_c:
            best_c = c_new
            best = h.copy()
        # Periodic perturbation
        if it % 500 == 400:
            h = _project(best + rng.normal(0, 0.03, n), target)
    return best, best_c

def _subgrad_step(h, dx, n, target, lr, top_k=10):
    ac = np.correlate(h, h, mode="full")
    m = ac.min()
    # Focus on the top_k smallest autocorrelations
    idx_sorted = np.argsort(ac)[:top_k]
    rng2 = np.zeros(len(ac))
    rng2[idx_sorted] = 1.0 / top_k
    j = np.arange(n)
    g = np.zeros(n)
    for idx in idx_sorted:
        s = idx - (n - 1)
        wt = -dx * rng2[idx]
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] += wt * h[jp[vp]]
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * h[jm[vm]]
    return _project(h - lr * g, target)

def _subgrad_loop(h, dx, n, target, t_end, n_iter=4000):
    best = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    lr = 0.5
    for it in range(n_iter):
        if time.time() > t_end - 5:
            break
        top_k = min(2 ** (it // 200) + 2, 30)
        h_new = _subgrad_step(h, dx, n, target, lr, top_k)
        c_new = _c5(h_new, dx)
        c_cur = _c5(h, dx)
        if c_new <= c_cur:
            h = h_new
            lr = min(lr * 1.01, 2.0)
        else:
            lr = max(lr * 0.97, 0.01)
        if c_new < best_c:
            best_c = c_new
            best = h_new.copy()
        if it % 300 == 200:
            h = _project(best + rng_normal(n, 0.04, it), target)
    return best, best_c

def rng_normal(n, scale, seed):
    r = np.random.RandomState(seed)
    return r.normal(0, scale, n)

def _binary_init(n, target):
    idx = np.arange(n)
    np.random.shuffle(idx)
    h = np.zeros(n)
    h[idx[:int(target)]] = 1.0
    return h

def _binary_local_search(h_bin, dx, n, target, t_end, max_iter=30000):
    best = h_bin.copy()
    best_c = _c5(h_bin, dx)
    h = h_bin.copy()
    rng_ls = np.random.RandomState(123)
    for it in range(max_iter):
        if time.time() > t_end - 3:
            break
        # Pick a 1 and a 0 to swap
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i1 = ones[rng_ls.randint(len(ones))]
        i0 = zeros[rng_ls.randint(len(zeros))]
        h[i1] = 0.0
        h[i0] = 1.0
        c_new = _c5(h, dx)
        if c_new < best_c or (it < 100 and c_new < best_c + 0.01):
            best_c = c_new
            best = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
    return best, best_c

def _slsqp_sequence(h, dx, n, target, t_end, eps_list):
    best = h.copy()
    best_c = _c5(h, dx)
    h = h.copy()
    for eps in eps_list:

        c = _c5(h_new, dx)

            best_c = c
            best = h_new.copy()
        h = h_new
    return best, best_c
