import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_gdx = 1.0
_gtau = 1.0
_cN = 0
_gactive = None  # indices of active lags, None means all

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    if _gactive is not None:
        corr = corr[_gactive]
    m = np.max(corr)
    sh = np.clip((corr - m) / _gtau, -50.0, 0.0)
    s = np.sum(np.exp(sh))
    if s < 1e-300:
        s = 1.0
    return (m + _gtau * np.log(s)) * _gdx

def _cfun(x):
    return np.sum(x) - _cN / 2.0

def _cjac(x):
    return np.ones(_cN)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if s > 1e-12:
        x = x * (tgt / s)
        x = np.clip(x, 0.0, 1.0)
    s2 = np.sum(x)
    if abs(s2 - tgt) > 1e-8 and s2 > 1e-12:
        x = x * (tgt / s2)
        x = np.clip(x, 0.0, 1.0)
    # Final fix: if still off, add/subtract small amount
    diff = tgt - np.sum(x)
    if abs(diff) > 1e-10:
        # Distribute evenly where possible
        mask = (x > 1e-6) & (x < 1 - 1e-6)
        if np.any(mask):
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def slsqp_opt(x0, n, dx, tau, mi=500, ft=1e-14, active=None):
    global _gdx, _gtau, _cN, _gactive
    _cN = n
    _gdx = dx
    _gtau = tau
    _gactive = active
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        r = minimize(_obj, x0, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': mi, 'ftol': ft})
        return repair(r.x, n)
    except Exception:
        return repair(x0, n)

def refine_1d(x0, n, dx, time_limit, taus=None):
    t0 = time.time()
    if taus is None:
        taus = [0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    x = repair(x0, n).copy()
    best = x.copy()
    best_val = compute_c5(x, dx)
    for tau in taus:
        if time.time() - t0 > time_limit - 0.5:
            break
        x = slsqp_opt(x, n, dx, tau, 500, 1e-14)
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best = x.copy()
    return best, best_val

def refine_active(x0, n, dx, time_limit, top_k=8):
    """Optimize focusing on top-k correlation lags."""
    t0 = time.time()
    x = repair(x0, n).copy()
    best = x.copy()
    best_val = compute_c5(x, dx)
    taus = [0.05, 0.02, 0.01, 0.005, 0.001, 0.0005]
    for tau in taus:
        if time.time() - t0 > time_limit - 0.5:
            break
        # Identify active lags
        corr = np.correlate(x, 1.0 - x, mode="full")
        idx = np.argsort(corr)[::-1][:top_k]
        # Optimize on active set
        x = slsqp_opt(x, n, dx, tau, 300, 1e-14, active=idx)
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best = x.copy()
        # Also try full
        x_full = slsqp_opt(best, n, dx, tau * 0.5, 200, 1e-14, active=None)
        val_full = compute_c5(x_full, dx)
        if val_full < best_val:
            best_val = val_full
            best = x_full.copy()
    return best, best_val

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, x_src, kind='linear', fill_value='extrapolate')(xd)
    return repair(xp, n_dst)

def perturb(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def structured_perturb(x, n, rng):
    """Create diverse starting points."""
    starts = []
    # Small Gaussian
    starts.append(perturb(x, n, 0.05, rng))
    # Medium Gaussian
    starts.append(perturb(x, n, 0.15, rng))
    # Sinusoidal
    t = np.linspace(0, 2, n, endpoint=False)
    for freq in [2, 4, 6]:
        starts.append(repair(x + 0.1 * np.sin(2 * np.pi * freq * t / 2.0), n))
    # Boundary push: randomly set some to 0 or 1
    y = x.copy()
    n_push = n // 10
    idx = rng.choice(n, n_push, replace=False)
    y[idx] = rng.random(n_push) > 0.5  # 0 or 1
    starts.append(repair(y, n))
    # Large Gaussian
    starts.append(perturb(x, n, 0.25, rng))
    return starts

        corr = np.correlate(best, 1.0 - best, mode="full")
        idx = np.argsort(corr)[::-1][:top_k]
        x = slsqp_opt(best, n, dx, tau, 300, 1e-14, active=idx)
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best = x.copy()
        x_full = slsqp_opt(best, n, dx, max(tau * 0.5, 0.0002), 200, 1e-14, active=None)
        val_full = compute_c5(x_full, dx)
        if val_full < best_val:
            best_val = val_full
            best = x_full.copy()
    return best, best_val

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, x_src, kind='linear', fill_value='extrapolate')(xd)
    return repair(xp, n_dst)

def perturb(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def gen_starts(x, n, rng):
    starts = [x.copy()]
    starts.append(perturb(x, n, 0.05, rng))
    starts.append(perturb(x, n, 0.12, rng))
    starts.append(perturb(x, n, 0.20, rng))
    starts.append(perturb(x, n, 0.30, rng))
    t = np.linspace(0, 2, n, endpoint=False)
    for freq in [2, 5, 8]:
        starts.append(repair(x + 0.08 * np.sin(2 * np.pi * freq * t / 2.0 + rng.random()), n))
    y = x.copy()
    n_push = max(3, n // 8)
    idx = rng.choice(n, n_push, replace=False)
    y[idx] = (rng.random(n_push) > 0.5).astype(float)
    starts.append(repair(y, n))
    y2 = x.copy()
    idx2 = rng.choice(n, max(2, n // 12), replace=False)
    y2[idx2] = 0.0
    starts.append(repair(y2, n))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0_raw = np.array(initial_h_values, dtype=float)
    else:
        h0_raw = np.full(191, 0.5)
    n0 = len(h0_raw)
    dx0 = 2.0 / n0
    h0 = repair(h0_raw, n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    # Phase 1: Multi-start at base grid
    p1_end = t_start + budget_s * 0.42
    starts = gen_starts(h0, n0, rng)
    per_start = (p1_end - t_start) / len(starts)

    for idx, xs in enumerate(starts):
        elapsed = time.time() - t_start
        if elapsed > p1_end:
            break
        alloc = min(per_start, p1_end - elapsed)
        if alloc < 3:
            break
        x_best, val = refine_1d(xs, n0, dx0, alloc)
        if val < best_val:
            best_val = val
            best_h = x_best
            best_n = n0

    # Phase 2: Active-set at base
    p2_end = t_start + budget_s * 0.55
    if time.time() < p2_end:
        p2_budget = p2_end - time.time()
        if p2_budget > 8:
            x_b, v = refine_active(best_h, n0, dx0, p2_budget, top_k=10)
            if v < best_val:
                best_val = v
                best_h = x_b
                best_n = n0

    # Phase 3: Grid refinement to 2n+1
    p3_end = t_start + budget_s * 0.82
    n1 = 2 * best_n + 1
    if n1 < 1000 and time.time() < p3_end:
        dx1 = 2.0 / n1
        x_p = project_grid(best_h, best_n, n1)
        p3_budget = p3_end - time.time()
        if p3_budget > 15:
            x_b1, v1 = refine_1d(x_p, n1, dx1, p3_budget * 0.45)
            if v1 < best_val:
                best_val = v1
                best_h = x_b1
                best_n = n1
            if time.time() < p3_end - 5:
                x_p2 = perturb(x_p, n1, 0.04, rng)
                x_b2, v2 = refine_1d(x_p2, n1, dx1, p3_budget * 0.20)
                if v2 < best_val:
                    best_val = v2
                    best_h = x_b2
                    best_n = n1
            if time.time() < p3_end - 3:
                rem = p3_end - time.time()
                if rem > 8:
                    x_b3, v3 = refine_active(best_h, best_n, 2.0 / best_n, rem, top_k=12)
                    if v3 < best_val:
                        best_val = v3
                        best_h = x_b3

    # Phase 4: Final polish at best grid
    rem = budget_s - (time.time() - t_start) - 5
    if rem > 8:
        dx_best = 2.0 / best_n
        x = best_h.copy()
        for tau in [0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]:
            if time.time() - t_start > budget_s - 5:
                break
            x = slsqp_opt(x, best_n, dx_best, tau, 400, 1e-15)
            val = compute_c5(x, dx_best)
            if val < best_val:
                best_val = val
                best_h = x.copy()
        # Small perturbation restart
        rem2 = budget_s - (time.time() - t_start) - 3
        if rem2 > 6:
            x_pert = perturb(best_h, best_n, 0.02, rng)
            x_b4, v4 = refine_1d(x_pert, best_n, dx_best, rem2,
                                 taus=[0.005, 0.002, 0.001, 0.0005, 0.0002])
            if v4 < best_val:
                best_val = v4
                best_h = x_b4
        # Active set final
        rem3 = budget_s - (time.time() - t_start) - 2
        if rem3 > 5:
            x_b5, v5 = refine_active(best_h, best_n, dx_best, rem3, top_k=6)
            if v5 < best_val:
                best_val = v5
                best_h = x_b5

    # Final evaluation
    dx_f = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_f)
    return (best_h, final_val, best_n)
