import numpy as np
from scipy.optimize import minimize
import time

_dx_g = 1.0
_tau_g = 1.0
_n_g = 0

def _smooth_max_obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    shifted = np.clip((corr - m) / _tau_g, -45.0, 0.0)
    return (m + _tau_g * np.log(np.sum(np.exp(shifted)))) * _dx_g

def _con_fun(x):
    return np.sum(x) - _n_g / 2.0

def _con_jac(x):
    return np.ones(_n_g)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n, iters=20):
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(iters):
        s = np.sum(x)
        if abs(s - tgt) < 1e-13:
            break
        err = tgt - s
        if err > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            x += err / n
        else:
            x[mask] += err / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def slsqp_solve(x0, n, dx, tau, maxiter=400, ftol=1e-13):
    global _dx_g, _tau_g, _n_g
    _dx_g = dx
    _tau_g = tau
    _n_g = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_smooth_max_obj, x0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return repair(res.x, n)
    except Exception:
        return repair(x0, n)

def project_h(x_src, n_src, n_dst):
    if n_src == n_dst:
        return repair(x_src.copy(), n_dst)
    xs = np.arange(n_src) * (2.0 / n_src)
    xd = np.arange(n_dst) * (2.0 / n_dst)
    x_new = np.interp(xd, xs, x_src)
    return repair(x_new, n_dst)

def perturb(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def refine_tau_schedule(x0, n, dx, time_limit):
    t0 = time.time()
    x = repair(x0, n).copy()
    best = x.copy()
    best_val = compute_c5(x, dx)
    # Focused tau schedule: spend more time in the critical 0.01-0.05 range
    schedule = [
        (0.15, 200), (0.07, 300), (0.03, 400), (0.015, 500),
        (0.008, 500), (0.004, 600), (0.002, 400), (0.001, 300)
    ]
    for tau, mit in schedule:
        elapsed = time.time() - t0
        if elapsed > time_limit - 2.0:
            break
        x = slsqp_solve(x, n, dx, tau, mit, 1e-14)
        val = compute_c5(x, dx)
        if val < best_val - 1e-10:
            best_val = val
            best = x.copy()
    return best, best_val

def targeted_polish(x, n, dx, n_rounds=50):
    """Reduce the top few correlations via subgradient-like steps."""
    best = x.copy()
    best_val = compute_c5(x, dx)
    for r in range(n_rounds):
        corr = np.correlate(x, 1.0 - x, mode="full")
        top_k = 3
        top_idx = np.argsort(corr)[-top_k:]
        max_val = np.max(corr)
        grad = np.zeros(n)
        n_pts = n
        for ti in top_idx:
            lag = ti - (n_pts - 1)
            if lag == 0:
                grad = grad + (1.0 - 2.0 * x)
            elif lag > 0:
                valid = n_pts - lag
                grad[lag:] += (1.0 - x[:valid])
                grad[:valid] += (-x[lag:])
            else:
                a = -lag
                valid = n_pts - a
                grad[:valid] += (1.0 - x[a:a+valid])
                grad[a:a+valid] += (-x[:valid])
        grad *= dx
        # Try several step sizes
        improved = False
        for ss in [0.3, 0.1, 0.05, 0.02]:
            x_try = repair(x - ss * grad, n)
            v = compute_c5(x_try, dx)
            if v < best_val - 1e-9:
                best_val = v
                best = x_try.copy()
                x = x_try.copy()
                improved = True
                break
        if not improved:
            break
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    global initial_h_values
    t_start = time.time()
    rng = np.random.default_rng(seed)

    try:

    best_n = n0

    # Phase 1: Optimize at parent grid size with multiple restarts
    p1_budget = budget_s * 0.45
