import numpy as np
from scipy.optimize import minimize
import time

_opt_state = {}

def _obj_func(x):
    return eval_c5(x, _opt_state['dx'])

def _eq_func(x):
    return x.sum() - _opt_state['target']

def _eq_jac(x):
    return np.ones_like(x)

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def lse_value_and_weights(h, dx, eps):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    shifted = np.clip((corr - max_c) / eps, -700.0, 0.0)
    exp_s = np.exp(shifted)
    total = exp_s.sum()
    weights = exp_s / total
    lse_val = max_c + eps * np.log(total)
    return lse_val, weights, corr

def lse_gradient(h, dx, eps, weights):
    n = len(h)
    grad = np.zeros(n)
    m_arr = np.arange(n)
    n_lags = 2 * n - 1
    for k_idx in range(n_lags):
        w = weights[k_idx] * dx
        if w < 1e-12:
            continue
        lag = k_idx - (n - 1)
        i1 = m_arr + lag
        i2 = m_arr - lag
        t1 = np.zeros(n)
        mask1 = (i1 >= 0) & (i1 < n)
        t1[mask1] = 1.0 - h[i1[mask1]]
        t2 = np.zeros(n)
        mask2 = (i2 >= 0) & (i2 < n)
        t2[mask2] = h[i2[mask2]]
        grad += w * (t1 - t2)
    return grad

    best_c5 = eval_c5(h, dx)
    best_h = h.copy()

        g = lse_gradient(h, dx, eps, weights)
        g = g - g.mean()
        h_trial = None
        best_trial_obj = lse_v
        for lr in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            h_t = project_feasible(h - lr * g, len(h))
            lse_t, _, _ = lse_value_and_weights(h_t, dx, eps)
            if lse_t < best_trial_obj - 1e-12:
                best_trial_obj = lse_t
                h_trial = h_t
                break
        if h_trial is None:
            break
        h = h_trial
        c5_now = eval_c5(h, dx)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5

def slsqp_opt(h, n, dx, maxiter=500, ftol=1e-12):
    _opt_state['dx'] = dx
    _opt_state['target'] = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_func, 'jac': _eq_jac}]
    try:
        res = minimize(_obj_func, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def structured_perturb(h0, n, rng):
    ptype = rng.integers(0, 4)
    h = h0.copy()
    if ptype == 0:
        lo = rng.integers(0, n - 5)

    elif ptype == 2:

    else:
        kernel = np.array([0.2, 0.3, 0.3, 0.2])
        h = np.convolve(h0, kernel, mode='same')
    return project_feasible(h, n)
