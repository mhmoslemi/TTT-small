def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def _project(h, target):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt == 0:
            h[:] = target / len(h)
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _coord_descent(h, max_sweeps=30, grid_pts=70):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    h = _project(h, target)
    bc5 = _c5(h)
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]

                bc5 = best_c
        h = _project(h, target)

            break
