```python
def subgradient_step(h, step_size):
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    # Find top lags
    max_corr = corr.max()
    # Active set: lags within epsilon of max
    active = corr > max_corr - 0.05
    # Compute gradient
    grad = np.zeros(n)
    for lag_idx in range(len(corr)):
        if active[lag_idx]:
            m = lag_idx - (n-1)  # actual lag
            for i in range(n):
                # contribution to corr[lag_idx] from h[i]
                j = i + m
                if 0 <= j < n:
                    grad[i] += (1 - h[j])
                j2 = i - m
                if 0 <= j2 < n:
                    grad[i] -= h[j2]
    # Normalize active by count
    grad /= active.sum()
    # Step
    h_new = h - step_size * grad
    # Project
    h_new = project(h_new, n)
    return h_new
```

```python
def project_feasible(h, target):
    """Project h onto {0<=h<=1, sum(h)=target}."""
    n = len(h)
    # Bisection on lambda
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_p = np.clip(h - mid, 0, 1)
        s = h_p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix exact sum
    diff = target - h_p.sum()
    if abs(diff) > 1e-12:
        mask = (h_p > 1e-10) & (h_p < 1 - 1e-10)
        if mask.sum() > 0:
            h_p[mask] += diff / mask.sum()
        else:
            h_p += diff / n
        h_p = np.clip(h_p, 0, 1)
    return h_p

def c5_value(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1-h, mode='full'))) * (2.0/n)

def subgrad_c5(h, tol=0.05):
    """Compute subgradient of max correlation (smoothed over active set)."""
    n = len(h)
    corr = np.correlate(h, 1-h, mode='full')
    max_c = corr.max()
    active = np.where(corr >= max_c - tol)[0]
    
    # Accumulate gradient
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)  # lag
        j = np.arange(n)
        # (1-h[j-m]) for valid j-m
        idx1 = j - m
        mask1 = (idx1 >= 0) & (idx1 < n)
        grad[mask1] += (1 - h[idx1[mask1]])
        # -h[j+m] for valid j+m
        idx2 = j + m
        mask2 = (idx2 >= 0) & (idx2 < n)
        grad[mask2] -= h[idx2[mask2]]
    
    grad /= len(active)
    return grad, max_c, len(active)

def subgradient_optimize(h0, n_steps=500, tol=0.05):
    """Run subgradient method."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()

        grad, _, _ = subgrad_c5(h, tol)

        c = c5_value(h_new)
        if c < best_c:

        h = h_new
```

- Find the vertex of the feasible polytope that minimizes the linear approximation
- Take a convex combination

The vertices of {0≤h≤1, sum(h)=n/2} are the 0-1 vectors with exactly n/2 ones. So the Frank-Wolfe step is: find which set of n/2 coordinates to set to 1 that minimizes the linear approximation.

This is equivalent to: given the subgradient g, find the subset S of size n/2 that minimizes sum_{i∈S} g[i]. This is just picking the n/2 smallest entries of g!

This is O(n log n) per step - very fast!

```python
def frank_wolfe_step(h, g):
    """Given h in feasible set and subgradient g, compute FW direction."""
    n = len(h)
    # Find n/2 indices with smallest g values
    k = n // 2
    idx = np.argsort(g)[:k]
    v = np.zeros(n)
    v[idx] = 1.0
    return v
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(h, target):
    n = len(h)
    lo, hi = -1.5, 2.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0.0, 1.0)
    diff = target - h_p.sum()
    if abs(diff) > 1e-13:
        m = (h_p > 1e-9) & (h_p < 1.0 - 1e-9)
        if m.sum() > 0:
            h_p[m] += diff / m.sum()
        else:
            h_p += diff / n
    return np.clip(h_p, 0.0, 1.0)

def c5_val(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * (2.0 / n)

def subgrad_c5(h, tol=0.04):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    active = np.where(corr >= max_c - tol)[0]
    j = np.arange(n)
    grad = np.zeros(n)
    for k in active:
        m = k - (n - 1)
        idx1 = j - m
        mk1 = (idx1 >= 0) & (idx1 < n)
        grad[mk1] += (1.0 - h[idx1[mk1]])
        idx2 = j + m
        mk2 = (idx2 >= 0) & (idx2 < n)
        grad[mk2] -= h[idx2[mk2]]
    grad /= len(active)
    return grad, max_c, len(active)

def fw_vertex(g, n, k):
    idx = np.argpartition(g, k)[:k]
    v = np.zeros(n)
    v[idx] = 1.0
    return v

def fw_optimize(h0, n_iters=600, tol_start=0.08, tol_end=0.02):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = c5_val(h)
    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        tol = tol_start + (tol_end - tol_start) * frac
        t_step = 2.0 / (it + 2.0)
        grad, _, _ = subgrad_c5(h, tol)
        v = fw_vertex(grad, n, n // 2)
        h_new = (1.0 - t_step) * h + t_step * v
        h_new = project_feasible(h_new, target)
        c = c5_val(h_new)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def subgrad_optimize(h0, n_iters=400):
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c = c5_val(h)
    for it in range(n_iters):
        alpha = 0.15 / np.sqrt(it + 1.0)
        grad, _, _ = subgrad_c5(h, tol=0.05)
        h_new = project_feasible(h - alpha * grad, target)
        c = c5_val(h_new)
        if c < best_c:
            best_c = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def slsqp_polish(h0, n):
    target = n / 2.0
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        m = corr.max()
        sh = np.clip(corr - m, -0.8, 0.8)
        al = 2000.0
        return (m + np.log(np.sum(np.exp(al * sh))) / al) * dx + 5000.0 * (np.sum(x) - target) ** 2

    try:

                       options={'maxiter': 80, 'ftol': 1e-11})
        h_p = project_feasible(res.x, target)
        return h_p, c5_val(h_p)

        return h0, c5_val(h0)
```
