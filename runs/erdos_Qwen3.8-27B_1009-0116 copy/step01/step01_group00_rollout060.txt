When we change h[i] += δ, h[j] -= δ:
- c[m] changes by: δ * (∂c[m]/∂h[i] - ∂c[m]/∂h[j])
- So c[m](δ) = c[m](0) + δ * d[m] where d[m] = ∂c[m]/∂h[i] - ∂c[m]/∂h[j]

We want to minimize max_m [c[m](0) + δ*d[m]] subject to 0≤h[i]+δ≤1, 0≤h[j]-δ≤1.

This is a 1D problem! The optimum δ* is where the max is minimized, which occurs at the intersection of the two "highest" lines.

This is a classic 1D minimax problem. We can solve it by:
1. Compute d[m] for all m
2. The function f(δ) = max_m (c[m] + δ*d[m]) is piecewise linear and convex in δ
3. Find the δ that minimizes f(δ) by finding the "knee" of the upper envelope

This can be done efficiently:
- Find the m* that achieves the max at δ=0
- The optimal δ moves in the direction that reduces c[m*]
- But another m might become the new max
- Iterate: this is like a 1D active-set method

For n=70, 2n-1=139 correlation values. The 1D minimax can be solved by finding the intersection points of all pairs of lines and checking which interval contains the minimum of the upper envelope.

Actually, a simpler approach: since f(δ) is convex piecewise linear, we can use ternary search or bisection on the subgradient. The subgradient at δ is {d[m*] : m* is an argmax}. f'(δ-) = min{d[m] : m in argmax}, f'(δ+) = max{d[m] : m in argmax}. The optimal δ is where 0 is in the subgradient, i.e., min d[argmax] ≤ 0 ≤ max d[argmax].

This can be solved in O(log) iterations of a bisection.

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h, dx):
    return np.max(_corr(h)) * dx

def _repair(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    s = h.sum()
    if s < 1e-12:
        return np.full(n, 0.5)
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if diff > 0:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
            else:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def _pair_delta(h, n, i, j, corr):
    hi, hj = h[i], h[j]
    dlo = max(-hi, hj - 1.0)
    dhi = min(1.0 - hi, hj)
    if dlo >= dhi - 1e-14:
        return 0.0
    d = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        L = m - (n - 1)
        vi = 0.0
        if 0 <= i + L < n:
            vi += 1.0 - h[i + L]
        if 0 <= i - L < n:
            vi -= h[i - L]
        vj = 0.0
        if 0 <= j + L < n:
            vj += 1.0 - h[j + L]
        if 0 <= j - L < n:
            vj -= h[j - L]
        d[m] = vi - vj
    lo, hi2 = dlo, dhi
    best_d = 0.0
    best_val = np.max(corr)
    for _ in range(45):
        mid = (lo + hi2) / 2.0
        vals = corr + mid * d
        mv = np.max(vals)
        if mv < best_val:
            best_val = mv
            best_d = mid
        active = np.where(vals >= mv - 1e-10)[0]
        dmin = d[active].min()
        dmax = d[active].max()
        if dmin <= 0.0 <= dmax:
            best_d = mid
            best_val = mv
            break
        elif dmax < 0.0:
            lo = mid
        else:
            hi2 = mid
    for td in (dlo, dhi):
        v = np.max(corr + td * d)
        if v < best_val - 1e-12:
            best_val = v
            best_d = td
    return best_d

def _sweep_pairs(h, n, dx, best_c5, best_h, rng, max_rng_pairs, t_limit):
    corr = _corr(h)
    improved = False
    for _ in range(max_rng_pairs):
        if time.time() > t_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        delta = _pair_delta(h, n, i, j, corr)
        if abs(delta) > 1e-11:
            h[i] += delta
            h[j] -= delta
            c5_new = _c5(h, dx)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h[:] = h
                corr = _corr(h)
                improved = True
            else:
                h[i] -= delta
                h[j] += delta
    return best_c5, best_h, improved

def _full_sweep(h, n, dx, best_c5, t_limit):
    corr = _corr(h)
    any_imp = False
    for i in range(n):
        if time.time() > t_limit:
            break
        for j in range(i + 1, n):
            delta = _pair_delta(h, n, i, j, corr)
            if abs(delta) > 1e-11:
                h[i] += delta
                h[j] -= delta
                c5_new = _c5(h, dx)

                    best_c5 = c5_new
                    corr = _corr(h)
                    any_imp = True
                else:
                    h[i] -= delta
                    h[j] += delta
    return best_c5, any_imp

def _slsqp_refine(h, n, dx, best_c5, t_limit):
    bnds = [(0.0, 1.0)] * n
    target = n / 2.0
    def cons_fun(x):
        return x.sum() - target
    cons = [{'type': 'eq', 'fun': cons_fun}]
    for alpha in [30, 60, 120]:
        if time.time() > t_limit:
            break
        def obj(x, a=alpha):
            c = _corr(x)
            m = np.max(c)
            s = np.clip(a * (c - m), -300, 300)
            return m + np.log(np.sum(np.exp(s))) / a
        try:
            res = minimize(obj, h, method='SLSQP', bounds=bnds,
                           constraints=cons, options={'maxiter': 250, 'ftol': 1e-15})
            hc = _repair(res.x, n)
            c5_new = _c5(hc, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                h[:] = hc
        except Exception:
            pass
    return best_c5, h
