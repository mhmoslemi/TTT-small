        dx_b = 2.0 / n_b
        tgt_b = n_b / 2.0
        sub_end = deadline - 20
        for r in range(10):
            if time.time() > sub_end:
                break
            amp = 0.02 + 0.01 * rng.random()
            h_p = _proj(h_b + rng.normal(0, amp, n_b), tgt_b)
            c_p = _c5(h_p, dx_b)
            if c_p < best_c5:
                h_b = h_p.copy()
                best_c5 = c_p
            sub_t = min(time.time() + 15, sub_end - 5)
            if sub_t - time.time() > 4:
                hp, cp = _opt_full(h_b, dx_b, n_b, tgt_b, rng, sub_t)
                if cp < best_c5:
                    best_c5 = cp
                    h_b = hp
                    best_h = (hp, n_b)

    # Final polish
    if best_h is not None and time.time() < deadline - 8:
        h_f, n_f = best_h
        dx_f = 2.0 / n_f
        tgt_f = n_f / 2.0
        hf, cf = _opt_full(h_f, dx_f, n_f, tgt_f, rng, deadline)
        if cf < best_c5:
            best_c5 = cf
            best_h = (hf, n_f)

    if best_h is None:
        n = 70
        h_r = np.full(n, 0.5)
        best_c5 = _c5(h_r, 2.0 / n)
        best_h = (h_r, n)

    h_out, n_out = best_h
    return h_out, best_c5, n_out
