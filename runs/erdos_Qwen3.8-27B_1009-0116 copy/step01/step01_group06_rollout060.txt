import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Shift to avoid overflow
    corr_shift = corr - np.max(corr)
    log_sum = np.log(np.sum(np.exp(beta * corr_shift))) + beta * np.max(corr)
    return (log_sum / beta) * dx

def smooth_c5_grad(h, dx, beta, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_shift = corr - np.max(corr)
    exp_vals = np.exp(beta * corr_shift)
    probs = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k_idx in range(2*n - 1):
        if probs[k_idx] < 1e-15:
            continue
        lag = k_idx - (n - 1)
        w = probs[k_idx]
        if lag == 0:

            # d/dh_i = (1-h[i]) - h[i] = 1 - 2*h[i]

            for m in range(n - lag):

            grad[:n-lag] += w * (1.0 - h[lag:])

            grad[lag2:] += w * (1.0 - h[:n-lag2])
            grad[:n-lag2] += w * h[lag2:]
    
    return dx * grad

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-14:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-14:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-14:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_corr(h, n):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_grad(h, n, dx, beta):
    corr = compute_corr(h, n)
    m = np.max(corr)
    # Compute softmax weights
    shifted = beta * (corr - m)
    exps = np.exp(shifted)
    s = np.sum(exps)
    probs = exps / s
    obj = (m + np.log(s) / beta) * dx
    
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        p = probs[ki]
        if p < 1e-14:
            continue

        elif k > 0:

            grad[k:] += -p * h[:n-k]

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, n, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    s = np.sum(exps)
    probs = exps / s
    obj = (m + np.log(s) / beta) * dx
    active = np.where(probs > 1e-10)[0]
    grad = np.zeros(n)
    for ki in active:
        p = probs[ki]
        k = ki - (n - 1)
        if k == 0:
            grad += p * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += p * (1.0 - h[k:])
            grad[k:] -= p * h[:n-k]
        else:
            l = -k
            grad[l:] += p * (1.0 - h[:n-l])
            grad[:n-l] -= p * h[l:]
    return obj, dx * grad

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = float(n) / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s > 1e-10:
                h += diff * room / s
        else:
            s = h.sum()
            if s > 1e-10:
                h += diff * h / s
        h = np.clip(h, 0.0, 1.0)
        d2 = target - h.sum()
        if abs(d2) > 1e-10:
            idx = int(np.argmax(1.0 - h)) if d2 > 0 else int(np.argmin(h))
            h[idx] = float(np.clip(h[idx] + d2, 0.0, 1.0))
    return h

def slsqp_opt(h, n, dx, beta, maxiter=200):
    target = float(n) / 2.0
    bounds = [(0.0, 1.0)] * n
    def f_obj(x):
        return smooth_obj_grad(x, n, dx, beta)[0]
    def f_grad(x):
        return smooth_obj_grad(x, n, dx, beta)[1]
    def f_eq(x):
        return x.sum() - target
    def f_eqj(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': f_eq, 'jac': f_eqj}]
    try:
        res = minimize(f_obj, h, jac=f_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)

def lbgfgs_opt(h, n, dx, beta, lam, maxiter=400):
    target = float(n) / 2.0
    def p_obj(x):
        v, g = smooth_obj_grad(x, n, dx, beta)
        return v + lam * (x.sum() - target) ** 2
    def p_grad(x):
        _, g = smooth_obj_grad(x, n, dx, beta)
        return g + 2.0 * lam * (x.sum() - target)
    try:
        res = minimize(p_obj, h, jac=p_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n, options={'maxiter': maxiter, 'ftol': 1e-16})
        ho = project_feasible(res.x, n)
        return ho, eval_c5(ho, dx)
    except Exception:
        return h, eval_c5(h, dx)
