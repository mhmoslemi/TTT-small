def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h

def smooth_max(c, t):
    """Log-sum-exp approximation of max(c)"""
    cmax = np.max(c)
    return cmax + np.log(np.sum(np.exp(c - cmax))) / t

def corr_and_grad(h, n, dx, t=100.0):
    """Compute smoothed max correlation and its gradient."""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Smooth max
    cshift = c - np.max(c)
    w = np.exp(cshift * t)
    w /= w.sum()
    obj = (np.max(c) + np.log(np.sum(np.exp(cshift * t))) / t) * dx
    # Gradient: sum_k w[k] * dc[k]/dh
    # dc[k]/dh[j] for correlation at lag (k - n + 1):
    # For full correlation of a and b: c[m] = sum_i a[i] * b[i - m + (n_b - 1)]
    # Here a=h, b=1-h, both length n.
    # c[m] for m=0,...,2n-2
    # c[m] = sum_i h[i] * (1-h[i - m + n - 1]) where 0<=i<n and 0<=i-m+n-1<n
    # Let k = m - n + 1 (the "shift"): c[n-1+k] = sum_i h[i+k]*(1-h[i]) for valid i
    
    # For shift k: dc/dh[j] = (1-h[j-k])*I[valid] - h[j+k]*I[valid]
    # Weighted sum over k with weight w[n-1+k]
    
    grad = np.zeros(n)
    for k in range(-n+1, n):
        idx = n - 1 + k
        wk = w[idx] * dx
        if wk < 1e-10:
            continue
        # Terms where j appears as h[i+k]: i=j-k, need 0<=j-k<n and 0<=j<n
        # Contribution: wk * (1 - h[j-k])
        i1 = j - k  # not vectorized, let me vectorize
        # Actually let me vectorize:
        # For shift k, valid i: max(0,-k) <= i <= min(n-1, n-1-k)
        # h[i+k]*(1-h[i]):
        #   d/dh[j]: (1-h[j-k]) if j-k in [0,n-1] and j in [max(0,k), min(n-1, n-1+k)]
        #            -h[j+k] if j in [max(0,-k), min(n-1, n-1-k)] and j+k in [0,n-1]
        
        # Vectorized:
        valid_j_first = np.arange(n)
        # j-k in [0,n-1]: j in [k, k+n-1] intersect [0,n-1]
        if k >= 0:
            j_start1, j_end1 = k, n-1
        else:
            j_start1, j_end1 = 0, n-1+k
        # j in [0,n-1] and j+k in [0,n-1]: j in [-k, n-1-k] intersect [0,n-1]
        if k >= 0:
            j_start2, j_end2 = 0, n-1-k
        else:
            j_start2, j_end2 = -k, n-1
        
        mask1 = np.zeros(n, dtype=bool)
        mask1[j_start1:j_end1+1] = True
        mask2 = np.zeros(n, dtype=bool)
        mask2[j_start2:j_end2+1] = True
        
        # (1-h[j-k]) for j in mask1: j-k ranges from j_start1-k to j_end1-k
        # h[j-k] where j is in mask1:
        idx1 = np.arange(j_start1, j_end1+1) - k
        grad[j_start1:j_end1+1] += wk * (1 - h[idx1])
        
        # -h[j+k] for j in mask2: j+k ranges from j_start2+k to j_end2+k
        idx2 = np.arange(j_start2, j_end2+1) + k
        grad[j_start2:j_end2+1] -= wk * h[idx2]
    
    return obj, grad

def compute_grad_topk(h, n, dx, k_list):
    """Gradient using top-k active lags (unnormalized weights)."""
    grad = np.zeros(n)
    for k in k_list:
        # k is the shift (can be negative)
        # c[n-1+k] = sum_i h[i+k]*(1-h[i])
        # d/dh[j] = (1-h[j-k])*I[0<=j-k<n] - h[j+k]*I[0<=j+k<n]
        if k >= 0:
            # j-k in [0,n-1] means j in [k, n-1]
            # j+k in [0,n-1] means j in [0, n-1-k]
            grad[k:] += (1 - h[:n-k])  # (1-h[j-k]) for j in [k, n-1], j-k in [0, n-1-k]
            grad[:n-k] -= h[k:]  # h[j+k] for j in [0, n-1-k], j+k in [k, n-1]

            grad[:n-kk] += (1 - h[kk:])  # (1-h[j+kk]) for j in [0, n-1-kk]
            grad[kk:] -= h[:n-kk]  # h[j-kk] for j in [kk, n-1], j-kk in [0, n-1-kk]
    return grad * dx

```python
def projected_gradient(h, n, t0, max_iter=5000, top_k=5):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    h = h.copy()
    lr = 0.01  # initial learning rate
    
    for it in range(max_iter):
        c = np.correlate(h, 1.0 - h, mode='full')
        c_scaled = c * dx
        # Find top k lags
        top_indices = np.argsort(c_scaled)[-top_k:]
        top_lags = top_indices - (n - 1)  # convert to shift
        weights = c_scaled[top_indices] - c_scaled[top_indices].min() + 0.1
        weights /= weights.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for w, lag in zip(weights, top_lags):
            if lag >= 0:
                grad[lag:] += w * (1 - h[:n-lag])
                grad[:n-lag] -= w * h[lag:]
            else:
                kk = -lag
                grad[:n-kk] += w * (1 - h[kk:])
                grad[kk:] -= w * h[:n-kk]
        grad *= dx
        
        # Gradient step
        h_new = h - lr * grad
        h_new = repair(h_new, n)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h = h_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.8
            if lr < 1e-6:
                break
    
    return h, best_c5
```

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:

def smooth_obj_and_grad(h, n, t, pen):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    shifted = t * (c - cmax)
    shifted -= shifted.max()
    e = np.exp(shifted)
    w = e / e.sum()
    val = (cmax + shifted.max() + np.log(e.sum() + 1e-300)) / t
    pen_val = pen * (h.sum() - n * 0.5) ** 2
    obj = (val + pen_val) * dx
    # Gradient of smooth max part
    grad = np.zeros(n)
    # Use all lags weighted by softmax (top-20 for speed)
    top_m = min(20, len(c))
    top_idx = np.argsort(c)[-top_m:]
    tw = w[top_idx].sum()
    if tw > 1e-15:
        tw = w[top_idx] / tw
    else:
        tw = w[top_idx]
    for wi, mi in zip(tw, top_idx):
        k = mi - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[k:] += wi * (1.0 - h[:n - k])
                grad[:n - k] -= wi * h[k:]
        else:
            kk = -k
            if n - kk > 0:
                grad[:n - kk] += wi * (1.0 - h[kk:])
                grad[kk:] -= wi * h[:n - kk]
    grad *= dx
    # Penalty gradient
    grad += 2.0 * pen * (h.sum() - n * 0.5) * dx
    # Zero out where at bounds
    grad[(h <= 0.0) | (h >= 1.0)] = 0.0
    return obj, grad


def lbfgsb_optimize(h_init, n, t, pen, max_iter, t_limit):
    t_start = time.time()
    x0 = h_init.copy()
    res = minimize(lambda x: smooth_obj_and_grad(x, n, t, pen)[0],
                   x0,
                   jac=lambda x: smooth_obj_and_grad(x, n, t, pen)[1],
                   method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
    return res.x, time.time() - t_start


def subgradient_polish(h, n, max_iter=3000, top_k=8, t_limit=None):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    h = h.copy()
    lr = 0.005
    stall = 0
    t_start = time.time()
    for it in range(max_iter):
        if t_limit is not None and time.time() - t_start > t_limit:

        c = np.correlate(h, 1.0 - h, mode='full')
        cs = c * dx
        top_idx = np.argsort(cs)[-top_k:]
        cmax_val = cs.max()
        # Weights: proportional to how close to max
        diffs = cmax_val - cs[top_idx]
        w = np.exp(-diffs * 200.0)
        w /= w.sum()
        grad = np.zeros(n)
        for wi, mi in zip(w, top_idx):
            k = mi - (n - 1)
            if k >= 0:

                    grad[:n - k] -= wi * h[k:]
            else:
                kk = -k
                if n - kk > 0:
                    grad[:n - kk] += wi * (1.0 - h[kk:])
                    grad[kk:] -= wi * h[:n - kk]
        grad *= dx
        # Step

            h = h_new

        else:
            h = h_new
            stall += 1
            lr *= 0.9
            if lr < 1e-7 or stall > 200:
                break
    return h, best_c5
