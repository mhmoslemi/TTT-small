import time

def _compute_c5(h, dx):
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _fix_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    for idx in order:
        if diff > 0:
            add = min(diff, 1.0 - h[idx])
            h[idx] += add

                s = np.clip(s, -500, 500)
                return m + np.log(np.sum(np.exp(s))) / a

            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target}]

            for alpha_f in [200, 500, 1000]:
                if time.time() > t_start + budget_s - 15:
                    break
                res = minimize(
                    _smooth_obj, global_best_h, args=(alpha_f,),
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 300, 'ftol': 1e-14}
                )
                hc = _fix_sum(res.x, target)
                c5 = _compute_c5(hc, dx)
                if c5 < global_best_c5:
                    global_best_c5 = c5
                    global_best_h = hc
        except Exception:
            pass

    # Final coordinate polish
    if time.time() < t_start + budget_s - 5:
        global_best_h, global_best_c5 = _coord_search(
            global_best_h, n, dx, target, offsets,
            t_start + budget_s - 5, rng
        )

    global_best_h = _fix_sum(global_best_h, target)
    global_best_c5 = _compute_c5(global_best_h, dx)
    return (global_best_h, global_best_c5, n)
