def project_onto_simplex_box(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}"""
    n = len(h)
    # Bisection on lambda
    lo, hi = -1.0, 1.0  # lambda range (h - lambda clipped to [0,1])
    # Actually the lambda might need to be outside [-1, 1] in theory
    # But since h is in [0,1] and we're projecting, lambda should be moderate
    
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fix tiny numerical errors
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        # Distribute to non-bound coordinates
        mask = (proj > 1e-12) & (proj < 1 - 1e-12)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
        else:
            # All at boundary, adjust one
            if diff > 0:
                idx = np.argmax(1.0 - proj)
            else:
                idx = np.argmin(proj)
            proj[idx] += diff
    return proj

    target_sum = n * 0.5
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    alpha = initial_alpha

        c = np.correlate(h, 1.0 - h, mode='full')
        # Find argmax lags
        max_val = np.max(c)
        argmax_mask = c >= max_val - 1e-10
        argmax_indices = np.where(argmax_mask)[0]
        
        # Convert array indices to lags
        # c[k_array] corresponds to lag k_array - (n-1)
        lags = argmax_indices - (n - 1)
        
        # Subgradient: average gradient over active lags
        g = np.zeros(n)
        for lag in lags:
            g += grad_c_at_lag(h, int(lag))
        g /= len(lags)
        
        # Gradient step
        h_new = h - alpha * g
        
        # Project
        h_new = project_onto_simplex_box(h_new, target_sum)
        
        # Evaluate
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        
        h = h_new
        
        # Decay alpha
        alpha *= 0.999
    
    return best_h, best_val

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if mask.any():
            h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-14:
            idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
            h[idx] = float(np.clip(h[idx] + diff2, 0.0, 1.0))
    return h

def grad_c_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    lo = max(0, k); hi = min(n-1, n-1+k)
    if lo <= hi:
        g[np.arange(lo,hi+1)] += 1.0 - h[np.arange(lo,hi+1)-k]
    lo = max(0, -k); hi = min(n-1, n-1-k)
    if lo <= hi:
        g[np.arange(lo,hi+1)] -= h[np.arange(lo,hi+1)+k]
    return g

    c = np.correlate(x, 1.0-x, mode='full')
    cm = np.max(c)
    return (cm + np.log(np.sum(np.exp((c-cm)*t)))/t) * dx

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * 2.0 / n)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    tgt = n * 0.5
    d = tgt - h.sum()
    if abs(d) > 1e-10:
        m = (h < 1.0 - 1e-10) if d > 0 else (h > 1e-10)
        if m.any():
            h[m] += d / m.sum()
            np.clip(h, 0.0, 1.0, out=h)
        d2 = tgt - h.sum()
        if abs(d2) > 1e-14:
            j = int(np.argmin(h)) if d2 > 0 else int(np.argmax(h))
            h[j] = float(np.clip(h[j] + d2, 0.0, 1.0))
    return h

def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    lo, hi = max(0, k), min(n-1, n-1+k)
    if lo <= hi:
        r = np.arange(lo, hi+1)
        g[r] += 1.0 - h[r-k]
    lo, hi = max(0, -k), min(n-1, n-1-k)
    if lo <= hi:
        r = np.arange(lo, hi+1)
        g[r] -= h[r+k]
    return g

def smo_max(x, n, tp):
    c = np.correlate(x, 1.0-x, mode='full')
    cm = np.max(c)
    return (cm + np.log(np.sum(np.exp((c-cm)*tp)))/tp) * (2.0/n)

def subgrad(h0, n, mit, tlim):
    t0 = time.time()
    h = h0.copy(); bh = h.copy(); bc = compute_c5(h)
    dg = np.zeros(n)
    for it in range(mit):
        if time.time()-t0 > tlim: break
        c = np.correlate(h, 1.0-h, mode='full')
        mx = float(np.max(c)); cv = mx*2.0/n
        if cv < bc: bc, bh = cv, h.copy()
        tol = 0.15*mx
        ac = np.where(c >= mx-tol)[0] - (n-1)
        g = np.zeros(n)
        for lag in ac: g += grad_lag(h, int(lag))
        g /= len(ac)
        dg = 0.7*dg + 0.3*g

        for _ in range(12):
            r = repair(h - a*dg, n)
            if compute_c5(r) < cv - 1e-13: tr = r; break
            a *= 0.5
        h = tr
    return bh, bc

    bs = [(0.0,1.0)]*n
    r = minimize(lambda x: smo_max(x,n,tp), h0, method='SLSQP',
                 bounds=bs, constraints=ds, options={'maxiter':mi,'ftol':1e-13})
    h = repair(r.x, n); return h, compute_c5(h)

            o = h[i]; bl, bv = b, o

                if v < bl - 1e-12: bl, bv = v, f
            h[i] = bv; h = repair(h, n)
            if bl < b: b = bl; imp = True
        if not imp: break
    return h, b
