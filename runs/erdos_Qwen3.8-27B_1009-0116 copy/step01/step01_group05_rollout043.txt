    try:
        res = minimize(f, h0, jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        if res.success or res.fun < 1e6:
            return _proj(res.x, n / 2.0)
    except Exception:
        pass
    return _proj(h0, n / 2.0)


def _proj_moment_grad(h, dx, n, eps, velocity, lr, beta=0.85):
    g = _smograd(h, dx, n, eps)
    velocity[:] = beta * velocity + (1.0 - beta) * g
    return _proj(h - lr * velocity, n / 2.0)


def _optimize_n(h_init, dx, n, t_end):
    best_h = h_init.copy()
    best_c = _c5(h_init, dx)
    h = h_init.copy()

    # Phase 1: SLSQP with decreasing epsilon
    for eps in [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() > t_end - 25:
            break
        h_new = _slsqp_smooth(h, dx, n, eps, maxiter=350)
        c = _c5(h_new, dx)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h_new.copy()
        h = h_new

    # Phase 2: Nesterov-momentum projected gradient
    vel = np.zeros(n)
    lr = 0.4
    eps_pg = 0.003
    for it in range(8000):
        if time.time() > t_end - 12:
            break
        h_new = _proj_moment_grad(h, dx, n, eps_pg, vel, lr)
        c_new = _c5(h_new, dx)
        if c_new <= _c5(h, dx) + 1e-12:
            h = h_new
            lr = min(lr * 1.015, 2.5)
        else:
            lr = max(lr * 0.94, 0.02)
            vel *= 0.5
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()

    # Phase 3: Final SLSQP polish
    if time.time() < t_end - 5:
        for eps in [0.0005, 0.0002, 0.0001, 0.00005]:
            if time.time() > t_end - 2:
                break
            h2 = _slsqp_smooth(best_h, dx, n, eps, maxiter=250)
            c2 = _c5(h2, dx)
            if c2 < best_c:
                best_c = c2
                best_h = h2.copy()

    return best_h, best_c


def _make_initial(n, dx, target, rng):
    parents = []
    if 'initial_h_values' in globals() and initial_h_values is not None:
        iv = np.array(initial_h_values, dtype=float).copy()
        if len(iv) == n:
            parents.append(_proj(iv, target))
        else:
            parents.append(_proj(np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(iv)), iv), target))
    # Symmetric step pattern
    hs = np.zeros(n)
    n_ones = int(target)
    idx = np.arange(n)
    order = np.argsort(np.abs(idx - n / 2.0))
    hs[order[:n_ones]] = 1.0
    parents.append(_proj(hs + rng.normal(0, 0.1, n), target))
    # Alternating blocks
    ha = np.zeros(n)
    blk = max(n // 70, 2)
    for i in range(0, n, blk):
        ha[i:i + blk] = 1.0
    parents.append(_proj(ha, target))
    # Random
    parents.append(_proj(rng.uniform(0.1, 0.9, n), target))
    return parents


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.RandomState(seed)

    global_best_h = None
    global_best_c = np.inf
    global_best_n = 70

    n_list = [70, 50, 64, 80, 90]

    for n in n_list:
        if time.time() > deadline - 80:
            break
        dx = 2.0 / n
        target = n / 2.0
        inits = _make_initial(n, dx, target, rng)

        # Allocate time: first init gets 50%, rest split remaining
        t_phase1 = min(time.time() + (deadline - time.time()) * 0.55, deadline - 40)
        for i, h_init in enumerate(inits):
            if i == 0:
                t_sub = t_phase1
            else:
                t_sub = min(time.time() + (deadline - time.time()) * 0.2, deadline - 40)
            if time.time() > deadline - 50:
                break
            c0 = _c5(h_init, dx)
            if c0 < global_best_c:
                global_best_c = c0
                global_best_h = h_init.copy()
                global_best_n = n
            t_end_sub = min(t_sub, deadline - 30)
            if t_end_sub - time.time() < 8:
                continue
            h_opt, c_opt = _optimize_n(h_init, dx, n, t_end_sub)
            if c_opt < global_best_c:
                global_best_c = c_opt
                global_best_h = h_opt
                global_best_n = n

        # Restart phase for this n
        if time.time() < deadline - 50:
            base = global_best_h.copy() if global_best_n == n else inits[0].copy()
            for r in range(8):
                if time.time() > deadline - 40:
                    break
                amp = rng.uniform(0.03, 0.15)
                h_p = _proj(base + rng.normal(0, amp, n), target)
                t_re = min(time.time() + 18, deadline - 35)
                if t_re - time.time() < 5:
                    break
                h_r, c_r = _optimize_n(h_p, dx, n, t_re)
                if c_r < global_best_c:
                    global_best_c = c_r
                    global_best_h = h_r
                    global_best_n = n
                    base = h_r.copy()

    # Final polish on global best
    if time.time() < deadline - 10 and global_best_h is not None:
        n_final = global_best_n
        dx_final = 2.0 / n_final
        t_pol = deadline - 2
        h_f, c_f = _optimize_n(global_best_h, dx_final, n_final, t_pol)
        if c_f < global_best_c:
            global_best_c = c_f
            global_best_h = h_f

    # Ensure feasibility
    n_ret = global_best_n
    dx_ret = 2.0 / n_ret
    global_best_h = _proj(global_best_h, n_ret / 2.0)
    # Exact integer sum correction
    s = global_best_h.sum()
    if abs(s - n_ret / 2.0) > 1e-10:
        global_best_h = _proj(global_best_h, n_ret / 2.0)

    return global_best_h, float(_c5(global_best_h, dx_ret)), n_ret
