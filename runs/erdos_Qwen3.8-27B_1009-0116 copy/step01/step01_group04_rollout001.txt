            gl[:n - ak] += wm * (1.0 - h[ak:])
    return gl * dx

def proj_grad_step(h, n, dx, tau, lr):
    g = grad_logsumexp(h, n, dx, tau)
    h_new = h - lr * g
    return repair(h_new, n)

def proj_grad_refine(h, n, dx, time_limit, tau=0.02, lr0=0.003):
    t0 = time.time()
    h = h.copy()
    best = h.copy()
    best_val = compute_c5(h, dx)
    lr = lr0
    it = 0
    while time.time() - t0 < time_limit - 0.3:
        h_new = proj_grad_step(h, n, dx, tau, lr)
        val = compute_c5(h_new, dx)
        if val < best_val:
            best_val = val
            best = h_new.copy()
            h = h_new.copy()
            lr = min(lr * 1.03, 0.015)
        else:
            lr *= 0.6
            if lr < 1e-7:
                h = repair(h, n)
        it += 1
    return best, best_val

def refine_full(x0, n, dx, time_limit):
    t0 = time.time()
    x = repair(x0, n).copy()
    best = x.copy()
    best_val = compute_c5(x, dx)
    taus = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for tau in taus:
        remaining = time_limit - (time.time() - t0)
        if remaining < 3.0:
            break
        mi = int(min(800, max(200, remaining * 30)))
        x = slsqp_opt(x, n, dx, tau, mi)
        val = compute_c5(x, dx)
        if val < best_val:
            best_val = val
            best = x.copy()
        remaining = time_limit - (time.time() - t0)
        if remaining > 2.0:
            x, val = proj_grad_refine(x, n, dx, remaining * 0.25, tau=max(tau, 0.01))
            if val < best_val:
                best_val = val
                best = x.copy()
    return best, best_val

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    xp = interp1d(xs, x_src, kind='linear', fill_value='extrapolate')(xd)
    return repair(xp, n_dst)

def perturb(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def smooth_restart(h, n, rng):
    k = 5
    kernel = np.ones(k) / k
    hs = np.convolve(h, kernel, mode='same')
    return repair(hs, n)

def sharpen_restart(h, n, alpha):
    hc = (h - h.min()) / (h.max() - h.min() + 1e-12)
    hs = hc ** alpha
    return repair(hs, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0_raw = np.array(initial_h_values, dtype=float)
    else:
        h0_raw = np.full(95, 0.5)
    n0 = len(h0_raw)
    dx0 = 2.0 / n0
    h0 = repair(h0_raw, n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    # Phase 1: Thorough SLSQP + grad refinement on base grid
    p1_t = budget_s * 0.38
    x_best, v1 = refine_full(h0, n0, dx0, p1_t)
    if v1 < best_val:
        best_val = v1
        best_h = x_best

    # Phase 2: Diverse restarts on base grid
    t2_end = t_start + budget_s * 0.55
    restarts = [
        perturb(best_h, n0, 0.06, rng),
        smooth_restart(best_h, n0, rng),
        perturb(best_h, n0, 0.12, rng),
        sharpen_restart(best_h, n0, 1.5),
    ]
    for i, xs in enumerate(restarts):
        if time.time() > t2_end:
            break
        alloc = min(budget_s * 0.06, t2_end - time.time())
        if alloc < 8:
            break
        x_r, v_r = refine_full(xs, n0, dx0, alloc)
        if v_r < best_val:
            best_val = v_r
            best_h = x_r

    # Phase 3: Finer grid (2n+1)
    if time.time() - t_start < t_start + budget_s * 0.60:
        n1 = 2 * best_n + 1
        dx1 = 2.0 / n1
        x_p = project_grid(best_h, best_n, n1)
        p3_t = min(budget_s * 0.20, t_start + budget_s * 0.78 - time.time())
        if p3_t > 15:
            x_b3, v3 = refine_full(x_p, n1, dx1, p3_t)
            if v3 < best_val:
                best_val = v3
                best_h = x_b3
                best_n = n1
        # Second attempt on finer grid with perturbation
        if time.time() - t_start < t_start + budget_s * 0.78:
            x_p2 = perturb(x_p, n1, 0.04, rng)
            p3b = min(budget_s * 0.07, t_start + budget_s * 0.78 - time.time())
            if p3b > 10:
                x_b3b, v3b = refine_full(x_p2, n1, dx1, p3b)
                if v3b < best_val:
                    best_val = v3b
                    best_h = x_b3b
                    best_n = n1

    # Phase 4: Final polish on best grid
    remaining = budget_s - (time.time() - t_start) - 5
    if remaining > 10:
        dx_b = 2.0 / best_n
        x_p4 = perturb(best_h, best_n, 0.02, rng)
        x_b4, v4 = refine_full(x_p4, best_n, dx_b, remaining * 0.85)
        if v4 < best_val:
            best_val = v4
            best_h = x_b4

    # Also try one more from scratch at best_n with different perturb
    remaining = budget_s - (time.time() - t_start) - 3
    if remaining > 12:
        dx_b = 2.0 / best_n
        x_p5 = perturb(best_h, best_n, 0.09, rng)
        x_b5, v5 = refine_full(x_p5, best_n, dx_b, remaining * 0.9)
        if v5 < best_val:
            best_val = v5
            best_h = x_b5

    # Final repair and evaluation
    dx_f = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_f)

    return (best_h, final_val, best_n)
