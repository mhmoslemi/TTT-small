        # Find positions of 0s and 1s
        zeros = np.where(b == 0)[0]
        ones = np.where(b == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        # Random swap
        i = zeros[rng.integers(len(zeros))]
        j = ones[rng.integers(len(ones))]

        new_val = compute_c5(b.astype(float), dx)
        delta = new_val - best_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):

                best_val = new_val
                best = b.copy()
        else:
            b[i], b[j] = b[j], b[i]  # revert
    return best, best_val

    if rng is None:
        rng = np.random.default_rng()
    b = b.copy()
    best_val = bin_eval(b, dx)

    for p in range(max_passes):
        zeros = np.where(b == 0)[0]
        ones = np.where(b == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        best_delta = 1e-12
        best_i, best_j = -1, -1
        n_z, n_o = len(zeros), len(ones)

        if n_z * n_o <= 10000:
            pairs_z = np.repeat(np.arange(n_z), n_o)
            pairs_o = np.tile(np.arange(n_o), n_z)
        else:
            ns = 5000
            pairs_z = rng.integers(0, n_z, ns)
            pairs_o = rng.integers(0, n_o, ns)

        for ia, ib in zip(pairs_z, pairs_o):
            i, j = zeros[ia], ones[ib]
            b[i], b[j] = b[j], b[i]
            v = bin_eval(b, dx)
            b[i], b[j] = b[j], b[i]
            d = best_val - v
            if d > best_delta:
                best_delta = d
                best_i, best_j = i, j

        if best_i >= 0:
            b[best_i], b[best_j] = b[best_j], b[best_i]
            best_val -= best_delta
        else:
            break
    return b, best_val

def binary_sa(b, n, dx, t0, n_iter, rng):
    """Simulated annealing on binary sequence."""
    b = b.copy()
    cur_val = bin_eval(b, dx)
    best_val = cur_val
    best_b = b.copy()
    for it in range(n_iter):
        T = t0 * (0.001 / t0) ** (it / max(n_iter, 1))
        zi = np.where(b == 0)[0]
        oi = np.where(b == 1)[0]
        if len(zi) == 0 or len(oi) == 0:
            break
        i = zi[rng.integers(len(zi))]
        j = oi[rng.integers(len(oi))]
        b[i], b[j] = b[j], b[i]
        nv = bin_eval(b, dx)
        if nv < cur_val or rng.random() < np.exp(-(nv - cur_val) / max(T, 1e-10)):
            cur_val = nv
            if nv < best_val:
                best_val = nv
                best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]
    return best_b, best_val

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in dir() or 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = np.full(95, 0.5)
    n0 = len(h0)
    dx0 = 2.0 / n0
    h0 = repair(h0, n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    # Phase 1: Binary local search (fast, finds good combinatorial structure)
    t_p1 = t_start + budget_s * 0.30
    b0 = (h0 > 0.45).astype(np.int8)
    n_ones = int(b0.sum())
    tgt_ones = n0 // 2
    if n_ones > tgt_ones:
        oi = np.where(b0 == 1)[0]
        b0[rng.choice(oi, n_ones - tgt_ones, replace=False)] = 0
    elif n_ones < tgt_ones:
        zi = np.where(b0 == 0)[0]
        b0[rng.choice(zi, tgt_ones - n_ones, replace=False)] = 1

    for rnd in range(12):
        if time.time() > t_p1:
            break
        b_r = b0.copy()
        if rnd > 0:
            nswaps = rnd * 8
            for _ in range(nswaps):
                zi = np.where(b_r == 0)[0]
                oi = np.where(b_r == 1)[0]
                a = zi[rng.integers(len(zi))]
                c = oi[rng.integers(len(oi))]
                b_r[a], b_r[c] = b_r[c], b_r[a]
        b_res, b_val = binary_greedy(b_r, n0, dx0, max_passes=40, rng=rng)
        # Add SA escape
        if time.time() < t_p1 - 5:
            b_sa, sa_val = binary_sa(b_res, n0, dx0, 0.005, 3000, rng)
            if sa_val < b_val:
                b_res, b_val = b_sa, sa_val
        if b_val < best_val:
            best_val = b_val
            best_h = b_res.astype(float)
            best_n = n0

    # Phase 2: SLSQP continuous refinement from best solution
    t_p2 = t_start + budget_s * 0.68
    x = best_h.copy()
    dx_cur = 2.0 / best_n
    taus = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    for tau in taus:
        if time.time() > t_p2:
            break
        x = slsqp_step(x, best_n, dx_cur, tau, 2500)
        v = compute_c5(x, dx_cur)
        if v < best_val:
            best_val = v
            best_h = x.copy()

    # Phase 3: Finer grid (2n+1)
    t_p3 = t_start + budget_s * 0.86
    if time.time() < t_p3:
        n1 = 2 * best_n + 1
        dx1 = 2.0 / n1
        x1 = project_grid(best_h, best_n, n1)
        # Also try binary search at finer grid
        b1 = (x1 > 0.45).astype(np.int8)
        no1 = int(b1.sum())
        tg1 = n1 // 2
        if no1 > tg1:
            oi = np.where(b1 == 1)[0]
            b1[rng.choice(oi, no1 - tg1, replace=False)] = 0
        elif no1 < tg1:
            zi = np.where(b1 == 0)[0]
            b1[rng.choice(zi, tg1 - no1, replace=False)] = 1
        b1r, b1v = binary_greedy(b1, n1, dx1, max_passes=20, rng=rng)
        if b1v < compute_c5(x1, dx1):
            x1 = b1r.astype(float)
        taus2 = [0.01, 0.005, 0.002, 0.001, 0.0005]
        for tau in taus2:
            if time.time() > t_p3:
                break
            x1 = slsqp_step(x1, n1, dx1, tau, 1500)
            v = compute_c5(x1, dx1)
            if v < best_val:
                best_val = v
                best_h = x1.copy()
                best_n = n1

    # Phase 4: Final polish at best grid
    t_end = t_start + budget_s - 5
    if time.time() < t_end - 10:
        dx_b = 2.0 / best_n
        x = best_h.copy()
        for tau in [0.0005, 0.0002, 0.0001]:
            if time.time() > t_end:
                break
            x = slsqp_step(x, best_n, dx_b, tau, 3000)
            v = compute_c5(x, dx_b)
            if v < best_val:
                best_val = v
                best_h = x.copy()

    # Additional: try a fresh SLSQP from SLSQP result with perturbation
    if time.time() < t_end - 10:
        dx_b = 2.0 / best_n
        xp = np.clip(best_h + 0.03 * rng.standard_normal(best_n), 0, 1)
        xp = repair(xp, best_n)
        x = slsqp_step(xp, best_n, dx_b, 0.001, 2000)
        v = compute_c5(x, dx_b)
        if v < best_val:
            best_val = v
            best_h = x.copy()

    best_h = repair(best_h, best_n)
    dx_f = 2.0 / best_n
    final_val = compute_c5(best_h, dx_f)
    return (best_h, final_val, best_n)
