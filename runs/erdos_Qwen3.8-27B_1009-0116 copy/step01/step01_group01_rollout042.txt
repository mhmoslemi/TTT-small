def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-14:
                idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
                h[idx] += diff2
    return h


def gold_search(eval_fn, a, b, iters=35):
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = eval_fn(c)
    fd = eval_fn(d)
    for _ in range(iters):

            b = d; d = c; fd = fc; c = b - gr * (b - a); fc = eval_fn(c)
        else:
            a = c; c = d; fc = fd; d = a + gr * (b - a); fd = eval_fn(d)
    return (a + b) * 0.5
