def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-13:
            if diff2 > 0:
                idx = int(np.argmin(h))
            else:
                idx = int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def softmax_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(c)
    return (cmax + np.log(np.sum(np.exp((c - cmax) * t))) / t)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def repair(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        diff2 = target - h.sum()
        if abs(diff2) > 1e-13:
            idx = int(np.argmin(h)) if diff2 > 0 else int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff2, 0.0, 1.0)
    return h

def softmax_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = np.max(c)
    scaled = (c - cmax) * t
    scaled = np.clip(scaled, -500, 500)
    return cmax + np.log(np.sum(np.exp(scaled))) / t

def slsqp_opt(h, n, t_param, maxiter):
    cons = [{'type': 'eq', 'fun': lambda x, n=n: x.sum() - n * 0.5,
             'jac': lambda x, n=n: np.ones(n)}]
    bnds = [(0.0, 1.0)] * n
    res = minimize(softmax_obj, h.copy(), args=(t_param,),
                   method='SLSQP', bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    return res.x

            break
        order = rng.permutation(n)
        for i in order:

            idx_im = np.clip(im, 0, n - 1).astype(np.intp)
            a = ((1.0 - h[idx_ip]) * v_ip - h[idx_im] * v_im) * dx

            xc = h[i]
            b = corr - a * xc

            pos = a > 1e-14
            neg = a < -1e-14
            zer = ~(pos | neg)

            best_x = xc
            best_v = float(np.max(a * xc + b))
            v0 = float(np.max(b))
            if v0 < best_v:

            break
