    global _g_alpha, _g_dx, _g_target
    t0 = time.time()
    rng = np.random.RandomState(seed)
    remaining = lambda: budget_s - (time.time() - t0)

    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values, dtype=float)
    else:
        parent = np.full(70, 0.5)
    pn = len(parent)
    best_c5 = _c5(parent)
    best_h = parent.copy()
    best_n = pn
    print(f"Start: n={pn}, C5={best_c5:.6f}")

    # Local search on parent
    h_ls, c_ls = _local_search(parent.copy(), float(pn)/2.0, 35.0, rng)
    if c_ls < best_c5:
        best_c5 = c_ls
        best_h = h_ls.copy()
        best_n = pn
        print(f"Parent LS: C5={c_ls:.6f}")

    # Optimize at multiple resolutions
    for n in [70, 80, 90, 100, 120, 140, 160]:
        if remaining() < 70:
            break
        if n == pn and best_n == pn:
            h_init = best_h.copy()
        elif n == pn:
            h_init = _interp_periodic(best_h, n)
            h_init = _project(h_init, float(n)/2.0)
        else:
            h_init = _interp_periodic(parent, n)
            h_init = _project(h_init, float(n)/2.0)
        target = float(n) / 2.0
        _g_dx = 2.0 / n
        _g_target = target
        bounds = [(0.0, 1.0)] * n

        for al in [200, 1000, 5000, 20000]:
            if remaining() < 15:
                break
            _g_alpha = float(al)
            try:
                mi = 400 if n <= 100 else 250
                res = minimize(_smooth_obj, h_init.copy(), method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': mi, 'ftol': 1e-14, 'gtol': 1e-10})
                hn = _project(res.x, target)
                cn = _c5(hn)
                if cn < best_c5:
                    best_c5 = cn
                    best_h = hn.copy()
                    best_n = n
                    print(f"  n={n} a={al}: C5={cn:.6f}")
                h_init = hn
            except Exception:
                pass

        # Local search after gradient
        if remaining() > min(50, 0.15 * budget_s):
            tlim = min(25.0, remaining() * 0.25)
            h_ls, c_ls = _local_search(h_init.copy(), target, tlim, rng)
            if c_ls < best_c5:
                best_c5 = c_ls
                best_h = h_ls.copy()
                best_n = n
                print(f"  n={n} LS: C5={c_ls:.6f}")

    # Final polish: more local search on best
    if remaining() > 30:
        target = float(best_n) / 2.0
        tlim = min(45.0, remaining() * 0.5)
        h_ls, c_ls = _local_search(best_h.copy(), target, tlim, rng)
        if c_ls < best_c5:
            best_c5 = c_ls
            best_h = h_ls.copy()
            print(f"Polish: C5={c_ls:.6f}")

    # Final sum correction
    h_ret = _project(best_h.copy(), float(best_n) / 2.0)
    c5f = _c5(h_ret)
    print(f"Final: n={best_n}, C5={c5f:.6f}")
    return h_ret, c5f, best_n
