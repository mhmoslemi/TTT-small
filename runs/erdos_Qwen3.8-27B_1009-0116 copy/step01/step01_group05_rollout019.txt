import numpy as np
import time

def _c5_scalar(h, n):
    return float(np.correlate(h, 1.0 - h, "full").max() * (2.0 / n))

def _c5_vec(h, n):
    return np.correlate(h, 1.0 - h, "full") * (2.0 / n)

def _make_binary(h, n, target):
    """Convert a [0,1]-valued array to binary with exactly 'target' ones."""
    h = np.clip(h, 0, 1)
    b = (h >= 0.5).astype(float)
    # Adjust to get exactly target ones
    diff = target - b.sum()
    if diff > 0:
        # Need more ones: pick the highest-valued zeros
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])[:int(diff)]
        b[zeros[order]] = 1
    elif diff < 0:
        # Need fewer ones: pick the lowest-valued ones
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])[:int(-diff)]
        b[ones[order]] = 0
    return b

def _local_search(h, n, t_end, rng=None, max_sweeps=500):
    """Pair-flip local search for binary sequence."""
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = _c5_scalar(best_h, n)
    cur_h = h.copy()
    cur_c5 = best_c5
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        improved = False
        # Shuffle for diversification
        if rng is not None:
            rng.shuffle(ones)
            rng.shuffle(zeros)
        for i in ones:
            if time.time() > t_end:
                return best_h, best_c5
            for j in zeros:
                if i == j:
                    continue
                cur_h[i] = 0
                cur_h[j] = 1
                c = _c5_scalar(cur_h, n)
                if c < cur_c5 - 1e-12:
                    cur_c5 = c
                    if c < best_c5 - 1e-12:
                        best_c5 = c
                        best_h = cur_h.copy()
                    improved = True
                else:
                    cur_h[i] = 1
                    cur_h[j] = 0
        if not improved:
            break
    return best_h, best_c5

def _anneal(h, n, t_end, rng):
    """Simulated annealing with pair flips."""
    best_h = h.copy()
    best_c5 = _c5_scalar(best_h, n)
    cur_h = h.copy()
    cur_c5 = best_c5
    T = 0.03
    while time.time() < t_end and T > 1e-5:
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur_h[i] = 0
        cur_h[j] = 1
        new_c5 = _c5_scalar(cur_h, n)
        d = new_c5 - cur_c5

                best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
        T *= 0.998
    return best_h, best_c5

= h[j] - h[j]*h[j+s] + h[j-s] - h[j]*h[j-s] + rest

This IS linear in h[j]! Because h[j+s] and h[j-s] are fixed (they're other coordinates).

import numpy as np
import time


def _c5_scalar(h, n):
    return float(np.correlate(h, 1.0 - h, "full").max() * (2.0 / n))


def _c5_vec(h, n):
    return np.correlate(h, 1.0 - h, "full") * (2.0 / n)


def _make_binary(h_cont, n, target):
    h = np.clip(h_cont, 0, 1)
    b = (h >= 0.5).astype(float)
    diff = int(target - b.sum())
    if diff > 0:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])[:diff]
        b[zeros[order]] = 1.0
    elif diff < 0:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])[:(-diff)]
        b[ones[order]] = 0.0
    return b


def _local_search(h, n, t_end, rng, max_sweeps=2000):
    best_h = h.copy()
    best_c5 = _c5_scalar(best_h, n)
    cur = h.copy()
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if rng is not None:
            rng.shuffle(ones)
            rng.shuffle(zeros)
        improved = False
        for i in ones:
            if time.time() > t_end:
                return best_h, best_c5
            for j in zeros:
                cur[i] = 0.0
                cur[j] = 1.0
                c = _c5_scalar(cur, n)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = cur.copy()
                    improved = True
                else:
                    cur[i] = 1.0
                    cur[j] = 0.0
        if not improved:
            break
    return best_h, best_c5


def _anneal(h, n, t_end, rng, t0_init=0.04, t_min=1e-5, cool=0.9985):
    best_h = h.copy()
    best_c5 = _c5_scalar(best_h, n)
    cur = h.copy()
    cur_c5 = best_c5
    T = t0_init
    while time.time() < t_end and T > t_min:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur[i] = 0.0
        cur[j] = 1.0
        nc = _c5_scalar(cur, n)
        d = nc - cur_c5
        if d < 0 or rng.random() < np.exp(-d / T):
            cur_c5 = nc
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
        T *= cool
    return best_h, best_c5


def _perturb(h, n, rng, k=3):
    hp = h.copy()
    for _ in range(k):
        ones = np.where(hp == 1)[0]
        zeros = np.where(hp == 0)[0]
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        hp[a] = 0.0
        hp[b] = 1.0
    return hp


def _gen_golden(n, target):
    phi = 0.6180339887498949
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    return _make_binary(h, n, target)


def _gen_blocks(n, target, block_size):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i // block_size) % 2 == 0 else 0.0
    return _make_binary(h, n, target)


def _gen_random(n, target, rng):
    idx = rng.choice(n, target, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s * 0.96
    rng = np.random.RandomState(seed)
    n = 70
    target = n // 2

    candidates = []

    # Parent thresholded
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_p = np.array(initial_h_values, dtype=float)
        if len(h_p) != n:

        candidates.append(_make_binary(h_p, n, target))

    # Golden ratio
    candidates.append(_gen_golden(n, target))
    # Various block sizes
    for bs in [3, 4, 5, 7, 10, 14]:
        candidates.append(_gen_blocks(n, target, bs))
    # Random starts
    for _ in range(8):
        candidates.append(_gen_random(n, target, rng))

    # Phase 1: local search on all candidates (45% of time)
    t1 = t0 + (deadline - t0) * 0.45
    per_cand = max(3.0, (t1 - t0) / len(candidates))
    best_h, best_c5 = None, np.inf
    for h0 in candidates:
        if time.time() > t1:
            break
        sub_end = min(time.time() + per_cand, t1)
        sh, sc = _local_search(h0, n, sub_end, rng, max_sweeps=800)
        if sc < best_c5:
            best_c5 = sc
            best_h = sh.copy()

    # Phase 2: anneal from best (25%)
    t2 = t0 + (deadline - t0) * 0.70
    if best_h is not None and time.time() < t2:
        ah, ac = _anneal(best_h, n, t2, rng)
        if ac < best_c5:
            best_c5 = ac
            best_h = ah.copy()

    # Phase 3: iterated perturbation + local search + short anneal (28%)
    t3 = deadline
    while time.time() < t3 - 15:
        hp = _perturb(best_h, n, rng, k=rng.randint(2, 6))

        if sc < best_c5:
            best_c5 = sc
            best_h = sh.copy()
        a_end = min(time.time() + 12, t3 - 15)
        if a_end > time.time() + 4:
            ah2, ac2 = _anneal(best_h, n, a_end, rng, t0_init=0.025, cool=0.999)
            if ac2 < best_c5:
                best_c5 = ac2
                best_h = ah2.copy()

    # Phase 4: final polish local search
    if time.time() < deadline - 3 and best_h is not None:
        sh, sc = _local_search(best_h, n, deadline - 2, None, max_sweeps=500)
        if sc < best_c5:
            best_c5 = sc
            best_h = sh

    return best_h, best_c5, n
