import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_gdx = 1.0
_gtau = 1.0
_gn = 0

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full") * _gdx
    m = np.max(corr)
    sh = np.clip((corr - m) / _gtau, -50.0, 0.0)
    return m + _gtau * np.log(np.sum(np.exp(sh)))

def _cfun(x):
    return np.sum(x) - _gn / 2.0

def _cjac(x):
    return np.ones(_gn)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def repair(x, n):
    x = np.clip(x, 0.0, 1.0)
    s = np.sum(x)
    tgt = n / 2.0
    if abs(s - tgt) > 1e-12 and s > 1e-12:
        x = x * (tgt / s)
        x = np.clip(x, 0.0, 1.0)
    s2 = np.sum(x)
    if abs(s2 - tgt) > 1e-10:
        mask = (x > 0.01) & (x < 0.99)
        if mask.sum() > 0:
            x[mask] += (tgt - s2) / mask.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def slsqp_optimize(x0, n, dx, tau, maxiter=200):
    global _gdx, _gtau, _gn
    _gn = n
    _gdx = dx
    _gtau = tau
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        r = minimize(_obj, x0, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': maxiter, 'ftol': 1e-15})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src)
    xd = np.linspace(0.0, 2.0, n_dst)
    f = interp1d(xs, x_src, kind='linear', fill_value='extrapolate')
    return repair(f(xd), n_dst)

def perturb(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def target_adjust(x, n, dx, lr=0.005, steps=50):
    """Reduce the peak correlation by targeted coordinate moves."""
    x = x.copy()
    tgt = n / 2.0
    for _ in range(steps):
        corr = np.correlate(x, 1.0 - x, mode="full") * dx
        k_star = int(np.argmax(corr))
        shift = k_star - (n - 1)
        # Find coordinates that most positively contribute to c[k_star]
        # c[k] = sum_j h[j]*(1-h[j-shift]) for valid indices
        contrib = np.zeros(n)
        for j in range(n):
            j2 = j - shift
            if 0 <= j2 < n:
                contrib[j] += x[j] * (1.0 - x[j2])
            j3 = j + shift
            if 0 <= j3 < n:
                contrib[j] += -x[j3] * (1.0 - x[j])
        # Reduce high-contribution coords, increase low ones
        top_k = max(1, n // 10)
        top_idx = np.argsort(contrib)[-top_k:]
        bot_idx = np.argsort(contrib)[:top_k]
        for ti in top_idx:
            for bi in bot_idx:
                if x[ti] > 0.05 and x[bi] < 0.95:
                    d = lr * min(x[ti] - 0.02, 1.0 - x[bi] - 0.02)
                    x[ti] -= d
                    x[bi] += d
                    break
        x = np.clip(x, 0.0, 1.0)
        s = np.sum(x)
        if abs(s - tgt) > 1e-10 and s > 1e-12:
            x = x * (tgt / s)
            x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        h0_raw = np.array(initial_h_values, dtype=float)
    else:
        h0_raw = np.full(191, 0.5)
    n0 = len(h0_raw)
    dx0 = 2.0 / n0
    h0 = repair(h0_raw, n0)

    best_h = h0.copy()
    best_n = n0
    best_val = compute_c5(h0, dx0)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Deep polish of parent (40% budget)
    taus1 = [0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    x = h0.copy()
    for tau in taus1:
        if time_left() < 50:
            break
        x = slsqp_optimize(x, n0, dx0, tau, maxiter=300)
        v = compute_c5(x, dx0)
        if v < best_val:
            best_val = v
            best_h = x.copy()

    # Phase 2: Targeted adjustment + re-optimize (15% budget)
    if time_left() > 60:
        x_t = target_adjust(best_h, best_n, 2.0 / best_n, lr=0.003, steps=30)
        x_t = slsqp_optimize(x_t, best_n, 2.0 / best_n, 0.01, maxiter=200)
        v = compute_c5(x_t, 2.0 / best_n)
        if v < best_val:
            best_val = v
            best_h = x_t.copy()

    # Phase 3: Diverse restarts (25% budget)
    restart_configs = [
        (0.04, 0.02), (0.08, 0.01), (0.15, 0.03),
        (0.03, 0.01), (0.12, 0.05), (0.20, 0.01),
    ]
    for sc, tau_r in restart_configs:
        if time_left() < 25:
            break
        x_s = perturb(best_h, best_n, sc, rng)
        x_e = slsqp_optimize(x_s, best_n, 2.0 / best_n, tau_r, maxiter=200)
        v = compute_c5(x_e, 2.0 / best_n)
        if v < best_val:
            x_e = slsqp_optimize(x_e, best_n, 2.0 / best_n, 0.003, maxiter=200)
            v2 = compute_c5(x_e, 2.0 / best_n)
            if v2 < best_val:
                best_val = v2
                best_h = x_e.copy()

    # Phase 4: Finer grid 2n+1 (20% budget)
    if time_left() > 80:
        n1 = 2 * best_n + 1
        dx1 = 2.0 / n1
        x_p = project_grid(best_h, best_n, n1)
        for tau in [0.08, 0.03, 0.01, 0.003, 0.001]:
            if time_left() < 15:
                break
            x_p = slsqp_optimize(x_p, n1, dx1, tau, maxiter=200)
            v = compute_c5(x_p, dx1)
            if v < best_val:
                best_val = v
                best_h = x_p.copy()
                best_n = n1

    # Phase 5: Final polish with smallest tau
    if time_left() > 20:
        dx_b = 2.0 / best_n
        x_f = slsqp_optimize(best_h, best_n, dx_b, 0.0005, maxiter=500)
        v = compute_c5(x_f, dx_b)
        if v < best_val:
            best_val = v
            best_h = x_f.copy()

    dx_f = 2.0 / best_n
    best_h = repair(best_h, best_n)
    final_val = compute_c5(best_h, dx_f)
    return (best_h, final_val, best_n)
