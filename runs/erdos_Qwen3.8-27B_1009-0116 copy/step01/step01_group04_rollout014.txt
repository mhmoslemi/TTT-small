import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

_gdx = 1.0
_gtau = 1.0
_cN = 0

def _obj(x):
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    sh = np.clip((corr - m) / _gtau, -50.0, 0.0)
    return (m + _gtau * np.log(np.sum(np.exp(sh)))) * _gdx

def _cfun(x):
    return np.sum(x) - _cN / 2.0

def _cjac(x):
    return np.ones(_cN)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_sum_box(h, n):
    for _ in range(100):
        s = np.sum(h) - n / 2.0
        h = h - s / n
        h = np.clip(h, 0.0, 1.0)
        if abs(np.sum(h) - n / 2.0) < 1e-10:
            break
    return h

def repair(x, n):
    return project_sum_box(np.clip(x, 0.0, 1.0), n)

def slsqp_step(x0, n, dx, tau, mi=400, ft=1e-14):
    global _gdx, _gtau, _cN
    _cN = n
    _gdx = dx
    _gtau = tau
    bd = [(0.0, 1.0)] * n
    cs = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    try:
        r = minimize(_obj, x0, method='SLSQP', bounds=bd,
                     constraints=cs, options={'maxiter': mi, 'ftol': ft})
        return repair(r.x, n)
    except Exception:
        return x0.copy()

def project_grid(x_src, n_src, n_dst):
    xs = np.linspace(0.0, 2.0, n_src, endpoint=True)
    xd = np.linspace(0.0, 2.0, n_dst, endpoint=True)
    xp = interp1d(xs, x_src, kind='linear', fill_value='extrapolate',
                  bounds_error=False)(xd)
    return repair(xp, n_dst)

def perturb_gauss(x, n, scale, rng):
    return repair(x + scale * rng.standard_normal(n), n)

def perturb_binary(x, n, frac, rng):
    x2 = x.copy()

    k = max(1, int(mask.sum()))
    x2[mask] = np.clip(x2[mask] + rng.standard_normal(k) * 0.3, 0, 1)
    return repair(x2, n)

def refine_schedule(x0, n, dx, time_limit, rng):
    t0 = time.time()
    x = repair(x0, n).copy()
    best = x.copy()
    best_val = compute_c5(x, dx)

    taus = [1.0, 0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

    for tau in taus:
        if time.time() - t0 > time_limit - 1.5:
            break
        # Pass 1: optimize current
        x_new = slsqp_step(x, n, dx, tau, 400, 1e-14)
        val = compute_c5(x_new, dx)
        if val < best_val:
            best_val = val
            best = x_new.copy()
            x = x_new.copy()
        # Pass 2: slight perturbation then optimize (if time allows)
        if time.time() - t0 < time_limit - 4.0:
            scale = min(0.06, 0.15 * tau)
            x_p = perturb_gauss(x, n, scale, rng)
            x_p = slsqp_step(x_p, n, dx, tau, 250, 1e-14)
            val_p = compute_c5(x_p, dx)
            if val_p < best_val:
                best_val = val_p
                best = x_p.copy()
                x = x_p.copy()

    return best, best_val
