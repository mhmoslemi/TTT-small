import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fast_project(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-9
    else:
        mask = h > 1e-9
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Fix any residual
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        idx = np.argmin(h) if diff2 > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff2, 0, 1)
    return h

def sa_step(h, rng, n):
    """Generate a perturbed copy of h."""
    h_new = h.copy()
    strat = rng.randint(4)
    if strat == 0:
        # Flip one bit
        i = rng.randint(n)
        h_new[i] = 1.0 - h_new[i]
    elif strat == 1:
        # Flip two bits
        i, j = rng.choice(n, 2, replace=False)
        h_new[i] = 1.0 - h_new[i]
        h_new[j] = 1.0 - h_new[j]
    elif strat == 2:
        # Small nudge on 1-3 points
        k = rng.randint(1, 4)
        idx = rng.choice(n, k, replace=False)
        h_new[idx] += rng.uniform(-0.3, 0.3, k)
    else:
        # Swap-like: move value from one point to another
        i, j = rng.choice(n, 2, replace=False)
        delta = rng.uniform(-0.5, 0.5)
        h_new[i] += delta
        h_new[j] -= delta
    return h_new

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fast_project(h, ts):
    h = np.clip(h, 0.0, 1.0)
    diff = ts - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-9
    else:
        mask = h > 1e-9
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff2 = ts - h.sum()
    if abs(diff2) > 1e-10:
        idx = np.argmin(h) if diff2 > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff2, 0, 1)
    return h

def sa_perturb(h, rng, n):
    h_new = h.copy()
    s = rng.randint(4)
    if s == 0:
        i = rng.randint(n)
        h_new[i] = 1.0 - h_new[i]
    elif s == 1:
        i, j = rng.choice(n, 2, replace=False)
        h_new[i] = 1.0 - h_new[i]
        h_new[j] = 1.0 - h_new[j]
    elif s == 2:
        k = rng.randint(1, 4)
        idx = rng.choice(n, k, replace=False)
        h_new[idx] += rng.uniform(-0.25, 0.25, k)
    else:
        i, j = rng.choice(n, 2, replace=False)
        d = rng.uniform(-0.6, 0.6)
        h_new[i] += d
        h_new[j] -= d
    return h_new

def sa_optimize(h, n, ts, T0, T_min, cool, max_iter, rng, tlim):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur = best_c5
    T = T0
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > tlim:
            break
        T = max(T * cool, T_min)
        hn = sa_perturb(h, rng, n)
        hn = fast_project(hn, ts)
        cn = compute_c5(hn)
        d = cn - cur
        if d < 0 or (T > T_min * 1.01 and rng.random() < np.exp(-d / T)):
            h = hn
            cur = cn
            if cn < best_c5 - 1e-14:
                best_c5 = cn
                best_h = h.copy()
        if it > 2000 and it % 8000 == 0 and cur > best_c5 + 0.004:
            h = best_h.copy()
            cur = best_c5
            T = min(T * 3.0, T0 * 0.4)
    return best_h, best_c5

def local_search(h, n, ts, tlim):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    while time.time() - t0 < tlim:
        improved = False
        for j in range(n):
            if time.time() - t0 > tlim:
                break
            old = h[j]
            for val in (0.0, 1.0):
                if abs(val - old) < 1e-6:
                    continue
                h[j] = val
                h = fast_project(h, ts)
                c = compute_c5(h)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = h.copy()
                    improved = True
                else:
                    h[j] = old
                    h = fast_project(h, ts)
        if not improved:
            break
    return best_h, best_c5
