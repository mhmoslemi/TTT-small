    cv = _corr(h, dx)
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

    cv = _corr(h, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()

        jp = j + s
        vp = (jp >= 0) & (jp < n)

3. If h[i] was the only interior value being changed, and it moved from 0.5 to 0.3, the sum decreased by 0.2. To restore the sum, λ becomes more negative (or less positive), pushing other values up and h[i] back up slightly.

So after projection, h[i] might be 0.32 instead of 0.3. The 1-D optimum was for h[i]=0.3 with all other values fixed, but after projection, other values changed too.

This is inherent to the projected approach. The coordinate descent will still make progress overall, just not exactly optimize each coordinate.

To mitigate: I can do multiple passes. In each pass, the effect of the projection gets smaller as the solution converges.

import numpy as np
import time

def _corr(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _c5(h, dx):
    return _corr(h, dx).max()

def _proj(h, target):
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _coord_pass(h, dx, target, grid_size=100, rng=None):
    n = len(h)
    h = h.copy()
    grid = np.linspace(0.0, 1.0, grid_size)
    order = np.arange(n) if rng is None else rng.permutation(n)
    for i in order:
        old = h[i]
        c_old = _corr(h, dx)
        coeffs = np.zeros(2 * n - 1)
        for idx in range(2 * n - 1):
            s = idx - (n - 1)
            if s == 0:
                continue
            c = 0.0
            if 0 <= i + s < n:
                c += (1.0 - h[i + s]) * dx
            if 0 <= i - s < n:
                c -= h[i - s] * dx
            coeffs[idx] = c
        mask = np.ones(2 * n - 1, dtype=bool)
        mask[n - 1] = False
        deltas = grid - old
        c_nz = c_old[mask][None, :] + deltas[:, None] * coeffs[mask][None, :]
        c_z = c_old[n - 1] + deltas * (1.0 - grid - old)
        cmax = np.maximum(c_nz.max(axis=1), c_z)
        bi = cmax.argmin()
        h[i] = grid[bi]
    return _proj(h, target)

def _smooth_grad(h, dx, eps):
    cv = _corr(h, dx)
    mx = cv.max()
    w = np.exp((cv - mx) / eps)
    w /= w.sum()
    n = len(h)
    g = np.zeros(n)
    j = np.arange(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-15:
            continue
        wt = w[idx] * dx
        s = idx - (n - 1)
        jm = j - s
        vm = (jm >= 0) & (jm < n)
        if vm.any():
            g[vm] += wt * (1.0 - h[jm[vm]])
        jp = j + s
        vp = (jp >= 0) & (jp < n)
        if vp.any():
            g[vp] -= wt * h[jp[vp]]
    return g

def _smooth_obj(h, dx, eps):
    cv = _corr(h, dx)
    mx = cv.max()
    return mx + eps * np.log(np.sum(np.exp((cv - mx) / eps)))

def _pgd(h, dx, target, eps, lr, n_iters):
    for _ in range(n_iters):
        g = _smooth_grad(h, dx, eps)
        f_old = _smooth_obj(h, dx, eps)
        h_new = _proj(h - lr * g, target)
        f_new = _smooth_obj(h_new, dx, eps)
        if f_new <= f_old:
            h = h_new
            lr = min(lr * 1.03, 1.5)
        else:
            lr = max(lr * 0.7, 0.01)
    return h

def _optimize_n(h_init, dx, target, rng, t_end):
    n = len(h_init)
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = _c5(h, dx)
    
    # Phase 1: Coordinate descent with increasing grid resolution
    for gs in [50, 100, 150, 200]:
        for _ in range(3):
            if time.time() > t_end - 20:
                return best_h, best_c5
            h = _coord_pass(h, dx, target, grid_size=gs, rng=rng)
            c = _c5(h, dx)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    
    # Phase 2: Smoothed gradient + coordinate
    for eps in [0.05, 0.02, 0.005, 0.002]:
        if time.time() > t_end - 15:
            return best_h, best_c5
        h = _pgd(h, dx, target, eps, lr=0.3, n_iters=20)
        h = _coord_pass(h, dx, target, grid_size=200, rng=rng)
        c = _c5(h, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    
    # Phase 3: Perturbations + re-optimize
    stall = 0
    while time.time() < t_end - 10:
        amp = 0.03 + 0.1 * (stall / 10.0)
        h_pert = _proj(best_h + rng.normal(0, amp, n), target)
        h_pert = _coord_pass(h_pert, dx, target, grid_size=100, rng=rng)
        h_pert = _coord_pass(h_pert, dx, target, grid_size=200, rng=rng)
        c = _c5(h_pert, dx)
        if c < best_c5:
            best_c5 = c
            best_h = h_pert.copy()
            stall = 0
        else:
            stall += 1
        if stall > 30:
            break
    
    return best_h, best_c5

def _try_structured_init(n, rng):
    dx = 2.0 / n
    target = n / 2.0
    # Try a few structured patterns
    candidates = []
    # Blurred ramp
    x = np.linspace(0, 1, n)
    for t in [0.3, 0.5, 0.7]:
        f = 0.5 + 0.4 * np.sin(2 * np.pi * x * (1 + rng.randint(1, 4)))
        f = np.clip(f * t + 0.5 * (1 - t), 0, 1)
        candidates.append(_proj(f, target))
    # Two-block
    for r in [0.2, 0.3, 0.35]:
        blk = np.zeros(n)
        w = int(n * r)
        blk[:w] = 1.0
        blk2 = np.zeros(n)
        blk2[n - w + int(n * 0.1):n - int(n * 0.1)] = 1.0
        for f in [blk, blk2]:
            s = f.sum()
            if s > 0:
                f = f * (target / s)
                candidates.append(_proj(f, target))
    # Near-uniform with noise
    for amp in [0.1, 0.2, 0.3]:
        f = 0.5 + amp * rng.randn(n)
        candidates.append(_proj(f, target))
    return candidates

    t0 = time.time()
