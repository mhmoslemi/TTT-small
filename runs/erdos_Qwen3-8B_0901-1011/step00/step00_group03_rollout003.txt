```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
import sys

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is a pre-loaded numpy array

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can be increased if time allows
    max_n_points = 200  # Max number of points to consider

    # Try different numbers of points up to max_n_points
    for n_points in range(100, max_n_points + 1, 50):
        # Create initial guess based on the provided initial_h_values
        if n_points <= len(initial_h_values):
            h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

        # Define variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2]
        objective = cp.Minimize(0)  # We'll use constraints to enforce the objective

        # Define the overlap integral for all possible shifts
        # We'll use a simplified approach to approximate the max overlap
        # By evaluating the overlap for a few key shifts
        shifts = np.linspace(-2, 2, 100)
        overlap_values = []
        for shift in shifts:
            # Create a shifted version of h
            h_shifted = np.zeros(n_points)
            for i in range(n_points):
                x = np.linspace(0, 2, n_points)[i]
                x_shifted = x + shift
                if 0 <= x_shifted <= 2:
                    idx = np.searchsorted(np.linspace(0, 2, n_points), x_shifted)
                    if idx < n_points:
                        h_shifted[idx] = 1 - h[i]
            overlap = np.dot(h.value, h_shifted) if h.value is not None else 0
            overlap_values.append(overlap)

        # Use the maximum overlap as the objective
        # We'll use a simplified approach to minimize the maximum overlap
        # By enforcing a constraint that all overlaps are below a certain threshold
        # We'll use a binary search to find the minimal threshold
        # Start with a threshold that is lower than the current best
        low = 0.0
        high = 1.0
        for _ in range(50):  # Perform a few iterations of binary search
            mid = (low + high) / 2
            # Define constraints for this threshold
            constraints = [cp.sum(h) == n_points / 2]
            for i in range(len(shifts)):
                # Create a shifted version of h
                h_shifted = np.zeros(n_points)
                for j in range(n_points):
                    x = np.linspace(0, 2, n_points)[j]
                    x_shifted = x + shifts[i]
                    if 0 <= x_shifted <= 2:
                        idx = np.searchsorted(np.linspace(0, 2, n_points), x_shifted)
                        if idx < n_points:
                            h_shifted[idx] = 1 - h[j]
                # Add constraint that the overlap is less than mid
                constraints.append(cp.sum(h * h_shifted) <= mid)
            # Solve the optimization problem
            try:
                problem = cp.Problem(objective, constraints)
                problem.solve(solver=cp.SCS, eps=1e-5)
                if problem.status == 'optimal':
                    current_c5 = mid
                    best_c5 = min(best_c5, current_c5)
                    best_h = h.value
                    break
            except:
                pass
            # Update the bounds based on the result
            if problem.status == 'optimal':
                high = mid
            else:
                low = mid

        # Evaluate the current best solution
        if best_h is not None:
            current_c5 = evaluate_erdos_solution(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = best_h

        # Check if we've exceeded the budget
        if time.time() - start_time > budget_s:
            break

    # Return the best solution found
    return best_h, best_c5, n_points
```
