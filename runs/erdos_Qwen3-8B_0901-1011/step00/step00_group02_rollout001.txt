import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 points for speed
    dx = 2.0 / n_points
    max_iterations = 100
    improvement_threshold = 1e-5
    best_c5 = float('inf')
    best_h = None
    
    # Initial guess
    h = np.array(initial_h_values)
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        # Evaluate current C5
        c5 = evaluate_erdos_solution(h, n_points, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Check if we've improved enough
        if c5 - best_c5 < improvement_threshold:
            break
        
        # Try to improve the solution with optimization
        # Define variables
        x = cp.Variable(n_points, lb=0, ub=1)
        # Constraint: integral of h is 1
        constraint = cp.sum(x) == n_points / 2
        
        # Objective: minimize the maximum overlap
        # We use a convex relaxation by minimizing the maximum of several expressions
        # We approximate the max by using a convex hull trick
        # For simplicity, we use a heuristic to minimize the maximum overlap
        # This is a simplified version of the problem
        
        # We will use a simple gradient descent approach for speed
        # Define the overlap function
        def overlap_func(x, dx, n_points):
            h = x
            # Compute the correlation
            corr = np.correlate(h, 1 - h, mode="full")
            # Take the maximum value
            return np.max(corr) * dx
        
        # Define the objective function for optimization
        def objective_func(x_val):
            return overlap_func(x_val, dx, n_points)
        
        # Use a simple gradient descent to minimize the overlap
        # We use a fixed step size for simplicity
        learning_rate = 0.01
        for _ in range(100):
            current_c5 = objective_func(h)
            # Compute gradient numerically
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                grad[i] = (objective_func(h_plus) - current_c5) / 1e-5
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Check if we've improved enough
        current_c5 = objective_func(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Check time budget
        if time.time() - start_time > budget_s - 10:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    return best_h, final_c5, n_points
