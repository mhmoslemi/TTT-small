import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    
    # Ensure the sum is exactly n_points / 2
    if not np.isclose(sum_h, n_points / 2):
        h = h / sum_h * (n_points / 2)
    
    def compute_c5(h):
        dx = 2.0 / len(h)
        h_reshaped = np.concatenate([h, np.zeros_like(h)])
        one_minus_h = 1 - h
        one_minus_h_reshaped = np.concatenate([one_minus_h, np.zeros_like(one_minus_h)])
        corr = signal.correlate(h_reshaped, one_minus_h_reshaped, mode='full')
        c5 = np.max(corr * dx)
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define optimization problem
    h_var = cp.Variable(n_points, lb=0, ub=1)
    objective_func = compute_c5(h_var)
    constraints = [constraint_sum(h_var) == 0]
    
    # Solve using CVXPY
    prob = cp.Problem(cp.Minimize(objective_func), constraints)
    try:
        result = prob.solve(solver=cp.SCS, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
        if result is not None:
            h_opt = h_var.value
        else:
            h_opt = h
    except:
        h_opt = h
    
    # Ensure the sum is correct
    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
    
    # Evaluate the result
    c5_bound = compute_c5(h_opt)
    
    # Return the result
    return (h_opt.tolist(), c5_bound, n_points)
