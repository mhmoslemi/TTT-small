```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Start with 100 points for faster evaluation
    max_iterations = 100
    tolerance = 1e-4
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points)
    
    # Initial solution
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute the overlap integral
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Try to improve the solution
        # Define the optimization problem
        h_var = cp.Variable(n_points, bounds=(0, 1))
        constraints = [cp.sum(h_var) == n_points / 2]
        
        # Objective: minimize the maximum overlap
        overlap = signal.correlate(h_var.value if h_var.value is not None else h, 1 - h_var.value if h_var.value is not None else h, mode="full") * dx
        max_overlap = cp.Max(overlap)
        objective = cp.Minimize(max_overlap)
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=tolerance)
            if prob.status == 'OPTIMAL':
                h = h_var.value
        except:
            # If optimization fails, use a gradient descent approach
            h = h + 0.01 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping if no improvement
        if np.abs(c5 - best_c5) < tolerance:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    
    return best_h, final_c5, n_points
```
