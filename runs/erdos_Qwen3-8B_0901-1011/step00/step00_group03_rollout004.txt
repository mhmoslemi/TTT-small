```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points for faster initial evaluation
    dx = 2.0 / n_points
    max_iter = 100  # Max number of iterations for optimization
    tolerance = 1e-4  # Tolerance for convergence
    
    # Initial guess
    h = initial_h_values.copy()[:n_points]
    h = h / (np.sum(h) * dx) * (n_points / 2)  # Normalize to satisfy the integral constraint
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points, dx)
    
    # Optimization loop
    for _ in range(max_iter):
        # Compute current C5
        current_c5 = best_c5
        
        # Define the optimization variables
        h_var = cp.Variable(n_points, name='h')
        constraints = [
            cp.Minimize(current_c5),
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]
        
        # Define the objective function to minimize the maximum overlap
        # We use a convex relaxation by minimizing the maximum using a convex constraint
        # This is a simplified approach for demonstration; for better performance, consider more sophisticated methods
        
        # Compute the overlap for each shift
        overlaps = []
        for k in range(1, n_points):
            shifted_h = np.roll(h_var.value, k)
            overlap = cp.sum(h_var * (1 - shifted_h))
            overlaps.append(overlap)
        
        # Maximize the overlap
        max_overlap = cp.Max(overlaps)
        
        # Add the constraint to minimize the maximum overlap
        constraints.append(cp.Minimize(max_overlap))
        
        # Solve the optimization problem
        try:
            prob = cp.Problem(cp.Minimize(max_overlap), constraints)
            result = prob.solve(solver=cp.SCS, max_iter=100, eps=1e-6)
            
            if result is not None and result < current_c5:
                best_h = h_var.value
                best_c5 = result
                print(f"New C5: {best_c5} at iteration {_}")
        except:
            print("Optimization failed, using previous solution")
            break
        
        # Early stopping if the improvement is below tolerance
        if abs(best_c5 - current_c5) < tolerance:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    print(f"Final C5: {final_c5}")
    
    # Ensure the solution is within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:
        print("Time budget exceeded, returning best solution found so far")
    
    return best_h, final_c5, n_points
```
