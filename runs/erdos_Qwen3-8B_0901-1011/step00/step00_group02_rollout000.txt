import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 45  # Starting with the initial construction
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    # Function to compute C5
    def compute_c5(h_values):
        h = h_values.copy()
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to evaluate the solution
    def evaluate(h_values):
        return compute_c5(h_values)
    
    # Function to optimize the solution using gradient descent with momentum
    def optimize_h(h_initial, max_iter=1000, lr=0.01, momentum=0.9, tol=1e-6):
        h = h_initial.copy()
        v = np.zeros_like(h)
        best_h = h.copy()
        best_c5 = evaluate(h)
        
        for i in range(max_iter):
            # Compute gradient
            grad = np.zeros_like(h)
            for k in range(1, len(h) - 1):
                # Compute the derivative of the overlap integral with respect to h[i]
                # This is a simplified approximation using finite differences
                h_plus = h.copy()
                h_plus[i] += 1e-6
                h_minus = h.copy()
                h_minus[i] -= 1e-6
                c5_plus = compute_c5(h_plus)
                c5_minus = compute_c5(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-6)
            
            # Update velocity and position
            v = momentum * v - lr * grad
            h = h + v
            
            # Ensure h is within [0, 1]
            h = np.clip(h, 0, 1)
            
            # Update best solution
            current_c5 = evaluate(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Check for convergence
            if np.abs(current_c5 - best_c5) < tol:
                break
        
        return best_h, best_c5
    
    # Function to optimize the solution using CVXPY
    def optimize_h_cvxpy(h_initial, max_iter=100, tol=1e-6):
        h = cp.Variable(h_initial.shape)
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        obj = cp.Maximize(cp.max(signal.correlate(h.value, 1 - h.value, mode="full") * dx))
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.SCS, max_iter=max_iter, eps=tol)
        if result is not None:
            return h.value, obj.value
        return h_initial, evaluate(h_initial)
    
    # Try gradient descent optimization
    optimized_h, c5 = optimize_h(h, max_iter=500, lr=0.01, momentum=0.9, tol=1e-6)
    if time.time() - start_time > budget_s - 10:
        return optimized_h, c5, n_points
    
    # Try CVXPY optimization
    optimized_h, c5 = optimize_h_cvxpy(optimized_h, max_iter=100, tol=1e-6)
    if time.time() - start_time > budget_s - 10:
        return optimized_h, c5, n_points
    
    # Try further refinement with gradient descent
    optimized_h, c5 = optimize_h(optimized_h, max_iter=500, lr=0.005, momentum=0.8, tol=1e-6)
    if time.time() - start_time > budget_s - 10:
        return optimized_h, c5, n_points
    
    return optimized_h, c5, n_points
