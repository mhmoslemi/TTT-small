import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points
    max_points = 1000  # Maximum number of points to consider
    max_iterations = 100  # Maximum number of iterations
    improvement_threshold = 1e-5  # Stop if improvement is below threshold
    
    # Initial guess
    h = initial_h_values.copy()
    h = h[:n_points]  # Truncate to n_points
    h = h / (n_points / 2)  # Normalize to satisfy the integral constraint
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h = h / (len(h) / 2)  # Normalize to ensure the integral is 1
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to optimize h using gradient descent
    def optimize_h(h, n_points, max_iter=100, lr=0.01):
        h = np.array(h)
        h = h / (n_points / 2)  # Normalize to ensure the integral is 1
        h = h.clip(0, 1)  # Ensure values are in [0, 1]
        
        for _ in range(max_iter):
            # Compute the overlap integral
            dx = 2.0 / len(h)
            corr = signal.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            
            # Compute the gradient (approximate using finite differences)
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = h_plus / (len(h) / 2)
                h_plus = h_plus.clip(0, 1)
                corr_plus = signal.correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / 1e-5
            
            # Update h
            h -= lr * grad
            h = h.clip(0, 1)
            h = h / (len(h) / 2)
        
        return h
    
    # Main optimization loop
    for n_points in range(100, max_points + 1, 50):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Optimize h for current n_points
        h_opt = optimize_h(h, n_points)
        c5 = evaluate_overlap(h_opt)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        
        # Print progress
        print(f"n_points={n_points}, c5={best_c5:.6f}")
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h)
    
    # Return the best solution
    return best_h, final_c5, len(best_h)
