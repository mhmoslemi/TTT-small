import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from functools import lru_cache

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are available

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    n_points = 70
    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial[:n_points]
    
    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / len(h)
        h = np.clip(h, 0, 1)
        h_shifted = np.zeros(len(h) * 2 - 1)
        h_shifted[:len(h)] = h
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute the overlap integral for a given h
    def objective(h):
        return compute_c5(h)
    
    # Function to compute the constraint (integral of h is 1)
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization with scipy.optimize
    def optimize_h(h_initial):
        h = cp.Variable(n_points)
        objective = cp.Minimize(compute_c5(h))
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        problem = cp.Problem(objective, constraints)
        result = problem.solve(solver=cp.SCS, max_iter=1000)
        if result is None:
            return h_initial
        return h.value
    
    # Run the optimization
    h_opt = optimize_h(h_initial)
    
    # Evaluate the result
    c5_bound = compute_c5(h_opt)
    
    # If we have time, try a larger n_points
    if time.time() - start_time < budget_s - 10:
        n_points = 100
        h_initial = np.random.rand(n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        def compute_c5_large(h):

            h = np.clip(h, 0, 1)

            h_shifted[:len(h)] = h
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        def objective_large(h):
            return compute_c5_large(h)
        
        def constraint_large(h):
            return np.sum(h) - n_points / 2
        
        def optimize_h_large(h_initial):
            h = cp.Variable(n_points)

                return h_initial

            h_opt = h_opt_large
            c5_bound = c5_bound_large

    if time.time() - start_time < budget_s:

            c5_candidate = compute_c5(h_opt)
            if c5_candidate < c5_bound:
                c5_bound = c5_candidate
    
    return h_opt, c5_bound, n_points
