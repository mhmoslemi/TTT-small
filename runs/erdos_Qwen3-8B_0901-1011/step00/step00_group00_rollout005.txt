```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Start with the initial construction
    n_points_initial = len(initial_h_values)
    h_initial = initial_h_values.copy()
    c5_initial = evaluate_erdos_solution(h_initial)
    best_c5 = c5_initial
    best_h = h_initial
    best_n_points = n_points_initial

    # Try increasing n_points in steps
    for n_points in [n_points_initial * 2, n_points_initial * 3, n_points_initial * 4]:
        if time.time() - start_time > budget_s - 10:
            break

        # Check if we can fit the samples in the interval [0, 2]
        if n_points > 1000:
            continue

        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy the integral constraint

        # Define the objective function
        def objective(h_vec):
            h = h_vec.reshape(n_points)
            # Compute the overlap integral for all shifts
            corr = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5

        # Define the constraint: sum(h) == n_points / 2
        def constraint(h_vec):
            return np.sum(h_vec) - n_points / 2

        # Use scipy.optimize to minimize the objective
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[{'type': 'eq', 'fun': constraint}]
        )

        if result.success:
            h_opt = result.x
            c5 = objective(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n_points = n_points

    # Try a few local refinements
    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break

        # Perturb the current best solution
        perturbation = np.random.normal(0, 0.05, best_n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (best_n_points / 2) / np.sum(h_perturbed)

        # Evaluate the perturbed solution
        c5 = evaluate_erdos_solution(h_perturbed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed
            best_n_points = len(h_perturbed)

    # Try a few convex optimization steps
    for _ in range(2):
        if time.time() - start_time > budget_s - 10:
            break

        n_points = best_n_points
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        h = cp.Variable(n_points)
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]

        # Define the objective
        def objective(h_vec):
            h = h_vec.reshape(n_points)
            corr = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5

        # Use CVXPY to minimize the objective
        prob = cp.Problem(cp.Minimize(objective(h)), constraints)
        try:
            result = prob.solve(solver=cp.SCS, eps=1e-4)
            if result is not None and result.success:
                h_opt = h.value
                c5 = objective(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                    best_n_points = n_points
        except:
            pass

    return best_h, best_c5, best_n_points
```
