import numpy as np
import scipy.optimize as optimize
import time
import copy

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    target_sum = n_points / 2
    if not np.isclose(sum_h, target_sum):
        initial_h = (initial_h * target_sum) / sum_h

    def compute_c5(h):
        h = np.array(h)
        shifted = np.roll(h, shift=1, axis=0)
        overlap = np.correlate(h, 1 - shifted, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    best_h = copy.deepcopy(initial_h)
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        try:
            result = optimize.minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                current_h = result.x
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
        except:
            pass

        # Add some random perturbations to escape local minima
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturb, 0, 1)
        perturbed_c5 = compute_c5(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5

    return best_h, best_c5, n_points
