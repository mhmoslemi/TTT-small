import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=10, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values.copy()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    target_sum = n_points / 2
    initial_h = initial_h * (target_sum / sum_h) if sum_h != 0 else np.zeros(n_points) + 0.5

    def objective(h):
        h = h.reshape(n_points)
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    initial_guess = initial_h.copy()

    start_time = time.time()
    result = opt.minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    elapsed = time.time() - start_time

    if elapsed >= budget_s:
        return (initial_h, objective(initial_h), n_points)

    best_h = result.x
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
