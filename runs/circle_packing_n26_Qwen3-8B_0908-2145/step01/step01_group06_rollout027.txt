import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import NonlinearConstraint

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: place circles in a hexagonal-like grid
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Estimate approximate grid dimensions
    row_count = int(np.ceil(np.sqrt(num_circles)))
    col_count = int(np.ceil(num_circles / row_count))
    
    # Place circles in a hexagonal grid
    for i in range(num_circles):
        col = i % col_count
        row = i // col_count
        x = col / (col_count - 1) if col_count > 1 else 0.5
        y = row / (row_count - 1) if row_count > 1 else 0.5
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = 0.05  # Initial guess for radii

    # Define objective function to maximize sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Minimize negative sum to maximize sum

    # Define constraint: circles must not overlap
    def constraint_overlap(x):
        centers_vec = x[:num_circles * 2].reshape(num_circles, 2)
        radii_vec = x[num_circles * 2:]
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_vec[i, 0] - centers_vec[j, 0]
                dy = centers_vec[i, 1] - centers_vec[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii_vec[i] + radii_vec[j]) >= -1e-6)
        return np.array(constraints)

    # Define constraint: circles must be inside the unit square
    def constraint_boundary(x):
        centers_vec = x[:num_circles * 2].reshape(num_circles, 2)
        radii_vec = x[num_circles * 2:]
        constraints = []
        for i in range(num_circles):
            x_center, y_center = centers_vec[i]
            r = radii_vec[i]
            constraints.append(x_center - r >= -1e-6)
            constraints.append(x_center + r <= 1.0 + 1e-6)
            constraints.append(y_center - r >= -1e-6)
            constraints.append(y_center + r <= 1.0 + 1e-6)
        return np.array(constraints)

    # Define bounds: center coordinates in [0, 1], radius >= 0
    bounds = Bounds([0.0, 0.0], [1.0, 1.0])  # for centers
    bounds = Bounds(np.concatenate([np.repeat([0.0, 0.0], num_circles), np.zeros(num_circles)]),
                    np.concatenate([np.repeat([1.0, 1.0], num_circles), np.inf * np.ones(num_circles)]))

    # Initial guess for optimization
    x0 = np.concatenate([centers.flatten(), radii])

    # Define constraints
    nonlinear_constraint_overlap = NonlinearConstraint(constraint_overlap, -np.inf, 0)
    nonlinear_constraint_boundary = NonlinearConstraint(constraint_boundary, -np.inf, 0)

    # Run the optimization
    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=[nonlinear_constraint_overlap, nonlinear_constraint_boundary],
        method='SLSQP',
        tol=1e-8
    )

    # Extract final results

    if not valid:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
