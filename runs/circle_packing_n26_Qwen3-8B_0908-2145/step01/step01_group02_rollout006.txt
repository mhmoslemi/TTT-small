import numpy as np
import scipy.optimize as optimize

def run_packing():

    n = 26

    # Initial hexagonal packing: approximate number of rows
    rows = 5
    cols_per_row = 5
    col_spacing = 1.0 / cols_per_row
    row_spacing = np.sqrt(3) / 2.0 / rows

    # Generate initial circle centers
    centers = np.zeros((n, 2))
    index = 0
    for row in range(rows):
        for col in range(cols_per_row):
            x = col * col_spacing
            y = row * row_spacing
            centers[index] = [x, y]
            index += 1

    # Initial estimate for radii (Maximum possible radius for the first circle)
    radii = np.zeros(n)
    radii[0] = 0.25  # Initial estimate, will be optimized

    # Define the optimization variables: [x1, y1, r1, x2, y2, r2, ...]
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, 1.0)])  # x, y, r

    # Objective function: maximize the sum of radii
    def objective(x):
        return -np.sum(x[2::3])  # Negative for minimization

    # Constraints: Each circle must be inside the square
    def constraint_inside(x):
        return np.array([x[i] - x[i+2] for i in range(0, len(x), 3)])
    constraints_inside = optimize.constraints.Bounds([0.0, 0.0, 0.0], [1.0, 1.0, 1.0])

    # Constraints: Each pair of circles must not overlap
    def constraint_no_overlap(x):
        distances = np.zeros(n * (n - 1) // 2)
        index = 0
        for i in range(n):
            for j in range(i + 1, n):
                xi, yi, ri = x[i*3], x[i*3 + 1], x[i*3 + 2]
                xj, yj, rj = x[j*3], x[j*3 + 1], x[j*3 + 2]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                dist -= ri + rj
                distances[index] = dist
                index += 1
        return distances

    constraints_no_overlap = optimize.constraints.LinearConstraint(
        np.repeat([1.0 if i % 3 == 0 or i % 3 == 1 else 0.0 for i in range(n * 3)], n), 
        lb=np.repeat(-np.inf, n * 3), 
        ub=np.repeat(0.0, n * 3)

        method='SLSQP',
        bounds=bounds,
        constraints=[constraints_inside, constraints_no_overlap]

    # Extract results
    x = result.x

    radii = np.array([x[i*3+2] for i in range(n)])

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
