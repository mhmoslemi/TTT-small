import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26
    rows = int(np.sqrt(num_circles))
    cols = (num_circles + rows - 1) // rows

    # Initialize centers in a staggered grid
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]

    # Scale centers to fit within the unit square
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    scale = 1.0 / max(max_x, max_y)
    centers *= scale

    # Initial radii as a uniform value (adjustable for optimization)
    radii = np.full(num_circles, 0.05)

    # Define the objective function to maximize the sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Negative for minimization

    # Define constraints: 
    # 1. All circles must be within the unit square
    # 2. No two circles can overlap

    def constraint_radius_bounds(radii_vec):
        return radii_vec >= 0

    def constraint_circle_bounds(radii_vec, centers):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii_vec[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return np.inf
        return 0

    def constraint_circle_overlap(radii_vec, centers):
        for i in range(num_circles):

                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii_vec[i] + radii_vec[j]:
                    return np.inf
        return 0

    # Define bounds for each radius
    bounds = [(0, 1) for _ in range(num_circles)]

    # Initial guess
    initial_guess = radii.copy()

    # Optimize
    result = opt.minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda x: constraint_radius_bounds(x)},
            {'type': 'ineq', 'fun': lambda x: constraint_circle_bounds(x, centers)},
            {'type': 'ineq', 'fun': lambda x: constraint_circle_overlap(x, centers)},

        method='SLSQP',
        tol=1e-8

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
