def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial configuration: place 26 circles in a grid-like formation
    n = 26
    # Start with hexagonal-like arrangement in the unit square
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Distribute circles with hexagonal pattern
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 5.0
        y = row / 5.0
        # Adjust slightly to avoid overlap
        x += np.sin(np.pi * row / 5.0) * 0.1
        y += np.cos(np.pi * col / 5.0) * 0.1
        centers[i] = [x, y]

    # Initial radius based on minimal distance between centers
    min_dist = np.inf
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < min_dist:
                min_dist = dist
    initial_radius = min_dist / 2.0
    radii[:] = initial_radius

    # Optimization objective: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints
    def constraint_bound(radii):
        # Ensure centers are within [0,1]^2
        return np.concatenate([centers[i] - radii[i] for i in range(n)]) >= -1e-8

    def constraint_overlap(radii):
        # Ensure circles do not overlap
        res = []
        for i in range(n):

                dist = np.linalg.norm(centers[i] - centers[j])
                res.append(dist - radii[i] - radii[j])
        return res

    # Reformulate variables
    x = np.concatenate([centers.flatten(), radii])

    bounds = []

        bounds.extend([(0, 1) for _ in range(2)])  # center coordinates
        bounds.extend([(0, 1) for _ in range(1)])  # radius

    # Run optimization
    constraints = [
        opt.SparseConstraint(constraint_bound, [], x),
        opt.SparseConstraint(constraint_overlap, [], x)
    ]

    result = opt.minimize(
        objective,
        x,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,

    if not result.success:

    # Extract results
    centers_final = result.x[:2 * n].reshape(n, 2)
