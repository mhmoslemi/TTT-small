import numpy as np
import scipy.optimize as opt
from typing import Tuple
import warnings

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial grid layout for 26 circles
    # Trying a configuration around 5x5 grid with a small adjustment
    num_circles = 26
    grid_size = 5
    grid = np.array([(i / (grid_size - 1), j / (grid_size - 1)) for i in range(grid_size) for j in range(grid_size)])
    extra_circle = np.array([0.5, 0.5])  # Central position for the extra circle

    # Initial centers (first 25 from grid and one extra)
    centers = np.vstack([grid, extra_circle])
    
    # Initial guess for radii: assume uniform size and scale down to fit geometry
    initial_radius = 1 / (2 * np.sqrt(2))  # safe estimate for a square grid
    radii = np.full(num_circles, initial_radius)

    # Objective function to maximize the sum of radii
    def objective(x):
        # x is a 1D array with dimensions (2*num_circles + num_circles) = 3*num_circles
        # reshape x into (centers, radii)
        centers_flat = x[:2*num_circles].reshape(num_circles, 2)
        radii_flat = x[2*num_circles:].reshape(num_circles)
        
        # Ensure non-negative radii
        radii_flat = np.maximum(radii_flat, 0)
        
        # Objective: maximize the sum of radii
        obj = -np.sum(radii_flat)  # negative for minimization
        return obj

    # Constraints: positions within [0,1]^2 and non-overlapping
    constraints = []

    # Position constraints
    def position_bounds(x):
        centers_flat = x[:2*num_circles].reshape(num_circles, 2)
        radii_flat = x[2*num_circles:].reshape(num_circles)
        for i in range(num_circles):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if not (0 - r_i <= x_i <= 1 + r_i and 0 - r_i <= y_i <= 1 + r_i):
                return np.inf
        return 0

    # Non-overlap constraints
    def non_overlap(x):
        centers_flat = x[:2*num_circles].reshape(num_circles, 2)
        radii_flat = x[2*num_circles:].reshape(num_circles)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-9:
                    return np.inf
        return 0

    # Add constraints
    constraints.append({'type': 'ineq', 'fun': position_bounds})
    constraints.append({'type': 'ineq', 'fun': non_overlap})

    # Initial guess
    x0 = np.hstack([centers.flatten(), radii.flatten()])

    # Set bounds for position coordinates and radii
    bounds = []
    for i in range(num_circles):
        bounds.append((0 - 1, 1 + 1))  # x and y bounds
        bounds.append((0, 1))          # radius bounds

    # Optimization settings
    options = {
        'maxiter': 1000,
        'ftol': 1e-9,
        'disp': False
    }

    # Run optimization
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        result = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, options=options)

    # Extract the result
    x_opt = result.x
    centers_opt = x_opt[:2*num_circles].reshape(num_circles, 2)
    radii_opt = x_opt[2*num_circles:].reshape(num_circles)
    sum_radii = np.sum(radii_opt)

    # Validate the solution

    if not valid:
        raise ValueError(msg)

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
