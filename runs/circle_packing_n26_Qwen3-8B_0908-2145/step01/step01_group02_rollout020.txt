from scipy.optimize import minimize, Bounds, NonlinearConstraint
import random

def run_packing():
    # Initial guess: Random positions and equal radii
    n = 26
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.04)  # Initial estimate

    # Objective function: Maximize sum of radii (minimize negative sum)
    def objective(x):

        return -np.sum(radii_flat)

    # Bounds: [x, y, r] for each circle
    bounds = Bounds(
        np.array([0.0] * (n * 2) + [0.0] * n),
        np.array([1.0] * (n * 2) + [1.0] * n)

    def constraints(x):

        x_flat = x[:n*2]
        r_flat = x[n*2:]
        centers = x_flat.reshape(n, 2)
        radii = r_flat

        # Check boundary constraints
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if not (0 <= x_i - r_i and x_i + r_i <= 1 and 0 <= y_i - r_i and y_i + r_i <= 1):
                return np.array([1.0])  # Violation

        # Check non-overlapping constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return np.array([1.0])  # Violation

        return np.array([0.0])

    # Define nonlinear constraint for boundaries and non-overlaps
    nonlin_cons = NonlinearConstraint(constraints, 0, 0)

    x0 = np.concatenate([centers.reshape(n * 2), radii])

    # Run the optimization
    result = minimize(objective, x0, bounds=bounds, constraints=nonlin_cons, method='SLSQP', tol=1e-8)

    # Validate packing
    centers_opt = result.x[:n*2].reshape(n, 2)
    radii_opt = result.x[n*2:]
    sum_radii = np.sum(radii_opt)

    return (centers_opt, radii_opt, sum_radii)
