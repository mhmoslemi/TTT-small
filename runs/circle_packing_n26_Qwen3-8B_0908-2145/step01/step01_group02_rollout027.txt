from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    # Initial hexagonal grid layout with spacing
    spacing = 0.15  # Initial spacing
    x_centers = np.arange(0.05, 1 - 0.05, spacing)
    y_centers = np.arange(0.05, 1 - 0.05, spacing)

    x_grid, y_grid = np.meshgrid(x_centers, y_centers)
    initial_centers = np.column_stack((x_grid.ravel(), y_grid.ravel()))

    # Limit the number of initial centers to 26
    initial_centers = initial_centers[:n]
    initial_radii = np.full(n, 0.05)

        return -np.sum(radii)

    def constraint_centers(radii):
        return np.array([0.05 <= r <= 0.95 for r in radii])

    def constraint_distances(radii, centers):
        dist_matrix = cdist(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-6:
                    return np.inf
        return 0

    # Optimization problem
    bounds = [(0.0, 0.9) for _ in range(n)]
    constraints = (
        {'type': 'ineq', 'fun': lambda r: constraint_centers(r)},
        {'type': 'ineq', 'fun': lambda r: constraint_distances(r, initial_centers)}
    )

        objective,
        initial_radii,
        args=(initial_centers,),
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )

    centers = initial_centers
    radii = result.x

    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
