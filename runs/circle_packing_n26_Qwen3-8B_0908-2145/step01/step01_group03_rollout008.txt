from scipy.optimize import Bounds, NonlinearConstraint

    rows, cols = 5, 6
    spacing = 1.0 / (rows + 1)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col * spacing + spacing * 0.5
        y = row * spacing + spacing * 0.5
        centers[i] = [x, y]

        return -np.sum(radii)  # Negative for minimization

    def constraints(radii):

        border_constraints = []
        for i in range(n):
            r = radii[i]

            border_constraints.append(lambda r: r - (x - 1e-6))  # x - r >= 0
            border_constraints.append(lambda r: 1 - x - r)  # x + r <= 1
            border_constraints.append(lambda r: r - (y - 1e-6))  # y - r >= 0
