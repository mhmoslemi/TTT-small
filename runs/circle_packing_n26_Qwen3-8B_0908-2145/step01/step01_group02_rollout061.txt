        return -np.sum(radii)  # Negative because we minimize
    
    # Define constraints
    def constraint_circle_inside(radii, centers):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return 1e6  # Penalize for being out of bounds
        return 0
    
    def constraint_circle_overlap(radii, centers):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return 1e6  # Penalize for overlap
        return 0
    
    # Optimize
    bounds = [(0, 1) for _ in range(n)]
    res = minimize(obj, radii, constraints=[
        {'type': 'ineq', 'fun': lambda r: -constraint_circle_inside(r, centers)},
        {'type': 'ineq', 'fun': lambda r: -constraint_circle_overlap(r, centers)},
    ], bounds=bounds, method='SLSQP', tol=1e-6)
    
    centers = res.x.reshape(-1, 2)  # Reshape for consistency

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
