import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():

    # Starting with 10 rows, 3 columns, then adjusting
    # This is a rough initial guess that can be optimized

    num_cols = 5
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

        y_offset = (row % 2) * (np.sqrt(3) / 2) * 0.8  # Adjusted for spacing

        y = row * (1.0 / (num_rows - 1)) + y_offset
        centers[i] = [x, y]

        return np.all((x >= 0) & (x <= 1))

    def no_overlap(x):

            for j in range(i + 1, n):
                dx = x[i, 0] - x[j, 0]
                dy = x[i, 1] - x[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)

                    return False
        return True

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
