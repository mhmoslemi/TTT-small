import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Number of circles
    n_circles = 26

    # Initial guess: hexagonal packing inside the unit square
    centers = []
    r = 0.04  # Initial radius
    for i in range(n_circles):
        row = i // 7
        col = i % 7
        x = col * (1 + r * np.sqrt(3)) / 7  # Horizontal spacing
        y = row * (1 + r * 2) / 7  # Vertical spacing
        centers.append([x, y])
    centers = np.array(centers)

    # All circles have the same initial radius for easier initialization
    radii = np.full(n_circles, 0.03)

    # Objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x[n_circles:])  # Negative sum for minimization

    # Bounds: x and y are [0,1], radii are [0,1]
    bounds = []
    for i in range(n_circles):
        # Center bounds
        bounds.append((0, 1))
        bounds.append((0, 1))
        # Radius bounds
        bounds.append((0, 1))

    # Constraints: (1) within square (x - r > 0, x + r < 1, y - r > 0, y + r < 1)
    #              (2) distance between centers >= sum of radii
    def constraint_square(x, i):
        x_center = x[i * 2]
        y_center = x[i * 2 + 1]
        r = x[i * 2 + 2]
        return (x_center - r >= 0) * (x_center + r <= 1) * (y_center - r >= 0) * (y_center + r <= 1)

    def constraint_overlap(x, i, j):
        x_i = x[i * 2]
        y_i = x[i * 2 + 1]
        r_i = x[i * 2 + 2]
        x_j = x[j * 2]
        y_j = x[j * 2 + 1]
        r_j = x[j * 2 + 2]
        dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
        return dist >= r_i + r_j - 1e-8

    # Create constraints
    constraints = []
    for i in range(n_circles):
        constraints.append({
            'type': 'ineq',
            'fun': lambda x, i=i: constraint_square(x, i)
        })

    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: constraint_overlap(x, i, j)
            })

    # Optimize
    result = opt.minimize(
        objective,
        np.concatenate([centers.flatten(), radii]),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints
    )

    # Extract centers and radii
    centers_opt = result.x[:n_circles*2].reshape(n_circles, 2)
    radii_opt = result.x[n_circles*2:]

    # Check if optimization was successful
    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    # Return the centers, radii, and sum of radii
    return centers_opt, radii_opt, np.sum(radii_opt)
