import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    num_circles = 26
    rows = int(np.sqrt(num_circles))
    cols = (num_circles + rows - 1) // rows

    spacing = 1.0 / (cols - 1) if cols > 1 else 1.0
    hex_radius = spacing * np.sqrt(3) / 2
    radius_base = hex_radius

    centers = np.zeros((num_circles, 2))

        centers[i] = [x, y]

    radii = np.full(num_circles, radius_base)

    x_vars = np.concatenate([centers.reshape(-1), radii])

    def objective(x):
        return -np.sum(x[-num_circles:])  # Maximize by minimizing negative sum
