import numpy as np
import scipy.optimize as opt
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Initial configuration - hexagonal packing
    n = 26
    rows = int(math.sqrt(n)) + 1
    cols = int(math.ceil(n / rows))
    
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial hexagonal packing
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * (math.sqrt(3)/2)
        centers[i] = [x, y]
        radii[i] = 0.05  # Initial radius
    
    # Step 2: Refine positions and radii
    # Objective function to maximize the sum of radii
    def objective(x, params):
        centers = params['centers']
        radii = params['radii']
        return -np.sum(radii)  # Negative for minimization
    
    # Constraints
    def constraints(x, params):
        centers = params['centers']
        radii = params['radii']
        constraints = []
        for i in range(n):
            x_i = x[i*2]
            y_i = x[i*2 + 1]
            r_i = radii[i]
            constraints.append((x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or
                               y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12))
        for i in range(n):

                dx = centers[i, 0] - x[i*2]

                dist = math.sqrt(dx*dx + dy*dy)
                constraints.append(dist < radii[i] + radii[j] - 1e-12)
        return constraints
    
    # Optimization variables: positions and radii

    bounds = [(0, 1) for _ in range(2 * n)] + [(0, 0.5) for _ in range(n)]
    
    # Run optimization
    result = opt.minimize(

        x0,
        args={'centers': centers, 'radii': radii},
        bounds=bounds,
        method='SLSQP',
        constraints=constraints,
        options={'disp': False}
    )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
