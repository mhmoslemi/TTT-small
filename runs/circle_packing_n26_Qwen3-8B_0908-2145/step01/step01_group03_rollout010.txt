import numpy as np
from scipy.optimize import minimize
import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal grid with 26 circles
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))

    # Generate hexagonal grid positions
    centers = np.zeros((n, 2))
    k = 0
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * (1 + np.sqrt(3)/2) / cols
            else:
                x = (j + 0.5) * (1 + np.sqrt(3)/2) / cols
            y = i * (1/2) / rows
            centers[k, 0] = x
            centers[k, 1] = y
            k += 1

    # Scaling to fit in [0,1]^2
    centers = centers * (1 / np.max(centers))

    # Initial radii estimation
    min_dist = np.min(np.sqrt(np.sum((centers[i] - centers[j])**2 for i in range(n) for j in range(i+1, n))))
    initial_radius = min_dist / 2

    # Initial radii
    radii = np.full(n, initial_radius)

    # Define objective function
    def objective(radii_vec, centers):
        return -np.sum(radii_vec)  # Maximize sum of radii

    # Define constraints
    def constraint_overlap(i, j, centers, radii):
        def cons(radii_vec):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            return dist - (radii_vec[i] + radii_vec[j])
        return cons

    # Define bounds for radii
    bounds = [(0, 1) for _ in range(n)]

    # Define constraints for overlaps
    constraints = []
    for i in range(n):

            cons = constraint_overlap(i, j, centers, radii)
            constraints.append({'type': 'ineq', 'fun': cons})

    # Function to validate packing

                return None, None, None
            sum_radii = np.sum(radii)
            return centers, radii, sum_radii
        except:
            return None, None, None

    # Optimization
    result = minimize(
        fun=lambda r: objective(r, centers),
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints
    )

    radii_opt = result.x
    centers_opt = centers
    centers_opt, radii_opt, sum_radii = validate_and_return(centers_opt, radii_opt)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
