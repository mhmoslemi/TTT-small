import numpy as np
import scipy.optimize as opt

def run_packing():

    # 5 rows in the center (maximum with full hexagonal packing), and some extra near boundaries
    
    # Define initial positions
    centers = []
    radii = []

            x = col + 0.5 * row

            y = (y - 2.5 * np.sqrt(3) / 2) / 2  # Normalize to [0,1]
            centers.append([x, y])
            radii.append(0.1)

    radii.append(0.05)

    radii.append(0.05)
    # Right side
    centers.append([0.95, 0.5])
    radii.append(0.05)
    # Left side
    centers.append([0.05, 0.5])
    radii.append(0.05)
    # Top
    centers.append([0.5, 0.95])
    radii.append(0.05)
    # Bottom
    centers.append([0.5, 0.05])
    radii.append(0.05)
    # Corners
    centers.append([0.05, 0.05])
    radii.append(0.05)
    centers.append([0.95, 0.95])
    radii.append(0.05)
    centers.append([0.05, 0.95])
    radii.append(0.05)
    centers.append([0.95, 0.05])
    radii.append(0.05)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative for minimization

    # Constraints: circles don't overlap and stay within [0,1]^2
    constraints = []
    
    # Boundary constraints
    def boundary(x, i):
        r = x[i]
        x_pos, y_pos = centers[i]
        if x_pos - r < 0 or x_pos + r > 1 or y_pos - r < 0 or y_pos + r > 1:
            return -1  # Violation
        return 0
    
    for i in range(len(radii)):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: boundary(x, i)})
    
    # Non-overlap constraints
    def non_overlap(x, i, j):
        r_i = x[i]
        r_j = x[j]
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist_sq = dx**2 + dy**2
        if dist_sq < (r_i + r_j)**2 - 1e-12:
            return -1  # Violation
        return 0
    
    for i in range(len(radii)):
        for j in range(i+1, len(radii)):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: non_overlap(x, i, j)})
    
    # Optimization
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 0.2) for _ in range(len(radii))],  # Upper bound for radii
        options={'ftol': 1e-9, 'maxiter': 1000}
    )
    
    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")
    
    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Final validation
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Packing invalid: {message}")
    
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
