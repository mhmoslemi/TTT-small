from scipy.optimize import minimize

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing (optimization starting point)
    num_circles = 26
    base_radius = 0.03  # Initial guess for radius (low value for safe starting point)
    centers = np.zeros((num_circles, 2))
    radii = np.full(num_circles, base_radius)

        return -np.sum(radii_opt)

            if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
                return 1e6  # Penalize violation of boundary
        return 0.0

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):

            return False, f"Circle {i} has negative radius {radii[i]}"
