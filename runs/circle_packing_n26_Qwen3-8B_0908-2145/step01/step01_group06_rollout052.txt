import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26

    # We provide a hexagonal grid, but adjust for boundaries
    # We will use a 5x5 grid (with 25 circles) and add one more at the center
    # This is just an initial guess; the optimizer will refine it
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Define hexagonal grid parameters
    rows = 5
    cols = 5
    spacing = 0.2  # Initial spacing between centers
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * (np.sqrt(3) * spacing) / 2
            if i % 2 == 0:
                x += spacing / 2
            centers[i * cols + j] = [x, y]
    
    # Add one more circle in the middle
    centers[-1] = [0.5, 0.5]
    
    # Initial radius for all circles (with boundary constraint)
    radii[:] = 0.1  # Start small and let the optimizer grow them

    # Define objective function: maximize sum of radii
    def objective(x):
        # x is a flattened array containing centers and radii
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:].reshape(n)
        return -np.sum(radii_flat)  # We minimize the negative of the sum

    # Define constraints
    def constraints(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:].reshape(n)
        
        # Constraint 1: Circles must be inside the unit square
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12
                or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return np.inf  # Penalize for going outside unit square
        # Constraint 2: Circles must not overlap

                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return np.inf  # Penalize for overlap
        return 0  # No constraint violation

    # Combine the objective and constraints for optimization
    def objective_and_constraints(x):
        obj = objective(x)
        cons = constraints(x)
        return obj + cons  # Add constraints to the objective

    # Bounds for the optimization variables
    bounds = []
    for i in range(n):
        # Bounds for x and y coordinates of each center
        bounds.append((0, 1))
        bounds.append((0, 1))
        # Bounds for radius (non-negative)
        bounds.append((0, 1))  # Radius can't exceed 1

        bounds=bounds,
        tol=1e-6,

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
