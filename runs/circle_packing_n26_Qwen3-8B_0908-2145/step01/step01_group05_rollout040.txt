from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run_packing():

    # We’ll place the circles in rows with alternating offset

        cols = int(np.ceil(np.sqrt(n_circles)))

                idx += 1

        max_y = np.max(centers[:, 1])
        centers /= np.max([max_x, max_y])
        return centers

    centers_initial = initial_guess(26)

    bounds = Bounds(np.zeros(52 + 26), np.array([1.0] * 52 + [1.0] * 26))
