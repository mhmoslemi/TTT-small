import numpy as np
from scipy.optimize import minimize
import random
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial population of 100 configurations
    n_pop = 100
    n_gen = 100  # Number of generations
    n_circles = 26

    def generate_random_config():
        centers = np.random.rand(n_circles, 2)
        radii = np.random.rand(n_circles) * 0.1
        return centers, radii

    # Initialize population
    population = [generate_random_config() for _ in range(n_pop)]

    def evaluate(config):
        centers, radii = config
        # If any circle is outside the square, penalize
        violation = 0
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            if not (0 - 1e-6 <= x - r <= 1 + 1e-6 and 0 - 1e-6 <= y - r <= 1 + 1e-6):
                violation += 100  # Penalty for boundary violation
            for j in range(i + 1, n_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = math.hypot(dx, dy)
                if dist < radii[i] + radii[j] - 1e-6:
                    violation += 100  # Penalty for overlap
        # Fitness is sum of radii minus violation penalty
        return -np.sum(radii) + violation  # Minimize this

    # Evolutionary algorithm
    for gen in range(n_gen):
        # Evaluate all configurations
        fitness = [evaluate(config) for config in population]
        # Select top 20% as parents
        parents = np.argsort(fitness)[:int(0.2 * n_pop)]
        # Generate new population
        new_population = []
        for _ in range(n_pop):
            # Randomly select parents
            parent1, parent2 = random.sample(parents, 2)
            config1 = population[parent1]
            config2 = population[parent2]
            # Crossover
            centers = np.zeros((n_circles, 2))
            radii = np.zeros(n_circles)
            for i in range(n_circles):
                if random.random() < 0.5:
                    centers[i] = config1[0][i]
                    radii[i] = config1[1][i]
                else:
                    centers[i] = config2[0][i]
                    radii[i] = config2[1][i]
            # Apply mutation
            for i in range(n_circles):
                if random.random() < 0.1:

                    radii[i] += np.random.normal(0, 0.01)
            # Ensure radii and centers are valid

                x, y = centers[i]
                r = radii[i]
                centers[i] = np.clip(centers[i], [r, r], [1 - r, 1 - r])
            new_population.append((centers, radii))
        population = new_population

    # Find the best configuration
    best_config = min(population, key=evaluate)
    centers, radii = best_config
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
