import numpy as np
import scipy.optimize as opt
from typing import Tuple
import multiprocessing

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Constants
    NUM_CIRCLES = 26
    EPS = 1e-6  # A small tolerance for floating point errors

    # Initial grid parameters
    rows = 5
    cols = 5
    gaps = 1.0 / (rows + 1)  # Adjust spacing to fit in the unit square
    radius_initial = 0.05  # Initial guess for all circles

    # Define the centers of the initial grid
    centers = np.zeros((NUM_CIRCLES, 2))
    counter = 0
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * gaps
            else:
                x = (j + 0.5) * gaps
            y = i * gaps
            centers[counter] = [x, y]
            counter += 1

    # Initial guess for radii
    radii = np.full(NUM_CIRCLES, radius_initial)

    # Objective function to maximize (sum of radii)
    def objective(x: np.ndarray) -> float:
        # x has shape (2*NUM_CIRCLES,)
        new_centers = x[:2*NUM_CIRCLES].reshape(NUM_CIRCLES, 2)
        new_radii = x[2*NUM_CIRCLES:]
        return -np.sum(new_radii)  # Negative for minimization

    # Constraints
    def constraint_centers(x: np.ndarray) -> np.ndarray:
        # Ensure centers are within the unit square
        new_centers = x[:2*NUM_CIRCLES].reshape(NUM_CIRCLES, 2)
        return np.concatenate([
            new_centers[:, 0] - EPS,  # x >= 0 - EPS
            1.0 + EPS - new_centers[:, 0],  # x <= 1 + EPS
            new_centers[:, 1] - EPS,  # y >= 0 - EPS
            1.0 + EPS - new_centers[:, 1]  # y <= 1 + EPS
        ])

    def constraint_distances(x: np.ndarray) -> np.ndarray:
        # Ensure all pairs of circles are non-overlapping
        new_centers = x[:2*NUM_CIRCLES].reshape(NUM_CIRCLES, 2)
        new_radii = x[2*NUM_CIRCLES:]
        distances = np.zeros(NUM_CIRCLES * (NUM_CIRCLES - 1) // 2)
        indices = 0
        for i in range(NUM_CIRCLES):
            for j in range(i+1, NUM_CIRCLES):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                distances[indices] = dist - new_radii[i] - new_radii[j]
                indices += 1
        return distances

    # Prepare initial guess
    x0 = np.concatenate([centers.flatten(), radii])

    # Bounds for centers (0 <= x <= 1, 0 <= y <= 1)
    bounds_centers = [(0 - EPS, 1 + EPS) for _ in range(2 * NUM_CIRCLES)]
    # Bounds for radii (0 <= r <= 1)
    bounds_radii = [(0, 1) for _ in range(NUM_CIRCLES)]
    bounds = bounds_centers + bounds_radii

    # Constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_centers},
        {'type': 'ineq', 'fun': constraint_distances}
    ]

    # Run optimization
    result = opt.minimize(
        fun=objective,
        x0=x0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        tol=1e-10
    )

    # Extract the optimized centers and radii
    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    centers_opt = result.x[:2*NUM_CIRCLES].reshape(NUM_CIRCLES, 2)
    radii_opt = result.x[2*NUM_CIRCLES:]
    sum_radii = np.sum(radii_opt)

    # Validate the configuration to ensure no errors
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise RuntimeError(f"Validation failed: {message}")

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
