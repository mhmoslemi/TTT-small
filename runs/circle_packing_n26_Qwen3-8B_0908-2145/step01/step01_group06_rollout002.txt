import scipy.optimize as opt
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based configuration
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Create a basic grid with a hexagonal pattern
    for i in range(num_circles):
        row = i // 5
        col = i % 5
        x = col / 4.5
        y = row / 4.5
        centers[i] = [x, y]

    # Start with uniform radii
    initial_radius = 0.05
    radii[:] = initial_radius

    # Objective function to maximize sum of radii
    def objective(radii_flat):
        return -np.sum(radii_flat)

    # Constraints
    def constraint_center(centers_flat, radii_flat):
        # Ensure that circles are fully inside the square
        for i in range(num_circles):
            x = centers_flat[2*i]
            y = centers_flat[2*i + 1]
            r = radii_flat[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return -1
        return 0

    def constraint_distance(centers_flat, radii_flat):
        # Ensure that no two circles overlap
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                x1 = centers_flat[2*i]
                y1 = centers_flat[2*i + 1]
                x2 = centers_flat[2*j]
                y2 = centers_flat[2*j + 1]
                r1 = radii_flat[i]

                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)

                    return -1
        return 0

    # Flatten centers and radii for optimization
    centers_flat = centers.flatten()
    radii_flat = radii.copy()

    # Optimization using SLSQP
    bounds = [(0, 1) for _ in range(2*num_circles)] + [(0, 1) for _ in range(num_circles)]

        'type': 'ineq',
        'fun': lambda x: constraint_center(x[:2*num_circles], x[2*num_circles:])
