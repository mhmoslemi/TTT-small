import scipy.optimize as optimize
import random

def compute_hexagonal_grid(n_circles, square_size=1.0, max_circle_radius=0.1):
    """Generate a hexagonal grid of circles in a square."""
    circles = []
    row = 0
    col = 0
    radius = max_circle_radius
    while len(circles) < n_circles:

        y = row * (np.sqrt(3) * radius) + radius

            circles.append((x, y, radius))
        col += 1
        if col % 2 == 0:
            row += 1
    return np.array(circles)
