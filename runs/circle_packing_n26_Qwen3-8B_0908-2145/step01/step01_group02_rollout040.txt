import scipy.optimize as optimize
from multiprocessing import Pool, cpu_count

                x = col + 0.5 * row
                y = row * np.sqrt(3) / 2

                    idx += 1
        return centers

    initial_centers = hexagonal_packing(n_circles)

    def distance(c1, c2):
        return np.sqrt(np.sum((c1 - c2) ** 2))

    # We assume centers are fixed initially, but this may be adjusted
    initial_radii = np.full(n_circles, 1e-3)

        return -np.sum(radii)

    # Constraint: distance between centers >= sum of radii
    def constraint_dist(radii, centers):

        dists = np.zeros(n * (n - 1) // 2)
        for i in range(n):
            for j in range(i + 1, n):
                dists[j - i - 1] = distance(centers[i], centers[j])
        return dists - (radii[:n] + radii[1:])

    # Constraint: all circles are within the unit square
    def constraint_bounds(radii, centers):
        n = len(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
