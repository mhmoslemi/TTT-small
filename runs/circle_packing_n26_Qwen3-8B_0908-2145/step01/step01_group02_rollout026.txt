import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    # Pack as tightly as possible first, then optimize
    # Start with a hexagonal grid and adjust radii to avoid overlaps

    # Parameters
    num_circles = 26
    tolerance = 1e-6  # for numerical stability
    max_radius_initial = 0.15  # Initial guess for radius

    # Generate an initial grid of centers based on hexagonal packing
    centers = np.zeros((num_circles, 2))
    row_count = int(np.ceil(np.sqrt(num_circles)))
    col_count = int(np.ceil(num_circles / row_count))
    radius = max_radius_initial
    offset_y = np.sqrt(3) / 2 * radius  # vertical spacing for hexagonal packing

    for i in range(num_circles):
        row = i // col_count
        col = i % col_count
        x = col * (1.5 * radius)
        y = row * offset_y + (col % 2) * (radius / 2)  # offset rows for hexagonal
        centers[i] = [x, y]

    # Scale the centers to fit in a unit square
    scaling_factor = 1.0 / (np.max(centers[:, 0]) + np.max(centers[:, 1]))
    centers *= scaling_factor

    # Initial guess for radii
    radii = np.full(num_circles, max_radius_initial)

    # Define objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative because we use minimize

    # Define constraints:
    # 1. Each circle must be within the unit square
    # 2. Distance between any two circles must be at least the sum of their radii

    def constraint_center_bounds(x):
        # x is a 2D vector of centers, but we flatten it
        centers_flattened = x[:2*num_circles].reshape((num_circles, 2))
        radii_flattened = x[2*num_circles:]

            r_i = radii_flattened[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1  # Violated constraint
        return 0

    def constraint_distances(x):
        # x is a 2D vector of centers, but we flatten it

                r_i = radii_flattened[i]
                r_j = radii_flattened[j]
                if dist < r_i + r_j - tolerance:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
