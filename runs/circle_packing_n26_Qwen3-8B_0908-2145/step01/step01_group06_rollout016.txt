import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based placement
    num_circles = 26
    grid_rows = 5
    grid_cols = 5
    spacing = 0.2  # Initial spacing

    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // grid_cols
        col = i % grid_cols
        x = col * spacing
        y = row * spacing
        centers[i, 0] = x
        centers[i, 1] = y

    # Initial radii
    radii = np.zeros(num_circles)
    radii[:] = 0.1  # Initial uniform radius

    # Objective function: maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize the negative of sum (to maximize sum)

    # Variables: x = [r0, r1, ..., r25]
    x0 = radii.copy()

    # Constraints
    # Constraint 1: All radii >= 0
    bounds = Bounds(0, np.inf)

    # Constraint 2: Circles within unit square
    def circle_in_square(radii, centers):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        return True

    def constraint_circle_in_square(x):
        return circle_in_square(x, centers)

    # Constraint 3: No overlap between circles
    def no_overlap(x):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                xi, yi = centers[i]
                xj, yj = centers[j]
                dist = np.sqrt((xi - xj) ** 2 + (yi - yj) ** 2)
                if dist < x[i] + x[j] - 1e-8:
                    return False
        return True

    def constraint_no_overlap(x):
        return no_overlap(x)

    # Constraints object
    cons = [
        LinearConstraint(np.ones(num_circles), 0, np.inf),  # radii >= 0
        NonlinearConstraint(constraint_circle_in_square, 0, 1),
        NonlinearConstraint(constraint_no_overlap, 0, 1)
    ]

    # Optimization
    result = minimize(
        objective,
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        tol=1e-6
    )

    # Extract results
    radii_opt = result.x
    centers_opt = centers
    sum_radii = np.sum(radii_opt)

    # Validate the packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise RuntimeError(f"Packing invalid: {msg}")

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
