import numpy as np
import scipy.optimize as optimize
import cvxpy as cp

def generate_hexagonal_layout(n, square_size):
    """
    Generate an initial hexagonal layout of n circles in a square.
    """
    radius = 0.1
    spacing = 2 * radius
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = []
    
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * np.sqrt(3) / 2
            centers.append([x, y])
    
    # Adjust to fit in square
    scale = square_size / np.max(centers, axis=0)
    centers = np.array(centers) * scale
    return np.array(centers)

def compute_sum_radii(radii):
    """Returns the sum of the radii."""
    return np.sum(radii)

def compute_overlap_penalty(centers, radii):
    """Penalty for overlapping circles."""
    n = centers.shape[0]
    penalty = 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            distance = np.sqrt(dx**2 + dy**2)
            if distance < radii[i] + radii[j] - 1e-6:
                penalty += 1e6 * (radii[i] + radii[j] - distance)
    return penalty

def optimize_packing(centers, square_size=1.0):
    """
    Optimize the positions and radii of the circles to maximize the total area
    under the constraint of non-overlapping and fitting within the square.
    """
    n = centers.shape[0]
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Objective: Maximize total area
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []
    for i in range(n):
        # Boundaries
        constraints.append(x[i, 0] - r[i] >= 0)
        constraints.append(x[i, 0] + r[i] <= square_size)
        constraints.append(x[i, 1] - r[i] >= 0)
        constraints.append(x[i, 1] + r[i] <= square_size)

    for i in range(n):
        for j in range(i + 1, n):
            # Non-overlapping constraint
            dx = x[i, 0] - x[j, 0]
            dy = x[i, 1] - x[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= r[i] + r[j] - 1e-6)

    # Initial guess
    x_initial = np.copy(centers)
    r_initial = np.full(n, 0.1)

    # Convert to CVXPY problem
    prob = cp.Problem(objective, constraints)
    try:
        result = prob.solve(solver=cp.OSQP, verbose=False, warm_start=True, x0=x_initial, r0=r_initial)
        if result is None:
            raise ValueError("Optimization failed")
        return x.value, r.value
    except:
        raise

def run_packing():
    """Main function to pack 26 circles in a unit square."""
    n = 26
    initial_centers = generate_hexagonal_layout(n, square_size=1.0)
    
    # Optimize layout
    try:
        centers, radii = optimize_packing(initial_centers)

        return (centers, radii, sum_radii)
