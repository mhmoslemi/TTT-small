import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26
    radius = 0.03  # Initial guess for radius
    centers = np.zeros((n, 2))
    for i in range(n):
        # Hexagonal grid with spacing of about 0.2 units
        r = i // 4
        c = i % 4
        x = c * 0.25 + 0.1
        y = r * 0.25 + 0.1
        centers[i] = [x, y]

    # Define the objective function (sum of radii)
    def objective(x):
        return -np.sum(x)  # Minimize negative sum of radii (equivalent to maximizing sum)

    # Define constraints
    def constraint1(x):
        return np.sum(x)  # Constraint: avoid empty circles

    def constraint2(x):
        # Constraint: all circles must be within the unit square
        return np.sum((x[i] < 0) + (x[i] > 1) for i in range(n))

    def constraint3(x):
        # Constraint: circles cannot overlap
        dist_sq = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dist_sq[i, j] = np.sum((centers[i] - centers[j]) ** 2)
        min_dist_sq = np.min(dist_sq)
        return min_dist_sq - (x[i] + x[j]) ** 2 + 1e-8  # If min_dist_sq < (x[i] + x[j])^2, then constraint is violated

    # Initial guess for radii
    radii = np.full(n, radius)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
