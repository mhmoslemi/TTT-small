import cvxpy as cp

def run_packing():

    n = 26

    for i in range(n):
        row = i // num_cols
        col = i % num_cols
        x = col * spacing
        y = row * spacing
        centers[i, 0] = x
        centers[i, 1] = y

    # Define the optimization variables
    radii = cp.Variable(n, nonneg=True)

    # Objective function: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

        x, y = centers[i]
        r = radii[i]
        constraints.append(x - r >= 0)

        constraints.append(y - r >= 0)
        constraints.append(y + r <= 1)

    # No overlap between circles
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= radii[i] + radii[j])

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-6, max_iters=1000)
