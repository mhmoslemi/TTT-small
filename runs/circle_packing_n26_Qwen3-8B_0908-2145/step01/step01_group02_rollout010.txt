import numpy as np
import cvxpy as cp

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    # Initial number of circles
    n_circles = 26

    # Initial configuration: grid-based with a hexagonal layout in the center
    # and some additional circles in edges and corners for refinement
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)

    # Step 1: Initialize with a grid layout
    # Row 0 to row 4 (5 rows)
    spacing = 0.2
    row_count = 5
    col_count = 5
    for i in range(row_count):
        for j in range(col_count):
            idx = i * col_count + j
            if idx >= n_circles:
                break
            x = 0.1 + j * spacing
            y = 0.1 + i * spacing
            centers[idx] = [x, y]
            radii[idx] = 0.06

    # Step 2: Optimization using CVXPY
    # Define variables
    x = cp.Variable((n_circles, 2))
    r = cp.Variable(n_circles)

    # Objective: maximize total radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints:
    # 1. Each circle is within the unit square
    constraints = []
    for i in range(n_circles):
        x_i, y_i = x[i]
        r_i = r[i]
        constraints.append(x_i >= r_i - 1e-8)
        constraints.append(x_i <= 1 - r_i + 1e-8)
        constraints.append(y_i >= r_i - 1e-8)
        constraints.append(y_i <= 1 - r_i + 1e-8)

    # 2. No two circles overlap (distance >= sum of radii)
    for i in range(n_circles):
        for j in range(i + 1, n_circles):
            dist = cp.norm(x[i] - x[j])
            constraints.append(dist >= r[i] + r[j] - 1e-8)

    # 3. Radii must be non-negative
    constraints.append(r >= 0)

    # Solve the optimization problem
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve()
    except Exception as e:
        raise RuntimeError(f"Optimization failed: {e}")

    # Step 3: Extract the optimal solution
    centers_opt = x.value
    radii_opt = r.value
    sum_radii = cp.sum(r).value

    # Step 4: Validate the solution
    if not validate_packing(centers_opt, radii_opt)[0]:
        raise RuntimeError("Invalid packing found after optimization")

    # Step 5: Return the result
    return centers_opt, radii_opt, sum_radii
