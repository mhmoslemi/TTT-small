import scipy.optimize as opt

def run_packing():

    num_circles = 26

    # First, create a hexagonal packing of circles with equal radii for a rough initial positioning
    # This is a basic grid and we'll later optimize it
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Let's start with a fixed radius for the first pass and then optimize
    initial_radius = 0.15
    for i in range(num_circles):
        row = i // 6
        col = i % 6
        x = col * (1 + initial_radius * 2) / 5
        y = row * (1 + initial_radius * 2) / 5
        centers[i] = [x, y]
        radii[i] = initial_radius

    # Optimization objective: Maximize sum of radii
    def objective(radii_flat):
        return -np.sum(radii_flat)  # Minimize negative sum

    # Optimization constraints
    def constraints(radii_flat):
        # Convert flat radii to radii array
        radii = radii_flat.reshape(-1)

        # Ensure all radii are non-negative

            if radii[i] < 0:
                return np.inf

        # Ensure circles are within the unit square
        for i in range(num_circles):
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12
                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
                return np.inf
