import cvxpy as cp

    radius = 0.1

    radii = []

    x = cp.Variable((total_circles, 2))
    r = cp.Variable(total_circles)

    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []

    problem = cp.Problem(objective, constraints)
