import scipy.optimize as opt

def validate_packing(centers, radii):

        return False, "NaN values present"

    return True, "ok"

def initial_config(n):
    # Create a grid-based initial configuration for n circles
    # This is a basic hexagonal packing approach
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):

        x = col / 4.0 + 0.25
        y = row / 4.0 + 0.25

        radii[i] = 0.05

    return centers, radii

def objective_function(x, n):

    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:].reshape(n)

    return -np.sum(radii)  # we minimize the negative sum to maximize the sum

def constraint_overlap(x, n):
    # Constraint: circles don't overlap
    centers = x[:2*n].reshape(n, 2)
    radii = x[2*n:].reshape(n)
