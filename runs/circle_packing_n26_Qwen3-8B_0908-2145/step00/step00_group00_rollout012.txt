import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Step 1: Initial hexagonal packing
    num_circles = 26
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))
    
    radius_initial = 0.05
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * (2 * radius_initial + 1e-6) + 0.5 * radius_initial
            y = i * (np.sqrt(3) * radius_initial + 1e-6) + 0.5 * radius_initial
            centers.append([x, y])
    
    centers = np.array(centers)
    radii = np.full(num_circles, radius_initial)

    # Step 2: Define objective function
    def objective(x):
        x_flat = x.reshape(-1)
        centers_flat = x_flat[:2 * num_circles].reshape(num_circles, 2)
        radii_flat = x_flat[2 * num_circles:]
        return -np.sum(radii_flat)  # Minimize negative sum to maximize sum

    # Step 3: Define constraints
    def constraints(x):
        x_flat = x.reshape(-1)
        centers_flat = x_flat[:2 * num_circles].reshape(num_circles, 2)
        radii_flat = x_flat[2 * num_circles:]
        constraints = []

        # Bound constraints
        for i in range(num_circles):
            x, y = centers_flat[i]
            r = radii_flat[i]
            constraints.append(x - r >= -1e-6)
            constraints.append(x + r <= 1 + 1e-6)
            constraints.append(y - r >= -1e-6)
            constraints.append(y + r <= 1 + 1e-6)

        # Non-overlap constraints
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraints.append(dist >= radii_flat[i] + radii_flat[j] - 1e-6)

        return np.array(constraints)

    # Step 4: Optimize
    bounds = []
    for i in range(num_circles):
        bounds.append((0, 1))  # x-coordinate
        bounds.append((0, 1))  # y-coordinate
        bounds.append((0, 0.5))  # radius (limit to 0.5 to keep feasible)
    
    initial_guess = np.concatenate([centers.flatten(), radii])
    result = optimize.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Step 5: Validate and format output
    centers_opt = result.x[:2 * num_circles].reshape(num_circles, 2)
    radii_opt = result.x[2 * num_circles:]
    sum_radii = np.sum(radii_opt)

    # Ensure non-negative radii
    radii_opt = np.maximum(radii_opt, 0)

    # Validate the packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(msg)

    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
