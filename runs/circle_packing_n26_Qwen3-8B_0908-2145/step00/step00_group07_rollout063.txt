import numpy as np
from scipy.optimize import minimize, Bounds, NonlinearConstraint
import math

def run_packing():
    # Initial hexagonal grid configuration
    num_circles = 26
    grid_rows = int(math.ceil(math.sqrt(num_circles)))
    grid_cols = int(math.ceil(math.sqrt(num_circles)))

    # Generate a hexagonal grid and then refine it
    centers = np.zeros((num_circles, 2))
    radii = np.ones(num_circles) * 0.05  # Initial small radius

    # Generate initial hexagonal pattern, adjusting for square
    for i in range(num_circles):
        row = i // grid_cols
        col = i % grid_cols
        x = col + 0.5 * (row % 2)
        y = row * (np.sqrt(3) / 2)
        centers[i] = [x, y]
        radii[i] = 0.05  # Initial radius

    # Normalize to fit in unit square
    def scale_to_unit_square(centers):
        max_x = np.max(centers[:, 0])
        max_y = np.max(centers[:, 1])
        scaling = min(1.0 / max_x, 1.0 / max_y)
        scaled_centers = centers * scaling
        return scaled_centers

    centers = scale_to_unit_square(centers)

    # Function to compute sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: circles must not overlap and stay inside the square
    def constraint_overlap(centers, radii):
        n = centers.shape[0]
        dists = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dists[i, j] = np.sqrt(dx**2 + dy**2)
        return dists - (radii[:, np.newaxis] + radii[np.newaxis, :]) + 1e-8

    def constraint_boundary(radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        return True

    # Create a problem for optimization
    bounds = Bounds(0, 1)  # Radii must be non-negative

    def constraint_overlap_func(radii):
        return constraint_overlap(centers, radii)

    nonlcon = NonlinearConstraint(constraint_overlap_func, 0, 0)

    # Optimization
    result = minimize(
        objective,
        radii,
        method="SLSQP",
        bounds=bounds,
        constraints=nonlcon
    )

    radii_opt = result.x
    centers_opt = centers.copy()
    sum_radii = np.sum(radii_opt)

    # Validate the packing
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Packing validation failed: {message}")

    # Final result
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
