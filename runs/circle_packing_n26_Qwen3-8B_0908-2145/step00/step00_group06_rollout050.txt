import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial square packing: place 26 circles in grid format
    # Determine the grid dimensions
    rows = 5
    cols = 6  # 5x6 = 30 positions, but we only need 26, so reduce one row
    
    # Initial grid of centers
    centers_initial = np.zeros((26, 2))
    for i in range(26):
        row = i // cols
        col = i % cols
        # Place with center of circle at (col * 1.0/(cols-1), row * 1.0/(rows-1))
        x = col * 1.0 / (cols - 1)
        y = row * 1.0 / (rows - 1)
        centers_initial[i] = [x, y]

    # Initial radii (equal for all)
    radii_initial = np.ones(26) * 0.05

    # Optimization bounds
    bounds = [(0.0, 1.0) for _ in range(26 * 2)]  # x and y for each circle

    # Optimization variables
    x = np.hstack([centers_initial.flatten(), radii_initial])

    # Objective: maximize sum of radii (minimize -sum of radii)
    def objective(x_flat):
        centers = x_flat[:26*2].reshape(26, 2)
        radii = x_flat[26*2:].reshape(26)
        return -np.sum(radii)  # Negative because we minimize

    # Constraints: non-overlap between circles and within unit square
    def constraint_overlap(x_flat):
        centers = x_flat[:26*2].reshape(26, 2)
        radii = x_flat[26*2:].reshape(26)
        for i in range(26):
            for j in range(i+1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1  # Violation of constraint
        return 0

    def constraint_boundary(x_flat):
        centers = x_flat[:26*2].reshape(26, 2)
        radii = x_flat[26*2:].reshape(26)
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1.0 + 1e-8 or y - r < -1e-8 or y + r > 1.0 + 1e-8:
                return -1
        return 0

    # Create optimization problem
    constraints = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    # Perform optimization
    result = opt.minimize(
        objective,
        x,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        tol=1e-6
    )

    # Extract results
    centers = result.x[:26*2].reshape(26, 2)
    radii = result.x[26*2:].reshape(26)
    sum_radii = np.sum(radii)

    # Validate packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
