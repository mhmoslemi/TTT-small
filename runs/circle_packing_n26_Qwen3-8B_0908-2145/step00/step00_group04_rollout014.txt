import scipy.optimize as opt
import math

def run_packing():

    radii = np.zeros(n)

    grid_count = 0
    grid_cols = int(math.ceil(n ** 0.5))
    grid_rows = int(math.ceil(n / grid_cols))

    for row in range(grid_rows):
        for col in range(grid_cols):
            if grid_count >= n:
                break
            x = grid_size * col
            y = grid_size * row + (grid_size / 2) * row
            if row % 2 == 1:
                x += grid_size / 2
            centers[grid_count] = [x, y]
            grid_count += 1

    # Define the optimization problem
    def objective(x):
        # x: flattened array (centers and radii)
        centers_flat = x[:n*2].reshape((n, 2))
        radii_flat = x[n*2:].reshape((n,))
        total_radius = np.sum(radii_flat)
        return -total_radius  # Minimize the negative of the total radius

    def constraints(x):
        centers_flat = x[:n*2].reshape((n, 2))
        radii_flat = x[n*2:].reshape((n,))
        cons = []

        # Boundary constraints: all circles must be within [0,1]×[0,1]
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            cons.append(0.0 - x_i + r_i <= 1e-8)
            cons.append(1.0 - x_i - r_i <= 1e-8)
            cons.append(0.0 - y_i + r_i <= 1e-8)
            cons.append(1.0 - y_i - r_i <= 1e-8)

        # Non-overlap constraints: distance between centers >= sum of radii
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                cons.append(dist - (radii_flat[i] + radii_flat[j]) <= 1e-8)

        return cons

    # Initial guess

    problem = opt.OptimizationProblem()
    problem.minimize(objective)
