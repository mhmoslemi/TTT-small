import scipy.optimize as opt

def run_packing():

    initial_radii = np.zeros(num_circles)

    for i in range(num_circles):
        row = i // 5
        col = i % 5
        initial_centers[i, 0] = col * 0.2 + 0.1

        initial_radii[i] = 0.05

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_centers(x):
        centers = x[:num_circles*2].reshape(num_circles, 2)
        radii = x[num_circles*2:]
        return np.concatenate([centers.flatten(), radii])

    def constraint_bounds(x):
        centers = x[:num_circles*2].reshape(num_circles, 2)
        radii = x[num_circles*2:]
        bounds = []
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            bounds.append((0.0, 1.0))  # x_i
            bounds.append((0.0, 1.0))  # y_i
            bounds.append((0.0, 1.0 - r_i))  # r_i must be <= 1 - r_i (margin)
        return bounds

    def constraint_non_overlap(x):
        centers = x[:num_circles*2].reshape(num_circles, 2)
        radii = x[num_circles*2:]
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]) >= -1e-8)
        return constraints

    def constraint_boundary(x):
        centers = x[:num_circles*2].reshape(num_circles, 2)

            # Check if any part of the circle is outside the square
