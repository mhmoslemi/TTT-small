import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid layout with equal radii
    num_circles = 26
    grid_size = int(np.ceil(np.sqrt(num_circles)))
    circle_radius = 0.15  # Initial radius (precomputed for grid layout)

    # Create initial grid layout
    centers = np.zeros((num_circles, 2))
    for i in range(num_circles):
        row = i // grid_size
        col = i % grid_size
        x = col * (1.0 / (grid_size - 1))
        y = row * (1.0 / (grid_size - 1))
        centers[i] = [x, y]

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # minimize the negative sum

    # Constraints: non-overlap between circles and inside the square
    def constraint_overlap(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint_bounds(radii):
        for r in radii:
            if r < 0:
                return -1
        return 0

    # Initial guess of radii
    initial_radii = np.full(num_circles, circle_radius)

    # Optimization bounds for radii
    bounds = Bounds(0, 1.0)

    # Use SLSQP for optimization
    result = opt.minimize(
        objective,
        initial_radii,
        method="SLSQP",
        bounds=bounds,
        constraints=[
            {"type": "ineq", "fun": constraint_overlap, "args": (centers,)},
            {"type": "ineq", "fun": constraint_bounds}
        ],
        tol=1e-8,
        options={"maxiter": 1000, "ftol": 1e-8, "eps": 1e-8}
    )

    radii = result.x
    centers = centers  # Adjust centers if needed during optimization (optional)
    sum_radii = np.sum(radii)

    # Validate the packing
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(f"Packing is invalid: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
