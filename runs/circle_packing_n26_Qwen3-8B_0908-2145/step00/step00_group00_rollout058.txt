import scipy.optimize as optimize
from typing import Tuple

    n = 26

    centers = np.zeros((n, 2))

    cols = int(np.sqrt(n))
    rows = (n + cols - 1) // cols

            centers[i * cols + j] = [x, y]

    max_dist = np.max(np.sqrt(np.sum(centers**2, axis=1)))

    radii = np.full(n, radius_initial / max_dist)

    x0 = np.concatenate([centers.flatten(), radii])

    # Optimization bounds: centers in [0,1]^2, radii >= 0
    bounds = []
