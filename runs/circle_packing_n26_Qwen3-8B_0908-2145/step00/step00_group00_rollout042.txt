from scipy.optimize import minimize
import math

def run_packing():
    # Initial guess for centers in a hexagonal grid with 26 circles
    # Hexagonal grid for 26 points
    rows = 6
    cols = [4, 5, 6, 5, 4, 4]
    centers = np.zeros((26, 2))

    # Hexagonal grid dimensions
    radius_guess = 0.1
    horizontal_spacing = 2 * radius_guess
    vertical_spacing = math.sqrt(3) * radius_guess

    # Place the grid
    idx = 0
    for r in range(rows):
        for c in range(cols[r]):
            x = c * horizontal_spacing
            y = r * vertical_spacing if r % 2 == 0 else (r + 1) * vertical_spacing
            centers[idx] = [x, y]
            idx += 1

    # Initial radii guess
    radii = np.full(26, radius_guess)

    # Optimization function: maximize sum of radii
    def objective(x):
        # x is a flat array of radii -> reshape it
        radii = x.reshape((26,))
        # Compute the total sum
        return -np.sum(radii)  # Minimize the negative to maximize

    # Constraints:
    # 1. Each circle must be within the unit square (x - r >= 0 and x + r <= 1, etc.)
    # 2. Distance between any two circles must be >= sum of radii
    # 3. Radii must be non-negative

    def constraint1(x):
        radii = x.reshape((26,))
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return -1
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:

        return 0

                dist = math.hypot(dx, dy)

        return 0

    # bounds: radii >= 0
    bounds = [(0.0, 1.0) for _ in range(26)]

    result = minimize(

        method='SLSQP',
        bounds=bounds,
        constraints=cons
