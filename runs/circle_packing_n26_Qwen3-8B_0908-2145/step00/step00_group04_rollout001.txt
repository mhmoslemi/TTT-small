import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial positions using hexagonal packing
    n = 26
    # Define the number of rows and columns in the hexagonal grid
    rows = 6
    cols = 5
    # Define the spacing for the hexagonal grid
    spacing = 0.25
    # Generate initial positions
    centers = np.zeros((n, 2))
    idx = 0
    for i in range(rows):
        for j in range(cols):
            # Offset for even rows
            if i % 2 == 0:
                x = j * spacing
            else:
                x = (j + 0.5) * spacing
            y = i * spacing * np.sqrt(3) / 2
            centers[idx] = [x, y]
            idx += 1

    # Initial radii: all circles start with the same radius
    radii = np.full(n, 0.05)

    # Objective function to maximize the sum of radii
    def objective(x):
        # x is a flattened 1D array of radii
        radii_flat = x
        radii = radii_flat.reshape((n,))
        # Ensure each circle stays inside the square
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -np.inf  # Penalize if any circle goes out of bounds
        # Check for overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf  # Penalize if any circles overlap
        # Maximize the sum of radii
        return -np.sum(radii)

    # Bounds and constraints for the optimization
    bounds = [(0.0, 0.5) for _ in range(n)]

    initial_guess = radii.copy()

        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    # Extract optimized radii and centers
    optimized_radii = result.x
    sum_radii = np.sum(optimized_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
