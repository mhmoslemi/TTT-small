import numpy as np
import scipy.optimize as opt

def run_packing():

    def generate_hexagonal_config(n_circles):
        centers = []
        radii = []
        r = 0.1  # Start with a small radius
        x = 0.5
        y = 0.5
        step = 0.6
        for i in range(n_circles):
            x += (i % 2) * step
            y += (i // 2) * step
            # Adjust coordinates to stay within unit square
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            centers.append((x, y))
            radii.append(r)
        return np.array(centers), np.array(radii)

    # Optimization function to maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)

    def constraint_1(radii, centers):
        # Total area constraint: sum of πr² ≤ 1
        return np.sum(np.pi * radii**2) - 1

    def constraint_2(radii, centers):
        # Distance between circles must be ≥ r_i + r_j
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    def constraint_3(radii, centers):
        # All circles must be within the unit square
        for i in range(len(radii)):
            if centers[i, 0] - radii[i] < 0 or centers[i, 0] + radii[i] > 1:
                return -1
            if centers[i, 1] - radii[i] < 0 or centers[i, 1] + radii[i] > 1:
                return -1
        return 0

    # Initial configuration
    centers, radii = generate_hexagonal_config(26)

    # Optimization
    bounds = [(0.0, 1.0) for _ in range(26)]  # All radii between 0 and 1
    cons = [
        {'type': 'ineq', 'fun': constraint_1, 'args': (centers,)},
        {'type': 'ineq', 'fun': constraint_2, 'args': (radii, centers)},
        {'type': 'ineq', 'fun': constraint_3, 'args': (radii, centers)}
    ]
    result = opt.minimize(
        objective,
        radii,
        args=(centers,),
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-6
    )

    if not result.success:
        raise ValueError("Optimization failed")

    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
