import numpy as np
import scipy.optimize as opt

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

def run_packing():
    n = 26

    # Initial guess: grid-based placement with spacing
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * 0.3 + 0.15
        y = row * 0.3 + 0.15
        centers[i] = [x, y]
        radii[i] = 0.1

    def objective(radii):
        # Objective is to maximize the sum of radii (so minimize -sum(radii))
        return -np.sum(radii)

    def constraints(radii):
        # Constraint 1: All circles must be within the unit square
        for i in range(n):
            r = radii[i]
            if r < 0:
                return False
            # Check if circle at centers[i] with radius r stays within the square
            x, y = centers[i]
            if not (0 - r <= x - r <= 1 + r and 0 - r <= y - r <= 1 + r):
                return False
        # Constraint 2: Circles must not overlap
        for i in range(n):
            for j in range(i + 1, n):
                r_i = radii[i]
                r_j = radii[j]
                x_i, y_i = centers[i]
                x_j, y_j = centers[j]
                dx = x_i - x_j
                dy = y_i - y_j
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < r_i + r_j - 1e-12:
                    return False
        return True

    # Setup constraints for optimization
    bounds = [(0, 1.0) for _ in range(n)]

    # Optimization
    result = opt.minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraints}],
        tol=1e-8
    )

    if not result.success:
        raise RuntimeError("Optimization failed with message: " + result.message)

    optimal_radii = result.x
    optimal_centers = centers.copy()

    # Validate the packing
    valid, msg = validate_packing(optimal_centers, optimal_radii)
    if not valid:
        raise RuntimeError("Packing is invalid: " + msg)

    return optimal_centers, optimal_radii, np.sum(optimal_radii)
