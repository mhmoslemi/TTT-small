import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based configuration
    n = 26
    cols = int(np.ceil(np.sqrt(n)))
    rows = int(np.ceil(n / cols))

    # Distribute circles in a grid with some spacing
    centers = np.zeros((n, 2))
    for i in range(n):
        x = i % cols / cols
        y = i // cols / rows
        centers[i] = [x, y]

    # Initial radius (very small to avoid overlap)
    radii = np.full(n, 0.01)

    # Define a function to compute the cost (sum of radii)
    def objective(x):
        # x contains flattened centers and radii
        x_flat = x.reshape((n, 4))  # 2 coordinates + radius
        centers_opt = x_flat[:, :2]
        radii_opt = x_flat[:, 2]
        return -np.sum(radii_opt)  # Minimize negative sum

    # Define constraints
    def constraint_center(x):
        x_flat = x.reshape((n, 4))
        centers_opt = x_flat[:, :2]
        radii_opt = x_flat[:, 2]
        return np.concatenate([
            centers_opt[:, 0] - radii_opt,
            1 - centers_opt[:, 0] - radii_opt,
            centers_opt[:, 1] - radii_opt,
            1 - centers_opt[:, 1] - radii_opt
        ])

    def constraint_overlap(x):
        x_flat = x.reshape((n, 4))
        centers_opt = x_flat[:, :2]
        radii_opt = x_flat[:, 2]
        dists = np.sqrt(np.sum((centers_opt[:, np.newaxis] - centers_opt[np.newaxis, :]) ** 2, axis=2))
        return np.concatenate([dists[i, j] - (radii_opt[i] + radii_opt[j]) for i in range(n) for j in range(i + 1, n)])

    # Define bounds for each circle (center coordinates and radius)
    bounds = []
    for _ in range(n):
        bounds += [(0.0, 1.0), (0.0, 1.0), (0.0, 1.0)]  # x, y, radius

    # Initial guess for the optimization
    x0 = np.concatenate([centers.flatten(), radii])

    # Set up the optimization problem
    cons = [
        {'type': 'ineq', 'fun': constraint_center},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Run the optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        tol=1e-8,
        options={'maxiter': 1000}
    )

    # Extract the solution
    centers_opt = result.x[:2 * n].reshape((n, 2))
    radii_opt = result.x[2 * n:].reshape(n)
    sum_radii = np.sum(radii_opt)

    # Validate the packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    return centers_opt, radii_opt, sum_radii
