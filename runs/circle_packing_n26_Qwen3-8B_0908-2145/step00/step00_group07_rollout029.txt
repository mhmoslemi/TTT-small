import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint

def run_packing():
    # Step 1: Generate an initial hexagonal packing in the unit square
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Base spacing for hexagonal grid
    spacing = 1.0 / np.sqrt(3)
    rows = int(np.ceil(np.sqrt(num_circles)))
    cols = int(np.ceil(num_circles / rows))

    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * spacing
            else:
                x = (j + 0.5) * spacing
            y = i * spacing * np.sqrt(3) / 2
            if i * cols + j < num_circles:
                centers[i * cols + j] = [x, y]
                radii[i * cols + j] = 0.1  # Start with small initial radius

    # Normalize to unit square
    centers /= max(np.max(centers, axis=0), 1.0)

    # Step 2: Define objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize

    # Step 3: Define constraints for non-overlapping and within square
    def constraint_overlap(radii, centers):
        n = centers.shape[0]
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(radii, centers):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1
        return 0

    # Step 4: Optimization using constrained optimization
    bounds = Bounds(0, 0.5)  # Radii must be non-negative, capped at 0.5

    # Objective function with parameters
    def objective_with_radii(x):
        return objective(x)

    # Constraints
    cons = [
        LinearConstraint(lambda x: np.array([constraint_overlap(x, centers)]), [-np.inf], [0]),
        LinearConstraint(lambda x: np.array([constraint_boundary(x, centers)]), [-np.inf], [0]),
    ]

    # Initial guess for radii
    initial_radii = radii.copy()

    # Optimize
    result = opt.minimize(
        fun=objective_with_radii,
        x0=initial_radii,
        bounds=bounds,
        constraints=cons,

    )

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
