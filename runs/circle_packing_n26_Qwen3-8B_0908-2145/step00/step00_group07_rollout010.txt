import numpy as np
import scipy.optimize as optimize
import random

def run_packing():

    n = 26
    # Initial radii estimate (based on total area and total sum)
    total_sum = 2.636
    initial_radii = np.full(n, total_sum / n)

    # Initial centers arranged in hexagonal grid
    # First row: 1 circle
    # Second row: 2 circles
    # Third row: 3 circles
    # Fourth row: 4 circles
    # Fifth row: 5 circles
    # Sixth row: 6 circles
    # Seventh row: 5 circles
    # Eighth row: 4 circles
    # Ninth row: 3 circles
    # Tenth row: 2 circles
    # Eleventh row: 1 circle

    # This will give us 26 circles
    rows = [1, 2, 3, 4, 5, 6, 5, 4, 3, 2, 1]
    centers = np.zeros((n, 2))
    idx = 0
    for i, row in enumerate(rows):
        row_centers = np.zeros((row, 2))
        for j in range(row):
            x = j + 0.5 * (1 - (i % 2))  # Offset for even rows
            y = i * (np.sqrt(3) * initial_radii[0]) + initial_radii[0]
            row_centers[j] = [x, y]
        centers[idx:idx+row] = row_centers
        idx += row

    # Ensure all circles fit within the square
    max_radius = max(initial_radii)
    padding = max_radius
    scale_factor = (1 - 2 * padding) / (np.max(centers[:, 0]) - np.min(centers[:, 0]))

    # Scale and shift the centers so they fit within [0, 1]
    centers = (centers - np.min(centers, axis=0)) * scale_factor + padding

    # Initial radii are adjusted to fit within the square
    initial_radii = initial_radii * (1 - 2 * padding) / (np.max(centers[:, 0]) - np.min(centers[:, 0]))

    # Objective function: maximize sum of radii while minimizing overlaps and staying within bounds
    def objective(params):
        radii = params[:n]
        centers = params[n:]
        centers = centers.reshape(n, 2)
        centers = (centers - np.min(centers, axis=0)) * scale_factor + padding
        radii = radii * (1 - 2 * padding) / (np.max(centers[:, 0]) - np.min(centers[:, 0]))

        # Check constraints
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return np.inf

            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)

                    return np.inf

        return -np.sum(radii)  # Negative for minimization

    # Initial parameters: radii and centers
    initial_params = np.concatenate((initial_radii, centers.flatten()))

    # Simulated annealing optimization
    result = optimize.minimize(
        objective,
        initial_params,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n * 3)]  # n radii and 2n centers
    )

    # Extract the result
    final_radii = result.x[:n]
    final_centers = result.x[n:].reshape(n, 2)

    # Final validation
    assert validate_packing(final_centers, final_radii)[0], "Invalid packing"

    return final_centers, final_radii, np.sum(final_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
