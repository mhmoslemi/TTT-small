def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing parameters
    num_circles = 26
    max_radius = 0.1  # Start with small radii for iterative optimization
    max_radius_initial = 0.1

    # Initial hexagonal packing on a square
    circles_per_row = [5, 5, 5, 5, 6]  # Total 26 circles
    rows = sum(circles_per_row)
    radius = max_radius_initial / (1 + 0.5)  # Conservative starting radius

    centers = []
    for i, cnt in enumerate(circles_per_row):
        for j in range(cnt):
            x = j * (3 * radius) + 0.5 * radius  # Shift horizontally to avoid edges
            y = i * (math.sqrt(3) * radius) + 0.5 * radius  # Shift vertically
            centers.append([x, y])

    # Convert list of centers to numpy array
    centers = np.array(centers)
    radii = np.full(num_circles, radius)

    # Optimization objective function
    def objective(radii, centers):
        return -np.sum(radii)  # Maximize sum of radii

    # Constraint function: define for scipy.optimize
    def constraint_circle_overlap(radii, centers):
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = math.hypot(dx, dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    # Constraint function for boundaries
    def constraint_boundary(radii, centers):
        for i in range(len(centers)):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0

    # Setup bounds for radii (non-negative)
    bounds = Bounds(lb=np.zeros(num_circles), ub=np.full(num_circles, 1.0))

    # Initial guess for optimization
    initial_guess = radii.copy()

    # Optimization parameters
    method = 'SLSQP'  # Sequential Least Squares Programming
    tol = 1e-10

    # Run optimization

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
