import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial hexagonal grid packing (approximation)
    n = 26
    rows = int(np.sqrt(n))
    cols = n // rows + (1 if n % rows != 0 else 0)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Set initial hexagonal grid
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]
    
    # Normalize to fit in [0, 1] x [0, 1]
    min_x, max_x = centers[:, 0].min(), centers[:, 0].max()
    min_y, max_y = centers[:, 1].min(), centers[:, 1].max()
    centers[:, 0] = (centers[:, 0] - min_x) / (max_x - min_x)
    centers[:, 1] = (centers[:, 1] - min_y) / (max_y - min_y)
    
    # Initial guess for radii (all equal)
    # We'll normalize to fit in the square by dividing by max distance
    max_distance = np.sqrt((max_x - min_x)**2 + (max_y - min_y)**2)
    radii = np.full(n, 1.0 / max_distance)
    
    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize -sum to maximize sum
    
    # Define constraints
    def constraint_overlap(x, y, r, idx):
        # Check if circle at idx overlaps with others
        for j in range(n):
            if j == idx:
                continue
            dx = x[idx] - x[j]
            dy = y[idx] - y[j]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < r[idx] + r[j] - 1e-8:
                return 1.0
        return 0.0
    
    def constraint_boundary(r, idx):
        x, y = centers[idx]
        if x - r < -1e-8 or x + r > 1.0 + 1e-8 or y - r < -1e-8 or y + r > 1.0 + 1e-8:
            return 1.0
        return 0.0
    
    # Optimization setup
    bounds = [(0, 1.0) for _ in range(n)]
    
    constraints = []
    for i in range(n):
        # Add constraint for boundary
        def boundary_constraint(r):
            return constraint_boundary(r, i)
        constraints.append({'type': 'ineq', 'fun': boundary_constraint})
        
        # Add constraint for overlaps
        def overlap_constraint(r):
            return constraint_overlap(centers, radii, r, i)
        constraints.append({'type': 'ineq', 'fun': overlap_constraint})
    
    # Run the optimization
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )
    
    # Extract results

        raise ValueError(f"Invalid packing: {msg}")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
