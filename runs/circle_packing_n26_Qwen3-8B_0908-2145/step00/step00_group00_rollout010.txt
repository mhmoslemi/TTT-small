import numpy as np
import scipy.optimize as optimize
import math

def run_packing():

    n = 26
    centers = np.zeros((n, 2))
    dx = 1.0 / math.sqrt(3)  # Spacing for hexagonal packing
    dy = 1.0 / 2.0           # Vertical spacing

    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * dx + 0.5 * dx  # Shift offset for even rows
        y = row * dy + 0.5 * dy  # Shift offset
        centers[i, 0] = x
        centers[i, 1] = y

    radii = np.full(n, 0.04)

    def constraints(radii_flat):

        for i in range(n):

            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return np.inf  # Violation of boundary constraints

                dy = centers[i, 1] - centers[j, 1]
                dist = math.hypot(dx, dy)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):

            return False, f"Circle {i} has negative radius {radii[i]}"

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12

            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
