import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import warnings

def run_packing():
    # Initial guess: Hexagonal grid with 5 rows and 5 columns
    num_circles = 26
    rows = 5
    cols = 5
    grid = np.zeros((num_circles, 2))
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * 0.5
            else:
                x = (j + 0.5) * 0.5
            y = i * np.sqrt(3) / 2
            idx = i * cols + j
            if idx < num_circles:
                grid[idx] = [x, y]

    # Scale the grid to fit in the unit square
    grid /= np.max(grid)

    # Initial radii: assume uniform radius
    initial_radius = 0.04

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Define constraints
    def constraint_position(radii, centers):
        return np.concatenate([
            centers[:, 0] - radii >= -1e-12,
            1 + 1e-12 - centers[:, 0] >= -1e-12,
            centers[:, 1] - radii >= -1e-12,
            1 + 1e-12 - centers[:, 1] >= -1e-12
        ])

    def constraint_overlap(radii, centers):
        dists = squareform(pdist(centers))
        return np.array([dists[i, j] - radii[i] - radii[j] for i in range(num_circles) for j in range(i + 1, num_circles)]).reshape(-1)

    # Define bounds: radii must be non-negative
    bounds = [(0, 1) for _ in range(num_circles)]

    # Set up the optimization problem
    centers = grid.copy()
    radii = np.full(num_circles, initial_radius)

    # Constraints for optimization
    cons = []
    cons.append({'type': 'ineq', 'fun': lambda r: constraint_position(r, centers)})
    cons.append({'type': 'ineq', 'fun': lambda r: constraint_overlap(r, centers)})

    # Optimization
    res = minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 1000}
    )

    # Extract results
    centers = grid.copy()
    radii = res.x
    sum_radii = np.sum(radii)
