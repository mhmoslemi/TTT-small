import scipy.optimize as optimize
from typing import Tuple, List

def objective_function(params, n_circles):

    radii = params[n_circles * 2:]
    return -np.sum(radii)

def constraint_non_overlap(params, n_circles):

    radii = params[n_circles * 2:]
