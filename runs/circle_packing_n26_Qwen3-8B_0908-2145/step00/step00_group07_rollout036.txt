import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial hexagonal packing with 26 circles
    # We will arrange them in a hexagonal grid
    # We need to estimate the number of rows and columns
    # A hexagonal grid has roughly sqrt(n) rows and sqrt(2n) columns
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(np.sqrt(2 * n)))
    
    # For simplicity, we'll start with a grid-like structure
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create a hexagonal grid
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (row % 2)
        y = row * np.sqrt(3) / 2
        centers[i] = [x, y]
    
    # Adjust centers to fit in the unit square
    centers = centers / (np.max(centers, axis=0) + 1e-12)
    
    # Radius for all circles initially
    initial_radius = 0.2
    radii = np.full(n, initial_radius)
    
    # Define the optimization function
    def objective(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:]
        
        # Check constraints
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.inf  # Penalize boundary violation

                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return np.inf  # Penalize overlap
        
        # Objective is to maximize the total radius
        return -np.sum(radii_flat)
    
    # Initial guess
    initial_guess = np.concatenate([centers.flatten(), radii])
    
    # Optimization using L-BFGS-B
    result = opt.minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(2 * n)] + [(0, 1) for _ in range(n)],
        options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
    )

    sum_radii = np.sum(final_radii)
    print(f"Total sum of radii: {sum_radii}")
    
    return final_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
