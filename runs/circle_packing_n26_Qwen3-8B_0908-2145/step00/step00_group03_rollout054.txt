import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run_packing():
    # Define number of circles
    n_circles = 26

    # Define the initial hexagonal grid layout
    def hexagonal_grid(n):
        num_rows = int(np.ceil(np.sqrt(n)))
        num_cols = int(np.ceil(n / num_rows))
        centers = np.zeros((n, 2))
        index = 0
        for r in range(num_rows):
            for c in range(num_cols):
                x = c + 0.5 * (r % 2)
                y = r
                if r % 2 == 1:
                    x += 0.5
                centers[index, 0] = x
                centers[index, 1] = y
                index += 1
        return centers

    # Initial guess: flatten the hexagonal grid into a vector
    centers = hexagonal_grid(n_circles)

    radii = np.ones(n_circles) * 0.05

    # Objective function: maximize sum of radii (minimize -sum(radii))
    def objective(x):
        return -np.sum(x[::2] + x[1::2])  # x is arranged as [x1, y1, r1, x2, y2, r2, ...]

    # Unpack the variables
    def unpack(x):
        centers_flat = x[::3]
        radii_flat = x[2::3]
        centers = centers_flat.reshape((-1, 2))
        radii = radii_flat
        return centers, radii

    # Constraint 1: boundary constraints
    def boundary_constraints(x):
        centers_flat = x[::3]
        radii_flat = x[2::3]
        centers = centers_flat.reshape((-1, 2))
        radii = radii_flat
        constraints = []
        for i in range(n_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            # x must be in [r_i, 1 - r_i]
            constraints.append((x_i - r_i >= -1e-12))
            constraints.append((x_i + r_i <= 1 + 1e-12))
            # y must be in [r_i, 1 - r_i]
            constraints.append((y_i - r_i >= -1e-12))
            constraints.append((y_i + r_i <= 1 + 1e-12))
        return constraints

    # Constraint 2: non-overlapping constraints
    def non_overlap_constraints(x):
        centers_flat = x[::3]
        radii_flat = x[2::3]

        radii = radii_flat
        constraints = []
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                x_i, y_i = centers[i]
                x_j, y_j = centers[j]
                r_i = radii[i]
                r_j = radii[j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                constraints.append((dist - r_i - r_j >= -1e-12))
        return constraints

    # Flatten the initial guess into a 1D vector
    x0 = np.concatenate([centers.flatten(), radii])

    bounds = Bounds(0.0, 1.0)

    constraints = []
    for c in boundary_constraints(x0):

    # Run optimization
    result = minimize(

        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-9, 'maxiter': 1000}

    # Extract the result
    x_opt = result.x

    sum_radii = np.sum(radii_opt)

    if not result.success:

    return centers_opt, radii_opt, sum_radii
