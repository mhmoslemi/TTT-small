from scipy.optimize import minimize

def generate_hexagonal_grid(n_circles, side_length=1.0):

    Generate a hexagonal grid of circles with given number of circles.
    """
    # Calculate number of rows in the hexagonal grid
    rows = int(np.ceil(np.sqrt(2 * n_circles)))
    cols = int(np.ceil(n_circles / rows))
    
    centers = []
    radii = []

        for col in range(cols):
            x = col * (1 + side_length * np.sqrt(3)) / 2

                radii.append(0.1)  # Initial guess for radius
            
            if len(centers) == n_circles:
                break
        if len(centers) == n_circles:
            break
    
    return np.array(centers), np.array(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
