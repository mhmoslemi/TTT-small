import scipy.optimize as optimize

def run_packing():

    n = 26

    circles_per_col = int(np.ceil(n / circles_per_row))

    centers = np.zeros((n, 2))
    radii = np.zeros(n)

        row = i // circles_per_row

        y = row * np.sqrt(3) / 2
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = 0.1  # Initial radius

    # Normalize to fit in unit square
    max_x = centers[:, 0].max()
    max_y = centers[:, 1].max()
    centers /= np.max([max_x, max_y])
    radii *= 0.5  # Adjust for unit square

    # Define the objective function
    def objective(x):
        # x is a flat array of [x1, y1, r1, x2, y2, r2, ...]
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        total_radius = np.sum(radii_flat)
        return -total_radius  # Minimize negative total radius

    # Define the constraints
    def constraint_boundary(x):
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
