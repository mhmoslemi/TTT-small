import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Number of circles
    n = 26
    # Create a hexagonal grid layout as our initial guess
    num_rows = int(np.ceil(np.sqrt(n)))
    num_cols = n // num_rows + 1

    # Define initial centers and radii
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    for i in range(n):
        row = i // num_cols
        col = i % num_cols

        # Hexagonal grid layout
        x = col + 0.5 * row
        y = row * np.sqrt(3) / 2

        # Adjust to fit within the unit square
        x_adj = x / max(x, 1 - x)
        y_adj = y / max(y, 1 - y)
        x = x_adj / (1 + 0.5 * row)
        y = y_adj / (1 + np.sqrt(3) / 2 * row)

        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = 0.04  # Initial guess for radius

    # Objective function to maximize total radius
    def objective(radii_flat):
        return -np.sum(radii_flat)

    # Constraint: circles must be within the unit square
    def in_square(radii_flat):
        centers_flat = np.reshape(centers, (-1, 2))
        r = np.reshape(radii_flat, -1)
        for i in range(n):
            x, y = centers_flat[i]
            if x - r[i] < 0 or x + r[i] > 1 or y - r[i] < 0 or y + r[i] > 1:
                return -1
        return 0

    # Constraint: circles must not overlap
    def no_overlap(radii_flat):
        centers_flat = np.reshape(centers, (-1, 2))
        r = np.reshape(radii_flat, -1)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < r[i] + r[j] - 1e-8:
                    return -1
        return 0

    # Define optimization variables
    x0 = radii.copy()

    # Define bounds: radii must be non-negative
    bounds = [(0, 1) for _ in range(n)]

    # Define constraints
    constraints = [
        {'type': 'ineq', 'fun': in_square, 'args': (radii,)},
        {'type': 'ineq', 'fun': no_overlap, 'args': (radii,)}
    ]

    # Optimize
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract final centers and radii
    final_centers = centers.copy()
    final_radii = result.x

    # Ensure the solution is valid (because of optimization)
    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    # Calculate total radius
    sum_radii = np.sum(final_radii)

    return final_centers, final_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
