import numpy as np
import cvxpy as cp

def run_packing():
    n = 26
    # Define variables for centers and radii
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n, non_negative=True)

    # Objective: maximize the total sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Each circle must lie within the unit square
    for i in range(n):
        x, y = centers[i]
        constraints.append(x >= 0)
        constraints.append(x <= 1)
        constraints.append(y >= 0)
        constraints.append(y <= 1)

    # Each circle has a non-negative radius
    constraints.append(cp.constraints.NonNegativity(radii))

    # Circles must not overlap
    for i in range(n):
        for j in range(i + 1, n):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            dist = cp.sqrt((x1 - x2) ** 2 + (y1 - y2) ** 2)
            constraints.append(dist >= radii[i] + radii[j])

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CVXPY_MI, verbose=True)

    if prob.status != "optimal":
        raise RuntimeError(f"Optimization failed with status: {prob.status}")

    centers_np = centers.value
    radii_np = radii.value
    sum_radii = cp.sum(radii).value

    return centers_np, radii_np, sum_radii
