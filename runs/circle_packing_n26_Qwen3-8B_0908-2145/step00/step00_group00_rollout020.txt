```python
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Step 1: Generate an initial hexagonal grid of 26 circles
    num_circles = 26
    rows = int(np.sqrt(num_circles))
    cols = (num_circles + rows - 1) // rows

    # Generate initial hexagonal grid
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            x = j + 0.5 * i
            y = i
            centers[idx] = [x, y]
            radii[idx] = 0.03 * (num_circles - idx)  # Initial small radii

    # Scale the grid within [0,1]^2
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers /= max_x if max_x > 0 else 1.0
    centers /= max_y if max_y > 0 else 1.0

    # Step 2: Define variables for optimization
    x = cp.Variable((num_circles, 2))
    r = cp.Variable(num_circles, nonneg=True)

    # Objective: Maximize total radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []
    for i in range(num_circles):
        x_i, y_i = x[i]
        r_i = r[i]
        constraints.append(x_i >= r_i)
        constraints.append(x_i <= 1 - r_i)
        constraints.append(y_i >= r_i)
        constraints.append(y_i <= 1 - r_i)

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dx = x[i][0] - x[j][0]
            dy = x[i][1] - x[j][1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= r[i] + r[j] - 1e-5)

    # Step 3: Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    result = prob.solve(solver=cp.CVXOPT, verbose=False)

    if result is None or result < 1e-6:
        return centers, radii, 0.0

    # Step 4: Extract result
```

```python
def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
```
