import numpy as np
import scipy.optimize as optimize
import math

def run_packing():
    # Initial grid setup for 26 circles
    # 5 rows, 5 columns with one extra circle added in the last row
    n = 26
    rows = 5
    cols = 5
    # Adjust last row to accommodate 26 circles
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Generate initial positions in grid
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col / (cols - 1)
        y = row / (rows - 1)
        centers[i] = [x, y]
    
    # Initial guess for radii (based on grid spacing)
    spacing = 1.0 / (cols - 1)
    initial_radius = spacing * 0.5
    radii[:] = initial_radius
    
    # Define objective function: maximize sum of radii
    def objective(x):
        # x[0:26] = centers
        # x[26:52] = radii
        centers_eval = x[:26].reshape(n, 2)
        radii_eval = x[26:26 + n]
        
        # Ensure radii are non-negative
        if np.any(radii_eval < 0):
            return -np.inf
        
        # Check if all circles are within the square
        for i in range(n):
            x, y = centers_eval[i]
            r = radii_eval[i]
            if (x - r < -1e-12) or (x + r > 1 + 1e-12) or (y - r < -1e-12) or (y + r > 1 + 1e-12):
                return -np.inf
        
        # Check for circle overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_eval[i][0] - centers_eval[j][0]
                dy = centers_eval[i][1] - centers_eval[j][1]
                dist = math.hypot(dx, dy)
                if dist < radii_eval[i] + radii_eval[j] - 1e-12:
                    return -np.inf
        
        # Objective function: maximize sum of radii
        return -np.sum(radii_eval)
    
    # Bounds for centers (must be within [0,1]^2)
    bounds_centers = [(0.0, 1.0) for _ in range(2 * n)]
    # Bounds for radii (non-negative)
    bounds_radii = [(0.0, 1.0) for _ in range(n)]
    bounds = bounds_centers + bounds_radii
    
    # Initial guess: centers and radii as above
    x0 = np.concatenate((centers.flatten(), radii))
    
    # Run optimization
    result = optimize.minimize(objective, x0, bounds=bounds, method='SLSQP', tol=1e-8, options={'maxiter': 200})

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
