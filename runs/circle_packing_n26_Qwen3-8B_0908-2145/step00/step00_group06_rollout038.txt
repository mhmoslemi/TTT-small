import numpy as np
import scipy.optimize as opt
import warnings

def run_packing():

    centers = np.random.rand(26, 2)

    # Objective function: maximize sum of radii

        return -np.sum(r)  # Negative for minimization

    def constraints(variables):
        x = variables[:52].reshape(26, 2)
        r = variables[52:]
        constraints = []

        # Bounding box constraint
        for i in range(26):
            x_i, y_i = x[i]
            constraints.append((x_i - r[i] >= -1e-12) * 1.0)
            constraints.append((x_i + r[i] <= 1 + 1e-12) * 1.0)
            constraints.append((y_i - r[i] >= -1e-12) * 1.0)
            constraints.append((y_i + r[i] <= 1 + 1e-12) * 1.0)

        # Non-negativity of radii
        for i in range(26):
            constraints.append(r[i] >= -1e-12)

        # Circle-circle distance constraint
        for i in range(26):
            for j in range(i + 1, 26):
                dx = x[i, 0] - x[j, 0]
                dy = x[i, 1] - x[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist >= r[i] + r[j] - 1e-12)

    # Optimization setup
    bounds = []

        result = opt.minimize(
