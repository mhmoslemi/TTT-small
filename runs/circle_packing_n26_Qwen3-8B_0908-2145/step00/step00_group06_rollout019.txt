import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

from functools import partial

def run_packing():

    # Calculate initial positions based on hexagonal grid

    # Number of rows in hexagonal packing
    num_rows = 6
    num_cols_per_row = 5

    # Positions of circles in hexagonal grid
    positions = []
    for i in range(num_rows):
        for j in range(num_cols_per_row):
            x = j - (num_cols_per_row // 2)
            y = i - (num_rows // 2)
            positions.append((x, y))
    
    # Convert to numpy array
    centers = np.array(positions)
    
    # Normalize to unit square
    centers /= max(np.abs(centers).max(axis=0))
    
    # Initial guess for radii
    # Assume all circles have radius 0.1 initially
    radii = np.full(centers.shape[0], 0.1)
    
    # Calculate initial sum of radii
    sum_radii = sum(radii)

    # Define constraints for optimization
    def constraint_non_overlap(x, centers, radii):
        # x is a flattened array of [x1, y1, r1, x2, y2, r2, ...]
        x_flat = x.reshape(-1)
        centers_flat = x_flat[:2 * centers.shape[0]]
        radii_flat = x_flat[2 * centers.shape[0]:]

        centers_opt = centers_flat.reshape(centers.shape)
        radii_opt = radii_flat.reshape(radii.shape)

        # Check for overlaps between every pair of circles
        for i in range(centers_opt.shape[0]):
            for j in range(i + 1, centers_opt.shape[0]):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_opt[i] + radii_opt[j] - 1e-12:
                    return dist - (radii_opt[i] + radii_opt[j] - 1e-12)
        return 0.0

    def constraint_boundary(x, centers, radii):
        x_flat = x.reshape(-1)
        centers_flat = x_flat[:2 * centers.shape[0]]
        radii_flat = x_flat[2 * centers.shape[0]:]

        centers_opt = centers_flat.reshape(centers.shape)
        radii_opt = radii_flat.reshape(radii.shape)

        for i in range(centers_opt.shape[0]):
            x, y = centers_opt[i]
            r = radii_opt[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return 1.0
        return 0.0

    def constraint_radius_non_negative(x, _, _):
        x_flat = x.reshape(-1)
        radii_flat = x_flat[2 * centers.shape[0]:]
        for r in radii_flat:
            if r < -1e-12:
                return r + 1e-12
        return 0.0

        centers_opt = centers_flat.reshape(centers.shape)
        radii_opt = radii_flat.reshape(radii.shape)
        return -sum(radii_opt)  # Maximize sum of radii

    # Define optimization bounds

    bounds = Bounds([0.0] * 2 * centers.shape[0] + [0.0] * centers.shape[0],

    constraint_radius_func = partial(constraint_radius_non_negative, centers=centers, radii=radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
