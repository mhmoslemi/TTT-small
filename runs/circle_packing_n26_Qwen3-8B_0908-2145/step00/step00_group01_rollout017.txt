import numpy as np
import scipy.optimize as opt

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal grid packing
    num_circles = 26
    cols = 5  # 5 columns
    rows = 6  # 6 rows

    # Start with a hexagonal grid
    radius_initial = 0.04
    centers = np.zeros((num_circles, 2))
    radii = np.ones(num_circles) * radius_initial

    # Generate initial hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i * np.sqrt(3) / 2
            idx = i * cols + j
            centers[idx] = [x, y]

    # Define optimization variables
    x = np.concatenate([centers.flatten(), radii])

    # Define objective: minimize the total radius (equivalently maximize it)
    def objective(x):
        return -np.sum(x[26*2:])  # Minimize -sum(radii), which is equivalent to maximizing sum(radii)

    # Define constraints: non-overlap and boundary
    def constraint_non_overlap(x):
        centers = x[:26*2].reshape(26, 2)
        radii = x[26*2:]
        for i in range(26):
            for j in range(i+1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1  # Violation
        return 0

    def constraint_boundary(x):
        centers = x[:26*2].reshape(26, 2)
        radii = x[26*2:]
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-8 or x + r > 1 + 1e-8
                or y - r < -1e-8 or y + r > 1 + 1e-8):
                return -1  # Violation
        return 0

    # Define bounds
    bounds = []
    for _ in range(26*2):
        bounds.append((0, 1))  # Centers must be in [0, 1]^2
    for _ in range(26):
        bounds.append((0, 1))  # Radii must be >= 0

    # Define constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_non_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    # Optimize
    result = opt.minimize(
        objective,
        x,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-6,
        maxiter=100
    )

    # Extract centers and radii
    centers = result.x[:26*2].reshape(26, 2)
    radii = result.x[26*2:]

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
