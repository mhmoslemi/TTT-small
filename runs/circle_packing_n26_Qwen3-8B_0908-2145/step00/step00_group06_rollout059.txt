import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    n = 26

    # Start with center circle at (0.5, 0.5)

    # For 26 circles, we can pack a few concentric layers

    radii = []
    radius = 0.1  # Starting radius for outer circles

    radii.append(0.1)

        for angle in angles:
            x = 0.5 + layer * distance * np.cos(angle)
            y = 0.5 + layer * distance * np.sin(angle)
            centers.append([x, y])
            radii.append(0.1)

        # Add inner points of hexagonal layer
        for i in range(6):
            x = 0.5 + (layer - 0.5) * distance * np.cos(angles[i])
            y = 0.5 + (layer - 0.5) * distance * np.sin(angles[i])
            centers.append([x, y])
            radii.append(0.1)

    # Trim to exactly 26 circles
    centers = np.array(centers[:26])
    radii = np.array(radii[:26])

        return -np.sum(x)  # Negative because we use minimization

        return np.sum((centers_flat - np.array([0.5, 0.5])) ** 2)  # Just for demonstration

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):

        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
