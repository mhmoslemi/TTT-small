import numpy as np
import scipy.optimize as optimize

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    # Initial grid of circles in a staggered grid pattern
    rows = 5
    cols = 5
    radius_initial = 0.12
    centers = []
    radii = []

    # Create a grid of centers
    for i in range(rows):
        for j in range(cols):
            x = j * (1 / (cols - 1)) if j < cols - 1 else 1.0
            y = i * (1 / (rows - 1)) if i < rows - 1 else 1.0
            centers.append([x, y])
            radii.append(radius_initial)

    # Add one more circle at a corner
    centers.append([0.0, 0.0])
    radii.append(0.0)

    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function to maximize the sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Negative because we minimize

    # Constraint: circles must not overlap
    def constraint_overlap(radii, centers):
        n = len(radii)
        dists = np.zeros(n * (n - 1) // 2)
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                dists[idx] = dist - (radii[i] + radii[j])
                idx += 1
        return dists

    # Constraint: circles must be within the unit square
    def constraint_boundary(radii, centers):
        n = len(radii)
        dists = np.zeros(n)
        for i in range(n):
            x, y = centers[i]
            dists[i] = np.min([x - radii[i], 1.0 - x - radii[i], y - radii[i], 1.0 - y - radii[i]])
        return dists

    # Optimization
    bounds = [(0.0, 1.0) for _ in range(len(radii))]  # Radii bounds
    cons = [
        {'type': 'ineq', 'fun': lambda r: constraint_overlap(r, centers)},
        {'type': 'ineq', 'fun': lambda r: constraint_boundary(r, centers)}
    ]
    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'disp': False, 'maxiter': 1000}
    )

    # Extract the optimized results

    # Return the result
    return optimal_centers, optimal_radii, np.sum(optimal_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
