import numpy as np
import time
import math
import random

def _project_h(h, target_sum):
    """Clip to [0,1] and adjust to hit the exact sum target."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s > 0:
        h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)

    # iterative residual correction
    for _ in range(10):
        residual = target_sum - h.sum()
        if abs(residual) < 1e-12:
            break
        add = residual / h.size
        h += add
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h, dx):
    """Return the discrete C5 = max_k ∫ h(x)(1-h(x+k)) dx."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.

    Returns:
        (h_values, c5_bound, n_points)
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # Determine number of samples
    n_points = kwargs.get('n_points')
    if n_points is None:
        init_global = globals().get('initial_h_values')
        if init_global is not None:
            n_points = len(init_global)
        else:
            n_points = 70  # fallback default

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialise h
    init_global = globals().get('initial_h_values')
    if init_global is not None:
        h_current = np.array(init_global, dtype=float)
    else:
        h_current = rng.random(n_points)
    h_current = _project_h(h_current, target_sum)

    best_h = h_current.copy()
    best_val = _c5_value(best_h, dx)

    # Simulated annealing parameters
    max_iter = 200000  # safety cap; actual loop stops by time
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:
            break
        # temperature decreasing linearly with remaining time
        remaining = max(budget_s - elapsed, 1e-6)
        temp = remaining / budget_s  # in (0,1]
        sigma = 0.05 * temp + 0.001  # perturbation scale

        # propose a new candidate by perturbing a few entries
        cand = best_h.copy()
        # number of coordinates to change (1–3)
        k = rng.integers(1, 4)
        idx = rng.choice(n_points, size=k, replace=False)
        cand[idx] += rng.normal(0.0, sigma, size=k)
        cand = _project_h(cand, target_sum)

        cand_val = _c5_value(cand, dx)

        # Acceptance test
        if cand_val < best_val:
            best_h, best_val = cand, cand_val
        else:
            # Metropolis criterion
            prob = math.exp(-(cand_val - best_val) / (temp + 1e-12))
            if rng.random() < prob:
                best_h, best_val = cand, cand_val

        iter_count += 1
        if iter_count >= max_iter:
            break

    return best_h, best_val, n_points
