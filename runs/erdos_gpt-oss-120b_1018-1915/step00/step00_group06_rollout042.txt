def run(seed=42, budget_s=1100, **kwargs):
    # seed reproducibility
    np.random.seed(seed)

    # ------------------------------------------------------------------

    def repair_vec(v, target_sum):

        np.clip(v, 0.0, 1.0, out=v)
        # set mean to 0.5 (so sum ≈ target)
        v -= v.mean()
        v += 0.5
        np.clip(v, 0.0, 1.0, out=v)

        if cur != 0.0:

        diff = target_sum - v.sum()
        if abs(diff) > 1e-12:
            free = (v > 0.0) & (v < 1.0)
            cnt = free.sum()
            if cnt > 0:

    start_time = time.time()
