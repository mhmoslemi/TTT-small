import time
import numpy as np

def compute_c5(h, dx):
    """Return max overlap integral C5 for step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def resample_to_n(source, n):
    """Linear interpolation of source (defined on [0,2]) to n points."""
    m = source.shape[0]
    xs = np.linspace(0.0, 2.0, m, endpoint=False)
    xt = np.linspace(0.0, 2.0, n, endpoint=False)
    return np.interp(xt, xs, source)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 to minimize C5.

    Returns:
        h_values (np.ndarray) : optimized values (length n_points)
        c5_bound (float)      : achieved C5 value
        n_points (int)        : number of discretization points
    """
    rnd = np.random.RandomState(seed)
    start_time = time.time()

    # choose discretization size (even for convenience)
    n_points = 200  # < 1k and provides decent resolution
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # initialise h
    if 'initial_h_values' in globals():
        h = resample_to_n(initial_h_values, n_points)
        # enforce sum exactly by a simple scaling + projection onto bounds
        cur_sum = h.sum()
        if cur_sum != 0:
            h *= target_sum / cur_sum
        h = np.clip(h, 0.0, 1.0)
        # after clipping the sum may drift; fix with a quick redistribution
        deficit = target_sum - h.sum()
        if abs(deficit) > 1e-12:
            # add/subtract uniformly on non‑saturated entries
            free = np.where((h > 0.0) & (h < 1.0))[0]
            if free.size > 0:
                delta = deficit / free.size
                h[free] += delta
                h = np.clip(h, 0.0, 1.0)
    else:
        # simple half‑ones pattern shuffled
        h = np.zeros(n_points)
        h[:int(target_sum)] = 1.0
        rnd.shuffle(h)

    # ensure feasibility exactly
    h = np.clip(h, 0.0, 1.0)
    # final adjustment to match sum precisely using pairwise swaps
    cur_sum = h.sum()
    while abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # pick an index to increase (if diff>0) or decrease (if diff<0)
        if diff > 0:
            inc_idx = rnd.choice(np.where(h < 1.0)[0])
            dec_idx = rnd.choice(np.where(h > 0.0)[0])
        else:
            inc_idx = rnd.choice(np.where(h > 0.0)[0])

        h[dec_idx] -= delta
        cur_sum = h.sum()

    # annealing parameters
    T = 0.5            # initial temperature
    T_end = 1e-4

    max_step = 0.2     # maximal amount moved between two coordinates
