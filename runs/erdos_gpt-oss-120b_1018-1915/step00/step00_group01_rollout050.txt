import time
import numpy as np
import random

# helpers (provided by harness)
# evaluate_erdos_solution(h)  -> float (the C5 value)
# initial_h_values           -> optional starting array

def _project_to_constraints(h, target_sum):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = target_sum - total
    # iterative redistribution among non‑boundary entries
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        nfree = free.sum()
        if nfree == 0:
            # distribute to all entries proportionally (rare)
            h += diff / n
        else:
            add = diff / nfree
            h[free] += add
            # re‑clip
            h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        diff = target_sum - total
    # final tiny adjustment to guarantee exact sum (may push a value slightly out of bounds)
    if abs(diff) > 1e-12:
        # pick a non‑boundary index if possible
        idx = np.where((h > 0.0) & (h < 1.0))[0]
        if idx.size == 0:
            idx = np.arange(n)
        i = idx[0]
        h[i] = min(1.0, max(0.0, h[i] + diff))
    return h

def _cost(h, dx):
    """Maximum overlap integral for discrete h."""
    # correlation of h with (1-h), full mode
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _initial_vector(n):
    """Create a starting vector of length n."""
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # interpolate to new size n
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        # simple half‑filled uniform pattern
        h = np.zeros(n, dtype=float)
        half = n // 2
        h[:half] = 1.0
    # enforce exact integral
    target = n / 2.0          # because dx = 2/n, sum must be n/2
    h = _project_to_constraints(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # choose a resolution; 200 gives a good trade‑off
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0

    h_current = _initial_vector(n)
    best_h = h_current.copy()
    best_c = _cost(best_h, dx)

    # simple simulated‑annealing / hill‑climbing loop
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95:
        iter_count += 1
        # propose a transfer of mass between two coordinates
        i, j = random.sample(range(n), 2)
        hi, hj = h_current[i], h_current[j]

        # maximum admissible transfer from i to j
        max_delta = min(hi, 1.0 - hj)
        if max_delta <= 0:
            continue
        # step size shrinks slowly with iterations
        scale = 0.5 * (1.0 - iter_count / (budget_s * 1000.0))
        delta = max_delta * scale * random.random()
        if delta <= 0:
            continue

        h_new = h_current.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # compute cost
        c_new = _cost(h_new, dx)

        # accept if improvement, else small probability
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h_current = h_new
        else:
            # temperature‑like acceptance
            temperature = max(0.001, 0.01 * (1.0 - (time.time() - start_time) / (budget_s * 0.95)))
            if random.random() < temperature:
                h_current = h_new

    # final projection to guarantee constraints (might be slightly violated by numeric noise)
    best_h = _project_to_constraints(best_h, target_sum)
    best_c = _cost(best_h, dx)

    return best_h, float(best_c), n
