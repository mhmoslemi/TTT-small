import time
import random
import numpy as np

def _repair(vec, target_sum):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    v = np.clip(vec, 0.0, 1.0)
    s = v.sum()
    if s == 0.0:
        v[:] = target_sum / v.size
        return v
    # scale to hit the target sum, then re‑clip
    v = v * (target_sum / s)
    v = np.clip(v, 0.0, 1.0)
    # final tiny correction on the first free entry
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        # find an index that can absorb the diff without breaking bounds
        for i in range(v.size):
            if (diff > 0 and v[i] < 1.0) or (diff < 0 and v[i] > 0.0):
                v[i] = np.clip(v[i] + diff, 0.0, 1.0)
                break
    return v

def _objective(h, dx):
    """Maximum overlap integral for a candidate step function."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C₅ problem."""
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80                       # fallback size
        h0 = np.full(n, 0.5)         # crude feasible start (sum = n/2)
    dx = 2.0 / n
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # DE parameters
    # ------------------------------------------------------------------
    pop_size = max(30, n // 2)       # a modest, dimension‑aware population
    F = 0.5
    CR = 0.9
    max_gen = 5000                  # upper bound, actual stop is time‑based

    # ------------------------------------------------------------------
    # initialise population (include the provided parent if feasible)
    # ------------------------------------------------------------------
    pop = []
    scores = []
    # helper to create a feasible random individual
    def _rand_ind():
        v = np.random.rand(n)
        return _repair(v, target_sum)

    # add initial vector if it's feasible; otherwise replace with repaired one
    if np.all((h0 >= 0) & (h0 <= 1)) and abs(h0.sum() - target_sum) < 1e-9:
        pop.append(h0.copy())
    else:
        pop.append(_rand_ind())
    # fill the rest
    while len(pop) < pop_size:
        pop.append(_rand_ind())
    pop = np.stack(pop)
    # evaluate
    for i in range(pop_size):
        scores.append(_objective(pop[i], dx))
    scores = np.array(scores)

    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    # ------------------------------------------------------------------
    # main DE loop
    # ------------------------------------------------------------------
    gen = 0
    while gen < max_gen and (time.time() - start_time) < (budget_s - 0.5):
        for i in range(pop_size):
            # mutation: select three distinct others
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = random.sample(idxs, 3)
            mutant = pop[a] + F * (pop[b] - pop[c])
            # crossover (binomial)
            cross_mask = np.random.rand(n) < CR
            # ensure at least one dimension from mutant
            j_rand = random.randrange(n)
            cross_mask[j_rand] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = _repair(trial, target_sum)

            trial_score = _objective(trial, dx)
            # selection
            if trial_score <= scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = float(trial_score)
                    best_h = trial.copy()
        gen += 1

    # final sanity clip (should already hold)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair(best_h, target_sum)

    return (best_h, best_score, n)
