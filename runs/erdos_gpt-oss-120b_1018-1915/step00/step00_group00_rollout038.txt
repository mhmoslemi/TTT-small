import time, random, math
import numpy as np

def _project_feasible(v, target):
    """Project vector v onto [0,1] with sum == target using simple clipping+redistribute."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    # If we need to add mass
    if cur < target:
        deficit = target - cur
        # indices not at upper bound
        free = np.where(v < 1.0 - 1e-12)[0]
        if free.size == 0:
            return v
        add = deficit / free.size
        v[free] += add
        # clip again and recurse if needed
        return _project_feasible(v, target)
    else:
        excess = cur - target
        free = np.where(v > 1e-12)[0]
        if free.size == 0:
            return v
        sub = excess / free.size
        v[free] -= sub
        return _project_feasible(v, target)

def _score(h, dx):
    """Maximum overlap integral for a sampled step function."""
    cm = np.correlate(h, 1.0 - h, mode='full')
    return np.max(cm) * dx

def _mutate(h, rng):
    n = h.size
    i, j = rng.randint(0, n, size=2)
    while i == j:
        j = rng.randint(0, n)
    # decide direction of transfer
    if rng.random() < 0.5:
        # i -> j
        max_transfer = min(0.5, h[i], 1.0 - h[j])
        if max_transfer <= 0:
            return h
        delta = rng.random() * max_transfer
        h[i] -= delta
        h[j] += delta
    else:
        # j -> i
        max_transfer = min(0.5, h[j], 1.0 - h[i])
        if max_transfer <= 0:
            return h
        delta = rng.random() * max_transfer
        h[j] -= delta
        h[i] += delta
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # Determine n_points and initial vector
    try:
        init = initial_h_values  # provided by harness if exists
        h0 = np.array(init, dtype=float)
        n = h0.size
    except NameError:
        n = 45
        h0 = np.zeros(n, dtype=float)

    target_sum = n / 2.0
    if h0.sum() == 0:
        # start with random feasible vector
        rnd = rng.random(n)
        h0 = _project_feasible(rnd, target_sum)
    else:
        # ensure feasibility
        h0 = _project_feasible(h0, target_sum)

    dx = 2.0 / n
    best_h = h0.copy()
    best_score = _score(best_h, dx)

    # annealing parameters
    temp = 0.02
    decay = 0.9995
    min_temp = 1e-6
    max_iter = 2000000  # safety cap
    iter_count = 0
    no_improve = 0
    restart_budget = 5  # number of possible restarts
    restarts_done = 0

    while time.time() - start_time < budget_s - 5 and iter_count < max_iter:
        # if stagnated, consider restart
        if no_improve > 20000 and restarts_done < restart_budget:
            # re‑initialize
            h_curr = _project_feasible(rng.random(n), target_sum)
            temp = 0.02
            no_improve = 0
            restarts_done += 1
        else:
            h_curr = best_h.copy()

        # propose mutation
        h_new = _mutate(h_curr.copy(), rng)

        # enforce feasibility (project only if numerical drift)
        # mutation keeps sum, but clip may be needed
        h_new = np.clip(h_new, 0.0, 1.0)

        new_score = _score(h_new, dx)

        # acceptance
        if new_score < best_score or rng.random() < math.exp((best_score - new_score) / max(temp, 1e-12)):
            best_h = h_new
            best_score = new_score
            no_improve = 0
        else:
            no_improve += 1

        # temperature decay
        temp = max(min_temp, temp * decay)
        iter_count += 1

    # Final projection to guarantee exact sum (tiny numerical error)
    best_h = _project_feasible(best_h, target_sum)

    c5_bound = _score(best_h, dx)
    return best_h, float(c5_bound), n
