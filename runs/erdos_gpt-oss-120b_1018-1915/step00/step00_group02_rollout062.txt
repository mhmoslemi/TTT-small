import time
import numpy as np

def _compute_c5(h):
    """Return the overlap constant for a discrete step function h."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_h(rng, n, target_sum):
    """Generate a random feasible h with entries in [0,1] and the prescribed sum."""
    # start from a Dirichlet-like sample
    x = rng.random(n)
    x = x / x.sum() * target_sum
    # enforce the box constraints by redistributing any overflow
    for _ in range(5):
        over = x > 1.0
        if not over.any():
            break
        excess = x[over] - 1.0
        total_excess = excess.sum()
        x[over] = 1.0
        under = x < 1.0
        if not under.any():
            break
        # distribute excess proportionally to the under‑filled entries
        x[under] += total_excess * (x[under] / x[under].sum())
    # final renormalisation to hit the exact sum (tiny numerical error)
    cur = x.sum()
    if cur != 0.0:
        x *= target_sum / cur
    return x

def _perturb(rng, h, step):
    """Propose a feasible neighbour by moving a small amount between two indices."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    delta = rng.uniform(-step, step)
    if delta > 0:
        # add to i, subtract from j
        max_add = 1.0 - h[i]
        max_sub = h[j]
        if delta > max_add or delta > max_sub:
            return None
        new = h.copy()
        new[i] += delta
        new[j] -= delta
    elif delta < 0:
        delta = -delta
        # add to j, subtract from i
        max_add = 1.0 - h[j]
        max_sub = h[i]
        if delta > max_add or delta > max_sub:
            return None
        new = h.copy()
        new[j] += delta
        new[i] -= delta
    else:
        return None
    return new
