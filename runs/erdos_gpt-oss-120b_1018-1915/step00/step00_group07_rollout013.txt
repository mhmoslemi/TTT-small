import time
import numpy as np

def _project_to_bounds(v, target_sum, tol=1e-10, max_iter=100):
    """
    Project vector v onto the set {x | 0<=x_i<=1, sum x_i = target_sum}.
    Uses a simple water-filling iterative scheme.
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        current = x.sum()
        diff = target_sum - current
        if abs(diff) <= tol:
            break
        # indices that are not at a bound and can receive/take mass
        free = (x > 0.0) & (x < 1.0)
        nfree = free.sum()
        if nfree == 0:
            # all at bounds, distribute uniformly and re‑clip
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
            continue
        delta = diff / nfree
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final tiny adjustment to hit the sum exactly (avoid rounding drift)
    residual = target_sum - x.sum()
    if abs(residual) > tol:
        # add residual to first non‑saturated entry
        idx = np.where((x > 0.0) & (x < 1.0))[0]
        if idx.size == 0:
            # all at bounds – distribute to any index safely
            idx = np.arange(len(x))
        x[idx[0]] = np.clip(x[idx[0]] + residual, 0.0, 1.0)
    return x

def _evaluate_c5(h, dx):
    """Return max overlap integral for step function h."""
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Differential‑Evolution optimisation for the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Best step‑function values (length n_points).
    c5_bound : float
        Corresponding upper bound on C5.
    n_points : int
        Number of discretisation points (length of h_values).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem data (either supplied via initial_h_values or default)
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)
        n_points = init.size
    except NameError:
        # fallback: 80 uniform points with half ones, half zeros (valid)
        n_points = 80
        init = np.concatenate([np.ones(n_points // 2), np.zeros(n_points // 2)])
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # integral = 1

    # ------------------------------------------------------------------
    # DE parameters
    # ------------------------------------------------------------------
    pop_size = max(30, n_points // 2)           # modest population
    F = 0.7                                      # mutation factor
    CR = 0.9                                     # crossover probability
    max_gen = 5000                               # will be cut by time

    # ------------------------------------------------------------------
    # Initialise population (include the provided initial vector)
    # ------------------------------------------------------------------
    population = rng.random((pop_size, n_points))
    population = np.apply_along_axis(_project_to_bounds, 1, population, target_sum)
    # replace first individual with the known feasible init (projected just in case)
    population[0] = _project_to_bounds(init, target_sum)

    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        scores[i] = _evaluate_c5(population[i], dx)

    best_idx = np.argmin(scores)
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # ------------------------------------------------------------------
    # Main DE loop
    # ------------------------------------------------------------------
    gen = 0
    while True:
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break
        if gen >= max_gen:
            break
        gen += 1

        for i in range(pop_size):
            # select three distinct others
            idxs = rng.choice(pop_size, 3, replace=False)
            while i in idxs:
                idxs = rng.choice(pop_size, 3, replace=False)
            r1, r2, r3 = idxs

            mutant = population[r1] + F * (population[r2] - population[r3])
            # binomial crossover
            cross_mask = rng.random(n_points) < CR
            # ensure at least one dimension from mutant
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, n_points)] = True
            trial = np.where(cross_mask, mutant, population[i])
            trial = _project_to_bounds(trial, target_sum)

            trial_score = _evaluate_c5(trial, dx)

            # Selection
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()

        # occasional shuffling to avoid stagnation
        if gen % 50 == 0:
            order = rng.permutation(pop_size)
            population = population[order]
            scores = scores[order]

    return best_h, float(best_score), int(n_points)
