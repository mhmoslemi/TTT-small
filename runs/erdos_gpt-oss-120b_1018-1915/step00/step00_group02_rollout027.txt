import numpy as np
import time

def _c5_of(h):
    """Compute C5 for discrete step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _project(h):
    """Clip to [0,1] and enforce sum(h)=len(h)/2 exactly."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative correction
    for _ in range(5):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        adj = (target - s) / n
        h = h + adj
        h = np.clip(h, 0.0, 1.0)
    # Final tiny correction if needed
    s = h.sum()
    if abs(s - target) > 1e-10:
        # distribute remaining error to a random feasible index
        idx = np.random.randint(0, n)
        delta = target - s
        h[idx] = np.clip(h[idx] + delta, 0.0, 1.0)
    return h
