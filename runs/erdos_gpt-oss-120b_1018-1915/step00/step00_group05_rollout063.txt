import time
import numpy as np
from scipy.optimize import minimize

def _c5_value(h, dx):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx for discrete h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def _project_feasible(h, target_sum):
    """Clip to [0,1] and then adjust uniformly to meet the exact sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target_sum
    if abs(diff) < 1e-12:
        return h
    # distribute the difference proportionally to the slack
    slack_up = 1.0 - h
    slack_down = h
    if diff > 0:  # need to decrease sum
        total_slack = np.sum(slack_down)
        if total_slack == 0:
            return h
        factor = diff / total_slack
        h -= factor * slack_down
    else:        # need to increase sum
        total_slack = np.sum(slack_up)
        if total_slack == 0:
            return h
        factor = -diff / total_slack
        h += factor * slack_up
    # final clip (tiny numerical errors)
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] (represented by n points) that
    minimizes C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # Determine number of points
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=float)
        n = init_h.size
    else:
        n = kwargs.get('n_points', 70)
        init_h = np.full(n, 0.5, dtype=float)  # feasible start (sum = n/2)

    # Ensure feasibility of the starting vector
    target_sum = n / 2.0
    init_h = _project_feasible(init_h, target_sum)

    dx = 2.0 / n
    best_h = init_h.copy()
    best_val = _c5_value(best_h, dx)

    # Simple hill‑climbing with pairwise mass transfers
    start_time = time.time()
    iters_without_improve = 0
    max_no_improve = 5000

    while time.time() - start_time < budget_s * 0.93:
        i, j = rng.choice(n, size=2, replace=False)
        # i will give, j will receive
        give = best_h[i]
        recv = best_h[j]
        if give == 0.0 or recv == 1.0:
            continue
        delta_max = min(give, 1.0 - recv)
        if delta_max <= 0.0:
            continue
        # Random fraction of the possible transfer
        delta = rng.random() * delta_max * 0.5  # move at most half the slack
        if delta == 0.0:
            continue
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _c5_value(cand, dx)
        if val < best_val - 1e-9:
            best_h = cand
            best_val = val
            iters_without_improve = 0
            continue
        iters_without_improve += 1
        # occasional random perturbation to escape plateaus
        if iters_without_improve >= max_no_improve:
            # restart from current best with small noise
            noise = rng.normal(scale=0.02, size=n)
            cand = best_h + noise
            cand = _project_feasible(cand, target_sum)
            val = _c5_value(cand, dx)
            if val < best_val:
                best_h = cand
                best_val = val
            iters_without_improve = 0

    # One round of constrained continuous optimization (SLSQP)
    # Using the best discrete solution as starting point
    bounds = [(0.0, 1.0)] * n
    eq_cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}

    def obj(x):
        return _c5_value(x, dx)

    try:
        res = minimize(
            obj,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=[eq_cons],
            options={'maxiter': 300, 'ftol': 1e-12, 'disp': False}
        )
        if res.success and res.fun < best_val:
            best_h = _project_feasible(res.x, target_sum)
            best_val = _c5_value(best_h, dx)
    except Exception:
        pass  # if optimizer fails, keep current best

    return best_h.tolist(), float(best_val), n
