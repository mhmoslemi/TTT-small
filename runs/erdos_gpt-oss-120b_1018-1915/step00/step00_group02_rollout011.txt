from numpy import correlate

# helpers are pre‑imported:
#   evaluate_erdos_solution(h) -> float
#   initial_h_values (may be undefined)

def _objective(h):
    """Return the C5 value for a step vector h."""
    return evaluate_erdos_solution(h)

def _max_shift_gradient(h):
    """Sub‑gradient of max_k ∫ h(x)(1‑h(x+k))dx at the current max shift."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h and (1-h)
    corr = correlate(h, 1.0 - h, mode='full') * dx
    s_max = int(np.argmax(corr))               # index in full array
    shift = s_max - (n - 1)                     # signed integer shift
    grad = np.zeros_like(h)

    # contribution where h is the left factor
    if 0 <= shift < n:
        grad[: n - shift] += 1.0 - h[shift:]
    elif -n < shift < 0:
        grad[-shift:] += 1.0 - h[: n + shift]

    # contribution where h appears in the right factor (inside the 1‑h)
    if 0 <= shift < n:
        grad[shift:] -= h[: n - shift]
    elif -n < shift < 0:
        grad[: n + shift] -= h[-shift:]

    return grad, corr.max()

def _project_feasible(h, target_sum):
    """Project onto [0,1]^n while keeping sum close to target_sum."""
    h = np.clip(h, 0.0, 1.0)
    # simple iterative affine‑box projection
    for _ in range(5):

        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _optimize(h0, budget, start_time):
