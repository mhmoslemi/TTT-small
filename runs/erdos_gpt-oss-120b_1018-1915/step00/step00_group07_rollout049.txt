import time
import numpy as np

def _repair(vec, target_sum):
    """Project a vector onto [0,1] with a prescribed sum (approx)."""
    v = np.clip(vec, 0.0, 1.0)
    s = v.sum()
    if s == 0.0:
        v[:] = target_sum / v.size
        return v
    # scale to hit the target sum
    v = v * (target_sum / s)
    v = np.clip(v, 0.0, 1.0)
    # final small adjustment on free variables
    s = v.sum()
    diff = target_sum - s
    if abs(diff) > 1e-12:
        mask = (v > 0.0) & (v < 1.0)
        if mask.any():
            v[mask] += diff / mask.sum()
            v = np.clip(v, 0.0, 1.0)
    return v


def _objective(h, dx):
    """Return max overlap integral for a candidate step function."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def run(seed=42, budget_s=1100, **kwargs):
    """
    Differential‑evolution search for a step function h on [0,2] with
    ∫ h = 1 that minimizes max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem dimensions
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None:
        h0 = np.asarray(init, dtype=float)
        n_points = h0.size
    else:
        n_points = 80
        h0 = rng.random(n_points)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because ∑h * dx = 1

    # ------------------------------------------------------------------
    # Population initialization
    # ------------------------------------------------------------------
    popsize = max(20, 5 * n_points // 10)   # modest size
    population = []
    fitness = []
    # ensure the parent (if any) is feasible
    h0 = _repair(h0, target_sum)
    population.append(h0)
    fitness.append(_objective(h0, dx))
    # fill rest of population randomly
    while len(population) < popsize:
        cand = rng.random(n_points)
        cand = _repair(cand, target_sum)
        population.append(cand)
        fitness.append(_objective(cand, dx))

    population = np.stack(population)          # shape (popsize, n_points)
    fitness = np.array(fitness)                # lower is better

    # keep track of the best ever seen
    best_idx = int(fitness.argmin())
    best_h = population[best_idx].copy()
    best_score = float(fitness[best_idx])

    # ------------------------------------------------------------------
    # DE parameters
    # ------------------------------------------------------------------
    F = 0.5            # mutation factor
    CR = 0.9           # crossover probability
    max_gen = 200

    # ------------------------------------------------------------------
    # Evolution loop
    # ------------------------------------------------------------------
    gen = 0
    while gen < max_gen and (time.time() - start_time) < budget_s:
        for i in range(popsize):
            # select three distinct individuals different from i
            idxs = list(range(popsize))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])

            # binomial crossover
            j_rand = rng.integers(0, n_points)
            cross_mask = rng.random(n_points) < CR
            cross_mask[j_rand] = True
            trial = np.where(cross_mask, mutant, population[i])

            # repair to satisfy constraints
            trial = _repair(trial, target_sum)

            # evaluate
            trial_score = _objective(trial, dx)

            # selection
            if trial_score <= fitness[i]:
                population[i] = trial
                fitness[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        gen += 1
        # optional early stop if budget low
        if (time.time() - start_time) > budget_s * 0.95:
            break

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return (best_h, best_score, n_points)
