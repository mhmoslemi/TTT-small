import time
import numpy as np

def _evaluate(h, dx):
    """Return the worst overlap C5 for a sampled step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _repair(h, target_sum):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    diff = target_sum - current
    if abs(diff) < 1e-12:
        return h
    # Distribute the remaining difference greedily
    for i in range(len(h)):
        if diff == 0.0:
            break
        if diff > 0:
            room = 1.0 - h[i]
            add = min(room, diff)
        else:
            room = h[i]
            add = max(-room, diff)
        h[i] += add
        diff -= add
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a low‑C5 step function on [0,2].

    Returns
    -------
    h_values : ndarray
        Sampled values of the best function found.
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of samples used (length of h_values).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine initial vector
    try:
        base_h = np.array(initial_h_values, dtype=float)  # provided by harness
        n = base_h.size
    except NameError:
        n = 70
        base_h = np.full(n, 0.5, dtype=float)  # uniform satisfies the integral
    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility of the starting point
    h = _repair(base_h, target_sum)
    best_h = h.copy()
    best_score = _evaluate(best_h, dx)

    # Simple simulated‑annealing / hill‑climbing loop
    temp = 0.05  # initial temperature for stochastic acceptance
    cooling = 0.9995
    min_temp = 1e-5

    # Pre‑compute index pairs for efficiency
    idx = np.arange(n)
    while time.time() - start_time < budget_s - 0.2:
        # Propose a move: transfer a small amount between two positions
        i, j = rng.choice(idx, size=2, replace=False)
        delta = rng.uniform(-0.07, 0.07)  # magnitude of transfer
        h_candidate = best_h.copy()
        h_candidate[i] += delta
        h_candidate[j] -= delta

        # Clip and fix the sum constraint
        h_candidate = _repair(h_candidate, target_sum)

        # Evaluate
        score = _evaluate(h_candidate, dx)

        # Acceptance rule
        if score < best_score:
            best_h, best_score = h_candidate, score
        else:
            if temp > min_temp and rng.random() < np.exp((best_score - score) / temp):
                best_h = h_candidate
        # Temperature schedule
        temp = max(min_temp, temp * cooling)

        # Occasionally try a random reshuffle to escape stagnation
        if rng.random() < 0.001:
            rnd = rng.random(n)
            rnd = _repair(rnd, target_sum)
            rnd_score = _evaluate(rnd, dx)
            if rnd_score < best_score:
                best_h, best_score = rnd, rnd_score

    # Final repair before returning
    best_h = _repair(best_h, target_sum)
    best_score = _evaluate(best_h, dx)

    return best_h, best_score, n
