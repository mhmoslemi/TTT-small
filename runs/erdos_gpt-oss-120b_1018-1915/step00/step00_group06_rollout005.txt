import time
import math
import random
import numpy as np

def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx using FFT correlation."""
    n = h.size
    dx = 2.0 / n
    # zero‑pad to length 2n for linear (non‑circular) correlation
    size = 2 * n
    fh = np.fft.rfft(h, n=size)
    fg = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(fh * np.conj(fg), n=size)
    # correlation length is 2n-1; take the first 2n-1 values
    corr = corr[: 2 * n - 1]
    return float(np.max(corr) * dx)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise C5.

    Returns
    -------
    h_values : list of float
        Optimised values of h on a uniform grid.
    c5_bound : float
        The achieved upper bound for C5.
    n_points : int
        Number of grid points (len(h_values)).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # Initial construction
    # ----------------------------------------------------------------------
    try:
        # provided by the harness if available
        h0 = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        # fallback: uniform 0.5 on a modest grid (≈100 points)
        n0 = 100
        h0 = np.full(n0, 0.5, dtype=np.float64)

    n = h0.size
    target_sum = n / 2.0
    # enforce exact integral constraint (might be off by rounding)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h0 *= target_sum / cur_sum
    h0 = np.clip(h0, 0.0, 1.0)

    # ----------------------------------------------------------------------
    # Simulated annealing / hill‑climbing loop
    # ----------------------------------------------------------------------
    h = h0.copy()
    best_h = h.copy()
    best_score = _compute_c5(best_h)

    start_time = time.time()
    max_time = budget_s * 0.95           # keep a safety margin
    max_temp = 0.08                      # initial temperature
    min_temp = 0.0005                    # final temperature

    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break

        # linear temperature schedule
        t_frac = elapsed / max_time
        temp = max_temp * (1.0 - t_frac) + min_temp * t_frac

        # --------------------------------------------------------------
        # propose a elementary move: transfer δ from i to j
        # --------------------------------------------------------------
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]

        # amount that can be moved while respecting bounds
        max_delta = min(hi, 1.0 - hj)
        if max_delta <= 0.0:
            continue

        # step size proportional to temperature, but never larger than max_delta
        delta = random.random() * max_delta * (0.1 + temp)

        # apply move
        h[i] = hi - delta
        h[j] = hj + delta

        # evaluate
        new_score = _compute_c5(h)

        # acceptance rule (Metropolis)
        accept = False
        if new_score < best_score:
            accept = True
        else:
            prob = math.exp(-(new_score - best_score) / (temp + 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert move
            h[i] = hi
            h[j] = hj

        it += 1
