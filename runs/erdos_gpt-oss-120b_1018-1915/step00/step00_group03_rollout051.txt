import time
import numpy as np

def _project_bounds_sum(v: np.ndarray, target: float, eps: float = 1e-12) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the sum == target (water‑filling)."""
    x = np.clip(v, 0.0, 1.0)
    while True:
        s = x.sum()
        delta = target - s
        if abs(delta) < 1e-9:
            break
        if delta > 0:
            free = np.where(x < 1.0 - eps)[0]
            if free.size == 0:
                break
            inc = delta / free.size
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
        else:
            free = np.where(x > eps)[0]
            if free.size == 0:
                break
            dec = (-delta) / free.size
            x[free] -= dec
            x = np.clip(x, 0.0, 1.0)
    return x


def _evaluate(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Stochastic local search for a low‑C5 step function.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Choose discretisation size (power of two, ≤ 1024 for speed)
    # ------------------------------------------------------------------
    n = 256                     # 256 points → dx = 2/256
    target_sum = n / 2.0        # integral = 1

    # ------------------------------------------------------------------
    # 2. Initialise from provided construction if available
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = globals()["initial_h_values"]
        m = init.size
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        h = rng.random(n)

    h = _project_bounds_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _evaluate(best_h)

    # ------------------------------------------------------------------
    # 3. Simple simulated‑annealing style optimisation
    # ------------------------------------------------------------------
    sigma = 0.25                         # initial mutation scale
    decay = 0.9995                       # exponential cooling
    max_iter = 50000
    iter_cnt = 0

    while (time.time() - start_time) < budget_s * 0.95 and iter_cnt < max_iter:
        iter_cnt += 1
        sigma *= decay

        # mutate a small random subset (≈1 % of coordinates)
        cand = best_h.copy()
        k = max(1, n // 100)
        idx = rng.integers(0, n, size=k)
        cand[idx] += rng.normal(0.0, sigma, size=k)

        cand = _project_bounds_sum(cand, target_sum)
        cand_c5 = _evaluate(cand)

        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand
            sigma = 0.25                     # reset temperature on improvement

        # occasional full random restart to escape stagnation
        if iter_cnt % 2000 == 0:
            rnd = rng.random(n)
            rnd = _project_bounds_sum(rnd, target_sum)
            rnd_c5 = _evaluate(rnd)
            if rnd_c5 < best_c5:
                best_c5 = rnd_c5
                best_h = rnd

    # ------------------------------------------------------------------
    # 4. Return the best feasible construction found
    # ------------------------------------------------------------------
    final_c5 = _evaluate(best_h)
    return (best_h, final_c5, n)
