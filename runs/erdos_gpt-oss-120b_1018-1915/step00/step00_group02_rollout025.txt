import numpy as np
import time
import math
from scipy.optimize import minimize

def _compute_c5(h):
    """Return the C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def _project_h(h, target_sum, eps=1e-12, max_iter=100):
    """
    Project h onto the box [0,1] and enforce sum(h)=target_sum.
    Uses an iterative redistribution of the slack on free coordinates.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) <= eps:
            break
        free = (h > 0.0 + eps) & (h < 1.0 - eps)
        nfree = free.sum()
        if nfree == 0:
            # all at bounds – distribute uniformly (may violate bounds a bit)
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / nfree
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def _initialize_random(n, rng):
    """Generate a random feasible h with values in [0,1] and correct sum."""

    h = np.full(n, 0.5, dtype=np.float64)
    h += rng.normal(scale=0.2, size=n)
    target = n / 2.0
    h = _project_h(h, target)
    return h
