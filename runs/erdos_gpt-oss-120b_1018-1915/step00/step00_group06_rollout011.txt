import numpy as np
import time

def _project_feasible(h, desired_sum, eps=1e-10, max_iter=100):
    """Clip h to [0,1] and adjust to attain the desired L1 sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = h.sum()
        diff = desired_sum - cur_sum
        if abs(diff) <= eps:
            break
        # free indices are those not at a bound
        free = np.where((h > eps) & (h < 1 - eps))[0]
        if free.size == 0:
            # distribute diff uniformly and re‑clip
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        inc = diff / free.size
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    return h

def _evaluate_c5(h):
    """Return the C5 value for a discrete step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / h.size) * np.max(corr)

def _initial_interpolate(target_n):
    """Interpolate the provided initial_h_values to target_n points."""
    src = np.array(initial_h_values, dtype=float)
    src_n = src.size
    xs_src = np.linspace(0.0, 2.0, src_n, endpoint=False)
    xs_tgt = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h = np.interp(xs_tgt, xs_src, src)
    # enforce feasibility
    desired = target_n / 2.0
    h = _project_feasible(h, desired)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # -------------------------------------------------------------
    # Choose a moderate resolution; 300 points balances speed & quality.
    # -------------------------------------------------------------
    n_points = 300
    h = _initial_interpolate(n_points)
    desired_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)

    # Simple hill‑climbing with pairwise swaps that preserve the sum.
    # Each step perturbs two coordinates by a small amount.
    max_step = 0.02          # maximal absolute change per move
    check_interval = 2000   # frequency of time‑check to avoid overhead

    iteration = 0
    while True:
        if iteration % check_interval == 0:
            if time.time() - start_time > budget_s - 0.5:
                break

        # propose a move
        i, j = rng.choice(n_points, size=2, replace=False)
        delta = (rng.random() - 0.5) * max_step  # in [-max_step/2, max_step/2]

        h_new = h.copy()
        # apply delta to i, opposite to j
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)

        # enforce exact sum constraint (project)
        h_new = _project_feasible(h_new, desired_sum)

        c5_new = _evaluate_c5(h_new)

        # accept only if strictly better (greedy)
        if c5_new < best_c5:

            h = h_new  # continue from the improved point

                h = _initial_interpolate(n_points)
