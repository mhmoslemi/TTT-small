def project_bounds_sum(v, target):
    # clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # water-filling
    eps = 1e-12
    while True:
        total = np.sum(v)

            # cannot adjust, break

        delta = diff / nfree
        v[free] -= delta
        # clip again
        v = np.clip(v, 0.0, 1.0)
    return v

import time
import numpy as np

def _project_bounds_sum(v, target):
    """Clip to [0,1] and adjust to make sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    eps = 1e-12
    # water‑filling loop
    for _ in range(1000):
        total = v.sum()
        diff = total - target
        if abs(diff) < eps:
            break
        free = (v > eps) & (v < 1.0 - eps)
        nfree = free.sum()
        if nfree == 0:
            # all at bounds, cannot fix further
            break
        delta = diff / nfree
        v[free] -= delta
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_score(h):
    """Return the C5 value for a step vector h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 constant.

    Returns

    h_values : np.ndarray
        Optimized step values (length n_points).

        Achieved C5 value.

        Number of discretisation points.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)
