import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    """Return the C5 value for a step function sampled by h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * corr.max()

def _repair(h, target_sum):
    """Clip to [0,1] and rescale to hit the required integral."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        h[:] = target_sum / h.size
        return h
    # simple iterative scaling + clipping
    for _ in range(5):
        factor = target_sum / s
        h *= factor
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
    return h

def _interpolate(h, new_n):
    """Linearly interpolate h to a different resolution."""
    old_n = h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    # enforce the integral constraint after interpolation
    target = new_n / 2.0
    h_new = _repair(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimizes C5."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s * 0.95  # keep a safety margin

    # ------------------------------------------------------------------
    # 1. Obtain an initial construction
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, copy=False, dtype=np.float64)

    except NameError:
        # no parent supplied – start from a uniform 0.5 function
        n0 = 120  # a modest size that fits easily in the budget
        init_h = np.full(n0, 0.5, dtype=np.float64)
        init_h = _repair(init_h, n0 / 2.0)

    best_h = init_h.copy()
    best_score = _evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # 2. Optionally upscale to a higher resolution (more degrees of freedom)
    # ------------------------------------------------------------------
    # we try a few resolutions that are multiples of the base size
    candidate_ns = [n0, int(n0 * 1.5), int(n0 * 2), int(n0 * 3)]
    candidate_ns = [n for n in candidate_ns if n <= 750 and n > 0]

    for n in candidate_ns:
        if time.time() - start_time > time_limit:
            break
        if n != best_h.size:
            h = _interpolate(best_h, n)
        else:
            h = best_h.copy()

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------
        target_sum = n / 2.0
        bounds = [(0.0, 1.0) for _ in range(n)]
        cons = {"type": "eq", "fun": lambda x, ts=target_sum: x.sum() - ts}

        def obj(x):
            return _evaluate_c5(x)

        # SLSQP may be expensive; limit iterations
        try:
            res = minimize(
                obj,
                h,
                method="SLSQP",
