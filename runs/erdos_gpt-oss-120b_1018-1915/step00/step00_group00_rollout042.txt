import time, random, math
import numpy as np

def _c5_of(h):
    """Compute the C5 value for a step function h sampled over [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation with zero padding (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_sum_bounds(x, target):
    """Adjust a vector x (0<=x<=1) so that its sum equals target.
    Simple linear correction on a random index; assumes feasible."""
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Find indices that can absorb the diff without breaking bounds
    if diff > 0:
        candidates = np.where(x < 1.0)[0]
    else:
        candidates = np.where(x > 0.0)[0]
    if candidates.size == 0:
        return x
    idx = np.random.choice(candidates)
    adj = np.clip(diff, -x[idx], 1.0 - x[idx])
    x[idx] += adj
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_score = math.inf
    best_n = None

    # Try a few resolutions; start with the size of any provided parent.
    candidate_sizes = [45, 90, 180]
    # If an initial construction exists, use its length first.
    if 'initial_h_values' in globals():
        try:
            init_len = len(initial_h_values)
            if init_len in candidate_sizes:
                candidate_sizes = [init_len] + [s for s in candidate_sizes if s != init_len]
        except Exception:
            pass

    for n in candidate_sizes:
        target_sum = n / 2.0               # because ∫h = 1  ⇒  sum(h)*dx = 1
        # ---- initialise h -------------------------------------------------
        if 'initial_h_values' in globals() and n == len(initial_h_values):
            h = np.array(initial_h_values, dtype=float).copy()
            # Clip to [0,1] and re‑project sum just in case.
            np.clip(h, 0.0, 1.0, out=h)
            h = _project_sum_bounds(h, target_sum)
        else:
            # Start from a random feasible point: half 1’s, half 0’s, with jitter.
            h = np.zeros(n, dtype=float)
            ones = n // 2
            h[np.random.choice(n, ones, replace=False)] = 1.0
            jitter = np.random.uniform(-0.08, 0.08, size=n)
            h = np.clip(h + jitter, 0.0, 1.0)
            h = _project_sum_bounds(h, target_sum)

        current_score = _c5_of(h)
        if current_score < best_score:
            best_h, best_score, best_n = h.copy(), current_score, n

        # ---- simulated‑annealing loop ------------------------------------
        T = 1.0                     # initial temperature
        iteration = 0
        while True:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 4.0:        # reserve a few seconds for cleanup
                break

            # geometric cooling
            T = max(1e-5, T * 0.9995)

            # propose a transfer of mass between two coordinates
            i, j = np.random.choice(n, 2, replace=False)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer <= 1e-12:
                continue
            eps = np.random.uniform(0.0, min(0.05, max_transfer))
            h_new = h.copy()
            h_new[i] -= eps
            h_new[j] += eps

            new_score = _c5_of(h_new)
            delta = new_score - current_score
            accept = delta < 0.0 or random.random() < math.exp(-delta / T)
            if accept:
                h, current_score = h_new, new_score
                if new_score < best_score:
                    best_h, best_score, best_n = h.copy(), new_score, n

            iteration += 1
            # occasional random restart from the current best to diversify
            if iteration % 7000 == 0:
                h = best_h.copy()
                current_score = best_score

        # early exit if we already beat the target bound
        if best_score <= 0.3808:
            break

    # Return plain Python list for compatibility
    return (best_h.tolist(), float(best_score), int(best_n))
