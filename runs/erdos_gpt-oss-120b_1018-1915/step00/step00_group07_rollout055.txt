import time
import numpy as np

def _repair_vector(vec, target_sum):
    """Project vec onto the box [0,1] and affine hyperplane sum=target_sum."""
    v = np.clip(vec, 0.0, 1.0)
    # Simple iterative redistribution
    for _ in range(5):
        s = v.sum()
        if abs(s - target_sum) < 1e-10:
            break
        diff = target_sum - s
        # indices that are not at bounds can receive adjustment
        free = (v > 0.0) & (v < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds, distribute uniformly then clip again
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / cnt
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final tiny scaling if still off (rare)
    s = v.sum()
    if abs(s - target_sum) > 1e-8:
        v *= target_sum / s
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h, dx):
    """Compute C5 for a step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 80
        h0 = np.full(n_points, 0.5, dtype=float)  # feasible uniform start

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure the initial vector respects constraints
    h0 = _repair_vector(h0, target_sum)

    # Parameters for Differential Evolution
    pop_size = max(30, int(0.5 * n_points))   # modest population
    F = 0.5
    CR = 0.9
    max_gen = 2000  # generous upper bound, will stop by time

    # Initialise population
    population = np.empty((pop_size, n_points), dtype=float)
    scores = np.empty(pop_size, dtype=float)

    # First individual is the provided (or uniform) feasible vector
    population[0] = h0
    scores[0] = _c5_value(h0, dx)

    for i in range(1, pop_size):
        rand_vec = rng.random(n_points)
        population[i] = _repair_vector(rand_vec, target_sum)
        scores[i] = _c5_value(population[i], dx)

    best_idx = np.argmin(scores)
    best_vec = population[best_idx].copy()
    best_score = scores[best_idx]

    gen = 0
    while gen < max_gen:
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break
        for i in range(pop_size):
            # mutation: select three distinct indices different from i
            idxs = rng.choice(pop_size, 3, replace=False)

                idxs = rng.choice(pop_size, 3, replace=False)
            r1, r2, r3 = idxs
            mutant = population[r1] + F * (population[r2] - population[r3])
