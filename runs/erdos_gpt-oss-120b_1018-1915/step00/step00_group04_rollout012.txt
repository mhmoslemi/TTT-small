import time
import numpy as np

def _compute_c5(h):
    """Return the Erdős C5 value for a candidate step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _repair(h, target_sum):
    """Project h onto the feasible box [0,1]^n with exact sum=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    n = h.shape[0]
    # first uniform shift
    diff = target_sum - h.sum()
    h += diff / n
    h = np.clip(h, 0.0, 1.0)

    # iterative refinement on free variables
    for _ in range(5):
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        delta = target_sum - cur
        free = (h > 0.0) & (h < 1.0)
        if not free.any():
            break
        h[free] += delta / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimized vector sampled uniformly on [0,2].
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of samples (len(h_values)).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Determine problem size and initial vector
    # ------------------------------------------------------------------
    try:
        init = initial_h_values  # supplied by harness if available
        n = int(init.shape[0])
        h0 = np.array(init, dtype=float)
    except Exception:
        # fallback: modest size, balanced step (half 1, half 0)
        n = 84
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        h0 = _repair(h0, n / 2)

    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # 2. Genetic algorithm parameters
    # ------------------------------------------------------------------
    pop_size = max(10, n // 4)          # modest population
    elite_frac = 0.1
    elite_cnt = max(1, int(pop_size * elite_frac))
    max_gen = 10000                     # safety cap
    sigma0 = 0.12                       # initial mutation std
    sigma_min = 0.005
    decay_rate = 0.995                  # per generation decay

    # ------------------------------------------------------------------
    # 3. Initialize population
    # ------------------------------------------------------------------
    population = np.empty((pop_size, n), dtype=float)
    population[0] = _repair(h0.copy(), target_sum)
    for i in range(1, pop_size):
        rand_vec = np.random.rand(n)
        population[i] = _repair(rand_vec, target_sum)

    # evaluate initial fitness
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = _compute_c5(population[i])

    best_idx = int(np.argmin(fitness))
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    sigma = sigma0

    # ------------------------------------------------------------------
    # 4. Main evolution loop
    # ------------------------------------------------------------------
    gen = 0
    while time.time() - start_time < budget_s * 0.95 and gen < max_gen:
        gen += 1

        # elite preservation
        elite_idx = np.argsort(fitness)[:elite_cnt]
        new_pop = population[elite_idx].copy()

        # fill the rest with mutated offspring
        while new_pop.shape[0] < pop_size:
            # tournament selection
            participants = np.random.choice(pop_size, size=3, replace=False)
            parent_idx = participants[np.argmin(fitness[participants])]
            parent = population[parent_idx]

            # mutation
            child = parent + sigma * np.random.randn(n)
            child = _repair(child, target_sum)

            new_pop = np.vstack((new_pop, child))

        # truncate in case we overshoot
        if new_pop.shape[0] > pop_size:
            new_pop = new_pop[:pop_size]

        # evaluate new generation
        new_fitness = np.empty(pop_size, dtype=float)
        for i in range(pop_size):
            new_fitness[i] = _compute_c5(new_pop[i])

        # update best
        cur_best_idx = int(np.argmin(new_fitness))
        cur_best_c5 = new_fitness[cur_best_idx]
        if cur_best_c5 < best_c5:
            best_c5 = float(cur_best_c5)
            best_h = new_pop[cur_best_idx].copy()

        # replace population
        population = new_pop
        fitness = new_fitness

        # decay mutation strength
        sigma = max(sigma_min, sigma * decay_rate)

        # early exit if we already beat target comfortably
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # 5. Return result
    # ------------------------------------------------------------------
    return best_h, float(best_c5), int(n)
