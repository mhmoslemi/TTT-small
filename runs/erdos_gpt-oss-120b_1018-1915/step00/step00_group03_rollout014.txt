import time
import numpy as np

# The harness provides these globals when available
# evaluate_erdos_solution(h)  -> returns the C5 value for a given h
# initial_h_values            -> optional starting array

def _project_to_constraints(h, target_sum, max_iter=15):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # direction of adjustment
        if diff > 0:
            free = h < 1.0
        else:
            free = h > 0.0
        if not np.any(free):
            break
        delta = diff / free.sum()
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h


def _evaluate(h, dx):
    """Return max overlap C5 for the discretised function."""
    # Using full correlation; the result length is 2*n-1
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _upscale(h, new_n):
    """Linear interpolation of a step function to a larger grid."""
    old_n = h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimizes
    C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, best_C5, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine initial resolution and starting vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
        n = h.size
    else:
        n = 140                     # a moderate starting size (<1k)
        h = rng.random(n)          # random start in [0,1)
    dx = 2.0 / n
    target_sum = n / 2.0           # because ∫ h = 1  =>  sum(h)*dx = 1

    h = _project_to_constraints(h, target_sum)
    best_h = h.copy()
    best_score = _evaluate(best_h, dx)

    # ------------------------------------------------------------------
    # 2. Core simulated‑annealing / stochastic descent loop
    # ------------------------------------------------------------------
    # Adaptive step size schedule
    sigma = 0.15
    sigma_min = 0.003
    no_improve = 0

    while True:
        now = time.time()
        if now - start_time > budget_s - 2.0:   # keep a small safety margin
            break

        # propose a new candidate
        cand = best_h + sigma * rng.standard_normal(n)
        cand = _project_to_constraints(cand, target_sum)

        score = _evaluate(cand, dx)

        if score < best_score:
            best_h, best_score = cand, score
            sigma = min(sigma * 1.1, 0.4)   # slightly enlarge step after improvement
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 200:
                sigma = max(sigma * 0.7, sigma_min)
                no_improve = 0

        # occasional random restart to escape flat regions
        if rng.random() < 0.002:

            cand = _project_to_constraints(cand, target_sum)
            cand_score = _evaluate(cand, dx)
            if cand_score < best_score:
                best_h, best_score = cand, cand_score
                sigma = 0.15

            best_score = _evaluate(best_h, dx)

            no_improve = 0

    # ------------------------------------------------------------------
    # 4. Return the final best construction
    # ------------------------------------------------------------------
    return best_h, float(best_score), n
