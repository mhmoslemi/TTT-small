import time
import numpy as np

def _c5_cost(h, dx):
    """Return max overlap integral for discrete h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points).
    Uses a simple stochastic local search to lower the max overlap.
    """
    rnd = np.random.RandomState(seed)
    start_time = time.time()

    # Determine initial discretisation
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.shape[0]
    else:
        n = 200                     # fallback size
        h = np.full(n, 0.5, dtype=float)   # sum = n/2 satisfies the integral
    dx = 2.0 / n

    # Ensure feasibility (clipping and normalising)
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if not np.isclose(total, n / 2.0):
        h *= (n / 2.0) / total

    best_h = h.copy()
    best_cost = _c5_cost(best_h, dx)

    # Parameters for the stochastic search
    max_iters = 200000
    step_initial = 0.05
    step_final = 0.001
    schedule_len = max_iters // 2

    for it in range(max_iters):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        step = step_initial - (step_initial - step_final) * min(1.0, it / schedule_len)

        i, j = rnd.choice(n, size=2, replace=False)

        # only move if source has room to give and target can receive

            continue
