import time
import numpy as np

def _c5_score(h, dx):
    """Return max overlap integral for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_swap_move(h, eps):
    """Propose a feasible move that keeps sum(h) constant and respects bounds."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # compute admissible delta range
    max_pos = min(eps, 1.0 - h[i], h[j])          # add to i, subtract from j
    max_neg = min(eps, h[i], 1.0 - h[j])          # subtract from i, add to j
    if max_pos <= 0 and max_neg <= 0:
        return None, None, None
    if max_pos > 0 and max_neg > 0:
        delta = (np.random.rand() * (max_pos + max_neg)) - max_neg
    elif max_pos > 0:
        delta = np.random.rand() * max_pos
    else:
        delta = -np.random.rand() * max_neg
    # apply move
    h[i] += delta
    h[j] -= delta
    return i, j, delta

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimised sample values (length n_points).
    c5_bound : float
        Corresponding C5 bound (lower is better).
    n_points : int
        Number of discretisation points.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Initialise h
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # default: 80 points, uniform 0.5 (satisfies integral constraint)
        n_points = 80
        h = np.full(n_points, 0.5, dtype=float)

    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure exact integral constraint (tiny numeric drift possible)
    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-12:
        h *= target_sum / current_sum
        np.clip(h, 0.0, 1.0, out=h)

    # ----------------------------------------------------------------------
    # Simple stochastic improvement (pairwise swaps)
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_score = _c5_score(best_h, dx)

    eps = 0.12            # initial maximal perturbation
    decay = 0.9995        # exponential decay of eps
    no_improve = 0
    max_no_improve = 2000

    while time.time() - start_time < budget_s * 0.95:
        move = _random_swap_move(h, eps)
        if move[0] is None:
            continue
        i, j, delta = move
        score = _c5_score(h, dx)

        if score < best_score:
            best_score = score
            best_h = h.copy()
            no_improve = 0
        else:
            # revert the move
            h[i] -= delta
            h[j] += delta
            no_improve += 1

        # occasional larger random refresh if stuck
        if no_improve >= max_no_improve:
            # perform a few random resets preserving sum
            for _ in range(5):
                a, b = np.random.choice(n, 2, replace=False)
                delta_big = np.random.uniform(-0.3, 0.3)
                # clip to keep feasibility

                    h[a] += delta_big

            no_improve = 0

    return best_h, best_score, n
