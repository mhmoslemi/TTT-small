import numpy as np
import time

def _project_to_constraints(vec, target_sum, eps=1e-9, max_iter=20):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target_sum
        if abs(diff) <= eps:
            break
        # redistribute the difference proportionally on the interior (0,1) entries
        interior = (v > 0.0) & (v < 1.0)
        if not np.any(interior):
            # all at bounds, just scale globally and re‑clip
            v = np.clip(v * (target_sum / s), 0.0, 1.0)
            continue
        # compute adjustment per interior element
        adj = diff / interior.sum()
        v[interior] -= adj
        v = np.clip(v, 0.0, 1.0)
    # final tiny mass‑correction if needed
    residual = target_sum - v.sum()
    if abs(residual) > eps:
        # add residual to first non‑full entry
        idx = np.where(v < 1.0)[0]
        if idx.size:
            v[idx[0]] = np.clip(v[idx[0]] + residual, 0.0, 1.0)
    return v

def _c5_value(h):
    """Compute the discrete C5 for a step vector h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * (2.0 / h.size)
