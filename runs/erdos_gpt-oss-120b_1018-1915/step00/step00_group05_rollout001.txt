import time, numpy as np
from scipy.optimize import minimize, differential_evolution

def _project(h, target, eps=1e-9, max_iter=10):
    """Clip to [0,1] and enforce sum == target (within eps)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target - s
        if abs(diff) <= eps:
            break
        # simple proportional scaling
        scale = target / (s if s != 0 else 1.0)
        h = np.clip(h * scale, 0.0, 1.0)
    return h

def _objective(h, dx):
    """Maximum overlap integral for a given step function h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        n = kwargs.get('n_points', 70)
        h0 = rng.random(n)
    n_points = h0.size
    dx = 2.0 / n_points
    sum_target = n_points / 2.0

    # Make the initial vector feasible
    h0 = _project(h0, sum_target)

    best_h = h0.copy()
    best_val = _objective(best_h, dx)

    # ------------------------------------------------------------------
    # Equality constraint for SLSQP
    # ------------------------------------------------------------------
    cons = ({'type': 'eq',
             'fun': lambda x: np.sum(x) - sum_target},)

    bounds = [(0.0, 1.0)] * n_points

    # ------------------------------------------------------------------
    # Local refinement (SLSQP)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        opts = {'maxiter': 800, 'ftol': 1e-12, 'disp': False}
        res = minimize(lambda x: _objective(x, dx),
                       best_h,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=cons,
                       options=opts)
        h_loc = _project(res.x, sum_target)
        val_loc = _objective(h_loc, dx)
        if val_loc < best_val:
            best_h, best_val = h_loc, val_loc

    # ------------------------------------------------------------------
    # Global search (Differential Evolution) – penalised equality
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 15.0:
        # Penalty weight large enough to force the sum constraint
        penalty = 1e5

        def de_obj(x):
            s = x.sum()
            pen = penalty * (s - sum_target) ** 2
            return _objective(x, dx) + pen

        de_bounds = bounds
        # Very modest population to stay within time
        popsize = max(5, n_points // 15)
        maxiter = max(5, int(remaining // 0.3))  # crude time → iteration estimate
        result = differential_evolution(de_obj,
                                        de_bounds,
                                        strategy='best1bin',
                                        maxiter=maxiter,
                                        popsize=popsize,
                                        tol=1e-6,
                                        mutation=(0.5, 1.0),
                                        recombination=0.7,
                                        seed=rng,
                                        polish=False,
                                        disp=False)
        h_de = _project(result.x, sum_target)
        val_de = _objective(h_de, dx)

    # ------------------------------------------------------------------
    # Final projection and return
    # ------------------------------------------------------------------
    best_h = _project(best_h, sum_target)
    best_val = _objective(best_h, dx)

    return best_h, float(best_val), int(n_points)
