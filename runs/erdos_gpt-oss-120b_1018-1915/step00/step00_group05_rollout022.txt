import time
import numpy as np

# Pre‑imported helpers (evaluate_erdos_solution, initial_h_values) are available in the environment.

def _project_box_sum(v, desired_sum, low=0.0, high=1.0, iters=60):
    """Project vector v onto the box [low,high] while forcing sum(v)=desired_sum via bisection."""
    v = np.clip(v, low, high)
    total = v.sum()
    if abs(total - desired_sum) < 1e-12:
        return v
    # binary search for λ so that sum(clip(v+λ)) = desired_sum
    lo, hi = -1.0, 1.0
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, low, high)
        s = w.sum()
        if s > desired_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, low, high)


def _fitness(h, dx):
    """C5 value for a candidate step function h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx


def _initialize_population(n_points, desired_sum, seed, init_vec=None, pop_size=50):
    """Create initial population respecting the constraints."""
    rng = np.random.default_rng(seed)
    pop = []
    if init_vec is not None:
        pop.append(_project_box_sum(init_vec.copy(), desired_sum))
    # a few perturbed copies of the initial vector (if any)
    for _ in range(4):
        if init_vec is not None:
            pert = init_vec + rng.uniform(-0.1, 0.1, n_points)
            pop.append(_project_box_sum(pert, desired_sum))
    # fill the rest with random feasible vectors
    while len(pop) < pop_size:
        cand = rng.random(n_points)
        pop.append(_project_box_sum(cand, desired_sum))
    return np.stack(pop[:pop_size])


def _crossover(p1, p2, rng):
    """Uniform crossover."""
    mask = rng.random(p1.shape) < 0.5
    child = np.where(mask, p1, p2)
    return child


def _mutate(ind, rng, mutation_rate=0.1, magnitude=0.15):
    """Additive Gaussian mutation on a subset of coordinates."""
    child = ind.copy()
    mutate_mask = rng.random(ind.shape) < mutation_rate
    noise = rng.normal(scale=magnitude, size=ind.shape)
    child[mutate_mask] += noise[mutate_mask]
    return child

    """

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Grid and constraints
    # ------------------------------------------------------------------
    n_points = kwargs.get('n_points')
    if n_points is None:
        # try to infer from the provided initial vector
        try:

    init_vec = None
    try:

    population = _initialize_population(n_points, desired_sum, seed, init_vec, pop_size)
    fitness_vals = np.array([_fitness(ind, dx) for ind in population])

    # Keep track of the best solution seen so far
    best_idx = np.argmin(fitness_vals)
    best_h = population[best_idx].copy()
    best_f = fitness_vals[best_idx]

        new_fit.append(best_f)
