import time, math, random
import numpy as np

def _evaluate(h):
    """Return the C5 value for a step vector h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _repair(h, target_sum):
    """Clip to [0,1] and rescale to hit the exact sum (within tolerance)."""
    np.clip(h, 0.0, 1.0, out=h)
    s = h.sum()
    if s == 0:
        # degenerate, fall back to uniform 0.5
        h.fill(0.5)
        return
    # simple proportional scaling; after scaling we may violate bounds slightly,
    # so iterate a tiny bit.
    for _ in range(3):
        scale = target_sum / h.sum()
        h *= scale
        np.clip(h, 0.0, 1.0, out=h)
    # final tiny adjustment to hit the sum exactly (distributed uniformly)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff over all entries that are not at a hard bound
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
        else:
            # fallback to uniform
            h.fill(target_sum / len(h))

def _upscale(h):
    """Double the resolution by linear interpolation and re‑repair."""
    n = len(h)
    new_n = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    _repair(h_new, new_n / 2.0)
    return h_new
