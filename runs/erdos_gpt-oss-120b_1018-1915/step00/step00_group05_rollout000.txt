import time
import numpy as np

def _project_to_bounds_and_sum(v, target_sum, eps=1e-12, max_iter=100):
    """
    Project vector v onto the box [0,1]^n intersecting the hyperplane sum = target_sum.
    Uses simple water-filling iterations.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = v.sum()
        diff = cur_sum - target_sum
        if abs(diff) <= eps:
            break
        if diff > 0:
            # need to decrease the sum: reduce from elements that are >0
            mask = v > 0.0
            if not np.any(mask):
                break
            reduction = diff / mask.sum()
            v[mask] -= reduction
        else:
            # need to increase the sum: add to elements that are <1
            mask = v < 1.0
            if not np.any(mask):
                break
            addition = (-diff) / mask.sum()
            v[mask] += addition
        v = np.clip(v, 0.0, 1.0)
    # final clip in case of tiny numerical overshoot
    v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    """Discrete C5 for a step vector h on [0,2] with uniform spacing."""
    n = h.shape[0]
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a tuple (h_values, c5_bound, n_points).

    h_values : ndarray of shape (n_points,) satisfying the constraints.
    c5_bound : the achieved C5 value (the objective to be minimized).
    n_points : number of discretization points used.
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals() and initial_h_values is not None:
        n_points = int(initial_h_values.shape[0])
        h_current = np.array(initial_h_values, dtype=float)
        # enforce exact sum constraint
        h_current = _project_to_bounds_and_sum(h_current, n_points / 2.0)
    else:
        n_points = kwargs.get("n_points", 70)
        # start from a uniform distribution that satisfies sum constraint
        h_current = rng.random(n_points)
        h_current = _project_to_bounds_and_sum(h_current, n_points / 2.0)

    best_h = h_current.copy()
    best_val = _c5_value(best_h)

    start_time = time.time()
    time_limit = budget_s * 0.95  # keep a safety margin
    # simple annealing schedule
    init_step = 0.2
    min_step = 0.005

    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break
        # decay step size
        progress = elapsed / time_limit
        step_size = init_step * (1 - progress) + min_step * progress

        # propose a random perturbation respecting bounds
        proposal = best_h + rng.normal(scale=step_size, size=n_points)
        proposal = _project_to_bounds_and_sum(proposal, n_points / 2.0)

        val = _c5_value(proposal)
        if val < best_val - 1e-9:
            best_val = val
            best_h = proposal

        # occasional random restart to avoid stagnation
        if iteration % 2000 == 0 and iteration > 0:
            if rng.random() < 0.2:
                cand = rng.random(n_points)
                cand = _project_to_bounds_and_sum(cand, n_points / 2.0)
                cand_val = _c5_value(cand)
                if cand_val < best_val:
                    best_val = cand_val
                    best_h = cand

        iteration += 1

    # Final safety projection (should already be satisfied)
    best_h = _project_to_bounds_and_sum(best_h, n_points / 2.0)

    return best_h, best_val, n_points
