import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Simulated‑annealing search for a step function h on [0,2] with
    ∫h = 1 that minimizes  C₅ = max_k ∫ h(x)(1‑h(x+k)) dx .
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # --- initialise h -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)   # provided by parent, if any
        n = h0.size
    except NameError:
        n = 45
        h0 = np.full(n, 0.5, dtype=float)               # uniform feasible start

    target_sum = n / 2.0          # because ∫_0^2 h = 1  and  dx = 2/n
    # enforce exact sum while staying in [0,1]
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h0 *= target_sum / cur_sum
        h0 = np.clip(h0, 0.0, 1.0)
        diff = target_sum - h0.sum()
        h0[0] += diff
        h0 = np.clip(h0, 0.0, 1.0)

    dx = 2.0 / n

    def c5_of(arr):
        """compute C₅ for a given discretised h."""
        corr = np.correlate(arr, 1.0 - arr, mode='full')
        return corr.max() * dx

    # current state and best record
    cur_h = h0.copy()
    cur_c5 = c5_of(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # --- simulated annealing parameters -------------------------------
    T = 1.0                     # initial temperature
    cooling = 0.9995           # multiplicative cooling per iteration
    min_T = 1e-6               # floor to avoid division by zero
    iter_cnt = 0

    # main loop, respecting the time budget
    while time.time() - start_time < budget_s * 0.95:
        # propose a feasible move: transfer δ from i to j
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = cur_h[i], cur_h[j]

        # admissible interval for δ (cur_h[i] - δ, cur_h[j] + δ stay in [0,1])
        lo = max(-hj, hi - 1.0)
        hi_bound = min(hi, 1.0 - hj)
        if lo >= hi_bound:                     # no feasible transfer, skip
            continue
        delta = np.random.uniform(lo, hi_bound)

        new_h = cur_h.copy()
        new_h[i] = hi - delta
        new_h[j] = hj + delta

        new_c5 = c5_of(new_h)

        # Metropolis acceptance
        if new_c5 < cur_c5:
            accept = True
        else:
            prob = np.exp(-(new_c5 - cur_c5) / max(T, min_T))
            accept = np.random.rand() < prob

        if accept:
            cur_h, cur_c5 = new_h, new_c5
            if cur_c5 < best_c5:
                best_h, best_c5 = cur_h.copy(), cur_c5

        # temperature schedule
        T *= cooling
        if T < min_T:
            T = min_T

        iter_cnt += 1

        # occasional random shake‑up to escape stagnation
        if iter_cnt % 5000 == 0:
            perturb = np.random.normal(scale=0.02, size=n)
            cand = cur_h + perturb
            cand = np.clip(cand, 0.0, 1.0)
            # enforce exact integral again
            diff = target_sum - cand.sum()
            cand[0] += diff
            cand = np.clip(cand, 0.0, 1.0)
            cand_c5 = c5_of(cand)
            if cand_c5 < best_c5:
                best_h, best_c5 = cand.copy(), cand_c5
                cur_h, cur_c5 = cand.copy(), cand_c5

    return (best_h, best_c5, n)
