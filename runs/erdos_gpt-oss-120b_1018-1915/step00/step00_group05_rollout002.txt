import time
import numpy as np
from scipy.optimize import minimize

def _repair_candidate(x, target_sum, eps=1e-12, max_iter=10):
    """Project x onto the box [0,1] and enforce sum == target_sum."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        if abs(s - target_sum) <= eps:
            break
        # scale uniformly; if scaling pushes outside bounds we clip again afterward
        if s > 0:
            x = x * (target_sum / s)
        else:
            x = np.full_like(x, target_sum / x.size)
        x = np.clip(x, 0.0, 1.0)
    # final adjustment of any remaining tiny discrepancy by adding/subtracting uniformly
    diff = target_sum - x.sum()
    if abs(diff) > eps:
        # distribute diff over entries not at bounds
        free = (x > eps) & (x < 1 - eps)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
    return x

def _objective(h, dx):
    """Compute max_k ∫ h(x) (1-h(x+k)) dx using discrete correlation."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return float(np.max(corr) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns:
        h_values (np.ndarray) : the best function values found (length n_points)
        c5_bound (float)      : the corresponding C5 upper bound
        n_points (int)        : number of discretisation points used
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    n_points = kwargs.get('n_points')
    if n_points is None:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            n_points = len(initial_h_values)
        else:
            n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum(h)*dx = 1

    # Initial candidate
    if 'initial_h_values' in globals() and initial_h_values is not None:
        x0 = np.asarray(initial_h_values, dtype=float)
        if x0.size != n_points:
            # truncate or pad with uniform values to match n_points
            if x0.size > n_points:
                x0 = x0[:n_points]
            else:
                pad = np.full(n_points - x0.size, target_sum / n_points)
                x0 = np.concatenate([x0, pad])
    else:
        x0 = np.full(n_points, target_sum / n_points)

    best_h = _repair_candidate(x0, target_sum)
    best_val = _objective(best_h, dx)

    # Equality constraint for SLSQP
    eq_cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * n_points

    start_time = time.time()
    time_limit = budget_s * 0.95  # stop a bit early to avoid cut‑off

    # Main optimization loop
    while time.time() - start_time < time_limit:
        # Random perturbation around current best
        pert = best_h + rng.normal(scale=0.07, size=n_points)
        pert = _repair_candidate(pert, target_sum)

        res = minimize(
            fun=lambda h: _objective(h, dx),
            x0=pert,
            method='SLSQP',
            bounds=bounds,
            constraints=eq_cons,
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        if not res.success:
            continue

        cand = _repair_candidate(res.x, target_sum)
        cand_val = _objective(cand, dx)

        if cand_val + 1e-9 < best_val:
            best_h, best_val = cand, cand_val
            # early exit if we already beat the target threshold
            if best_val <= 0.3808:
                break

    return best_h, best_val, n_points
