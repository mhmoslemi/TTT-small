import time
import math
import numpy as np

def _c5_value(h, dx):
    """Compute C5 = max_x ∫ h(t) (1 - h(t+x)) dt for discrete h."""
    # correlation of h with (1-h) over full range, then scale by dx
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
    - h_values : ndarray of length n_points with values in [0,1] and sum = n_points/2
    - c5_bound : the achieved C5 upper bound (float)
    - n_points : number of discretisation points used
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 45                     # default size when no parent is supplied
        target = n / 2.0
        # simple feasible construction: 1's, a single fractional entry, rest 0's
        h = np.zeros(n, dtype=float)
        whole = int(math.floor(target))
        h[:whole] = 1.0
        if whole < n:
            h[whole] = target - whole
        np.random.shuffle(h)

    dx = 2.0 / n

    # Ensure feasibility (tiny numerical drift may appear)
    current_sum = h.sum()
    if not np.isclose(current_sum, n / 2.0):
        # renormalise by scaling the non‑boundary entries proportionally
        # while keeping 0/1 bounds intact
        excess = current_sum - n / 2.0
        mask = (h > 0) & (h < 1)
        if mask.any():
            h[mask] -= excess / mask.sum()
            h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c = _c5_value(best_h, dx)

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    T = 1.0                      # initial temperature
    cooling = 0.9995
    max_iter = 10_000_000        # safety cap
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.95 and it < max_iter:
        # propose a swap (preserves sum and bounds automatically)
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        h_new = best_h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]

        c_new = _c5_value(h_new, dx)

        delta = c_new - best_c
        if delta < 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            best_h = h_new
            best_c = c_new

        T *= cooling
        it += 1

    # ------------------------------------------------------------------
    # Return the best found solution
    # ------------------------------------------------------------------
    return (best_h, best_c, n)
