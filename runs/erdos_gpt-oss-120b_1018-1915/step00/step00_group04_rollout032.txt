import numpy as np
import time
import random

def _compute_c5(h, dx):
    """Maximum overlap integral for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _initial_binary(n_points, rng):
    """Create a random binary vector with exactly half ones."""
    half = n_points // 2
    h = np.zeros(n_points, dtype=np.float64)
    h[:half] = 1.0
    rng.shuffle(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points).

    h_values : list of floats, length n_points, each in [0,1]
    c5_bound : float, the achieved upper bound C5
    n_points : int, number of discretization points (even)
    """
    start_time = time.time()
    rng = random.Random(seed)
    np_rng = np.random.default_rng(seed)

    # Choose a modest resolution; increase if time permits.
    n_points = 200  # even number, gives dx = 0.01
    dx = 2.0 / n_points
    target_sum = n_points / 2  # because integral must be 1

    # Try to use a provided parent construction if available.
    global_vars = globals()
    if "initial_h_values" in global_vars:
        parent = np.array(initial_h_values, dtype=np.float64)
        # Resize / trim / pad to our n_points while keeping total mass.
        if parent.size != n_points:
            # simply resample using linear interpolation
            x_old = np.linspace(0, 2, parent.size, endpoint=False)
            x_new = np.linspace(0, 2, n_points, endpoint=False)
            parent = np.interp(x_new, x_old, parent)
        # Project to satisfy bounds and sum constraint
        parent = np.clip(parent, 0.0, 1.0)
        cur_sum = parent.sum()
        if cur_sum != 0:
            parent *= (target_sum / cur_sum)
        # final clipping may break the sum a little; fix by a simple redistribution
        diff = target_sum - parent.sum()
        if abs(diff) > 1e-12:
            # add/subtract uniformly and clip again
            parent += diff / n_points
            np.clip(parent, 0.0, 1.0, out=parent)
        h = parent.copy()
    else:
        # start from a random binary pattern
        h = _initial_binary(n_points, np_rng)

    best_h = h.copy()
    best_score = _compute_c5(best_h, dx)

    # Simulated annealing parameters
    max_iters = 200000
    temp_start = 1.0
    temp_end = 1e-4
    decay = (temp_end / temp_start) ** (1.0 / max_iters)

    temp = temp_start
    it = 0

    while True:
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break
        if it >= max_iters:
            break

        # propose a swap of a 1 and a 0 (binary case)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            # fallback: random re‑initialisation
            h = _initial_binary(n_points, np_rng)
            continue
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        h[i], h[j] = h[j], h[i]

        new_score = _compute_c5(h, dx)

        # acceptance criterion
        delta = new_score - best_score
        if delta < 0 or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            # accept move
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]

        # temperature schedule
        temp *= decay
        it += 1

    # Final polishing: try a few random local perturbations that keep sum exact.
    # Small continuous adjustments may improve the bound slightly.
    for _ in range(5000):
        if time.time() - start_time > budget_s * 0.98:
            break
        # pick two indices i!=j
        i, j = rng.sample(range(n_points), 2)
        # move epsilon from i to j, respecting bounds
        eps = (rng.random() - 0.5) * 0.02  # [-0.01,0.01]
        if best_h[i] - eps < 0 or best_h[i] - eps > 1:
            continue
        if best_h[j] + eps < 0 or best_h[j] + eps > 1:
            continue
        candidate = best_h.copy()
        candidate[i] -= eps
        candidate[j] += eps
        new_score = _compute_c5(candidate, dx)
        if new_score < best_score:
            best_score = new_score
            best_h = candidate

    # Return as list for compatibility with the harness
    return best_h.tolist(), best_score, n_points
