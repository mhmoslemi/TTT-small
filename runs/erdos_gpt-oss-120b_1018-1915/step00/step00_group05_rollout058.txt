import time
import numpy as np
import random
from math import exp

def _objective(h, dx):
    """Return max overlap integral for step function h."""
    g = 1.0 - h
    # full convolution gives length 2n-1, max corresponds to all integer shifts
    return np.max(np.correlate(h, g, mode="full")) * dx

def _random_feasible(n):
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    # Dirichlet gives positive components summing to 1
    rnd = np.random.rand(n) + 1e-8
    rnd = rnd / rnd.sum() * (n / 2.0)
    # Clip to [0,1] while preserving sum by a simple projection loop
    for _ in range(5):
        excess = rnd - 1.0
        excess = excess.clip(min=0)
        deficit = -rnd
        deficit = deficit.clip(min=0)
        if excess.sum() == 0 and deficit.sum() == 0:
            break
        # redistribute excess to deficit proportionally
        total_excess = excess.sum()
        total_deficit = deficit.sum()
        if total_excess > 0 and total_deficit > 0:
            transfer = np.minimum(excess, (total_excess / total_deficit) * deficit)
            rnd -= transfer
            rnd += transfer * (total_excess / total_deficit)
        rnd = rnd.clip(0, 1)
    # final scaling to exact sum
    current_sum = rnd.sum()
    if abs(current_sum - n / 2.0) > 1e-12:
        rnd *= (n / 2.0) / current_sum
    return rnd

    Optimize a step function h on [0,2] to minimize
    C5 = max_k ∫ h(x)(1-h(x+k))dx.

    Returns (h_values, c5_bound, n_points).
    """

    np.random.seed(seed)

    # Number of sampling points (use parent length if provided)
    n_points = kwargs.get("n_points")
    if n_points is None:
        try:
            n_points = len(initial_h_values)  # type: ignore[name-defined]
        except Exception:
            n_points = 70
    dx = 2.0 / n_points

    # starting vector
    try:
        h0 = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        if h0.shape[0] != n_points:
            raise ValueError
        # project onto feasible set (clip & renormalise)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 *= (n_points / 2.0) / h0.sum()
    except Exception:
        h0 = _random_feasible(n_points)

    best_h = h0.copy()
    best_val = _objective(best_h, dx)

    current_h = best_h.copy()
    current_val = best_val

    t_start = time.time()
    # annealing parameters
    temp = 0.02
    decay = 0.9995
    min_temp = 1e-6

    while True:
        elapsed = time.time() - t_start
        if elapsed > budget_s - 0.5:
            break

        # pick donor (decrease) and receiver (increase) indices
        donors = np.where(current_h > 1e-12)[0]
        receivers = np.where(current_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            # fallback to full re‑randomisation
            current_h = _random_feasible(n_points)
            current_val = _objective(current_h, dx)
            continue

        i = int(np.random.choice(donors))
        j = int(np.random.choice(receivers))
        # maximal shift respecting bounds
        max_delta = min(current_h[i], 1.0 - current_h[j])
        if max_delta <= 0:
            continue
        delta = np.random.rand() * max_delta

        # propose new candidate
        cand = current_h.copy()
        cand[i] -= delta
