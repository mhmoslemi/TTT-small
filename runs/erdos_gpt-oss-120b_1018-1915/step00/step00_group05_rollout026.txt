import time
import numpy as np

def _dx(n):
    return 2.0 / n

def _compute_c5(h):
    """discrete C5 for step vector h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * _dx(len(h))

def _project_feasible(vec, target_sum):
    """clip to [0,1] and scale to hit exact target sum (within tiny tolerance)."""
    v = np.clip(vec, 0.0, 1.0)
    total = v.sum()
    if total == 0:
        return np.full_like(v, target_sum / len(v))
    factor = target_sum / total
    v = v * factor
    # a second clip may be needed; rebalance any overflow/underflow on first element
    v = np.clip(v, 0.0, 1.0)
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        # distribute the small remainder to the first coordinate respecting bounds
        add = min(diff, 1.0 - v[0])
        add = max(add, -v[0])
        v[0] += add
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    rnd = np.random.RandomState(seed)
    start = time.time()

    # Determine resolution
    if "initial_h_values" in globals():
        parent = globals()["initial_h_values"]
        n_points = int(len(parent))
        h = np.array(parent, dtype=float)
    else:
        n_points = 70
        h = rnd.rand(n_points)

    target_sum = n_points / 2.0          # because ∫h dx = 1  ⇒ sum(h)=n/2
    h = _project_feasible(h, target_sum)

    best_h = h.copy()
    best_score = _compute_c5(best_h)

    # annealing parameters
    max_iters = 20000
    temp0 = 0.02
    temp_decay = 0.9995

    i = 0
    while i < max_iters and (time.time() - start) < budget_s * 0.90:
        i += 1
        # pick two distinct indices
        idx = rnd.choice(n_points, size=2, replace=False)
        i1, i2 = idx[0], idx[1]

        # propose a small transfer
        eps = (rnd.rand() * 2.0 - 1.0) * 0.05   # magnitude up to ±0.05
        proposal = best_h.copy()
        proposal[i1] = np.clip(proposal[i1] - eps, 0.0, 1.0)
        proposal[i2] = np.clip(proposal[i2] + eps, 0.0, 1.0)

        # enforce exact integral constraint
        proposal = _project_feasible(proposal, target_sum)

        score = _compute_c5(proposal)

        # acceptance test (simulated annealing)
        if score < best_score:
            best_h, best_score = proposal, score
        else:
            temp = temp0 * (temp_decay ** i)
            prob = np.exp(-(score - best_score) / max(temp, 1e-12))
            if rnd.rand() < prob:
                best_h, best_score = proposal, score

    # final safety projection & score recompute
    best_h = _project_feasible(best_h, target_sum)
    best_score = _compute_c5(best_h)

    # sanity checks
    assert len(best_h) == n_points
    assert np.all((best_h >= 0.0) & (best_h <= 1.0))
    assert np.isclose(best_h.sum(), target_sum, atol=1e-9)

    return best_h, float(best_score), int(n_points)
