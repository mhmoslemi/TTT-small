import time
import numpy as np

def _project_capped_simplex(v, target_sum, max_iter=50, tol=1e-9):
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum x_i = target_sum}.
    Uses a simple bisection on the shifting parameter.
    """
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _c5_value(h):
    """Compute C5 = max correlation of h and (1-h) multiplied by dx."""
    n = h.shape[0]
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx

def _initial_vector(n):
    """Fallback random start respecting constraints."""
    v = np.random.rand(n)
    target = n / 2.0
    return _project_capped_simplex(v, target)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that minimizes the
    Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    t_start = time.time()

    # Determine problem size
    try:
        init = initial_h_values  # provided by the harness
        n = init.shape[0]
        h_current = np.copy(init)
    except NameError:
        n = kwargs.get("n_points", 70)
        h_current = _initial_vector(n)

    target_sum = n / 2.0
    # Ensure the starting vector satisfies constraints exactly
    h_current = _project_capped_simplex(h_current, target_sum)
    best_h = h_current
    best_val = _c5_value(best_h)

    # Simulated‑annealing style parameters
    temperature = 0.5
    decay = 0.9995               # exponential cooling
    sigma = 0.2                  # initial Gaussian perturbation scale
    sigma_decay = 0.9999

    # Main loop
    iteration = 0
    while True:
        if time.time() - t_start > budget_s * 0.95:
            break

        # Generate a candidate by Gaussian perturbation
        noise = np.random.randn(n) * sigma
        cand = best_h + noise
        cand = _project_capped_simplex(cand, target_sum)

        cand_val = _c5_value(cand)

        # Acceptance test
        if cand_val < best_val or np.random.rand() < np.exp(-(cand_val - best_val) / max(temperature, 1e-12)):
            best_h = cand
            best_val = cand_val

        # Cool down
        temperature *= decay
        sigma *= sigma_decay
        iteration += 1

    return best_h, best_val, n
