import time
import numpy as np
import random

def compute_c5(h):
    """Discrete evaluation of C5 for a step function sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_binary_pattern(n, ones):
    """Create a binary (0/1) vector of length n with exactly `ones` entries equal to 1."""
    h = np.zeros(n, dtype=float)
    h[:ones] = 1.0
    np.random.shuffle(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points).
    h_values : 1‑D numpy array of the step function samples on [0,2]
    c5_bound : the achieved C5 value for this h
    n_points : length of h_values
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # --------------------------------------------------------------
    # 1. Choose a convenient even resolution (binary pattern works best)
    # --------------------------------------------------------------
    n = 100                     # even so that sum = n/2 can be achieved with exactly n/2 ones
    target_ones = n // 2       # 50 ones, 50 zeros give sum = 50 = n/2

    # --------------------------------------------------------------
    # 2. Initialise a random binary pattern
    # --------------------------------------------------------------
    h = project_binary_pattern(n, target_ones)
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # --------------------------------------------------------------
    # 3. Simple swap‑based stochastic hill‑climbing
    # --------------------------------------------------------------
    # Parameters
    max_no_improve = 5000          # restart after many stagnations
    no_improve = 0
    temperature = 0.01            # for occasional uphill moves
    cooling_rate = 0.9999

    while time.time() - start_time < budget_s:
        # propose a swap of a 0 and a 1
        idx0 = np.where(best_h == 0)[0]
        idx1 = np.where(best_h == 1)[0]
        if idx0.size == 0 or idx1.size == 0:
            break
        i = int(np.random.choice(idx0))
        j = int(np.random.choice(idx1))

        h_candidate = best_h.copy()
        h_candidate[i], h_candidate[j] = h_candidate[j], h_candidate[i]   # swap

        score = compute_c5(h_candidate)

        # accept if improvement or with Metropolis probability
        if score < best_score or random.random() < np.exp(-(score - best_score) / max(temperature, 1e-12)):
            best_h, best_score = h_candidate, score
            no_improve = 0
        else:
            no_improve += 1

        # cooling
        temperature *= cooling_rate

        # occasional random restart to escape deep local minima
        if no_improve >= max_no_improve:
            h = project_binary_pattern(n, target_ones)
            best_h = h
            best_score = compute_c5(best_h)
            no_improve = 0
            temperature = 0.01

        # tiny chance to try a fractional perturbation around the current best
        if random.random() < 0.001:
            # pick a 1 -> lower slightly, give the amount to a 0
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            if ones.size and zeros.size:
                i1 = int(np.random.choice(ones))
                i0 = int(np.random.choice(zeros))
                eps = 0.1 * random.random()   # 0 < eps <= 0.1
                h_frac = best_h.copy()
                h_frac[i1] = max(0.0, 1.0 - eps)
                h_frac[i0] = min(1.0, eps)
                score_frac = compute_c5(h_frac)
                if score_frac < best_score:
                    best_h, best_score = h_frac, score_frac

    # --------------------------------------------------------------
    # 4. Final tiny local refinement using continuous moves
    # --------------------------------------------------------------
    # Try to improve the best solution by moving small amounts between
    # a random pair while respecting bounds.
    refine_iters = 2000
    for _ in range(refine_iters):
        if time.time() - start_time >= budget_s:
            break
        i, j = np.random.choice(n, 2, replace=False)
        # amount we can transfer from i to j without breaking bounds
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        eps = 0.02 * random.random() * max_transfer
        h_new = best_h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        score = compute_c5(h_new)
        if score < best_score:
            best_h, best_score = h_new, score

    # --------------------------------------------------------------
    # 5. Return result
    # --------------------------------------------------------------
    return best_h, best_score, n
