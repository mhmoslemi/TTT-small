import numpy as np
import time

def _objective(h, dx):
    """Return the C5 value for a candidate step function."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def _initialize_h(n_points, target_sum):
    """Create a feasible start: uniform 0.5 satisfies the constraints."""
    h = np.full(n_points, 0.5, dtype=np.float64)
    # Adjust tiny rounding error to meet exact sum
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly (will stay within [0,1] because diff is tiny)
        h += diff / n_points
    return h

def _perturb(h, step):
    """Return a new h with the same sum, keeping values in [0,1]."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    # feasible delta bounds
    up_i = 1.0 - hi
    down_i = hi
    up_j = 1.0 - hj
    down_j = hj

    lower = max(-down_i, -up_j)
    upper = min(up_i, down_j)

    if lower >= upper:
        # cannot move these two entries; return original copy
        return h.copy()

    # limit magnitude by current step size
    bound = min(step, upper - lower)
    delta = np.random.uniform(-bound, bound)
    # ensure within [lower,upper]
    delta = np.clip(delta, lower, upper)

    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new
