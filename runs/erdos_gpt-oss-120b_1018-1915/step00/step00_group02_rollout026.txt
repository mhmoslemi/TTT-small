def compute_c5(h):
    """Return max overlap integral for step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _feasible_random(rng, n):
    """Generate a random vector in [0,1]^n with exact sum n/2."""
    # start from uniform samples
    x = rng.random(n)
    # scale to hit the required sum
    scale = (n / 2) / x.sum()
    x = np.clip(x * scale, 0.0, 1.0)
    # fix possible residual error
    diff = n / 2 - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 0) & (x < 1))[0]
        if free.size:
            add = diff / free.size
            x[free] += add
