import numpy as np
import time
import random

def _project(h, target_sum):
    """Project vector h onto [0,1]^n with given L1 target sum."""
    # Simple iterative clipping+scaling
    n = h.size
    for _ in range(5):
        np.clip(h, 0.0, 1.0, out=h)
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        delta = (target_sum - cur) / n
        h += delta
    # final clip to ensure feasibility
    np.clip(h, 0.0, 1.0, out=h)
    # if still off due to bounds, redistribute the remaining mass
    cur = h.sum()
    if cur < target_sum - 1e-12:
        # add remaining to interior points
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size:
            inc = (target_sum - cur) / idx.size
            h[idx] = np.clip(h[idx] + inc, 0.0, 1.0)
    elif cur > target_sum + 1e-12:
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size:
            dec = (cur - target_sum) / idx.size
            h[idx] = np.clip(h[idx] - dec, 0.0, 1.0)
    return h
