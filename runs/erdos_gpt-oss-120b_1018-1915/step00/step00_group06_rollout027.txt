import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx using linear correlation."""
    n = h.size
    size = 2 * n - 1
    # linear correlation via FFT (real signals)
    fa = np.fft.rfft(h, n=size)
    fb = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(fa * np.conj(fb), n=size)
    return np.max(corr) * (2.0 / n)


def _repair(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h) = n/2 exactly."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    for _ in range(6):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        # capacity to add (if diff>0) or remove (if diff<0)
        if diff > 0:
            cap = 1.0 - h
        else:
            cap = h
        total = cap.sum()
        if total == 0:
            break
        delta = cap * (diff / total)
        h += delta
        h = np.clip(h, 0.0, 1.0)
    return h


def _resample(h_old: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation from old grid (period 2) to a new grid."""
    x_old = np.linspace(0.0, 2.0, h_old.size, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return _repair(h_new)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimizes C₅."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_target = max(120, int(h0.size * 1.2))  # a modest increase in resolution
        h = _resample(h0, n_target)
    else:
        n_target = 120
        h = np.full(n_target, 0.5, dtype=float)

    h = _repair(h)
    cur_score = _compute_c5(h)
    best_h = h.copy()
    best_score = cur_score

    n = h.size

    # ----- optimisation loop -------------------------------------------------
    # Simple stochastic hill‑climb with a mild simulated‑annealing acceptance rule
    while time.time() - start_time < budget_s * 0.95:
        # annealing temperature (decays linearly)
        elapsed = time.time() - start_time
        temp = max(1e-4, (budget_s - elapsed) / budget_s) * 0.02

        # step size adapts with temperature
        max_delta = max(0.05, 0.3 * temp)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = (rng.random() * 2.0 - 1.0) * max_delta
        hi = h[i] + delta
        hj = h[j] - delta

        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue

        h_candidate = h.copy()
        h_candidate[i] = hi
        h_candidate[j] = hj

        cand_score = _compute_c5(h_candidate)

        # keep the globally best solution
        if cand_score < best_score:
            best_score = cand_score
            best_h = h_candidate.copy()

        # accept if improvement, otherwise with Metropolis probability
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / temp):
            h = h_candidate
            cur_score = cand_score

    return best_h, best_score, best_h.size
