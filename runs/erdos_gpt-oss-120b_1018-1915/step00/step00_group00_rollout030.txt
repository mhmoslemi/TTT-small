import numpy as np
import time

def _project_to_feasible(h, target_sum):
    """Clip to [0,1] and adjust to achieve exactly target_sum."""
    h = np.clip(h, 0.0, 1.0)
    # Simple redistribution loop – converges in a few passes for small n
    for _ in range(10):
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            free = h < 1.0
        else:
            free = h > 0.0
        if not np.any(free):
            break
        adjust = diff / free.sum()
        h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
    return h

def _worst_shift_and_gradient(h):
    """Return worst lag k (relative index) and its gradient."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    k_idx = np.argmax(corr)                     # index in full correlation array
    k = k_idx - (n - 1)                         # convert to lag (can be negative)
    # Build gradient for this lag
    if k >= 0:
        part1 = np.concatenate((np.zeros(k), 1.0 - h[: n - k]))
        part2 = np.concatenate((h[k:], np.zeros(k)))
    else:
        kpos = -k
        part1 = np.concatenate((1.0 - h[kpos:], np.zeros(kpos)))
        part2 = np.concatenate((np.zeros(kpos), h[: n - kpos]))
    grad = part1 - part2
    return k, grad, corr.max() * (2.0 / n)
