import time
import numpy as np

def _compute_c5(h):
    """Return the max overlap integral approximation for step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_sum(h, target):
    """Scale h to meet the sum constraint while staying in [0,1]."""
    # Simple iterative clipping + scaling
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        return np.full_like(h, target / h.size)
    scale = target / cur
    h *= scale
    # Clip again and re‑distribute any excess/deficit
    for _ in range(5):
        excess = h.sum() - target
        if abs(excess) < 1e-12:
            break
        # distribute excess proportionally to interior entries
        mask = (h > 0.0) & (h < 1.0)
        if not np.any(mask):
            break
        adj = excess / mask.sum()
        h[mask] -= adj
        h = np.clip(h, 0.0, 1.0)
    # final renormalisation (tiny error tolerated)
    h *= target / h.sum()
    return h

def _random_tweak(h, step):
    """Return a new h after moving a small amount from one index to another."""
    n = h.size
    i = np.random.randint(n)
    j = np.random.randint(n)
    while j == i:
        j = np.random.randint(n)
    # move from i (which must have mass) to j (which must have room)
    if h[i] <= 0.0 or h[j] >= 1.0:
        return None
    delta = min(step, h[i], 1.0 - h[j])
    if delta <= 0.0:
        return None
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def _local_search(initial_h, budget, target_sum):
    """Simple stochastic hill‑climb with annealing‑like step decay."""
    h = _project_sum(initial_h, target_sum)
    best_h = h.copy()
    best_val = _compute_c5(best_h)
    start = time.time()
    step = 0.05  # initial transfer amount
    it = 0
    while time.time() - start < budget:
        it += 1
        # slowly decay step size
        step = max(0.001, step * 0.9995)
        cand = _random_tweak(h, step)
        if cand is None:
            continue
        val = _compute_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand.copy()
            h = cand
        else:
            # occasional uphill move to escape local minima
            if np.random.rand() < 0.001:
                h = cand
    return best_h, best_val

def _upscale(h, factor):
    """Linear interpolation to increase resolution by an integer factor."""
    n = h.size
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, n * factor, endpoint=False)
    return np.interp(x_new, x_old, h)
