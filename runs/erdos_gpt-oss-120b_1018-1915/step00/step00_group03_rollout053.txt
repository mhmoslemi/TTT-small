import numpy as np
import time

def _compute_c5(h):
    """Return the overlap constant C5 for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _repair(h):
    """Project h onto the feasible set 0≤h≤1, sum(h)=n/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(15):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # indices that can still move in the needed direction
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        adjust = diff / np.count_nonzero(mask)
        h[mask] += adjust
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction to hit the sum exactly (might exceed bounds by epsilon)
    s = h.sum()
    if abs(s - target) > 1e-12:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _upscale(vec, new_n):
    """Linear interpolation of vec (defined on [0,2)) to length new_n."""
    old_n = vec.size
    if new_n == old_n:
        return vec.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, vec)

def _random_feasible(n, rng):
    """Generate a random feasible vector of length n."""
    h = rng.random(n)
    h = _repair(h)
    return h
