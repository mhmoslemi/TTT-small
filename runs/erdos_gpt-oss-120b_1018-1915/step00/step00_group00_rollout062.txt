import time
import numpy as np

def _c5_value(h, dx):
    """Compute C5 for a given step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound and number of points.

    The function uses a simple simulated‑annealing search that respects
    the constraints:
        0 ≤ h[i] ≤ 1,
        sum(h) * dx = 1   (i.e. sum(h) = n/2, where dx = 2/n).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine discretisation size and initial vector
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)  # provided by the harness
        n = init.size
        h_current = init.copy()
    except NameError:
        n = 45
        h_current = np.full(n, 0.5, dtype=float)  # uniform satisfies the integral

    dx = 2.0 / n
    target_sum = n / 2.0
    # Minor correction in case of numerical drift
    h_current *= target_sum / h_current.sum()

    # ------------------------------------------------------------------
    # Initialise best solution
    # ------------------------------------------------------------------
    best_h = h_current.copy()
    best_c = _c5_value(best_h, dx)

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    T = 1.0                     # initial temperature
    cooling = 0.9995            # exponential cooling factor
    max_iter = 10_000_000       # safety cap
    iter_count = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        # Time check (leave a small margin)
        if time.time() - start_time > budget_s * 0.96:
            break
        if iter_count >= max_iter:
            break
        iter_count += 1

        # Choose two distinct indices
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)

        hi, hj = h_current[i], h_current[j]

        # Determine feasible delta range that keeps both entries in [0,1]
        max_up = min(1.0 - hi, hj)          # increase hi, decrease hj
        max_down = min(hi, 1.0 - hj)        # decrease hi, increase hj
        if max_up + max_down == 0.0:
            continue  # cannot move mass between these two points

        delta = rng.uniform(-max_down, max_up)

        # Propose new vector
        h_new = h_current.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta

        # Evaluate
        new_c = _c5_value(h_new, dx)

        # Acceptance test
        if new_c < best_c:
            best_c = new_c
            best_h = h_new.copy()
        if new_c < _c5_value(h_current, dx) or rng.random() < np.exp(-(new_c - _c5_value(h_current, dx)) / T):
            h_current = h_new

        # Temperature update
        T *= cooling
        if T < 1e-8:
            T = 1e-8

        # Occasionally restart from best to escape stagnation
        if iter_count % 200000 == 0:
            h_current = best_h.copy()
            T = 1.0

    # Final safeguard: enforce exact sum constraint (tiny numerical errors)
    best_h *= target_sum / best_h.sum()
    best_h = np.clip(best_h, 0.0, 1.0)

    final_c = _c5_value(best_h, dx)
    return best_h, final_c, n
