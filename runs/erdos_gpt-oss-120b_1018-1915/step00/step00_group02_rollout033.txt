import time
import numpy as np

def _integral_target(n):
    """Desired sum of h values for given n (since dx=2/n and integral=1)."""
    return n / 2.0

def _project(h):
    """Clamp to [0,1] and adjust to satisfy the exact integral constraint."""
    n = h.shape[0]
    target = _integral_target(n)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds – distribute uniformly then reclamp
            h += diff / n
        else:
            h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h):
    """Compute the discretized C5 value for a candidate h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _neighbor(h, scale):
    """Generate a neighboring candidate by random perturbations."""
    n = h.shape[0]
    k = max(1, int(0.04 * n))          # perturb ~4% of entries
    idx = np.random.choice(n, k, replace=False)
    perturb = np.random.normal(scale=scale, size=k)
    h_new = h.copy()
    h_new[idx] += perturb
    return _project(h_new)

def _anneal(initial_h, time_limit):
    """Simple simulated‑annealing improvement within a fixed time budget."""
    start = time.time()
    best_h = _project(initial_h)
    best_score = _c5_value(best_h)

    n = best_h.shape[0]
    while time.time() - start < time_limit:
        # linear cooling from 0.1 to 0.0005
        elapsed = time.time() - start
        T = 0.1 * (1.0 - elapsed / time_limit) + 0.0005
        step = 0.08 * T + 0.001          # perturbation scale adapts to temperature
        cand = _neighbor(best_h, step)
        cand_score = _c5_value(cand)

        if cand_score < best_score or np.random.rand() < np.exp(-(cand_score - best_score) / max(T, 1e-12)):
            best_h, best_score = cand, cand_score
    return best_h, best_score

def _upsample(h):
    """Double the resolution using linear interpolation."""
    n = h.shape[0]
    new_n = min(n * 2, 400)   # cap size to keep work reasonable

    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _project(h_new)
