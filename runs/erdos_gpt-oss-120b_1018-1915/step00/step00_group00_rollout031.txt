import time, math
import numpy as np

def _compute_c5(h, dx):
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx for discrete h."""
    # correlation h with (1-h) over full range
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_feasible(v):
    """Clip to [0,1] and renormalize to have sum = len(v)/2."""
    v = np.clip(v, 0.0, 1.0)
    target = v.size / 2.0
    cur = v.sum()
    if cur == 0:
        return np.full_like(v, target / v.size)
    return v * (target / cur)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] giving a low C5 bound.

    Returns
    -------
    h_values : list of float
        The sampled values of h (length n_points).
    c5_bound : float
        The achieved C5 upper bound.
    n_points : int
        Number of samples (length of h_values).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # Starting point --------------------------------------------------- #
    # ------------------------------------------------------------------ #
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: uniform 0.5 satisfies the integral constraint
        n = 45
        h = np.full(n, 0.5, dtype=float)

    n = h.size
    dx = 2.0 / n
    h = _project_to_feasible(h)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    # ------------------------------------------------------------------ #
    # Simulated‑annealing style search --------------------------------- #
    # ------------------------------------------------------------------ #
    # Temperature schedule (linear from T0 to Tf)
    T0 = 1.0
    Tf = 1e-4

    # Parameters controlling move magnitude
    max_step = 0.25          # maximum absolute ε for a single swap
    local_step_factor = 0.5 # typical step size as a fraction of max_step

    # Main loop – stop a bit before budget to avoid a hard kill
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        # linear temperature decay
        frac = elapsed / budget_s
        T = T0 * (1.0 - frac) + Tf * frac

        # ----- propose a move -------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # bound the amount we may transfer while staying in [0,1]
        available_i = 1.0 - h[i] if h[i] < 0.5 else h[i]
        available_j = 1.0 - h[j] if h[j] < 0.5 else h[j]
        bound = min(max_step, available_i, available_j)
        if bound <= 0:
            continue

        eps = (rng.random() - 0.5) * 2.0 * bound * local_step_factor
        # transfer ε from j to i (so i gains, j loses)
        new_i = h[i] + eps
        new_j = h[j] - eps
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue

        h_candidate = h.copy()
        h_candidate[i] = new_i
        h_candidate[j] = new_j

        # ----- evaluate -------------------------------------------------------
        cand_c5 = _compute_c5(h_candidate, dx)
        delta = cand_c5 - best_c5

        # ----- acceptance ------------------------------------------------------
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = h_candidate.copy()
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            h = h_candidate

        # Occasionally perform a larger random reshuffle to escape plateaus
        if rng.random() < 0.001:
            perm = rng.permutation(n)
            h = h[perm]
            h = _project_to_feasible(h)

    # ------------------------------------------------------------------ #
    # Return ------------------------------------------------------------ #
    # ------------------------------------------------------------------ #
    return best_h.tolist(), float(best_c5), n
