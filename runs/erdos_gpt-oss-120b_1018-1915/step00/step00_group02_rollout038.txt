def project(h, target):
    h = np.clip(h,0,1)
    current = h.sum()
    # adjust using simple proportional scaling for interior points
    # compute mask where 0<h<1
    mask = (h>0) & (h<1)
    if mask.any():
        # distribute difference uniformly among mask entries
        diff = target - current
        h[mask] += diff / mask.sum()
        # clamp again
        h = np.clip(h,0,1)
        # if still not meet target, iterate a few times
        for _ in range(10):
            current = h.sum()
            if abs(current - target) < 1e-9:
                break
            mask = (h>0) & (h<1)
            if not mask.any():
                break
            diff = target - current
            h[mask] += diff / mask.sum()
            h = np.clip(h,0,1)
    # final correction: if still off, adjust smallest needed at extremes
    current = h.sum()
    diff = target - current
    if abs(diff)>1e-9:
        # adjust first element within bounds
        for i in range(len(h)):
            avail_up = 1 - h[i]
            avail_down = h[i]
            if diff>0 and avail_up>0:
                delta = min(diff, avail_up)
                h[i] += delta
                diff -= delta
            elif diff<0 and avail_down>0:
                delta = min(-diff, avail_down)
                h[i] -= delta
                diff += delta
            if abs(diff)<1e-12:
                break
    return h

import time, random, math
import numpy as np

def _project_to_constraints(vec, target_sum):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    vec = np.clip(vec, 0.0, 1.0)
    # iterative water-filling
    for _ in range(20):
        cur = vec.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # eligible indices (still have room to move)
        if diff > 0:
            mask = vec < 1.0
        else:
            mask = vec > 0.0
        if not np.any(mask):
            break
        # distribute evenly among eligible entries
        delta = diff / mask.sum()
        vec[mask] += delta
        vec = np.clip(vec, 0.0, 1.0)
    # final tiny correction if needed
    cur = vec.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        for i in range(len(vec)):
            if diff > 0 and vec[i] < 1.0:
                add = min(diff, 1.0 - vec[i])
                vec[i] += add
                diff -= add
            elif diff < 0 and vec[i] > 0.0:
                sub = min(-diff, vec[i])
                vec[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return vec

def _c5_value(h):
    """Compute the discrete C5 value for a step vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized step values (length n_points), each in [0,1].
    c5_bound : float
        The achieved upper bound C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    n_points : int
        Number of discretisation points used.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise base vector (use provided parent if available)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
    else:
        # a simple feasible start: uniform 0.5 values (integral = n/2)
        base = np.full(60, 0.5, dtype=float)

    n = len(base)
    target = n / 2.0
    h_best = _project_to_constraints(base.copy(), target)
    best_val = _c5_value(h_best)

    # ------------------------------------------------------------------
    # 2. Main stochastic local search
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.9:
        # Random perturbation of a fraction of coordinates
        cand = h_best.copy()
        k = max(1, int(0.07 * n))                 # about 7 % of entries
        idx = np.random.choice(n, k, replace=False)
        cand[idx] += np.random.normal(scale=0.12, size=k)
        cand = _project_to_constraints(cand, target)

        val = _c5_value(cand)
        if val < best_val:
            h_best, best_val = cand, val

        # Occasional global restart (tiny probability)
        if random.random() < 0.02:
            cand = np.random.rand(n)
            cand = _project_to_constraints(cand, target)
            val = _c5_value(cand)
            if val < best_val:
                h_best, best_val = cand, val

        # ------------------------------------------------------------------
        # 3. Try increasing resolution if we have made good progress
        # ------------------------------------------------------------------
        if n < 180 and best_val < 0.39 and (time.time() - start_time) > budget_s * 0.3:
            n2 = n * 2
            # Linear interpolation onto the finer grid
            x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
            x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0 / n2
            h_fine = np.interp(x_new, x_old, h_best)
            target2 = n2 / 2.0
            h_fine = _project_to_constraints(h_fine, target2)
            # reset search state on the finer grid
            h_best = h_fine
            n = n2
            target = target2
            best_val = _c5_value(h_best)

    return h_best, best_val, n
