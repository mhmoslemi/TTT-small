import time
import numpy as np

def _c5_score(h):
    """Compute C5 for a discretized step function."""
    # Use built‑in evaluator if present (more reliable)
    if 'evaluate_erdos_solution' in globals():
        return evaluate_erdos_solution(h)
    # fallback implementation
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _feasible_start(n, target_sum):
    """Return a simple feasible vector of length n with sum = target_sum."""
    h = np.zeros(n, dtype=float)
    ones = int(np.floor(target_sum))
    h[:ones] = 1.0
    remainder = target_sum - ones
    if remainder > 0:
        h[ones] = remainder
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # problem size – keep it modest for speed
    n_points = 45
    target_sum = n_points / 2.0   # because ∫ h = 1 over length 2

    # initialise h
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - target_sum) > 1e-9:
            # simple rescaling, then clip and distribute leftover
            h *= target_sum / cur
            h = np.clip(h, 0.0, 1.0)
            diff = target_sum - h.sum()
            if abs(diff) > 1e-9:
                free = np.where((h > 0.0) & (h < 1.0))[0]
                if free.size:
                    h[free] += diff / free.size
                h = np.clip(h, 0.0, 1.0)
    else:
        h = _feasible_start(n_points, target_sum)

    best_h = h.copy()
    best_score = _c5_score(best_h)

    start_time = time.time()
    iteration = 0
    step_size = 0.12   # typical magnitude of a perturbation

    # main stochastic local‑search loop
    while time.time() - start_time < budget_s * 0.95:
