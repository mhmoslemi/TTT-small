import numpy as np

def compute_c5(h):
    """Return the C5 value for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def repair(h, target_sum):
    """Clip to [0,1] and renormalise to exact sum (if possible)."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        return np.full_like(h, target_sum / h.size)
    factor = target_sum / s
    h *= factor
    # a second clip may be needed; if it changes the sum, distribute the leftover uniformly
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny remainder to the interior points
        interior = (h > 0) & (h < 1)
        if interior.any():
            h[interior] += diff / interior.sum()
        else:
            h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    """Linear interpolation of h (defined on [0,2]) to new_n points preserving integral."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
