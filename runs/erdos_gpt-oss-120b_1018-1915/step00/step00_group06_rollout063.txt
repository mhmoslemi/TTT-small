import time
import math
import numpy as np

def _c5_value(h):
    """Compute the discrete C5 value for a step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _project_to_feasible(h, target_sum):
    """Project vector onto the box [0,1] and enforce the linear sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    # simple redistribution: add/subtract uniformly, then clip again
    delta = target_sum - cur
    h += delta / h.size
    h = np.clip(h, 0.0, 1.0)
    # if still off due to clipping, correct the residual on the largest free entries
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual on entries that are not at the bounds
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # fall back to scaling
            h *= target_sum / h.sum()
        else:
            add = residual / free.size
            h[free] += add
            h = np.clip(h, 0.0, 1.0)
    # final safety scaling
    if h.sum() == 0:
        h[:] = target_sum / h.size
    else:
        h *= target_sum / h.sum()
    h = np.clip(h, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimizes the
    Erdős C5 overlap constant.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : list of floats, length n_points, satisfying the constraints
        c5_bound : float, the obtained upper bound C5
        n_points : int, the discretisation size used
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ------------------------------------------------------------
    # choose a modest resolution; higher gives finer search but slower
    n = 200                     # 200 points gives dx = 0.01
    target_sum = n / 2.0       # because integral must be 1 over [0,2]
    # ------------------------------------------------------------
    # initialise h
    try:
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            # interpolate to required grid
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, init)
        else:
            h0 = init.copy()
    except Exception:
        # fallback: random values
        h0 = rng.random(n)

    # project initial vector to satisfy constraints (box + sum)
    h0 = _project_to_feasible(h0, target_sum)

    # keep the best solution seen
    best_h = h0.copy()
    best_c = _c5_value(best_h)

    # ------------------------------------------------------------
    # Simulated‑annealing / stochastic hill‑climbing on a binary backbone.
    # We keep a binary mask of which coordinates are at 0 or 1; the remaining
    # coordinates are allowed to be fractional and are adjusted by the
    # projection routine after each perturbation.
    # ------------------------------------------------------------
    # start from a binary version of h0 (take the largest entries)
    binary_h = np.zeros_like(h0)
    idx = np.argpartition(-h0, int(target_sum))[:int(target_sum)]
    binary_h[idx] = 1.0

    current_h = binary_h.copy()
    current_c = _c5_value(current_h)

    # parameters for the search
    max_iters = 200000
    report_interval = 2.0               # seconds
    next_report = start_time + report_interval

    while time.time() - start_time < 0.95 * budget_s and max_iters > 0:
        max_iters -= 1

        # propose a swap of a 1 and a 0 (maintains exact sum)
        ones = np.where(current_h == 1.0)[0]
        zeros = np.where(current_h == 0.0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        trial = current_h.copy()
        trial[i] = 0.0
        trial[j] = 1.0

        # occasional small random nudges to escape plateaus
        if rng.random() < 0.03:
            idx = rng.integers(0, n)
            delta = rng.uniform(-0.2, 0.2)
            trial[idx] = np.clip(trial[idx] + delta, 0.0, 1.0)
            trial = _project_to_feasible(trial, target_sum)

        c_trial = _c5_value(trial)

        # accept if better, otherwise with a temperature‑dependent probability
        temperature = max(0.001, (budget_s - (time.time() - start_time)) / budget_s)
        accept = False
        if c_trial < current_c:
            accept = True
        else:
            prob = math.exp(-(c_trial - current_c) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            current_h = trial
            current_c = c_trial
            if c_trial < best_c:
                best_c = c_trial
                best_h = trial.copy()

        # periodic reporting (helps to keep within time)
        if time.time() >= next_report:
            next_report = time.time() + report_interval

    # final projection of the best vector (in case of numeric drift)
    best_h = _project_to_feasible(best_h, target_sum)

    # compute final C5
    final_c = _c5_value(best_h)

    return (best_h.tolist(), float(final_c), n)
