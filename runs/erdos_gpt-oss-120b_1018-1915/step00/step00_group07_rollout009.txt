import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize
    C5 = max_k ∫ h(x) (1 - h(x+k)) dx.

    Returns:
        h_values : np.ndarray of shape (n_points,)
        c5_bound : float, the achieved C5 value
        n_points : int, number of discretization points
    """
    rng = np.random.default_rng(seed)

    # Determine number of points; use initial_h_values if present
    try:
        init_h = np.array(initial_h_values, dtype=float)
        n_points = init_h.size
    except NameError:
        # fallback: 80 points (as in the statement)
        n_points = 80
        init_h = None

    dx = 2.0 / n_points
    target_sum = n_points / 2.0          # because sum(h)*dx = 1

    # Helper: project a random vector to the feasible set via simple swaps
    if init_h is not None:
        # Ensure feasibility (might be slightly off due to rounding)
        h = np.clip(init_h, 0.0, 1.0)
        current_sum = h.sum()
        if not np.isclose(current_sum, target_sum):
            # Rescale uniformly while respecting bounds
            scale = target_sum / current_sum
            h *= scale
            h = np.clip(h, 0.0, 1.0)
            # Fix any residual sum error by distributing to non‑boundary entries
            residual = target_sum - h.sum()
            if abs(residual) > 1e-12:
                free = (h > 0.0) & (h < 1.0)
                if free.any():
                    h[free] += residual / free.sum()
                else:
                    # fallback to uniform distribution
                    h = np.full(n_points, target_sum / n_points)
        # final clip to stay within [0,1]
        h = np.clip(h, 0.0, 1.0)
    else:
        # Uniform start (sum = target_sum)
        h = np.full(n_points, target_sum / n_points)

    # Evaluate function (pre‑imported)
    def eval_h(vec):
        # evaluate_erdos_solution is assumed to return (c5, …)
        return evaluate_erdos_solution(vec)[0]

    best_h = h.copy()
    best_score = eval_h(best_h)

    # Simulated annealing parameters
    start_temp = 0.02
    end_temp = 0.0002
    max_time = budget_s * 0.95   # keep a safety margin
    start_time = time.time()
    it = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break
        # linear temperature schedule
        progress = elapsed / max_time
        temp = start_temp * (1 - progress) + end_temp * progress

        # propose a swap move between two random indices
        i, j = rng.choice(n_points, size=2, replace=False)
        hi, hj = h[i], h[j]

        # amount that can be transferred from i to j while staying in [0,1]
        max_transfer = min(hi, 1.0 - hj)
        min_transfer = -min(hj, 1.0 - hi)   # negative means transfer from j to i
        if max_transfer == min_transfer:
            continue  # nothing to move

        delta = rng.random() * (max_transfer - min_transfer) + min_transfer
        if abs(delta) < 1e-15:
            continue

        h_candidate = h.copy()
        h_candidate[i] -= delta
        h_candidate[j] += delta

        # (feasibility is guaranteed by construction)
        cand_score = eval_h(h_candidate)

        # Metropolis acceptance
        if cand_score < best_score or rng.random() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):

            if cand_score < best_score:
                best_score = cand_score
                best_h = h_candidate.copy()

        it += 1

    # Final evaluation to be safe
    final_score = eval_h(best_h)

    return best_h, float(final_score), n_points
