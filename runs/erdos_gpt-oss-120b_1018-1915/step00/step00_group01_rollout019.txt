import numpy as np
import time
import math

def evaluate_c5(h):
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def random_binary_initial(n, rng):
    """Create a binary vector of length n with exactly n/2 ones."""
    half = n // 2
    arr = np.zeros(n, dtype=np.float64)
    arr[:half] = 1.0
    rng.shuffle(arr)
    return arr

def simulated_annealing(n, rng, time_limit):
    """Run SA to minimize C5 for binary sequences of length n."""
    best_h = random_binary_initial(n, rng)
    best_c5 = evaluate_c5(best_h)

    current_h = best_h.copy()
    current_c5 = best_c5

    # Temperature schedule parameters
    start_temp = 0.02   # scale of C5 difference
    end_temp = 1e-5
    iter_cnt = 0
    start_time = time.time()

    # Pre‑compute indices of zeros and ones for fast swaps
    zeros = np.flatnonzero(current_h == 0)
    ones = np.flatnonzero(current_h == 1)

    while True:
        now = time.time()
        if now - start_time >= time_limit:
            break

        iter_cnt += 1
        # linear cooling in log‑space
        frac = iter_cnt / (time_limit * 200)  # rough scaling to iterations
        if frac > 1.0:
            frac = 1.0
        temp = start_temp * (end_temp / start_temp) ** frac

        # propose a swap: pick random one and zero
        if zeros.size == 0 or ones.size == 0:
            break
        i0 = rng.choice(zeros)
        i1 = rng.choice(ones)

        # perform swap
        current_h[i0] = 1.0
        current_h[i1] = 0.0

        new_c5 = evaluate_c5(current_h)

        accept = False
        if new_c5 < current_c5:
            accept = True
        else:
            # Metropolis criterion
            delta = new_c5 - current_c5
            if rng.random() < math.exp(-delta / temp):
                accept = True

        if accept:
            current_c5 = new_c5
            # update zero/one index lists
            zeros = np.where(current_h == 0)[0]
            ones = np.where(current_h == 1)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = current_h.copy()
        else:
            # revert swap
            current_h[i0] = 0.0
            current_h[i1] = 1.0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (binary values) to lower the
    Erdős C5 bound. Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # Choose a resolution; larger n gives finer control but costs more time.
    # 400 points is a good trade‑off for the given budget.
    n = 400
    # Ensure n is even for the half‑ones constraint.
    if n % 2 != 0:
        n += 1

    # Allocate time: leave a small safety margin.
    time_limit = budget_s - 0.5
    if time_limit < 1.0:
        time_limit = budget_s

    best_h, best_c5 = simulated_annealing(n, rng, time_limit)

    # Ensure the integral constraint holds exactly (within floating error).
    # Because the vector is binary with exactly n/2 ones, the sum equals n/2.
    # Convert to float64 before returning.
    best_h = best_h.astype(np.float64)

    return best_h, best_c5, n
