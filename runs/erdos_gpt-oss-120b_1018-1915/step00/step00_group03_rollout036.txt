import time
import math
import random
import numpy as np

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Return C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * float(np.max(corr))

def project_to_constraints(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and renormalize to hit the exact integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # distribute uniformly
        h[:] = target_sum / len(h)
        return h
    # Scale to match the required sum while staying in bounds
    scale = target_sum / cur
    h *= scale
    # After scaling some entries may exceed limits; fix iteratively
    for _ in range(3):
        over = np.maximum(h - 1.0, 0.0)
        under = np.maximum(-h, 0.0)
        if over.sum() == 0 and under.sum() == 0:
            break
        excess = over.sum()
        deficit = under.sum()
        if excess > 0:
            # pull excess down uniformly from the over‑filled entries
            h[over > 0] -= excess / (over > 0).sum()
        if deficit > 0:
            # add deficit uniformly to the under‑filled entries
            h[under > 0] += deficit / (under > 0).sum()
    # final clip
    return np.clip(h, 0.0, 1.0)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimizes C5."""
    random.seed(seed)
    np.random.seed(seed)

    # --- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # fallback: start from a flat profile (integral 1)
        n_default = 70
        h = np.full(n_default, 0.5, dtype=float)

    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0                # because sum(h)*dx == 1
    h = project_to_constraints(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    start_time = time.time()
    max_time = budget_s * 0.95          # keep a safety margin

    # --- optimisation loop -------------------------------------------------
    iteration = 0
    while time.time() - start_time < max_time:
        iteration += 1

        # temperature‑like schedule (simple linear decay)
        elapsed = time.time() - start_time
        remain_frac = max(0.0, 1.0 - elapsed / max_time)
        temp = max(0.01, remain_frac)

        # propose a random mass transfer between two coordinates
        i, j = random.randrange(n), random.randrange(n)
        if i == j:
            continue

        # maximal admissible transfer respecting bounds
        max_transfer = min(h[i], 1.0 - h[j], 0.02 * temp)
        if max_transfer <= 1e-12:
            continue
        delta = random.random() * max_transfer

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # (sum is preserved, bounds already respected)
        new_c5 = compute_c5(h_new, dx)

            best_h = h_new.copy()
