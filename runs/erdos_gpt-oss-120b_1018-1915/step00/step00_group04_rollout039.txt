def _project_to_feasible(x, target_sum):
    """Project vector x onto [0,1] box with given sum using a simple water‑filling."""
    n = x.size
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Iteratively adjust to hit the sum
    for _ in range(10):
        cur_sum = x.sum()
        if abs(cur_sum - target_sum) < 1e-12:
            break
        # Distribute the excess uniformly, then re‑clip
        delta = (target_sum - cur_sum) / n
        x += delta
        x = np.clip(x, 0.0, 1.0)
    # Final correction if needed (rare)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        # Add diff to the first non‑saturated entry
        idx = np.where((x > 0) & (x < 1))[0]
        if idx.size:
            x[idx[0]] = np.clip(x[idx[0]] + diff, 0.0, 1.0)
    return x

def _c5_score(h, dx):
    """Return the C5 value for a discretised step function h."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def _initial_vector(n_points):
    """Create a feasible starting vector of length n_points."""
    # binary pattern with half ones, half zeros, alternating for good mixing
    half = n_points // 2
    h = np.zeros(n_points, dtype=float)
    h[:half] = 1.0
    np.random.shuffle(h)  # randomise positions
    return h

def _upgrade_initial(old_h, new_n):
    """Resize an existing construction to a new resolution by linear interpolation."""
