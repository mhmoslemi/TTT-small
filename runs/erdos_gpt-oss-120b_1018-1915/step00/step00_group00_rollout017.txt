import time
import numpy as np

# evaluate_erdos_solution and initial_h_values are provided by the harness

def _project_feasible(v, target_sum, max_iter=10):
    """Clip to [0,1] and adjust to meet the target sum using a simple water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = v.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-9:
            break
        free = (v > 0.0) & (v < 1.0)
        cnt = free.sum()
        if cnt == 0:                     # all at bounds, distribute uniformly then re‑clip
            v += diff / v.size
            v = np.clip(v, 0.0, 1.0)
        else:
            v[free] += diff / cnt
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_score(h, dx):
    """Maximum overlap integral for a discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with n_points samples
    to lower the Erdős C5 bound.
    Returns (best_h, best_c5, n_points).
    """
    np.random.seed(seed)
    n_points = 45                       # default size (fast to evaluate)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0          # because ∫ h = 1  →  sum(h)*dx = 1

    # ---------- initial feasible vector ----------
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
        if h.size != n_points:
            # resample to correct length if needed
            h = np.resize(h, n_points)
    else:
        h = np.random.rand(n_points)

    h = _project_feasible(h, target_sum)

    best_h = h.copy()
    best_score = _c5_score(best_h, dx)

    start_time = time.time()
    cur_time = start_time

    # ---------- annealing parameters ----------
    max_iters = int(2e6)                 # generous upper bound, actual stop by time
    step_initial = 0.3                  # maximal transfer amount per move
    step_min = 1e-4
    decay_rate = 0.9995                 # exponential decay per iteration
    temperature_initial = 0.05
    temperature_min = 1e-4
    temp_decay = 0.9997

    step = step_initial
    temperature = temperature_initial

    i = 0
    while cur_time - start_time < budget_s and i < max_iters:
        # propose a mass transfer between two random indices
        a, b = np.random.randint(0, n_points, size=2)
        if a == b:
            continue

        # allowed transfer respecting bounds
        max_decrease = min(step, h[a])           # cannot go below 0
        max_increase = min(step, 1.0 - h[b])     # cannot exceed 1
        delta = np.random.uniform(-max_decrease, max_increase)

        if delta == 0.0:
            continue

        h_new = h.copy()
        h_new[a] -= delta
        h_new[b] += delta

        # re‑project to exact sum (should already be conserved, but clip may have nudged)
        h_new = _project_feasible(h_new, target_sum)

        score_new = _c5_score(h_new, dx)

        # accept if better, else probabilistically
        if score_new < best_score:
            best_h, best_score = h_new.copy(), score_new

        if score_new < _c5_score(h, dx) or np.random.rand() < np.exp(-(score_new - _c5_score(h, dx)) / max(temperature, 1e-12)):
            h = h_new

        # schedule decay
        step = max(step * decay_rate, step_min)
        temperature = max(temperature * temp_decay, temperature_min)

        i += 1
        cur_time = time.time()

    # final safety projection (ensure constraints)
    best_h = _project_feasible(best_h, target_sum)
    best_score = _c5_score(best_h, dx)

    return best_h, best_score, n_points
