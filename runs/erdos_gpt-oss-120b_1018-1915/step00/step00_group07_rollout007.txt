import time
import numpy as np

def _c5_from_h(h):
    """Compute C5 for a discrete step function h on [0,2]."""
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / h.size
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound and the number of samples.

    The function searches for a low‑C5 construction using simulated annealing.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # 1. Initialise h
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        # enforce exact mass sum = n/2 (integral 1)
        target_sum = n / 2.0
        current_sum = h.sum()
        if not np.isclose(current_sum, target_sum):
            h *= target_sum / current_sum
        # clip to stay feasible after possible rescaling
        np.clip(h, 0.0, 1.0, out=h)
    else:
        # default: 100 points, simple two‑block shape
        n = 100
        h = np.zeros(n, dtype=float)
        half = n // 4
        h[:half] = 1.0
        h[-half:] = 1.0
        # adjust to exact sum = n/2
        target_sum = n / 2.0
        current_sum = h.sum()
        h *= target_sum / current_sum
    # Ensure feasibility
    np.clip(h, 0.0, 1.0, out=h)

    cur_c5 = _c5_from_h(h)

    best_c5 = cur_c5

        if time.time() - start_time > budget_s * 0.94:
            break

        # linear temperature schedule
        temp = temp_start + (temp_end - temp_start) * (i / max_iters)

        # pick two distinct indices
        idx = np.random.choice(n, 2, replace=False)
        a, b = idx[0], idx[1]

        ha, hb = h[a], h[b]

        # maximum permissible transfer delta (positive means a→b)
        delta_max = min(ha, 1.0 - hb)
        delta_min = -min(hb, 1.0 - ha)

        if delta_max <= delta_min:

            continue  # nothing can be moved between these positions

        h[a] = ha - delta
        h[b] = hb + delta
