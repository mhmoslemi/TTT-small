import time
import numpy as np

def _evaluate(h):
    """Return C5 value for a step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_bounds_sum(h, target):
    """Clip to [0,1] and enforce sum == target (iterative water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(15):
        s = h.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h > 0
            if not mask.any():
                break
            delta = diff / mask.sum()
            h[mask] -= delta
        else:
            mask = h < 1
            if not mask.any():
                break
            delta = -diff / mask.sum()
            h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def _mutate(parent, sigma=0.07):
    """Gaussian mutation with repair."""
    child = parent + np.random.normal(0.0, sigma, size=parent.shape)
    target = parent.size / 2.0
    child = _project_bounds_sum(child, target)
    return child

def _interpolate(h, new_n):
    """Upsample h to length new_n by linear interpolation and repair."""
    old_n = h.size
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, h)
    target = new_n / 2.0
    new_h = _project_bounds_sum(new_h, target)
    return new_h
