import numpy as np, time, math, random

def _project_to_feasible(h, target_sum):
    """Clip to [0,1] and adjust to hit exact sum by a simple water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    # iterative scaling+clipping (few iterations enough for moderate n)
    for _ in range(5):
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        # scale proportionally, then re‑clip
        scale = target_sum / cur if cur != 0 else 0.0
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    # final fix: distribute remaining deficit/excess uniformly on free variables
    deficit = target_sum - h.sum()
    if abs(deficit) > 1e-12:
        free = (h > 0) & (h < 1)
        if free.any():
            h[free] += deficit / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _evaluate(h, dx):
    """Return max overlap C5 for step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _initial_feasible(n):
    """Create a random feasible vector of length n."""
    rng = np.random.default_rng()
    vec = rng.random(n)
    target = n / 2.0
    return _project_to_feasible(vec, target)
