import numpy as np
import time
import random

# Helper: evaluate overlap either via provided function or manual computation
def _score_h(h, dx):
    # h is a 1‑D numpy array
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _evaluate(h, dx):
    try:
        # evaluate_erdos_solution may be pre‑imported by the harness
        return evaluate_erdos_solution(h)
    except Exception:
        return _score_h(h, dx)


def _repair(vec, target_sum):
    """Project vec onto [0,1] box and enforce sum == target_sum."""
    v = np.clip(vec, 0.0, 1.0)
    s = v.sum()
    if abs(s - target_sum) < 1e-12:
        return v
    # iterative redistribution
    for _ in range(10):
        diff = target_sum - v.sum()
        free = (v > 0) & (v < 1)
        if not free.any():
            # all at bounds – scale globally
            v = np.clip(v * (target_sum / v.sum()), 0.0, 1.0)
            break
        adj = diff / free.sum()
        v[free] += adj
        v = np.clip(v, 0.0, 1.0)
        if abs(v.sum() - target_sum) < 1e-12:
            break
    return v

    Differential‑Evolution optimisation for the Erdős C₅ bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
    """
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size

        n = init.size
    else:
        # fallback: use a simple symmetric start with n=80
        n = 80
        init = np.full(n, 0.5)  # will be repaired to sum=n/2
    dx = 2.0 / n
    target_sum = n / 2.0

    # Repair the supplied initial vector
    init = _repair(init, target_sum)

    # DE parameters
    pop_size = max(20, int(0.7 * n))          # moderate population
    F = 0.5                                   # mutation factor
    CR = 0.9                                  # crossover probability
    max_gen = 1000                            # generous upper bound; time will stop earlier

    # Initialise population
    population = np.random.rand(pop_size, n)
    for i in range(pop_size):
        population[i] = _repair(population[i], target_sum)
    # Insert the known initial construction
    population[0] = init.copy()

    # Evaluate initial scores
    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):

    best_idx = np.argmin(scores)
    best_vec = population[best_idx].copy()
    best_score = scores[best_idx]

    start_time = time.time()
    time_limit = budget_s - 0.5   # safety margin

    gen = 0
    while gen < max_gen and (time.time() - start_time) < time_limit:
        gen += 1
        for i in range(pop_size):

            idxs = list(range(pop_size))
