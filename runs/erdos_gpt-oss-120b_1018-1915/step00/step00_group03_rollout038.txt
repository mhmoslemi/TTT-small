import numpy as np
import time

def _evaluate(h, dx):
    """Return max overlap integral C5 for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _fix_sum(vec, target, max_iter=10):
    """Project vec onto [0,1]^n while enforcing sum(vec)=target."""
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    n = len(vec)
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # indices that can move in the needed direction
        if diff > 0:
            cand = np.where(vec < 1.0)[0]
        else:
            cand = np.where(vec > 0.0)[0]
        if cand.size == 0:
            break
        i = np.random.choice(cand)
        if diff > 0:
            cap = 1.0 - vec[i]
            change = min(diff, cap)
        else:
            cap = -vec[i]
            change = max(diff, cap)
        vec[i] += change
        diff -= change
    # final clipping (safety)
    np.clip(vec, 0.0, 1.0, out=vec)
    return vec

def _interpolate(old, new_n):
    """Linear interpolation of old vector to length new_n."""
    old_n = len(old)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new = np.interp(x_new, x_old, old)
    return new

def _random_feasible(n, target_sum):
    """Generate a random vector in [0,1]^n with given sum."""
    vec = np.random.rand(n)
    # scale to target sum ignoring bounds, then project
    vec = vec / vec.sum() * target_sum
    return _fix_sum(vec, target_sum)
