import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def _project_to_feasible(x, desired_sum):
    """Clip to [0,1] and scale to meet the exact sum constraint."""
    v = np.clip(x, 0.0, 1.0)
    s = v.sum()
    if s == 0:
        return v
    scale = desired_sum / s
    v = v * scale
    # second clipping may disturb the sum slightly; renormalize once more
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if s == 0:
        return v
    v = v * (desired_sum / s)
    return v

def _c5_of_h(h, dx):
    """Return the overlap constant for a given discrete step function."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx

def _objective(x, n_points, dx, target_sum):
    """Objective for optimizers: feasibility‑projected C5."""
    h = _project_to_feasible(x, target_sum)
    return _c5_of_h(h, dx)
