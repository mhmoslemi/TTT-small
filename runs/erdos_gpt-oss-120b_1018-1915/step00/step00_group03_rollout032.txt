import time
import numpy as np
from math import exp

# Helper: compute C5 value for a candidate h
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# Helper: project h to [0,1] and enforce exact sum = n/2
def repair_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # all at bounds, distribute uniformly (may violate bounds a bit)
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            break
        add = diff / free.size
        h[free] += add
        h = np.clip(h, 0.0, 1.0)
    return h

# Simulated annealing on a fixed resolution
def simulated_annealing(h_start, time_budget, rng):
    n = h_start.shape[0]
    target_sum = n / 2.0
    h_cur = repair_h(h_start.copy(), target_sum)
    cur_score = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score

    start_time = time.time()
    max_move = 0.2  # maximal amount transferred per move
    while True:
        now = time.time()
        if now - start_time > time_budget:
            break
        elapsed = now - start_time
        # linear cooling
        T = max(1e-6, 0.2 * (1.0 - elapsed / time_budget))

        # propose a move: transfer delta from i to j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.uniform(-max_move, max_move)
        # adjust delta to respect bounds
        delta = max(delta, -h_cur[i])               # cannot go below 0 at i
        delta = min(delta, 1.0 - h_cur[i])          # cannot exceed 1 at i
        delta = max(delta, -(1.0 - h_cur[j]))       # cannot exceed 1 at j
        delta = min(delta, h_cur[j])                # cannot go below 0 at j
        if abs(delta) < 1e-12:
            continue

        h_new = h_cur.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # no need for extra repair: sum unchanged and bounds respected
        new_score = compute_c5(h_new)

        # accept if better or by Metropolis criterion
        if new_score < cur_score or rng.random() < exp(-(new_score - cur_score) / T):
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score

    return best_h, best_score
