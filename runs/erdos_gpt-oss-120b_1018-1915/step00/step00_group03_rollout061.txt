import time
import numpy as np

# Helper: compute the C5 value for a given step function h
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

# Project a vector onto the feasible set: 0<=h<=1 and sum(h)=n/2
def project_feasible(h):
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    # Distribute the remaining mass uniformly, respecting bounds
    # Simple iterative fix – works because diff is small after clipping
    while abs(diff) > 1e-12:
        adjustable = (h > 0.0) & (h < 1.0)
        if not np.any(adjustable):
            # fallback: add to first element within bounds
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            break
        delta = diff / adjustable.sum()
        # limit each step to keep within bounds
        step = np.where(adjustable, delta, 0.0)
        step = np.clip(step, -h, 1.0 - h)
        h += step
        cur = h.sum()
        diff = target - cur
    return h

# Produce a neighbor preserving the integral exactly via a mass transfer
def neighbor_transfer(h, rng):
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # decide direction randomly
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_transfer = min(h[src], 1.0 - h[dst])
    if max_transfer <= 0.0:
        # try opposite direction
        src, dst = dst, src
        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer <= 0.0:
            return h.copy()
    delta = rng.uniform(0.0, max_transfer)
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new

# Simple stochastic local search for a fixed time budget (seconds)
def hill_climb(start_h, time_limit, rng):
    best_h = project_feasible(start_h)
    best_val = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    start = time.time()
    no_improve = 0
    while time.time() - start < time_limit:
        cand = neighbor_transfer(cur_h, rng)
        cand_val = compute_c5(cand)
        if cand_val < cur_val:
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_h, best_val = cand.copy(), cand_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        # occasional random restart if stuck
        if no_improve >= 2000:
            cur_h = best_h + rng.uniform(-0.1, 0.1, size=best_h.shape)
            cur_h = project_feasible(cur_h)
            cur_val = compute_c5(cur_h)
            no_improve = 0
    return best_h, best_val

# Upsample a vector to a new length using linear interpolation
def upscale(h, new_n):
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new)
