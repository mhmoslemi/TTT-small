import time
import numpy as np

def _c5_value(h, dx):
    """Compute C5 for a given h vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project(h):
    """Project h onto [0,1] and enforce sum(h)=len(h)/2."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # Scale towards target then clip again
        factor = target / (s if s != 0 else 1.0)
        h *= factor
        h = np.clip(h, 0.0, 1.0)
    return h

def _initial_h(n_points):
    """Create an initial h of length n_points."""
    try:
        init = initial_h_values  # provided by harness if available
        init = np.asarray(init, dtype=float)
        if init.size == n_points:
            return _project(init.copy())
        # up‑sample / down‑sample linearly
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h_new = np.interp(x_new, x_old, init)
        return _project(h_new)
    except NameError:
        # fallback: uniform 0.5 satisfies the integral
        return np.full(n_points, 0.5, dtype=float)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] minimizing the overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best step function found (values in [0,1]).
    c5_bound : float
        The achieved C5 value.
    n_points : int
        Number of discretisation points used.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Choose a moderate resolution that balances speed and expressive power
    n_points = 200
    dx = 2.0 / n_points

    # Starting vector
    h_curr = _initial_h(n_points)
    best_h = h_curr.copy()
    best_c5 = _c5_value(best_h, dx)

    # Simulated‑annealing parameters
    temp_start = 1.0
    temp_end = 1e-4
    max_iters = 500000  # upper bound; loop will stop earlier by time budget
    step_std = 0.04      # scale of Gaussian proposals

    for it in range(max_iters):
        # Time‑budget check
        if time.time() - start_time > budget_s - 0.5:
            break

        # Linear cooling
        t = temp_start * (1 - it / max_iters) + temp_end * (it / max_iters)

        # Propose a new candidate
        proposal = h_curr + step_std * np.random.randn(n_points)
        proposal = _project(proposal)

        # Evaluate
        cand_c5 = _c5_value(proposal, dx)

        # Acceptance test
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = proposal.copy()
        delta = cand_c5 - _c5_value(h_curr, dx)
        if delta < 0 or np.exp(-delta / max(t, 1e-12)) > np.random.rand():
            h_curr = proposal

    return best_h, float(best_c5), n_points
