import time
import numpy as np

def _compute_c5(h):
    """Return the C5 value for a given step vector h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _repair(h, target_sum):
    """Project h onto the feasible set {0<=h<=1, sum=h_target}."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # distribute the difference uniformly, iterating to respect bounds
    for _ in range(15):
        if abs(diff) < 1e-12:
            break
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        new_cur = h.sum()
        new_diff = target_sum - new_cur
        if abs(new_diff - diff) < 1e-12:
            break
        diff = new_diff
    # final scaling if still off (rare)
    cur = h.sum()
    if abs(cur - target_sum) > 1e-8:
        h = np.clip(h * (target_sum / cur), 0.0, 1.0)
    return h

def _random_feasible(n):
    """Generate a random feasible vector of length n."""
    # start with uniform random numbers and scale to required sum
    h = np.random.rand(n)
    h = _repair(h, n / 2.0)
    return h

def _upscale(h, new_n):
    """Linear interpolation to increase resolution of h to length new_n."""
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.95  # safety margin

    # ------------------------------------------------------------------
    # Determine initial resolution and vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base_h = globals()["initial_h_values"].astype(float)
    else:
        base_h = _random_feasible(42)

    base_n = len(base_h)
    # Target resolution (keep reasonably small for speed)
    target_n = max(200, base_n)
    if target_n % 2:
        target_n += 1

    if base_n != target_n:
        h = _upscale(base_h, target_n)
        h = _repair(h, target_n / 2.0)
    else:
        h = base_h.copy()

    n = len(h)
    target_sum = n / 2.0

    best_h = h
    best_score = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # Simple stochastic local search
    # ------------------------------------------------------------------
    sigma = 0.2  # initial mutation strength
    iter_count = 0
    while time.time() - start_time < time_limit:

        # Adaptive sigma: decay slowly

            sigma = max(0.02, sigma * 0.9)

        # Propose mutation

        cand = _repair(cand, target_sum)
        cand_score = _compute_c5(cand)

        if cand_score < best_score:
            best_h, best_score = cand, cand_score
