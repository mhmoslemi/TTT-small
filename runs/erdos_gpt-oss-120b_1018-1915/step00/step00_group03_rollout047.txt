import time
import numpy as np

def _eval_c5(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_sum_box(x, target):
    """Project vector x onto the intersection of box[0,1] and sum=target."""
    # Clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # Simple iterative redistribution to satisfy sum
    for _ in range(10):
        s = x.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # indices that can move in direction of diff
        if diff > 0:
            idx = np.where(x < 1.0)[0]
        else:
            idx = np.where(x > 0.0)[0]
        if len(idx) == 0:
            break
        # distribute equally, but respect bounds
        inc = diff / len(idx)
        inc = np.clip(inc, -0.5, 0.5)  # safety
        x[idx] += inc
        x = np.clip(x, 0.0, 1.0)
    # final tiny correction on a random free index
    s = x.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        free = np.where((x > 0.0) & (x < 1.0))[0]
        if free.size == 0:
            free = np.arange(len(x))
        i = np.random.choice(free)
        x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def _random_feasible(n, rng):
    target = n / 2.0
    # start with Dirichlet then scale
    x = rng.dirichlet(np.ones(n))
    x = x * target
    x = np.clip(x, 0.0, 1.0)
    return _project_sum_box(x, target)

def _perturb(h, rng, max_step):
    """Move a small amount from i to j while keeping constraints."""
    n = len(h)
    # pick i with positive value, j with room
    pos_idx = np.where(h > 1e-12)[0]
    if pos_idx.size == 0:
        return h.copy()
    i = rng.choice(pos_idx)
    neg_idx = np.where(h < 1.0 - 1e-12)[0]
    if neg_idx.size == 0:
        return h.copy()
    j = rng.choice(neg_idx)
    delta = max_step * rng.random()
    delta = min(delta, h[i])
    delta = min(delta, 1.0 - h[j])
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def _upscale(h):
    """Double resolution by repeating each sample."""
    return np.repeat(h, 2)
