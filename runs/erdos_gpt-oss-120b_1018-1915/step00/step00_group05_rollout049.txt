import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def _repair_solution(h, target_sum):
    """Project h onto [0,1] and enforce exact sum by a simple redistribution."""
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target_sum
    if abs(diff) > 1e-12:
        # distribute the excess uniformly (may push some values out of bounds)
        h -= diff / h.size
        h = np.clip(h, 0.0, 1.0)
        # final correction of remaining tiny error
        diff = np.sum(h) - target_sum
        if abs(diff) > 1e-12:
            h -= diff / h.size
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the step function h on [0,2] to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine resolution
    n_points = kwargs.get('n_points')
    if n_points is None:
        if 'initial_h_values' in globals():
            n_points = len(initial_h_values)
        else:
            n_points = 120  # default moderate size

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum(h)*dx = 1

    # Initial candidate
    if 'initial_h_values' in globals() and len(initial_h_values) == n_points:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # random start respecting bounds, then project to sum
        h0 = rng.random(n_points)
        h0 = _repair_solution(h0, target_sum)

    # Objective with penalty for sum constraint
    penalty_weight = 5e2

    def objective_vec(vec):
        # vec is raw DE variable inside [0,1]
        h = np.array(vec, dtype=float)
        c5 = _compute_c5(h, dx)
        sum_err = np.sum(h) - target_sum
        return c5 + penalty_weight * (sum_err ** 2)

    # Time budgeting
    time_left = lambda: budget_s - (time.time() - start_time)

    # --- Global search with Differential Evolution ---
    de_maxiter = 200  # generous but will be cut by time check
    popsize = 10

    bounds = [(0.0, 1.0)] * n_points
    de_result = None

    # Run DE respecting remaining time
    if time_left() > 5.0:
        de_result = differential_evolution(
            objective_vec,
            bounds,
            strategy='best1bin',
            maxiter=de_maxiter,
            popsize=popsize,
            polish=False,
            seed=seed,
            updating='deferred',
            workers=1,
            tolerance=1e-6,
            init='random',
        )
        h_candidate = _repair_solution(de_result.x, target_sum)
    else:
        h_candidate = h0.copy()

    # --- Local refinement with SLSQP ---
    def local_obj(h):
        return _compute_c5(h, dx)

    eq_cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
    bounds_slsqp = [(0.0, 1.0)] * n_points

    if time_left() > 2.0:
        res = minimize(
            local_obj,
            h_candidate,
            method='SLSQP',
            bounds=bounds_slsqp,
            constraints=[eq_cons],
            options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False},
        )
        if res.success:
            h_candidate = _repair_solution(res.x, target_sum)

    # Final evaluation
    c5_bound = _compute_c5(h_candidate, dx)

    # Ensure feasibility before returning
    h_candidate = _repair_solution(h_candidate, target_sum)

    return h_candidate, float(c5_bound), n_points
