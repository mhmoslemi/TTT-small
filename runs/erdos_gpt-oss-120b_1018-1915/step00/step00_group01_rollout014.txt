import time
import numpy as np

def _project_to_constraints(v, target_sum):
    """Project vector v onto the box [0,1] and the affine constraint sum = target_sum."""
    h = np.clip(v, 0.0, 1.0)
    # Iteratively adjust using a water‑filling style update
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            # no free variables – distribute diff uniformly and re‑clip
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free.sum()
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # Final tiny correction by adding a uniform offset and re‑clipping once more
    s = h.sum()
    if abs(s - target_sum) > 1e-12:
        offset = (target_sum - s) / len(h)
        h += offset
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def _initial_h(n_points):

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with values in [0,1] and integral 1

    """
    start_time = time.time()
    np.random.seed(seed)

    # grid size – a compromise between resolution and speed
    n_points = 300  # even, gives target sum = 150.0
    target_sum = n_points / 2.0

    # initialise from parent if available
    current = _initial_h(n_points)
    best = current.copy()
    best_val = _c5_value(best)

    # annealing parameters
    max_iters = 30000
    sigma = 0.15          # initial perturbation scale
    decay = 0.9995        # exponential decay per iteration

    for it in range(max_iters):
        # time guard
        if time.time() - start_time > budget_s * 0.95:

        proposal = _project_to_constraints(proposal, target_sum)
