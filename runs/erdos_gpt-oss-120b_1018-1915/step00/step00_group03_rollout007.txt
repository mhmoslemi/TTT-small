import time
import numpy as np

def _evaluate(h, dx):
    """Return max overlap C5 for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_initial(h):
    """Adjust h minimally so that sum(h)=len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    # try to add diff to a single entry that can accommodate it
    for i in range(n):
        val = h[i] + diff
        if 0.0 <= val <= 1.0:
            h[i] = val
            return h
    # fallback: distribute diff uniformly (diff is tiny)
    h += diff / n
    np.clip(h, 0.0, 1.0, out=h)
    # re‑balance if needed
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        # simple two‑point redistribution
        i, j = np.argmax(h), np.argmin(h)
        delta = (target - cur) / 2.0
        h[i] = np.clip(h[i] + delta, 0.0, 1.0)
        h[j] = np.clip(h[j] - delta, 0.0, 1.0)
    return h

def _propose_transfer(h, rng, max_step):
    """Return a new h where a mass delta is moved from j to i, preserving sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # maximum admissible positive delta (i gets, j loses)
    max_pos = min(max_step, 1.0 - h[i], h[j])
    # maximum admissible negative delta (i loses, j gets)
    max_neg = min(max_step, h[i], 1.0 - h[j])
    if max_pos <= 0 and max_neg <= 0:
        return h.copy()          # no feasible move
    if rng.random() < 0.5:
        delta = rng.random() * max_pos
    else:
        delta = -rng.random() * max_neg
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős C5 bound.

    Returns
    -------
    h_values : list of floats
        The best discovered h sampled at n points.
    c5_bound : float
        The corresponding C5 value.
    n_points : int
        Number of discretisation points.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: uniform distribution respecting the integral constraint
        n = 70
        h = np.full(n, 0.5, dtype=np.float64)   # sum = 0.5*n = n/2  OK
    n = h.size
    h = _project_initial(h)
    dx = 2.0 / n

    best_h = h.copy()
    best_val = _evaluate(best_h, dx)

    # ------------------------------------------------------------------
    # Annealing parameters
    # ------------------------------------------------------------------
    max_iter = 10**7                     # safety upper bound
    temp0 = 0.005                        # initial temperature
    temp_final = 1e-6
    step0 = 0.2                          # initial max transfer amount
    step_final = 0.001

    for it in range(max_iter):
        # time check
        if time.time() - start_time > budget_s * 0.97:
            break

        # linear cooling schedules
        frac = it / max_iter
        temp = temp0 * (1 - frac) + temp_final * frac
        max_step = step0 * (1 - frac) + step_final * frac

        # propose a neighbour
        h_new = _propose_transfer(h, rng, max_step)
        val_new = _evaluate(h_new, dx)

        # acceptance test (minimise)
        delta = val_new - best_val
        if val_new < best_val:
            best_h = h_new.copy()
            best_val = val_new
        if val_new < _evaluate(h, dx) or rng.random() < np.exp(-delta / max(temp, 1e-12)):
            h = h_new

    return best_h.tolist(), best_val, n
