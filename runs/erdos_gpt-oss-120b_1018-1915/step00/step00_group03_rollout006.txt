import time, math, random
import numpy as np

def _evaluate(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated by discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that lowers the
    Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Array of length n (values in {0,1}) with sum = n/2.
    c5_bound : float
        Achieved upper bound C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = initial_h_values.astype(float)
        base_n = init.shape[0]
    else:
        init = None
        base_n = 70

    # use a modest but sufficiently fine grid
    n = max(140, base_n * 2)          # at least 140 points
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Helper to create an initial binary vector with half ones
    # ------------------------------------------------------------------
    def make_initial() -> np.ndarray:
        if init is not None and init.shape[0] == n:
            h0 = init.copy()
        elif init is not None:
            # simple up‑sampling by repetition
            factor = n // init.shape[0]
            remainder = n % init.shape[0]
            h0 = np.repeat(init, factor)
            if remainder:
                h0 = np.concatenate([h0, init[:remainder]])
        else:
            h0 = np.zeros(n)

        # enforce binary 0/1 and exact half‑ones
        h0 = np.where(h0 > 0.5, 1.0, 0.0)
        ones_needed = n // 2
        current_ones = int(h0.sum())
        if current_ones > ones_needed:
            idx = np.flatnonzero(h0)[: current_ones - ones_needed]
            h0[idx] = 0.0
        elif current_ones < ones_needed:
            idx = np.flatnonzero(1.0 - h0)[: ones_needed - current_ones]
            h0[idx] = 1.0
        return h0

    # ------------------------------------------------------------------
    # Simulated‑annealing style search on binary vectors
    # ------------------------------------------------------------------
    best_h = make_initial()
    best_score = _evaluate(best_h, dx)

    # keep a working copy that will be mutated
    h = best_h.copy()

    # annealing schedule
    T0 = 0.02
    T = T0
    it = 0
    last_improve_it = 0

    # main loop, stop a little before the allotted time
    while time.time() - start_time < budget_s * 0.95:
        # pick two distinct indices with opposite values
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        if h[i] == h[j]:
            continue                     # no effect, skip

        # propose a swap
        h[i], h[j] = h[j], h[i]

        new_score = _evaluate(h, dx)
        delta = new_score - best_score

        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
            last_improve_it = it
        else:
            # Metropolis acceptance
            prob = math.exp(-delta / T) if T > 1e-12 else 0.0
            if random.random() >= prob:
                # reject: revert swap
                h[i], h[j] = h[j], h[i]

        it += 1

        # cooling every n swaps
        if it % (n * 10) == 0:
            T *= 0.99
            # occasional temperature reset if frozen
            if T < 1e-6:
                T = T0

        # restart if no improvement for a while
        if it - last_improve_it > n * 2000:
            # random re‑initialisation preserving the half‑ones constraint
            h = make_initial()
            T = T0
            last_improve_it = it

    return best_h, float(best_score), n
