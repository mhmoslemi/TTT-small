import time
import numpy as np
from scipy import optimize as opt

def _project_to_feasible(y, target):
    """
    Project vector y onto the convex set:
        0 <= x_i <= 1  and  sum(x) = target
    Using a simple bisection on the Lagrange multiplier.
    """
    # start with unclipped y
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)


def _objective(raw_vec, target):
    """
    Objective for DE: project raw_vec to feasible region then evaluate C5.
    """
    vec = _project_to_feasible(raw_vec, target)
    # Pre‑imported helper evaluates the overlap integral.
    return evaluate_erdos_solution(vec)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size from the provided initial vector if present.
    try:
        n_points = int(len(initial_h_values))
        init_vector = np.asarray(initial_h_values, dtype=float)
    except NameError:
        n_points = 80
        init_vector = np.full(n_points, 0.5)  # fallback uniform start

    target_sum = n_points / 2.0          # because integral must be 1, dx = 2/n
    bounds = [(0.0, 1.0)] * n_points

    # Callback to respect the overall time budget.
    def _time_callback(xk, convergence):
        return (time.time() - start_time) > budget_s * 0.95

    # Build an initial population: first individual = feasible projection of the given vector,
    # the rest are random Latin‑Hypercube points.
    popsize = 15
    init_pop = opt._differential_evolution._init_pop(
        bounds, popsize, np.random, init='latinhypercube')
    init_pop[0] = _project_to_feasible(init_vector, target_sum)

    # Run Differential Evolution.
    result = opt.differential_evolution(
        func=lambda v: _objective(v, target_sum),
        bounds=bounds,
        strategy='best1bin',
        maxiter=200,
        popsize=popsize,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        polish=False,
        init=init_pop,
        callback=_time_callback,
        disp=False,
        atol=0,
        tol=0,
    )

    # Ensure feasibility of the returned solution.
    best_vec = _project_to_feasible(result.x, target_sum)
    best_score = evaluate_erdos_solution(best_vec)

    # In the unlikely case time ran out before DE produced a result,
    # fall back to the initial feasible vector.
    if best_score is None or np.isnan(best_score):
        best_vec = _project_to_feasible(init_vector, target_sum)
        best_score = evaluate_erdos_solution(best_vec)

    return best_vec, float(best_score), n_points
