import time
import math
import numpy as np

# The harness provides evaluate_erdos_solution and possibly initial_h_values
# We use them directly without redefining.

def _objective(h, dx):
    """Compute C5 for sample vector h."""
    # Use the provided helper if available, otherwise compute directly.
    try:
        return evaluate_erdos_solution(h)
    except Exception:
        # fallback direct computation
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

def _make_feasible(vec, target_sum):
    """Scale and clip to satisfy sum and bounds exactly."""
    s = vec.sum()
    if s == 0:
        vec = np.full_like(vec, target_sum / vec.size)
    else:
        vec = vec * (target_sum / s)
    np.clip(vec, 0.0, 1.0, out=vec)
    # After clipping, renormalize the small residual to keep sum exact
    residual = target_sum - vec.sum()
    if abs(residual) > 1e-12:
        # distribute residual uniformly (will stay within bounds because residual is tiny)
        vec += residual / vec.size
        np.clip(vec, 0.0, 1.0, out=vec)
    return vec

def _mutate(h):
    """Return a new feasible vector by a small random redistribution."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # propose a new value for position i
    new_i = np.random.rand()
    delta = new_i - h[i]
    new_j = h[j] - delta
    # if new_j stays in [0,1], accept the move; otherwise fall back to simple swap
    if 0.0 <= new_j <= 1.0:
        new_h = h.copy()
        new_h[i] = new_i
        new_h[j] = new_j
        return new_h
    # swap as safe fallback
    new_h = h.copy()
    new_h[i], new_h[j] = h[j], h[i]
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) after a timed simulated‑annealing search."""
    np.random.seed(seed)
    start_time = time.time()
    # Determine discretisation
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 45
        h0 = np.full(n_points, 0.5, dtype=float)  # dummy uniform start
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure feasibility of the starting vector
    h_cur = _make_feasible(h0.copy(), target_sum)
    cur_score = _objective(h_cur, dx)

    best_h = h_cur.copy()
    best_score = cur_score

    # Simulated‑annealing parameters
    T0 = 5.0
    cooling = 0.9995
    iter_count = 0
    max_iter = 10_000_000  # safety cap

    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        if iter_count >= max_iter:
            break
        T = T0 * (cooling ** iter_count)
        # generate neighbour
        h_new = _mutate(h_cur)
        # enforce feasibility (sum may drift by tiny epsilon due to floating error)
        h_new = _make_feasible(h_new, target_sum)
        new_score = _objective(h_new, dx)

        # acceptance test
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability of accepting worse move
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            h_cur = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score

        iter_count += 1

    # final sanity: ensure exact sum constraint
    best_h = _make_feasible(best_h, target_sum)

    return best_h, best_score, n_points
