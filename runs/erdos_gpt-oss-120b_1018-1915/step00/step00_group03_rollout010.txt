import time
import numpy as np

def _compute_c5(h, dx):
    """Maximum overlap integral for a given step function h."""
    # correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _initial_binary(n, rng):
    """Create a binary vector with exactly n/2 ones, shuffled."""
    ones = np.ones(n // 2, dtype=np.float64)
    zeros = np.zeros(n - n // 2, dtype=np.float64)
    h = np.concatenate([ones, zeros])
    rng.shuffle(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimises
    C_5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, best_c5, n_points).
    """
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine number of sampling points.
    # Prefer the provided initial_h_values if present, otherwise fall back
    # to a modest size (n = 70, an even number).
    # ------------------------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        n = init.size
        # enforce even size for exact half‑integral constraint
        if n % 2 == 1:
            n += 1
    except NameError:
        n = 70
    dx = 2.0 / n
    target_sum = n / 2  # because ∫ h = 1 ⇒ Σ h * dx = 1 ⇒ Σ h = n/2

    # ------------------------------------------------------------------
    # Initialise a binary candidate (exactly half ones) – this satisfies
    # the constraints automatically.
    # ------------------------------------------------------------------
    h = _initial_binary(n, rng)
    best_h = h.copy()
    best_c = _compute_c5(best_h, dx)

    # ------------------------------------------------------------------
    # Simple stochastic hill‑climbing with occasional random restarts.
    # ------------------------------------------------------------------
    iteration = 0
    temp_start = 0.02
    temp_end = 0.0005
    while time.time() < deadline:
        iteration += 1

        # linear cooling schedule
        t_frac = iteration / 50000.0
        temp = temp_start * (1 - t_frac) + temp_end * t_frac

        # propose a swap of a 1 and a 0
        ones_idx = np.flatnonzero(h == 1.0)
        zeros_idx = np.flatnonzero(h == 0.0)
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        h_trial = h.copy()
        h_trial[i] = 0.0
        h_trial[j] = 1.0

        c_trial = _compute_c5(h_trial, dx)

        # accept if improvement or by Metropolis criterion
        if c_trial < best_c or rng.random() < np.exp((best_c - c_trial) / max(temp, 1e-12)):
            h = h_trial
            if c_trial < best_c:
                best_c = c_trial
                best_h = h_trial.copy()

        # periodic random restart to escape deep local minima
        if iteration % 20000 == 0:
            h = _initial_binary(n, rng)

        # stop early if we already beat the target bound
        if best_c < 0.3808:
            break

    # ------------------------------------------------------------------
    # Return the best found solution.
    # ------------------------------------------------------------------
    return best_h, best_c, n
