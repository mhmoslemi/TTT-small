import time
import numpy as np
from scipy.optimize import minimize

def _objective(h, dx):
    """Return the overlap constant C5 for a given h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx

def _sum_constraint(h, target):
    return np.sum(h) - target

def _project(h, target):
    """Project vector onto box [0,1] and exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    total = np.sum(h)
    if abs(total - target) < 1e-12:
        return h
    diff = total - target
    # indices that are not at a bound can absorb the difference
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size:
        adj = diff / free.size
        h[free] -= adj
        h = np.clip(h, 0.0, 1.0)
    else:
        # fallback: uniform scaling
        h *= target / total
    # final tiny correction on the first element
    final_diff = np.sum(h) - target
    if abs(final_diff) > 1e-12:
        h[0] -= final_diff
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    np.random.seed(seed)
    start_time = time.time()

    # ----- problem size -------------------------------------------------
    if "initial_h_values" in globals():
        init_vec = np.asarray(initial_h_values, dtype=float)
        n_points = init_vec.size
    else:
        n_points = kwargs.get("n_points", 70)
        init_vec = np.full(n_points, 0.5, dtype=float)  # feasible constant start

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ----- make sure initial vector satisfies constraints -------------
    h_current = _project(init_vec, target_sum)
    best_h = h_current.copy()
    best_c5 = _objective(best_h, dx)

    # ----- local SLSQP refinement ----------------------------------------
    bounds = [(0.0, 1.0)] * n_points
    cons = {"type": "eq", "fun": lambda x: _sum_constraint(x, target_sum)}
    opts = {"maxiter": 1500, "ftol": 1e-12, "disp": False}

    res = minimize(
        lambda x: _objective(x, dx),
        h_current,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    if res.success:
        h_opt = _project(res.x, target_sum)
        c5_opt = _objective(h_opt, dx)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
            h_current = h_opt.copy()

    # ----- simple stochastic hill‑climbing --------------------------------
    # parameters for the random walk
    max_iters = 500000
    step_scale = 0.2  # typical magnitude of a change

    i = 0
    while time.time() - start_time < budget_s * 0.94 and i < max_iters:
        i += 1
        # propose a move that keeps the sum exactly constant
        idx_i, idx_j = np.random.choice(n_points, 2, replace=False)
        new_val = np.random.rand()  # in [0,1]
        delta = new_val - h_current[idx_i]
        # adjust j to keep total sum unchanged
