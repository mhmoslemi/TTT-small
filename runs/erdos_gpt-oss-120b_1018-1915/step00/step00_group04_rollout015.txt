def _project_to_bounds_sum(v, lower, upper, target_sum, max_iter=1000, tol=1e-9):
    """
    Project vector v onto the box [lower, upper] with a fixed L1 sum = target_sum.
    Uses a simple water-filling algorithm.
    """
    v = np.clip(v, lower, upper)
    for _ in range(max_iter):
        cur_sum = v.sum()
        diff = cur_sum - target_sum
        if abs(diff) <= tol:
            break
        # indices that can still move (not at a bound)
        free = (v > lower + tol) & (v < upper - tol)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds; distribute remainder uniformly (may violate bounds slightly)
            v += diff / len(v)
            v = np.clip(v, lower, upper)
            break
        adjustment = diff / n_free
        v[free] -= adjustment
        v = np.clip(v, lower, upper)
    # final tiny correction to meet the sum exactly (if possible within bounds)
    cur_sum = v.sum()
    if abs(cur_sum - target_sum) > tol:
        # distribute the remaining error uniformly and clip again
        v += (target_sum - cur_sum) / len(v)
        v = np.clip(v, lower, upper)
    return v
