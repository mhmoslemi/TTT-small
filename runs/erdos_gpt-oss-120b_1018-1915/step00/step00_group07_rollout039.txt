import time
import numpy as np

def _repair(h, target_sum):
    """Project h onto [0,1]^n with sum = target_sum using simple iterative scaling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # indices that are not at a bound
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size == 0:
            # distribute uniformly if everything is clamped
            h += diff / h.size
        else:
            h[free] += diff / free.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _objective(h, dx):
    """Return the C5 value for a given h vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Differential‑evolution search for a step function h on [0,2]
    that minimizes the Erdős overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points) :
        h_values – numpy array of the best h found
        c5_bound – corresponding C5 value
        n_points – number of discretisation points (len(h_values))
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 80
        h0 = np.full(n_points, 0.5, dtype=float)  # feasible start

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Parameters for DE
    pop_size = max(30, n_points // 2)          # modest population
    F = 0.5                                     # mutation factor
    CR = 0.9                                    # crossover probability
    max_gen = 5000                              # safety upper bound

    # Initialise population around the seed vector
    population = np.empty((pop_size, n_points), dtype=float)
    for i in range(pop_size):
        if i == 0:
            candidate = h0.copy()
        else:
            candidate = rng.random(n_points)
            candidate = _repair(candidate, target_sum)
        population[i] = candidate

    # Pre‑evaluate
    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        scores[i] = _objective(population[i], dx)

    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    start_time = time.time()
    gen = 0
    while gen < max_gen and (time.time() - start_time) < budget_s:
        gen += 1
        for i in range(pop_size):
            # mutation: choose three distinct others
            idxs = rng.choice(pop_size, 3, replace=False)
            a, b, c = idxs
            mutant = population[a] + F * (population[b] - population[c])
            # crossover (binomial)
            cross = np.empty(n_points, dtype=float)
            jrand = rng.integers(n_points)
            for j in range(n_points):
                if rng.random() < CR or j == jrand:
                    cross[j] = mutant[j]
                else:
                    cross[j] = population[i, j]
            child = _repair(cross, target_sum)
            child_score = _objective(child, dx)

            # selection
            if child_score <= scores[i]:
                population[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_h = child.copy()
                    # early stop if target achieved
                    if best_score < 0.3808:
                        return best_h, best_score, n_points
        # optional: reshuffle best to front (not required)

    return best_h, best_score, n_points
