import numpy as np, time, math, random

def compute_c5(h):
    """Return the discretized C5 for a step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def repair_sum(h, desired):
    """Adjust h in-place so that sum(h) == desired while keeping bounds [0,1]."""
    total = h.sum()
    diff = desired - total
    # simple iterative correction on random positions
    for _ in range(10):

            break

        if diff > 0:
            # need to add mass

        else:
            # need to remove mass

    total = h.sum()
    if abs(total - desired) > 1e-9:
        h *= (desired / total)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample_h(parent_h, new_n):
    """Linear interpolation of parent_h (length m) to length new_n, then repair sum."""
