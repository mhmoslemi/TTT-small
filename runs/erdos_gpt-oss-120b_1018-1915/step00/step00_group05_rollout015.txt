import time
import numpy as np

def _project_onto_feasible(v, target):
    """Project vector v onto the set {0<=x<=1, sum=x_target} using bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return proj

def _make_feasible(vec, n):
    """Clip to [0,1] and enforce exact sum = n/2."""
    target = n * 0.5
    vec = np.clip(vec, 0.0, 1.0)
    return _project_onto_feasible(vec, target)

def _c5_score(h):
    """Discrete version of C5 = max_k ∫ h(x)(1‑h(x+k)) dx."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1, 0≤h≤1
    to minimize C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine resolution
    n_points = kwargs.get("n_points", None)
    if n_points is None:
        # Prefer the length of the provided parent if present
        try:
            n_points = len(initial_h_values)
        except NameError:
            n_points = 70
    else:
        n_points = int(n_points)

    # Build initial feasible candidate
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.size == n_points:
            h_cur = _make_feasible(parent, n_points)
        else:
            # size mismatch – fall back to random
            h_cur = np.random.rand(n_points)
            h_cur = _make_feasible(h_cur, n_points)
    else:
        h_cur = np.random.rand(n_points)
        h_cur = _make_feasible(h_cur, n_points)

    best_h = h_cur.copy()
    best_score = _c5_score(best_h)

    # Simple stochastic hill‑climbing within time budget
    # Parameters
    max_iter = 200000
    delta_scale = 0.05  # typical move size
    target_sum = n_points * 0.5

    it = 0
    while time.time() - start_time < budget_s * 0.95 and it < max_iter:
        # propose a small transfer between two coordinates
        i, j = np.random.randint(0, n_points, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * 2.0 * delta_scale  # in [-scale,scale]
        candidate = best_h.copy()
        candidate[i] -= delta
        candidate[j] += delta
        # enforce box then exact sum
        candidate = np.clip(candidate, 0.0, 1.0)
        # after clipping the sum may deviate; correct with projection
        candidate = _make_feasible(candidate, n_points)

        score = _c5_score(candidate)
        if score < best_score:
            best_h = candidate
            best_score = score
        it += 1

    # Final sanity checks
    assert best_h.shape[0] == n_points
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
