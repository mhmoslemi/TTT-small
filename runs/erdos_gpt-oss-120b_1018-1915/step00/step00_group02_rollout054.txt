import time
import numpy as np
from math import exp

def _c5_score(h):
    """Return C5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_move(h, rng):
    """Propose a sum‑preserving move: transfer ε from j to i."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    hi, hj = h[i], h[j]
    # feasible ε so that 0≤hi+ε≤1 and 0≤hj-ε≤1
    low = max(-hi, hj - 1.0)
    high = min(hj, 1.0 - hi)
    if high <= low:
        return None, None, None
    eps = rng.uniform(low, high)
    return i, j, eps
