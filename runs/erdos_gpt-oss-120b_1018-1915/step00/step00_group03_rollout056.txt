import time
import numpy as np

def _evaluate(h, dx):
    """Return max overlap C5 for step function h."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def _local_search(h, dx, budget, start_time, rng):
    """Simple hill‑climbing with random mass transfers."""
    n = h.size
    best_h = h.copy()
    best_score = _evaluate(best_h, dx)

    step_size = 0.02          # typical transfer magnitude
    no_improve = 0
    max_no_improve = 2000    # restart if stuck

    while time.time() - start_time < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a transfer delta from i to j
        delta = (rng.random() * 2 - 1) * step_size
        # clip to keep both variables in [0,1]
        delta = np.clip(delta, -best_h[i], 1.0 - best_h[j])

        if abs(delta) < 1e-12:
            continue

        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        score = _evaluate(new_h, dx)
        if score < best_score - 1e-12:
            best_h = new_h
            best_score = score
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max_no_improve:
            # random perturbation to escape stagnation
            idx = rng.integers(0, n, size=5)
            perturb = (rng.random(5) - 0.5) * 0.1
            new_h = best_h.copy()
            new_h[idx] += perturb
            new_h = np.clip(new_h, 0.0, 1.0)
            # re‑project to satisfy sum constraint exactly
            excess = new_h.sum() - n / 2.0
            if abs(excess) > 1e-12:
                new_h -= excess / n
                new_h = np.clip(new_h, 0.0, 1.0)
            score = _evaluate(new_h, dx)
            if score < best_score:
                best_h = new_h
                best_score = score
            no_improve = 0

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with integral 1 to minimize
    C5 = max_k ∫ h(x)(1-h(x+k))dx.

    Returns
    -------
    h_values : list of float
        Optimized discretized values of h.
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1. Choose base resolution (70 points if an initial solution is given,
    #    otherwise 80 points) and obtain a start vector.
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        n0 = len(init)
        h0 = np.array(init, dtype=float)
    else:
        n0 = 80
        h0 = rng.random(n0)
        h0 = np.clip(h0, 0.0, 1.0)
        # enforce the integral constraint exactly
        h0 -= (h0.sum() - n0 / 2.0) / n0
        h0 = np.clip(h0, 0.0, 1.0)

    # ensure sum constraint after clipping
    h0 -= (h0.sum() - n0 / 2.0) / n0
    h0 = np.clip(h0, 0.0, 1.0)

    # ------------------------------------------------------------------
    # 2. Run local search at this resolution until half of the budget.
    # ------------------------------------------------------------------
    half_budget = budget_s * 0.5
    dx0 = 2.0 / n0
    h_best, best_score = _local_search(h0, dx0, half_budget, start, rng)

    # ------------------------------------------------------------------
    # 3. If time permits, double the resolution and continue refining.
    # ------------------------------------------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:
        n1 = n0 * 2
        # linear interpolation to higher resolution
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n1, endpoint=False)

        # project onto feasible set
        h_interp = np.clip(h_interp, 0.0, 1.0)

        h_interp = np.clip(h_interp, 0.0, 1.0)

            h_best = h_refined
            best_score = refined_score
            n0 = n1
            dx0 = dx1

    # ------------------------------------------------------------------
    # 4. Return results in the required format.
    # ------------------------------------------------------------------
    return (h_best.tolist(), float(best_score), int(n0))
