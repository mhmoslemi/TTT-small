import time
import numpy as np
from scipy.signal import correlate

# evaluate_erdos_solution and initial_h_values are provided in the environment.

def compute_c5(h):
    """Return the overlap bound C5 for a discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def upsample_h(old_h, new_n):
    """Linearly interpolate old_h onto a new grid of size new_n and renormalise."""
    old_n = len(old_h)
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, old_h)
    h_new = np.clip(h_new, 0.0, 1.0)
    target_sum = new_n / 2.0
    cur_sum = h_new.sum()
    if cur_sum > 0:
        h_new *= target_sum / cur_sum
    else:
        h_new.fill(target_sum / new_n)
    h_new = np.clip(h_new, 0.0, 1.0)
    # final correction to enforce exact sum (tiny numeric drift)
    deficit = target_sum - h_new.sum()
    if abs(deficit) > 1e-12:
        # distribute to first element that can absorb it
        for i in range(new_n):
            add = np.clip(deficit, -h_new[i], 1.0 - h_new[i])
            h_new[i] += add
            deficit -= add
            if abs(deficit) < 1e-12:
                break
    return h_new

def random_feasible_h(n, rng):
    """Generate a random vector in [0,1] with sum = n/2."""
    # start with uniform random then project to simplex with upper bound 1
    # Simple rejection method: draw until sum is close.
    target = n / 2.0
    while True:
        h = rng.random(n)
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s == 0:
            continue
        h *= target / s
        if h.max() <= 1.0:
            return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős overlap constant C5.

    Returns:
        h_values (np.ndarray): the best step heights found.
        c5_bound (float): the achieved C5 value.
        n_points (int): number of discretisation points.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a working resolution.
    # Use the provided construction if present, otherwise fall back to 200 points.
    if 'initial_h_values' in globals():
        base_h = np.asarray(initial_h_values, dtype=float)
        base_n = len(base_h)
        n_points = max(base_n, 200)
    else:
        base_h = None
        n_points = 200

    # Prepare initial vector
    if base_h is not None:
        h_curr = upsample_h(base_h, n_points)
    else:
        h_curr = random_feasible_h(n_points, rng)

    best_h = h_curr.copy()
    best_c5 = compute_c5(best_h)

    # Parameters for the simple stochastic local search
    max_iters = 10_000_000  # effectively unlimited, bounded by time
    iter_count = 0
    no_improve = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # propose a move: transfer a small amount between two coordinates
        i, j = rng.integers(0, n_points, size=2)
        if i == j:
            continue
        # random transfer magnitude

                best_c5 = cand_c5

    for _ in range(3):
