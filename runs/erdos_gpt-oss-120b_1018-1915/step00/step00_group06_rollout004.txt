import time
import numpy as np
import math

def _compute_c5(h):
    """Return the max overlap integral for a given discrete h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _initial_vector(n, seed):
    """Create a feasible starting vector with sum == n/2 and values in [0,1]."""
    rng = np.random.default_rng(seed)
    # Dirichlet ensures positivity; scale to meet sum constraint
    v = rng.random(n)
    v = np.clip(v, 0.0, 1.0)
    # Adjust to exact sum n/2 by linear scaling and residual redistribution
    target = n / 2.0
    scale = target / v.sum()
    v = v * scale
    # Clip again and fix any leftover mass
    excess = v.sum() - target
    if abs(excess) > 1e-12:
        # distribute excess uniformly respecting bounds
        mask = (v > 0) & (v < 1)
        if mask.any():
            adjust = excess / mask.sum()
            v[mask] -= adjust
            v = np.clip(v, 0.0, 1.0)
    # final safety renorm
    v = v * (target / v.sum())
    return v
