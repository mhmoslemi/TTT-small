import time
import numpy as np

def compute_c5(h):
    """Compute the C5 overlap bound for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_onto_bounds_simplex(v, target_sum, lower=0.0, upper=1.0, max_iter=20):
    """
    Project vector v onto the intersection of the box [lower, upper]^n
    and the affine hyperplane sum(v)=target_sum.
    Simple water‑filling style iteration.
    """
    v = np.clip(v, lower, upper)
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target_sum
        if abs(diff) < 1e-12:
            break
        if diff > 0:  # sum too large, need to decrease
            idx = np.where(v > lower)[0]
            if len(idx) == 0:
                break
            dec = diff / len(idx)
            v[idx] = np.maximum(v[idx] - dec, lower)
        else:          # sum too small, need to increase
            idx = np.where(v < upper)[0]
            if len(idx) == 0:
                break
            inc = -diff / len(idx)
            v[idx] = np.minimum(v[idx] + inc, upper)
    # final tiny correction if needed
    s = v.sum()
    if abs(s - target_sum) > 1e-9:
        correction = (target_sum - s) / v.size
        v += correction
        v = np.clip(v, lower, upper)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] with ∫h=1 that minimizes
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 80
        h0 = np.full(n, 0.5, dtype=float)  # uniform start satisfies integral

    target_sum = n / 2.0          # because ∫h dx = 1 and dx = 2/n
    h0 = project_onto_bounds_simplex(h0, target_sum)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    # current state for SA
    cur_h = best_h.copy()
    cur_score = best_score

    # Annealing schedule parameters
    max_iters = 500000            # large safety cap
    temp_start = 0.2
    temp_end = 1e-4
    step_start = 0.3
    step_end = 1e-4

    for it in range(1, max_iters + 1):
        # time check
        if time.time() - start_time > budget_s - 0.1:
            break

        # linear cooling
        frac = it / max_iters
        temp = temp_start * (1 - frac) + temp_end * frac
        step = step_start * (1 - frac) + step_end * frac

        # propose new candidate by Gaussian perturbation
        proposal = cur_h + step * np.random.randn(n)
        proposal = project_onto_bounds_simplex(proposal, target_sum)

        prop_score = compute_c5(proposal)

        # Metropolis acceptance
        if prop_score < cur_score or np.random.rand() < np.exp(-(prop_score - cur_score) / max(temp, 1e-12)):

            cur_score = prop_score
