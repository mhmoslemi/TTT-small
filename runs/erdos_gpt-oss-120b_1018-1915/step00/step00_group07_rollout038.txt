import time
import numpy as np

def _repair(h, target_sum):
    """Project h onto [0,1] with sum = target_sum using simple iterative water-filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        total = h.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        if not free.any():
            # no free variables, distribute uniformly (may violate bounds a bit)
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h, dx):
    """Compute the C5 overlap for a given step vector."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Differential‑evolution search for a step function h on [0,2] that lowers the
    Erdős C5 overlap bound.

    Returns
    -------
    h_values : np.ndarray
        The best step vector found (length n_points, values in [0,1]).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    rng = np.random.default_rng(seed)

    # Determine discretisation size
    if 'initial_h_values' in globals():
        n_points = int(len(initial_h_values))
        h_start = _repair(np.asarray(initial_h_values, dtype=float), n_points / 2.0)
    else:
        n_points = 80
        h_start = _repair(rng.random(n_points), n_points / 2.0)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # DE parameters
    pop_size = max(30, n_points // 2)
    F = 0.5
    CR = 0.9
    max_gen = 10_000  # generous; loop will stop by time

    # Initialise population
    pop = np.empty((pop_size, n_points), dtype=float)
    pop[0] = h_start
    for i in range(1, pop_size):
        pop[i] = _repair(rng.random(n_points), target_sum)

    # Evaluate initial population
    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        scores[i] = _c5_value(pop[i], dx)

    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    start_time = time.time()
    generation = 0
    while generation < max_gen:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        generation += 1

        for i in range(pop_size):
            # Choose three distinct indices different from i
            idxs = rng.choice(pop_size, 3, replace=False)
            while i in idxs:
                idxs = rng.choice(pop_size, 3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]

            mutant = a + F * (b - c)
            # Binomial crossover
            cross_mask = rng.random(n_points) < CR
            # Ensure at least one component from mutant
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, n_points)] = True
            child = np.where(cross_mask, mutant, pop[i])
            child = _repair(child, target_sum)

            child_score = _c5_value(child, dx)

            if child_score < scores[i]:
                pop[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_h = child.copy()

    # Final projection to ensure feasibility (tiny numerical drift)
    best_h = _repair(best_h, target_sum)
    best_score = _c5_value(best_h, dx)

    return best_h, float(best_score), n_points
