import numpy as np

def _evaluate_c5(h):
    """Return max overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _mutate(h, step):
    """Propose a new h by moving a small amount of mass between two points."""
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    # allowed transfer respecting bounds
    max_transfer = min(step, 1.0 - h[i], h[j])
    if max_transfer <= 0:
        return None
    delta = np.random.uniform(0, max_transfer)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new
