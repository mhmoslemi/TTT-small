import time
import numpy as np
import random
from math import exp

def _compute_c5(h, dx):
    """Return max overlap integral for discrete step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_bounds_sum(v, target):
    """Clip to [0,1] and adjust to hit exact sum=target (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    max_iter = 1000
    for _ in range(max_iter):
        s = v.sum()
        diff = s - target
        if abs(diff) < 1e-12:
            break
        # free variables not at a bound
        free = (v > 0.0) & (v < 1.0)
        if not np.any(free):
            # all at bounds, distribute remaining diff uniformly (will stay at bounds)
            break
        # distribute diff uniformly over free variables
        adjust = diff / free.sum()
        v[free] -= adjust
        v = np.clip(v, 0.0, 1.0)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 to minimize
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine number of points; use provided initial if present
    h0 = globals().get("initial_h_values")
    if h0 is not None:
        n = int(len(h0))
        h = np.asarray(h0, dtype=float).copy()
        # Ensure feasibility
        target_sum = n / 2.0
        h = _project_bounds_sum(h, target_sum)
    else:
        n = 80  # modest size if no parent is available
        h = np.full(n, 0.5, dtype=float)  # satisfies sum=n/2
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = h.copy()
    best_c = _compute_c5(best_h, dx)

    # Parameters for stochastic search
    max_move = 0.02          # max amount transferred between two points per trial
    temp_start = 1e-2       # annealing temperature start
    temp_end = 1e-5
    total_iters = 200000    # upper bound; loop will break earlier by time

    for it in range(total_iters):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # linear cooling schedule
        t = temp_start + (temp_end - temp_start) * (it / total_iters)

        # propose a transfer between two distinct indices
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        delta = rng.uniform(-max_move, max_move)

        if delta > 0.0:
            delta = min(delta, best_h[i], 1.0 - best_h[j])
        else:
            delta = max(delta, -(1.0 - best_h[i]), -best_h[j])

        if abs(delta) < 1e-12:
            continue

        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        # No need to renormalize; sum unchanged, bounds respected by construction
        new_c = _compute_c5(new_h, dx)

        # Accept if improvement or with Metropolis probability
        if new_c < best_c or rng.random() < exp(-(new_c - best_c) / max(t, 1e-12)):
            best_h = new_h
            best_c = new_c

        # Occasionally try a random reset around current best (to escape plateaus)
        if it % 5000 == 0 and rng.random() < 0.05:

    # Final projection (safety)
    best_h = _project_bounds_sum(best_h, target_sum)
    best_c = _compute_c5(best_h, dx)

    # Return as plain Python list for compatibility
    return best_h.tolist(), float(best_c), int(n)
