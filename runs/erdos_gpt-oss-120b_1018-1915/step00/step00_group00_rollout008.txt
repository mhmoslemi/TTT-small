import numpy as np
import time
import math

def compute_c5(h):
    """Return max overlap integral C5 for step function defined by samples h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def perturb(h, rng):
    """
    Return a perturbed copy of h by moving a random amount of mass
    from index i to j while respecting bounds [0,1].
    """
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    hi = h[i]
    hj = h[j]
    # maximum transferable mass without breaking bounds
    max_delta = min(hi, 1.0 - hj)
    if max_delta <= 0.0:
        # try opposite direction
        max_delta = min(hj, 1.0 - hi)
        if max_delta <= 0.0:
            return None, None, None
        # swap roles
        i, j = j, i
        hi, hj = h[i], h[j]
    delta = rng.random() * max_delta
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new, i, j

def run(seed=42, budget_s=1100, **kwargs):

    n = h0.size
    # ensure feasibility

    target_sum = n / 2.0
    cur_sum = h0.sum()
    if not np.isclose(cur_sum, target_sum):
        # rescale around 0.5 to meet sum constraint while staying in [0,1]
        diff = target_sum - cur_sum
        # distribute diff uniformly, then clip
        adjustment = diff / n
        h0 += adjustment
        h0 = np.clip(h0, 0.0, 1.0)
        # fix any residual error by scaling the interior values
        excess = h0.sum() - target_sum
        if abs(excess) > 1e-12:
            interior = (h0 > 0.0) & (h0 < 1.0)
            if interior.any():
                scale = (h0[interior].sum() - excess) / h0[interior].sum()
                h0[interior] *= scale
                h0 = np.clip(h0, 0.0, 1.0)
    # initial evaluation
