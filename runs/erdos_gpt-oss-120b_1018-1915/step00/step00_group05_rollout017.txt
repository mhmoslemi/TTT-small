import time
import numpy as np

def compute_c5(h):
    """Return C5 for a step function h on [0,2] with uniform grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()
