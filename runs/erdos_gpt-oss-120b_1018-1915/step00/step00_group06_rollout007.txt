import time
import numpy as np

# Helper: compute C5 for a discretized h
def _c5_value(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)

# Simple repair that only clips to [0,1]; sum is preserved by construction
def _clip(h):
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -----------------------------------------------------------------
    # Initialise h
    # -----------------------------------------------------------------
    try:
        # Use the provided parent construction if present
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        # No parent: start with a uniform half‑density function
        n = 200
        h0 = np.full(n, 0.5, dtype=float)

    n = h0.shape[0]
    # Ensure feasibility (clip and adjust sum)
    h0 = _clip(h0)
    sum_needed = n / 2.0
    cur_sum = h0.sum()
    if cur_sum != 0:
        h0 *= (sum_needed / cur_sum)
    h0 = _clip(h0)

    best_h = h0.copy()
    best_score = _c5_value(best_h)

    # -----------------------------------------------------------------
    # Optimization parameters
    # -----------------------------------------------------------------
    max_iter = 10_000_000  # large upper bound; loop stops by time budget
    max_temp = 0.02        # initial temperature for simulated annealing
    min_temp = 0.0001
    decay = 0.999995       # exponential decay per iteration

    # Pre‑allocate a working copy to avoid repeated allocations
    h = best_h.copy()

    it = 0
    while it < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.96:
            break

        # Linear temperature schedule
        T = max(min_temp, max_temp * (decay ** it))

        # ---------- propose a random transfer ----------
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # maximal amount we can move without violating bounds
        max_move = min(h[i], 1.0 - h[j])
        if max_move <= 0:
            continue
        # random epsilon up to a fraction of the allowed move
        eps = rng.random() * max_move * 0.5  # 0‑50% of possible transfer

        # Apply transfer
        h[i] -= eps
        h[j] += eps

        # Evaluate
        score = _c5_value(h)

        # Acceptance test
        accept = False
        if score < best_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(score - best_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            # Keep the new state; also update best if improved
            if score < best_score:
                best_score = score
                best_h = h.copy()
        else:
            # Revert the transfer
            h[i] += eps
            h[j] -= eps

        it += 1

    # Final clipping (should already be within bounds)
    best_h = _clip(best_h)

    # Return results
    return best_h, best_score, n
