def _project_to_feasible(vec):
    """Clip to [0,1] and enforce sum(vec) = len(vec)/2."""
    n = vec.size
    target = n / 2.0
    vec = np.clip(vec, 0.0, 1.0)

    # Simple iterative scaling to hit the target sum while staying in bounds.
    for _ in range(5):
        s = vec.sum()
        if abs(s - target) < 1e-10:
            break
        # Scale uniformly, then re‑clip.
        scale = target / s if s != 0 else 0.0
        vec *= scale
        vec = np.clip(vec, 0.0, 1.0)
    # One final correction distributing the residual uniformly.
    residual = target - vec.sum()
    if abs(residual) > 1e-12:
        vec += residual / n
        vec = np.clip(vec, 0.0, 1.0)
    return vec
