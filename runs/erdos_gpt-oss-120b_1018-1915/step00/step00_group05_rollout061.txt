import time
import numpy as np
from math import exp

def compute_c5(h):
    """Return the overlap constant C5 for a given step function h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * (2.0 / h.size)

def perturb(h, rng, max_step):
    """Transfer a random amount of mass from one index to another while keeping
    each entry inside [0,1] and preserving the total sum."""
    n = h.size
    i = rng.integers(n)
    j = rng.integers(n)
    while i == j:
        j = rng.integers(n)

    # Try moving mass i -> j
    delta = min(max_step, h[i], 1.0 - h[j])
    if delta <= 0.0:
        # Try the opposite direction j -> i
        delta = min(max_step, h[j], 1.0 - h[i])
        if delta <= 0.0:
            return h.copy()          # no feasible move
        i, j = j, i                 # swap so we always move i -> j

    delta *= rng.random()           # random fraction of the feasible amount
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns:
        h_values : 1‑D numpy array with values in [0,1] and sum = n/2
        c5_bound : the achieved C5 value (float)
        n_points : length of the array
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = kwargs.get('n_points', 70)
        # random feasible start
        h0 = rng.random(n_points)
        # project to required sum n/2
        target_sum = n_points / 2.0
        cur_sum = h0.sum()
        if cur_sum == 0:
            h0[:] = target_sum / n_points
        else:
            h0 *= target_sum / cur_sum
        # clip to [0,1] and re‑balance if necessary
        h0 = np.clip(h0, 0.0, 1.0)
        deficit = target_sum - h0.sum()
        while abs(deficit) > 1e-12:
            idx = rng.choice(np.where((deficit > 0) & (h0 < 1.0) |
                                      (deficit < 0) & (h0 > 0.0))[0])
            adjust = min(abs(deficit), 1.0 - h0[idx]) if deficit > 0 else min(abs(deficit), h0[idx])
            adjust = adjust if deficit > 0 else -adjust
            h0[idx] += adjust
            deficit = target_sum - h0.sum()

    # ----------------------------------------------------------------------
    # Baseline evaluation
    # ----------------------------------------------------------------------
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    target_sum = n_points / 2.0

    # ----------------------------------------------------------------------
    # Simulated‑annealing style hill climbing
    # ----------------------------------------------------------------------
    start_time = time.time()
    end_time = start_time + budget_s
    max_step_initial = 0.2            # maximal mass that may be moved in one step
    while time.time() < end_time:
        # linearly decay step size with remaining time
        remaining = end_time - time.time()

        max_step = max_step_initial * frac

        cand_val = compute_c5(cand)

        # accept if better, else maybe accept probabilistically
