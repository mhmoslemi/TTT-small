import time, math, random
import numpy as np

def _evaluate(h):
    """Return C5 for discretized h (dx factor included)."""
    n = h.size
    # correlation length 2n-1
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)

def _project_box_sum(v, lower, upper, target):
    """Project onto [lower,upper] box with a prescribed sum (water‑filling)."""
    v = np.clip(v, lower, upper)
    # iterative water‑filling
    while True:
        total = v.sum()
        diff = target - total
        if abs(diff) < 1e-12:
            break
        free = (v > lower + 1e-12) & (v < upper - 1e-12)
        if not np.any(free):
            # cannot move any further
            break
        add = diff / free.sum()
        v[free] += add
        v = np.clip(v, lower, upper)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized values on a uniform grid.
    c5_bound : float
        Achieved upper bound.
    n_points : int
        Number of discretisation points.
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # resolution (adjustable, stays ≤1000)
    n = 300
    target_sum = n / 2.0

    # ----- initial construction ------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)
        if init.size != n:
            # linear interpolation to new grid
            old_x = np.linspace(0.0, 2.0, init.size, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(new_x, old_x, init)
    except Exception:
        init = np.full(n, 0.5)

    h = _project_box_sum(init, 0.0, 1.0, target_sum)

    best_h = h.copy()
    best_score = _evaluate(h)

    # ----- simple simulated annealing -----------------------------------------
    max_iters = 200_000
    iteration = 0
    while iteration < max_iters and (time.time() - start_time) < 0.95 * budget_s:
        # pick two distinct indices and shift opposite amounts
        i, j = np.random.randint(n, size=2)

        h_candidate = h.copy()

        h_candidate[j] -= delta
        h_candidate = _project_box_sum(h_candidate, 0.0, 1.0, target_sum)

        # temperature schedule (linear cooling)
