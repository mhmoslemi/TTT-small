import time
import numpy as np

def _evaluate(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project(h, target_sum, max_iter=15):
    """Clip to [0,1] and adjust to meet exact sum using a simple water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual fix
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-10:
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                step = min(diff, 1.0 - h[i])
                h[i] += step
                diff -= step
            elif diff < 0 and h[i] > 0.0:
                step = min(-diff, h[i])
                h[i] -= step
                diff += step
            if abs(diff) < 1e-12:
                break
    return h

def _propose(h, eps_scale=0.05):
    """Generate a nearby feasible candidate by moving mass between two cells."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    epsilon = np.random.uniform(-eps_scale, eps_scale)
    h_new = h.copy()
    # transfer epsilon from i to j
    h_new[i] = np.clip(h_new[i] - epsilon, 0.0, 1.0)
    h_new[j] = np.clip(h_new[j] + epsilon, 0.0, 1.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    - h_values : list of step‑function samples on [0,2]
    - c5_bound : achieved upper bound for C₅
    - n_points : number of samples (length of h_values)
    """
    rng = np.random.default_rng(seed)
    np.random.seed(seed)          # for any legacy calls
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 70
        h0 = np.full(n_points, 0.5)   # uniform satisfies integral constraint

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial feasible point
    h = _project(h0, target_sum)

    best_h = h.copy()
    best_score = _evaluate(best_h, dx)

    # Annealing schedule parameters
    T_start = 0.2
    T_end = 1e-5
    max_iters = 500000   # safety cap; loop will break on time budget

    for it in range(1, max_iters + 1):
        # time check
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        # linear cooling
        T = T_start + (T_end - T_start) * (elapsed / budget_s)

        # propose a neighbour
        h_candidate = _propose(h, eps_scale=0.07)
        h_candidate = _project(h_candidate, target_sum)

        cand_score = _evaluate(h_candidate, dx)

        # Accept if better, else with Metropolis probability
        if cand_score < best_score:
            best_score = cand_score
            best_h = h_candidate.copy()

        if cand_score < _evaluate(h, dx) or rng.random() < np.exp(( _evaluate(h, dx) - cand_score) / (T + 1e-12)):
            h = h_candidate

        # occasional aggressive random restart if stuck
        if it % 20000 == 0:
            if rng.random() < 0.2:
                h = _project(rng.random(n_points), target_sum)

    # Final evaluation on best solution
    final_score = _evaluate(best_h, dx)

    return (best_h.tolist(), final_score, n_points)
