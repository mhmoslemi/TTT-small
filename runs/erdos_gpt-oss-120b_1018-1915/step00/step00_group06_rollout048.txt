import numpy as np, time, math, random

def _evaluate_c5(h):
    """Return C5 for a step function sampled as h on [0,2]."""
    n = h.shape[0]
    # full correlation of h with 1-h
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)

def _project_feasible(h, target_sum, max_iter=10):
    """Clip to [0,1] and adjust to meet sum==target_sum (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) <= 1e-12:
            break
        # distribute diff uniformly over non‑boundary entries
        mask = (h > 0.0) & (h < 1.0)
        cnt = mask.sum()
        if cnt == 0:  # all at bounds, fallback to scaling whole vector
            factor = target_sum / s if s != 0 else 1.0
            h = np.clip(h * factor, 0.0, 1.0)
            continue
        delta = diff / cnt
        # apply limited step so we stay within [0,1]
        h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction by scaling if still off (rare)
    s = h.sum()
    if s > 0:
        h *= target_sum / s
    h = np.clip(h, 0.0, 1.0)
    return h

def _interpolate_parent(parent, n_target):
    """Linear interpolation of parent vector to a new length."""
    x_parent = np.linspace(0.0, 2.0, len(parent), endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_new = np.interp(x_new, x_parent, parent)
    # enforce integral constraint
    target_sum = n_target / 2.0
    h_new = _project_feasible(h_new, target_sum)
    return h_new

    base_n = 95
