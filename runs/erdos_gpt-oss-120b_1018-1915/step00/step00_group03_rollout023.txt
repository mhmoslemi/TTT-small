import time
import numpy as np
import random

try:
    import cvxpy as cp
except Exception:  # pragma: no cover
    cp = None


def _feasible_initial(n, rng):
    """Return a feasible h vector (0<=h<=1, sum = n/2) using cvxpy if possible,
    otherwise a simple projection method."""
    if cp is not None:
        h = cp.Variable(n)
        constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2]
        prob = cp.Problem(cp.Minimize(0), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if h.value is not None:
            return np.clip(h.value, 0.0, 1.0)
    # fallback: simple projection
    x = rng.random(n)
    x = x / x.sum() * (n / 2)                     # enforce sum
    # iterative clipping to satisfy bounds while keeping sum
    for _ in range(20):
        over = x - 1.0
        under = -x
        excess = over.clip(min=0).sum()
        deficit = under.clip(min=0).sum()
        if excess == 0 and deficit == 0:
            break
        # redistribute excess to deficit proportionally
        if excess > 0:
            x[over > 0] -= excess / (over > 0).sum()
        if deficit > 0:
            x[under > 0] += deficit / (under > 0).sum()
        x = np.clip(x, 0.0, 1.0)
    return x


def _max_overlap(h, dx):
    """Compute C5 for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 constant.

    Returns:
        h_values (list of float): optimized samples of h.
        c5_bound (float): achieved upper bound for C5.
        n_points (int): number of discretisation points used.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    try:
        n = len(initial_h_values)  # provided by harness if available
        h_current = np.array(initial_h_values, dtype=float)
        # ensure feasibility (might be already feasible)
        if not (np.all((0.0 <= h_current) & (h_current <= 1.0)) and
                abs(h_current.sum() - n / 2) < 1e-9):
            h_current = _feasible_initial(n, rng)
    except Exception:
        n = 200  # default resolution
        h_current = _feasible_initial(n, rng)

    dx = 2.0 / n
    best_h = h_current.copy()
    best_c5 = _max_overlap(best_h, dx)

    # Simple simulated annealing / hill‑climbing
    max_iter = 10_000_000  # upper bound, actual loop stops by time
    iter_count = 0
    while time.time() - start_time < budget_s * 0.94 and iter_count < max_iter:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = best_h[i], best_h[j]
        # maximal allowed epsilon keeping within [0,1]
        eps_max = min(1.0 - hi, hj) if hi <= hj else min(hi, 1.0 - hj)
        if eps_max <= 0:
            continue
        eps = rng.uniform(0, eps_max * 0.5)  # moderate step size
        # propose swap of mass eps
        new_h = best_h.copy()
        new_h[i] += eps
        new_h[j] -= eps

        # evaluate
        new_c5 = _max_overlap(new_h, dx)

        # accept if improvement or with small probability (annealing)
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
        else:
            # occasional uphill moves to escape local minima
            temp = max(0.01, 0.5 * (budget_s - (time.time() - start_time)) / budget_s)
            if rng.random() < np.exp((best_c5 - new_c5) / temp):
