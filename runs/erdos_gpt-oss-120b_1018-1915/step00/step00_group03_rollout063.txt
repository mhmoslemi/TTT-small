import time
import random
import numpy as np

def evaluate(h: np.ndarray) -> float:
    """Return the maximal overlap C5 for the step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def perturb(h: np.ndarray, delta_max: float) -> np.ndarray:
    """Transfer a small amount of mass between two random coordinates,
    keeping each entry in [0,1] and preserving the total sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # propose a signed transfer
    delta = (np.random.rand() * 2.0 - 1.0) * delta_max
    # clamp to feasible interval so bounds are respected
    delta = np.clip(delta, - (1.0 - h[j]), h[i])
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Optimize the step function to obtain a low C5 bound.

    Returns
    -------
    (list_of_h, c5_bound, n_points)
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ----- initialise -------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h_best = np.array(init, dtype=float)
    else:
        n0 = 70
        # alternating 1/0 gives the required integral automatically
        h_best = np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n0)], dtype=float)

    best_score = evaluate(h_best)

    # ----- main (low‑resolution) search ------------------------------------
    while time.time() - start_time < budget_s - 2.0:
        # gradually shrink the perturbation magnitude
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        delta_max = 0.2 * (1.0 - frac) + 0.01

        h_candidate = perturb(h_best, delta_max)
        score = evaluate(h_candidate)
        if score < best_score:
            h_best = h_candidate
            best_score = score

    # ----- if time permits, double the resolution and continue -------------
    if time.time() - start_time < budget_s - 2.0:
        # repeat each sample, preserving the total integral automatically
        h_best = np.repeat(h_best, 2)
        best_score = evaluate(h_best)

        while time.time() - start_time < budget_s - 2.0:
            elapsed = time.time() - start_time
            frac = elapsed / budget_s
            delta_max = 0.1 * (1.0 - frac) + 0.005

            h_candidate = perturb(h_best, delta_max)
            score = evaluate(h_candidate)
            if score < best_score:
                h_best = h_candidate
                best_score = score

    # ----- output -----------------------------------------------------------
    return (h_best.tolist(), float(best_score), len(h_best))
