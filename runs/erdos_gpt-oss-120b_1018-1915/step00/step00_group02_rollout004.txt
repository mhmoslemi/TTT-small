def _project_feasible(h, target_sum, max_iter=10):
    """Clip to [0,1] and enforce sum == target_sum (within numerical tolerance)."""
    # Simple iterative scaling with clipping
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        if np.isclose(cur_sum, target_sum, atol=1e-9):
            break
        # Scale the interior (non‑clipped) part; if all clipped, just distribute uniformly
        scale = target_sum / cur_sum if cur_sum > 0 else 1.0
        h *= scale
    # Final clip to guarantee bounds
    h = np.clip(h, 0.0, 1.0)
    # Tiny correction for any residual sum error
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # Distribute residual uniformly over entries that are not at a bound
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += residual / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h
