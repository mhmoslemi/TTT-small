import time
import numpy as np

def evaluate_h(h):
    """Return the C5 value for a step function sampled in h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def normalize_sum(h):
    """Scale h so that sum(h) == len(h)/2 while keeping values in [0,1]."""
    target = h.size / 2.0
    cur = h.sum()
    if cur == 0:
        return np.full_like(h, 0.5)
    scale = target / cur
    h = h * scale
    # Clip any drift outside [0,1] and re‑adjust with a simple projection
    h = np.clip(h, 0.0, 1.0)
    # Re‑balance the total after clipping
    excess = h.sum() - target
    if abs(excess) > 1e-12:
        # Distribute excess uniformly (could be improved)
        h -= excess / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def random_transfer(h, rng, max_step):
    """Propose a new h by moving a small amount from j to i."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # decide direction: i will receive, j will give
    cap_i = 1.0 - h[i]
    cap_j = h[j]
    amount = rng.random() * max_step * min(cap_i, cap_j)
    if amount <= 0:
        return h  # no change possible
    new_h = h.copy()
    new_h[i] += amount
    new_h[j] -= amount
    return new_h

def local_search(h, rng, budget, start_time):
    """Simple hill‑climbing / simulated‑annealing loop."""
    best_h = h.copy()
    best_val = evaluate_h(best_h)
    n = h.size
    max_iters = 500000
    max_step = 0.2  # maximum transferred mass per move
    T0 = 0.005
    for it in range(max_iters):
        if time.time() - start_time > budget * 0.98:
            break
        T = T0 * (1.0 - it / max_iters)
        cand = random_transfer(best_h, rng, max_step)
        cand_val = evaluate_h(cand)
        delta = cand_val - best_val
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            best_h, best_val = cand, cand_val
    return best_h, best_val

def upscale(h):
    """Double the resolution using linear interpolation and re‑normalize."""
    n = h.size
    new_n = n * 2
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, h)
    return normalize_sum(new_h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = normalize_sum(h0)
    else:
        n0 = 64
        h0 = rng.random(n0)
        h0 = normalize_sum(h0)

    # ------------------------------------------------------------------
    # First level optimisation (base resolution)
    # ------------------------------------------------------------------
    h_best, val_best = local_search(h0, rng, budget_s, start_time)

    # ------------------------------------------------------------------
    # Optional up‑scaling passes if time permits
    # ------------------------------------------------------------------
    h_curr = h_best
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        # double resolution
        h_up = upscale(h_curr)
        h_curr, val_curr = local_search(h_up, rng, budget_s - elapsed, start_time)
        if val_curr < val_best:
            h_best, val_best = h_curr, val_curr
        else:
            # if no improvement, stop up‑scaling
            break

    n_points = h_best.size
    c5_bound = float(val_best)
    return (h_best, c5_bound, n_points)
