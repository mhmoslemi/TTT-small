import time
import numpy as np

def _score(h, dx):
    """Return max overlap C5 for step values h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _propose(h, rng, max_move=0.2):
    """Transfer a random amount of mass between two indices, preserving sum."""
    n = h.size
    i, j = rng.choice(n, 2, replace=False)
    # how much can we add to i and remove from j while staying in [0,1]?
    upper_i = 1.0 - h[i]
    lower_j = h[j]                     # amount we can take from j
    max_possible = min(upper_i, lower_j, max_move)
    if max_possible <= 0:
        return h.copy()
    delta = rng.uniform(0.0, max_possible)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] with integral 1
    minimizing max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine problem size
    try:
        init_h = np.array(initial_h_values, copy=False, dtype=float)
        n = init_h.size
        h = init_h.copy()
    except NameError:
        n = 45
        h = rng.random(n)
    dx = 2.0 / n
    target_sum = n / 2.0

    # Project initial h onto constraints (sum = target, 0≤h≤1)
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if cur_sum == 0:
        h = np.full(n, target_sum / n)
    else:
        h *= target_sum / cur_sum
    # after scaling some entries may exceed 1, clip and renormalize once
    overflow = h > 1.0
    if overflow.any():
        excess = h[overflow] - 1.0
        h[overflow] = 1.0
        # distribute excess uniformly to other positions that are below 1
        under = h < 1.0
        if under.any():
            h[under] -= excess.sum() / under.sum()
            h = np.clip(h, 0.0, 1.0)
    # Ensure exact sum (tiny numerical error)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # add/subtract diff to a random non‑boundary element
        cand = np.where((h > 0.0) & (h < 1.0))[0]
        if cand.size == 0:
            cand = np.arange(n)
        idx = rng.choice(cand)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_score = _score(best_h, dx)

    # Simulated annealing parameters
    temp = 0.05
    cooling = 0.995
    inner_iters = max(1000, n * 10)

    while time.time() - start_time < budget_s - 5.0:
        for _ in range(inner_iters):
            h_new = _propose(best_h, rng, max_move=0.2)
            score_new = _score(h_new, dx)
            # Accept if better or with Metropolis probability
            if score_new < best_score or rng.random() < np.exp((best_score - score_new) / max(temp, 1e-12)):
                best_h = h_new
                best_score = score_new
        temp *= cooling
        # occasional random restart if stagnating
        if rng.random() < 0.02:
            h_restart = rng.random(n)
            h_restart = np.clip(h_restart, 0.0, 1.0)
            h_restart *= target_sum / h_restart.sum()
            best_h = h_restart
            best_score = _score(best_h, dx)

    return best_h, best_score, n
