import numpy as np, time, math, random

def _project_bounds_sum(x, target, max_iter=10):
    """Project vector x onto [0,1] box with given sum target."""
    for _ in range(max_iter):
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < 1e-9:
            break
        diff = target - s
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds, distribute proportionally
            x += diff / len(x)
            continue
        x[free] += diff / cnt
    # final clip to be safe
    return np.clip(x, 0.0, 1.0)

def _evaluate(h, dx):
    """Return max overlap C5 for step vector h."""
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return conv.max()

def _optimize(initial_h, dx, budget_sec, rng):
    """Simple stochastic improvement respecting constraints."""
    n = len(initial_h)
    target_sum = n / 2.0
    best_h = initial_h.copy()
    best_score = _evaluate(best_h, dx)
    cur_h = best_h.copy()
    cur_score = best_score

    start = time.time()
    it = 0
    # schedule for step size (standard deviation)
    step = 0.1
    while time.time() - start < budget_sec and it < 200000:
        it += 1
        # annealing schedule
        step = 0.15 * (0.99 ** (it // 1000))
        proposal = cur_h + rng.normal(scale=step, size=n)
        proposal = _project_bounds_sum(proposal, target_sum)
        score = _evaluate(proposal, dx)
        if score < cur_score or rng.random() < math.exp(-(score - cur_score) / (0.001 + 0.01 * it / 10000)):
            cur_h, cur_score = proposal, score
            if score < best_score:
                best_h, best_score = proposal.copy(), score
        # occasional random restart to escape local minima
        if it % 5000 == 0 and rng.random() < 0.2:
            cur_h = best_h.copy()
            cur_score = best_score
    return best_h, best_score
