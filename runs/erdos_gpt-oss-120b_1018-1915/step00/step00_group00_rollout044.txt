import time
import numpy as np

def evaluate_c5(h, dx):
    """Return max overlap C5 for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def propose_mutation(h, rng):
    """Return a new h after a feasible transfer between two positions."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # amount we can move from i to j without breaking bounds
    max_down = min(h[i], 1.0 - h[j])
    max_up = min(1.0 - h[i], h[j])
    # choose direction randomly
    if rng.random() < 0.5:
        delta = rng.random() * max_down
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        delta = rng.random() * max_up
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
    return new_h
