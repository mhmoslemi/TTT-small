                break

            v = np.clip(v,0.0,1.0)

                    v[i] += inc
                    diff -= inc

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    n_points = kwargs.get('n_points', 70)
    desired_sum = n_points / 2.0          # because ∫h = 1 and dx = 2/n
    dx = 2.0 / n_points

    # ----------------------------------------------------------------------
    # feasibility projection (keeps sum exactly desired_sum, respects [0,1])
    # ----------------------------------------------------------------------
    def project(v):
        v = np.clip(v, 0.0, 1.0)
        # iterative scaling + clipping
        for _ in range(8):
            s = v.sum()
            if abs(s - desired_sum) < 1e-12:
                break
            if s == 0:
                break
            v = v * (desired_sum / s)
            v = np.clip(v, 0.0, 1.0)
        # tiny residual correction
        diff = desired_sum - v.sum()
        if abs(diff) > 1e-12:
            # distribute diff to free entries
            for idx in np.argsort(-np.abs(v - 0.5)):
                if diff > 0 and v[idx] < 1.0:
                    inc = min(diff, 1.0 - v[idx])
                    v[idx] += inc
                    diff -= inc
                elif diff < 0 and v[idx] > 0.0:
                    dec = min(-diff, v[idx])
                    v[idx] -= dec
                    diff += dec
                if abs(diff) < 1e-12:
                    break
        return v

    # ----------------------------------------------------------------------
    # objective: maximal overlap C5
    # ----------------------------------------------------------------------
    def c5_value(vec):
        h = project(vec)
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    # ----------------------------------------------------------------------
    # initial vector
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        if len(init) != n_points:
            # resample by linear interpolation to required size
            old_x = np.linspace(0, 2, len(init), endpoint=False)
            new_x = np.linspace(0, 2, n_points, endpoint=False)
            init = np.interp(new_x, old_x, init)
        init = project(init)
    else:
        # binary half‑ones, random order
        init = np.zeros(n_points, dtype=float)
        init[:int(desired_sum)] = 1.0
        rng.shuffle(init)

    best_vec = init.copy()
    best_score = c5_value(best_vec)

    # ----------------------------------------------------------------------
    # Phase 1: local refinement with SLSQP (small portion of budget)
    # ----------------------------------------------------------------------
    elapsed = time.time() - start_time
    if budget_s - elapsed > 5.0:
        cons = ({'type': 'eq',
                 'fun': lambda x: np.sum(project(x)) - desired_sum},)
        bounds = [(0.0, 1.0)] * n_points
        res = minimize(c5_value, best_vec, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 600, 'ftol': 1e-9, 'disp': False})
        if res.success:
            cand = project(res.x)
            sc = c5_value(cand)
            if sc < best_score:
                best_vec, best_score = cand, sc

    # ----------------------------------------------------------------------
    # Phase 2: global search with Differential Evolution
    # ----------------------------------------------------------------------
    elapsed = time.time() - start_time
    time_left = budget_s - elapsed
    if time_left > 5.0:
        # rough estimate of iterations within remaining time
        max_iter = max(1, int((time_left * 0.9) / 0.05))
        de_bounds = [(0.0, 1.0)] * n_points
        de_res = differential_evolution(c5_value, de_bounds,
                                         strategy='best1bin',
                                         maxiter=max_iter,
                                         popsize=15,
                                         mutation=(0.5, 1.0),
                                         recombination=0.7,
                                         seed=seed,
                                         polish=False,
                                         updating='deferred',
                                         workers=1)
        cand = project(de_res.x)
        sc = c5_value(cand)
        if sc < best_score:
            best_vec, best_score = cand, sc

    # ----------------------------------------------------------------------
    # Phase 3: cheap coordinate‑swap hill climbing until budget expires
    # ----------------------------------------------------------------------
    while time.time() - start_time < budget_s - 0.1:
        i, j = rng.choice(n_points, size=2, replace=False)
        delta = (rng.random() - 0.5) * 0.2   # +/-0.1 at most
        vi, vj = best_vec[i], best_vec[j]
        if not (0.0 <= vi + delta <= 1.0 and 0.0 <= vj - delta <= 1.0):
            continue
        trial = best_vec.copy()
        trial[i] = vi + delta
        trial[j] = vj - delta
        sc = c5_value(trial)
        if sc < best_score:
            best_vec, best_score = trial, sc

    # final projection to guarantee constraints
    best_vec = project(best_vec)
    best_score = c5_value(best_vec)

    # final sanity checks (will raise if violated)
    assert len(best_vec) == n_points
    assert np.all(best_vec >= -1e-12) and np.all(best_vec <= 1.0 + 1e-12)
    assert abs(best_vec.sum() - desired_sum) < 1e-8

    return best_vec, float(best_score), n_points
