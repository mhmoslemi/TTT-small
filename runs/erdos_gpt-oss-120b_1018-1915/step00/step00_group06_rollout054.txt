import numpy as np
import time

def _compute_c5(h):
    """Return C5 for a given step function h sampled uniformly on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _interpolate_h(old_h, old_n, new_n):
    """Linear interpolation of old_h (length old_n) onto a new grid of size new_n."""
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    # Renormalize to satisfy the integral constraint exactly
    desired_sum = new_n / 2.0
    cur_sum = new_h.sum()
    if cur_sum > 0:
        new_h *= desired_sum / cur_sum
    # Clip to [0,1] and re‑adjust if clipping broke the sum
    new_h = np.clip(new_h, 0.0, 1.0)
    cur_sum = new_h.sum()
    diff = desired_sum - cur_sum
    if abs(diff) > 1e-12:
        # distribute the remaining mass uniformly over non‑saturated entries
        free = (new_h > 0.0) & (new_h < 1.0)
        if free.any():
            add = diff / free.sum()
            new_h[free] += add
        new_h = np.clip(new_h, 0.0, 1.0)
    return new_h
