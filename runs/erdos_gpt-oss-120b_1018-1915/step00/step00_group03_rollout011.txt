import numpy as np
import time

def compute_c5(h, dx):
    """Return the max overlap C5 for a step function h."""
    # correlation of h with (1-h)
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def repair_sum_and_bounds(h, target_sum):
    """Clip to [0,1] and enforce exact sum by adjusting the last element."""
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    diff = target_sum - cur_sum
    # adjust last element, then clip again
    h[-1] += diff
    h[-1] = np.clip(h[-1], 0.0, 1.0)
    # if still off (rare), scale whole vector proportionally
    if not np.isclose(h.sum(), target_sum, atol=1e-8):
        scale = target_sum / h.sum()
        h *= scale
        h = np.clip(h, 0.0, 1.0)
        # final tweak on last entry
        cur_sum = h.sum()
        h[-1] += target_sum - cur_sum
        h[-1] = np.clip(h[-1], 0.0, 1.0)
    return h

def random_neighbor(h, step, target_sum, rng):
    """Generate a neighbor by moving a small amount between two positions."""
    n = h.size
    a, b = rng.integers(0, n, size=2)
    while a == b:
        b = rng.integers(0, n)
    delta = rng.uniform(-step, step)
    h_new = h.copy()
    h_new[a] += delta
    h_new[b] -= delta
    h_new = repair_sum_and_bounds(h_new, target_sum)
    return h_new

    n_points = h0.size
    dx = 2.0 / n_points

    # Ensure feasibility of the initial vector
    h_current = repair_sum_and_bounds(h0, target_sum)
    best_h = h_current.copy()

    max_iters = 200000  # hard cap; loop will exit early on time
