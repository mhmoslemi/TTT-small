import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős overlap value C5 for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def repair_h(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to equal target_sum."""
    np.clip(h, 0.0, 1.0, out=h)
    # simple uniform shift iterations
    for _ in range(10):
        cur = h.sum()
        delta = target_sum - cur
        if abs(delta) < 1e-12:
            break
        shift = delta / h.size
        h += shift
        np.clip(h, 0.0, 1.0, out=h)

    # final proportional correction for non‑boundary entries
    cur = h.sum()
    if abs(cur - target_sum) > 1e-9:
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            extra = target_sum - cur
            h[mask] += extra / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 to minimise the Erdős
    overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimised values of h on a uniform grid.
    c5_bound : float
        The achieved upper bound for C5.
    n_points : int
        Number of samples used (grid size).
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # initialise h
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: a uniform 0.5 function (satisfies the integral constraint)
        n_fallback = 84
        h0 = np.full(n_fallback, 0.5, dtype=float)

    n = h0.size
    target_sum = n / 2.0
    h_current = repair_h(h0.copy(), target_sum)
    best_h = h_current.copy()
    best_c5 = compute_c5(best_h)
    current_c5 = best_c5

    # ----------------------------------------------------------------------
    # optimisation loop
    # ----------------------------------------------------------------------
    start = time.time()
    time_limit = budget_s - 0.5          # safety margin
    sigma = 0.05                         # mutation scale
    no_improve = 0
    max_no_improve = 5000                # when to re‑seed from best

    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break

        # propose a mutation
        k = rng.integers(1, 4)                     # 1‑3 coordinates
        idx = rng.choice(n, size=k, replace=False)
        h_candidate = h_current.copy()
        h_candidate[idx] += rng.normal(0.0, sigma, size=k)
        repair_h(h_candidate, target_sum)

                best_c5 = cand_c5
