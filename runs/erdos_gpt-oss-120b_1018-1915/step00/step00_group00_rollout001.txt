import time
import numpy as np

def compute_c5(h, dx):
    """Return C5 for sample vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def mutate_swap_preserve_sum(h, rng):
    """Return a new h where mass is transferred between two indices,
    keeping each entry in [0,1] and preserving the total sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    hi, hj = h[i], h[j]

    # feasible delta so that new values stay in [0,1]
    lower = max(-hi, hj - 1.0)
    upper = min(1.0 - hi, hj)
    if lower >= upper:
        # cannot move mass between these indices; return original copy
        return h.copy()
    delta = rng.uniform(lower, upper)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new
