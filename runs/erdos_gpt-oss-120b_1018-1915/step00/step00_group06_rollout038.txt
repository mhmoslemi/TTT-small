def _repair(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h) = len(h)/2 exactly."""
    n = len(h)
    desired = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        # equally distribute the required mass
        h[:] = desired / n
        return h
    # First scaling to get close to the target sum
    factor = desired / total
    h = np.clip(h * factor, 0.0, 1.0)
    # Fix any remaining discrepancy with a water‑filling step
    diff = desired - h.sum()
    if abs(diff) > 1e-12:
        # indices that are not at a bound can absorb the residue
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # fall back to uniform adjustment (will stay in bounds because diff is tiny)
            h += diff / n
        else:
            add = diff / free.size
            h[free] += add
        h = np.clip(h, 0.0, 1.0)  # final safety clip
    return h


def _evaluate(h: np.ndarray) -> float:
    """Return C5 = (2/n) * max_k convolution of h with 1-h."""
    n = len(h)
    conv = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * np.max(conv)
