import numpy as np
import time

def _compute_c5(h, dx):
    # correlation of h with (1-h) over full range, then max * dx
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # ----- initial construction ------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n_points = 70
        # simple balanced start: half ones, half zeros, shuffled
        h0 = np.zeros(n_points, dtype=float)
        h0[:n_points // 2] = 1.0
        rng.shuffle(h0)
    n_points = h0.shape[0]
    dx = 2.0 / n_points

    # enforce exact integral constraint (sum = n_points/2)
    target_sum = n_points / 2.0
    cur_sum = h0.sum()
    if not np.isclose(cur_sum, target_sum):
        # scale then clip and re‑project onto simplex with box constraints
        h0 = h0 * (target_sum / cur_sum)
        # clip to [0,1]
        h0 = np.clip(h0, 0.0, 1.0)
        # simple redistribution for any deficit/excess
        deficit = target_sum - h0.sum()
        while abs(deficit) > 1e-12:
            idx = rng.choice(n_points, size=n_points, replace=False)
            for i in idx:
                avail = 1.0 - h0[i] if deficit > 0 else h0[i]
                if avail <= 0:
                    continue
                delta = min(abs(deficit), avail) * np.sign(deficit)
                h0[i] += delta
                deficit -= delta
                if abs(deficit) < 1e-12:
                    break

    # ----- simulated annealing -------------------------------------------------
    current = h0.copy()
    current_score = _compute_c5(current, dx)
    best = current.copy()
    best_score = current_score

    start_time = time.time()
    max_time = budget_s * 0.95  # safety margin

    # annealing parameters
    init_temp = 0.02
    min_temp = 1e-5
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > max_time:
            break
        # linear temperature schedule
        T = max(min_temp, init_temp * (1.0 - elapsed / max_time))
        # step size scales with temperature
        step = 0.3 * T + 1e-4

        # propose a swap of mass between two coordinates
        i, j = rng.choice(n_points, size=2, replace=False)
        delta = (rng.random() - 0.5) * 2.0 * step  # in [-step, step]
        hi = current[i] + delta
        hj = current[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue  # reject out‑of‑bounds move

        new = current.copy()
        new[i] = hi
        new[j] = hj
        new_score = _compute_c5(new, dx)

        # Metropolis acceptance
        if new_score < current_score or rng.random() < np.exp(-(new_score - current_score) / (T + 1e-12)):
            current = new
            current_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        it += 1

    # ----- return result -------------------------------------------------------
    return best, best_score, n_points
