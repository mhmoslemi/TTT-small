import time
import numpy as np
import random

def _evaluate(h):
    """Return C5 for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _feasible_random(n):

    v = np.random.rand(n)
    v = v / v.sum() * (n / 2.0)          # impose integral constraint
    # enforce box constraints, re‑scale if needed
