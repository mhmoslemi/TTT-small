import time
import numpy as np

def _c5_of(h, dx):
    """Compute the C5 value for a given step function h."""
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimises C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # problem dimensions
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
        n_points = h.size
    else:
        # fallback: 45 points, average value 0.5 (integral 1)
        n_points = 45
        h = np.full(n_points, 0.5, dtype=float)

    dx = 2.0 / n_points

    # enforce the exact sum constraint (allow tiny numerical error)
    target_sum = n_points / 2.0
    cur_sum = h.sum()
    if cur_sum != 0:
        h *= target_sum / cur_sum
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = _c5_of(best_h, dx)

    # ------------------------------------------------------------------
    # simple simulated‑annealing / coordinate‑descent loop
    # ------------------------------------------------------------------
    start_time = time.time()
    elapsed = 0.0

    # annealing schedule
    temperature = 0.02          # initial acceptance temperature
    temp_decay = 0.9995
    step_size = 0.1             # maximal transfer between two coordinates
    step_decay = 0.9995

    # pre‑allocate a mask of indices for faster random draws
    idx = np.arange(n_points)

    while elapsed < budget_s:
        # pick two distinct indices
        i, j = np.random.choice(idx, size=2, replace=False)

        # maximum amount we may transfer from j to i while staying in [0,1]
        max_transfer = min(step_size, 1.0 - h[i], h[j])

        if max_transfer <= 0:
            # nothing can be moved; try another pair next iteration
            elapsed = time.time() - start_time
            continue

        eps = (np.random.rand() * 2.0 - 1.0) * max_transfer  # signed move
        h_prop = h.copy()
        h_prop[i] += eps
        h_prop[j] -= eps

        # maintain feasibility (clipping should be unnecessary but safe)
        np.clip(h_prop, 0.0, 1.0, out=h_prop)

        # evaluate
        c5_prop = _c5_of(h_prop, dx)

        # acceptance test
        accept = False
        if c5_prop < best_c5:
            accept = True
        else:
            # metropolis criterion
            prob = np.exp(-(c5_prop - best_c5) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            h = h_prop
            if c5_prop < best_c5:
                best_c5 = c5_prop
                best_h = h.copy()

        # schedule updates
        temperature *= temp_decay
        step_size *= step_decay

        elapsed = time.time() - start_time

    # final projection to exact sum (tiny drift may have accumulated)
    cur_sum = best_h.sum()
    if cur_sum != 0:
        best_h *= target_sum / cur_sum
    best_h = np.clip(best_h, 0.0, 1.0)

    # recompute final bound for safety
    final_c5 = _c5_of(best_h, dx)

    return best_h, final_c5, n_points
