import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        parent = np.asarray(globals()['initial_h_values'], dtype=float)
        n = len(parent)
        init_vec = parent.copy()
    else:
        n = kwargs.get('n_points', 70)
        init_vec = np.random.rand(n)

    dx = 2.0 / n
    desired_sum = n / 2.0  # because ∫ h = 1 ⇒ sum(h)*dx = 1

    # ------------------------------------------------------------------
    # Helper: project a vector onto the feasible set 0≤h≤1, sum = desired_sum
    def project_to_feasible(v):
        v = np.clip(v, 0.0, 1.0)
        for _ in range(10):
            diff = desired_sum - v.sum()
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                mask = v < 1.0 - 1e-12
            else:
                mask = v > 1e-12
            if not np.any(mask):
                break
            adjust = diff / mask.sum()
            v[mask] += adjust
            v = np.clip(v, 0.0, 1.0)
        return v

    # Ensure initial vector is feasible
    init_vec = project_to_feasible(init_vec)

    # ------------------------------------------------------------------
    # Objective: compute the discrete C5 value for a candidate h
    def compute_c5(h):
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    # ------------------------------------------------------------------
    # Phase 1: local refinement with SLSQP (handles bounds + equality)
    bounds = [(0.0, 1.0)] * n
    eq_constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - desired_sum}
    local_res = minimize(
        compute_c5,
        init_vec,
        method='SLSQP',
        bounds=bounds,
        constraints=eq_constraint,
        options={'maxiter': 800, 'ftol': 1e-9, 'disp': False}
    )
    if local_res.success:
        best_h = project_to_feasible(local_res.x)
    else:
        best_h = init_vec.copy()
    best_score = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Phase 2: simple stochastic perturbation search within remaining budget
    # Parameters for perturbations
    sigma = 0.2          # initial noise level
    decay = 0.995        # exponential decay per iteration
    min_sigma = 0.01

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.5:
            break

        # Generate a perturbed candidate
        cand = best_h + sigma * np.random.randn(n)
        cand = np.clip(cand, 0.0, 1.0)
        cand = project_to_feasible(cand)

        cand_score = compute_c5(cand)
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score

        sigma = max(min_sigma, sigma * decay)

    # Final safety projection and verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_feasible(best_h)
    final_score = compute_c5(best_h)

    # Consistency checks (won't raise unless something is seriously wrong)
    assert best_h.shape[0] == n
    assert np.all(best_h >= -1e-9) and np.all(best_h <= 1.0 + 1e-9)
    assert abs(best_h.sum() - desired_sum) < 1e-8

    return best_h, float(final_score), n
