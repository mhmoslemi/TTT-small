import time
import numpy as np

def _compute_c5(h, dx):
    """Maximum overlap integral for discrete step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that minimizes
    C5 = max_k ∫ h(x)(1-h(x+k))dx.

    Returns (h_values, c5_bound, n_points)
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine initial size and vector
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        # Ensure the integral constraint holds (might be slightly off)
        target_sum = n / 2.0
        cur_sum = h0.sum()
        if cur_sum != 0:
            h0 = h0 * (target_sum / cur_sum)
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        # fallback: 47 points, uniform distribution respecting the integral
        n = 47
        h0 = np.full(n, 0.5)
        h0 = h0 * (n / 2.0) / h0.sum()
        h0 = np.clip(h0, 0.0, 1.0)

    dx = 2.0 / n

    # current and best solutions
    cur_h = h0.copy()
    cur_c5 = _compute_c5(cur_h, dx)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    # Parameters for the local search
    max_step = 0.07          # max amount moved between two coordinates
    try_interval = 0.002     # time between occasional checks (seconds)
    next_check = start_time + try_interval

    while time.time() - start_time < budget_s - 0.05:
        # Propose a small transfer of mass between two coordinates
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = rng.uniform(0, max_step)

        # Ensure bounds after transfer
        if cur_h[i] + eps > 1.0:
            eps = 1.0 - cur_h[i]
        if cur_h[j] - eps < 0.0:
            eps = cur_h[j]

            continue

        cur_h[i] += eps
        cur_h[j] -= eps

        new_c5 = _compute_c5(cur_h, dx)

        # Accept if improvement; otherwise revert (simple hill‑climb)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if new_c5 < best_c5:

                best_h = cur_h.copy()
        else:

            cur_h[i] -= eps
            cur_h[j] += eps

        if time.time() >= next_check:
