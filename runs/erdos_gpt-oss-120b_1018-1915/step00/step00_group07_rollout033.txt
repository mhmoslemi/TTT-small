import random
import numpy as np

def _compute_c5(h):
    """Return the overlap constant C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())
