import time
import numpy as np
from scipy import optimize as opt

def _compute_c5(h):
    """Return the max overlap integral for a given discretized h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def _project_to_feasible(h, n):
    """Clamp to [0,1] and enforce the exact integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    target = n / 2.0
    if total > 0:
        h = h * (target / total)
    # after scaling some entries may have left the box; clip again and renorm once more
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total > 0:
        h = h * (target / total)
    return h


def _optimize_once(start_h, maxiter=600):
    """Run a single SLSQP optimisation from start_h."""
    n = start_h.shape[0]

    cons = ({
        "type": "eq",
        "fun": lambda x: x.sum() - n / 2.0,
    })
    bounds = [(0.0, 1.0)] * n

    res = opt.minimize(
        _compute_c5,
        start_h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": maxiter, "ftol": 1e-10, "disp": False},
    )
    if not res.success:
        # fall back to projection of the initial point
        return _project_to_feasible(start_h, n), _compute_c5(start_h)
    h = _project_to_feasible(res.x, n)
    return h, _compute_c5(h)


def _multi_start_optimize(h0, budget, rng):
    """Perform several restarts within the given time budget."""
    best_h, best_val = _optimize_once(h0)
    start = time.time()
    while time.time() - start < budget:
        # perturbation magnitude shrinks as we run
        eps = 0.05 * rng.random()
        pert = h0 + eps * (rng.random(h0.shape) - 0.5)
        pert = _project_to_feasible(pert, h0.shape[0])
        h_candidate, val = _optimize_once(pert, maxiter=300)
        if val < best_val:
            best_h, best_val = h_candidate, val
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) respecting the time budget."""
    rng = np.random.default_rng(seed)
    t_start = time.time()

    best_h = None

    # ------------------------------------------------------------------ #

        level_budget = max(5.0, 0.20 * remaining)
