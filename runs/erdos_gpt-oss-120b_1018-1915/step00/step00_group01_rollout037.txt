import numpy as np
import time
import math
import random

def evaluate_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated by correlation."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def project_onto_constraints(h, target_sum):
    """Clip to [0,1] and adjust to achieve exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    diff = target_sum - current
    # simple adjustment by moving mass between admissible entries
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass: choose indices that can increase
            idxs = np.where(h < 1.0 - 1e-12)[0]
        else:
            # need to remove mass: choose indices that can decrease
            idxs = np.where(h > 1e-12)[0]
        if idxs.size == 0:
            break
        # distribute diff uniformly over chosen indices
        delta = diff / idxs.size
        # enforce bounds during update
        for i in idxs:
            if diff > 0:
                inc = min(delta, 1.0 - h[i])
                h[i] += inc
                diff -= inc
            else:
                dec = min(-delta, h[i])
                h[i] -= dec
                diff += dec
        # recompute remaining diff
        diff = target_sum - h.sum()
    # final tiny correction by adding to first free slot if needed
    if abs(diff) > 1e-9:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:
            i = free[0]
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def upscale(h, new_n):
    """Linear interpolation of h defined on [0,2) to a new grid of size new_n."""
    old_n = h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    # re‑project to satisfy sum constraint exactly
    target = new_n / 2.0
    return project_onto_constraints(h_new, target)

def hill_climb(initial_h, target_sum, max_iters, start_temp, end_temp, deadline):
    """Simple simulated‑annealing hill climb respecting the sum constraint."""
    n = initial_h.size
    best = initial_h.copy()
    best_val = evaluate_c5(best)

    for it in range(max_iters):
        if time.time() > deadline:
            break
        # linear temperature schedule
        t = start_temp + (end_temp - start_temp) * (it / max_iters)
        # propose a move: transfer ε from i to j (sum unchanged)
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = np.random.uniform(-0.05, 0.05) * t  # scale with temperature
        # compute feasible transfer respecting bounds
        if eps > 0:
            max_inc = 1.0 - best[j]
            max_dec = best[i]
            delta = min(eps, max_inc, max_dec)
        else:
            max_dec = best[j]
            max_inc = 1.0 - best[i]
            delta = max(eps, -max_dec, -max_inc)
        if abs(delta) < 1e-12:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        val = evaluate_c5(cand)
        # accept if better or by Metropolis criterion
        if val < best_val or np.random.rand() < math.exp((best_val - val) / (t + 1e-12)):
            best = cand
            best_val = val
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed)
