import numpy as np
from scipy.optimize import minimize

def _c5_cost(h, dx):
    """Return C5 value for discretized h."""
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _integral_constraint(h, dx):
    """Equality constraint: integral equals 1."""
    return np.sum(h) * dx - 1.0

def _prepare_initial(old_h, new_n):
    """Upscale old_h (length m) to length new_n by linear interpolation,
    then re‑scale to satisfy the integral exactly and clip to [0,1]."""
    m = len(old_h)
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, old_h)
    # enforce sum constraint
    dx_new = 2.0 / new_n
    total = np.sum(h_new) * dx_new
