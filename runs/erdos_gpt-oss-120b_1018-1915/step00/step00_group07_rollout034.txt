import time
import numpy as np

def _compute_c5(h, dx):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by full correlation."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)

def _repair_bounds(h):
    """Clip to [0,1] and renormalize sum to target (n/2)."""
    n = h.size
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        # fallback uniform
        return np.full_like(h, target_sum / n)
    # scale to meet sum, then clip again (rarely needed)
    h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    # final fix for tiny rounding errors
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly keeping bounds
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size:
            h[idx[0]] += diff
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h on [0,2] to minimize the Erdős C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,)
        c5_bound : float, the achieved maximal overlap
        n_points : int, number of discretization points (length of h_values)
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Prepare initial vector
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.asarray(init, dtype=float).copy()
        n = h.size
    else:
        # fallback: uniform 0.5 (satisfies integral)
        n = 80
        h = np.full(n, 0.5, dtype=float)

    # Ensure feasibility
    h = _repair_bounds(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_score = _compute_c5(best_h, dx)

    # ------------------------------------------------------------------
    # Simulated annealing / random walk
    # ------------------------------------------------------------------
    start_time = time.time()
    it = 0
    # parameters
    max_iter = 10_000_000  # safety cap
    init_step = 0.3  # max delta per move
    min_step = 0.001
    step_decay = 0.9995
    temperature = 0.05
    temp_decay = 0.9999

    step = init_step

    while it < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # propose delta
        delta = (rng.random() * 2 - 1) * step  # in [-step, step]
        new_i = h[i] + delta
        new_j = h[j] - delta
        # feasibility check
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            # reject and continue
            it += 1
            continue

        h_candidate = h.copy()
        h_candidate[i] = new_i
        h_candidate[j] = new_j

        # (no need for further repair, sum unchanged)
        score = _compute_c5(h_candidate, dx)

        # acceptance

        temperature *= temp_decay

    best_score = _compute_c5(best_h, dx)

    return best_h, float(best_score), int(n)
