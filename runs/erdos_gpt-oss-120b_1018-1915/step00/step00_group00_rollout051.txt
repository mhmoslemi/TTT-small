import time, math
import numpy as np

def _c5_value(h, dx):
    """Compute C5 for a given step function sample vector."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Simulated‑annealing optimizer for the Erdős C₅ overlap constant.

    Returns
    -------
    h_values : numpy.ndarray
        Best discovered step‑function samples (length n_points).
    c5_bound : float
        Corresponding upper bound on C₅.
    n_points : int
        Number of discretisation points (samples over [0,2]).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # 1. Initialise problem size and starting vector
    # ----------------------------------------------------------------------
    # Prefer the parent's coarse construction if present.
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
        n_points = h.shape[0]
    else:
        # default to a modest 45‑point uniform start (sum = n/2 = 22.5)
        n_points = 45
        h = np.full(n_points, 0.5, dtype=float)

    dx = 2.0 / n_points

    # Ensure feasibility (clipping and renormalising if needed)
    np.clip(h, 0.0, 1.0, out=h)
    current_sum = h.sum()
    target_sum = n_points / 2.0
    if not np.isclose(current_sum, target_sum):
        # simple rescaling while respecting bounds
        factor = target_sum / current_sum
        h *= factor
        np.clip(h, 0.0, 1.0, out=h)
        # final correction by adding a tiny uniform offset
        deficit = target_sum - h.sum()
        if abs(deficit) > 1e-12:
            h += deficit / n_points
        np.clip(h, 0.0, 1.0, out=h)

    cur_c5 = _c5_value(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5

    # ----------------------------------------------------------------------
    # 2. Simulated‑annealing parameters
    # ----------------------------------------------------------------------
    T = 1.0               # initial temperature
    cooling = 0.9995      # geometric cooling factor
    min_T = 1e-6
    max_delta_frac = 0.08   # max fraction of a point that can be moved

    # ----------------------------------------------------------------------
    # 3. Main optimisation loop
    # ----------------------------------------------------------------------
    iteration = 0
    while True:
        # time guard – leave a thin margin for returning results
        if (time.time() - start_time) > budget_s * 0.95:
            break

        # ---- mutation: transfer a small amount from i to j (preserves sum) ----
        i, j = rng.choice(n_points, size=2, replace=False)
        # maximal safe transfer without breaking bounds
        max_delta = min(
            max_delta_frac,          # global step size ceiling
            h[i],                    # cannot subtract more than present
            1.0 - h[j]               # cannot exceed 1 at destination
        )
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # evaluate
        new_c5 = _c5_value(h_new, dx)

        # ---- acceptance test ----
        improve = new_c5 < cur_c5
        if improve or (T > min_T and rng.random() < math.exp(-(new_c5 - cur_c5) / T)):
            h = h_new
            cur_c5 = new_c5

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()

        # temperature schedule
        T *= cooling
        iteration += 1

        # occasional random restart to diversify search (probability decays)
        if iteration % 2000 == 0:
            if rng.random() < 0.02 * (T / 1.0):

    # 4. Return best discovered solution
    # ----------------------------------------------------------------------
    return best_h, float(best_c5), n_points
