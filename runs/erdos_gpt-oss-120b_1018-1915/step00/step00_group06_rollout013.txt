def _eval_score(h):
    """Return C5 score for a given h array."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

    """Create a simple feasible start: n/2 ones, n/2 zeros shuffled."""

    np.random.shuffle(h)
