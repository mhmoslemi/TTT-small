import time
import numpy as np

def _evaluate(h):
    """Return C5 value for step function h sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_bounds_sum(h, target):
    """Clip to [0,1] and adjust to hit the required sum (iterative)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            free = h < 1.0
            if not np.any(free):
                break
            delta = diff / free.sum()
            h[free] += delta
        else:
            free = h > 0.0
            if not np.any(free):
                break
            delta = diff / free.sum()
            h[free] += delta  # diff negative, so this subtracts
        h = np.clip(h, 0.0, 1.0)
    return h

def _upscale(h):
    """Double the resolution by linear interpolation and re‑project."""

    new_h = np.interp(new_x, old_x, h)
    target = new_n / 2.0
    return _project_bounds_sum(new_h, target)
