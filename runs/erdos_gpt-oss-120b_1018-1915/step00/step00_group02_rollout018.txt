import time
import numpy as np

# helper to compute C5 bound for a discrete step function
def _c5_score(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx


# up‑sample by integer factor (simple repeat, preserves total integral)
def _upsample_repeat(h, factor):
    return np.repeat(h, factor)


# perform a bounded pairwise move that keeps the total sum constant
def _propose_move(h, rng, max_step):
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # random signed step, limited by bounds
    delta = (rng.random() * 2 - 1) * max_step
    if delta > 0:
        delta = min(delta, 1.0 - h[i], h[j])
    else:
        delta = -min(-delta, h[i], 1.0 - h[j])
    if delta == 0:
        return None, None, None
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h, i, j


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples.

    The routine uses the global `initial_h_values` if present; otherwise it
    builds a uniform start.  A simple stochastic hill‑climbing search with
    pairwise moves is carried out under the time budget.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- start vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
    else:
        # fallback: uniform 0.5 on a modest grid
        n0 = 45
        h0 = np.full(n0, 0.5, dtype=np.float64)

    # up‑sample to gain resolution (factor 4 -> 180 points)
    factor = 4
    h = _upsample_repeat(h0, factor)
    n = h.shape[0]

    # ensure exact integral (sum = n/2) – the repeat already satisfies it
    target_sum = n / 2.0
    cur_sum = h.sum()
    if not np.isclose(cur_sum, target_sum):
        # small uniform correction (should be rare)
        h += (target_sum - cur_sum) / n
        np.clip(h, 0.0, 1.0, out=h)

    best_h = h.copy()
    best_score = _c5_score(best_h)

    # ----- optimization parameters ---------------------------------------
    max_step_initial = 0.08          # max absolute transfer per move
    temperature_start = 0.02
    temperature_end = 0.0002
    total_iters_est = 200_000       # rough upper bound; loop stops by time

    # exponential cooling schedule
    def _temp(it, max_it):
        frac = it / max_it
        return temperature_start * (temperature_end / temperature_start) ** frac

    it = 0
    while True:
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break
        if it >= total_iters_est:
            break

        # gradually shrink step size
        max_step = max_step_initial * (1 - it / total_iters_est)

        cand, _, _ = _propose_move(best_h, rng, max_step)
        if cand is None:
            it += 1
            continue

        cand_score = _c5_score(cand)

        # accept if better or with Metropolis probability
        if cand_score < best_score:
            best_h, best_score = cand, cand_score
        else:
            temp = _temp(it, total_iters_est)
            if rng.random() < np.exp(-(cand_score - best_score) / max(temp, 1e-12)):
                best_h = cand
                best_score = cand_score

        it += 1

    # final evaluation (ensuring constraints)
    np.clip(best_h, 0.0, 1.0, out=best_h)

        cur = best_h.sum()
        if cur != 0:
            best_h *= target_sum / cur

    final_score = _c5_score(best_h)
    return best_h, float(final_score), int(n)
