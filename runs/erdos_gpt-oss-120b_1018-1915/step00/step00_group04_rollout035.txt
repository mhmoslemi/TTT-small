def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random

    # ------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------
    def repair(vec, target_sum):
        """Clip to [0,1] and enforce sum == target_sum (within tolerance)."""
        vec = np.clip(vec, 0.0, 1.0)
        s = vec.sum()
        if s == 0.0:
            vec = np.full_like(vec, target_sum / len(vec))
        else:
            vec = vec * (target_sum / s)
        # One more clip and tiny redistribution if needed
        vec = np.clip(vec, 0.0, 1.0)
        s = vec.sum()
        diff = target_sum - s
        if abs(diff) > 1e-12:
            mask = (vec > 0) & (vec < 1)
            if mask.any():
                vec[mask] += diff / mask.sum()
                vec = np.clip(vec, 0.0, 1.0)
        return vec

    def compute_c5(vec):
        """Maximum overlap integral for step function defined by vec."""
        return np.max(np.correlate(vec, 1.0 - vec, mode='full')) * (2.0 / len(vec))

    # ------------------------------------------------------------
    # Initialization
    # ------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # target resolution – modest size for speed but enough freedom
    n_points = 256
    target_sum = n_points / 2.0

    init = globals().get('initial_h_values', None)
    if init is not None and len(init) > 0:
        if len(init) == n_points:
            h = init.astype(float).copy()
        else:
            x_old = np.linspace(0.0, 2.0, len(init), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
            h = np.interp(x_new, x_old, init).astype(float)
    else:
        h = np.random.rand(n_points)

    h = repair(h, target_sum)

    best_h = h.copy()
    best_score = compute_c5(best_h)
    current_h = h.copy()
    current_score = best_score

    # ------------------------------------------------------------
    # Simple stochastic hill‑climbing / simulated annealing
    # ------------------------------------------------------------
    it = 0
    while time.time() - start_time < budget_s - 0.5:
        it += 1
        # temperature slowly cools; never below tiny floor
        temp = max(1e-4, 0.1 * (1.0 - it / 1_000_000.0))

        # Gaussian perturbation
        cand = current_h + np.random.normal(0.0, 0.05, size=n_points)
        cand = repair(cand, target_sum)

        cand_score = compute_c5(cand)

        # Accept if better, else with Metropolis probability
        if cand_score < current_score or np.random.rand() < math.exp(-(cand_score - current_score) / temp):
            current_h = cand
            current_score = cand_score

        # Track global best
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand.copy()

    # Final safety repair and score recompute
    best_h = repair(best_h, target_sum)
    best_score = compute_c5(best_h)

    return (best_h, float(best_score), n_points)
