import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h: np.ndarray) -> float:
    """Return C5 for a step function sampled in h (uniform grid on [0,2])."""
    n = h.shape[0]
    m = 2 * n  # zero‑pad length for linear correlation
    # rfft works on real input and is faster
    a = np.fft.rfft(h, m)
    b = np.fft.rfft(1.0 - h, m)
    corr = np.fft.irfft(a * np.conj(b), m)  # linear cross‑correlation
    max_val = np.max(corr[: 2 * n - 1])
    return max_val * (2.0 / n)


def interpolate_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate old_h (any length) to length new_n and renormalise."""
    old_n = old_h.shape[0]
    old_pos = np.arange(old_n, dtype=float) / old_n * 2.0
    new_pos = np.arange(new_n, dtype=float) / new_n * 2.0
    new_h = np.interp(new_pos, old_pos, old_h)
    new_h = np.clip(new_h, 0.0, 1.0)

    desired = new_n / 2.0
    total = new_h.sum()
    if total > 0:
        new_h *= desired / total
        new_h = np.clip(new_h, 0.0, 1.0)
        total = new_h.sum()
        new_h *= desired / total
    else:
        new_h = np.full(new_n, 0.5)

    return new_h


def local_refine(h_start: np.ndarray, time_left: float) -> tuple:
    """Run a brief SLSQP refinement if time permits."""
    n = h_start.shape[0]
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - n / 2.0}
    opts = {"maxiter": 1500, "ftol": 1e-12}
    try:
        res = minimize(
            compute_c5,
            h_start,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options=opts,
        )
        if res.success:
            return res.x, compute_c5(res.x)
    except Exception:
        pass
    return h_start, compute_c5(h_start)

    else:

    # Ensure feasibility exactly
    h = np.clip(h, 0.0, 1.0)
    h *= (n_points / 2.0) / h.sum()

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # --------------------------------------------------------------
    # 2. Simulated‑annealing style random walk
    # --------------------------------------------------------------
    max_iter = 20000
