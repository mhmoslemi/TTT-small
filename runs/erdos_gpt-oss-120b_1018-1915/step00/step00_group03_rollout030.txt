import numpy as np, time, math, random

def _evaluate_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _repair(h, target_sum, max_iter=10):
    # Clip bounds
    np.clip(h, 0.0, 1.0, out=h)
    # Adjust sum using simple capacity transfers
    cur = h.sum()
    if abs(cur - target_sum) <= 1e-12:
        return h
    delta = target_sum - cur
    for _ in range(max_iter):
        if abs(delta) < 1e-12:
            break
        # Choose indices that can absorb the delta
        if delta > 0:
            candidates = np.where(h < 1.0)[0]
            if candidates.size == 0:
                break
            idx = random.choice(candidates)
            cap = 1.0 - h[idx]
            add = min(delta, cap)
        else:
            candidates = np.where(h > 0.0)[0]
            if candidates.size == 0:
                break
            idx = random.choice(candidates)
            cap = h[idx]
            add = -min(-delta, cap)
        h[idx] += add
        delta -= add
    # One final clip for safety
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _random_feasible(n, target_sum):
    h = np.zeros(n, dtype=float)
    remaining = target_sum
    indices = list(range(n))
    random.shuffle(indices)
    for i in indices:
        if remaining <= 0:
            break
        # allowed max at this position
        max_val = min(1.0, remaining)
        val = random.random() * max_val
        h[i] = val
        remaining -= val
    # If we undershoot due to rounding, distribute leftover uniformly
    if remaining > 1e-12:
        add = remaining / n
        h += add
        np.clip(h, 0.0, 1.0, out=h)
        # final repair to fix any tiny integral drift
        h = _repair(h, target_sum)
    else:
        h = _repair(h, target_sum)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # Target number of points (balance quality & speed)
    n = 256
    target_sum = n / 2.0
    # Build initial vector
    if 'initial_h_values' in globals() and isinstance(initial_h_values, np.ndarray):
        old = initial_h_values.astype(float)
        m = old.size
        x_old = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h_cur = np.interp(x_new, x_old, old)
    else:
        h_cur = _random_feasible(n, target_sum)

    h_cur = _repair(h_cur, target_sum)
    best_h = h_cur.copy()
    best_c5 = _evaluate_c5(best_h)

    # Simple simulated annealing parameters
    init_temp = 0.05
    min_temp = 1e-4
    scale = 0.04  # mutation step size
    iter_count = 0
    # Main loop
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        iter_count += 1
        # Linear cooling
        temp = max(min_temp, init_temp * (1.0 - elapsed / (budget_s * 0.9)))
        # Propose new candidate
        cand = h_cur + scale * np.random.randn(n)
        cand = _repair(cand, target_sum)
        cand_c5 = _evaluate_c5(cand)
