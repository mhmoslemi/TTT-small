import time
import numpy as np

# evaluate_erdos_solution is provided in the execution environment.
# It returns a tuple where the first element is the C5 value for a given h.

def _initial_h(n):
    """Create a simple feasible starter: 0.5 everywhere."""
    h = np.full(n, 0.5, dtype=np.float64)
    return h

def _perturb_swap(rng, h, max_frac=0.5):
    """
    Return a new h obtained by moving some mass from one index to another.
    The total sum is preserved exactly and the bounds [0,1] are respected.
    max_frac limits the transferred amount relative to the maximum possible.
    """
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    # transfer from i to j
    avail_i = h[i]               # amount we can take from i (>=0)
    space_j = 1.0 - h[j]         # capacity we can add to j (>=0)
    max_transfer = min(avail_i, space_j)
    if max_transfer <= 0:
        return None  # impossible move
    # choose a fraction of the allowed transfer
    t = rng.random() * max_transfer * max_frac
    if t <= 0:
        return None
    new_h = h.copy()
    new_h[i] -= t
    new_h[j] += t
    return new_h

def _local_search(rng, h0, budget_s, start_time):
    """
    Simple hill‑climbing with random swaps until time expires.
    Returns the best vector and its C5 value found.
    """
    h = h0.copy()
    n = h.size
    best_h = h.copy()
    best_score = evaluate_erdos_solution(h)[0]

    # Parameters
    max_frac = 0.7          # allow fairly large moves initially
    iter_batch = 2000       # swaps per batch before checking time
    decay = 0.995           # slowly reduce max_frac to fine‑tune
    min_frac = 0.05

    while True:
        # time check
        if time.time() - start_time > budget_s * 0.97:
            break

        for _ in range(iter_batch):
            cand = _perturb_swap(rng, h, max_frac)
            if cand is None:
                continue
            cand_score = evaluate_erdos_solution(cand)[0]
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
                h = cand
            # else keep current h (greedy)

        # gradually reduce step size to refine
        max_frac = max(min_frac, max_frac * decay)

    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h sampled uniformly (length n_points).
    c5_bound : float
        The achieved C5 upper bound for the returned h.
    n_points : int
        Number of sampled points (size of h_values).
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
        # Ensure feasibility (average must be 0.5). Slight adjustments if needed.
        cur_sum = h0.sum()
        target_sum = n / 2.0
        if not np.isclose(cur_sum, target_sum):
            # Simple scaling while keeping within bounds
            scale = target_sum / cur_sum
            h0 = np.clip(h0 * scale, 0.0, 1.0)
            # Re‑balance any residual error by adding/subtracting uniformly
            residual = target_sum - h0.sum()
            if abs(residual) > 1e-9:
                h0 += residual / n
                h0 = np.clip(h0, 0.0, 1.0)
    else:

    start_time = time.time()

    # Run the local search within the time budget
