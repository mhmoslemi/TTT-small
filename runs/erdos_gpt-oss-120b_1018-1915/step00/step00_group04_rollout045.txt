import time
import random
import numpy as np

def _compute_c5(h):
    """Calculate C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _score(h):
    """Wrapper that uses the provided evaluator if present."""
    try:
        return evaluate_erdos_solution(h)  # pre‑imported by the harness
    except Exception:
        return _compute_c5(h)

def _project_sum(h, target):
    """Adjust h so that its sum equals target while staying in [0,1]."""
    current = h.sum()
    diff = target - current
    if abs(diff) < 1e-12:
        return h
    # Distribute diff uniformly, then clip and iterate a few times
    h = h + diff / h.size
    for _ in range(5):
        over = h - 1.0
        under = -h
        if over.max() > 0:
            excess = over.max()
            idx = np.argmax(over)
            h[idx] = 1.0
            diff = diff - excess
        if under.max() > 0:
            shortage = under.max()
            idx = np.argmin(h)
            h[idx] = 0.0
            diff = diff + shortage
        if abs(diff) < 1e-12:
            break
        h = h + diff / h.size
    # Final clip (tiny numerical overshoot)
    return np.clip(h, 0.0, 1.0)

def _random_feasible(n, seed):
    """Generate a random feasible h of length n."""
    rng = np.random.default_rng(seed)
    h = rng.random(n)
    target = n / 2.0
    h = _project_sum(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a feasible step function h, its C5 bound, and sample count."""
    start = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        h_current = np.array(initial_h_values, dtype=float)
        n = h_current.size
        # enforce exact sum constraint
        target_sum = n / 2.0
        h_current = _project_sum(h_current, target_sum)
    else:
        n = 84          # modest size for quick search
        h_current = _random_feasible(n, seed)

    best_h = h_current.copy()
    best_score = _score(best_h)

    target_sum = n / 2.0
    max_iter = 200000  # safety cap
    iter_count = 0
    while time.time() - start < budget_s * 0.95 and iter_count < max_iter:
        iter_count += 1

        # Propose a neighbor by moving a small amount of mass
        i, j = random.sample(range(n), 2)
        hi, hj = best_h[i], best_h[j]
        max_move = min(hi, 1.0 - hj)
        if max_move <= 0:
            continue
        delta = random.uniform(0.0, max_move * 0.5)  # move at most half of allowed
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # occasional jitter to escape flat regions
        if random.random() < 0.01:
            idx = random.randrange(n)
            perturb = random.uniform(-0.1, 0.1)
            h_new[idx] = np.clip(h_new[idx] + perturb, 0.0, 1.0)

        # Repair sum (should already be exact, but numerical drift possible)
        h_new = _project_sum(h_new, target_sum)

        score = _score(h_new)
        if score < best_score:
            best_h, best_score = h_new, score

        # occasional random restart if stuck
        if iter_count % 5000 == 0 and time.time() - start < budget_s * 0.9:
            # keep the best found, try new random around it
            h_current = best_h + np.random.normal(0, 0.02, size=n)
            h_current = np.clip(h_current, 0.0, 1.0)
            h_current = _project_sum(h_current, target_sum)
            # accept if not worse
            sc = _score(h_current)
            if sc < best_score:
                best_h, best_score = h_current, sc

    # Final safeguard: ensure feasibility
    best_h = _project_sum(best_h, target_sum)

    return best_h, best_score, n
