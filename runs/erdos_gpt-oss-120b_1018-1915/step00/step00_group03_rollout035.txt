import numpy as np
import time

def _evaluate(h, dx):
    """Return the overlap constant C5 for a given step function h."""
    # full convolution, then scale by dx
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _repair(h, target_sum, max_iter=12):
    """Project h onto the feasible set: 0≤h≤1 and sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        if abs(s - target_sum) <= 1e-10:
            break
        scale = target_sum / s if s != 0 else 0.0
        h = h * scale
        h = np.clip(h, 0.0, 1.0)
    # final residual fix (very small)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) > 1e-8:
        # distribute the residual greedily while respecting bounds
        for i in range(len(h)):
            if diff == 0:
                break
            if diff > 0:
                cap = 1.0 - h[i]
                adj = min(diff, cap)
            else:
                cap = -h[i]
                adj = max(diff, cap)
            h[i] += adj
            diff -= adj
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] with integral 1 and values in [0,1]
    to minimise the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best-found discretised step function (length n_points).
    c5_bound : float
        The evaluated C5 value for h_values.
    n_points : int
        Length of the returned vector.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise h
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)  # provided by harness
    except NameError:
        # if no parent, start from the uniform feasible vector
        n0 = 70
        h0 = np.full(n0, 0.5, dtype=float)

    n = h0.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0
    h0 = _repair(h0, target_sum)

    best_h = h0.copy()
    best_score = _evaluate(best_h, dx)

    # ------------------------------------------------------------------
    # 2. Simple stochastic local search with annealing‑like schedule
    # ------------------------------------------------------------------
    iteration = 0
    # keep a small archive of good candidates for occasional recombination
    archive = [best_h.copy()]

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:   # stop a bit before hard limit
            break

        # annealing schedule for mutation strength
        frac = elapsed / budget_s
        sigma = 0.30 * (1.0 - frac) + 0.008   # decays from ~0.30 to ~0.008

        # ----- generate candidate -----
        if np.random.rand() < 0.15 and len(archive) > 1:
            # recombine two archive members (crossover‑like)
            a, b = np.random.choice(archive, 2, replace=False)
            mask = np.random.rand(n) < 0.5
            cand = np.where(mask, a, b)
        else:
            cand = best_h + np.random.normal(scale=sigma, size=n)

        cand = _repair(cand, target_sum)
        score = _evaluate(cand, dx)

        iteration += 1

        # ----- accept improvements -----
        if score < best_score - 1e-7:

            best_h = cand.copy()

                best_score, best_h = rand_score, rand_vec.copy()
                archive.append(rand_vec.copy())
                if len(archive) > 20:
                    archive.pop(0)
