import numpy as np
import time

def _project_bounds_sum(h, target):
    """Project h onto [0,1] with sum == target using simple iterative clipping."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # iterate a few times; convergence is fast for this small problem
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        mask = (diff > 0) & (h < 1.0) | (diff < 0) & (h > 0.0)
        if not np.any(mask):
            break
        adjustment = diff / np.count_nonzero(mask)
        h[mask] += adjustment
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # final tiny correction by uniform shift if still off (should be negligible)
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _evaluate(h):
    """Return C5 for a discretized step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

    """Simple stochastic hill‑climbing within the given time budget."""
    rng = np.random.default_rng()
    best_h = start_h.copy()
    best_c = _evaluate(best_h)
    n = len(best_h)
