import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for a step function h sampled on [0,2]."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * (2.0 / n)

def repair(h, target_sum):
    """Clip to [0,1] and adjust to meet the exact integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    # Distribute the difference proportionally over free entries
    free = (h > 0.0) & (h < 1.0)
    if not free.any():
        # fallback: scale everything and clip again
        h = np.clip(h * (target_sum / cur), 0.0, 1.0)
        return h
    diff = target_sum - cur
    while abs(diff) > 1e-12:
        free_idx = np.where(free)[0]
        if diff > 0:
            capacity = 1.0 - h[free_idx]
            add = np.minimum(diff / free_idx.size, capacity)
            h[free_idx] += add
            diff -= add.sum()
        else:
            capacity = h[free_idx]
            sub = np.minimum(-diff / free_idx.size, capacity)
            h[free_idx] -= sub
            diff += sub.sum()
        free = (h > 0.0) & (h < 1.0)
        if not free.any():
            break
    # tiny numerical clean‑up
    h = np.clip(h, 0.0, 1.0)
    return h

def upscale(h, new_n):
    """Interpolate to a higher resolution while preserving the integral."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    target = new_n / 2.0
    h_new = repair(h_new, target)
    return h_new

def random_move(h, rng):
    """Propose a feasible move that keeps sum constant and respects bounds."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    # feasible delta interval
    delta_pos = min(1.0 - hi, hj)   # increase i, decrease j
    delta_neg = -min(hi, 1.0 - hj)  # decrease i, increase j
    if delta_pos == delta_neg == 0:
        return None, None, None
    delta = rng.uniform(delta_neg, delta_pos)
    return i, j, delta

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h_cur = np.asarray(initial_h_values, dtype=float).copy()
        n_cur = h_cur.shape[0]
        # ensure feasibility
        target = n_cur / 2.0
        h_cur = repair(h_cur, target)
    else:
        n_cur = 84                     # a modest default size
        h_cur = rng.uniform(0.0, 1.0, n_cur)
        h_cur = repair(h_cur, n_cur / 2.0)

    best_h = h_cur.copy()
    best_score = compute_c5(best_h)

    # ----- optimisation parameters ------------------------------------
    max_n = 256                      # highest resolution we will attempt
    upscale_interval = 30.0          # seconds between possible up‑scales
    next_upscale = start_time + upscale_interval
    iteration_batch = 2000          # moves before checking time

    while True:
        now = time.time()
        if now - start_time >= budget_s - 0.5:
            break

        # ----- local search moves ------------------------------------
        for _ in range(iteration_batch):
            move = random_move(h_cur, rng)
            if move[0] is None:
                continue
            i, j, delta = move
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta
            # repair (only clipping needed because sum is preserved)
            h_new = np.clip(h_new, 0.0, 1.0)

            # re‑balance any tiny sum error
            target = h_new.shape[0] / 2.0
            cur_sum = h_new.sum()
            if abs(cur_sum - target) > 1e-12:
                h_new = repair(h_new, target)

            new_score = compute_c5(h_new)

            # simulated‑annealing acceptance
            temp = max(0.001, (budget_s - (now - start_time)) / budget_s)
            accept = False
            if new_score < best_score:
                accept = True
            else:
                prob = np.exp(-(new_score - best_score) / temp)
                if rng.random() < prob:
                    accept = True

            if accept:
                h_cur = h_new
                if new_score < best_score:
                    best_score = new_score
                    best_h = h_new.copy()

        # ----- possible resolution increase ---------------------------
        now = time.time()
        if now >= next_upscale and n_cur < max_n:
            # interpolate to double resolution (or the next power of two)
            new_n = min(max_n, n_cur * 2)
            best_h = upscale(best_h, new_n)
            h_cur = best_h.copy()
            n_cur = new_n
            best_score = compute_c5(best_h)
            next_upscale = now + upscale_interval

    # final repair to guarantee exact integral
    final_n = best_h.shape[0]
    best_h = repair(best_h, final_n / 2.0)
    final_score = compute_c5(best_h)

    return best_h, float(final_score), int(final_n)
