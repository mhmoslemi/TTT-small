import time
import numpy as np

def _eval_c5(h):
    """Return the C5 value for step function values h over [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h * (1-h) with full mode
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_feasible(n, rng):
    """Generate a random feasible h of length n (values in [0,1] and sum = n/2)."""
    # start with n/2 ones and n/2 zeros, shuffled, then add small noise and clip
    half = n // 2
    h = np.zeros(n, dtype=np.float32)
    h[:half] = 1.0
    rng.shuffle(h)
    # add small uniform noise while keeping the sum exact
    noise = rng.uniform(-0.02, 0.02, size=n).astype(np.float32)
    h = np.clip(h + noise, 0.0, 1.0)
    # adjust to meet exact sum constraint
    current = h.sum()
    target = n / 2.0
    diff = target - current
    # distribute the difference uniformly (will stay in bounds because diff is tiny)
    h += diff / n
    np.clip(h, 0.0, 1.0, out=h)
    # final renormalisation to guarantee exact sum (tiny numeric tweak)
    scale = target / h.sum()
    h *= scale
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _improve(h, time_budget, max_iter=5000, rng=None):
    """Local improvement by random mass transfers preserving constraints."""
    if rng is None:
        rng = np.random.default_rng()
    n = h.shape[0]
    dx = 2.0 / n
    best = h.copy()
    best_score = _eval_c5(best)
    start = time.time()
    it = 0
    while it < max_iter and (time.time() - start) < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a small amount delta from i to j
        delta = rng.uniform(-0.04, 0.04)
        hi, hj = best[i], best[j]
        new_hi = hi - delta
        new_hj = hj + delta
        if not (0.0 <= new_hi <= 1.0 and 0.0 <= new_hj <= 1.0):
            continue
        cand = best.copy()
        cand[i] = new_hi
        cand[j] = new_hj
        score = _eval_c5(cand)
        if score < best_score:
            best = cand
            best_score = score
        it += 1
    return best, best_score
