import time
import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds

def _dx(n):
    return 2.0 / n

def _evaluate_c5(h):
    n = h.size
    dx = _dx(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _make_feasible_random(n, target_sum, rng):
    # start with Dirichlet-like random vector
    v = rng.random(n)
    v = v / v.sum() * target_sum
    for _ in range(10):
        np.clip(v, 0.0, 1.0, out=v)
        cur = v.sum()
        if abs(cur - target_sum) < 1e-8:
            break
        # distribute the difference uniformly
        v += (target_sum - cur) / n
    # final clip
    np.clip(v, 0.0, 1.0, out=v)
    return v

def _upsample(old_h, new_n):
    old_n = old_h.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + _dx(old_n) / 2
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + _dx(new_n) / 2
    return np.interp(new_x, old_x, old_h)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine base resolution and starting vector
    try:
        base_h = np.array(initial_h_values, dtype=float)  # provided by harness
        base_n = base_h.size
    except Exception:
        base_h = None
        base_n = None

    # Candidate resolutions (balance quality vs time)
    resolutions = [47, 94, 188, 376]  # up to ~400 points
    best_h = None
    best_c5 = np.inf
    best_n = None

    for n in resolutions:
        # Time check
        if time.time() - start_time > budget_s * 0.95:
            break

        dx = _dx(n)
        target_sum = n / 2.0

        # Prepare initial guess
        if base_h is not None:
            if base_n == n:
                h0 = base_h.copy()
            else:
                h0 = _upsample(base_h, n)
        else:
            h0 = _make_feasible_random(n, target_sum, rng)

        # Ensure feasibility
        np.clip(h0, 0.0, 1.0, out=h0)
        # Adjust sum to target (simple projection)
        diff = target_sum - h0.sum()
        h0 += diff / n
        np.clip(h0, 0.0, 1.0, out=h0)

        # Linear equality constraint: sum(h) = target_sum
        lin_con = LinearConstraint(np.ones((1, n)), [target_sum], [target_sum])
        bounds = Bounds(0.0, 1.0)

        # Objective wrapper
        def obj(v):
            return _evaluate_c5(v)

        # Run a few restarts per resolution
        attempts = 3
        for _ in range(attempts):
            if time.time() - start_time > budget_s * 0.95:
                break
            res = minimize(
                obj,
                h0,
                method="SLSQP",
                bounds=bounds,
                constraints=[lin_con],
                options={"maxiter": 200, "ftol": 1e-9, "disp": False},
            )
            if not res.success:
                # try a fresh random start if failed
                h0 = _make_feasible_random(n, target_sum, rng)
                continue

            h_opt = np.clip(res.x, 0.0, 1.0)
            c5_val = _evaluate_c5(h_opt)

            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                best_n = n

            # Prepare next start close to current optimum
            h0 = h_opt + rng.normal(scale=0.02, size=n)
            np.clip(h0, 0.0, 1.0, out=h0)
            # re‑project sum
            diff = target_sum - h0.sum()
            h0 += diff / n
            np.clip(h0, 0.0, 1.0, out=h0)

    # Fallback: if nothing succeeded, return a simple half‑ones vector
    if best_h is None:
        best_n = 100
        best_h = np.full(best_n, 0.5)
        best_c5 = _evaluate_c5(best_h)

    return (best_h, float(best_c5), int(best_n))
