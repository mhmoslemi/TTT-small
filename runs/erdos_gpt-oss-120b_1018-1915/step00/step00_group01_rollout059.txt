import time
import math
import random
import numpy as np

def _evaluate(h):
    """Return max overlap integral C5 for discretized h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_constraints(h, target_sum):
    """
    Clip values to [0,1] and adjust the total sum to target_sum
    (the integral constraint). Uses a simple scaling + redistribution.
    """
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        # distribute uniformly
        h[:] = target_sum / len(h)
        return h

    # scale to hit the target, then re‑clip
    h *= target_sum / cur
    h = np.clip(h, 0.0, 1.0)

    # redistribute any small residual error
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # free indices are those not at a bound
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size > 0:
            h[free] += residual / free.size
            h = np.clip(h, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimizes
    C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise a feasible solution.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
        target = n / 2.0  # because ∫ h = 1 ⇒ sum h * dx = 1 ⇒ sum h = n/2
        h = _project_to_constraints(h, target)
    else:
        # fallback: start from a random feasible vector of moderate size
        n = 200
        h = np.random.rand(n)
        target = n / 2.0
        h = _project_to_constraints(h, target)

    best_h = h.copy()
    best_cost = _evaluate(best_h)

    # ------------------------------------------------------------------
    # Simple stochastic local search within the time budget.
    # ------------------------------------------------------------------
    target_sum = n / 2.0
    step_size = 0.05          # initial magnitude of random perturbations
    max_iters = 200000        # generous upper bound; loop stops by time

    while time.time() - start_time < budget_s * 0.90 and max_iters > 0:
        max_iters -= 1

        # Propose a random perturbation
        delta = (np.random.rand(n) - 0.5) * 2.0 * step_size
        h_candidate = best_h + delta

        # Enforce constraints
        h_candidate = _project_to_constraints(h_candidate, target_sum)

        # Evaluate
        cand_cost = _evaluate(h_candidate)

        # Accept improvements
        if cand_cost < best_cost:
            best_cost = cand_cost
            best_h = h_candidate
            # gradually reduce step size to fine‑tune
            step_size = max(step_size * 0.99, 1e-4)
        else:
            # occasional slight increase to escape flat regions
            step_size = min(step_size * 1.001, 0.2)

    # ------------------------------------------------------------------
    # Return the best discovered solution.
    # ------------------------------------------------------------------
    return best_h.tolist(), float(best_cost), len(best_h)
