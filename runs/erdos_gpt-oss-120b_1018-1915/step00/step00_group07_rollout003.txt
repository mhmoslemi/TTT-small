import numpy as np
import time
import math
import random

# Pre‑imported in the harness:
# evaluate_erdos_solution(h) -> float
# initial_h_values                (optional)

def _compute_c5(h, dx):
    """Raw C5 computation (fallback if evaluate_erdos_solution unavailable)."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _evaluate(h, dx):
    """Use the harness helper if present, else fall back."""
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        return _compute_c5(h, dx)

def _feasible_delta(h_i, h_j):
    """Return a feasible interval [low, high] for moving mass from i to j."""
    low = max(-h_i, h_j - 1.0)      # cannot go below 0 or above 1
    high = min(h_i, 1.0 - h_j)
    return low, high

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------
    # 1. Initialise h
    # ------------------------------------------------------------

        h = np.full(n_points, 0.5, dtype=float)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0          # because sum(h)*dx = 1
    cur_sum = h.sum()
    if not np.isclose(cur_sum, target_sum):
        # Simple scaling then clipping to stay in [0,1] while preserving total mass
        scaling = target_sum / cur_sum
        h *= scaling
        # redistribute any excess/deficit caused by clipping
        excess = h.sum() - target_sum
        if abs(excess) > 1e-12:
            # distribute excess uniformly over entries that are not at a bound
            mask = (h > 0.0) & (h < 1.0)
            if mask.any():
                adjustment = -excess / mask.sum()
                h[mask] += adjustment
        # final clip
        h = np.clip(h, 0.0, 1.0)

    # ------------------------------------------------------------
    # 2. Baseline evaluation
    # ------------------------------------------------------------
    best_h = h.copy()
    best_score = _evaluate(best_h, dx)

    start_time = time.time()

    while elapsed < budget_s and max_iters > 0:

        # pick two distinct indices

        h[j] += delta
