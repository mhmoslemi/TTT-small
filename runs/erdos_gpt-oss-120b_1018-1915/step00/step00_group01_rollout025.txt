import numpy as np
import time
import random

# helper: compute C5 for a discrete h
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

# project a vector onto the feasible set 0≤h≤1 and sum = target
def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to fix with two indices
    max_iter = 100
    for _ in range(max_iter):
        i, j = np.random.choice(h.size, 2, replace=False)
        # move amount from j to i
        move = diff
        # bound the move so both stay in [0,1]
        lo = max(-h[i], move - (1.0 - h[j]))
        hi = min(1.0 - h[i], move + h[j])
        if lo <= hi:
            move = np.clip(move, lo, hi)
            h[i] += move
            h[j] -= move
            diff = target_sum - h.sum()
            if abs(diff) < 1e-12:
                break
    # if still not exact, scale uniformly (will stay close)
    if abs(diff) > 1e-9:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    TIME_BUFFER = 0.5  # leave a little slack

    # choose resolution (trade‑off between quality and speed)
    n_points = 200  # even for exact target sum = n/2
    target_sum = n_points / 2.0

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        # up‑sample / interpolate to desired resolution
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x, x0, h0, period=2.0)
    else:
        # start from a simple alternating pattern (roughly 0/1)
        h = np.zeros(n_points, dtype=float)
        half = n_points // 2
        h[:half] = 1.0
        h[half:] = 0.0
        np.random.shuffle(h)

    h = project_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Simple simulated‑annealing hill‑climber
    # ------------------------------------------------------------------
    max_iter = 200000
    temp0 = 0.02
    temp_end = 5e-4
    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s - TIME_BUFFER:
            break

        # linear temperature schedule
        t = temp0 + (temp_end - temp0) * (it / max_iter)

        # propose a local move affecting two positions
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = (np.random.rand() - 0.5) * 0.06  # ±0.03
        hi_new = h[i] + delta
        hj_new = h[j] - delta
        if not (0.0 <= hi_new <= 1.0 and 0.0 <= hj_new <= 1.0):
            continue

        h_candidate = h.copy()
        h_candidate[i] = hi_new
        h_candidate[j] = hj_new

        c5_candidate = compute_c5(h_candidate)
