import time
import numpy as np

def _compute_score(h, dx):
    """Return max overlap integral for a given h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx

def _enforce_constraints(h, target_sum):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target_sum - cur
    # Distribute the difference proportionally to available slack
    if abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - h
        else:
            slack = h
        cap = slack.sum()
        if cap > 0:
            h += diff * slack / cap
        # one more clip to guard numeric overshoot
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h improving the Erdős C5 bound.

    Returns:
        h_values (list of float): h sampled on a uniform grid of length n_points.
        c5_bound (float): the achieved C5 value (the max overlap integral).
        n_points (int): number of sampling points (grid size).

    rng = np.random.default_rng(seed)

        i, j = rng.integers(0, n_points, size=2)
