import time
import numpy as np

def _evaluate_h(h, dx):
    """Return the C5 value for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _mutate(h):
    """Transfer a random amount of mass between two coordinates while keeping
    each entry inside [0,1] and preserving the total sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    # feasible delta range so that both stay in [0,1]
    delta_max = min(1.0 - hi, hj)          # amount moved from j to i
    delta_min = -min(hi, 1.0 - hj)         # amount moved from i to j
    if delta_max <= delta_min:
        return h  # nothing feasible; extremely unlikely

    delta = np.random.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    np.random.seed(seed)
    start_time = time.time()

    # ----- discretization -----
    n_points = 45                     # preferred small size
    dx = 2.0 / n_points
    required_sum = n_points / 2.0    # because ∫h = 1

    # ----- initial solution -----
    if "initial_h_values" in globals():
        h_current = np.array(initial_h_values, dtype=float)
        # ensure correct length (parent may provide different size)
        if h_current.size != n_points:
            # resample or truncate/pad uniformly to required size
            if h_current.size > n_points:
                h_current = h_current[:n_points]
            else:
                pad = np.full(n_points - h_current.size, 0.5)
                h_current = np.concatenate([h_current, pad])
        # force sum to required value by a simple scaling and clipping
        diff = required_sum - h_current.sum()
        if abs(diff) > 1e-12:
            # distribute diff uniformly while respecting bounds
            inc = diff / n_points
            h_current = np.clip(h_current + inc, 0.0, 1.0)
            # renormalize if needed
            correction = required_sum - h_current.sum()
            if abs(correction) > 1e-12:
                # small final adjustment on a random index
                idx = np.random.randint(n_points)
                h_current[idx] = np.clip(h_current[idx] + correction, 0.0, 1.0)
    else:
        h_current = np.full(n_points, 0.5, dtype=float)  # uniform satisfies sum

    best_h = h_current.copy()

    # ----- simulated annealing parameters -----
