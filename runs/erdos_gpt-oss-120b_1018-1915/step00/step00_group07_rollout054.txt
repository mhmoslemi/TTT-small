import time
import numpy as np

def _repair(h, target_sum, max_iter=10):
    """Project h onto [0,1] and enforce sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        if abs(s - target_sum) <= 1e-9:
            break
        if s == 0.0:  # avoid division by zero
            h[:] = target_sum / len(h)
            break
        scale = target_sum / s
        h = np.clip(h * scale, 0.0, 1.0)
    # final scaling to hit target exactly (may slightly violate bounds, fix)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 0) & (h < 1)
        if free.any():
            add = diff / free.sum()
            h[free] += add
            h = np.clip(h, 0.0, 1.0)
    return h

def _evaluate(h, dx):
    """Compute C5 for a given step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
