import numpy as np
import time, math, random

def _capped_simplex_projection(v, s, max_iter=50):
    """Project v onto {0 <= x_i <= 1, sum x_i = s}."""
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        total = w.sum()
        if total < s:
            hi = lam
        else:
            lo = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h):
    """Compute C5 for a step function sampled in h."""
    n = h.shape[0]
    dx = 2.0 / n
    size = 2 * n - 1
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    conv = np.fft.irfft(H * G, size)
    return dx * conv.max()


def _interpolate(initial, new_n):
    """Linear interpolation of the initial vector to length new_n."""
    old_n = initial.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    interp = np.interp(new_x, old_x, initial)
    # enforce the integral constraint exactly
    desired = new_n / 2.0
    return _capped_simplex_projection(interp, desired)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)

    h_values : 1‑D numpy array of length n_points, entries in [0,1],
               sum(h_values) = n_points/2.
    c5_bound : the achieved C5 value (float).
    n_points : number of samples used.
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # 1. Choose a working resolution (balance quality vs speed)
    # ------------------------------------------------------------------
    n_points = 380                     # 4 × the provided 95‑point parent
    # ------------------------------------------------------------------
    # 2. Initialise from the supplied parent if available
    # ------------------------------------------------------------------
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.ndim != 1:
            raise ValueError
        h = _interpolate(parent, n_points)
    except Exception:
        # fallback: simple half‑ones, half‑zeros pattern
        h = np.zeros(n_points, dtype=float)
        h[: n_points // 2] = 1.0
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ------------------------------------------------------------------
    # 3. Simulated‑annealing style local search
    # ------------------------------------------------------------------
    max_iter = 200000                     # hard cap; loop ends earlier by time
    step_size = 0.12                      # typical amount moved per swap
    temp0 = 0.02                          # initial temperature
    for it in range(1, max_iter + 1):
        # time guard
        if time.time() - start_time > budget_s * 0.94:
            break

        # temperature schedule (linear decay)
        t = it / max_iter
        temp = temp0 * (1.0 - t)

        # propose a transfer between two random indices
        i, j = random.sample(range(n_points), 2)
        # amount that can be moved without breaking bounds
        max_sub = min(step_size, h[i])
        max_add = min(step_size, 1.0 - h[j])
        amount = min(max_sub, max_add)
        if amount <= 0.0:
            continue
        # randomise the exact transferred mass
        amount *= random.random()
        h_new = h.copy()
        h_new[i] -= amount
        h_new[j] += amount

        cand_c5 = _c5_value(h_new)

        # acceptance rule

                best_h = h_jit.copy()
                best_c5 = c5_jit

    # final projection (safety)
    best_h = _capped_simplex_projection(best_h, n_points / 2.0)

    return best_h, float(best_c5), int(n_points)
