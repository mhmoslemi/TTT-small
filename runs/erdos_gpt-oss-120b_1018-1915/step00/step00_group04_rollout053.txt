import time
import numpy as np

def _repair(h):
    """Project h onto [0,1] and enforce sum(h)=len(h)/2."""
    n = h.shape[0]
    target = n * 0.5
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        mask = (h > 0.0) & (h < 1.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[mask] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny adjustment if needed
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-10:
        for i in range(n):
            if 0.0 <= h[i] + diff <= 1.0:
                h[i] += diff
                break
    return h

def _random_valid(n):
    """Create a random feasible vector of length n."""
    h = np.random.rand(n)
    return _repair(h)

def _objective(h):
    """Compute C5 for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # resolutions to explore (must be even)
    resolutions = [42, 84, 168, 336]
    # start with parent if available
    if 'initial_h_values' in globals():
        h_cur = np.asarray(initial_h_values, dtype=float)

        best_h = h_cur.copy()
        best_score = _objective(best_h)
    else:
        n0 = resolutions[0]
        h_cur = _random_valid(n0)
        best_h = h_cur.copy()
        best_score = _objective(best_h)

    for n in resolutions:
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break
        # upscale if needed
        if h_cur.shape[0] != n:
            x_old = np.linspace(0, 2, h_cur.shape[0], endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h_cur = np.interp(x_new, x_old, h_cur)
            h_cur = _repair(h_cur)
        # local search parameters
