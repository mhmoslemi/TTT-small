import time
import numpy as np

def _compute_c5(h, dx):
    """Return C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(h0, target_sum, max_iter=60, tol=1e-10):
    """
    Project vector h0 onto the box [0,1] with the linear equality sum(h)=target_sum.
    This is done by finding λ such that h = clip(h0+λ,0,1) satisfies the sum.
    """
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        h = np.clip(h0 + lam, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) <= tol:
            return h
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    # final clipping
    return np.clip(h0 + (lo + hi) / 2.0, 0.0, 1.0)

def _initial_h(n_points):
    """Create an initial feasible h using the provided global initial_h_values if
    it matches the desired length, otherwise interpolate it."""
    try:
        base = np.asarray(initial_h_values, dtype=float)
    except NameError:
        # fallback: uniform distribution
        base = np.full(47, 1.0 / 47)
    if len(base) == n_points:
        h = base.copy()
    else:
        x_old = np.linspace(0.0, 2.0, len(base), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_old, base)
    # enforce exact integral constraint
    target = n_points / 2.0  # because sum(h) * dx = 1 and dx = 2/n
    h = _project_to_constraints(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimizer for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Choose a modest resolution that balances quality & speed
    n_points = 200  # can be increased if time permits
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Start from an interpolated version of the global initial solution
    h_best = _initial_h(n_points)
    c5_best = _compute_c5(h_best, dx)

    # Parameters for the stochastic search
    max_step = 0.08          # maximum perturbation magnitude
    init_temp = 0.02
    min_temp = 0.0005

    # Main loop respecting the time budget
    while time.time() - start_time < budget_s * 0.94:
        elapsed = time.time() - start_time
        # linear temperature schedule
        temp = init_temp * (1 - elapsed / budget_s) + min_temp * (elapsed / budget_s)

        # propose a small move that preserves the sum after projection
        h_candidate = h_best.copy()
        i, j = rng.choice(n_points, size=2, replace=False)
        delta = rng.uniform(-max_step, max_step)
        h_candidate[i] += delta
        h_candidate[j] -= delta

        # project back onto feasible set
        h_candidate = _project_to_constraints(h_candidate, target_sum)

        c5_candidate = _compute_c5(h_candidate, dx)

        # accept if better, otherwise Metropolis acceptance
