import time
import numpy as np

def _project_to_bounds(v, target_sum, max_iter=20):
    """Project vector v onto the box [0,1] with given sum."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = target_sum - s
        if abs(diff) < 1e-10:
            break
        free = ((x > 0.0) & (x < 1.0)).astype(float)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds – distribute uniformly and re‑clip
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
            continue
        delta = diff / cnt
        x += delta * free
        x = np.clip(x, 0.0, 1.0)
    return x

def _evaluate_c5(h, dx):
    """Return max overlap C5 for a given step vector."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def _upscale(h):
    """Double resolution by linear interpolation."""
    n = len(h)
    new_n = n * 2
    # points are mid‑interval samples
    x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(x_new, x_old, h)
    return np.clip(h_new, 0.0, 1.0)

def _random_mutation(parent, scale=0.2, k=3):
    """Perturb k random coordinates of parent."""
    h = parent.copy()
    n = len(h)
    idx = np.random.choice(n, size=k, replace=False)
    h[idx] += np.random.normal(0.0, scale, size=k)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function that (approximately) minimizes C5.

    Returns
    -------
    h_values : np.ndarray
        Length‑n array with values in [0,1] and integral 1 over [0,2].
    c5_bound : float
        Estimated C5 for the returned function.
    n_points : int
        Length of the returned array.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # initial construction
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.array(init, dtype=float)
    else:
        # fallback: uniform 0.5 values (integral satisfied automatically)
        n0 = 128
        h = np.full(n0, 0.5, dtype=float)

    n = len(h)
    target_sum = n / 2.0
    h = _project_to_bounds(h, target_sum)

    dx = 2.0 / n
    best_h = h.copy()
    best_val = _evaluate_c5(best_h, dx)

    # ------------------------------------------------------------------
    # optimisation parameters
    mutation_scale = 0.25
    mutation_decay = 0.99
    evals = 0
    no_improve = 0

    while time.time() < deadline:
        # basic hill‑climbing step
        cand = _random_mutation(best_h, scale=mutation_scale, k=4)
        cand = _project_to_bounds(cand, target_sum)

        val = _evaluate_c5(cand, dx)
        evals += 1

        if val < best_val:
            best_h, best_val = cand, val
            no_improve = 0
        else:
            no_improve += 1

        # decay mutation scale gradually
        mutation_scale *= mutation_decay

        # occasional restart from a perturbed best (to escape flat regions)
        if evals % 500 == 0:
            restart = _random_mutation(best_h, scale=0.5, k=8)
            restart = _project_to_bounds(restart, target_sum)
            r_val = _evaluate_c5(restart, dx)
            evals += 1
            if r_val < best_val:
                best_h, best_val = restart, r_val
                no_improve = 0

        # if stagnation occurs, try to increase resolution
        if no_improve >= 2000 and n < 1024:
            # upscale
            best_h = _upscale(best_h)
            n = len(best_h)
            target_sum = n / 2.0
            dx = 2.0 / n
            best_h = _project_to_bounds(best_h, target_sum)
            best_val = _evaluate_c5(best_h, dx)
            # reset parameters for finer grid
            mutation_scale = 0.2
            no_improve = 0

    return best_h, best_val, len(best_h)
