import time
import random
import numpy as np

def _c5_value(h):
    """Return the C5 value for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _hill_climb(initial_h, time_limit, max_iters=20000):
    """Simple hill‑climbing that preserves sum and box constraints."""
    h = initial_h.copy()
    n = h.size
    best_h = h.copy()
    best_c = _c5_value(h)

    # pre‑compute index pools for quick selection
    idxs = np.arange(n)
    start = time.time()
    for _ in range(max_iters):
        if time.time() >= time_limit:
            break

        # choose donor (with positive mass) and receiver (with room)
        donor = random.choice(idxs[h > 1e-12])
        receiver = random.choice(idxs[h < 1.0 - 1e-12])
        if donor == receiver:
            continue

        # transfer a small random amount, keeping bounds
        max_eps = min(0.05, h[donor], 1.0 - h[receiver])
        if max_eps <= 1e-12:
            continue
        eps = random.uniform(1e-4, max_eps)

        # perform move
        h[donor] -= eps
        h[receiver] += eps

        c = _c5_value(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        else:
            # revert move
            h[donor] += eps
            h[receiver] -= eps
    return best_h, best_c

def _interpolate_to_n(source_h, source_n, target_n):
    """Linearly interpolate a step vector to a different resolution."""
    if source_n == target_n:
        return source_h.copy()
    xs = np.linspace(0.0, 2.0, source_n, endpoint=False)
    xt = np.linspace(0.0, 2.0, target_n, endpoint=False)
    interp = np.interp(xt, xs, source_h, period=2.0)
    # enforce sum constraint exactly by a tiny redistribution
    target_sum = target_n / 2.0
    cur_sum = interp.sum()
    if cur_sum == 0:
        interp[:] = 0.5
    else:
        interp *= target_sum / cur_sum
    # clip and fix any tiny violations of the box constraints
    np.clip(interp, 0.0, 1.0, out=interp)
    # final renormalisation by a simple redistribution loop
    deficit = target_sum - interp.sum()
    if abs(deficit) > 1e-12:
        # add/subtract uniformly on non‑saturated entries
        free = np.where((interp > 1e-12) & (interp < 1.0 - 1e-12))[0]
        if free.size > 0:
            adj = deficit / free.size
            interp[free] += adj
            np.clip(interp, 0.0, 1.0, out=interp)
    return interp
