import time
import numpy as np
import random

def _project_to_constraints(h, target_sum):
    """
    Project array h onto the box [0,1] and the affine equality sum(h)=target_sum
    using a water-filling style algorithm.
    """
    n = h.size
    # Initial clipping
    h = np.clip(h, 0.0, 1.0)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) <= 1e-12:
        return h

    # Determine direction of adjustment
    if cur_sum < target_sum:
        # Need to increase some entries that are not at 1
        free = np.ones(n, dtype=bool)
        while True:
            free = (h < 1.0 - 1e-12)
            free_cnt = free.sum()
            if free_cnt == 0:
                break
            need = target_sum - h.sum()
            add = need / free_cnt
            h[free] += add
            # Clip newly exceeded entries
            h = np.clip(h, 0.0, 1.0)
            if abs(h.sum() - target_sum) < 1e-12:
                break
    else:
        # Need to decrease some entries that are not at 0
        while True:
            free = (h > 1e-12)
            free_cnt = free.sum()
            if free_cnt == 0:
                break
            need = h.sum() - target_sum
            sub = need / free_cnt
            h[free] -= sub
            h = np.clip(h, 0.0, 1.0)
            if abs(h.sum() - target_sum) < 1e-12:
                break
    # Final tiny correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly on free variables (should be tiny)
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h):
    """Compute C5 for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(conv.max())

def _mutate(h, sigma):

    n = h.size

    h_new = h + pert
    target = n / 2.0  # because integral 1 -> sum = n/2
    return _project_to_constraints(h_new, target)

    h_new = np.interp(x_new, x_old, h)
    # enforce sum constraint after interpolation
    return _project_to_constraints(h_new, new_n / 2.0)
