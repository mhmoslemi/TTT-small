import numpy as np
import time
from math import floor, ceil

# Helper: evaluate the C5 value for a step function h on [0,2]
def _evaluate_c5(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

# Helper: project a vector onto the box [0,1] while enforcing a fixed sum.
def _project_sum_bounds(h: np.ndarray, target_sum: float) -> np.ndarray:
    h = np.clip(h, 0.0, 1.0)
    # Iteratively correct the sum using free variables (those not at a bound)
    for _ in range(15):
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0:          # everything stuck at a bound – spread uniformly
            h += diff / h.size
        else:
            inc = diff / free.size
            h[free] += inc
        h = np.clip(h, 0.0, 1.0)

    # One final scaling step if still off (very unlikely)
    cur_sum = h.sum()
    if cur_sum == 0:
        h[:] = target_sum / h.size
    else:
        factor = target_sum / cur_sum
        h *= factor
    return np.clip(h, 0.0, 1.0)

# Main entry point required by the harness
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # 1. Initialise the problem size and a feasible starting vector
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        # fallback: 128 points, uniform average 0.5 (integral = 1)
        n = 128
        h0 = np.full(n, 0.5, dtype=float)

    # Ensure the integral constraint is satisfied exactly
    target_sum = n / 2.0            # because dx = 2/n, so ∑h * dx = 1 ⇔ ∑h = n/2
    h0 = _project_sum_bounds(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _evaluate_c5(best_h)

    # ----------------------------------------------------------------------
    # 2. Simple stochastic hill‑climbing under the time budget
    # ----------------------------------------------------------------------
    # Parameters controlling the search
    max_iter = 10_000_000           # hard cap – loop stops earlier by time
    mutation_rate = 0.04            # proportion of entries mutated each trial
    sigma_initial = 0.25            # std. dev. of Gaussian perturbation
    sigma_final = 0.005
    anneal_start = 0.2 * budget_s   # start cooling after 20% of budget elapsed
    anneal_end = 0.9 * budget_s     # finish cooling by 90% of budget

    for it in range(max_iter):
        # ------------------------------------------------------------------
        # Time check
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        # ------------------------------------------------------------------
        # Adaptive mutation strength (linear annealing)
        # ------------------------------------------------------------------
        if elapsed < anneal_start:
            sigma = sigma_initial
        elif elapsed > anneal_end:
            sigma = sigma_final
        else:
            frac = (elapsed - anneal_start) / (anneal_end - anneal_start)
            sigma = sigma_initial * (1 - frac) + sigma_final * frac

        # ------------------------------------------------------------------
        # Generate a candidate by mutating the current best solution
        # ------------------------------------------------------------------
        cand = best_h.copy()
        # pick a subset of indices to mutate
        n_mut = max(1, int(floor(mutation_rate * n)))
        idx = rng.choice(n, size=n_mut, replace=False)
        perturb = rng.normal(loc=0.0, scale=sigma, size=n_mut)
        cand[idx] += perturb

        # ------------------------------------------------------------------
        # Repair to satisfy constraints
        # ------------------------------------------------------------------
        cand = _project_sum_bounds(cand, target_sum)

        # ------------------------------------------------------------------
        # Evaluate and accept if better
        # ------------------------------------------------------------------
        cand_c5 = _evaluate_c5(cand)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand

    # ----------------------------------------------------------------------
    # 3. Return the best solution discovered
    # ----------------------------------------------------------------------
    return best_h, best_c5, int(n)
