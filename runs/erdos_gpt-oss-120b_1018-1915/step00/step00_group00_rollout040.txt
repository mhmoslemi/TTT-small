import time, math, random
import numpy as np

def compute_c5(h):
    """Return the C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h * (1-h) (full linear convolution)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_swap_move(h, i, j, eps):
    """Transfer eps from index i to j preserving bounds."""
    h[i] -= eps
    h[j] += eps

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] that lowers the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        The best function values found (length n_points).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ----- initialise starting vector -----
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.shape[0]
        # sanity: enforce sum constraint if violated (rare)
        target_sum = n / 2.0
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            # simple projection: add/subtract uniform delta while respecting bounds
            delta = target_sum - cur_sum
            while abs(delta) > 1e-12:
                # move mass from high to low entries
                if delta > 0:
                    idx_low = np.where(h < 1.0)[0]
                    if len(idx_low) == 0:
                        break
                    step = min(delta / len(idx_low), 1.0 - h[idx_low])
                    h[idx_low] += step
                else:
                    idx_high = np.where(h > 0.0)[0]
                    if len(idx_high) == 0:
                        break
                    step = min(-delta / len(idx_high), h[idx_high])
                    h[idx_high] -= step
                cur_sum = h.sum()
                delta = target_sum - cur_sum
    else:
        n = 45
        target = n / 2.0
        ones = int(math.floor(target))
        remainder = target - ones
        h = np.zeros(n, dtype=float)
        h[:ones] = 1.0
        if remainder > 1e-12:
            h[ones] = remainder
        np.random.shuffle(h)

    # ----- optimisation parameters -----
    start_time = time.time()
    time_limit = budget_s * 0.95   # keep a safety margin
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    current_h = h.copy()
    current_c5 = best_c5

    # Simple simulated‑annealing loop with mass‑swap moves
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit:
            break

        # linear temperature schedule (higher early, close to zero later)
        T = max(1e-4, 1.0 - elapsed / time_limit)

        # propose a swap move
        i, j = np.random.randint(n), np.random.randint(n)
        while i == j:
            j = np.random.randint(n)

        # maximum transferable mass respecting bounds
        max_transfer = min(current_h[i], 1.0 - current_h[j])
        if max_transfer <= 1e-12:
            continue

        # random fraction of the admissible transfer (up to half of max)
        eps = max_transfer * np.random.rand() * 0.5
        new_h = current_h.copy()
        project_swap_move(new_h, i, j, eps)

        new_c5 = compute_c5(new_h)
        delta = new_c5 - current_c5

        if delta < 0 or random.random() < math.exp(-delta / (T + 1e-12)):
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                # early exit if we beat the target record
                if best_c5 <= 0.3808:
                    break

        iteration += 1

    # Return the best found solution
    return best_h, float(best_c5), n
