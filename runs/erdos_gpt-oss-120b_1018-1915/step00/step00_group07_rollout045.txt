import numpy as np
import time
import random

# Helper to enforce constraints: values in [0,1] and sum = target_sum
def repair_h(vec, target_sum):
    h = np.clip(vec, 0.0, 1.0)
    s = h.sum()
    # Simple proportional scaling, iterated a couple of times to respect bounds
    for _ in range(3):
        if abs(s - target_sum) < 1e-9:
            break
        scale = target_sum / s if s != 0 else 0.0
        h = np.clip(h * scale, 0.0, 1.0)
        s = h.sum()
    # If still off due to clipping, distribute remaining mass uniformly over non‑fixed entries
    if abs(s - target_sum) > 1e-6:
        deficit = target_sum - s
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += deficit / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h, dx):
    # correlation length 2*len-1, max corresponds to any shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def de_optimize(initial, n_points, budget_s, seed=42):
    rng = np.random.default_rng(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # DE parameters
    pop_size = max(20, n_points)          # at least 20 individuals
    F = 0.5
    CR = 0.9
    max_gen = 1000                         # will be limited by time

    # Initialize population
    pop = []
    scores = []
    # include the given initial if feasible
    if initial is not None and len(initial) == n_points:
        h0 = repair_h(initial, target_sum)
        pop.append(h0)
        scores.append(compute_c5(h0, dx))
    while len(pop) < pop_size:
        cand = rng.random(n_points)  # uniform [0,1)
        cand = repair_h(cand, target_sum)
        pop.append(cand)
        scores.append(compute_c5(cand, dx))

    pop = np.array(pop)
    scores = np.array(scores)

    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    start_time = time.time()
    gen = 0
    while gen < max_gen and (time.time() - start_time) < budget_s:
        gen += 1
        for i in range(pop_size):
            # mutation: select three distinct others
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # crossover (binomial)
            cross_mask = rng.random(n_points) < CR
            # ensure at least one dimension from mutant
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, n_points)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = repair_h(trial, target_sum)

            trial_score = compute_c5(trial, dx)
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = float(trial_score)
                    best_h = trial.copy()

        # optional early stop if very good
        if best_score <= 0.3808:
            break

    return best_h, best_score, n_points

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)

    h_values : numpy array of shape (n_points,) with entries in [0,1]
    c5_bound : float, the achieved max overlap integral
    n_points : int, number of discretization points used
    """
    # n_points is fixed by the provided initial vector if present,
    # otherwise we choose a modest size (80) as in the statement.
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_points = len(initial_h_values)
        init = initial_h_values
    else:
        n_points = 80
        init = None

    best_h, best_score, n = de_optimize(init, n_points, budget_s, seed=seed)
    return best_h, best_score, n
