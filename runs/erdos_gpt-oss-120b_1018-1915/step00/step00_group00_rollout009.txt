import time, math
import numpy as np

def _c5_of(h, dx):
    """Compute C5 = max correlation * dx."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _mutate_pair(h, rng):
    """Return a copy of h with a feasible perturbation on a random pair."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    a, b = h[i], h[j]
    # feasible delta range keeping both in [0,1] and preserving sum
    max_pos = min(1.0 - a, b)          # increase a, decrease b
    max_neg = min(a, 1.0 - b)          # decrease a, increase b
    if max_pos + max_neg == 0.0:
        return None, None, None
    delta = rng.uniform(-max_neg, max_pos)
    h_new = h.copy()
    h_new[i] = a + delta
    h_new[j] = b - delta
    return h_new, i, j

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # obtain initial vector if provided, else uniform 0.5
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 60
        h = np.full(n, 0.5, dtype=float)

    n = h.shape[0]
    dx = 2.0 / n

    # sanity: enforce exact integral constraint (sum = n/2)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h *= target_sum / cur_sum

    cur_c5 = _c5_of(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5

    # SA parameters
    T = 1.0
    cooling = 0.9995
    iter_count = 0

    while time.time() - start < budget_s * 0.98:
        mut = _mutate_pair(h, rng)
        if mut[0] is None:
            continue
        h_new = mut[0]
        new_c5 = _c5_of(h_new, dx)

        # accept if better or by Metropolis criterion
        if new_c5 < cur_c5 or rng.random() < math.exp(-(new_c5 - cur_c5) / T):
            h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()

        # temperature update
        T *= cooling
        iter_count += 1

        # occasional restart if stuck
        if iter_count % 5000 == 0 and T < 1e-6:
            T = 1.0  # reset temperature
            # optionally perturb current solution slightly
            for _ in range(10):
                tmp = _mutate_pair(h, rng)
                if tmp[0] is not None:
                    h = tmp[0]
                    cur_c5 = _c5_of(h, dx)
                    break

    return best_h, best_c5, n
