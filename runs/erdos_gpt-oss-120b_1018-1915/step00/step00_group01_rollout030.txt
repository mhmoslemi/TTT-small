import time, math, numpy as np

def compute_c5(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    # full linear correlation of discrete arrays
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_balanced_binary(n, rng):
    """Return a binary vector of length n with exactly n/2 ones."""
    half = n // 2
    arr = np.zeros(n, dtype=np.float64)
    arr[:half] = 1.0
    rng.shuffle(arr)
    return arr

def hill_climb(initial_h, dx, budget, rng):
    """Simple improvement loop: swap a 1 with a 0 to reduce C5."""
    h = initial_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)

    n = len(h)
    # pre‑compute indices of ones and zeros for quick sampling
    ones = np.flatnonzero(h == 1.0)
    zeros = np.flatnonzero(h == 0.0)

    start = time.time()
    it = 0
    while time.time() - start < budget:
        # propose a swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        h[i], h[j] = 0.0, 1.0
        # update index lists
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]

        val = compute_c5(h, dx)
        if val + 1e-12 < best_val:
            best_val = val
            best_h = h.copy()
            ones = ones_idx
            zeros = zeros_idx
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
        it += 1
        # occasional random restart if stuck
        if it % 2000 == 0:
            if rng.random() < 0.1:
                h = random_balanced_binary(n, rng)
                ones = np.flatnonzero(h == 1.0)
                zeros = np.flatnonzero(h == 0.0)
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Choose a moderate resolution (even number for exact half‑sum)
    n_points = 400
    dx = 2.0 / n_points

    # Start from the provided parent if it exists and respects the constraints
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64)
        if len(parent) == n_points and abs(parent.sum() * dx - 1.0) < 1e-12:
            best_h = parent.copy()
            best_val = compute_c5(best_h, dx)
        else:
            # Resample parent onto our grid (simple linear interpolation)
            x_src = np.linspace(0, 2, len(parent), endpoint=False)
            x_tgt = np.linspace(0, 2, n_points, endpoint=False)
            best_h = np.interp(x_tgt, x_src, parent)
            # Project onto feasible set: clip and adjust sum to n/2
            best_h = np.clip(best_h, 0.0, 1.0)
            target_sum = n_points / 2
            cur_sum = best_h.sum()
            if cur_sum != target_sum:
                # scale then shift to meet exact sum while staying in [0,1]
                # simple proportional scaling with clipping
                scale = target_sum / cur_sum
                best_h = np.clip(best_h * scale, 0.0, 1.0)
                # adjust remaining discrepancy by adding/subtracting tiny epsilon
                diff = target_sum - best_h.sum()
                if abs(diff) > 1e-9:
                    # distribute diff uniformly
                    best_h += diff / n_points
                    best_h = np.clip(best_h, 0.0, 1.0)
            best_val = compute_c5(best_h, dx)
    except NameError:
        # No parent supplied; start from a random binary sequence
        best_h = random_balanced_binary(n_points, rng)
        best_val = compute_c5(best_h, dx)

    # Main optimization budget: leave a small margin for final steps
    remaining = max(0.0, budget_s - (time.time() - kwargs.get('start_time', time.time())) - 5.0)
    # Run several independent hill‑climb attempts
    attempts = 5
    per_attempt = remaining / attempts if attempts else 0
    for _ in range(attempts):
        init = random_balanced_binary(n_points, rng)
        h_opt, val_opt = hill_climb(init, dx, per_attempt, rng)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()

    # Return as Python list for safety
    return (best_h.tolist(), float(best_val), n_points)
