import time
import numpy as np

def _project_bounds_sum(x, target, lo=0.0, hi=1.0, max_iter=15):
    """Project vector onto [lo,hi] with exact sum = target."""
    x = np.clip(x, lo, hi)
    total = x.sum()
    for _ in range(max_iter):
        if abs(total - target) <= 1e-12:
            break
        diff = target - total
        free = (x > lo + 1e-12) & (x < hi - 1e-12)
        if not np.any(free):
            free = np.ones_like(x, dtype=bool)
        adjust = diff / free.sum()
        x[free] += adjust
        x = np.clip(x, lo, hi)
        total = x.sum()
    return x

def _evaluate(h):
    """Return C5 for step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _upscale(h, new_n):
    """Linearly interpolate h to length new_n (periodic on [0,2))."""
    n = h.shape[0]
    old_x = np.linspace(0.0, 2.0, n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    # periodic interpolation via concatenation of one extra period
    extended_h = np.concatenate([h, h[:1]])
    extended_x = np.concatenate([old_x, [2.0]])
    return np.interp(new_x, extended_x, extended_h)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise starting vector
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    except NameError:
        n = 128
        h0 = np.full(n, 0.5, dtype=float)   # sum = n/2 automatically
    target_sum = n / 2.0
    h_curr = _project_bounds_sum(h0, target_sum)

    best_h = h_curr.copy()
    best_c5 = _evaluate(best_h)

    # ------------------------------------------------------------------
    # optimisation parameters
    sigma = 0.2                     # mutation scale
    it = 0
    max_iters = 10**7
    last_upscale = 0

    while time.time() - start < budget_s * 0.90 and it < max_iters:
        # annealing schedule
        sigma = max(0.005, sigma * 0.9997)

        # propose a new candidate
        cand = h_curr + sigma * np.random.randn(n)
        cand = _project_bounds_sum(cand, target_sum)

        cand_c5 = _evaluate(cand)

        # accept if improvement; else keep current (simple hill‑climb)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h_curr = cand
        else:
            # occasional random jump to escape plateaus
            if np.random.rand() < 0.001:
                h_curr = cand

        # periodically upscale to a finer grid (up to 280 points)
        if n < 280 and (it - last_upscale) >= 25000:
            new_n = min(2 * n, 280)
            up_h = _upscale(best_h, new_n)
            n = new_n
            target_sum = n / 2.0
            h_curr = _project_bounds_sum(up_h, target_sum)
            best_h = h_curr.copy()
            best_c5 = _evaluate(best_h)
            last_upscale = it

        it += 1

    return best_h, best_c5, best_h.shape[0]
