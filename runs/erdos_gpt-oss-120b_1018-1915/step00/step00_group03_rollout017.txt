import time, math, random
import numpy as np

def _compute_c5(h):
    """Fallback C5 evaluator using correlation."""
    n = len(h)
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()

def _evaluate(h):
    """Use the provided evaluator if available, otherwise fallback."""
    return evaluate_erdos_solution(h) if "evaluate_erdos_solution" in globals() else _compute_c5(h)

def _project_to_constraints(h):
    """Clip to [0,1] and renormalise the total integral exactly."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # Simple redistribution of excess/deficit while keeping bounds.
    deficit = target - h.sum()
    if abs(deficit) < 1e-12:
        return h
    # indices that are not at a bound can receive/take mass
    free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
    if free.size == 0:
        # all at bounds – just spread uniformly (rare)
        h[:] = target / n
        return h
    # distribute linearly
    delta = deficit / free.size
    h[free] += delta
    # clip again and recurse if needed (very limited recursion)
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-9:
        # final correction on a single free coordinate
        diff = target - h.sum()
        idx = free[0]
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _random_transfer(h, max_step=0.1):
    """Propose a new h by moving a small amount of mass between two indices."""
    n = len(h)
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None
    # direction: move from j to i or reverse
    if np.random.rand() < 0.5:
        src, dst = j, i
    else:
        src, dst = i, j
    # feasible amount we can move
    move_max = min(h[src], 1.0 - h[dst])
    if move_max <= 0:
        return None
    delta = np.random.rand() * move_max * max_step
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new

    """Search for a step function on [0,2] with integral 1 that minimises C5.

    Returns:

        c5_bound (float): the corresponding C5 value.
        n_points (int): number of samples used.
    """

    h = _project_to_constraints(h)

    best_h = h.copy()
    best_c5 = float(_evaluate(best_h))

    start_time = time.time()
    time_limit = budget_s * 0.95   # safety margin

    # Main simulated‑annealing loop

    T0 = 0.02                      # initial temperature

    while time.time() - start_time < time_limit:
