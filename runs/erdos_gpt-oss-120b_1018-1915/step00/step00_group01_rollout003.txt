def _compute_c5(h):
    """Return C5 for a given step vector h (values in [0,1])."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' gives length 2n-1
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())
