import numpy as np
import time

# Helper: project a vector onto the capped simplex {0<=x<=1, sum=x_target}
def project_capped_simplex(v, target_sum, tol=1e-9, max_iter=100):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target_sum) <= tol:
            return x
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

# Compute C5 for a step vector h (array)
def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

# Build initial feasible h of length n
def get_initial_h(n):
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        # simple alternating 1/0 pattern with correct average
        h = np.zeros(n, dtype=float)
        half = n // 2
        h[:half] = 1.0
        h[half:] = 0.0
    # project to satisfy sum constraint
    target = n / 2.0
    h = project_capped_simplex(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s - 5.0  # safety margin

    # Choose resolution (trade-off between quality and speed)
    n = 300                       # can be adjusted; keeps memory low
    dx = 2.0 / n
    target_sum = n / 2.0

    # Initial solution
    h_current = get_initial_h(n)
    best_h = h_current.copy()
    best_c5 = compute_c5(best_h, dx)

    # Simulated annealing parameters
    T0 = 0.1
    Tf = 1e-4
    max_iters = 200000

    for it in range(max_iters):
        # time check
        if time.time() - start_time > time_limit:
            break

        # exponential cooling
        t = it / max_iters
        T = T0 * (Tf / T0) ** t

        # propose move: transfer delta between two random indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # max amount we can move without breaking bounds
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        # scale by temperature
        delta = rng.uniform(-max_transfer, max_transfer) * T * 10
        h_prop = h_current.copy()
        h_prop[i] -= delta
        h_prop[j] += delta

        # project to satisfy sum and box constraints
        h_prop = project_capped_simplex(h_prop, target_sum)

        # evaluate
        c5_prop = compute_c5(h_prop, dx)

        # accept if better or by Metropolis criterion
        if c5_prop < best_c5:
            best_h = h_prop.copy()
            best_c5 = c5_prop
        if c5_prop < compute_c5(h_current, dx) or rng.random() < np.exp(-(c5_prop - best_c5) / (T + 1e-12)):
            h_current = h_prop

    return best_h, float(best_c5), int(n)
