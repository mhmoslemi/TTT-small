import time
import numpy as np
from scipy.optimize import minimize

def _c5_cost(h, dx):
    """Return the overlap max for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def _project_to_constraints(h, target_sum):
    """
    Clip to [0,1] and then affinely rescale to hit the exact integral sum.
    """
    h = np.clip(h, 0.0, 1.0)
    cur_sum = np.sum(h)
    if cur_sum == 0:
        # distribute uniformly
        return np.full_like(h, target_sum / h.size)
    # affine shift to meet target sum while staying in [0,1] as much as possible
    shift = (target_sum - cur_sum) / h.size
    h = h + shift
    # after shift some entries may leave the box; clip and repeat once
    h = np.clip(h, 0.0, 1.0)
    # final correction to enforce exact sum (small residual)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        # distribute residual uniformly over non‑saturated entries
        free = (h > 0.0) & (h < 1.0)
        if np.any(free):
            add = residual / free.sum()
            h[free] += add
            h = np.clip(h, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):

    Optimize a step function h on [0,2] to minimise the Erdős overlap constant C5.

        The achieved upper bound C5.
    n_points : int
        Number of discretisation points used.

    start = time.time()

    dx = 2.0 / n
    target_sum = n / 2.0           # because sum(h)*dx = 1

    # enforce constraints on the starter
    h = _project_to_constraints(h0, target_sum)
    best_h = h.copy()
    best_c5 = _c5_cost(best_h, dx)

    # ------------------------------------------------------------------ #
    # 2. Simple stochastic local search (pairwise mass transfers).
    # ------------------------------------------------------------------ #
    max_iter = 10_000_000          # safety cap; actual stop is time based
    attempt = 0

    while time.time() - start < budget_s * 0.92 and attempt < max_iter:
        attempt += 1

        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)

        # decide a transfer amount delta (positive moves mass from i to j)
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        # limit step size to keep search stable
        delta = rng.uniform(0.0, min(max_transfer, 0.02))

        # propose new vector
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # no need to re‑project: construction guarantees bounds and sum
        c5_new = _c5_cost(h_new, dx)

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
