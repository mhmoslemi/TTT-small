import time
import numpy as np

def _evaluate(h):
    """Return C5 for a given step function samples."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_delta_mutation(h, rng):
    """Transfer a random amount delta from one entry to another while staying in [0,1]."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # maximum delta that can be added to i (taken from j)
    max_pos = min(1.0 - h[i], h[j])
    # maximum delta that can be subtracted from i (given to j)
    max_neg = min(h[i], 1.0 - h[j])
    if max_pos + max_neg == 0.0:
        return h.copy()  # nothing can move
    delta = rng.uniform(-max_neg, max_pos)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def _swap_mutation(h, rng):
    """Swap two entries."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new
