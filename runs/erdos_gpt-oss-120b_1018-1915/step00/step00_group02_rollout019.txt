import time
import numpy as np
from scipy import optimize

def _evaluate(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * (2.0 / h.size)


def _project_to_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Adjust a vector x in [0,1] so that its sum equals target.
    Simple iterative redistribution that respects the bounds.
    """
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
        add = diff / x.size
        x = np.clip(x + add, 0.0, 1.0)
    return x


def _optimize_at_resolution(h0: np.ndarray, time_limit: float, start: float) -> tuple:
    """
    Perform a few SLSQP runs (one deterministic, two random restarts) starting
    from h0, respecting the time budget. Returns the best vector found and its
    C5 value.
    """
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": lambda v: np.sum(v) - n / 2.0}

    best_h = h0.copy()
    best_val = _evaluate(best_h)

    # three restarts (first is the provided start)
    for restart in range(3):
        if time.time() - start > time_limit:
            break

        if restart == 0:
            x0 = best_h.copy()
        else:
            rnd = np.random.rand(n)
            rnd = _project_to_sum(rnd, n / 2.0)
            x0 = rnd

        res = optimize.minimize(
            _evaluate,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=eq_cons,
            options={"maxiter": 300, "ftol": 1e-9, "disp": False},
        )
        if not res.success:
            continue

        val = _evaluate(res.x)
        if val < best_val:
            best_val = val
            best_h = res.x.copy()

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where:
      - h_values : ndarray of the step function values on a uniform grid of [0,2]
      - c5_bound : the achieved max overlap integral (the C5 upper bound)
      - n_points : number of discretisation points used
    """
    np.random.seed(seed)
    start_time = time.time()
    time_limit = float(budget_s) - 0.5   # small safety margin

    # ----------------------------------------------------------------------
    # 1. Obtain an initial vector (use provided one if present)
    # ----------------------------------------------------------------------
    try:
        h_start = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        # fallback: uniform 0.5 on 45 points satisfies the integral constraint
        h_start = np.full(45, 0.5, dtype=np.float64)

    # Ensure the integral constraint exactly (possible tiny numerical drift)
    target_sum = h_start.size / 2.0
    h_start = _project_to_sum(h_start, target_sum)

    best_h = h_start.copy()
    best_val = _evaluate(best_h)

    # ----------------------------------------------------------------------
    # 2. Hierarchical optimisation with resolution doubling
    # ----------------------------------------------------------------------
    max_points = 2000  # keep evaluation cheap
    while True:
        # optimise at current resolution
        h_opt, val_opt = _optimize_at_resolution(best_h, time_limit, start_time)

    return best_h, best_val, best_h.size
