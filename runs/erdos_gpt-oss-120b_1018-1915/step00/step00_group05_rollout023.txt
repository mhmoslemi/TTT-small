import numpy as np
import time

def _repair_sum(vec, desired_sum, eps=1e-12):
    """Adjust vec in-place so that sum(vec) == desired_sum while staying in [0,1]."""
    cur = vec.sum()
    diff = desired_sum - cur
    if abs(diff) <= eps:
        return vec
    # If we need to add mass
    if diff > 0:
        # indices that can receive more
        idx = np.where(vec < 1 - 1e-12)[0]
        while diff > eps and idx.size:
            i = np.random.choice(idx)
            inc = min(diff, 1.0 - vec[i])
            vec[i] += inc
            diff -= inc
            if vec[i] >= 1 - 1e-12:
                idx = np.delete(idx, np.where(idx == i))
    else:  # diff < 0, need to remove mass
        idx = np.where(vec > 1e-12)[0]
        while diff < -eps and idx.size:
            i = np.random.choice(idx)
            dec = min(-diff, vec[i])
            vec[i] -= dec
            diff += dec
            if vec[i] <= 1e-12:
                idx = np.delete(idx, np.where(idx == i))
    # If still off due to rounding, distribute tiny residual uniformly
    if abs(vec.sum() - desired_sum) > 1e-8:
        vec += (desired_sum - vec.sum()) / vec.size
        np.clip(vec, 0.0, 1.0, out=vec)
    return vec

def _fitness(h, dx):
    """Return C5 value for a given step function h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def _random_feasible(n, desired_sum):
    """Generate a random vector in [0,1]^n with given sum."""
    # Start with uniform random numbers
    v = np.random.rand(n)
    # Scale to match sum, then repair clipping
    v *= desired_sum / v.sum()
    v = np.clip(v, 0.0, 1.0)
    # Repair any residual discrepancy
    return _repair_sum(v, desired_sum)

    Optimize the Erdős C5 bound.

    np.random.seed(seed)
    start_time = time.time()

    dx = 2.0 / n_points
    desired_sum = n_points / 2.0  # because integral = 1

    # Initialise best solution
    if h0 is not None:
        best = np.clip(h0, 0.0, 1.0)
        best = _repair_sum(best, desired_sum)
    else:
        best = _random_feasible(n_points, desired_sum)

    best_score = _fitness(best, dx)

    # Parameters for the stochastic search
    max_iter = 200000  # safety cap
    iter_cnt = 0
    noise_scale = 0.2  # initial magnitude of perturbations
    decay_rate = 0.9995  # slowly reduce step size

    while True:

        if elapsed >= budget_s or iter_cnt >= max_iter:
