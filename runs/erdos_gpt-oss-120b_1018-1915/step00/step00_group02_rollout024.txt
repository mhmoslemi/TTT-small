import numpy as np
import time
import random

def _c5_value(h):
    """Return the C5 value for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project(h, target_sum, max_iter=20):
    """Project h onto [0,1]^n and the hyperplane sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            # all at bounds – distribute uniformly (may violate bounds a bit)
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction to hit target exactly (may push a tiny bit out of bounds)
    s = h.sum()
    if abs(s - target_sum) > 1e-12:
        h += (target_sum - s) / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _random_feasible(n, rng):
    """Generate a random feasible vector of length n."""
    vals = rng.uniform(0.0, 1.0, size=n)
    target = n / 2.0
    vals = _project(vals, target)
    return vals

def _upscale(h):
    """Double the resolution by linear interpolation on the torus."""
    n = h.size
    new_n = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    # periodic interpolation (period 2)
    new_h = np.interp(x_new, x_old, h, period=2.0)
    target = new_n / 2.0
    new_h = _project(new_h, target)
    return new_h

    """
    Optimize a step function for the Erdős C5 problem.

    Returns

    h_values : np.ndarray

    max_levels = 3          # 45 → 90 → 180 → 360
