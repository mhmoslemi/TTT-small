import time
import numpy as np

def _project_to_bounds_and_sum(vec, target, max_iter=5):
    """Project vector onto [0,1] with prescribed sum = target."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        if abs(s - target) < 1e-12:
            break
        # scale the free part
        scale = target / s if s != 0 else 0.0
        v *= scale
        v = np.clip(v, 0.0, 1.0)
    # final correction for tiny residuals
    resid = target - v.sum()
    if abs(resid) > 1e-12:
        free = (v > 0.0) & (v < 1.0)
        if free.any():
            v[free] += resid / free.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_score(h, dx):
    """Return max overlap C5 for a discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

        population[i] = _project_to_bounds_and_sum(rand_vec, target_sum)

    scores = np.array([_c5_score(ind, dx) for ind in population])
