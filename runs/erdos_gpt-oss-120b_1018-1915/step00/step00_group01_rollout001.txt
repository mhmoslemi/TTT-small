import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Return the overlap bound C5 for a discretized step function h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _project_to_constraints(x, target_sum):
    """Clip to [0,1] and then adjust to hit the exact sum using a simple redistribution."""
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    diff = target_sum - s
    # distribute the difference uniformly; repeat a few times to respect bounds
    for _ in range(5):
        if abs(diff) < 1e-12:
            break
        inc = diff / x.size
        x = x + inc
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
        diff = target_sum - s
    return x
