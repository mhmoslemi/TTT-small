def get_feasible_neighbor(h: List[float], n: int) -> List[float]:
    """Generate neighbor that always forie and preserves the sum.
    
    Args:
        h: current vector
        n: dimension
        
    Returns:
        list: new vector ensuring 0<=values<=1 and sum=n/2.
    """
    # Simple implementation using bound checks and repair
    h_copy = np.array(h, copy=True)
    # Randomly choose two indices different from any fixed point
    i, j = np.random.choice(n, 2, replace=False)
    # Swap values and clamp if needed
    h_copy[i], h_copy[j] = h_copy[j], h_copy[i]
    # Adjust for any sum deviation caused by clamping
    current_sum = sum(h_copy)
    desired_sum = n/2.0
    if abs(current_sum - desired_sum) > 1e-5:
        # Use direct value compensation
        idx = np.random.randint(0, n)
        diff = desired_sum - current_sum
        h_copy[idx] += diff
        # Reclamp the modified value
        h_copy = np.clip(h_copy, 0, 1).tolist()
    return h_copy.tolist()

def adaptive_cooling_duration(iteration_count: int) -> int:
    """Determine cooling steps based on iteration count
    - First 1000 cycles: full steps
    - As warming stabilizes, reduce steps per cycle to save time
    
    Args:
        iteration_count: current total iterations
        
    Returns:
        int: number of additional iterations to allocate to cooling
    """
    # Heuristic: faster cooling as warming progresses
    if iteration_count < 1000:
        return 1000
    elif iteration_count < 5000:
        return 500
    elif iteration_count < 10000:
        return 200
    else:
        if iteration_count % 1000 == 0:
            # Occasionally slow down further
            return 100
        return 10

def calculate_optimal_speed(target_speed=1.0, temperature_hint=None):
    """Temporarily set optimization speed using cooling rate info

    - Slow cooling: higher temperature -> slower optimization

        target_speed: (0.0-1.0) speed relative factor
        temperature_hint: current temperature to inform speed

def progressive_mutation(h, n, mutation_rate=0.3):
    """Generate either a swap or global perturbation based on condition
    
    Args:
        h: current solution
        n: number of points
        mutation_rate: probability of more complex mutation
        
    Returns:
        list: mutated solution
    """
    if np.random.random() < mutation_rate and n > 20:
        # Global perturbation: small flips in several points
        indices = np.random.choice(range(n), size=3, replace=False)
        h_copy = np.array(h)
        h_copy[indices] += np.random.normal(size=3) * 0.2
        h_copy = np.clip(h_copy, 0, 1)
        return h_copy.tolist()
    else:
        # Standard bound mutation
        return get_feasible_neighbor(h, n)

def upscale_representation(h, new_n, dx):
    """Increase resolution from n_old to new_n (should be twice)
    Returns a new representation by piecewise linear interpolation.
    
    Args:
        h: original step function
        new_n: target higher resolution
        dx: original grid spacing
        
    Returns:
        list: h_new with improved resolution
    """
    original_length = sum(h)
    h = np.array(h)
    h_interpolated = []
    interpolation_steps = new_n // original_length
    
    for i in range(len(h) - 1):
        # Interpolate between consecutive points
        for k in range(interpolation_steps):
            interp_value = h[i] + (h[i+1] - h[i]) * (k / interpolation_steps)
            h_interpolated.append(interp_value)
    # Add the last point if needed
    if len(h_interpolated) < new_n:
        h_interpolated += [h[-1]] * (new_n - len(h_interpolated))
    return np.clip(h_interpolated, 0, 1).tolist()
