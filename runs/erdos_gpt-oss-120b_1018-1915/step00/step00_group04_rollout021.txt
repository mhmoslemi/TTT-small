import time
import numpy as np

def _compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _binarize(arr):
    """Convert an arbitrary array to a binary vector with exactly half ones."""
    n = arr.shape[0]
    ones_needed = n // 2
    idx = np.argsort(-arr)  # descending
    h = np.zeros(n, dtype=int)
    h[idx[:ones_needed]] = 1
    return h

def _anneal(h, time_limit, start_time):
    """Simple simulated‑annealing on a binary vector preserving the half‑ones constraint."""
    n = h.shape[0]
    cur = h.copy()
    cur_score = _compute_c5(cur.astype(float))
    best = cur.copy()
    best_score = cur_score

    max_iter = 200000
    T0, Tmin = 0.2, 1e-4

    # pre‑compute index lists to avoid repeated scans
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        # linear temperature decay
        T = max(Tmin, T0 * (1.0 - it / max_iter))

        # choose a random 1‑position and a random 0‑position to swap
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)

        # propose swap
        cur[i1], cur[i0] = 0, 1
        new_score = _compute_c5(cur.astype(float))
        delta = new_score - cur_score

        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            # accept
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = cur.copy()
        else:
            # reject – revert swap
            cur[i1], cur[i0] = 1, 0

    return best.astype(float), best_score
