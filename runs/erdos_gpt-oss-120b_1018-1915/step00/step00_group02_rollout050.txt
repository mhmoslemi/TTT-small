import time
import numpy as np
from scipy.optimize import minimize

# Helper: compute the Erdős C5 value for a discrete step function h
def _c5_value(h):
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# Project a random vector onto the feasible set (box + integral)
def _random_feasible(n, rng):
    # start from uniform [0,1]
    x = rng.random(n)
    # enforce the integral with a simple scaling, then clip, iterate a few times
    for _ in range(5):
        s = x.sum()
        target = n / 2.0
        x = x * (target / s)            # scale to hit the sum
        np.clip(x, 0.0, 1.0, out=x)     # enforce box
    # final correction for any tiny residual error
    deficit = target - x.sum()
    if abs(deficit) > 1e-12:
        # distribute uniformly while staying in bounds
        free = (x > 0) & (x < 1)
        if free.any():
            adjustment = deficit / free.sum()
            x[free] += adjustment
            np.clip(x, 0.0, 1.0, out=x)
    return x

# Perform a constrained SLSQP optimisation for a given start vector
def _optimize_start(x0, maxiter, ftol, time_limit, start_time):
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq',
             'fun': lambda x: x.sum() - n / 2.0},)

    def obj(x):
        # early abort if time exceeded
        if time.time() - start_time > time_limit:
            raise TimeoutError
        return _c5_value(x)

    try:
        res = minimize(obj,
                       x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        if not res.success:
            return x0, _c5_value(x0)
        return res.x, _c5_value(res.x)
    except TimeoutError:
        # Return the best known (current x0) if time ran out
        return x0, _c5_value(x0)
