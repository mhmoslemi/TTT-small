import time
import numpy as np

def _target_sum(n):
    """Integral constraint: sum h = n/2 because dx = 2/n and ∫h = 1."""
    return n / 2.0

def _project_to_constraints(h, target):
    """
    Clip to [0,1] and then adjust to satisfy the exact sum constraint.
    Simple redistribution algorithm that iterates until the sum matches.
    """
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(200):
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        adjust = diff / cnt
        h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    # final tiny correction by adding uniformly (won't break bounds by design)
    if abs(diff) > 1e-12:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _evaluate(h):

def _random_feasible(n, rng):
    """Generate a random vector in [0,1]^n satisfying the sum constraint."""
    h = rng.random(n)
    return _project_to_constraints(h, _target_sum(n))

def _perturb(h, rng, scale):
    """
    Produce a neighbour by moving a small amount between two coordinates.
    The sum stays unchanged; after the move we re‑project to the feasible set.

    h_new[j] -= delta
    h_new = np.clip(h_new, 0.0, 1.0)
