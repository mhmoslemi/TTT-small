import time
import numpy as np

def _compute_c5(h):
    """Return the overlap maximum C5 for a given step vector h."""
    if "evaluate_erdos_solution" in globals():
        # Use the provided exact evaluator if available.
        return float(evaluate_erdos_solution(h))
    dx = 2.0 / h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _make_feasible(h):
    """Force 0≤h≤1 and sum(h)=n/2 (exactly) while keeping h close to original."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute the deficit uniformly, then repair overflow if needed
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        # final exact fix on first element
        h[0] = np.clip(h[0] + (target - cur), 0.0, 1.0)
    return h

def _mutate(h, rng, scale):
    """Move a small amount of mass from one index to another, preserving sum."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    delta = rng.normal(0.0, scale)
    # Clamp delta so both entries stay in [0,1]
    delta = np.clip(delta, -h[i], 1.0 - h[j])
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def _upscale(h):
    """Double the resolution by nearest‑neighbour repeat and re‑normalize."""
    n = h.shape[0]
    h_big = np.repeat(h, 2)
    target = (2 * n) / 2.0
    h_big = h_big / h_big.sum() * target
    h_big = np.clip(h_big, 0.0, 1.0)
    # exact sum fix
    h_big[0] = np.clip(h_big[0] + (target - h_big.sum()), 0.0, 1.0)
    return h_big
