import time
import numpy as np

def evaluate_corr_max(h):
    """Return max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def repair_sum_and_bounds(h, target_sum):
    """Clip to [0,1] and scale to exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # distribute uniformly
        h[:] = target_sum / len(h)
    else:
        h *= target_sum / cur
        h = np.clip(h, 0.0, 1.0)
        # second pass to fix possible tiny drift
        cur = h.sum()
        if not np.isclose(cur, target_sum):
            h *= target_sum / cur
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of sample points."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # 1. Choose resolution and initialise h
    # --------------------------------------------------------------
    init = kwargs.get('initial_h_values')
    if init is not None:
        init = np.asarray(init, dtype=float)
        base_n = len(init)
    else:
        base_n = 42  # fallback size
        init = rng.uniform(0.0, 1.0, base_n)

    # target resolution (moderate, fits budget)
    target_n = max(84, base_n * 2)          # at least 84 points
    target_n = min(target_n, 336)          # keep size reasonable

    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h = np.interp(x_new, x_old, init)

    # Enforce integral constraint (sum = n/2)
    target_sum = target_n / 2.0
    h = repair_sum_and_bounds(h, target_sum)

    best_h = h.copy()
    best_score = evaluate_corr_max(best_h)

    # --------------------------------------------------------------
    # 2. Simple stochastic hill‑climbing

    iteration = 0
    max_iter = 200000   # generous bound, loop will break on time

    while True:

            break
        if iteration >= max_iter:

        iteration += 1

        # ----- elementary transfer move (preserves sum) -----
        i, j = rng.integers(0, target_n, size=2, endpoint=False)
        if i == j:
            continue
        # amount we can move from j to i without breaking bounds
        delta_max = min(1.0 - best_h[i], best_h[j])
        delta_min = -min(best_h[i], 1.0 - best_h[j])
