import time
import numpy as np
from scipy.optimize import minimize

# ---------- helper functions (top‑level) ----------
def evaluate_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function discretized by h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the exact sum = target (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(15):
        s = v.sum()
        if abs(s - target) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        if not np.any(free):
            # all at bounds – uniform scaling and re‑clip
            v *= target / s
            v = np.clip(v, 0.0, 1.0)
            break
        diff = target - s
        v[free] += diff / free.sum()
        v = np.clip(v, 0.0, 1.0)
    return v


# globals used by the optimizer (set inside run)
_target_sum = None
_penalty = 1e6


def objective(h: np.ndarray) -> float:
    """Penalized objective: C5 + huge penalty for violating the integral."""
    # h is already bounded by the optimizer
    c5 = evaluate_c5(h)
    penalty = _penalty * (h.sum() - _target_sum) ** 2
    return c5 + penalty


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where
        h_values : list of floats in [0,1] with sum = n_points/2,
        c5_bound : the achieved upper bound C5,
        n_points : number of discretisation points used.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 100  # fallback size
        h0 = np.full(n, 0.5)  # uniform start satisfies the integral

    target_sum = n / 2.0
    global _target_sum
    _target_sum = target_sum

    # enforce feasibility of the starting vector
    h0 = project_bounds_sum(h0, target_sum)

    # optimizer settings
    bounds = [(0.0, 1.0)] * n
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # keep a small budget for the first deterministic run
    max_iter = 400
    res = minimize(
        objective,
        best_h,
        method="L-BFGS-B",
        bounds=bounds,
        options={"maxiter": max_iter, "ftol": 1e-12},
    )
    if res.success:
        cand = project_bounds_sum(res.x, target_sum)
        cand_c5 = evaluate_c5(cand)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5

    # stochastic restarts while time permits
    restart_limit = 12
    attempts = 0
    while attempts < restart_limit and (time.time() - start_time) < (budget_s - 2):
        attempts += 1
        # random feasible start
        rand_vec = rng.random(n)
        rand_vec = project_bounds_sum(rand_vec, target_sum)

        res = minimize(
            objective,
            rand_vec,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 300, "ftol": 1e-11},
        )
        if not res.success:
            continue
        cand = project_bounds_sum(res.x, target_sum)
        cand_c5 = evaluate_c5(cand)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5

    # final safety projection (ensure exact sum)
    best_h = project_bounds_sum(best_h, target_sum)
    best_c5 = evaluate_c5(best_h)

    # return as plain python list for compatibility
    return best_h.tolist(), float(best_c5), int(n)
