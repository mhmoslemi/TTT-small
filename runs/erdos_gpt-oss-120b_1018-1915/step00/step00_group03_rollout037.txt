import numpy as np
import time
import math
import random

def _compute_c5(arr, dx):
    """Maximum overlap integral for a given step function."""
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return np.max(corr) * dx

def _binary_projection(vals, half):
    """Return a 0/1 vector with exactly `half` ones, choosing the largest entries."""
    idx = np.argsort(-vals)  # descending
    out = np.zeros_like(vals, dtype=float)
    out[idx[:half]] = 1.0
    return out

def _anneal_optimize(initial_h, max_seconds):
    """Simple simulated‑annealing on binary vectors with fixed number of ones."""
    n = initial_h.size
    dx = 2.0 / n
    half = n // 2

    # enforce binary feasibility
    cur = _binary_projection(initial_h, half)
    cur_c5 = _compute_c5(cur, dx)

    best = cur.copy()
    best_c5 = cur_c5

    # indices of the two groups – will be refreshed after each accepted move
    ones_idx = np.where(cur == 1)[0]
    zeros_idx = np.where(cur == 0)[0]

    start = time.time()
    it = 0
    while time.time() - start < max_seconds:
        # temperature slowly cools to a small positive value
        elapsed = time.time() - start
        progress = elapsed / max_seconds
        temp = 0.1 * (1.0 - progress) + 0.001

        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        new = cur.copy()
        new[i], new[j] = 0.0, 1.0

        new_c5 = _compute_c5(new, dx)
        delta = new_c5 - cur_c5

        accept = False
        if delta < 0:
            accept = True
        else:
            if math.exp(-delta / (temp + 1e-12)) > random.random():
                accept = True

        if accept:
            cur = new
            cur_c5 = new_c5
            ones_idx = np.where(cur == 1)[0]
            zeros_idx = np.where(cur == 0)[0]
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best = cur.copy()
                if best_c5 <= 0.3808:      # target reached → stop early
                    break
        it += 1

    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points).

    - h_values : list of float samples in [0,1] satisfying the constraints.
    - c5_bound : the achieved upper bound for C₅.
    - n_points : number of discretisation points used.
    """
    np.random.seed(seed)
    random.seed(seed)

    total_budget = budget_s * 0.95          # leave a safety margin
    start_all = time.time()

    # Determine a base resolution from the parent construction if present
    if "initial_h_values" in globals():
        base_h = np.asarray(initial_h_values, dtype=float)
        base_n = base_h.size
    else:
        base_n = 120
        base_h = np.zeros(base_n, dtype=float)
        base_h[: base_n // 2] = 1.0
        np.random.shuffle(base_h)

    # Candidate resolutions (do not exceed 500 points for speed)
    resolutions = [base_n]
    if base_n * 2 <= 500:
        resolutions.append(base_n * 2)
    if base_n * 3 <= 500:
        resolutions.append(base_n * 3)

    best_overall_h = None
    best_overall_c5 = math.inf
    best_overall_n = None

    # Allocate equal time to each resolution
    per_res_budget = total_budget / len(resolutions)

    for n in resolutions:
        # Initialise a binary vector with exactly n/2 ones
        half = n // 2
        if n == base_n and "initial_h_values" in globals():
