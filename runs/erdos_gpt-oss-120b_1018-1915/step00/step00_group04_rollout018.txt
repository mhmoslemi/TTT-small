def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np

    # ---------- helper functions ----------
    def compute_c5(h):
        """Maximum overlap integral for a given step function h."""
        n = h.size
        dx = 2.0 / n
        # correlation of h with (1-h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    def repair(h, n):
        """Project h onto the feasible set 0≤h≤1, sum(h)=n/2."""
        target = n * 0.5
        h = np.clip(h, 0.0, 1.0)
        # iterative redistribution of the excess/deficit
        for _ in range(15):
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
            # allowable movement for each coordinate
            if diff > 0:
                cap = 1.0 - h          # how much we can increase
            else:
                cap = -h                # how much we can decrease (negative)
            total_cap = cap.sum()
            if total_cap == 0:
                break
            # distribute proportionally to the available capacity
            delta = diff * (cap / total_cap)
            h += delta
            h = np.clip(h, 0.0, 1.0)
        return h

    # ---------- initialization ----------
    np.random.seed(seed)

    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 42
        h0 = np.full(n, 0.5)               # sum = n/2 satisfies the integral constraint

    h0 = repair(h0, n)
    best_h = h0.copy()
    best_score = compute_c5(best_h)

    start_time = time.time()
    max_iters = 200000                     # generous upper bound; real stop is time‑based
    iter_cnt = 0

    # ---------- main stochastic search ----------
    while time.time() - start_time < budget_s * 0.94 and iter_cnt < max_iters:
        # decreasing mutation strength
        sigma = 0.25 * (1.0 - iter_cnt / max_iters)
        cand = best_h + np.random.normal(scale=sigma, size=n)
        cand = repair(cand, n)
        cand_score = compute_c5(cand)

        if cand_score < best_score:
            best_h, best_score = cand, cand_score

        # occasional random restart to escape stagnation
        if iter_cnt % 5000 == 0 and iter_cnt > 0:
            rnd = np.random.rand(n)
            rnd = repair(rnd, n)
            rnd_score = compute_c5(rnd)
            if rnd_score < best_score:
                best_h, best_score = rnd, rnd_score

        iter_cnt += 1

    # ---------- optional resolution refinement ----------
    # If time permits, double the resolution and continue refinement.
    if time.time() - start_time < budget_s * 0.9 and n < 672:
        n2 = n * 2
        # linear interpolation of the current best vector onto the finer grid
        x_old = np.linspace(0, 2, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0, 2, n2, endpoint=False) + 1.0 / n2
        h_fine = np.interp(x_new, x_old, best_h)
        h_fine = repair(h_fine, n2)

        best_h = h_fine
        n = n2
        best_score = compute_c5(best_h)

        # continue stochastic search on the finer grid
        while time.time() - start_time < budget_s * 0.98 and iter_cnt < max_iters * 2:
            sigma = 0.15 * (1.0 - iter_cnt / (max_iters * 2))
