import time
import numpy as np
import random

def _evaluate_h(h, dx):
    """Return max overlap C5 for a given step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def _initial_vector(n):
    """Create a feasible start: half the interval with value 1, rest 0,
    slightly smoothed to avoid strict 0/1 plateaus."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[:half] = 1.0
    # smooth a few points near the jump to give more flexibility
    eps = 0.05
    for i in range(max(0, half - 3), min(n, half + 3)):
        h[i] = max(0.0, min(1.0, h[i] * (1 - eps) + eps * random.random()))
    # ensure exact integral
    h *= (n / 2) / h.sum()
    return h

    Returns (h_values, c5_bound, n_points).

    random.seed(seed)

        n = init.size
        h_current = init.astype(float).copy()

    dx = 2.0 / n
    target_sum = n / 2.0          # because sum(h)*dx = 1

    # ensure feasibility of the initial vector
    h_current = np.clip(h_current, 0.0, 1.0)
    cur_sum = h_current.sum()
    if cur_sum == 0:
        h_current[:] = 1.0 / n
        cur_sum = h_current.sum()
    h_current *= target_sum / cur_sum

    best_h = h_current.copy()
    best_c5 = _evaluate_h(best_h, dx)

    # parameters for the stochastic search

    temperature_start = 0.02

    elapsed = lambda: time.time() - start_time
    while elapsed() < budget_s * 0.95:   # keep a safety margin
        # linear cooling schedule
