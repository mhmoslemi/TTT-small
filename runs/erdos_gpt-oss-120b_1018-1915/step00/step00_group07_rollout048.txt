import time
import numpy as np
import random

def _project_to_feasible(v, target_sum):
    """Project vector v onto [0,1]^n with exact sum = target_sum."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(20):
        s = v.sum()
        if abs(s - target_sum) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        if not free.any():
            # all at bounds – distribute remaining mass uniformly then clip
            v += (target_sum - s) / len(v)
            v = np.clip(v, 0.0, 1.0)
            break
        delta = (target_sum - s) / free.sum()
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    return v

def _evaluate(h, dx):
    """Return C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = len(h0)
    else:
        n = 80
        h0 = None
    dx = 2.0 / n
    target_sum = n / 2.0  # because integral must be 1

    # Helper to create a feasible random vector
    def random_feasible():
        v = np.random.rand(n)
        return _project_to_feasible(v, target_sum)

    # Initialise population
    pop_size = max(30, n)          # modest size, keeps runtime low
    F = 0.5                         # mutation factor
    CR = 0.9                        # crossover probability
    max_gen = 5000                  # generous upper bound, will be cut by time

    population = []
    scores = []

    # first individual = provided feasible one if present
    if h0 is not None:
        h0 = _project_to_feasible(h0, target_sum)
        population.append(h0)
        scores.append(_evaluate(h0, dx))
    # fill rest
    while len(population) < pop_size:
        cand = random_feasible()
        population.append(cand)
        scores.append(_evaluate(cand, dx))

    population = np.stack(population)
    scores = np.array(scores)
    best_idx = int(np.argmin(scores))
    best = population[best_idx].copy()
    best_score = float(scores[best_idx])

    start_time = time.time()
    gen = 0
    # main DE loop
    while gen < max_gen and (time.time() - start_time) < budget_s - 0.5:
        for i in range(pop_size):
            # mutation: select three distinct indices different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = random.sample(idxs, 3)
            mutant = population[r1] + F * (population[r2] - population[r3])
            mutant = _project_to_feasible(mutant, target_sum)

            # crossover (binomial)
            jrand = np.random.randint(n)
            trial = np.empty(n, dtype=float)
            for j in range(n):
                if np.random.rand() < CR or j == jrand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i, j]
            trial = _project_to_feasible(trial, target_sum)

            # selection
            trial_score = _evaluate(trial, dx)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best = trial.copy()
                    best_score = float(trial_score)
        gen += 1

    # Return result
    return (best.tolist(), best_score, n)
