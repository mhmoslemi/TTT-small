import time
import numpy as np

# Helper to evaluate the Erdős overlap for a candidate
def _c5_value(h):
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


# Project a vector onto the feasible set:
#   0 ≤ h_i ≤ 1  and  sum(h) = n/2
def _repair(h):
    n = h.size
    target = n / 2.0
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)

    # Simple scaling to hit the sum, iterating to respect bounds
    for _ in range(5):
        total = h.sum()
        if total == 0:
            # fill the first target entries with 1 (if integer) else proportionally
            ones = int(target)
            h[:ones] = 1.0
            if target - ones > 0:
                h[ones] = target - ones
            break
        scale = target / total
        h = h * scale
        # enforce bounds again
        over = h > 1.0
        under = h < 0.0
        if over.any():
            excess = (h[over] - 1.0).sum()
            h[over] = 1.0
            cnt = (~over).sum()
            if cnt:
                h[~over] += excess / cnt
        if under.any():
            deficit = (-h[under]).sum()
            h[under] = 0.0
            cnt = (~under).sum()
            if cnt:
                h[~under] -= deficit / cnt
        h = np.clip(h, 0.0, 1.0)

    # Final tiny adjustment to hit the target sum exactly (within float tolerance)
    total = h.sum()
    if total != 0:
        h *= target / total
    h = np.clip(h, 0.0, 1.0)
    return h


# Create a random feasible individual of length n
def _random_individual(n, rng):
    h = rng.random(n)          # uniform in [0,1)
    return _repair(h)


# Mutate an individual by Gaussian noise then repair
def _mutate(parent, rng, sigma=0.07):
    child = parent + rng.normal(0.0, sigma, size=parent.size)
    return _repair(child)


# Crossover two parents (average) then repair
def _crossover(p1, p2, rng):
    alpha = rng.random(p1.size)
    child = alpha * p1 + (1 - alpha) * p2
    return _repair(child)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points)
    where h_values is a NumPy 1‑D array representing the step function.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Desired resolution – keep it modest for speed
    n = 200                       # 200 samples on [0,2]
    dx = 2.0 / n

    # Initialise population
    pop_size = 30
    population = []
    scores = []

    # If a parent solution is provided, interpolate it to the current n
    if "initial_h_values" in globals():
        parent = globals()["initial_h_values"]
        if parent.ndim != 1:
            parent = parent.ravel()
        # Interpolate linearly to length n
        x_old = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, parent)
        h0 = _repair(h0)
        population.append(h0)
        scores.append(_c5_value(h0))

    # Fill the remaining slots with random individuals
    while len(population) < pop_size:
        h = _random_individual(n, rng)
        population.append(h)
        scores.append(_c5_value(h))

    # Track the best found
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # Simple (μ+λ) evolutionary loop
    while time.time() - start_time < budget_s - 0.5:   # small safety margin
        # tournament selection
        idx1, idx2 = rng.choice(pop_size, size=2, replace=False)
        p1 = population[idx1]
        p2 = population[idx2]

        # produce offspring
        if rng.random() < 0.5:
            child = _mutate(p1, rng)
        else:
            child = _crossover(p1, p2, rng)

        child_score = _c5_value(child)

        # replace worst individual if child is better
        worst_idx = int(np.argmax(scores))
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score
            if child_score < best_score:
                best_score = child_score
                best_h = child.copy()

        # occasional diversification to avoid stagnation
        if rng.random() < 0.02:
            idx = rng.integers(pop_size)
            population[idx] = _random_individual(n, rng)
            scores[idx] = _c5_value(population[idx])
            if scores[idx] < best_score:
                best_score = scores[idx]
                best_h = population[idx].copy()

    # Final polishing: slight local search around the best
    for _ in range(5):
        candidate = _mutate(best_h, rng, sigma=0.03)
        cand_score = _c5_value(candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_h = candidate

    # Ensure exact feasibility before returning
    best_h = _repair(best_h)

    return best_h, float(best_score), n
