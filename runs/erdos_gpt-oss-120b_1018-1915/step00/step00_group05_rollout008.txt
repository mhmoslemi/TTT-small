import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def _project_feasible(vec, target_sum):
    """Clip to [0,1] and adjust to hit the exact sum constraint."""
    v = np.clip(vec, 0.0, 1.0)
    cur = v.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return v
    # indices that are not at a bound
    free = (v > 0.0) & (v < 1.0)
    if free.any():
        # distribute diff uniformly among free variables
        adjust = diff / free.sum()
        v[free] += adjust
        # if we pushed some out of bounds, recurse once
        v = np.clip(v, 0.0, 1.0)
        # final scaling if still off
        cur = v.sum()
        if abs(cur - target_sum) > 1e-9:
            v *= target_sum / cur
        return v
    # fallback: scale whole vector then clip
    if cur > 0:
        v *= target_sum / cur
    v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h, dx):
    """Compute the overlap integral C5 for a candidate h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size

    if n_points is None:
        if "initial_h_values" in globals():
            n_points = len(initial_h_values)
        else:
            n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    bounds = [(0.0, 1.0)] * n_points

    # Starting point – use provided construction if present
    if "initial_h_values" in globals():
        x0 = np.array(initial_h_values, dtype=float)
        x0 = _project_feasible(x0, target_sum)
    else:
        # random feasible start
        rand = rng.random(n_points)
        x0 = _project_feasible(rand, target_sum)

    best_h = x0.copy()
    best_val = _c5_value(best_h, dx)

    # Penalty‑augmented objective for DE
    penalty_factor = 1e3

    def penalized_obj(x):
        s = x.sum()
        penalty = penalty_factor * abs(s - target_sum)
        c5 = _c5_value(x, dx)
        return c5 + penalty

    # Callback to respect the time budget

            return True  # stop DE
        return False

        atol=0,

        disp=False,
    )
    cand = _project_feasible(de_result.x, target_sum)
    cand_val = _c5_value(cand, dx)
    if cand_val < best_val:
        best_h, best_val = cand, cand_val
