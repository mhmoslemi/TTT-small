import numpy as np
import time

def _project_to_bounds_and_sum(x, target_sum, tol=1e-8, max_iter=50):
    """Project vector x onto the box [0,1] and the hyperplane sum = target_sum.
    Uses a simple binary search for the additive shift λ such that
    sum(clip(x + λ, 0, 1)) = target_sum.
    """
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if abs(s - target_sum) <= tol:
            return y
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def _c5_value(h):
    """Compute the C5 overlap value for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimizes C5.

    Returns
    -------
    h_values : np.ndarray
        Optimized step values (length n_points) in [0,1].
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Length of the returned array.
    """
    rnd = np.random.RandomState(seed)
    start_time = time.time()

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 70
        h0 = np.full(n, 0.5, dtype=float)  # uniform feasible start

    target_sum = n / 2.0  # because integral 1 ⇒ sum * (2/n) = 1

    # Ensure feasibility of the initial vector
    h_best = _project_to_bounds_and_sum(h0, target_sum)
    c5_best = _c5_value(h_best)

    # Simple hill‑climbing / simulated‑annealing loop
    # Parameters that adapt with the remaining time
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 0.2:  # tiny safety margin
            break

        # annealing schedule for step size
        progress = iter_count / 20000.0
        sigma = max(0.001, 0.2 * (1.0 - progress))

        # propose a new candidate
        proposal = h_best + sigma * rnd.randn(n)
        proposal = _project_to_bounds_and_sum(proposal, target_sum)

        c5_prop = _c5_value(proposal)

        # accept if improvement or with small probability (simulated annealing)
        if c5_prop < c5_best or rnd.rand() < 0.001:
            h_best, c5_best = proposal, c5_prop

        # occasional random restart to avoid stagnation
        if iter_count % 5000 == 0 and iter_count > 0:
            rand_vec = rnd.rand(n)
            rand_vec = _project_to_bounds_and_sum(rand_vec, target_sum)
            rand_c5 = _c5_value(rand_vec)
            if rand_c5 < c5_best:
                h_best, c5_best = rand_vec, rand_c5

        iter_count += 1

    return h_best, c5_best, n
