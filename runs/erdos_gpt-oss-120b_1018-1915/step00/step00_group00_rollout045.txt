import time
import numpy as np

def _c5_of_h(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _neighbor(rng, h):
    """Generate a neighboring h by moving a small amount epsilon from i to j."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # amount we can transfer without breaking bounds
    max_down = h[i]          # cannot go below 0
    max_up = 1.0 - h[j]      # cannot exceed 1
    eps_max = min(max_down, max_up)
    if eps_max <= 0:
        return h.copy()
    eps = rng.random() * eps_max * 0.5  # up to half of the feasible range
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h
