import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_c5(h):
    """Return C5 for discrete step function h on [0,2]."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_simplex(x, total):
    """Project onto box [0,1] and adjust to hit exact sum `total`."""
    # clip to [0,1]
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if np.isclose(s, total):
        return x
    # simple linear correction preserving bounds (may need iteration)
    # Distribute deficit/excess uniformly then clip again.
    delta = (total - s) / len(x)
    x += delta
    x = np.clip(x, 0.0, 1.0)
    # second pass to correct residual error
    s = x.sum()
    if not np.isclose(s, total):
        # scale uniformly (since after clipping values are interior)
        scale = total / s if s != 0 else 0.0
        x *= scale
        x = np.clip(x, 0.0, 1.0)
    return x
