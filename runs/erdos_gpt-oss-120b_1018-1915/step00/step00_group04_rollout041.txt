import time
import numpy as np

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _project_to_bounds_sum(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    current = v.sum()
    diff = target - current
    # Simple water‑filling loop
    max_iter = 100
    it = 0
    while abs(diff) > 1e-10 and it < max_iter:
        if diff > 0:
            free = v < 1.0 - 1e-12
            cnt = free.sum()
            if cnt == 0:
                break
            inc = min(diff / cnt, 1e-2)
            v[free] += inc
            v = np.clip(v, 0.0, 1.0)
        else:
            free = v > 1e-12
            cnt = free.sum()
            if cnt == 0:
                break
            dec = min(-diff / cnt, 1e-2)
            v[free] -= dec
            v = np.clip(v, 0.0, 1.0)
        current = v.sum()
        diff = target - current
        it += 1
    # Final small correction to hit target exactly (if feasible)
    if abs(diff) > 1e-12:
        # Distribute the remaining diff uniformly over non‑boundary entries
        mask = (v > 0.0 + 1e-12) & (v < 1.0 - 1e-12)
        inds = np.where(mask)[0]
        if inds.size > 0:
            v[inds] += diff / inds.size
            v = np.clip(v, 0.0, 1.0)
    return v

def _mutate(h, rng, step):
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    delta = rng.uniform(-step, step)
    new = h.copy()
    new[i] += delta
    new[j] -= delta
    target = n / 2.0
