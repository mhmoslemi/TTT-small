import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

# Helper: compute C5 for a candidate h (assumes correct sum)
def _c5_value(h):
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# Project a vector onto [0,1] with the exact sum constraint ∑h = n/2
def _project_h(vec, target_sum, eps=1e-9, max_iter=10):
    h = np.clip(vec, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        if abs(s - target_sum) <= eps:
            break
        # Scale globally, then re‑clip
        scale = target_sum / s if s != 0 else 0.0
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    # Final tiny adjustment by distributing the residual uniformly where possible
    diff = target_sum - h.sum()
    if abs(diff) > eps:
        mask = (diff > 0) * (h < 1.0) + (diff < 0) * (h > 0.0)
        free = np.where(mask)[0]
        if free.size:
            delta = diff / free.size
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
    return h

# Penalized objective for DE (soft equality)
def _penalized_obj(x, target_sum):
    h = np.clip(x, 0.0, 1.0)
    penalty = (h.sum() - target_sum) ** 2
    return _c5_value(h) + 1e4 * penalty

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # Determine problem size
    init = globals().get("initial_h_values", None)
    n = init.size if isinstance(init, np.ndarray) else kwargs.get("n_points", 70)
    target_sum = n / 2.0

    # Starting candidate
    if isinstance(init, np.ndarray) and init.size == n:
        h0 = _project_h(init.astype(float), target_sum)
    else:
        h0 = rng.random(n)
        h0 = _project_h(h0, target_sum)

    best_h = h0.copy()
    best_val = _c5_value(best_h)

    # ---- Differential Evolution (global) ----
    bounds = [(0.0, 1.0)] * n
    max_iter = max(1, int(0.25 * budget_s))  # allocate part of the budget
    de_res = differential_evolution(
        func=lambda x: _penalized_obj(x, target_sum),
        bounds=bounds,
        strategy="best1bin",
        maxiter=max_iter,
        polish=False,
        updating="deferred",
        seed=rng,
        disp=False,
        workers=1,
    )
    cand = _project_h(de_res.x, target_sum)
    cand_val = _c5_value(cand)
    if cand_val < best_val:
        best_h, best_val = cand, cand_val

    # ---- Local refinement (SLSQP) ----
    # Use remaining time budget, but cap iterations
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 2.0:
        cons = {"type": "eq", "fun": lambda x: x.sum() - target_sum}
        bounds_opt = [(0.0, 1.0)] * n
        opt_res = minimize(
            fun=lambda x: _c5_value(_project_h(x, target_sum)),
            x0=best_h,
            method="SLSQP",
            bounds=bounds_opt,
            constraints=[cons],
            options={"maxiter": int(2000 * remaining / budget_s), "ftol": 1e-9},
        )
        if opt_res.success:
            refined = _project_h(opt_res.x, target_sum)
            refined_val = _c5_value(refined)
            if refined_val < best_val:
                best_h, best_val = refined, refined_val

    # Ensure final feasibility
    best_h = _project_h(best_h, target_sum)
    best_val = _c5_value(best_h)

    return best_h, float(best_val), int(n)
