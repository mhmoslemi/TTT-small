import numpy as np
import time
import math

def _compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _initial_h(n: int) -> np.ndarray:
    """Obtain a feasible starting vector."""
    try:
        init = np.array(initial_h_values, copy=True)  # type: ignore[name-defined]
        if init.size == n:
            # ensure feasibility (may be slightly off due to numeric noise)
            init = np.clip(init, 0.0, 1.0)
            target = n / 2.0
            diff = target - init.sum()
            if abs(diff) > 1e-12:
                # distribute the tiny difference uniformly and clip again
                init += diff / n
                init = np.clip(init, 0.0, 1.0)
                # final projection onto the capped simplex
                init = _project_onto_capped_simplex(init, target)
            return init
    except Exception:
        pass
    # fallback: uniform 0.5 (feasible because sum = n*0.5 = n/2)
    return np.full(n, 0.5, dtype=float)


def _project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0<=x_i<=1, sum x_i = target}.
    Implements a simple iterative clipping algorithm (guaranteed to converge
    for this modest size).
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        excess = x.sum() - target
        if abs(excess) < 1e-12:
            break
        # indices that are not at a bound can be adjusted
        free = (x > 0.0) & (x < 1.0)
        if not np.any(free):
            # all at bounds – distribute uniformly (will stay at bounds)
            break
        delta = excess / np.count_nonzero(free)
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.

    Returns
    -------
    h_values : np.ndarray
        The best found vector (length n_points) with entries in [0,1] and
        sum(h)*dx = 1.
    c5_bound : float
        The corresponding C5 value.
    n_points : int
        Number of discretisation points used.
    """
    np.random.seed(seed)
    # Determine problem size
    try:
        n = len(initial_h_values)  # type: ignore[name-defined]
    except Exception:
        n = 70
    dx = 2.0 / n
    target_sum = n / 2.0

    # Initialise feasible h
    h = _initial_h(n)
    best_h = h.copy()
    best_score = _compute_c5(best_h, dx)

    start = time.time()
    # Annealing parameters
    T0 = 0.05
    min_temp = 1e-4
    iter_count = 0
    no_improve = 0

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        # temperature schedule (linear cooling)
        temp = max(min_temp, T0 * (1.0 - elapsed / budget_s))

        # propose a transfer between two coordinates (preserves sum exactly)
        i, j = np.random.choice(n, 2, replace=False)
        # maximum amount we can move from i to j without violating bounds
        max_forward = min(h[i], 1.0 - h[j])
        max_backward = -min(h[j], 1.0 - h[i])
        if max_forward == max_backward:
            continue
        eps = np.random.uniform(max_backward, max_forward)
        if abs(eps) < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps

        # evaluate
        score = _compute_c5(h_new, dx)

        # Metropolis acceptance
        if score < best_score or np.random.rand() < math.exp(-(score - best_score) / temp):
            h = h_new
            if score < best_score:
                best_score = score
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional diversification if stuck
        if no_improve > 20000:
            # random perturbation that respects bounds and sum:
            # pick a random index and re‑sample its value, fixing the sum by adjusting another.
            k = np.random.randint(n)
            new_val = np.random.rand()
            diff = new_val - h[k]
            # find a partner to absorb the diff
            partners = np.where((h > 0) & (h < 1))[0]
            partners = partners[partners != k]
            if partners.size == 0:
                # fallback to simple swap with another random index
                l = (k + 1) % n
                h[k], h[l] = h[l], h[k]
            else:
                l = np.random.choice(partners)

                if -diff >= -h[l] and -diff <= (1.0 - h[l]):

                else:
                    # otherwise just keep previous state
                    pass
            no_improve = 0

        iter_count += 1

    # final projection (safety)
    best_h = np.clip(best_h, 0.0, 1.0)
