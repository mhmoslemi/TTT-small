import numpy as np
import time
import random

def _evaluate(h):
    """Return C5 value for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _project_feasible(h, target):
    """Clip to [0,1] and adjust to meet sum == target."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if np.isclose(total, target, atol=1e-10):
        return h
    if total < target:
        deficit = target - total
        mask = h < 1.0
        if not np.any(mask):
            return h
        cap = (1.0 - h[mask]).sum()
        if cap > 0:
            inc = deficit * (1.0 - h[mask]) / cap
            h[mask] += inc
    else:  # total > target
        excess = total - target
        mask = h > 0.0
        if not np.any(mask):
            return h
        cap = h[mask].sum()
        if cap > 0:
            dec = excess * h[mask] / cap
            h[mask] -= dec
    # final clip (tiny rounding issues)
    h = np.clip(h, 0.0, 1.0)
    # If still off by a negligible amount, distribute uniformly
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        h -= diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _initial_h(n_points):
    """Create an initial feasible h of size n_points."""
    # Use provided initial_h_values if size matches, otherwise upscale.
    try:
        parent = initial_h_values  # global provided by harness
        parent = np.asarray(parent, dtype=np.float64)
        n_parent = parent.size
        if n_parent == n_points:
            return parent.copy()
        # linear interpolation onto new grid
        x_parent = np.linspace(0.0, 2.0, n_parent, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h_new = np.interp(x_new, x_parent, parent, period=2.0)
        h_new = _project_feasible(h_new, n_points / 2.0)
        return h_new
    except NameError:
        # No parent supplied – use uniform 0.5 which satisfies constraints.
        h = np.full(n_points, 0.5, dtype=np.float64)
        return _project_feasible(h, n_points / 2.0)

def _hill_climb(h_start, time_limit, max_iter=50000):
    """Simple stochastic hill‑climbing respecting constraints."""
    n = h_start.size
    target_sum = n / 2.0
    best_h = h_start.copy()
    best_val = _evaluate(best_h)
    h = best_h.copy()
    start_time = time.time()
    iter_count = 0
    while time.time() - start_time < time_limit and iter_count < max_iter:
        iter_count += 1
        # pick donor and receiver indices
        i = random.randrange(n)
        j = random.randrange(n)
        if i == j:
            continue
        donor = h[i]
        recv = h[j]
        if donor <= 0.0 or recv >= 1.0:
            continue
        # propose a transfer delta
        max_delta = min(donor, 1.0 - recv, 0.05)  # limit step size
        if max_delta <= 0:
            continue
        delta = random.random() * max_delta
        h[i] -= delta
        h[j] += delta
        # cheap feasibility check (bounds already respected)
        val = _evaluate(h)
        if val < best_val - 1e-9:   # improvement
            best_val = val
            best_h = h.copy()
        else:
            # revert move
            h[i] += delta
            h[j] -= delta
    return best_h, best_val
