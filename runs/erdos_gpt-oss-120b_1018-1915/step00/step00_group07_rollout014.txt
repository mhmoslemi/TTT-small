import time
import numpy as np

def _evaluate(h, dx):
    """Return C5 value for a given h."""
    # correlation with (1-h), full mode includes all shifts
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_to_feasible(h, target_sum):
    """
    Project vector h onto the box [0,1] and the affine constraint sum(h)=target_sum.
    Simple iterative clipping and scaling (works well for modest dimensions).
    """
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s == 0:
        # distribute uniformly
        return np.full_like(h, target_sum / h.size)
    # scale to hit the target sum, then clip again if needed
    h = h * (target_sum / s)
    # final clipping (any small overshoot)
    h = np.clip(h, 0.0, 1.0)
    # adjust any residual error by adding/subtracting a tiny amount from a random index
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.random.randint(0, h.size)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):

    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized samples (length n_points) in [0,1] with sum equal to n_points/2.
    c5_bound : float
        The achieved upper bound C5.

        Number of discretisation points (size of h_values).
    """

    np.random.seed(seed)

        h[: n_points // 2] = 1.0
        np.random.shuffle(h)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because ∑h * dx = 1

    # Ensure feasibility of the starting point
    h = _project_to_feasible(h, target_sum)

    best_h = h.copy()
    best_score = _evaluate(best_h, dx)

    # current state for simulated annealing / hill climbing
    cur_h = h.copy()
    cur_score = best_score

    # annealing schedule parameters
    max_iters = 10_000_000  # generous upper bound; loop will exit by time
    iter_count = 0
    temperature = 0.02  # initial temperature
    temp_decay = 0.99995  # decay per iteration

    # main optimisation loop
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.5:  # leave a tiny margin
            break
        if iter_count >= max_iters:

        iter_count += 1

        # propose a move: transfer a random amount between two indices
        i, j = np.random.choice(n_points, size=2, replace=False)
