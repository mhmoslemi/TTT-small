import numpy as np
import time

def _repair(h, target_sum, max_iter=10):
    """Clip to [0,1] and enforce the sum constraint."""
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        # Simple proportional adjustment
        if s > 0:
            h *= target_sum / s
        else:
            # If all zeros, distribute evenly
            h[:] = target_sum / h.size
    # Final clip
    h = np.clip(h, 0.0, 1.0)
    # Tiny redistribution to fix residual sum error
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # add diff uniformly and re‑clip
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h, dx):
    """Compute the overlap constant for a given step vector."""
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return conv.max()
