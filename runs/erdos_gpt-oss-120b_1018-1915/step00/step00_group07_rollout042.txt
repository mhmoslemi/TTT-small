import time
import numpy as np
import random

def _repair(h, target_sum):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = target_sum - total
    # Iterative proportional adjustment
    for _ in range(15):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            cap = room.sum()
            if cap < 1e-12:
                break
            delta = diff * (room / cap)
            h += delta
        else:
            room = h
            cap = room.sum()
            if cap < 1e-12:
                break
            delta = (-diff) * (room / cap)
            h -= delta
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        diff = target_sum - total
    return h

def _objective(h, dx):
    """Maximum overlap value for a candidate h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.

    Returns
    -------
    h_values : list of float
        Optimised values of h at the discretisation points.
    c5_bound : float
        Achieved upper bound for C5.
    n_points : int
        Number of discretisation points used.
    """
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 80
        h0 = np.full(n_points, 0.5)  # dummy feasible start

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum * dx = 1

    # DE parameters
    pop_size = max(20, n_points // 2)
    F = 0.5
    CR = 0.9
    max_generations = 2000  # safety upper bound; loop will stop by time

    # Initialise population
    pop = []
    # Include the provided initial vector (repair it to be feasible)
    pop.append(_repair(h0, target_sum))
    while len(pop) < pop_size:
        cand = np.random.rand(n_points)
        cand = _repair(cand, target_sum)
        pop.append(cand)
    pop = np.array(pop)

    # Pre‑evaluate
    scores = np.array([_objective(ind, dx) for ind in pop])
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_score = float(scores[best_idx])

    generation = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s or generation >= max_generations:
            break
        generation += 1

        for i in range(pop_size):
            # Choose distinct indices for mutation
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = random.sample(idxs, 3)

            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            # Binomial crossover
            cross_mask = np.random.rand(n_points) < CR
            # Ensure at least one component from mutant
            cross_mask[random.randrange(n_points)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            trial = _repair(trial, target_sum)

            trial_score = _objective(trial, dx)

            # Selection
            if trial_score < scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = float(trial_score)
                    best_h = trial.copy()

        # Early exit if we already beat the target threshold
        if best_score <= 0.3808:
            break

    return best_h.tolist(), best_score, n_points
