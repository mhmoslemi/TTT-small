import time
import math
import random
import numpy as np

def compute_c5(h):
    """Return max overlap integral for step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def repair_vector(v, target_sum):
    """Project v onto [0,1] box with exact sum = target_sum."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if s == 0.0:
        v[:] = target_sum / len(v)
        return v
    # Scale to meet the sum, then clip again
    v = v * (target_sum / s)
    v = np.clip(v, 0.0, 1.0)
    # Redistribute any residual error iteratively
    for _ in range(10):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = v < 1.0 - 1e-12
        else:
            idx = v > 1e-12
        if not np.any(idx):
            break
        delta = diff / np.count_nonzero(idx)
        v[idx] += delta
        v = np.clip(v, 0.0, 1.0)
    return v

def random_feasible(n, target_sum):
    """Generate a random feasible vector of length n."""
    rnd = np.random.rand(n)
    return repair_vector(rnd, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
