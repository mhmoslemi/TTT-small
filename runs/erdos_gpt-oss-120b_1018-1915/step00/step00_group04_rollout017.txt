import time
import numpy as np

def _evaluate(h):
    """Return C5 for a step function h sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _repair(h):
    """Project h onto the feasible set: 0≤h≤1 and sum(h)=n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # iterative affine correction
    for _ in range(5):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        # shift to fix the sum, then re‑clip
        shift = (target - s) / n
        h += shift
        h = np.clip(h, 0.0, 1.0)
    # final scaling if still off (rare)
    s = h.sum()
    if s > 0:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    # one last linear offset to be safe
    s = h.sum()
    if abs(s - target) > 1e-9:
        h += (target - s) / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _upsample(h, new_n):
    """Linear interpolation of h from its current grid to new_n points."""
    old_n = h.shape[0]
    xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(xs_new, xs_old, h)
    return _repair(h_new)
