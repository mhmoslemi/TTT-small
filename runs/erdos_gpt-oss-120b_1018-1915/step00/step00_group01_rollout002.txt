def _evaluate(h):
    """Return C5 for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project(h, target_sum):
    """Clip to [0,1] and adjust to hit the required sum (target_sum)."""
    h = np.clip(h, 0.0, 1.0)
    # simple iterative equal redistribution
    for _ in range(5):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # distribute equally to all entries, then re‑clip
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)

    cur = h.sum()
    if abs(cur - target_sum) > 1e-9:
