def project_to_bounds_and_sum(v, S):
    # water-filling: ensure 0<=v<=1 and sum=v=S
    v = np.clip(v, 0.0, 1.0)
    # iterative
    for _ in range(10):
        total = v.sum()
        diff = S - total
        if abs(diff) < 1e-9:
            break
        # indices that are not at bounds
        free = (v > 1e-12) & (v < 1-1e-12)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds, cannot adjust exactly, distribute proportionally to those not at 0 if need increase, or not at 1 if need decrease.
            # fallback: adjust all with tiny epsilon then clip again
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / cnt
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final clip
    v = np.clip(v, 0.0, 1.0)
    # final scaling adjustment if still off: adjust uniformly among non-boundary
    total = v.sum()
    if abs(total - S) > 1e-6:
        # simple scaling of whole vector and reclip
        v = v * (S / total)
        v = np.clip(v, 0.0, 1.0)
    return v

import time
import numpy as np
import random

def _project(v, target_sum):
    """Clip to [0,1] and adjust to meet the required sum (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(10):
        total = v.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        free = (v > 1e-12) & (v < 1 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds – distribute uniformly then clip again
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / cnt
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final safeguard scaling if still off
    total = v.sum()
    if abs(total - target_sum) > 1e-8:
        v *= target_sum / total
        v = np.clip(v, 0.0, 1.0)
    return v

def _score(h):
    """Compute C₅ for a discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function h, its C₅ bound, and the number of samples.
    The algorithm starts from `initial_h_values` (if present) and performs
    stochastic local search under the given time budget.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 100
        h = np.full(n, 0.5, dtype=float)

    n = len(h)
    target_sum = n / 2.0
    h = _project(h, target_sum)

    best_h = h.copy()
    best_score = _score(best_h)

    # ------------------------------------------------------------------
    # 2. Search loop
    # ------------------------------------------------------------------
    start_time = time.time()
    it = 0
    # parameters
    max_iter_without_check = 1000
    while time.time() - start_time < budget_s * 0.95:
        it += 1
        # linear cooling schedule for a simple temperature
        elapsed = time.time() - start_time
        temp = max(0.01, 1.0 - elapsed / budget_s)

        # ----- propose a new candidate ---------------------------------
        if np.random.rand() < 0.7:
            # small random perturbation on a random subset
            cand = best_h.copy()
            k = max(1, int(0.05 * n))
            idx = np.random.choice(n, k, replace=False)
            noise = np.random.normal(scale=0.12 * temp, size=k)
            cand[idx] += noise
        else:
            # smoothing move via a 3‑point kernel
            kernel = np.array([0.25, 0.5, 0.25])
            cand = np.convolve(best_h, kernel, mode='same')
        cand = _project(cand, target_sum)

        # ----- evaluate -------------------------------------------------
        cand_score = _score(cand)

        # ----- accept / reject ------------------------------------------
        if cand_score < best_score:
            best_h, best_score = cand, cand_score
        else:
            # occasional uphill moves (simulated annealing)
            if np.random.rand() < np.exp(-(cand_score - best_score) / (0.02 + temp)):
                best_h, best_score = cand, cand_score

        # ----- occasional diversification -------------------------------
        if it % max_iter_without_check == 0:
            # try a fresh random start
            rand = np.random.rand(n)
            rand = _project(rand, target_sum)
            rand_score = _score(rand)
            if rand_score < best_score:
                best_h, best_score = rand, rand_score

    # ------------------------------------------------------------------
    # 3. Return result
    # ------------------------------------------------------------------
    return (best_h, float(best_score), n)
