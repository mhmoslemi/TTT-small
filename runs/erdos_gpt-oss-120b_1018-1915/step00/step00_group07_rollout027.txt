import time
import numpy as np

# The harness provides evaluate_erdos_solution and possibly initial_h_values.
# We will compute the objective ourselves for speed.

def _c5_of_h(h, dx):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    # correlate h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_swap_move(rng, h):
    """Propose a feasible swap move: transfer mass from i to j."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None, None, None
    # amount we can move from i to j while staying in [0,1]
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return None, None, None
    delta = rng.random() * max_delta
    return i, j, delta

def _apply_swap(h, i, j, delta):
    """Return a new vector after moving delta from i to j."""
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
