import time
import numpy as np

def compute_c5(h, dx):
    """Return the overlap constant C5 for step values h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def propose_move(h, max_transfer=0.05):
    """Propose a feasible move that transfers a small amount of mass
    from index j to i while preserving the sum and staying in [0,1]."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # maximum that can be moved without violating bounds
    avail_i = 1.0 - h[i]
    avail_j = h[j]
    max_eps = min(avail_i, avail_j, max_transfer)
    if max_eps <= 1e-12:
        return None  # no feasible move
    eps = np.random.uniform(0.0, max_eps)
    h_new = h.copy()
    h_new[i] += eps
    h_new[j] -= eps
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns:
        h_values (np.ndarray): optimal values on the discretization grid.
        c5_bound (float): the achieved C5 upper bound.
        n_points (int): number of discretization samples.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine discretization size
    if 'initial_h_values' in globals():
        h_current = np.asarray(initial_h_values, dtype=np.float64).copy()
        n_points = h_current.size
    else:
        n_points = 90  # default modest size (<1k)
        # random feasible start: draw from Dirichlet then scale to [0,1] bounds
        raw = np.random.rand(n_points)
        target_sum = n_points / 2.0
        scale = target_sum / raw.sum()
        h_current = raw * scale
        # clip to [0,1] and re‑project onto the hyperplane sum = target_sum
        # simple iterative fix (few passes suffice)
        for _ in range(5):
            excess = h_current - 1.0
            deficit = -h_current
            excess_mask = excess > 0
            deficit_mask = deficit > 0
            if not excess_mask.any() and not deficit_mask.any():
                break
            # distribute excess uniformly to deficits
            excess_total = excess[excess_mask].sum()
            deficit_total = -deficit[deficit_mask].sum()
            if excess_total > 0:
                adjust = excess_total / deficit_mask.sum()
                h_current[deficit_mask] += adjust
                h_current[excess_mask] = 1.0
            if deficit_total > 0:
                adjust = deficit_total / excess_mask.sum()
                h_current[excess_mask] -= adjust

        # final normalization to exact sum

            sub = diff / down_idx.size
            best_h[down_idx] += sub
        best_h = np.clip(best_h, 0.0, 1.0)

    return best_h, best_score, n_points
