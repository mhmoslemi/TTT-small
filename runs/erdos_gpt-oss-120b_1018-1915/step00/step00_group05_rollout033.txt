import time
import numpy as np
from scipy.optimize import minimize

def _project_to_feasible(vec, desired_sum):
    """Clip to [0,1] and rescale to achieve the exact desired sum."""
    v = np.clip(vec, 0.0, 1.0)
    total = v.sum()
    if total == 0:
        # uniform fill that satisfies the sum
        v.fill(desired_sum / len(v))
        return v
    # First rescale
    v = v * (desired_sum / total)
    # After rescaling some entries may exceed 1; iterate a couple of times
    for _ in range(3):
        over = v > 1.0
        if not np.any(over):
            break
        excess = v[over] - 1.0
        v[over] = 1.0
        # redistribute excess uniformly over the under‑filled entries
        under = ~over
        if np.any(under):
            v[under] -= excess.sum() / under.sum()
        v = np.clip(v, 0.0, 1.0)
    # Final renormalisation to correct tiny drift
    v = v * (desired_sum / v.sum())
    return v


def _c5_score(h, dx):
    """Return the discrete C5 value for a step vector h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def _local_refine(h0, dx, desired_sum, max_iter=500):
    """SLSQP refinement keeping the equality constraint exactly."""
    n = len(h0)

    def obj(x):
        return _c5_score(x, dx)

    cons = ({
        'type': 'eq',
        'fun': lambda x: np.sum(x) - desired_sum
    })
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    if res.success:
        return _project_to_feasible(res.x, desired_sum)
    return h0


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------
    # problem size
    n_points = 70                     # prefer small, uses provided parent size
    dx = 2.0 / n_points
    desired_sum = n_points / 2.0      # because ∫ h = 1  ⇒  sum(h)*dx = 1
    # ------------------------------------------------------------
    # initial candidate
    parent = globals().get('initial_h_values', None)
    if parent is not None and len(parent) == n_points:
        cur = _project_to_feasible(np.asarray(parent, dtype=float), desired_sum)
    else:
        cur = np.random.rand(n_points)
        cur = _project_to_feasible(cur, desired_sum)

    best_h = cur.copy()
    best_score = _c5_score(best_h, dx)

    # ------------------------------------------------------------
    # main stochastic hill‑climbing loop
    max_iters = 20000
    sigma = 0.3               # initial perturbation magnitude
    decay = 0.9995            # gradual cooling
    it = 0
    while it < max_iters:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.5:   # keep a tiny margin
            break

        # propose a neighbour
        proposal = best_h + sigma * np.random.randn(n_points)
        proposal = _project_to_feasible(proposal, desired_sum)

        score = _c5_score(proposal, dx)
        if score < best_score:
            best_h, best_score = proposal, score

        sigma *= decay
        it += 1

        # occasional deeper local refinement (every 2000 iterations)
        if it % 2000 == 0:
            refined = _local_refine(best_h, dx, desired_sum, max_iter=300)
            refined_score = _c5_score(refined, dx)
            if refined_score < best_score:
                best_h, best_score = refined, refined_score

    # ------------------------------------------------------------
    # final safety projection and score recomputation
    best_h = _project_to_feasible(best_h, desired_sum)
    best_score = _c5_score(best_h, dx)

    # final assertions (will raise if violated)
    assert best_h.shape == (n_points,)
    assert np.all((best_h >= 0.0) & (best_h <= 1.0))
    assert np.isclose(best_h.sum(), desired_sum, atol=1e-9)

    return best_h, float(best_score), n_points
