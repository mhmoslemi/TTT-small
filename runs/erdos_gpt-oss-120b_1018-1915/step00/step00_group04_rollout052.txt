import time
import numpy as np
import random
import math

def _repair(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=len(h)/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # iterative rescaling to respect bounds and sum
    for _ in range(5):
        s = h.sum()
        if s == 0:
            h[:] = target / n
            break
        scale = target / s
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    # final tiny adjustment to hit target sum exactly (within machine eps)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny difference uniformly
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_score(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _mutate(parent: np.ndarray) -> np.ndarray:
    """Gaussian mutation of a subset of entries, then repair."""
    n = parent.size
    child = parent.copy()
    # mutate ~5% of entries (at least one)
    k = max(1, n // 20)
    idx = np.random.choice(n, k, replace=False)
    # noise magnitude proportional to current spread
    spread = child.max() - child.min()
    sigma = 0.2 * max(spread, 0.1)
    child[idx] += np.random.normal(0.0, sigma, size=k)
    return _repair(child)

def _crossover(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Simple uniform crossover with repair."""
    mask = np.random.rand(a.size) < 0.5
    child = np.where(mask, a, b)
    return _repair(child)
