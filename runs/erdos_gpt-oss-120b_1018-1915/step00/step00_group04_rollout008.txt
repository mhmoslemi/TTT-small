import time
import numpy as np

def _c5_of_h(h, dx):
    # correlation of h with (1-h); full mode gives all shifts
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _upscale(h, target_n):
    """Linear interpolation to change number of points."""
    n = h.size
    if n == target_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    return np.interp(x_new, x_old, h)
