import time
import numpy as np

def _compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx  approximated on the grid."""
    n = h.size
    # convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # dx = 2 / n
    return conv.max() * (2.0 / n)

def _thue_morse(n):
    """Return a binary Thue‑Morse sequence of length n (n must be a power of two)."""
    seq = np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=float)
    return seq

def run(seed=42, budget_s=1100, **kwargs):

        n = base.size
        # Force binary representation while respecting the integral constraint.
        h0 = (base > 0.5).astype(float)
    else:
        # fallback size
        n = 128
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)

    target_ones = n // 2
    current_ones = int(h0.sum())
