import numpy as np
import time
import random

def _repair_vector(vec, target_sum):
    """Project vec onto [0,1]^n with exact sum = target_sum."""
    v = np.clip(vec, 0.0, 1.0)
    total = v.sum()
    if abs(total - target_sum) < 1e-10:
        return v
    # Distribute the deficit/excess uniformly over free coordinates
    free = (v > 0) & (v < 1)
    while True:
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            # fallback: adjust first coordinate
            v[0] = np.clip(v[0] + (target_sum - v.sum()), 0, 1)
            break
        diff = target_sum - v.sum()
        inc = diff / free_idx.size
        v[free_idx] += inc
        # re‑clip and recompute free set
        np.clip(v, 0.0, 1.0, out=v)
        free = (v > 0) & (v < 1)
        if abs(v.sum() - target_sum) < 1e-10:
            break
    return v

def _objective(vec, dx):
    """Return the C5 value for a candidate step function."""
    # Ensure feasibility
    target = len(vec) / 2.0
    h = _repair_vector(vec, target)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()
