import time
import numpy as np
from math import exp

def _evaluate_c5(h):
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx

def _project(h):
    """Clip to [0,1] and enforce sum(h)=n/2 by redistributing excess."""
    n = h.size
    need = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        cur = h.sum()
        diff = need - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds – stretch uniformly and clip again
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / cnt
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def _upscale(h, new_n):
    old_n = h.size
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
