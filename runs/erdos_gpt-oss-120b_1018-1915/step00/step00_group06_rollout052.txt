import time
import numpy as np
import cvxpy as cp

def evaluate_h(h: np.ndarray) -> float:
    """Return the C5 value for a step function sampled in h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_bounds_sum(y: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of y onto {0<=x<=1, sum(x)=target}."""
    n = y.size
    x = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(x - y)),
                      [cp.sum(x) == target,
                       x >= 0,
                       x <= 1])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    except Exception:
        prob.solve(solver=cp.ECOS, warm_start=False, verbose=False)
    if x.value is not None:
        return np.asarray(x.value).reshape(-1)
    # fallback: simple clipping & scaling
    z = np.clip(y, 0.0, 1.0)
    s = z.sum()
    if s == 0:
        return np.full_like(z, target / n)
    z = np.clip(z * (target / s), 0.0, 1.0)
    diff = target - z.sum()
    if abs(diff) > 1e-12:
        # distribute the remaining mass uniformly on non‑boundary entries
        mask = (z > 0.0) & (z < 1.0)
        if mask.any():
            z[mask] += diff / mask.sum()
            z = np.clip(z, 0.0, 1.0)
    return z

    """

    start_time = time.time()

    # ------------------------------------------------------------------

    target_sum = n / 2.0

    # Ensure feasibility of the starting point
    h = project_bounds_sum(h0, target_sum)
