import numpy as np, time, random, math

def _evaluate(h, dx):
    """Return max overlap C5 for step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project(h, target_sum):
    """Clip to [0,1] and adjust to satisfy sum == target_sum."""
    # clip first
    h = np.clip(h, 0.0, 1.0)
    # simple iterative scaling to meet the sum constraint
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        # scale uniformly, then clip again
        if s > 0:
            h *= target_sum / s
        else:
            h[:] = target_sum / h.size
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction by adding a constant offset if needed
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff uniformly, respecting bounds
        add = diff / h.size
        h += add
        h = np.clip(h, 0.0, 1.0)
        # re‑normalize if still off
        s = h.sum()
        if abs(s - target_sum) > 1e-12 and s > 0:
            h *= target_sum / s

def _interpolate_parent(parent, n):

    return np.interp(x_new, x_old, parent)
