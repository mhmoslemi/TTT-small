import time
import math
import numpy as np

# helpers are pre‑imported in the environment:
#   evaluate_erdos_solution(h)  – returns C5 for a given h vector
#   initial_h_values            – optional starting vector (length may differ)

def _compute_c5(h):
    """Return the C5 value for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def _project_sum_bounds(v, target, lo=0.0, hi=1.0, tol=1e-9, max_iter=100):
    """
    Euclidean projection of v onto the box [lo,hi] with a fixed sum.
    Solution has the form w_i = clip(v_i - λ, lo, hi) where λ is found
    by bisection so that sum w_i = target.
    """
    # initial λ bounds
    low = min(v) - hi
    high = max(v) - lo
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        w = np.clip(v - lam, lo, hi)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(v - (low + high) * 0.5, lo, hi)

def _initial_h(n):
    """Create an initial h of length n, using the provided parent if any."""
    if "initial_h_values" in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        # simple linear interpolation onto n points
        x_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_parent, parent)
    else:
        # uniform start: 0.5 on half the interval, 0 elsewhere
        h = np.zeros(n)
        half = n // 2
        h[:half] = 1.0
        h[half:] = 0.0
    # enforce exact integral constraint
    target = n / 2.0
    h = _project_sum_bounds(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] that minimizes C5.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # resolution (may be increased later if time permits)
    n = 200
    target_sum = n / 2.0
    dx = 2.0 / n

    # initialise
    h_current = _initial_h(n)
    best_h = h_current.copy()
    best_score = _compute_c5(best_h)

    # annealing schedule
    max_iters = 200000
    t_start = 0.5
    t_end = 1e-4

    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        if it >= max_iters:
            break

        # temperature decreasing linearly
        temp = t_start + (t_end - t_start) * (it / max_iters)

        # propose a perturbation
        proposal = h_current + rng.normal(scale=temp, size=n)
        proposal = _project_sum_bounds(proposal, target_sum)

        # evaluate
        score = _compute_c5(proposal)

        # acceptance test (Metropolis)
        if score < best_score:
            best_h = proposal.copy()
            best_score = score
        if score < _compute_c5(h_current) or rng.random() < math.exp(
            (_compute_c5(h_current) - score) / max(temp, 1e-12)
        ):
            h_current = proposal

        it += 1

    # Optional refinement at higher resolution if time remains
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30.0:
        # upsample to a finer grid (e.g., 400 points) using interpolation
        n_fine = 400
        x_coarse = np.linspace(0.0, 2.0, n, endpoint=False)
        x_fine = np.linspace(0.0, 2.0, n_fine, endpoint=False)
        h_fine = np.interp(x_fine, x_coarse, best_h)
        target_fine = n_fine / 2.0
        h_fine = _project_sum_bounds(h_fine, target_fine)

        # brief local search on fine grid
        max_iters_fine = 50000
        t_start_fine = 0.3
        t_end_fine = 5e-5
        for it_f in range(max_iters_fine):
            if time.time() - start_time >= budget_s * 0.99:
                break
            temp = t_start_fine + (t_end_fine - t_start_fine) * (it_f / max_iters_fine)
            prop = h_fine + rng.normal(scale=temp, size=n_fine)
            prop = _project_sum_bounds(prop, target_fine)
            sc = _compute_c5(prop)
            if sc < best_score:
                best_h = prop.copy()
                best_score = sc
                h_fine = prop
            elif sc < _compute_c5(h_fine) or rng.random() < math.exp(
                (_compute_c5(h_fine) - sc) / max(temp, 1e-12)
            ):
                h_fine = prop

        # return fine solution
        return best_h, best_score, n_fine

    return best_h, best_score, n
