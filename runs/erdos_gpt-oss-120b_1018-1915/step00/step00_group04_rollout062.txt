import time
import numpy as np

def _repair(h, n):
    """Force h into [0,1] and make sum exactly n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Simple proportional scaling, then iterative fix for clipping side‑effects
    total = h.sum()
    if total != 0:
        h *= target / total
    # Iteratively distribute the residual while respecting bounds
    for _ in range(10):
        total = h.sum()
        diff = target - total
        if abs(diff) < 1e-10:
            break
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_score(h):
    """Compute C_5 for a given step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _neighbor(h):
    """Generate a neighbour by moving a small amount from one index to another."""
    n = h.size
    a, b = np.random.choice(n, 2, replace=False)
    # amount that can be moved without violating bounds
    max_inc = 1.0 - h[a]
    max_dec = h[b]
    delta = np.random.uniform(-max_dec, max_inc)
    h2 = h.copy()
    h2[a] += delta
    h2[b] -= delta
    return h2

def _initial_binary(n):
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def _upscale(parent, new_n):
    """Linear interpolation of a parent vector to a larger resolution."""
    old_n = parent.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h = np.interp(x_new, x_old, parent)
    h = _repair(h, new_n)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best found vector (values in [0,1]).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Length of the vector (grid size).
    """
    np.random.seed(seed)
    start = time.time()
    # ------------------------------------------------------------
    # Choose resolution and starting point
    # ------------------------------------------------------------
    TARGET_N = 200                     # modest size, fast evaluation
    if "initial_h_values" in globals():
        parent = initial_h_values
        # upscale / downscale to TARGET_N
        h_cur = _upscale(parent, TARGET_N)
    else:
        h_cur = _initial_binary(TARGET_N)

    h_cur = _repair(h_cur, TARGET_N)
    best_h = h_cur.copy()
    best_score = _c5_score(best_h)

    # ------------------------------------------------------------
    # Simulated annealing loop
    # ------------------------------------------------------------
    cur_score = best_score
    temp = best_score * 0.2           # initial temperature proportional to score
    decay = 0.9995                    # geometric cooling
    it = 0
    while time.time() - start < budget_s * 0.95:
        it += 1
        h_new = _neighbor(h_cur)
        h_new = _repair(h_new, TARGET_N)
        new_score = _c5_score(h_new)

        if new_score < best_score:
            best_score = new_score
            best_h = h_new.copy()

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
            h_cur = h_new
            cur_score = new_score

        temp *= decay
        if temp < 1e-8:
            temp = 1e-8

    # ------------------------------------------------------------
    # Optional refinement at higher resolution if time permits
    # ------------------------------------------------------------
    if time.time() - start < budget_s * 0.95:
        # try one refinement step to 400 points
        refine_n = 400
        h_ref = _upscale(best_h, refine_n)
        cur_score = _c5_score(h_ref)
        best_score_ref = cur_score
        best_h_ref = h_ref.copy()
        temp = cur_score * 0.2
        decay = 0.999
        while time.time() - start < budget_s * 0.95:
            h_new = _neighbor(h_ref)
            h_new = _repair(h_new, refine_n)
            new_score = _c5_score(h_new)
            if new_score < best_score_ref:
                best_score_ref = new_score
                best_h_ref = h_new.copy()
            if new_score < cur_score or np.random.rand() < np.exp(-(new_score - cur_score) / max(temp, 1e-12)):
                h_ref = h_new

            temp *= decay

        if best_score_ref < best_score:
            best_h = best_h_ref
            best_score = best_score_ref
            TARGET_N = refine_n

    return best_h, best_score, TARGET_N
