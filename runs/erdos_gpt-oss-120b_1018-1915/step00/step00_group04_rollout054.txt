import time
import numpy as np

def _compute_c5(h):
    """Return max overlap integral for step function h on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _repair(h, target_sum):
    """Project h onto [0,1] and adjust to meet exact sum target."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # Distribute diff to variables not at bounds
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size == 0:
        # all at bounds, just scale globally (might violate bounds a little)
        h = h * (target_sum / cur)
        return np.clip(h, 0.0, 1.0)
    # simple proportional distribution
    add = diff / free.size
    h[free] += add
    # Clip again and, if still off, recurse once (tiny residuals)
    return _repair(h, target_sum)

def _neighbor(h, rng):
    """Generate a feasible neighbor by moving mass between two indices."""
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    eps_max_pos = min(hi, 1.0 - hj)          # move from i to j
    eps_max_neg = min(hj, 1.0 - hi)          # move from j to i (negative)
    eps = rng.uniform(-eps_max_neg, eps_max_pos)
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h

def _interpolate_up(h):
    """Double the resolution using linear interpolation and repair."""
    n_old = len(h)
    n_new = n_old * 2

    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    target = n_new / 2.0
    h_new = _repair(h_new, target)
