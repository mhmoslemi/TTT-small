import time
import numpy as np
import math

def _c5_value(h):
    """Return the overlap integral C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h on the chosen grid.
    c5_bound : float
        The achieved upper bound C5.
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.shape[0]
    else:
        n = kwargs.get('n_points', 70)
        # Random feasible start: uniform then scale to satisfy sum = n/2
        h = rng.random(n)
        target_sum = n / 2.0
        current_sum = h.sum()
        if current_sum == 0:
            h[:] = target_sum / n
        else:
            h *= target_sum / current_sum
        # Clip to [0,1] while preserving sum (simple projection)
        excess = h - 1.0
        excess[excess < 0] = 0.0
        if excess.sum() > 0:
            # move excess down uniformly
            h -= excess * (excess.sum() / (n - (h > 1).sum()))
        h = np.clip(h, 0.0, 1.0)

    target_sum = n / 2.0

    best_h = h.copy()
    best_val = _c5_value(best_h)

    max_iter = 10_000_000  # safety upper bound
