import numpy as np
import time

def _c5_value(h, dx):
    """Compute C5 for a step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def _initial_feasible(n, rng):
    """Create a feasible start: exactly half the mass, respecting bounds."""
    # start with binary half‑ones, half‑zeros and shuffle
    h = np.zeros(n, dtype=np.float64)
    h[:n // 2] = 1.0
    rng.shuffle(h)
    # allow a little randomness while keeping the sum exact
    for _ in range(5):
        i, j = rng.choice(n, 2, replace=False)
        delta = (rng.random() - 0.5) * 0.2  # small tweak
        # clamp to bounds and keep sum
        max_inc_i = min(1.0 - h[i], 0.1)
        max_dec_j = min(h[j], 0.1)
        max_delta = min(max_inc_i, max_dec_j)
        min_inc_i = -min(h[i], 0.1)
        min_dec_j = -min(1.0 - h[j], 0.1)
        min_delta = max(min_inc_i, min_dec_j)
        delta = np.clip(delta, min_delta, max_delta)
        h[i] += delta
        h[j] -= delta
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : numpy.ndarray
        Optimized step function values on the uniform grid.
    c5_bound : float
        Achieved upper bound for C5.
    n_points : int
        Number of discretisation points (grid size).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values, dtype=np.float64).copy()
        n_points = h_current.size
    else:
        n_points = kwargs.get('n_points', 70)
        h_current = _initial_feasible(n_points, rng)

    dx = 2.0 / n_points

    # Ensure feasibility of the starting vector
    total = h_current.sum()
    desired = n_points / 2.0
    if abs(total - desired) > 1e-9:
        # simple scaling + clipping loop to repair exact sum
        h_current = np.clip(h_current * (desired / total), 0.0, 1.0)
        # fix any residual discrepancy by distributing it over random indices
        diff = desired - h_current.sum()
        while abs(diff) > 1e-9:
            idx = rng.integers(0, n_points)
            adj = np.sign(diff) * min(abs(diff), 0.001, 1.0 - h_current[idx] if diff > 0 else h_current[idx])
            h_current[idx] += adj
            diff = desired - h_current.sum()

    best_h = h_current.copy()
    best_score = _c5_value(best_h, dx)

    # Simulated annealing parameters
    temperature = 0.1
    cooling = 0.9995
    min_temp = 1e-5
    max_iters = 500000

    n = n_points
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s or temperature < min_temp:
            break

        # propose a move that preserves the total sum
        i, j = rng.choice(n, 2, replace=False)
        # allowable delta range respecting bounds and a max step size
        max_inc_i = min(1.0 - best_h[i], 0.1)
        max_dec_j = min(best_h[j], 0.1)
        max_delta = min(max_inc_i, max_dec_j)
        max_inc_j = min(1.0 - best_h[j], 0.1)
        max_dec_i = min(best_h[i], 0.1)
        min_delta = -min(max_inc_j, max_dec_i)  # negative of symmetric bound
        delta = rng.uniform(min_delta, max_delta)

        # construct candidate
        cand = best_h.copy()
        cand[i] += delta
        cand[j] -= delta

        # evaluate
        cand_score = _c5_value(cand, dx)

        # acceptance

            best_h, best_score = cand, cand_score
        else:
            prob = np.exp(-(cand_score - best_score) / temperature)
            if rng.random() < prob:
                best_h, best_score = cand, cand_score

        # anneal
        temperature *= cooling

        # occasional random restart to escape stagnation

            restart_score = _c5_value(restart, dx)
            if restart_score < best_score:

    return best_h, float(final_score), n_points
