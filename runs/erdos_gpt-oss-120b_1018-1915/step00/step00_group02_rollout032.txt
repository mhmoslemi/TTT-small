import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for a discrete step function h on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def adjust_sum(vec):
    """Scale / shift a vector so that its sum equals len(vec)/2 while staying in [0,1]."""
    target = len(vec) / 2.0
    s = vec.sum()
    if abs(s - target) < 1e-12:
        return vec
    diff = target - s
    # try to fix by a single entry
    for _ in range(200):
        idx = np.random.randint(len(vec))
        new = vec[idx] + diff
        if 0.0 <= new <= 1.0:
            vec[idx] = new
            return vec
    # fallback: proportional scaling and clipping
    if s > 0:
        vec = vec * (target / s)
    vec = np.clip(vec, 0.0, 1.0)
    # final tiny correction if needed
    s = vec.sum()
    if abs(s - target) > 1e-12:
        vec[-1] += target - s
        vec[-1] = np.clip(vec[-1], 0.0, 1.0)
    return vec

def improve_h(h0, time_budget, start_time):
    """Simple stochastic local search keeping the integral constraint."""
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    n = len(best_h)
    # step size schedule
    max_step = 0.04
    min_step = 0.001
    it = 0
    while time.time() - start_time < time_budget:
        it += 1
        # gradually shrink step size
        step = max_step * (0.5 ** (it // 2000))
        step = max(step, min_step)
        i, j = np.random.choice(n, 2, replace=False)
        delta = (np.random.rand() - 0.5) * step
        hi, hj = best_h[i], best_h[j]
        new_hi = hi + delta
        new_hj = hj - delta
        if not (0.0 <= new_hi <= 1.0 and 0.0 <= new_hj <= 1.0):
            continue
        cand = best_h.copy()
        cand[i] = new_hi
        cand[j] = new_hj
        c5 = compute_c5(cand)
        if c5 < best_c5 - 1e-9:          # accept only strict improvement
            best_h = cand
            best_c5 = c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Starting vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = len(h0)
        # enforce the exact integral constraint
        h0 = adjust_sum(h0)
    else:
        # fall back to a simple rectangular step function
        n = 100  # even for convenience
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        h0 = adjust_sum(h0)

    # ------------------------------------------------------------------
    # Optimization phase (stochastic local search)
    # ------------------------------------------------------------------
    remaining = budget_s - 0.5  # leave a tiny margin
    best_h, best_c5 = improve_h(h0, remaining, start)

    # ------------------------------------------------------------------
    # Return results
    # ------------------------------------------------------------------
    return best_h.tolist(), float(best_c5), len(best_h)
