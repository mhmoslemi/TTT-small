import time, math
import numpy as np

def compute_c5(h, n):
    """Return max overlap integral for discrete h on [0,2]."""
    dx = 2.0 / n
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()

def upscale_h(init, n_target):
    """Linear interpolation of a coarse vector onto n_target points."""
    N0 = len(init)
    x0 = np.linspace(0.0, 2.0, N0, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(xt, x0, init)

def project_bounds_sum(h, target_sum, max_iters=10):
    """Project h into [0,1] with exact sum = target_sum (bounded simplex)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iters):
        total = h.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            cap = 1.0 - h
        else:
            cap = h
        cap_sum = cap.sum()
        if cap_sum == 0:
            break
        h += diff * (cap / cap_sum)
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős overlap constant."""
    rnd = np.random.RandomState(seed)
    start_time = time.time()

    # discretisation (keep modest size for speed)
    n = 200                       # number of samples in [0,2]
    target_sum = n / 2.0          # because ∫ h = 1 ⇒ sum(h)=n/2

    # initialise h
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        h = upscale_h(init, n)
    else:
        h = rnd.rand(n)
    h = project_bounds_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)

    # annealing parameters
    max_iters = 250000
    it = 0

    while it < max_iters:
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # simple linear cooling: temperature proportional to remaining time
        remaining = max(0.0, budget_s - (time.time() - start_time))
        temp = remaining / budget_s

        # propose a mass transfer between two positions (preserves sum)
        i, j = rnd.randint(0, n, size=2)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue
        delta = rnd.rand() * max_delta

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        c5_new = compute_c5(h_new, n)

        # accept if better or with Metropolis probability
        if c5_new < best_c5 or rnd.rand() < math.exp(-(c5_new - best_c5) / (0.001 + temp)):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        it += 1

    return (best_h, best_c5, n)
