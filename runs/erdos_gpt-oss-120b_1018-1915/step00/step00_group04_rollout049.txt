import time
import numpy as np

def _compute_c5(h):
    """Return the Erdős overlap constant for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _repair(h, target_sum):
    """Clip to [0,1] and adjust to hit target_sum exactly (within eps)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # Distribute the difference over non‑saturated entries
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size == 0:  # all at bounds – fallback to uniform scaling
        return np.clip(h * (target_sum / cur), 0.0, 1.0)
    # add/subtract equally
    delta = diff / free.size
    h[free] += delta
    # recurse a few times to settle rounding issues
    return _repair(h, target_sum)

def _initial_vector(n):
    """Create a simple feasible start: half ones, half zeros, shuffled."""
    h = np.zeros(n)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h.astype(float)
