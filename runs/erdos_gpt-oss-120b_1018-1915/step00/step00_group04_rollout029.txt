import time
import numpy as np

def _repair(h):
    """Clip to [0,1] and enforce sum(h)=len(h)/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        h.fill(target / n)
        return h
    h *= target / cur
    # iterative project to bounds & sum
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        h += diff / n
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur != 0:
        h *= target / cur
    return h

def _c5_value(h):
    """Return C5 for a discretised step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def _interpolate(h, new_n):
    """Linear interpolation of h to length new_n, preserving domain [0,2]."""
    n = h.shape[0]
    old_x = np.linspace(0.0, 2.0, n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, h)
    return _repair(new_h)

def run(seed=42, budget_s=1100, **kwargs):

    Returns a tuple (h_values, c5_bound, n_points).
    h_values : np.ndarray with values in [0,1] satisfying sum(h)=n/2
    c5_bound : float, the achieved upper bound for C5
    n_points : int, length of h_values
    """
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        base_n = h0.shape[0]
    else:
        base_n = 42
        h0 = np.full(base_n, 0.5)  # trivial feasible start
    # Aim for a modest resolution that improves the bound but stays fast.
    # Double the size if we have enough budget left after a quick warm‑up.
    target_n = min(168, base_n * 2)   # 84 for the usual case
    h0 = _interpolate(h0, target_n)
    n = h0.shape[0]

    # ------------------------------------------------------------------
    # GA parameters (tuned for speed)
    pop_size = max(20, n // 3)            # at least 20 individuals
    elite_frac = 0.05
    elite_cnt = max(1, int(pop_size * elite_frac))
    max_gen = 5000                       # generous upper bound, will stop by time
    sigma_init = 0.07                    # mutation strength
    tournament_k = 3

    # ------------------------------------------------------------------
    # Initialise population
    population = []
    # include the interpolated parent
    population.append(_repair(h0.copy()))
    while len(population) < pop_size:
        cand = np.random.rand(n)
        population.append(_repair(cand))
    population = np.array(population, dtype=float)

    best_h = None
    best_c5 = np.inf

    gen = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:      # leave tiny margin for final steps
            break
        if gen >= max_gen:
            break
        # ------------------------------------------------------------------
        # Evaluate
        values = np.array([_c5_value(ind) for ind in population])
