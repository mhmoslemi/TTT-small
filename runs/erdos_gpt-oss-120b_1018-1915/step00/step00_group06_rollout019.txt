import time
import numpy as np

def evaluate_c5(h):
    """Return C5 for a discretized step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def repair(h):
    """Project h onto the feasible set: 0<=h<=1 and sum(h)=n/2."""
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    # Enforce the integral constraint by simple iterative redistribution.
    for _ in range(10):
        s = h.sum()
        target = n / 2.0
        diff = target - s
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def hill_climb(start_h, budget_s, start_time, rng):
    """Simple stochastic local search that respects the time budget."""
    h = start_h.copy()
    n = h.size
    best_h = h.copy()
    best_score = evaluate_c5(best_h)

    # initial step size; will decay
    step = 0.2
    it = 0
    while time.time() - start_time < budget_s:
        # propose a single-coordinate perturbation
        i = rng.integers(0, n)
        delta = (rng.random() * 2.0 - 1.0) * step
        h_trial = best_h.copy()
        h_trial[i] += delta
        h_trial = repair(h_trial)

        score = evaluate_c5(h_trial)

        # accept if improvement
        if score < best_score:
            best_h = h_trial
            best_score = score

        it += 1
        if it % 1000 == 0:
            step *= 0.9   # slowly reduce the search radius
    return best_h, best_score

def upsample(h, new_n):
    """Linear interpolation of a periodic function defined on [0,2]."""
    old_n = h.size
    # sample points including the endpoint to keep periodicity
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    # periodic extension via modulo
    h_ext = np.concatenate([h, h[:1]])  # one extra point to close the period
    old_x_ext = np.concatenate([old_x, [2.0]])
    return np.interp(new_x, old_x_ext, h_ext)
