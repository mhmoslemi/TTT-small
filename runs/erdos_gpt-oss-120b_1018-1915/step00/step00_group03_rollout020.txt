import time
import numpy as np

def _compute_c5(h):
    """Return the overlap bound C5 for a given step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _make_binary_initial(h_float):
    """Project a real‑valued vector onto the {0,1} simplex with sum = n/2."""
    n = h_float.shape[0]
    target = n // 2
    # start from a simple rounding
    h_bin = (h_float > 0.5).astype(float)
    cur = int(h_bin.sum())
    rng = np.random.default_rng()
    # fix the sum by random flips
    while cur != target:
        if cur > target:
            # turn a random 1 into 0
            ones = np.flatnonzero(h_bin == 1.0)
            idx = rng.choice(ones)
            h_bin[idx] = 0.0
            cur -= 1
        else:
            zeros = np.flatnonzero(h_bin == 0.0)
            idx = rng.choice(zeros)
            h_bin[idx] = 1.0
            cur += 1
    return h_bin

def _random_binary(n, rng):
    """Generate a random binary vector with exactly n/2 ones."""
    target = n // 2
    h = np.zeros(n, dtype=float)
    h[:target] = 1.0
    rng.shuffle(h)
    return h

    best_score = _compute_c5(best_h)

    # --------------------------------------------------------------
    # Helper for cheap constrained swaps
    # --------------------------------------------------------------
    def _propose_swap(h):
        """Swap a random 1 with a random 0."""
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if len(ones) == 0 or len(zeros) == 0:
            return h.copy()  # shouldn't happen
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i], h_new[j] = 0.0, 1.0
        return h_new

    # --------------------------------------------------------------
    # Simple simulated‑annealing style search
    # --------------------------------------------------------------
    iteration = 0
    temperature = 0.1  # initial acceptance probability for worses
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.2:  # keep a tiny safety margin
            break

        iteration += 1
        # occasional random restart to diversify
        if iteration % 500 == 0:
            h_candidate = _random_binary(n, rng)

        cand_score = _compute_c5(h_candidate)
