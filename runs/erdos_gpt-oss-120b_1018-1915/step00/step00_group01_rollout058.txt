import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _initial_binary(n, rng):
    """Create a binary vector of length n with exactly n/2 ones."""
    half = n // 2
    h = np.zeros(n, dtype=np.float64)
    ones_idx = rng.choice(n, size=half, replace=False)
    h[ones_idx] = 1.0
    return h

def _prepare_initial_h(n, rng):
    """Return a binary starting vector, using initial_h_values if possible."""
    # try to use the global initial_h_values if it matches the size
    glob = globals()
    if "initial_h_values" in glob:
        init = np.asarray(glob["initial_h_values"], dtype=np.float64)
        if init.shape[0] == n:
            # binarise by rounding, then fix the exact count
            h = (init >= 0.5).astype(np.float64)
            cur = int(h.sum())
            need = n // 2 - cur
            if need > 0:
                zero_idx = np.where(h == 0)[0]
                flip = rng.choice(zero_idx, size=need, replace=False)
                h[flip] = 1.0
            elif need < 0:
                one_idx = np.where(h == 1)[0]
                flip = rng.choice(one_idx, size=-need, replace=False)
                h[flip] = 0.0
            return h
    # fallback random binary
    return _initial_binary(n, rng)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where:
      * h_values – numpy array of step heights in [0,1] (length n_points)
      * c5_bound – the achieved max overlap integral
      * n_points – number of discretisation points used

    rng = np.random.default_rng(seed)

    # Choose a working resolution (balance quality vs speed)
    n_points = 200                       # dx = 2 / n_points
    dx = 2.0 / n_points

    # Initial candidate (binary, respecting the integral constraint)
    h_current = _prepare_initial_h(n_points, rng)
    best_h = h_current.copy()
    best_c = _compute_c5(best_h, dx)

    start = time.time()
    # Main stochastic hill‑climbing loop
    while time.time() - start < budget_s * 0.94:
        # Perform a batch of random swaps
        for _ in range(500):
            # indices of a 1 and a 0
            ones = np.where(h_current == 1)[0]
            zeros = np.where(h_current == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:

            i = rng.choice(ones)
            j = rng.choice(zeros)

            # swap them
            h_candidate = h_current.copy()
            h_candidate[i] = 0.0
            h_candidate[j] = 1.0

            c_candidate = _compute_c5(h_candidate, dx)
            if c_candidate < best_c:
                best_c = c_candidate

            # accept only if improves the current state
