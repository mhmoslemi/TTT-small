import time
import numpy as np
from scipy import optimize

# Global helpers are assumed to be pre‑imported:
#   evaluate_erdos_solution(h)  -> returns C5 bound
#   initial_h_values            -> numpy array if provided

def _integral_target(n):
    """Required sum of samples for integral = 1 on [0,2] with n points."""
    return n / 2.0

def _project_to_feasible(h):
    """Clip to [0,1] and adjust to satisfy the exact integral constraint."""
    n = h.size
    target = _integral_target(n)
    h = np.clip(h, 0.0, 1.0)

    for _ in range(8):               # a few correction passes are enough
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-10:
            break
        # distribute the difference uniformly over non‑boundary entries
        mask = (h > 0.0) & (h < 1.0)
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _evaluate(h):
    """Compute the C5 quantity for a candidate vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _random_feasible(n, rng):
    """Generate a random feasible vector (bounds + exact integral)."""
    target = _integral_target(n)
    h = rng.random(n)
    h = h * (target / h.sum())
    h = _project_to_feasible(h)
    return h

def _local_optimize(start):
    """Run a small SLSQP refinement starting from `start`."""
    n = start.size
    target = _integral_target(n)

    bounds = [(0.0, 1.0)] * n
    cons = ({
        "type": "eq",
        "fun": lambda x: x.sum() - target,
    },)

    def obj(x):
        return _evaluate(x)

    res = optimize.minimize(
        obj,
        start,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 200, "ftol": 1e-9, "disp": False},
    )
    if not res.success:
        return start
    h = _project_to_feasible(res.x)
    return h

def _upsample(h):
    """Double the resolution using linear interpolation (periodic on [0,2])."""
    n = h.size
    new_n = n * 2
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    h_new = _project_to_feasible(h_new)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """Return a feasible step function, its C5 bound, and the sample count."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise with the provided parent or a random feasible vector.
    # ------------------------------------------------------------------
    if "initial_h_values" in globals() and isinstance(initial_h_values, np.ndarray):
        h_curr = _project_to_feasible(initial_h_values.astype(float))
    else:
        n0 = 45
        h_curr = _random_feasible(n0, rng)

    best_h = h_curr.copy()
    best_c = _evaluate(best_h)

    # ------------------------------------------------------------------
    # 2. Main optimization loop – random search + occasional local refinement.
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.85:
            break

        # generate a candidate
        cand = _random_feasible(best_h.size, rng)
        cand_c = _evaluate(cand)

        if cand_c < best_c:
            best_h, best_c = cand, cand_c

        # every few seconds try a local refinement from the current best
        if int(elapsed) % 30 == 0:
            refined = _local_optimize(best_h)
            refined_c = _evaluate(refined)
            if refined_c < best_c:
                best_h, best_c = refined, refined_c

    # ------------------------------------------------------------------
    # 3. Attempt a resolution increase if time permits.
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        # double resolution
        h_high = _upsample(best_h)
        n_high = h_high.size
        if n_high > 2000:          # keep size modest for speed
            break

        # random search at higher resolution (short budget)
        local_best = h_high.copy()
        local_c = _evaluate(local_best)
        limit = 2000  # max evaluations at this level
        for _ in range(limit):
            cand = _random_feasible(n_high, rng)
            c = _evaluate(cand)
            if c < local_c:
                local_best, local_c = cand, c

        # one refinement step
        refined = _local_optimize(local_best)
        refined_c = _evaluate(refined)
        if refined_c < local_c:
            refined, refined_c = refined, refined_c

        # accept if improvement over previous best
        if refined_c < best_c:
            best_h, best_c = refined, refined_c
        else:
            break   # no progress → stop increasing resolution

    # ------------------------------------------------------------------
    # 4. Return the final solution.
    # ------------------------------------------------------------------
    return best_h, float(best_c), int(best_h.size)
