import scipy.optimize

# initial_h_values – optional starting vector


def _objective(x):
    """Objective for the optimizer: the C5 bound."""
    # evaluate_erdos_solution may return a tuple, we need the first entry
    return evaluate_erdos_solution(x)[0]


def _feasible_projection(vec, target_sum):
    """
    Project a vector onto the hyperplane sum(vec)=target_sum while keeping
    0≤vec≤1. Uses a simple pair‑wise redistribution that preserves feasibility.
    """
    # Clip first
    vec = np.clip(vec, 0.0, 1.0)
    cur = np.sum(vec)
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return vec
    # Distribute the difference by moving mass between random pairs
    n = len(vec)
    rng = np.random.default_rng()
    for _ in range(10 * n):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount we could move from i to j while staying in bounds
        max_move = min(vec[i], 1.0 - vec[j])
        min_move = -min(vec[j], 1.0 - vec[i])
        # Desired move amount to fix the total difference (distributed equally)
        move = diff / n
        move = np.clip(move, min_move, max_move)
        if abs(move) < 1e-12:
            continue
        vec[i] -= move
        vec[j] += move
        diff -= (move - (-move))  # net change is zero, only sum unchanged
        if abs(diff) < 1e-12:
            break
    # final safety clip
    return np.clip(vec, 0.0, 1.0)


def _random_feasible(n, target_sum):
    """Generate a random feasible h vector (0‑1 bounds, fixed sum)."""
    rng = np.random.default_rng()
    while True:
        v = rng.random(n)
        v = v / v.sum() * target_sum
        if np.all(v <= 1.0):
            return v


def _local_optimize(start_h, target_sum, time_deadline):
    """
    Perform a short SLSQP refinement starting from start_h.
    Stops early if the time budget is exceeded.
    """
    n = len(start_h)
    bounds = [(0.0, 1.0)] * n
    constraints = ({
        "type": "eq",
        "fun": lambda x: np.sum(x) - target_sum,
    },)

    options = {
        "maxiter": 200,
        "ftol": 1e-8,
        "disp": False,
    }

    # wrap objective to respect the deadline
    def timed_obj(x):
        if time.time() > time_deadline:
            raise TimeoutError
        return _objective(x)

    try:
        res = scipy.optimize.minimize(
            timed_obj,
            x0=start_h,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options=options,
        )
        if not res.success:
            # fall back to the start point
            return start_h, _objective(start_h)
        return res.x, res.fun
    except TimeoutError:
        # budget ran out during this call
        return start_h, _objective(start_h)
