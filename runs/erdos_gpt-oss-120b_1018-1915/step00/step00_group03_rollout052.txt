import numpy as np
import time

def _evaluate_c5(h):
    """Return the C5 value for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_bounds_sum(h, target_sum, tol=1e-9, max_iter=100):
    """
    Project vector h onto the box [0,1] and the affine hyperplane sum(h)=target_sum.
    Simple iterative scheme that respects bounds.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        # indices that can still move without hitting a bound
        free = np.where((h > 0.0 + 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0:
            # all at bounds, distribute remaining diff uniformly (may violate bounds)
            # fallback: distribute proportionally to capacity
            caps_up = 1.0 - h
            caps_down = h
            if diff > 0:
                cap = caps_up.sum()
                if cap > 0:
                    h += diff * caps_up / cap
            else:
                cap = caps_down.sum()
                if cap > 0:
                    h += diff * caps_down / cap
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final clamp for safety
    h = np.clip(h, 0.0, 1.0)

            h[i] = np.clip(h[i] + final_diff, 0.0, 1.0)
    return h
