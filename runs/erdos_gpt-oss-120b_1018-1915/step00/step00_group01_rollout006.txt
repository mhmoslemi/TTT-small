import time
import numpy as np
from math import inf

def _c5_bound(h):
    """Return the C5 bound for a step vector h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _random_neighbor(h, step, rng):
    """Propose a neighbour keeping sum and bounds by moving mass between two indices."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # max delta respecting bounds
    max_up = min(step, 1.0 - h[i])
    max_down = min(step, h[j])
    # choose delta sign: we will add to i, subtract from j
    delta = rng.uniform(-max_up, max_down)
    # ensure resulting values stay in [0,1]
    if not (0.0 <= h[i] + delta <= 1.0 and 0.0 <= h[j] - delta <= 1.0):
        return None, None, None
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, i, j
