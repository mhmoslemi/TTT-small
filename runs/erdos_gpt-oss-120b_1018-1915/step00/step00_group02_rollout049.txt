import time
import numpy as np
from scipy import optimize as opt

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def compute_c5(h):
    """Return C5 value for discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def upsample_half(h):
    """Double resolution while preserving integral: repeat each entry twice and halve."""
    return np.repeat(h, 2) * 0.5


def random_perturb(h, scale=0.12, rng=None):
    """Add Gaussian noise and clip to [0,1]."""
    if rng is None:
        rng = np.random.default_rng()
    pert = h + scale * rng.normal(size=h.shape)
    return np.clip(pert, 0.0, 1.0)


def optimise_once(h0, max_iter=250, ftol=1e-7):
    """Run SLSQP from initial point h0 respecting bounds and integral constraint."""
    n = h0.shape[0]
    target = n / 2.0

    def obj(x):
        return compute_c5(x)

    cons = {"type": "eq", "fun": lambda x: np.sum(x) - target}
    bounds = [(0.0, 1.0)] * n

    res = opt.minimize(
        obj,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": max_iter, "ftol": ftol, "disp": False},
    )
    return res.x, res.fun, res.success


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial construction
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_initial = np.asarray(initial_h_values, dtype=np.float64)
    else:
        # fallback: half the interval set to 1, half to 0 (integral = n/2)
        n_fallback = 45
        h_initial = np.concatenate(
            [np.ones(n_fallback // 2), np.zeros(n_fallback - n_fallback // 2)]
        ).astype(np.float64)

    best_h = h_initial.copy()
    best_score = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Multi‑resolution search
    # ------------------------------------------------------------------
    max_levels = 5                     # up to 45 * 2^5 = 1440 points (still cheap)
    level = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:          # keep margin for final return
            break
        n = best_h.shape[0]
        # try several starts at this resolution
        candidates = [best_h]
        for _ in range(3):
            candidates.append(random_perturb(best_h, rng=rng))

        for cand in candidates:
            # quick feasibility correction (project onto sum constraint)
            # scale to hit target sum, then clip; SLSQP will enforce equality later.
            target = n / 2.0
            cur = np.sum(cand)
            if cur > 0:
                cand = cand * (target / cur)
            cand = np.clip(cand, 0.0, 1.0)

            h_opt, score, ok = optimise_once(cand)
            if ok and score < best_score:
                best_h, best_score = h_opt, score

        # prepare next resolution if time permits
        level += 1
        if level >= max_levels:
            break
        # upsample current best for next level
        best_h = upsample_half(best_h)

    # ------------------------------------------------------------------
    # Final result
    # ------------------------------------------------------------------
    final_h = best_h.astype(np.float64)
    final_score = compute_c5(final_h)
    return final_h, final_score, final_h.shape[0]
