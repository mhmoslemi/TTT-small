import time
import numpy as np

def _evaluate_c5(h, dx):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated by discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _project(h, target_sum):
    """Project h onto [0,1] box and enforce sum == target_sum (up to tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    for _ in range(15):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # final tiny adjustment if still off (distribution may violate bounds)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-10:
        # distribute remaining diff proportionally within free slack
        slack_low = h - 0.0
        slack_high = 1.0 - h
        slack = slack_low + slack_high
        total_slack = slack.sum()
        if total_slack > 0:
            h += diff * (slack / total_slack)
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback uniform correction
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def _random_neighbor(best, step_scale, rng):
    """Generate a random neighbor of best by Gaussian perturbation."""
    cand = best + step_scale * rng.standard_normal(best.shape)
    return cand
