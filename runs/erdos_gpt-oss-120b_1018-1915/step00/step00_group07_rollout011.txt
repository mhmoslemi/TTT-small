import time
import math
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _initial_vector(n, seed):
    """Create a feasible initial vector (binary, sum = n/2)."""
    rng = np.random.default_rng(seed)
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    rng.shuffle(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 bound.

    Returns
    -------
    h_values : list of float
        Optimized samples (length n_points).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of sampled points (length of h_values).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Prepare initial feasible vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values, dtype=float).copy()
        n = h_current.size
        # enforce exact sum constraint (may be slightly off due to float)
        target_sum = n / 2.0
        current_sum = h_current.sum()
        if abs(current_sum - target_sum) > 1e-9:
            # simple scaling + clipping repair
            h_current *= target_sum / current_sum
            np.clip(h_current, 0.0, 1.0, out=h_current)
            # final re‑scale to hit the exact sum if needed
            excess = h_current.sum() - target_sum
            if abs(excess) > 1e-12:
                # adjust a random non‑boundary element
                idx = rng.choice(np.where((h_current > 0) & (h_current < 1))[0])
                h_current[idx] -= excess
                h_current[idx] = np.clip(h_current[idx], 0.0, 1.0)
    else:
        n = 80
        h_current = _initial_vector(n, seed)

    dx = 2.0 / n
    best_h = h_current.copy()
    best_score = _compute_c5(best_h, dx)

    # ------------------------------------------------------------------
    # Simulated‑annealing style local search
    # ------------------------------------------------------------------
    max_step = 0.05          # maximal perturbation magnitude
    temp_start = 0.02
    temp_end = 0.0001
    total_iters = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:   # leave a small safety margin
            break

        # temperature linear schedule
        t = elapsed / budget_s
        temperature = temp_start * (1 - t) + temp_end * t

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)

        hi, hj = best_h[i], best_h[j]

        # feasible delta range respecting bounds
        delta_min = max(-max_step, -hj)            # cannot go below 0 for j

        h_new[i] = hi + delta
        h_new[j] = hj - delta

        # evaluate
        new_score = _compute_c5(h_new, dx)

        accept = False
