import time, random, math
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a step function h (as a list), its C5 bound, and the number of points.
    """
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size
    n = len(initial_h_values) if 'initial_h_values' in globals() else 80
    target_sum = n / 2.0          # integral constraint (dx = 2/n)
    dx = 2.0 / n

    # --- projection onto [0,1]^n with exact sum ------------------------------
    def project(v):
        """Clip to [0,1] and iteratively enforce the sum constraint."""
        v = np.clip(v, 0.0, 1.0)
        for _ in range(15):
            s = v.sum()
            diff = target_sum - s
            if abs(diff) < 1e-12:
                break
            free = (v > 0) & (v < 1)
            cnt = free.sum()
            if cnt == 0:               # all at bounds – distribute uniformly
                v += diff / n
            else:
                v[free] += diff / cnt
            v = np.clip(v, 0.0, 1.0)
        return v

    # --- final exact‑sum adjustment ----------------------------------------
    def adjust_exact(v):
        """If rounding errors remain, fix the sum by a single safe tweak."""
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            return v
        for i in range(len(v)):
            if 0.0 <= v[i] + diff <= 1.0:
                v[i] += diff
                return v
        # fallback: tiny uniform redistribution then re‑project
        v = v + diff / len(v)
        return project(v)

    # --- objective evaluation ------------------------------------------------
    def evaluate(h_vec):
        """C5 value for a feasible h."""
        corr = np.correlate(h_vec, 1.0 - h_vec, mode='full')
        return np.max(corr) * dx

    # --- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        best = project(np.asarray(initial_h_values, dtype=float))
    else:
        best = np.full(n, target_sum / n)
    best_score = evaluate(best)

    start_time = time.time()
    sigma = 0.12                     # initial perturbation magnitude
    iter_count = 0

    # ------------------------------------------------------------------------
    # Simple stochastic local search with occasional random restarts.
    # ------------------------------------------------------------------------
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break

        # candidate generation
        if random.random() < 0.04:               # random restart
            cand = np.random.rand(n)
        else:
            cand = best + sigma * np.random.randn(n)

        cand = project(cand)
        cand_score = evaluate(cand)

        # accept improvement
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            sigma = max(0.02, sigma * 0.94)      # finer search
        else:
            # occasional exploration boost
            if random.random() < 0.01:
                sigma = min(0.4, sigma * 1.07)

        iter_count += 1

    # Final sanity projection
    best = adjust_exact(best)
    final_score = evaluate(best)

    return best.tolist(), final_score, n
