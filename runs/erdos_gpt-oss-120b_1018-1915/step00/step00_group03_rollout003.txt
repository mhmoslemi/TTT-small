def _project_to_constraints(vec):
    """Clip to [0,1] and enforce sum(vec) = len(vec)/2."""
    n = vec.size
    target = n / 2.0
    vec = np.clip(vec, 0.0, 1.0)
    s = vec.sum()
    if abs(s - target) < 1e-12:
        return vec
    # iterative redistribution among non‑boundary entries
    for _ in range(12):
        diff = target - s
        free = np.where((vec > 0.0) & (vec < 1.0))[0]
        if free.size == 0:
            break
        delta = diff / free.size
        vec[free] += delta
        vec = np.clip(vec, 0.0, 1.0)
        s = vec.sum()
        if abs(s - target) < 1e-12:
            break
    # final scaling if needed (rare)
    if abs(s - target) > 1e-8:
        vec *= target / s
        vec = np.clip(vec, 0.0, 1.0)
    return vec


def _c5_of(h):
    """Return the overlap constant for a discretised h."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)
