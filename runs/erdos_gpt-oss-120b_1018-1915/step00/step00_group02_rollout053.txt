import numpy as np

def _adjust_sum(arr: np.ndarray, target: float) -> np.ndarray:
    """Project arr onto the simplex with sum=target while respecting 0‑1 bounds."""
    n = arr.size
    for _ in range(5):
        diff = arr.sum() - target
        if abs(diff) < 1e-12:
            break
        arr -= diff / n
        np.clip(arr, 0.0, 1.0, out=arr)
    # tiny residual correction
    diff = arr.sum() - target
    if abs(diff) > 1e-9:
        free = (arr > 0.0) & (arr < 1.0)
        cnt = int(free.sum())
        if cnt:
            arr[free] -= diff / cnt
            np.clip(arr, 0.0, 1.0, out=arr)
    return arr

def _c5_score(h: np.ndarray) -> float:
    """Return the overlap constant for a discretised step function."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())
