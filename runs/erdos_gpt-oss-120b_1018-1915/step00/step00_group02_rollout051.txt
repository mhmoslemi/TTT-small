import time
import numpy as np
from scipy.optimize import minimize

def _c5_objective(h, dx):
    """Return the max overlap for step function h (dx is grid spacing)."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _optimize_at_resolution(h0, dx, time_deadline, max_iter=300):
    """Run a short SLSQP optimization from start h0 within the remaining time."""
    n = h0.size
    target_sum = n / 2.0

    bounds = [(0.0, 1.0) for _ in range(n)]
    constraints = ({
        "type": "eq",
        "fun": lambda h: np.sum(h) - target_sum,
    },)

    # wrapper that also checks the time budget
    def obj_wrap(h):
        return _c5_objective(h, dx)

    options = {"maxiter": max_iter, "ftol": 1e-8, "disp": False}

    res = minimize(
        obj_wrap,
        h0,
        method="SLSQP",
        bounds=bounds,
        constraints=constraints,
        options=options,
    )
    # If optimizer stopped early because of time, we still have a feasible point
    return res.x, _c5_objective(res.x, dx)

def _double_resolution(h):
    """Create a twice‑as‑fine representation preserving the integral."""
    # Linear interpolation to 2n points
    n = h.size
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, 2 * n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # After interpolation the total sum may have changed; renormalise exactly.
    target_sum = (2 * n) / 2.0
    diff = target_sum - h_new.sum()
    # Distribute the difference uniformly, then clip back into bounds.
    h_new += diff / (2 * n)
    np.clip(h_new, 0.0, 1.0, out=h_new)
    # Re‑adjust to satisfy the equality constraint exactly (simple projection)
    excess = h_new.sum() - target_sum
    if abs(excess) > 1e-12:
        # shift uniformly on free indices
        free = np.where((h_new > 0.0) & (h_new < 1.0))[0]
        if free.size == 0:
            # fallback: scale the whole vector
            h_new *= target_sum / h_new.sum()
        else:
            h_new[free] -= excess / free.size
            np.clip(h_new, 0.0, 1.0, out=h_new)
    return h_new
