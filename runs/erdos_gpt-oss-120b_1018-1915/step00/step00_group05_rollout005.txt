import numpy as np
import time
import random

# ------------------------------------------------------------
# Helper: evaluate C5 for a discrete step function h
# ------------------------------------------------------------
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# ------------------------------------------------------------
# Projection onto the box [0,1] with a prescribed sum
# Finds w = clip(v - lambda, 0, 1) such that sum(w) = target
# ------------------------------------------------------------
def project_onto_box_sum(v, target, eps=1e-9, max_iter=60):
    n = v.size
    # If the simple clipping already satisfies the sum, return it
    w = np.clip(v, 0.0, 1.0)
    s = w.sum()
    if abs(s - target) <= eps:
        return w
    # Binary search on lambda
    lo, hi = -1.0, 2.0  # bounds that surely contain the solution
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < eps:
            break
    return w

# ------------------------------------------------------------
# Simple stochastic local search with simulated annealing
# ------------------------------------------------------------
def stochastic_search(init_h, budget_s, seed=42):
    random.seed(seed)
    np.random.seed(seed)
    n = init_h.size
    target_sum = n / 2.0
    best = init_h.copy()
    best_score = compute_c5(best)

    start_time = time.time()
    # annealing schedule parameters
    T0 = 0.1
    Tf = 0.001
    max_iter = 200000  # hard cap; loop will stop earlier due to time

    for it in range(1, max_iter + 1):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        # temperature decay
        T = T0 * (Tf / T0) ** (it / max_iter)

        # propose a perturbation
        cand = best.copy()
        # pick a random subset size (1 to 5) to modify
        k = random.randint(1, min(5, n))
        idx = np.random.choice(n, k, replace=False)
        # add Gaussian noise
        noise = np.random.normal(scale=0.2, size=k)
        cand[idx] += noise
        # project back to feasible region
        cand = project_onto_box_sum(cand, target_sum)

        cand_score = compute_c5(cand)

        # accept if better, else maybe accept with prob exp(-(Δ)/T)
        if cand_score < best_score or random.random() < np.exp(-(cand_score - best_score) / max(T, 1e-12)):
            best, best_score = cand, cand_score

    return best, best_score

# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
        h_values : 1‑D numpy array of length n_points, values in [0,1]
        c5_bound : the obtained max overlap integral C5
        n_points : number of discretisation points (len(h_values))
    """
    # Determine problem size
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = kwargs.get('n_points', 70)
        # simple feasible start: half of the entries are 1, rest 0, uniformly spread
        h0 = np.zeros(n, dtype=float)
        idx = np.arange(n)
        h0[idx % 2 == 0] = 1.0
        # enforce exact sum
        target_sum = n / 2.0
        h0 = project_onto_box_sum(h0, target_sum)

    # Ensure feasibility of the initial vector
    target_sum = n / 2.0
    h0 = project_onto_box_sum(h0, target_sum)

    # Allocate a fraction of the budget for the search
    search_budget = budget_s * 0.9

    best_h, best_c5 = stochastic_search(h0, search_budget, seed=seed)

    if not (0.0 <= best_h).all() or not (best_h <= 1.0).all() or abs(best_h.sum() - target_sum) > 1e-6:
        best_h = h0
