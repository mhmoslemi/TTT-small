import time
import random
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize the
    Erdős overlap constant C5.

    Returns:
        (h_values, c5_bound, n_points)
    """
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    n_points = (
        len(initial_h_values)
        if 'initial_h_values' in globals()
        else 80
    )
    dx = 2.0 / n_points
    target_sum = n_points / 2.0        # because ∫ h = 1  →  sum(h)*dx = 1

    # ---------- Projection onto box [0,1] with sum constraint ----------
    def project(v):
        """Project vector v onto {0≤x_i≤1, sum x_i = target_sum}."""
        lo, hi = -1.0, 1.0                      # bounds for Lagrange multiplier
        for _ in range(60):                     # sufficient for double precision
            mid = (lo + hi) * 0.5
            w = np.clip(v + mid, 0.0, 1.0)
            s = w.sum()
            if s > target_sum:
                hi = mid
            else:
                lo = mid
        return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    # ---------- Objective evaluation ----------
    def evaluate(h):
        """Compute C5 for a feasible h."""
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # ---------- Initial population ----------
    pop_size = 30
    population = []
    scores = []

    # include provided initial solution if present
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = project(h0)
        population.append(h0)
        scores.append(evaluate(h0))

    # fill the rest with random feasible vectors
    while len(population) < pop_size:
        v = np.random.rand(n_points)
        v = project(v)
        population.append(v)
        scores.append(evaluate(v))

    # current best
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # ---------- Differential Evolution ----------
    F = 0.5          # mutation factor
    CR = 0.9         # crossover probability
    start_time = time.time()
    generation = 0
    max_generations = 2000

    while time.time() - start_time < budget_s - 5 and generation < max_generations:
        generation += 1
        for i in range(pop_size):
            # mutation: select three distinct individuals different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = random.sample(idxs, 3)
            mutant = population[r1] + F * (population[r2] - population[r3])

            # binomial crossover
            trial = np.empty_like(mutant)
            jrand = random.randrange(n_points)
            for j in range(n_points):
                if random.random() < CR or j == jrand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]

            trial = project(trial)
            trial_score = evaluate(trial)

            # selection
            if trial_score <= scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()

    return (best_h, float(best_score), n_points)
