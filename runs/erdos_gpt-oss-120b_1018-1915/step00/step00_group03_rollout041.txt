import time
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral for step function h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def enforce_constraints(h, target_sum):
    """Project h onto [0,1] and adjust to exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0.0:
        h[:] = target_sum / h.size
        return h
    scale = target_sum / cur
    h *= scale
    h = np.clip(h, 0.0, 1.0)

    # fine‑tune remaining discrepancy
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            idx = np.argmin(h)  # pick smallest entry
        else:
            idx = free[0]
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def improve(h, budget, start_step=0.3):
    """Simple stochastic hill‑climbing within given time budget."""
    n = h.size
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    step = start_step
    t0 = time.time()
    while time.time() - t0 < budget:
        cand = best_h + step * np.random.randn(n)
        cand = enforce_constraints(cand, target)
        c5 = compute_c5(cand, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand
        # gradual cooling
        step *= 0.9995
        # occasional random restart to avoid stagnation
        if np.random.rand() < 0.001:
            cand = np.random.rand(n)
            cand = enforce_constraints(cand, target)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    np.random.seed(seed)
    start_all = time.time()

    # --- initial feasible vector -----------------------------------------
    if "initial_h_values" in globals():
        base_h = np.array(initial_h_values, dtype=float)
    else:
        base_n = 64
        base_h = np.full(base_n, 0.5)
    base_h = enforce_constraints(base_h, base_h.size / 2.0)

    best_h = base_h
    best_c5 = compute_c5(best_h, 2.0 / best_h.size)

    # --- multiresolution schedule (up to ~512 points) --------------------
    ns = [best_h.size]
    while ns[-1] < 512:
        ns.append(ns[-1] * 2)

    for level, n in enumerate(ns):
        elapsed = time.time() - start_all
        remaining = budget_s - elapsed
        if remaining <= 0:
            break

        # allocate a fair share of the remaining time to this level
        level_budget = remaining / (len(ns) - level)

        # upscale if needed
        if n != best_h.size:
            factor = n // best_h.size
            h = np.repeat(best_h, factor)
            h = enforce_constraints(h, n / 2.0)
        else:
            h = best_h.copy()

        h, c5 = improve(h, level_budget, start_step=0.3)
