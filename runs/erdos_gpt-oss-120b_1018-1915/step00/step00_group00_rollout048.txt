import time
import numpy as np

def _compute_c5(h, dx):
    """Return C5 = max correlation * dx for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _feasible_mutation(h, rng):
    """Return a copy of h after a small feasible two‑coordinate move."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    hi, hj = h[i], h[j]

    # allowable increase for i (decrease for j) and opposite direction
    max_pos = min(1.0 - hi, hj)          # delta > 0 (i up, j down)
    max_neg = min(hi, 1.0 - hj)          # delta < 0 (i down, j up)

    if max_pos + max_neg == 0.0:
        return h.copy()                 # no move possible

    delta = rng.random() * (max_pos + max_neg) - max_neg
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

        h_curr += (target_sum - cur_sum) / n

    best_h = h_curr.copy()

    cur_val = best_val

    # ----- simulated annealing parameters -----
