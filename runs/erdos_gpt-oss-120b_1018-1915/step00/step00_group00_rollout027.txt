import time
import math
import numpy as np

def _calc_c5(h, dx):
    """Return max overlap integral for step values h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _initial_binary(n):
    """Return a feasible binary‑like start: 1's, 0's and one 0.5 if needed."""
    target = n / 2.0
    h = np.zeros(n, dtype=np.float64)
    half = int(target)               # floor(target)
    h[:half] = 1.0
    remainder = target - half        # 0 or 0.5 for odd n
    if remainder > 0:
        h[half] = remainder
    return h

    where h_values is a list of length n_points satisfying the constraints.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    except NameError:

        h_curr = _initial_binary(n_points)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ensure feasibility of the starting vector

    cur_sum = h_curr.sum()
    if abs(cur_sum - target_sum) > 1e-12:

        diff = target_sum - cur_sum

        cur_sum = h_curr.sum()

    # annealing parameters
