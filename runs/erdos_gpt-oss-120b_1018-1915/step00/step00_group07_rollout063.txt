import time
import numpy as np

def _project_to_capped_simplex(v, target_sum, cap=1.0, iters=5):
    """Return a vector in [0,cap] with sum == target_sum, close to v."""
    x = np.clip(v, 0.0, cap)
    for _ in range(iters):
        total = x.sum()
        if abs(total - target_sum) < 1e-10:
            break
        if total > target_sum:
            over = total - target_sum
            mask = x > 0
            if mask.sum() == 0:
                break
            x[mask] -= over * x[mask] / x[mask].sum()
        else:
            under = target_sum - total
            mask = x < cap
            if mask.sum() == 0:
                break
            x[mask] += under * (cap - x[mask]) / (cap - x[mask]).sum()
        x = np.clip(x, 0.0, cap)
    return x

def _evaluate_candidate(h_vec, dx):
    """Maximum overlap C5 for a given step vector."""
    corr = np.correlate(h_vec, 1.0 - h_vec, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] giving a low C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : ndarray of shape (n_points,)
            The step values in [0,1] respecting the integral constraint.
        c5_bound : float
            The achieved C5 upper bound.
        n_points : int
            Number of discretisation points (length of h_values).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------------
    # Determine problem size (use provided initial_h_values if present)
    # -------------------------------------------------------------
    try:
        n_points = int(len(initial_h_values))
        base_vec = np.asarray(initial_h_values, dtype=float)
    except NameError:
        n_points = 80
        base_vec = rng.random(n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because integral must be 1

    # -------------------------------------------------------------
    # Helper: repair to feasible region
    # -------------------------------------------------------------
    def repair(vec):
        return _project_to_capped_simplex(vec, target_sum, cap=1.0)

    # -------------------------------------------------------------
    # Evaluation wrapper (includes repair)
    # -------------------------------------------------------------
    def evaluate(vec):
        h = repair(vec)
        return _evaluate_candidate(h, dx)

    # -------------------------------------------------------------
    # Initialise population
    # -------------------------------------------------------------
    pop_size = max(20, n_points // 2)          # modest size for speed
    population = np.empty((pop_size, n_points), dtype=float)

    # first individual = repaired initial (or random if none)
    population[0] = repair(base_vec)

    # remaining individuals = random feasible vectors
    for i in range(1, pop_size):
        rand_vec = rng.random(n_points)
        population[i] = repair(rand_vec)

    scores = np.array([evaluate(ind) for ind in population])
    best_idx = int(np.argmin(scores))
    best_vec = population[best_idx].copy()
    best_score = float(scores[best_idx])

    # -------------------------------------------------------------
    # Differential Evolution core loop
    # -------------------------------------------------------------
    F = 0.5          # mutation factor
    CR = 0.9         # crossover probability
    max_gen = 500   # safety cap; loop will break on time limit

    for gen in range(max_gen):
        # time‑check at start of each generation
        if time.time() - start_time > budget_s * 0.95:
            break

        for i in range(pop_size):
            # select three distinct individuals different from i
            idxs = rng.choice(pop_size, 3, replace=False)
            while i in idxs:
                idxs = rng.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs]

            mutant = a + F * (b - c)

            # binomial crossover
            cross_mask = rng.random(n_points) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(n_points)] = True
            trial = np.where(cross_mask, mutant, population[i])
            trial = repair(trial)

            trial_score = _evaluate_candidate(trial, dx)

            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = float(trial_score)
                    best_vec = trial.copy()

    # -------------------------------------------------------------
    # Final output (ensure feasibility one last time)
    # -------------------------------------------------------------
    final_h = repair(best_vec)
    final_score = _evaluate_candidate(final_h, dx)

    return final_h, final_score, n_points
