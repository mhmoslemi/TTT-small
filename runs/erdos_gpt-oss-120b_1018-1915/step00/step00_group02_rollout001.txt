import time
import numpy as np
from scipy.optimize import minimize

# helper: evaluate C5 for a given h vector
def _c5_of_h(h):
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

# project onto equality constraint sum(h)=n/2 while keeping bounds
def _project_sum(h, target_sum):
    # simple affine projection: shift by average delta, then clip
    delta = (target_sum - h.sum()) / h.size
    h = h + delta
    np.clip(h, 0.0, 1.0, out=h)
    # if still not exact, rescale the interior points proportionally
    cur_sum = h.sum()
    if not np.isclose(cur_sum, target_sum):
        # distribute remaining difference proportionally to distance from bounds
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            scale = (target_sum - cur_sum) / free.sum()
            h[free] += scale
            np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # starting resolution
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
    else:
        n0 = 45
        h0 = np.full(n0, 0.5)  # dummy, will be projected
        h0 = _project_sum(h0, n0 / 2)

    best_h = h0.copy()
    best_score = _c5_of_h(best_h)
    best_n = best_h.size

    # parameters
    max_levels = 4          # how many doublings
    max_iter_per_level = 500
    ftol = 1e-5

    for level in range(max_levels + 1):
        n = best_h.size
        target_sum = n / 2.0
        bounds = [(0.0, 1.0) for _ in range(n)]

        # equality constraint for sum
        cons = ({
            'type': 'eq',
            'fun': lambda x, ts=target_sum: np.sum(x) - ts,
        },)

        # objective wrapper for optimizer
        def obj(x):
            return _c5_of_h(x)

        # run SLSQP
        res = minimize(
            obj,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': max_iter_per_level, 'ftol': ftol, 'disp': False}
        )

        if res.success:
            cand_h = res.x
        else:
            cand_h = best_h  # fallback

        cand_score = _c5_of_h(cand_h)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
            best_n = n

        # time check
        if time.time() - start_time > budget_s * 0.95:
            break

        # prepare next resolution by doubling
        if level < max_levels:
            # upsample by repeating each point
            h_up = np.repeat(best_h, 2)
            # small random jitter to escape local minima
            jitter = rng.uniform(-0.02, 0.02, size=h_up.shape)
            h_up = np.clip(h_up + jitter, 0.0, 1.0)
            # project to exact sum (should already be correct but ensure)
            h_up = _project_sum(h_up, h_up.size / 2.0)
            best_h = h_up

    # final sanity: ensure constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum(best_h, best_h.size / 2.0)

    return best_h, float(best_score), best_h.size
