import time
import numpy as np
import scipy.optimize as opt

def _compute_c5(h, dx):
    """Return max overlap integral C5 for discrete h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _project_h(h, target_sum):
    """Clip to [0,1] and (approximately) enforce the linear sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        scale = target_sum / s if s != 0 else 0.0
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h on [0,2] to minimise the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimised samples of h (length n_points).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points (sample count).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Parameters
    # ------------------------------------------------------------------
    n_points = 84                     # chosen resolution (must be <1000)
    target_sum = n_points / 2.0       # integral constraint Σh = n/2
    dx = 2.0 / n_points

    # ------------------------------------------------------------------
    # Build initial guess
    # ------------------------------------------------------------------
    try:
        init_parent = np.asarray(initial_h_values, dtype=float)
        # Interpolate parent to the desired resolution
        x_old = np.linspace(0.0, 2.0, len(init_parent), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_old, init_parent)
    except NameError:
        # No parent supplied – generate a random feasible vector
        rand = rng.random(n_points)
        h0 = rand / rand.sum() * target_sum

    h0 = _project_h(h0, target_sum)

    # ------------------------------------------------------------------
    # Objective with penalty for violating the equality constraint
    # ------------------------------------------------------------------
    penalty_weight = 1e3

    def objective(x):
        # x is already within [0,1] due to bounds, but may deviate from sum
        c5 = _compute_c5(x, dx)
        pen = penalty_weight * abs(x.sum() - target_sum)
        return c5 + pen

    cons = ({
        "type": "eq",
        "fun": lambda x: x.sum() - target_sum,
    })

    bounds = [(0.0, 1.0)] * n_points

    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    # ------------------------------------------------------------------
    # Primary local optimisation (SLSQP)
    # ------------------------------------------------------------------
    opt_res = opt.minimize(
        objective,
        best_h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 500, "ftol": 1e-9, "disp": False},
    )
    if opt_res.success:
        cand = _project_h(opt_res.x, target_sum)
        cand_c5 = _compute_c5(cand, dx)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5

    # ------------------------------------------------------------------
    # Random‑restart SLSQP loop until time expires
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.95:

        r = r / r.sum() * target_sum

            objective,
            r,
            method="SLSQP",

    final_c5 = _compute_c5(best_h, dx)
