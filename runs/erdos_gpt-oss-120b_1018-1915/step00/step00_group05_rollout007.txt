import numpy as np
import time

# Helper: project a vector onto [0,1] with prescribed sum using binary search
def _project_sum_bounds(v, target_sum, tol=1e-9, max_iter=60):
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _fitness(h, dx):
    """Maximum overlap integral for a given step function h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full") * dx
    return np.max(corr)


def _initialize_population(pop_size, n, target_sum, rng):
    pop = []
    for _ in range(pop_size):
        vec = rng.random(n)
        vec = _project_sum_bounds(vec, target_sum)
        pop.append(vec)
    return np.array(pop)


def _perturb(vec, rng, sigma=0.1):
    """Add Gaussian noise then project back to feasible set."""
    noisy = vec + rng.normal(scale=sigma, size=vec.shape)
    return _project_sum_bounds(noisy, np.sum(vec))
