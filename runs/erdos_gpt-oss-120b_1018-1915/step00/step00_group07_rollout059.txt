import time
import numpy as np

def _repair(vec, target_sum, n):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    v = np.clip(vec, 0.0, 1.0)
    total = v.sum()
    if total == 0.0:
        v[:] = target_sum / n
        v = np.clip(v, 0, 1)
        total = v.sum()
    diff = target_sum - total
    # simple redistribution
    v += diff / n
    v = np.clip(v, 0, 1)
    # iterative refinement (few passes)
    for _ in range(5):
        total = v.sum()
        if abs(total - target_sum) < 1e-12:
            break
        diff = target_sum - total
        mask = (v > 0) & (v < 1)
        cnt = mask.sum()
        if cnt == 0:
            break
        v[mask] += diff / cnt
        v = np.clip(v, 0, 1)
    return v

def _evaluate(vec, dx):
    """Return the max overlap integral for a given step vector."""
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Differential‑Evolution search for a step function h on [0,2]
    improving the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem dimensions
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        # fallback: 80 uniform points
        n_points = 80
        h0 = np.full(n_points, 0.5)   # satisfies integral automatically
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ------------------------------------------------------------------
    # DE parameters
    # ------------------------------------------------------------------
    pop_size = max(20, n_points // 4)          # modest population
    F = 0.5                                     # mutation factor
    CR = 0.9                                    # crossover rate
    max_generations = 5000                     # generous upper bound, will be cut by time

    # ------------------------------------------------------------------
    # initialise population
    # ------------------------------------------------------------------
    population = np.empty((pop_size, n_points), dtype=float)
    population[0] = _repair(h0, target_sum, n_points)   # keep the provided seed
    for i in range(1, pop_size):
        rnd = rng.random(n_points)
        population[i] = _repair(rnd, target_sum, n_points)

    scores = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        scores[i] = _evaluate(population[i], dx)

    best_idx = np.argmin(scores)
    best_vec = population[best_idx].copy()
    best_score = scores[best_idx]

    # ------------------------------------------------------------------
    # main DE loop
    # ------------------------------------------------------------------
    gen = 0
    while gen < max_generations and (time.time() - start_time) < budget_s:
        for i in range(pop_size):
            # select three distinct others
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])

            # binomial crossover
            cross = rng.random(n_points) < CR
            if not np.any(cross):
                cross[rng.integers(0, n_points)] = True
            child = np.where(cross, mutant, population[i])

            # repair to feasibility
            child = _repair(child, target_sum, n_points)

            # evaluate
            child_score = _evaluate(child, dx)

            # selection
            if child_score < scores[i]:
                population[i] = child
                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_vec = child.copy()
        gen += 1

        # early exit if time exhausted
        if (time.time() - start_time) >= budget_s:
            break

    # ------------------------------------------------------------------
    # return results
    # ------------------------------------------------------------------
    return (best_vec.tolist(), float(best_score), n_points)
