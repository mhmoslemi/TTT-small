import numpy as np
import time
import math

def _evaluate_c5(h):
    """Return C5 for step function h on [0,2] with uniform spacing."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _make_binary_feasible(arr):
    """Convert array to a binary (0/1) vector with exactly n/2 ones."""
    n = arr.shape[0]
    half = n // 2
    # indices of largest values become 1
    idx = np.argsort(arr)[-half:]
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx] = 1.0
    return bin_vec

def _initial_solution(n):
    """Create an initial feasible binary vector."""
    # try to use provided initial_h_values if present and compatible
    try:
        init = initial_h_values
        if init.shape[0] == n:
            return _make_binary_feasible(init)
    except NameError:
        pass
    # otherwise random binary
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] minimizing the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # Determine problem size
    try:
        n0 = int(initial_h_values.shape[0])
    except NameError:
        n0 = 128                     # fallback size (power of two)
    # enforce even size for binary construction
    if n0 % 2 != 0:
        n0 += 1

    # start with the base resolution
    n = n0
    h = _initial_solution(n)
    best_h = h.copy()
    best_score = _evaluate_c5(best_h)

    ones_pos = np.flatnonzero(best_h == 1.0)
    zeros_pos = np.flatnonzero(best_h == 0.0)

    # parameters for stochastic hill‑climbing
    max_iter = 10_000_000            # safety upper bound
    iter_count = 0
    # temperature schedule (simulated annealing)
    T0 = 0.02
    decay = 0.9995

    while True:
        # budget check
        if time.time() - start_time > budget_s * 0.94:
            break
        if iter_count >= max_iter:
            break
        iter_count += 1

        # propose a simple swap of a 1 and a 0
        if len(ones_pos) == 0 or len(zeros_pos) == 0:
            # should never happen, but re‑initialise if it does
            h = _initial_solution(n)
            ones_pos = np.flatnonzero(h == 1.0)
            zeros_pos = np.flatnonzero(h == 0.0)
            continue

        i = np.random.choice(ones_pos)
        j = np.random.choice(zeros_pos)

        # perform swap
        h[i], h[j] = 0.0, 1.0

        T = T0 * (decay ** iter_count)
