import time
import numpy as np

def evaluate_c5(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return float(corr.max())

def proj_box_simplex(x, target):
    """Project onto {0<=y<=1, sum(y)=target} via bisection on lambda."""
    lo, hi = -1e5, 1e5
    for _ in range(50):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 70
        h0 = np.full(n, 0.5)  # placeholder; will be projected to satisfy sum
    dx = 2.0 / n
    target_sum = n / 2.0

    # ensure feasibility of the start
    h = proj_box_simplex(h0, target_sum)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)

    start_time = time.time()
    time_limit = budget_s * 0.95  # safety margin

    # schedule for step size decay
    step = 0.3
    decay_iter = max(1, n * 200)  # roughly 200 passes over vector

    iteration = 0
    while time.time() - start_time < time_limit:
        iteration += 1

        # random perturbation with zero sum
        delta = rng.normal(scale=step, size=n)
        # enforce zero-mean perturbation
        delta -= delta.mean()

        cand = h + delta
        cand = proj_box_simplex(cand, target_sum)

        c5 = evaluate_c5(cand, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()

        # occasionally accept worse solutions to escape local minima (simulated annealing)
        else:
            # simple temperature schedule
            T = max(0.001, (time_limit - (time.time() - start_time)) / time_limit)
            if rng.random() < np.exp(-(c5 - best_c5) / (T * 0.01)):
                h = cand
                continue

        # update current point
        h = cand

        # decay step size
        if iteration % decay_iter == 0:
            step *= 0.7
            step = max(step, 1e-4)

        # occasional global restart from a smoothed version of best_h

            smooth = np.convolve(best_h, kernel, mode='same')
            h = proj_box_simplex(smooth, target_sum)
            step = 0.3
