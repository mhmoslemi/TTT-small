import numpy as np, time, math, random
# evaluate_erdos_solution and initial_h_values are pre‑imported by the harness

def max_overlap(h):
    """Return C5 value for a given step function h (numpy 1‑D array)."""
    n = h.size
    dx = 2.0 / n
    # full cross‑correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def random_binary(n, ones):
    """Random binary vector of length n with exactly `ones` ones."""
    a = np.zeros(n, dtype=np.float64)
    a[:ones] = 1.0
    np.random.shuffle(a)
    return a

def neighbor_swap(h):
    """Produce a neighbour by swapping a 1 and a 0."""
    idx_one = np.where(h == 1.0)[0]
    idx_zero = np.where(h == 0.0)[0]
    i = np.random.choice(idx_one)
    j = np.random.choice(idx_zero)
    h_new = h.copy()
    h_new[i], h_new[j] = 0.0, 1.0
    return h_new

def hill_climb(initial_h, max_iters, time_limit, start_time):
    """Simple stochastic hill‑climbing limited by iteration count or time."""
    best_h = initial_h
    best_score = max_overlap(best_h)
    cur_h = best_h
    cur_score = best_score
    for _ in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        cand = neighbor_swap(cur_h)
        cand_score = max_overlap(cand)
        if cand_score < cur_score:
            cur_h, cur_score = cand, cand_score
            if cand_score < best_score:
                best_h, best_score = cand, cand_score
    return best_h, best_score

def upsample_initial(target_n):
    """Create an initial vector of length target_n using the provided
    initial_h_values (if any) by linear interpolation and then projecting
    to a binary vector with the correct number of ones."""
    if 'initial_h_values' not in globals():
        return None
    src = np.asarray(initial_h_values, dtype=np.float64)
    src_n = src.size
    # interpolate onto target grid
    x_src = np.linspace(0, 2, src_n, endpoint=False)
    x_tgt = np.linspace(0, 2, target_n, endpoint=False)
    interp = np.interp(x_tgt, x_src, src)
    # project to binary keeping exactly target_n//2 ones
    ones = target_n // 2
    idx = np.argpartition(-interp, ones)[:ones]
    h = np.zeros(target_n, dtype=np.float64)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # parameters --------------------------------------------------------------
    n_points = 200                     # resolution (adjustable)
    ones_needed = n_points // 2       # exactly half the interval must be 1
    max_total_iters = 200000          # safety cap
    time_per_restart = 4.0            # seconds allocated to each hill‑climb

    # best solution storage
    best_h = None
    best_score = math.inf

    # try to use the provided initial construction
    init = upsample_initial(n_points)
    if init is not None:
        h0, s0 = hill_climb(init, max_total_iters, time_per_restart, start_time)
        if s0 < best_score:
            best_h, best_score = h0, s0

    # stochastic restarts ------------------------------------------------------
    while time.time() - start_time < budget_s - 5.0:   # keep a small safety margin
        # generate a fresh random binary vector
        h_start = random_binary(n_points, ones_needed)
        # perform hill climbing
        rem_time = budget_s - (time.time() - start_time) - 1.0
        iters = int(min(max_total_iters, rem_time * 5000))   # crude scaling
        h_candidate, score_candidate = hill_climb(
            h_start, iters, rem_time, time.time()
        )
        if score_candidate < best_score:
            best_h, best_score = h_candidate, score_candidate
        # early exit if we already beat the target bound
        if best_score <= 0.3808:

    # final verification (ensure constraint exactly satisfied)
    # enforce exact sum constraint by fixing any rounding issues
    current_sum = best_h.sum()
    target_sum = ones_needed
    if current_sum != target_sum:
        # adjust by flipping the smallest number of entries

            flip = np.random.choice(ones, -diff, replace=False)
            best_h[flip] = 0.0

    return best_h, final_score, n_points
