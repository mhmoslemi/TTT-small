import numpy as np
import time
from scipy.optimize import minimize, Bounds

def _objective(h, dx):
    """Return max correlation * dx."""
    # ensure array type
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _sum_constraint(h, target_sum):
    """Equality constraint: sum(h) - target_sum == 0."""
    return np.sum(h) - target_sum

def _feasible_start(n, rng):
    """Generate a random feasible vector (0<=h_i<=1, sum = n/2)."""
    # start with uniform random numbers
    h = rng.random(n)
    # project to sum constraint by scaling, then clip iteratively
    target = n / 2.0
    # simple scaling; if clipping needed, adjust proportionally
    scale = target / np.sum(h)
    h *= scale
    # ensure within [0,1]
    h = np.clip(h, 0.0, 1.0)
    # if clipping altered the sum, redistribute the deficit/excess uniformly
    deficit = target - np.sum(h)
    if abs(deficit) > 1e-12:
        # distribute only to entries not at bounds
        free = (h > 0.0) & (h < 1.0)
        if np.any(free):
            h[free] += deficit / np.count_nonzero(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
