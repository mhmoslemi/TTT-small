import time
import numpy as np
from scipy.optimize import differential_evolution, minimize

def _c5_value(h, dx):
    """Compute C5 = max correlation of h and (1-h) times dx."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _objective_penalty(x, target_sum, dx, penalty_coeff=1e5):
    """C5 with a penalty for violating the sum constraint."""
    c5 = _c5_value(x, dx)
    sum_diff = np.sum(x) - target_sum
    return c5 + penalty_coeff * abs(sum_diff)

def _sum_constraint(x, target_sum):
    """Equality constraint for scipy.optimize (must be zero)."""
    return np.sum(x) - target_sum

def _local_refine(x0, target_sum, dx, time_limit):
    """Run a constrained local optimizer with a time guard."""
    bounds = [(0.0, 1.0)] * len(x0)
    cons = {"type": "eq", "fun": _sum_constraint, "args": (target_sum,)}
    opts = {"maxiter": 1000, "ftol": 1e-12, "disp": False}
    start = time.time()
    res = minimize(
        lambda v: _c5_value(v, dx),
        x0,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    # If we ran out of time, simply return the current best feasible point
    if time.time() - start > time_limit:
        return x0
    # Project onto feasible set (clip then adjust sum)
    proj = np.clip(res.x, 0.0, 1.0)
    diff = np.sum(proj) - target_sum
    if abs(diff) > 1e-12:
        # Simple proportional rescaling to fix the sum, staying in [0,1]
        scale = target_sum / np.sum(proj)
        proj = proj * scale
        proj = np.clip(proj, 0.0, 1.0)
        # If still off, distribute the remainder uniformly
        remainder = target_sum - np.sum(proj)
        if abs(remainder) > 1e-12:
            idx = np.where((proj > 0) & (proj < 1))[0]
            if idx.size:
                add = remainder / idx.size
                proj[idx] = np.clip(proj[idx] + add, 0.0, 1.0)
    return proj

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize C5.

    Returns:
        h_values : numpy.ndarray of shape (n_points,)
        c5_bound : float, the achieved C5 upper bound
        n_points : int, number of discretisation points
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    n_points = kwargs.get("n_points", None)
    if n_points is None:
        # Use the length of the supplied parent if it exists
        if "initial_h_values" in globals():
            n_points = len(initial_h_values)
        else:
            n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum(h)*dx = 1

    # Prepare initial candidate
    if "initial_h_values" in globals():
        h_current = np.array(initial_h_values, dtype=float)
        # Ensure feasibility by a quick projection
        h_current = np.clip(h_current, 0.0, 1.0)
        sum_diff = np.sum(h_current) - target_sum
        if abs(sum_diff) > 1e-12:
            h_current *= target_sum / np.sum(h_current)
            h_current = np.clip(h_current, 0.0, 1.0)
    else:
        # Random feasible start via normalization
        r = np.random.rand(n_points)
        h_current = r / np.sum(r) * target_sum
        h_current = np.clip(h_current, 0.0, 1.0)

    best_h = h_current.copy()
    best_c5 = _c5_value(best_h, dx)

    # ---------- Global search with Differential Evolution ----------
