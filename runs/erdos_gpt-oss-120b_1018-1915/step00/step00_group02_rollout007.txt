import time
import random
import numpy as np

# assume evaluate_erdos_solution is pre‑imported
# assume initial_h_values may be provided globally

def _initial_h():
    """Return initial feasible h array."""
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.asarray(init, dtype=float).copy()
    else:
        n = 45
        h = np.full(n, 0.5, dtype=float)  # sum = n/2 gives integral 1
    # enforce exact sum = n/2 (tiny numerical drift possible)
    n = h.size
    target_sum = n / 2.0
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # adjust first element within [0,1]
        adj = np.clip(h[0] + diff, 0.0, 1.0) - h[0]
        h[0] += adj
        diff -= adj
        if abs(diff) > 1e-12:
            # distribute remaining diff uniformly (should be negligible)
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    return h

def _repeat_twice(h):
    """Double resolution by repeating each entry."""
    return np.repeat(h, 2)

def _score(h):
    """Compute C5 bound using the provided evaluator."""
    return evaluate_erdos_solution(h)

def _local_search(h, time_limit, start_time):
    """Projected random transfer local search within given time budget."""
    n = h.size
    best_h = h.copy()
    best_val = _score(best_h)
    cur_h = h.copy()
    cur_val = best_val

    step = 0.05  # maximum transfer amount
    decay = 0.9995

    while time.time() - start_time < time_limit:
        # pick source (with >0) and sink (with <1)
        i, j = random.sample(range(n), 2)
        if cur_h[i] <= 0.0 or cur_h[j] >= 1.0:
            continue
        max_transfer = min(step, cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * random.random()
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        val = _score(new_h)
        if val < cur_val:
            cur_h, cur_val = new_h, val
            if val < best_val:
                best_h, best_val = new_h.copy(), val
        else:
            # occasional uphill move
            if random.random() < 0.001:
                cur_h, cur_val = new_h, val

        step *= decay  # slowly shrink step size
        if step < 1e-4:
            step = 0.05

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    # Phase 1: start at base resolution
    h0 = _initial_h()
    time_for_phase = budget_s * 0.6
    best_h, best_val = _local_search(h0, time_for_phase, start)

    # Phase 2: if time remains, try double resolution
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:  # need a small buffer
        h2 = _repeat_twice(best_h)
        # re‑project to exact sum after repetition (should already be correct)
        n2 = h2.size
        target_sum = n2 / 2.0
        diff = target_sum - h2.sum()
        if abs(diff) > 1e-12:
            h2[0] = np.clip(h2[0] + diff, 0.0, 1.0)

        best_h2, best_val2 = _local_search(h2, remaining, start)
        if best_val2 < best_val:
            best_h, best_val = best_h2, best_val2

    # final sanity check
    n_final = best_h.size
    # enforce feasibility one last time
    best_h = np.clip(best_h, 0.0, 1.0)
    target_sum = n_final / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # distribute correction uniformly, then clip again
        best_h += diff / n_final
        best_h = np.clip(best_h, 0.0, 1.0)
        # final renormalize if still off
        cur_sum = best_h.sum()
        if cur_sum != 0:
            best_h *= target_sum / cur_sum

    return best_h.tolist(), float(best_val), n_final
