import time
import numpy as np

def _evaluate(h, dx):
    """Return max overlap C5 for step function h."""
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _make_binary_feasible(h):
    """Force h to be binary 0/1 with exactly half the entries equal to 1."""
    n = h.size
    target_ones = n // 2
    # start from current rounding
    bin_h = (h >= 0.5).astype(float)
    current_ones = int(bin_h.sum())
    if current_ones > target_ones:
        # turn some ones to zero
        idx = np.where(bin_h == 1)[0]
        np.random.shuffle(idx)
        bin_h[idx[: current_ones - target_ones]] = 0.0
    elif current_ones < target_ones:
        # turn some zeros to one
        idx = np.where(bin_h == 0)[0]
        np.random.shuffle(idx)
        bin_h[idx[: target_ones - current_ones]] = 1.0
    return bin_h


def _random_binary(n):
    """Create a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : numpy.ndarray
        Optimized values of h on a uniform grid.
    c5_bound : float
        The achieved upper bound on C5.
    n_points : int
        Number of discretisation points used.
    """
    np.random.seed(seed)
    t_start = time.time()

    # --- Determine problem size ------------------------------------------------
    if "initial_h_values" in globals():
        init_h = np.asarray(initial_h_values, dtype=float)
        n = init_h.size
        # enforce feasibility (binary) for simplicity
        h_current = _make_binary_feasible(init_h)
    else:
        # default modest size, still below 1000
        n = 200  # even
        h_current = _random_binary(n)

    dx = 2.0 / n
    best_h = h_current.copy()
    best_score = _evaluate(best_h, dx)

    # --- Simple stochastic local search ----------------------------------------
    # Parameters
    max_iter = 10_000_000  # large cap, will be limited by time
    temperature_start = 0.1
    temperature_end = 1e-4

    # Pre‑compute indices of ones and zeros for fast swaps
    def _update_indices(arr):
        ones = np.where(arr == 1)[0]
        zeros = np.where(arr == 0)[0]
        return ones, zeros

    ones_idx, zeros_idx = _update_indices(h_current)

    iteration = 0
    while True:
        # time check
        if time.time() - t_start > budget_s - 0.5:
            break
        if iteration >= max_iter:
            break
        iteration += 1

        # temperature schedule (linear cooling)
        t_frac = iteration / max_iter
        temperature = temperature_start * (1 - t_frac) + temperature_end * t_frac

        # propose a swap between a 1 and a 0
        if ones_idx.size == 0 or zeros_idx.size == 0:
            # should not happen, but rebuild indices
            ones_idx, zeros_idx = _update_indices(h_current)
            continue
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        # perform swap
        h_candidate = h_current.copy()
        h_candidate[i], h_candidate[j] = 0.0, 1.0

        cand_score = _evaluate(h_candidate, dx)

        # accept if better or with Metropolis probability
        if cand_score < best_score or np.random.rand() < np.exp((best_score - cand_score) / max(temperature, 1e-12)):
            h_current = h_candidate
            # update index sets efficiently
            ones_idx = ones_idx[ones_idx != i]
            zeros_idx = zeros_idx[zeros_idx != j]
            ones_idx = np.append(ones_idx, j)
            zeros_idx = np.append(zeros_idx, i)

            if cand_score < best_score:
                best_score = cand_score
                best_h = h_candidate.copy()

    # Final evaluation (in case of early exit)
    final_score = _evaluate(best_h, dx)

    return best_h, final_score, n
