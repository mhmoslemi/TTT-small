import time
import math
import random
import numpy as np

# helpers are pre‑imported in the execution environment:
#   evaluate_erdos_solution(h)  – returns the C5 value for a given h
#   initial_h_values           – optional numpy array with a starter construction

def _initial_h(n):
    """Create a feasible random start (sum = n/2, values in [0,1])."""
    # start with uniform random numbers, then project to the simplex with box constraints
    h = np.random.rand(n)
    # enforce box constraints by simple clipping then scaling
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    target = n / 2.0
    if total == 0:
        # fallback to half ones, half zeros
        h[:n // 2] = 1.0
    else:
        h *= target / total
        # after scaling some values may exceed 1 due to rounding; fix by iterative projection
        for _ in range(5):
            excess = np.maximum(h - 1.0, 0.0).sum()
            deficit = np.maximum(-h, 0.0).sum()
            if excess == 0 and deficit == 0:
                break
            if excess > 0:
                over_idx = h > 1.0
                h[over_idx] = 1.0
                h[~over_idx] -= excess / (~over_idx).sum()
            if deficit > 0:
                under_idx = h < 0.0
                h[under_idx] = 0.0
                h[~under_idx] += deficit / (~under_idx).sum()
        # final renormalisation to exact sum
        h *= target / h.sum()
    return h

def _c5_value(h, dx):
    """Exact objective: max_k ∫ h(x)(1‑h(x+k)) dx, discretised."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

    Returns a tuple (h_values, c5_bound, n_points).

    - h_values : numpy array of length n_points, values in [0,1]

    """

    np.random.seed(seed)

    # Determine problem size

        n = h0.size
        # enforce exact integral constraint
        target = n / 2.0
        h0 *= target / h0.sum()
        h0 = np.clip(h0, 0.0, 1.0)          # safety
        # fix any small sum drift
        h0 *= target / h0.sum()
    else:
        n = 70           # default modest size
        h0 = _initial_h(n)

    dx = 2.0 / n

    # Initial evaluation
    best_h = h0.copy()
    best_c5 = _c5_value(best_h, dx)

    # Simulated‑annealing / hill‑climbing parameters
    temp = 0.2
    min_temp = 1e-5
    decay = 0.9995
    max_step = 0.4          # maximal fraction of movable mass per move
    start_time = time.time()
    time_limit = budget_s * 0.95   # keep a safety margin

    # Main optimisation loop
    while time.time() - start_time < time_limit:
        # propose a move that preserves the total mass
        i, j = random.sample(range(n), 2)
        # amount we can shift from i to j without violating bounds
        delta_max = min(best_h[i], 1.0 - best_h[j])
        if delta_max <= 0:
            continue
        # random step size
        delta = delta_max * random.random() * max_step
        if delta <= 0:
            continue

        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # compute new objective
        new_c5 = _c5_value(h_new, dx)

        # acceptance rule
        if new_c5 < best_c5:
            best_h, best_c5 = h_new, new_c5
        else:
            # Metropolis criterion
