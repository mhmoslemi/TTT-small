import time
import random
import numpy as np
from scipy.optimize import minimize

def _project_to_constraints(h, target_sum):
    """
    Clip h to [0,1] and adjust its sum to target_sum (sum of entries, not multiplied by dx).
    Simple water-filling iteration.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        if cur < target_sum:
            # need to increase some entries that are not at 1
            idx = np.where(h < 1.0)[0]
            if len(idx) == 0:
                break
            delta = (target_sum - cur) / len(idx)
            # cap increase so we don't exceed 1
            increase = np.minimum(delta, 1.0 - h[idx])
            h[idx] += increase
        else:
            # need to decrease entries that are not at 0
            idx = np.where(h > 0.0)[0]
            if len(idx) == 0:
                break
            delta = (cur - target_sum) / len(idx)
            decrease = np.minimum(delta, h[idx])
            h[idx] -= decrease
    # final clipping to be safe
    return np.clip(h, 0.0, 1.0)

def _c5_value(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 to minimize the
    overlap constant C5.

    Returns:
        h_values (list of float): the optimized discretized function.
        c5_bound (float): the achieved C5 value.
        n_points (int): number of discretization samples used.
    """
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # Determine discretization size
    if 'initial_h_values' in globals():
        n_points = int(len(initial_h_values))
        h_current = np.array(initial_h_values, dtype=float)
        # Ensure the sum constraint holds for the given n_points
        target_sum = n_points / 2.0
        h_current = _project_to_constraints(h_current, target_sum)
    else:
        n_points = 200  # fallback resolution
        target_sum = n_points / 2.0
        h_current = np.random.rand(n_points)
        h_current = _project_to_constraints(h_current, target_sum)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial evaluation
    best_h = h_current.copy()
    best_c5 = _c5_value(best_h, dx)

    # Parameters for stochastic search
    sigma = 0.25          # initial perturbation magnitude
    sigma_min = 1e-4
    decay = 0.995        # multiplicative decay per iteration
    max_iter = 20000

    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:  # leave a little margin
            break
        if iteration >= max_iter:
            break

        # Propose a new candidate by Gaussian perturbation
        proposal = best_h + sigma * np.random.randn(n_points)
        proposal = _project_to_constraints(proposal, target_sum)

        # Evaluate
        cand_c5 = _c5_value(proposal, dx)

        # Accept if improvement
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = proposal

        # Decay step size
        sigma = max(sigma * decay, sigma_min)
        iteration += 1

    # As a safety net, run a simple local refinement using scipy L-BFGS-B
    # (only if time permits)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5.0:
        def obj(vec):
            return _c5_value(vec, dx)

        cons = ({'type': 'eq',

        bounds = [(0.0, 1.0)] * n_points
        try:

                    best_c5 = refined_c5
                    best_h = refined
        except Exception:
            pass  # ignore any optimizer failures

    # Prepare output
    h_values = best_h.tolist()
    c5_bound = float(best_c5)
    return h_values, c5_bound, n_points
