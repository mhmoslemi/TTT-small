import time
import numpy as np

def _evaluate(h):
    """Compute C5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project(h):
    """Project array h onto [0,1] with the integral constraint sum(h)=len(h)/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(8):
        s = h.sum()
        if s == 0:
            h[:] = target / n
            break
        factor = target / s
        h *= factor
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction
    diff = target - h.sum()
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns:
        (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    start_time = time.time()
    max_time = budget_s

    # ----- initial construction ------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n_points = h0.size
        h0 = _project(h0.copy())
    else:
        n_points = 100
        h0 = np.full(n_points, 0.5)   # sum = n_points/2 meets the integral
        h0 = _project(h0)

    best_h = h0
    best_c = _evaluate(best_h)

    # ----- main stochastic local search ----------------------------------------
    while time.time() - start_time < max_time - 0.5:
        # occasional resolution increase (up to ~200 points)
        if n_points < 200 and np.random.rand() < 0.03:
            new_n = min(200, n_points * 2)
            x_old = np.linspace(0, 2, n_points, endpoint=False)
            x_new = np.linspace(0, 2, new_n, endpoint=False)
            h_interp = np.interp(x_new, x_old, best_h, period=2)
            best_h = _project(h_interp)
            n_points = new_n
            best_c = _evaluate(best_h)
            continue

        # step size decays with time
        elapsed = time.time() - start_time
        decay = 1.0 - elapsed / max_time
        step = 0.03 * decay   # max perturb per coordinate

        perturb = (np.random.rand(n_points) - 0.5) * 2.0 * step
        cand = best_h + perturb
        cand = _project(cand)
        cand_c = _evaluate(cand)

        if cand_c < best_c:
            best_h, best_c = cand, cand_c
            continue

        # occasional random restart in case of stagnation
        if np.random.rand() < 0.005:
            rand = np.random.rand(n_points)
            rand = _project(rand)
            rand_c = _evaluate(rand)
            if rand_c < best_c:
                best_h, best_c = rand, rand_c

    return best_h.tolist(), float(best_c), n_points
