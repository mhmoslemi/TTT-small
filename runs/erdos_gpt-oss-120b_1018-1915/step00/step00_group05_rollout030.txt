import time
import numpy as np
from numpy.random import default_rng

def _project_to_feasible(x, target_sum):
    """Clip to [0,1] and adjust to meet exact sum using water-filling."""
    x = np.clip(x, 0.0, 1.0)
    # iterative redistribution
    for _ in range(10):
        total = x.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size == 0:
            free = np.arange(x.size)
        delta = diff / free.size
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    return x

def _c5_value(h, dx):
    """Compute discrete C5 = max_k ∫ h(x)(1-h(x+k)) dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h sampled uniformly on [0,2].
    c5_bound : float
        Achieved upper bound for C5.
    n_points : int
        Number of discretization points used.
    """
    rng = default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n_points = init.size
    else:
        n_points = 70
        init = None

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because integral = 1

    # Initial candidate
    if init is not None and init.size == n_points:
        cand = _project_to_feasible(init.copy(), target_sum)
    else:
        # random feasible vector
        raw = rng.random(n_points)
        cand = _project_to_feasible(raw, target_sum)

    best_h = cand.copy()
    best_score = _c5_value(best_h, dx)

    start_time = time.time()
    time_limit = budget_s * 0.95  # keep margin for final checks

    # Simple stochastic pairwise transfer search
    while time.time() - start_time < time_limit:
        i, j = rng.integers(0, n_points, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]

        # feasible delta range to keep both in [0,1]
        low = -min(hi, 1.0 - hj)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = rng.uniform(low, high)

        trial = best_h.copy()
        trial[i] = hi + delta
        trial[j] = hj - delta

        # No need to project: construction guarantees feasibility
        score = _c5_value(trial, dx)
        if score < best_score - 1e-12:
            best_h = trial
            best_score = score

    # Final feasibility enforcement and score recomputation
    best_h = _project_to_feasible(best_h, target_sum)
    best_score = _c5_value(best_h, dx)

    # Safety assertions (will raise if violated)
    assert best_h.shape[0] == n_points
    assert np.all((best_h >= -1e-9) & (best_h <= 1.0 + 1e-9))
    assert abs(best_h.sum() - target_sum) < 1e-8

    return best_h, float(best_score), int(n_points)
