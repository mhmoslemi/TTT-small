import time
import numpy as np
import random

def _compute_c5(h, dx):
    """Return C5 for a given step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def _random_transfer(h, i, j):
    """Transfer a random feasible amount from h[i] to h[j] and return new array."""
    hi, hj = h[i], h[j]
    # max amount we can move from i to j without violating bounds
    max_move = min(hi, 1.0 - hj)
    if max_move <= 0.0:
        return None
    # also allow moving opposite direction (j->i)
    max_move_rev = min(hj, 1.0 - hi)
    # choose direction
    if random.random() < 0.5 and max_move_rev > 0:
        delta = -random.random() * max_move_rev
    else:
        delta = random.random() * max_move
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):

    Returns

    h_values : np.ndarray

            h[0] += diff

        h = np.full(n_points, 0.5, dtype=np.float64)  # satisfies sum = n/2

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    start_time = time.time()
    # annealing schedule parameters
    T0 = 0.1
    T_end = 1e-4
