import time
import numpy as np

def _evaluate_c5(h, dx):
    """Return C5 for a given step vector h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _random_feasible(n, target_sum, rng):
    """Generate a random feasible vector with entries in [0,1] and given sum."""
    # start from uniform and then random redistribution
    h = rng.uniform(low=0.0, high=1.0, size=n)
    # project to required sum by scaling and clipping iteratively
    for _ in range(10):
        total = h.sum()
        if total == 0:
            h = rng.uniform(low=0.0, high=1.0, size=n)
            continue
        scale = target_sum / total
        h *= scale
        # clip and renormalise the excess/deficit
        excess = h - 1.0
        deficit = -h
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-10:
            break
        # distribute the remaining diff uniformly
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # final correction to hit the sum exactly (within tolerance)
    if abs(h.sum() - target_sum) > 1e-8:
        # adjust a random element within bounds
        idx = rng.integers(0, n)
        correction = target_sum - h.sum()
        new_val = np.clip(h[idx] + correction, 0.0, 1.0)
        h[idx] = new_val
    return h

def _pair_move(h, step, target_sum, rng):
    """Propose a feasible neighbour by transferring a small delta between two indices."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    delta = (rng.random() * 2.0 - 1.0) * step
    hi = h[i] + delta
    hj = h[j] - delta
    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
        new_h = h.copy()
        new_h[i] = hi
        new_h[j] = hj
        # sum is preserved exactly
        return new_h
    return None

    """
    start_time = time.time()
