import time
import numpy as np
from scipy import optimize as opt

def run(seed=42, budget_s=1100, **kwargs):
    # -------------------- setup --------------------
    np.random.seed(seed)
    start_time = time.time()

    # number of discretization points
    n_points = kwargs.get("n_points")
    if n_points is None:
        n_points = (len(initial_h_values)
                    if "initial_h_values" in globals()
                    else 70)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ------------ feasibility helpers ------------
    def repair(x):
        """Project a vector onto [0,1]^n and the linear equality sum = target_sum."""
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
        if s == 0.0:
            x = np.full_like(x, target_sum / n_points)
        else:
            x = x * (target_sum / s)
        x = np.clip(x, 0.0, 1.0)

        # small residual correction
        diff = target_sum - x.sum()
        if abs(diff) > 1e-9:
            free = np.where((x > 0.0) & (x < 1.0))[0]
            if free.size:
                x[free] += diff / free.size
                x = np.clip(x, 0.0, 1.0)
        return x

    def objective_raw(vec):
        """Objective evaluated on a raw (possibly infeasible) vector."""
        h = repair(vec)
        g = 1.0 - h
        corr = np.correlate(h, g, mode="full") * dx
        return np.max(corr)

    # --------------- initial guess ----------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n_points:
            # resample linearly to required size
            src_x = np.linspace(0.0, 1.0, h0.size, endpoint=False)
            dst_x = np.linspace(0.0, 1.0, n_points, endpoint=False)
            h0 = np.interp(dst_x, src_x, h0)
        h0 = repair(h0)
    else:
        h0 = np.full(n_points, target_sum / n_points)

    bounds = [(0.0, 1.0)] * n_points

    # --------------- time‑controlled DE -------------
    max_iter_de = 200  # generous upper bound; will be cut by time budget

    def time_stop(*_):
        return (time.time() - start_time) >= budget_s

    de_result = opt.differential_evolution(
        objective_raw,
        bounds,
        seed=seed,
        maxiter=max_iter_de,
        polish=False,
        updating="deferred",
        workers=1,
        callback=lambda x, convergence: time_stop(),

    # best from DE (already feasible via repair)

    # --------------- optional local refinement -----

        pass
