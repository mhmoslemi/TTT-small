import time
import numpy as np
from scipy.optimize import minimize, Bounds

# Helper to compute the C5 bound for a discrete step function
def compute_c5(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def feasible_start(n, rng):
    """Generate a random feasible vector of length n."""
    # draw uniform, then scale to have correct sum, then clip and re‑project
    x = rng.random(n)
    target = n / 2.0
    # scale to target sum
    x = x * (target / x.sum())
    # simple projection onto [0,1] while preserving sum
    for _ in range(5):
        x = np.clip(x, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < 1e-12:
            break
        # distribute the deficit/excess uniformly over non‑bound entries
        free = (x > 0.0) & (x < 1.0)
        if not np.any(free):
            # all at bounds – adjust all equally (will stay in bounds because target is feasible)
            x += (target - s) / n
        else:
            delta = (target - s) / free.sum()
            x[free] += delta
    # final clip to guarantee bounds
    return np.clip(x, 0.0, 1.0)

def optimize_h(initial_h, budget, rng):
    """Run a short SLSQP optimisation starting from initial_h."""
    n = len(initial_h)
    dx = 2.0 / n
    target_sum = n / 2.0

    # objective
    def obj(vec):
        return compute_c5(vec, dx)

    # equality constraint for the integral
    cons = ({
        "type": "eq",
        "fun": lambda v: v.sum() - target_sum,
    },)

    bounds = Bounds(0.0, 1.0)

    # use SLSQP with a modest iteration limit
    res = minimize(
        obj,
        initial_h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 250, "ftol": 1e-9, "disp": False},
    )
    if not res.success:
        # fallback to the starting point
        return initial_h, compute_c5(initial_h, dx)
    h_opt = np.clip(res.x, 0.0, 1.0)  # safety clip
    # enforce exact sum (tiny numerical drift)
    h_opt *= target_sum / h_opt.sum()
    return h_opt, compute_c5(h_opt, dx)

def upscale(h):
    """Double the resolution by linear interpolation."""
    n = len(h)
    x = np.arange(n)
    x2 = np.linspace(0, n - 1, 2 * n)
    h2 = np.interp(x2, x, h)

    target = (2 * n) / 2.0

    h2 *= target / h2.sum()
    return h2
