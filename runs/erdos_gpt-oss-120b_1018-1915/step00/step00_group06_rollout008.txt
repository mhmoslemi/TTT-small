import time
import numpy as np
import random

def _evaluate(h):
    """Return C5 value for discretized h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * np.max(corr)


def _project(h, target_sum):
    """Clip to [0,1] and enforce sum == target_sum by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    # simple scaling first
    total = h.sum()
    if total > 0:
        scale = target_sum / total
        h *= scale
        h = np.clip(h, 0.0, 1.0)

    # iterative redistribution for remaining error
    for _ in range(10):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        # indices that are not at a bound
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            # distribute to all and clip again
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        inc = diff / free.sum()
        h[free] += inc
        h = np.clip(h, 0.0, 1.0)
    return h


def _upsample(parent, new_n):
    """Linear interpolation of parent (size m) to new_n points, then project."""
    m = parent.size
    x_parent = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    interp = np.interp(x_new, x_parent, parent)
    target = new_n / 2.0
    return _project(interp, target)
