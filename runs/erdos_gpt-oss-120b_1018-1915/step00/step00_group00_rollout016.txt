import time
import numpy as np

def _project_onto_constraints(vec, target_sum, max_iter=20):
    """Project vec onto [0,1] box and sum == target_sum."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # distribute the difference uniformly, then re‑clip
        v += diff / len(v)
        v = np.clip(v, 0.0, 1.0)
    # final scaling if still off (rare)
    s = v.sum()
    if s > 0:
        v *= target_sum / s
    return np.clip(v, 0.0, 1.0)

def _c5_value(h, dx):
    """Return max overlap integral for given step values h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] giving a low C5 bound.

    Returns
    -------
    h_values : ndarray
        The best found values (length n_points, each in [0,1]).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of sample points used.
    """
    np.random.seed(seed)

    # Determine number of points
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n = len(initial_h_values)
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 45                     # a modest size, works fast
        h = np.full(n, 0.5)        # uniform start (sum = n/2)

    target_sum = n / 2.0           # because integral of h over length 2 must be 1
    dx = 2.0 / n

    # Ensure feasibility of the start
    h = _project_onto_constraints(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_value(best_h, dx)

    # Simple simulated‑annealing style search
    start_time = time.time()
    temp = 0.1                     # temperature for probabilistic acceptance
    cooling = 0.9995
    step_scale = 0.2               # magnitude of random perturbation

    while time.time() - start_time < budget_s:
        # Propose a new candidate by adding Gaussian noise
        cand = h + step_scale * np.random.randn(n)
        cand = _project_onto_constraints(cand, target_sum)

        cand_c5 = _c5_value(cand, dx)

        # Accept if better, otherwise maybe accept according to temperature
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand.copy()
            h = cand
        else:
            if np.random.rand() < np.exp(-(cand_c5 - best_c5) / max(temp, 1e-12)):
                h = cand

        # Anneal and slowly shrink step size
        temp *= cooling
        step_scale *= cooling

    return best_h, best_c5, n
