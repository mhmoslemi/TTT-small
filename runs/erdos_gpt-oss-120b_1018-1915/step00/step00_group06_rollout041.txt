import time
import numpy as np
from scipy.optimize import minimize

# helper: compute C5 for a given h vector
def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)

# helper: project onto [0,1] box and affine sum constraint
def project_box_sum(v, target, eps=1e-9, max_iter=100):
    """Find λ such that sum(clip(v+λ,0,1)) = target."""
    lo = -np.max(v)          # makes all ≤0 → sum=0
    hi = 1.0 - np.min(v)     # makes all ≥1 → sum=n
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) <= eps:
            return proj
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + mid, 0.0, 1.0)

# helper: generate an initial h at given resolution
def initialise_h(n_points):
    # use provided initial_h_values if its length matches or can be up‑sampled
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.size == n_points:
            h0 = parent.copy()
        else:
            # linear interpolation on uniform grid [0,2)
            x_parent = np.linspace(0.0, 2.0, parent.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
            h0 = np.interp(x_new, x_parent, parent)
    else:
        # random start
        rng = np.random.default_rng()
        h0 = rng.random(n_points)
    # enforce box and sum constraints
    target = n_points / 2.0
    h0 = project_box_sum(h0, target)
    return h0

# main run function
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # configuration
    n_points = 200                     # moderate resolution
    target_sum = n_points / 2.0
    # initial solution
    h_curr = initialise_h(n_points)
    best_h = h_curr.copy()
    best_score = compute_c5(best_h)
    # simple stochastic local search
    step = 0.2                         # initial perturbation scale
    evals = 0
    while time.time() - start_time < 0.95 * budget_s:
        # propose perturbation on a random subset
        h_prop = best_h.copy()
        idx = rng.choice(n_points, size=5, replace=False)
        delta = rng.normal(scale=step, size=5)
        h_prop[idx] += delta
        # project back to feasible set
        h_prop = project_box_sum(h_prop, target_sum)
        score = compute_c5(h_prop)
        evals += 1
        if score < best_score - 1e-7:
            best_h = h_prop
            best_score = score
            # gradually reduce step when improvement found
            step = max(0.02, step * 0.95)
        else:
            # occasional step increase to escape plateaus
            if evals % 500 == 0:
                step = min(0.4, step * 1.2)
    # final polishing with a few SLSQP iterations (optional, cheap)
    def objective(h):
        return compute_c5(h)
    cons = ({'type': 'eq',
             'fun': lambda h: np.sum(h) - target_sum},)
    bounds = [(0.0, 1.0)] * n_points
    res = minimize(objective, best_h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': 200, 'ftol': 1e-9})
    if res.success and res.fun < best_score:
        best_h = res.x
        best_score = res.fun
    # ensure exact feasibility
    best_h = project_box_sum(best_h, target_sum)
    best_score = compute_c5(best_h)
    return best_h, float(best_score), n_points
