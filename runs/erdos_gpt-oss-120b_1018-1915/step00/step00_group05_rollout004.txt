from scipy.optimize import minimize, differential_evolution

# ------------------------------------------------------------
# Helper: evaluate the C5 value for a discrete step function h
def _c5_value(h, dx):
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return float(np.max(corr) * dx)
