import time
import numpy as np

def _evaluate(h):
    """Return max overlap integral C5 for discretized h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _repair(h, target_sum, max_iter=500):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            add = min(diff / idx.size, 1.0 - h[idx].min())
            inc = np.full(idx.shape, add)
            # make sure we don't overshoot any entry
            inc = np.minimum(inc, 1.0 - h[idx])
            h[idx] += inc
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            sub = min(-diff / idx.size, h[idx].max())
            dec = np.full(idx.shape, sub)
            dec = np.minimum(dec, h[idx])
            h[idx] -= dec
    return np.clip(h, 0.0, 1.0)

def _perturb(h, rng, step=0.02):
    """Move a small amount of mass from one coordinate to another."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    # amount that can be transferred without violating bounds
    max_from_i = min(step, h[i])
    max_to_j = min(step, 1.0 - h[j])
    delta = rng.random() * min(max_from_i, max_to_j)
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    return h2
