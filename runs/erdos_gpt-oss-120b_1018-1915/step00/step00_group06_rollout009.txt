import time
import math
import random
import numpy as np

def _compute_c5(h):
    """Return the C5 bound for a discrete h on [0,2]."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)

def _normalize_sum(h, target_sum):
    """Scale h to have exact sum target_sum while staying in [0,1]."""
    cur = h.sum()
    if cur == 0:
        # distribute uniformly
        h[:] = target_sum / h.size
        return h
    scale = target_sum / cur
    h *= scale
    # Clip and re‑adjust any overflow/underflow
    np.clip(h, 0.0, 1.0, out=h)
    # final fix for any small residual error
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute the residual to a random index that can absorb it
        idx = np.argwhere((h > 0) & (h < 1)).flatten()
        if idx.size == 0:
            # fallback: adjust the first element within bounds
            idx = np.array([0])
        i = random.choice(idx.tolist())
        h[i] = min(1.0, max(0.0, h[i] + diff))
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimizes a step function h on [0,2] to minimise the Erdős C5 overlap bound.

    Returns
    -------
    h_values : np.ndarray
        Optimised values of h on a uniform grid.
    c5_bound : float
        Corresponding C5 bound.
    n_points : int
        Number of discretisation points.
    """
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # Initialise grid
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: 100 points, uniform 0.5 (satisfies integral)
        h0 = np.full(100, 0.5, dtype=float)

    n = h0.size
    target_sum = n / 2.0          # because integral of h must be 1, dx = 2/n
    h = _normalize_sum(h0.copy(), target_sum)

    best_h = h.copy()
    best_score = _compute_c5(best_h)

    # ----------------------------------------------------------------------
    # Simulated annealing parameters
    # ----------------------------------------------------------------------
    T0 = 0.05                     # initial temperature
    min_T = 1e-6
    iteration = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:      # stop a bit early to be safe
            break

        # linear cooling schedule
        T = max(min_T, T0 * (1.0 - elapsed / budget_s))

        # Propose a move: pick two distinct indices and transfer mass
        i, j = random.sample(range(n), 2)
        new_i = random.random()            # uniform candidate in [0,1]
        delta = new_i - h[i]
        new_j = h[j] - delta

        # Feasibility test
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue

        h_candidate = h.copy()
        h_candidate[i] = new_i
        h_candidate[j] = new_j

        # (optional) small random jitter to avoid plateau
        if random.random() < 0.02:
            idx = random.randrange(n)
            jitter = (random.random() - 0.5) * 0.02
            val = h_candidate[idx] + jitter
            if 0.0 <= val <= 1.0:
                h_candidate[idx] = val

        # Ensure exact sum (tiny numeric drift)
        h_candidate = _normalize_sum(h_candidate, target_sum)

        cand_score = _compute_c5(h_candidate)

        # Acceptance rule
        if cand_score < best_score or random.random() < math.exp((best_score - cand_score) / T):
            h = h_candidate
            if cand_score < best_score:
                best_score = cand_score
                best_h = h_candidate.copy()

        iteration += 1

    # Final safety clamp and exact normalisation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _normalize_sum(best_h, target_sum)
    final_score = _compute_c5(best_h)

    return best_h, final_score, n
