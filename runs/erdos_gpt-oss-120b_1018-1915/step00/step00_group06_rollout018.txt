import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def evaluate_h(h):
    """Return C5 value for a given step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * np.max(corr)


def binary_initial(n, rng):
    """Create a binary vector with exactly n//2 ones."""
    h = np.zeros(n, dtype=np.float64)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    return h


def simulated_annealing(n, budget, rng, start_h=None):
    """Simple SA that swaps a 1 with a 0 to improve the max‑overlap."""
    if start_h is None:
        cur = binary_initial(n, rng)
    else:
        cur = start_h.copy()
    best = cur.copy()
    cur_score = best_score = evaluate_h(cur)

    start = time.time()
    T0 = 1.0
    Tmin = 1e-4
    while True:
        elapsed = time.time() - start
        if elapsed >= budget:
            break
        # temperature schedule (linear)
        T = max(Tmin, T0 * (1.0 - elapsed / budget))

        # pick indices to swap
        ones_idx = np.flatnonzero(cur)
        zeros_idx = np.flatnonzero(1.0 - cur)
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        # perform swap
        cur[i], cur[j] = cur[j], cur[i]

        new_score = evaluate_h(cur)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur_score = new_score
            if new_score < best_score:
                best = cur.copy()
                best_score = new_score
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]

    return best, best_score


def upsample_to_n(parent, n):
    """Linear interpolation of a parent vector to a new size n, then
    project to a binary vector with exact n/2 ones."""
    x_old = np.linspace(0, 2, len(parent), endpoint=False)
    x_new = np.linspace(0, 2, n, endpoint=False)
    interp = np.interp(x_new, x_old, parent)
    # threshold to obtain binary while preserving count
    thresh = np.percentile(interp, 50)  # median gives roughly half ones
    binary = (interp >= thresh).astype(np.float64)
    # fix count if off by one due to ties
    cur_ones = int(binary.sum())
    target = n // 2
    if cur_ones > target:
        # turn some ones to zero
        idx = np.flatnonzero(binary)
        np.random.shuffle(idx)
        binary[idx[: cur_ones - target]] = 0.0
    elif cur_ones < target:
        idx = np.flatnonzero(1.0 - binary)
        np.random.shuffle(idx)
        binary[idx[: target - cur_ones]] = 1.0
    return binary

            h_candidate, score = simulated_annealing(
                n, attempt_budget, rng, start_h=init
            )
            if score < best_score:
                best_h, best_score, best_n = h_candidate, score, n
            # stop early if we already beat the target
            if best_score <= 0.3808:
                break
        if best_score <= 0.3808:

    # Fallback: if nothing improved, use a simple balanced binary vector
    if best_h is None:
        n = 200
        best_h = binary_initial(n, rng)
        best_score = evaluate_h(best_h)
        best_n = n

    # Ensure exact constraints (numerical safety)

    target = best_n / 2.0
