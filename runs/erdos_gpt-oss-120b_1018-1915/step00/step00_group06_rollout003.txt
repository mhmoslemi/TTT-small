import time
import math
import numpy as np

def _c5_of_h(h: np.ndarray) -> float:
    """Return the C5 value for a discrete step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation h * (1-h) shifted; mode='full' yields length 2n-1
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)


def _ensure_constraints(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and renormalise to exact integral = 1 (sum = n/2)."""
    n = h.size
    np.clip(h, 0.0, 1.0, out=h)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if cur_sum == 0:
        # fallback to uniform 0.5
        h.fill(0.5)
    else:
        h *= target_sum / cur_sum
    # a final clipping may break the sum; perform a simple redistribution
    excess = h.sum() - target_sum
    if abs(excess) > 1e-12:
        # spread the excess uniformly while keeping bounds
        delta = excess / n
        h -= delta
        np.clip(h, 0.0, 1.0, out=h)
        # renormalise again (very cheap, only a few iterations needed)
        cur_sum = h.sum()
        if cur_sum != target_sum:
            h *= target_sum / cur_sum
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 overlap bound.

    Returns
    -------
    h_values : list of float
        Optimised values on a uniform grid.
    c5_bound : float
        Corresponding C5 bound.
    n_points : int
        Number of samples used.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine grid size and initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h = _ensure_constraints(h0.copy())
    else:
        # fallback: modest size to keep runtime short, yet enough resolution
        n = 150
        h = np.full(n, 0.5, dtype=float)   # already satisfies integral
    # make sure constraints hold
    h = _ensure_constraints(h)

    # ------------------------------------------------------------------
    # 2. Simple simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = _c5_of_h(best_h)

    start_time = time.time()
    time_limit = budget_s * 0.94   # a safety margin
    # annealing schedule parameters
    max_iter = 200000
    T0 = 0.5                      # initial temperature (controls acceptance)
    decay = 0.9999                # per‑iteration temperature decay

    T = T0
    iter_count = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit or iter_count >= max_iter:
            break

        # ---- propose a move -------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # maximum amount we can shift from i to j without leaving [0,1]
        max_drop_i = h[i]                     # can't go below 0
        max_raise_j = 1.0 - h[j]              # can't exceed 1
        max_delta = min(max_drop_i, max_raise_j)

        if max_delta <= 0:
            continue

        delta = rng.random() * max_delta * 0.5   # limit step size
        # direction random: sometimes move from j to i
        if rng.random() < 0.5:
            delta = -delta

        # apply tentative move
        h[i] -= delta
        h[j] += delta

        # ---- evaluate -------------------------------------------------------
        new_score = _c5_of_h(h)

        # acceptance rule
        accept = False
        if new_score < best_score:
            accept = True
        else:
            prob = math.exp(-(new_score - best_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert move
            h[i] += delta
            h[j] -= delta

        # temperature decay
        T *= decay
        iter_count += 1

    # ------------------------------------------------------------------
    # 3. Final clean‑up: ensure feasibility and recompute final score
    # ------------------------------------------------------------------
    best_h = _ensure_constraints(best_h)
    final_score = _c5_of_h(best_h)

    return best_h.tolist(), final_score, n
