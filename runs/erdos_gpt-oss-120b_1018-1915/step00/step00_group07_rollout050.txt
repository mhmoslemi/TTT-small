import time
import numpy as np

def _repair(h, target_sum):
    """Project h onto [0,1] with exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    s = target_sum - h.sum()
    # iterative water-filling style
    for _ in range(10):
        if abs(s) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            # no room to adjust, distribute proportionally ignoring bounds
            h += s / len(h)
            h = np.clip(h, 0.0, 1.0)
            s = target_sum - h.sum()
            continue
        delta = s / free.sum()
        h[free] += delta
        # re‑clip and recompute deficit
        h = np.clip(h, 0.0, 1.0)
        s = target_sum - h.sum()
    # final tiny correction by uniform scaling if still off
    if abs(h.sum() - target_sum) > 1e-9:
        h *= target_sum / h.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _objective(h, dx):
    """Return max overlap C5 for a given step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 80
        h0 = np.full(n_points, 0.5)  # dummy feasible start
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure the start vector satisfies constraints
    h0 = _repair(h0, target_sum)
    best_h = h0.copy()
    best_score = _objective(best_h, dx)

    # DE parameters
    pop_size = max(20, n_points // 2)
    F = 0.5
    CR = 0.9
    max_gen = 500  # generous upper bound; time check will stop earlier

    # Initialise population (include the known start)
    pop = np.empty((pop_size, n_points), dtype=float)
    pop[0] = best_h
    for i in range(1, pop_size):
        cand = rng.random(n_points)
        pop[i] = _repair(cand, target_sum)

    scores = np.array([_objective(ind, dx) for ind in pop])
    # Keep track of best ever
    idx_best = np.argmin(scores)
    best_h = pop[idx_best].copy()
    best_score = scores[idx_best]

    start_time = time.time()
    for gen in range(max_gen):
        if time.time() - start_time > budget_s * 0.95:
            break
        for i in range(pop_size):
            # mutation: select three distinct indices different from i
            idxs = rng.choice(pop_size, 3, replace=False)
            while i in idxs:
                idxs = rng.choice(pop_size, 3, replace=False)
            a, b, c = pop[idxs]
            mutant = a + F * (b - c)
            # crossover (binomial)
            cross = rng.random(n_points) < CR
            if not np.any(cross):
                cross[rng.integers(0, n_points)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = _repair(trial, target_sum)
            trial_score = _objective(trial, dx)

            # Selection

                scores[i] = trial_score

                    best_score = trial_score
                    best_h = trial.copy()
                    if best_score < 0.3808:
                        # excellent; can stop early

        # optional: reshuffle best individuals to maintain diversity
        # (not strictly necessary)
