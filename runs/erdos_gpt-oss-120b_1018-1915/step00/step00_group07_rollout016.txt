import time
import numpy as np

def _project_to_feasible(v, target_sum):
    """
    Clip to [0,1] and adjust to hit the target sum using a simple iterative
    redistribution scheme.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(15):
        s = v.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        free = (v > 0.0) & (v < 1.0)
        if not free.any():
            # all at bounds – scale the whole vector (may break bounds a bit)
            if s > 0:
                v *= target_sum / s
            v = np.clip(v, 0.0, 1.0)
            continue
        adjust = diff / free.sum()
        v[free] += adjust
        v = np.clip(v, 0.0, 1.0)
    return v

def _random_feasible(n, target_sum):
    """Generate a random vector in [0,1]^n with the required sum."""
    v = np.random.rand(n)
    v = _project_to_feasible(v, target_sum)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

        n_points = h0.size
    else:

        h0 = _random_feasible(n_points, n_points / 2.0)

    target_sum = n_points / 2.0          # because sum(h)*dx = 1, dx=2/n
    best_h = _project_to_feasible(h0, target_sum)
    best_score = evaluate_erdos_solution(best_h)[0]

    # Helper to evaluate and possibly update the incumbent
    def try_candidate(cand):
        nonlocal best_h, best_score
        sc = evaluate_erdos_solution(cand)[0]
        if sc < best_score:
            best_score = sc
            best_h = cand.copy()

    # Simple stochastic hill‑climbing loop
    while time.time() - start_time < budget_s * 0.98:
        cand = best_h.copy()

        # Perform a few random transfers between coordinates
        for _ in range(np.random.randint(1, 4)):
            i, j = np.random.choice(n_points, 2, replace=False)
            max_transfer = min(cand[i], 1.0 - cand[j])
            # propose a change within ±0.05 (adjusted to feasible range)
            delta = (np.random.rand() * 0.1) - 0.05
            delta = np.clip(delta, -max_transfer, max_transfer)
            cand[i] -= delta
            cand[j] += delta

        # Occasionally add tiny Gaussian noise to a handful of entries
        if np.random.rand() < 0.2:
            idx = np.random.choice(n_points, size=np.random.randint(1, 5), replace=False)
            noise = np.random.randn(idx.size) * 0.02
            cand[idx] += noise

        cand = _project_to_feasible(cand, target_sum)

            try_candidate(restart)

    return best_h, best_score, n_points
