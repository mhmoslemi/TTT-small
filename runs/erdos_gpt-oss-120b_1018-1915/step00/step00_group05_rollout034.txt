import time, math
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine starting vector and size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 70
        h0 = np.full(n, 0.5, dtype=float)

    dx = 2.0 / n
    target_sum = n / 2.0  # because ∫h = 1  →  sum(h)*dx = 1

    # Projection onto box [0,1] and affine sum constraint
    def project(v):
        y = np.clip(v, 0.0, 1.0)
        for _ in range(100):
            diff = target_sum - y.sum()
            if abs(diff) < 1e-12:
                break
            free = (y > 0.0) & (y < 1.0)
            cnt = free.sum()
            if cnt == 0:
                y += diff / n
                y = np.clip(y, 0.0, 1.0)
                continue
            adjust = diff / cnt
            y[free] += adjust
            y = np.clip(y, 0.0, 1.0)
        # tiny residual correction
        residual = target_sum - y.sum()
        if abs(residual) > 1e-9:
            free = (y > 0.0) & (y < 1.0)
            cnt = free.sum()
            if cnt:
                y[free] += residual / cnt
                y = np.clip(y, 0.0, 1.0)
        return y

    # Objective: maximal overlap integral
    def evaluate(v):
        corr = np.correlate(v, 1.0 - v, mode='full') * dx
        return np.max(corr)

    current = project(h0)
    cur_val = evaluate(current)

    best = current.copy()
    best_val = cur_val

    start_time = time.time()
    temp_start = 0.2
    temp_end = 1e-4
    max_iter = 200000

    for it in range(max_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break

        # linear cooling
        elapsed = time.time() - start_time
        t = temp_start * (1.0 - elapsed / budget_s)
        if t < temp_end:
            t = temp_end

        # propose a perturbation
        prop = current.copy()
        k = max(1, int(0.1 * n))                     # 10% of entries
        idx = rng.choice(n, size=k, replace=False)
        sigma = 0.2 * (t / temp_start)               # shrink with temperature
        prop[idx] += rng.normal(scale=sigma, size=k)
        prop = project(prop)

        prop_val = evaluate(prop)
        delta = prop_val - cur_val

        if delta < 0 or rng.random() < math.exp(-delta / t):
            current, cur_val = prop, prop_val
            if prop_val < best_val:
                best, best_val = prop.copy(), prop_val

    return best, best_val, n
