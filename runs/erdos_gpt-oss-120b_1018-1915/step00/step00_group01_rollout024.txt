import time
import numpy as np

def _project_to_constraints(v, target):
    """Project vector v onto the box [0,1]^n with sum == target."""
    # Clip to [0,1] first
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= 1e-12:
        return v
    # Find lambda such that sum(clip(v - lam,0,1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        sw = w.sum()
        if sw > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _upscale(h, new_n):
    """Linearly interpolate h (defined on [0,2)) to length new_n."""
    n = h.shape[0]
    if n == new_n:
        return h.copy()
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    import random

    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ----- initial vector -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.shape[0]
    else:
        n0 = 200
        h0 = np.full(n0, 0.5, dtype=float)

    # ensure feasibility
    target0 = n0 / 2.0
    h0 = _project_to_constraints(h0, target0)

    # evaluate helper (provided by harness)
    best_h = h0.copy()
    best_score = float(evaluate_erdos_solution(best_h))
    best_n = n0

    # ----- local search routine -------------------------------------------
    def local_search(h_init, n):
        """Simple stochastic improvement on a fixed resolution."""
        target = n / 2.0
        h_cur = _project_to_constraints(h_init, target)
        cur_score = float(evaluate_erdos_solution(h_cur))
        step = 0.05
        itr = 0
        # limit iterations loosely; stop early if out of time
        while time.time() - start_time < budget_s * 0.9 and itr < 3000:
            itr += 1
            cand = h_cur + step * np.random.randn(n)
            cand = _project_to_constraints(cand, target)
            cand_score = float(evaluate_erdos_solution(cand))
            if cand_score < cur_score:
                h_cur, cur_score = cand, cand_score
                step = max(step * 0.99, 1e-4)
                # early exit if very good

                    break
            else:
                step *= 0.995
        return h_cur, cur_score

    # ----- multi‑resolution exploration ------------------------------------
    candidate_ns = [best_n, best_n * 2, int(best_n * 1.5), 300, 400, 500]
