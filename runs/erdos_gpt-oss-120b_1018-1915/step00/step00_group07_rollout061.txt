import time
import random
import numpy as np

def _project_to_bounds_sum(v, target, lo=0.0, hi=1.0):
    """Project vector v onto [lo,hi] with sum exactly target.
    Simple iterative water-filling."""
    n = len(v)
    x = np.clip(v, lo, hi)
    current_sum = x.sum()
    if abs(current_sum - target) < 1e-12:
        return x
    # Distribute the deficit/excess uniformly to non‑boundary entries
    # Perform up to a few passes; convergence is fast for our small n.
    for _ in range(10):
        deficit = target - x.sum()
        if abs(deficit) < 1e-12:
            break
        free = (x > lo) & (x < hi)
        if not np.any(free):
            # all at bounds, just scale uniformly ignoring bounds
            x = np.clip(v * (target / current_sum), lo, hi)
            break
        adjust = deficit / free.sum()
        x[free] += adjust
        # enforce bounds again
        x = np.clip(x, lo, hi)
    # final correction if tiny numerical error
    x = np.clip(x, lo, hi)
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute the tiny remainder to any non‑full entries
        for idx in np.where(x < hi)[0]:
            add = min(diff, hi - x[idx])
            x[idx] += add
            diff -= add
            if abs(diff) < 1e-12:
                break
    return x

def _evaluate_erdos(h, dx):
    """Return C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
