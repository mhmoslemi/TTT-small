import time
import numpy as np
import random

# The harness supplies evaluate_erdos_solution and possibly initial_h_values.

def _project_to_constraints(h, target_sum, max_iter=10, eps=1e-9):
    """Clip h to [0,1] and enforce sum == target_sum via iterative scaling."""
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) <= eps:
            break
        if s == 0:
            # all zeros, distribute uniformly
            h[:] = target_sum / len(h)
            break
        scale = target_sum / s
        h *= scale
    # final clip to be safe
    h = np.clip(h, 0.0, 1.0)
    return h

def _random_step(h, step_std=0.1, n_change=5):
    """Return a perturbed copy of h."""
    h_new = h.copy()
    idx = np.random.choice(len(h), size=n_change, replace=False)
    h_new[idx] += np.random.randn(n_change) * step_std
    return h_new

    n = len(h_initial)
