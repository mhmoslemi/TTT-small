import time, random, math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx using FFT convolution."""
    n = h.shape[0]
    dx = 2.0 / n
    m = n * 2                # zero‑pad length for linear convolution
    H = np.fft.fft(h, m)
    G = np.fft.fft(1.0 - h, m)
    conv = np.fft.ifft(H * np.conj(G)).real   # length m
    # linear convolution of length 2n‑1 lives in the first 2n‑1 entries
    max_val = np.max(conv[: 2 * n - 1])
    return max_val * dx

def _jitter(h: np.ndarray, sigma: float) -> np.ndarray:
    """Add Gaussian noise and re‑project onto the feasible set."""
    n = h.shape[0]
    target = n / 2.0
    h_new = h + np.random.normal(0.0, sigma, size=n)
    h_new = np.clip(h_new, 0.0, 1.0)
    total = h_new.sum()
    if total > 0:
        h_new *= target / total
    else:
        h_new[:] = target / n
    # final clipping & tiny mass fix
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        # distribute the leftover mass to a random free coordinate
        idx = np.random.randint(n)
        add = np.clip(h_new[idx] + diff, 0.0, 1.0) - h_new[idx]
        h_new[idx] += add
    return h_new
