import numpy as np
import time
from scipy.optimize import differential_evolution

def _dx(n):
    return 2.0 / n

def _compute_c5(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _project_to_feasible(h, desired_sum, tol=1e-9, max_iter=1000):
    """Clip to [0,1] and adjust to hit the exact sum using a simple water‐filling."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        total = h.sum()
        diff = desired_sum - total
        if abs(diff) <= tol:
            break
        # indices that are not at a bound
        free = np.where((h > 0.0 + tol) & (h < 1.0 - tol))[0]
        if free.size == 0:
            # all at bounds, distribute diff uniformly and clip again
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction by scaling the free part if needed
    total = h.sum()
    if abs(total - desired_sum) > tol:
        scale = desired_sum / total
        h = np.clip(h * scale, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    if "initial_h_values" in globals():
        init_h = np.asarray(initial_h_values, dtype=float)
        n_points = init_h.size
    else:
        n_points = kwargs.get("n_points", 70)
        init_h = None

    dx = _dx(n_points)
    desired_sum = n_points / 2.0

    # penalty weights
    PEN_SUM = 100.0
    PEN_BND = 100.0

    def objective(x):
        # x is in [0,1] due to bounds, but we add soft penalties for safety
        c5 = _compute_c5(x, dx)
        sum_err = x.sum() - desired_sum
        # bound penalty – small because DE respects bounds, but keep for numerical safety
        bnd_err = np.maximum(-x, 0.0).sum() + np.maximum(x - 1.0, 0.0).sum()
        return c5 + PEN_SUM * (sum_err ** 2) + PEN_BND * bnd_err

    bounds = [(0.0, 1.0)] * n_points

    # Prepare initial population (optional)
    popsize = kwargs.get("popsize", 12)  # default DE popsize (12 = 12*d)
    init_pop = None
    if init_h is not None:
        # Ensure the initial vector satisfies bounds; it already should.
        init_pop = np.random.rand(popsize, n_points)
        init_pop[0] = np.clip(init_h, 0.0, 1.0)

    # Callback to respect the time budget
    def stop_callback(xk, convergence):
        return (time.time() - start_time) >= budget_s

    # Run differential evolution with modest iteration budget
    maxiter = kwargs.get("maxiter", 30)
    result = differential_evolution(
        objective,
        bounds,
        strategy="best1bin",
        maxiter=maxiter,
        popsize=popsize,
        mutation=(0.5, 1.0),
        recombination=0.7,
        polish=False,
        init=init_pop,
        callback=stop_callback,
        disp=False,
        seed=seed,
    )

    # Take the best found vector
    best_vec = result.x
    # Project to exact feasible region
    best_vec = _project_to_feasible(best_vec, desired_sum)

    # Final C5 evaluation
    c5_bound = _compute_c5(best_vec, dx)

    # Ensure constraints (sanity check)
    assert np.all(best_vec >= -1e-9) and np.all(best_vec <= 1.0 + 1e-9)
    assert abs(best_vec.sum() - desired_sum) <= 1e-6

    return best_vec, float(c5_bound), n_points
