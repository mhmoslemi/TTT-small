import time
import numpy as np

def evaluate(h):
    """Return the C5 value for a discretized step function h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def upsample_h(old_h, new_n):
    """Linearly interpolate old_h to length new_n and fix integral & bounds."""
    old_n = len(old_h)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # enforce sum = new_n/2
    desired = new_n / 2.0
    total = new_h.sum()
    if total > 0:
        new_h *= desired / total
    new_h = np.clip(new_h, 0.0, 1.0)
    # tiny correction if clipping altered the sum
    deficit = desired - new_h.sum()
    if abs(deficit) > 1e-12:
        free = np.where((new_h > 0.0) & (new_h < 1.0))[0]
        if free.size:
            add = deficit / free.size
            new_h[free] += add
            new_h = np.clip(new_h, 0.0, 1.0)
        else:
            new_h[0] = np.clip(new_h[0] + deficit, 0.0, 1.0)
    return new_h

def local_search(start_h, time_limit, max_step=0.12):
    """Simple hill‑climbing with mass transfers that keep the integral fixed."""
    n = len(start_h)
    best_h = start_h.copy()
    best_score = evaluate(best_h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit and it < 200_000:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = best_h[i], best_h[j]
        # feasible transfer delta from i to j
        lower = max(-hj, hi - 1.0)          # ensure hi‑delta ≤1 and hj+delta ≥0
        upper = min(hi, 1.0 - hj)           # ensure hi‑delta ≥0 and hj+delta ≤1
        if lower >= upper:
            continue
        delta = lower + np.random.rand() * (upper - lower)
        # optionally scale delta to stay within a modest step size
        if abs(delta) > max_step:
            delta = np.sign(delta) * max_step
        if abs(delta) < 1e-12:
            continue
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_score = evaluate(new_h)
        if new_score < best_score:
            best_h = new_h
            best_score = new_score
        it += 1
        # occasional large random perturbation to escape plateaus
        if it % 5000 == 0:
            mask = np.random.rand(n) < 0.02
            perturb = (np.random.rand(mask.sum()) - 0.5) * 0.2
            cand = best_h.copy()
            cand[mask] += perturb
            cand = np.clip(cand, 0.0, 1.0)
            # re‑normalize integral
            desired = n / 2.0
            diff = desired - cand.sum()
            free = np.where((cand > 0.0) & (cand < 1.0))[0]
            if free.size:
                cand[free] += diff / free.size
                cand = np.clip(cand, 0.0, 1.0)
            cand_score = evaluate(cand)
            if cand_score < best_score:
                best_h, best_score = cand, cand_score
    return best_h, best_score

    remaining = total_budget
