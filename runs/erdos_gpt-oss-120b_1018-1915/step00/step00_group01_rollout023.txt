import time
import numpy as np

def _max_overlap(h):
    """Return C5 bound for a given discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlate as defined in the statement
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _binary_from_fractional(arr, target_ones):
    """Convert a fractional array to binary with exactly target_ones ones."""
    # greedy threshold then fix count
    idx = np.argsort(-arr)  # descending
    bin_arr = np.zeros_like(arr, dtype=float)
    bin_arr[idx[:target_ones]] = 1.0
    return bin_arr

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function h (as a numpy array), its C5 bound, and n_points.
    The algorithm uses a simple simulated‑annealing over binary vectors
    with exactly half the entries equal to 1 (so the integral constraint holds).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Choose resolution
    # ------------------------------------------------------------------
    n_points = 400                     # even, gives dx = 0.005
    target_ones = n_points // 2       # integral = 1  <=> sum(h)=n/2

    # ------------------------------------------------------------------
    # 2. Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        # Upsample / interpolate to the chosen resolution
        x_old = np.linspace(0.0, 2.0, base.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_old, base)
        # Force binary while preserving the total number of ones
        h = _binary_from_fractional(h0, target_ones)
    else:
        # pure random binary start
        h = np.zeros(n_points, dtype=float)
        h[:target_ones] = 1.0
        np.random.shuffle(h)

    best_h = h.copy()
    best_score = _max_overlap(best_h)

    # ------------------------------------------------------------------
    # 3. Simulated annealing parameters
    # ------------------------------------------------------------------
    temperature = 0.02                # initial temperature (in bound units)
    cooling = 0.995                   # multiplicative cooling per iteration
    max_iter = 200000                # safety cap
    iter_count = 0

    # Pre‑compute indices of ones and zeros for fast swaps
    ones_idx = np.flatnonzero(h == 1.0)
    zeros_idx = np.flatnonzero(h == 0.0)

    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        if iter_count >= max_iter:
            break
        iter_count += 1

        # ------------------------------------------------------------------
        # 3a. Propose a swap of a 1 and a 0
        # ------------------------------------------------------------------
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        # perform tentative swap
        h[i], h[j] = 0.0, 1.0

        # ------------------------------------------------------------------
        # 3b. Evaluate
        # ------------------------------------------------------------------
        new_score = _max_overlap(h)

        # ------------------------------------------------------------------
        # 3c. Acceptance test
        # ------------------------------------------------------------------
        accept = False
        if new_score < best_score:
            accept = True
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(new_score - best_score) / temperature):
                accept = True

        if accept:
            # keep the modification, update supporting structures
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
            # update index lists (swap them)
            ones_idx = np.where(h == 1.0)[0]
            zeros_idx = np.where(h == 0.0)[0]
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0

        # ------------------------------------------------------------------
        # 3d. Cool down
        # ------------------------------------------------------------------
        temperature *= cooling
        if temperature < 1e-6:
            temperature = 1e-6

    # ------------------------------------------------------------------
    # 4. Return best found solution
    # ------------------------------------------------------------------
    final_c5 = best_score
    return best_h, final_c5, n_points
