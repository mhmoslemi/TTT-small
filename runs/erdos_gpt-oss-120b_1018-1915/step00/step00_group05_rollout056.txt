import time
import random
import numpy as np

def _project_to_bounds_and_sum(v, target):
    """Project vector v onto [0,1]^n with sum == target using water‑filling."""
    v = np.clip(v, 0.0, 1.0)
    # iterative water‑filling
    for _ in range(10):
        excess = v.sum() - target
        if abs(excess) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        if not np.any(free):
            # all at bounds – distribute excess uniformly and re‑clip
            v += excess / len(v)
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = -excess / free.sum()
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h):
    """Discrete C5 = max correlation of h with (1-h) multiplied by dx."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    h_values: ndarray of length n_points, values in [0,1] and sum == n_points/2
    c5_bound: the achieved max overlap integral
    n_points: number of samples used
    """
    random.seed(seed)
    np.random.seed(seed)

    # Determine dimension
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = kwargs.get('n_points', 70)
        h0 = np.random.rand(n)

    target_sum = n / 2.0
    h = _project_to_bounds_and_sum(h0, target_sum)

    best_h = h.copy()
    best_val = _c5_value(best_h)

    t0 = time.time()
    end_time = t0 + budget_s * 0.95  # small safety margin

    # annealing schedule
    init_temp = 0.02
    final_temp = 0.0005
    total_iters = 0

    while time.time() < end_time:
        # linear temperature decay based on elapsed proportion
        elapsed = (time.time() - t0) / (end_time - t0)
        temp = init_temp * (1 - elapsed) + final_temp * elapsed

        # pick two distinct indices
        i, j = random.sample(range(n), 2)

        # compute feasible delta range for h[i] += d, h[j] -= d
        d_min = max(-h[i], h[j] - 1.0)          # cannot go below 0 or above 1
        d_max = min(1.0 - h[i], h[j])          # cannot exceed bounds
        if d_max <= d_min:
            continue

        # propose a change; magnitude biased by temperature
        d = (random.random() * 2 - 1) * temp
        d = np.clip(d, d_min, d_max)

        h[i] += d
        h[j] -= d

        val = _c5_value(h)

        # accept if better or with Metropolis probability
        if val < best_val or random.random() < np.exp(-(val - best_val) / max(temp, 1e-12)):
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            # revert
            h[i] -= d
            h[j] += d

        total_iters += 1

        # occasional full re‑projection to curb drift
        if total_iters % 5000 == 0:
            h = _project_to_bounds_and_sum(h, target_sum)

    # Final projection of best solution (ensures exact sum)
    best_h = _project_to_bounds_and_sum(best_h, target_sum)
    best_val = _c5_value(best_h)

    return best_h, best_val, n
