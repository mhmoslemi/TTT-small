import time
import numpy as np
import random

def evaluate(h):
    """Return C5 for discrete h on [0,2]."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)

def repair(h):
    """Project h onto the feasible set 0≤h≤1, sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    desired = n / 2.0
    for _ in range(10):
        total = h.sum()
        diff = desired - total
        if abs(diff) < 1e-10:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = int(free.sum())
        if cnt == 0:
            # all at bounds – distribute uniformly and re‑clip
            h = np.clip(h + diff / n, 0.0, 1.0)
        else:
            adjust = diff / cnt
            h[free] += adjust
            h = np.clip(h, 0.0, 1.0)
    return h
