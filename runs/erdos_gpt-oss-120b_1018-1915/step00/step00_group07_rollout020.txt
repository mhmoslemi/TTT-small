import time
import numpy as np

def _objective(h, dx):
    """Return max overlap C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h, target_sum):
    """Clip to [0,1] and enforce exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    # iterative proportional scaling with re‑clipping
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        if s == 0.0:
            h = np.full_like(h, target_sum / h.size)
        else:
            h = h * (target_sum / s)
        h = np.clip(h, 0.0, 1.0)
    # final fine‑tune (distribute tiny residual uniformly)
    s = h.sum()
    if s != 0.0:
        h = h * (target_sum / s)
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a feasible step function h, its C5 bound, and number of sample points.
    """
    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size
    try:
        n_points = len(initial_h_values)      # provided by harness
        h0 = np.asarray(initial_h_values, dtype=float)
    except Exception:
        # fallback: use 80 points with uniform distribution
        n_points = 80
        h0 = np.full(n_points, 0.5)  # will be repaired below

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure starting vector is feasible
    best_h = _repair(h0, target_sum)
    best_score = _objective(best_h, dx)

    # Parameters for stochastic search
    noise_scale = 0.15          # magnitude of random perturbations
    restart_prob = 0.02        # chance to try a fresh random vector
    iteration = 0

    while True:
        if time.time() - start_time > budget_s * 0.99:
            break
        iteration += 1

        # Occasionally restart from a random feasible point
        if np.random.rand() < restart_prob:
            cand = np.random.rand(n_points)
        else:
            # Perturb current best solution
            cand = best_h + noise_scale * np.random.randn(n_points)

        cand = _repair(cand, target_sum)
        cand_score = _objective(cand, dx)

        if cand_score < best_score:
            best_h, best_score = cand, cand_score
            # Gradually reduce perturbation size to fine‑tune
            noise_scale = max(0.01, noise_scale * 0.99)

    return best_h, best_score, n_points
