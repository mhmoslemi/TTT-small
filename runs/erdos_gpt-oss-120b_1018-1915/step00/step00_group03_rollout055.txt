import time
import numpy as np

def _repair(h, target_sum):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) == 0:
            break
        delta = diff / len(idx)
        h[idx] += delta
        h = np.clip(h, 0.0, 1.0)
    # fix any tiny residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        for _ in range(100):
            j = np.random.randint(len(h))
            if residual > 0 and h[j] < 1.0:
                add = min(residual, 1.0 - h[j])
                h[j] += add
                residual -= add
            elif residual < 0 and h[j] > 0.0:
                sub = min(-residual, h[j])
                h[j] -= sub
                residual += sub
            if abs(residual) < 1e-12:
                break
    return h

def _c5_value(h):
    """Compute the overlap constant C5 for a given step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
