import time
import numpy as np

def _project_onto_constraints(h):
    """Clip to [0,1] and enforce sum(h) = len(h)/2."""
    n = h.shape[0]
    desired = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative redistribution
    for _ in range(10):
        s = h.sum()
        diff = desired - s
        if abs(diff) < 1e-10:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h):
    """Return the overlap constant for a given discrete step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)

def _interpolate_h(old_h, new_n):
    """Linear interpolation of a discrete function to a new resolution."""
    old_n = old_h.shape[0]
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    return _project_onto_constraints(new_h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h:[0,2]->[0,1] to minimise the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best found discrete representation of h.
    c5_bound : float
        Corresponding upper bound C5.
    n_points : int
        Number of samples (length of h_values).
    """
    rnd = np.random.RandomState(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # a simple feasible start: first half 1, second half 0
        n_default = 96
        h0 = np.zeros(n_default, dtype=float)
        h0[: n_default // 2] = 1.0
        h0 = _project_onto_constraints(h0)

    n = h0.shape[0]

    # Optional upscale if time permits (up to ~400 points)
    target_n = min(400, max(200, n))
    if target_n != n:
        h0 = _interpolate_h(h0, target_n)
        n = target_n

    best_h = _project_onto_constraints(h0.copy())
    best_score = _c5_value(best_h)

    # ------------------------------------------------------------------
    # 2. Stochastic local search (simulated annealing style)
    # ------------------------------------------------------------------
    max_iters = 10_000_000  # safety cap; loop breaks on time.
    sigma_start = 0.15
    sigma_end = 0.001
    temp_start = 0.02
    temp_end = 0.0002

    for it in range(max_iters):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # schedule
        frac = it / max_iters
        sigma = sigma_start * (sigma_end / sigma_start) ** frac
        temp = temp_start * (temp_end / temp_start) ** frac

        # propose
        proposal = best_h + rnd.normal(scale=sigma, size=n)
        proposal = _project_onto_constraints(proposal)

        score = _c5_value(proposal)

        # accept if better, otherwise with prob. exp(-(Δ)/temp)
        if score < best_score:
            best_h, best_score = proposal, score
        else:
            delta = score - best_score
            if rnd.rand() < np.exp(-delta / max(temp, 1e-12)):
                best_h, best_score = proposal, score

    # ------------------------------------------------------------------
    # 3. Final polishing with smaller perturbations
    # ------------------------------------------------------------------
    for it in range(2000):

        proposal = _project_onto_constraints(proposal)
        score = _c5_value(proposal)
        if score < best_score:
            best_h, best_score = proposal, score

    return best_h, float(best_score), int(n)
