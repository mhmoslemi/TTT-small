import time
import numpy as np

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_sum(h, target):
    """Adjust h minimally to make sum exactly target while staying in [0,1]."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # distribute diff uniformly, then clip and repeat a few times
    for _ in range(5):
        h += diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize C5.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h at the discretisation points.
    c5_bound : float
        The achieved C5 value.
    n_points : int
        Number of discretisation points.
    """
    rng = np.random.default_rng(seed)

    # Determine dimension
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = kwargs.get('n_points', 70)
        h0 = rng.random(n_points)

    # Ensure feasibility
    target_sum = n_points / 2.0
    h = h0.copy()
    h = np.clip(h, 0.0, 1.0)
    h = project_sum(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    start = time.time()
    elapsed = lambda: time.time() - start
    time_limit = budget_s - 0.5  # safety margin

    # Simulated‑annealing / coordinate‑swap scheme
    temp = 0.02          # initial temperature
    step_scale = 0.2     # initial step size as a fraction of allowed move
    min_temp = 1e-5
    while elapsed() < time_limit and temp > min_temp:
        # inner loop count adapts to remaining time
        inner_iters = min(2000, int(1e5 * (time_limit - elapsed()) / budget_s))
        for _ in range(inner_iters):
            i, j = rng.choice(n_points, size=2, replace=False)
            # maximum feasible delta that keeps both entries in [0,1]
            max_pos = min(h[i], 1.0 - h[j])
            max_neg = -min(h[j], 1.0 - h[i])
            if max_pos <= 0 and max_neg >= 0:
                continue
            delta = rng.uniform(max_neg, max_pos) * step_scale
            if abs(delta) < 1e-12:
                continue

            h_new[j] += delta
            val = compute_c5(h_new)

                h = h_new
                if improve:

                    best_val = val
        # cooling
        temp *= 0.985
        step_scale *= 0.990

    # final projection (should already be exact)
    best_h = project_sum(best_h, target_sum)
    best_val = compute_c5(best_h)

    return best_h, float(best_val), int(n_points)
