import time
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, target_sum):
    """Clip to [0,1] and renormalize to achieve the required sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        return np.full_like(h, target_sum / len(h))
    h = h * (target_sum / s)
    # second clipping to avoid tiny violations after scaling
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if not np.isclose(s, target_sum):
        # distribute the remaining discrepancy uniformly over non‑boundary entries
        diff = target_sum - s
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of discretisation points."""
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Initialise h
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h0 = project_to_constraints(h0, target)
    else:
        n = 100                     # default resolution if no parent is given
        h0 = np.full(n, 0.5)        # already satisfies the integral constraint
        target = n / 2.0

    dx = 2.0 / n
    best_h = h0.copy()
    best_c = compute_c5(best_h, dx)

    # ----------------------------------------------------------------------
    # Local search parameters
    # ----------------------------------------------------------------------
    max_delta = 0.05               # maximal mass moved in a single proposal
    restart_prob = 0.001          # probability of a random restart each iteration
    time_limit = budget_s * 0.95   # stop slightly before the allotted budget

    while time.time() - start_time < time_limit:
        # ----- propose a small redistribution between two positions -------------
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if best_h[i] <= 0.0 or best_h[j] >= 1.0:
            continue
        allowable = min(max_delta, best_h[i], 1.0 - best_h[j])
        if allowable <= 0.0:
            continue
        delta = np.random.rand() * allowable
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # (sum is preserved, bounds respected by construction)
        c_new = compute_c5(h_new, dx)
        if c_new < best_c:
            best_h, best_c = h_new, c_new

        # ----- occasional random restart ---------------------------------------
        if np.random.rand() < restart_prob:
            h_rand = np.full(n, 0.5)
            perturb = (np.random.rand(n) - 0.5) * 0.1
            h_rand += perturb
            h_rand = project_to_constraints(h_rand, target)
            c_rand = compute_c5(h_rand, dx)
            if c_rand < best_c:
                best_h, best_c = h_rand, c_rand

    # ----------------------------------------------------------------------
    # Return the best result found
    # ----------------------------------------------------------------------
    return (best_h.tolist(), float(best_c), int(n))
