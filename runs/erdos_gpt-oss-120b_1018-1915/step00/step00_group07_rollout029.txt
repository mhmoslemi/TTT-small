import time
import numpy as np

def _evaluate_c5(h):
    """Return C5 for a given h (numpy 1‑d array)."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    dx = 2.0 / h.size
    return corr.max() * dx

def _repair(h, target_sum):
    """Clip to [0,1] and rescale to meet the integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0.0:
        # give equal mass to random positions
        idx = np.random.choice(h.size, int(round(target_sum)), replace=False)
        h[idx] = 1.0
        return h
    h *= target_sum / s
    # second clipping may disturb the sum, do a tiny redistribution
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-8:
        diff = target_sum - s
        # spread the diff over a random index that is not at a bound
        free = np.where((h > 0) & (h < 1))[0]
        if free.size == 0:
            free = np.arange(h.size)
        i = np.random.choice(free)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h

def _hill_climb(start_h, target_sum, time_budget):
    """Simple hill‑climbing with pairwise mass transfers and occasional perturbations."""
    best = start_h.copy()
    best_score = _evaluate_c5(best)
    n = best.size
    t0 = time.time()

        improved = False
        # try a batch of swaps
