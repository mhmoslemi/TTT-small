import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_h(h, target_sum):
    """Clip to [0,1] and adjust to hit target_sum (preserving feasibility)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # all at bounds – scale everything uniformly
            h = h * (target_sum / s)
            h = np.clip(h, 0.0, 1.0)
    return h

def _propose_move(h, step):
    """Generate a feasible neighbour by transferring mass between two coordinates."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # decide direction randomly
    if np.random.rand() < 0.5:
        i, j = j, i
    max_inc = min(step, 1.0 - h[i])
    max_dec = min(step, h[j])
    if max_inc <= 0.0 or max_dec <= 0.0:
        return None
    delta = np.random.rand() * min(max_inc, max_dec)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def _random_perturb(h, magnitude, target_sum):
    """Add zero‑mean noise and re‑project to feasibility."""
    noise = (np.random.rand(h.size) - 0.5) * 2.0 * magnitude
    h_pert = h + noise - noise.mean()
    return _project_h(h_pert, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best step‑function values found (length n_points).
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of discretisation points (len(h_values)).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size and starting vector
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n_points = h.size
    else:
        # fallback: 80 points, half mass placed uniformly at value 0.5
        n_points = 80
        h = np.full(n_points, 0.5, dtype=float)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because sum(h)*dx == 1

    # Ensure feasibility of the initial vector
    h = _project_h(h, target_sum)

    best_h = h.copy()
    best_score = _compute_c5(best_h, dx)

    current_h = h.copy()
    current_score = best_score

    step_size = 0.05          # mass transferred in a single move
    no_improve = 0
    iteration = 0

    while time.time() - start_time < budget_s:
        iteration += 1

        # Attempt a local move
        cand = _propose_move(current_h, step_size)
        if cand is None:

        cand_score = _compute_c5(cand, dx)

        # Accept if strictly better
        if cand_score < current_score:
