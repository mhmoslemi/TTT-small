import time
import numpy as np

def evaluate_c5(h, dx):
    """Compute C5 for a given step function sample array."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def mutate(h, rng):
    """Return a new h obtained by moving a small amount between two entries,
    preserving the sum and staying in [0,1]."""
    n = h.size
    max_tries = 20
    for _ in range(max_tries):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        # delta added to i, subtracted from j
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue
        # pick a non‑zero delta
        delta = rng.uniform(delta_min, delta_max)
        if np.isclose(delta, 0.0):
            continue
        new_h = h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        return new_h
    # fallback: no viable move found, return original
    return h.copy()
