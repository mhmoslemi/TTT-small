import time
import numpy as np

def _evaluate(h, dx):
    """Return C5 for a given step function sample array h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _mutate(h, rng):
    """Perform a feasible two‑index transfer mutation preserving sum and bounds."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # feasible delta such that h[i] - delta ∈ [0,1] and h[j] + delta ∈ [0,1]
    delta_min = -min(h[j], 1.0 - h[i])
    delta_max = min(h[i], 1.0 - h[j])
    if delta_max <= delta_min:
        return h  # no feasible move
    delta = rng.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initialization -------------------------------------------------
    if 'initial_h_values' in globals():
        h_curr = np.asarray(initial_h_values, dtype=float)
        n_points = h_curr.size
    else:
        n_points = 45
        h_curr = np.full(n_points, 0.5, dtype=float)

    # enforce exact integral constraint sum(h) = n_points/2
    target_sum = n_points / 2.0
    cur_sum = h_curr.sum()
    if not np.isclose(cur_sum, target_sum):
        h_curr *= target_sum / cur_sum
    # clipping may disturb the sum slightly; correct with a tiny adjustment
    diff = target_sum - h_curr.sum()
    if abs(diff) > 1e-12:
        # add the residual to the first element without breaking bounds
        add = np.clip(diff, -h_curr[0], 1.0 - h_curr[0])
        h_curr[0] += add

    dx = 2.0 / n_points

    best_h = h_curr.copy()
    best_val = _evaluate(best_h, dx)
    curr_val = best_val

    # ----- simulated annealing --------------------------------------------
    temp = 1.0            # initial temperature
    cooling = 0.9995      # geometric cooling factor
    min_temp = 1e-6

    while True:
        if time.time() - start_time > budget_s:
            break
        # propose mutation
        h_new = _mutate(h_curr, rng)
        new_val = _evaluate(h_new, dx)

        # accept if better or by Metropolis criterion
        accept = False
        if new_val < curr_val:
            accept = True
        else:
            if temp > 0:
                prob = np.exp(-(new_val - curr_val) / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            h_curr = h_new
            curr_val = new_val
            if new_val < best_val:
                best_h = h_new.copy()
                best_val = new_val

        # temperature update
        temp = max(temp * cooling, min_temp)

    return best_h, best_val, n_points
