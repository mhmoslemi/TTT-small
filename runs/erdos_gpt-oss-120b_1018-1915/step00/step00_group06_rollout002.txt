import time
import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def compute_c5(h):
    """Return the C5 value for a step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def repair(h, desired_sum):
    """Clip to [0,1] and adjust to meet the exact integral constraint."""
    h = np.clip(h, 0.0, 1.0)
    # Simple redistribution of the sum error.
    for _ in range(20):
        diff = desired_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        add = diff / h.size
        h = np.clip(h + add, 0.0, 1.0)
    return h


def objective(h, n):
    """Objective for the optimizer: the C5 value."""
    return compute_c5(h)


def optimize_h(initial_h, max_iter=2000, ftol=1e-9):
    """Run a constrained SLSQP optimization starting from initial_h."""
    n = initial_h.size
    desired_sum = n / 2.0

    bounds = Bounds(0.0, 1.0)
    linear_cons = LinearConstraint(np.ones((1, n)), [desired_sum], [desired_sum])

    res = minimize(
        fun=objective,
        x0=initial_h,
        args=(n,),
        method="SLSQP",
        bounds=bounds,
        constraints=[linear_cons],
        options={"maxiter": max_iter, "ftol": ftol, "disp": False},
    )
    h_opt = repair(res.x, desired_sum)
    return h_opt, compute_c5(h_opt)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():

        n_points = h0.size
        # Ensure feasibility of the supplied parent
        h0 = repair(h0, n_points / 2.0)
    else:
        # Fallback: modest size, uniform 0.5 satisfies the integral exactly
        n_points = 200
        h0 = np.full(n_points, 0.5, dtype=np.float64)

    best_h = h0
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Optimization loop with a few random restarts
    # ------------------------------------------------------------------
    # Parameters governing the search

    time_per_restart = max(0.1, budget_s / (max_restarts + 1))

            break

        # Random perturbation of the current best

        cand = repair(best_h + perturb, n_points / 2.0)

        # Local SLSQP refinement
