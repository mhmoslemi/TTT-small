import numpy as np
import time

def _evaluate_c5(h, dx):
    """Return C5 = max_k integral h(x)*(1-h(x+k)) dx for discrete h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_feasible(h, target_sum):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    if current == 0:
        # fallback uniform fill
        h = np.full_like(h, target_sum / h.size)
        return h

    # Simple uniform correction, repeat a few times to settle within bounds
    for _ in range(5):
        diff = target_sum - current
        if abs(diff) < 1e-12:
            break
        # distribute only to interior (non‑saturated) entries if possible
        interior = (h > 0.0) & (h < 1.0)
        if interior.any():
            delta = diff / interior.sum()
            h[interior] += delta
        else:
            # all entries at bounds – distribute uniformly and re‑clip
            h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        current = h.sum()
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h sampled uniformly (length n_points).
    c5_bound : float
        Corresponding C5 value (max overlap integral).
    n_points : int
        Number of discretization points used.
    """
    rng = np.random.default_rng(seed)

    # ------------------- problem dimensions -------------------
    n_points = 70                     # start with the provided size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0       # because integral must be 1

    # ------------------- initial candidate -------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.array(init, dtype=float)
        if h.size != n_points:
            # resize by simple interpolation if size mismatch
            x_old = np.linspace(0, 2, h.size, endpoint=False)
            x_new = np.linspace(0, 2, n_points, endpoint=False)
            h = np.interp(x_new, x_old, h)
    else:
        h = np.full(n_points, 0.5, dtype=float)  # uniform feasible start

    h = _project_to_feasible(h, target_sum)
    best_h = h.copy()
    best_c = _evaluate_c5(best_h, dx)

    start_time = time.time()
    time_limit = budget_s * 0.95  # leave a small safety margin

    # ------------------- main stochastic search -------------------
    # Parameters controlling mutation size and frequency
    mutation_frac = 0.07
    base_scale = 0.12

    while time.time() - start_time < time_limit:
        # Random perturbation of a subset of entries
        h_candidate = best_h.copy()
        k = max(1, int(mutation_frac * n_points))
        idx = rng.choice(n_points, k, replace=False)
        noise = rng.normal(scale=base_scale, size=k)
        h_candidate[idx] += noise

        # Project back to the feasible region
        h_candidate = _project_to_feasible(h_candidate, target_sum)

        # Evaluate
        cand_c = _evaluate_c5(h_candidate, dx)

        # Accept if improvement
        if cand_c < best_c:
            best_c = cand_c
            best_h = h_candidate
            # occasionally tighten mutation scale after good steps
            base_scale = max(0.02, base_scale * 0.98)

        # Occasionally perform a targeted adjustment using the worst shift
        if rng.random() < 0.1:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            s_idx = np.argmax(corr)                # index of worst shift
            shift = s_idx - (n_points - 1)          # convert to offset
            if shift != 0:
                # Gradient‑like tweak: reduce product h[i]*h[i+shift]
                grad = np.zeros_like(best_h)
                for i in range(n_points):

                h_candidate = best_h - step * grad

                cand_c2 = _evaluate_c5(h_candidate, dx)
                if cand_c2 < best_c:
                    best_c = cand_c2
                    best_h = h_candidate

    return best_h, float(best_c), n_points
