import numpy as np
import time
import math

def _compute_c5(h):
    """Return the C5 value for a step function h on [0,2] with uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h) over all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _mutate(h):
    """Randomly move a feasible amount of mass from one index to another."""
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while j == i:
        j = np.random.randint(0, n)
    # how much can we add to i and remove from j while staying in [0,1] ?
    max_add = 1.0 - h[i]
    max_sub = h[j]
    max_delta = min(max_add, max_sub)
    if max_delta <= 0.0:
        return h  # nothing to do
    delta = np.random.rand() * max_delta
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new


def _initialize_h(n_points):
    """Return a feasible starting vector (uniform 0.5)."""
    h = np.full(n_points, 0.5, dtype=np.float64)
    # enforce exact sum constraint (sum = n_points/2) – already satisfied.
    return h
