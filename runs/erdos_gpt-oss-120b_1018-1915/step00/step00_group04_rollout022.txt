import time
import numpy as np

def _c5_of(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _initialize_binary(n, rng):
    # exactly n/2 ones, n even
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Use provided construction if present
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.size
        # Ensure constraints: clip and adjust sum to n/2
        h = np.clip(h, 0.0, 1.0)
        target_sum = n / 2.0
        cur_sum = h.sum()
        if not np.isclose(cur_sum, target_sum):
            # Simple proportional scaling then rounding to binary while preserving sum
            if cur_sum == 0:
                h[:] = 0.0
            else:
                h *= target_sum / cur_sum
            # force binary by thresholding at 0.5; then fix count
            h = (h >= 0.5).astype(float)
            cur_ones = int(h.sum())
            diff = int(target_sum - cur_ones)
            if diff > 0:
                zero_idx = np.where(h == 0)[0]
                pick = rng.choice(zero_idx, size=diff, replace=False)
                h[pick] = 1.0
            elif diff < 0:
                one_idx = np.where(h == 1)[0]
                pick = rng.choice(one_idx, size=-diff, replace=False)
                h[pick] = 0.0
    else:
        # Choose a modest resolution
        n = 200  # even, <1000
        h = _initialize_binary(n, rng)

    dx = 2.0 / h.size
    best_h = h.copy()
    best_c5 = _c5_of(best_h, dx)

    # Temperature schedule
    temp = best_c5 * 0.2 + 1e-3
    min_temp = 1e-6
    decay = 0.9995

    # Pre‑compute index lists for fast swaps
    while time.time() - start_time < budget_s * 0.95:
        # keep at least one 0 and one 1
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break

        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)

        # propose swap
        h[i], h[j] = 0.0, 1.0
        new_c5 = _c5_of(h, dx)
        delta = new_c5 - best_c5

        accept = False
        if delta < 0:
            accept = True
        else:
            if temp > min_temp and rng.random() < np.exp(-delta / temp):
                accept = True

        if accept:
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1.0, 0.0

        # cool down
        temp = max(min_temp, temp * decay)

    # Final evaluation
    final_c5 = _c5_of(best_h, dx)
    return best_h, final_c5, best_h.size
