import time
import numpy as np
from scipy.optimize import minimize

# Helper: compute the C5 value for a candidate step vector
def _c5_value(h):
    g = 1.0 - h
    # full-mode correlation (same as np.correlate)
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * (2.0 / len(h))

# Project any vector onto the feasible set:
#   0 <= h[i] <= 1  and  sum(h) = n/2   (integral = 1)
def _project_to_feasible(v, target_sum):
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if s == 0.0:
        # fallback: set first target_sum entries to 1
        v[:int(round(target_sum))] = 1.0
        return v
    return v * (target_sum / s)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine number of sample points
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        # ensure it satisfies the sum constraint (it should)
        target_sum = n / 2.0
        h_cur = _project_to_feasible(init.copy(), target_sum)
    else:
        # default size (even) – 120 gives a fine resolution but stays <1k
        n = 120
        target_sum = n / 2.0
        # start with a simple half‑ones / half‑zeros block
        h_cur = np.zeros(n, dtype=float)
        h_cur[:int(target_sum)] = 1.0

    best_h = h_cur.copy()
    best_score = _c5_value(best_h)

    # ---------- Random binary search ----------
    # generate many binary vectors (exactly target_sum ones) and keep the best
    max_iter_random = 20000  # upper bound; loop will break on time
    while time.time() - start_time < budget_s - 5.0 and max_iter_random > 0:
        max_iter_random -= 1
        perm = np.random.permutation(n)
        cand = np.zeros(n, dtype=float)
        cand[perm[:int(target_sum)]] = 1.0

        # occasional small continuous perturbation to escape plateaus
        if np.random.rand() < 0.08:
            noise = np.random.uniform(-0.15, 0.15, size=n)
            cand = np.clip(cand + noise, 0.0, 1.0)
            cand = _project_to_feasible(cand, target_sum)

        score = _c5_value(cand)
        if score < best_score:
            best_score = score
            best_h = cand.copy()
            # keep current point for possible local refinement later
            h_cur = cand.copy()

    # ---------- Local continuous refinement (SLSQP) ----------
    # Use the best binary vector as a warm start; allow continuous values.
    remaining = budget_s - (time.time() - start_time)
    if remaining > 3.0:
        target_sum = n / 2.0

        def obj(x):
            # x is unconstrained; project inside objective to keep feasibility
            h = _project_to_feasible(x, target_sum)
            return _c5_value(h)

        # equality constraint to force the sum (handled by projection but kept for optimizer)
        cons = {"type": "eq", "fun": lambda x: _project_to_feasible(x, target_sum).sum() - target_sum}
        bounds = [(0.0, 1.0) for _ in range(n)]

        # SLSQP often converges quickly; limit iterations to stay within budget
        opt_res = minimize(
            obj,
            best_h,  # start from best binary candidate
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 500, "ftol": 1e-9, "disp": False},
        )
        if opt_res.success:
            refined = _project_to_feasible(opt_res.x, target_sum)
            refined_score = _c5_value(refined)
            if refined_score < best_score:
                best_score = refined_score
                best_h = refined.copy()

    # Final safety checks
    assert best_h.shape == (n,)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(best_h.sum() - n / 2.0) < 1e-8

    return best_h, float(best_score), int(n)
