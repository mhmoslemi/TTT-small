import time
import numpy as np

def _project_onto_capped_simplex(v, target, max_iter=60):
    """Project vector v onto {0<=x_i<=1, sum x_i = target} using binary search for lambda."""
    low, high = -1.0, 1.0
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target:
            low = lam
        else:
            high = lam
    return w

def _c5_value(h):
    """Compute C5 for discretized h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _initialize_h(n_points):
    """Create initial h using provided initial_h_values if present, otherwise a simple half‑interval."""
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.size
        if n0 == n_points:
            h = init.copy()
        else:
            # interpolate linearly onto new grid
            old_x = np.arange(n0, dtype=float) / n0 * 2.0
            new_x = np.arange(n_points, dtype=float) / n_points * 2.0
            h = np.interp(new_x, old_x, init, period=2.0)
    else:
        # simple step: first half ones, second half zeros
        h = np.zeros(n_points, dtype=float)
        h[: n_points // 2] = 1.0
    # enforce feasibility
    target = n_points / 2.0
    h = _project_onto_capped_simplex(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap bound C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    end_time = start_time + budget_s * 0.99  # slight safety margin

    # resolution (trade‑off between flexibility and speed)
    n_points = 250  # < 1k as requested
    target_sum = n_points / 2.0

    # initial candidate
    h_curr = _initialize_h(n_points)
    best_h = h_curr.copy()
    best_c5 = _c5_value(best_h)

    # simulated‑annealing parameters
    sigma = 0.12                # std. dev. of Gaussian proposal
    T0 = 0.12
    Tf = 1e-4
    max_iters = 200000         # safety upper bound
    it = 0

    while time.time() < end_time and it < max_iters:
        it += 1
        # exponential cooling
        t = it / max_iters
        T = T0 * (Tf / T0) ** t

        # propose new candidate
        proposal = h_curr + sigma * np.random.randn(n_points)
        proposal = _project_onto_capped_simplex(proposal, target_sum)

        cand_c5 = _c5_value(proposal)
        delta = cand_c5 - best_c5

        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            h_curr = proposal
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = proposal.copy()

        # occasional restart to avoid stagnation
        if it % 50000 == 0:
            h_curr = best_h.copy()
            sigma *= 0.9  # gradually reduce step size

    # final projection to guarantee feasibility (numeric safety)
    best_h = _project_onto_capped_simplex(best_h, target_sum)

    return best_h, float(best_c5), n_points
