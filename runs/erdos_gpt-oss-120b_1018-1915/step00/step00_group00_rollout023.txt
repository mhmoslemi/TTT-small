import time
import numpy as np
import random

def compute_c5(h):
    """Return the Erdős C5 value for a step function h on [0,2] sampled uniformly."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def neighbor_step(h, epsilon):
    """Generate a neighbor by moving a small amount epsilon from one index to another,
    preserving sum and bounds."""
    n = h.shape[0]
    i, j = random.sample(range(n), 2)
    # amount we can actually transfer without violating bounds
    max_move = min(epsilon, h[i])               # cannot take more than we have at i
    max_move = min(max_move, 1.0 - h[j])        # cannot overflow at j
    if max_move <= 0.0:
        return None
    new_h = h.copy()
    new_h[i] -= max_move
    new_h[j] += max_move
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h = 1 that minimizes C5.
    Returns (h_values, c5_bound, n_points)."""
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 60                     # modest size when no parent is supplied
        h0 = np.full(n, 0.5)       # uniform satisfies the integral constraint

    target_sum = n / 2.0
    # Ensure the starting vector respects the constraints (might be slightly off due to float)
    diff = target_sum - h0.sum()
    if abs(diff) > 1e-12:
        # Shift the first element to fix the sum while staying in bounds
        adj = min(max(diff, -h0[0]), 1.0 - h0[0])
        h0[0] += adj

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    start_time = time.time()
    # Parameters for the annealing loop
    max_elapsed = budget_s * 0.95      # leave a small safety margin
    epsilon = 0.02                     # typical transfer amount
    init_temp = 0.1

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed >= max_elapsed:
            break

        # Temperature decays linearly to near zero
        temp = init_temp * max(0.0, 1.0 - elapsed / max_elapsed)

        # Propose a neighbor
        cand = neighbor_step(best_h, epsilon)
        if cand is None:
            continue
        cand_score = compute_c5(cand)

        # Accept if improvement or according to Metropolis rule
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score
        else:
            # Probabilistic acceptance
            delta = cand_score - best_score
            if temp > 0 and random.random() < np.exp(-delta / temp):
                best_h = cand
                best_score = cand_score

        # Occasionally shrink epsilon to fine‑tune
        if elapsed % 10 < 0.01:  # every ~10 seconds
            epsilon = max(0.001, epsilon * 0.9)

        # Occasionally random restart if stuck

            idx = random.randrange(n)
            perturb = (random.random() - 0.5) * 0.1
            new_val = np.clip(best_h[idx] + perturb, 0.0, 1.0)
