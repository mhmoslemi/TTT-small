import time
import numpy as np

def evaluate_c5(h, dx):
    """Return C5 = max correlation * dx for given h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def random_swap(h):
    """Return a copy of h with two positions swapped."""
    i, j = np.random.choice(len(h), size=2, replace=False)
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def perturb_pair(h, eps=0.01):
    """Add eps to one entry and subtract from another, respecting bounds."""
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    delta = (np.random.rand() * 2 - 1) * eps  # in [-eps, eps]
    # propose new values
    a_new = h[i] + delta
    b_new = h[j] - delta
    if 0.0 <= a_new <= 1.0 and 0.0 <= b_new <= 1.0:
        h_new = h.copy()
        h_new[i] = a_new
        h_new[j] = b_new
        return h_new
    return None  # invalid move

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure feasibility (clip and renormalize if needed)
    h_current = np.clip(h_current, 0.0, 1.0)
    cur_sum = h_current.sum()
    if not np.isclose(cur_sum, target_sum):
        # simple scaling to meet sum while staying in [0,1]
        scale = target_sum / cur_sum
        h_current = np.clip(h_current * scale, 0.0, 1.0)
        # adjust residual mass by adding to first element within bounds
        residual = target_sum - h_current.sum()
        for idx in range(n_points):
            add = min(1.0 - h_current[idx], residual)
            h_current[idx] += add
            residual -= add
            if abs(residual) < 1e-12:

    best_h = h_current.copy()
    best_c5 = evaluate_c5(best_h, dx)
