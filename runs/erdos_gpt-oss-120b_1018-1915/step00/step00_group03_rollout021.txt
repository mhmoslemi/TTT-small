import numpy as np
import time

def _evaluate_c5(h, dx):
    """Return max overlap C5 for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _repair(h, target_sum, max_iter=10, tol=1e-8):
    """Clip to [0,1] and enforce sum == target_sum (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        if abs(s - target_sum) <= tol:
            break
        # Scale uniformly then re‑clip
        factor = target_sum / s if s != 0 else 0.0
        h *= factor
        h = np.clip(h, 0.0, 1.0)
    # Final tiny adjustment on a random index if needed
    diff = target_sum - h.sum()
    if abs(diff) > tol:
        idx = np.random.randint(len(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős minimum‑overlap step function.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h on the grid.
    c5_bound : float
        Corresponding C5 upper bound (smaller is better).
    n_points : int
        Number of discretisation points.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Set up problem size and initial candidate
    # ------------------------------------------------------------------
    try:
        # use the provided initial construction if present
        h0 = np.array(initial_h_values, copy=True)
        n_points = h0.size
    except NameError:
        # fall back to a simple uniform start
        n_points = 70
        h0 = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0               # because ∫ h = 1 → sum(h)*dx = 1

    # ensure feasibility of the starting vector
    h0 = _repair(h0, target_sum)

    best_h = h0
    best_score = _evaluate_c5(best_h, dx)

    # ------------------------------------------------------------------
    # 2. Simple simulated‑annealing / random‑walk search
    # ------------------------------------------------------------------
    # parameters (tuned heuristically)
    temperature = 0.2
    decay = 0.9995
    sigma = 0.1

    iteration = 0
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1

        # propose a neighbour by perturbing a few coordinates
        h_candidate = best_h.copy()
        # number of coordinates to touch (1‑3)
        n_touch = np.random.randint(1, 4)
        idxs = np.random.choice(n_points, size=n_touch, replace=False)
        deltas = np.random.normal(scale=sigma, size=n_touch)
        h_candidate[idxs] += deltas
        # repair bounds and the integral constraint
        h_candidate = _repair(h_candidate, target_sum)

        cand_score = _evaluate_c5(h_candidate, dx)

        # accept if better, else probabilistically
        if cand_score < best_score:
            best_h, best_score = h_candidate, cand_score
        else:
            prob = np.exp(-(cand_score - best_score) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                best_h, best_score = h_candidate, cand_score

        # temperature schedule
        temperature *= decay
        sigma *= decay

        # occasional random restart to avoid stagnation
        if iteration % 5000 == 0:

            restart = _repair(restart, target_sum)
            restart_score = _evaluate_c5(restart, dx)
            if restart_score < best_score:
                best_h, best_score = restart, restart_score

    # ------------------------------------------------------------------
    # 3. Return the best found solution
    # ------------------------------------------------------------------
    return best_h, float(best_score), n_points
