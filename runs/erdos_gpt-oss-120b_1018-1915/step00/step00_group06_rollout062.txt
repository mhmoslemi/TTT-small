import time
import numpy as np
from math import exp

def _evaluate(h):
    """Return C5 bound for array h (values in [0,1])."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _project(h, desired_sum):
    """Clip to [0,1] and adjust to satisfy sum == desired_sum."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        # avoid division by zero, distribute uniformly
        h = np.full_like(h, desired_sum / h.size)
        return np.clip(h, 0.0, 1.0)
    # simple affine scaling, then clip again
    scale = desired_sum / total
    h = h * scale
    h = np.clip(h, 0.0, 1.0)
    # second correction if needed
    total2 = h.sum()
    if abs(total2 - desired_sum) > 1e-8:
        # distribute the difference uniformly (tiny adjustments)
        diff = desired_sum - total2
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _gradient_step(h, step):
    """One projected gradient step for the max‑overlap objective."""
    n = h.size
    # correlation to find worst shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    k = np.argmax(corr) - (n - 1)          # shift index (can be negative)
    # gradient: g_j = 1 - h_{j+k} - h_{j-k}
    idx_plus = (np.arange(n) + k) % n
    idx_minus = (np.arange(n) - k) % n
    grad = 1.0 - h[idx_plus] - h[idx_minus]
    h_new = h - step * grad
    return h_new
