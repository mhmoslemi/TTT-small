import numpy as np
import time

def _c5_of_h(h, dx):
    """Compute the C5 value for a given step function vector h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _feasible_projection(h, target_sum):
    """
    Project a vector onto the simplex {x>=0, sum(x)=target_sum}
    and clip to [0,1]. Simple iterative scaling works for our small,
    well‑behaved vectors.
    """
    # clip to [0,1] first
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) < 1e-12:
        return h
    # scale uniformly and re‑clip until sum matches (few iterations suffice)
    for _ in range(10):
        if s == 0:
            h = np.full_like(h, target_sum / len(h))
            break
        h *= target_sum / s
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 constant.

    Returns
    -------
    h_values : list of float
        The optimized h sampled at n_points points.
    c5_bound : float
        The achieved upper bound C5.
    n_points : int
        Number of discretization points (length of h_values).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem setup
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.asarray(init, dtype=float).copy()
        n_points = h.shape[0]
    else:
        n_points = 80
        # start from a Dirichlet sample that respects the integral constraint
        raw = rng.gamma(1.0, size=n_points)
        h = raw / raw.sum() * (n_points / 2.0)   # because sum(h)=n_points/2
        h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure feasibility of the initial point
    h = _feasible_projection(h, target_sum)

    best_h = h.copy()
    best_score = _c5_of_h(best_h, dx)

    # ------------------------------------------------------------------
    # 2. Simulated annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    start_time = time.time()
    end_time = start_time + budget_s * 0.95  # safety margin

    # Annealing schedule
    T0 = 0.02
    Tf = 1e-5
    max_iters = 200000

    for it in range(max_iters):
        now = time.time()
        if now >= end_time:
            break

        # temperature linearly decays in log‑space
        progress = (now - start_time) / (end_time - start_time)
        T = np.exp(np.log(T0) * (1 - progress) + np.log(Tf) * progress)

        # propose a move: transfer mass between two random indices
        i, j = rng.integers(0, n_points, size=2)
        if i == j:
            continue

        # maximum transferable amount respecting bounds
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            continue

        # sample a fraction of that amount
        frac = rng.random() * 0.5  # up to 50% of the allowable transfer
        delta = max_delta * frac

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # project back to feasibility (maintains sum and clips)
        h_new = _feasible_projection(h_new, target_sum)

        # evaluate
        score_new = _c5_of_h(h_new, dx)

        # accept if better, or with Metropolis probability
        if score_new < best_score:
            best_h = h_new.copy()
            best_score = score_new
        if score_new < _c5_of_h(h, dx) or rng.random() < np.exp(-(score_new - _c5_of_h(h, dx)) / T):
            h = h_new

    # ------------------------------------------------------------------
    # 3. Return result
    # ------------------------------------------------------------------
    return best_h.tolist(), float(best_score), int(n_points)
