        bnds = [(0.0,1.0)] * n

def run(seed=42, budget_s=1100, **kwargs):
    import time, math
    import numpy as np
    from scipy.optimize import minimize

    # Optional cvxpy for a feasible start when needed
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        cp = None

    start_time = time.time()
    rnd = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # Helper: compute the Erdős overlap constant for a discrete h
    # ------------------------------------------------------------------
    def score_vec(h):
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        return np.max(corr)

    # ------------------------------------------------------------------
    # Helper: produce a feasible vector (0<=h_i<=1, sum = n/2)
    # ------------------------------------------------------------------
    def feasible_start(n, seed_local):
        # try to use cvxpy if available, otherwise a simple projection loop
        rnd_local = np.random.RandomState(seed_local)
        x0 = rnd_local.rand(n)

        if cp is not None:
            x = cp.Variable(n)
            constraints = [x >= 0, x <= 1, cp.sum(x) == n / 2.0]
            prob = cp.Problem(cp.Minimize(cp.sum_squares(x - x0)), constraints)
            try:
                prob.solve(solver=cp.SCS, eps=1e-5, verbose=False)
                if x.value is not None:
                    return np.asarray(x.value).flatten()
            except Exception:  # pragma: no cover
                pass

        # fallback: simple iterative clipping + global shift
        y = np.clip(x0, 0.0, 1.0)
        for _ in range(10):
            cur = y.sum()
            delta = (n / 2.0 - cur) / n
            y = np.clip(y + delta, 0.0, 1.0)
        return y

    # ------------------------------------------------------------------
    # Acquire an initial vector (prefer the provided one)
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        # ensure feasibility (tiny numerical drift)
        n0 = h0.shape[0]
        cur = h0.sum()
        if not (0 <= h0).all() or not (h0 <= 1).all() or abs(cur - n0 / 2.0) > 1e-8:
            # project onto feasible set using the helper
            h0 = feasible_start(n0, seed + 1)
    except NameError:
        # no supplied construction – start from a random feasible point
        n0 = 45
        h0 = feasible_start(n0, seed)

    best_h, best_score, best_n = None, math.inf, None

    # ------------------------------------------------------------------
    # Core optimizer (SLSQP respects bounds & equality)
    # ------------------------------------------------------------------
    def optimize(h_start):
        n = h_start.shape[0]

        cons = {"type": "eq", "fun": lambda x: x.sum() - n / 2.0}
        bnds = [(0.0, 1.0)] * n

        res = minimize(
            lambda x: score_vec(x),
            h_start,
            method="SLSQP",
            bounds=bnds,
            constraints=[cons],
            options={"maxiter": 250, "ftol": 1e-7, "disp": False},
        )

        if not res.success:
            # simple repair of the result
            x = np.clip(res.x, 0.0, 1.0)
            # enforce the sum by a uniform shift (iterated)
            for _ in range(5):
                delta = (n / 2.0 - x.sum()) / n
                x = np.clip(x + delta, 0.0, 1.0)
            return x, score_vec(x)

        return res.x, res.fun

    # ------------------------------------------------------------------
    # Multi‑resolution search (doubling grid size)
    # ------------------------------------------------------------------
    h_curr = h0.copy()
    n_curr = h0.shape[0]

    while True:
        # optimise at current resolution
        h_opt, sc = optimize(h_curr)

        if sc < best_score:
            best_score, best_h, best_n = sc, h_opt.copy(), n_curr

        # time check
        if time.time() - start_time > budget_s - 5:

        # decide whether to double resolution
        if n_curr * 2 > 2000:  # keep size modest for speed

        h_curr, n_curr = h_next, n_next

    # return the best discovered construction
    return best_h, best_score, best_n
