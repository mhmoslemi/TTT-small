import time
import numpy as np
import random

def _c5_score(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project(h, target_sum, iters=10):
    """Project vector onto [0,1] box and exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(iters):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        if not free.any():
            # all at bounds – distribute uniformly (will be re‑clipped)
            h += diff / h.size
        else:
            adjust = diff / free.sum()
            h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) after a timed local search."""
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------
    # initialise h
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # fallback: binary step function with exactly half the mass
        n = 200
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0
        np.random.shuffle(h)

    n = h.shape[0]
    target_sum = n / 2.0                     # because ∫h = 1 and dx = 2/n
    h = _project(h, target_sum)              # ensure feasibility

    best_h = h.copy()
    best_score = _c5_score(best_h)

    # ----------------------------------------------------------------------
    # simulated‑annealing style local search
    # ----------------------------------------------------------------------
    start = time.time()
    temp = 0.02                               # initial temperature
    decay = 0.9999
    max_iter = 10 ** 7

    for iteration in range(max_iter):
        if time.time() - start > budget_s * 0.95:
            break

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)

        # decide direction of transfer
        if random.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        max_transfer = min(best_h[src], 1.0 - best_h[dst])
        if max_transfer <= 0.0:
            continue

        # random proportion of the allowable transfer (up to 50%)
        eps = max_transfer * np.random.rand() * 0.5
        if eps < 1e-12:
            continue

        h_candidate = best_h.copy()
        h_candidate[src] -= eps
        h_candidate[dst] += eps

        # quick feasibility check (box already satisfied)
        score = _c5_score(h_candidate)

        # accept if better or Metropolis criterion
        improve = score < best_score
        if improve or np.random.rand() < np.exp((best_score - score) / max(temp, 1e-12)):
            best_h = h_candidate
            best_score = score

        # temperature schedule
        temp *= decay

        # occasional global perturbation to avoid stagnation
        if iteration % 5000 == 0 and iteration > 0:
            # pick a small random block and shuffle its values
            block_len = max(2, int(n * 0.03))
            start_idx = np.random.randint(0, n - block_len + 1)
            block = best_h[start_idx:start_idx + block_len].copy()
            np.random.shuffle(block)
            best_h[start_idx:start_idx + block_len] = block
            best_h = _project(best_h, target_sum)   # re‑project
            best_score = _c5_score(best_h)

    # final projection to guarantee constraints
    best_h = _project(best_h, target_sum)
    final_score = _c5_score(best_h)

    return best_h, final_score, n
