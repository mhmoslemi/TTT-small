import time
import numpy as np

def _c5_score(h):
    """Return C5 for the discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _local_search(h, time_budget):
    """Simple transfer‑based hill‑climbing respecting 0‑1 bounds and sum."""
    n = h.size
    best_h = h.copy()
    best_score = _c5_score(best_h)
    start = time.time()
    # parameters
    max_iter = 10_000_000  # safety upper bound
    while time.time() - start < time_budget and max_iter:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # maximum amount we can move from j to i while staying in [0,1]
        max_move = min(1.0 - best_h[i], best_h[j])
        if max_move <= 0.0:
            continue
        # propose a random fraction of that amount
        delta = np.random.rand() * max_move * 0.5
        cand = best_h.copy()
        cand[i] += delta
        cand[j] -= delta
        sc = _c5_score(cand)
        if sc < best_score:
            best_h, best_score = cand, sc
        max_iter -= 1
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C₅ bound."""
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Obtain an initial feasible vector
    # ------------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
    except NameError:
        n0 = 45
        h0 = np.full(n0, 0.5, dtype=np.float64)

    n0 = h0.size
    target_sum = n0 / 2.0
    cur_sum = h0.sum()
    if not np.isclose(cur_sum, target_sum, atol=1e-9):
        # scale uniformly then clip to keep feasibility
        h0 = h0 * (target_sum / cur_sum)
        h0 = np.clip(h0, 0.0, 1.0)
        # simple correction for any remaining deficit/excess
        deficit = target_sum - h0.sum()
        if abs(deficit) > 1e-12:
            free = (h0 > 0.0) & (h0 < 1.0)
            if free.any():
                adjust = deficit / free.sum()
                h0[free] += adjust
                h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0
    best_score = _c5_score(best_h)

    start_total = time.time()

    # ------------------------------------------------------------------
    # 2. Define resolution ladder (doubling up to a reasonable size)
    # ------------------------------------------------------------------
    resolutions = [best_h.size]
    max_len = 720  # keep evaluation cheap
    while resolutions[-1] * 2 <= max_len:
        resolutions.append(resolutions[-1] * 2)

    # ------------------------------------------------------------------
    # 3. Iterate over resolutions, performing local search at each level
    # ------------------------------------------------------------------
    for level, n in enumerate(resolutions):
        # if we moved to a larger grid, repeat values (preserves sum)
        if best_h.size != n:
            best_h = np.repeat(best_h, 2)

        # allocate remaining time fairly among remaining levels
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining <= 0:
            break
        alloc = remaining / (len(resolutions) - level)

        # run hill‑climbing within allocated slice of the budget
        best_h, best_score = _local_search(best_h, alloc)

    # ------------------------------------------------------------------
    # 4. Return the best solution found
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_score), best_h.size)
