import time
import numpy as np

def _project_to_capped_simplex(v, target_sum):
    """Project vector v onto {x: 0<=x<=1, sum x = target_sum}."""
    # Simple iterative projection: clip, scale, then fix residual.
    x = np.clip(v, 0.0, 1.0)
    cur = x.sum()
    if abs(cur - target_sum) < 1e-9:
        return x
    # Scale towards target while staying in bounds
    scale = target_sum / cur if cur > 0 else 0.0
    x = np.clip(x * scale, 0.0, 1.0)
    cur = x.sum()
    # Adjust residual by adding/subtracting from a non‑saturated coordinate
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # find indices that can absorb the diff
        if diff > 0:
            candidates = np.where(x < 1.0)[0]
            if candidates.size == 0:
                return x
            i = candidates[0]
            add = min(diff, 1.0 - x[i])
            x[i] += add
        else:
            candidates = np.where(x > 0.0)[0]
            if candidates.size == 0:
                return x
            i = candidates[0]
            sub = min(-diff, x[i])
            x[i] -= sub
    return x

def _evaluate_h(h, dx):
    """Return the C5 value for step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    Returns

        init_h[: n_points // 2] = 1.0

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because integral = 1

    init_h = _project_to_capped_simplex(init_h, target_sum)

    scores = []
    for _ in range(pop_size):
        cand = rng.random(n_points)
        cand = _project_to_capped_simplex(cand, target_sum)

        scores.append(_evaluate_h(cand, dx))
    population = np.array(population)

    population[0] = init_h

    best_score = float(scores[best_idx])

    generation = 0
    while True:
        generation += 1
        # time guard
