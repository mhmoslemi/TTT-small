def _project(h):
    """Clamp to [0,1] and enforce exact integral constraint sum(h)=len(h)/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Iteratively adjust by a uniform offset and reclamp.
    for _ in range(5):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        offset = (target - s) / n
        h += offset
        h = np.clip(h, 0.0, 1.0)
    # Tiny final correction to hit the target exactly (may push a few values slightly out of bounds)
    s = h.sum()
    if abs(s - target) > 1e-9:
        offset = (target - s) / n
        h += offset
    # final clamp to stay inside [0,1]
    h = np.clip(h, 0.0, 1.0)
    return h

def _local_search(h0, budget, start_time):
    """Simple stochastic descent from h0 within remaining budget."""
    n = h0.shape[0]
    best_h = h0.copy()
    best_c = evaluate_erdos_solution(best_h)

        cand = _project(cand)
        c_val = evaluate_erdos_solution(cand)
        if c_val < best_c:

            best_h = cand
    return best_h, best_c

def _double_resolution(h):
    """Interpolate h to double number of points preserving the integral."""
    n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, 2 * n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    h_new = _project(h_new)
    return h_new
