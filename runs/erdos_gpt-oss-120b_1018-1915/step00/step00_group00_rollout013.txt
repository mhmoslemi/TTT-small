import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _propose_move(h, max_step):
    """Generate a new feasible h by moving mass between two coordinates."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # amount to transfer from i to j (positive) or opposite (negative)
    delta_pos = min(max_step, h[i], 1.0 - h[j])
    delta_neg = -min(max_step, h[j], 1.0 - h[i])
    delta = np.random.uniform(delta_neg, delta_pos)
    if abs(delta) < 1e-12:
        return h  # no effective change
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
