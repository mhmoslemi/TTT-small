import time
import numpy as np

# Helper: evaluate C5 for a given step function h
def _c5_score(h):
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

# Helper: project a vector onto [0,1] box and enforce the exact sum constraint
def _project_sum_box(h, target):
    """Clip to [0,1] then adjust to achieve the exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):  # small fixed‑point iteration
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:                # need to increase some entries
            free = h < 1.0
            if not free.any():
                break
            avail = np.sum(1.0 - h[free])
            inc = diff * (1.0 - h[free]) / avail
            h[free] += inc
        else:                         # need to decrease some entries
            free = h > 0.0
            if not free.any():
                break
            avail = np.sum(h[free])
            dec = (-diff) * h[free] / avail
            h[free] -= dec
        h = np.clip(h, 0.0, 1.0)
    return h

# Helper: generate a perturbation of h by moving a small amount between two coordinates
def _perturb_transfer(h, rng, target_sum, max_step=0.05):
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    # amount that can be taken from i (respecting lower bound 0)
    max_take = min(max_step, h[i])
    if max_take <= 0.0:
        return h.copy()
    delta = rng.random() * max_take
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    h_new = _project_sum_box(h_new, target_sum)
    return h_new

# Helper: small Gaussian perturbation on a random subset
def _perturb_gauss(h, rng, target_sum, sigma=0.02, frac=0.1):
    n = h.shape[0]
    mask = rng.random(n) < frac
    if not mask.any():
        return h.copy()
    h_new = h.copy()
    h_new[mask] += rng.normal(0.0, sigma, size=mask.sum())
    h_new = _project_sum_box(h_new, target_sum)
    return h_new
