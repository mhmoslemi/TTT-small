import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _propose_swap(h, rng):
    """Generate a feasible neighbour by moving mass between two positions."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]

    # allowable delta so that both stay in [0,1]
    max_pos = min(1.0 - hi, hj)
    max_neg = -min(hi, 1.0 - hj)
    if max_pos <= max_neg:
        return None  # no feasible move

    delta = rng.uniform(max_neg, max_pos)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize the Erdős C₅ bound using simulated annealing.

    Returns
    -------
    h_best : np.ndarray
        Step‑function values (length n_points) achieving the best bound found.
    c5_best : float
        Corresponding C₅ upper bound.
    n_points : int
        Number of discretisation points (length of h_best).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # initialise discretisation
    # ------------------------------------------------------------
    n_points = 45                     # default resolution (given by initial_h_values)
    dx = 2.0 / n_points

    # start from provided initial_h_values when available
    try:
        h0 = np.array(initial_h_values, dtype=float)
        if h0.size != n_points:
            # fall back to uniform if size mismatches
            raise ValueError
    except Exception:
        # uniform distribution with required integral
        h0 = np.full(n_points, 0.5, dtype=float)

    # enforce exact sum constraint (might be slightly off due to rounding)
    target_sum = n_points / 2.0
    current_sum = h0.sum()
    if not np.isclose(current_sum, target_sum):
        # scale and then clip to keep within [0,1]; finally adjust one entry to hit exact sum
        h0 = np.clip(h0 * (target_sum / current_sum), 0.0, 1.0)
        diff = target_sum - h0.sum()
        # add the remaining mass to the first element that can accept it
        for idx in range(n_points):
            add = min(diff, 1.0 - h0[idx])
            if add > 0:
                h0[idx] += add
                diff -= add
                if np.isclose(diff, 0):
                    break

    # ------------------------------------------------------------
    # simulated annealing
    # ------------------------------------------------------------
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    current_h = best_h.copy()
    current_c5 = best_c5

    T0 = 1.0                # initial temperature
    cooling = 0.9995        # multiplicative cooling factor
    T = T0

    start_time = time.time()
    it = 0
    max_iters = 10_000_000   # safety upper bound

    while time.time() - start_time < budget_s and it < max_iters:
        it += 1
        # propose a neighbour
        h_candidate = _propose_swap(current_h, rng)
        if h_candidate is None:
            continue

        cand_c5 = _compute_c5(h_candidate, dx)

        # acceptance rule
        if cand_c5 < current_c5:
            accept = True
        else:
            # Metropolis criterion
            accept = rng.random() < np.exp(-(cand_c5 - current_c5) / max(T, 1e-12))

        if accept:
            current_h = h_candidate
            current_c5 = cand_c5
            # update best if improved
            if cand_c5 < best_c5:
                best_h = h_candidate.copy()
                best_c5 = cand_c5
                # early exit if target reached
                if best_c5 <= 0.3808:
                    break

        # temperature schedule
        T *= cooling

        # occasional re‑temperature reset to avoid stagnation
        if it % 50000 == 0:
            T = T0

    # ------------------------------------------------------------
    # final return
    # ------------------------------------------------------------
    return best_h, float(best_c5), n_points
