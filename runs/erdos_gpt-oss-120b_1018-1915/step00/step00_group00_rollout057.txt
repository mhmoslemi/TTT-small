def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial feasible step function
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, copy=True, dtype=float)
        n_points = h0.size
        h_curr = h0.copy()
    except Exception:  # initial_h_values not defined
        n_points = 45
        h_curr = np.full(n_points, 0.5, dtype=float)

    dx = 2.0 / n_points  # grid spacing, ensures sum(h)*dx == 1 when sum(h)=n/2

    # ------------------------------------------------------------------
    # Objective: maximal overlap integral C_5
    # ------------------------------------------------------------------
    def evaluate(arr):
        # correlation of h with (1-h); mode='full' yields all integer shifts
        return np.max(np.correlate(arr, 1.0 - arr, mode='full')) * dx

    cur_score = evaluate(h_curr)
    best_h = h_curr.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    temperature = 1.0
    cooling_rate = 0.9995
    iteration = 0

    while time.time() - start_time < budget_s:
        iteration += 1

        # ----- propose a feasible move ---------------------------------
        if rng.random() < 0.5:
            # Swap two entries (preserves sum and bounds)
            i, j = rng.integers(n_points, size=2)
            if i == j:
                continue
            h_new = h_curr.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # Transfer a small amount δ from j to i while staying in [0,1]
            i, j = rng.integers(n_points, size=2)
            if i == j:
                continue
            hi, hj = h_curr[i], h_curr[j]
            max_inc = min(1.0 - hi, hj)          # how much i can increase
            max_dec = min(hi, 1.0 - hj)          # how much i can decrease
            if max_inc + max_dec == 0.0:
                continue
            delta = rng.uniform(-max_dec, max_inc) * 0.5  # damped step
            h_new = h_curr.copy()
            h_new[i] = hi + delta
            h_new[j] = hj - delta

        new_score = evaluate(h_new)

        # ----- Metropolis acceptance ------------------------------------
        delta_score = new_score - cur_score
        if delta_score < 0 or rng.random() < np.exp(-delta_score / (temperature + 1e-12)):
            h_curr = h_new
            cur_score = new_score
            if new_score < best_score:
                best_h = h_new.copy()
                best_score = new_score

        # ----- temperature update ---------------------------------------
        temperature *= cooling_rate

        # occasional mild restart to avoid stagnation
        if iteration % 20000 == 0:
            h_curr = best_h.copy()
            cur_score = evaluate(h_curr)

    return (best_h, float(best_score), n_points)
