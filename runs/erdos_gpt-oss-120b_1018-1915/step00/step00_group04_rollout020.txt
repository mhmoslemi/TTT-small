import numpy as np
import time
import math

def compute_c5(h):
    """Return the C5 value for a step function represented by array h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def repair_feasible(h):
    """Project h onto the feasible set 0<=h<=1 and sum(h)=n/2."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # simple iterative uniform adjustment
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-8:
            break
        h += diff / n
        h = np.clip(h, 0.0, 1.0)

    # if still off, do a binary‑search scaling
    s = h.sum()
    if abs(s - target) > 1e-6:
        lo, hi = 0.0, 2.0
        for _ in range(30):
            a = (lo + hi) / 2.0
            h2 = np.clip(a * h, 0.0, 1.0)
            s2 = h2.sum()
            if s2 < target:
                lo = a
            else:
                hi = a
        h = np.clip(a * h, 0.0, 1.0)
    return h


def random_feasible(n, rng):
    """Generate a random feasible vector of length n."""
    # start from uniform then project
    h = rng.random(n)
    return repair_feasible(h)


def upsample(h):
    """Double the resolution by linear interpolation over [0,2)."""
    n = len(h)
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n * 2, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return repair_feasible(h_new)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # parameters for the stochastic search
    step_scale = 0.05          # typical step size (fraction of [0,1])
    decay = 0.9995            # exponential decay of step_scale
    min_step = 0.001
    max_iters = 200000        # safety cap
    upsamples_done = False

    for iteration in range(max_iters):
        # time guard
        if time.time() - start_time > budget_s * 0.97:
            break

        # propose a move that preserves sum in expectation
        v = rng.standard_normal(n)
        v -= v.mean()
        eps = step_scale * rng.random()
        h_candidate = best_h + eps * v
