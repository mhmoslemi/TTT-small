def _project_feasible(h, target_sum):
    """Project h onto [0,1] box and linear equality sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative redistribution of surplus/deficit
    for _ in range(10):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        nfree = free.sum()
        if nfree == 0:  # all at bounds, distribute uniformly then re‑clip
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / nfree
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final correction if tiny residual remains
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # Add residual to a non‑saturated element if possible

            h += residual / len(h)
            h = np.clip(h, 0.0, 1.0)

def _c5_metric(h):
    """Compute C5 for a discretised h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _random_neighbor(h, step_scale):
    """Generate a nearby feasible vector by moving mass between two coordinates."""
    n = h.size
