import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for step values h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def _initial_h(n):
    """Provide a feasible starting vector."""
    # If a parent construction is present, use it (must have correct length)
    try:
        if len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
            # ensure feasibility in case of tiny drift
            h0 = np.clip(h0, 0.0, 1.0)
            cur_sum = h0.sum()
            target = n / 2.0
            if not np.isclose(cur_sum, target):
                # rescale linearly while keeping bounds
                scale = target / cur_sum
                h0 = np.clip(h0 * scale, 0.0, 1.0)
                # final adjustment to meet exact sum
                diff = target - h0.sum()
                if abs(diff) > 1e-12:
                    # distribute diff to a random index that can absorb it
                    idx = np.random.randint(n)
                    h0[idx] = np.clip(h0[idx] + diff, 0.0, 1.0)
            return h0
    except NameError:
        pass
    # fallback: uniform 0.5 satisfies sum=n/2
    return np.full(n, 0.5, dtype=float)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    np.random.seed(seed)
    start_time = time.time()
    n = 45                     # primary discretisation
    dx = 2.0 / n
    h = _initial_h(n)
    best_h = h.copy()
    best_c5 = _compute_c5(h, dx)

    # SA parameters
    T = 0.5                    # initial temperature
    cooling = 0.9995
    min_T = 1e-6
    it = 0

    while time.time() - start_time < budget_s:
        # propose a move: transfer eps from i to j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # max transferable while staying feasible
        eps_max = min(0.2, h[i], 1.0 - h[j])
        if eps_max <= 0:
            continue
        eps = np.random.rand() * eps_max
        h_new = h.copy()
        h_new[i] -= eps
        h_new[j] += eps

        new_c5 = _compute_c5(h_new, dx)

        # acceptance test
        if new_c5 < best_c5:
            best_h, best_c5 = h_new.copy(), new_c5
        if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / max(T, 1e-12)):
            h = h_new

        # temperature schedule
        T = max(min_T, T * cooling)
        it += 1

        # occasional occasional re‑randomisation to escape stagnation
        if it % 20000 == 0 and time.time() - start_time < budget_s:
            # perturb a few random entries
            for _ in range(5):
                a, b = np.random.randint(n, size=2)
                eps = min(0.1, h[a], 1.0 - h[b])
                if eps > 0:
                    h[a] -= eps
                    h[b] += eps

    # final sanity: ensure constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    target = n / 2.0
    if not np.isclose(cur_sum, target, atol=1e-9):
        # adjust a random coordinate
        idx = np.random.randint(n)
        best_h[idx] += target - cur_sum
        best_h[idx] = np.clip(best_h[idx], 0.0, 1.0)

    best_c5 = _compute_c5(best_h, dx)
    return best_h, float(best_c5), n
