def run(seed=42, budget_s=1100, **kwargs):
    import time, random, math
    import numpy as np

    random.seed(seed)
    np.random.seed(seed)

    # ---------- helper utilities ----------
    def compute_c5(h):
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    def project_bounds_and_sum(vec, target):
        """Iteratively clip to [0,1] and renormalise to hit the target sum."""
        v = vec.copy()
        for _ in range(8):
            v = np.clip(v, 0.0, 1.0)
            s = v.sum()
            if s == 0:
                v[:] = target / v.size
                break
            v *= target / s
        v = np.clip(v, 0.0, 1.0)
        return v

    def upscale(h, new_n):
        """Linear interpolation to a finer grid."""
        old_n = h.shape[0]
        old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
        new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
        new_h = np.interp(new_x, old_x, h)
        new_h = project_bounds_and_sum(new_h, new_n / 2.0)
        return new_h

    # ---------- initialise ----------
    if "initial_h_values" in globals():
        h_cur = np.array(initial_h_values, dtype=float)
    else:
        # start with a modest size if no parent is available
        n0 = 84
        h_cur = np.random.rand(n0)
        h_cur = project_bounds_and_sum(h_cur, n0 / 2.0)

    n = h_cur.shape[0]
    target_sum = n / 2.0
    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    # ---------- optimisation loop ----------
    while time.time() - start_time < budget_s * 0.97:
        elapsed = time.time() - start_time
        # gradually shrink mutation strength
        sigma = 0.12 * (1.0 - elapsed / budget_s) + 0.02

        proposal = best_h + np.random.normal(0.0, sigma, size=n)
        proposal = project_bounds_and_sum(proposal, target_sum)

        cand_c5 = compute_c5(proposal)
        if cand_c5 < best_c5:
            best_h = proposal
            best_c5 = cand_c5

        # occasional up‑sampling to a finer resolution
        if n < 512 and elapsed > budget_s * 0.3 and random.random() < 0.015:

    return (best_h.tolist(), float(final_c5), n)
