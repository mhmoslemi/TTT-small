import time, math
import numpy as np

def _compute_c5(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_transfer(h, rng):
    """Return a new h where a random amount is transferred from one index to another,
    preserving the sum and staying in [0,1]."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # amount to move from j to i
    max_inc = 1.0 - h[i]
    max_dec = h[j]
    max_delta = min(max_inc, max_dec)
    if max_delta <= 0.0:
        return h  # no feasible move
    delta = rng.random() * max_delta
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h
