def project_to_simplex_bounds(v, S, lo, hi):
    import cvxpy as cp
    n = v.shape[0]
    x = cp.Variable(n)
    objective = cp.Minimize(cp.sum_squares(x - v))
    constraints = [cp.sum(x) == S, x >= lo, x <= hi]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)

import time, math, numpy as np, cvxpy as cp

def _project_to_bounds_sum(v, target, lo=0.0, hi=1.0):
    """Euclidean projection of v onto {x: sum x = target, lo <= x <= hi}."""
    n = v.shape[0]
    x = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(x - v)),
                      [cp.sum(x) == target, x >= lo, x <= hi])
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    return np.asarray(x.value).reshape(v.shape)

def _compute_c5(h):
    """Maximum overlap integral for step function h on [0,2]."""
    n = h.shape[0]
    N = 2 * n
    fh = np.fft.fft(h, N)
    fg = np.fft.fft(1.0 - h, N)
    corr = np.fft.ifft(fh * np.conj(fg)).real
    return (2.0 / n) * np.max(corr)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # resolution (trade‑off between quality and speed)
    n_points = 200
    target_sum = n_points / 2.0

    # ------------------------------------------------------------------
    # initialise from the provided parent if it exists
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        x_parent = np.linspace(0.0, 2.0, len(parent), endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h0 = np.interp(x_new, x_parent, parent)
    else:
        h0 = np.full(n_points, 0.5)

    # enforce feasibility exactly
    h0 = _project_to_bounds_sum(h0, target_sum, 0.0, 1.0)

    best_h = h0.copy()
    best_val = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # Simulated annealing / stochastic hill climbing
    max_iters = 200000
    T0, T_end = 0.15, 1e-4
    max_delta = 0.12   # maximal amount moved between two coordinates

    h = best_h.copy()
    for it in range(max_iters):
        # time check
        if time.time() - start_time > budget_s * 0.94:
            break

        # temperature schedule (linear)
        t_frac = it / max_iters
        T = T0 * (1.0 - t_frac) + T_end * t_frac

        # propose a swap that preserves the total sum
        i, j = rng.integers(0, n_points, size=2)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 2.0 * max_delta
        hi = h[i] + delta
        hj = h[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h_new = h.copy()
        h_new[i] = hi
        h_new[j] = hj

        val_new = _compute_c5(h_new)

        # Metropolis acceptance
        if val_new < best_val or rng.random() < math.exp((best_val - val_new) / T):
            h = h_new
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()

    # final feasibility projection (numerical drift)
    best_h = _project_to_bounds_sum(best_h, target_sum, 0.0, 1.0)
    final_val = _compute_c5(best_h)

    return (best_h, float(final_val), n_points)
