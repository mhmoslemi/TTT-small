import numpy as np
import random
import time

# Helper: project a vector onto the box-simplex {0<=x_i<=1, sum x_i = target}
def _project_box_simplex(v, target):
    lower, upper = 0.0, 1.0
    x = np.clip(v, lower, upper)
    free = np.ones_like(v, dtype=bool)
    # Iteratively adjust the free components
    for _ in range(10 * len(v)):  # safe upper bound
        s = x.sum()
        if abs(s - target) < 1e-12:
            break
        delta = target - s
        free_idx = np.where(free)[0]
        if free_idx.size == 0:
            break
        add = delta / free_idx.size
        x[free_idx] += add
        # Re‑apply bounds and update free mask
        over = x > upper
        under = x < lower
        if over.any():
            x[over] = upper
            free[over] = False
        if under.any():
            x[under] = lower
            free[under] = False
    # Final clip to be safe
    return np.clip(x, lower, upper)

def _evaluate_c5(h, dx):
    # correlation as described in the task
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fallback: uniform 0.5 vector with 80 points
        n_points = 80
        h0 = np.full(n_points, 0.5, dtype=float)

    n_points = h0.size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because ∫h = 1 → sum(h)=n/2

    # Ensure initial vector satisfies constraints
    h0 = _project_box_simplex(h0, target_sum)

    # DE parameters
    pop_size = max(30, 4 * n_points // 10)   # modest population
    F = 0.5
    CR = 0.9
    max_generations = 2000  # generous upper bound; will stop by time

    # Initialise population
    pop = np.empty((pop_size, n_points), dtype=float)
    pop[0] = h0.copy()
    for i in range(1, pop_size):
        rand_vec = np.random.rand(n_points)
        pop[i] = _project_box_simplex(rand_vec, target_sum)

    # Fitness of initial population
    fitness = np.empty(pop_size, dtype=float)
    for i in range(pop_size):
        fitness[i] = _evaluate_c5(pop[i], dx)

    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_score = fitness[best_idx]

    generation = 0
    while generation < max_generations:
        # time guard
        if time.time() - start_time > budget_s * 0.98:
            break

        for i in range(pop_size):
            # mutation: select three distinct indices different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = random.sample(idxs, 3)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])

            # crossover (binomial)
            trial = np.empty_like(mutant)
            j_rand = random.randrange(n_points)
            for j in range(n_points):
                if random.random() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i][j]

            # repair to satisfy constraints
            trial = _project_box_simplex(trial, target_sum)

            # evaluate
            trial_score = _evaluate_c5(trial, dx)

            # selection
            if trial_score < fitness[i]:
                pop[i] = trial
                fitness[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()

        generation += 1

    # Final safeguard: ensure best_h respects constraints exactly
    best_h = _project_box_simplex(best_h, target_sum)
    best_score = _evaluate_c5(best_h, dx)

    return best_h, float(best_score), n_points
