import numpy as np
import time

def _project_to_capped_simplex(v, target, max_iter=20, tol=1e-9):
    """
    Project vector v onto the set {x | 0<=x<=1, sum(x)=target}.
    Simple iterative water-filling algorithm.
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) <= tol:
            break
        # indices that are not at a bound
        free = (x > 0.0) & (x < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # nothing can move; break
            break
        delta = -diff / n_free
        x[free] += delta
        # re‑clip
        x = np.clip(x, 0.0, 1.0)
    # final correction if still off by tiny amount
    if abs(x.sum() - target) > tol:
        # distribute remaining error uniformly on free variables
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += (target - x.sum()) / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def _repair(h, target_sum):
    """Bring h into [0,1] and enforce exact sum."""
    return _project_to_capped_simplex(h, target_sum)

def _evaluate(h, dx):
    """Maximum overlap value for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
