import time
import numpy as np

def _compute_c5(h, dx):
    """Maximum overlap integral for a discrete h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _enforce_constraints(h, target_sum):
    """Clip to [0,1] and adjust to hit the required sum."""
    h = np.clip(h, 0.0, 1.0)
    # iterative Dykstra-like projection
    for _ in range(5):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        nfree = free.sum()
        if nfree == 0:
            # all at bounds – distribute uniformly (will be re‑clipped)
            h += diff / len(h)
        else:
            h[free] += diff / nfree
        h = np.clip(h, 0.0, 1.0)
    return h
