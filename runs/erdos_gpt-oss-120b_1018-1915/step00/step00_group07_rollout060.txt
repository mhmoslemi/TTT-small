import numpy as np, time, random, math

def _repair(vec, target_sum):
    """Project vec onto [0,1] box and enforce sum == target_sum."""
    h = np.clip(vec, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        h[:] = target_sum / len(h)
        return h
    # scale to target sum
    h = h * (target_sum / total)
    # iterative clipping & redistribution
    for _ in range(5):
        over = h > 1.0
        if not np.any(over):
            break
        excess = h[over].sum() - over.sum()  # amount above 1 summed
        h[over] = 1.0
        below = h < 1.0
        if np.any(below):
            h[below] += excess / np.sum(below)
        h = np.clip(h, 0.0, 1.0)
    # final scaling to hit target precisely (within tolerance)
    total = h.sum()
    if not math.isclose(total, target_sum, rel_tol=1e-9, abs_tol=1e-12):
        h = h * (target_sum / total)
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_score(h, dx):
    """Compute C5 for a given h vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **_):
    """Return (best_h, best_score, n_points) within the time budget."""
    random.seed(seed)
    np.random.seed(seed)

    # ----- problem dimensions -----
    try:
        n_points = len(initial_h_values)          # supplied by harness
    except NameError:
        n_points = 80                               # fallback size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0                     # integral = 1

    # ----- initial population -----
    pop_size = max(30, n_points // 2)               # modest pop
    pop = np.empty((pop_size, n_points), dtype=float)
    scores = np.empty(pop_size, dtype=float)

    # first individual = provided construction (if any)
    try:
        pop[0] = _repair(initial_h_values, target_sum)
    except NameError:
        pop[0] = np.full(n_points, target_sum / n_points)
    scores[0] = _c5_score(pop[0], dx)

    # remaining individuals = random feasible vectors
    for i in range(1, pop_size):
        rand_vec = np.random.rand(n_points)
        pop[i] = _repair(rand_vec, target_sum)
        scores[i] = _c5_score(pop[i], dx)

    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_score = scores[best_idx]

    # ----- DE parameters -----
    F = 0.5
    CR = 0.9
    max_gen = 2000  # generous upper bound; will stop by time
    start_time = time.time()
    time_limit = budget_s * 0.95  # safety margin

    gen = 0
    while gen < max_gen and (time.time() - start_time) < time_limit:
        for i in range(pop_size):
            # mutation: select three distinct indices different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = random.sample(idxs, 3)
            mutant = pop[a] + F * (pop[b] - pop[c])

            # crossover (binomial)
            j_rand = random.randrange(n_points)
            trial = np.empty_like(mutant)
            for j in range(n_points):
                if random.random() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = pop[i, j]

            # repair to feasibility
            trial = _repair(trial, target_sum)

            # evaluate
            trial_score = _c5_score(trial, dx)

            # selection
            if trial_score <= scores[i]:
                pop[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_h = trial.copy()
        gen += 1
        # early exit if we already beat target bound
        if best_score < 0.3808:
            break

    return (best_h, float(best_score), n_points)
