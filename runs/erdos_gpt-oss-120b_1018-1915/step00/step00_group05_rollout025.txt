import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return C5 value for step function h on [0,2] with uniform grid."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)  # dx = 2/n


def sa_step(h: np.ndarray, cur_c5: float, target_sum: float, rng: np.random.Generator,
            temperature: float) -> tuple[np.ndarray, float]:
    """Perform one Metropolis step that conserves the sum exactly."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)

    # allowed delta so that 0<=h[i]+delta<=1 and 0<=h[j]-delta<=1
    delta_max = min(1.0 - h[i], h[j])
    delta_min = -min(h[i], 1.0 - h[j])
    if delta_max <= delta_min:
        return h, cur_c5  # no feasible move (should be rare)

    # propose a delta scaled by temperature
    delta = rng.uniform(delta_min, delta_max) * min(1.0, temperature / 0.1)
    if abs(delta) < 1e-12:
        return h, cur_c5

    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta

    new_c5 = compute_c5(h_new)
    if new_c5 < cur_c5 or rng.random() < np.exp(-(new_c5 - cur_c5) / max(temperature, 1e-12)):
        return h_new, new_c5
    return h, cur_c5

    target_sum = n / 2.0
