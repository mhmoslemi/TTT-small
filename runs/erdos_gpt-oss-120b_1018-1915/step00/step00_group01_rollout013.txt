import time
import numpy as np

def evaluate_c5(h):
    """Compute C5 for a given h vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def random_binary_vector(n, rng):
    """Generate a binary vector with exactly n/2 ones."""
    half = n // 2
    vec = np.zeros(n, dtype=float)
    vec[:half] = 1.0
    rng.shuffle(vec)
    return vec

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h

    if 'initial_h_values' in globals():

        n = h.size
    else:
        n = 200                     # fallback size (must be even)
        h = random_binary_vector(n, rng)

    # Ensure the sum constraint exactly (might be violated by rounding)
    target_sum = n / 2.0
    current_sum = h.sum()
    if not np.isclose(current_sum, target_sum):
        # Simple proportional scaling then clipping, re‑balance to meet sum
        h = np.clip(h * (target_sum / current_sum), 0.0, 1.0)
        deficit = target_sum - h.sum()
        # Distribute the deficit uniformly on entries that are not at a bound
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # fallback to binary construction
            h = random_binary_vector(n, rng)
        else:
            add = deficit / free.size
            h[free] += add
            h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    max_delta_initial = 0.3   # initial max change for a pair
    max_delta = max_delta_initial
    iteration = 0

    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        # gradually shrink step size
        max_delta = max_delta_initial * (1.0 - iteration / 1e6)
        i, j = rng.choice(n, size=2, replace=False)

        new_i = h[i] + delta
