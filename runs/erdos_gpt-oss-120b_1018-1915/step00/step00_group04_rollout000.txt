import time
import numpy as np

def _objective(h):
    """Return C5 for a candidate h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _repair(h, target_sum):
    """Project h onto [0,1] and enforce sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative linear adjustment
    for _ in range(10):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = h < 1.0
        else:
            idx = h > 0.0
        if not np.any(idx):
            break
        delta = diff / np.count_nonzero(idx)
        h[idx] += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def _perturb(h, scale=0.05):
    """Create a perturbed copy of h and repair it."""
    n = h.size
    new_h = h.copy()
    # perturb a random subset (about 10% of entries)
    k = max(1, n // 10)
    idx = np.random.choice(n, k, replace=False)
    noise = np.random.normal(0.0, scale, size=k)
    new_h[idx] += noise
    target = n / 2.0
    new_h = _repair(new_h, target)
    return new_h
