import time
import numpy as np
import math

def _c5_score(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _mutate(h):
    """Return a new h after moving a random amount of mass between two entries."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    h_new = h.copy()
    # decide direction of transfer
    if np.random.rand() < 0.5:
        # move from j to i
        max_transfer = min(h_new[j], 1.0 - h_new[i])
        if max_transfer <= 0:
            return h_new
        delta = np.random.rand() * max_transfer
        h_new[i] += delta
        h_new[j] -= delta
    else:
        # move from i to j
        max_transfer = min(h_new[i], 1.0 - h_new[j])
        if max_transfer <= 0:
            return h_new
        delta = np.random.rand() * max_transfer
        h_new[i] -= delta
        h_new[j] += delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] subject to:
        0 ≤ h[i] ≤ 1,
        sum(h) * dx = 1   (dx = 2/n_points)
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine discretisation size
    try:
        n_points = int(initial_h_values.size)  # provided by harness if any
        h = np.array(initial_h_values, copy=True)
    except Exception:
        n_points = 45
        h = np.full(n_points, 0.5)  # feasible start (average = 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    # Slight correction in case initial vector is not perfectly feasible
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h = h * (target_sum / cur_sum)

    best_h = h.copy()
    best_score = _c5_score(best_h, dx)

    current_h = best_h.copy()
    current_score = best_score

    # Simulated‑annealing schedule
    T = 0.05          # initial temperature
    cooling = 0.9995  # geometric cooling factor
    iter_count = 0

    while time.time() - start_time < budget_s - 0.5:
        new_h = _mutate(current_h)
        new_score = _c5_score(new_h, dx)

        # Metropolis acceptance
        if new_score < current_score or np.random.rand() < math.exp((current_score - new_score) / T):
            current_h, current_score = new_h, new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score

        iter_count += 1
        if iter_count % 1000 == 0:
            T *= cooling
            # occasional re‑heat to avoid stagnation
            if T < 1e-6:
                T = 0.02

        # optional refinement: double resolution after finding a decent bound
        if (best_score < 0.382) and (n_points == 45) and (time.time() - start_time > budget_s * 0.3):
