import time
import numpy as np

def _evaluate(h, dx):
    """Return max overlap C5 for step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def _mutate(h, rng, target_sum):
    """Propose a feasible mutation that preserves the total sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # move a random amount delta from i to j
    max_down = h[i]               # how much we can decrease at i
    max_up   = 1.0 - h[j]         # how much we can increase at j
    max_move = min(max_down, max_up)
    if max_move == 0.0:
        return None  # mutation not possible
    delta = rng.random() * max_move
    # with 50% chance reverse the direction
    if rng.random() < 0.5:
        delta = -delta
        # now moving from j to i
        max_down = h[j]
        max_up   = 1.0 - h[i]
        max_move = min(max_down, max_up)
        if max_move == 0.0:
            return None
        delta = rng.random() * max_move
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
    else:
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    # numeric safety: clip tiny violations
    np.clip(h_new, 0.0, 1.0, out=h_new)
    # renormalize tiny drift
    drift = target_sum - h_new.sum()
    if abs(drift) > 1e-12:
        h_new[0] += drift
        if h_new[0] < 0.0:
            h_new[0] = 0.0
        elif h_new[0] > 1.0:
            h_new[0] = 1.0
    return h_new
