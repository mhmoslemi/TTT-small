import time
import numpy as np

def _repair_bounds_and_sum(h, target_sum, eps=1e-12):
    """Clip to [0,1] and then adjust to meet exact sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) <= eps:
        return h
    # distribute the difference greedily
    if diff > 0:
        # need to add mass
        idx = np.where(h < 1.0)[0]
        for i in idx:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
            if diff <= eps:
                break
    else:
        # need to remove mass
        idx = np.where(h > 0.0)[0]
        for i in idx:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
            if diff >= -eps:
                break
    # final clip (just in case)
    h = np.clip(h, 0.0, 1.0)
    return h


def _propose(h, step):
    """Create a neighbour by moving a small amount of mass between two cells."""
    n = h.shape[0]
    new = h.copy()
    # perform a few small transfers
    for _ in range(5):
        a, b = np.random.choice(n, 2, replace=False)
        # max amount we can move from a to b without breaking bounds
        max_forward = min(step, new[a], 1.0 - new[b])
        max_backward = min(step, new[b], 1.0 - new[a])
        # choose direction uniformly
        if np.random.rand() < 0.5:
            amt = np.random.uniform(0, max_forward)
            new[a] -= amt
            new[b] += amt
        else:
            amt = np.random.uniform(0, max_backward)
            new[a] += amt
            new[b] -= amt
    return new


def _initialize_h(n):
    """Uniform start: each entry 0.5 gives correct integral."""
    return np.full(n, 0.5)


def _interpolate(old_h, new_n):
    """Linear interpolation of a step function to a finer grid."""
    old_n = old_h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # enforce exact integral after interpolation
    target_sum = new_n / 2.0
    new_h = _repair_bounds_and_sum(new_h, target_sum)
    return new_h
