import time
import numpy as np

# helpers are supplied in the execution environment:
# - evaluate_erdos_solution(h): returns C5 bound for a given step vector
# - initial_h_values (optional): starting vector (length 45)

def compute_c5(h):
    """Direct computation of C5 using correlation (for safety)."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def project_pair(h, i, j, delta):
    """Attempt to add delta to h[i] and subtract from h[j] keeping 0‑1 bounds."""
    hi, hj = h[i] + delta, h[j] - delta
    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
        new_h = h.copy()
        new_h[i], new_h[j] = hi, hj
        return new_h
    return None

def local_search(h, time_budget):
    """Simple stochastic pairwise hill‑climbing within a time budget."""
    best = h.astype(float)
    best_score = evaluate_erdos_solution(best)
    n = best.size
    rnd = np.random.default_rng()
    start = time.time()
    while time.time() - start < time_budget:
        i, j = rnd.choice(n, 2, replace=False)
        delta = rnd.uniform(-0.02, 0.02)  # modest step
        cand = project_pair(best, i, j, delta)
        if cand is None:
            continue
        score = evaluate_erdos_solution(cand)
        if score < best_score - 1e-7:          # strict improvement
            best, best_score = cand, score
    return best, best_score

def repeat_resolution(h):
    """Double the number of samples by repeating each entry."""
    return np.repeat(h, 2)
