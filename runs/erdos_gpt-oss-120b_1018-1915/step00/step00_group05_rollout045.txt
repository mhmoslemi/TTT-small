import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint

def _objective(h, dx):
    """Return max correlation h with (1-h) times dx."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx
