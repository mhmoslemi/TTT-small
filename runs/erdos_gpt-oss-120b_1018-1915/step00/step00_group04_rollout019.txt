import numpy as np
import time

def _overlap_bound(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx via discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()


def _repair(h, target_sum):
    """
    Project h onto the feasible set:
      0 <= h_i <= 1  and  sum h_i = target_sum
    A simple iterative redistribution scheme is used; it is fast for the
    modest dimensions of this problem.
    """
    h = np.clip(h, 0.0, 1.0)

    # First try a uniform redistribution of the total error.
    for _ in range(10):
        err = target_sum - h.sum()
        if abs(err) < 1e-12:
            break
        # choose indices that can move in the needed direction
        if err > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        idx = np.where(mask)[0]
        if len(idx) == 0:
            break
        delta = err / len(idx)
        h[idx] += delta
        h = np.clip(h, 0.0, 1.0)

    # Final correction (if any) by scaling the interior values.
    err = target_sum - h.sum()
    if abs(err) > 1e-12:
        interior = (h > 0.0) & (h < 1.0)
        idx = np.where(interior)[0]
        if len(idx) > 0:
            h[idx] += err / len(idx)
            h = np.clip(h, 0.0, 1.0)

    # As a safety net, if the sum is still off, perform a simple least‑squares
    # projection onto the hyperplane sum h = target_sum while respecting bounds.
    # The projection for small dimensions can be done analytically.
    err = target_sum - h.sum()
    if abs(err) > 1e-12:
        # shift all entries uniformly, then clip again.
        h += err / len(h)
        h = np.clip(h, 0.0, 1.0)
        # final uniform tweak to hit the sum exactly
        err = target_sum - h.sum()
        if abs(err) > 1e-12:
            h += err / len(h)
            h = np.clip(h, 0.0, 1.0)

    return h

    Optimize a step function h on [0,2] (discretized) to lower the Erdős
    overlap constant C5.

    Returns
    -------

        Corresponding upper bound for C5.
    n_points : int
        Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size.  If a parent solution is provided use its size,
    # otherwise fall back to a modest default.
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        n = parent.size

    except NameError:
        n = 84                     # reasonable small grid if no parent exists
        current = rng.uniform(0.0, 1.0, n)

    target_sum = n / 2.0           # because ∫ h = 1 and dx = 2/n
    current = _repair(current, target_sum)

    cur_c5 = _overlap_bound(current)
    best_h = current.copy()
    best_c5 = cur_c5

    # Simulated‑annealing parameters
    T = 0.02                       # initial temperature
    T_final = 1e-5
    cooling_rate = 0.9995
    iteration = 0

    while True:

        if now - start_time > budget_s - 0.2:   # leave a small safety margin

        iteration += 1

        # Propose a new candidate by Gaussian perturbation
