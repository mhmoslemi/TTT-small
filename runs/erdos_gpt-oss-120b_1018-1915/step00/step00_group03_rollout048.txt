    if abs(current - target) > 1e-9:
        # adjust by adding delta equally to all entries within bounds

        h += delta
        # after shift may violate bounds; clip again and redistribute residual
        # Iterate few times.

import time
import numpy as np

def _repair(h):
    """Clip to [0,1] and enforce sum(h)=len(h)/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Simple redistribution to meet the integral constraint
    for _ in range(5):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        delta = diff / n
        h += delta
        # respect bounds again
        low = h < 0.0
        high = h > 1.0
        if not low.any() and not high.any():
            break
        excess = h[high].sum() - high.sum() * 1.0
        deficit = -h[low].sum()
        h[high] = 1.0
        h[low] = 0.0
        free = ~(low | high)
        free_cnt = free.sum()
        if free_cnt == 0:
            break
        # distribute net excess/deficit over free entries
        h[free] += (excess - deficit) / free_cnt
    return np.clip(h, 0.0, 1.0)

def _c5_value(h):
    """Compute the max overlap integral for a given step vector."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

def _neighbor(h, sigma):
    """Create a perturbed neighbor respecting the constraints."""
    n = h.size
    # modify about 5 % of entries (at least one)
    k = max(1, n // 20)
    idx = np.random.choice(n, k, replace=False)
    perturb = np.random.normal(0.0, sigma, size=k)
    h_new = h.copy()
    h_new[idx] += perturb
    return _repair(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős C5 bound.

    Returns
    -------
    h_values : np.ndarray
        Step function samples on [0,2] (values in [0,1]).
    c5_bound : float
        Achieved max overlap integral.
    n_points : int
        Length of the returned array.
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------ #
    # 1. Initial construction
    # ------------------------------------------------------------------ #
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fallback: uniform 0.5, then repaired to the correct integral
        n0 = 70
        h0 = np.full(n0, 0.5, dtype=float)
    h0 = _repair(h0)
    n = h0.size
    best_h = h0
    best_c5 = _c5_value(best_h)

    # ------------------------------------------------------------------ #
    # 2. Simple simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------ #
    # schedule: sigma decays from 0.3 to 0.01
    sigma_start, sigma_end = 0.3, 0.01
    iter_count = 0
    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:
            break
        # linear decay based on elapsed proportion
        elapsed = (now - start_time) / (budget_s * 0.95)
        sigma = sigma_start * (1 - elapsed) + sigma_end * elapsed

        cand = _neighbor(best_h, sigma)
        cand_c5 = _c5_value(cand)

        if cand_c5 < best_c5:
            best_h, best_c5 = cand, cand_c5

        # occasional random restart to avoid stagnation
        if iter_count % 5000 == 0 and iter_count > 0:
            rnd = np.random.uniform(0.0, 1.0, size=n)
            rnd = _repair(rnd)
            rnd_c5 = _c5_value(rnd)
            if rnd_c5 < best_c5:
                best_h, best_c5 = rnd, rnd_c5

        iter_count += 1

    # ------------------------------------------------------------------ #
    # 3. Optional refinement by doubling resolution (if time permits)
    # ------------------------------------------------------------------ #
    # Try one refinement step to a finer grid (up to ~200 points)
    if time.time() - start_time < budget_s * 0.85:
        n2 = min(200, n * 2)
        # linear interpolation to new grid
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n2, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_ref = _repair(h_interp)
        best_h, best_c5 = h_ref, _c5_value(h_ref)

        # brief local search on the finer grid
        sigma_start_f, sigma_end_f = 0.2, 0.005
        while True:
            now = time.time()
            if now - start_time > budget_s * 0.98:
                break
            elapsed = (now - start_time) / (budget_s * 0.98)
            sigma = sigma_start_f * (1 - elapsed) + sigma_end_f * elapsed
            cand = _neighbor(best_h, sigma)
            cand_c5 = _c5_value(cand)
            if cand_c5 < best_c5:
                best_h, best_c5 = cand, cand_c5
            if np.random.rand() < 0.001:
                break

    # ------------------------------------------------------------------ #
    # 4. Return result
    # ------------------------------------------------------------------ #
    return best_h, float(best_c5), best_h.size
