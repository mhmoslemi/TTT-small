import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for step function h defined on [0,2]."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def repair(h):
    """Project h onto the feasible set: 0≤h≤1 and sum(h)=len(h)/2."""
    np.clip(h, 0.0, 1.0, out=h)
    target = h.size / 2.0
    for _ in range(15):          # a few passes are enough
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = 1.0 - h
            total = free.sum()
            if total == 0:
                break
            h += diff * free / total
        else:
            avail = h
            total = avail.sum()
            if total == 0:
                break
            h += diff * avail / total   # diff is negative → subtraction
        np.clip(h, 0.0, 1.0, out=h)
    return h


def mutate(h, rng):
    """Create a neighbour by moving a small amount of mass between two cells."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h.copy()
    # maximum amount we can transfer without breaking bounds
    max_from_j = h[j]                     # cannot take more than it has
    max_to_i = 1.0 - h[i]                 # cannot give more than i can receive
    max_transfer = min(max_from_j, max_to_i, 0.1)  # limit step size
    if max_transfer <= 0:
        return h.copy()
    delta = rng.random() * max_transfer
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    """Optimise a step function h on [0,2] to minimise the Erdős C5 overlap."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # decide on resolution (must stay < 1000 samples for speed)
    n_points = 200

    # initialise from the provided parent if it exists
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        # linear interpolation to the desired size

        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_old, parent)
    else:
        h = rng.random(n_points)   # uniform start
    h = repair(h)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # simulated‑annealing parameters
    T = 1.0
    cooling = 0.9995

    # main loop – stop a few seconds before budget to avoid abrupt termination
    while time.time() - start_time < budget_s - 5.0:
        # perform a batch of proposals per temperature update
        for _ in range(200):
            cand = mutate(best_h, rng)
            cand = repair(cand)
            cand_score = compute_c5(cand)

            # accept if better, otherwise with Metropolis probability
            if cand_score < best_score or rng.random() < np.exp((best_score - cand_score) / max(T, 1e-12)):
                best_h = cand
                best_score = cand_score

            # early exit if we already beat the target bound
            if best_score <= 0.3808:
                break

        T *= cooling
        # optional early stop if temperature is tiny
        if T < 1e-6:

    # final projection to guarantee feasibility
    best_h = repair(best_h)
    final_score = compute_c5(best_h)

    # return as plain python list for the harness
    return best_h.tolist(), float(final_score), n_points
