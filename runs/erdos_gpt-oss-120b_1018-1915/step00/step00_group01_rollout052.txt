import time
import numpy as np
import random

def _evaluate(h, dx):
    """Maximum overlap integral for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _resample_initial(target_n):
    """Return an initial vector of length target_n based on `initial_h_values`."""
    if 'initial_h_values' not in globals():
        # uniform start (value 0.5 gives the required integral)
        return np.full(target_n, 0.5, dtype=float)
    src = np.asarray(initial_h_values, dtype=float)
    src_n = src.size
    if src_n == target_n:
        return src.copy()
    # linear interpolation on the circle [0,2)
    x_src = np.linspace(0.0, 2.0, src_n, endpoint=False)
    x_tgt = np.linspace(0.0, 2.0, target_n, endpoint=False)
    return np.interp(x_tgt, x_src, src)

def _project_sum_clip(h, target_sum):
    """
    Clip values to [0,1] and adjust to satisfy the sum constraint.
    Simple iterative scaling that respects bounds.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        total = h.sum()
        if np.isclose(total, target_sum):
            break
        # scale the free (non‑boundary) entries
        free = (h > 0.0) & (h < 1.0)
        if not free.any():
            # all at bounds – distribute uniformly
            h = np.full_like(h, target_sum / h.size)
            break
        scale = (target_sum - h[~free].sum()) / h[free].sum()
        h[free] *= scale
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : ndarray
        Optimized values of h on the chosen uniform grid.
    c5_bound : float
        Corresponding upper bound C5 = max_k ∫ h(x)(1‑h(x+k))dx.
    n_points : int
        Number of discretisation points used.
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # Choose a moderate resolution that balances speed and expressiveness.
    n_points = 200                       # can be changed if time permits
    dx = 2.0 / n_points
    target_sum = n_points / 2.0           # because ∫ h = 1

    # Initialise from the provided parent or a uniform start.
    h = _resample_initial(n_points)
    h = _project_sum_clip(h, target_sum)

    best_h = h.copy()
    best_val = _evaluate(best_h, dx)

    # Parameters for the stochastic local search.
    max_iter = 10_000_000                 # generous upper bound
    report_interval = 0.5                 # seconds
    next_report = start_time + report_interval

    while True:
        # Time‑budget check (leave a small safety margin).
        if time.time() - start_time > budget_s * 0.95:
            break

        # Propose a mass‑transfer move that keeps the total sum unchanged.
        i, j = np.random.randint(0, n_points, size=2)
        if i == j:
            continue
        hi, hj = h[i], h[j]

        # Feasible delta range so that both stay in [0,1].
        delta_low  = -min(1.0 - hi, hj)   # move from j to i
        delta_high =  min(hi, 1.0 - hj)   # move from i to j
        if delta_low >= delta_high:
            continue
        delta = np.random.uniform(delta_low, delta_high)

        # Apply move.
        h[i] = hi - delta
        h[j] = hj + delta

        # Evaluate.
        val = _evaluate(h, dx)

        # Accept if improvement (greedy hill‑climb). Occasionally accept
        # worse moves to escape shallow traps (10 % probability).
        if val < best_val or np.random.rand() < 0.10:
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            # Reject: revert the move.
            h[i] = hi
            h[j] = hj

        # Periodic reporting (optional; does not affect output).
        if time.time() >= next_report:
            next_report = time.time() + report_interval

    # Final projection to guarantee feasibility (useful if the last move was
    # accepted with a slight numerical drift).
    best_h = _project_sum_clip(best_h, target_sum)
    best_val = _evaluate(best_h, dx)

    return best_h, float(best_val), int(n_points)
