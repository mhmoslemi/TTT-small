import time
import numpy as np

def compute_c5(h, dx):
    """Return the C5 value for step function samples h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def mutate(h, rng):
    """Transfer a random amount of mass from i to j while keeping bounds."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    # maximum transferable amount respecting [0,1] bounds
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return h  # nothing to do
    eps = rng.random() * max_eps
    h[i] -= eps
    h[j] += eps
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] that lowers the Erdős C5 bound.

    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # discretization
    n_points = 45
    dx = 2.0 / n_points

    # initialise h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        if h.size != n_points:
            # adapt size if needed (very rare)
            h = np.resize(h, n_points).astype(float)
    else:
        h = np.full(n_points, 0.5, dtype=float)

    # enforce exact integral constraint
    target_sum = n_points / 2.0
    cur_sum = h.sum()
    if cur_sum != 0:
        h *= target_sum / cur_sum
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)

    # simulated annealing parameters
    T = 0.05                # initial temperature
    cooling = 0.9995        # geometric cooling factor
    iter_count = 0
    max_iter = 10_000_000   # safety ceiling

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95 or iter_count >= max_iter:
            break

        # propose a move
        cand = best_h.copy()
        cand = mutate(cand, rng)

        cand_c5 = compute_c5(cand, dx)
        delta = cand_c5 - best_c5

        # accept if better, otherwise with Boltzmann probability
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            best_h = cand
            best_c5 = cand_c5

        # temperature decay
        T *= cooling
        iter_count += 1

    return best_h, best_c5, n_points
