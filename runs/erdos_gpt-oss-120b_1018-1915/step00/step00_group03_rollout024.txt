import time
import numpy as np

def _c5_of_h(h):
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _random_feasible_vector(n, rng):
    """Generate a random vector in [0,1]^n with sum = n/2."""
    # start from 0.5 + zero‑mean perturbation
    delta = rng.uniform(-0.5, 0.5, size=n)
    delta -= delta.mean()                     # enforce zero sum
    h = 0.5 + delta
    # clamp and re‑project to exact sum
    h = np.clip(h, 0.0, 1.0)
    # fix any small sum error by distributing the residual uniformly
    residual = (n / 2.0) - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual over entries that are not at bounds
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += residual / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback to uniform vector
            h[:] = 0.5
    return h
