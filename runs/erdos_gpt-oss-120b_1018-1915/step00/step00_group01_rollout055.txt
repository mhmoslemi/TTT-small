import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h, dx):
    """Return the overlap bound C5 for a discrete step function."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_feasible(h, target_sum):
    """Clip to [0,1] and renormalize so that sum(h) == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if s == 0:
        # avoid division by zero – spread mass uniformly
        h[:] = target_sum / len(h)
    else:
        h *= target_sum / s
    # a final clipping in case rounding pushed values out of bounds
    h = np.clip(h, 0.0, 1.0)
    return h

def _interpolate_h(src_h, src_n, tgt_n):
    """Periodically interpolate a vector from src_n points to tgt_n points."""
    if src_n == tgt_n:
        return src_h.copy()
    src_x = np.arange(src_n) * (2.0 / src_n)
    tgt_x = np.arange(tgt_n) * (2.0 / tgt_n)

    # create periodic extension for interpolation
    src_x_ext = np.concatenate([src_x - 2.0, src_x, src_x + 2.0])
    src_h_ext = np.concatenate([src_h, src_h, src_h])
    return np.interp(tgt_x, src_x_ext, src_h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] that lowers the Erdős C5 bound."""
    rnd = np.random.RandomState(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # 1.  Determine a starting construction.
    # ----------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # provided by harness
        n0 = init_h.size
    except Exception:
        # fallback: uniform function (integral 1 => sum = n/2)
        n0 = 47
        init_h = np.full(n0, 0.5)  # sum = 0.5*n0 = n0/2 → satisfies integral
    dx0 = 2.0 / n0
    target_sum0 = n0 / 2.0
    init_h = _project_to_feasible(init_h, target_sum0)

    best_h = init_h
    best_n = n0
    best_dx = dx0
    best_c5 = _compute_c5(best_h, best_dx)

    # ----------------------------------------------------------------------
    # 2.  Define a generic optimisation routine.
    # ----------------------------------------------------------------------
    def optimise_at_resolution(n_points, x0):
        dx = 2.0 / n_points
        target_sum = n_points / 2.0

        # equality constraint: sum(h) = target_sum
        eq_cons = {'type': 'eq',
                   'fun': lambda h: np.sum(h) - target_sum}

        bounds = [(0.0, 1.0)] * n_points

        # Scipy's SLSQP works well with simple bounds & equality.
        res = minimize(lambda h: _compute_c5(h, dx),
                       x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=[eq_cons],
                       options={'maxiter': 250, 'ftol': 1e-10, 'disp': False})
        h_opt = _project_to_feasible(res.x, target_sum)
        c5_opt = _compute_c5(h_opt, dx)
        return h_opt, c5_opt, dx

    # ----------------------------------------------------------------------
    # 3.  Exploration schedule (increasing resolution, few random restarts).
    # ----------------------------------------------------------------------
    # Resolutions to try – start from the initial size and go up.
    schedule = [47, 94, 188, 300]  # keep total work modest
