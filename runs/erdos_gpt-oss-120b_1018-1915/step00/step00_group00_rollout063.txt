import time
import numpy as np

def compute_c5(h, dx):
    """Compute the overlap bound C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def mutate(h):
    """Return a new h obtained by moving some mass between two positions."""
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    hi, hj = h[i], h[j]

    # feasible epsilon so that both stay in [0,1]
    lower = max(-hi, hj - 1.0)
    upper = min(1.0 - hi, hj)
    if lower >= upper:
        return h.copy()

    eps = np.random.uniform(lower, upper)
    new_h = h.copy()
    new_h[i] = hi + eps
    new_h[j] = hj - eps
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] (with n points) minimizing the
    Erdős overlap constant C5.

    Returns:
        (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine discretisation size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 45
        target_sum = n / 2.0
        # Dirichlet gives a random simplex point; scale to target sum
        h0 = np.random.dirichlet(np.ones(n)) * target_sum

    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility of the start vector
    h0 = np.clip(h0, 0.0, 1.0)
    # Small correction for floating‑point drift
    diff = target_sum - h0.sum()
    if abs(diff) > 1e-12:
        # adjust a random entry that can absorb the diff
        idxs = np.where((h0 + diff >= 0.0) & (h0 + diff <= 1.0))[0]
        if idxs.size:
            h0[idxs[0]] += diff
        else:
            h0 += diff / n  # fallback; will be re‑clipped later

    current_h = h0.copy()
    current_c = compute_c5(current_h, dx)

    best_h = current_h.copy()
    best_c = current_c

    # Simulated annealing parameters
    T = 0.5
    cooling = 0.9995
    iter_no_improve = 0
    max_no_improve = 2000

    while time.time() - start_time < budget_s * 0.97:
        new_h = mutate(current_h)
        new_c = compute_c5(new_h, dx)

        delta = new_c - current_c
        accept = False
        if delta <= 0:
            accept = True
        else:
            prob = np.exp(-delta / max(T, 1e-12))
            if prob > np.random.rand():
                accept = True

        if accept:

            if new_c < best_c:
                best_h, best_c = new_h.copy(), new_c
                iter_no_improve = 0
            else:
                iter_no_improve += 1
        else:
            iter_no_improve += 1

        # Cooling schedule
        T *= cooling

        # Random restart if stuck
        if iter_no_improve >= max_no_improve:
