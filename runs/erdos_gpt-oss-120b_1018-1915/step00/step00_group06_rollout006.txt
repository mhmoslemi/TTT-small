import time
import numpy as np

def _evaluate(h):
    """Return C5 value for sampled step function h."""
    n = h.size
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # max correlation multiplied by dx = 2/n
    return np.max(corr) * (2.0 / n)

def _project(h, target_sum, max_iter=20, tol=1e-9):
    """Clip to [0,1] and adjust to hit the required sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < tol:
            break
        # distribute uniformly, then re‑clip
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final tiny fix if still off (should be rare)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > tol:
        # distribute proportionally on non‑border elements
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def _perturb(h, scale, rng):
    """Create a random perturbation and project back to feasibility."""
    n = h.size
    # add Gaussian noise to a random subset (≈30% of entries)
    mask = rng.random(n) < 0.3
    delta = rng.normal(0.0, scale, size=n) * mask
    h_new = h + delta
    target = n / 2.0          # because integral h dx = 1, dx=2/n
    h_new = _project(h_new, target)
    return h_new
