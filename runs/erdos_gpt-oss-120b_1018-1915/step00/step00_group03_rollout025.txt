import time
import numpy as np
from math import exp

def _evaluate_c5(h, dx):
    """Return max overlap C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _repair_sum(h, target_sum):
    """Adjust h so that its sum equals target_sum while keeping 0<=h<=1."""
    current = np.sum(h)
    diff = target_sum - current
    # Simple greedy redistribution
    while abs(diff) > 1e-12:
        if diff > 0:
            # need to add mass
            candidates = np.where(h < 1.0)[0]
            if candidates.size == 0:
                break
            idx = np.random.choice(candidates)
            allowance = 1.0 - h[idx]
            add = min(allowance, diff)
            h[idx] += add
            diff -= add
        else:
            # need to remove mass
            candidates = np.where(h > 0.0)[0]
            if candidates.size == 0:
                break
            idx = np.random.choice(candidates)
            allowance = h[idx]
            sub = min(allowance, -diff)
            h[idx] -= sub
            diff += sub
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] minimizing the Erdős C5 overlap."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # choose resolution (prefer 140 points, keep under 1k)
    n_points = 140
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because integral should be 1

    # ----- initial construction -----
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = len(h0)
        x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        x = (np.arange(n_points) + 0.5) * dx
        h = np.interp(x, x0, h0)
        # enforce sum constraint by scaling then repairing
        h = _repair_sum(h * (target_sum / np.sum(h)), target_sum)
    else:
        # triangular shape: area exactly 1
        x = (np.arange(n_points) + 0.5) * dx
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
        h = _repair_sum(h, target_sum)

    # initial evaluation
    best_h = h.copy()
    best_score = _evaluate_c5(best_h, dx)

    # parameters for stochastic hill climbing
    max_step_fraction = 0.15   # max proportion of transferable mass per move
    temp_start = 0.5
    temp_end = 0.001

    while time.time() - start_time < budget_s * 0.96:
        # temperature schedule (linear decay)
        elapsed = time.time() - start_time
        temp = temp_start + (temp_end - temp_start) * (elapsed / budget_s)

        # propose a move: shift some mass from i to j
        i, j = rng.integers(0, n_points, size=2)
        if i == j:
            continue
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random() * max_step_fraction
        if delta <= 0:
            continue

        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        # evaluate
        score = _evaluate_c5(new_h, dx)

        # accept if better or with Metropolis probability
        if score < best_score:
            best_h = new_h
            best_score = score
        else:
            if rng.random() < exp(-(score - best_score) / max(temp, 1e-12)):
                best_h = new_h

        # occasional larger perturbation to escape plateaus
        if rng.random() < 0.005:
            perm = rng.permutation(n_points)
            perturbed = best_h[perm]  # shuffle values
            perturbed = _repair_sum(perturbed, target_sum)
            pert_score = _evaluate_c5(perturbed, dx)
            if pert_score < best_score:
                best_h = perturbed
                best_score = pert_score

    # final repair to guarantee constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_sum(best_h, target_sum)

    # final evaluation (in case repair altered score)
    final_score = _evaluate_c5(best_h, dx)

    return best_h, final_score, n_points
