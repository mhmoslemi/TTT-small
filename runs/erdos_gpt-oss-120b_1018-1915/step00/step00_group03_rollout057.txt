import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def evaluate(h):
    """Return C5 for a step function h sampled on [0,2]."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def project_sum_bounds(x, lower, upper, target_sum, tol=1e-10, max_iter=200):
    """
    Euclidean projection of x onto the box [lower,upper] with an exact sum constraint.
    Implements a simple water‑filling scheme.
    """
    x = np.clip(x, lower, upper)
    diff = target_sum - x.sum()
    for _ in range(max_iter):
        if abs(diff) <= tol:
            break
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            # all variables are at a bound, cannot adjust further
            break
        add = diff / nfree
        x[free] += add
        x = np.clip(x, lower, upper)
        diff = target_sum - x.sum()
    return x


def upsample(h, new_n):
    """Linear interpolation of h from its current length to new_n points."""
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    # enforce exact sum after interpolation
    target_sum = new_n / 2.0
    h_new = project_sum_bounds(h_new, 0.0, 1.0, target_sum)
    return h_new


def local_search(h_start, time_budget):
    """
    Simple stochastic local optimisation with projection.
    Returns the best vector found and its C5 value.
    """
    start_time = time.time()
    n = len(h_start)
    target_sum = n / 2.0
    best_h = h_start.copy()
    best_c = evaluate(best_h)

    # scaling schedule
    scale = 0.25
    iteration = 0

    while time.time() - start_time < time_budget:
        iteration += 1
        if iteration % 100 == 0:
            scale *= 0.9  # gradual cooling

        # occasional full random restart
        if np.random.rand() < 0.01:
            cand = np.random.rand(n)
        else:
            cand = best_h + np.random.normal(scale=scale, size=n)

        cand = project_sum_bounds(cand, 0.0, 1.0, target_sum)

        c_val = evaluate(cand)
        if c_val < best_c:
            best_c = c_val
            best_h = cand

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimises a step function h on [0,2] to minimise the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initial construction
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        # ensure feasibility (it should already be feasible)
        target_sum0 = n0 / 2.0
        h0 = project_sum_bounds(h0, 0.0, 1.0, target_sum0)
    else:
        n0 = 70
        h0 = np.full(n0, 0.5)
        h0 = project_sum_bounds(h0, 0.0, 1.0, n0 / 2.0)

    best_overall_h = h0
    best_overall_c = evaluate(h0)

    # ------------------------------------------------------------------
    # Multi‑resolution schedule (up to ~800 points, respecting budget)
    # ------------------------------------------------------------------
    resolution_factors = [1, 2, 4]  # 70, 140, 280 (or less if budget tight)
    for factor in resolution_factors:
        # check remaining time
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        target_n = n0 * factor
        if target_n > 800:          # keep evaluation cheap
            target_n = min(target_n, 800)

        # Upsample current best to target_n
        h_current = upsample(best_overall_h, target_n)

        # Allocate a proportion of the remaining budget
        # First level gets ~50%, second ~30%, third ~20% of what is left
        if factor == 1:
            alloc = remaining * 0.5
        elif factor == 2:
            alloc = remaining * 0.3
        else:
            alloc = remaining * 0.2

        # Run local search at this resolution
        h_opt, c_opt = local_search(h_current, alloc)

        # Keep the best over all levels
        if c_opt < best_overall_c:
            best_overall_c = c_opt
            best_overall_h = h_opt

    # ------------------------------------------------------------------
    # Final sanity check before returning
    # ------------------------------------------------------------------
    n_final = len(best_overall_h)
    # Project once more to guarantee constraints (numerical drift)
    best_overall_h = project_sum_bounds(best_overall_h, 0.0, 1.0, n_final / 2.0)
    best_overall_c = evaluate(best_overall_h)

    return best_overall_h, best_overall_c, n_final
