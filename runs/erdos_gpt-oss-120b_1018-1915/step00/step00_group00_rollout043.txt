import time
import numpy as np
import random

def _compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / h.size))

def _repair(h: np.ndarray) -> np.ndarray:
    """Adjust h to satisfy 0≤h≤1 and sum(h)=len(h)/2 (within tiny epsilon)."""
    n = h.size
    target = n / 2.0
    # Clip to feasible box
    np.clip(h, 0.0, 1.0, out=h)
    # Fix the linear constraint by distributing the difference
    diff = target - h.sum()
    # Redistribute in a loop to handle large diffs
    while abs(diff) > 1e-12:
        if diff > 0:  # need to add mass
            candidates = np.where(h < 1.0)[0]
            if candidates.size == 0:
                break
            idx = random.choice(candidates)
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
            diff -= add
        else:          # need to remove mass
            candidates = np.where(h > 0.0)[0]
            if candidates.size == 0:
                break
            idx = random.choice(candidates)
            sub = min(-diff, h[idx])
            h[idx] -= sub
            diff += sub
    return h

def _mutate(h: np.ndarray) -> np.ndarray:
    """Create a nearby feasible h by moving a small amount of mass."""
    n = h.size
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return h.copy()
    # Determine how much can be transferred from i to j while staying in [0,1]
    max_transfer = min(h[i], 1.0 - h[j])
    min_transfer = -min(h[j], 1.0 - h[i])  # negative means opposite direction
    if max_transfer == 0 and min_transfer == 0:
        return h.copy()
    # Propose a delta in a modest range
    delta = np.random.uniform(-0.1, 0.1)
    delta = np.clip(delta, min_transfer, max_transfer)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that minimizes the
    Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Array of length n_points with values in [0,1] and sum = n_points/2.
    c5_bound : float
        Corresponding maximal overlap integral.
    n_points : int
        Number of discretisation points.
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise problem size and starting vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        h = _repair(h)
        n = h.size
    else:
        n = 45                     # default size (odd length is allowed)
        h = np.random.rand(n)      # random start in [0,1]
        h = _repair(h)

    best_h = h.copy()
    best_score = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # Simulated‑annealing loop
    # ------------------------------------------------------------------
    temperature = 1.0
    T_min = 1e-4
    decay = 0.995
    iteration = 0

    while time.time() - start_time < budget_s * 0.95:
        # generate neighbour and repair feasibility
        cand = _mutate(best_h)
        cand = _repair(cand)

        cand_score = _compute_c5(cand)

        # Acceptance test
        if cand_score < best_score or np.random.rand() < np.exp((best_score - cand_score) / temperature):
            best_h = cand
            best_score = cand_score

        # Temperature schedule
        temperature = max(T_min, temperature * decay)
        iteration += 1

        # occasional random restart to escape deep local minima
        if iteration % 5000 == 0:
            rand_h = np.random.rand(n)
            rand_h = _repair(rand_h)
            rand_score = _compute_c5(rand_h)
            if rand_score < best_score:
                best_h = rand_h
                best_score = rand_score

    # Final clean‑up: ensure exact constraint (tiny numerical drift)
    best_h = _repair(best_h)

    return (best_h.astype(float), best_score, n)
