import time
import numpy as np

def _compute_c5(h):
    """Fallback C5 computation if evaluate_erdos_solution is unavailable."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **_):
    """
    Return a step function h (as a NumPy array), its C5 bound, and the sample count.
    The routine performs a simple constrained hill‑climbing search starting from
    an available parent solution or from a binary half‑filled vector.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise h
    # ------------------------------------------------------------------
    try:
        # Use the provided parent construction if it exists.
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        # Clip to [0,1] and enforce the integral constraint ∑h = n/2.
        h = np.clip(h, 0.0, 1.0)

        cur = h.sum()
        if abs(cur - target) > 1e-12:
            # Simple affine correction then clip again.
            h += (target - cur) / n
            h = np.clip(h, 0.0, 1.0)
    except NameError:
        # No parent; build a binary vector with exactly half the entries =1.
        n = 200  # modest size, ensures quick evaluation
        h = np.zeros(n, dtype=float)
        h[: n // 2] = 1.0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Pre‑compute index masks for speed.
    def _indices(mask):
        return np.flatnonzero(mask)

    while time.time() - start < budget_s * 0.9:
