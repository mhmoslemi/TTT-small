def _evaluate_c5(h):
    """Discrete evaluation of C5 for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_onto_bounds_and_sum(h, target_sum, max_iter=10):
    """
    Project vector h onto the box [0,1] and enforce sum(h) == target_sum.
    Simple iterative scaling with clipping.
    """
    h = np.clip(h, 0.0, 1.0)
