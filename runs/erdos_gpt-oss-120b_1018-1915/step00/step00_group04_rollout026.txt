import time
import numpy as np

def _c5_value(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _repair(h, target_sum, eps=1e-9, max_iter=100):
    # clip bounds first
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) <= eps:
            break
        if diff > 0:   # need to increase
            idx = h < 1.0
            cnt = idx.sum()
            if cnt == 0:
                break
            inc = diff / cnt
            # cap by remaining capacity
            add = np.minimum(inc, 1.0 - h[idx])
            h[idx] += add
        else:          # need to decrease
            idx = h > 0.0
            cnt = idx.sum()
            if cnt == 0:
                break
            dec = (-diff) / cnt
            sub = np.minimum(dec, h[idx])
            h[idx] -= sub
    # final clip to be safe
    return np.clip(h, 0.0, 1.0)

def _tournament(pop, scores, k):
    sel = np.random.choice(len(pop), k, replace=False)
    best = sel[0]
    best_score = scores[best]
    for i in sel[1:]:
        if scores[i] < best_score:  # lower C5 is better
            best = i
            best_score = scores[i]
    return pop[best]

def _crossover(p1, p2):
    return 0.5 * (p1 + p2)

def _mutate(child, sigma):
    child += np.random.normal(0.0, sigma, size=child.shape)
    return child

def _optimize(h0, budget, max_pop=40, sigma0=0.05, sigma_decay=0.99):
    n = h0.shape[0]
    target_sum = n / 2.0
    pop = []
    # initialise population around h0
    pop.append(_repair(h0.copy(), target_sum))
    for _ in range(max_pop - 1):
        pert = h0 + np.random.normal(0.0, sigma0, size=n)
        pop.append(_repair(pert, target_sum))
    pop = np.array(pop)
    scores = np.array([_c5_value(ind) for ind in pop])

    start = time.time()
    sigma = sigma0
    elite_frac = 0.1
    elite_cnt = max(1, int(elite_frac * max_pop))

    while time.time() - start < budget:
        # select elites
        elite_idx = np.argsort(scores)[:elite_cnt]
        new_pop = pop[elite_idx].copy()
        # generate offspring

            p1 = _tournament(pop, scores, 3)
            p2 = _tournament(pop, scores, 3)
            child = _crossover(p1, p2)
            child = _mutate(child, sigma)
            child = _repair(child, target_sum)
            new_pop = np.vstack([new_pop, child])
        pop = new_pop[:max_pop]
        scores = np.array([_c5_value(ind) for ind in pop])
        sigma = max(1e-4, sigma * sigma_decay)
    best_idx = np.argmin(scores)
    return pop[best_idx], scores[best_idx]

def _upscale(h, factor=2):
    n = h.shape[0]
    new_n = n * factor
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # enforce sum after upscale
    target_sum = new_n / 2.0
    h_new = _repair(h_new, target_sum)
    return h_new
