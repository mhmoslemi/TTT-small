import time
import numpy as np

def _c5_value(h, dx):
    """Compute the C5 value for a discrete h vector."""
    # correlation of h with (1-h); mode='full' gives length 2n-1, step accounts for dx
    conv = np.correlate(h, 1 - h, mode="full")
    return np.max(conv) * dx

def _make_initial(n):
    """Create a feasible starting vector if no parent is provided."""
    # simple binary pattern: first n/2 entries = 1, rest = 0
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimized h sampled at n_points equally spaced points.
    c5_bound : float
        Corresponding max overlap integral.
    n_points : int
        Number of discretization points (len(h_values)).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Use the parent's construction if it exists, otherwise build a simple one.
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        # default to 80 points as suggested by the problem statement
        n_default = 80
        h = _make_initial(n_default)

    n = h.shape[0]
    dx = 2.0 / n

    # Ensure feasibility: clip and rescale to satisfy sum(h)*dx = 1 exactly.
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        h[: n // 2] = 1.0
        total = h.sum()
    # scale to exact sum = n/2 (since sum * dx = 1)
    target_sum = n / 2.0
    h *= target_sum / total
    h = np.clip(h, 0.0, 1.0)  # possible tiny overshoot after scaling

    best_h = h.copy()
    best_score = _c5_value(best_h, dx)

    # Parameters for the local search
    max_step = 0.05          # maximal mass moved in one perturbation
    patience = 0            # counts consecutive non‑improving attempts
    max_patience = 2000     # stop early if stuck for long

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:   # leave a little margin
            break

        # Choose two distinct indices where a transfer is possible
        i, j = rng.choice(n, size=2, replace=False)
        if h[i] <= 0.0 or h[j] >= 1.0:
            continue

        delta = min(h[i], 1.0 - h[j], max_step)
        if delta <= 0:
            continue

        h_candidate = best_h.copy()
        h_candidate[i] -= delta
        h_candidate[j] += delta

        # Small numerical cleanup
        h_candidate = np.clip(h_candidate, 0.0, 1.0)

        # Verify the mass constraint (allow tiny tolerance)
        if not np.isclose(h_candidate.sum(), target_sum, atol=1e-9):

            h_candidate = np.clip(h_candidate, 0.0, 1.0)

            best_score = cand_score
