import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint

def _c5_value(h):
    """Compute the C5 bound for a discrete h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_constraints(h, target_sum):
    """Clip to [0,1] and then adjust to hit the exact sum (in expectation)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # all zeros, just fill uniformly
        h[:] = target_sum / h.size
        return h
    # simple affine scaling to match the sum, then clip again
    scale = target_sum / cur
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    # if clipping broke the sum, distribute the residual uniformly
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        h += residual / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _initial_on_grid(n, seed):
    """Create an initial guess on a grid of size n."""
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        if n == n0:
            return _project_to_constraints(h0.copy(), n / 2.0)
        # interpolate to new grid
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h_new = np.interp(x_new, x_old, h0)
        return _project_to_constraints(h_new, n / 2.0)
    # fallback: uniform distribution
    rng = np.random.default_rng(seed)
    h = rng.random(n)
    return _project_to_constraints(h, n / 2.0)

    start_time = time.time()

    grid_sizes = [47, 94, 188, 250, 376]

    def _optimize_on_grid(n, h_start):
        dx = 2.0 / n
        target_sum = n / 2.0

        bounds = Bounds(0.0, 1.0)
        lin_cons = LinearConstraint(np.ones((1, n)), [target_sum], [target_sum])

        # The optimizer works on a flat array
