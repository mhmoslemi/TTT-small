import numpy as np
import time
import math
import random

def _compute_c5(h):
    """Return C5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _project_feasible(h, target_sum, eps=1e-12):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    # Simple redistribution loop (will converge quickly for modest n)
    while abs(diff) > eps:
        if diff > 0:
            # need to add mass
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
        else:
            # need to remove mass
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
        # Choose a random admissible index
        i = random.choice(idx.tolist())
        allowable = (1.0 - h[i]) if diff > 0 else (-h[i])
        delta = min(abs(diff), abs(allowable))
        delta = delta if diff > 0 else -delta
        h[i] += delta
        diff = target_sum - h.sum()
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : list of float
        Optimized samples of h (length n_points).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of discretisation points.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise problem size and a feasible starting vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 80
        h = np.zeros(n, dtype=float)

    target_sum = n / 2.0          # because dx = 2/n and integral must be 1
    if h.sum() == 0.0:
        # start from a random binary vector with exactly n/2 ones
        ones = random.sample(range(n), n // 2)
        h[ones] = 1.0
    else:
        # project any supplied vector onto the feasible polytope
        h = _project_feasible(h, target_sum)

    # ------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop
    # ------------------------------------------------------------------
    start_time = time.time()
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # annealing schedule parameters
    max_iters = 200000
    temp_start = 0.02
    temp_end = 1e-4

    for it in range(1, max_iters + 1):
        # time check every 1000 iterations
        if it % 1000 == 0:
            if time.time() - start_time > budget_s * 0.95:
                break

        # temperature according to a simple exponential decay
        tau = it / max_iters
        temp = temp_start * (temp_end / temp_start) ** tau

        # propose a move: transfer delta from i to j
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]

        # feasible delta range (moving mass from i to j)
        lo = max(-hi, - (1.0 - hj))   # negative delta means opposite direction
        hi_delta = min(1.0 - hi, hj)  # positive delta moves from i to j

        if lo >= hi_delta:
