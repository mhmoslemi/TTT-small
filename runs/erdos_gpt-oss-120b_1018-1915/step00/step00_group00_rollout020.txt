import time
import numpy as np

# evaluate_erdos_solution and initial_h_values are injected by the harness

def _project_box_simplex(v, S, lower=0.0, upper=1.0, tol=1e-9, max_iter=60):
    """Project vector v onto {x: lower<=x_i<=upper, sum x_i = S}."""
    # initial clamp
    x = np.clip(v, lower, upper)
    total = x.sum()
    if abs(total - S) <= tol:
        return x
    # Find lambda via bisection so that sum_i min(upper, max(lower, v_i + lam)) = S
    lo = -np.max(v) - 1.0  # ensure lower bound
    hi = upper - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, lower, upper)
        s = x.sum()
        if abs(s - S) <= tol:
            break
        if s > S:
            hi = lam
        else:
            lo = lam
    return x

def _compute_c5(h, dx):
    """Return C5 for given h (numpy array) and dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx, corr

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ---- parameters ----
    n_points = 45                     # default size (can be changed)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0       # because integral should be 1

    # ---- initialization ----

        h = np.array(initial_h_values, dtype=float)
        if h.size != n_points:
            # resize by simple interpolation if size differs

            x_new = np.linspace(0, 2, n_points, endpoint=False)
            h = np.interp(x_new, x_old, h)
    else:
        h = rng.random(n_points)

    # enforce constraints exactly
    h = _project_box_simplex(h, target_sum)

    best_h = h.copy()
    best_score, _ = _compute_c5(best_h, dx)

    start_time = time.time()
    elapsed = 0.0

    # ---- optimization loop ----
    max_iters = 5000
    lr = 0.04            # base learning rate
    decay = 0.999        # learning rate decay
    patience = 0
    for it in range(max_iters):

        idx_j = idx_i + k

        # part from h[i-k] * (1-h[i])
