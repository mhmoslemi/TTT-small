import time
import numpy as np

def _repair(vec, target_sum):
    """Project vec onto [0,1]^n with sum = target_sum."""
    v = np.clip(vec.astype(float), 0.0, 1.0)
    for _ in range(10):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = v < 1.0 - 1e-12
        else:
            mask = v > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        adjust = diff / cnt
        v[mask] += adjust
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction to hit target_sum as close as possible
    residual = target_sum - v.sum()
    if abs(residual) > 1e-12:
        # distribute residual uniformly respecting bounds
        for _ in range(5):
            if residual > 0:
                mask = v < 1.0 - 1e-12
            else:
                mask = v > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            delta = residual / cnt
            v[mask] += delta
            v = np.clip(v, 0.0, 1.0)
            residual = target_sum - v.sum()
            if abs(residual) < 1e-12:
                break
    return v
