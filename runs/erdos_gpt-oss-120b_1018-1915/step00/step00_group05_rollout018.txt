import time
import numpy as np

def _evaluate_max_overlap(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def _project_bounds_and_sum(h, target_sum):
    """Clip to [0,1] and then adjust to hit the exact sum using a simple redistribution."""
    h = np.clip(h, 0.0, 1.0)
    current = h.sum()
    if np.isclose(current, target_sum):
        return h
    # redistribute the difference uniformly while keeping bounds
    diff = target_sum - current
    # distribute proportionally to free space
    free_up = 1.0 - h
    free_down = h
    # first try to add where possible
    if diff > 0:
        space = free_up.sum()
        if space > 0:
            add = diff * (free_up / space)
            h += add
    else:
        space = free_down.sum()
        if space > 0:
            sub = (-diff) * (free_down / space)
            h -= sub
    # final clip to safeguard rounding issues
    h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # adjust the first element that can take the residual
        for i in range(len(h)):
            allowed = (1.0 - h[i]) if residual > 0 else (-h[i])
            step = np.clip(residual, -allowed, allowed)
            h[i] += step
            residual -= step
            if abs(residual) < 1e-12:
                break
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.

    Returns (h_values, c5_bound, n_points).

    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 70                      # fallback size
        h = np.full(n, 0.5, dtype=float)   # satisfies sum constraint exactly

    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility of the initial point
    h = _project_bounds_and_sum(h, target_sum)

    best_h = h.copy()
    best_score = _evaluate_max_overlap(best_h, dx)

    # Simulated annealing parameters
    max_iter = 200000                # generous upper bound; actual stop by time
    init_temp = 0.02
    final_temp = 0.0005
    decay = (final_temp / init_temp) ** (1.0 / max_iter)

    temp = init_temp
    i = 0
    while i < max_iter:
        # time check
        if time.time() - start_time > budget_s - 0.2:
            break

        # propose a two‑index move preserving the total sum
        idx = np.random.choice(n, size=2, replace=False)
        i1, i2 = idx[0], idx[1]

        # random delta limited to keep values inside [0,1]
        max_delta = min(0.1, h[i1], 1.0 - h[i2])  # move from i1 to i2
        delta = (np.random.rand() - 0.5) * 2.0 * max_delta

        if delta == 0.0:
            i += 1
            continue

        new_h = h.copy()
        new_h[i1] -= delta
        new_h[i2] += delta

        # quick feasibility test
        if (new_h[i1] < 0.0) or (new_h[i1] > 1.0) or (new_h[i2] < 0.0) or (new_h[i2] > 1.0):

        temp *= decay
        i += 1

    for _ in range(5000):

            break

        delta = (np.random.rand() - 0.5) * 2.0 * max_delta
        if delta == 0.0:

        new_h[i1] -= delta
        new_h[i2] += delta
