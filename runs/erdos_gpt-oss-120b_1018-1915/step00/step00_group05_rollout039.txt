import time
import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a step function h (as a numpy array), its C5 bound, and the number of sample points.
    The algorithm starts from an optional parent solution, performs a constrained local
    optimization and then a lightweight stochastic refinement while respecting the time budget.
    """
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # Determine problem size and initial candidate
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
    else:
        n = int(kwargs.get('n_points', 70))
        init = None

    dx = 2.0 / n
    desired_sum = n / 2.0  # because ∫h dx = 1  =>  sum(h)*dx = 1

    # ----------------------------------------------------------------------
    # Helper: project any vector onto the feasible set {0<=h<=1, sum = desired_sum}
    # ----------------------------------------------------------------------
    def project(vec):
        v = np.clip(vec, 0.0, 1.0)
        # simple iterative scaling to hit the sum constraint
        for _ in range(5):
            total = v.sum()
            if abs(total - desired_sum) < 1e-12:
                break
            if total == 0:
                break
            factor = desired_sum / total
            v = np.clip(v * factor, 0.0, 1.0)
        # final tiny correction on a free coordinate if needed
        diff = desired_sum - v.sum()
        if abs(diff) > 1e-12:
            free_idx = np.where((v > 0) & (v < 1))[0]
            if free_idx.size:
                v[free_idx[0]] = np.clip(v[free_idx[0]] + diff, 0.0, 1.0)
        return v

    # Initial feasible vector
    if init is not None and init.size == n:
        h0 = project(init.copy())
    else:
        h0 = project(np.random.rand(n))

    # ----------------------------------------------------------------------
    # Compute C5 for a given h
    # ----------------------------------------------------------------------
    def c5_of(h):
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    # ----------------------------------------------------------------------
    # Constrained local optimization (SLSQP)
    # ----------------------------------------------------------------------
    bounds = [(0.0, 1.0)] * n
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - desired_sum},)

    best_h = h0
    best_val = c5_of(best_h)

    start_time = time.time()
    time_elapsed = lambda: time.time() - start_time
    time_remaining = lambda: max(0.0, budget_s - time_elapsed())

    try:
        res = minimize(lambda x: c5_of(x), best_h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 500, 'ftol': 1e-9, 'disp': False})
        if res.success:
            cand = project(res.x)
            val = c5_of(cand)
            if val < best_val:
                best_h, best_val = cand, val
    except Exception:
        pass

    # ----------------------------------------------------------------------
    # Light stochastic refinement while time permits
    # ----------------------------------------------------------------------
    it = 0
    while time_remaining() > 0.5 and it < 20000:
        # magnitude shrinks as time runs out
        scale = 0.15 * (time_remaining() / budget_s)
        perturb = np.random.randn(n) * scale
        cand = project(best_h + perturb)
        val = c5_of(cand)
        if val < best_val:
            best_h, best_val = cand, val
        it += 1

    # Final safety projection
    best_h = project(best_h)
    best_val = c5_of(best_h)

    # ----------------------------------------------------------------------
    # Return result
    # ----------------------------------------------------------------------
    return best_h, best_val, n
