import time
import math
import numpy as np

def _evaluate(h):
    """Return the C5 value for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project(h):
    """Project h onto the feasible set:
       0 ≤ h_i ≤ 1 and sum(h) = n/2 (average = 0.5)."""
    n = h.size
    target_sum = n * 0.5
    h = np.clip(h, 0.0, 1.0)

    # simple iterative offset‑and‑clip to satisfy the sum constraint
    for _ in range(10):
        cur_sum = h.sum()
        delta = (target_sum - cur_sum) / n
        if abs(delta) < 1e-12:
            break
        h += delta
        h = np.clip(h, 0.0, 1.0)

    # final tiny redistribution if needed
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            adjust = (target_sum - cur_sum) / mask.sum()
            h[mask] += adjust
            h = np.clip(h, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):

            h = np.interp(x_new, x_old, initial_h_values)
        else:

        h = _project(h)
