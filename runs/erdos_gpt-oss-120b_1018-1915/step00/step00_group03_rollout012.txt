def _evaluate(h):
    """Wrapper for the pre‑imported evaluator."""
    return evaluate_erdos_solution(h)


def _normalize(h, target_sum):
    """Clip to [0,1] and enforce exact sum by distributing residual."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = target_sum - h.sum()
    # distribute diff to non‑saturated entries
    for _ in range(5):
        mask = (h > 0.0) & (h < 1.0)
        if not mask.any():
            break
        adjust = diff / mask.sum()
        h[mask] += adjust
        np.clip(h, 0.0, 1.0, out=h)
        diff = target_sum - h.sum()
    return h
