import time
import numpy as np

def _compute_c5(h):
    """Return max overlap integral for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def _repair(h):
    """Project h into [0,1] and enforce sum(h)=len(h)/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    S = h.sum()
    if S == 0:
        h[:] = target / n
        return h
    # scale to hit target roughly
    h = h * (target / S)
    # iterative clip-and‑redistribute (few passes suffice)
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        S = h.sum()
        if abs(S - target) < 1e-10:
            break
        h += (target - S) / n
    return h

def _local_search(start_h, time_limit, sigma0=0.05):
    """Simple hill‑climbing with Gaussian perturbations."""
    best_h = start_h.copy()
    best_score = _compute_c5(best_h)
    cur_h = best_h.copy()
    cur_score = best_score
    n = best_h.shape[0]
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # decay sigma a bit
        sigma = sigma0 * (0.99 ** (it // 100))
        cand = cur_h + np.random.normal(0.0, sigma, size=n)
        cand = _repair(cand)
        cand_score = _compute_c5(cand)
        if cand_score < cur_score:
            cur_h, cur_score = cand, cand_score
            if cand_score < best_score:
                best_h, best_score = cand, cand_score
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : list of float
        Optimized discretised values of h.
    c5_bound : float
        Achieved upper bound for C5.
    n_points : int
        Number of discretisation points used.
    """
    np.random.seed(seed)
    t_start = time.time()
    # ------------------------------------------------------------------
    # 1. Initial solution (use provided parent if any)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        # ensure feasibility
        h0 = _repair(h0)
    else:
        n0 = 42  # even
        h0 = np.random.rand(n0)
        h0 = _repair(h0)

    best_h, best_score = _local_search(
        h0, time_limit=min(budget_s * 0.2, 30.0)
    )  # spend a short slice on the base resolution

    # ------------------------------------------------------------------
    # 2. Multi‑resolution refinement (doubling points each level)
    # ------------------------------------------------------------------
    max_levels = 5          # 42 → 84 → 168 → 336 → 672 → 1344 (if time permits)
    level = 0
    while True:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break
        if level >= max_levels:

        # Determine next resolution
        n_next = best_h.shape[0] * 2
        # Interpolate to new grid

        x_new = np.linspace(0.0, 2.0, n_next, endpoint=False)
