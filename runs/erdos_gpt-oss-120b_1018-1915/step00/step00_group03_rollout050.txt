import numpy as np
import time
import random

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def repair_sum_bounds(h, desired_sum, max_iter=10, eps=1e-9):
    """
    Project h onto the feasible set:
        0 <= h_i <= 1  and  sum(h) = desired_sum
    Uses a simple water-filling / redistribution scheme.
    """
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = h.sum()
        diff = desired_sum - cur_sum
        if abs(diff) <= eps:
            break
        # indices that can still move in the needed direction
        if diff > 0:
            free = h < 1.0 - eps
        else:
            free = h > eps
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        # apply delta only to free entries
        h[free] += delta
        # re‑clip to stay in bounds
        h = np.clip(h, 0.0, 1.0)
    # final tweak: if still off by a tiny amount, distribute uniformly
    cur_sum = h.sum()
    if abs(cur_sum - desired_sum) > eps:
        h += (desired_sum - cur_sum) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def evaluate_c5(h):
    """Return max overlap integral for step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float((corr * dx).max())

def upsample_vector(v, target_n):
    """Linear interpolation from len(v) to target_n points on [0,2)."""
    n = len(v)
    if n == target_n:
        return v.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    return np.interp(x_new, x_old, v)

def mutate(h, scale):
    """Gaussian mutation with given stddev, keeping within [0,1] then repairing sum."""
    n = h.size
    # Add zero‑mean Gaussian noise
    noise = np.random.normal(loc=0.0, scale=scale, size=n)
    h_new = h + noise
    # clip to bounds
    h_new = np.clip(h_new, 0.0, 1.0)
    # repair the sum constraint
    desired = n / 2.0
    h_new = repair_sum_bounds(h_new, desired)
    return h_new

    max_iter = 200000          # safety cap
