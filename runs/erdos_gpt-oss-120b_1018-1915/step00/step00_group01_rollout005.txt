import numpy as np
import random

def compute_c5(h):
    """Maximum overlap integral for discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_bounds_sum(v, target_sum, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    """
    Project vector v onto the hyperplane sum(v)=target_sum with box bounds.
    Uses a simple iterative redistribution algorithm.
    """
    v = np.clip(v, lower, upper)
    for _ in range(max_iter):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        # indices that are not at a bound can receive/take the adjustment
        free = (v > lower + tol) & (v < upper - tol)
        if not np.any(free):
            # all variables stuck; distribute uniformly then clip again
            v += diff / len(v)
            v = np.clip(v, lower, upper)
            continue
        delta = diff / free.sum()
        v[free] += delta
        v = np.clip(v, lower, upper)
    # Final clip to avoid tiny numerical overshoots
    return v
