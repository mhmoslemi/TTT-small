import time
import numpy as np

def _evaluate(h, dx):
    """Return max overlap C5 for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _swap_mutation(h, rng):
    """Swap two entries (preserves sum, bounds automatically)."""
    i, j = rng.integers(0, h.size, size=2)
    if i == j:
        return h  # no change
    h_new = h.copy()
    h_new[i], h_new[j] = h_new[j], h_new[i]
    return h_new

def _transfer_mutation(h, rng):
    """Move a small amount from i to j while staying in [0,1]."""
    i, j = rng.integers(0, h.size, size=2)
    if i == j:
        return h
    # maximum amount we can transfer without breaking bounds
    max_down = h[i]                # amount we can take from i
    max_up = 1.0 - h[j]            # amount we can add to j
    delta_max = min(max_down, max_up)
    if delta_max <= 0:
        return h
    # random transfer up to delta_max
    delta = rng.random() * delta_max
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
