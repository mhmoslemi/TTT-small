import time
import numpy as np

def _evaluate(h, dx):
    """Return C5 value for step function h (numpy array)."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_swap(h):
    """Return a new array with two entries swapped."""
    i, j = np.random.randint(0, h.size, size=2)
    while i == j:
        j = np.random.randint(0, h.size)
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h

    np.random.seed(seed)
    start_time = time.time()

        if not np.isclose(cur_sum, target_sum, atol=1e-8):
            # Simple scaling then clipping + redistribution to satisfy constraints
            scale = target_sum / cur_sum if cur_sum != 0 else 0.0
            h0 = h0 * scale
