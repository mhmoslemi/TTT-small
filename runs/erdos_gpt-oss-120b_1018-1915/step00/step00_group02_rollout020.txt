def project(h):
    # enforce [0,1]
    h = np.clip(h,0,1)
    # enforce sum = n/2
    desired = len(h)/2
    # simple algorithm: while sum not close
    for _ in range(10):
        diff = desired - h.sum()
        if abs(diff) < 1e-8:
            break
        # distribute equally to non-boundary points
        free = ((h>0) & (h<1))
        if free.any():
            delta = diff / free.sum()
            h[free] += delta
            # clip again
            h = np.clip(h,0,1)
        else:
            # all at bounds, adjust by adding to all then clip
            h += diff/len(h)
            h = np.clip(h,0,1)
    return h

import time
import numpy as np

def _project_to_feasibility(vec):
    """Clip to [0,1] and enforce sum(vec) = len(vec)/2 (average 0.5)."""
    n = len(vec)
    desired_sum = n / 2.0
    vec = np.clip(vec, 0.0, 1.0)
    # iterative affine‑box projection
    for _ in range(15):
        cur_sum = vec.sum()
        diff = desired_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        free = (vec > 0.0) & (vec < 1.0)
        if free.any():
            delta = diff / free.sum()
            vec[free] += delta
            vec = np.clip(vec, 0.0, 1.0)
        else:
            # all at bounds – distribute uniformly then clip again
            vec += diff / n
            vec = np.clip(vec, 0.0, 1.0)
    return vec

def _c5_from_h(h):
    """Compute max_k ∫ h(x)*(1‑h(x+k)) dx using discrete correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

    np.random.seed(seed)

    best_h = h.copy()
    best_score = _c5_from_h(best_h)
