import time
import math
import random
import numpy as np

def _c5_score(h):
    """Return C5 for step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def _random_move(h, rng):
    """Return a new feasible h after moving mass between two positions."""
    n = h.size
    i, j = rng.choice(n, 2, replace=False)
    # amount that can be taken from i and added to j while staying in [0,1]
    max_take = h[i]
    max_give = 1.0 - h[j]
    max_delta = min(max_take, max_give)
    if max_delta <= 0.0:
        return None
    delta = rng.random() * max_delta
    if delta == 0.0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def _anneal(initial_h, budget_s, seed):
    rng = np.random.default_rng(seed)
    n = initial_h.size
    target_sum = n / 2.0
    # ensure feasibility of the start
    h = np.clip(initial_h, 0.0, 1.0)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple proportional rescaling then clip and fix residual
        h = h * (target_sum / cur_sum)
        # clip again
        h = np.clip(h, 0.0, 1.0)
        # fix any remaining discrepancy by distributing uniformly
        diff = target_sum - h.sum()
        while abs(diff) > 1e-12:
            free = np.where((diff > 0) & (h < 1.0) | (diff < 0) & (h > 0.0))[0]
            if free.size == 0:
                break
            adjust = diff / free.size
            adjust = np.clip(adjust, -h[free], 1.0 - h[free])
            h[free] += adjust
            diff = target_sum - h.sum()
    best_h = h.copy()
    best_score = _c5_score(best_h)

    start = time.time()
    temp0 = 0.02
    temp_end = 0.0005
    it = 0
    while True:
        it += 1
        # linear cooling
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break
        t = elapsed / budget_s
        temp = temp0 * (1 - t) + temp_end * t

        cand = _random_move(h, rng)
        if cand is None:
            continue
        cand_score = _c5_score(cand)
        delta = cand_score - best_score
        if delta < 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            h = cand
            if cand_score < best_score:
                best_score = cand_score
                best_h = cand.copy()
    return best_h, best_score
