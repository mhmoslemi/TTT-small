import time
import numpy as np

def _c5_value(h, dx):
    """Compute C5 for a given step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (represented by n points) to lower the
    Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h (length n_points).
    c5_bound : float
        Achieved upper bound for C5.
    n_points : int
        Number of discretisation points (size of h_values).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Determine problem size and initial vector
    # ------------------------------------------------------------------
    try:
        # Provided by the harness (e.g., n = 45)
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        # Fallback: use a uniform function satisfying the integral constraint
        n_fallback = 45
        h0 = np.full(n_fallback, 0.5, dtype=float)

    n = h0.size
    dx = 2.0 / n

    # Verify feasibility; if not, project onto feasible set
    total = h0.sum()
    target = n / 2.0
    if not np.isclose(total, target, atol=1e-9):
        # Scale uniformly and clip to keep within [0,1]
        h0 = np.clip(h0 * (target / total), 0.0, 1.0)
        # Adjust residual error by adding to the first element (keeps sum exact)
        residual = target - h0.sum()
        h0[0] = np.clip(h0[0] + residual, 0.0, 1.0)

    # ------------------------------------------------------------------
    # 2. Simulated annealing configuration
    # ------------------------------------------------------------------
    T = 50.0                # initial temperature
    T_min = 1e-4
    cooling = 0.995
    max_iter = 10**9        # effectively unlimited, bounded by time
    best_h = h0.copy()
    best_score = _c5_value(best_h, dx)
    cur_h = best_h.copy()
    cur_score = best_score

    # Helper for quick swap mutation
    def propose_swap(arr):
        i, j = np.random.choice(n, size=2, replace=False)
        new = arr.copy()
        new[i], new[j] = new[j], new[i]
        return new

    # ------------------------------------------------------------------
    # 3. Main SA loop
    # ------------------------------------------------------------------
    it = 0
    while time.time() - start_time < budget_s - 0.05 and it < max_iter:
        it += 1
        cand_h = propose_swap(cur_h)
        cand_score = _c5_value(cand_h, dx)

        # Accept if better, otherwise probabilistically
        if cand_score < cur_score:
            accept = True
        else:
            delta = cur_score - cand_score
            accept = np.exp(delta / T) > np.random.rand()

        if accept:
            cur_h = cand_h
            cur_score = cand_score
            if cand_score < best_score:
                best_h = cand_h.copy()
                best_score = cand_score

        # Temperature schedule
        if T > T_min:
            T *= cooling

        # Occasionally randomize to escape deep plateaus
        if it % 5000 == 0:
            if np.random.rand() < 0.05:
                # Random shuffle while preserving sum
                shuffled = np.random.permutation(cur_h)
                cur_h = shuffled
                cur_score = _c5_value(cur_h, dx)

    # ------------------------------------------------------------------
    # 4. Return result
    # ------------------------------------------------------------------
    return best_h, best_score, n
