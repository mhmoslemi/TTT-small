def _project_bounds_sum(v, target_sum, tol=1e-9, max_iter=100):
    """Project vector v onto [0,1] with specified L1 sum = target_sum."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = v.sum()
        diff = cur_sum - target_sum
        if abs(diff) <= tol:
            break
        # indices that can move (not at a bound)
        free = (v > 0.0 + tol) & (v < 1.0 - tol)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds, just distribute the remaining diff uniformly
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
            break
        adjust = diff / n_free
        v[free] -= adjust
        v = np.clip(v, 0.0, 1.0)
    # final clipping to enforce bounds
    return np.clip(v, 0.0, 1.0)

def _evaluate(h):
    """Return the C5 value for a discretised step function h."""
    n = len(h)
    dx = 2.0 / n
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
