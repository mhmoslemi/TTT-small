    import time, math
    import numpy as np

    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialization
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 100
        h = np.full(n, 0.5, dtype=float)  # uniform satisfies the integral constraint
    n = h.size

    # Ensure the basic feasibility (should already hold)
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum exactly to n/2 by a simple redistribution if needed
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # add diff to a random index that can absorb it
        for _ in range(10):
            idx = rng.integers(0, n)
            cap = 1.0 - h[idx] if diff > 0 else h[idx]
            if abs(cap) >= abs(diff):
                h[idx] += diff
                break
        h = np.clip(h, 0.0, 1.0)

    # ------------------------------------------------------------------
    # Helper: objective evaluation (uses the pre‑imported evaluator)
    # ------------------------------------------------------------------
    def score_of(arr):
        return evaluate_erdos_solution(arr)

    best_h = h.copy()
    best_score = score_of(best_h)

    # ------------------------------------------------------------------
    # Stochastic search parameters
    # ------------------------------------------------------------------
    temperature = 0.12          # initial step scale
    min_temp = 1e-5
    decay = 0.99995
    iter_count = 0
    stale = 0                    # iterations without improvement

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.95:
        iter_count += 1

        # propose a new candidate by swapping mass between two points
        cand = best_h.copy()
        # determine number of elementary swaps (more later in the run)
        swaps = 1
        if rng.random() < 0.08:          # occasional larger move
            swaps = rng.integers(2, 5)

        for _ in range(swaps):
            i, j = rng.integers(0, n, size=2)
            while j == i:
                j = rng.integers(0, n)
            # feasible epsilon range preserving bounds
            max_inc = 1.0 - cand[i]
            max_dec = cand[j]
            max_eps = min(max_inc, max_dec)
            if max_eps <= 0:
                continue
            eps = rng.normal(0.0, temperature)
            eps = max(-max_dec, min(eps, max_inc))   # clamp to feasible
            cand[i] += eps
            cand[j] -= eps

        # safety clipping (should be unnecessary but cheap)
        np.clip(cand, 0.0, 1.0, out=cand)

        cand_score = score_of(cand)

        # Acceptance rule: improve always, otherwise Metropolis
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score
            stale = 0
        else:
            stale += 1
            prob = math.exp((best_score - cand_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                best_h = cand
                best_score = cand_score

        # Temperature schedule
        temperature = max(min_temp, temperature * decay)

        # Restart heuristics if stuck
        if stale >= 20000:
            # small random perturbation preserving sum
            idx = rng.integers(0, n)
            delta = rng.uniform(-0.05, 0.05)
            delta = max(-best_h[idx], min(delta, 1.0 - best_h[idx]))
            best_h[idx] += delta
            best_h = np.clip(best_h, 0.0, 1.0)
