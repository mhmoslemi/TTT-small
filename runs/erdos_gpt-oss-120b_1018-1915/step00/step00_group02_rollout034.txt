def projected_gradient_descent(h0, budget, max_iter=5000):
    h = h0.copy()
    n = len(h)
    target = n/2
    start = time.time()
    best_h = h.copy()
    best_score = evaluate_erdos_solution(h)
    step = 0.5
    for it in range(max_iter):
        if time.time() - start > budget:
            break
        # compute overlap array
        O = np.correlate(h, 1-h, mode='full')
        k = np.argmax(O)
        # compute gradient
        g = np.zeros_like(h)
        # shift arrays
        shift = k - (n-1)  # because correlation indices offset
        # correlation index i corresponds to shift = i - (n-1)
        shift_val = shift
        # compute (1 - h shifted by shift) - h shifted by -shift
        # implement shift function
        def shift_arr(arr, s):
            if s>0:
                res = np.concatenate((np.zeros(s), arr[:-s]))
            elif s<0:
                res = np.concatenate((arr[-s:], np.zeros(-s)))
            else:
                res = arr.copy()
            return res
        g = (1 - shift_arr(h, shift_val)) - shift_arr(h, -shift_val)
        # gradient step
        y = h - step * g

def project_to_feasibility(y, target):
    x = np.clip(y, 0.0, 1.0)
    # iterative adjustment
    for _ in range(10):
        s = x.sum()
        if abs(s - target) < 1e-9:
            break
        diff = s - target
        free = (x > 0) & (x < 1)
        if not np.any(free):
            # distribute diff equally ignoring bounds
            x = np.clip(x - diff/len(x), 0, 1)
            continue
        adjust = diff / free.sum()
        x[free] -= adjust
        x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np

# helpers assumed to be pre‑imported:
# evaluate_erdos_solution, initial_h_values (optional)

def _shift(arr, s):
    """Shift arr by s positions with zero padding (linear, not circular)."""
    n = arr.size
    if s > 0:
        return np.concatenate((np.zeros(s, dtype=arr.dtype), arr[:n - s]))
    if s < 0:
        s = -s
        return np.concatenate((arr[s:], np.zeros(s, dtype=arr.dtype)))
    return arr.copy()


def _project_feasible(y, target):
    """Project y onto {0≤x≤1, sum(x)=target} using simple water‑filling."""
    x = np.clip(y, 0.0, 1.0)
    for _ in range(15):
        s = x.sum()
        if abs(s - target) < 1e-12:
            break
        diff = s - target
        free = (x > 0) & (x < 1)
        if not free.any():
            # all at bounds – distribute remaining diff uniformly and re‑clip
            x = np.clip(x - diff / x.size, 0.0, 1.0)
            continue
        adjust = diff / free.sum()
        x[free] -= adjust
        x = np.clip(x, 0.0, 1.0)
    return x


def _objective_max_overlap(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx using discrete correlation."""
    dx = 2.0 / h.size
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _pgd_optimize(h0, time_budget):
    """Projected gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.size
    target = n / 2.0
    best_h, best_score = h.copy(), _objective_max_overlap(h)
    step = 0.5
    start = time.time()
    max_iter = 2000

    for _ in range(max_iter):
        if time.time() - start > time_budget:
            break

        corr = np.correlate(h, 1.0 - h, mode="full")
        k_idx = int(np.argmax(corr))
        shift_val = k_idx - (n - 1)          # shift corresponding to max overlap

        grad = (1.0 - _shift(h, shift_val)) - _shift(h, -shift_val)

        y = h - step * grad
        h = _project_feasible(y, target)

        score = _objective_max_overlap(h)
        if score < best_score - 1e-7:
            best_score, best_h = score, h.copy()

        # simple step‑size adaptation
        if score > best_score:
            step *= 0.5
        else:
            step = min(step * 1.02, 2.0)

    return best_h, best_score
