import time
import numpy as np

def _evaluate_h(h):
    """Return C5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _project_h(h):
    """
    Project vector h onto the feasible set:
      0 <= h_i <= 1   and   sum(h) = n/2  (i.e. average = 0.5)
    Uses an iterative redistribution scheme.
    """
    h = np.clip(h, 0.0, 1.0)
    n = h.shape[0]
    target = n * 0.5
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # free indices are those not at a bound that would be violated by moving
        free = (h > 0.0) & (h < 1.0)
        # if no free variable, relax bounds a little
        if not np.any(free):
            free = np.ones_like(h, dtype=bool)
        delta = diff / free.sum()
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final tiny mass‑balance correction (adds to first non‑bound element)
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        idx = np.where((h > 0.0) & (h < 1.0))[0]
        if idx.size == 0:
            idx = np.arange(n)
        h[idx[0]] = np.clip(h[idx[0]] + diff, 0.0, 1.0)
    return h

def _local_search(start_h, max_iters, time_limit, start_time):
    """
    Simple stochastic hill‑climbing with projection.
    Returns the best vector and its C5 value found.
    """
    best = _project_h(start_h.copy())
    best_score = _evaluate_h(best)
    n = best.shape[0]
    while max_iters > 0:
        if time.time() - start_time > time_limit:
            break
        # random perturbation
        h_new = best.copy()
        # number of coordinates to modify (1‑5% of n)
        k = np.random.randint(1, max(2, n // 20) + 1)
        idx = np.random.choice(n, k, replace=False)
        h_new[idx] += np.random.normal(scale=0.08, size=k)
        h_new = _project_h(h_new)
        score = _evaluate_h(h_new)
        if score < best_score:
            best, best_score = h_new, score
        max_iters -= 1
    return best, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.

    Returns
    -------
    h_values : np.ndarray
        The best discrete function values found.
    c5_bound : float
        The corresponding upper bound C5.
    n_points : int
        Length of the discretisation.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        h = np.full(45, 0.5, dtype=np.float64)
    h = _project_h(h)

    best_h = h.copy()
    best_score = _evaluate_h(best_h)
    n = h.shape[0]

    # ----- multi‑resolution optimisation -------------------------------
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining < 5.0:               # keep a small safety margin
            break

        # allocate a chunk of the remaining budget to this resolution
        # (roughly 3000 evaluations or all remaining time, whichever is smaller)
        max_iters = 3000
        time_limit = start_time + remaining

        h_opt, score_opt = _local_search(best_h, max_iters, time_limit, start_time)
        if score_opt < best_score:
            best_h, best_score = h_opt, score_opt

        # try to refine by doubling resolution (stop if too large)
        if n * 2 > 480:   # keep vectors reasonably small for the time budget
            break
        h2 = np.empty(n * 2, dtype=np.float64)
        h2[::2] = best_h
        # odd points get the average of neighbours (non‑periodic edge handling)
        shifted = np.empty_like(best_h)
        shifted[:-1] = best_h[1:]
        shifted[-1] = best_h[-1]          # repeat last value for the final odd slot
        h2[1::2] = (best_h + shifted) * 0.5
        h2 = _project_h(h2)

        best_h = h2
        n = best_h.shape[0]

    return best_h, best_score, n
