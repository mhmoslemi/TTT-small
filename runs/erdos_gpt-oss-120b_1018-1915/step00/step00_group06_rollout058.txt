def project_bounds_sum(h, target):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # iterative water filling
    lower = np.zeros(n, dtype=bool)
    upper = np.zeros(n, dtype=bool)
    while True:
        free = ~(lower | upper)
        if not np.any(free):
            break
        total_free = np.sum(h[free])
        needed = target - (np.sum(h[lower]) + np.sum(h[upper]))
        # distribute equally
        if np.abs(needed) < 1e-12:
            break
        increment = needed / np.sum(free)
        h[free] += increment
        # enforce bounds
        newly_lower = h < 0
        newly_upper = h > 1
        if np.any(newly_lower):
            h[newly_lower] = 0
            lower[newly_lower] = True
        if np.any(newly_upper):
            h[newly_upper] = 1
            upper[newly_upper] = True
    # final clip
    h = np.clip(h, 0, 1)
    return h

def project_feasible(h, target, tol=1e-8, max_iter=100):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        total = h.sum()
        diff = target - total
        if abs(diff) < tol:
            break
        # indices not at bound
        free = (h > 0 + 1e-12) & (h < 1 - 1e-12)
        if not np.any(free):
            # cannot adjust, break
            break
        inc = diff / np.sum(free)
        h[free] += inc
        # clip again
        h = np.clip(h, 0.0, 1.0)
    # final correction if still off (rare)
    total = h.sum()
    if abs(total - target) > 1e-6:
        # distribute residual uniformly and clip
        h += (target - total) / len(h)
        h = np.clip(h, 0.0, 1.0)
        # renormalize linearly
        # final scaling factor
        scale = target / h.sum()
        h *= scale
        h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the discretized overlap constant for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * float(np.max(corr))


def project_feasible(x: np.ndarray, target_sum: float, tol: float = 1e-8, max_iter: int = 100) -> np.ndarray:
    """
    Project vector x onto the feasible set:
        0 <= v_i <= 1  and  sum(v) = target_sum
    Uses a simple water‑filling scheme.
    """
    v = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        total = v.sum()
        diff = target_sum - total
        if abs(diff) < tol:
            break
        free = (v > 0.0) & (v < 1.0)
        if not np.any(free):
            break
        inc = diff / np.sum(free)
        v[free] += inc
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    total = v.sum()
    if abs(total - target_sum) > 1e-6:
        v += (target_sum - total) / len(v)
        v = np.clip(v, 0.0, 1.0)
        scale = target_sum / v.sum()
        v *= scale
        v = np.clip(v, 0.0, 1.0)
    return v
