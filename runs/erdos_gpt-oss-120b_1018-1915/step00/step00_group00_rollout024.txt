import numpy as np
import time
from math import exp

def _project_to_constraints(x, target_sum):
    """Project vector x onto [0,1] box and sum == target_sum."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(20):
        total = x.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = x < 1.0
        else:
            free = x > 0.0
        if not np.any(free):
            break
        adjust = diff / free.sum()
        x[free] += adjust
        x = np.clip(x, 0.0, 1.0)
    return x
