import time
import numpy as np

def _project_to_sum(vec, target):
    """Clip to [0,1] and adjust to meet exact sum (simple iterative)."""
    v = np.clip(vec, 0.0, 1.0)
    total = v.sum()
    if abs(total - target) <= 1e-12:
        return v
    for _ in range(20):
        diff = target - total
        # indices that can still move
        mask = (v > 0.0) & (v < 1.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        v[mask] += delta
        v = np.clip(v, 0.0, 1.0)
        total = v.sum()
        if abs(total - target) <= 1e-12:
            break
    return v


def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimizer for the Erdős C5 constant.

    Returns
    -------
    h_values : list of float
        The step‑function values on the chosen discretisation.
    c5_bound : float
        The achieved upper bound (the evaluated C5).
    n_points : int
        Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.shape[0]
        # enforce exact integral constraint
        h = _project_to_sum(h, n / 2.0)
    else:
        n = 80
        h = np.full(n, 0.5, dtype=float)            # average = 0.5 gives integral 1
        h = _project_to_sum(h, n / 2.0)

    # ------------------------------------------------------------------
    # Helper to evaluate the current vector
    # ------------------------------------------------------------------
    def _score(vec):
        # evaluate_erdos_solution is provided by the harness
        return float(evaluate_erdos_solution(vec)[0])

    best_h = h.copy()
    best_score = _score(best_h)

    # ------------------------------------------------------------------
    # Simple pairwise transfer hill‑climbing with annealing‑like step decay
    # ------------------------------------------------------------------
    step = 0.2                     # maximal transfer amount per move
    it = 0
    # parameters controlling schedule
    reduction_interval = 2000
    min_step = 1e-4

    while time.time() - start_time < budget_s - 0.2:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # maximum transferable amount respecting bounds
        max_transfer = min(step, best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0.0:
            continue

        delta = rng.random() * max_transfer

        # propose move
        h[i] -= delta
        h[j] += delta

        # evaluate
        cur_score = _score(h)

        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
        else:
            # revert move
            h[i] += delta
            h[j] -= delta

        it += 1
        # gradually shrink the step size
        if it % reduction_interval == 0:
            step = max(min_step, step * 0.9)

    # ------------------------------------------------------------------
    # Return the best found solution
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_score), n)
