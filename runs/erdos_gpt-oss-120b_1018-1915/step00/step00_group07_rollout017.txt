import time
import numpy as np
import math

def _evaluate_c5(h, dx):
    """Return max overlap integral for a given h vector."""
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def _initial_vector(n):
    """Create a simple feasible start if no parent is supplied."""
    # Uniform distribution that satisfies the integral constraint.
    # Sum must be n/2, so each entry = 0.5.
    return np.full(n, 0.5, dtype=np.float64)
