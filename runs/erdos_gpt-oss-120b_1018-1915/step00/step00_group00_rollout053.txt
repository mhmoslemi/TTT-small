import time
import numpy as np

def _compute_c5(h, dx):
    """Return the overlap constant C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _perturb(h, rng):
    """Return a new h obtained by moving a random amount from i to j while
    keeping 0≤h≤1 and preserving the total sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # choose direction: move from i to j or opposite
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_transfer = min(h[src], 1.0 - h[dst])
    if max_transfer <= 0:
        return h  # no feasible move
    eps = rng.random() * max_transfer
    new_h = h.copy()
    new_h[src] -= eps
    new_h[dst] += eps
    return new_h
