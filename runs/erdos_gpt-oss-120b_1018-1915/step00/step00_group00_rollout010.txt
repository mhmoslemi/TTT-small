import numpy as np
import time
import math

# Helper: project onto [0,1] with a given sum (target_sum)
def _project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        if s < target_sum:
            deficit = target_sum - s
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0:
                break
            inc = deficit / cnt
            h = np.where(mask, np.minimum(h + inc, 1.0), h)
        else:
            excess = s - target_sum
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0:
                break
            dec = excess / cnt
            h = np.where(mask, np.maximum(h - dec, 0.0), h)
    return h

# Helper: propose a small perturbation that respects the sum constraint after projection
def _perturb(h, target_sum, step):
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = (np.random.rand() * 2.0 - 1.0) * step
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    new_h = _project_h(new_h, target_sum)
    return new_h

# Main entry point
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # Determine problem size
    if 'initial_h_values' in globals() and initial_h_values is not None:
        n_points = int(len(initial_h_values))
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.random.rand(n_points)
    target_sum = n_points / 2.0          # because integral must be 1, dx = 2/n
    h = _project_h(h, target_sum)

    # Evaluation function (fallback if evaluate_erdos_solution is missing)
    def _evaluate(vec):
        try:
            return evaluate_erdos_solution(vec)
        except Exception:
            dx = 2.0 / n_points
            corr = np.correlate(vec, 1.0 - vec, mode='full')
            return np.max(corr) * dx

    best_h = h.copy()
    best_score = _evaluate(best_h)

    start = time.time()
    cur_h = h.copy()
    cur_score = best_score

    # Simulated‑annealing parameters
    step_size = 0.07          # magnitude of random moves
    temp = 0.02
    temp_decay = 0.9995

    while time.time() - start < budget_s:
        # generate neighbour
        cand_h = _perturb(cur_h, target_sum, step_size)
        cand_score = _evaluate(cand_h)

        # accept if better or according to temperature
        if cand_score < cur_score or np.random.rand() < math.exp(-(cand_score - cur_score) / max(temp, 1e-12)):
            cur_h, cur_score = cand_h, cand_score

        # keep global best
        if cand_score < best_score:
            best_h, best_score = cand_h.copy(), cand_score

        # anneal
        temp *= temp_decay
        # occasionally reduce step size to fine‑tune
        if temp < 1e-4:
            step_size *= 0.9
            temp = 0.02

    return best_h, best_score, n_points
