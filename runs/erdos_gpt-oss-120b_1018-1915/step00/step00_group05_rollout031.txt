import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretized) to minimize the
    Erdős overlap constant C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n_points = init.size
    else:
        n_points = kwargs.get('n_points', 70)
        init = None

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because ∫h = 1 and dx = 2/n

    def repair(v):
        """Clip to [0,1] and enforce exact sum constraint."""
        v = np.clip(v, 0.0, 1.0)
        s = v.sum()
        if s == 0.0:
            v[:] = target_sum / n_points
            return v
        v *= (target_sum / s)
        v = np.clip(v, 0.0, 1.0)
        # distribute any residual uniformly (few iterations suffice)
        for _ in range(5):
            diff = target_sum - v.sum()
            if abs(diff) < 1e-12:
                break
            add = diff / n_points
            v += add
            v = np.clip(v, 0.0, 1.0)
        return v

    # Initial candidate
    if init is not None and init.size == n_points:
        current = repair(init.copy())
    else:
        current = repair(rng.random(n_points))

    def c5_of(h):
        """Compute discrete C5."""
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    best = current.copy()
    best_score = c5_of(best)

    start_time = time.time()
    # Simple simulated annealing / stochastic hill‑climb
    while time.time() - start_time < budget_s * 0.9:
        cand = best.copy()
        # Choose two distinct indices and move a small amount
        i, j = rng.integers(0, n_points, size=2)
        if i == j:
            continue
        eps = rng.random() * 0.05  # max transferred mass
        delta = min(eps, cand[i])   # cannot go below 0
        cand[i] -= delta
        cand[j] = min(1.0, cand[j] + delta)
        cand = repair(cand)

        score = c5_of(cand)

        # Accept improvements; occasionally accept worse moves (temperature schedule)
        if score < best_score - 1e-10:
            best, best_score = cand, score
        else:
            # temperature decays linearly
            elapsed = time.time() - start_time
            T = max(0.0, 1.0 - elapsed / (budget_s * 0.9))
            if rng.random() < np.exp((best_score - score) / (T + 1e-12)):
                best, best_score = cand, score

    # Final feasibility check
    best = repair(best)
    best_score = c5_of(best)

    return best, float(best_score), n_points
