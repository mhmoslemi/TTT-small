import time
import numpy as np

# evaluate_erdos_solution and initial_h_values are provided in the execution environment.

def _project_feasible(vec, target_sum):
    """Clip to [0,1] and adjust to meet the linear equality sum(vec)=target_sum."""
    v = np.clip(vec, 0.0, 1.0)
    # water‑filling style adjustment
    tol = 1e-12
    max_iter = 10
    for _ in range(max_iter):
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        # indices that can still move
        free = np.where((v > tol) & (v < 1 - tol))[0]
        if free.size == 0:
            # all at bounds – distribute remaining diff proportionally to capacity
            caps = np.where(v < 0.5, 1 - v, v)  # capacity to move without breaking bounds
            total_cap = caps.sum()
            if total_cap == 0:
                break
            delta = diff * caps / total_cap
            v += delta
            v = np.clip(v, 0.0, 1.0)
            break
        delta = diff / free.size
        # keep within bounds for the free set
        delta = np.clip(delta, -np.min(v[free]), np.min(1 - v[free]))
        v[free] += delta
    # final clip to guard against tiny numerical drift
    return np.clip(v, 0.0, 1.0)

def _initial_h(n_points):
    """Create a starting vector of length n_points.
    If a parent construction is available, up‑sample it; otherwise use a simple
    two‑block pattern that satisfies the integral constraint."""
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        # linear interpolation to the target length
        x_parent = np.linspace(0, 2, len(parent), endpoint=False)
        x_target = np.linspace(0, 2, n_points, endpoint=False)
        h = np.interp(x_target, x_parent, parent)
    else:
        # simple construction: half of the interval at value 1, rest 0
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
    # enforce exact integral
    target = n_points / 2.0
    h = _project_feasible(h, target)
    return h

def _c5_bound(h):
    """Return the overlap bound for a candidate step function."""
    # evaluate_erdos_solution is assumed to follow the specification:
    #   max(np.correlate(h, 1-h, mode="full")) * dx
    return evaluate_erdos_solution(h)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] improving the Erdős C5 bound."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a modest resolution that balances speed and quality.
    n_points = 96  # multiple of 2, gives dx = 2/96 ≈ 0.0208

    target_sum = n_points / 2.0
    h_current = _initial_h(n_points)
    best_h = h_current.copy()
    best_score = _c5_bound(best_h)

    # Simulated‑annealing style parameters
    max_iter = 50000
    temp_start = 0.05
    temp_end = 1e-4

    for it in range(1, max_iter + 1):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # linear temperature schedule
        t = temp_start + (temp_end - temp_start) * (it / max_iter)

        # propose a perturbation
        proposal = h_current + rng.normal(scale=0.02, size=n_points)
        proposal = _project_feasible(proposal, target_sum)

        score = _c5_bound(proposal)

        # accept if improvement or with probability exp(-(Δ)/t)
        if score < best_score:
            best_h, best_score = proposal.copy(), score
            h_current = proposal.copy()
        else:
            delta = score - best_score

            best_h, best_score = cand, sc

    return (best_h.tolist(), float(best_score), int(n_points))
