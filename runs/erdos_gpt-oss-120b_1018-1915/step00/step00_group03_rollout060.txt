import numpy as np, time, math, random

def _max_corr(arr):
    """Maximum of correlation h with (1-h) over all integer shifts."""
    n = arr.size
    m = 1
    while m < 2 * n:
        m <<= 1
    H = np.fft.rfft(arr, m)
    G = np.fft.rfft(np.flip(1.0 - arr), m)
    conv = np.fft.irfft(H * G, m)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)   # dx = 2/n


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h on [0,2] (as a vector), its C5 bound, and n."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # -------------------------------------------------------------
    # Choose a moderate resolution (power of two for FFT convenience)
    n_points = 256                     # 256 samples → dx = 2/256
    target_ones = n_points // 2       # integral constraint: sum(h)=n/2
    dx = 2.0 / n_points

    # -------------------------------------------------------------
    # Initialise h
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # Linear interpolation onto the new grid
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        # Pure binary start if no parent is present
        h = np.zeros(n_points)
        h[np.random.choice(n_points, target_ones, replace=False)] = 1.0

    # Force binary values and exact integral (sum = target_ones)
    # Take the largest entries as ones, rest as zeros
    idx_desc = np.argsort(-h)
    h[:] = 0.0
    h[idx_desc[:target_ones]] = 1.0

    # -------------------------------------------------------------
    # Simulated‑annealing optimisation on the binary space.
    best_h = h.copy()
    best_val = _max_corr(best_h)

    cur_h = h.copy()
    cur_val = best_val

    # Annealing schedule parameters
    max_iter = 500_000
    temp_start = 0.5
    temp_end = 0.001

    # Pre‑compute index arrays for speed (will be refreshed after each swap)
    ones_idx = np.flatnonzero(cur_h > 0.5)
    zeros_idx = np.flatnonzero(cur_h < 0.5)

    it = 0
    while True:
        # Time‑out guard – keep a safety margin
        if time.time() - start_time > budget_s * 0.95:
            break
        if it >= max_iter:
            break

        # Geometric temperature decay
        t = temp_start * ((temp_end / temp_start) ** (it / max_iter))

        # Choose a random 1‑position and a random 0‑position to swap
        if ones_idx.size == 0 or zeros_idx.size == 0:
            break
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)

        # Perform the swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0

        new_val = _max_corr(cur_h)
        delta = new_val - cur_val

        # Metropolis acceptance
        if delta < 0 or np.random.rand() < math.exp(-delta / t):

            # Update index lists after an accepted move
            ones_idx = np.flatnonzero(cur_h > 0.5)

            if cur_val < best_val:

                best_h = cur_h.copy()
        else:
            # Revert the swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0

        it += 1

    # Return the best binary construction found
    return (best_h, float(best_val), n_points)
