import time
import numpy as np
import random

# evaluate_erdos_solution is pre‑imported by the harness.
# initial_h_values may be provided; otherwise we start from a random binary vector.

def _make_binary(vec):
    """Return a binary (0/1) vector with the same length and exactly half ones."""
    n = vec.size
    target = n // 2
    # indices of the largest target values become 1
    idx = np.argpartition(-vec, target)[:target]
    binary = np.zeros(n, dtype=np.float64)
    binary[idx] = 1.0
    return binary

def _initial_vector(n, seed):
    """Create a starting binary vector of length n."""
    rng = np.random.default_rng(seed)
    vec = rng.random(n)
    return _make_binary(vec)

def _objective(h):
    """Compute C5 for a candidate step function h."""
    # The evaluator provided by the harness already respects the exact definition.
    return evaluate_erdos_solution(h)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) minimizing the Erdős C5 bound.

    The algorithm performs a simple stochastic hill‑climb on binary vectors
    with exactly half the entries equal to 1 (integral constraint).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        n = init.size
        h = _make_binary(init)                     # enforce binary feasibility
    else:
        n = 100                                     # reasonable default size
        h = _initial_vector(n, seed)

    # Ensure exact integral constraint (half ones)
    assert h.sum() == n // 2

    best_h = h.copy()
    best_score = _objective(best_h)

    # Pre‑compute indices of ones and zeros for fast swapping
    ones = np.flatnonzero(best_h == 1.0)
    zeros = np.flatnonzero(best_h == 0.0)

    # Main stochastic search loop
    ITER_BATCH = 5000   # evaluate this many proposals before checking time
    while True:
        # stop if budget exhausted
        if time.time() - start_time > budget_s * 0.99:
            break

        # Propose a batch of swaps
        for _ in range(ITER_BATCH):
            # choose a random pair (i from ones, j from zeros)
            i = rng.choice(ones)
            j = rng.choice(zeros)

            # perform swap
            best_h[i], best_h[j] = 0.0, 1.0

            # evaluate
            score = _objective(best_h)

            # Accept if improvement; otherwise revert
            if score < best_score:
                best_score = score
                # update index lists
                ones = np.append(ones[ones != i], j)
                zeros = np.append(zeros[zeros != j], i)
            else:
                # revert swap
                best_h[i], best_h[j] = 1.0, 0.0

        # occasional random restart if stagnation detected
        if time.time() - start_time > budget_s * 0.5:
            # keep the current best, but occasionally try a fresh random start
            if rng.random() < 0.1:
                h = _initial_vector(n, rng.integers(1 << 30))
                score = _objective(h)
                if score < best_score:
                    best_h = h.copy()
                    best_score = score
                    ones = np.flatnonzero(best_h == 1.0)
                    zeros = np.flatnonzero(best_h == 0.0)

    # final check: ensure constraints exactly satisfied
    best_h = np.clip(best_h, 0.0, 1.0)
