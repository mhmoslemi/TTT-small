import time
import numpy as np

def _c5_value(h, dx):
    """Return the overlap bound C5 for a given step vector h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _random_transfer(rng, h):
    """Propose a new h by moving mass between two coordinates while
    respecting the [0,1] bounds and preserving the total sum."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # direction: from i to j
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        return None, None, None
    # pick a fraction of the allowable transfer
    delta = rng.random() * max_delta
    if delta == 0.0:
        return None, None, None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new, i, j


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (discretised) to minimise the
    Erdős C5 overlap bound.

    Returns
    -------

        Optimised values of h (length n_points, each in [0,1]).
    c5_bound : float
        Corresponding C5 bound.
    n_points : int
        Number of discretisation points used.
    """

    rng = np.random.default_rng(seed)

    # initialisation

        # initial_h_values may be provided by the harness

        n = 70
        h0 = np.full(n, 0.5, dtype=float)

    # enforce exact sum constraint (might be off by tiny numerical error)
    target_sum = n / 2.0
    current_sum = h0.sum()
    if abs(current_sum - target_sum) > 1e-12:
        # simple rescale then clip and re‑adjust via water‑filling
        h0 = np.clip(h0 * (target_sum / current_sum), 0.0, 1.0)
        deficit = target_sum - h0.sum()
        # distribute the deficit uniformly over non‑saturated entries
        free = np.where((h0 > 0) & (h0 < 1))[0]

            inc = deficit / free.size

    h_best = h0.copy()
