def _initial_h(n):

    return np.full(n, 0.5, dtype=float)

def _feasible_swap(h, i, j, delta):
    """Apply a swap of *delta* from i to j, staying within [0,1]. Returns actual delta."""
    # amount we can actually move without violating bounds
    max_move = min(delta, h[i], 1.0 - h[j])
    if max_move <= 0.0:
        return 0.0
    h[i] -= max_move
    h[j] += max_move
    return max_move
