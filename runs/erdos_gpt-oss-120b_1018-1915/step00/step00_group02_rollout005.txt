import time
import numpy as np
from scipy import optimize

# Helper: evaluate C5 for a discrete h
def _c5_value(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

# Project a vector onto the feasible set (box + exact integral)
def _project(h):
    h = np.clip(h, 0.0, 1.0)
    # Adjust to meet the exact integral by adding a constant offset then clipping again
    target_sum = h.size / 2.0
    current_sum = h.sum()
    offset = (target_sum - current_sum) / h.size
    h = np.clip(h + offset, 0.0, 1.0)
    # If still off (rare), rescale linearly
    cur = h.sum()
    if abs(cur - target_sum) > 1e-9:
        factor = target_sum / cur if cur != 0 else 0.0
        h = np.clip(h * factor, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # use the provided initial vector if present, else start from a simple rectangle
    try:
        init = np.array(initial_h_values, dtype=float)  # type: ignore
        base_n = init.size

        base_n = 45
        init = np.concatenate([np.ones(base_n // 2), np.zeros(base_n // 2)])
        init = _project(init)

    best_h = init.copy()
    best_val = _c5_value(best_h)
    best_n = base_n

    # resolution schedule (do not grow beyond a few thousand points)
    schedule = [base_n, base_n * 2, base_n * 4, base_n * 8]
    schedule = [n for n in schedule if n <= 2000]

    for n in schedule:
        if time.time() - start_time > budget_s * 0.95:
            break

        # generate initial guess for this resolution by linear interpolation of best_h
        x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x_new, x_old, best_h)
        h0 = _project(h0)

        # prepare bounds and equality constraint
        bounds = [(0.0, 1.0)] * n
        eq_cons = {"type": "eq", "fun": lambda x: x.sum() - n / 2.0}

        # local optimizer – we will call it a few times with random perturbations

            res = optimize.minimize(
                _c5_value,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=[eq_cons],
                options={"maxiter": 250, "ftol": 1e-9, "disp": False},
            )
            if not res.success:
                # try a cheap projection fallback
                cand = _project(res.x)
                val = _c5_value(cand)
                return cand, val
            cand = _project(res.x)  # ensure feasibility
            return cand, _c5_value(cand)
