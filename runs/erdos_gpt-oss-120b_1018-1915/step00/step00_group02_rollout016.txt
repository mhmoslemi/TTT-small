import time
import random
import numpy as np

# helpers are pre‑imported: evaluate_erdos_solution, initial_h_values (optional)

def _objective(h):
    """Return the C5 value for a candidate h."""
    return float(evaluate_erdos_solution(h))

def _project_transfer(h, i, j, delta):
    """Transfer delta from index i to j while staying in [0,1]."""
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def _upscale(h):
    """Double the resolution by repeating each sample."""
    h2 = np.repeat(h, 2)
    return h2.astype(np.float64)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the time budget."""
    random.seed(seed)
    np.random.seed(seed)

    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # simple feasible start: half ones, half zeros
        n0 = 60
        h = np.zeros(n0, dtype=np.float64)
        h[: n0 // 2] = 1.0
    n = len(h)
    # guarantee exact sum constraint (should already hold)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h *= target_sum / cur_sum

    best_h = h.copy()
    best_score = _objective(best_h)

    start_time = time.time()
    time_limit = budget_s * 0.95  # safety margin

    # ----- main stochastic improvement loop ---------------------------
    while time.time() - start_time < time_limit:
        # occasional up‑sampling when we seem stuck
        if n < 720 and random.random() < 0.001:
            best_h = _upscale(best_h)
            n = len(best_h)
            best_score = _objective(best_h)
            continue

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # maximal admissible transfer from i to j
        delta_max = min(best_h[i], 1.0 - best_h[j])
        if delta_max <= 1e-12:
            continue

        # propose a random fraction of the allowed transfer
        delta = delta_max * random.random() * 0.5  # ≤ ½ of the max
        cand_h = _project_transfer(best_h, i, j, delta)

        # evaluate candidate
        cand_score = _objective(cand_h)

        # accept if better (strict improvement)
        if cand_score < best_score:
            best_h = cand_h
            best_score = cand_score
            # early exit if target reached
            if best_score <= 0.3808:
                break

    return (best_h, best_score, len(best_h))
