import numpy as np
import time

def _evaluate_h(h, dx):
    """Return max overlap integral C5 for a given h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_sum_bound(h, target_sum):
    """
    Project vector h onto the simplex defined by:
        0 <= h_i <= 1
        sum(h) = target_sum
    Simple heuristic: clip to [0,1] then adjust uniformly.
    """
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # distribute the difference equally, then clip again
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        # if still not exact, rescale the interior
        cur = h.sum()
        if abs(cur - target_sum) > 1e-9:
            # distribute remaining error proportionally to free variables
            free = (h > 0) & (h < 1)
            if free.any():
                adjust = (target_sum - cur) / free.sum()
                h[free] += adjust
                h = np.clip(h, 0.0, 1.0)
    # final correction by brute force if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-9:
        # fix by adding/subtracting tiny epsilon to first element
        h[0] += target_sum - cur
        h[0] = np.clip(h[0], 0.0, 1.0)
    return h

def _random_feasible(n, rng):
    """Generate a random feasible h: values in [0,1] with sum = n/2."""
    # start with half ones, half zeros and shuffle
    h = np.zeros(n)
    half = n // 2
    h[:half] = 1.0
    rng.shuffle(h)
    # optionally add small noise and re‑project
    noise = rng.uniform(-0.05, 0.05, size=n)
    h = _project_sum_bound(h + noise, n / 2.0)
    return h

def _pairwise_move(h, rng):
    """Perform a random mass transfer between two coordinates while staying feasible."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # decide direction
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_transfer = min(h[src], 1.0 - h[dst])
    if max_transfer <= 0:
        return h  # no feasible move
    delta = rng.uniform(0, max_transfer)
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new
