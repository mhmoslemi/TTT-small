import time
import numpy as np
from scipy.ndimage import zoom

# Helper to compute the C5 value for a given discretized h
def _c5_value(h):
    # Use the provided evaluator if available (faster / exact)
    if 'evaluate_erdos_solution' in globals():
        return evaluate_erdos_solution(h)
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

# Project a vector onto the feasible set:
#   0 <= h <= 1  and  mean(h) == 0.5   (integral == 1 on [0,2])
def _project(h):
    h = np.clip(h, 0.0, 1.0)
    target_mean = 0.5
    # Adjust the mean by a uniform shift, then re‑clip.
    # Iterate a few times to reduce boundary effects.
    for _ in range(3):
        cur_mean = np.mean(h)
        shift = target_mean - cur_mean
        h = h + shift
        h = np.clip(h, 0.0, 1.0)
    # Final fine‑tune with a scaling around the mean if needed.
    cur_mean = np.mean(h)
    if abs(cur_mean - target_mean) > 1e-9:
        # Scale the deviation from the mean.
        dev = h - cur_mean
        scale = (target_mean) / (cur_mean) if cur_mean != 0 else 1.0
        h = target_mean + dev * scale
        h = np.clip(h, 0.0, 1.0)
        # Ensure the mean one last time
        h = h - (np.mean(h) - target_mean)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points)
    where h_values is a 1‑D numpy array of the step function samples.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise a feasible starting vector.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        # Upsample if the base size is modest (to give the optimiser more freedom)
        if base.size < 120:
            # zoom with order=0 (nearest‑neighbour) keeps the step‑function shape
            factor = max(2, int(np.ceil(180 / base.size)))
            h0 = zoom(base, factor, order=0)
        else:
            h0 = base.copy()
    else:
        # No parent supplied: start from a simple 50%–50% step.
        n0 = 120
        h0 = np.zeros(n0, dtype=float)
        h0[: n0 // 2] = 1.0
        h0 = _project(h0)

    h0 = _project(h0)
    n = h0.size
    best_h = h0
    best_score = _c5_value(best_h)

    # ------------------------------------------------------------------
    # 2. Simple stochastic hill‑climbing with decreasing step size.
    # ------------------------------------------------------------------
    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.96:
            break

        # Temperature / step‑size schedule: linear decay.
        temp = max(0.02, 0.3 * (1.0 - elapsed / budget_s))

        # Propose a new candidate by adding Gaussian noise.
        cand = best_h + rng.normal(scale=temp, size=n)
        cand = _project(cand)

        score = _c5_value(cand)

        if score < best_score:
            best_h, best_score = cand, score
            # occasional aggressive restart if we improve a lot
            if rng.random() < 0.05:
                # perturb the current best heavily to explore new basins
                perturb = rng.normal(scale=0.2, size=n)
                best_h = _project(best_h + perturb)

        # Every few hundred iterations try a completely random restart.
        if iteration % 300 == 0 and iteration > 0:
            rand = rng.random(n)
            rand = _project(rand)
            rand_score = _c5_value(rand)
            if rand_score < best_score:
                best_h, best_score = rand, rand_score

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Return the best solution found.
    # ------------------------------------------------------------------
    return best_h, best_score, n
