import time
import numpy as np

def _compute_c5(h):
    """Return C5 for discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, dx

def _project(h, target_sum):
    """Clip to [0,1] and renormalise to exact target_sum (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        h[:] = target_sum / h.shape[0]
        return h
    h *= target_sum / total
    h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 0) & (h < 1))[0]
        if free.size:
            idx = np.random.choice(free)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _local_search(h0, time_limit):
    """Simple simulated‑annealing style optimisation for C5."""
    start = time.time()
    n = h0.shape[0]
    target_sum = n / 2.0
    cur_h = _project(h0.copy(), target_sum)
    cur_val, dx = _compute_c5(cur_h)
    best_h, best_val = cur_h.copy(), cur_val
    T0 = 0.1

    while time.time() - start < time_limit:
        # propose a small transfer between two positions
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * 0.02  # ±0.01 roughly
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_h = _project(new_h, target_sum)

        new_val, _ = _compute_c5(new_h)

        # acceptance rule (simulated annealing)
        if new_val < cur_val or np.random.rand() < np.exp((cur_val - new_val) / max(T0, 1e-12)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val

        # linear cooling
        elapsed = time.time() - start
        T0 = 0.1 * (1.0 - elapsed / time_limit)

    return best_h, best_val

def _interpolate_initial(old_h, new_n):
    """Resize old_h (length m) to new_n points by linear interpolation."""
    m = old_h.shape[0]
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, old_h)
