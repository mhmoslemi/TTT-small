import time
import numpy as np

def _c5_score(h):
    # evaluate_erdos_solution may be provided by the harness
    try:
        return evaluate_erdos_solution(h)  # type: ignore
    except Exception:
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        return float(np.max(corr))

def _mass_transfer_step(h, rng, target_sum):
    """Return a new vector after moving a random amount of mass from one index to another."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h, None
    # decide direction: move from i to j or j to i
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_delta = min(h[src], 1.0 - h[dst])
    if max_delta <= 0.0:
        return h, None
    delta = rng.random() * max_delta
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    # numerical safety: enforce bounds and exact sum
    new_h = np.clip(new_h, 0.0, 1.0)
    cur_sum = new_h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # rescale uniformly then re‑clip
        scale = target_sum / cur_sum
        new_h *= scale
        new_h = np.clip(new_h, 0.0, 1.0)
        # final fix for any tiny deviation
        diff = target_sum - new_h.sum()
        if abs(diff) > 1e-12:
            # distribute the tiny remainder to a random non‑saturated entry
            idxs = np.where((new_h > 0) & (new_h < 1))[0]
            if idxs.size:
                new_h[rng.choice(idxs)] += diff
                new_h = np.clip(new_h, 0.0, 1.0)
    return new_h, delta

def _hill_climb(start_h, rng, budget_s, start_time):
    """Simple stochastic hill‑climbing with mass‑transfer moves."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    best_h = start_h.copy()
    best_score = _c5_score(best_h)
    # iteration budget loosely tied to time
    while time.time() - start_time < budget_s * 0.95:
        new_h, delta = _mass_transfer_step(best_h, rng, target_sum)
        if delta is None:
            continue
        new_score = _c5_score(new_h)
        if new_score < best_score - 1e-12:
            best_h, best_score = new_h, new_score
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)  # provided by harness
    else:
        n = 80
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)
    n = h0.shape[0]
    target_sum = n / 2.0

    # Ensure feasibility of the initial vector
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Simple scaling then clipping to satisfy constraints
        h0 = np.clip(h0 * (target_sum / cur_sum), 0.0, 1.0)
        diff = target_sum - h0.sum()
        if abs(diff) > 1e-12:
            idxs = np.where((h0 > 0) & (h0 < 1))[0]
            if idxs.size:
                h0[rng.choice(idxs)] += diff
                h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_score = _c5_score(best_h)

    # Main multi‑restart loop
    while time.time() - start_time < budget_s * 0.9:
        # random binary start (n/2 ones, n/2 zeros) shuffled
        start_vec = np.zeros(n, dtype=float)
        start_vec[: n // 2] = 1.0
        rng.shuffle(start_vec)
        # local search
        h_candidate, score_candidate = _hill_climb(start_vec, rng, budget_s, start_time)
        if score_candidate < best_score:
            best_h, best_score = h_candidate, score_candidate

    # Final refinement starting from the best found so far
    refined_h, refined_score = _hill_climb(best_h, rng, budget_s, start_time)
    if refined_score < best_score:
        best_h, best_score = refined_h, refined_score

    # Return as plain Python list for maximum compatibility
    return best_h.tolist(), float(best_score), n
