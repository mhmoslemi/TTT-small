import time
import numpy as np
from scipy.optimize import minimize

def _project_onto_bounds_sum(v, target_sum):
    """Project vector v onto [0,1] box with given sum using bisection."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def _c5_value(h, dx):
    """Compute C5 = max_k ∫ h(x)(1‑h(x+k)) dx via discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())
