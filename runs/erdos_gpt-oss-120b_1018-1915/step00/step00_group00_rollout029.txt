def _compute_c5(arr, dx):
    """Return max overlap integral for step function arr."""
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * dx

def _project_to_feasible(vec, target_sum):
    """Clip to [0,1] and adjust to hit target_sum (simple iterative projection)."""
    vec = np.clip(vec, 0.0, 1.0)
    for _ in range(10):
        cur = vec.sum()
        diff = cur - target_sum
        if abs(diff) < 1e-9:
            break

        interior = (vec > 1e-8) & (vec < 1.0 - 1e-8)

            vec -= diff / len(vec)

            vec[interior] -= diff / cnt
        vec = np.clip(vec, 0.0, 1.0)
