import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_feasible(n, target_sum, rng):
    """Generate a random vector in [0,1] with given sum."""
    # start from Dirichlet then clip and renormalize
    v = rng.random(n)
    v = v / v.sum() * target_sum
    # enforce box constraints by simple iterative projection
    for _ in range(10):
        over = v > 1.0
        if not np.any(over):
            break
        excess = v[over] - 1.0
        v[over] = 1.0
        # redistribute excess uniformly
        v[~over] -= excess.sum() / (~over).sum()
    # final clipping
    v = np.clip(v, 0.0, 1.0)
    # adjust tiny residual to meet exact sum
    diff = target_sum - v.sum()
    if abs(diff) > 1e-12:
        # add diff to a random non‑saturated entry
        free = np.where((v > 0) & (v < 1))[0]
        if free.size:
            idx = rng.choice(free)
            v[idx] = np.clip(v[idx] + diff, 0.0, 1.0)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős C5 constant.

    Returns
    -------
    h_vals : np.ndarray
        The best step values found (shape (n_points,)).
    c5_bound : float
        Corresponding max overlap integral.
    n_points : int
        Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None:
        h_curr = np.array(init, dtype=float)
        n_points = h_curr.shape[0]
    else:
        n_points = kwargs.get("n_points", 70)
        target_sum = n_points / 2.0
        h_curr = _random_feasible(n_points, target_sum, rng)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # ------------------------------------------------------------
    # initial evaluation
    # ------------------------------------------------------------
    best_h = h_curr.copy()
    best_c5 = _compute_c5(best_h, dx)

    # ------------------------------------------------------------
    # local search parameters
    # ------------------------------------------------------------
    max_iter = 10 ** 7               # safety upper bound
    start_time = time.time()
    elapsed = 0.0
    step_scale = 0.2                 # typical transferred amount as fraction of 1
    accept_prob = 0.0                # pure hill‑climbing (can be raised to anneal)

    # ------------------------------------------------------------
    # main loop
    # ------------------------------------------------------------
    for _ in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # propose a neighbour by moving mass between two indices
        i, j = rng.choice(n_points, size=2, replace=False)
        max_transfer = min(best_h[i], 1.0 - best_h[j])
        if max_transfer <= 0:
            continue
        # sample a delta in [-step, step] intersect feasible interval
        step = step_scale * max_transfer
        delta = rng.uniform(-step, step)
        # keep delta feasible
        delta = np.clip(delta, -best_h[i], 1.0 - best_h[j])

        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate
        cand_c5 = _compute_c5(cand, dx)

        if cand_c5 < best_c5 - 1e-12:
            best_h = cand
            best_c5 = cand_c5
        elif accept_prob > 0 and rng.random() < accept_prob:
            # occasional uphill move (not used here)
            best_h = cand
            best_c5 = cand_c5

        # optional simple temperature decay for future use
        # accept_prob *= 0.99999

    # ------------------------------------------------------------
    # return result
    # ------------------------------------------------------------
    return best_h, float(best_c5), int(n_points)
