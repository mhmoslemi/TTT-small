import time
import numpy as np

# --- helper functions ---------------------------------------------------------

def _project_onto_box_and_sum(y: np.ndarray, target: float) -> np.ndarray:
    """Project y onto {0<=x<=1, sum(x)=target}."""
    lo = -np.max(y)          # enough to push all to 0
    hi = 1.0 - np.min(y)     # enough to push all to 1
    for _ in range(60):      # binary search for the shift lambda
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_max_overlap(h: np.ndarray) -> np.ndarray:
    """Sub‑gradient of max_k  sum_i h[i]*(1-h[i+k]) for the worst shift."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(corr.argmax())
    shift = idx - (n - 1)                     # shift in index units
    g = np.zeros_like(h)

    if shift >= 0:
        s = shift
        # first term: (1 - h[i+s]) for i=0..n-s-1
        g[: n - s] += 1.0 - h[s:]
        # second term: -h[i] contributes to position i+s
        g[s:] -= h[: n - s]
    else:
        s = -shift
        # first term: (1 - h[i]) for i=s..n-1 (i = j+s)
        g[s:] += 1.0 - h[: n - s]
        # second term: -h[i+s] contributes to position i
        g[: n - s] -= h[s:]

    return g


def _optimize_at_resolution(h_start: np.ndarray, time_budget: float):
    """Projected gradient descent for a fixed resolution."""
    n = h_start.size
    target = n * 0.5
    h = h_start.copy()
    best_h = h.copy()
    best_score = evaluate_erdos_solution(best_h)

    start = time.time()
    lr = 0.6
    while time.time() - start < time_budget:
        g = _gradient_max_overlap(h)
        h_new = h - lr * g
        h_new = _project_onto_box_and_sum(h_new, target)
        score = evaluate_erdos_solution(h_new)

        if score < best_score:
            best_score = score
            best_h = h_new.copy()

        # accept if not worse
        if score <= best_score + 1e-12:
            h = h_new

        lr *= 0.9995                   # gentle decay
    return best_h, best_score


# --- main entry point ---------------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
      h_values : ndarray of the step function values on a uniform grid of [0,2]
      c5_bound : the achieved upper bound C₅
      n_points : number of samples used (len(h_values))
    """
    np.random.seed(seed)
    start_total = time.time()

    # ---- initial construction ------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        # fallback: half ones, half zeros, shuffled
        n0 = 45
        h0 = np.zeros(n0, dtype=float)
        h0[: n0 // 2] = 1.0
        np.random.shuffle(h0)

    h0 = _project_onto_box_and_sum(h0, h0.size * 0.5)
    best_h = h0
    best_score = evaluate_erdos_solution(best_h)

    # ---- multiresolution iterative improvement -------------------------------
    while True:
        elapsed = time.time() - start_total
        remaining = budget_s - elapsed
        if remaining < 5.0:
            break

        # allocate a portion of the remaining budget to the current resolution
        sub_budget = min(remaining * 0.55, 60.0)   # cap to avoid long single runs
        h_opt, score_opt = _optimize_at_resolution(best_h, sub_budget)

        if score_opt < best_score - 1e-9:
            best_h, best_score = h_opt, score_opt

        # attempt a finer grid if time permits and we are not already too large
        if remaining > 30.0 and best_h.size < 720:
            best_h = np.repeat(best_h, 2)          # double the resolution
            # after repetition the sum constraint is automatically satisfied
            best_h = _project_onto_box_and_sum(best_h, best_h.size * 0.5)
            best_score = evaluate_erdos_solution(best_h)
        else:
            break

    n_final = int(best_h.size)
    return best_h, best_score, n_final
