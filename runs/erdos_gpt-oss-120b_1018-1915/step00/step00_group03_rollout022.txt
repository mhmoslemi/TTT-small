import time, math
import numpy as np

def _normalize(h):
    """Clip to [0,1] and enforce sum(h) = len(h)/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        h[:] = target / n
        return h
    h = h * (target / total)

    # iterative correction for bound violations
    for _ in range(5):
        over = h > 1.0
        if over.any():
            excess = (h[over] - 1.0).sum()
            h[over] = 1.0
            under = h < 1.0
            if under.any():
                h[under] += excess / under.sum()
        under = h < 0.0
        if under.any():
            deficit = (-h[under]).sum()
            h[under] = 0.0
            above = h > 0.0
            if above.any():
                h[above] -= deficit / above.sum()
        h = np.clip(h, 0.0, 1.0)

    # tiny final adjustment
    diff = h.sum() - target
    if abs(diff) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] -= diff / free.sum()
        else:
            h -= diff / n
        h = np.clip(h, 0.0, 1.0)
    return h


def _c5_score(h):
    """Return max overlap integral for step function sampled in h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _neighbor(h, rng):
    """Produce a nearby feasible h by moving mass between two positions."""
    h2 = h.copy()
    n = h2.size
    i, j = rng.choice(n, size=2, replace=False)

    # decide direction randomly
    if rng.random() < 0.5:
        # move from i to j
        delta_max = min(h2[i], 1.0 - h2[j])
        if delta_max <= 0.0:
            return h2
        delta = rng.uniform(0.0, delta_max)
        h2[i] -= delta
        h2[j] += delta
    else:
        # move from j to i
        delta_max = min(h2[j], 1.0 - h2[i])
        if delta_max <= 0.0:
            return h2
        delta = rng.uniform(0.0, delta_max)
        h2[i] += delta
        h2[j] -= delta
    return h2


def _upscale(h):
    """Double the resolution by linear interpolation and renormalise."""
    n0 = h.size
    n1 = n0 * 2
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n1, endpoint=False)
    h1 = np.interp(x1, x0, h)
    return _normalize(h1)
