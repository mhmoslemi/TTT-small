def project_feasible(h, target_sum):
    n = h.shape[0]
    x = cp.Variable(n)
    obj = cp.Minimize(cp.sum_squares(x - h))
    constraints = [x >= 0, x <= 1, cp.sum(x) == target_sum]
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    return np.clip(x.value, 0, 1)  # fallback

import time
import numpy as np

def _project_feasible(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clamp to [0,1] and adjust to meet the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        nfree = free.sum()
        if nfree == 0:
            # push a tiny amount to all entries then reclamp
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / nfree
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final sanity scaling if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-6 and cur > 0:
        h *= target_sum / cur
        h = np.clip(h, 0.0, 1.0)
    return h
