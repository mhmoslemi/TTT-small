import time
import numpy as np

# helpers are pre‑imported in the harness: evaluate_erdos_solution, initial_h_values

def _project_box_sum(y, target, lo=0.0, hi=1.0, tol=1e-9, max_iter=60):
    """Euclidean projection of y onto {x: lo<=x<=hi, sum(x)=target}."""
    # binary search for λ such that sum(clamp(y+λ)) = target
    low, high = -1e6, 1e6
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        x = np.clip(y + lam, lo, hi)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s < target:
            low = lam
        else:
            high = lam
    return np.clip(y + (low + high) * 0.5, lo, hi)

def _c5_score(h, dx):
    """Maximum overlap integral for a discretised step function."""
    # correlate h with (1-h); mode='full' yields length 2*n-1
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimises C₅."""
    rng = np.random.default_rng(seed)

    n_points = 45                     # modest size, fast evaluation
    dx = 2.0 / n_points
    target_sum = n_points / 2.0       # because ∑h * dx = 1

    # ----- initial solution -------------------------------------------------
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
        if h.size != n_points:
            # adjust size by simple interpolation if needed
            idx = np.linspace(0, h.size - 1, n_points).astype(int)
            h = h[idx]
    else:
        h = rng.random(n_points)

    # enforce feasibility
    h = np.clip(h, 0.0, 1.0)
    h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_score = _c5_score(best_h, dx)

    start = time.time()
    it = 0
    # annealing schedule
    T0 = 0.05
    T_end = 1e-4

    while True:
        now = time.time()
        if now - start > budget_s:
            break
        it += 1
        # temperature decays linearly in iteration count
        progress = it / 50000.0
        T = T0 * (1 - progress) + T_end * progress

        # propose a perturbation
        proposal = best_h + rng.normal(scale=0.02, size=n_points)
        proposal = np.clip(proposal, 0.0, 1.0)
        proposal = _project_box_sum(proposal, target_sum)

        score = _c5_score(proposal, dx)

        # accept if better or with Metropolis probability
        if score < best_score or rng.random() < np.exp(-(score - best_score) / max(T, 1e-12)):
            best_h, best_score = proposal, score

        # occasional random restart to avoid stagnation
        if it % 8000 == 0:
            h_restart = rng.random(n_points)
            h_restart = _project_box_sum(h_restart, target_sum)
            r_score = _c5_score(h_restart, dx)
            if r_score < best_score:
                best_h, best_score = h_restart, r_score

    return best_h, best_score, n_points
