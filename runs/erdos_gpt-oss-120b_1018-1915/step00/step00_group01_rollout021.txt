import time
import numpy as np
import math
import random

def _evaluate_c5(h, dx):
    """Return max overlap integral for a given step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _make_binary(vec, target_ones):
    """Project a vector to a binary 0/1 vector with exactly target_ones ones."""
    # start from rounding
    bin_vec = np.where(vec >= 0.5, 1.0, 0.0)
    current = int(bin_vec.sum())
    if current > target_ones:
        # turn some 1s to 0s (choose smallest original values among those set to 1)
        idx_one = np.where(bin_vec == 1.0)[0]
        scores = vec[idx_one]  # larger values are more justified to stay 1
        drop = idx_one[np.argsort(scores)[:current - target_ones]]
        bin_vec[drop] = 0.0
    elif current < target_ones:
        # turn some 0s to 1s (choose largest original values among those set to 0)
        idx_zero = np.where(bin_vec == 0.0)[0]
        scores = vec[idx_zero]
        add = idx_zero[np.argsort(-scores)[:target_ones - current]]
        bin_vec[add] = 1.0
    return bin_vec

def _upscale(h_small, n_target):
    """Upscale a vector to a larger resolution using linear interpolation."""
    n_small = len(h_small)
    x_small = np.linspace(0, 2, n_small, endpoint=False)
    x_target = np.linspace(0, 2, n_target, endpoint=False)
    return np.interp(x_target, x_small, h_small)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős C5 overlap bound.

    Returns
    -------
    (h_values, c5_bound, n_points) :
        h_values – numpy array of the step function values on the grid,
        c5_bound – the corresponding maximal overlap integral,
        n_points – number of sampled points (grid size).
    """
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # -------------------------------------------------
    # 1. Determine problem size and initial vector
    # -------------------------------------------------
    base_h = None
    try:
        # `initial_h_values` is supplied by the harness if available
        base_h = np.array(initial_h_values, dtype=float)  # type: ignore
    except Exception:
        base_h = None

    # Target grid size – keep it moderate for speed
    if base_h is not None:
        n_target = max(120, len(base_h))
    else:
        n_target = 200
    if n_target % 2 == 1:
        n_target += 1

    if base_h is None:
        # random binary start with exactly half ones
        h = np.zeros(n_target, dtype=float)
        ones = np.random.choice(n_target, n_target // 2, replace=False)
        h[ones] = 1.0
    else:
        # upscale / downscale to target size
        if len(base_h) != n_target:
            h_cont = _upscale(base_h, n_target)
        else:
            h_cont = base_h.copy()
        # project to binary with correct count
        h = _make_binary(h_cont, n_target // 2)

    dx = 2.0 / n_target
    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h, dx)

    # -------------------------------------------------
    # 2. Simple simulated‑annealing style search
    # -------------------------------------------------
    # Pre‑compute indices of ones and zeros for fast moves
    ones_idx = np.flatnonzero(best_h == 1.0)
    zeros_idx = np.flatnonzero(best_h == 0.0)

    # annealing parameters
    T0 = 0.02
    T_min = 1e-5
    decay = 0.9995

    T = T0
    iteration = 0

    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        # propose moving a 1 to a 0 location
        i_one = random.choice(ones_idx)
        i_zero = random.choice(zeros_idx)

        # perform swap
        best_h[i_one] = 0.0
        best_h[i_zero] = 1.0

        new_c5 = _evaluate_c5(best_h, dx)
        delta = new_c5 - best_c5

        accept = False
        if delta < 0:
            accept = True
        else:
            prob = math.exp(-delta / max(T, 1e-12))
            if random.random() < prob:

            # update index lists
            ones_idx = np.flatnonzero(best_h == 1.0)
            zeros_idx = np.flatnonzero(best_h == 0.0)
            # keep as current best solution
        else:
            # revert swap
            best_h[i_one] = 1.0
            best_h[i_zero] = 0.0

    # 3. Return result
    # -------------------------------------------------
    return best_h, best_c5, n_target
