import numpy as np, time, math, random

def _evaluate(h):
    """Return C5 for a given step function vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _pairwise_move(h, step):
    """Return a new candidate by moving a small amount from one index to another."""
    n = h.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    if h[i] <= 0.0 or h[j] >= 1.0:
        return None
    delta = min(step, h[i], 1.0 - h[j])
    if delta <= 0.0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


def _initialize_h(n):
    """Create a feasible initial vector of length n."""
    # uniform 0.5 satisfies the integral constraint for any n
    return np.full(n, 0.5, dtype=np.float64)


def _upscale(h, new_n):
    """Linearly interpolate h to a larger size new_n (must be multiple of len(h))."""
    old_n = h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # enforce bounds and exact sum
    h_new = np.clip(h_new, 0.0, 1.0)
    target = new_n / 2.0
    s = h_new.sum()
    if s != 0:
        h_new *= target / s
    # final clipping + small correction to meet the sum exactly
    h_new = np.clip(h_new, 0.0, 1.0)
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny residual uniformly over non‑saturated entries
        free = (h_new > 0.0) & (h_new < 1.0)
        if free.any():
            h_new[free] += diff / free.sum()
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new
