import numpy as np
import time
import random

def _compute_c5(h, dx):
    """Return C5 for step function h given spacing dx."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _repair_sum(h, target_sum):
    """Scale h to satisfy exact sum while keeping within [0,1]."""
    current = h.sum()
    if np.isclose(current, target_sum):
        return h
    diff = target_sum - current
    # distribute diff proportionally to free room
    free_up = (1.0 - h).sum()
    free_down = h.sum()
    if diff > 0 and free_up > 0:
        h += diff * (1.0 - h) / free_up
    elif diff < 0 and free_down > 0:
        h += diff * h / free_down
    # clip any tiny violations
    np.clip(h, 0.0, 1.0, out=h)
    # final tiny adjustment to hit target exactly
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # spread residual uniformly on indices not at bounds
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += residual / mask.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _sa_optimize(h0, dx, budget_s, start_time):
    """Simple simulated‑annealing optimizer for the C5 objective."""
    n = h0.size
    target_sum = n / 2.0  # because integral = 1 and dx = 2/n
    h = h0.copy()
    best_h = h.copy()
    best_val = _compute_c5(h, dx)

    # annealing schedule parameters
    T0 = 0.05
    Tf = 1e-4
    max_iter = 200000

    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s:
            break
        # linear cooling
        T = T0 * (1 - it / max_iter) + Tf * (it / max_iter)

        # propose a move: transfer mass from j to i
        i, j = random.sample(range(n), 2)
        # max transferable amount respecting bounds
        max_inc = min(0.2, 1.0 - h[i])          # limit step size for stability
        max_dec = min(0.2, h[j])
        if max_inc <= 0 or max_dec <= 0:
            continue
        delta = random.random() * min(max_inc, max_dec)

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # evaluate
        new_val = _compute_c5(h_new, dx)

        # accept if better or probabilistically
        if new_val < best_val or random.random() < np.exp((best_val - new_val) / (T + 1e-12)):
            h = h_new
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()

    # final repair to ensure exact sum (tiny numeric drift)
    best_h = _repair_sum(best_h, target_sum)
    best_val = _compute_c5(best_h, dx)
    return best_h, best_val

def _interpolate(h, new_n):
    """Linear interpolation of step vector h to a finer grid."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős C5 constant.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h on the chosen grid.
    c5_bound : float
        Corresponding upper bound C5.
    n_points : int
        Number of discretisation points used.
    """
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # Determine initial grid size and starting vector
    init = globals().get("initial_h_values", None)
    if init is not None:
        h0 = np.asarray(init, dtype=float)
        n0 = h0.size
    else:
        n0 = 84                     # modest size when no parent exists
        h0 = np.full(n0, 0.5)       # uniform start satisfies integral automatically

    # Ensure feasibility of the start
    dx0 = 2.0 / n0
    target_sum0 = n0 / 2.0
    h0 = _repair_sum(h0, target_sum0)

    # Phase 1: optimise at base resolution
    remaining = budget_s - (time.time() - start_time)
    h_opt, c5_opt = _sa_optimize(h0, dx0, remaining * 0.5, start_time)

    # Phase 2: double resolution and refine (if time permits)
    if time.time() - start_time < budget_s * 0.9:
