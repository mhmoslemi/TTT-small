import time
import numpy as np
import random

def _c5_value(h, dx):
    """Compute C5 = max correlation * dx for a given h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _mutate(h):
    """Return a new h by moving a small amount of mass from one index to another."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # amount to move from j to i while staying in [0,1]
    max_up = 1.0 - h[i]          # how much we can add to i
    max_down = h[j]              # how much we can subtract from j
    max_delta = min(max_up, max_down)

    max_down_i = h[i]            # how much we can subtract from i
    max_up_j = 1.0 - h[j]        # how much we can add to j
    min_delta = -min(max_down_i, max_up_j)   # negative direction (move from i to j)

    if max_delta <= min_delta:
        # no feasible move; return a copy unchanged
        return h.copy()

    delta = np.random.rand() * (max_delta - min_delta) + min_delta
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # clip for numerical safety
    new_h[i] = min(max(new_h[i], 0.0), 1.0)
    new_h[j] = min(max(new_h[j], 0.0), 1.0)
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] giving a low C5 bound.

    Returns:
        (h_values, c5_bound, n_points)
    """
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # Determine discretization size
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n_points = h0.size
    else:
        n_points = 45
        # start with half ones, half zeros (satisfies sum = n/2)
        h0 = np.zeros(n_points, dtype=float)
        h0[: n_points // 2] = 1.0
        np.random.shuffle(h0)

    dx = 2.0 / n_points

    # Ensure feasibility of initial vector (may be slightly off due to rounding)
    target_sum = n_points / 2.0
    current_sum = h0.sum()
    if not np.isclose(current_sum, target_sum, atol=1e-9):
        # Simple rescaling while keeping bounds; clip afterwards
        h0 = np.clip(h0 * (target_sum / current_sum), 0.0, 1.0)
        # Re‑adjust any residual to meet exact sum by adding to first element
        residual = target_sum - h0.sum()
        h0[0] = min(max(h0[0] + residual, 0.0), 1.0)

    best_h = h0.copy()
    best_c5 = _c5_value(best_h, dx)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # Simulated annealing schedule
    T = 1.0
    cooling = 0.995
    min_T = 1e-5

    iteration = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        iteration += 1
        # generate neighbor
        new_h = _mutate(cur_h)
        new_c5 = _c5_value(new_h, dx)

        # acceptance test
        if new_c5 < cur_c5:
            accept = True
        else:
            # probabilistic acceptance
            prob = np.exp(-(new_c5 - cur_c5) / max(T, 1e-12))
            accept = np.random.rand() < prob

        if accept:

            if new_c5 < best_c5:
