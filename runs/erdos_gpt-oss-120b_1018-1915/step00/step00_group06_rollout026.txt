def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def hill_climb(n, init_h, time_limit, start_time):
    best_h = init_h.copy()
    best_score = compute_c5(best_h)
    current_h = best_h.copy()
    current_score = best_score
    # precompute indices sets
    indices_one = set(np.where(current_h > 0.5)[0])
    indices_zero = set(np.where(current_h <= 0.5)[0])
    T0 = 1.0
    iter = 0
    while time.time() - start_time < time_limit:
        iter += 1
        # temperature schedule
        T = T0 * (1 - iter/1e6)  # degrade
        # pick random swap
        i = random.choice(tuple(indices_one))
        j = random.choice(tuple(indices_zero))
        # swap
        current_h[i], current_h[j] = 0.0, 1.0
        new_score = compute_c5(current_h)
        delta = new_score - current_score
        accept = False
        if new_score < best_score:
            best_score = new_score
            best_h = current_h.copy()
            accept = True
        if delta < 0 or random.random() < math.exp(-delta / max(T,1e-9)):
            accept = True
        if not accept:
            # revert swap
            current_h[i], current_h[j] = 1.0, 0.0
        else:
            # update sets
            indices_one.remove(i); indices_one.add(j)
            indices_zero.remove(j); indices_zero.add(i)
            current_score = new_score
    return best_h, best_score

import time
import random
import math
import numpy as np

def _c5_score(h: np.ndarray) -> float:
    """Return the C5 value for a given step function sampled on a uniform grid."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _binary_from_continuous(v: np.ndarray) -> np.ndarray:
    """Convert a real vector to a binary one with exactly half the entries equal 1."""
    n = v.size
    half = n // 2
    idx = np.argpartition(-v, half)[:half]
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h


def _hill_climb(n: int, h0: np.ndarray, time_limit: float, start_time: float):
    """Simple simulated‑annealing hill climber on binary vectors."""
    best_h = h0.copy()
    best_score = _c5_score(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    # keep sets of positions of 1s and 0s for fast random swaps
    ones = set(np.where(cur_h > 0.5)[0])
    zeros = set(np.where(cur_h <= 0.5)[0])

    it = 0
    T0 = 1.0  # initial temperature

    while time.time() - start_time < time_limit:
        it += 1
        # linear cooling
        T = T0 * max(0.0, 1.0 - it / 200000.0)

        i = random.choice(tuple(ones))
        j = random.choice(tuple(zeros))

        # propose swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_score = _c5_score(cur_h)
        delta = new_score - cur_score

        accept = False
        if new_score < best_score:
            best_score = new_score
            best_h = cur_h.copy()
            accept = True
        if delta < 0.0 or random.random() < math.exp(-delta / max(T, 1e-12)):
            accept = True

        if not accept:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        else:
            # keep swap, update sets and current score
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)
            cur_score = new_score

    return best_h, best_score


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a low upper bound for C5.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,) with entries in [0,1]
        c5_bound : float, the attained max overlap
        n_points : int, the number of samples used
    """
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # candidate lengths (must be even to allow exact half‑ones)
    lengths = [100, 120, 150, 180, 200]

    best_h = None
    best_score = float("inf")
    best_n = None

    # helper to obtain an initial vector for a given size
    def _initial_vector(n):
        if "initial_h_values" in globals():
            v = globals()["initial_h_values"]
            if v.size != n:
                # linear interpolation to required size
                x_old = np.linspace(0.0, 2.0, v.size, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                v = np.interp(x_new, x_old, v)
            return v.astype(np.float64)
        else:
            return np.random.rand(n)

    # allocate roughly equal time slices to each length, respecting total budget
    for n in lengths:
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        # time allowed for this length (10 % of remaining)
        remaining = budget_s - elapsed
        time_limit = remaining * 0.15

        # build starting binary vector
        init_vec = _initial_vector(n)
        h0 = _binary_from_continuous(init_vec)

        # run hill‑climb optimisation
        h_opt, score_opt = _hill_climb(n, h0, time_limit, start)

        if score_opt < best_score:
            best_score = score_opt
            best_h = h_opt
            best_n = n

    # final sanity checks
    if best_h is None:
        # fallback: simple alternating 0/1 pattern
        best_n = 100
        best_h = np.zeros(best_n, dtype=np.float64)
        best_h[::2] = 1.0
        best_score = _c5_score(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    # ensure exact integral (sum = n/2) – binary already satisfies it
    final_score = _c5_score(best_h)

    return best_h, final_score, best_n
