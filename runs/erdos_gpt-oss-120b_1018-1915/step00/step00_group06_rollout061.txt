import time, random, math
import numpy as np

def evaluate(h):
    """Return the discretized C5 value for a step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / h.size) * np.max(corr)

def enforce_bounds_and_sum(h, target):
    """Clip to [0,1] and adjust to meet the exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0.0:
        h[:] = target / h.size
        return h
    # iterative scaling + clipping
    for _ in range(5):
        factor = target / total
        h = np.clip(h * factor, 0.0, 1.0)
        total = h.sum()
    diff = target - total
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    """Linear interpolation of h (defined on [0,2)) to length n_new."""
    n_old = h.size
    if n_old == n_new:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = enforce_bounds_and_sum(h_new, n_new / 2.0)
    return h_new

def hill_climb(initial_h, budget, start_time):
    """Simple hill‑climbing with pairwise reallocation."""
    n = initial_h.size
    h = initial_h.copy()
    best_h = h.copy()
    best_score = evaluate(h)
    step = 0.2
    it = 0
    while time.time() - start_time < budget:
        i, j = random.sample(range(n), 2)
        delta = (random.random() * 2.0 - 1.0) * step
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        h_new = enforce_bounds_and_sum(h_new, n / 2.0)
        score = evaluate(h_new)
        if score < best_score:
            h = h_new
            best_h = h_new
            best_score = score
        it += 1
        if it % 200 == 0:
            step *= 0.95
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the given time budget."""
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    remaining = lambda: max(0.0, budget_s - (time.time() - start))

    # Determine the base construction
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=float)
    else:
        # fallback: 50% ones, 50% zeros shuffled
        n0 = 95
        base = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)])
        np.random.shuffle(base)

    # Candidate resolutions (keep modest to stay within budget)
    candidates = [base.size, base.size * 2, base.size * 3, base.size * 4]
    best_overall_h = None
    best_overall_score = float("inf")
    best_n = None

    # Allocate time roughly equally among resolutions
    per_res_budget = remaining() / len(candidates) if candidates else remaining()
    for n in candidates:
        if remaining() <= 0.5:
            break
        h0 = upsample(base, n)
        # Run hill climbing on this resolution
        h_opt, score = hill_climb(h0, per_res_budget, time.time())
        if score < best_overall_score:
            best_overall_score = score
            best_overall_h = h_opt.copy()
            best_n = n

    # If nothing succeeded (unlikely), fall back to the base construction
    if best_overall_h is None:
        best_overall_h = enforce_bounds_and_sum(base, base.size / 2.0)
        best_overall_score = evaluate(best_overall_h)
        best_n = best_overall_h.size

    return (best_overall_h, float(best_overall_score), int(best_n))
