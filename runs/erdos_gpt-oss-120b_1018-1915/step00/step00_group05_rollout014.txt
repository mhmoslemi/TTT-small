import time

def _repair(candidate, desired_sum, eps=1e-9):
    """Project candidate onto [0,1] box and sum constraint."""
    cand = np.clip(candidate, 0.0, 1.0)
    cur = cand.sum()
    diff = desired_sum - cur
    # simple iterative redistribution
    while abs(diff) > eps:
        if diff > 0:
            # need to add mass: choose indices with room <1
            room = 1.0 - cand
            idx = np.where(room > eps)[0]
            if idx.size == 0:
                break
            add = min(diff / idx.size, room[idx].max())
            inc = np.minimum(add, room[idx])
            cand[idx] += inc
        else:
            # need to remove mass: choose indices with >0
            room = cand
            idx = np.where(room > eps)[0]
            if idx.size == 0:
                break
            sub = min(-diff / idx.size, room[idx].max())
            dec = np.minimum(sub, room[idx])
            cand[idx] -= dec
        cur = cand.sum()
        diff = desired_sum - cur
    return cand

def _crossover(p1, p2):
    mask = np.random.rand(p1.size) < 0.5
    child = np.where(mask, p1, p2)
    return child

def _mutate(ind, mutation_rate=0.1, amplitude=0.1):
    child = ind.copy()
    mut = np.random.rand(child.size) < mutation_rate
    noise = np.random.uniform(-amplitude, amplitude, child.size)
    child[mut] += noise[mut]
    return child
