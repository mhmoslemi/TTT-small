import time
import numpy as np

def evaluate_c5(h):
    """Return C5 for step function h on [0,2]."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def upsample_h(old_h, new_n):
    """Linearly interpolate old_h (length m) to length new_n."""
    m = old_h.size
    old_x = np.linspace(0.0, 2.0, m, endpoint=False) + 1.0 / m  # centroids
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    return np.interp(new_x, old_x, old_h)

def mutate_preserve_sum(h, scale):
    """Pairwise mass transfer mutation preserving exact sum."""
    n = h.size
    new = h.copy()
    # number of elementary moves roughly proportional to scale
    moves = max(1, int(scale * n))
    for _ in range(moves):
        # choose donor (has room to give) and receiver (has room to receive)
        donors = np.where(new > 1e-12)[0]
        receivers = np.where(new < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = np.random.choice(donors)
        j = np.random.choice(receivers)
        max_delta = min(new[i], 1.0 - new[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * scale
        new[i] -= delta
        new[j] += delta
    return new
