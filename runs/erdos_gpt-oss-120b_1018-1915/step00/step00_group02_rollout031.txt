def _project_feasible(h):
    """Clip to [0,1] and enforce sum_i h_i = n/2 exactly."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    # Simple iterative redistribution to meet the target sum
    for _ in range(10):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        # indices that are not at a bound can receive/take mass
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size == 0:
            # all at bounds – distribute uniformly (may violate bounds a tiny bit)
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free.size
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # final scaling if still off by tiny amount
    cur = h.sum()
    if abs(cur - target) > 1e-12:
        h *= target / cur
        h = np.clip(h, 0.0, 1.0)
    return h

def _hill_climb(initial_h, time_limit, start_time):
    """Simple pairwise exchange hill‑climb respecting feasibility."""
    best_h = initial_h.copy()
    best_val = evaluate_erdos_solution(best_h)
    n = best_h.shape[0]

    # pre‑compute max iterations based on remaining time
    while time.time() - start_time < time_limit:
        i, j = random.sample(range(n), 2)
        delta = random.uniform(-0.05, 0.05)  # small move
        # propose exchange respecting bounds
        hi, hj = best_h[i], best_h[j]
        if not (0.0 <= hi + delta <= 1.0 and 0.0 <= hj - delta <= 1.0):
            continue
        new_h = best_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta
        # feasibility is preserved by construction, but re‑project in case of rounding
        new_h = _project_feasible(new_h)
        val = evaluate_erdos_solution(new_h)
        if val < best_val - 1e-9:
            best_h, best_val = new_h, val
    return best_h, best_val
