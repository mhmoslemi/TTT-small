import time
import numpy as np
from scipy.optimize import minimize

def _evaluate_max_overlap(h):
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def _project_to_sum(h, target_sum):
    """Clip to [0,1] then adjust linearly to meet target sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    # simple proportional scaling, then re‑clip if necessary
    scale = target_sum / cur if cur > 0 else 0.0
    h = h * scale
    h = np.clip(h, 0.0, 1.0)
    # final small correction distributed equally
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # last resort: renormalise if still off (rare)
    cur = h.sum()
    if abs(cur - target_sum) > 1e-9:
        h *= target_sum / cur
        h = np.clip(h, 0.0, 1.0)
    return h

def _make_initial(n_target, h_base):
    """Upsample/ downsample a base vector to length n_target."""
    n_base = h_base.size
    if n_target == n_base:
        return h_base.copy()
    x_base = np.linspace(0.0, 2.0, n_base, endpoint=False)
    x_target = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h_interp = np.interp(x_target, x_base, h_base)
    # enforce integral constraint
    target_sum = n_target / 2.0
    h_interp = _project_to_sum(h_interp, target_sum)
    return h_interp

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Base vector of length 47 – use provided parent if present
    if "initial_h_values" in globals():
        h47 = np.array(initial_h_values, dtype=float)
    else:
        h47 = np.full(47, 0.5)  # already satisfies the sum constraint

    ns = [47, 94, 188, 376]            # resolutions to try (doubling each step)
    best_val = float("inf")
    best_h = None
    best_n = None

    for n in ns:
        if time.time() - start_time > budget_s * 0.95:
            break

        # Prepare initial guess
        h0 = _make_initial(n, h47)

        # Equality constraint: sum(h) = n/2
        target_sum = n / 2.0
        constraints = ({
            "type": "eq",
            "fun": lambda h, ts=target_sum: float(h.sum() - ts)
        },)

        bounds = [(0.0, 1.0)] * n

        # Objective wrapper for optimizer
        def obj(h):
            return _evaluate_max_overlap(h)

        # Run SLSQP optimization
        res = minimize(
            obj,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 300, "ftol": 1e-9, "disp": False},
        )

        # If optimizer failed, fall back to the initial point
        h_opt = res.x if res.success else h0
        h_opt = _project_to_sum(h_opt, target_sum)  # safety projection

        val = _evaluate_max_overlap(h_opt)

        if val < best_val:
            best_val = val
            best_h = h_opt
            best_n = n

        # Small random perturbation stage to possibly escape local minima
        # (only if time permits)
        for _ in range(3):
            if time.time() - start_time > budget_s * 0.95:
                break
            perturb = rng.normal(scale=0.02, size=n)
            h_candidate = _project_to_sum(h_opt + perturb, target_sum)
            cand_val = _evaluate_max_overlap(h_candidate)
            if cand_val < best_val:
                best_val, best_h, best_n = cand_val, h_candidate, n
                h_opt = h_candidate  # continue from improved point

    # Ensure a valid return even if nothing succeeded
    if best_h is None:
        best_h = _make_initial(47, h47)
        best_val = _evaluate_max_overlap(best_h)
        best_n = 47

    return best_h, best_val, int(best_n)
