import numpy as np
import time
import math

# The harness provides `initial_h_values` if available and a helper
# `evaluate_erdos_solution` (not required here but kept for compatibility).

def compute_c5(h):
    """Return the C5 value for a discretized step function h."""
    n = h.size
    # correlation of h with (1-h); mode='full' gives length 2n-1
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)

def project_bounds_sum(h, desired_sum):
    """Clip to [0,1] and adjust to hit the desired sum (simple scaling)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # degenerate case – spread evenly
        h.fill(desired_sum / h.size)
        return np.clip(h, 0.0, 1.0)
    factor = desired_sum / cur
    h *= factor
    # second clip may disturb the sum a little; fix by a tiny redistribution
    h = np.clip(h, 0.0, 1.0)
    deficit = desired_sum - h.sum()
    if abs(deficit) > 1e-12:
        # distribute the tiny deficit uniformly over non‑boundary entries
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += deficit / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def propose_move(h, step):
    """Generate a new candidate by moving mass between two coordinates."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # raw delta, centered at 0
    delta = (np.random.rand() * 2.0 - 1.0) * step
    # respect bounds
    max_inc = 1.0 - h[i]
    max_dec = h[j]
    delta = np.clip(delta, -max_dec, max_inc)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def simulated_annealing(h0, time_limit, start_temp=0.02, end_temp=0.0005, step=0.2):
    """Simple SA that respects bounds and the integral constraint."""
    start = time.time()
    n = h0.size
    desired_sum = n * 0.5
    best_h = project_bounds_sum(h0, desired_sum)
    best_c = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c = best_c

    # exponential temperature schedule
    total_iters = 0
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # linear progress in [0,1] for temperature schedule
        progress = elapsed / time_limit
        temp = start_temp * (1 - progress) + end_temp * progress

        cand = propose_move(cur_h, step)
        cand = project_bounds_sum(cand, desired_sum)
        cand_c = compute_c5(cand)

        if cand_c < cur_c or np.exp(-(cand_c - cur_c) / max(temp, 1e-12)) > np.random.rand():
            cur_h, cur_c = cand, cand_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c

        total_iters += 1
        # gradually shrink step size to fine‑tune
        if total_iters % 2000 == 0:
            step = max(step * 0.95, 1e-4)

    return best_h, best_c

def upsample(h, new_n):
    """Linear interpolation from current grid to a finer grid."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # enforce exact integral after interpolation
    desired_sum = new_n * 0.5
    h_new = project_bounds_sum(h_new, desired_sum)
    return h_new
