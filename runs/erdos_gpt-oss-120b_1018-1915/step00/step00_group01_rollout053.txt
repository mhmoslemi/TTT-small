import time
import numpy as np
import random

def _compute_c5(h, dx):
    """
    Compute the C5 value for a discrete step function h.
    """
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_pairwise(h, i, j, eps):
    """
    Transfer eps from h[i] to h[j] respecting bounds.
    Returns the new array (or None if transfer impossible).
    """
    if eps <= 0:
        return None
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    if new_h[i] < 0.0 or new_h[j] > 1.0:
        return None
    return new_h
