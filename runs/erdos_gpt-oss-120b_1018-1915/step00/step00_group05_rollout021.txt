import numpy as np
import time
import math

def compute_c5(h, dx):
    """Return the overlap bound C5 for a sampled step function h."""
    # correlation of h with (1-h), full mode gives length 2*n-1
    corr = np.correlate(h, 1.0 - h, mode="full")
    # maximum absolute shift (the integral) multiplied by dx
    return np.max(corr) * dx

def project_transfer(h, i, j, delta):
    """Transfer delta from h[j] to h[i] respecting [0,1] bounds."""
    # assume delta > 0, h[i] + delta <= 1, h[j] - delta >= 0
    h[i] += delta
    h[j] -= delta

def random_feasible_initial(n, rng):
    """Create a random feasible vector of length n with sum = n/2 and values in [0,1]."""
    # start from uniform Dirichlet then clip and renormalize by transfers
    x = rng.random(n)
    x = x / x.sum() * (n / 2.0)  # enforce sum, may exceed 1
    # fix any entries >1 by moving excess to random undershooters
    excess = x - 1.0
    while np.any(excess > 1e-12):
        overs = np.where(excess > 1e-12)[0]
        unders = np.where(x < 1.0 - 1e-12)[0]
        if len(unders) == 0:
            break
        for o in overs:
            take = min(excess[o], 1.0 - x[unders].max())
            # distribute equally to unders
            share = take / len(unders)
            x[unders] += share
            x[o] -= take
        excess = x - 1.0
    # final clip and a small redistribution to hit exact sum
    x = np.clip(x, 0.0, 1.0)
    cur_sum = x.sum()
    target = n / 2.0
    diff = target - cur_sum
    # adjust by small transfers between random pairs
    while abs(diff) > 1e-10:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if diff > 0 and x[i] < 1.0:
            delta = min(diff, 1.0 - x[i])
            x[i] += delta
            diff -= delta
        elif diff < 0 and x[i] > 0.0:
            delta = min(-diff, x[i])
            x[i] -= delta
            diff += delta
    return x
