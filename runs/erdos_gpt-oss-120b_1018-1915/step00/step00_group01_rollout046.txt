import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h, dx):
    """Return max overlap integral for step values h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _project_to_bounds_and_sum(x, target_sum):
    """
    Project vector x onto the box [0,1] while forcing sum(x)=target_sum.
    Uses a simple iterative clipping algorithm that converges for our sizes.
    """
    n = x.shape[0]
    # start by clipping to [0,1]
    x = np.clip(x, 0.0, 1.0)
    # adjust to meet the sum using a water‑filling style loop
    for _ in range(10):
        cur_sum = x.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        # distribute diff uniformly to non‑saturated entries
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:  # all at bounds, distribute to all then clip again
            cnt = n
            free = np.ones(n, dtype=bool)
        delta = diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # final safety clip
    return np.clip(x, 0.0, 1.0)

def _initial_h(n_points):

    Build an initial feasible vector.
    If the global `initial_h_values` exists, up‑sample it;
    otherwise use a simple two‑block pattern (half ones, half zeros)
