import numpy as np, time, random, math
from scipy.fft import fft, ifft

def _evaluate_max_overlap(h, dx):
    n = h.size
    m = 2 * n - 1
    # FFT based correlation (full)
    H = fft(h, m)
    G = fft(1.0 - h, m)
    corr = np.real(ifft(H * np.conj(G)))[:m] * dx
    return corr.max()

def _binary_from_continuous(vals, n_one):
    """Return a binary (0/1) vector of same length as vals with exactly n_one ones,
    keeping the positions of the largest entries."""
    idx = np.argpartition(-vals, n_one - 1)[:n_one]
    out = np.zeros_like(vals, dtype=np.float64)
    out[idx] = 1.0
    return out

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # ----- choose discretisation size -----
    # target size – moderate but enough to give a good bound
    n_points = 240               # must be even, ≤ 1000
    dx = 2.0 / n_points
    half = n_points // 2

    # ----- initialise h -----
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        # upscale / interpolate to target resolution
        x_old = np.linspace(0, 2, init.size, endpoint=False) + 1.0 / init.size
        x_new = np.linspace(0, 2, n_points, endpoint=False) + dx / 2
        h0 = np.interp(x_new, x_old, init, left=init[0], right=init[-1])
    except NameError:
        # fallback: simple alternating pattern
        h0 = np.zeros(n_points, dtype=np.float64)
        h0[:half] = 1.0
        np.random.shuffle(h0)

    # force binary while preserving sum
    current = _binary_from_continuous(h0, half)

    best_h = current.copy()
    best_score = _evaluate_max_overlap(best_h, dx)

    start_time = time.time()
    # annealing parameters
    temp = 0.02
    min_temp = 1e-4
    decay = 0.9995

    # pre‑compute index lists for fast random swaps
    ones_idx = lambda arr: np.where(arr == 1)[0]
    zeros_idx = lambda arr: np.where(arr == 0)[0]

    while time.time() - start_time < budget_s * 0.95:
        # pick a random 1↔0 swap
        o = ones_idx(current)
        z = zeros_idx(current)
        if o.size == 0 or z.size == 0:
            break
        i = int(random.choice(o))
        j = int(random.choice(z))

        # propose move
        current[i] = 0.0
        current[j] = 1.0
        cand_score = _evaluate_max_overlap(current, dx)

        # Metropolis acceptance
        accept = False
        if cand_score < best_score:
            accept = True
            best_score = cand_score
            best_h = current.copy()
        else:
            prob = math.exp((best_score - cand_score) / max(temp, 1e-12))
            if random.random() < prob:
                accept = True

        if not accept:
            # revert
            current[i] = 1.0
            current[j] = 0.0

        # temperature schedule
        temp = max(min_temp, temp * decay)

    # final evaluation (ensure exact sum)
    # best_h already binary with exact half ones
    c5_bound = _evaluate_max_overlap(best_h, dx)

    return best_h, float(c5_bound), n_points
