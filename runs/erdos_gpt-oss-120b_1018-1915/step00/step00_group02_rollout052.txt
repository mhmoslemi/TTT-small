import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] (represented by n points) to minimize
    the Erdős overlap constant C5 = max_k ∫ h(x)(1‑h(x+k)) dx.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h on the uniform grid.
    c5_bound : float
        The achieved upper bound (the objective value).
    n_points : int
        Number of discretisation points used (length of h_values).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise a feasible h.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # default size – modest enough to stay fast
        h = np.full(180, 0.5, dtype=np.float64)   # sum = 0.5*180 = 90 = n/2

    n = h.shape[0]
    target_sum = n / 2.0

    # Repair any slight violation of the integral constraint.
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Simple redistribution respecting bounds.
        diff = target_sum - cur_sum
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        for _ in range(5):
            cur_sum = h.sum()
            diff = target_sum - cur_sum
            free = (h > 0.0) & (h < 1.0)
            if not free.any():
                break
            adjust = diff / free.sum()
            h[free] += adjust
            h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n

    def objective(vec):
        """Return max_k ∫ h(x)(1‑h(x+k)) dx for a given vector."""
        corr = np.correlate(vec, 1.0 - vec, mode='full') * dx
        return np.max(corr)

    best_h = h.copy()
    best_score = objective(best_h)

    # ------------------------------------------------------------------
    # Simple stochastic local search respecting the constraints.
    # ------------------------------------------------------------------
    # Maximum iterations is deliberately high; the loop exits on time.
    while True:
        # Time guard.
        if time.time() - start_time > budget_s * 0.95:
            break

        i, j = rng.integers(0, n, size=2)
