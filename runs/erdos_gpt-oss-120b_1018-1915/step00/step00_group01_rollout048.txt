import time
import numpy as np
from scipy.optimize import minimize

def _c5_value(h, dx):
    """Compute the C5 bound for a discretized step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_to_simplex_box(x, target_sum):
    """
    Project vector x onto the intersection of the box [0,1]^n and the affine
    hyperplane sum(x) = target_sum. Uses a simple iterative scheme that works
    well for our modest dimensions.
    """
    n = x.size
    # First clip to box
    x = np.clip(x, 0.0, 1.0)
    # Adjust to meet the sum constraint by a uniform shift,
    # then re‑clip, repeat a few times.
    for _ in range(5):
        cur_sum = x.sum()
        delta = (target_sum - cur_sum) / n
        x += delta
        x = np.clip(x, 0.0, 1.0)
    # Final tweak: if still off by a tiny amount, distribute it proportionally
    cur_sum = x.sum()
    if abs(cur_sum - target_sum) > 1e-8:
        diff = target_sum - cur_sum
        # Distribute diff proportionally to free slack (distance to bounds)
        slack_low = x - 0.0
        slack_high = 1.0 - x
        slack = slack_low + slack_high
        if slack.sum() > 0:
            x += diff * slack / slack.sum()
        x = np.clip(x, 0.0, 1.0)
    return x

def _optimize_h(initial_h, n_points, budget, start_time):

    def cost(h):
        return _c5_value(h, dx)

    # Equality constraint for the integral
    cons = ({
        "type": "eq",
        "fun": lambda h: np.sum(h) - target_sum,

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        cost,
        initial_h,
        method="SLSQP",
        bounds=bounds,

        options=opts,

    h_opt = _project_to_simplex_box(res.x, target_sum)
    bound = _c5_value(h_opt, dx)
