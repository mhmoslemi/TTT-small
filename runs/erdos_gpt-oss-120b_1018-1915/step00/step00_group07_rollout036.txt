import numpy as np

def _project_feasible(h, target_sum):
    """Project vector h onto [0,1] box with exact sum = target_sum."""
    for _ in range(10):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        free = (h > 0.0) & (h < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds – rescale globally (rare)
            h *= target_sum / s
            continue
        delta = (target_sum - s) / n_free
        h[free] += delta
    # final clean‑up
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-9:
        scale = target_sum / s
        h = np.clip(h * scale, 0.0, 1.0)
        # tiny residual fix
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            for idx in range(len(h)):
                if diff > 0 and h[idx] < 1.0:
                    add = min(diff, 1.0 - h[idx])
                    h[idx] += add
                    diff -= add
                elif diff < 0 and h[idx] > 0.0:
                    sub = min(-diff, h[idx])
                    h[idx] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
    return h
