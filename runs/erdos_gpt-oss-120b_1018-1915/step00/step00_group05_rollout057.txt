import time
import numpy as np
from scipy import optimize as opt

def _project_to_constraints(x, target_sum):
    """Clip to [0,1] and adjust to meet the sum constraint."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        diff = np.sum(x) - target_sum
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # all at bounds – redistribute uniformly
            x = np.full_like(x, target_sum / x.size)
            break
        delta = diff / n_free
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
    return x

def _evaluate(h, dx):
    """Return max correlation integral for a given h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize
    C5 = max_k ∫ h(x)(1‑h(x+k))dx.

    Returns (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    n = kwargs.get('n_points', 200)               # grid size (≤ 1000)
    dx = 2.0 / n
    target_sum = n / 2.0                           # ∑h = n/2  ⇔  ∫h = 1

    # ---------- initial candidate ----------
    if 'initial_h_values' in globals():
        init = initial_h_values.astype(float)
