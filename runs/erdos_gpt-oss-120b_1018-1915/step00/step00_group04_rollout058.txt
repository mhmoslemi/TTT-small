import time
import numpy as np

def compute_score(h, dx):
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    # Full linear correlation (no wrap)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_sum_constant(h, target_sum):
    """Adjust h minimally to satisfy sum == target_sum while staying in [0,1]."""
    # Simple iterative redistribution
    current = h.copy()
    diff = target_sum - current.sum()
    if abs(diff) < 1e-12:
        return current
    # Distribute diff to random free slots
    free = np.where((current > 0) & (current < 1))[0]
    if free.size == 0:
        # All at bounds – fallback to scaling (may violate bounds slightly)
        scale = target_sum / current.sum()
        current = np.clip(current * scale, 0.0, 1.0)
        # Rescale again if needed
        diff = target_sum - current.sum()
    attempts = 0
    while abs(diff) > 1e-12 and attempts < 1000:
        idx = np.random.randint(0, free.size)
        i = free[idx]
        delta = np.sign(diff) * min(abs(diff), 1.0 - current[i] if diff > 0 else current[i])
        current[i] += delta
        diff = target_sum - current.sum()
        attempts += 1
    # Final clip to be safe
    return np.clip(current, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):

    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.
    Returns (h_values, c5_bound, n_points).

    np.random.seed(seed)
    start_time = time.time()

    # ----- Determine resolution -----

        target_n = 200
        x_base = np.linspace(0.0, 2.0, base_n, endpoint=False)
        x_target = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h = np.interp(x_target, x_base, base_h)
    else:
        # No parent – start from a simple half‑filled rectangle
        target_n = 200
        h = np.zeros(target_n, dtype=float)
        h[:target_n // 2] = 1.0

    n = h.size
    dx = 2.0 / n
    target_sum = n / 2.0
    h = project_sum_constant(h, target_sum)

    best_h = h.copy()
    best_score = compute_score(best_h, dx)

    # ----- Local search parameters -----
    max_iter = 200000  # safety ceiling
    report_interval = 1.0  # seconds
    last_report = start_time

    i = 0
    while i < max_iter and (time.time() - start_time) < budget_s * 0.95:
        # Propose a move that preserves the total sum:
        # pick two distinct indices, transfer a small amount delta from j to i
        i_idx, j_idx = np.random.choice(n, 2, replace=False)

        if delta_max <= 0:

        h_new[i_idx] += delta
