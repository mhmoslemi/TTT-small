import numpy as np
import time

def _compute_c5(h):
    """Return the overlap bound C5 for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _initial_h(n, rng):

    h = np.clip(h, 0.0, 1.0)
    # enforce sum = n/2 exactly by scaling and clipping iteratively
    target = n / 2.0
    for _ in range(10):
        s = h.sum()
        if abs(s - target) < 1e-12:
            break
        factor = target / s
        h *= factor
        h = np.clip(h, 0.0, 1.0)
    # final projection onto the simplex with bounds [0,1]
    excess = h.sum() - target
    if excess > 1e-12:
        # remove excess uniformly from positive entries

        if pos.any():
            h[pos] -= excess / pos.sum()
            h = np.clip(h, 0.0, 1.0)

        neg = h < 1
        if neg.any():
