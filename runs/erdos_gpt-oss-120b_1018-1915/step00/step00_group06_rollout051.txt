import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return discretized C5 for a step function h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def repair_h(h, desired_sum, eps=1e-12):
    """Project h to [0,1] and enforce exact sum=desired_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = desired_sum - s
        if abs(diff) <= eps:
            break
        # distribute the difference uniformly
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final tiny scaling if still off
    s = h.sum()
    if abs(s - desired_sum) > 1e-9:
        factor = desired_sum / s if s != 0 else 0.0
        h = np.clip(h * factor, 0.0, 1.0)
        # one more uniform adjust
        s = h.sum()
        if abs(s - desired_sum) > eps:
            h += (desired_sum - s) / h.size
            h = np.clip(h, 0.0, 1.0)
    return h


def upsample_h(old_h, new_n):
    """Linear interpolation of old_h (defined on [0,2]) to new_n points."""
    old_n = old_h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    new_h = np.interp(x_new, x_old, old_h)
    # enforce sum constraint after interpolation
    desired = new_n / 2.0
    return repair_h(new_h, desired)


def random_perturb(h, scale=0.12, frac=0.07):
    """Add zero‑mean Gaussian noise to a fraction of entries."""
    n = h.size
    k = max(1, int(frac * n))
    idx = np.random.choice(n, k, replace=False)
    noise = np.random.randn(k) * scale
    h_new = h.copy()
    h_new[idx] += noise
    return h_new


def local_optimize(h, max_iter=120, tol=1e-6):
    """SLSQP refinement of h under box and linear constraints."""
    n = h.size
    desired = n / 2.0

    def obj(vec):
        return compute_c5(vec)

    cons = ({
        "type": "eq",
        "fun": lambda vec: vec.sum() - desired,
    })
    bounds = [(0.0, 1.0)] * n
    res = minimize(
        obj,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": max_iter, "ftol": tol, "disp": False},
    )
    if not res.success:
        return h
    return repair_h(res.x, desired)
