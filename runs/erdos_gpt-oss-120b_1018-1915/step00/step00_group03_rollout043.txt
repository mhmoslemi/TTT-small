import time
import numpy as np
import cvxpy as cp

def _compute_c5(h, dx):
    """Return C5 for a given h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

class _Projector:
    """Projection onto {h: sum(h)=n/2, 0<=h<=1} using a tiny QP."""
    def __init__(self, n):
        self.n = n
        self._y = cp.Parameter(n)
        self._x = cp.Variable(n)
        self._prob = cp.Problem(
            cp.Minimize(cp.sum_squares(self._x - self._y)),
            [cp.sum(self._x) == n / 2,
             self._x >= 0,
             self._x <= 1]
        )
        # warm start improves speed for repeated solves
        self._prob.solve(solver=cp.ECOS, warm_start=True)

    def __call__(self, v):
        self._y.value = np.asarray(v, dtype=float)
        self._prob.solve(solver=cp.ECOS, warm_start=True)
        return np.asarray(self._x.value, dtype=float)

def _gradient_worst_lag(h, lag):
    """Subgradient of the worst lag term w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if lag == 0:
        grad[:] = 1.0 - 2.0 * h
    elif lag > 0:
        l = lag
        idx1 = np.arange(l, n)            # i where i-l >=0
        grad[idx1] += 1.0 - h[idx1 - l]
        idx2 = np.arange(0, n - l)        # i where i+l < n
        grad[idx2] -= h[idx2 + l]
    else:  # lag < 0
        l = -lag
        idx1 = np.arange(0, n - l)        # i where i+l < n
        grad[idx1] += 1.0 - h[idx1 + l]
        idx2 = np.arange(l, n)            # i where i-l >=0
        grad[idx2] -= h[idx2 - l]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h for the Erdős C5 constant.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : numpy array of length n_points with values in [0,1]
        c5_bound : the achieved upper bound C5
        n_points : number of discretisation points
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise problem size and starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        h0 = np.full(70, 0.5, dtype=float)   # fallback uniform
    n = h0.shape[0]
    dx = 2.0 / n

    projector = _Projector(n)

    # ensure the start satisfies constraints (project if needed)
    h0 = projector(h0)

    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    max_restarts = 8
    max_iters_per_restart = 2000
    base_step = 0.8        # initial step size (scaled later by dx)
    step_decay = 0.9995    # multiplicative decay per iteration
    time_check_interval = 0.2  # seconds between budget checks

    # ------------------------------------------------------------------
    # main optimisation loop
    # ------------------------------------------------------------------
    for restart in range(max_restarts):
        # stop early if time is almost exhausted
        if time.time() - start_time > 0.95 * budget_s:
            break

        # starting point for this restart
        if restart == 0:
            h = h0.copy()
        else:
            rand_vec = rng.random(n)
            h = projector(rand_vec)

        step = base_step

        for it in range(max_iters_per_restart):
            # periodic time check
            if (it & 31) == 0:  # approx every 32 iterations
                if time.time() - start_time > 0.95 * budget_s:
                    break

            # worst lag and its gradient
            corr = np.correlate(h, 1.0 - h, mode="full")
            worst_idx = int(np.argmax(corr))
            lag = worst_idx - (n - 1)

            grad = _gradient_worst_lag(h, lag)

            # gradient descent step (scale by dx to match objective scale)
            y = h - step * grad * dx
            h = projector(y)

            # evaluate current candidate
            cur_c5 = _compute_c5(h, dx)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

            # simple step-size schedule
            step *= step_decay

            # occasional mild random perturbation to escape plateaus
            if (it + 1) % 200 == 0:
                perturb = rng.normal(scale=0.015, size=n)
                h = projector(h + perturb)

    return (best_h, best_c5, n)
