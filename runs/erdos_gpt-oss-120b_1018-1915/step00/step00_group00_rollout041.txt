import time, random, math
import numpy as np

def _compute_c5(arr, dx):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx for discrete step function."""
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step function h on [0,2] with ∫h = 1 and 0≤h≤1.

    Returns:
        h_values (list of float) – the optimized samples,
        c5_bound (float)        – the achieved C₅ value,
        n_points (int)          – number of samples used.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # 1. Initialise
    # ----------------------------------------------------------------------
    n_points = 45                     # default grid size
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        h = np.full(n_points, 0.5, dtype=float)   # uniform feasible start

    h = np.clip(h, 0.0, 1.0)
    target_sum = n_points / 2.0
    # Simple redistribution to hit the exact sum
    diff = target_sum - h.sum()
    for _ in range(n_points * 3):

        idx = random.randrange(n_points)

            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc

            dec = min(-diff, h[idx])
            h[idx] -= dec
            diff += dec

    dx = 2.0 / n_points
