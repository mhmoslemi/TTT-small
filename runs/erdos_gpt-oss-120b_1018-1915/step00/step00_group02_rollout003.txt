import numpy as np
import time
from scipy.optimize import minimize

# Helper: compute C5 for a discrete h
def compute_c5(h, dx):
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

# Upsample by simple linear interpolation (pairwise averaging)
def upsample(h):
    n = len(h)
    new_n = n * 2
    # repeat each value, then smooth odd positions by averaging neighbours
    up = np.empty(new_n, dtype=h.dtype)
    up[::2] = h
    # odd positions: average of surrounding even points (wrap around)
    up[1::2] = 0.5 * (np.concatenate(([h[-1]], h[:-1])) + h)
    return up

def run(seed=42, budget_s=1100, **kwargs):

    best_score = np.inf

    # resolution schedule: start with n0, double up to ~720 (under 1000)

        # Equality constraint for sum, box bounds for each entry
        cons = ({

        })
        bounds = [(0.0, 1.0) for _ in range(n)]

        # objective wrapper (needs only vector)
