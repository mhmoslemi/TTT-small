import numpy as np
import time
import random
from scipy.optimize import minimize

def _evaluate_c5(h):
    """Return C5 for a given h vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_bounds_and_sum(h, target_sum):
    """Clip to [0,1] and adjust to exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h
    # Simple redistribution: add/subtract uniformly on non‑saturated entries
    free = np.where((h > 0.0) & (h < 1.0))[0]
    if free.size == 0:
        # fallback: scale whole vector then clip again
        h = np.clip(h * (target_sum / cur), 0.0, 1.0)
        return h
    diff = target_sum - cur
    while abs(diff) > 1e-12 and free.size > 0:
        delta = diff / free.size
        # limit delta so we stay in bounds
        for idx in free:
            if delta > 0:
                max_inc = 1.0 - h[idx]
                adj = min(delta, max_inc)
            else:
                max_dec = -h[idx]
                adj = max(delta, max_dec)
            h[idx] += adj
            diff -= adj
        free = np.where((h > 0.0) & (h < 1.0))[0]
    return h

def _slqp_optimize(h0, target_sum, time_deadline):
    """Run a short SLSQP optimisation respecting bounds and sum."""
    n = h0.shape[0]
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - target_sum}

    def fun(x):
        return _evaluate_c5(x)

    opts = {"maxiter": 400, "ftol": 1e-9}
    res = minimize(fun, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options=opts)
    if not res.success:
        return h0
    return _project_bounds_and_sum(res.x, target_sum)

def _hill_climb(h, target_sum, time_deadline):
    """Simple stochastic hill‑climbing preserving sum."""
    n = h.shape[0]
    best_h = h.copy()
    best_score = _evaluate_c5(best_h)
    step = 0.05
    iter_count = 0
    while time.time() < time_deadline:
        iter_count += 1
        # periodically shrink step size
        if iter_count % 2000 == 0:
            step = max(step * 0.7, 0.005)

        i, j = random.sample(range(n), 2)
        # decide direction of transfer
        if random.random() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i

        delta = min(step, best_h[donor], 1.0 - best_h[receiver])
        if delta <= 0:
            continue

        new_h = best_h.copy()
        new_h[donor] -= delta
        new_h[receiver] += delta

        # evaluate
        new_score = _evaluate_c5(new_h)
        if new_score < best_score - 1e-10:
            best_h = new_h
            best_score = new_score
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    time_deadline = start_time + budget_s

    # obtain initial vector
    init = globals().get("initial_h_values")
    if init is not None:
        h0 = np.array(init, dtype=float)
    else:
        # fallback: uniform 0.5 (satisfies integral)
        n_fallback = 45
        h0 = np.full(n_fallback, 0.5, dtype=float)

    n = h0.shape[0]
    target_sum = n / 2.0
    h0 = _project_bounds_and_sum(h0, target_sum)

    # First stage: SLSQP local optimisation
    if time.time() < time_deadline:
        h_opt = _slqp_optimize(h0, target_sum, time_deadline)
    else:
        h_opt = h0

    # Second stage: hill climbing refinement
    if time.time() < time_deadline:
        h_best, best_score = _hill_climb(h_opt, target_sum, time_deadline)
    else:
        h_best = h_opt
        best_score = _evaluate_c5(h_best)

    # Optional: try a few random restarts if time permits
    while time.time() + 0.5 < time_deadline:
        # generate random feasible vector
        rand = np.random.rand(n)
        rand = _project_bounds_and_sum(rand, target_sum)
        # quick SLSQP
        rand_opt = _slqp_optimize(rand, target_sum, time_deadline)
        # hill climb
        rand_best, rand_score = _hill_climb(rand_opt, target_sum, time_deadline)
        if rand_score < best_score:
            h_best = rand_best
            best_score = rand_score

    # Ensure final solution respects constraints exactly
    h_best = _project_bounds_and_sum(h_best, target_sum)
    final_c5 = _evaluate_c5(h_best)

    return h_best, float(final_c5), int(n)
