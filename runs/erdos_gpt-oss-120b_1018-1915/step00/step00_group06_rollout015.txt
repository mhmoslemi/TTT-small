import numpy as np, time, math, random

def compute_c5(h):
    """Return discrete C5 for array h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(corr)

def repair_h(h, target_sum):
    """Project h onto [0,1] and enforce sum == target_sum."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = target_sum - total
    if abs(diff) > 1e-12:
        # distribute uniformly, then re‑clip (few iterations suffice)
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        # second pass for any remaining tiny error
        total = h.sum()
        diff = target_sum - total
        if abs(diff) > 1e-12:
            h += diff / h.size
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(old_h, new_n):
    """Linear interpolation of old_h (sampled on [0,2]) to new_n points."""
    old_n = old_h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    # enforce integral constraint after interpolation
    return repair_h(new_h, target_sum=new_n / 2)

def local_search(init_h, budget, start_time):
    """Simple stochastic hill‑climbing under time budget."""
    n = init_h.size
    target = n / 2.0
    h = init_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    step = 0.1
    iter_cnt = 0
    max_iter = 30000
    while time.time() - start_time < budget and iter_cnt < max_iter:
        pert = np.random.randn(n) * step
        cand = repair_h(h + pert, target)
        val = compute_c5(cand)
        if val < best_val:
            best_val, best_h = val, cand.copy()
        if val < compute_c5(h):
            h = cand
            step = min(step * 1.03, 0.5)
        else:
            step *= 0.95
        if step < 1e-4:
            step = 0.1
        iter_cnt += 1
    return best_h, best_val
