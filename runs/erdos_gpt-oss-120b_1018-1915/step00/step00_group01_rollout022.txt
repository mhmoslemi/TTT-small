import time
import numpy as np

def _project_to_box_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum(x)=target_sum}.
    Uses a simple bisection on the Lagrange multiplier for the affine constraint.
    """
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + mid, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return the max overlap integral for a discrete h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Heuristic optimizer for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)

    # -------------------------------------------------------------
    # Parameters
    # -------------------------------------------------------------
    n_points = 200                     # < 1000 for speed
    target_sum = n_points / 2.0        # because ∫h = 1  →  sum(h)*dx = 1
    max_time = budget_s * 0.95         # safety margin

    # -------------------------------------------------------------
    # Initial construction
    # -------------------------------------------------------------
    try:
        # initial_h_values is provided by the harness (length 47)
        init = np.asarray(initial_h_values, dtype=float)
        # Resample linearly to the working grid
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_old, init)
    except NameError:
        # Fallback: simple half‑filled block
        h = np.concatenate([np.ones(n_points // 2), np.zeros(n_points - n_points // 2)])

    # Enforce the constraints exactly
    h = _project_to_box_and_sum(h, target_sum)

    best_h = h.copy()
    best_score = _c5_value(best_h)

    start = time.time()
    iteration = 0

    # -------------------------------------------------------------
    # Stochastic hill‑climbing loop
    # -------------------------------------------------------------
    while time.time() - start < max_time:
        iteration += 1

        # gradually shrink the proposal step size
        step_scale = 0.07 / (1.0 + iteration * 0.001)

        proposal = best_h + np.random.normal(scale=step_scale, size=n_points)
        proposal = _project_to_box_and_sum(proposal, target_sum)

        score = _c5_value(proposal)

        if score < best_score:
            best_score = score
            best_h = proposal

    # Final evaluation (ensuring constraints are still met)
    best_h = _project_to_box_and_sum(best_h, target_sum)
    final_c5 = _c5_value(best_h)

    return best_h, final_c5, n_points
