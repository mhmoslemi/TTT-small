import time
import numpy as np

def _evaluate_c5(h, dx):
    """Return max overlap integral for discretized h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_to_bounds_and_sum(v, target_sum, eps=1e-12):
    """Clip v to [0,1] and adjust to meet exact target sum while staying in bounds."""
    v = np.clip(v, 0.0, 1.0)
    current = v.sum()
    deficit = target_sum - current
    if abs(deficit) < eps:
        return v
    # Iteratively distribute the deficit among free variables
    for _ in range(1000):
        if deficit > 0:
            free = np.where(v < 1.0)[0]
            capacity = (1.0 - v[free]).sum()
            if capacity < eps:
                break
            inc = np.minimum(deficit * (1.0 - v[free]) / capacity, 1.0 - v[free])
            v[free] += inc
            deficit = target_sum - v.sum()
        else:
            free = np.where(v > 0.0)[0]
            capacity = v[free].sum()
            if capacity < eps:
                break
            dec = np.minimum(-deficit * v[free] / capacity, v[free])
            v[free] -= dec
            deficit = target_sum - v.sum()
        if abs(deficit) < eps:
            break
    # final clipping to guard against tiny numerical errors
    return np.clip(v, 0.0, 1.0)

def _upsample_initial(h0, n_new):
    """Linear interpolation of h0 (defined on [0,2)) to n_new points."""
    n0 = len(h0)
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x0, h0)
    target = n_new / 2.0
    # Scale to roughly meet the integral, then project precisely
    scale = target / h_new.sum()
    h_new *= scale
    return _project_to_bounds_and_sum(h_new, target)

    """
    Optimize a step function h on [0,2] (discretized) to minimize
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """

    start_time = time.time()

    # Choose a working resolution (higher gives more flexibility)
    n_points = max(200, int(n0 * 4))  # at least 200 samples
    h = _upsample_initial(h0, n_points)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h, dx)

    # Simple stochastic hill‑climb with occasional annealing
    step_scale = 0.1  # typical magnitude of delta adjustments
    iter_count = 0
    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:  # keep margin for return
            break
        iter_count += 1
        # Propose a move: adjust two coordinates preserving the sum
        i, j = rng.choice(n_points, size=2, replace=False)
        delta = rng.uniform(-step_scale, step_scale)
        # Clamp to keep within [0,1]
