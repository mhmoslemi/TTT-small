import time
import random
import math
import numpy as np

def compute_max_overlap(h: np.ndarray, dx: float) -> float:
    """
    Compute   max_k  ∫ h(x) (1 - h(x+k)) dx
    via discrete correlation using FFT.
    """
    n = h.size
    size = 2 * n
    # linear convolution via zero‑padding
    fft_h = np.fft.fft(h, size)
    fft_g = np.fft.fft(1.0 - h, size)
    conv = np.fft.ifft(fft_h * np.conj(fft_g)).real
    corr = conv[: 2 * n - 1]          # length 2n‑1, same as np.correlate(full)
    return np.max(corr) * dx


def hill_climb(initial_h: np.ndarray, dx: float,
               time_limit: float, seed: int) -> tuple:
    """
    Simple simulated‑annealing hill‑climb on binary vectors.
    Returns (best_h, best_score).
    """
    rnd = random.Random(seed)
    n = initial_h.size
    half = n // 2

    # project to binary with exactly half ones
    order = np.argsort(-initial_h)          # descending
    h = np.zeros(n, dtype=float)
    h[order[:half]] = 1.0

    best_h = h.copy()
    best_score = compute_max_overlap(best_h, dx)

    # annealing schedule
    T = 0.12
    Tmin = 1e-6
    decay = 0.9996

    start = time.time()
    while time.time() - start < time_limit:
        # choose a random 1‑position and a random 0‑position to swap
        ones = np.flatnonzero(h == 1.0)
        zeros = np.flatnonzero(h == 0.0)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rnd.choice(ones)
        j = rnd.choice(zeros)

        # perform swap
        h[i], h[j] = 0.0, 1.0
        new_score = compute_max_overlap(h, dx)

        # acceptance test
        if new_score < best_score:
            best_score = new_score
            best_h = h.copy()
            accept = True
        else:
            prob = math.exp(-(new_score - best_score) / T) if T > 0 else 0.0
            accept = rnd.random() < prob

        if not accept:
            # revert swap
            h[i], h[j] = 1.0, 0.0

        T = max(T * decay, Tmin)

    return best_h, best_score

    # choose a moderate resolution; larger n gives more flexibility

    dx = 2.0 / n_points

    # try to initialise from the provided parent if it exists

        parent = np.asarray(initial_h_values, dtype=float)

        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        init_h = np.interp(x_new, x_old, parent)
    except Exception:
        # fallback: random start
        init_h = np.random.rand(n_points)

    # time budgeting: allocate most of the budget to repeated restarts
    overall_start = time.time()
    best_h_overall = None

    while time.time() - overall_start < budget_s * 0.94:
