import numpy as np, time, math

def compute_c5(h):
    """Return C5 for step function h sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def repair(h, target_sum):
    """Clip to [0,1] and enforce exact sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    # Distribute diff greedily while respecting bounds
    if abs(diff) < 1e-12:
        return h
    # indices that can still move up or down
    up = np.where(h < 1.0)[0]
    down = np.where(h > 0.0)[0]
    # iterate small adjustments
    it = 0
    while abs(diff) > 1e-12 and it < 1000:
        if diff > 0 and up.size:
            idx = np.random.choice(up)
            step = min(diff, 1.0 - h[idx])
            h[idx] += step
            diff -= step
            if h[idx] >= 1.0 - 1e-12:
                up = up[up != idx]
        elif diff < 0 and down.size:
            idx = np.random.choice(down)
            step = min(-diff, h[idx])
            h[idx] -= step
            diff += step
            if h[idx] <= 1e-12:
                down = down[down != idx]
        else:
            break
        it += 1
    # final tiny correction by adding to a random feasible entry
    if abs(diff) > 1e-12:
        idx = np.random.randint(len(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def perturb(h, sigma=0.05):
    """Gaussian perturbation with repair."""
    n = h.shape[0]
    new = h + np.random.normal(0.0, sigma, size=n)
    return new

def upscale(h, new_n):
    """Linear interpolation of h to length new_n (must be even)."""
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)
