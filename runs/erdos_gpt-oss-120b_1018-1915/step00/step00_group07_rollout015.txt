import time
import math
import numpy as np

def _evaluate_c5(h, dx):
    """Return max overlap C5 for step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

    c5_bound : float
        Corresponding C5 value.
    n_points : int

    """
