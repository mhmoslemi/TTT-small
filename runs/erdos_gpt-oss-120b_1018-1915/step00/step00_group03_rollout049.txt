import numpy as np
import time
import random

# Helper to compute C5 directly if the external evaluator is unavailable
def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

# Repair a vector so that 0<=h[i]<=1 and sum(h)=target
def _repair_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    # Simple iterative projection onto the affine hyperplane with box constraints
    for _ in range(15):
        current = h.sum()
        diff = target - current
        if abs(diff) < 1e-10:
            break
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # Final tiny adjustment if needed
    residual = target - h.sum()
    if abs(residual) > 1e-8:
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            h[mask] += residual / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    # ------------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    time_limit = budget_s * 0.95  # safety margin

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        n = 256                       # power of two, modest size
        # start with half of the entries at 1, half at 0, shuffled
        h0 = np.zeros(n, dtype=np.float64)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

    target_sum = n / 2.0               # because dx = 2/n and integral must be 1
    h_best = _repair_sum(h0.copy(), target_sum)

    # Initial evaluation
    eval_fn = globals().get("evaluate_erdos_solution", _compute_c5)
    c5_best = eval_fn(h_best)

    # ------------------------------------------------------------------
    # Optimization parameters
    # ------------------------------------------------------------------
    sigma = 0.20                       # initial mutation scale
    min_sigma = 0.005
    iter_count = 0
    no_improve = 0

    # Main loop
    while time.time() - start < time_limit:
        iter_count += 1

        # Mutate
        cand = h_best + sigma * np.random.randn(n)
        cand = _repair_sum(cand, target_sum)

        # Evaluate
        c5_cand = eval_fn(cand)

        # Accept if better
        if c5_cand < c5_best:
            h_best = cand
            c5_best = c5_cand
            sigma = max(min_sigma, sigma * 0.95)   # tighten mutations
            no_improve = 0
        else:
            no_improve += 1

        # Periodic diversification
        if iter_count % 200 == 0:
            # occasional random restart around the best point
            sigma = min(0.3, sigma * 1.1)
            if no_improve > 1000:
                # perturb heavily to escape possible plateau
                cand = h_best + 0.5 * np.random.randn(n)
                cand = _repair_sum(cand, target_sum)
                c5_cand = eval_fn(cand)
                if c5_cand < c5_best:
                    h_best = cand
                    c5_best = c5_cand
                    sigma = max(min_sigma, sigma * 0.9)
                no_improve = 0

        # Optional up‑sampling step if we still have generous time left

    # Return result
    # ------------------------------------------------------------------
    return (h_best, c5_best, n)
