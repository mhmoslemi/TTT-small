def _c5_score(h):
    """Return max overlap integral for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _neighbor(h, step):

    n = h.shape[0]
