import numpy as np
import time
import math

# -------------------------------------------------------
# Helper: projection of a vector onto the box [0,1]^n
# with a prescribed sum S using a simple bisection on the
# Lagrange multiplier lambda: w_i = clip(v_i - λ, 0, 1)
# -------------------------------------------------------
def _project(v, target_sum, tol=1e-9, max_iter=60):
    low, high = -np.max(v) - 1.0, np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if abs(s - target_sum) <= tol:
            return w
        if s > target_sum:
            low = lam
        else:
            high = lam
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)


# -------------------------------------------------------
# Evaluate C5 for a discrete step function h
# -------------------------------------------------------
def _c5_of(h, dx):
    # full convolution h * (1-h)
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx


# -------------------------------------------------------
# Simulated‑annealing / hill‑climb optimizer
# -------------------------------------------------------
def _optimize(n_points, budget, init_h=None, seed=0):
    rng = np.random.default_rng(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # initialise
    if init_h is not None:
        h = np.array(init_h, copy=True, dtype=float)
        # if length differs, interpolate linearly
        if h.size != n_points:
            xs_old = np.linspace(0, 2, h.size, endpoint=False)
            xs_new = np.linspace(0, 2, n_points, endpoint=False)
            h = np.interp(xs_new, xs_old, h)
        h = _project(h, target_sum)
    else:
        # random start respecting bounds then project
        h = rng.random(n_points)
        h = _project(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_of(best_h, dx)

    start_time = time.time()
    # annealing schedule parameters
    T0 = 0.5
    T_end = 1e-4
    decay = 0.9995

    T = T0
    it = 0
    while time.time() - start_time < budget:
        it += 1
        # propose: perturb a small random subset
        h_new = best_h.copy()
        # pick k indices (≈sqrt(n) or at least 1)
        k = max(1, int(math.sqrt(n_points)))
        idx = rng.choice(n_points, size=k, replace=False)
        # add Gaussian noise
        h_new[idx] += rng.normal(scale=0.2, size=k)
        # project back to feasible set
        h_new = _project(h_new, target_sum)

        c5_new = _c5_of(h_new, dx)
        delta = c5_new - best_c5

        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            best_h, best_c5 = h_new, c5_new

        # temperature decay
        T = max(T_end, T * decay)

        # occasional restart if stuck
        if it % 5000 == 0:
            if rng.random() < 0.3:
                h_restart = rng.random(n_points)
                h_restart = _project(h_restart, target_sum)
                c5_restart = _c5_of(h_restart, dx)
                if c5_restart < best_c5:
                    best_h, best_c5 = h_restart, c5_restart

    return best_h, best_c5


# -------------------------------------------------------
# Main entry point required by the harness
# -------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)

    The optimizer works in three stages of increasing resolution,
    spending the available time budget among them.
    """
    rng = np.random.default_rng(seed)
    total_start = time.time()

    # stage 1: coarse resolution (n=90)
    n1 = 90
    t1 = budget_s * 0.35
    h1, c5_1 = _optimize(n1, t1, init_h=globals().get("initial_h_values", None), seed=rng.integers(1 << 30))

    # stage 2: medium resolution (n=180)
    n2 = 180
    t2 = budget_s * 0.35
    h2, c5_2 = _optimize(n2, t2, init_h=h1, seed=rng.integers(1 << 30))

    # stage 3: fine resolution (n=360) – if time remains
    elapsed = time.time() - total_start
    remaining = max(0.0, budget_s - elapsed)
    n3 = 360
    if remaining > 5.0:  # need a few seconds to be worthwhile
        h3, c5_3 = _optimize(n3, remaining, init_h=h2, seed=rng.integers(1 << 30))
        best_h, best_c5, best_n = h3, c5_3, n3
    else:
        # pick the better of stage 1 and 2
        if c5_1 <= c5_2:
            best_h, best_c5, best_n = h1, c5_1, n1
        else:
            best_h, best_c5, best_n = h2, c5_2, n2

    # Final projection to guarantee feasibility (numerical drift)
    best_h = _project(best_h, best_n / 2.0)

    return best_h, float(best_c5), int(best_n)
