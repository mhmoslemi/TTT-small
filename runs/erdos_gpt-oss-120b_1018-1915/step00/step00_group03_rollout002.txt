import numpy as np
import time
import math

def _evaluate(h):
    """Return max overlap C5 for a given h (numpy array)."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def _repair(h, target_sum, max_iter=20, tol=1e-9):
    """Project h onto [0,1] box and sum=h_target."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:            # everything at a bound, just shift uniformly
            h += diff / h.size
        else:
            h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _perturb(h, sigma, rng):
    """Add Gaussian noise and repair."""
    cand = h + rng.normal(scale=sigma, size=h.shape)
    return cand

def _interpolate(h, new_n):
    """Linear interpolation of h from its current size to new_n points."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0/old_n
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0/new_n
    return np.interp(x_new, x_old, h)
