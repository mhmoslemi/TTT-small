import time, random, numpy as np

def _evaluate(h):
    """Return C5 for sample array h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_simplex(v, total):
    """Euclidean projection of v onto {0<=x<=1, sum x = total} using cvxpy."""
    import cvxpy as cp
    n = v.shape[0]
    x = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(x - v)),
                      [x >= 0, x <= 1, cp.sum(x) == total])
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    return x.value

    np.random.seed(seed)

        h = _project_to_simplex(h, target)

    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = h.copy()
