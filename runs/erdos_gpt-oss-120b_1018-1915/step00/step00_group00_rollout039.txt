import time
import numpy as np

# Helper: project onto [0,1] box and enforce exact sum = target
def _project_box_sum(x, target):
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    for _ in range(10):
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        x[free] += delta
        np.clip(x, 0.0, 1.0, out=x)
        diff = target - x.sum()
        if abs(diff) < 1e-12:
            break
    if abs(diff) > 1e-12:
        # fallback simple scaling
        x = x / x.sum() * target
        np.clip(x, 0.0, 1.0, out=x)
    return x


def run(seed=42, budget_s=1100, **kwargs):
