def _initial_h(n):

    h = np.full(n, 0.5, dtype=float)
    return h

def _make_start():
    """Return a feasible starting vector and its length."""
    try:
        h0 = np.array(initial_h_values, copy=True, dtype=float)
        n = h0.size
        # Ensure exact feasibility (numeric drift)
        desired = n / 2.0
        cur = h0.sum()
        if abs(cur - desired) > 1e-9:
            # redistribute the difference uniformly then clip
            h0 += (desired - cur) / n
            np.clip(h0, 0.0, 1.0, out=h0)
            # final correction by scaling the deficit/excess
            diff = desired - h0.sum()
            if abs(diff) > 1e-9:
                # distribute remaining diff to a random index with capacity
                idx = np.where((diff > 0) & (h0 < 1.0))[0]
                if idx.size:
                    inc = min(diff, 1.0 - h0[idx[0]])
                    h0[idx[0]] += inc
                else:
                    idx = np.where((diff < 0) & (h0 > 0.0))[0]
                    if idx.size:
                        dec = min(-diff, h0[idx[0]])
                        h0[idx[0]] -= dec
        return h0, n
    except NameError:
        # no parent supplied – fall back to 200‑point uniform start
        n = 200
        return _initial_h(n), n

def _random_move(h, step):
    """Perform a feasible transfer of mass between two coordinates."""
    n = h.size
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None, None, 0.0
    # move from i to j (positive delta) – bounded by availability and capacity
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        return None, None, 0.0
    delta = np.random.rand() * max_delta * step
    h[i] -= delta
    h[j] += delta
    return i, j, delta

    """Search for a step function h on [0,2] with improved C5 bound.

    Returns

    c5_bound : float
        Corresponding C5 value (the objective).
    n_points : int
        Number of discretisation points (len(h_values)).

    # Parameters for the stochastic hill‑climber

    attempts = 0
    successes = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:

        i, j, delta = _random_move(h, step_factor)
        if i is None:

        attempts += 1
        cur_score = evaluate_erdos_solution(h, n)

        if cur_score < best_score - 1e-12:

            best_h = h.copy()
            successes += 1
        else:

        # Periodically adapt step size based on recent acceptance ratio
