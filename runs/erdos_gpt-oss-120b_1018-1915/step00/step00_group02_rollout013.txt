import time
import numpy as np

# evaluate_erdos_solution is pre‑imported; we will also compute C5 directly for speed.
def _c5_of(h, dx):
    """Return max correlation integral for a candidate h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _make_feasible(h):
    """Clip to [0,1] and enforce exact integral (sum = len(h)/2)."""
    h = np.clip(h, 0.0, 1.0)
    n = h.size
    target = n / 2.0
    cur = h.sum()
    # Simple redistribution to fix the sum while staying in bounds
    # Move excess/deficit uniformly over free slots.
    if np.isclose(cur, target):
        return h
    diff = target - cur
    # indices that can still move in the needed direction
    if diff > 0:
        free = np.where(h < 1.0)[0]
        cap = 1.0 - h[free]
        total_cap = cap.sum()
        if total_cap < diff:  # should not happen
            diff = total_cap
        h[free] += diff * cap / total_cap
    else:
        free = np.where(h > 0.0)[0]
        cap = h[free]
        total_cap = cap.sum()
        if total_cap < -diff:
            diff = -total_cap
        h[free] += diff * cap / total_cap  # diff is negative
    return h

def _initial_h(n):
    """Create a simple feasible start: half 1's, half 0's shuffled."""
    h = np.zeros(n, dtype=float)
    h[: n // 2] = 1.0
    rng = np.random.default_rng()
    rng.shuffle(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the allotted budget."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Base resolution coming from the provided parent, if any.
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        base_n = parent.size
    else:
        parent = None
        base_n = 45

    # Choose a working resolution (multiple of the base, not too large).
    # 360 points give decent granularity while staying cheap.
    work_n = 360
    dx = 2.0 / work_n

    # Build initial candidate.
    if parent is not None:
        # Upsample parent to work_n points via linear interpolation.
        x_parent = np.linspace(0.0, 2.0, base_n, endpoint=False)
        x_work = np.linspace(0.0, 2.0, work_n, endpoint=False)
        h = np.interp(x_work, x_parent, parent)
        h = _make_feasible(h)
    else:
        h = _initial_h(work_n)

    best_h = h.copy()
    best_c5 = _c5_of(best_h, dx)

    # Parameters for the stochastic local search
    max_iter = 2000000  # safety cap; loop will break on time.
    eps_start = 0.02
    eps_end = 0.001
    eps_decay = 0.9995

    eps = eps_start
    it = 0
    while True:
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break
        if it >= max_iter:
            break
        it += 1

        # pick donor (i) with h>0 and receiver (j) with h<1
        i = rng.integers(work_n)
        if best_h[i] == 0.0:
            continue
        j = rng.integers(work_n)
        if best_h[j] == 1.0 or i == j:
            continue

        delta = min(eps, best_h[i], 1.0 - best_h[j])
        if delta <= 0:
            continue

        # propose move
        h_candidate = best_h.copy()
        h_candidate[i] -= delta
        h_candidate[j] += delta

    official_c5 = evaluate_erdos_solution(best_h)
