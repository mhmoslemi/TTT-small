import time
import numpy as np

def _project_onto_bounds_and_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the box [0,1]^n with the additional linear
    constraint sum(x)=target_sum.  Uses a simple bisection on the Lagrange
    multiplier λ for the equality constraint.
    """
    lo, hi = -2.0, 2.0  # wide enough for the multiplier range
    for _ in range(60):   # enough for double precision
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray, dx: float) -> float:
    """
    Compute the C5 value for a discrete step function h.
    """
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def run(seed: int = 42, budget_s: float = 1100.0, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that gives a small
    C5 bound.  Returns (h_values, c5_bound, n_points).
    """
    rnd = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # problem parameters (fixed)
    # --------------------------------------------------------------
    n_points = 45                      # allowed small size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0        # because integral of h must be 1

    # --------------------------------------------------------------
    # initialise h
    # --------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # ensure feasibility (tiny numerical drift may exist)
        h = _project_onto_bounds_and_sum(h, target_sum)
    else:
        # uniform start that satisfies the sum constraint
        h = np.full(n_points, target_sum / n_points)

    best_h = h.copy()
    best_score = _c5_value(best_h, dx)

    start_time = time.time()
    end_time = start_time + budget_s

    # --------------------------------------------------------------
    # optimisation loop (simple stochastic descent with annealing)
    # --------------------------------------------------------------
    iteration = 0
    while time.time() < end_time:
        # time‑dependent step size (larger at start, shrink later)
        t = (time.time() - start_time) / budget_s
        sigma = 0.25 * (1.0 - t) + 0.02   # between 0.27 and 0.02

        # propose a new candidate by Gaussian perturbation
        cand = best_h + sigma * rnd.standard_normal(n_points)
        cand = _project_onto_bounds_and_sum(cand, target_sum)

        cand_score = _c5_value(cand, dx)

        # accept if better; otherwise accept with small probability
        if cand_score < best_score:
            best_h, best_score = cand, cand_score
        else:
            # simulated‑annealing style acceptance
            temp = 0.01 + 0.5 * (1.0 - t)   # temperature decays
            prob = np.exp(-(cand_score - best_score) / temp)
            if rnd.random() < prob:
