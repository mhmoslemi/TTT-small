import time
import numpy as np
import random
import math

def _compute_c5(h, dx):
    """Maximum overlap integral for a discrete step function h."""
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------
    # Initialise h
    # ----------------------------------------------------------------------
    try:
        h = np.array(initial_h_values, dtype=float)  # provided by harness
    except NameError:
        # fallback: 80 uniform values (sum must be n/2)
        n = 80
        h = np.full(n, 0.5, dtype=float)
    else:
        n = h.size

    dx = 2.0 / n
    target_sum = n / 2.0
    # Ensure the integral constraint exactly (tiny numerical drift may happen)
    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-12:
        h *= target_sum / current_sum

    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)
    cur_c5 = best_c5

    # ----------------------------------------------------------------------
    # Simulated‑annealing style search
    # ----------------------------------------------------------------------
    T = 0.5               # initial temperature
    T_min = 1e-5
    decay = 0.9999        # exponential cooling per iteration
    iter_count = 0
    max_iter = 10_000_000  # generous upper bound; we'll stop by time

    while time.time() - start_time < budget_s - 0.5:  # small safety margin
        # pick two distinct indices
        i, j = random.sample(range(n), 2)
        hi, hj = h[i], h[j]

        # convex mixing preserves sum and bounds
        alpha = random.random()
        new_hi = (1 - alpha) * hi + alpha * hj
        new_hj = (1 - alpha) * hj + alpha * hi

        # apply tentative move
        h[i], h[j] = new_hi, new_hj
        new_c5 = _compute_c5(h, dx)

        # acceptance test
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # Metropolis criterion
            if T > 0:
                prob = math.exp(-(new_c5 - cur_c5) / T)
                if random.random() < prob:
                    accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = hi, hj

        # temperature schedule
        T = max(T_min, T * decay)

        iter_count += 1
        # occasional restart from the best found to avoid stagnation
        if iter_count % 200_000 == 0:
            h[:] = best_h
            cur_c5 = best_c5

    return (best_h, best_c5, n)
