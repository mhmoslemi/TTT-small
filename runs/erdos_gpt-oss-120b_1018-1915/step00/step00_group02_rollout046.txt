import time
import math
import numpy as np

def compute_c5(h):
    """Return the C5 value for a given step function h (numpy array)."""
    n = h.size
    dx = 2.0 / n
    # correlation h with (1-h) using FFT
    size = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(h, n=size)
    O = np.fft.rfft(1.0 - h, n=size)
    corr = np.fft.irfft(H * np.conj(O), n=size)[: 2 * n - 1]
    return corr.max() * dx
