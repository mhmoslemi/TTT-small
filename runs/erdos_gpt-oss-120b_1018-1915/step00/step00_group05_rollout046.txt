import time
import numpy as np
from scipy.optimize import minimize

def _objective(h, dx):
    """Return max overlap C5 for step function h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx

def _repair(candidate, target_sum):
    """Clip to [0,1] and enforce exact sum by proportional scaling."""
    cand = np.clip(candidate, 0.0, 1.0)
    s = np.sum(cand)
    if s == 0:
        # fallback: uniform fill
        cand[:] = target_sum / len(cand)
        return cand
    factor = target_sum / s
    cand *= factor
    # second clipping may disturb sum; iterate once more
    cand = np.clip(cand, 0.0, 1.0)
    s = np.sum(cand)
    if abs(s - target_sum) > 1e-10:
        # distribute residual uniformly within bounds
        diff = target_sum - s
        cand += diff / len(cand)
        cand = np.clip(cand, 0.0, 1.0)
    return cand

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        Optimized values of h on the discretization.
    c5_bound : float
        Corresponding upper bound C5 = max_k ∫ h(x)(1-h(x+k))dx.
    n_points : int
        Number of discretization points used.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine number of points and initial vector
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n_points = init.size
        h0 = init.copy()
    else:
        n_points = kwargs.get("n_points", 70)
        h0 = np.random.rand(n_points)

    # Enforce the integral constraint sum(h)=n/2  (dx = 2/n)
    target_sum = n_points / 2.0
    h0 = _repair(h0, target_sum)

    dx = 2.0 / n_points
    best_h = h0
    best_val = _objective(best_h, dx)

    # Equality constraint for optimizer
    eq_cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
    bounds = [(0.0, 1.0)] * n_points

    # Helper to run a single local optimisation from a start point
    def _local_opt(start_vec):
        res = minimize(
            fun=lambda x: _objective(x, dx),
            x0=start_vec,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 300, "ftol": 1e-12, "disp": False},
        )
        x = _repair(res.x, target_sum)
        val = _objective(x, dx)
        return x, val

    # Main search loop – alternate random perturbations and local refinements
    iteration = 0
    while time.time() - start_time < budget_s - 5.0:  # keep a few seconds margin
        iteration += 1

        # Random perturbation of the current best solution
        pert = best_h + 0.05 * np.random.randn(n_points)
        pert = _repair(pert, target_sum)

        # Local optimisation from perturbed point
        h_candidate, val_candidate = _local_opt(pert)

        # Accept if improvement
        if val_candidate < best_val:
            best_h, best_val = h_candidate, val_candidate

        # Occasionally restart from a fresh random feasible point
        if iteration % 10 == 0:
            rand_start = np.random.rand(n_points)
            rand_start = _repair(rand_start, target_sum)
            h_candidate, val_candidate = _local_opt(rand_start)
            if val_candidate < best_val:
                best_h, best_val = h_candidate, val_candidate

    return best_h, best_val, n_points
