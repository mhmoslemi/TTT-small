import time
import numpy as np

def compute_c5(h):
    """Return the C5 value for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr.max() * dx).item()

def random_transfer(h, max_delta):
    """Propose a new vector by moving mass from one index to another."""
    n = h.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n)
    if i == j:
        return None, None
    # feasible delta respecting bounds
    delta = np.random.rand() * max_delta
    delta = min(delta, h[i])               # cannot go below 0
    delta = min(delta, 1.0 - h[j])         # cannot exceed 1
    if delta <= 0.0:
        return None, None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new, delta

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 constant.

    Returns
    -------
    h_best : np.ndarray
        Optimized vector (length n_points) with values in [0,1] and sum = n/2.
    c5_best : float
        Corresponding C5 bound.
    n_points : int
        Number of discretization points.
    """
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size and initial vector
    if 'initial_h_values' in globals():
        h_curr = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: equal 0.5 everywhere (satisfies sum constraint)
        n_default = 90
        h_curr = np.full(n_default, 0.5, dtype=np.float64)

    n = h_curr.shape[0]

    # Ensure feasibility of the initial vector (clamp and renormalise)
    h_curr = np.clip(h_curr, 0.0, 1.0)
    total = h_curr.sum()
    target = n / 2.0
    if abs(total - target) > 1e-12:
        # Scale uniformly then correct residual by distributing to feasible entries
        h_curr *= target / total
        # Small correction loop
        for _ in range(5):
            excess = h_curr.sum() - target
            if abs(excess) < 1e-12:
                break
            # Pull from entries >0, push to those <1
            if excess > 0:
                mask = h_curr > 0
                adjust = excess / mask.sum()
                h_curr[mask] -= adjust
                h_curr = np.clip(h_curr, 0.0, 1.0)
            else:
                mask = h_curr < 1
                adjust = (-excess) / mask.sum()
                h_curr[mask] += adjust
                h_curr = np.clip(h_curr, 0.0, 1.0)

    best_h = h_curr.copy()
    best_score = compute_c5(best_h)

    # Parameters for the annealing loop
    max_delta = 0.05          # maximum mass moved per proposal
    temp = 0.02               # initial temperature
    cooling = 0.99995         # exponential cooling factor

            if jitter_score < best_score:
                best_h = h_jit
                best_score = jitter_score
