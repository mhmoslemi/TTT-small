import time
import numpy as np
import random

def _compute_c5(h):
    """Return C5 for a candidate h (numpy array)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _repair(h, target_sum):
    """
    Project h onto the feasible set:
      0 <= h <= 1   and   sum(h) == target_sum
    Uses a few scaling/clipping passes.
    """
    for _ in range(5):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if s == 0.0:
            h[:] = target_sum / h.size
            continue
        h *= target_sum / s
    # final adjustment for any tiny residual
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # add/subtract residual to a coordinate that can move
        if residual > 0:
            idx = np.argmax(h < 1.0)
            add = min(residual, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.argmax(h > 0.0)
            sub = min(-residual, h[idx])
            h[idx] -= sub
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic optimisation for the Erdős C5 constant.

    Returns
    -------
    h_values : list of floats
        Optimised step function sampled on a uniform grid of n points.
    c5_bound : float
        The obtained upper bound C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    n_points : int
        Number of samples (length of h_values).
    """
    rnd = random.Random(seed)
    np.random.seed(seed)

    # Determine problem size
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
        h = h0.copy()
    else:
        n = 200                       # modest size when no starter is given
        h = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    h = _repair(h, target_sum)

    best_h = h.copy()
    best_c = _compute_c5(best_h)

    start_time = time.time()
    max_time = budget_s * 0.95  # safety margin

    # Annealing / hill‑climbing parameters
    max_iter = 200000
    sigma_init = 0.25
    sigma_min = 0.005

    iteration = 0
    while time.time() - start_time < max_time and iteration < max_iter:
        # gradually shrink mutation strength
        sigma = sigma_min + (sigma_init - sigma_min) * max(0.0, 1.0 - iteration / max_iter)

        cand = best_h + sigma * np.random.randn(n)
        cand = _repair(cand, target_sum)

        cand_c = _compute_c5(cand)

        if cand_c < best_c:

                restart_c = _compute_c5(restart)
                if restart_c < best_c:
