import time, math, numpy as np
from scipy.optimize import minimize, differential_evolution

def _dx(n):
    return 2.0 / n

def _c5_of_h(h):
    """Return C5 value for a step vector h."""
    n = h.size
    dx = _dx(n)
    g = 1.0 - h
    # full discrete cross‑correlation
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def _project_sum(h, target):
    """Scale h to hit the target sum while staying in [0,1]."""
    s = np.sum(h)
    if s == 0:
        # uniform fill
        h[:] = target / h.size
        return h
    h = h * (target / s)
    # clip then renormalise if necessary
    np.clip(h, 0.0, 1.0, out=h)
    diff = np.sum(h) - target
    if abs(diff) > 1e-12:
        # distribute the tiny discrepancy uniformly
        h -= diff / h.size
        np.clip(h, 0.0, 1.0, out=h)
    return h
