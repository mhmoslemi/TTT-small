import time
import numpy as np

def _repair(h: np.ndarray, n: int) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h) = n/2 by water‑filling."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            inc = np.minimum(inc, 1.0 - h[idx])
            h[idx] += inc
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            dec = (-diff) / idx.size
            dec = np.minimum(dec, h[idx])
            h[idx] -= dec
    # final tiny adjustment to hit the target exactly (within float eps)
    cur = h.sum()
    if abs(cur - target) > 1e-10:
        # distribute the remaining tiny error uniformly
        h += (target - cur) / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h: np.ndarray) -> float:
    """Compute C₅ = max_k ∫ h(x)(1‑h(x+k))dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx

def _initial_h(n_target: int) -> np.ndarray:
    """Create a starting vector, interpolating from initial_h_values if present."""
    if "initial_h_values" in globals():
        base = globals()["initial_h_values"]
        n0 = base.shape[0]
        # linear interpolation on the circle [0,2)
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x1 = np.linspace(0.0, 2.0, n_target, endpoint=False)
        h = np.interp(x1, x0, base)
        h = _repair(h, n_target)
        return h
    # fallback: random start
    h = np.random.rand(n_target)
    return _repair(h, n_target)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h that (approximately) minimizes C₅.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : list of floats, length n_points, each in [0,1]
        c5_bound : float, the achieved C₅ value
        n_points : int, the discretisation size used
    """
    np.random.seed(seed)
    start_time = time.time()

    # ---- choose a modest resolution -------------------------------------------------
    base_n = 42
    if "initial_h_values" in globals():
        base_n = globals()["initial_h_values"].shape[0]
    # aim for about 80–120 points (fast enough, yet finer than the parent)
    n = max(84, 2 * base_n)          # at least double the parent
    if n > 200:                     # keep it small
        n = 200

    # ---- initialise -----------------------------------------------------------------
    best_h = _initial_h(n)
    best_score = _c5_value(best_h)

    # ---- optimisation parameters ----------------------------------------------------
    sigma = 0.15                     # mutation scale
    min_sigma = 0.02
    decay = 0.9995                  # geometric decay per iteration
    max_iter = 10_000_000           # safety cap

    iter_count = 0
    while True:
        if time.time() - start_time > budget_s * 0.95:
            break
        if iter_count >= max_iter:
            break
        iter_count += 1

        # propose a mutated candidate
        cand = best_h + np.random.normal(0.0, sigma, size=n)
        cand = _repair(cand, n)
        cand_score = _c5_value(cand)

        # accept if better (simple (1+1)‑ES)
        if cand_score < best_score:
            best_h, best_score = cand, cand_score

        # gradually reduce mutation strength
        sigma = max(min_sigma, sigma * decay)

        # occasional random restart to escape plateaus
        if iter_count % 5000 == 0 and np.random.rand() < 0.1:
            rnd = _initial_h(n)
            rnd_score = _c5_value(rnd)
            if rnd_score < best_score:
                best_h, best_score = rnd, rnd_score

    # final repair (ensure exact integral)
    best_h = _repair(best_h, n)
    best_score = _c5_value(best_h)

    return best_h.tolist(), float(best_score), int(n)
