import time, math, numpy as np, random

def _project_to_constraints(vec, n):
    """Clip to [0,1] and enforce sum(vec)=n/2 exactly."""
    vec = np.clip(vec, 0.0, 1.0)
    target = n / 2.0
    # Simple iterative redistribution to meet the exact sum while respecting bounds
    for _ in range(15):
        diff = target - vec.sum()
        if abs(diff) < 1e-12:
            break
        vec += diff / n
        vec = np.clip(vec, 0.0, 1.0)
    # Final scaling if still off (rare)
    if abs(vec.sum() - target) > 1e-8:
        vec *= target / vec.sum()
        vec = np.clip(vec, 0.0, 1.0)
    return vec

def _evaluate(vec, dx):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for the discrete vector."""
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx

    """Search for a step function h on [0,2] that lowers the Erdős overlap bound.

    h_values : np.ndarray
