import time
import numpy as np
from scipy.optimize import minimize

def _compute_c5(h):
    """Return the discretized C5 value for a step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / h.size
    return dx * np.max(corr)

def _project(h, target_sum):
    """Clip to [0,1] and enforce the linear sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    # Simple affine correction; iterate a couple of times for stability
    for _ in range(3):
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        delta = (target_sum - cur) / h.size
        h += delta
        h = np.clip(h, 0.0, 1.0)
    return h

def _interpolate_parent(parent, n):
    """Resize parent vector to length n using linear interpolation."""
    m = parent.size
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n, endpoint=False)
    new = np.interp(x_new, x_old, parent)
    # rescale to satisfy the integral constraint exactly
    desired = n / 2.0
    new = _project(new, desired)
    return new

def _optimize_start(start, max_iter=200):
    """Run a bounded SLSQP optimisation starting from `start`."""
    n = start.size
    target_sum = n / 2.0

    bounds = [(0.0, 1.0) for _ in range(n)]

    cons = ({
        "type": "eq",
        "fun": lambda h: np.sum(h) - target_sum,
    },)

    def obj(h):
        return _compute_c5(h)

    opts = {
        "maxiter": max_iter,
        "ftol": 1e-9,
        "disp": False,
    }

    res = minimize(
        obj,
        start,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=opts,
    )
    h_opt = _project(res.x, target_sum)
    return h_opt, _compute_c5(h_opt)

def _random_local_search(best_h, best_val, time_limit, start_time):
    """Perform cheap random perturbations while time permits."""
    n = best_h.size
    target_sum = n / 2.0
    rng = np.random.default_rng()
    while time.time() - start_time < time_limit:
        pert = rng.normal(scale=0.05, size=n)
        cand = best_h + pert
        cand = _project(cand, target_sum)
        val = _compute_c5(cand)
        if val < best_val:
            best_h, best_val = cand, val
    return best_h, best_val

    best_h = None
