import time
import numpy as np
from scipy.optimize import minimize

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h, dx):
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx approximated by correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def repair_vector(h, target_sum, eps=1e-12):
    """
    Project h onto the feasible set:
      0 <= h_i <= 1  and  sum(h) = target_sum
    Simple iterative redistribution; sufficient for moderate dimensions.
    """
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    if abs(cur - target_sum) <= eps:
        return h

    # If we need to add mass, distribute to entries <1
    # If we need to remove mass, subtract from entries >0
    for _ in range(10):
        diff = target_sum - cur
        if abs(diff) <= eps:
            break
        if diff > 0:
            mask = h < 1.0 - eps
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            add = diff / cnt
            h[mask] = np.minimum(1.0, h[mask] + add)
        else:
            mask = h > eps
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            sub = (-diff) / cnt
            h[mask] = np.maximum(0.0, h[mask] - sub)
        cur = np.sum(h)
    return h


def initial_interpolated_h(n_points, rng):
    """Create an initial feasible h using the global initial_h_values if present."""
    try:
        base = np.asarray(initial_h_values, dtype=float)
        m = base.size
        x_base = np.linspace(0.0, 2.0, m, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(x_new, x_base, base)
    except NameError:
        # No parent construction – start from a random Dirichlet‑like vector
        raw = rng.rand(n_points)
        h = raw / raw.sum()
    # Scale to satisfy the integral constraint (sum = n/2)
    target = n_points / 2.0
    h = h * (target / np.sum(h))
    h = repair_vector(h, target)
    return h

# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫ h = 1, 0≤h≤1
    to minimize C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Choose a modest resolution – enough to beat the current record but fast.
    n_points = 120          # 0.01666… spacing, well under 1k samples
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initial candidate (interpolated from parent if available)
    best_h = initial_interpolated_h(n_points, rng)
    best_score = compute_c5(best_h, dx)

    # Equality constraint for SLSQP
    eq_cons = {"type": "eq", "fun": lambda h: np.sum(h) - target_sum}
    bounds = [(0.0, 1.0)] * n_points

    # Optimization options – keep iterations modest for time budget
    opt_options = {"maxiter": 500, "ftol": 1e-9, "disp": False}

    # Helper to run a single local optimisation from a feasible start
    def local_optimize(start_vec):
        res = minimize(
            fun=lambda h: compute_c5(h, dx),
            x0=start_vec,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options=opt_options,
        )
        if not res.success:
            return None, np.inf
        h_opt = repair_vector(res.x, target_sum)  # safeguard feasibility
        score = compute_c5(h_opt, dx)
        return h_opt, score

    # First deterministic local optimisation (starting from interpolated)
    h_opt, score = local_optimize(best_h)
    if h_opt is not None and score < best_score:
        best_h, best_score = h_opt, score

    # Stochastic restarts – limited by remaining time
    restart = 0
    while time.time() - start_time < budget_s * 0.9:  # keep margin
        restart += 1
        # Random perturbation of the current best solution
        pert = rng.normal(scale=0.07, size=n_points)
        cand = best_h + pert
        cand = np.clip(cand, 0.0, 1.0)
        cand = repair_vector(cand, target_sum)

        h_opt, score = local_optimize(cand)
        if h_opt is not None and score < best_score:
            best_h, best_score = h_opt, score

        # Abort if we already beat the target bound comfortably
        if best_score < 0.3808:
            break

        # Safety net: avoid infinite loop
        if restart >= 20:
            break

    # Final sanity check – ensure constraints are met
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = repair_vector(best_h, target_sum)
    best_score = compute_c5(best_h, dx)

    return best_h, float(best_score), n_points
