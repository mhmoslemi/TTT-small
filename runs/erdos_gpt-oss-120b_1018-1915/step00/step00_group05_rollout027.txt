import time
import numpy as np

def _compute_c5(h, dx):
    """Return C5 for a discrete step vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_feasible(vec, target_sum):
    """Clip to [0,1] and adjust to meet the exact sum (water‑filling)."""
    x = np.clip(vec, 0.0, 1.0)
    total = x.sum()
    if np.isclose(total, target_sum):
        return x
    # simple scaling with iterative clipping
    while True:
        if total == 0:
            # nothing to scale, just distribute uniformly
            x[:] = target_sum / len(x)
            break
        scale = target_sum / total
        x *= scale
        over = x > 1.0
        if not np.any(over):
            break
        excess = np.sum(x[over] - 1.0)
        x[over] = 1.0
        under = x < 1.0
        cnt_under = np.count_nonzero(under)
        if cnt_under == 0:
            break
        x[under] += excess / cnt_under
        total = x.sum()
    # final safety clip
    x = np.clip(x, 0.0, 1.0)
    # tiny adjustment to hit target_sum exactly (within floating error)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        # distribute diff to non‑boundary entries
        free = (x > 0.0) & (x < 1.0)
        if np.any(free):
            x[free] += diff / np.count_nonzero(free)
    return x

def _random_feasible(n, target_sum, rng):
    """Generate a random feasible vector under 0≤h≤1 and sum=target_sum."""
    x = rng.random(n)
    x = x / x.sum() * target_sum
    return _project_feasible(x, target_sum)

def run(seed=42, budget_s=1100, **kwargs):

    best_h = h_curr.copy()
