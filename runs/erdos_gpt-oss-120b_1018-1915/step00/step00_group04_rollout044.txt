def _compute_c5(h):
    """Return the overlap bound for a step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _repair(h):
    """Project h into [0,1] and enforce sum(h)=len(h)/2."""
    n = h.size
    # Clip to bounds
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

    # Simple redistribution loop
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Distribute diff uniformly over non‑boundary entries
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:  # all at bounds, fallback to scaling
            h = h * (target / s) if s != 0 else np.full_like(h, 0.5)
            break
        inc = diff / cnt
        h[free] += inc
        # Re‑clip
        h = np.clip(h, 0.0, 1.0)
    # Final tiny adjustment by scaling if still off
    s = h.sum()
    if abs(s - target) > 1e-9:
        h *= target / s
        h = np.clip(h, 0.0, 1.0)
    return h

def _mutate(parent, scale=0.05):
    """Gaussian mutation respecting bounds, then repair."""
    n = parent.size
    child = parent + np.random.normal(0.0, scale, size=n) * (parent.max() - parent.min())
    return _repair(child)
