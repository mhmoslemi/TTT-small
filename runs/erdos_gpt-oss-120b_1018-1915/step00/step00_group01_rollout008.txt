import numpy as np
import time
import random

def _compute_c5(h):
    """Return max overlap integral for discretized step function h."""
    n = h.size
    dx = 2.0 / n
    # correlate h with (1-h) in full mode gives length 2n-1, shift k corresponds to index offset n-1
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _make_feasible(h):
    """Clip to [0,1] and renormalize sum to n/2 by a simple projection."""
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    current = h.sum()
    # scale uniformly if possible
    if current > 0:
        h *= target / current
    # after scaling some values may violate bounds; fix iteratively
    for _ in range(5):
        over = h - 1.0
        under = -h
        over_idx = over > 0
        under_idx = under > 0
        excess = over[over_idx].sum()
        deficit = -under[under_idx].sum()
        if excess > 0 and deficit > 0:
            # move mass from overfull to underfull proportionally
            h[over_idx] -= excess * (over[over_idx] / excess)
            h[under_idx] += excess * (under[under_idx] / deficit)
        else:
            break
    h = np.clip(h, 0.0, 1.0)
    # final tiny correction to hit target sum exactly
    diff = target - h.sum()
