import numpy as np
import time
import random

def _project_to_bounds_sum(x, target, lo=0.0, hi=1.0, max_iter=20, eps=1e-9):
    """Project vector x onto { lo <= xi <= hi, sum xi = target }."""
    x = np.clip(x, lo, hi)
    for _ in range(max_iter):
        s = x.sum()
        diff = target - s
        if abs(diff) <= eps:
            break
        free = (x > lo + eps) & (x < hi - eps)
        n_free = free.sum()
        if n_free == 0:
            # all at bounds, distribute remaining uniformly within bounds
            n_free = len(x)
            free = np.ones_like(x, dtype=bool)
        adjust = diff / n_free
        x[free] += adjust
        x = np.clip(x, lo, hi)
    # final correction for tiny residuals
    residual = target - x.sum()
    if abs(residual) > eps:
        # distribute residual to any free slots
        free = (x > lo + eps) & (x < hi - eps)
        if free.any():
            x[free] += residual / free.sum()
            x = np.clip(x, lo, hi)
    return x

def _evaluate(h, dx):
    """Return the C5 value for a candidate h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):

    Returns (h_values, c5_bound, n_points).

    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because integral must be 1

    # DE hyper‑parameters
    popsize = max(20, 4 * n_points)  # enough diversity, but not huge
    F = 0.5
    CR = 0.9
    max_gen = 5000  # will be limited by time

    # Initial population

    # include the given construction if feasible (project just in case)

        rand_vec = rnd.rand(n_points)
        proj = _project_to_bounds_sum(rand_vec, target_sum)

    pop = np.array(pop)
    scores = np.array(scores)

    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
