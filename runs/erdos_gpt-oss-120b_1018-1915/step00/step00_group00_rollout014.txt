import time
import math
import random
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap C5 for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _project_feasible(vec, target_sum):
    """Clip to [0,1] and adjust to hit target_sum preserving feasibility."""
    v = np.clip(vec, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target_sum) < 1e-12:
        return v
    # Simple redistributive fix: iterate a few times
    for _ in range(10):
        diff = target_sum - v.sum()
        free = (v > 0.0) & (v < 1.0)
        if not np.any(free):
            break
        adjust = diff / free.sum()
        v[free] += adjust
        v = np.clip(v, 0.0, 1.0)
    # Final tiny correction on a random free entry
    if abs(v.sum() - target_sum) > 1e-9:
        free = (v > 0.0) & (v < 1.0)
        if np.any(free):
            i = np.flatnonzero(free)[0]
            v[i] += target_sum - v.sum()
            v[i] = min(max(v[i], 0.0), 1.0)
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        # ensure feasibility of the supplied start
        target_sum = n / 2.0
        h0 = _project_feasible(h0, target_sum)
    else:
        n = 45
        target_sum = n / 2.0
        h0 = np.full(n, 0.5)  # uniform feasible start

    dx = 2.0 / n
    target_sum = n / 2.0

    # keep best solution found
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    # parameters for stochastic search
    start_time = time.time()
    max_iters = 300000
    T0 = 0.02
    T_min = 1e-5

    h = best_h.copy()
    for it in range(1, max_iters + 1):
        # time guard
        if it % 2000 == 0:
            if time.time() - start_time > budget_s * 0.95:
                break

        # temperature schedule (linear cooling)
        T = max(T_min, T0 * (1.0 - it / max_iters))

        # propose a move: transfer a small mass between two positions
        i, j = np.random.choice(n, size=2, replace=False)
        if h[i] <= 0.0 or h[j] >= 1.0:
            continue
        # magnitude limited by available slack and a base step size
        base_step = 0.05
        delta = base_step * random.random()
        delta = min(delta, h[i], 1.0 - h[j])
        if delta <= 0:
            continue

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # evaluate
        c5_new = _compute_c5(h_new, dx)

        # accept if better or by Metropolis criterion
        improve = c5_new < best_c5
        if improve or random.random() < math.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

        # occasional random restart to escape plateaus
        if it % 12000 == 0:
            rand_vec = np.random.rand(n)
            rand_vec = _project_feasible(rand_vec, target_sum)
            h = rand_vec.copy()
            c5_rand = _compute_c5(h, dx)
            if c5_rand < best_c5:
                best_c5 = c5_rand
                best_h = h.copy()

    # final sanity projection (should already be feasible)
    best_h = _project_feasible(best_h, target_sum)

    return best_h, best_c5, n
