import time

def _project_feasible(v, target_sum):
    """Euclidean projection of v onto the box [0,1]^n with sum == target_sum."""
    lo, hi = -2.0, 2.0  # wide enough for the Lagrange multiplier search
    for _ in range(60):  # binary search for lambda
        lam = (lo + hi) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    # final projection
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Numerical fix: if sum is slightly off, redistribute tiny amount
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute proportionally among non‑tight entries
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            h[free] += diff / free.sum()
            np.clip(h, 0.0, 1.0, out=h)
    return h
