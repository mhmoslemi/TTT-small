import time
import numpy as np

def _evaluate_c5(h):
    """Return C5 for a step function h sampled on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h):
    """Project h onto the box [0,1] and the affine hyperplane sum(h)=len(h)/2."""
    n = h.shape[0]
    target = n * 0.5
    # box projection
    h = np.clip(h, 0.0, 1.0)
    # affine projection by iterative redistribution
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # indices that can move in the needed direction
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        # apply limited delta
        for i in idx:
            if diff > 0:
                max_inc = 1.0 - h[i]
                inc = min(delta, max_inc)
            else:
                max_dec = -h[i]
                inc = max(delta, max_dec)
            h[i] += inc
    # final clip to avoid tiny numerical overshoots
    return np.clip(h, 0.0, 1.0)

def _upscale(h, new_n):
    """Interpolate h (length m) to length new_n preserving the integral."""
    if h.shape[0] == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, h.shape[0], endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return _repair(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C₅ overlap bound.

    Returns
    -------
    h_values : np.ndarray
        The sampled step function (values in [0,1]).
    c5_bound : float
        The achieved C₅ value for the returned h.
    n_points : int
        Number of samples (length of h_values).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise ---------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)  # provided by harness
    except NameError:
        # fall back to a uniform feasible start (average 0.5)
        n0 = 70
        h0 = np.full(n0, 0.5, dtype=float)

    h0 = _repair(h0)
    best_h = h0.copy()
    best_c5 = _evaluate_c5(best_h)

    # ---- multi‑resolution schedule -----------------------------------------
    # start size, then progressively double (capped at ~1500 to stay fast)
    sizes = [best_h.shape[0]]
    while sizes[-1] < 1200:
        sizes.append(min(sizes[-1] * 2, 1200))

    # ---- optimisation loop --------------------------------------------------
    for n in sizes[1:]:
        # stop if budget is almost used
        if time.time() - start_time > budget_s * 0.95:
            break

        # upscale current best to new resolution
        cur_h = _upscale(best_h, n)

        # parameters for stochastic descent
        step_scale = 0.2
        min_scale = 1e-4
        decay = 0.995
        max_iters = 5000

        for it in range(max_iters):
            # budget check inside the inner loop
            if time.time() - start_time > budget_s * 0.99:
                break

            # propose a mutated candidate
            candidate = cur_h + rng.normal(scale=step_scale, size=n)

        best_h = _upscale(best_h, n)

    best_h = _repair(best_h)
    final_c5 = _evaluate_c5(best_h)

    return best_h, float(final_c5), int(best_h.shape[0])
