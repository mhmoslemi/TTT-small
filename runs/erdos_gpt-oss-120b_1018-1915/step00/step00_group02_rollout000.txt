import numpy as np, time, random, math
# evaluate_erdos_solution and possibly initial_h_values are provided by the harness.

def _c5_score(h):
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx via discrete correlation."""
    # full correlation length = 2*n-1, shift k corresponds to index offset n-1
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _random_move(h, eps):
    """Return a new h after moving a small mass eps from i to j respecting bounds."""
    n = h.size
    # choose donor with positive mass, receiver with room
    donors = np.where(h > 1e-12)[0]
    receivers = np.where(h < 1 - 1e-12)[0]
    if donors.size == 0 or receivers.size == 0:
        return h.copy()
    i = random.choice(donors)
    j = random.choice(receivers)
    delta = min(eps, h[i], 1.0 - h[j])
    if delta <= 0:
        return h.copy()
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    return h2

def _refine_resolution(h):
    """Double resolution by repeating each entry twice, then renormalize."""
    h2 = np.repeat(h, 2)
    # after repetition, integral doubled, so scale to maintain sum = 1
    h2 *= 0.5
    return h2
