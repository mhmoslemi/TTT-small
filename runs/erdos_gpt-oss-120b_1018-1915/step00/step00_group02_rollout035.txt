import numpy as np
import time

def _evaluate_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    cor = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(cor.max())

def _project_bounds(h):
    """Clip to [0,1] while preserving the total sum exactly."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        return np.full_like(h, target / n)
    # simple scaling to hit target, then re‑clip if needed
    h *= target / s
    h = np.clip(h, 0.0, 1.0)
    # final tiny correction for any residual error
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # distribute residual uniformly within bounds
        free = (h > 0) & (h < 1)
        if free.any():
            add = residual / free.sum()
            h[free] += add
            h = np.clip(h, 0.0, 1.0)
    # enforce exact sum by a final scaling of a single element if needed
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff < 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h
