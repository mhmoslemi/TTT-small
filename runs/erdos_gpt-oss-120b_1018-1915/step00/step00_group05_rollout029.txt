import time
import numpy as np
from scipy.optimize import minimize

def _evaluate(h, dx):
    """Return max overlap C5 for a given step vector h."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a low‑overlap step function h on [0,2].

    Returns
    -------
    h_values : ndarray
        Best step vector found (values in [0,1]).
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of discretisation points used.
    """
    np.random.seed(seed)

    # ----- configuration -------------------------------------------------
    n = int(kwargs.get('n_points', 70))          # default resolution
    dx = 2.0 / n
    target_ones = n // 2                         # exact integral constraint

    # ----- initialise h --------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size != n:
            # linear interpolation to required size
            x_old = np.linspace(0, 2, init.size, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        h = np.clip(init, 0.0, 1.0)
    else:
        h = np.random.rand(n)

    # project to a feasible binary vector with exactly n/2 ones
    idx = np.argsort(-h)               # descending
    h_bin = np.zeros(n, dtype=float)
    h_bin[idx[:target_ones]] = 1.0
    h = h_bin

    # ----- helper --------------------------------------------------------
    def evaluate(vec):
        return _evaluate(vec, dx)

    # ----- stochastic swap search ----------------------------------------
    best_h = h.copy()
    best_val = evaluate(best_h)
    cur_h = best_h.copy()

    start = time.time()
    no_improve = 0
    max_no_improve = 3000

    while time.time() - start < budget_s * 0.95:
        # select a random 1‑position and 0‑position and swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        new_h = cur_h.copy()
        new_h[i] = 0.0
        new_h[j] = 1.0

        val = evaluate(new_h)
        if val < best_val - 1e-9:
            best_h, best_val = new_h, val
            cur_h = new_h
            no_improve = 0
        else:
            no_improve += 1
            # occasional uphill move to escape plateaus
            if np.random.rand() < 0.001:
                cur_h = new_h

        if no_improve >= max_no_improve:
            # random reshuffle of current vector
            np.random.shuffle(cur_h)
            no_improve = 0

    # ----- local continuous refinement ------------------------------------
    # Use SLSQP to fine‑tune the real‑valued vector while keeping constraints.
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target_ones},)
    bounds = [(0.0, 1.0)] * n

    res = minimize(evaluate, best_h, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': 200, 'ftol': 1e-12, 'disp': False})

    refined = np.clip(res.x, 0.0, 1.0)

    # enforce the exact sum constraint (tiny numerical drift)
    diff = np.sum(refined) - target_ones
    if abs(diff) > 1e-8:
        # shift uniformly then renormalise within bounds
        refined -= diff / n
        refined = np.clip(refined, 0.0, 1.0)
        cur_sum = np.sum(refined)
        if cur_sum > 0:
            refined *= target_ones / cur_sum

    refined_val = evaluate(refined)
    if refined_val < best_val:
        best_h, best_val = refined, refined_val

    return best_h, best_val, n
