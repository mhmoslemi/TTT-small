def project_to_sum_and_box(y, target_sum):
    import cvxpy as cp
    n = len(y)
    x = cp.Variable(n)
    prob = cp.Problem(cp.Minimize(cp.sum_squares(x - y)),
                      [x >= 0, x <= 1, cp.sum(x) == target_sum])
    prob.solve(solver=cp.ECOS, warm_start=True)

import time
import numpy as np
import scipy.optimize as opt

def _project_to_sum_and_box(y, target_sum):
    """Simple iterative projection onto box [0,1] and sum constraint."""
    x = np.clip(y, 0.0, 1.0)
    for _ in range(100):
        total = x.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        nfree = int(free.sum())
        if nfree == 0:
            break
        delta = diff / nfree
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    return x

def _objective(h, dx):
    """Objective = max overlap + penalty for violating the integral constraint."""
    # max correlation term
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr) * dx
    # penalty (large enough to keep sum ≈ 1)
    penalty = 1e4 * abs(h.sum() * dx - 1.0)
    return max_val + penalty

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) within the allotted time."""
    np.random.seed(seed)
    start_time = time.time()

    best_h = None
    best_score = np.inf
    best_n = None

    # Possible grid sizes (start small, then refine)
    grid_sizes = [45, 90, 180]

    # Grab any provided initial vector
    parent = globals().get("initial_h_values", None)

    for n in grid_sizes:
        if time.time() - start_time > budget_s * 0.95:
            break
        dx = 2.0 / n
        target_sum = n / 2.0  # because sum(h)*dx = 1

        # Build a list of starting points
        initials = []

        # 1) use parent if compatible or upscale/downscale it
