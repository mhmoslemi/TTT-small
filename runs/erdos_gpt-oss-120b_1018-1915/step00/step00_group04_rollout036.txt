import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant for a discretized step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def local_search(h: np.ndarray, max_iters: int, time_limit: float):
    """Simple simulated‑annealing on the sum‑preserving hyperplane."""
    start = time.time()
    n = h.size
    best_h = h.copy()
    best_val = compute_c5(best_h)
    T0 = 0.1  # initial temperature

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue

        # propose moving a small amount epsilon from j to i
        eps = (np.random.rand() * 2.0 - 1.0) * 0.2  # |eps| ≤ 0.2
        hi, hj = h[i], h[j]
        new_hi = hi + eps
        new_hj = hj - eps

        # stay inside [0,1]
        if not (0.0 <= new_hi <= 1.0 and 0.0 <= new_hj <= 1.0):
            continue

        h[i], h[j] = new_hi, new_hj
        val = compute_c5(h)
        delta = val - best_val
        T = T0 * (1.0 - it / max_iters)  # linear cooling

        if delta < 0.0 or np.exp(-delta / (T + 1e-12)) > np.random.rand():
            # accept
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            # reject
            h[i], h[j] = hi, hj

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of samples."""
    np.random.seed(seed)
    start_all = time.time()

    # ----- initial construction ------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 42
        h = np.random.rand(n0)
        # scale to meet the integral constraint (sum = n/2)
        h *= (n0 / 2) / h.sum()
        h = np.clip(h, 0.0, 1.0)

    # enforce exact sum = n/2 (tiny correction)
    target = h.size / 2
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)

    best_h, best_val = h.copy(), compute_c5(h)

    # ----- multiscale optimisation -------------------------------------------
    # build a list of increasingly finer discretisations (up to ~300 points)
    sizes = [best_h.size]
    while sizes[-1] < 300:
        sizes.append(sizes[-1] * 2)

    for n_target in sizes:
        # respect overall budget
        if time.time() - start_all > budget_s * 0.95:
            break

        # upscale if needed
        if n_target != best_h.size:
            xs_old = np.linspace(0, 2, best_h.size, endpoint=False)
            xs_new = np.linspace(0, 2, n_target, endpoint=False)
            h_interp = np.interp(xs_new, xs_old, best_h)
            # re‑scale to satisfy the sum constraint
            h_interp *= (n_target / 2) / h_interp.sum()
            h_interp = np.clip(h_interp, 0.0, 1.0)
            diff = (n_target / 2) - h_interp.sum()
            if abs(diff) > 1e-12:
                h_interp += diff / n_target
                h_interp = np.clip(h_interp, 0.0, 1.0)
            best_h = h_interp

        # allocate iterations proportionally to remaining time
        remaining = budget_s - (time.time() - start_all)
        max_iters = int(4e4 * (remaining / budget_s))
        max_iters = max(2000, max_iters)  # ensure a minimal effort

        best_h, best_val = local_search(
            best_h,
            max_iters=max_iters,
            time_limit=budget_s - (time.time() - start_all),
        )

    # final evaluation (ensures consistency)
    best_val = compute_c5(best_h)
    return (best_h.tolist(), float(best_val), best_h.size)
