import time
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_sum_bounds(h, target_sum, max_iter=100):
    """Clip to [0,1] and then adjust to hit the target sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        total = h.sum()
        if abs(total - target_sum) < 1e-12:
            break
        diff = target_sum - total
        free = (h > 0.0) & (h < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds – distribute proportionally
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            continue
        add = diff / cnt
        h[free] += add
        h = np.clip(h, 0.0, 1.0)
    return h

def upscale(h_old, new_n):
    """Linear interpolation of an old grid onto a finer grid."""
    old_n = len(h_old)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h_old)
    target = new_n / 2.0
    h_new = project_sum_bounds(h_new, target)
    return h_new

def propose_move(h, max_step=0.02):
    """Create a new candidate by moving mass between two positions."""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # maximum transferable mass respecting bounds
    max_transfer = min(max_step, h[i], 1.0 - h[j])
    if max_transfer <= 0:
        return None
    delta = np.random.rand() * max_transfer
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def local_search(start_h, dx, budget, start_time):
    """Simple hill‑climbing with mass‑transfer moves."""
    best_h = start_h.copy()
    best_c = compute_c5(best_h, dx)
    n = len(start_h)
    # number of iterations per second estimate (very rough)
    iters = 0
    while time.time() - start_time < budget:
        cand = propose_move(best_h)
        if cand is None:
            continue
        cand_c = compute_c5(cand, dx)
        if cand_c < best_c:
            best_h, best_c = cand, cand_c
        iters += 1
        # occasional random perturbation to escape plateaus
        if iters % 5000 == 0:
            rand_idx = np.random.randint(n)
            rand_val = np.random.rand()
            cand = best_h.copy()
            cand[rand_idx] = np.clip(rand_val, 0.0, 1.0)
            # re‑project to keep the integral exact
            cand = project_sum_bounds(cand, n / 2.0)
            cand_c = compute_c5(cand, dx)
            if cand_c < best_c:
                best_h, best_c = cand, cand_c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h=1 to minimise
    C5 = max_k ∫ h(x)(1‑h(x+k)) dx.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # Base resolution from the provided parent construction, if any.
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values, dtype=float)
        base_n = len(base_h)
    else:
        # fallback: uniform distribution satisfying the integral
        base_n = 47
        base_h = np.full(base_n, 0.5)
    # Ensure the base respects the exact integral.
    base_h = project_sum_bounds(base_h, base_n / 2.0)

    # Candidate resolutions (keep them modest to stay within time budget)
    resolutions = [base_n, base_n * 2, base_n * 4, base_n * 8]
    # Trim to a reasonable maximum (≈500 points)
    resolutions = [r for r in resolutions if r <= 500]

    best_overall_h = None
    best_overall_c = np.inf
    best_overall_n = None

    # Divide time roughly equally among resolutions
    remaining_budget = budget_s * 0.95  # a safety margin
    per_res_budget = remaining_budget / len(resolutions)

    for n in resolutions:
        dx = 2.0 / n
        # Initialise at this resolution
        if n == base_n:
            h0 = base_h.copy()
        else:
            h0 = upscale(base_h, n)

        # Local optimisation within the allotted time slice
        h_opt, c_opt = local_search(h0, dx, per_res_budget, time.time())

        if c_opt < best_overall_c:
            best_overall_h = h_opt
            best_overall_c = c_opt
            best_overall_n = n

        # Update remaining budget
        elapsed = time.time() - start_time
        remaining_budget = budget_s * 0.95 - elapsed
        if remaining_budget <= 0:
            break
        per_res_budget = max(0.1, remaining_budget / (len(resolutions) - resolutions.index(n) - 1))

    # Final safety projection (should already be satisfied)
    best_overall_h = project_sum_bounds(best_overall_h, best_overall_n / 2.0)

    return best_overall_h, best_overall_c, int(best_overall_n)
