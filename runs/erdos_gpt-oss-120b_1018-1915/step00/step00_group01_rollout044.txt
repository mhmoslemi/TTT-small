import numpy as np
import time
from itertools import repeat

# Helper: evaluate C5 for a discrete h on [0,2]
def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

# Project vector onto [0,1] and enforce sum = target (n/2) via simple scaling+clipping
def project(h, target):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        return np.full_like(h, target / len(h))
    # Scale to hit target, then clip again; repeat a few times to stay in bounds
    for _ in range(5):
        h *= target / total
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
        if abs(total - target) < 1e-12:
            break
    # Final fine‑adjustment: distribute any tiny residual uniformly
    residual = target - total
    if abs(residual) > 1e-12:
        h += residual / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

# Simple hill‑climbing improvement for a fixed n
def improve_h(h0, dx, budget, rng):
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h, dx)
    n = len(h)
    target_sum = n / 2.0
    step_size = 0.02  # typical move magnitude
    start = time.time()
    while time.time() - start < budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 2 * step_size
        hi, hj = h[i] + delta, h[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h_candidate = h.copy()
        h_candidate[i], h_candidate[j] = hi, hj
        # projection only needed for numerical drift
        h_candidate = project(h_candidate, target_sum)
        val = evaluate_c5(h_candidate, dx)
        if val <= best_val:
            h = h_candidate
            if val < best_val:
                best_val = val
                best_h = h_candidate.copy()
        # occasional random restart to escape plateaus
        if rng.random() < 0.001:
            h = project(rng.random(n), target_sum)
    return best_h, best_val

# Upsample a vector to a new resolution using linear interpolation
def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)
