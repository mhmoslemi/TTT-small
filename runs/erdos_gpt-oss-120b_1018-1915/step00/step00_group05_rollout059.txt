import time
import numpy as np
from scipy.optimize import minimize

# The harness provides `initial_h_values` (may be undefined) and
# `evaluate_erdos_solution` for final evaluation.

def _objective(h, dx):
    """Return max overlap C5 for a given step vector h."""
    g = 1.0 - h
    # full convolution length = 2*n-1, but we only need the max value
    corr = np.correlate(h, g, mode='full')
    return np.max(corr) * dx

def _sum_constraint(h, target_sum):
    return np.sum(h) - target_sum

def _run_once(x0, dx, target_sum, bounds, cons, maxiter):
    """Run a single constrained local optimisation and return (h, C5)."""
    res = minimize(
        fun=lambda x: _objective(x, dx),
        x0=x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': maxiter, 'ftol': 1e-9, 'disp': False}
    )
    h_opt = np.clip(res.x, 0.0, 1.0)          # safety clip
    # enforce the equality constraint by a simple projection if needed
    diff = np.sum(h_opt) - target_sum
    if abs(diff) > 1e-8:
        h_opt -= diff / len(h_opt)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        # renormalise once more
        scale = target_sum / np.sum(h_opt) if np.sum(h_opt) > 0 else 1.0
        h_opt *= scale
        h_opt = np.clip(h_opt, 0.0, 1.0)
    return h_opt, _objective(h_opt, dx)
