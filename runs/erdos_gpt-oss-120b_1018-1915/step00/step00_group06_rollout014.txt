import time
import random
import numpy as np
from scipy import optimize

def _objective(h):
    """Compute C5 for a discretized step function h."""
    n = h.size
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * np.max(corr)

def _project(h, desired_sum):
    """Clip to [0,1] and renormalise to the required integral."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if cur == 0:
        # fallback: uniform fill
        h[:] = desired_sum / h.size
        return h
    scale = desired_sum / cur
    h *= scale
    # after scaling some entries may breach bounds – clip again and re‑balance
    for _ in range(2):
        h = np.clip(h, 0.0, 1.0)
        excess = h.sum() - desired_sum
        if abs(excess) < 1e-12:
            break
        # distribute the excess uniformly on interior points that are not at a bound
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            break
        adjust = excess / free.sum()
        h[free] -= adjust
    return h

def _optimize_once(start_h, max_iter=250):
    """Run a single SLSQP optimisation starting from start_h."""
    n = start_h.size
    desired = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - desired}
    res = optimize.minimize(
        _objective,
        start_h,
        method="SLSQP",
        bounds=bounds,
        constraints=[cons],
        options={"maxiter": max_iter, "ftol": 1e-12, "disp": False},
    )
    if res.success:
        h_opt = _project(res.x, desired)
        val = _objective(h_opt)
        return h_opt, val
    # on failure fall back to the start point (projected)
    h_proj = _project(start_h, desired)
    return h_proj, _objective(h_proj)

def _random_perturb(h, scale=0.02):
    """Create a nearby feasible vector by moving a small amount of mass."""
    n = h.size
    i, j = random.sample(range(n), 2)
    # maximum amount we can shift from i to j without breaking bounds
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0:
        return h.copy()
    delta = random.random() * max_transfer * scale
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
