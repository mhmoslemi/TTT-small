import time, math, numpy as np

def _c5_score(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    # np.correlate with mode='full' gives linear (zero‑padded) correlation
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h=1, 0≤h≤1,
    minimizing C₅ = max_k ∫ h(x)(1‑h(x+k)) dx.
