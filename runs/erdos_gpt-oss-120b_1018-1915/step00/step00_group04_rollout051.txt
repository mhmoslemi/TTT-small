import time
import numpy as np

def _compute_c5(h):
    """Return C5 for step function values h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair(h, target_sum, max_iter=5):
    """Clip to [0,1] and enforce sum == target_sum (approx)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        if abs(s - target_sum) < 1e-9:
            break
        # adjust proportionally, respecting bounds
        diff = target_sum - s
        # distribute diff equally to non‑boundary entries
        free = (h > 0.0) & (h < 1.0)
        if not free.any():
            # all at bounds: distribute to all then clip again
            h += diff / len(h)
        else:
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def _mutate(parent, sigma):
    child = parent + sigma * np.random.randn(parent.size)
    return child

def _crossover(p1, p2):
    mask = np.random.rand(p1.size) < 0.5
    child = np.where(mask, p1, p2)
    return child
