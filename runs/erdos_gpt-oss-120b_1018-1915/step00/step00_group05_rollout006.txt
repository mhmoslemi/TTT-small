import time
import numpy as np

def _make_initial(n, init_vals=None):
    """Create a binary feasible h with exactly n/2 ones."""
    target_ones = n // 2
    if init_vals is not None:
        h = np.clip(init_vals, 0.0, 1.0)
        # turn into binary by thresholding
        h = (h > 0.5).astype(float)
    else:
        h = np.zeros(n, dtype=float)
        h[:target_ones] = 1.0
    # repair count if needed
    cur = int(h.sum())
    if cur > target_ones:
        # turn some ones to zero
        idx = np.where(h == 1.0)[0]
        np.random.shuffle(idx)
        h[idx[:cur - target_ones]] = 0.0
    elif cur < target_ones:
        idx = np.where(h == 0.0)[0]
        np.random.shuffle(idx)
        h[idx[:target_ones - cur]] = 1.0
    return h

def _c5_value(h, dx):
    """Compute C5 for a binary (or general) step function."""
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C5 bound."""
    np.random.seed(seed)
    start_time = time.time()

    # obtain problem size
    if 'initial_h_values' in globals():
        init_vals = initial_h_values
        n_points = int(init_vals.shape[0])
    else:
        init_vals = None
        n_points = kwargs.get('n_points', 70)

    dx = 2.0 / n_points

    # initial feasible solution
    h_current = _make_initial(n_points, init_vals)
    best_h = h_current.copy()
    best_c5 = _c5_value(best_h, dx)

    # annealing schedule parameters
    max_iter = 200000
    temp_start = 1.0
    temp_end = 1e-3

    for it in range(1, max_iter + 1):
        # time check
        if time.time() - start_time > budget_s:
            break

        # temperature linear decay
        t = temp_start + (temp_end - temp_start) * (it / max_iter)

        # propose a neighbor by swapping a 0 and a 1
        ones = np.where(h_current == 1.0)[0]
        zeros = np.where(h_current == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            # unlikely, but restart
            h_current = _make_initial(n_points, init_vals)
            continue
        i_swap = np.random.choice(ones)
        j_swap = np.random.choice(zeros)

        h_candidate = h_current.copy()
        h_candidate[i_swap] = 0.0
        h_candidate[j_swap] = 1.0

        cand_c5 = _c5_value(h_candidate, dx)

        delta = cand_c5 - best_c5
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = h_candidate.copy()
        # Metropolis acceptance
        if delta < 0 or np.random.rand() < np.exp(-delta / t):
            h_current = h_candidate

        # occasional random restart to escape plateaus
        if it % 20000 == 0:
            h_current = _make_initial(n_points, init_vals)

    return best_h, best_c5, n_points
