def _repair(vec):
    """Project vec onto [0,1] box and sum(vec)=len(vec)/2."""
    v = np.asarray(vec, dtype=float)
    target = len(v) / 2.0
    for _ in range(10):
        np.clip(v, 0.0, 1.0, out=v)
        s = v.sum()
        diff = target - s
        if abs(diff) < 1e-10:
            break
        v += diff / len(v)
    np.clip(v, 0.0, 1.0, out=v)
    # tiny residual – fix a random coordinate if needed
    residual = target - v.sum()
    if abs(residual) > 1e-6:
        idx = np.random.randint(len(v))
        cap = (1.0 - v[idx]) if residual > 0 else v[idx]
        adj = np.clip(residual, -cap, cap)
        v[idx] += adj
    return v
