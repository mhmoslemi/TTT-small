import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project_to_feasible(x):
    """Clip to [0,1] and enforce sum = n/2 by scaling within bounds."""
    n = x.size
    # Clip
    x = np.clip(x, 0.0, 1.0)
    target = n * 0.5
    current = x.sum()
    if np.isclose(current, target):
        return x
    # Simple proportional scaling while keeping within [0,1]
    # iteratively adjust
    for _ in range(10):
        diff = target - current
        # distribute proportionally to distance from bounds
        free = np.where((diff > 0) & (x < 1.0) | (diff < 0) & (x > 0.0))[0]
        if free.size == 0:
            break
        adjust = diff / free.size
        adjust = np.clip(adjust, -np.min(x[free]), np.max(1.0 - x[free]))
        x[free] += adjust
        x = np.clip(x, 0.0, 1.0)
        current = x.sum()
        if np.isclose(current, target, atol=1e-9):
            break
    # final tiny correction
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        # add to first element that can absorb it
        for i in range(n):
            if (residual > 0 and x[i] < 1.0) or (residual < 0 and x[i] > 0.0):
                delta = min(abs(residual), (1.0 - x[i]) if residual > 0 else x[i])
                x[i] += np.sign(residual) * delta
                break
    return x
