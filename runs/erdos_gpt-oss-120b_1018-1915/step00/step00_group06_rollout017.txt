import time
import numpy as np

def _evaluate(h):
    """Return C5 value for a given step function sampled in h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project_box_sum(v, target_sum, max_iter=10, atol=1e-9):
    """
    Project vector v onto the set {0 <= x_i <= 1, sum x_i = target_sum}.
    Simple water‑filling algorithm.
    """
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        cur = x.sum()
        diff = target_sum - cur
        if abs(diff) < atol:
            break
        free = (x > 0.0) & (x < 1.0)
        if not free.any():
            # all components at a bound – cannot satisfy exactly, break
            break
        delta = diff / free.sum()
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    return x

def _upscale(h, new_n):
    """
    Linear interpolation of a periodic step function from len(h) to new_n.
    The resulting vector is projected onto the constraints.
    """
    old_n = h.size
    # periodic coordinates
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return _project_box_sum(h_new, new_n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] minimizing the Erdős C5 overlap.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Initialise starting vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
        h0 = _project_box_sum(h0, n / 2.0)
    else:
        n = 100
        h0 = _project_box_sum(rng.random(n), n / 2.0)

    current_h = h0.copy()
    current_score = _evaluate(current_h)
    best_h = current_h.copy()
    best_score = current_score

    # ------------------------------------------------------------------
    # Main stochastic optimisation loop
    # ------------------------------------------------------------------
    max_iter = 12000
    for it in range(max_iter):
        # time guard
        if time.time() - start_time > budget_s * 0.96:
            break

        # step size schedule
        frac = it / max_iter
        step = 0.2 * (1.0 - frac) + 0.005

        # candidate generation
        cand = current_h + step * rng.standard_normal(n)
        cand = _project_box_sum(cand, n / 2.0)

        cand_score = _evaluate(cand)

        # keep the best ever found
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand.copy()

        # simple greedy acceptance
        if cand_score < current_score:
            current_h = cand
            current_score = cand_score

        # occasional random restart to escape local minima
        if (it % 500 == 0) and (rng.random() < 0.07):
            current_h = _project_box_sum(rng.random(n), n / 2.0)
            current_score = _evaluate(current_h)

        # occasional up‑sampling to allow finer adjustments
        if (it % 3000 == 0) and (n < 300):
            # keep the best found so far, upscale and continue
            new_n = min(2 * n, 300)
            best_h = _upscale(best_h, new_n)
            best_score = _evaluate(best_h)
            current_h = best_h.copy()
            current_score = best_score
            n = new_n

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return (best_h.tolist(), best_score, best_h.size)
