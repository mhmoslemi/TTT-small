import time
import numpy as np
from scipy import optimize

# The harness provides evaluate_erdos_solution and possibly initial_h_values.


def _c5_from_h(h):
    """Compute the C5 value for a discrete h on [0,2]."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)


def _project_to_feasible(h, n):
    """Clamp to [0,1] and enforce the integral constraint sum(h)=n/2."""
    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    current_sum = h.sum()
    if current_sum == 0:
        # avoid division by zero: fill uniformly
        h[:] = target_sum / n
        return h
    # adjust by adding a constant to meet the sum, then re‑clip and repeat a few times
    for _ in range(5):
        diff = target_sum - h.sum()
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # final renormalisation if still off by tiny amount
    h *= target_sum / h.sum()
    return h


def _objective_flat(x, n):
    """Objective for the optimizer (flattened h)."""
    h = _project_to_feasible(x, n)
    return _c5_from_h(h)


def _optimize_start(h0, max_iter, time_deadline):
    """Run a short SLSQP optimisation starting from h0."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {
        "type": "eq",
        "fun": lambda x: x.sum() - n / 2.0,
    }

    # wrapper that respects the time budget

        if time.time() > time_deadline:
            raise TimeoutError

    try:
        res = optimize.minimize(
            fun=lambda x: _c5_from_h(_project_to_feasible(x, n)),
            x0=h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": max_iter, "ftol": 1e-9, "disp": False},
            callback=callback,

            # fallback to projected point
            best = _project_to_feasible(res.x, n)
        else:
            best = _project_to_feasible(res.x, n)
    except TimeoutError:
        # time ran out – return current feasible point
        best = _project_to_feasible(h0, n)

    return best, _c5_from_h(best)


def _upsample(h, new_n):
    """Linearly up‑sample h from its current size to new_n points on [0,2]."""

    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h)
