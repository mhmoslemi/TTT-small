import time
import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    """Return the C5 value for a discrete step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)

def project_h(h):
    """Clip to [0,1] and enforce the integral constraint sum(h)=len(h)/2."""
    n = h.size
    desired = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        # distribute uniformly
        h[:] = desired / n
        return h
    if abs(total - desired) > 1e-12:
        factor = desired / total
        h = h * factor
        h = np.clip(h, 0.0, 1.0)
        # tiny residual correction
        residual = desired - h.sum()
        if abs(residual) > 1e-12:
            # distribute residual greedily
            for i in range(n):
                if residual > 0:
                    add = min(residual, 1.0 - h[i])
                else:
                    add = -min(-residual, h[i])
                h[i] += add
                residual -= add
                if abs(residual) < 1e-12:
                    break
    return h

def upsample(h, new_n):
    """Linear interpolation to a larger grid, then re‑project to satisfy constraints."""
    old_n = h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_h(h_new)

def slsqp_optimize(h0, max_iter, rng):
    """Run a short SLSQP optimization from start h0."""
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}
    # SLSQP may need a feasible start; ensure it
    h0 = project_h(h0)
    res = minimize(compute_c5, h0, method='SLSQP', bounds=bounds,
                   constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    if res.success:
        h_opt = project_h(res.x)
        return h_opt, compute_c5(h_opt)
    else:
        return h0, compute_c5(h0)

def random_move(h, rng):
    """Transfer a random amount of mass between two indices, preserving the sum."""
    n = h.size
    i, j = rng.choice(n, 2, replace=False)
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0:
        return h  # nothing feasible
    eps = rng.uniform(0.0, max_transfer)
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] minimizing the Erdős C5 overlap."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---- initial solution -------------------------------------------------
    if 'initial_h_values' in globals():
        h_cur = np.array(initial_h_values, dtype=np.float64)
    else:
        # uniform 0.5 satisfies the integral constraint for any even length
        n0 = 95
        h_cur = np.full(n0, 0.5, dtype=np.float64)

    h_cur = project_h(h_cur)
    best_h = h_cur.copy()
    best_c = compute_c5(best_h)

    # ---- multi‑resolution loop --------------------------------------------
    max_n = 750  # upper bound on samples
    n = best_h.size
    while True:
        # time guard

        # 1) local SLSQP refinement

            if c_trial < best_c:

        # 3) increase resolution if possible

        n = n * 2
        best_h = upsample(best_h, n)
        best_c = compute_c5(best_h)

    # final check (project one last time for safety)
    best_h = project_h(best_h)
    best_c = compute_c5(best_h)

    return best_h, float(best_c), best_h.size
