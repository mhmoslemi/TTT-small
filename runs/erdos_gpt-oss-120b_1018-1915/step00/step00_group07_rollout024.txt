import time
import numpy as np

# Helpers ---------------------------------------------------------

def project_box_sum(v, target):
    """Project vector v onto [0,1]^n with sum == target."""
    v = np.clip(v, 0.0, 1.0)
    if abs(v.sum() - target) < 1e-12:
        return v
    free = (v > 0.0) & (v < 1.0)
    # iterative redistribution
    while True:
        total = v.sum()
        deficit = target - total
        n_free = free.sum()
        if n_free == 0:
            # cannot satisfy sum exactly; return clipped version
            return v
        delta = deficit / n_free
        v[free] += delta
        # enforce bounds again
        over = v > 1.0
        under = v < 0.0
        if not np.any(over) and not np.any(under):
            break
        v[over] = 1.0
        v[under] = 0.0
        free = (v > 0.0) & (v < 1.0)
    return v


def evaluate_h(h, dx):
    """Return max overlap C5 for a given step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def initialize_population(rng, dim, target_sum, pop_size, init_vec=None):
    """Create a population of feasible vectors."""
    pop = []
    if init_vec is not None:
        pop.append(project_box_sum(np.array(init_vec, dtype=float), target_sum))
    while len(pop) < pop_size:
        rand = rng.random(dim)  # uniform in [0,1)
        proj = project_box_sum(rand, target_sum)
        pop.append(proj)
    return np.stack(pop)


def de_optimize(
    n_points,
    init_vec,
    budget_s,
    max_gen=200,
    F=0.5,
    CR=0.9,
    pop_factor=5,
    seed=42,
):
    rng = np.random.default_rng(seed)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    dim = n_points

    pop_size = max(15, pop_factor * dim)  # ensure enough diversity
    population = initialize_population(rng, dim, target_sum, pop_size, init_vec)
    fitness = np.apply_along_axis(evaluate_h, 1, population, dx)

    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_score = float(fitness[best_idx])

    start_time = time.time()
    gen = 0
    while gen < max_gen and (time.time() - start_time) < budget_s * 0.95:
        for i in range(pop_size):
            # select three distinct others
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            # crossover (binomial)
            cross_mask = rng.random(dim) < CR
            # ensure at least one dimension from mutant
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, dim)] = True
            trial = np.where(cross_mask, mutant, population[i])
            trial = project_box_sum(trial, target_sum)

            trial_score = evaluate_h(trial, dx)

            if trial_score <= fitness[i]:
                population[i] = trial
                fitness[i] = trial_score
                if trial_score < best_score:
                    best_score = float(trial_score)
                    best_h = trial.copy()
        gen += 1

    return best_h, best_score, n_points
