import time
import numpy as np
import scipy.optimize as opt

def _objective(h):
    """Compute C5 for a given step function h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _project_to_feasible(h, desired_sum):
    """Clip to [0,1] and adjust to meet the sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - desired_sum) < 1e-12:
        return h
    # simple scaling then redistribution
    h = h * (desired_sum / cur)
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = desired_sum - cur
    if abs(diff) < 1e-12:
        return h
    # distribute the remaining diff to non‑saturated entries
    free = (h > 0.0) & (h < 1.0)
    if free.any():
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function giving a low C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of length n_points, values in [0,1] and sum = n/2
        c5_bound : float, the computed C5 value for this h
        n_points : int, number of discretisation points
    """
    start_time = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Acquire initial construction
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: uniform 0.5 on a modest grid
        h = np.full(100, 0.5, dtype=np.float64)

    n = len(h)
    desired_sum = n / 2.0
    h = _project_to_feasible(h, desired_sum)

    # ------------------------------------------------------------------
    # First stage: SLSQP continuous optimisation
    # ------------------------------------------------------------------
    cons = ({'type': 'eq', 'fun': lambda x: x.sum() - desired_sum},)
    bounds = [(0.0, 1.0)] * n

    # Give the optimiser a modest iteration budget; stop early if time runs out
    slsqp_options = {'maxiter': 500, 'ftol': 1e-12, 'disp': False}
    res = opt.minimize(
        _objective,
        h,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options=slsqp_options,
    )
    if res.success:
        h = _project_to_feasible(res.x, desired_sum)

    best_h = h.copy()
    best_c = _objective(best_h)

    # ------------------------------------------------------------------
    # Second stage: simple stochastic local search within remaining time
    # ------------------------------------------------------------------
    time_allow = budget_s * 0.90  # keep a safety margin
    while time.time() - start_time < time_allow:
        # propose a random transfer of mass between two coordinates
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = best_h[i], best_h[j]

        # feasible delta range so 0 ≤ hi−δ ≤ 1 and 0 ≤ hj+δ ≤ 1
        lower = max(hi - 1.0, -hj)
        upper = min(hi, 1.0 - hj)
        if lower >= upper:
            continue
        delta = np.random.uniform(lower, upper)

        trial = best_h.copy()
        trial[i] = hi - delta
        trial[j] = hj + delta

        c_trial = _objective(trial)
        if c_trial < best_c - 1e-9:          # accept only strict improvements
            best_h = trial
            best_c = c_trial

    # Final projection (numerical drift can happen)
    best_h = _project_to_feasible(best_h, desired_sum)
    best_c = _objective(best_h)

    return best_h, float(best_c), n
