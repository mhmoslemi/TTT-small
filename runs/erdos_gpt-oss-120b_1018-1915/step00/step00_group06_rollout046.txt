import numpy as np
import time
from scipy.optimize import minimize

def _compute_c5(h):
    """Return the C5 value for a step function sampled in h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)

def _repair(h, desired_sum, eps=1e-9, max_iter=10):
    """Project h onto [0,1] with the linear equality sum(h)=desired_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        total = h.sum()
        diff = desired_sum - total
        if abs(diff) < eps:
            break
        # distribute diff only to interior points
        interior = (h > 0.0) & (h < 1.0)
        cnt = interior.sum()
        if cnt == 0:               # all at bounds – fall back to uniform
            h[:] = desired_sum / h.size
            break
        add = diff / cnt
        h[interior] += add
        h = np.clip(h, 0.0, 1.0)
    # final safety scaling (rare)
    total = h.sum()
    if abs(total - desired_sum) > eps:
        factor = desired_sum / total
        h = np.clip(h * factor, 0.0, 1.0)
        # final tiny adjustment
        total = h.sum()
        if abs(total - desired_sum) > eps:
            h += (desired_sum - total) / h.size
            h = np.clip(h, 0.0, 1.0)
    return h

def _random_initial(n, rng):
    """Generate a feasible random start (mean 0.5)."""
    h = rng.random(n)
    h = _repair(h, n / 2.0)
    return h

def _perturb(h, scale, rng):
    """Create a neighbour by moving mass between two coordinates."""
    n = h.size
    i, j = rng.choice(n, 2, replace=False)
    delta = rng.uniform(-scale, scale)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    h_new = _repair(h_new, n / 2.0)
    return h_new
