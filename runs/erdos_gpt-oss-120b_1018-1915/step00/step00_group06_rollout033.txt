import time
import math
import random
import numpy as np

def evaluate_erdos_solution(h):
    """Return C5 for the given step function h (numpy array)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)

def project_to_feasible(h, target):
    """Clip to [0,1] and adjust to hit the target sum (within tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    # Simple redistribution loop
    max_iter = 10
    eps = 1e-12
    for _ in range(max_iter):
        diff = target - total
        if abs(diff) < eps:
            break
        # indices that can still move in the needed direction
        if diff > 0:
            movable = np.where(h < 1.0 - eps)[0]
            if movable.size == 0:
                break
            inc = np.minimum(diff / movable.size, 1.0 - h[movable])
            h[movable] += inc
            total = h.sum()
        else:
            movable = np.where(h > eps)[0]
            if movable.size == 0:
                break
            dec = np.minimum(-diff / movable.size, h[movable])
            h[movable] -= dec
            total = h.sum()
    # Final tiny correction by uniform scaling (may violate bounds a bit, re‑clip)
    if abs(h.sum() - target) > 1e-8:
        scale = target / h.sum()
        h = np.clip(h * scale, 0.0, 1.0)
        # One more redistribution if needed
        h = project_to_feasible(h, target)
    return h

def random_feasible(n, rng):
    """Generate a random feasible vector of length n."""
    # start with uniform numbers
    h = rng.random(n)
    h = project_to_feasible(h, n / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine initial dimension
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        h = project_to_feasible(h0, n / 2.0)
    else:
        n = 200                     # fallback size
        h = random_feasible(n, rng)

    target_sum = n / 2.0
    best_h = h.copy()
    best_score = evaluate_erdos_solution(best_h)

    # Parameters for the stochastic search
    max_iter = 10_000_000          # very large upper bound; actual stop by time
    temperature = 0.02
    decay = 0.999999                # slow exponential decay

    i = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break

        # propose a move: transfer epsilon from i to j
        i_idx, j_idx = rng.integers(0, n, size=2)
        if i_idx == j_idx:
            continue
        eps = rng.uniform(-0.05, 0.05)   # signed amount
        # apply move
        h_new = best_h.copy()
        h_new[i_idx] -= eps
        h_new[j_idx] += eps
        h_new = project_to_feasible(h_new, target_sum)

        new_score = evaluate_erdos_solution(h_new)

        # Accept if improvement or with Metropolis probability
        if new_score < best_score:
            best_h, best_score = h_new, new_score
        else:
            prob = math.exp(-(new_score - best_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                best_h = h_new
                best_score = new_score

        # temperature schedule
        temperature *= decay

        # occasional random restart to escape stagnation
        if i % 500_000 == 0 and i > 0:
            h_restart = random_feasible(n, rng)
            score_restart = evaluate_erdos_solution(h_restart)
            if score_restart < best_score:
                best_h, best_score = h_restart, score_restart

        i += 1

    # Final projection to guarantee feasibility
    best_h = project_to_feasible(best_h, target_sum)
    best_score = evaluate_erdos_solution(best_h)

    return best_h, float(best_score), int(n)
