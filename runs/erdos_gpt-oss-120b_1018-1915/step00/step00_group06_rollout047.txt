import numpy as np
import time
import random
from math import isfinite

def _compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation (length 2n-1)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _repair(h, target_sum):
    """Clip to [0,1] and renormalize to achieve exact target sum."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if total == 0:
        # distribute uniformly
        h[:] = target_sum / h.size
        return h
    # scale to hit target sum, then clip again and iterate a couple times
    for _ in range(3):
        factor = target_sum / total
        h *= factor
        h = np.clip(h, 0.0, 1.0)
        total = h.sum()
    # final tiny adjustment if needed
    diff = target_sum - total
    if abs(diff) > 1e-12:
        # distribute diff uniformly respecting bounds
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            inc = diff / free.sum()
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
    # ensure sum very close
    h *= target_sum / h.sum()
    return h

def _interpolate_parent(parent, new_n):
    """Linear interpolation of parent vector to new grid size."""
    old_n = parent.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)

    interp = np.interp(new_x, old_x, parent)
    # enforce exact integral
    target = new_n / 2.0
    return _repair(interp, target)
