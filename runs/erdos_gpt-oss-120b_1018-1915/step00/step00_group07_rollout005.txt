import numpy as np
import time
import math

# Helper to compute the C5 value for a given step function h.
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise h
    # ------------------------------------------------------------------
    try:
        h = np.array(initial_h_values, copy=True)  # type: ignore[name-defined]
        n = h.shape[0]
    except Exception:
        # fall back to a modest size if parent not provided
        n = 80
        h = np.full(n, 0.5, dtype=float)

    # Ensure feasibility: 0≤h≤1 and exact integral constraint
    dx = 2.0 / n
    target_sum = n / 2.0  # because sum(h)*dx = 1
    # Clip to [0,1] first
    h = np.clip(h, 0.0, 1.0)
    # Adjust total mass if necessary by adding/subtracting a tiny amount
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # distribute the discrepancy uniformly (will stay in bounds because
        # discrepancy is tiny for a reasonable start)
        h += (target_sum - cur_sum) / n
        h = np.clip(h, 0.0, 1.0)

    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # 2. Optimisation parameters
    # ------------------------------------------------------------------
    max_iters = 200000  # generous upper bound; loop will stop by time
    T0 = 0.5            # initial temperature
    Tmin = 1e-4

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (simulated annealing with pair swaps)
    # ------------------------------------------------------------------
    for it in range(max_iters):
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        while j == i:
            j = np.random.randint(0, n)

        hi, hj = h[i], h[j]

        # feasible delta interval so that both stay in [0,1]
        delta_low = max(-hi, hj - 1.0)
        delta_high = min(1.0 - hi, hj)

        if delta_low >= delta_high:
            continue  # no feasible move

        delta = np.random.uniform(delta_low, delta_high)

        # propose new h
        h[i] = hi + delta
        h[j] = hj - delta

        new_c5 = compute_c5(h)

        # temperature schedule (linear cooling)
        elapsed = time.time() - start_time
        T = max(T0 * (1.0 - elapsed / budget_s), Tmin)

        # acceptance rule
        accept = False
        if new_c5 <= cur_c5:
            accept = True
        else:
            prob = math.exp(-(new_c5 - cur_c5) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert move
            h[i] = hi
            h[j] = hj

    # ------------------------------------------------------------------
    # 4. Return result
    # ------------------------------------------------------------------
    # Ensure the returned vector satisfies the exact sum constraint (possible
    # tiny drift due to floating point).

    best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n)
