import time

def _evaluate(h, dx):
    """Return C5 bound for discrete step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _project(h, target_sum):
    """Project h onto [0,1] with exact sum = target_sum."""
    # Clip to bounds
    h = np.clip(h, 0.0, 1.0)
    # Iteratively adjust the mean while staying in bounds
    for _ in range(5):
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        # shift all non‑boundary entries uniformly
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            # fallback: scale whole vector then clip again
            h = h * (target_sum / cur_sum)
            h = np.clip(h, 0.0, 1.0)
            continue
        delta = diff / free.sum()
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # Final fix with simple scaling if needed
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        h = h * (target_sum / cur_sum)
        h = np.clip(h, 0.0, 1.0)
    return h
