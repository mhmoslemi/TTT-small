import time
import random
import numpy as np

def _project_to_feasible(h):
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # simple redistribution of the surplus/deficit
    # first try uniform shift, then clamp and redistribute remainder
    h += -diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # fix any leftover by adjusting the first free element
    for i in range(n):
        if diff > 0 and h[i] > 0.0:
            take = min(diff, h[i])
            h[i] -= take
            diff -= take
        elif diff < 0 and h[i] < 1.0:
            take = min(-diff, 1.0 - h[i])
            h[i] += take
            diff += take
        if abs(diff) < 1e-12:
            break
    return h

def _c5_score(h):
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)
