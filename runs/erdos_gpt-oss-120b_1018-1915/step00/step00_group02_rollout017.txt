import time
import numpy as np
from scipy.optimize import minimize

# helper: ensure sum constraint exactly by simple projection
def enforce_sum_and_bounds(x, target):
    x = np.clip(x, 0.0, 1.0)
    # simple iterative shift to meet sum
    for _ in range(10):
        cur = x.sum()
        if abs(cur - target) < 1e-12:
            break
        x += (target - cur) / x.size
        x = np.clip(x, 0.0, 1.0)
    return x

def evaluate(h):
    # pre‑imported in the harness
    return evaluate_erdos_solution(h)

def local_optimize(h_start, time_limit):
    n = h_start.size
    target = n / 2.0
    best_h = h_start.copy()
    best_val = evaluate(best_h)
    start = time.time()

    # SLSQP options
    opts = {"maxiter": 200, "ftol": 1e-9, "disp": False}

    while time.time() - start < time_limit:
        # random perturbation
        pert = np.random.randn(n) * 0.07
        h0 = best_h + pert
        h0 = enforce_sum_and_bounds(h0, target)

        cons = ({
            "type": "eq",
            "fun": lambda x: x.sum() - target,
        })
        bounds = [(0.0, 1.0)] * n

        res = minimize(
            evaluate,
            h0,
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options=opts,
        )

        if not res.success:
            continue

        val = evaluate(res.x)
        if val < best_val:
            best_val, best_h = val, res.x.copy()

    return best_h, best_val

def upscale(h):
    n = h.size
    new_n = n * 2
    # linear interpolation onto doubled grid
    x_old = np.linspace(0, 1, n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # enforce sum constraint for new size
    target = new_n / 2.0
    h_new = enforce_sum_and_bounds(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_left = lambda: max(0.0, budget_s - (time.time() - start_time))

    # initial construction
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: uniform 0.5 values (satisfies sum automatically)
        n0 = 45
        h0 = np.full(n0, 0.5)

    h0 = enforce_sum_and_bounds(h0, h0.size / 2.0)

    # start with the base resolution
    best_h, best_val = local_optimize(h0, time_left() * 0.4)

    # progressive resolution increase
    max_n = 360  # keep size moderate for speed
    while time_left() > 5.0 and best_h.size < max_n:
        # allocate a fraction of remaining time to this level
        level_time = time_left() * 0.3
        # generate finer grid from current best
        h_fine = upscale(best_h)
        # local search on finer grid

        if cand_val < best_val:

    # final safeguard: one more quick refinement at current resolution

    # ensure final vector respects constraints exactly
    best_h = enforce_sum_and_bounds(best_h, best_h.size / 2.0)
