import numpy as np
import time
import math
import random

# the harness provides evaluate_erdos_solution, and possibly initial_h_values
# we only need to compute C5 ourselves for internal decisions

def compute_c5(h, dx):
    """Compute C5 = max_k ∫ h(x)*(1-h(x+k)) dx via discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_two_indices(h, i, j, delta):
    """
    Add delta to h[i] and subtract delta from h[j] while keeping the vector
    within [0,1]. Returns the new vector and a boolean indicating if the move
    was feasible.
    """
    hi, hj = h[i], h[j]
    # compute feasible range for delta
    lo = -hi          # cannot go below 0 at i
    hi_max = 1.0 - hi # cannot exceed 1 at i
    lo_j = - (1.0 - hj)  # cannot exceed 1 at j (since we subtract)
    hi_j = hj           # cannot go below 0 at j
    delta_min = max(lo, -hi_j)
    delta_max = min(hi_max, -lo_j)
    if delta < delta_min:
        delta = delta_min
    if delta > delta_max:
        delta = delta_max
    if delta == 0.0:
        return h, False
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new, True
