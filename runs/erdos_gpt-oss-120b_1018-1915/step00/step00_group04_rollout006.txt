import time
import numpy as np

def _objective(h):
    """Return C5 for step function h (array of length n)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair(h):
    """Project h into feasible set: 0<=h<=1 and sum(h)=n/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)

    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h

    # First, shift to correct mean if possible
    mean_shift = (target / n) - h.mean()
    h += mean_shift
    h = np.clip(h, 0.0, 1.0)

    # If still off, redistribute the remaining error on non‑boundary entries
    cur = h.sum()
    diff = cur - target
    if abs(diff) > 1e-9:
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            adjust = diff / mask.sum()
            h[mask] -= adjust
            h = np.clip(h, 0.0, 1.0)

    # Final tiny fix on a single entry
    cur = h.sum()
    if abs(cur - target) > 1e-9:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += (target - cur)
        h = np.clip(h, 0.0, 1.0)

    return h

def _mutate(parent, sigma=0.08):
    """Gaussian mutation with repair."""
    child = parent.astype(float)
    noise = np.random.normal(0.0, sigma, size=parent.size)
    child += noise
    return _repair(child)

def _upsample(h, new_n):
    """Linear interpolation of h from its current length to new_n points."""
    old_n = h.size
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 overlap bound.

    Returns
    -------
    h_values : numpy.ndarray
        Optimized values of h sampled uniformly (length n_points).
    c5_bound : float
        Corresponding C5 upper bound.
    n_points : int
        Number of sample points used (grid size).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----- Determine grid size ------------------------------------------------
    # target size: 336 points (≈0.006 resolution, well below 1k limit)
    target_n = 336

    # ----- Initialise first solution -----------------------------------------
    if 'initial_h_values' in globals():
        init_h = np.asarray(initial_h_values, dtype=float)
        init_h = _upsample(init_h, target_n)
    else:
        # random feasible start
        init_h = np.random.uniform(0.0, 1.0, size=target_n)
        init_h = _repair(init_h)

    best_h = init_h
    best_score = _objective(best_h)

    # ----- Simple population (size 12) ---------------------------------------
    pop_size = 12
    population = [best_h]
    while len(population) < pop_size:
        cand = np.random.uniform(0.0, 1.0, size=target_n)
        cand = _repair(cand)
        population.append(cand)

    # ----- Main optimisation loop ---------------------------------------------
    # allocate roughly half of budget to optimisation; rest for safety
    max_iters = 200000
    iter_count = 0
    while time.time() - start_time < budget_s * 0.95 and iter_count < max_iters:
        iter_count += 1

        # select a parent (biased towards better individuals)
        scores = np.array([_objective(ind) for ind in population])
        probs = np.exp(-scores)  # lower score => higher prob
        probs /= probs.sum()
        parent_idx = np.random.choice(len(population), p=probs)
        parent = population[parent_idx]

        # produce offspring
        child = _mutate(parent, sigma=0.07)

        child_score = _objective(child)

        # replace worst if child is better
        worst_idx = np.argmax(scores)
        if child_score < scores[worst_idx]:
            population[worst_idx] = child
            scores[worst_idx] = child_score

        # update global best
        if child_score < best_score:
            best_h = child
            best_score = child_score

    # final sanity repair (ensures exact integral)
    best_h = _repair(best_h)
    best_score = _objective(best_h)

    return best_h, best_score, target_n
