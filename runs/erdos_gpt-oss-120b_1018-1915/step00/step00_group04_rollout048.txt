import time
import numpy as np

# Helper: compute the C5 objective for a given step function h
def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

# Helper: project vector onto the box [0,1] with a fixed sum = target
def repair(h, target):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    # Simple iterative redistribution
    eps = 1e-12
    if abs(total - target) <= eps:
        return h
    if total > target:  # need to subtract excess
        excess = total - target
        while excess > eps and np.any(h > 0):
            idx = h > 0
            amount = excess / np.count_nonzero(idx)
            h[idx] -= amount
            # clamp negatives and recompute excess
            neg = h < 0
            if np.any(neg):
                excess = -h[neg].sum()
                h[neg] = 0.0
            else:
                excess = 0.0
    else:               # need to add deficit
        deficit = target - total
        while deficit > eps and np.any(h < 1):
            idx = h < 1
            amount = deficit / np.count_nonzero(idx)
            h[idx] += amount
            over = h > 1
            if np.any(over):
                deficit = h[over].sum() - np.count_nonzero(over)
                h[over] = 1.0
            else:
                deficit = 0.0
    return h

# Helper: produce a random neighbour of h
def neighbour(h, scale=0.2):
    n = h.size
    new = h.copy()
    # perturb a modest fraction of entries
    k = max(1, n // 15)
    idx = np.random.choice(n, k, replace=False)
    noise = np.random.normal(0.0, scale, size=k) * (h.max() - h.min() + 1e-9)
    new[idx] += noise
    return new

# Main entry point required by the harness
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    # ------------------------------------------------------------------
    # Choose starting resolution and initial h
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 84                        # fallback size (even)
        h = np.zeros(n, dtype=float)
        # start with half ones, half zeros (any order)
        h[: n // 2] = 1.0
        np.random.shuffle(h)
    # enforce feasibility
    target_sum = n / 2.0
    h = repair(h, target_sum)
    best_h = h.copy()
    best_score = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Main optimisation loop (hill‑climbing with random restarts)
    while time.time() - t_start < 0.85 * budget_s:
        cand = neighbour(best_h, scale=0.15)
        cand = repair(cand, target_sum)
        score = compute_c5(cand)
        if score < best_score:
            best_h, best_score = cand, score

        # occasional random restart to escape plateaus
        if np.random.rand() < 0.005:
            rnd = np.random.rand(n)
            rnd = repair(rnd, target_sum)
            if compute_c5(rnd) < best_score:
                best_h, best_score = rnd, compute_c5(rnd)

        # time‑based promotion to higher resolution
        elapsed = time.time() - t_start
        if elapsed > 0.55 * budget_s and n < 672:
            # double resolution via linear interpolation
            new_n = min(n * 2, 672)
