import numpy as np
import time
from math import inf

def _project_feasible(vec, target_sum, eps=1e-12, max_iter=20):
    """Project a vector onto the box [0,1] and the affine sum constraint."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = v.sum()
        diff = target_sum - cur_sum
        if abs(diff) < eps:
            break
        # indices that can move without hitting a bound
        free = (v > 0.0) & (v < 1.0)
        n_free = free.sum()
        if n_free == 0:
            # distribute proportionally to distance from bounds
            caps_up = 1.0 - v
            caps_down = v
            caps = caps_up + caps_down
            prob = caps / caps.sum()
            delta = diff * prob
            v = np.clip(v + delta, 0.0, 1.0)
        else:
            delta = diff / n_free
            v[free] += delta
            v = np.clip(v, 0.0, 1.0)
    # final tiny correction by uniform scaling (preserves bounds)
    cur_sum = v.sum()
    if abs(cur_sum - target_sum) > eps:
        scale = target_sum / cur_sum if cur_sum != 0 else 0.0
        v = np.clip(v * scale, 0.0, 1.0)
    return v

def _c5_value(h):
    """Return the C5 bound for a discrete step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
