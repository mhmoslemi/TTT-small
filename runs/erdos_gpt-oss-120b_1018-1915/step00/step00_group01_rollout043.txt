import time
import numpy as np

def compute_c5(h):
    """Return C_5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, target_sum):
    """Clamp to [0,1] and adjust to hit the exact sum (within tiny tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    # simple redistribution loop
    for _ in range(10):
        residual = target_sum - h.sum()
        if abs(residual) < 1e-12:
            break
        # find an index that can absorb the residual
        if residual > 0:
            # need to increase some entry
            caps = 1.0 - h
            if caps.max() == 0:
                break
            idx = np.argmax(caps)
            add = min(residual, caps[idx])
            h[idx] += add
        else:
            # need to decrease some entry
            caps = h
            if caps.max() == 0:
                break
            idx = np.argmax(caps)
            sub = min(-residual, caps[idx])
            h[idx] -= sub
    return h

def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

    # resolution (keep below 1000 as requested)
    n = 200
    dx = 2.0 / n
    target_sum = n / 2  # because integral must be 1

    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():

        init = np.asarray(initial_h_values, dtype=float)
        x_old = np.linspace(0.0, 2.0, init.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        h = np.full(n, 0.5)  # uniform feasible start

    h = project_to_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ----- simulated annealing ----------------------------------------
    total_time = budget_s * 0.95  # keep a safety margin
    while time.time() - start_time < total_time:

        frac = elapsed / total_time

        # propose a move: transfer mass from i to j
