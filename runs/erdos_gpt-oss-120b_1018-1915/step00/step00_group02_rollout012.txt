import time
import numpy as np
from scipy.optimize import minimize

# Helper: compute the C5 bound for a discrete h
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr.max()

# Project a vector onto the feasible set {0<=h<=1, sum(h)=n/2}
def repair(h):
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # simple iterative redistribution
    for _ in range(10):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        adjust = diff / mask.sum()
        h[mask] += adjust
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction by scaling (won't violate bounds much)
    cur = h.sum()
    if cur != 0:
        h *= target / cur
        h = np.clip(h, 0.0, 1.0)
    # ensure exact sum by a last tiny tweak on a free coordinate
    cur = h.sum()
    if abs(cur - target) > 1e-9:
        free = np.where((h > 0.0) & (h < 1.0))[0]
        if free.size:
            idx = free[0]
            h[idx] += target - cur
            h[idx] = np.clip(h[idx], 0.0, 1.0)
    return h

# Upsample a vector to a new grid via linear interpolation
def upscale(h, new_n):
    old_n = h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    h_new = repair(h_new)
    return h_new

# Generate a random feasible vector of length n
def random_feasible(n, rng):
    h = rng.random(n)
    h = repair(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine base size
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values, dtype=float)
        base_n = base_h.shape[0]
    else:
        base_n = 45
        base_h = np.full(base_n, 0.0)
        # simple alternating pattern respecting integral
        half = base_n // 2
        base_h[:half] = 1.0
        base_h = repair(base_h)

    best_h = base_h
    best_score = compute_c5(best_h)

    # List of target resolutions (increase gradually but stay modest)
    resolutions = [base_n]
    while resolutions[-1] < 800:
        resolutions.append(resolutions[-1] * 2)

    for n in resolutions:
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break

        # Prepare starting points
        starts = []

        # Upscaled previous best
        if n != base_n:
            starts.append(upscale(best_h, n))

        # Random starts
        for _ in range(4):
            starts.append(random_feasible(n, rng))

        bounds = [(0.0, 1.0)] * n
        constraints = ({
            'type': 'eq',
            'fun': lambda h, target=n/2.0: h.sum() - target
        },)

        for x0 in starts:
            if time.time() - start_time > budget_s * 0.95:
                break
            res = minimize(

                method='SLSQP',

            score = compute_c5(h_candidate)
            if score < best_score:
                best_score = score
                best_h = h_candidate
