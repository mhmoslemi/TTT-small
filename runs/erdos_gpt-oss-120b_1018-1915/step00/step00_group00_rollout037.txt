import time
import numpy as np

def _compute_c5(h, dx):
    # correlation of h with (1-h); mode='full' returns length 2*n-1
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _initial_h(n):
    # generate a feasible random start with sum = n/2 and values in [0,1]
    h = np.random.rand(n)
    h = h / np.sum(h) * (n / 2.0)
    for _ in range(10):
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if s == 0:
            h = np.full(n, 0.5)
            break
        h = h * (n / 2.0) / s
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    if 'initial_h_values' in globals():

        n = h_cur.size
        # Ensure exact sum constraint; tiny adjustment if needed
        diff = (n / 2.0) - np.sum(h_cur)
        if abs(diff) > 1e-12:
            h_cur += diff / n
            h_cur = np.clip(h_cur, 0.0, 1.0)
            # re‑balance if clipping disturbed the sum
            s = np.sum(h_cur)
            if s != 0:
                h_cur *= (n / 2.0) / s
    else:
        n = 45
        h_cur = _initial_h(n)

    dx = 2.0 / n
    cur_c5 = _compute_c5(h_cur, dx)
    best_h = h_cur.copy()
    best_c5 = cur_c5

    max_step = 0.2          # maximum mass moved in one mutation
    max_T = 0.03            # initial temperature
    min_T = 1e-5            # final temperature

    # Main simulated‑annealing loop
    while time.time() - start_time < budget_s * 0.94:
        elapsed = time.time() - start_time
        # Linear cooling schedule
        T = max_T * (1.0 - elapsed / (budget_s * 0.94))
        T = max(T, min_T)

        # Propose a move: transfer delta from i to j
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        hi, hj = h_cur[i], h_cur[j]
        delta_max = min(max_step, hi, 1.0 - hj)
        if delta_max <= 0.0:
            continue
        delta = np.random.rand() * delta_max

        h_new = h_cur.copy()
        h_new[i] = hi - delta
        h_new[j] = hj + delta

        new_c5 = _compute_c5(h_new, dx)
        deltaE = new_c5 - cur_c5

        # Accept if better or with Boltzmann probability
        if deltaE < 0.0 or np.random.rand() < np.exp(-deltaE / (T + 1e-12)):

        # Track the best solution seen
        if new_c5 < best_c5:
            best_c5 = new_c5

            cur_c5 = _compute_c5(h_cur, dx)
