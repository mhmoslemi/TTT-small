def project_bounds_sum(h, total):
    # ensure 0<=h_i<=1 and sum = total
    n = len(h)
    # first clip to [0,1]
    h = np.clip(h, 0, 1)
    # water-filling: if sum > total, reduce proportionally among those >0
    # if sum < total, increase proportionally among those <1
    # iterate few times
    for _ in range(10):
        cur = h.sum()
        if abs(cur - total) < 1e-12:
            break
        diff = total - cur
        # identify free indices that can move in direction of diff
        if diff > 0:
            free = h < 1 - 1e-12
        else:
            free = h > 1e-12
        if not np.any(free):
            break
        # distribute diff equally among free
        delta = diff / np.sum(free)
        h[free] += delta
        h = np.clip(h, 0, 1)
    return h

import time
import numpy as np

def _project_bounds_sum(vec, total):
    """Clip vec to [0,1] and adjust to have exact sum `total` (water‑filling)."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(20):
        cur = v.sum()
        diff = total - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0
        else:
            free = v > 0.0
        if not np.any(free):
            break
        delta = diff / free.sum()
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    return v

def _c5_score(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = len(h)
    dx = 2.0 / n
    # full correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _propose_move(h, step):
    """Randomly transfer a small amount between two indices while respecting bounds."""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    max_sub = min(step, h[i])
    max_add = min(step, 1.0 - h[j])
    delta = np.random.uniform(0.0, min(max_sub, max_add))
    if delta <= 0.0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

        Corresponding value of C5.
    n_points : int
        Number of samples (grid size).
    """
    start_time = time.time()
    np.random.seed(seed)

    n = len(h0)

    best_h = cur_h.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
