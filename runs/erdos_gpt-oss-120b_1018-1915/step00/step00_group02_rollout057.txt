import time
import numpy as np

def _compute_c5(h):
    """Return max overlap integral for step function samples h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _project(h, target_sum):
    """Clamp to [0,1] and adjust to hit exact sum target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        delta = (target_sum - h.sum()) / h.size
        h += delta
        h = np.clip(h, 0.0, 1.0)
    # final linear scaling of interior values if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-12:
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            interior_sum = h[mask].sum()
            exterior_sum = h[~mask].sum()
            scale = (target_sum - exterior_sum) / interior_sum
            h[mask] *= scale
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initial construction
    # ------------------------------------------------------------------
    try:
        h = np.asarray(initial_h_values, dtype=float)  # provided by harness
    except NameError:
        # fallback: uniform 0.5 satisfies integral exactly
        h = None

    if h is None:
        n0 = 45
        h = np.full(n0, 0.5, dtype=float)
    else:
        n0 = h.shape[0]

    # ensure feasibility
    target_sum = n0 / 2.0
    h = _project(h, target_sum)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # Multi‑resolution stochastic descent
    # ------------------------------------------------------------------
    n = n0
    while True:
        # time guard
        if time.time() - start > budget_s * 0.95:
            break
        # parameters adapting to size
        steps = int(1500 * (n / 45.0))
        noise_scale = 0.15 / np.sqrt(n)  # smaller noise for larger n
        target_sum = n / 2.0

        for _ in range(steps):
            if time.time() - start > budget_s * 0.95:
                break
            # random perturbation of a subset
            h_candidate = best_h.copy()
            k = max(1, int(0.12 * n))          # number of altered entries
            idx = rng.choice(n, size=k, replace=False)
            h_candidate[idx] += rng.normal(scale=noise_scale, size=k)
            h_candidate = _project(h_candidate, target_sum)

            c5 = _compute_c5(h_candidate)
            if c5 < best_c5:
                best_h, best_c5 = h_candidate, c5

        # ------------------------------------------------------------------
        # Try to increase resolution if still time and not too large
        # ------------------------------------------------------------------
        if n * 2 > 2000:
            break
        if time.time() - start > budget_s * 0.9:
            break

        # double resolution by simple repetition (preserves integral)
        h_high = np.repeat(best_h, 2)
        n = h_high.shape[0]
        # recompute C5 at new resolution (no projection needed)
        c5_high = _compute_c5(h_high)

        # keep the finer version if it is not worse
        if c5_high <= best_c5:
            best_h, best_c5 = h_high, c5_high
        else:
            # still move to finer grid to give optimizer more freedom
            best_h = h_high
            best_c5 = c5_high
        # loop continues with larger n

    return best_h, best_c5, best_h.shape[0]
