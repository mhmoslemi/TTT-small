import numpy as np
import time
import random

def _evaluate(h: np.ndarray) -> float:
    """Return C5 for a discrete step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def _make_feasible(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 exactly."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # indices that can still move a little
    movable = np.where((h > 0.0) & (h < 1.0))[0]
    if movable.size == 0:
        movable = np.arange(n)
    inc = diff / movable.size
    h[movable] += inc
    h = np.clip(h, 0.0, 1.0)
    # Fix any tiny residual by adjusting the first element
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        h[0] = np.clip(h[0] + residual, 0.0, 1.0)
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.

    Returns
    -------
    h_values : list[float]
        The optimized sample values.
    c5_bound : float
        The achieved C5 upper bound.
    n_points : int
        Number of discretisation points used.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ---- initialise --------------------------------------------------------
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        n0 = parent.size
        # choose a higher resolution for more flexibility
        n = max(200, n0)
        if n != n0:
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, parent)
        else:
            h = parent.copy()
    else:
        n = 200
        h = np.full(n, 0.5, dtype=float)   # integral = 1 automatically

    h = _make_feasible(h)

    best_h = h.copy()
    best_score = _evaluate(best_h)

    # ---- optimisation loop -------------------------------------------------
    start_time = time.time()
    max_time = budget_s * 0.95          # leave margin for cleanup
    step = 0.02                         # typical magnitude of a transfer
    no_improve = 0

    while time.time() - start_time < max_time:
        # propose a transfer of mass delta from j to i
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() * 2.0 - 1.0) * step

        # enforce bounds before constructing the candidate
        if not (0.0 <= h[i] + delta <= 1.0):
            continue
        if not (0.0 <= h[j] - delta <= 1.0):
            continue

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
