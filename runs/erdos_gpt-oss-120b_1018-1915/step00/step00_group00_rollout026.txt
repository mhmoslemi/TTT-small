import time
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _mutate(h, rng):
    """Transfer a small amount of mass between two positions, preserving sum."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return h
    # decide transfer direction randomly
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_transfer = min(h[src], 1.0 - h[dst])
    if max_transfer <= 0.0:
        return h
    # transfer up to half of the allowed amount
    delta = rng.random() * max_transfer * 0.5
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős C5 constant for a step function on [0,2].

    Returns:
        h_values (np.ndarray): optimized values in [0,1] with sum = n_points/2
        c5_bound (float): corresponding C5 upper bound
        n_points (int): number of discretization points used
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initial construction
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)  # provided by harness
        n_points = h0.size
        # Ensure exact sum; small numeric drift is corrected by scaling the
        # deviation uniformly while staying inside [0,1].
        target_sum = n_points / 2.0
        diff = target_sum - h0.sum()
        if abs(diff) > 1e-12:
            # distribute diff proportionally, clip to [0,1], then renormalize
            h0 += diff / n_points
            np.clip(h0, 0.0, 1.0, out=h0)
            # final exact fix by adjusting a random index
            cur = h0.sum()
            correction = target_sum - cur
            idx = rng.integers(0, n_points)
            h0[idx] = np.clip(h0[idx] + correction, 0.0, 1.0)
    except NameError:
        # No supplied parent – start from a simple half‑ones pattern.
        n_points = 90  # even, modest size
        h0 = np.zeros(n_points, dtype=float)
        half = n_points // 2
        h0[:half] = 1.0
        rng.shuffle(h0)

    dx = 2.0 / n_points

    # ------------------------------------------------------------------
    # Simulated annealing / hill climbing
    # ------------------------------------------------------------------
    best = h0.copy()
    best_score = _compute_c5(best, dx)

    current = best.copy()
    current_score = best_score

    iteration = 0
    # Simple linear cooling: temperature proportional to remaining time fraction
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.05:
            break
        # temperature falls from 0.1 to 1e-4 linearly
        T = 0.1 * (1.0 - elapsed / budget_s) + 1e-4

        candidate = _mutate(current, rng)
        cand_score = _compute_c5(candidate, dx)

        # Accept if better, else with Metropolis probability
        if cand_score < current_score or rng.random() < np.exp((current_score - cand_score) / T):
            current, current_score = candidate, cand_score

        # Record global best
        if cand_score < best_score:
            best, best_score = candidate, cand_score

        iteration += 1
        # occasional random restart to avoid stagnation
        if iteration % 20000 == 0:
            if rng.random() < 0.2:
                current = best.copy()
                current_score = best_score

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return best, best_score, n_points
