import time
import numpy as np
from scipy.optimize import minimize

def _project_to_constraints(v, target):
    """Clip to [0,1] and adjust to exact sum `target`."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if s == 0:
        v[:] = target / len(v)
    else:
        v = v * (target / s)
    v = np.clip(v, 0.0, 1.0)
    diff = v.sum() - target
    if abs(diff) > 1e-12:
        # simple walk to fix the remaining error
        for i in range(len(v)):
            if diff > 0 and v[i] > 0:
                take = min(diff, v[i])
                v[i] -= take
                diff -= take
            elif diff < 0 and v[i] < 1:
                give = min(-diff, 1.0 - v[i])
                v[i] += give
                diff += give
            if abs(diff) < 1e-12:
                break
    return v

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and initial vector
    # ------------------------------------------------------------------
    if 'n_points' in kwargs:
        n_points = int(kwargs['n_points'])
    else:
        # use length of supplied parent if it exists, else default 70
        try:
            n_points = len(initial_h_values)  # type: ignore[name-defined]
        except Exception:
            n_points = 70

    dx = 2.0 / n_points
    sum_target = n_points / 2.0

    # initial vector
    try:
        h0 = np.array(initial_h_values, dtype=float)  # type: ignore[name-defined]
        if len(h0) != n_points:
            # resample parent to required size via linear interpolation
            xs = np.linspace(0, 1, len(h0), endpoint=False)
            xs_new = np.linspace(0, 1, n_points, endpoint=False)
            h0 = np.interp(xs_new, xs, h0)
    except Exception:
        h0 = rng.random(n_points)
    h0 = _project_to_constraints(h0, sum_target)

    # ------------------------------------------------------------------
    # Objective and constraints
    # ------------------------------------------------------------------
    def objective(h):
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        return np.max(corr) * dx

    eq_constraint = {'type': 'eq',
                     'fun': lambda x: np.sum(x) - sum_target}

    bounds = [(0.0, 1.0)] * n_points

    # ------------------------------------------------------------------
    # Local optimization helper
    # ------------------------------------------------------------------
    def local_optimize(x0, max_iter=300):
        res = minimize(objective, x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=eq_constraint,
                       options={'maxiter': max_iter, 'ftol': 1e-10, 'disp': False})
        return _project_to_constraints(res.x, sum_target)

    # ------------------------------------------------------------------
    # First (parent) refinement
    # ------------------------------------------------------------------
    best_h = local_optimize(h0, max_iter=500)
    best_val = objective(best_h)

    # ------------------------------------------------------------------
    # Iterative random‑restart local search while time permits
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.9:
        # small perturbation of the current best
        perturb = rng.normal(scale=0.05, size=n_points)
        cand = _project_to_constraints(best_h + perturb, sum_target)
        cand_opt = local_optimize(cand, max_iter=200)

    return best_h, best_val, n_points
