import time
import numpy as np

def _project_feasible(h):
    """Clip to [0,1] and enforce sum(h)=len(h)/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size == 0:
            # nothing free, distribute uniformly and re‑clip
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
            continue
        add = diff / free.size
        h[free] += add
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argmin(h)  # safe place to adjust
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _c5_value(h):
    """Compute the C5 objective for a discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    # ------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()

    # Determine problem size
    try:
        init = initial_h_values  # provided by harness, may be None
        n = len(init)
        h = np.array(init, dtype=float)
    except NameError:
        n = 70
        h = np.random.rand(n)

    # Make the initial vector feasible
    h = _project_feasible(h)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ------------------------------------------------------------
    # Simulated annealing / hill‑climbing loop
    # ------------------------------------------------------------
    target_sum = n / 2.0
    dx = 2.0 / n
    max_elapsed = budget_s * 0.95   # safety margin
    T0 = 0.02                       # initial temperature
    step0 = 0.05                    # initial transfer magnitude
    iteration = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_elapsed:
            break
        iteration += 1

        # temperature and step size decay
        progress = elapsed / max_elapsed
        T = T0 * (1.0 - progress)
        step = step0 * (1.0 - progress)

        # pick a donor (h>0) and a receiver (h<1)
        donors = np.where(h > 1e-8)[0]
        receivers = np.where(h < 1.0 - 1e-8)[0]
        if donors.size == 0 or receivers.size == 0:
            continue
        i = np.random.choice(donors)
        j = np.random.choice(receivers)
        if i == j:
            continue

        # maximum transferable amount respecting bounds
        max_transfer = min(step, h[i], 1.0 - h[j])
        if max_transfer <= 0:
            continue
        delta = np.random.rand() * max_transfer

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        # evaluate
        new_c5 = _c5_value(h_new)

        # acceptance test
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        if new_c5 <= _c5_value(h) or np.exp(-(new_c5 - _c5_value(h)) / max(T, 1e-12)) > np.random.rand():
            h = h_new

    # Final projection (numerical drift)
    best_h = _project_feasible(best_h)

    return best_h.tolist(), float(best_c5), n
