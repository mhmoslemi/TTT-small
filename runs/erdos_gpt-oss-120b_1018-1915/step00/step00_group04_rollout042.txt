import time
import numpy as np

def _project_h(h, target_sum, max_iter=10):
    """Project h onto [0,1] box and sum==target_sum by simple iterative clipping."""
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - target_sum) < 1e-12:
            break
        # distribute the difference uniformly
        h += (target_sum - cur) / h.size
    # final clip
    return np.clip(h, 0.0, 1.0)

def _make_initial_h(n_points):
    """Create a sinusoidal initial vector with exact integral 1."""
    xs = np.linspace(0.0, 2.0, n_points, endpoint=False)
    h = (1.0 + np.sin(np.pi * xs)) / 2.0          # values in [0,1], avg 0.5
    target_sum = n_points / 2.0
    h = _project_h(h, target_sum)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function h (as a NumPy 1‑D array), its C5 bound, and the number of points.
    The algorithm builds a sinusoidal construction and, if time permits,

    start_time = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals():
        base_n = int(initial_h_values.shape[0])
        n_points = max(84, base_n * 2)          # at least ~80 points for decent resolution
    else:
        n_points = 84

    target_sum = n_points / 2.0                 # because integral must be 1, dx = 2/n

    # initialise h

        # upscale the provided vector by linear interpolation
        old = np.asarray(initial_h_values, dtype=float)

        xs_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.interp(xs_new, xs_old, old)
        h = _project_h(h, target_sum)
    else:
        h = _make_initial_h(n_points)

    # base evaluation
    def evaluate(vec):

    best_h = h.copy()
