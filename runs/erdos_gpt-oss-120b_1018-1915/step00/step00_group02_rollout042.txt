import time, math, random
import numpy as np
from scipy import optimize  # noqa: F401


def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for discrete h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_constraints(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce exact integral = n/2."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        h[:] = target / n
        return h
    h *= target / s
    # fix any tiny rounding errors
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the residual to a random feasible index
        for _ in range(10):
            i = np.random.randint(n)
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff = target - h.sum()
            if abs(diff) < 1e-12:
                break
    return h


def _local_search(h0: np.ndarray, start: float, budget: float,
                  best_c5: float) -> tuple[np.ndarray, float]:
    """Simple hill‑climbing with random two‑index transfers."""
    n = h0.size
    cur_h = h0.copy()
    cur_c5 = best_c5
    best_h = cur_h.copy()
    best_val = cur_c5

    max_iter = 80000
    for it in range(max_iter):
        if time.time() - start > budget * 0.88:
            break

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # random transfer from i to j (or vice‑versa)
        if np.random.rand() < 0.5:
            max_move = min(cur_h[i], 1.0 - cur_h[j])
            if max_move == 0:
                continue
            delta = np.random.rand() * max_move
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            max_move = min(cur_h[j], 1.0 - cur_h[i])
            if max_move == 0:
                continue
            delta = np.random.rand() * max_move
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        new_h = _project_to_constraints(new_h)
        new_val = _compute_c5(new_h)

        # accept only improvements
        if new_val + 1e-12 < cur_c5:
            cur_h, cur_c5 = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()

    # ---- initial construction ------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n0 = 60
        h = np.full(n0, 0.5, dtype=float)

    h = _project_to_constraints(h)
    best_val = _compute_c5(h)

    # ---- progressive refinement ------------------------------------------------
    # try three resolutions: current, doubled, doubled again
    for factor in (1, 2, 2):
        if time.time() - start_time > budget_s * 0.94:
            break

        if factor > 1:
            # linear interpolation to higher resolution
            n_old = h.size
            n_new = n_old * factor
            x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
            h = np.interp(x_new, x_old, h)
            h = _project_to_constraints(h)

        # hill‑climbing at this resolution
        h, best_val = _local_search(h, start_time, budget_s, best_val)

    # final evaluation
    final_c5 = _compute_c5(h)
    return (h.tolist(), float(final_c5), h.size)
