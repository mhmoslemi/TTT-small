import numpy as np, time, math, random
from scipy.optimize import minimize

# ------------------------------------------------------------
def _objective(h):
    """Return C5 bound for step function h on [0,2]."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * np.max(corr)


def _feasible_projection(h, target_sum):
    """
    Project h onto the box [0,1]^n while preserving (approximately) the total sum.
    Simple iterative clipping + uniform redistribution.
    """
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # distribute diff uniformly over all entries that are not at a bound
        free = (h > 0.0) & (h < 1.0)
        if not np.any(free):
            # all at bounds – just add diff to first element and re‑clip
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        else:
            add = diff / free.sum()
            h[free] += add
            h = np.clip(h, 0.0, 1.0)
    # final tiny correction to enforce sum exactly (won't break bounds appreciably)
    cur = h.sum()
    if abs(cur - target_sum) > 1e-10:
        h[0] = np.clip(h[0] + (target_sum - cur), 0.0, 1.0)
    return h


def _interpolate_parent(parent, new_n):
    """Linear interpolation of parent vector to new_n points."""
    old_n = parent.size
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, parent)
    # enforce integral constraint
    target = new_n / 2.0
    new_h = _feasible_projection(new_h, target)
    return new_h


def _local_search(h0, budget_s, start_time):
    """Simple random two‑coordinate exchange hill‑climb with temperature decay."""
    n = h0.size
    target_sum = n / 2.0
    best_h = h0.copy()
    best_val = _objective(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
        # temperature schedule (log‑linear)
        temp = max(1e-4, (budget_s - elapsed) / budget_s)
        # pick two distinct indices
        i, j = random.sample(range(n), 2)
        hi, hj = cur_h[i], cur_h[j]
        # allowed move range so both stay in [0,1]
        d_max = min(hi, 1.0 - hj)
        d_min = -min(1.0 - hi, hj)
        if d_max <= d_min:
            continue
        # magnitude scales with temperature
        max_step = (d_max - d_min) * 0.5 * temp
        d = (np.random.rand() - 0.5) * 2.0 * max_step
        d = np.clip(d, d_min, d_max)
        if abs(d) < 1e-12:
            continue
        new_h = cur_h.copy()
        new_h[i] -= d
        new_h[j] += d
        # no need to re‑project; sum preserved and bounds respected by construction
        new_val = _objective(new_h)
        # accept if better or with Metropolis probability
        if new_val < cur_val or np.exp(-(new_val - cur_val) / max(temp, 1e-8)) > np.random.rand():
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
        it += 1
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
