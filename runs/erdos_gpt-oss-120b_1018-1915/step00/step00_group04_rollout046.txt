import numpy as np
import time

def _compute_score(h):
    """Return max overlap integral C5 for step values h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _repair(h, target_sum, max_iter=5):
    """Project h into [0,1] and enforce exact sum = target_sum."""
    for _ in range(max_iter):
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if cur == 0:
            h[:] = target_sum / h.size
            break
        factor = target_sum / cur
        h *= factor
    # final clip to guarantee bounds
    h = np.clip(h, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _upscale(old_h, new_n):
    """Linear interpolation of a step vector to a larger resolution."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    # enforce sum after interpolation
    target = new_n / 2.0
    return _repair(new_h, target)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with values in [0,1] and integral 1
    to minimise the Erdős overlap constant C5.

    Returns

        Optimised step values (length n_points).

        Corresponding C5 value.

        Number of discretisation points used.
    """
    np.random.seed(seed)
    t_start = time.time()

    # ------------------------------------------------------------------

    n_points = 400                     # moderate size (<1k)
    target_sum = n_points / 2.0
    max_no_improve = 2000              # stop after many rejects
    temp_start = 0.02
    temp_end = 0.001
    total_iters = 200000               # generous upper bound

    # ------------------------------------------------------------------
    # Initialise population (single current solution)
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():

    else:
        h_curr = np.random.rand(n_points)
        h_curr = _repair(h_curr, target_sum)

    best_h = h_curr.copy()
    best_score = _compute_score(best_h)

    # ------------------------------------------------------------------
    # Simple stochastic hill‑climbing with occasional temperature decay
    # ------------------------------------------------------------------
    it_no_improve = 0
    for it in range(total_iters):
        # time check
        if time.time() - t_start > budget_s * 0.95:
            break

        # temperature schedule (linear)
        temp = temp_start + (temp_end - temp_start) * (it / total_iters)

        # propose a small exchange between two positions
        i, j = np.random.choice(n_points, 2, replace=False)
        delta = np.random.normal(scale=0.05) * (1.0 - 0.0)  # scale ~0.05
