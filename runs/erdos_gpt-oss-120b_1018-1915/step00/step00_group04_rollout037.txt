import time
import numpy as np
from scipy.optimize import differential_evolution

def compute_c5(h):
    """Return the Erdős overlap bound for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def propose_move(h, max_delta):
    """Generate a feasible neighbour by moving mass between two coordinates."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    delta = np.random.uniform(-max_delta, max_delta)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    if 0.0 <= new_h[i] <= 1.0 and 0.0 <= new_h[j] <= 1.0:
        # sum is preserved exactly because we added and subtracted the same delta
        return new_h
    return None


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős C5 constant.

    Returns
    -------
    h_values : list of float
        Optimized discretized h values.
    c5_bound : float
        The corresponding C5 upper bound.
    n_points : int
        Number of discretization points used.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Choose a problem size (keep < 1000). 100 points offers a good trade‑off.
    # ----------------------------------------------------------------------
    n = 100

    # ----------------------------------------------------------------------
    # Initialise h. Use the provided global `initial_h_values` if present,
    # otherwise fall back to a uniform 0.5 function.
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(initial_h_values, dtype=float)  # noqa: F821
        # Resample to the required length via linear interpolation.
        x_old = np.linspace(0, 2, base.size, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h0 = np.interp(x_new, x_old, base)
    except NameError:
        h0 = np.full(n, 0.5)

    # Enforce the exact integral constraint (sum == n/2).
    # Simple scaling to achieve the right sum while staying in [0,1].
    total = h0.sum()
    target = n / 2.0
    if total != 0:
        h0 = h0 * (target / total)
    h0 = np.clip(h0, 0.0, 1.0)

    # ----------------------------------------------------------------------
    # Differential Evolution as a global search.
    # ----------------------------------------------------------------------
    bounds = [(0.0, 1.0)] * n
    penalty_weight = 1e3

    def objective(x):
        # Clip to keep within bounds (DE may propose values slightly outside).
        h = np.clip(x, 0.0, 1.0)
        # Penalise violation of the integral constraint.
        sum_violation = np.abs(h.sum() - target)
        return compute_c5(h) + penalty_weight * sum_violation

    # Callback to respect the time budget.
    def de_callback(xk, convergence):
        if time.time() - start_time > budget_s * 0.85:
            return True  # stop the optimizer
        return False

    # Run DE with a limited number of generations to stay within budget.
    result = differential_evolution(
        objective,
        bounds,
        strategy="best1bin",
        maxiter=200,
        popsize=15,
        mutation=(0.5, 1.0),
        recombination=0.7,
        polish=False,
        init="latinhypercube",
        updating="deferred",
        seed=seed,
        callback=de_callback,
        disp=False,
        workers=1,
    )

    best_h = np.clip(result.x, 0.0, 1.0)
    # Project exactly onto the sum constraint by a simple redistribution.
    diff = best_h.sum() - target
    if abs(diff) > 1e-12:
        # Distribute the excess uniformly to entries that stay in bounds.
        # This loop terminates quickly because diff is tiny after DE.
        for _ in range(1000):
            mask = (best_h > 0.0) & (best_h < 1.0)
            if not mask.any():
                break
            adjust = diff / mask.sum()
            best_h[mask] -= adjust
            # Clip again and recompute diff.
            best_h = np.clip(best_h, 0.0, 1.0)
            diff = best_h.sum() - target
            if abs(diff) < 1e-12:
                break

    best_c5 = compute_c5(best_h)

    # ----------------------------------------------------------------------
    # Local refinement: stochastic pairwise moves.
    # ----------------------------------------------------------------------
    max_delta = 0.05
    while time.time() - start_time < budget_s * 0.98:
        new_h = propose_move(best_h, max_delta)
        if new_h is None:
            continue
        c5 = compute_c5(new_h)
        if c5 < best_c5:
            best_h = new_h
            best_c5 = c5
        # Occasionally shrink the mutation size to fine‑tune.
        if np.random.rand() < 0.01:
            max_delta *= 0.9
            max_delta = max(max_delta, 1e-4)

    return best_h.tolist(), float(best_c5), n
