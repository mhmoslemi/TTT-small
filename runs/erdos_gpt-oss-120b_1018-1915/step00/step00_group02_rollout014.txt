def _score(h):
    """Return C5 for a discrete step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # full cross‑correlation of h with (1‑h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _enforce_bounds_and_sum(h, target_sum):
    """Clip to [0,1] and then adjust one coordinate to hit the exact sum."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # find an index with slack to absorb the difference
        for idx in range(h.shape[0]):
            low = -h[idx]
            high = 1.0 - h[idx]
            if low <= diff <= high:
                h[idx] += diff
                break
        else:
            # fallback: distribute uniformly (should be rare)
            h += diff / h.shape[0]
            h = np.clip(h, 0.0, 1.0)
    # final clip to be safe
    return np.clip(h, 0.0, 1.0)


def _local_search(h, time_budget, max_iter=5000, init_step=0.2):
    """Simple hill‑climbing that preserves the integral."""
    n = h.shape[0]
    target_sum = n / 2.0
    best = h.copy()
    best_score = _score(best)
    step = init_step
    start = time.time()
    it = 0
    while it < max_iter and time.time() - start < time_budget:

            continue

            step = max(step * 0.95, 1e-4)
        it += 1
