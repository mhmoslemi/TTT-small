import time
import numpy as np

def _c5_score(h):
    """Return the C5 value for a given step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _initial_binary(n, rng):
    """Create a binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] that minimizes C5.

    Returns
    -------
    h_values : list of float
        The best step function found (values in [0,1]).
    c5_bound : float
        Corresponding C5 value.
    n_points : int
        Number of samples used.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine problem size and starting vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.size
        # Project onto feasible binary space if not already binary
        # Count how many entries are close to 1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        # Adjust count to exactly n/2 ones
        target_ones = n // 2
        if ones.size > target_ones:
            # demote excess ones to zero
            drop = rng.choice(ones, ones.size - target_ones, replace=False)
            h[drop] = 0.0
            ones = np.setdiff1d(ones, drop, assume_unique=True)
        elif ones.size < target_ones:
            # promote zeros to one
            add = rng.choice(zeros, target_ones - ones.size, replace=False)
            h[add] = 1.0
            ones = np.union1d(ones, add)
        # Ensure exact binary values
        h[ones] = 1.0
        zero_idx = np.setdiff1d(np.arange(n), ones, assume_unique=True)
        h[zero_idx] = 0.0
    else:
        n = 200  # moderate resolution
        h = _initial_binary(n, rng)

    best_h = h.copy()
    best_score = _c5_score(best_h)

    # Pre‑compute index pools
    def refresh_pools(vec):
        ones = np.where(vec > 0.5)[0]
        zeros = np.where(vec <= 0.5)[0]
        return ones, zeros

    ones_pool, zeros_pool = refresh_pools(best_h)

    # Simple simulated‑annealing style loop
    iteration = 0
    temperature = 0.05  # acceptance probability scale
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        # Propose a swap between a 1 and a 0
        i = rng.choice(ones_pool)
        j = rng.choice(zeros_pool)
        # Perform swap
        best_h[i], best_h[j] = 0.0, 1.0

        score = _c5_score(best_h)

        accept = False
        if score < best_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(score - best_score) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:

                best_score = score
