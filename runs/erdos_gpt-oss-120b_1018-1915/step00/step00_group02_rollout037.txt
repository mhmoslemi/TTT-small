import time
import numpy as np
import scipy.optimize as opt

def _c5_of_h(h):
    """Compute the overlap constant C5 for a discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def _make_feasible(x, target_sum, max_iter=10):
    """Project vector x onto [0,1] box and enforce exact sum = target_sum."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        diff = target_sum - x.sum()
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            # all at bounds – distribute uniformly (may violate bounds a little)
            x += diff / x.size
            x = np.clip(x, 0.0, 1.0)
            continue
        x[free] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize the Erdős overlap constant C5.

    Returns
    -------
    h_values : list of float
        The optimal discretised function values.
    c5_bound : float
        The achieved upper bound for C5.
    n_points : int
        Number of sample points used.
    """
    start_time = time.time()
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # Starting construction
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: half‑ones, half‑zeros (satisfies the integral)
        n0 = 45
        h0 = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)])
    n0 = h0.size
    target_sum0 = n0 / 2.0
    h0 = _make_feasible(h0, target_sum0)

    best_h = h0.copy()
    best_c5 = _c5_of_h(best_h)

    # ----------------------------------------------------------------------
    # Resolution schedule (start at current size, then double up to ~400)
    # ----------------------------------------------------------------------
    schedule = [n0, 90, 180, 360]
    # ensure monotonic schedule without repeats
    schedule = [s for s in schedule if s >= n0]

    for n_target in schedule:
        # time guard
        if time.time() - start_time > 0.95 * budget_s:
            break

        # Upsample previous best if needed
        if n_target != best_h.size:
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)

        else:
            h_start = best_h.copy()

        target_sum = n_target / 2.0
        h_start = _make_feasible(h_start, target_sum)

        # ------------------------------------------------------------------
        # Multiple local optimisation attempts at this resolution
        # ------------------------------------------------------------------
        attempts = 6
        for _ in range(attempts):

            # Random perturbation of the start point
            init = h_start + 0.04 * np.random.randn(n_target)
            init = _make_feasible(init, target_sum)
