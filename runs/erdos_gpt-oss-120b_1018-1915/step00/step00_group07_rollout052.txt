import time
import numpy as np

# The harness provides `initial_h_values` when available.
# It also provides `evaluate_erdos_solution` but we implement our own fast evaluator.

def _repair(vec, target_sum):
    """Clip to [0,1] and rescale to meet the target sum (approximate)."""
    v = np.clip(vec, 0.0, 1.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, target_sum / v.size)
    # scale to hit the target sum, then clip again
    v = v * (target_sum / s)
    v = np.clip(v, 0.0, 1.0)
    # a second scaling to reduce any residual error
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, target_sum / v.size)
    v = v * (target_sum / s)
    return np.clip(v, 0.0, 1.0)


def _objective(h, dx):
    """Return the C5 value for a candidate step function."""
    # h already respects bounds and sum ≈ target
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def run(seed=42, budget_s=1100, **kwargs):
    """Differential evolution search for a low C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : ndarray of shape (n_points,)
        c5_bound : float, the best C5 found
        n_points : int, number of discretisation points used
    """
    start_time = time.time()
    # problem size
    n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # integral = 1

    # initialise random generator
    rng = np.random.default_rng(seed)

    # base population
    pop_size = 24
    # include parent if given
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        base = _repair(base, target_sum)
        population = [base]
    else:
        population = []

    # fill remaining members randomly
    while len(population) < pop_size:
        cand = rng.random(n_points)  # uniform in [0,1)
        cand = _repair(cand, target_sum)
        population.append(cand)

    population = np.stack(population)
    scores = np.array([_objective(ind, dx) for ind in population])

    # keep the best ever
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = float(scores[best_idx])

    # DE parameters
    F = 0.5          # mutation factor
    CR = 0.9         # crossover probability
    max_gen = 500    # generous upper bound, will stop earlier by time

    for gen in range(max_gen):
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break

        new_pop = []
        new_scores = []

        for i in range(pop_size):
            # choose three distinct indices different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = population[r1] + F * (population[r2] - population[r3])

            # binomial crossover
            cross_mask = rng.random(n_points) < CR
            # ensure at least one dimension from mutant
            cross_mask[rng.integers(0, n_points)] = True
            trial = np.where(cross_mask, mutant, population[i])

            trial = _repair(trial, target_sum)

            trial_score = _objective(trial, dx)

            # selection
            if trial_score <= scores[i]:
                new_pop.append(trial)
                new_scores.append(trial_score)
                if trial_score < best_score:
                    best_score = float(trial_score)
                    best_h = trial.copy()
            else:
                new_pop.append(population[i])
                new_scores.append(scores[i])

        population = np.stack(new_pop)
        scores = np.array(new_scores)

    # final return
    return best_h, float(best_score), n_points
