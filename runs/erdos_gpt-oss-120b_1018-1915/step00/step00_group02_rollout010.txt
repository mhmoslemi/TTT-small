import numpy as np
from scipy.optimize import minimize, LinearConstraint, Bounds

def _c5_of_h(h):
    """Compute the C5 value for a step function sampled in h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def _optimize_h(start_h, max_time, max_iter=250):

        return _c5_of_h(x)
