import numpy as np
import time

def _repair(vec, target_sum):
    """Project vec onto [0,1] box and enforce sum == target_sum."""
    v = np.clip(vec, 0.0, 1.0)
    for _ in range(5):
        s = v.sum()
        if abs(s - target_sum) < 1e-9:
            break
        scale = target_sum / s if s != 0 else 0.0
        v = np.clip(v * scale, 0.0, 1.0)
    # final correction for tiny residual by adding/subtracting uniformly
    diff = target_sum - v.sum()
    if abs(diff) > 1e-9:
        # distribute diff over non‑boundary entries
        free = (v > 0.0) & (v < 1.0)
        if free.any():
            v[free] += diff / free.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _objective(h, dx):
    """Return max overlap C5 for a candidate h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _initialize_population(pop_size, n, target_sum, rng):
    """Create a feasible population."""
    pop = np.empty((pop_size, n), dtype=np.float64)
    # include a deterministic half‑ones vector (feasible)
    half = np.zeros(n, dtype=np.float64)
    half[:int(target_sum)] = 1.0
    rng.shuffle(half)
    pop[0] = half
    for i in range(1, pop_size):
        # start from uniform random then repair
        rand = rng.random(n)
        pop[i] = _repair(rand, target_sum)
    return pop

def _de_optimize(initial, n, dx, budget_s, seed):
    rng = np.random.default_rng(seed)
    target_sum = n / 2.0
    pop_size = 30
    F = 0.5
    CR = 0.9
    max_gen = 2000   # generous, will stop by time
    # population

        init_vec = np.array(initial_h_values, dtype=np.float64)
        init_vec = _repair(init_vec, target_sum)
        pop = _initialize_population(pop_size, n, target_sum, rng)
        pop[0] = init_vec
    else:
        pop = _initialize_population(pop_size, n, target_sum, rng)

    scores = np.empty(pop_size, dtype=np.float64)
    for i in range(pop_size):
        scores[i] = _objective(pop[i], dx)

    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_score = scores[best_idx]

    start = time.time()
    gen = 0
    while gen < max_gen and (time.time() - start) < budget_s:
        for i in range(pop_size):
            idxs = rng.choice(pop_size, 3, replace=False)
            a, b, c = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]
            mutant = a + F * (b - c)
            # binomial crossover
            cross = np.empty(n, dtype=np.float64)
            j_rand = rng.integers(n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    cross[j] = mutant[j]
                else:
                    cross[j] = pop[i][j]
            child = _repair(cross, target_sum)
            child_score = _objective(child, dx)

                scores[i] = child_score
                if child_score < best_score:
                    best_score = child_score
                    best_vec = child.copy()
        gen += 1
    return best_vec, best_score
