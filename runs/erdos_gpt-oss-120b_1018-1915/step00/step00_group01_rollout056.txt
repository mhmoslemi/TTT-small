import time
import numpy as np
from scipy import optimize

# Helper: enforce 0≤h≤1 and ∫h = 1  (dx = 2 / n)
def _project(h, dx):
    h = np.clip(h, 0.0, 1.0)
    total = h.sum() * dx
    if total == 0.0:
        return h
    scale = 1.0 / total
    h = h * scale
    # iterative fix for any values that exceed 1 after scaling
    for _ in range(5):
        over = h > 1.0
        if not np.any(over):
            break
        excess = (h[over] - 1.0).sum() * dx
        h[over] = 1.0
        under = h < 1.0
        if np.any(under):
            h[under] += excess / (under.sum() * dx)
            h = np.clip(h, 0.0, 1.0)
    # final normalization (tiny drift)
    total = h.sum() * dx
    if not np.isclose(total, 1.0, rtol=1e-9, atol=1e-12):
        h = h * (1.0 / total)
        h = np.clip(h, 0.0, 1.0)
    return h

# Objective: max overlap C5 for a given h
def _c5_value(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _objective(h, dx):
    return _c5_value(h, dx)

def _optimize(initial_h, dx, time_deadline):
    n = len(initial_h)
    bounds = [(0.0, 1.0)] * n
    cons = ({
        'type': 'eq',
        'fun': lambda h: np.sum(h) * dx - 1.0
    },)
    # SLSQP works with bound and equality constraints
    opts = {'maxiter': 200, 'ftol': 1e-9, 'disp': False}
    best_h = None
    best_val = np.inf

    # try a few random restarts plus the supplied start
    starts = [initial_h]
    rng = np.random.default_rng(42)
    for _ in range(3):
        rnd = rng.random(n)
        rnd = _project(rnd, dx)
        starts.append(rnd)

    for x0 in starts:
        if time.time() >= time_deadline:
            break
        res = optimize.minimize(
            lambda h: _objective(h, dx),
            x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options=opts
        )
        if not res.success:
            continue
        h_opt = _project(res.x, dx)
        val = _c5_value(h_opt, dx)
        if val < best_val:
            best_val = val
            best_h = h_opt
    return best_h, best_val
