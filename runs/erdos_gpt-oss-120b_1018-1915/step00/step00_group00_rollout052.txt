import time
import math
import random
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def propose_swap(h, rng):
    """Return a new h obtained by moving a feasible delta between two indices."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    hi, hj = h[i], h[j]

    # feasible delta range: keep both in [0,1]
    delta_max = min(1.0 - hi, hj)
    delta_min = -min(hi, 1.0 - hj)
    if delta_max <= delta_min:
        return None  # no feasible move (should be rare)

    delta = rng.uniform(delta_min, delta_max)
    new_h = h.copy()
    new_h[i] = hi + delta
    new_h[j] = hj - delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # obtain initial vector
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 45
        h0 = np.full(n, 0.5, dtype=float)  # satisfies sum condition automatically

    dx = 2.0 / n
    target_sum = n / 2.0
    # enforce exact sum (tiny numerical error correction)
    diff = target_sum - h0.sum()
    if abs(diff) > 1e-12:
        h0 += diff / n

    # initial evaluation
    cur_h = h0.copy()
    cur_c5 = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # annealing schedule
    T = 0.05 * cur_c5  # initial temperature proportional to objective
    cooling = 0.9995
    iteration = 0
    start = time.time()
    time_limit = start + budget_s * 0.95  # keep a safety margin

    while time.time() < time_limit:
        iteration += 1
        cand_h = propose_swap(cur_h, rng)
        if cand_h is None:
            continue
        cand_c5 = compute_c5(cand_h, dx)

        # accept if better or with Metropolis probability
        if cand_c5 < cur_c5:
            accept = True
        else:
            prob = math.exp(-(cand_c5 - cur_c5) / (T + 1e-12))
            accept = rng.random() < prob

        if accept:

            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
