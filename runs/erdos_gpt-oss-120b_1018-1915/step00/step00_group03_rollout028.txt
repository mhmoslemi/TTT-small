import time
import numpy as np

def _evaluate_c5(h):
    """Return the C5 value for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _repair(h, target_sum, max_iter=15):
    """Project h onto the box [0,1] with the given sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        # Determine where we can move in the needed direction
        if diff > 0.0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        # Distribute the needed adjustment uniformly over the free entries
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h
