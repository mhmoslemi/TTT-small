        if abs(s - target) < 1e-8:
            break
        diff = target - s
        free = (v > 0) & (v < 1)
        if not np.any(free):
            # all at bounds, distribute using scaling
            v = v * (target / s)
            v = np.clip(v, 0, 1)
            continue

        v[free] += adjust
        v = np.clip(v, 0, 1)

import time
import numpy as np

def _project_to_bounds_sum(v, target, max_iter=20, tol=1e-9):
    """Project vector v onto the box [0,1]^n intersect {sum = target}."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target - s
        if abs(diff) <= tol:
            break
        free = (v > 0.0) & (v < 1.0)
        n_free = np.count_nonzero(free)
        if n_free == 0:                      # all at a bound, scale globally
            v = np.clip(v * (target / s), 0.0, 1.0)
            continue
        adjust = diff / n_free
        v[free] += adjust
        v = np.clip(v, 0.0, 1.0)
    return v

def _evaluate(h, dx):
    """Return the maximal overlap integral for a candidate step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] with ∫h = 1 that minimizes C₅.

    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # Determine discretisation size
    if "initial_h_values" in globals():
        n_points = int(len(initial_h_values))
    else:
        n_points = 80
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # because integral = 1

    # ----- helper functions -----
    def repair(vec):
        return _project_to_bounds_sum(vec, target_sum)

    def score(vec):
        return _evaluate(vec, dx)

    # ----- initialise population -----
    pop_size = max(30, n_points // 2)   # modest size, works for ≤100 points
    population = []
    scores = []

    # include provided parent if available
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        h0 = repair(h0)
        population.append(h0)
        scores.append(score(h0))

    while len(population) < pop_size:
        cand = rng.random(n_points)
        cand = repair(cand)
        population.append(cand)
        scores.append(score(cand))

    population = np.array(population)
    scores = np.array(scores)

    # ----- differential evolution core -----
    F = 0.5      # mutation factor
    CR = 0.9     # crossover probability
    best_idx = int(np.argmin(scores))
    best_h = population[best_idx].copy()
    best_score = scores[best_idx]

    # iterate until time runs out or target reached
    while time.time() - start_time < budget_s * 0.95:
        for i in range(pop_size):
            # choose three distinct individuals different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)

            mutant = population[r1] + F * (population[r2] - population[r3])

            # binomial crossover
            cross = np.empty(n_points)
            jrand = rng.integers(n_points)
            for j in range(n_points):
                if rng.random() < CR or j == jrand:
                    cross[j] = mutant[j]

            # selection
