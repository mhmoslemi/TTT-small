import time
import numpy as np

def evaluate_erdos(h, n):
    # correlation of h with (1-h), full mode
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)


def repair(h, desired_sum):
    """
    Clip to [0,1] and then adjust to hit the exact sum by distributing the
    residual to any coordinates that can still move.
    """
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = desired_sum - cur
    # simple redistribution loop (few iterations are enough)
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass: pick indices with room <1
            room = 1.0 - h
            if np.all(room <= 1e-12):
                break
            idx = np.argmax(room)
            add = min(diff, room[idx])
            h[idx] += add
            diff -= add
        else:
            # need to remove mass
            room = h
            if np.all(room <= 1e-12):
                break
            idx = np.argmax(room)
            sub = min(-diff, room[idx])
            h[idx] -= sub
            diff += sub
    # final clip (should already be feasible)
    return np.clip(h, 0.0, 1.0)


def interpolate_parent(parent, n_target):
    """Linearly interpolate a parent vector to a new size."""
    m = len(parent)
    x_old = np.linspace(0.0, 2.0, m, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(x_new, x_old, parent)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- choose resolution -------------------------------------------------
    n = 200  # a modest size: fast eval, enough freedom
    desired_sum = n / 2.0

    # ----- initial vector ----------------------------------------------------
    try:
        parent = initial_h_values  # provided by harness if available
        h = interpolate_parent(parent, n)
    except NameError:
        # start from a simple half‑filled block (measure 1)
        h = np.zeros(n)
        h[: n // 2] = 1.0
    h = repair(h, desired_sum)

    best_h = h.copy()
    best_score = evaluate_erdos(best_h, n)

    # ----- annealing parameters ---------------------------------------------
    temp0 = 0.05
    temp_final = 0.0005
    max_iter = 200_000
    # schedule exponential decay based on elapsed fraction
    for it in range(max_iter):
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break
        # temperature
        frac = it / max_iter
        temp = temp0 * (temp_final / temp0) ** frac

        # propose a move: pick two distinct indices and transfer mass
        i, j = rng.choice(n, size=2, replace=False)
        delta = (rng.random() - 0.5) * 0.2  # up to ±0.1
        # apply tentative changes
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # repair to stay feasible
        h_new = repair(h_new, desired_sum)

        score_new = evaluate_erdos(h_new, n)

        # accept if better or with Metropolis probability
        improve = score_new < best_score
        if improve or rng.random() < np.exp(-(score_new - best_score) / max(temp, 1e-12)):
            h = h_new
            if score_new < best_score:
                best_score = score_new
                best_h = h_new.copy()
