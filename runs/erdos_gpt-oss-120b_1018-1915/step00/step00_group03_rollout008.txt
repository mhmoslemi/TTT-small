import numpy as np
import time

def _c5_score(h):
    """Return max overlap integral for step function h."""
    dx = 2.0 / h.size
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()

def _transfer(h, i, j, delta):
    """Transfer delta from i to j, keeping bounds [0,1] and preserving sum."""
    hi, hj = h[i], h[j]

    # adjust delta so hi - delta stays in [0,1]
    if hi - delta < 0.0:
        delta = hi
    elif hi - delta > 1.0:
        delta = hi - 1.0

    # adjust delta so hj + delta stays in [0,1]
    if hj + delta < 0.0:
        delta = -hj
    elif hj + delta > 1.0:
        delta = 1.0 - hj

    h[i] = hi - delta
    h[j] = hj + delta
    return h

def _perturb(h, scale):
    """Randomly move a small amount between two coordinates."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    delta = scale * (np.random.rand() - 0.5) * 2.0  # uniform in [-scale,scale]
    return _transfer(h.copy(), i, j, delta)
