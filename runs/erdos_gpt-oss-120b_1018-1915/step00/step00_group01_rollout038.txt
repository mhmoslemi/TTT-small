import numpy as np, time, math, random

def _project_bounds_sum(v, target, max_iter=100):
    """Project vector v onto the box [0,1] with sum == target."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target
        if abs(diff) < 1e-10:
            break
        if diff > 0:  # need to reduce
            mask = x > 0.0
            if not mask.any():
                break
            reduction = diff / mask.sum()
            x[mask] = np.maximum(0.0, x[mask] - reduction)
        else:          # need to increase
            mask = x < 1.0
            if not mask.any():
                break
            increase = -diff / mask.sum()
            x[mask] = np.minimum(1.0, x[mask] + increase)
    return x

def _c5_value(h):
    """Compute C5 for a discretised step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _initial_h(n):
    """Create an initial vector of length n using the global initial_h_values if present."""
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.size == n:
            h0 = init.copy()
        else:
            old_x = np.linspace(0.0, 2.0, init.size, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(new_x, old_x, init)
    else:
        # Random binary pattern with exactly half the mass
        h0 = np.zeros(n, dtype=float)
        ones = np.random.choice(n, n // 2, replace=False)
        h0[ones] = 1.0
    # Project to satisfy sum constraint exactly
    target = n / 2.0
    return _project_bounds_sum(h0, target)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_val = float('inf')
    # Candidate resolutions (keep small for speed)
    candidates = [47, 94]
    for n in candidates:
        if time.time() - start_time > budget_s * 0.95:
            break
        h_cur = _initial_h(n)
        cur_val = _c5_value(h_cur)
        target_sum = n / 2.0
        if cur_val < best_val:
            best_val, best_h = cur_val, h_cur.copy()
        # Simulated‑annealing style search
        max_iter = 6000
        T0, T_end = 0.5, 1e-4
        for i in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
            # temperature schedule
            T = T0 * (T_end / T0) ** (i / max_iter)
            # propose a random perturbation
            proposal = h_cur + np.random.normal(scale=0.12, size=n)
            proposal = _project_bounds_sum(proposal, target_sum)
            prop_val = _c5_value(proposal)
            # accept if better or with Metropolis probability
            if prop_val < cur_val or random.random() < math.exp((cur_val - prop_val) / (T + 1e-12)):
                h_cur, cur_val = proposal, prop_val
                if prop_val < best_val:
                    best_val, best_h = prop_val, proposal.copy()
        # early stop if we already beat the target bound
        if best_val <= 0.3808:
            break
    # final safety projection (ensures exact sum)
    if best_h is not None:
        best_h = _project_bounds_sum(best_h, best_h.size / 2.0)
        best_val = _c5_value(best_h)
    return (best_h, best_val, int(best_h.size) if best_h is not None else 0)
