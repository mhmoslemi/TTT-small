import time, random, math
import numpy as np

def evaluate(h):
    """Return C5 for discrete h on [0,2] with step dx = 2/n."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (2.0 / n) * np.max(corr)

def project_feasible(h):
    """Clamp to [0,1] and enforce sum(h)=n/2 by simple iterative redistribution."""
    n = len(h)
    desired = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(5):
        cur = h.sum()
        diff = desired - cur
        if abs(diff) < 1e-10:
            break
        # distribute uniformly; this keeps average correct
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
    # final fine‑tune: if still off, scale the interior points
    cur = h.sum()
    if abs(cur - desired) > 1e-8:
        # adjust only those not at bounds
        free = (h > 0) & (h < 1)
        if free.any():
            adjust = (desired - cur) / free.sum()
            h[free] += adjust
            h = np.clip(h, 0.0, 1.0)
    # ensure exactness (tiny rounding)
    h *= desired / h.sum()
    return h

def upsample(h, new_n):
    """Linear interpolation of h from its current length to new_n samples."""
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, h)

def random_feasible(n, rng):
    """Generate a random feasible vector with entries in [0,1] and sum=n/2."""
    h = rng.random(n)
    h = project_feasible(h)
    return h

def simulated_anneal(h, budget, start_time, rng):
    """Simple SA that mutates h while respecting constraints."""
    n = len(h)
    best_h = h.copy()
    best_score = evaluate(best_h)
    temperature = 0.5
    step_scale = 0.2  # max absolute delta per move
    iter_count = 0

    while time.time() - start_time < budget * 0.98:
        iter_count += 1
        # gradually cool
        if iter_count % (n * 10) == 0:
            temperature *= 0.9
            step_scale *= 0.9
            if temperature < 1e-4:
                temperature = 1e-4
        # propose a change
        i = rng.integers(0, n)
        delta = (rng.random() * 2 - 1) * step_scale
        h_new = best_h.copy()
        h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
        h_new = project_feasible(h_new)

        new_score = evaluate(h_new)

        # accept if better or by Metropolis criterion
        if new_score < best_score or rng.random() < math.exp(-(new_score - best_score) / temperature):
            best_h = h_new
            best_score = new_score

    return best_h, best_score

    start = time.time()
    rng = np.random.default_rng(seed)

    # 1. initialise base vector

        n0 = len(h0)
        h0 = project_feasible(h0)
    else:
        n0 = 200
        h0 = np.full(n0, 0.5, dtype=float)  # already feasible

    best_h = h0
    best_score = evaluate(best_h)

    # 2. progressive refinement at larger resolutions
    target_sizes = [n0, 300, 500, 750]  # keep below 1000 for speed
    for sz in target_sizes:
        if time.time() - start >= budget_s * 0.95:
            break
        # upscale current best to new size
        h_up = upsample(best_h, sz)
        h_up = project_feasible(h_up)

        # allocate a fraction of remaining time
        remaining = budget_s - (time.time() - start)
        if remaining < 5.0:

        alloc = min(remaining * 0.4, 30.0)  # keep each stage short

        h_opt, sc_opt = simulated_anneal(h_up, alloc, start, rng)

        if sc_opt < best_score:
            best_h, best_score = h_opt, sc_opt

    final_budget = budget_s - (time.time() - start)
    if final_budget > 2.0:
        best_h, best_score = simulated_anneal(best_h, final_budget, start, rng)

    # ensure feasibility one last time
    best_h = project_feasible(best_h)
    best_score = evaluate(best_h)

    return best_h, float(best_score), int(len(best_h))
