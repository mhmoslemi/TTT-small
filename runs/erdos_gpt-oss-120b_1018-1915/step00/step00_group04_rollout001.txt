import time
import numpy as np
import random

def compute_c5(h):
    """Return C5 for a step function h on [0,2] with uniform sampling."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def repair(h, target_sum, tol=1e-9, max_iter=10):
    """Project h onto the feasible set: 0≤h≤1 and sum = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) <= tol:
            break
        # Distribute diff uniformly, but keep within bounds
        add = diff / h.size
        h += add
        # Clip again
        h = np.clip(h, 0.0, 1.0)
    # Final scaling if still off (rare)
    s = h.sum()
    if s != 0:
        h *= target_sum / s
    h = np.clip(h, 0.0, 1.0)
    return h

def mutate(h, sigma, target_sum):
    """Gaussian mutation with repair."""
    new_h = h + np.random.normal(0.0, sigma, size=h.shape)
    return repair(new_h, target_sum)

def upscale(h):
    """Double the resolution by linear interpolation."""
    n = h.size
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, 2 * n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # enforce integral constraint for new size
    target = (2 * n) / 2.0
    return repair(h_new, target)

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimize C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    t_start = time.time()
    time_limit = budget_s * 0.95  # safety margin

    # Starting solution
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
        target_sum = n / 2.0
        h = repair(h, target_sum)
        best_h = h.copy()
        best_score = compute_c5(best_h)
    else:
        # default 42-point random feasible start
        n = 42
        target_sum = n / 2.0
        h = np.random.uniform(0.0, 1.0, n)
        h = repair(h, target_sum)
        best_h = h.copy()
        best_score = compute_c5(best_h)

    # Optimization parameters
    max_iter = 200000
    sigma_initial = 0.2
    sigma_final = 0.005
    iter_count = 0

    while time.time() - t_start < time_limit and iter_count < max_iter:
        # annealing schedule
        sigma = sigma_initial * (1 - iter_count / max_iter) + sigma_final * (iter_count / max_iter)

        # occasional random restart
