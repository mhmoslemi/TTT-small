# The harness provides:
#   evaluate_erdos_solution(h_values) -> float (the C5 bound)
#   initial_h_values (optional) -> ndarray of shape (n_points,)

def _feasible_from_initial(init):
    """Return a feasible vector with sum = n/2 and 0<=h<=1."""
    h = np.asarray(init, dtype=float).copy()
    n = h.size
    target = n / 2.0

    # clip first
    h = np.clip(h, 0.0, 1.0)

    # adjust sum while respecting bounds using a simple water‑filling loop
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0)[0]
            if free.size == 0:
                break
            add = diff / free.size
            # do not exceed 1
            inc = np.minimum(add, 1.0 - h[free])
            h[free] += inc
        else:
            free = np.where(h > 0.0)[0]
            if free.size == 0:
                break
            sub = diff / free.size
            dec = np.maximum(sub, -h[free])
            h[free] += dec
    # final clip (tiny numerical drift)
    h = np.clip(h, 0.0, 1.0)
    return h

    h = np.full(n, 0.5, dtype=float)   # sum = 0.5*n = n/2, respects bounds
    return h

    # ------------------------------------------------------------------

        if time.time() - start_time > budget_s * 0.95:

        # pick two distinct indices

        # decide direction of transfer (+d from i1 to i2, -d the opposite)
        # compute feasible interval for d
        d_max = min(h[i1], 1.0 - h[i2])     # positive transfer amount
        d_min = -min(h[i2], 1.0 - h[i1])    # negative transfer amount
        if d_max <= d_min:                 # no feasible move (should not happen)
            i += 1
            continue

        h_new = h.copy()
        h_new[i1] -= d
