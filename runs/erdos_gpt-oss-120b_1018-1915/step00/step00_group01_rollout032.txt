import numpy as np, time, math

def compute_c5(h):
    """Return max overlap integral for discretized h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def random_move(h):
    """Perform a sum‑preserving random perturbation within [0,1]."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # delta added to i, subtracted from j
    delta_min = max(-h[i], h[j] - 1.0)          # keep both in [0,1]
    delta_max = min(1.0 - h[i], h[j])
    if delta_min >= delta_max:
        return h
    delta = np.random.uniform(delta_min, delta_max)
    h2 = h.copy()
    h2[i] += delta
    h2[j] -= delta
    return h2

def hill_climb(n, h0, time_budget):
    """Simple simulated‑annealing style hill climbing."""
    start = time.time()
    h_curr = h0.copy()
    best_h = h_curr.copy()
    best_c = compute_c5(best_h)
    T0 = 0.02                      # initial temperature
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / time_budget)
        h_new = random_move(h_curr)
        c_new = compute_c5(h_new)
        if c_new < best_c or np.random.rand() < math.exp(-(c_new - best_c) / (T + 1e-12)):
            h_curr = h_new
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
    return best_h, best_c

def upscale(h, new_n):
    """Linear interpolation of h defined on [0,2) to new_n points."""
    old_n = h.size
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    # enforce exact integral constraint
    target = new_n / 2.0
    h_new *= target / h_new.sum()
    h_new = np.clip(h_new, 0.0, 1.0)
    # final tiny correction
    diff = target - h_new.sum()
    if abs(diff) > 1e-12:
        h_new[0] += diff
        h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

        cur = h_init.sum()
        if abs(cur - target0) > 1e-12:

            diff = target0 - h_init.sum()
            h_init[0] += diff

    else:
