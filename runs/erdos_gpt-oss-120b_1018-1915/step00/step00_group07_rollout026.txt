import numpy as np
import cvxpy as cp
import time
import math

def _project_onto_feasible(v, target):
    """Project vector v onto the feasible set {0<=h<=1, sum(h)=target}."""
    n = v.shape[0]
    param = cp.Parameter(n)
    h = cp.Variable(n)
    constraints = [h >= 0, h <= 1, cp.sum(h) == target]
    prob = cp.Problem(cp.Minimize(cp.sum_squares(h - param)), constraints)
    param.value = v
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    # In rare numerical cases ECOS may not converge; fall back to clipping & scaling
    if h.value is None:
        w = np.clip(v, 0.0, 1.0)
        s = w.sum()
        if s == 0:
            w[0] = target
        else:
            w *= target / s
        w = np.clip(w, 0.0, 1.0)
        # simple redistribution if still off
        diff = target - w.sum()
        if abs(diff) > 1e-9:
            idx = np.where((w > 0) & (w < 1))[0]
            if idx.size:
                w[idx] += diff / idx.size
                w = np.clip(w, 0.0, 1.0)
        return w
    return np.array(h.value).ravel()


def _c5_of_h(h, dx):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr.max()


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of discretization points.

    The function performs a simple simulated‑annealing search starting from
    `initial_h_values` when available.
    """
    np.random.seed(seed)

    # Determine discretization size and initial vector
    try:
        init = np.array(initial_h_values, dtype=float)  # provided by harness
        n_points = init.size
    except Exception:
        n_points = 80
        init = np.full(n_points, 0.5)  # placeholder, will be projected

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure the start point is feasible
    h_current = _project_onto_feasible(init, target_sum)
    best_h = h_current.copy()
    best_score = _c5_of_h(best_h, dx)

    # Annealing parameters
    max_iter = 200000  # generous upper bound; loop will stop by time
    sigma0 = 0.3       # initial perturbation magnitude
    sigma_decay = 0.9995
    T0 = 1e-3          # initial temperature for Metropolis acceptance
    T_decay = 0.9995

    start_time = time.time()
    time_limit = budget_s * 0.95  # keep a safety margin

    sigma = sigma0
    T = T0
    iteration = 0

    while time.time() - start_time < time_limit and iteration < max_iter:
        # Propose a new candidate by adding Gaussian noise
        proposal = h_current + sigma * np.random.randn(n_points)

        # Project onto feasible region
        proposal = _project_onto_feasible(proposal, target_sum)

        # Evaluate
        score = _c5_of_h(proposal, dx)

        # Acceptance rule
        if score < best_score:
            best_h = proposal.copy()
            best_score = score
        if score < _c5_of_h(h_current, dx) or np.random.rand() < math.exp(-(score - _c5_of_h(h_current, dx)) / max(T, 1e-12)):
            h_current = proposal

        # Annealing schedule
        sigma *= sigma_decay
        T *= T_decay
        iteration += 1

    # Final safety check: ensure returned vector respects constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_onto_feasible(best_h, target_sum)

    return best_h, best_score, n_points
