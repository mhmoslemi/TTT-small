import time
import numpy as np

def _cost(h, dx):
    """Maximum overlap integral for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return np.max(corr)

def _project_onto_bounds_and_sum(x, total, lower=0.0, upper=1.0, max_iter=20, eps=1e-9):
    """
    Euclidean projection of x onto the set:
        { y | lower <= y_i <= upper, sum(y) = total }.
    Simple iterative clipping / redistribution scheme.
    """
    y = np.clip(x, lower, upper)
    for _ in range(max_iter):
        cur = y.sum()
        diff = total - cur
        if abs(diff) < eps:
            break
        # indices that are not at a bound can receive adjustments
        free = (y > lower + eps) & (y < upper - eps)
        if free.any():
            inc = diff / free.sum()
            y[free] += inc
        else:
            # all variables are at a bound; distribute uniformly then clip again
            y += diff / len(y)
        y = np.clip(y, lower, upper)
    # final correction if slight sum error remains
    cur = y.sum()
    if abs(cur - total) > eps:
        # apply a small uniform shift and re‑clip
        y += (total - cur) / len(y)
        y = np.clip(y, lower, upper)
    return y

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with values in [0,1] and ∫h=1
    that minimizes C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    np.random.seed(seed)

    # Determine problem size
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        # fallback: start with 120 points, average value 0.5 (integral 1)
        n = 120
        h0 = np.full(n, 0.5)

    total_sum = n / 2.0          # because dx = 2/n, integral = sum*h*dx = 1
    dx = 2.0 / n

    # Ensure the initial vector satisfies constraints
    h_best = _project_onto_bounds_and_sum(h0, total_sum)
    best_cost = _cost(h_best, dx)

    # Simple stochastic improvement loop
    # sigma (noise amplitude) shrinks over time
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:   # keep a safety margin
            break

        # schedule sigma: start ~0.3, decay to ~0.02
        progress = elapsed / budget_s
        sigma = 0.3 * (1.0 - progress) + 0.02

        cand = h_best + np.random.randn(n) * sigma
        cand = _project_onto_bounds_and_sum(cand, total_sum)

        cand_cost = _cost(cand, dx)
        if cand_cost < best_cost:
            h_best = cand
            best_cost = cand_cost

    return h_best.tolist(), float(best_cost), n
