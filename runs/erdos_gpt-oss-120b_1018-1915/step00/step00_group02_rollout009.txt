import numpy as np
import time
import math
import random

# the harness provides evaluate_erdos_solution and possibly initial_h_values
# we only use them as described.

def _project_to_feasible(h, target_sum, max_iter=10):
    """Clip to [0,1] and adjust to meet exact sum target."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        if abs(diff) < 1e-12:
            break
        # distribute uniformly; this keeps us in bounds after another clip
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction by scaling if still off (unlikely)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-10:
        # scale around mean while keeping bounds
        mean = cur_sum / h.size
        factor = target_sum / cur_sum if cur_sum > 0 else 1.0
        h = (h - mean) * factor + mean
        h = np.clip(h, 0.0, 1.0)
        # last uniform shift if needed
        cur_sum = h.sum()
        h += (target_sum - cur_sum) / h.size
        h = np.clip(h, 0.0, 1.0)
    return h

def _c5_of_h(h):
    """Return the overlap constant for a given discretised h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _initial_vector(n):
    """Construct a simple feasible start: half 1's, half 0's (spread evenly)."""
    h = np.zeros(n)
    # place ones evenly to satisfy sum=n/2
    ones = n // 2
    indices = np.linspace(0, n - 1, ones, dtype=int)
    h[indices] = 1.0
    # project just in case n odd
    target = n / 2.0
    h = _project_to_feasible(h, target)
    return h
