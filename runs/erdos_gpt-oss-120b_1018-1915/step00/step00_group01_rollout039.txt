import numpy as np
import time
import random
from math import isfinite

def _compute_c5(h, dx):
    """Return max overlap integral for step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _upscale(old_h, new_n):
    """Linearly interpolate old_h (defined on [0,2]) to length new_n."""
    old_n = len(old_h)
    if old_n == new_n:
        return old_h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # enforce sum constraint
    target_sum = new_n / 2.0
    cur_sum = new_h.sum()
    if cur_sum > 0:
        new_h *= target_sum / cur_sum
    else:
        new_h[:] = 1.0 / 2.0
    # clip to [0,1] in case scaling overshoots
    np.clip(new_h, 0.0, 1.0, out=new_h)
    # final renorm (tiny adjustment)
    cur_sum = new_h.sum()
    if cur_sum > 0:
        new_h *= target_sum / cur_sum
    return new_h
