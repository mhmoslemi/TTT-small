import numpy as np
import time
from scipy import optimize

def _evaluate(h):
    """Return the C5 value for a discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(corr)


def _project(h, total):
    """Clip h to [0,1] and adjust its sum to equal total."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        return np.full_like(h, total / h.size)
    # First add a constant to hit the target sum, then re‑clip.
    h += (total - s) / h.size
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - total) > 1e-8:
        # Final proportional scaling (preserves bounds because total>0)
        h *= total / s
        h = np.clip(h, 0.0, 1.0)
    return h


def _upscale(h, new_n):
    """Linear interpolation of h onto a finer grid of size new_n."""
    old_n = h.shape[0]
    xs_old = np.linspace(0, 2, old_n, endpoint=False)
    xs_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(xs_new, xs_old, h)
    return _project(h_new, new_n / 2.0)


def _local_search(h0, budget, start_time):
    """
    Simple stochastic hill‑climbing on a fixed resolution.
    Returns best (h, C5) found within the allocated time.
    """
    n = h0.shape[0]
    total = n / 2.0
    best_h = _project(h0.copy(), total)
    best_c = _evaluate(best_h)

    # Adaptive step size
    step = 0.05
    patience = 0
    while time.time() - start_time < budget:
        cand = best_h + np.random.randn(n) * step
        cand = _project(cand, total)
        c = _evaluate(cand)
        if c < best_c:
            best_c, best_h = c, cand
            step = max(step * 0.9, 1e-4)
            patience = 0
        else:
            patience += 1
            if patience > 200:
                step = min(step * 1.2, 0.2)
                patience = 0
    return best_h, best_c
