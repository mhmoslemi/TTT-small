import time, numpy as np, math, random
from typing import Tuple

def _evaluate(h: np.ndarray) -> float:
    """Return the C5 value for a candidate h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(conv.max())

def _repair_bounds(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and return a copy."""
    return np.clip(h, 0.0, 1.0)

def _repair_sum(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Adjust h so that its sum equals target_sum while staying in [0,1]."""
    h = _repair_bounds(h)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # Distribute diff over non‑saturated entries
    mask = (h > 0.0) & (h < 1.0)
    if mask.any():
        h[mask] += diff / mask.sum()
        h = _repair_bounds(h)
        # if still off due to clipping, fall back to scaling
        if abs(h.sum() - target_sum) > 1e-9:
            h *= target_sum / h.sum()
    else:
        # all entries at bounds; scale uniformly
        h *= target_sum / cur
    return h

def _perturb(h: np.ndarray, max_frac: float = 0.3) -> np.ndarray:
    """Transfer a random amount of mass between two positions."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # amount that can be moved from i to j without breaking bounds
    max_move = min(h[i], 1.0 - h[j])
    if max_move <= 0:
        return h.copy()
    delta = max_move * max_frac * np.random.rand()
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def _simulated_annealing(start_h: np.ndarray,
                         budget: float,
                         init_temp: float = 1.0,
                         cooling: float = 0.995) -> Tuple[np.ndarray, float]:
    """Run a simple SA loop within the supplied time budget (seconds)."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    h = _repair_sum(_repair_bounds(start_h), target_sum)
    best_h = h.copy()
    best_score = _evaluate(best_h)
    temp = init_temp
    t0 = time.time()
    while time.time() - t0 < budget:
        # a small batch of proposals per temperature
        for _ in range(50):
            h_new = _perturb(h)
            h_new = _repair_sum(h_new, target_sum)
            score = _evaluate(h_new)
            if score < best_score or math.exp((best_score - score) / max(temp, 1e-12)) > np.random.rand():
                h = h_new
                if score < best_score:
                    best_score = score
                    best_h = h_new.copy()
        temp = max(temp * cooling, 1e-5)
    return best_h, best_score
