import time
import numpy as np

def _compute_c5(h, dx):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for discrete h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _mutate(h):
    """Perform a small transfer mutation preserving sum and bounds."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # decide direction: move amount from i to j
    if np.random.rand() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    # maximum transferable amount without violating bounds
    max_transfer = min(h[src], 1.0 - h[dst])
    if max_transfer <= 0:
        return h  # nothing to move
    delta = np.random.rand() * max_transfer * 0.5  # limit step size
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new
