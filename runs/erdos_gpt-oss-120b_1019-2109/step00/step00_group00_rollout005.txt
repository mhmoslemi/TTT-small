def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_feasible(n, target_sum):
    # fill with ones then remainder
    h = np.zeros(n)
    full = int(np.floor(target_sum))
    h[:full] = 1.0
    remainder = target_sum - full
    if full < n:
        h[full] = remainder
    # random permutation
    rng = np.random.default_rng()
    rng.shuffle(h)
    return h

import time
import math
import random
import numpy as np

def _corr_h_1mh(h):
    """Correlation used for the overlap integral."""
    n = h.shape[0]
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def _c5_from_corr(corr):
    return float(np.max(corr))

def _random_feasible(n, rng):
    """Return a feasible vector of length n (sum = n/2, values in [0,1])."""
    target = n / 2.0
    full = int(math.floor(target))
    h = np.zeros(n, dtype=np.float64)
    h[:full] = 1.0
    if full < n:
        h[full] = target - full
    rng.shuffle(h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that yields a low
    value of C5 = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise starting vector
    # ------------------------------------------------------------------
    try:
        init_h = initial_h_values  # provided by the harness, if any
    except NameError:
        init_h = None

    if init_h is not None:
        h = np.array(init_h, dtype=np.float64).copy()
        n = h.shape[0]
        target_sum = n / 2.0
        # tiny re‑normalisation in case of rounding drift
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            idx = rng.integers(n)
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    else:
        n = 45
        target_sum = n / 2.0
        h = _random_feasible(n, rng)

    # Ensure feasibility (might be slightly off because of rounding)
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        # distribute the tiny error uniformly
        h += (target_sum - s) / n

    best_h = h.copy()
    corr = _corr_h_1mh(h)
    best_c5 = _c5_from_corr(corr)

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    start = time.time()
    limit = budget_s * 0.96  # leave a small safety margin
    init_temp = 0.05
    final_temp = 1e-4
    init_step = 0.25
    final_step = 0.02

    # we keep the current state separately; it will be re‑initialised
    # after each successful restart.
    while time.time() - start < limit:
        # ---- start a new local search ---------------------------------
        # 20% of the time start from a slight perturbation of the best so far,
        # otherwise start from a fresh random feasible point.
        if rng.random() < 0.2:
            cur = best_h.copy()
            # apply a few small random pair‑moves to diversify
            for _ in range(5):
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                step = init_step * 0.5
                delta_i_min = max(-step, -cur[i])
                delta_i_max = min(step, 1.0 - cur[i])
                delta_j_min = cur[j] - 1.0
                delta_j_max = cur[j]
                lo = max(delta_i_min, delta_j_min)
                hi = min(delta_i_max, delta_j_max)
                if lo >= hi:
                    continue
                d = rng.uniform(lo, hi)
                cur[i] += d
                cur[j] -= d
        else:
            cur = _random_feasible(n, rng)

        cur_corr = _corr_h_1mh(cur)
        cur_c5 = _c5_from_corr(cur_corr)

        # ---- inner optimisation loop ----------------------------------
        it_without_improve = 0
        max_no_improve = 300  # stop early if stuck
        while time.time() - start < limit:
            elapsed = time.time() - start
            frac = elapsed / limit
            temp = init_temp * (1 - frac) + final_temp * frac
            step = init_step * (1 - frac) + final_step * frac

            # choose move type: 60% targeted, 40% random pair
            if rng.random() < 0.6:
                # ---------- targeted move (reduce the current worst lag) ----------
                k_max = int(np.argmax(cur_corr))
                lag = k_max - (n - 1)                # shift corresponding to max overlap
                if lag >= 0:
                    i_range = range(0, n - lag)
                else:
                    i_range = range(-lag, n)          # lag negative => shift left
                # metric favouring large h_i and small h_{i+lag}
                best_metric = -1.0
                best_i = None
                for i in i_range:
                    j = i + lag
                    metric = cur[i] + (1.0 - cur[j])      # proportional to possible reduction
                    if metric > best_metric:
                        best_metric = metric
                        best_i = i
                if best_i is None:
                    continue
                i = best_i
                j = i + lag
                # admissible delta respecting bounds
                max_dec_i = min(step, cur[i])
                max_inc_j = min(step, 1.0 - cur[j])
                delta = min(max_dec_i, max_inc_j)
                if delta <= 0:
                    continue
                # perform move
                new = cur.copy()
                new[i] -= delta
                new[j] += delta
            else:
                # ---------- random pair move ----------
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1
                delta_i_min = max(-step, -cur[i])
                delta_i_max = min(step, 1.0 - cur[i])
                delta_j_min = cur[j] - 1.0
                delta_j_max = cur[j]
                lo = max(delta_i_min, delta_j_min)
                hi = min(delta_i_max, delta_j_max)
                if lo >= hi:
                    continue
                delta = rng.uniform(lo, hi)
                new = cur.copy()
                new[i] += delta
                new[j] -= delta

            # evaluate new candidate
            new_corr = _corr_h_1mh(new)
            new_c5 = _c5_from_corr(new_corr)

            # acceptance rule (simulated annealing)
            improve = new_c5 < cur_c5 - 1e-12
            if improve or rng.random() < math.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
                cur = new
                cur_corr = new_corr
                cur_c5 = new_c5
                it_without_improve = 0
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    best_h = new.copy()
            else:
                it_without_improve += 1

            if it_without_improve >= max_no_improve:
                break

        # occasional re‑normalisation of the best solution (tiny drift)
        diff = target_sum - best_h.sum()
        if abs(diff) > 1e-12:
            idx = np.argmin(best_h) if diff > 0 else np.argmax(best_h)
            best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)

    # final evaluation
    final_corr = _corr_h_1mh(best_h)
    final_c5 = _c5_from_corr(final_corr)

    # ensure strict feasibility before returning
    s = best_h.sum()
    if abs(s - target_sum) > 1e-10:
        best_h += (target_sum - s) / n
        best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h, float(final_c5), n)
