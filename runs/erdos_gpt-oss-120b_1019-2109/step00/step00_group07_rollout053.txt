def upsample_h(initial, N_target):
    N0 = len(initial)
    x0 = np.linspace(0, 2, N0, endpoint=False) # or inclusive? We'll use same domain discretization for original: suppose original dx0 = 2/N0.
    x_target = np.linspace(0, 2, N_target, endpoint=False)
    h_target = np.interp(x_target, x0, initial, left=initial[0], right=initial[-1]) # linear interpolation.
    # ensure within [0,1]
    h_target = np.clip(h_target, 0, 1)
    # adjust sum to target = N_target/2
    total = h_target.sum()
    target = N_target / 2
    diff = target - total
    # distribute diff uniformly among indices not at bound in direction of diff.
    if abs(diff) > 1e-12:
        # find indices where adding diff possible:
        if diff > 0:
            candidates = np.where(h_target < 1)[0]
        else:
            candidates = np.where(h_target > 0)[0]
        if len(candidates) > 0:
            inc = diff / len(candidates)
            # ensure not exceed bounds:
            for idx in candidates:
                h_target[idx] += inc
            # clip after
            h_target = np.clip(h_target, 0, 1)
            # recompute sum and adjust small residual:
            total2 = h_target.sum()
            residual = target - total2
            # apply small correction more precisely:
            if abs(residual) > 1e-12:
                # Add residual to first variable that can be adjusted
                for idx in candidates:
                    max_add = (1 - h_target[idx]) if residual > 0 else -h_target[idx]
                    delta = np.sign(residual) * min(abs(residual), abs(max_add))
                    h_target[idx] += delta
                    residual -= delta
                    if abs(residual) < 1e-12:
                        break
    # final clipping ensure sum close
    # Might still slight off; we accept minor floating error.
    return h_target

def shape_h(a, N):
    dx = 2.0/N
    x = (np.arange(N) + 0.5) * dx  # midpoints
    # compute h
    # ramp region: 0<=x<a => linear increase from 0 to M # note a is actual length
    # plateau region: a <= x <= 2 - a => M
    # descending region: x > 2 - a => linear decreasing to 0.
    M = 1.0 / (2.0 - a)  # as derived
    h = np.empty(N, dtype=np.float64)
    # left ramp
    left = x < a
    h[left] = M * (x[left] / a) if a > 0 else 0.0

    mid = (x >= a) & (x <= 2 - a)

    # If a==0, M=0.5 and plateau region entire interval (x>=0 and <=2), use M.
    # For a close to zero we must avoid division by zero.
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np

def compute_c5(arr: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * dx)


def enforce_sum(vec: np.ndarray, target: float) -> np.ndarray:
    """Adjust a single entry so that sum(vec)=target while staying in [0,1]."""
    diff = target - vec.sum()
    if abs(diff) < 1e-12:
        return vec
    if diff > 0:
        cand = np.where(vec < 1.0 - 1e-12)[0]
    else:
        cand = np.where(vec > 1e-12)[0]
    if cand.size == 0:  # fallback – should not happen
        cand = np.arange(vec.size)
    i = cand[0]
    vec[i] = np.clip(vec[i] + diff, 0.0, 1.0)
    return vec


def shape_h(n: int, a: float) -> np.ndarray:
    """
    Symmetric piecewise‑linear shape.
    a – width of each ramp (0 ≤ a ≤ 1).
    """
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx  # mid‑points
    if a < 1e-12:
        h = np.full(n, 0.5, dtype=np.float64)
    else:
        M = 1.0 / (2.0 - a)          # peak height
        h = np.empty(n, dtype=np.float64)
        left = x < a
        h[left] = M * x[left] / a
        right = x > 2.0 - a
        h[right] = M * (2.0 - x[right]) / a
        middle = (~left) & (~right)
        h[middle] = M
    h = np.clip(h, 0.0, 1.0)
    # enforce exact integral = 1  (i.e. sum = n/2)
    target = n / 2.0
    return enforce_sum(h, target)


def upsample_initial(init: np.ndarray, n: int) -> np.ndarray:
    """Linear up‑sampling of a given construction to length n."""
    m = init.shape[0]
    if m == n:
        return enforce_sum(init.copy(), n / 2.0)
    x0 = np.linspace(0.0, 2.0, m, endpoint=False)
    x1 = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(x1, x0, init, left=0.0, right=0.0)
    h = np.clip(h, 0.0, 1.0)
    return enforce_sum(h, n / 2.0)


def initial_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a feasible starting vector (use parent if present, else scan shapes)."""
    init = globals().get("initial_h_values")
    if init is not None:
        return upsample_initial(np.asarray(init, dtype=np.float64), n)

    # no parent – search the one‑parameter family for a good start
    best_h = None
    best_val = np.inf
    for a in np.linspace(0.0, 1.0, 101):
        h = shape_h(n, a)
        val = compute_c5(h)
        if val < best_val:
            best_val, best_h = val, h.copy()
    # refine around the best a
    a0 = np.linspace(0.0, 1.0, 101)[np.argmin([compute_c5(shape_h(n, a)) for a in np.linspace(0.0, 1.0, 101)])]
    lo, hi = max(0.0, a0 - 0.02), min(1.0, a0 + 0.02)
    for a in np.linspace(lo, hi, 81):
        h = shape_h(n, a)
        val = compute_c5(h)
        if val < best_val:
            best_val, best_h = val, h.copy()
    return best_h


def propose_move(cur: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Generate a new feasible vector by moving mass between two coordinates."""
    n = cur.shape[0]
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    # decide direction
    if rng.random() < 0.5:
        # from j to i
        max_transfer = min(1.0 - cur[i], cur[j])
        if max_transfer <= 0.0:
            return cur
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[i] += delta
        new[j] -= delta
    else:
        # from i to j
        max_transfer = min(1.0 - cur[j], cur[i])
        if max_transfer <= 0.0:
            return cur
        delta = rng.random() * max_transfer
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
    return new


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    n_points = 200                      # a convenient size (< 1 k)
    target_sum = n_points / 2.0

    # ---- initial construction ------------------------------------------------
    cur_h = initial_candidate(n_points, rng)
    cur_val = compute_c5(cur_h)
    best_h, best_val = cur_h.copy(), cur_val

    start = time.time()
    end_time = start + budget_s * 0.95   # keep a safety margin

    # SA parameters
    T0 = 0.015
    T_min = 1e-6

    while time.time() < end_time:
        # linear temperature schedule according to elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / (budget_s * 0.95)
        T = max(T_min, T0 * (1.0 - frac))

        # propose a neighbour
        cand_h = propose_move(cur_h, rng)

        # occasional big jump to diversify
        if rng.random() < 0.001:
            cand_h = initial_candidate(n_points, rng)

        cand_val = compute_c5(cand_h)

        # Metropolis acceptance
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_h, cur_val = cand_h, cand_val
            if cand_val < best_val:
                best_h, best_val = cand_h.copy(), cand_val

    # final safety projection (tiny numerical drift)
    best_h = enforce_sum(best_h, target_sum)

    # final evaluation
    final_val = compute_c5(best_h)

    return best_h.tolist(), float(final_val), n_points
