def mismatches(h, shift):
    if shift >= 0:
        left = np.arange(0, n-shift)
        right = left + shift
    else:
        shift_abs = -shift
        left = np.arange(shift_abs, n)
        right = left - shift_abs
    # left corresponds to i, right to i+shift
    # mismatches where left=1 and right=0:
    mask1 = (h[left]==1) & (h[right]==0)
    # mismatches in opposite direction left=0 and right=1
    mask2 = (h[left]==0) & (h[right]==1)
    idx1 = left[mask1]
    idx2 = left[mask2]
    return idx1, idx2

import time
import math
import random
import numpy as np


def _c5_value(h: np.ndarray) -> float:
    """Return the C₅ value of a binary step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full").max() * dx


def _hill_climb(start_h: np.ndarray, time_budget: float, seed: int) -> tuple:
    """
    Simple stochastic hill‑climb / simulated‑annealing on binary vectors.
    Returns the best vector found and its C₅ value.
    """
    np.random.seed(seed)
    random.seed(seed)

    n = start_h.shape[0]
    dx = 2.0 / n

    cur_h = start_h.copy()
    cur_val = _c5_value(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    start_time = time.time()
    max_temp = 0.025  # start temperature for occasional uphill moves

    while time.time() - start_time < time_budget:
        elapsed = time.time() - start_time
        temperature = max_temp * (1.0 - elapsed / time_budget)

        # correlation of the current vector – used to locate the worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        O = corr * dx
        max_idx = int(O.argmax())
        max_val = float(O[max_idx])
        shift = max_idx - (n - 1)

        # indices that cause the maximal overlap for this shift
        if shift >= 0:
            left = np.arange(0, n - shift, dtype=int)
            right = left + shift
        else:
            left = np.arange(-shift, n, dtype=int)
            right = left + shift

        mism1 = left[(cur_h[left] == 1) & (cur_h[right] == 0)]
        mism2 = left[(cur_h[left] == 0) & (cur_h[right] == 1)]

        if mism1.size and mism2.size:
            i = int(np.random.choice(mism1))
            j = int(np.random.choice(mism2))
        else:
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))

        # try the swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_val = _c5_value(cur_h)

        accept = False
        if new_val <= cur_val:
            accept = True
        elif temperature > 1e-12:
            prob = math.exp(-(new_val - cur_val) / temperature)
            accept = random.random() < prob

        if accept:
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert the swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        # occasional tiny random perturbation to shake out of plateaus
        if random.random() < 0.001:
            for _ in range(3):

                z = np.where(cur_h == 0)[0]

                j2 = int(np.random.choice(z))
                cur_h[i2], cur_h[j2] = cur_h[j2], cur_h[i2]
            cur_val = _c5_value(cur_h)

    return best_h, best_val
