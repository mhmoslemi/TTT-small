    n = arr.shape[0]
    # full correlation between h and (1-h); max value scaled by dx = 2/n
    return np.max(np.correlate(arr, 1.0 - arr, mode="full")) * (2.0 / n)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function with a small Erdős overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,) with entries in [0,1]
        c5_bound : float, the maximal overlap for the returned h_values
        n_points : int, length of the discretisation (2 / n_points = dx)
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # -------------------------------------------------
    # Initialise problem size and starting configuration
    # -------------------------------------------------
    if "initial_h_values" in globals():
        # Use the supplied values as a guide: keep the largest half as 1,
        # the rest as 0 (ensures the integral constraint).
        init_arr = np.asarray(initial_h_values, dtype=np.float64).ravel()
        n = int(init_arr.size)
        target_ones = n // 2
        # sort indices by descending value, break ties arbitrarily
        order = np.argsort(-init_arr)
        h_cur = np.zeros(n, dtype=np.float64)
        h_cur[order[:target_ones]] = 1.0
    else:
        # Fallback: choose a modest size and a random binary start.
        n = 200  # < 1000 as required
        target_ones = n // 2
        h_cur = np.zeros(n, dtype=np.float64)
        h_cur[rng.choice(n, size=target_ones, replace=False)] = 1.0

    # Ensure exact integral (may be off by 1 due to rounding)
    cur_sum = int(np.sum(h_cur))
    target_sum = n // 2
    if cur_sum != target_sum:
        diff = target_sum - cur_sum
        if diff > 0:
            zeros = np.flatnonzero(h_cur == 0)
            flip = rng.choice(zeros, size=diff, replace=False)
            h_cur[flip] = 1.0
        else:
            ones = np.flatnonzero(h_cur == 1)
            flip = rng.choice(ones, size=-diff, replace=False)
            h_cur[flip] = 0.0

    best_h = h_cur.copy()
    best_cost = _eval_cost(best_h)
    cur_cost = best_cost

    # -------------------------------------------------
    # Simulated annealing on binary swaps
    # -------------------------------------------------
    T = 0.5                     # initial temperature
    cooling = 0.99995           # geometric cooling factor
    restart_limit = 20000       # steps without improvement before restart
    no_improve = 0
    max_iters = 5_000_000       # safety cap, real stop is time budget

    it = 0
    while True:
        # Time check
        if time.time() - start_time > budget_s * 0.97:
            break
        if it >= max_iters:
            break

        # Propose a swap of a 1‑position with a 0‑position
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j or h_cur[i] == h_cur[j]:
            # try a few more times; give up if unsuccessful
            for _ in range(3):
                i = rng.integers(n)
                j = rng.integers(n)
                if i != j and h_cur[i] != h_cur[j]:
                    break
            else:
                it += 1
                continue

        new_h = h_cur.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]   # swap entries

        new_cost = _eval_cost(new_h)
        delta = new_cost - cur_cost

        # Metropolis acceptance
        if delta <= 0.0 or rng.random() < math.exp(-delta / T):
            h_cur = new_h
            cur_cost = new_cost
            if new_cost < best_cost - 1e-12:
                best_h = new_h.copy()
                best_cost = new_cost
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Cool down
        T *= cooling

        # Random restart if stuck for a while
        if no_improve > restart_limit:
            h_cur = np.zeros(n, dtype=np.float64)
            h_cur[rng.choice(n, size=target_sum, replace=False)] = 1.0
            cur_cost = _eval_cost(h_cur)
            T = 0.5
            no_improve = 0

        it += 1

    # -------------------------------------------------
    # Greedy local refinement (pure downhill) with remaining time
    # -------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 0.5:
        # Number of greedy attempts roughly proportional to remaining time
        attempts = int(min(300_000, remaining * 5000))
        h_ref = best_h.copy()
        cost_ref = best_cost
        for _ in range(attempts):
            i = rng.integers(n)
            j = rng.integers(n)
            if h_ref[i] == h_ref[j]:
                continue
            # try swap
            h_ref[i], h_ref[j] = h_ref[j], h_ref[i]
            new_c = _eval_cost(h_ref)
            if new_c < cost_ref - 1e-12:
                cost_ref = new_c
                if new_c < best_cost:
                    best_h = h_ref.copy()
                    best_cost = new_c
                # keep swap
            else:
                # revert
                h_ref[i], h_ref[j] = h_ref[j], h_ref[i]

    # Final bound using the provided helper (if present)
    if "evaluate_erdos_solution" in globals():
        final_c5 = float(evaluate_erdos_solution(best_h))
    else:
        final_c5 = float(best_cost)

    return (best_h, final_c5, n)
