def project_onto_simplex_box(v, target_sum):
    # v: initial values
    # Bound lower bound 0, upper bound 1.
    # Want to find w s.t. w_i ∈ [0,1], sum w_i = target_sum, and minimize ||w - v||^2.
    # Equivalent to Euclidean projection onto bounded simplex.
    # Use iterative method: Start with w=v, clip to [0,1].
    # Then if sum w != target_sum, adjust the free variables (those not at bounds) by adding lambda.
    # Solve lambda such that sum_i min(1, max(0, v_i+lambda)) = target_sum.
    # This is monotonic in lambda, so we can bisection between lower = -max(v) to upper = 1 - min(v).
    lo = -max(v) - 1  # ensure enough
    hi = 1 - min(v) + 1
    for _ in range(50):
        mid = (lo + hi) / 2
        w = np.clip(v + mid, 0, 1)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo+hi)/2, 0, 1)
    # ensure sum exactly maybe adjust small residual by distributing among non-bound entries.
    diff = target_sum - w.sum()

        mask = (w > 0) & (w < 1)
        if mask.sum() > 0:
            w[mask] += diff / mask.sum()
            # Clip again (should maintain within bound)
            w = np.clip(w, 0, 1)
    # After final adjust may still not exactly sum but close enough
    return w

def project_to_constraints(v, target):
    # Euclidean projection onto simplex with upper bound 1.
    # Solve w = clip(v + λ, 0, 1) such that sum w = target.
    # Use bisection on λ.
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # after bisection sum may be slightly off; correct by distributing residual
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        # Distribute diff equally among non-bound entries
        free = (w > 0) & (w < 1)
        free_idx = np.where(free)[0]
        if free_idx.size > 0:
            # Add diff equally
            add = diff / free_idx.size
            w[free_idx] += add
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback: adjust any index not at bound arbitrarily
            # find any index with w < 1 if need to increase, else w >0
            if diff > 0:
                # increase smallest value
                idx = np.argmin(w)
                w[idx] = min(1.0, w[idx] + diff)
            else:
                idx = np.argmax(w)
                w[idx] = max(0.0, w[idx] + diff)
    return w

import time
import random
import numpy as np

def _project(v, target):
    """Euclidean projection of v onto {0≤x≤1, sum x = target}."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        free = (w > 0.0) & (w < 1.0)
        idx = np.where(free)[0]
        if idx.size:
            add = diff / idx.size
            w[idx] += add
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback: adjust a single coordinate
            if diff > 0:
                i = np.argmin(w)
                w[i] = min(1.0, w[i] + diff)
            else:
                i = np.argmax(w)
                w[i] = max(0.0, w[i] + diff)
    return w

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)

    # ---- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
        n = h.size
    else:
        n = 70
        h = np.full(n, 0.5, dtype=np.float64)
        h += np.random.uniform(-0.2, 0.2, size=n)
        h = np.clip(h, 0.0, 1.0)
        h = _project(h, n / 2.0)

    target_sum = n / 2.0
    dx = 2.0 / n

    def c5(val):
        return np.max(np.correlate(val, 1 - val, mode='full')) * dx

    cur_c5 = c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    start = time.time()
    # ------------------------------------------------------------------------
    while time.time() - start < budget_s * 0.97:          # safety margin
        # identify the worst shift
        corr_full = np.correlate(h, 1 - h, mode='full')
        max_idx = int(np.argmax(corr_full))
        s = max_idx - (n - 1)               # signed shift
        s_abs = abs(s)

        # choose indices for a local mass‑transfer move
        if s_abs == 0 or s_abs >= n:
            # fallback: random pairwise transfer
            i, j1, j2 = random.sample(range(n), 3)
            i2 = (i + 1) % n
        else:
            # indices that can be increased for this shift
            cand_i = np.arange(n - s_abs)
            up = np.minimum(1.0 - h[cand_i], 1.0 - h[cand_i + s_abs])
            good = np.where(up > 1e-6)[0]
            if good.size:
                idx = random.choice(good)
                i = int(cand_i[idx])
            else:
                i = random.randint(0, n - s_abs - 1)
            i2 = i + s_abs

            # two dec. indices not touching i,i2
            mask = (np.arange(n) != i) & (np.arange(n) != i2) & (h > 1e-6)
            dec_idx = np.where(mask)[0]
            if dec_idx.size >= 2:
                j1, j2 = random.sample(dec_idx.tolist(), 2)
            else:
                # any distinct pair
                rest = [idx for idx in range(n) if idx not in (i, i2)]
                j1, j2 = random.sample(rest, 2)

        # maximal admissible transfer
        delta = min(1.0 - h[i], 1.0 - h[i2], h[j1], h[j2])
        if delta < 1e-8:
            # very small, skip this iteration
            continue
        delta *= random.random() * 0.5    # random fraction

        h_new = h.copy()
        h_new[i]   += delta
        h_new[i2]  += delta
        h_new[j1]  -= delta
        h_new[j2]  -= delta

        # tiny numerical drift – re‑project occasionally
        if abs(h_new.sum() - target_sum) > 1e-8:
            h_new = _project(h_new, target_sum)

        new_c5 = c5(h_new)

        # accept if it improves the current state
        if new_c5 <= cur_c5:
            h = h_new
            cur_c5 = new_c5

        # keep the global best
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()

        # occasional global random restart (escape)
        if random.random() < 0.001:
            rnd = np.random.rand(n)
            h = _project(rnd, target_sum)
            cur_c5 = c5(h)

    # final clean‑up
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project(best_h, target_sum)
    final_c5 = c5(best_h)

    return best_h, float(final_c5), n
