def project_to_bounds_and_sum(v, target):
    # ensures each entry within 0,1 and sum = target.
    # v can be any real numbers.
    # Use simple iterative projection (like compute λ such that sum(np.clip(v - λ,0,1)) = target)
    # Use bisection for λ in [min(v-1), max(v)] maybe.
    lo = -1.0
    hi = 1.0
    # Actually λ is shift to subtract, each w_i = min(max(v_i - λ, 0), 1)
    # We'll find λ such that sum = target.
    # We'll set initial bounds: λ_lo = -max(v) (so v_i - λ always >1) leading to w_i=1; sum = n. λ_hi = max(v) (so v_i - λ <= 0) leading sum 0.
    lo = -np.max(v) - 1  # ensure w_i =1
    hi = np.max(v) + 1   # ensure w_i =0
    for _ in range(60):
        mid = (lo + hi) / 2
        w = np.clip(v - mid, 0, 1)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo+hi)/2, 0, 1)
    # after rounding errors may not precisely meet sum, adjust small residual
    residual = target - w.sum()
    if abs(residual) > 1e-8:
        # distribute residual to non-bound entries (if any)
        free = (w > 1e-12) & (w < 1 - 1e-12)
        if free.any():
            # distribute equally
            add = residual / free.sum()
            w[free] += add
            # clip again
            w = np.clip(w, 0, 1)
        else:
            # fallback: adjust one element
            idx = np.argmin(w) if residual > 0 else np.argmax(w)
            w[idx] = np.clip(w[idx] + residual, 0, 1)
    return w

import time
import random
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for discrete step h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def project_to_bounds_and_sum(v, target):
    """
    Projection of v onto the feasible set:
        0 ≤ w_i ≤ 1,   Σ w_i = target
    Uses a simple water‑filling (bisection on the Lagrange multiplier).
    """
    # bisection on λ in w_i = clip(v_i - λ, 0, 1)
    lo = -np.max(v) - 1.0   # makes all w_i = 1
    hi = np.max(v) + 1.0    # makes all w_i = 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    resid = target - w.sum()
    if abs(resid) > 1e-12:
        free = (w > 1e-12) & (w < 1.0 - 1e-12)
        if free.any():
            w[free] += resid / free.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            idx = 0 if resid > 0 else -1
            w[idx] = np.clip(w[idx] + resid, 0.0, 1.0)
    return w

# Gradient based refinement (moves mass from a positive‑gradient
# index to a negative‑gradient index for the currently worst shift)
# --------------------------------------------------------------

def gradient_refine(h, time_limit):
    n = h.size
    cur = h.copy()
    best = cur.copy()
    best_val = compute_c5(cur)
    cur_val = best_val
    start = time.time()
    step_factor = 1.0
    it = 0
    while time.time() - start < time_limit:
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)          # integer offset k
        # gradient of the worst term
        g = np.zeros(n, dtype=float)
        if shift >= 0:
            s = shift
            g[: n - s] += (1.0 - cur[s:])
            g[s:]      -= cur[: n - s]
        else:
            s = -shift
            g[s:]      += (1.0 - cur[: n - s])
            g[: n - s] -= cur[s:]
        p = int(np.argmax(g))   # removal candidate (largest positive gradient)
        q = int(np.argmin(g))   # addition candidate (most negative gradient)
        diff = g[p] - g[q]
        if diff <= 1e-12:
            break                     # no improving direction for this shift
        max_delta = min(cur[p], 1.0 - cur[q])
        if max_delta <= 0:
            break
        delta = max_delta * step_factor * 0.5   # take a safe fraction
        if delta < 1e-12:
            break
        # trial move
        cur[p] -= delta
        cur[q] += delta
        new_val = compute_c5(cur)
        if new_val < best_val - 1e-12:
            best_val = new_val
            best = cur.copy()
        if new_val < cur_val - 1e-12:
            cur_val = new_val
        else:
            # revert and shrink step
            cur[p] += delta
            cur[q] -= delta
            step_factor *= 0.7
            if step_factor < 1e-4:
                break
        it += 1
        if it % 20 == 0:
            step_factor = max(step_factor * 0.9, 0.01)
    return best, best_val

# Simple hill‑climbing with random pairwise transfers
# --------------------------------------------------------------

def hill_climb(h, max_iter, time_limit=None):
    n = h.size
    cur = h.copy()
    cur_val = compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    start = time.time()
    it = 0
    while it < max_iter:
        if time_limit is not None and (time.time() - start) >= time_limit:
            break
        # pick two distinct indices
        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1
        # move mass from i to j (or opposite if not feasible)
        max_delta = min(cur[i], 1.0 - cur[j])
        if max_delta <= 0:
            max_delta = min(cur[j], 1.0 - cur[i])
            if max_delta <= 0:
                it += 1
                continue
            i, j = j, i
        # random fraction of the allowed amount
        delta = max_delta * random.random() * 0.5
        if delta <= 0:
            it += 1
            continue
        cur[i] -= delta
        cur[j] += delta
        new_val = compute_c5(cur)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best = cur.copy()
        else:
            # undo
            cur[i] += delta
            cur[j] -= delta
        it += 1
    return best, best_val
