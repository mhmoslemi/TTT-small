def project_onto_simplex(y, target_sum):
    lo = -np.max(y)  # ensure all y+lo <=0 -> all clipped to 0 => sum=0
    hi = 1 - np.min(y)  # ensure all y+hi >=1 -> all clipped to 1 => sum = n
    for _ in range(40):
        mid = (lo + hi)/2
        x = np.clip(y + mid, 0, 1)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo+hi)/2, 0, 1)
    return x

import numpy as np
import time

def _compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using full mode (zero padding)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def _project_to_sum(v, target):
    """Project v onto the set {x: 0<=x<=1, sum(x)=target}."""
    # Find lambda such that sum(clip(v+lambda,0,1)) = target
    lo = -np.max(v) - 2.0   # guarantees all clipped to 0
    hi = 1.0 - np.min(v) + 2.0  # guarantees all clipped to 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_shift(h, shift):
    """Gradient of C_shift = sum_i h_i (1-h_{i+shift}) w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift == 0:
        g = 1.0 - 2.0 * h
    else:
        lhs = n - shift
        g[:lhs] += (1.0 - h[shift:])          # derivative from left term
        g[shift:] -= h[:lhs]                 # derivative from right term
    return g


def run(seed=42, budget_s=1100, **_kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # 1. initialise h
    # ------------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=float)  # provided by harness
        n = h0.shape[0]
        target_sum = n * 0.5
        h = np.clip(h0, 0.0, 1.0)
        h = _project_to_sum(h, target_sum)
    except NameError:
        # no parent; start from a moderate size, binary half‑filled
        n = 120                     # even for exact half‑sum
        target_sum = n * 0.5
        h = np.full(n, 0.5)
        # add a tiny random jitter then project to exact sum
        h += rng.uniform(-0.01, 0.01, size=n)
        h = np.clip(h, 0.0, 1.0)
        h = _project_to_sum(h, target_sum)

    best_h = h.copy()
    best_c = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # 2. optimisation loop (projected gradient on the worst shift)
    # ------------------------------------------------------------------
    max_iters = 200000
    alpha = 0.12               # initial step size
    decay = 0.9995
    n = h.shape[0]
    target_sum = n * 0.5

    for it in range(max_iters):
        # time guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # current worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        shift = abs(shift)               # use absolute value (symmetry)

        # gradient of the offending shift
        grad = _gradient_shift(h, shift)

        # gradient step + projection
        y = h - alpha * grad
        y = np.clip(y, 0.0, 1.0)
        y = _project_to_sum(y, target_sum)

        # evaluate
        cur_c = _compute_c5(y)

        # keep best ever
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = y.copy()

        # accept if improvement, otherwise keep current (occasionally escape)
        if cur_c < _compute_c5(h) or rng.rand() < 0.001:
            h = y

        # decay step size
        alpha *= decay

        # occasional random shake‑up to avoid stagnation
        if (it + 1) % 4000 == 0:
            # small random mass transfer between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = min(0.05, h[i], 1.0 - h[j])
            if eps > 0:
                h[i] -= eps
                h[j] += eps
                h = _project_to_sum(h, target_sum)

    # final safety projection (ensure sum constraint)
    best_h = _project_to_sum(best_h, target_sum)
    best_c = _compute_c5(best_h)

    return best_h, float(best_c), int(n)
