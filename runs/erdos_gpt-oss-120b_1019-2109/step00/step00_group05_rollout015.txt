def gradient_for_shift(h, k):
    N = len(h)
    grad = np.zeros(N)
    if k >= 0:
        if k < N: # else grad zero
            grad[:N - k] += 1 - h[k:]
            grad[k:] -= h[:N - k]
    else:
        kk = -k
        if kk < N:
            grad[kk:] += 1 - h[:N - kk]
            grad[:N - kk] -= h[kk:]
    return grad

import numpy as np, time

def compute_corr(h, dx):
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_cost(h, dx):
    return np.max(compute_corr(h, dx))

def projection_bounds_sum(x, target):
    # binary search lambda and clip
    low = -1.0
    high = 1.0
    for _ in range(60):
        mid = (low + high) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = mid
        else:
            high = mid
    # final projection with mid average
    proj = np.clip(x - ((low+high)/2.0), 0.0, 1.0)
    # small tolerance adjust
    diff = proj.sum() - target
    if abs(diff) > 1e-9:
        mask = (proj > 0) & (proj < 1)
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] -= diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def gradient_for_shift(h, k):
    N = h.shape[0]
    grad = np.zeros(N, dtype=h.dtype)
    if k >= 0:
        if k < N:
            grad[:N - k] += 1.0 - h[k:]
            grad[k:] -= h[:N - k]
    else:
        kk = -k
        if kk < N:
            grad[kk:] += 1.0 - h[:N - kk]
            grad[:N - kk] -= h[kk:]
    return grad

def ramp_shape(N, b):
    dx = 2.0 / N
    a = 2.0 - b
    x = (np.arange(N) + 0.5) * dx # midpoints
    h = np.empty(N, dtype=float)

    mask0 = x <= a
    mask1 = x >= b
    mask_mid = (~mask0) & (~mask1)
    h[mask0] = 0.0
    h[mask1] = 1.0
    # linear part
    h[mask_mid] = (x[mask_mid] - a) / (b - a)
    return h

import time
import math
import numpy as np

def _project_bounds_sum(x: np.ndarray, target: float, low: float = 0.0, high: float = 1.0) -> np.ndarray:
    """Projection of x onto the box [low,high] with the additional linear constraint sum(x)=target."""
    # binary search for the Lagrange multiplier λ such that sum(clamp(x-λ,low,high)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, low, high)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    proj = np.clip(x - lam, low, high)
    # tiny correction to meet the sum exactly (helps with numerical noise)
    diff = proj.sum() - target
    if abs(diff) > 1e-12:
        mask = (proj > low) & (proj < high)
        cnt = mask.sum()
        if cnt:
            proj[mask] -= diff / cnt
            proj = np.clip(proj, low, high)
    return proj


def _cost(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral C5 for a given step function h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _grad_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of the overlap term for a single shift k (no dx factor)."""
    N = h.shape[0]
    g = np.zeros(N, dtype=h.dtype)
    if k >= 0:
        if k < N:
            g[: N - k] += 1.0 - h[k:]
            g[k:] -= h[: N - k]
    else:
        kk = -k
        if kk < N:
            g[kk:] += 1.0 - h[: N - kk]
            g[: N - kk] -= h[kk:]
    return g
