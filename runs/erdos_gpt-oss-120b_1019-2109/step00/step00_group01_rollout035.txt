def adjust_sum(h):
    n = len(h)
    target = n/2
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    i = n // 2
    new_val = h[i] + diff
    new_val = min(max(new_val, 0.0), 1.0)
    # compute new diff after clamping
    diff2 = diff - (new_val - h[i])
    h[i] = new_val
    if abs(diff2) > 1e-12:
        # recursively adjust other index
        # find another index not at bounds
        for j in range(n):
            if diff2 == 0:
                break
            if diff2 > 0 and h[j] < 1.0:
                inc = min(diff2, 1.0 - h[j])
                h[j] += inc
                diff2 -= inc
            elif diff2 < 0 and h[j] > 0.0:
                dec = min(-diff2, h[j])
                h[j] -= dec
                diff2 += dec
    return h

import time
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """max_{k} ∫ h(x)(1‑h(x+k)) dx approximated on a uniform grid."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * (2.0 / h.size))

def _adjust_sum(h: np.ndarray) -> np.ndarray:
    """Force ∑h = n/2 while keeping 0≤h≤1 (tiny corrections)."""
    n = h.size
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Apply the whole correction to the centre element, clamp,
    # then distribute any residual on a sweep.
    i = n // 2
    inc = diff
    new = h[i] + inc
    if new > 1.0:
        inc = 1.0 - h[i]
    elif new < 0.0:
        inc = -h[i]
    h[i] += inc
    diff -= inc
    if abs(diff) > 1e-12:
        # spread the remainder over the rest of the array
        idx = np.arange(n)
        idx = idx[idx != i]
        for j in idx:
            if diff == 0.0:
                break
            if diff > 0.0:
                avail = 1.0 - h[j]
                add = min(avail, diff)
                h[j] += add
                diff -= add
            else:
                avail = h[j]
                sub = min(avail, -diff)
                h[j] -= sub
                diff += sub
    return h

def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def _random_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Symmetric 5‑interval step function:
        [e, p, q, p, e] on lengths [a, b, c, b, a]  (2a+2b+c=2)
    e can be zero or a small random value.
    """
    for _ in range(50):
        a = rng.random() * 0.5          # edge length
        b = rng.random() * (0.5 - a)    # middle length, ensure a+b<0.5
        if a + b >= 0.5:
            continue
        c = 2.0 - 2.0 * (a + b)         # centre length  (>0)
        e = rng.random() * 0.2 if rng.random() < 0.3 else 0.0
        p = rng.random()
        # enforce integral: 2ea + 2pb + qc = 1 → q = (1‑2ea‑2pb)/c
        q = (1.0 - 2.0 * e * a - 2.0 * p * b) / c
        if not (0.0 <= q <= 1.0):
            continue
        lengths = np.array([a, b, c, b, a])
        values = np.array([e, p, q, p, e])
        # discretise
        dx = 2.0 / n
        xs = (np.arange(n) + 0.5) * dx
        edges = np.concatenate(([0.0], np.cumsum(lengths)))
        idx = np.searchsorted(edges, xs, side='right') - 1
        h = values[idx]
        h = _adjust_sum(h)             # enforce exact integral
        return h
    # fallback – uniform 0.5
    return np.full(n, 0.5, dtype=np.float64)

def _neighbor(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap a small amount of mass between two positions, preserving sum."""
    n = h.size
    i, j = rng.choice(n, 2, replace=False)
    # move mass from j to i (or opposite) without breaking bounds
    maxi = 1.0 - h[i]
    maxj = h[j]
    max_up = min(maxi, maxj)          # how much we can transfer i←j
    max_down = min(h[i], 1.0 - h[j])  # how much we can transfer i→j
    if max_up <= 0 and max_down <= 0:
        return h
    if rng.random() < 0.5 and max_up > 0:
        delta = rng.random() * max_up
        nh = h.copy()
        nh[i] += delta
        nh[j] -= delta
    else:
        delta = rng.random() * max_down
        nh = h.copy()
        nh[i] -= delta
        nh[j] += delta
    return nh

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # base resolution (even, not too large – good trade‑off)
    n = 200

    # start from a provided construction if any
    if 'initial_h_values' in globals():
        h_best = np.array(initial_h_values, dtype=np.float64)
        # enforce constraints (tiny rounding may occur)
        h_best = _adjust_sum(h_best)
    else:
        # try a random shape first – often already decent
        h_best = _random_shape(n, rng)
    c_best = _c5_of(h_best)

    # keep a separate current point for the simulated annealing walk
    h_cur = h_best.copy()
    c_cur = c_best

    # annealing schedule
    temp = 0.12
    decay = 0.9995

    # main loop – stop a second before deadline
    while time.time() - start < budget_s - 1.0:
        # occasionally try a brand‑new random shape/binary candidate
        if rng.random() < 0.02:
            cand = _random_shape(n, rng) if rng.random() < 0.7 else _random_binary(n, rng)
            cand_c = _c5_of(cand)
            if cand_c < c_best:
                h_best, c_best = cand, cand_c
            continue

        # propose a neighbour move
        cand = _neighbor(h_cur, rng)
        cand_c = _c5_of(cand)

        # Metropolis acceptance
        if cand_c < c_cur or rng.random() < np.exp(-(cand_c - c_cur) / max(temp, 1e-12)):
            h_cur, c_cur = cand, cand_c
            if cand_c < c_best:
                h_best, c_best = cand, cand_c

        # temperature decay
        temp *= decay
        if temp < 1e-6:
            temp = 1e-6

    # final bound – use the helper if present (more precise than our own metric)
    if 'evaluate_erdos_solution' in globals():
        try:
            c_bound = float(evaluate_erdos_solution(h_best))
        except Exception:
            c_bound = _c5_of(h_best)
    else:
        c_bound = _c5_of(h_best)

    # guarantee constraints one last time
    h_best = _adjust_sum(h_best)
    return h_best, c_bound, h_best.size
