import numpy as np
import time
import random

def _compute_c5(h, dx):
    """C5 = max_x ∫ h(x) (1-h(x+dx)) dx approximated by discrete correlation."""
    # np.correlate implements the non‑periodic overlap (zero‑padding)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _trapezoid_h(a, N, dx):
    """Symmetric trapezoid: ramps of length a on each side, flat top of height hmax."""
    if a <= 1e-12:  # degenerate to constant 0.5
        return np.full(N, 0.5, dtype=np.float64)

    hmax = 1.0 / (2.0 - a)          # enforce total area = 1
    x = (np.arange(N) + 0.5) * dx   # mid‑point sample positions

    h = np.empty(N, dtype=np.float64)

    left = x <= a
    right = x >= 2.0 - a
    middle = ~(left | right)

    h[left] = hmax * x[left] / a
    h[right] = hmax * (2.0 - x[right]) / a
    h[middle] = hmax

    # fix any tiny drift in the integral (should be ~1e‑12)
    target_sum = N / 2.0
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # add the remainder to a non‑boundary entry
        idx = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if idx.size == 0:
            idx = np.arange(N)
        idx = idx[0]
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

    return h


def _gradient_for_lag(h, L):
    """Gradient of S(L)=∑ h_i(1-h_{i+L}) w.r.t. h (L≥0)."""
    N = h.size
    if L == 0:
        return 1.0 - 2.0 * h
    g = np.zeros_like(h)
    # contribution from the first factor h_i
    g[: N - L] += 1.0 - h[L:]
    # contribution where h_i appears as the second factor (with a minus)
    g[L:] -= h[: N - L]
    return g


def _initial_from_parent(N, dx):
    """Upsample `initial_h_values` if present; otherwise return None."""
    if "initial_h_values" not in globals():
        return None
    init = initial_h_values
    M = init.shape[0]
    # sample points are mid‑points of the intervals
    x_parent = (np.arange(M) + 0.5) * (2.0 / M)
    x_target = (np.arange(N) + 0.5) * dx
    h = np.interp(x_target, x_parent, init, left=0.0, right=0.0)
    # enforce the integral exactly
    target_sum = N / 2.0
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        idx = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if idx.size == 0:
            idx = np.arange(N)
        idx = idx[0]
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points).

    The algorithm:
      1. Scan a one‑parameter family of symmetric trapezoids.
      2. Take the best among them (and the optional parent construction).
      3. Refine the best candidate with a simple simulated‑annealing /
         gradient‑driven local search that respects the constraints.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Choose a modest but expressive discretisation.
    # ------------------------------------------------------------------
    n_points = 200                     # faster than a thousand‑point grid
    dx = 2.0 / n_points

    # ------------------------------------------------------------------
    # 2. Scan the trapezoidal family.
    # ------------------------------------------------------------------
    best_h = None
    best_val = np.inf

    # coarse scan
    for a in np.linspace(0.0, 1.0, 101):
        h = _trapezoid_h(a, n_points, dx)
        val = _compute_c5(h, dx)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # ------------------------------------------------------------------
    # 3. If a parent solution is supplied, consider it as well.
    # ------------------------------------------------------------------
    parent_h = _initial_from_parent(n_points, dx)
    if parent_h is not None:
        parent_val = _compute_c5(parent_h, dx)
        if parent_val < best_val:
            best_val = parent_val
            best_h = parent_h.copy()

    # ------------------------------------------------------------------
    # 4. Local optimisation: gradient‑driven transfers + occasional swaps.
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_val = best_val

    # parameters for the stochastic search
    max_step = 0.10            # maximum amount moved in one transfer
    min_step = 0.001           # final step size
    prob_swap = 0.15           # how often we do a pure swap (no sum change)

    while True:
        # time check
        if time.time() - start_time > budget_s * 0.96:
            break

        # temperature‑like schedule for step size
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        step_size = max_step * (1.0 - frac) + min_step * frac

        # decide move type
        if random.random() < prob_swap:
            # pure swap – does not change the sum
            i, j = np.random.choice(n_points, 2, replace=False)
            new_h = cur_h.copy()
            new_h[i], new_h[j] = new_h[j], new_h[i]
        else:
            # compute worst lag
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            lag_index = int(np.argmax(corr)) - (n_points - 1)
            L = abs(lag_index)                 # we only need the magnitude

            # gradient for that lag
            g = _gradient_for_lag(cur_h, L)

            # indices where we can move mass from (g>0) to (g<0)
            pos = np.where((g > 0) & (cur_h > 1e-12))[0]
            neg = np.where((g < 0) & (cur_h < 1.0 - 1e-12))[0]

            if pos.size == 0 or neg.size == 0:
                # fallback: random transfer
                up = np.where(cur_h < 1.0 - 1e-12)[0]
                down = np.where(cur_h > 1e-12)[0]
                if up.size == 0 or down.size == 0:
                    continue
                i = np.random.choice(up)
                j = np.random.choice(down)
                while i == j:
                    j = np.random.choice(down)
                delta = min(step_size, 1.0 - cur_h[i], cur_h[j])
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
            else:
                i = np.random.choice(pos)
                j = np.random.choice(neg)
                delta = min(step_size, cur_h[i], 1.0 - cur_h[j])
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta

        # evaluate candidate
        new_val = _compute_c5(new_h, dx)

        # accept improvement
        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
        else:
            # occasional uphill move (simulated annealing)
            # temperature decays linearly to 0
            T = max(1e-6, (1.0 - frac) * 0.02)
            if random.random() < np.exp(-(new_val - cur_val) / T):
                cur_h, cur_val = new_h, new_val
                if new_val < best_val:
                    best_h, best_val = new_h.copy(), new_val
            # otherwise reject (keep cur_h)

    # ------------------------------------------------------------------
    # 5. Final tidy‑up: enforce exact sum & bounds (tiny numerical drift)
    # ------------------------------------------------------------------
    target_sum = n_points / 2.0
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # adjust the first non‑boundary entry
        idx = np.where((best_h > 1e-12) & (best_h < 1 - 1e-12))[0]
        if idx.size == 0:
            idx = np.arange(n_points)
        idx = idx[0]
        best_h[idx] += diff
        best_h = np.clip(best_h, 0.0, 1.0)

    final_val = _compute_c5(best_h, dx)

    return best_h, float(final_val), n_points
