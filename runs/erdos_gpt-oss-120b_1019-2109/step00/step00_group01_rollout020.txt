def make_binary(arr):
    n = len(arr)
    target = n // 2
    # Determine current integer rounding of arr: treat as ones if >0.5 else zero
    binary = (arr > 0.5).astype(np.int8)
    current = binary.sum()
    # Adjust difference
    diff = target - current
    if diff > 0:
        # need to turn some zeros to ones
        zero_idxs = np.where(binary == 0)[0]
        chosen = rng.choice(zero_idxs, size=diff, replace=False)
        binary[chosen] = 1
    elif diff < 0:
        # need to turn some ones to zeros
        one_idxs = np.where(binary == 1)[0]
        chosen = rng.choice(one_idxs, size=-diff, replace=False)
        binary[chosen] = 0

def compute_c5(arr):
    n = arr.shape[0]
    corr = np.correlate(arr, 1 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def _compute_c5(arr):
    """C5 = max_k ∫ h(x)(1−h(x+k))dx  (discrete version)"""
    n = arr.shape[0]
    # linear correlation of h with (1−h)
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    return np.max(corr) * (2.0 / n)

def _worst_shift_and_scores(arr):
    """return (worst_overlap, shift_index, score_vector) where
    shift_index ∈ [-(n-1), n-1] and score_vector[i]=h[i+|k|]+h[i-|k|] (0 if out of range)."""
    n = arr.shape[0]
    corr = np.correlate(arr, 1.0 - arr, mode='full')
    max_idx = int(np.argmax(corr))
    worst = corr[max_idx] * (2.0 / n)
    shift = max_idx - (n - 1)          # actual integer shift
    k = abs(shift)
    scores = np.zeros_like(arr)
    if k < n:                         # otherwise scores stay zero
        # contribution from the forward neighbour
        scores[:n - k] += arr[k:]
        # contribution from the backward neighbour
        scores[k:] += arr[:n - k]
    return worst, shift, scores

    total_time = budget_s * 0.97        # keep a tiny safety margin
    T0, Tf = 0.02, 0.0002               # temperature schedule

    # ---------- main simulated‑annealing loop ----------
    while time.time() - start_time < total_time:
        # compute current worst shift and the score vector
        cur_val, cur_shift, scores = _worst_shift_and_scores(h)

        # -------------------------------------------------
        # choose a pair (i increase, j decrease) guided by scores
        n_quarter = max(1, n // 4)
        order = np.argsort(scores)

        low_cand = order[:n_quarter]          # low scores
        high_cand = order[-n_quarter:]        # high scores

        # filter by feasibility (cannot increase a 1 or decrease a 0)
        low_cand = low_cand[h[low_cand] > 0.0]
        high_cand = high_cand[h[high_cand] < 1.0]

        if low_cand.size == 0 or high_cand.size == 0:
            # fallback to a completely random feasible pair
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            # ensure we can move mass from j to i
            max_delta = min(1.0 - h[i], h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            i_inc, j_dec = i, j
        else:
            i_inc = rng.choice(high_cand)
            j_dec = rng.choice(low_cand)
            max_delta = min(1.0 - h[i_inc], h[j_dec])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

        # perform the mass transfer
        h[i_inc] += delta
        h[j_dec] -= delta

        new_val = _compute_c5(h)

        # temperature based acceptance
        elapsed = time.time() - start_time
        prog = elapsed / total_time
        T = T0 * (1.0 - prog) + Tf * prog
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # uphill move with probability exp(-(Δ)/T)
            if rng.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i_inc] -= delta
            h[j_dec] += delta

    # ---------- final polishing (greedy hill‑climb) ----------
    # a few deterministic improvements using the same guided move idea
    for _ in range(2000):
        _, _, scores = _worst_shift_and_scores(best_h)
        n_quarter = max(1, n // 4)
        order = np.argsort(scores)
        low_cand = order[:n_quarter]
        high_cand = order[-n_quarter:]

        low_cand = low_cand[best_h[low_cand] > 0.0]
        high_cand = high_cand[best_h[high_cand] < 1.0]
        if low_cand.size == 0 or high_cand.size == 0:
            break
        i_inc = rng.choice(high_cand)
        j_dec = rng.choice(low_cand)
        max_delta = min(1.0 - best_h[i_inc], best_h[j_dec])
        if max_delta <= 0.0:
            continue
        # try the full possible transfer – if it improves we keep it
        delta = max_delta
        best_h[i_inc] += delta
        best_h[j_dec] -= delta
        new_val = _compute_c5(best_h)
        if new_val < best_val:
            best_val = new_val
        else:
            # revert
            best_h[i_inc] -= delta
            best_h[j_dec] += delta

    # final evaluation via the provided helper (more reliable)
    try:
        c5_bound = float(evaluate_erdos_solution(best_h))
    except Exception:
        # fallback to our own computation
        c5_bound = _compute_c5(best_h)

    return best_h, c5_bound, int(n)
