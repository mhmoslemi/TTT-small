    n = len(h)
    T = n/2.

        i, j = np.random.choice(n, 2, replace=False)

import time
import numpy as np

def _project_to_box_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto the box [0,1]^n intersected with sum(v)=target_sum."""
    n = v.shape[0]
    # Find lower bound λ so that sum(clamp(v-λ,0,1)) >= target_sum
    low = -np.max(v) - 1.0
    # ensure low is low enough
    while True:
        proj = np.clip(v - low, 0.0, 1.0)
        if proj.sum() >= target_sum:
            break
        low -= 1.0
    # Find upper bound λ so that sum(clamp(v-λ,0,1)) <= target_sum
    high = np.max(v) + 1.0
    while True:
        proj = np.clip(v - high, 0.0, 1.0)
        if proj.sum() <= target_sum:
            break
        high += 1.0
    # Binary search for λ giving exact sum (within tolerance)
    for _ in range(60):
        mid = (low + high) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v - (low + high) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return C5 for a given step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _smooth_gradient(h: np.ndarray, alpha: float = 12.0) -> np.ndarray:
    """Gradient of the smooth approximation of max correlation."""
    n = h.shape[0]
    one_minus = 1.0 - h
    corr = np.correlate(h, one_minus, mode="full")
    max_c = np.max(corr)
    # weights for a smooth max (log‑sum‑exp)
    exp_vals = np.exp(alpha * (corr - max_c))
    w = exp_vals / exp_vals.sum()

    grad = np.zeros_like(h)
    shifts = np.arange(-(n - 1), n)          # same length as corr / w
    for idx, L in enumerate(shifts):
        weight = w[idx]
        if weight == 0.0:
            continue
        if L >= 0:
            k = L
            # term (1 - h[i+L]) for i = 0 .. n-L-1
            grad[: n - k] += weight * (1.0 - h[k:])
            # term -h[i-L] for i = L .. n-1
            grad[k:] += weight * (-h[: n - k])
        else:
            k = -L
            # term (1 - h[i+L]) -> (1 - h[i-k]) for i = k .. n-1
            grad[k:] += weight * (1.0 - h[: n - k])
            # term -h[i-L] -> -h[i+k] for i = 0 .. n-k-1
            grad[: n - k] += weight * (-h[k:])
    return grad


def _initial_vector(n: int) -> np.ndarray:
    """Create an initial feasible vector of length n."""
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        if n % len(base) == 0:
            factor = n // len(base)
            h0 = np.repeat(base, factor)
        else:
            # linear interpolation if sizes do not match exactly
            x_old = np.linspace(0.0, 2.0, len(base), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = np.interp(x_new, x_old, base)
    else:
        h0 = np.random.rand(n)
    # project onto feasible set
    target = n * 0.5
    return _project_to_box_simplex(h0, target)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a small
    upper bound for the Erdős minimum overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : list of floats – the step heights
        c5_bound : float – the achieved C5 value
        n_points : int – number of discretisation points used
    """
    start_time = time.time()
    np.random.seed(seed)

    # Choose number of points – double the size of the provided sample if any
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        n = n0 * 2                     # modest up‑sampling
    else:
        n = 140                         # fall‑back size
    target_sum = n * 0.5

    # Best solution seen so far
    best_h = _initial_vector(n)
    best_c5 = _c5_value(best_h)

    # Parameters for the outer loop
    max_restarts = 30
    time_limit = budget_s - 0.2          # leave a tiny margin

    restart = 0
    while time.time() - start_time < time_limit and restart < max_restarts:
        restart += 1

        # ------------------------------------------------------------------
        # 1) Choose a start point for this restart
        # ------------------------------------------------------------------
        if restart == 1 and "initial_h_values" in globals():
            h = _initial_vector(n)          # start from the given construction
        else:
            h = _initial_vector(n)          # random feasible start

        cur_c5 = _c5_value(h)

        # ------------------------------------------------------------------
        # 2) Gradient descent on the smooth maximum
        # ------------------------------------------------------------------
        step = 0.4                          # initial step size
        alpha = 12.0                        # smoothness of the max
        for it in range(2000):
            # break if time exhausted
            if time.time() - start_time > time_limit:
                break

            g = _smooth_gradient(h, alpha=alpha)
            v = h - step * g
            h_new = _project_to_box_simplex(v, target_sum)
            c5_new = _c5_value(h_new)

            if c5_new < cur_c5 - 1e-9:
                h = h_new
                cur_c5 = c5_new
                # slight step increase when we make progress
                step = min(step * 1.02, 1.0)
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()
            else:
                # unsuccessful step – shrink step size
                step *= 0.65
                if step < 5e-4:            # step too small → stop this descent
                    break

        # ------------------------------------------------------------------
        # 3) Small random pairwise swaps (local search)
        # ------------------------------------------------------------------
        for _ in range(500):
            if time.time() - start_time > time_limit:
                break
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # permissible movement amount
            up_max = min(0.12, 1.0 - h[i])
            down_max = min(0.12, h[j])
            if up_max <= 0 or down_max <= 0:
                continue
            delta = np.random.uniform(0.0, min(up_max, down_max))
            if delta <= 0.0:
                continue
            h2 = h.copy()
            h2[i] += delta
            h2[j] -= delta
            c5_h2 = _c5_value(h2)
            if c5_h2 < cur_c5:
                h = h2
                cur_c5 = c5_h2
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, h.copy()
            # early exit if we have already beaten the target
            if best_c5 < 0.3808:
                break

        # ------------------------------------------------------------------
        # 4) Check for early termination
        # ------------------------------------------------------------------
        if best_c5 < 0.3808:
            break

    # final projection (safety)
    best_h = _project_to_box_simplex(best_h, target_sum)
    best_c5 = _c5_value(best_h)

    return (best_h.tolist(), float(best_c5), n)
