def project_onto_bounds_and_sum(x, target, eps=1e-9):
    lo = -1.0
    hi = x.max()
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # due to rounding may be off a bit, enforce sum via small adjust on some entries not at bounds
    diff = target - proj.sum()
    if abs(diff) > eps:
        # adjust proportionally on entries not at bounds
        free = (proj > 0) & (proj < 1)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0, 1)
    return proj
```

import numpy as np
import time

def _compute_c(h):
    """Return the C5 bound for a step vector h (dx = 2/len(h))."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using full mode (zero padding outside [0,2])
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_bounds_sum(x, target, eps=1e-12):
    """
    Euclidean projection of x onto the set:
        {y : sum(y) = target, 0 <= y_i <= 1}
    Implements a simple bisection on the scalar shift lambda.
    """
    lo = -1.0            # enough to make all entries 1 after clipping
    hi = x.max()        # makes all entries <=0 after clipping
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    proj = np.clip(x - lam, 0.0, 1.0)
    # tiny correction for possible rounding error
    diff = target - proj.sum()
    if abs(diff) > eps:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _direction_for_shift(h, shift):
    """Gradient direction d = ∂/∂h Σ_i h_i h_{i+shift} for a given integer shift."""
    n = h.shape[0]
    d = np.zeros_like(h)
    if shift >= 0:
        k = shift
        # contribution from h[i+k]
        d[:n - k] += h[k:]
        # contribution from h[i-k]
        d[k:] += h[:n - k]
    else:
        k = -shift
        # contribution from h[i-k] (i+shift)
        d[k:] += h[:n - k]
        # contribution from h[i+k] (i-shift)
        d[:n - k] += h[k:]
    return d

def _gradient_descent(h0, max_iters, time_deadline, best_h, best_c):
    """Simple projected sub‑gradient descent on max overlap."""
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    alpha0 = 0.5       # initial step size
    for it in range(1, max_iters + 1):
        if time.time() >= time_deadline:
            break
        # compute full correlation and locate worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        # direction that increases the self‑correlation at this shift
        d = _direction_for_shift(h, shift)
        # decreasing step size
        alpha = alpha0 / np.sqrt(it)
        h_new = h + alpha * d
        h_new = _project_bounds_sum(h_new, target)
        c_new = _compute_c(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
        h = h_new
    return best_h, best_c

def _local_search(h_start, best_c, best_h, time_deadline):
    """Simple pairwise mass swaps that preserve the integral."""
    n = h_start.shape[0]
    target = n / 2.0
    h = h_start.copy()
    rng = np.random.default_rng()
    it = 0
    while time.time() < time_deadline:
        i, j = rng.choice(n, size=2, replace=False)
        # allowable delta keeping both variables in [0,1]
        max_inc = min(1.0 - h[i], h[j])
        max_dec = min(h[i], 1.0 - h[j])
        if max_inc + max_dec == 0:
            continue
        delta = rng.random() * (max_inc + max_dec) - max_dec
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        c_new = _compute_c(h_new)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
            h = h_new
        it += 1
    return best_h, best_c
