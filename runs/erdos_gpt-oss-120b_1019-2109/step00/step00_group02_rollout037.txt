def gradient_for_shift(h, L, dx):
    n = len(h)
    g = np.ones_like(h) * dx  # derivative of sum part = dx
    # subtract h[i+L] contributions
    if L >= 0:
        # h[i+L] when i+L < n => for i=0..n-1-L

        # h[i-L] when i-L >=0 => i = L..n-1 => use h[:n-L]

def project_to_box_and_sum(v, target, lo=0., hi=1., tol=1e-8, max_iter=50):
    lower = lo - np.max(v)
    upper = hi - np.min(v)
    # Ensure that target is reachable: sum of lo after shift = lo*n, sum after upper = hi*n.
    # Binary search.
    for _ in range(max_iter):
        mid = (lower + upper) / 2.0
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            lower = mid
        else:
            upper = mid
    x = np.clip(v + (lower + upper) / 2.0, lo, hi)
    return x

def subgradient_max_overlap(h, dx):
    n = h.shape[0]
    O = np.correlate(h, 1 - h, mode='full') * dx
    idx = int(np.argmax(O))
    L = idx - (n - 1)  # lag between -(n-1) and n-1
    # compute neighbor sum
    neighbor = np.zeros_like(h)
    if L >= 0:
        neighbor[:n-L] += h[L:]
        neighbor[L:] += h[:n-L]
    else:
        k = -L
        neighbor[k:] += h[:n-k]
        neighbor[:n-k] += h[k:]
    # gradient of O w.r.t. h: g = dx * (1 - neighbor)
    g = dx * (1.0 - neighbor)
    return g, O[idx]

import time, numpy as np

def project_to_sum_and_box(v, target):
    lo = 0.0
    hi = 1.0
    low = lo - v.max()
    high = hi - v.min()
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low+high)/2.0, lo, hi)

def compute_O_circular(h):
    n = h.shape[0]
    dx = 2.0 / n
    H = np.fft.rfft(h)
    corr = np.fft.irfft(H * np.conj(H), n=n)
    O = 1.0 - dx * corr
    return O

def subgradient_step(h, dx, alpha):
    # compute O via FFT
    n = len(h)
    H = np.fft.rfft(h)
    corr = np.fft.irfft(H * np.conj(H), n=n)
    O = 1.0 - dx * corr
    # worst shift index L
    L = int(np.argmax(O))
    # compute gradient
    neighbor = np.roll(h, -L) + np.roll(h, L)
    grad = -dx * neighbor + alpha * (2 * h - 1)  # gradient of f = max O - alpha * sum h(1-h)
    return grad, O.max(), L

def optimizer(initial_h, target_sum, dx, max_iters, alpha, eta0, eta_end):
    h = initial_h.copy()
    best_h = h.copy()
    best_c = compute_O_circular(h).max()
    n = len(h)
    for t in range(max_iters):
        # compute gradient and C
        grad, cur_c, L = subgradient_step(h, dx, alpha)
        # step size schedule linearly decreasing
        eta = eta0 * (1 - t / max_iters) + eta_end * (t / max_iters)
        # update
        h = h - eta * grad
        # project onto constraints
        h = project_to_sum_and_box(h, target_sum)
        # evaluate after projection
        cur_c = compute_O_circular(h).max()
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
        # early exit if cur_c < target (0.38)
        if cur_c < 0.3808:
            break
    return best_h, best_c

def _project_box_sum(v, target, lo=0.0, hi=1.0, max_iter=60):
    """Project vector v onto the set {x: lo<=x_i<=hi, sum(x)=target}."""
    low = lo - np.max(v)      # makes all entries <= lo
    high = hi - np.min(v)     # makes all entries >= hi
    for _ in range(max_iter):
        mid = (low + high) / 2.0
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) / 2.0, lo, hi)


def _circular_overlap(h):
    """Return the array O_k = ∫ h(1-h(·+k)) for all circular shifts k."""
    n = h.shape[0]
    dx = 2.0 / n
    # circular autocorrelation via FFT
    H = np.fft.rfft(h)
    corr = np.fft.irfft(H * np.conj(H), n=n)   # sum_i h_i * h_{i+k} (circular)
    O = 1.0 - dx * corr
    return O

    # gradient of the penalty  -α∑h(1-h)  (to push values to 0/1)
    grad_pen = alpha * (2.0 * h - 1.0)
    return grad_O + grad_pen, O.max(), L


# ------------------------------------------------------------
# Optimisation routine
# ------------------------------------------------------------
def _optimise(initial_h, target_sum, alpha=0.5,
              eta_start=0.7, eta_end=0.01, max_iters=3000,
              time_limit=None, t0=None):
    """Projected sub‑gradient descent from a given start."""
    h = initial_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _circular_overlap(h).max()
    for it in range(max_iters):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        # decreasing step size
        eta = eta_start * (1 - it / max_iters) + eta_end * (it / max_iters)
        grad, cur_val, _ = _worst_shift_and_grad(h, alpha)
        h = h - eta * grad
        h = _project_box_sum(h, target_sum)
        cur_val = _circular_overlap(h).max()
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # early exit if we already beat the target bound
        if best_val < 0.3808:
            break
    return best_h, best_val


# ------------------------------------------------------------
# Construction of an initial guess (interpolation of parent)
# ------------------------------------------------------------
def _initial_guess(n_points, rng):
    """Return a feasible start vector – interpolated parent if present,
    otherwise random."""
    target = n_points / 2.0
    if 'initial_h_values' in globals():
        parent = np.asarray(initial_h_values, dtype=float)
        n0 = parent.size
        dx0 = 2.0 / n0
        x0 = np.arange(n0) * dx0 + dx0 / 2.0
        dx = 2.0 / n_points
        x_new = np.arange(n_points) * dx + dx / 2.0
        h = np.interp(x_new, x0, parent)
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, target)
        return h
    # random start
    v = rng.random(n_points)
    return _project_box_sum(v, target)


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    # problem size – small enough for quick iterations but expressive
    n = 200                     # can be changed; < 1000 as required
    target_sum = n / 2.0

    # first attempt from the supplied parent (if any)
    h_start = _initial_guess(n, rng)
    best_h, best_val = _optimise(
        h_start,
        target_sum,
        alpha=0.6,
        eta_start=0.8,
        eta_end=0.02,
        max_iters=4000,
        time_limit=budget_s - 2.0,
        t0=t0,
    )

    # a few random restarts for robustness
    restarts = 5
    for _ in range(restarts):
        if time.time() - t0 > budget_s - 2.0:
            break
        h_rand = _initial_guess(n, rng)
        h_opt, val = _optimise(
            h_rand,
            target_sum,
            alpha=0.6,
            eta_start=0.8,
            eta_end=0.02,
            max_iters=3000,
            time_limit=budget_s - (time.time() - t0) - 0.5,
            t0=t0,
        )
        if val < best_val:
            best_val = val
            best_h = h_opt.copy()

    # final bound according to the provided evaluator
    final_bound = evaluate_erdos_solution(best_h)

    return best_h, float(final_bound), n
