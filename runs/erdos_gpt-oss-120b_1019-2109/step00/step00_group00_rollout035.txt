def build_h(a, n):
    dx = 2.0 / n
    a_len = a  # length in [0,2]
    b_len = 2 - 2 * a_len
    if a_len < 1e-12:  # effectively uniform
        h_max = 0.5
        # all values 0.5 everywhere
        return np.full(n, h_max, dtype=np.float64)
    h_max = 1.0 / (2 - a_len)  # compute height
    # Precompute positions
    xs = np.arange(n) * dx + dx/2.0  # midpoints
    # Use vectorized compute:
    h = np.empty_like(xs)
    # left region
    mask_left = xs <= a_len
    # ramp up linear
    h[mask_left] = (h_max / a_len) * xs[mask_left]
    # middle region
    mask_mid = (xs > a_len) & (xs < a_len + b_len)
    h[mask_mid] = h_max
    # right region
    mask_right = xs >= a_len + b_len
    # For right region descending
    h[mask_right] = h_max - (h_max / a_len) * (xs[mask_right] - (a_len + b_len))
    # Clip to [0,1] for floating errors.
    np.clip(h, 0, 1, out=h)
    return h

import numpy as np, time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def shape_h(a, n):
    dx = 2.0 / n
    a_len = a  # length of ramp on each side
    if a_len <= 0:
        # uniform shape with height = 0.5
        return np.full(n, 0.5)
    b_len = 2.0 - 2.0 * a_len
    h_max = 1.0 / (2.0 - a_len)
    xs = (np.arange(n) + 0.5) * dx  # midpoints
    h = np.empty_like(xs)
    mask_left = xs <= a_len

    h[mask_left] = (h_max / a_len) * xs[mask_left]

    h[mask_mid] = h_max
    # right ramp
    h[mask_right] = h_max - (h_max / a_len) * (xs[mask_right] - (a_len + b_len))
    np.clip(h, 0, 1, out=h)
    return h

import time
import numpy as np

def _compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _shape_h(a, n):
    """Symmetric piece‑wise linear shape with ramp length a (0≤a≤1)."""
    dx = 2.0 / n
    if a <= 0.0:
        # uniform (height 0.5 gives integral 1)
        return np.full(n, 0.5, dtype=np.float64)

    a_len = a                     # length of each ramp
    b_len = 2.0 - 2.0 * a_len     # length of central plateau (may be zero)
    h_max = 1.0 / (2.0 - a_len)   # height to satisfy total area = 1

    # mid‑point sample in each sub‑interval
    xs = (np.arange(n) + 0.5) * dx
    h = np.empty_like(xs)

    left = xs <= a_len
    middle = (xs > a_len) & (xs < a_len + b_len)
    right = xs >= a_len + b_len

    # rising ramp
    h[left] = (h_max / a_len) * xs[left]
    # flat top
    h[middle] = h_max
    # falling ramp
    h[right] = h_max - (h_max / a_len) * (xs[right] - (a_len + b_len))

    np.clip(h, 0.0, 1.0, out=h)
    return h
