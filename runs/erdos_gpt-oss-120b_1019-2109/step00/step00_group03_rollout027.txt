def project_to_bounds(v, S, tol=1e-9, max_iter=60):
    lo = v.min() - 1.0  # ensures all x_i <= 1 after subtract λ?
    hi = v.max()
    for _ in range(max_iter):
        lam = (lo + hi) / 2
        x = np.clip(v - lam, 0, 1)
        total = x.sum()
        if total > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2, 0, 1)
    # correct possible sum error because of rounding:
    diff = S - x.sum()
    if abs(diff) > 1e-7:
        # Distribute diff over indices not at bounds within tolerance.
        # For diff>0, raise values of those not equal to 1 by small amount, else reduce not zero.

import time
import math
import numpy as np

def _project_to_bounds(v: np.ndarray, total: float, eps: float = 1e-12) -> np.ndarray:
    """Project v onto the box [0,1] while keeping sum(v) == total."""
    lo = v.min() - 1.0      # ensures all entries become 1 after shift
    hi = v.max()            # ensures all entries become 0 after shift
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = total - x.sum()
    if abs(diff) > eps:
        # distribute the tiny leftover mass over non‑boundary entries
        free = np.where((x > eps) & (x < 1 - eps))[0]
        if free.size:
            x[free] += diff / free.size
            np.clip(x, 0.0, 1.0, out=x)
    return x


def _gradient(h: np.ndarray, shift: int) -> np.ndarray:
    """Subgradient of the max‑overlap functional for the active shift."""
    n = h.size
    g = np.ones(n, dtype=np.float64)
    if shift >= 0:
        s = shift
        if s:
            g[: n - s] -= h[s:]
            g[s:] -= h[: n - s]
    else:
        s = -shift
        if s:
            g[s:] -= h[: n - s]
            g[: n - s] -= h[s:]
    return g
