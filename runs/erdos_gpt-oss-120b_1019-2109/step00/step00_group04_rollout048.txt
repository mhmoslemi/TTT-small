    import time

    rng = np.random.default_rng(seed)

    # ----- initial step function ------------------------------------

        init_h = np.array(initial_h_values, dtype=float)  # provided by harness
    except NameError:

        init_h = np.concatenate([np.ones(n0 // 2), np.zeros(n0 - n0 // 2)])

    n0 = init_h.shape[0]

    # increase resolution modestly (still well below 1000 points)
    factor = 5
    n = n0 * factor
    h0 = np.tile(init_h, factor).astype(float)
    np.clip(h0, 0.0, 1.0, out=h0)          # safety

    # ----- helper to compute the bound -----------------------------
    def compute_c(h):
        return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / len(h))

    best_h = h0.copy()
    best_c = compute_c(best_h)

    cur_h = best_h.copy()
    cur_c = best_c

    # ----- main stochastic improvement loop ------------------------
    start = time.time()
    opt_budget = budget_s * 0.98            # leave a tiny margin
    eps = 1e-12
    max_no_improve = 2000
    no_improve = 0
    iteration = 0

    while time.time() - start < opt_budget:
        # identify the worst lag (largest overlap)
        corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        worst_idx = int(np.argmax(corr_full))
        lag = worst_idx - (n - 1)           # lag = shift index (can be negative)

        # choose a random overlapping pair (i, i+lag)
        if lag >= 0:
            i_min, i_max = 0, n - 1 - lag
        else:
            i_min, i_max = -lag, n - 1
        if i_min > i_max:
            continue
        i = rng.integers(i_min, i_max + 1)
        j = i + lag

        # we want to increase the product at the pair; move mass to the smaller entry
        if cur_h[i] < cur_h[j]:
            target, partner = i, j
        else:
            target, partner = j, i

        # pick a source index with some mass to donate
        src = None
        for _ in range(10):
            cand = rng.integers(0, n)
            if cand != target and cur_h[cand] > eps:
                src = cand
                break
        if src is None:
            continue

        max_delta = min(cur_h[src], 1.0 - cur_h[target])
        if max_delta <= eps:
            continue

        # random fraction of the available mass (occasionally try a larger move)
        r = rng.random()
        if r < 0.2:
            delta = max_delta * r                # very small
        else:
            delta = max_delta * 0.5 * r          # up to half of max_delta

        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[target] += delta

        new_c = np.max(np.correlate(new_h, 1.0 - new_h, mode="full")) * (2.0 / n)

        # greedy acceptance, with a tiny simulated‑annealing fallback
        if new_c < cur_c:
