import numpy as np, time, math

def compute_c5(h):
    N = h.shape[0]
    # np.correlate returns length 2*N-1
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * (2.0 / N)

def project_to_box_sum(v, target, lower=0.0, upper=1.0):
    # binary search for lambda s.t. sum(clipped(v+lam)) = target
    # initial bounds
    lam_low = lower - np.max(v)
    lam_high = upper - np.min(v)
    # ensure that sum at low <= target <= sum at high
    # compute sum at low: all lower => n*lower
    # sum at high: n*upper
    # target is within those.

        s = x.sum()
        if s > target:

        else:
            lam_low = lam

import time, math
import numpy as np

def _max_overlap(h):
    """Return C5 for the step function represented by vector h."""
    n = h.size
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * (2.0 / n)


def _proj_box_sum(v, target, lo=0.0, hi=1.0):
    """Projection of v onto the box [lo,hi] with sum == target."""
    # Clip first, then adjust using a common offset lambda.
    x = np.clip(v, lo, hi)
    s = x.sum()
    if abs(s - target) < 1e-12:
        return x
    # binary search for lambda
    lam_low = lo - v.max()
    lam_high = hi - v.min()
    for _ in range(60):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v + lam, lo, hi)
        s = x.sum()
        if s > target:
            lam_high = lam
        else:
            lam_low = lam
    x = np.clip(v + lam_low, lo, hi)
    # tiny residual correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > lo + 1e-12) & (x < hi - 1e-12)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            x = np.clip(x, lo, hi)
    return x


def _random_feasible(n, target):
    v = np.random.rand(n)
    v *= target / v.sum()
    return _proj_box_sum(v, target)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ----- initial vector -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
        n = h0.size
        target = n * 0.5
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = _proj_box_sum(h0, target)
        candidates = [h0]
    else:
        n = 70
        target = n * 0.5
        candidates = []

    # a few random starts
    for _ in range(3):
        candidates.append(_random_feasible(n, target))

    # best so far
    best_h = None
    best_val = float("inf")

    # pick the best of the initial candidates as current
    cur = candidates[0].copy()
    cur_val = _max_overlap(cur)
    for cand in candidates[1:]:
        v = _max_overlap(cand)
        if v < cur_val:
            cur = cand.copy()
            cur_val = v
    best_h, best_val = cur.copy(), cur_val

    # ----- SA / directed search -----------------------------------------
    temp = 0.02                     # start temperature
    step = 0.25                     # max mass transferred in one move
    prob_invert = 0.003
    prob_random = 0.001
    no_imp = 0

    while time.time() - start < budget_s * 0.95:
        # compute worst lag for current solution
        conv = np.correlate(cur, 1.0 - cur, mode="full")
        idx_max = int(np.argmax(conv))
        cur_val = conv[idx_max] * (2.0 / n)
        lag = idx_max - (n - 1)          # signed shift giving the worst overlap

        # decide move type
        if np.random.rand() < prob_invert:
            cand = 1.0 - cur               # inversion keeps sum = target automatically
        elif np.random.rand() < prob_random:
            cand = _random_feasible(n, target)
        else:
            cand = cur.copy()
            if lag == 0:
                # balance around 0.5
                low = np.where(cand < 0.5 - 1e-6)[0]
                high = np.where(cand > 0.5 + 1e-6)[0]
                if low.size and high.size:
                    i = np.random.choice(low)
                    j = np.random.choice(high)
                    delta = min(step, cand[i], 1.0 - cand[j])
                    cand[i] -= delta
                    cand[j] += delta
                else:
                    # fallback: simple transfer
                    i, j = np.random.choice(n, 2, replace=False)
                    d_low = max(-step, -cand[i], cand[j] - 1.0)
                    d_high = min(step, 1.0 - cand[i], cand[j])
                    if d_low <= d_high:
                        delta = np.random.uniform(d_low, d_high)
                        cand[i] += delta
                        cand[j] -= delta
            else:

                if lag > 0:

                region = np.arange(start, end + 1, dtype=int)
                # gradient = 1 - h_{i+lag}
                grad = 1.0 - cand[region + lag]
                feasible_mask = cand[region] > 1e-8
                if feasible_mask.any():

                    i_local = np.argmax(grad[feasible_mask])

                    # choose j outside the region
                    mask = np.ones(n, dtype=bool)

                    outside = np.where(mask)[0]
                    outside = outside[cand[outside] < 1.0 - 1e-8]
                    if outside.size:

                        cand[i] -= delta
                        cand[j] += delta

                    d_high = min(step, 1.0 - cand[i], cand[j])
                    if d_low <= d_high:
                        delta = np.random.uniform(d_low, d_high)
                        cand[i] += delta
                        cand[j] -= delta

        # evaluate candidate
        cand_val = _max_overlap(cand)

        # acceptance (SA)
        if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / max(temp, 1e-12)):

            if cand_val < best_val - 1e-12:
                best_h, best_val = cand.copy(), cand_val
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        # cooling & step decay
        temp *= 0.9999
        step *= 0.9999
        step = max(step, 0.001)

        # occasional restart if stuck
        if no_imp > 60000:
            if np.random.rand() < 0.5:
                cur, cur_val = best_h.copy(), best_val
            else:
                cur = _random_feasible(n, target)
                cur_val = _max_overlap(cur)
            no_imp = 0

    # final projection to guarantee feasibility
    best_h = _proj_box_sum(best_h, target)
    best_val = _max_overlap(best_h)

    return (best_h, float(best_val), int(n))
