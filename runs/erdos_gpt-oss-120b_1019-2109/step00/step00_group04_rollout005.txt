def evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def random_feasible(n, target, rng, max_trials=2000):
    for _ in range(max_trials):
        # generate n random numbers, ensure sum = target after scaling
        y = rng.random(n)
        s = y.sum()
        if s == 0:
            continue
        y = y / s * target
        if np.all(y <= 1.0):
            return y
    # fallback to uniform 0.5 (will satisfy sum only if target == 0.5*n)
    return np.full(n, target / n)

import time
import numpy as np

def _evaluate(h: np.ndarray) -> float:
    """Return C₅ for a step function h on [0,2] sampled uniformly."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate implements the discrete correlation with zero padding outside
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def _random_feasible(n: int, target: float, rng: np.random.Generator, trials: int = 2000) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum equal to target."""
    for _ in range(trials):
        y = rng.random(n)          # uniform in (0,1)
        s = y.sum()
        if s == 0:
            continue
        y = y / s * target          # scale to required sum
        if np.all(y <= 1.0):
            return y
    # fall back to a simple deterministic feasible point (0.5 everywhere)
    return np.full(n, target / n)


def _hill_climb(initial_h: np.ndarray, deadline: float, rng: np.random.Generator):
    """Greedy + occasional stochastic hill climbing with restarts."""
    n = initial_h.shape[0]
    target = n / 2.0
    best_h = initial_h.copy()
    best_val = _evaluate(best_h)

    cur_h = best_h.copy()
    cur_val = best_val

    max_step = 0.2                 # maximal change transferred between two positions
    no_improve = 0
    max_no_improve = 20000        # restart after this many failed attempts
    start = time.perf_counter()

    while time.perf_counter() < deadline:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a mass transfer delta from j to i (preserves total sum)
        delta = rng.uniform(-max_step, max_step)
        new_i = cur_h[i] + delta
        new_j = cur_h[j] - delta
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue

        new_h = cur_h.copy()
        new_h[i] = new_i
        new_h[j] = new_j
        new_val = _evaluate(new_h)

        # simple simulated‑annealing style acceptance
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # temperature decreases linearly to a tiny floor
            elapsed = time.perf_counter() - start
            total = deadline - start
            temp = 0.05 * max(0.0, 1.0 - elapsed / total) + 1e-6
            prob = np.exp(-(new_val - cur_val) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_h = new_h
            cur_val = new_val
            no_improve = 0
            if new_val < best_val - 1e-12:
                best_h = new_h
                best_val = new_val
        else:
            no_improve += 1

        # restart if stuck
        if no_improve >= max_no_improve:
            if rng.random() < 0.5:
                # go back to the current best
                cur_h = best_h.copy()
                cur_val = best_val
            else:
                # random fresh feasible point
                cur_h = _random_feasible(n, target, rng)
                cur_val = _evaluate(cur_h)
            no_improve = 0

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Determine the basic size; use the provided construction if any.
    # -----------------------------------------------------------------
    if "initial_h_values" in globals():
        base_h = np.asarray(initial_h_values, dtype=float)
    else:
        # sensible fallback: 42 points, uniform density 0.5 (will be re‑scaled)
        base_h = np.full(42, 0.5, dtype=float)

    base_n = base_h.size
    candidates = [base_n]

    # also try a doubled resolution (still far below the 1 000 limit)
    if base_n * 2 <= 1000:
        candidates.append(base_n * 2)
    # and a triple, if still reasonable
    if base_n * 3 <= 1000:
        candidates.append(base_n * 3)

    start_time = time.perf_counter()
    global_deadline = start_time + budget_s - 0.05   # tiny safety margin

    best_overall_h = None
    best_overall_val = np.inf
    best_overall_n = None

    for n in candidates:
        # -----------------------------------------------------------------
        # Build an initial feasible vector for this resolution.
        # -----------------------------------------------------------------
        if n == base_n and "initial_h_values" in globals():
            h0 = base_h.copy()
        elif n == base_n * 2 and "initial_h_values" in globals():
            h0 = np.repeat(base_h, 2)
        elif n == base_n * 3 and "initial_h_values" in globals():
            h0 = np.tile(base_h, 3)[:n]
        else:
            target_sum = n / 2.0
            h0 = _random_feasible(n, target_sum, rng)

        # Ensure the exact integral constraint (tiny numerical correction)
        target_sum = n / 2.0
        diff = target_sum - h0.sum()
        if abs(diff) > 1e-12:
            # adjust the first component that can absorb the difference
            for idx in range(n):
                adj = h0[idx] + diff
                if 0.0 <= adj <= 1.0:
                    h0[idx] = adj
                    break

        # -----------------------------------------------------------------
        # Allocate a share of the remaining time to this size.
        # -----------------------------------------------------------------
        time_left = global_deadline - time.perf_counter()
        if time_left <= 0:
            break
        # give each candidate roughly equal remaining budget
        sub_deadline = time.perf_counter() + time_left * 0.9

        # -----------------------------------------------------------------
        # Optimize.
        # -----------------------------------------------------------------
        h_opt, val_opt = _hill_climb(h0, sub_deadline, rng)

        if val_opt < best_overall_val:
            best_overall_h = h_opt
            best_overall_val = val_opt
            best_overall_n = n

    # -----------------------------------------------------------------
    # Final sanity adjustments (ensure feasibility).
    # -----------------------------------------------------------------
    n = best_overall_n
    h = best_overall_h.copy()
    target = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target) > 1e-9:
        delta = target - cur_sum
        for i in range(n):
            cand = h[i] + delta
            if 0.0 <= cand <= 1.0:
                h[i] = cand
                break
    h = np.clip(h, 0.0, 1.0)

    final_val = _evaluate(h)

    return (h, float(final_val), int(n))
