from scipy.optimize import minimize

    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

    # ensure within [0,1] and sum = target_sum
    # Use simple projection: clip, then adjust mass.
    h = np.clip(h, 0.0, 1.0)
    cur = np.sum(h)
    diff = target_sum - cur
    # adjust uniformly into interior where possible
    # iterate
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        # find indices not at bound in direction of diff
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        # distribute equally
        d = diff / np.count_nonzero(mask)
        # add but keep within bounds
        adjustment = np.clip(h[mask] + d, 0.0, 1.0) - h[mask]
        h[mask] += adjustment
        diff = target_sum - np.sum(h)
    return h

def local_optimize(h0, max_iter=200):

import time, random
import numpy as np
from scipy.optimize import minimize

def _c5_of_h(h):
    """Compute the C5 value for a given step function h (array of values)."""
    n = h.shape[0]
    # correlation of h with (1-h); np.correlate does linear (zero‑padded) correlation
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)   # dx = 2/n

def _project_bounds_sum(x, target_sum, max_iter=20):
    """Clip to [0,1] and then adjust to satisfy the exact sum constraint."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = x.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        # indices that can move in the needed direction
        if diff > 0:
            mask = x < 1.0 - 1e-12
        else:
            mask = x > 1e-12
        if not np.any(mask):
            break
        d = diff / np.count_nonzero(mask)
        # apply step but stay inside bounds
        adjust = np.clip(x[mask] + d, 0.0, 1.0) - x[mask]
        x[mask] += adjust
    return x

def _local_optimize(start_h, max_iter=200):
    """Run a small SLSQP optimisation from `start_h`."""
    n = start_h.shape[0]
    target = n / 2.0
    cons = ({'type': 'eq', 'fun': lambda v: v.sum() - target},)
    bounds = [(0.0, 1.0)] * n
    opt = minimize(_c5_of_h, start_h, method="SLSQP",
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-9, 'disp': False})
    h_opt = _project_bounds_sum(np.asarray(opt.x, dtype=float), target)
    return h_opt, _c5_of_h(h_opt)

def _random_feasible(n, target_sum):
    """Generate a random point in [0,1]^n with sum = target_sum."""
    # Dirichlet‑like sample then project
    r = np.random.rand(n)
    r = r / r.sum() * target_sum
    return _project_bounds_sum(r, target_sum)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h that yields a low C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
    """
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------------
    # initialise problem size and a feasible starting vector
    # --------------------------------------------------------------
    init = globals().get("initial_h_values", None)
    if init is not None:
        h0 = np.array(init, dtype=float)
    else:
        # fall back to a uniform 0.5 vector (always feasible)
        n = 42
        h0 = np.full(n, 0.5, dtype=float)

    n = h0.shape[0]
    target_sum = n / 2.0

    # --------------------------------------------------------------
    # baseline optimisation from the provided initial vector
    # --------------------------------------------------------------
    best_h, best_c5 = _local_optimize(_project_bounds_sum(h0, target_sum), max_iter=300)

    start_time = time.time()
    # --------------------------------------------------------------
    # many short restarts with random starts
    # --------------------------------------------------------------
    restarts = 0
    while time.time() - start_time < budget_s * 0.65:
        # create a random feasible point; occasionally use a binary pattern
        if random.random() < 0.4:
            # binary pattern with exactly n/2 ones (n is even for the given records)
            idx = np.random.choice(n, n // 2, replace=False)
            h_rand = np.zeros(n, dtype=float)
            h_rand[idx] = 1.0
        else:
            h_rand = _random_feasible(n, target_sum)

        h_opt, c5_val = _local_optimize(h_rand, max_iter=120)
        if c5_val < best_c5:
            best_c5, best_h = c5_val, h_opt
        restarts += 1
        if restarts >= 120:        # a safety ceiling
            break

    # --------------------------------------------------------------
    # fine‑grained hill‑climbing from the current best solution
    # --------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c5 = best_c5
    step_factor = 0.12
    while time.time() - start_time < budget_s * 0.94:
        i, j = random.sample(range(n), 2)
        # decide direction of transfer
        if random.random() < 0.5:
            donor, recv = i, j
        else:
            donor, recv = j, i

        can_give = cur_h[donor]
        can_take = 1.0 - cur_h[recv]
        eps_max = min(can_give, can_take)
        if eps_max <= 1e-12:
            continue
        # random step size, shrinking slowly as time runs out
        eps = eps_max * random.random() * step_factor
        new_h = cur_h.copy()
        new_h[donor] -= eps
        new_h[recv] += eps
        # quick feasibility check (bounds already honoured)
        new_c5 = _c5_of_h(new_h)
        if new_c5 + 1e-12 < cur_c5:
            cur_h, cur_c5 = new_h, new_c5

        # occasional reduction of step size
        if random.random() < 0.001:

    # final projection onto the feasible set (tiny numerical drift may exist)
    best_h = _project_bounds_sum(best_h, target_sum)
    best_c5 = _c5_of_h(best_h)
