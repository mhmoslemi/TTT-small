import time
import math
import random
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a step function h."""
    n = h.size
    dx = 2.0 / n
    # full correlation of h with (1-h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx


def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto the hyperplane sum(h)=target while respecting 0≤h≤1."""
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # try to fix by adjusting a single free coordinate
    for i in range(h.size):
        new_val = h[i] + diff
        if 0.0 <= new_val <= 1.0:
            h[i] = new_val
            return h
    # fallback: simple proportional scaling then clipping
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        # make half of entries 1
        h[:int(target)] = 1.0
        return h
    h = h * (target / s)
    # final clip to stay in bounds
    h = np.clip(h, 0.0, 1.0)
    # small correction if still off
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny remainder to the first non‑saturated entry
        for i in range(h.size):
            if 0.0 < h[i] < 1.0:
                h[i] += diff
                break
    return h

    target_sum = n / 2.0
    h0 = _repair_sum(h0, target_sum)

    best_h = h0.copy()

    cur_h = best_h.copy()

    # ------------------------------------------------------------------

        # --------------------------------------------------------------
