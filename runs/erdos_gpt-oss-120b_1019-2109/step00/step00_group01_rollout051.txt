def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    c5 = conv.max() * dx
    return c5

def simulated_annealing(n, time_limit):
    cur_h = random binary with n/2 ones
    cur_c5 = compute_c5(cur_h)
    best_h, best_c5 = cur_h.copy(), cur_c5

import time, math
import numpy as np

def _calc_c5(vec: np.ndarray, dx: float) -> float:
    """Return C5 for a binary (or 0/1) vector."""
    conv = np.correlate(vec, 1 - vec, mode="full")
    return float(conv.max()) * dx


def _thue_morse(n: int) -> np.ndarray:
    """Generate a length‑n Thue‑Morse binary sequence (0/1)."""
    seq = np.arange(n, dtype=np.uint32)
    bits = seq ^ (seq >> 1)
    # parity of bits (popcount mod 2)
    parity = np.unpackbits(bits.view(np.uint8), count=32, bitorder='little')
    parity = parity.reshape(-1, 32).sum(axis=1) & 1
    return parity.astype(np.int8)


def _initial_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random binary vector of length n with exactly n/2 ones."""
    vec = np.zeros(n, dtype=np.int8)
    vec[: n // 2] = 1
    rng.shuffle(vec)
    return vec


def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function achieving a low C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,) with entries in [0,1]
        c5_bound : float, the computed max overlap integral
        n_points : int, length of the discretisation
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------
    # Choose problem size
    # -------------------------------------------------------
    target_n = 200                     # even, moderate size
    if target_n % 2:
        target_n += 1
    dx = 2.0 / target_n

    # -------------------------------------------------------
    # Build initial vector, using provided construction if handy
    # -------------------------------------------------------
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        # tile / truncate the parent pattern to the desired length
        pattern = np.array(init_vec, dtype=np.int8)
        pattern = (pattern > 0.5).astype(np.int8)
        repeats = (target_n + len(pattern) - 1) // len(pattern)
        h = np.tile(pattern, repeats)[:target_n].copy()
        # enforce exact half‑ones count
        cur_ones = int(h.sum())
        need = target_n // 2 - cur_ones
        if need > 0:
            zero_idx = np.where(h == 0)[0]
            rng.shuffle(zero_idx)
            h[zero_idx[:need]] = 1
        elif need < 0:
            one_idx = np.where(h == 1)[0]
            rng.shuffle(one_idx)
            h[one_idx[: -need]] = 0
    else:
        # fall back: use a short Thue‑Morse pattern as seed
        h = _thue_morse(target_n)
        if h.sum() != target_n // 2:
            # adjust to the exact weight
            cur = int(h.sum())
            need = target_n // 2 - cur
            if need > 0:
                zero_idx = np.where(h == 0)[0]
                rng.shuffle(zero_idx)
                h[zero_idx[:need]] = 1
            elif need < 0:
                one_idx = np.where(h == 1)[0]
                rng.shuffle(one_idx)
                h[one_idx[: -need]] = 0

    # ensure feasibility
    assert int(h.sum()) == target_n // 2
    cur_c5 = _calc_c5(h, dx)
    best_h = h.copy()
    best_c5 = cur_c5

    # -------------------------------------------------------
    # Simulated‑annealing style local search
    # -------------------------------------------------------
    max_no_improve = 40000          # restart after this many rejects
    no_improve = 0
    iteration = 0
    # temperature schedule parameters
    T0 = 0.03
    decay_factor = 1e-5

    while True:
        if time.time() - start > budget_s - 0.7:
            break

        # temperature slowly decays with iteration count
        T = T0 * math.exp(-decay_factor * iteration)

        # pick a random 1 and a random 0 to exchange
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(h == 0)
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # perform swap
        h[i] = 0
        h[j] = 1
        new_c5 = _calc_c5(h, dx)

        diff = new_c5 - cur_c5
        accept = diff < 0 or rng.random() < math.exp(-diff / max(T, 1e-9))

        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
                # early exit when we hit the target bound
                if best_c5 <= 0.3808:
                    break
            else:
                no_improve += 1
        else:
            # revert the swap
            h[i] = 1
            h[j] = 0
            no_improve += 1

        # occasional restart to escape deep local minima
        if no_improve >= max_no_improve:
            # restart from the best found so far, plus a few random swaps
            h = best_h.copy()
            # apply a small random perturbation
            pert = rng.integers(1, 5)
            for _ in range(pert):
                o = rng.choice(np.flatnonzero(h))
                z = rng.choice(np.flatnonzero(h == 0))
                h[o] = 0
                h[z] = 1
            cur_c5 = _calc_c5(h, dx)
            no_improve = 0

        iteration += 1

    # final verification (should already hold)
    final_c5 = _calc_c5(best_h, dx)
    return best_h.astype(float), final_c5, target_n
