import time, math
import numpy as np

def _evaluate(h, dx):
    """Return C5 for a given step function h."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def _project_capped_simplex(v, target_sum, lower=0.0, upper=1.0, max_iter=60):
    """Projection of v onto {x: lower<=x<=upper, sum(x)=target_sum}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, lower, upper)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, lower, upper)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ----- initialise -----
    if "initial_h_values" in globals():
        h_cur = np.array(initial_h_values, dtype=float)
        n = h_cur.size
        # enforce box constraints (they should already hold)
        h_cur = np.clip(h_cur, 0.0, 1.0)
        # ensure exact integral constraint
        target = n / 2.0
        if abs(h_cur.sum() - target) > 1e-12:
            h_cur = _project_capped_simplex(h_cur, target)
    else:
        # fallback: n=70, half ones binary start
        n = 70
        target = n / 2.0
        h_cur = np.zeros(n)
        ones = rng.choice(n, size=int(target), replace=False)
        h_cur[ones] = 1.0

    dx = 2.0 / n
    target = n / 2.0

    # best solution tracking
    best_h = h_cur.copy()
    best_val = _evaluate(best_h, dx)

    cur_val = best_val

    start = time.time()
    # leave a small safety margin
    time_limit = budget_s * 0.98

    # simulated‑annealing schedule
    start_temp = 0.02
    end_temp = 1e-4

    while True:
        now = time.time()
        if now - start >= time_limit:
            break

        # linear cooling
        progress = (now - start) / budget_s
        temp = start_temp * (1.0 - progress) + end_temp * progress

        i, j = rng.integers(n, size=2)

        s = h_cur[i] + h_cur[j]                       # constant sum for the pair
