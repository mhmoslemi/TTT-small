import numpy as np
import time
import math

def compute_overlap(h):
    """Return max_{k} ∫ h(x)(1−h(x+k))dx approximated by discrete correlation."""
    N = h.shape[0]
    dx = 2.0 / N
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def random_binary(N):
    """Random binary vector with exactly N/2 ones."""
    h = np.zeros(N, dtype=np.float64)
    h[np.random.choice(N, N // 2, replace=False)] = 1.0
    return h

def thue_morse_pattern(N):
    """Balanced Thue–Morse binary pattern of length N."""
    seq = np.fromiter((bin(i).count('1') % 2 for i in range(N)), dtype=np.float64)
    target = N // 2
    cur = int(seq.sum())
    if cur > target:
        ones = np.where(seq == 1)[0]
        drop = np.random.choice(ones, cur - target, replace=False)
        seq[drop] = 0.0
    elif cur < target:
        zeros = np.where(seq == 0)[0]
        add = np.random.choice(zeros, target - cur, replace=False)
        seq[add] = 1.0
    return seq

def simulated_annealing(N, start_h, max_iters, T0, Tf, target_c):
    """Simple SA that swaps a 0↔1 while preserving total sum."""
    h = start_h.copy()
    cur_c = compute_overlap(h)
    best_h = h.copy()
    best_c = cur_c
    half = N // 2
    for it in range(max_iters):
        # linear temperature schedule
        T = T0 * (1 - it / max_iters) + Tf * (it / max_iters)
        i = np.random.randint(N)
        j = np.random.randint(N)
        # need opposite values to keep the half‑ones invariant
        attempts = 0
        while h[i] == h[j] and attempts < 5:
            i = np.random.randint(N)
            j = np.random.randint(N)
            attempts += 1
        if h[i] == h[j]:
            continue
        # propose swap
        h[i], h[j] = h[j], h[i]
        new_c = compute_overlap(h)
        delta = new_c - cur_c
        if delta <= 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        if best_c <= target_c:
            break
    return best_h, best_c
