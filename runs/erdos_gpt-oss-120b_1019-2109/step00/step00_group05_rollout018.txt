import numpy as np
import time
import random
import math

def _c5_of(h: np.ndarray) -> float:
    """C5 value for a step function sampled in h."""
    N = h.shape[0]
    dx = 2.0 / N
    # np.correlate gives the discrete convolution needed for the overlap integral
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _sturmian(N: int, alpha: float) -> np.ndarray:
    """Balanced binary word of length N with density 1/2 using a Sturmian construction."""
    seq = ((np.arange(N, dtype=float) * alpha) % 1.0) < 0.5
    target = N // 2
    cur = int(seq.sum())
    if cur > target:
        idx = np.where(seq)[0]
        flip = np.random.choice(idx, cur - target, replace=False)
        seq[flip] = False
    elif cur < target:
        idx = np.where(~seq)[0]
        flip = np.random.choice(idx, target - cur, replace=False)
        seq[flip] = True
    return seq.astype(float)


def _random_binary(N: int) -> np.ndarray:
    """Uniform random binary vector with exactly N/2 ones."""
    h = np.zeros(N, dtype=float)
    ones = np.random.choice(N, N // 2, replace=False)
    h[ones] = 1.0
    return h


def _local_search(h: np.ndarray, max_iters: int, deadline: float) -> tuple:
    """
    Simple simulated‑annealing style improvement.
    Returns (best_h, best_c5) found during the run.
    """
    N = h.shape[0]
    cur_c5 = _c5_of(h)
    best_h = h.copy()
    best_c5 = cur_c5
    T = 0.005  # initial temperature

    it = 0
    while it < max_iters and time.time() < deadline:
        i = random.randrange(N)
        j = random.randrange(N)
        if h[i] == h[j]:
            continue
        # swap a 0 and a 1
        h[i], h[j] = h[j], h[i]
        new_c5 = _c5_of(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            # uphill move with probability
            if random.random() < math.exp(-(new_c5 - cur_c5) / T):
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
            else:
                # revert swap
                h[i], h[j] = h[j], h[i]
        # cool down
        T *= 0.9999
        it += 1
    return best_h, best_c5
