import time

# ---------- helper functions ----------
def compute_c5(h):
    """Exact max overlap C5 for a step function h represented by samples."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_onto_box_sum(v, target):
    """
    Euclidean projection of v onto the set {x | 0<=x_i<=1 , sum(x)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            hi = lam
        else:
            lo = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def smooth_max_overlap(h, beta, dx):
    """Smooth approximation of max_{k} overlap using log‑sum‑exp."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    # use stable log‑sum‑exp
    m = np.max(corr)
    return dx * (m + math.log(np.sum(np.exp(beta * (corr - m)))) / beta)

        n = 200
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = 0.5 + 0.5 * np.cos(np.pi * x / 2.0)   # values in [0,1]

    target_sum = n / 2.0
    h = project_onto_box_sum(h, target_sum)

    best_h = h.copy()
    best_score = compute_c5(best_h)

    # ----- simulated‑annealing style local search -----
    max_delta_start = 0.4
    beta_smooth = 20.0       # for the optional smooth optimisation later
    penalty = 1e3

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.96:
            break

        # cooling schedules
        t_frac = 1.0 - elapsed / budget_s
        temperature = 0.008 * t_frac          # annealing temperature
        max_delta = max_delta_start * t_frac   # decreasing step size

        # propose a move that preserves the total sum
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        hi, hj = h[i], h[j]

        # allowed delta range (increase i, decrease j)
        delta_up = min(1.0 - hi, hj)       # positive direction
        delta_down = min(hi, 1.0 - hj)     # negative direction
        if delta_up + delta_down == 0.0:
            continue
        delta = (rng.random() * (delta_up + delta_down) - delta_down)
        # limit magnitude for stability
        if abs(delta) > max_delta:
            delta = math.copysign(max_delta, delta)

        new_h = h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        # very small feasibility check (avoid round‑off violations)
        if new_h[i] < -1e-12 or new_h[i] > 1.0 + 1e-12:
            continue
        if new_h[j] < -1e-12 or new_h[j] > 1.0 + 1e-12:
            continue

        new_score = compute_c5(new_h)

        # Metropolis acceptance
        if new_score < best_score or rng.random() < math.exp(-(new_score - best_score) / (temperature + 1e-12)):
            h = new_h
            if new_score < best_score:
                best_score = new_score
                best_h = new_h.copy()

    # ----- optional smooth optimisation with scipy (if available) -----
    try:
        from scipy.optimize import minimize
        from scipy.special import logsumexp as _logsumexp

        def smooth_obj(x):
            """Smooth max‑overlap + quadratic penalty for the sum constraint."""
            dx = 2.0 / len(x)
            corr = np.correlate(x, 1.0 - x, mode="full")
            m = np.max(corr)
            s = m + (1.0 / beta_smooth) * _logsumexp(beta_smooth * (corr - m))
            return dx * s + penalty * (np.sum(x) - target_sum) ** 2

        bounds = [(0.0, 1.0) for _ in range(n)]
        res = minimize(
            smooth_obj,
            best_h,
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxiter": 300, "ftol": 1e-12},
        )
        cand = project_onto_box_sum(res.x, target_sum)
        cand_score = compute_c5(cand)
        if cand_score < best_score:
            best_h = cand
            best_score = cand_score
    except Exception:
        # scipy not available or failed – simply continue with SA result
        pass

    # final safety projection (tiny numerical drift)
    best_h = project_onto_box_sum(best_h, target_sum)
    final_score = compute_c5(best_h)

    return best_h, final_score, n
