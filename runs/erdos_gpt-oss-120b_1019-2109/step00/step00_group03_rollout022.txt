def sturmian_binary(n, slope, offset):
    idx = np.arange(n)
    v = (idx * slope + offset) % 1
    # sort indices by v ascending
    # ones are those in first n/2 indices when sorted
    order = np.argsort(v)
    h = np.zeros(n, dtype=float)

import time
import math
import random
import numpy as np

# ---------- helper functions ----------

def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)*(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def binarize_top_half(v: np.ndarray) -> np.ndarray:
    """Return a 0/1 vector with exactly half entries set to 1 (or floor(n/2))."""
    n = v.shape[0]
    k = n // 2
    order = np.argsort(v)[::-1]          # descending
    h = np.zeros(n, dtype=float)
    h[order[:k]] = 1.0
    return h


def generate_sturmian(n: int, slope: float, offset: float) -> np.ndarray:
    """
    Produce a balanced 0/1 word of length n with exactly n/2 ones.
    It is the characteristic word of a Beatty sequence with given slope.
    """
    idx = np.arange(n, dtype=float)
    vals = (idx * slope + offset) % 1.0
    order = np.argsort(vals)            # smallest fractional parts first
    h = np.zeros(n, dtype=float)
    h[order[: n // 2]] = 1.0
    return h

    """
    Simple deterministic hill climbing: try all single swaps until no improvement.
    The number of iterations is limited to avoid pathological loops.
    """
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(best_h)

    for _ in range(limit):
        improved = False
        ones = np.flatnonzero(best_h == 1)
        zeros = np.flatnonzero(best_h == 0)

        # try swaps; stop at first improvement
        for i in ones:
            for j in zeros:
                cand = best_h.copy()
                cand[i] = 0.0
                cand[j] = 1.0
                val = compute_c5(cand)
                if val < best_val - 1e-12:
                    best_h = cand
                    best_val = val
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return best_h, best_val


def simulated_annealing(start_h: np.ndarray,
                        time_limit: float,
                        start_time: float) -> (np.ndarray, float):
    """
    Simulated annealing on the space of constant‑weight binary vectors.
    Swaps a random 1 with a random 0 at each step.
    """
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    # temperature schedule (linear cooling)
    Tmax = 0.02
    Tmin = 1e-6

    # pre‑compute index lists – will be refreshed after each accepted move
    ones = np.flatnonzero(cur_h == 1)
    zeros = np.flatnonzero(cur_h == 0)

    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= time_limit - 0.2:   # keep a small margin
            break

        # temperature linearly decreasing
        t = elapsed / time_limit
        T = Tmax * (1.0 - t) + Tmin * t

        # propose a swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        cand = cur_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_val = compute_c5(cand)

        delta = cand_val - cur_val
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0 and np.random.rand() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_h = cand
            cur_val = cand_val
            # refresh the two index lists ( cheap for n~100 )
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)

            if cur_val < best_val:
                best_h = cur_h.copy()
                best_val = cur_val

        it += 1

        # occasional random restart to avoid being trapped too long
        if it % 25000 == 0 and np.random.rand() < 0.07:
            rnd = np.zeros(n, dtype=float)
            rnd[: n // 2] = 1.0
            np.random.shuffle(rnd)
            cur_h = rnd
            cur_val = compute_c5(cur_h)
            ones = np.flatnonzero(cur_h == 1)
            zeros = np.flatnonzero(cur_h == 0)

    return best_h, best_val


# ---------- main entry point ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    where h_values is a 0/1 numpy array with sum = n_points/2
    and c5_bound is the corresponding max overlap integral.
    """
    start_time = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # Determine problem size
    try:
        n_points = int(len(initial_h_values))
        use_initial = True
    except NameError:
        n_points = 70
        use_initial = False

    # Ensure even number of points (required for exact half‑weight)
    if n_points % 2 != 0:
        n_points += 1

    # ----- generate a collection of promising starts -----
    candidates = []

    # 1. deterministic Sturmian families (golden ratio and a few irrationals)
    irrationals = [(1 + 5 ** 0.5) / 2, np.sqrt(2), np.sqrt(3), np.e, np.pi]
    for slope in irrationals:
        for _ in range(3):
            offset = np.random.rand()
            cand = generate_sturmian(n_points, slope, offset)
            candidates.append(cand)

    # 2. a handful of purely random constant‑weight vectors
    for _ in range(12):
        rand_vec = np.zeros(n_points, dtype=float)
        rand_vec[: n_points // 2] = 1.0
        np.random.shuffle(rand_vec)
        candidates.append(rand_vec)

    # 3. if a parent construction exists, incorporate its binary version
    if use_initial:
        init_bin = binarize_top_half(np.asarray(initial_h_values, dtype=float))
        candidates.append(init_bin)

    # evaluate all candidates, keep the best
    best_h = None
    best_val = float("inf")
    for cand in candidates:
        val = compute_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand.copy()

    # ----- simulated annealing from the current best -----
    remaining_budget = budget_s - (time.time() - start_time)
    if remaining_budget > 10:   # keep a small reserve for final polishing
        best_h, best_val = simulated_annealing(best_h, remaining_budget, start_time)

    # ----- final greedy refinement (fast, deterministic) -----
    best_h, best_val = greedy_local_opt(best_h, limit=3000)

    # Final safety check
    best_val = compute_c5(best_h)

    return best_h, best_val, n_points
