def _max_corr(arr: np.ndarray) -> float:
    """Raw maximum correlation sum for the overlap integral (without dx)."""
    # correlation of h with (1-h) using zero padding (mode='full')
    return float(np.max(np.correlate(arr, 1.0 - arr, mode='full')))
