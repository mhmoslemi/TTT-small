import numpy as np, time

def evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    max_corr = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    return max_corr

def build_h(n,l):
    dx = 2.0 / n
    xs = np.arange(n) * dx + dx/2  # center of intervals
    high_mask = (xs < l) | (xs >= 2 - l)
    # compute needed u to meet integral exactly
    high_len = high_mask.sum() * dx
    middle_len = 2 - high_len
    if middle_len > 0:
        u = (1 - high_len) / middle_len
        u = max(0.0, min(1.0, u))
    else:
        u = 0.0
    h = np.where(high_mask, 1.0, u)
    # adjust any rounding error
    target_sum = n / 2.0
    current_sum = h.sum()
    err = target_sum - current_sum
    if abs(err) > 1e-12:
        # distribute to middle entries (where h not at bounds 0 or 1)
        free_idx = np.where((h > 0) & (h < 1))[0]
        if free_idx.size > 0:
            delta = err / free_idx.size
            h[free_idx] += delta
        else:
            # fallback distribute across all
            h += err / n
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # final fix sum if off due to clipping:
    # compute sum again, adjust smallest interior.
    current_sum = h.sum()
    err = target_sum - current_sum
    if abs(err) > 1e-12:
        # add to any index not at bound
        for idx in free_idx:
            max_inc = 1 - h[idx] if err > 0 else -h[idx]  # delta sign accordingly
            d = np.sign(err) * min(abs(err), abs(max_inc))
            h[idx] += d
            err -= d
            if abs(err) < 1e-12:
                break
    return h

import numpy as np, time, math

def evaluate(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def build_two_end_shape(n, l):
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    high = (xs < l) | (xs >= 2 - l)
    high_len = high.sum() * dx
    middle_len = 2.0 - high_len
    if middle_len <= 0:
        u = 0.0
    else:
        u = (1.0 - high_len) / middle_len
        u = max(0.0, min(1.0, u))
    h = np.where(high, 1.0, u)
    # adjust sum exactly
    target = n / 2.0
    err = target - h.sum()
    if abs(err) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size > 0:
            delta = err / free.size
            h[free] += delta
        else:
            h += err / n
    np.clip(h, 0.0, 1.0, out=h)
    return h

import time, math
import numpy as np

def _evaluate(h):
    """C₅ value for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def _build_two_end_shape(n, l):
    """Shape with value 1 on two end intervals of total length 2*l,
    constant u in the middle, adjusted to satisfy the integral = 1."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx                # sample centres
    high = (xs < l) | (xs >= 2.0 - l)             # ends where h = 1
    high_len = high.sum() * dx
    middle_len = 2.0 - high_len
    if middle_len <= 0.0:
        u = 0.0
    else:
        u = (1.0 - high_len) / middle_len
        u = max(0.0, min(1.0, u))
    h = np.where(high, 1.0, u)
    # enforce exact integral (sum = n/2)
    target = n / 2.0
    err = target - h.sum()
    if abs(err) > 1e-12:
        free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
        if free.size:
            h[free] += err / free.size
        else:
            h += err / n
        np.clip(h, 0.0, 1.0, out=h)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise ---------------------------------------------------------
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        n = len(initial_h_values)
        cur = np.array(initial_h_values, dtype=np.float64)
        cur = np.clip(cur, 0.0, 1.0)
        target = n / 2.0
        diff = target - cur.sum()
        if abs(diff) > 1e-12:
            free = np.where((cur > 0) & (cur < 1))[0]
            if free.size:
                cur[free] += diff / free.size
            else:
                cur += diff / n
            np.clip(cur, 0.0, 1.0, out=cur)
    else:
        n = 200
        cur = None

    best_h = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # a quick scan over simple two‑end shapes ---------------------------
    # ------------------------------------------------------------------
    steps = 401
    for step in range(steps):
        l = 0.5 * step / (steps - 1)
        h = _build_two_end_shape(n, l)
        val = _evaluate(h)
        if val < best_val:
            best_val, best_h = val, h.copy()
        if time.time() - start > budget_s * 0.25:
            break

    # consider the provided initial vector as an extra candidate
    if cur is not None:
        val = _evaluate(cur)
        if val < best_val:
            best_val, best_h = val, cur.copy()
        # a few more shapes to diversify the pool
        for step in range(steps):
            l = 0.5 * step / (steps - 1)
            h = _build_two_end_shape(n, l)
            val = _evaluate(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
            if time.time() - start > budget_s * 0.45:
                break

    # ------------------------------------------------------------------
    # local stochastic improvement ----------------------------------------
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_val = best_val
    n = cur_h.shape[0]

    while True:
        now = time.time()
        if now - start > budget_s * 0.98:
            break

        # shift that realises the current maximum overlap (the worst one)
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        worst_idx = np.argmax(corr)
        s = worst_idx - (n - 1)        # integer shift (positive → right)

        # gradient of the product sum for that shift
        grad = np.zeros_like(cur_h)
        if s >= 0:
            grad[: n - s] += cur_h[s:]
            grad[s:] += cur_h[: n - s]
        else:
            ss = -s
            grad[ss:] += cur_h[: n - ss]
            grad[: n - ss] += cur_h[ss:]

        # choose an index to increase (high gradient, not saturated)
        inc_mask = (grad > 0) & (cur_h < 1 - 1e-12)
        dec_mask = (grad > 0) & (cur_h > 1e-12)
        if inc_mask.any() and dec_mask.any():
            inc_idx = np.where(inc_mask)[0]
            dec_idx = np.where(dec_mask)[0]
            inc_sorted = inc_idx[np.argsort(-grad[inc_idx])]
            dec_sorted = dec_idx[np.argsort(grad[dec_idx])]
            topk = max(1, inc_sorted.size // 5)
            botk = max(1, dec_sorted.size // 5)
            i = inc_sorted[rng.integers(0, topk)]
            j = dec_sorted[rng.integers(0, botk)]
        else:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)

        # feasible transfer amount
        max_inc = 1.0 - cur_h[i]
        max_dec = cur_h[j]
        max_delta = min(max_inc, max_dec)
        if max_delta <= 0:
            continue

        # step size shrinks as time passes
        elapsed = now - start
        frac = elapsed / budget_s
        delta = max_delta * rng.random() * (0.5 * (1 - frac) + 0.02)
        if delta <= 0:
            continue

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_val = _evaluate(new_h)

        # deterministic improvement
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_val, best_h = new_val, new_h.copy()
            continue

        # simulated‑annealing acceptance
        T = 0.02 * (1 - frac) + 1e-6
        if rng.random() < math.exp(-(new_val - cur_val) / T):
            cur_h, cur_val = new_h, new_val

        # occasional tiny random perturbation to keep diversity
        if rng.random() < 0.01:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            while j == i:
                j = rng.integers(0, n)
            max_inc = 1.0 - cur_h[i]
            max_dec = cur_h[j]
            max_delta = min(max_inc, max_dec)
            if max_delta > 0:
                delta = max_delta * rng.random() * 0.02
                tmp = cur_h.copy()
                tmp[i] += delta
                tmp[j] -= delta
                tmp_val = _evaluate(tmp)
                if tmp_val < cur_val:
                    cur_h, cur_val = tmp, tmp_val
                    if tmp_val < best_val:
                        best_val, best_h = tmp_val, tmp.copy()

    # ------------------------------------------------------------------
    # final polishing ----------------------------------------------------
    # ------------------------------------------------------------------
    np.clip(best_h, 0.0, 1.0, out=best_h)
    target = best_h.shape[0] / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        free = np.where((best_h > 1e-12) & (best_h < 1 - 1e-12))[0]
        if free.size:
            best_h[free] += diff / free.size
        else:
            best_h += diff / best_h.shape[0]
        np.clip(best_h, 0.0, 1.0, out=best_h)

    final_val = _evaluate(best_h)
    return (best_h.tolist(), float(final_val), best_h.shape[0])
