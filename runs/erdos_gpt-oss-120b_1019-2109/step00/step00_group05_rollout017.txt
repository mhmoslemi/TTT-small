import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = corr.max()
    return max_corr * dx, corr

def proj_feasible(y, target):
    # solve x = clip(y + lam, 0,1) s.t. sum(x)=target
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(50):

        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(y + (lo + hi) / 2.0, 0, 1)

def gradient_subgrad(h, k):
    n = len(h)

        grad[k:] -= h[:n - k]

    n = h.shape[0]

def project(v, target):
    lo = -np.max(v) - 2.0   # guarantee below
    hi = 1.0 - np.min(v) + 2.0  # above
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # After binary search, may not exactly hit target due to clipping; adjust tiny diff with something.
    # Use small correction to ensure sum close.
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # adjust uniformly among elements not at bounds
        free = (x > 0) & (x < 1)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            # clip again
            x = np.clip(x, 0, 1)
        else:
            # fallback: scale entire vector
            x = np.clip(x + diff / len(x), 0, 1)
    return x
```

import numpy as np
import time
import math

def _project_to_box_sum(v, target):
    """Project v onto the set {x: 0<=x<=1, sum(x)=target}."""
    # binary search for lambda such that sum(clip(v+lam,0,1)) = target
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if needed
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            inc = diff / free.sum()
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback uniform adjustment (should not happen often)
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_and_corr(h):
    """Return C5 bound and full correlation array for vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = corr.max()
    c5 = max_corr * (2.0 / n)
    return c5, corr

def _subgrad(h, shift):
    """Subgradient of the active linear term for given shift."""
    n = h.shape[0]
    grad = np.ones(n)
    if shift == 0:
        grad = 1.0 - 2.0 * h
    elif shift > 0:
        k = shift
        grad[: n - k] -= h[k:]
        grad[k:] -= h[: n - k]
    else:
        k = -shift
        grad[k:] -= h[: n - k]
        grad[: n - k] -= h[k:]
    return grad

def _triangular_init(n):
    """Triangular function on [0,2] with height 1 and area 1."""
    dx = 2.0 / n
    # sample at cell centres
    xs = (np.arange(n) + 0.5) * dx
    h = np.maximum(0.0, 1.0 - np.abs(xs - 1.0))
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    time_limit = float(budget_s) - 0.05  # leave a tiny margin

    # ----- problem size -------------------------------------------------
    # use a modest but flexible size
    n = 200
    target_sum = n / 2.0

    # ----- initial vector ------------------------------------------------
    init_vec = None
    if 'initial_h_values' in globals():
        try:
            iv = np.asarray(initial_h_values, dtype=float)
            if iv.ndim != 1:
                iv = iv.ravel()
            init_vec = _project_to_box_sum(np.clip(iv, 0.0, 1.0), target_sum)
        except Exception:
            init_vec = None

    if init_vec is None:
        init_vec = _triangular_init(n)
        init_vec = _project_to_box_sum(np.clip(init_vec, 0.0, 1.0), target_sum)

    # ----- tracking best solution ----------------------------------------
    best_h = init_vec.copy()
    best_c5, _ = _c5_and_corr(best_h)

    # ----- main optimisation loop ----------------------------------------
    # We'll repeatedly perform a short deterministic descent followed by a
    # stochastic annealing phase, restarting when time permits.
    while time.time() - start < time_limit:
        # start from the current best with a small random perturbation
        cur = best_h.copy()
        # small random jitter to escape plateaus
        jitter = np.random.normal(scale=0.02, size=n)
        cur += jitter
        cur = np.clip(cur, 0.0, 1.0)
        cur = _project_to_box_sum(cur, target_sum)

        # ---- deterministic subgradient descent -------------------------
        max_iter = 4000
        for it in range(max_iter):
            if time.time() - start >= time_limit:
                break
            c5, corr = _c5_and_corr(cur)
            # update global best
            if c5 < best_c5:
                best_c5, best_h = c5, cur.copy()
            # locate active shift
            idx = int(np.argmax(corr))
            shift = idx - (n - 1)
            grad = _subgrad(cur, shift)
            # step size decreasing like 0.5 / sqrt(it+1)
            step = 0.5 / math.sqrt(it + 1.0)
            improved = False
            for factor in (1.0, 0.5, 0.25):
                cand = _project_to_box_sum(cur - step * factor * grad, target_sum)
                cand_c5, _ = _c5_and_corr(cand)
                if cand_c5 < c5 - 1e-12:
                    cur = cand
                    c5 = cand_c5
                    improved = True
                    break
            if not improved:

                continue

        cur_c5, _ = _c5_and_corr(cur)

        T0 = 0.02
        moves = 0
        max_moves = 25000
        while time.time() - start < time_limit and moves < max_moves:

            i, j = np.random.choice(n, size=2, replace=False)
            # allowed delta preserving bounds
            delta_max = min(0.12, 1.0 - cur[i], cur[j])
            delta_min = -min(0.12, cur[i], 1.0 - cur[j])

                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, cur.copy()
            moves += 1

        # quick escape if time is almost over
        if time.time() - start > time_limit - 0.1:
            break

    # final projection to guarantee feasibility (tiny numerical drift)
    best_h = _project_to_box_sum(best_h, target_sum)
    final_c5, _ = _c5_and_corr(best_h)
    return (best_h.tolist(), float(final_c5), n)
