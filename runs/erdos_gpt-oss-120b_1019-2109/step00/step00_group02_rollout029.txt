def project_onto_feasible(v, target):
    # Clip to [0,1]
    # Use binary search for lambda...
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo+hi)/2.0, 0, 1)
    # adjust residual if small deviation
    res = target - proj.sum()
    if abs(res) > 1e-8:
        # distribute residual among non-boundary entries
        # find indices where proj not at 0 or 1
        free = (proj > 0) & (proj < 1)
        if free.any():
            # distribute equally
            inc = res / free.sum()
            proj[free] += inc
            # clip again
            proj = np.clip(proj, 0, 1)
        else:
            # if no free, adjust any index with capacity
            # allocate to first with capacity
            for i in range(len(proj)):
                cap_up = 1 - proj[i]
                cap_down = proj[i]
                if res > 0 and cap_up > 1e-6:
                    add = min(res, cap_up)
                    proj[i] += add
                    res -= add
                elif res < 0 and cap_down > 1e-6:
                    sub = min(-res, cap_down)
                    proj[i] -= sub
                    res += sub
                if abs(res) < 1e-9:
                    break
        # final clip
        proj = np.clip(proj, 0, 1)
    return proj

def compute_total_gradient(h, C_vec, beta, dx):
    maxC = C_vec.max()
    # compute exponentials
    exps = np.exp(beta * (C_vec - maxC))
    weights = exps / exps.sum()
    N = len(h)
    g_total = np.zeros(N)
    # iterate through lags
    for idx, w in enumerate(weights):
        if w == 0:
            continue
        d = idx - (N - 1)
        # compute g_d as defined earlier, without dx factor (we'll multiply later)
        if d >= 0:
            g = np.zeros(N)
            g[:N-d] = 1 - h[d:]
            if d > 0:
                g[d:] -= h[:N-d]
        else:
            s = -d
            g = np.zeros(N)
            g[s:] = 1 - h[:N-s]
            g[:N-s] -= h[s:]
        g_total += w * g
    return g_total * dx

def optimize_h(h0, budget):
    h = h0.copy()
    N = len(h)
    target_sum = N / 2.0
    dx = 2.0 / N
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()
    # initial step size and decay
    lr = 0.5  # maybe adapt
    decay = 0.9995
    iter = 0
    while time.time() - start < budget:
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        max_idx = np.argmax(corr)
        cur_val = corr[max_idx]
        # update best
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # gradient for worst shift
        d = max_idx - (N - 1)
        # compute gradient
        g = np.zeros(N)
        if d == 0:
            g = 1 - 2 * h
        elif d > 0:
            g[:N-d] = 1 - h[d:]
            g[d:] -= h[:N-d]
        else:  # d < 0
            s = -d
            g[s:] = 1 - h[:N-s]
            g[:N-s] -= h[s:]
        g = g * dx
        # update with step
        h_new = h - lr * g
        # project to feasible
        h_new = project_onto_feasible(h_new, target_sum)
        # compute new corr for acceptance
        corr_new = np.correlate(h_new, 1 - h_new, mode='full') * dx
        new_val = np.max(corr_new)
        # accept if improved else maybe accept with probability
        if new_val <= cur_val:
            h = h_new
            cur_val = new_val
        else:
            # optionally accept with probability
            # compute temperature schedule:
            # T decays
            T = max(0.001, 0.1 * (1 - iter / 50000))
            if np.random.rand() < np.exp(-(new_val - cur_val) / T):
                h = h_new
                cur_val = new_val
        # decay step size
        lr *= decay
        # occasionally random perturb
        if iter % 1000 == 0 and np.random.rand() < 0.5:
            # random swap pair (i,j)
            i = np.random.randint(N)
            j = np.random.randint(N)
            if i != j:
                # propose moving epsilon from j to i
                eps = 0.1 * min(1 - h[i], h[j])
                if eps > 1e-8:
                    h2 = h.copy()
                    h2[i] += eps
                    h2[j] -= eps
                    h2 = project_onto_feasible(h2, target_sum) # maybe unnecessary as sum unchanged
                    # compute val
                    val2 = np.max(np.correlate(h2, 1 - h2, mode='full') * dx)
                    if val2 < cur_val:
                        h = h2
                        cur_val = val2
        iter += 1
    return best_h, best_val

    lo = -2.0
    hi = 2.0
