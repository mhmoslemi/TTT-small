import time
import numpy as np

def _compute_c5(vec, dx):
    """Return C5 for a given step function."""
    # correlation of h with (1‑h) using full mode (zero padding)
    return np.max(np.correlate(vec, 1.0 - vec, mode="full")) * dx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1 and 0≤h≤1
    to minimize  C5 = max_k ∫ h(x)(1‑h(x+k))dx.

    Returns
    -------
    h_values : np.ndarray
        Optimized values on a uniform grid.
    c5_bound : float
        The achieved C5 bound.
    n_points : int
        Number of discretisation points.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64)
    else:
        # fallback: 50 % of points set to 1 at random
        n = 70
        h = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0

    n = h.shape[0]
    dx = 2.0 / n

    # ensure feasibility (tiny numerical noise may appear)
    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # adjust a single free entry to fix the total mass
        diff = target_sum - cur_sum
        for i in range(n):
            if (diff > 0 and h[i] < 1.0) or (diff < 0 and h[i] > 0.0):
                adj = min(1.0 - h[i], diff) if diff > 0 else max(-h[i], diff)
                h[i] += adj
                diff -= adj
                if abs(diff) < 1e-12:
                    break
        if abs(diff) > 1e-12:   # last resort: uniform correction
            h += diff / n

    # ------------------------------------------------------------------
    # basic local search / simulated annealing
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _compute_c5(best_h, dx)

    start_time = time.time()
    end_time = start_time + budget_s - 0.1   # leave a small safety margin

    max_step = 0.08          # maximal amount transferred in one move
    init_temp = 0.012        # starting temperature for SA
    eps = 1e-12

    # keep current state values
    cur_h = h.copy()
    cur_c5 = best_c5

    while time.time() < end_time:
        # correlation of the current candidate
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)          # positive ⇒ right shift, negative ⇒ left

        # tail indices that contribute directly to the current worst overlap
        if shift > 0:
            tail = np.arange(n - shift, n)
        elif shift < 0:
            tail = np.arange(0, -shift)
        else:
            tail = np.empty(0, dtype=int)

        # try to move mass from a heavy tail element to a lighter interior one
        if tail.size > 0:
            # candidates in the tail that still hold some mass
            tail_candidates = tail[cur_h[tail] > eps]
            # interior indices (complement of tail)
            interior_mask = np.ones(n, dtype=bool)
            interior_mask[tail] = False
            interior = np.where(interior_mask)[0]
            interior_candidates = interior[cur_h[interior] < 1.0 - eps]
        else:
            tail_candidates = np.empty(0, dtype=int)
            interior_candidates = np.empty(0, dtype=int)

        if tail_candidates.size and interior_candidates.size:
            i = int(rng.choice(tail_candidates))
            j = int(rng.choice(interior_candidates))
        else:

            i, j = rng.choice(n, size=2, replace=False)

        # feasible transfer amount
        delta_max = min(cur_h[i], 1.0 - cur_h[j])
        if delta_max <= eps:

        # random step up to max_step (or the whole feasible amount if smaller)
        delta = min(delta_max, max_step) * rng.random()
        if delta <= eps:
            continue

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_c5 = _compute_c5(new_h, dx)

        elapsed = time.time() - start_time
        progress = elapsed / (budget_s - 0.1)
        temp = init_temp * (1.0 - progress)

        if new_c5 < cur_c5 - 1e-12:
            cur_h, cur_c5 = new_h, new_c5
        else:
            if temp > 0.0 and rng.random() < np.exp(-(new_c5 - cur_c5) / temp):
                cur_h, cur_c5 = new_h, new_c5

        # store global best
        if cur_c5 < best_c5 - 1e-12:
            best_h, best_c5 = cur_h.copy(), cur_c5

    if abs(diff) > 1e-10:

        if abs(diff) > 1e-12:
            best_h += diff / n

    final_c5 = _compute_c5(best_h, dx)

    return best_h, float(final_c5), n
