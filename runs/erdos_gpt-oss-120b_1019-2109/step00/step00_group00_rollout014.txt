import time
import numpy as np
import random
import math

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant for a given discretized function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h) – the same as used in the evaluation harness
    o = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(o.max())

def project_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the feasible set:
        0 <= x_i <= 1  and  sum x_i = target_sum
    using a binary‐search on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0          # wide enough bounds for the multiplier
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam            # need larger lambda to decrease the sum
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny numerical fix to hit the target sum exactly
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            add = diff / free.sum()
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x

def smooth_min_weights(S: np.ndarray, alpha: float) -> np.ndarray:
    """
    Return the normalized weights w_i = exp(-alpha * S_i) / sum ...
    using a numerically stable shift.
    """
    shift = S.max()
    exp_term = np.exp(-alpha * (S - shift))
    total = exp_term.sum()
    if total == 0.0:
        return np.full_like(S, 1.0 / S.size)
    return exp_term / total

def gradient_min_corr(h: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth minimum of the self‑correlation of h.
    The gradient is   g_j = sum_k w_k (h_{j+k}+h_{j-k}) .
    """
    N = h.shape[0]
    S = np.correlate(h, h, mode="full")          # length 2N-1
    w = smooth_min_weights(S, alpha)
    # convolution with w and reversed w gives the two summations
    conv1 = np.convolve(w, h, mode="full")       # w * h
    conv2 = np.convolve(w[::-1], h, mode="full")  # w_rev * h
    start = N - 1
    grad = conv1[start:start + N] + conv2[start:start + N]
    return grad

def local_optimize(start_h: np.ndarray,
                   max_iters: int,
                   time_deadline: float) -> tuple:
    """
    Perform a simple projected gradient ascent on the smooth‑min of the
    self‑correlation. Returns (best_h, best_c5) found during the run.
    """
    N = start_h.shape[0]
    target = N / 2.0
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # schedule for the smoothing parameter
    alpha_start = 5.0
    alpha_end = 30.0

    for it in range(max_iters):
        # time check
        if time.time() >= time_deadline:
            break

        # linearly increase smoothing (focus more on the worst shift later)
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        grad = gradient_min_corr(h, alpha)

        # simple diminishing step size
        eta = 0.6 * (1.0 - it / max_iters)

        h = project_box_sum(h + eta * grad, target)

        # periodic evaluation
        if (it & 63) == 0:          # every 64 iterations
            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
            # early exit if we have already broken the target record
            if best_c5 < 0.3808:
                break

    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a small upper bound
    for the Erdős C5 constant.  Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Determine problem size (use the provided initial vector if possible)
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)   # provided by the harness
        n = init.shape[0]
        # ensure feasibility
        init = project_box_sum(init, n / 2.0)
        best_h, best_c5 = init, compute_c5(init)
    except NameError:
        # no initial vector – fall back to a modest size
        n = 200
        best_h = np.full(n, 0.5)
        best_h = project_box_sum(best_h, n / 2.0)
        best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------------
    # Main optimisation loop – several restarts with short gradient runs
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # random feasible start
        h0 = np.random.rand(n)
        h0 = project_box_sum(h0, n / 2.0)

        # a quick gradient ascent (couple thousand iterations)
        h_opt, c5_opt = local_optimize(
            h0,
            max_iters=3000,
            time_deadline=deadline,
        )

        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt

        # if we already beat the advertised target stop early
        if best_c5 < 0.3808:
            break

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_c5), int(n))
