    N_GLOBAL = n
    DX_GLOBAL = 2.0 / n
    target_sum = n / 2.0

    best_h = None
    best_c5 = np.inf

    # -------------------------------------------------------------
    # 2. Start from the provided construction if available
    try:
        init = initial_h_values          # may be supplied by the harness
        n0 = init.shape[0]
        x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
        x = (np.arange(n) + 0.5) * DX_GLOBAL
        h0 = np.interp(x, x0, init)
        # scale to meet the integral constraint
        scale = target_sum / h0.sum()
        h0 = np.clip(h0 * scale, 0.0, 1.0)
        # final exact correction
        diff = target_sum - h0.sum()
        free = np.where((h0 > 1e-12) & (h0 < 1 - 1e-12))[0]
        if free.size:
            h0[free] += diff / free.size
            h0 = np.clip(h0, 0.0, 1.0)
        c5_0 = _c5_from_h(h0)
        best_h, best_c5 = h0, c5_0
    except NameError:
        pass   # no initial_h_values supplied

    # -------------------------------------------------------------
    # 3. Simple cosine bump (a1=0.5) as a deterministic baseline
    coeffs0 = np.array([0.5] + [0.0] * 6)   # 7 coefficients, only a1 non‑zero
    h_cos = _make_h_from_coeffs(coeffs0)
    c5_cos = _c5_from_h(h_cos)
    if c5_cos < best_c5:
        best_h, best_c5 = h_cos, c5_cos

    # -------------------------------------------------------------
    # 4. Global search using Differential Evolution on a few Fourier modes
    M = 7                                 # number of cosine terms
    bounds = [(-0.5, 0.5) for _ in range(M)]
    # time‑aware max iterations (roughly 15000 evaluations ≈ < 30 s)
    max_iter = 250
    de_result = _opt.differential_evolution(
        _objective,
        bounds,
        seed=seed,
        maxiter=max_iter,
        popsize=15,
        polish=False,
        disp=False,
        updating='deferred',
    )
    h_de = _make_h_from_coeffs(de_result.x)
    c5_de = _c5_from_h(h_de)
    if c5_de < best_c5:
        best_h, best_c5 = h_de, c5_de

    # -------------------------------------------------------------
    # 5. Local refinement on the best solution found so far
    h = best_h.copy()
    c5 = best_c5
    it = 0
    while time.time() - start_time < budget_s - 0.5:
        # compute full correlation and locate the worst lag
        corr = np.correlate(h, 1.0 - h, mode="full") * DX_GLOBAL
        max_idx = int(corr.argmax())
        lag = max_idx - (n - 1)                 # lag in [-(n-1), n-1]
        grad = _gradient_max_lag(h, lag)

        # step size schedule (decays to zero)
        alpha = 0.3 * (1.0 - it / 20000.0)
        if alpha <= 0:
            break

        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        # re‑impose the integral constraint
        diff = target_sum - h_new.sum()
        free = np.where((h_new > 1e-12) & (h_new < 1 - 1e-12))[0]
        if free.size:
            h_new[free] += diff / free.size
            h_new = np.clip(h_new, 0.0, 1.0)
        else:
            scale = target_sum / h_new.sum() if h_new.sum() > 0 else 1.0
            h_new = np.clip(h_new * scale, 0.0, 1.0)

        new_c5 = _c5_from_h(h_new)

        # accept if improvement; otherwise occasional uphill move (simulated annealing)
        if new_c5 < c5:
            h, c5 = h_new, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = h_new.copy(), new_c5
        else:
            # temperature schedule
            T = 0.02 * max(0.0, 1.0 - it / 40000.0)
            if T > 0 and np.random.rand() < np.exp(-(new_c5 - c5) / T):
                h, c5 = h_new, new_c5
        it += 1
        if time.time() - start_time > budget_s - 0.2:
            break

    # -------------------------------------------------------------
    # 6. Final sanity check – enforce the exact integral constraint
    s = best_h.sum()
    if abs(s - target_sum) > 1e-9:
        diff = target_sum - s
        free = np.where((best_h > 1e-12) & (best_h < 1 - 1e-12))[0]
        if free.size:
            best_h[free[0]] = np.clip(best_h[free[0]] + diff, 0.0, 1.0)
        else:
            factor = target_sum / s if s > 0 else 1.0
            best_h = np.clip(best_h * factor, 0.0, 1.0)

    final_c5 = _c5_from_h(best_h)

    return best_h, float(final_c5), n
