import numpy as np, random, time, math

def _raw_corr(h: np.ndarray) -> np.ndarray:
    n = h.size
    m = 2 * n - 1
    # convolution of h with reversed (1-h)
    return np.fft.ifft(np.fft.fft(h, m) * np.fft.fft((1 - h)[::-1], m)).real

def _eval_c5(h: np.ndarray) -> float:
    n = h.size
    return _raw_corr(h).max() * (2.0 / n)

def _hill_climb(start_h: np.ndarray, max_step: float, time_budget: float):
    n = start_h.size
    h = start_h.copy()
    raw = _raw_corr(h)
    best_h = h.copy()
    best_raw = raw.max()
    cur_raw = best_raw
    start = time.time()
    p_rand = 0.1                     # chance to do a completely random move
    while time.time() - start < time_budget:
        idx = int(np.argmax(raw))
        shift = idx - (n - 1)        # shift = k/dx, may be negative
        # choose donor/receiver
        if random.random() < p_rand:
            donor = random.randrange(n)
            receiver = random.randrange(n)
            while donor == receiver:
                receiver = random.randrange(n)
        else:
            if shift >= 0:
                lo, hi = 0, n - shift
            else:
                lo, hi = -shift, n
            if hi - lo <= 0:
                donor = random.randrange(n)
                receiver = random.randrange(n)
                while donor == receiver:
                    receiver = random.randrange(n)
            else:
                i = random.randrange(lo, hi)
                j = i + shift
                if random.random() < 0.5:
                    donor, receiver = i, j
                else:
                    donor, receiver = j, i
        down = h[donor]                     # amount we can take from donor
        up = 1.0 - h[receiver]              # room at receiver
        delta_max = min(down, up, max_step)
        if delta_max <= 1e-12:
