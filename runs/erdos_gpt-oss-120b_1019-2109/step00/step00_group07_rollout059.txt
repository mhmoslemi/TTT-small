def subgradient_descent(x0, max_iters, lr0):
    h = x0.copy()
    for it in range(max_iters):
        c_raw = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(c_raw)
        k = max_idx - (n - 1)
        # compute shifted lows
        if k > 0:
            shift_pos = np.concatenate([np.zeros(k), h[:-k]])
            shift_neg = np.concatenate([h[k:], np.zeros(k)])
        elif k < 0:
            kk = -k
            shift_pos = np.concatenate([h[kk:], np.zeros(kk)])
            shift_neg = np.concatenate([np.zeros(kk), h[:-kk]])
        else:
            shift_pos = h
            shift_neg = h
        grad = 1.0 - shift_pos - shift_neg
        lr = lr0 / np.sqrt(it + 1)
        h = h - lr * grad
        # clip and project
        h = np.clip(h, 0.0, 1.0)
        h = project(h)
    return h

import numpy as np, time, math

def shift(arr, k):
    n = len(arr)
    if k > 0:
        return np.concatenate([np.zeros(k), arr[:n-k]])
    elif k < 0:
        k = -k
        return np.concatenate([arr[k:], np.zeros(k)])
    else:
        return arr

def max_corr(h):
    c = np.correlate(h, 1 - h, mode='full')
    return np.max(c)  # raw, before scaling

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return max_corr(h) * dx

def project_onto_bounds_and_sum(h, target):
    lo, hi = -2.0, 2.0
    # binary search for lambda shift
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(h + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    return y

def gradient_softmax(h, alpha):
    n = len(h)
    c = np.correlate(h, 1 - h, mode='full')
    a = alpha * c
    max_a = np.max(a)
    exp_a = np.exp(a - max_a)
    sum_exp = np.sum(exp_a)
    p = exp_a / sum_exp
    grad = np.zeros_like(h)
    for idx, w in enumerate(p):
        k = idx - (n - 1)
        if k > 0:
            shifted = np.concatenate([np.zeros(k), h[:n - k]])
            shifted_neg = np.concatenate([h[k:], np.zeros(k)])
        elif k < 0:
            kk = -k
            shifted = np.concatenate([h[kk:], np.zeros(kk)])
            shifted_neg = np.concatenate([np.zeros(kk), h[:n - kk]])
        else:
            shifted = h
            shifted_neg = h
        grad += w * (1.0 - shifted - shifted_neg)
    return grad

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: shift an array with zero padding (linear correlation convention)
def _shift(arr: np.ndarray, k: int) -> np.ndarray:
    n = arr.size
    if k > 0:
        return np.concatenate([np.zeros(k, dtype=arr.dtype), arr[: n - k]])
    if k < 0:
        k = -k
        return np.concatenate([arr[k:], np.zeros(k, dtype=arr.dtype)])
    return arr


# ----------------------------------------------------------------------
# Projection onto the box [0,1] intersected with the hyperplane sum=h_target
def _project_box_sum(x: np.ndarray, h_target: float) -> np.ndarray:
    lo, hi = -2.0, 2.0               # guarantees lo -> sum=0, hi -> sum=n
    for _ in range(60):            # binary search for the Lagrange multiplier
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < h_target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)


# ----------------------------------------------------------------------
# Correlation (linear, not circular) and the C5 value
def _max_corr_raw(h: np.ndarray) -> float:
    return np.max(np.correlate(h, 1.0 - h, mode="full"))


def _c5_value(h: np.ndarray) -> float:
    n = h.size
    dx = 2.0 / n
    return _max_corr_raw(h) * dx


# ----------------------------------------------------------------------
# Soft‑max smoothed objective and its gradient
def _softmax_grad(h: np.ndarray, alpha: float):
    n = h.size
    # raw correlation values (no dx)
    c_raw = np.correlate(h, 1.0 - h, mode="full")
    a = alpha * c_raw
    a_max = np.max(a)
    exp_a = np.exp(a - a_max)
    sum_exp = np.sum(exp_a)
    probs = exp_a / sum_exp                     # soft‑max weights, sum to 1

    grad = np.zeros_like(h)
    for idx, w in enumerate(probs):
        k = idx - (n - 1)                       # shift corresponding to this entry
        if k > 0:
            sh_pos = np.concatenate([np.zeros(k), h[: n - k]])
            sh_neg = np.concatenate([h[k:], np.zeros(k)])
        elif k < 0:
            kk = -k
            sh_pos = np.concatenate([h[kk:], np.zeros(kk)])
            sh_neg = np.concatenate([np.zeros(kk), h[: n - kk]])
        else:
            sh_pos = h
            sh_neg = h
        grad += w * (1.0 - sh_pos - sh_neg)
    return grad


# ----------------------------------------------------------------------
# Simple projected gradient descent (soft‑max surrogate)
def _local_optimize(start_h: np.ndarray,
                    h_target: float,
                    max_iter: int = 2000,
                    lr0: float = 0.7,
                    alpha: float = 15.0) -> tuple[np.ndarray, float]:
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_value(h)

    for it in range(1, max_iter + 1):
        lr = lr0 / math.sqrt(it)
        g = _softmax_grad(h, alpha)
        h -= lr * g
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, h_target)

        cur = _c5_value(h)
        if cur < best_val:
            best_val = cur
            best_h = h.copy()
    return best_h, best_val
