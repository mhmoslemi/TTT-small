import numpy as np, time

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def random_move(h, max_step):
    N = len(h)
    i, j = np.random.choice(N, size=2, replace=False)
    # delta bounded such that after adding to i, subtract from j, both stay between 0 and 1
    # h_i can change by delta ∈ [ -h_i, 1 - h_i]
    # h_j changes opposite, must stay in [0,1] also
    delta_max_pos = min(1 - h[i], h[j])   # positive delta (increase i, decrease j)
    delta_max_neg = min(h[i], 1 - h[j])   # negative delta (decrease i, increase j)
    # choose delta uniformly within [-delta_max_neg, delta_max_pos], scaled by random factor
    if delta_max_pos + delta_max_neg == 0:
        return None, None
    delta = (np.random.rand() * (delta_max_pos + delta_max_neg) - delta_max_neg)
    # optionally limit magnitude to max_step (if needed)
    delta = np.clip(delta, -max_step, max_step)
    # ensure still within allowed range
    new_i = h[i] + delta
    new_j = h[j] - delta
    if not (0 <= new_i <= 1 and 0 <= new_j <= 1):
        return None, None
    h_new = h.copy()
    h_new[i] = new_i
    h_new[j] = new_j
    return h_new, (i, j, delta)

def propose_move(h, max_delta):
    n = len(h)
    # choose i and j distinct
    i, j = np.random.randint(n), np.random.randint(n-1)
    if j >= i:
        j += 1
    # The move delta can be positive (transfer from j->i) or negative (transfer from i->j)
    # Determine allowed delta min and max.
    # Transfer amount up to min(1 - h[i], h[j]) if moving from j to i
    # Or up to min(h[i], 1 - h[j]) if moving from i to j.
    # Choose direction randomly.
    if np.random.rand() < 0.5:
        # direction i gains
        delta_max = min(1.0 - h[i], h[j])
        delta = np.random.rand() * delta_max  # uniform in [0, delta_max]
    else:
        # direction j gains
        delta_max = min(h[i], 1.0 - h[j])
        delta = -np.random.rand() * delta_max  # negative delta (i loses)
    # optionally limit magnitude to max_delta
    if abs(delta) > max_delta:
        delta = np.sign(delta) * max_delta
        # after this scaling, ensure still within allowed bounds: might cause exceed, but we can simply recompute allowed bound based on max_delta
        if delta > 0:
            delta = min(delta, 1.0 - h[i], h[j])
        else:
            delta = -min(-delta, h[i], 1.0 - h[j])
    # apply
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    # maximum value
    max_val = np.max(corr)
    return max_val * dx

import time
import math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a step function h sampled on a uniform grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using zero padding (np.correlate does exactly this)
    val = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    return float(val)


def _project(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Project a vector onto the feasible set 0≤h≤1 and sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    # simple iterative redistribution of the residual
    for _ in range(20):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1.0 - h
            tot = slack.sum()
            if tot == 0.0:
                break
            h += diff * slack / tot
        else:
            slack = h
            tot = slack.sum()
            if tot == 0.0:
                break
            h += diff * slack / tot
        h = np.clip(h, 0.0, 1.0)
    # tiny final fix
    s = h.sum()
    if abs(s - target_sum) > 1e-9:
        idx = np.random.randint(h.size)
        adj = target_sum - s
        new_val = h[idx] + adj
        h[idx] = float(min(1.0, max(0.0, new_val)))
    return h


def _propose_move(h: np.ndarray, max_delta: float) -> np.ndarray:
    """Return a new feasible vector obtained by moving mass between two entries."""
    n = h.shape[0]
    # pick two distinct indices
    i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    # maximal transfer that keeps both entries in [0,1]
    inc_max = min(1.0 - h[i], h[j])      # amount we could add to i (remove from j)
    dec_max = min(h[i], 1.0 - h[j])      # amount we could remove from i (add to j)
    if inc_max <= 0.0 and dec_max <= 0.0:
        return h.copy()
    if np.random.rand() < 0.5:
        delta = np.random.rand() * inc_max          # i gains
    else:
        delta = -np.random.rand() * dec_max         # i loses
    # clamp to the supplied maximum step size
    if max_delta is not None:
        if delta > max_delta:
            delta = max_delta
        elif delta < -max_delta:
            delta = -max_delta
        # ensure still feasible after clamping
        if delta > 0.0:
            delta = min(delta, inc_max)
        else:
            delta = -min(-delta, dec_max)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new
