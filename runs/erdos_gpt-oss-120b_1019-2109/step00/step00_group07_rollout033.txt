import numpy as np
import random
import time
import math

def evaluate_vec(vec: np.ndarray) -> float:
    """Return C5 for a step function given by vec (dx = 2/len(vec))."""
    n = vec.shape[0]
    # correlation h * (1-h shifted) summed over all integer shifts
    return np.max(np.correlate(vec, 1.0 - vec, mode="full")) * (2.0 / n)


def random_binary(n: int) -> np.ndarray:
    """Binary vector of length n with exactly n//2 ones."""
    v = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, n // 2, replace=False)
    v[ones] = 1.0
    return v


def progression_binary(n: int, step: int) -> np.ndarray:
    """Arithmetic‑progression construction: take n//2 indices i, i+step, … (mod n)."""
    v = np.zeros(n, dtype=np.float64)
    idx = 0
    for _ in range(n // 2):
        v[idx] = 1.0
        idx = (idx + step) % n
    return v


def anneal(start_vec: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing improving the max‑overlap bound."""
    n = start_vec.shape[0]
    h = start_vec.copy()
    cur = evaluate_vec(h)
    best_h = h.copy()
    best_val = cur

    T0 = 5.0                 # initial temperature
    max_iter = 60000         # hard cap; loop may stop earlier by time

    for it in range(max_iter):
        if time.time() > time_limit:
            break

        # propose swapping a 1 and a 0
        i = random.randrange(n)
        j = random.randrange(n)
        while h[i] == h[j]:
            i = random.randrange(n)
            j = random.randrange(n)

        hi, hj = h[i], h[j]
        h[i], h[j] = hj, hi
        new_val = evaluate_vec(h)

        # linear cooling
        T = T0 * (1.0 - it / max_iter)

        accept = False
        if new_val < cur:
            accept = True
        else:
            if T > 1e-12 and random.random() < math.exp(-(new_val - cur) / T):
                accept = True

        if accept:
            cur = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert the swap
            h[i], h[j] = hi, hj

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a small Erdős C₅ bound."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # -----------------------------------------------------------------
    # Determine problem size.
    # If a parent vector exists, keep its length; otherwise use a generous n.
    # -----------------------------------------------------------------
    try:
        init_raw = initial_h_values               # provided by the harness
        init = np.asarray(init_raw, dtype=np.float64)
        n = init.shape[0]
        use_init = True
    except NameError:
        n = 800                                   # default length (even)
        use_init = False

    if n % 2 == 1:
        n += 1

    best_h = None
    best_val = float("inf")

    if use_init:
        try:
            val = evaluate_vec(init)
            best_h = init.copy()
            best_val = val
        except Exception:
            pass

    target = 0.3808
    if best_val <= target:
        return best_h, best_val, n

    # time budget for the optimisation loop (reserve a little margin)
    end_time = start_time + budget_s * 0.96

    while time.time() < end_time:
        # ------- generate a fresh starting point -----------------------
        if random.random() < 0.5:
            start_vec = random_binary(n)
        else:

        best_h = random_binary(n)
        best_val = evaluate_vec(best_h)

    return best_h, best_val, n
