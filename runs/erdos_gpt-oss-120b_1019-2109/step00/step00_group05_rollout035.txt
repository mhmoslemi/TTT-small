import cvxpy as cp
import numpy as np, time

def solve_ilp(n):
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == n/2]
    w_vars = {}
    # For shift d from 1 to n-1:
    for d in range(1, n):
        L = n - d
        # w_{i,d} for i=0..L-1
        w = cp.Variable(L, boolean=True)  # vector
        w_vars[(d)] = w
        # linking constraints
        constraints += [
            w <= x[:L],
            w <= x[d:],
            w >= x[:L] + x[d:] - 1
        ]
        # compute C_d and C_-d
        c_pos = cp.sum(x[:L]) - cp.sum(w)        # shift +d
        c_neg = cp.sum(x[d:]) - cp.sum(w)        # -d
        constraints += [t >= c_pos, t >= c_neg]
    # for d=0, C0=0 automatically satisfied by any t >=0
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.CBC, verbose=False, warm_start=True, max_iters=1000000)
    # Retrieve solution
    x_val = x.value
    t_val = t.value
    # x_val may be floating approx near 0/1
    return x_val, t_val
```

def shape_p(p, n):
    x = np.linspace(0, 2, n)
    t = np.abs(x - 1)  # distance from center
    s = (p + 1) / (2 * p)  # scaling factor
    h = s * (1 - t**p)
    h = np.clip(h, 0, 1)  # ensures within [0,1]
    return h

import numpy as np, time, random, math

def compute_c5(h):
    """C5 value for a discretized h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / h.shape[0]))

def project_onto_sum(v, target):
    """Euclidean projection of v onto [0,1]^n intersect {sum = target}."""
    v = np.clip(v, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) <= 1e-12:
        return v
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    y = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    return y

def shape_p(p, n):
    """Symmetric shape with parameter p (integral = 1 after scaling)."""
    x = np.linspace(0.0, 2.0, n)
    t = np.abs(x - 1.0)
    scale = (p + 1.0) / (2.0 * p)          # makes integral exactly 1
    h = scale * (1.0 - np.power(t, p))
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_sum(h, n / 2.0)
    return h

def random_binary(n):
