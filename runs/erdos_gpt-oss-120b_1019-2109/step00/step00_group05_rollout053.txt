import time, math, numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a binary step function h on [0,2] (sampled uniformly) that
    minimizes C5 = max_k ∫ h(x)(1-h(x+k))dx.
    The function returns the best h found, its C5 value and the number of points.
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------------ #
    # 1. Load / construct an initial binary vector with exactly half 1's.
    # ------------------------------------------------------------------ #
    try:
        # initial_h_values is provided by the harness (if any)
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        # force binary values (0/1) while keeping exactly n/2 ones
        bin_arr = (init_arr > 0.5).astype(int)
        if bin_arr.sum() != n // 2:
            # re‑assign half of the positions to 1 at random
            idx = np.arange(n)
            np.random.shuffle(idx)
            bin_arr[:] = 0
            bin_arr[idx[: n // 2]] = 1
        current_best = bin_arr.copy()
    except Exception:   # no initial_h_values supplied
        n = 70                     # default size (must be even)
        current_best = np.zeros(n, dtype=int)
        current_best[: n // 2] = 1
        np.random.shuffle(current_best)

    # --------------------------------------------------------------- #
    # 2. Helper to compute the maximal mismatch count for a binary h.
    #    C5 = max_mismatches * (2/n)
    # --------------------------------------------------------------- #
    def max_mismatches(h_vec: np.ndarray) -> int:
        """return max_{shift} Σ h[i]·(1−h[i+shift]) (integer count)."""
        # np.correlate with mode='full' gives exactly the needed counts.
        return int(np.correlate(h_vec, 1 - h_vec, mode='full').max())

    # --------------------------------------------------------------- #
    # 3. Target integer bound derived from the floating target.
    # --------------------------------------------------------------- #
    target_c5 = 0.38085
    # allowed mismatches ≤ floor(target * n/2)
    target_int = math.floor(target_c5 * (n / 2) + 1e-12)

    best_h = current_best.copy()
    best_val = max_mismatches(best_h)

    # if the initial candidate already meets the target, finish immediately
    if best_val <= target_int:
        return best_h.tolist(), float(best_val * 2.0 / n), int(n)

    # --------------------------------------------------------------- #
    # 4. Simulated‑annealing style exploration.
    #    We keep a current solution, propose random swaps (preserving the
    #    number of 1's), and accept them with a temperature schedule.
    # --------------------------------------------------------------- #
    # temperature parameters
    T_start = 2.5          # initial temperature
    T_end = 0.001          # final temperature
    # counters for stagnation handling
    no_improve_counter = 0
    RESTART_AFTER = 200000   # steps without improvement → restart from best

    cur_h = best_h.copy()
    cur_val = best_val

    # main loop – run until the global time budget is exhausted
    while time.time() - start_time < budget_s - 0.05:
        # ----------------------------------------------------------- #
        # propose a random 1↔0 swap
        # ----------------------------------------------------------- #
        ones = np.flatnonzero(cur_h == 1)
        zeros = np.flatnonzero(cur_h == 0)

        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        new_h = cur_h.copy()
        new_h[i] = 0
        new_h[j] = 1

        new_val = max_mismatches(new_h)
        delta = new_val - cur_val

        # ----------------------------------------------------------- #
        # temperature based on global elapsed proportion
        # ----------------------------------------------------------- #
        elapsed = time.time() - start_time
        frac = min(elapsed / budget_s, 1.0)
        T = T_start * (1.0 - frac) + T_end

        # decide acceptance
        if delta <= 0 or np.random.rand() < math.exp(-delta / T):
            cur_h = new_h
            cur_val = new_val
            no_improve_counter = 0
            # update global best if improved
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
                # early exit if we already satisfy the required bound
                if best_val <= target_int:
                    break
        else:
            no_improve_counter += 1

        # occasional restart to escape deep local minima
        if no_improve_counter >= RESTART_AFTER:
            cur_h = best_h.copy()
            cur_val = best_val
            no_improve_counter = 0

    # --------------------------------------------------------------- #
    # 5. Return the best solution found.
    # --------------------------------------------------------------- #
    final_c5 = best_val * (2.0 / n)
    return best_h.tolist(), float(final_c5), int(n)
