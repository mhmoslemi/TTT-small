import time
import numpy as np
import random

# ------------------------------------------------------------
# Helper: project a vector onto the box [0,1] and the hyperplane sum = target.
# ------------------------------------------------------------
def _proj_sum_box(x, target):
    """
    Find lambda such that y = clip(x - lambda, 0, 1) satisfies sum(y) = target.
    Binary search on lambda works because the sum is monotone decreasing in lambda.
    """
    lo, hi = -1.0, 1.0                # guarantees sum(lo) = len(x), sum(hi) = 0
    for _ in range(50):               # enough for double precision
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


# ------------------------------------------------------------
# Core optimizer: smoothed max gradient descent with projection.
# ------------------------------------------------------------
def _optimize(start_h, max_iter, base_lr, base_alpha, time_limit, start_time):
    N = len(start_h)
    target = N * 0.5
    dx = 2.0 / N

    h = start_h.copy()
    lr = base_lr
    alpha = base_alpha
    best_local_h = h.copy()
    best_local_val = np.inf

    # pre‑allocate gradient array
    grad = np.empty_like(h)

    for it in range(max_iter):
        # time check
        if time.time() - start_time > time_limit:
            break

        # correlation and current value
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        cur_val = corr.max()
        if cur_val < best_local_val:
            best_local_val = cur_val
            best_local_h = h.copy()

        # smoothed max weights
        m = corr.max()
        w = np.exp(alpha * (corr - m))
        w_sum = w.sum()
        w /= w_sum

        # gradient via vectorised slice updates
        grad.fill(0.0)
        # shifts k = -(N-1) .. (N-1)
        for k in range(-(N - 1), N):
            wk = w[k + N - 1]
            if wk == 0.0:
                continue
            if k >= 0:
                # term1: grad[0:N-k] += wk * (1 - h[k:])
                grad[: N - k] += wk * (1.0 - h[k:])
                # term2: grad[k:]   -= wk * h[:N-k]
                grad[k:] -= wk * h[: N - k]
            else:
                d = -k
                # term1: grad[d:]   += wk * (1 - h[:N-d])
                grad[d:] += wk * (1.0 - h[: N - d])
                # term2: grad[:N-d] -= wk * h[d:]
                grad[: N - d] -= wk * h[d:]

        # gradient step
        h = h - lr * grad
        # projection back onto feasible set
        h = _proj_sum_box(h, target)

        # simple schedules
        lr *= 0.998                # decay learning rate
        if (it + 1) % 250 == 0:    # slowly raise smoothing
            alpha = min(alpha * 1.3, 300.0)

    # final check
    final_corr = np.correlate(h, 1.0 - h, mode="full") * dx
    final_val = final_corr.max()
    if final_val < best_local_val:
        best_local_val = final_val
        best_local_h = h.copy()

    return best_local_h, best_local_val
