import time
import numpy as np

def _c5_value(h):
    """Return the C5 value for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation as defined in the task
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _random_initial(n):
    """Binary + one half entry, exact integral = 1."""
    h = np.zeros(n, dtype=np.float64)
    half = n // 2  # floor(n/2)
    # place `half` ones
    ones = np.random.choice(n, half, replace=False)
    h[ones] = 1.0
    if n % 2:  # odd length -> need a 0.5 to reach exact sum n/2
        remaining = np.setdiff1d(np.arange(n), ones)
        idx = np.random.choice(remaining)
        h[idx] = 0.5
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] with integral 1 that
    minimizes max_k ∫ h(x)(1‑h(x+k))dx.

    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    time_limit = start_time + budget_s * 0.95  # safety margin

    # Determine number of points and starting vector
    try:
        # `initial_h_values` is provided by the harness when available
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    except Exception:
        n = 45
        h = _random_initial(n)

    target_sum = n / 2.0          # ∑h_i must equal n/2
    # Tiny correction to satisfy the integral exactly (floating error)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # add the tiny difference to a non‑boundary entry
        cand = np.where((h > 0) & (h < 1))[0]
        if cand.size == 0:
            cand = np.arange(n)
        i = np.random.choice(cand)
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # If the starting construction already beats the target we can return it
    target = 0.3808
    if best_c5 <= target:

    # Parameters for the stochastic search
    T0 = 5e-3            # initial temperature for simulated annealing
    it = 0

    while time.time() < time_limit:

        # temperature decays slowly
        T = T0 * (0.9995 ** it)

        # compute the overlap vector once per iteration
        ov = np.correlate(h, 1.0 - h, mode='full')

        if np.random.rand() < 0.15:

            h_new[j] -= delta

            if shift >= 0:

                delta_max = min(h[j], 1.0 - h[i])
                delta = delta_max * np.random.rand()
                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta
            else:
                # fallback random feasible pair
                i, j = np.random.randint(n, size=2)
                lo = max(-h[i], h[j] - 1.0)
                hi = min(1.0 - h[i], h[j])
                if lo >= hi:
                    continue
                delta = lo + np.random.rand() * (hi - lo)
                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta

        # ----- evaluate -----
        new_c5 = _c5_value(h_new)

        # ----- acceptance -----
        accept = False
        if new_c5 < best_c5 - 1e-12:
            # any improvement over the global best is always accepted
            best_h = h_new.copy()
            best_c5 = new_c5
            accept = True
        if new_c5 < _c5_value(h) - 1e-12:
            accept = True
        elif T > 0.0:
            # Metropolis criterion for uphill moves
            prob = np.exp(-(new_c5 - _c5_value(h)) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            h = h_new

        # early exit if we have met the target
        if best_c5 <= target:
            break

    # Final small correction to guarantee the integral exactly equals 1
    diff_final = target_sum - best_h.sum()
    if abs(diff_final) > 1e-12:
        cand = np.where((best_h > 0) & (best_h < 1))[0]
        if cand.size == 0:
            cand = np.arange(n)
        i = np.random.choice(cand)
        best_h[i] = np.clip(best_h[i] + diff_final, 0.0, 1.0)

    # recompute final bound (ensuring no drift)
    final_c5 = _c5_value(best_h)

    return best_h.tolist(), float(final_c5), n
