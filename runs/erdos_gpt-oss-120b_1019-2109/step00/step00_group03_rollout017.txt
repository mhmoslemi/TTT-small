import numpy as np, time, math, random

def evaluate_c5(h):
    n = len(h)
    S = np.correlate(h, h, mode='full')
    min_S = S.min()
    return 1 - (2.0 / n) * min_S

def overlap_mask(n, shift):
    d = abs(shift)
    mask = np.zeros(n, dtype=bool)
    if d >= n:
        return mask
    mask[:n-d] = True
    mask[d:] = True
    return mask

def random_move(h):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # ensure can transfer from i to j
    delta_max = min(h[i], 1 - h[j])
    if delta_max <= 0:
        # try opposite direction
        delta_max = min(h[j], 1 - h[i])
        if delta_max <= 0:
            return None, None
        # swap i,j
        i, j = j, i
    delta = np.random.rand() * delta_max
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h, delta

def targeted_move(h, shift):
    n = len(h)
    mask = overlap_mask(n, shift)
    src_indices = np.where(~mask & (h > 1e-12))[0]  # can give mass
    tgt_indices = np.where(mask & (h < 1 - 1e-12))[0]
    if src_indices.size == 0 or tgt_indices.size == 0:
        return None
    i = np.random.choice(src_indices)
    j = np.random.choice(tgt_indices)
    delta_max = min(h[i], 1 - h[j])
    if delta_max <= 0:
        return None
    delta = np.random.rand() * delta_max * 0.5  # smaller step
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

import time, math, random
import numpy as np

def _evaluate_c5(h):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    # correlation of h with itself gives S_k = sum_i h_i h_{i+k}
    S = np.correlate(h, h, mode='full')
    min_S = S.min()
    return 1.0 - (2.0 / n) * min_S

def _gradient_shift(h, shift):
    """Gradient of S_shift = Σ h_i h_{i+shift}  w.r.t. h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        d = shift
        # contribution from h[i] * h[i+d]
        g[:n - d] += h[d:]
        # contribution from h[i-d] * h[i]
        g[d:] += h[:n - d]
    else:
        d = -shift
        # shift negative: same pattern with d
        g[d:] += h[:n - d]
        g[:n - d] += h[d:]
    return g

def _targeted_move(h, shift):
    """Move a little mass from a low‑gradient index to a high‑gradient one."""
    g = _gradient_shift(h, shift)
    src_mask = (h > 1e-12)
    tgt_mask = (h < 1.0 - 1e-12)
    if not src_mask.any() or not tgt_mask.any():
        return None
    src_candidates = np.where(src_mask)[0]
    tgt_candidates = np.where(tgt_mask)[0]
    src = src_candidates[np.argmin(g[src_candidates])]
    tgt = tgt_candidates[np.argmax(g[tgt_candidates])]
    delta_max = min(h[src], 1.0 - h[tgt])
    if delta_max <= 0:
        return None
    delta = np.random.rand() * delta_max * 0.5  # modest step
    nh = h.copy()
    nh[src] -= delta
    nh[tgt] += delta
    return nh

def _random_move(h):
    """Transfer a random amount of mass between two different cells."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    # direction i→j
    dm = min(h[i], 1.0 - h[j])
    if dm <= 0:
        # opposite direction
        dm = min(h[j], 1.0 - h[i])
        if dm <= 0:
            return None
        i, j = j, i
    delta = np.random.rand() * dm
    nh = h.copy()
    nh[i] -= delta
    nh[j] += delta
    return nh
