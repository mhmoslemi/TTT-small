import numpy as np
import random
import time
import math

def _c5_value(h: np.ndarray) -> float:
    """Return C5 for a step function h discretized on [0,2]."""
    n = h.shape[0]
    # product correlation (h with itself)
    prod = np.correlate(h, h, mode='full')
    min_prod = prod.min()
    return 1.0 - (2.0 / n) * min_prod


def _binary_from_cont(arr: np.ndarray) -> np.ndarray:
    """Convert a possibly fractional array to a binary (0/1) vector
    respecting the integral constraint exactly."""
    n = arr.shape[0]
    target = int(round(n / 2))
    bin_arr = (arr > 0.5).astype(np.int8)
    cur = int(bin_arr.sum())
    if cur == target:
        return bin_arr
    diff = target - cur
    if diff > 0:
        zeros = np.where(bin_arr == 0)[0]
        if len(zeros) >= diff:
            bin_arr[np.random.choice(zeros, diff, replace=False)] = 1
        else:
            bin_arr[zeros] = 1
            extra = diff - len(zeros)
            ones = np.where(bin_arr == 1)[0]
            if extra > 0:
                bin_arr[np.random.choice(ones, extra, replace=False)] = 0
    else:
        diff = -diff
        ones = np.where(bin_arr == 1)[0]
        if len(ones) >= diff:
            bin_arr[np.random.choice(ones, diff, replace=False)] = 0
        else:
            bin_arr[ones] = 0
            extra = diff - len(ones)
            zeros = np.where(bin_arr == 0)[0]
            if extra > 0:
                bin_arr[np.random.choice(zeros, extra, replace=False)] = 1
    return bin_arr


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that minimizes
    C5 = max_k ∫ h(x)(1‑h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed + 987654)

    # ------------------------------------------------------------------
    # 1. Load initial construction if available.
    # ------------------------------------------------------------------
    try:
        init_arr = np.asarray(initial_h_values, dtype=float)  # global variable
        n_points = init_arr.shape[0]
        init_arr = np.clip(init_arr, 0.0, 1.0)
        init_c5 = _c5_value(init_arr)
        # If the existing construction already meets the target, use it.
        if init_c5 <= 0.38085:
            return (init_arr.astype(float), float(init_c5), int(n_points))
    except NameError:
        # No pre‑supplied construction – fall back to a random binary start.
        n_points = 70
        init_arr = None
        init_c5 = None

    # ------------------------------------------------------------------
    # 2. Prepare a binary starting point (weight = n/2) for the search.
    # ------------------------------------------------------------------
    target_ones = int(round(n_points / 2))
    if init_arr is not None:
        cur_h = _binary_from_cont(init_arr)
    else:
        cur_h = np.zeros(n_points, dtype=np.int8)
        cur_h[np.random.choice(n_points, target_ones, replace=False)] = 1

    cur_c5 = _c5_value(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # Also keep the original (possibly fractional) construction as a candidate.
    if init_arr is not None and init_c5 is not None and init_c5 < best_c5:
        best_h = init_arr.astype(float)  # keep as float for final output
        best_c5 = init_c5

    # ------------------------------------------------------------------
    # 3. Simulated annealing on binary vectors.
    # ------------------------------------------------------------------
    start_time = time.time()
    end_time = start_time + budget_s * 0.95  # slight safety margin
    # SA schedule parameters
    T_start = 0.08
    T_end = 1e-5

    # Helper to perform a single swap (maintains weight)
    def _propose_swap(vec):
        ones_idx = np.where(vec == 1)[0]
        zeros_idx = np.where(vec == 0)[0]
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        vec[i], vec[j] = vec[j], vec[i]
        return i, j

    # Main loop
    while time.time() < end_time:

        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.95)
        T = T_start * (1 - frac) + T_end * frac

        # propose a neighbour by swapping a 1 and a 0
        i, j = _propose_swap(cur_h)
        new_c5 = _c5_value(cur_h)
        delta = new_c5 - cur_c5

                if best_c5 <= 0.3808:  # early stop if target reached

        else:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # Ensure the returned array respects the constraints (clip, correct sum)

        final_h *= scale
        final_h = np.clip(final_h, 0.0, 1.0)

    final_c5 = _c5_value(final_h)
