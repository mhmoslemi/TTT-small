def evaluate_h_params(lengths, heights, n=200):
    # lengths: array of size 5 (float, sum to 2)
    # heights: array size 5 (0 <= hi <= 1)
    x = np.linspace(0, 2, n, endpoint=False)
    cum_lengths = np.cumsum(lengths)
    # Determine interval for each x
    # Use np.searchsorted
    idx = np.searchsorted(cum_lengths, x, side='right')
    h = heights[idx]
    # Compute C5 via correlation
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    # maximum
    return c.max()

    start = time.time()
    np.random.seed(seed)
    N = 200
    dx = 2.0 / N

    bestC5 = np.inf
    best_L = None
    best_a = None

    # Define function to compute C5 for given L,a using N points (fast)
    def compute_c5(L,a, n_points):
        # compute lengths
        Z = 0.5 - L + a * L
        M = 1 - 2 * a * L
        # Ensure nonnegative lengths
        if Z < 0 or M < 0:
            return np.inf
        lengths = np.array([Z, L, M, L, Z])
        heights = np.array([0.0, a, 1.0, a, 0.0])

        cum_lengths = np.cumsum(lengths)
        x = (np.arange(n_points) + 0.5) * (2.0 / n_points)

def run(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    start = time.time()
    np.random.seed(seed)

    # Setup sample sizes
    N_COARSE = 120
    N_FINE = 300  # final; maybe 300
    # But we also may use N_FINE for local improvement evaluation, not final.

    dx_coarse = 2.0 / N_COARSE
    xs_coarse = (np.arange(N_COARSE) + 0.5) * dx_coarse

    # compute baseline if initial_h_values defined
    best_h = None
    best_C5 = np.inf

    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n0 = h0.shape[0]
        dx0 = 2.0 / n0
        C0 = np.max(np.correlate(h0, 1 - h0, mode='full') * dx0)
        best_h = h0
        best_C5 = C0

    # define evaluation function
    def compute_c5_sym(L,a,n,x):
        Z = 0.5 - L + a * L
        M = 1 - 2 * a * L
        if Z < 0 or M < 0:
            return np.inf
        lengths = np.array([Z, L, M, L, Z])
        heights = np.array([0.0, a, 1.0, a, 0.0])
        cum = np.cumsum(lengths)
        idx = np.searchsorted(cum, x, side='right')
        h = heights[idx]
        cfull = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
        return np.max(cfull)

    # Coarse grid search
    num_L = 200
    num_a = 200
    L_vals = np.linspace(0.001, 0.5, num=num_L)

            if C < best_C5:

                best_L = L
                best_a = a

import time, numpy as np

def evaluate_step_function(lengths, heights, n):
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    cum = np.cumsum(lengths)
    idx = np.searchsorted(cum, x, side='right')
    h = heights[idx]
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr), h

def compute_c5_sym(L, a, n):
    Z = 0.5 - L + a * L
    M = 1 - 2 * a * L
    if Z < 0 or M < 0:
        return np.inf
    lengths = np.array([Z, L, M, L, Z])
    heights = np.array([0.0, a, 1.0, a, 0.0])
    C, _ = evaluate_step_function(lengths, heights, n)
    return C

def project_onto_sum_box(v, target):
    lo, hi = -1.0, 1.0
    for _ in range(60):

        p = np.clip(v + mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np


def evaluate_step_function(lengths, heights, n):
    """Return max overlap integral and the discretized step function."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    cum = np.cumsum(lengths)
    idx = np.searchsorted(cum, x, side='right')
    h = heights[idx]
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr), h


def compute_c5_sym(L, a, n):
    """C5 for the symmetric 5‑step pattern 0‑a‑1‑a‑0."""
    Z = 0.5 - L + a * L
    M = 1.0 - 2.0 * a * L
    if Z < 0.0 or M < 0.0:
        return np.inf
    lengths = np.array([Z, L, M, L, Z])
    heights = np.array([0.0, a, 1.0, a, 0.0])
    C, _ = evaluate_step_function(lengths, heights, n)
    return C


def project_onto_sum_box(v, target):
    """Shift all entries by λ and clip to [0,1] so that sum equals target."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(v + mid, 0.0, 1.0)
        s = p.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
