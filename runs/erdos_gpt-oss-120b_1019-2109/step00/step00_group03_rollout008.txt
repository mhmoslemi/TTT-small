import numpy as np
import time, math, random

def _max_overlap_count(h: np.ndarray) -> int:
    """Return max_{k} sum_i h[i] * (1-h[i+k]) as an integer."""
    n = h.shape[0]
    g = 1 - h
    size = 2 * n - 1
    m = 1 << (size - 1).bit_length()          # next power of two
    H = np.fft.rfft(h.astype(np.float64), n=m)
    G = np.fft.rfft(g.astype(np.float64), n=m)
    conv = np.fft.irfft(H * G, n=m)[:size]
    # rounding eliminates tiny floating errors
    return int(np.rint(conv).max())
