def solve_milp(N, time_limit=1000):
    # create variables
    x = cp.Variable(N, boolean=True)
    total_ones = N // 2
    constraints = [cp.sum(x) == total_ones]
    # pairs
    pair_i = []
    pair_j = []
    shift_pair_idx = [[] for _ in range(N-1)]
    for shift in range(1, N):
        for i in range(N - shift):
            idx = len(pair_i)
            pair_i.append(i)
            pair_j.append(i + shift)
            shift_pair_idx[shift-1].append(idx)

    M = len(pair_i)
    y = cp.Variable(M, boolean=True)
    # product constraints
    for p in range(M):
        constraints.append(y[p] <= x[pair_i[p]])
        constraints.append(y[p] <= x[pair_j[p]])
        constraints.append(y[p] >= x[pair_i[p]] + x[pair_j[p]] - 1)

    # max mismatch variable
    t = cp.Variable(integer=True, nonneg=True)

    # constraints per shift
    for s in range(N-1):
        shift = s + 1
        idxs = shift_pair_idx[s]  # list of y indices for this shift
        if idxs:
            sum_y = cp.sum(cp.hstack([y[p] for p in idxs]))
        else:
            sum_y = 0
        # left side sum_x_i from i=0 to N-shift-1
        sum_x_left = cp.sum(x[:N-shift])
        sum_x_right = cp.sum(x[shift:])
        mismatch_pos = sum_x_left - sum_y
        mismatch_neg = sum_x_right - sum_y
        constraints.append(mismatch_pos <= t)
        constraints.append(mismatch_neg <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.CBC, timeLimit=int(time_limit), verbose=False)

import time
import math
import numpy as np
import cvxpy as cp

def evaluate_vector(h):
    """Helper that uses the pre‑imported evaluation routine."""
    # The harness provides `evaluate_erdos_solution`
    return evaluate_erdos_solution(h)

def solve_binary_milp(N, time_limit):
    """Solve the binary formulation via MILP.
    Returns a binary numpy array of length N or None on failure."""
    # variables
    x = cp.Variable(N, boolean=True)
    total = N // 2
    constraints = [cp.sum(x) == total]

    # build pair list (i, j=i+shift) for all positive shifts
    pair_i = []
    pair_j = []
    shift_pairs = [[] for _ in range(N - 1)]          # shift_pairs[s] holds indices of pairs with shift s+1
    for shift in range(1, N):
        for i in range(N - shift):
            idx = len(pair_i)
            pair_i.append(i)
            pair_j.append(i + shift)
            shift_pairs[shift - 1].append(idx)

    M = len(pair_i)
    y = cp.Variable(M, boolean=True)

    # product linearisation constraints
    for p in range(M):
        constraints.append(y[p] <= x[pair_i[p]])
        constraints.append(y[p] <= x[pair_j[p]])
        constraints.append(y[p] >= x[pair_i[p]] + x[pair_j[p]] - 1)

    # max‑mismatch variable
    t = cp.Variable(integer=True, nonneg=True)

    # constraints linking mismatches to t
    for s in range(N - 1):
        shift = s + 1
        idxs = shift_pairs[s]
        sum_y = cp.sum(y[idxs]) if idxs else 0
        sum_x_left = cp.sum(x[: N - shift])
        sum_x_right = cp.sum(x[shift:])
        mismatch_pos = sum_x_left - sum_y
        mismatch_neg = sum_x_right - sum_y
        constraints.append(mismatch_pos <= t)
        constraints.append(mismatch_neg <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # try CBC first, fall back to GLPK_MI
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, tm_lim=int(time_limit * 1000), verbose=False)
        except Exception:
            return None

    if prob.status not in ("optimal", "optimal_inaccurate", "optimal"):
        # infeasible or timeout
        return None
    if x.value is None:
        return None
    # Round to 0/1 (solver should already be integral)
    x_bin = np.rint(np.clip(x.value, 0, 1)).astype(float)
    return x_bin


def hill_climb_binary(start_h, time_budget, seed):
    """Simple stochastic hill‑climbing on binary vectors preserving the half‑sum."""
    rng = np.random.default_rng(seed)
    h = start_h.copy()
    best = h.copy()
    best_score = evaluate_vector(best)
    N = len(h)

    while time.time() < time_budget:
        i, j = rng.integers(0, N, size=2)
        if i == j or h[i] == h[j]:
            continue
        # swap a 1 with a 0
        cand = h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_score = evaluate_vector(cand)
        if cand_score < best_score:
            best, best_score = cand, cand_score
            h = cand
        else:
            # occasionally accept worse moves (simulated annealing style)
            # temperature decays with time
            elapsed = time.time() - start_time
            fraction = elapsed / time_budget
            temp = max(0.01, 0.5 * (1 - fraction))
            if rng.random() < math.exp(-(cand_score - best_score) / temp):
                h = cand
    return best, best_score
