def propose_move(h, step_factor):
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    max_plus = min(h[i], 1 - h[j])
    max_minus = min(h[j], 1 - h[i])
    if max_plus + max_minus == 0:
        return None
    # sample raw epsilon uniformly in [-max_minus, max_plus]
    epsilon_raw = (np.random.rand() * (max_plus + max_minus)) - max_minus
    epsilon = epsilon_raw * step_factor
    new_h = h.copy()
    new_h[i] -= epsilon
    new_h[j] += epsilon
    # ensure bounds (clamp)
    new_h[i] = min(max(new_h[i], 0.0), 1.0)
    new_h[j] = min(max(new_h[j], 0.0), 1.0)
    return new_h

import numpy as np, time, random, math

def compute_c5(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def make_feasible_from_initial(initial):
    v = np.asarray(initial, dtype=float)
    n = v.shape[0]
    target = n / 2.0
    idx = np.argsort(-v)  # descending
    h = np.zeros(n, dtype=float)
    acc = 0.0
    for i in idx:
        if acc + 1.0 <= target + 1e-12:
            h[i] = 1.0
            acc += 1.0
        else:
            remainder = target - acc
            if remainder > 0:
                h[i] = remainder
                acc += remainder
            break
    return h

def upsample(h, k):
    return np.repeat(h, k)

def random_initial(n):
    target = n / 2.0
    h = np.zeros(n, dtype=float)
    # choose indices for ones
    indices = np.arange(n)
    np.random.shuffle(indices)
    acc = 0.0
    for i in indices:
        if acc + 1.0 <= target:
            h[i] = 1.0
            acc += 1.0
        else:
            remainder = target - acc
            if remainder > 1e-12:
                h[i] = remainder
                acc += remainder
            break
    return h

def propose_move(h, step_factor):
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    max_plus = min(hi, 1.0 - hj)   # amount we can move from i to j (positive epsilon)
    max_minus = min(hj, 1.0 - hi)  # amount we can move from j to i (negative epsilon)
    if max_plus + max_minus <= 0.0:
        return None
    # raw epsilon uniform in [-max_minus, max_plus]
    eps_raw = (np.random.rand() * (max_plus + max_minus)) - max_minus
    eps = eps_raw * step_factor
    new_h = h.copy()
    new_h[i] = hi - eps
    new_h[j] = hj + eps
    # enforce bounds due to numeric errors
    new_h[i] = 0.0 if new_h[i] < 0 else (1.0 if new_h[i] > 1 else new_h[i])
    new_h[j] = 0.0 if new_h[j] < 0 else (1.0 if new_h[j] > 1 else new_h[j])
    return new_h

def anneal(start_h, max_seconds):
    start = time.time()
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    # annealing schedule parameters
    max_step = 1.0
    min_step = 0.01
    Tmax = 0.05
    Tmin = 1e-4
    while True:
        elapsed = time.time() - start
        if elapsed > max_seconds:
            break
        prog = elapsed / max_seconds
        step_factor = max_step * (1 - prog) + min_step * prog
        T = Tmax * (1 - prog) + Tmin * prog
        new_h = propose_move(cur_h, step_factor)
        if new_h is None:
            continue
        new_val = compute_c5(new_h)
        delta = new_val - cur_val
        if delta < 0 or random.random() < math.exp(-delta / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val

def local_refine(h):
    # attempt greedy full transfers that improve C5
    n = h.shape[0]
    improved = True
    best_val = compute_c5(h)
    while improved:
        improved = False
        # try a limited number of random pairs to avoid too much work
        for _ in range(n * 5):
            i, j = np.random.choice(n, size=2, replace=False)
            if h[i] <= 0 or h[j] >= 1:

            transfer = min(h[i], 1.0 - h[j])
            if transfer <= 0:

            new_h = h.copy()
            new_h[i] -= transfer
            new_h[j] += transfer
            new_val = compute_c5(new_h)
            if new_val + 1e-12 < best_val:
                h = new_h
                best_val = new_val
                improved = True
                break
    return h, best_val

import numpy as np
import time
import random
import math

# ---------- Core utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant for a step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))


def upsample(vec: np.ndarray, k: int) -> np.ndarray:
    """Repeat each entry k times (preserves the integral constraint)."""
    return np.repeat(vec, k)


def propose_move(h: np.ndarray, step_factor: float) -> np.ndarray | None:
    """
    Transfer a mass epsilon from index i to j while keeping sum and bounds.
    step_factor (0,1] scales the raw feasible epsilon.
    """
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]

    # amount we could move from i to j (positive) and j to i (negative)
    max_plus = min(hi, 1.0 - hj)      # i -> j
    max_minus = min(hj, 1.0 - hi)     # j -> i
    if max_plus + max_minus <= 0.0:
        return None

    # raw epsilon uniformly in the feasible interval
    eps_raw = (np.random.rand() * (max_plus + max_minus)) - max_minus
    eps = eps_raw * step_factor

    new_h = h.copy()
    new_h[i] = hi - eps
    new_h[j] = hj + eps
    # clamp for numerical safety
    new_h[i] = 0.0 if new_h[i] < 0 else (1.0 if new_h[i] > 1 else new_h[i])
    new_h[j] = 0.0 if new_h[j] < 0 else (1.0 if new_h[j] > 1 else new_h[j])
    return new_h


def anneal(start_h: np.ndarray, max_seconds: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing optimizer that respects the constraints.
    Returns the best vector found during the allotted time.
    """
    start_time = time.time()
    cur_h = start_h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # schedule parameters
    max_step, min_step = 1.0, 0.01
    Tmax, Tmin = 0.05, 1e-4

    while True:
        elapsed = time.time() - start_time
        if elapsed > max_seconds:
            break
        prog = elapsed / max_seconds
        step_factor = max_step * (1 - prog) + min_step * prog
        T = Tmax * (1 - prog) + Tmin * prog

        cand_h = propose_move(cur_h, step_factor)
        if cand_h is None:
            continue
        cand_val = compute_c5(cand_h)
        delta = cand_val - cur_val
        if delta < 0 or random.random() < math.exp(-delta / T):
            cur_h, cur_val = cand_h, cand_val
            if cand_val < best_val:
                best_h, best_val = cand_h.copy(), cand_val
    return best_h, best_val


def local_refine(h: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Greedy refinement: repeatedly transfer the full possible mass
    between a pair of indices if it improves the objective.
    """
    n = h.shape[0]
    best_val = compute_c5(h)
    improved = True

    while improved:
        improved = False
        for _ in range(n * 5):
            i, j = np.random.choice(n, size=2, replace=False)
            if h[i] <= 0.0 or h[j] >= 1.0:
                continue
            transfer = min(h[i], 1.0 - h[j])
            if transfer <= 0.0:
                continue
            new_h = h.copy()
            new_h[i] -= transfer
            new_h[j] += transfer
            new_val = compute_c5(new_h)
            if new_val + 1e-12 < best_val:
                h, best_val = new_h, new_val
                improved = True
                break
    return h, best_val


# ---------- Main entry point ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) for the Erdős C5 problem.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Obtain an initial vector. Use the provided one when available.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init_vec = np.asarray(initial_h_values, dtype=float)
    else:
        # fallback: 100 points, exactly half ones
        init_vec = np.concatenate([np.ones(50), np.zeros(50)], dtype=float)
        np.random.shuffle(init_vec)

    # ------------------------------------------------------------------
    # 2. Upsample to gain more degrees of freedom (still respects the
    #    integral constraint because repetition multiplies both length
    #    and total mass by the same factor).
    # ------------------------------------------------------------------
    up_factor = 8
    if init_vec.size * up_factor > 1000:
        up_factor = max(1, 1000 // init_vec.size)

    h0 = upsample(init_vec, up_factor)
    target_sum = h0.size / 2.0
    # Small numerical adjustment if needed
    diff = target_sum - h0.sum()
    if abs(diff) > 1e-12:
        for i in range(h0.size):
            if 0.0 < h0[i] < 1.0:
                h0[i] += diff
                break
        h0 = np.clip(h0, 0.0, 1.0)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    start_time = time.time()

    # ------------------------------------------------------------------
    # 3. Repeated annealing runs until budget expires.
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed
        if remaining <= 0.5:
            break

        # allocate a chunk (max 30 s) for a single annealing run
        chunk = min(30.0, remaining * 0.8)

        # start point: either a perturbed version of the current best
        # or a fresh random feasible vector (20 % chance)
        if random.random() < 0.8:
            cur = best_h.copy()
            for _ in range(10):
                tmp = propose_move(cur, step_factor=0.5)
                if tmp is not None:
                    cur = tmp
        else:
            cur = np.zeros(h0.size, dtype=float)
            # random binary (or fractional) feasible vector
            indices = np.arange(cur.size)
            np.random.shuffle(indices)
            acc = 0.0
            for idx in indices:
                if acc + 1.0 <= target_sum:
                    cur[idx] = 1.0
                    acc += 1.0
                else:
                    remainder = target_sum - acc
                    if remainder > 0:
                        cur[idx] = remainder
                    break

        # annealing on this start point
        local_h, local_val = anneal(cur, chunk)

        # deterministic greedy polishing
        local_h, local_val = local_refine(local_h)

        if local_val < best_val:
            best_h, best_val = local_h, local_val
            # early exit if we already beat the target bound
            if best_val < 0.3808:
                break

    final_val = compute_c5(best_h)
    n_points = int(best_h.shape[0])
    return best_h, final_val, n_points
