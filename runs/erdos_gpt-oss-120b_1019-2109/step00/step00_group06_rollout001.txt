def project_to_box_and_sum(v, target):
    # Ensure 0 <= v <= 1
    v = np.clip(v, 0.0, 1.0)
    # Use binary search for λ such that sum(clamp(v - λ, 0, 1)) = target
    lo = np.min(v) - 1.0  # ensure at high λ sum = 0
    hi = np.max(v)  # ensure at low λ sum = n (since v - lo large positive -> clamp to 1)
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # final fine adjustment to hit target exactly (within tolerance)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        # find indices not at bounds
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size > 0:
            # distribute diff evenly
            inc = diff / free.size
            x[free] += inc
            # clip again
            x = np.clip(x, 0.0, 1.0)
        # if still off, rescale directly maybe:
        # iterative approach to correct small errors
        # Could do another pass with binary search again.
    return x

def propose_move(h, max_step):
    n = h.shape[0]
    i = np.random.randint(0, n)
    j = np.random.randint(0, n-1)
    if j >= i:
        j += 1   # ensure j != i

    # compute feasible interval for delta: delta added to j and sub from i
    # delta >= -h_j (so h_j + delta >=0)
    # delta <= 1 - h_j (so h_j+delta <=1)
    # delta <= h_i (so h_i - delta >=0)

    low = max(-h[j], -max_step, h[i] - 1.0)
    high = min(h[i], max_step, 1.0 - h[j])

        return None
    delta = low + np.random.rand() * (high - low)
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta

import numpy as np, time, math

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] with sum(v)=target (Euclidean projection)."""
    v = np.clip(v, 0.0, 1.0)
    lo = np.min(v) - 1.0          # low λ gives all 1's
    hi = np.max(v)                # high λ gives all 0's
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1.0 - 1e-12))[0]
        if free.size:
            inc = diff / free.size
            x[free] += inc
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_value(h: np.ndarray) -> float:
    """Compute the C5 value for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _propose_move(h: np.ndarray, max_step: float) -> np.ndarray | None:
    """Generate a feasible transfer move (preserving the sum)."""
    n = h.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    # feasible interval for delta (transfer from i to j)
    lo = max(-h[j], -max_step, h[i] - 1.0)
    hi = min(h[i], max_step, 1.0 - h[j])
    if lo >= hi:
        return None
    delta = lo + np.random.rand() * (hi - lo)
    nh = h.copy()
    nh[i] -= delta
    nh[j] += delta
    return nh

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1. initialise a feasible vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
        target_sum = n / 2.0
        h = _project_box_sum(np.clip(h0, 0.0, 1.0), target_sum)
    else:
        n = 200                      # even size makes a binary solution possible
        target_sum = n / 2.0
        h = np.random.rand(n)
        h = _project_box_sum(np.clip(h, 0.0, 1.0), target_sum)

    # ------------------------------------------------------------------
    # 2. simulated‑annealing / stochastic descent
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c = _c5_value(h)

    cur_c = best_c
    max_iter = 500_000
    T0 = 0.08                     # initial temperature
    Tmin = 1e-6
    max_step0 = 0.5               # maximum mass transfer at start

    for it in range(max_iter):
        # time guard
        if time.time() - start > budget_s * 0.95:
            break

        # linear cooling
        T = max(T0 * (1.0 - it / max_iter), Tmin)
        step_scale = max_step0 * (1.0 - it / max_iter)

        nh = _propose_move(h, step_scale)
        if nh is None:
            continue
        new_c = _c5_value(nh)

        if new_c < cur_c or np.random.rand() < math.exp(-(new_c - cur_c) / T):
            h = nh
            cur_c = new_c

        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
            if best_c <= 0.3808:
                break

    # ------------------------------------------------------------------
    # 3. final evaluation and return
    # ------------------------------------------------------------------
    final_c = _c5_value(best_h)
    return best_h, float(final_c), int(n)
