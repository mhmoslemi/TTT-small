def shift(arr, k):
    n = len(arr)
    if k >= 0:
        out = np.empty(n)
        out[:k] = 0.0
        out[k:] = arr[:n-k]
    else:
        kp = -k
        out = np.empty(n)
        out[:n-kp] = arr[kp:]
        out[n-kp:] = 0.0
    return out

def shift(arr, k):
    n = arr.shape[0]
    out = np.empty_like(arr)
    if k > 0:
        out[:k] = 0.0
        out[k:] = arr[:n - k]
    elif k < 0:
        kp = -k
        out[:n - kp] = arr[kp:]
        out[n - kp:] = 0.0
    else:
        out[:] = arr
    return out

def gradient_move(h, k):
    n = len(h)
    h_plus = shift(h, -k)  # h_{i+k}
    h_minus = shift(h, k)  # h_{i-k}
    scores = h_plus + h_minus
    inc_idx = int(np.argmax(scores))
    dec_idx = int(np.argmin(scores))
    # ensure we have room
    max_inc = 1.0 - h[inc_idx]
    max_dec = h[dec_idx]
    delta = min(max_inc, max_dec)
    if delta <= 1e-12:
        return None
    # use fraction of delta
    delta = delta * (0.1 + 0.9 * random.random())
    new_h = h.copy()
    new_h[inc_idx] += delta
    new_h[dec_idx] -= delta
    return new_h
```

def random_move(h):
    n = len(h)
    # pick i and j distinct
    i, j = np.random.choice(n, 2, replace=False)
    # decide direction
    max_ij = min(h[i], 1.0 - h[j])
    max_ji = min(h[j], 1.0 - h[i])
    if max_ij + max_ji == 0:
        return None
    if random.random() < (max_ij / (max_ij + max_ji)):
        max_delta = max_ij
        delta = max_delta * random.random()
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        max_delta = max_ji
        delta = max_delta * random.random()
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
    return new_h
```

def random_binary_h(n, rng):
    target = n / 2.0
    floor = int(np.floor(target))
    frac = target - floor
    h = np.zeros(n)
    indices = rng.choice(n, floor, replace=False)
    h[indices] = 1.0
    if frac > 1e-12:
        remaining = list(set(range(n)) - set(indices))
        idx = rng.choice(remaining)
        h[idx] = frac
    return h

import numpy as np
import time
import random
import math

# ---------- helper functions ----------

def shift(arr, k):
    """Shift array by k positions with zero padding (right shift for k>0)."""
    n = arr.shape[0]
    if k > 0:
        out = np.empty_like(arr)
        out[:k] = 0.0
        out[k:] = arr[: n - k]
    elif k < 0:
        kp = -k
        out = np.empty_like(arr)
        out[: n - kp] = arr[kp:]
        out[n - kp :] = 0.0
    else:
        out = arr.copy()
    return out


def min_corr_info(h):
    """Return (min_corr, k) where min_corr = min_i sum_j h_j h_{j+i},
    k is the shift (index) giving that minimum."""
    corr = np.correlate(h, h, mode="full")
    min_idx = int(np.argmin(corr))
    min_val = corr[min_idx]
    k = min_idx - (len(h) - 1)
    return min_val, k, corr


def compute_c5(h, dx):
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx = 1 - min_corr*dx."""
    min_val, _, _ = min_corr_info(h)
    return 1.0 - min_val * dx


def random_binary_h(n, rng):
    """Binary (0/1) vector of length n with exact sum n/2 (fractional last entry if odd)."""
    target = n / 2.0
    whole = int(math.floor(target))
    frac = target - whole
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, whole, replace=False)
    h[idx] = 1.0
    if frac > 1e-12:
        remaining = list(set(range(n)) - set(idx))
        i = rng.choice(remaining)
        h[i] = frac
    return h


def thue_morse_h(n):
    """Thue–Morse binary word of length n (approximate)."""
    bits = np.fromiter(((bin(i).count("1") & 1) for i in range(n)), dtype=int)
    h = bits.astype(float)
    # adjust to exact sum n/2
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # add a fractional entry
        if diff > 0:
            zero_idx = np.where(h < 0.5)[0][0]
            h[zero_idx] = diff
        else:
            one_idx = np.where(h > 0.5)[0][0]
            h[one_idx] = 1.0 + diff
    return h


def gradient_move(h, k):
    """Move a small amount of mass from a low‑score index to a high‑score index for the worst shift k."""
    n = len(h)
    # shifted partners for the worst shift
    h_plus = shift(h, -k)   # h_{i+k}
    h_minus = shift(h, k)   # h_{i‑k}
    scores = h_plus + h_minus
    inc = int(np.argmax(scores))
    dec = int(np.argmin(scores))
    # ensure a beneficial direction
    if scores[inc] <= scores[dec] + 1e-12:
        return None
    max_inc = 1.0 - h[inc]
    max_dec = h[dec]
    delta = min(max_inc, max_dec, 0.5) * random.random()
    if delta < 1e-12:
        return None
    nh = h.copy()
    nh[inc] += delta
    nh[dec] -= delta
    return nh


def random_move(h):
    """Transfer a random amount of mass between two positions while keeping the sum fixed."""
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # direction i→j
    max_ij = min(h[i], 1.0 - h[j])
    max_ji = min(h[j], 1.0 - h[i])
    if max_ij + max_ji == 0.0:
        return None
    if random.random() < (max_ij / (max_ij + max_ji)):
        delta = max_ij * random.random()
        nh = h.copy()
        nh[i] -= delta
        nh[j] += delta
    else:
        delta = max_ji * random.random()
        nh = h.copy()
        nh[i] += delta
        nh[j] -= delta
    return nh

    phi = (1.0 + math.sqrt(5.0)) / 2.0          # ≈1.61803398875
