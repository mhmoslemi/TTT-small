import time
import random
import numpy as np
import math

def _max_corr_unscaled(h: np.ndarray) -> int:
    """Return the unscaled max of np.correlate(h,1-h,'full') for a binary array."""
    # np.correlate works with any numeric dtype
    conv = np.correlate(h, 1 - h, mode="full")
    return int(conv.max())

def _c5_value(h: np.ndarray) -> float:
    """Scaled C5 value for a step function h on [0,2]."""
    n = h.shape[0]
    return _max_corr_unscaled(h) * (2.0 / n)


def _simulated_annealing(n: int, time_limit: float, rng: random.Random):
    """Return a binary sequence of length n with sum=n/2 that (hopefully) has low C5."""
    half = n // 2
    # start from a random balanced vector
    h = np.zeros(n, dtype=np.int8)
    idx = np.arange(n)
    rng.shuffle(idx)
    h[idx[:half]] = 1

    best_h = h.copy()
    best_score = _max_corr_unscaled(h)

    cur_h = h.copy()
    cur_score = best_score

    # temperature schedule – start relatively warm
    temp = max(1.0, cur_score * 0.5)
    cooling = 0.9995

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # propose a swap of a 1 and a 0
        i = rng.randrange(n)
        j = rng.randrange(n)
        if cur_h[i] == cur_h[j]:
            continue
        # do swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        new_score = _max_corr_unscaled(cur_h)
        delta = new_score - cur_score

        # accept if improvement or by Metropolis criterion
        if delta <= 0 or rng.random() < math.exp(-delta / max(temp, 1e-12)):
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        # cool down
        temp *= cooling
        # occasional temperature reset to avoid being stuck too early
        if it % 5000 == 0:
            temp = max(temp, 0.01)

    return best_h.astype(float), best_score


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function h on [0,2] with ∫h=1 that minimises the overlap constant C₅."""
    rng = random.Random(seed)
    np.random.seed(seed)

    # initialise best solution with the provided construction, if any
    best_h = None
    best_score_unscaled = None
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        best_h = h0.copy()
        best_score_unscaled = _max_corr_unscaled(best_h)

    # candidate lengths (must be even for binary vectors)
    n_candidates = [44, 48, 52, 56, 60, 66, 72, 80, 96, 108, 120, 144, 168, 192]

    start_all = time.time()
    # allocate roughly equal time slices, but respect the global budget
    for idx, n in enumerate(n_candidates):
        # stop if we are out of budget
        if time.time() - start_all > budget_s * 0.97:
            break

        # remaining budget and remaining n's
        remaining = len(n_candidates) - idx
        remaining_time = budget_s - (time.time() - start_all)
        # give each remaining n at most  a fair share, but cap at 30 s
        this_budget = min(30.0, remaining_time / remaining)

        # run SA for this length
        h_opt, score_unscaled = _simulated_annealing(n, this_budget, rng)

        if best_score_unscaled is None or score_unscaled < best_score_unscaled:
            best_score_unscaled = score_unscaled
            best_h = h_opt.copy()
            # early exit if we already beat the target bound

    # final safety: if no solution was found (unlikely), fall back to a simple alternating pattern
    if best_h is None:

        best_h[:half] = 1.0
        best_score_unscaled = _max_corr_unscaled(best_h)

    # compute the scaled bound
    final_c5 = best_score_unscaled * (2.0 / best_h.shape[0])
