def hill_climb(bit, max_iter, init_temp, decay):
    n = len(bit)
    ones = np.flatnonzero(bit)
    zeros = np.flatnonzero(1-bit)
    best_c5 = compute_c5(bit)
    best_bit = bit.copy()
    cur_c5 = best_c5
    cur_bit = bit.copy()
    cur_ones = ones
    cur_zeros = zeros
    T = init_temp
    for it in range(max_iter):
        # random swap
        i = np.random.choice(cur_ones)
        j = np.random.choice(cur_zeros)
        # swap bits
        cur_bit[i], cur_bit[j] = cur_bit[j], cur_bit[i]
        # update ones/zeros
        # For performance, we can recompute ones/zeros from scratch every many steps.
        new_c5 = compute_c5(cur_bit)
        if new_c5 < cur_c5 or np.random.rand() < np.exp((cur_c5 - new_c5) / T):
            # accept
            cur_c5 = new_c5
            # update lists
            # swap indices in arrays
            # For simplicity, recompute after acceptance (slower)
            cur_ones = np.flatnonzero(cur_bit)
            cur_zeros = np.flatnonzero(1 - cur_bit)
        else:
            # revert swap
            cur_bit[i], cur_bit[j] = cur_bit[j], cur_bit[i]
        # update best
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_bit = cur_bit.copy()
        # temperature schedule
        T *= decay
    return best_bit, best_c5

def compute_c5(h):
    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * (2.0 / n)

def random_balanced(n):
    # n even
    ones = np.random.choice(n, n//2, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[ones] = 1.0
    return h

def solve_one(n, iter_swaps, temp_start=0.05, temp_decay=0.9995):
    # Simulated annealing
    h = random_balanced(n)
    best_h = h.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    T = temp_start
    # Precompute indices arrays:
    ones_idx = np.flatnonzero(h)
    zeros_idx = np.flatnonzero(1 - h)
    for i in range(iter_swaps):
        # randomly pick swap
        i1 = np.random.choice(ones_idx)
        i0 = np.random.choice(zeros_idx)
        # swap
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        # Decide acceptance
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            # accept
            cur_c5 = new_c5
            # update indices

            ones_idx = np.flatnonzero(h)

        # update best
        if cur_c5 < best_c5:

            best_h = h.copy()
        # temperature schedule
        T *= temp_decay
        if T < 1e-6:

    return best_h, best_c5

def simulated_annealing_binary(h, iterations, init_temp, decay):
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    T = init_temp
    n = len(h)
    for it in range(iterations):
        # choose random swap
        i1 = np.random.randint(n)
        # ensure i1 is 1 if we want swap with zero; we can randomly pick two indices and swap regardless: if they are same value, swap leads same config (no change). That's wasteful but fine. Let's ensure we pick different.
        i0 = np.random.randint(n)
        while i0 == i1:
            i0 = np.random.randint(n)
        # swap values (they could be same)
        if cur_h[i1] == cur_h[i0]:
            continue
        cur_h[i1], cur_h[i0] = cur_h[i0], cur_h[i1]
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i1], cur_h[i0] = cur_h[i0], cur_h[i1]
        T *= decay
        if T < 1e-6:
            T = 1e-6
    return best_h, best_c5

import time
import numpy as np

# ---------- basic utilities ----------
def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discrete step function."""
    n = h.size
    # correlation h * (1-h) with zero padding
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * (2.0 / n))


def random_balanced(n: int) -> np.ndarray:
    """Binary vector of length n with exactly n/2 ones."""
    idx = np.random.choice(n, n // 2, replace=False)
    h = np.zeros(n, dtype=np.float64)
    h[idx] = 1.0
    return h


def thue_morse(n: int) -> np.ndarray:
    """Thue–Morse binary pattern (0/1). Works exactly for powers of two."""
    return np.fromiter(((bin(i).count("1") & 1) for i in range(n)),
                       dtype=np.float64, count=n)


# ---------- simulated annealing ----------
def simulated_annealing(start_h: np.ndarray,
                       max_iters: int,
                       init_temp: float = 0.02,
                       decay: float = 0.9999) -> tuple[np.ndarray, float]:
    """Binary SA with swaps that preserves the weight."""
    h = start_h.copy()
    n = h.size
    cur = compute_c5(h)
    best_h, best = h.copy(), cur
    T = init_temp
    for _ in range(max_iters):
        # propose a swap of opposite bits
        i, j = np.random.randint(n, size=2)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]          # perform swap
        new = compute_c5(h)
        delta = new - cur
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            cur = new
            if cur < best:
                best, best_h = cur, h.copy()
        else:
            h[i], h[j] = h[j], h[i]      # revert
        T = max(T * decay, 1e-8)
    return best_h, best

        except Exception:

    # ----- lengths to explore (even numbers, modest size) -----
    lengths = [64, 80, 96, 112, 128, 144, 160, 176,
