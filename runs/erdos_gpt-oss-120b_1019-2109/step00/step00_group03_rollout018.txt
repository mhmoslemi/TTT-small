def hill_climb_binary(h, rng, max_steps, time_limit):
    n = len(h)
    best = h.copy()
    best_score = evaluate_score(h)
    steps = 0
    while steps < max_steps and time.time() < time_limit:
        # compute correlation and find max location
        c = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(c)
        max_val = c[max_idx]
        if max_val <= best_score: # maybe not needed
            pass
        if max_val <= target_val: # but we don't know target
            pass
        # compute shift k
        k = max_idx - (n-1)
        # find mismatched indices list
        if k >= 0:
            i_range = np.arange(0, n - k)
        else:
            i_range = np.arange(-k, n)
        j_range = i_range + k
        mismatched_mask = (h[i_range] == 1) & (h[j_range] == 0)
        mismatched_i = i_range[mismatched_mask]
        if mismatched_i.size == 0:
            # try random swap
            i = rng.integers(n)
            j = rng.integers(n)
            if h[i] == h[j]:
                continue
            # swap
            new_h = h.copy()
            new_h[i], new_h[j] = h[j], h[i]
            new_score = evaluate_score(new_h)
            if new_score < best_score:
                h = new_h
                best_score = new_score
                best = h.copy()
            steps += 1
            continue
        # choose random i from mismatched list
        i = rng.choice(mismatched_i)
        # choose random zero index not equal i
        zero_idx = np.where(h == 0)[0]
        # optionally avoid zeros that would create bigger mismatch.
        j = rng.choice(zero_idx)
        # swap i and j
        h[i], h[j] = 0, 1
        new_score = evaluate_score(h)
        if new_score <= best_score:
            best_score = new_score
            best = h.copy()
        else:
            # revert swap
            h[i], h[j] = 1, 0
        steps += 1
    return best, best_score
```

import time, math
import numpy as np


def max_overlap(h: np.ndarray) -> int:
    """Return max_{k} sum_i h[i]*(1-h[i+k]) (linear correlation)."""
    return int(np.max(np.correlate(h, 1 - h, mode="full")))


def max_overlap_info(h: np.ndarray):
    """Return (max_value, shift) where shift = index-(n-1) for the max."""
    c = np.correlate(h, 1 - h, mode="full")
    idx = int(c.argmax())
    n = h.shape[0]
    return int(c[idx]), idx - (n - 1)


def mismatched_indices_for_shift(h: np.ndarray, shift: int):
    """Indices i where h[i]=1 and h[i+shift]=0 (within the overlapping window)."""
    n = h.shape[0]
    if shift >= 0:
        i = np.arange(0, n - shift)
    else:
        i = np.arange(-shift, n)
    j = i + shift
    mask = (h[i] == 1) & (h[j] == 0)
    return i[mask]


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a length‑n binary vector with exactly n/2 ones."""
    half = n // 2
    a = np.zeros(n, dtype=np.int8)
    a[:half] = 1
    rng.shuffle(a)
    return a


def hill_climb_binary(start_h: np.ndarray, rng: np.random.Generator, end_time: float):
    """
    Simple hill‑climbing on a binary vector with fixed Hamming weight.
    Swaps a 1 and a 0 that belong to the shift producing the current worst overlap.
    Stops after time runs out or after a burst of unsuccessful attempts.
    """
    h = start_h.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = max_overlap(h)

    attempts_without_improve = 0
    max_stale = 3000  # stop after this many non‑improving trials

    while time.time() < end_time and attempts_without_improve < max_stale:
        cur_val, shift = max_overlap_info(h)

        # indices that contribute to the current worst overlap
        bad_i = mismatched_indices_for_shift(h, shift)

        if bad_i.size == 0:
            # degenerate (should only happen when cur_val == 0)
            attempts_without_improve += 1
            continue

        i = int(rng.choice(bad_i))
        zeros = np.where(h == 0)[0]
        j = int(rng.choice(zeros))

        # perform the swap (preserves total weight)
        h[i], h[j] = 0, 1
        new_val = max_overlap(h)

        if new_val < best_val:
            best_val = new_val
            best_h = h.copy()
            attempts_without_improve = 0
        else:
            # revert – the move was not helpful
            h[i], h[j] = 1, 0
            attempts_without_improve += 1

    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on a uniform grid that yields a small
    value of C5 = max_k ∫ h(x)(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s
    target_c5 = 0.3808

    # Choose a set of candidate discretisations (all even, small enough for speed)
    candidate_ns = [70, 84, 100, 120, 140, 160, 180, 200]

    # If a parent construction exists, use its length as an additional candidate
    global_initial = globals().get("initial_h_values", None)
    if global_initial is not None:
        n_parent = int(np.asarray(global_initial).shape[0])
        if n_parent % 2 == 0 and n_parent not in candidate_ns:
            candidate_ns.append(n_parent)

    # Initialise best solution with the parent if present
    if global_initial is not None:
        h_parent = np.array(global_initial, copy=False).astype(np.int8)
        # force values to {0,1} while keeping exactly half ones
        half_parent = h_parent.shape[0] // 2
        h_parent = (h_parent >= 0.5).astype(np.int8)
        cur = h_parent.sum()
        if cur != half_parent:
            diff = half_parent - cur
            if diff > 0:
                zero_idx = np.where(h_parent == 0)[0]
                flip = rng.choice(zero_idx, size=diff, replace=False)
                h_parent[flip] = 1
            else:
                one_idx = np.where(h_parent == 1)[0]
                flip = rng.choice(one_idx, size=-diff, replace=False)
                h_parent[flip] = 0
        best_h = h_parent.copy()
        best_val = max_overlap(best_h)
        best_n = best_h.shape[0]
    else:
        best_h = None
        best_val = math.inf
        best_n = None

    # Main search loop – many short hill‑climbs with random restarts
    while time.time() < deadline * 0.995:
        remaining = deadline - time.time()
        if remaining < 0.2:
            break

        # pick a problem size
        n = int(rng.choice(candidate_ns))
        # generate a random start point with the correct weight
        start_h = random_binary(n, rng)

        # allocate a slice of the remaining budget to this climb
        slice_time = min(remaining * 0.5, 5.0)
        end_climb = time.time() + slice_time
        h_opt, val_opt = hill_climb_binary(start_h, rng, end_climb)

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            best_n = n
            if best_val * 2.0 / best_n <= target_c5:
                break

        # occasional refinement from the current global best
        if best_h is not None and remaining > 0.5:
            refine_time = min(remaining * 0.2, 2.0)
            end_refine = time.time() + refine_time
            h_ref, val_ref = hill_climb_binary(best_h, rng, end_refine)
            if val_ref < best_val:
                best_val = val_ref
                best_h = h_ref.copy()
                best_n = n
                if best_val * 2.0 / best_n <= target_c5:
                    break

    # Final answer – ensure we have something
    if best_h is None:
        best_n = 70
        best_h = random_binary(best_n, rng)

    c5_bound = best_val * 2.0 / best_n
    return (best_h.astype(float), c5_bound, best_n)
