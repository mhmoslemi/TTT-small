def compute_c5(h):
    N = h.shape[0]
    # correlation via FFT
    L = 1 << ((2*N - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1 - h, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)
    # for full linear correlation, valid range 0..2N-2. ignore rest (zero padded)
    # The correlation peaks at index N-1 corresponding to shift 0.
    # We need max over all indices.
    max_val = np.max(corr[:2*N-1])
    return max_val * (2.0 / N)
```

def compute_c5(h):
    N = len(h)
    L = 1 << ((2*N - 1).bit_length())
    H = np.fft.fft(h, L)
    G = np.fft.fft(1 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real[:2*N-1]
    max_val = np.max(corr)
    return max_val * (2.0 / N)
```

import time, math, random
import numpy as np

def _correlation_vec(h):
    """Linear correlation of h with (1-h). Returns length 2*len(h)-1 array."""
    n = h.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    H = np.fft.fft(h, L)
    G = np.fft.fft(1.0 - h, L)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    return corr


def _c5_value(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k))dx  using the discrete model."""
    n = h.shape[0]
    corr = _correlation_vec(h)
    return np.max(corr) * (2.0 / n)


def _make_derivative(h, k):
    """
    For a given integer shift k (k = idx - (n-1)),
    return the array d_i = 1 - h_{i+k}  (with out‑of‑range treated as 0 ⇒ d_i=1).
    """
    n = h.shape[0]
    d = np.ones(n, dtype=h.dtype)
    if k >= 0:
        if k < n:
            d[: n - k] = 1.0 - h[k:]
    else:
        kk = -k
        if kk < n:
            d[n - kk :] = 1.0 - h[: n - kk]
    return d


def _normalize(h):
    """Scale and clip h so that 0≤h≤1 and sum(h)=len(h)/2 exactly."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == 0:
        h[:] = 0.5
        return h
    h = h * (target / s)
    h = np.clip(h, 0.0, 1.0)
    # tiny redistribution to hit the sum exactly
    residual = target - h.sum()
    for _ in range(100):
        if abs(residual) < 1e-12:
            break
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if len(idx) == 0:
                break
            add = residual / len(idx)
            add = np.minimum(add, 1.0 - h[idx])
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) == 0:
                break
            add = residual / len(idx)
            add = np.maximum(add, -h[idx])
        h[idx] += add
        residual = target - h.sum()
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that gives a low C5.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ---------- initialise -------------------------------------------------
    target_n = 256                     # a power of two, small enough & fast
    # use provided construction if present, otherwise start from random
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        if init.ndim != 1 or init.size == 0:
            init = np.full(target_n, 0.5)
        # up‑sample / down‑sample to target_n by linear interpolation
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        h = np.random.rand(target_n)
    h = _normalize(h)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ---------- optimisation parameters ------------------------------------
    start_time = time.time()
    max_time = budget_s * 0.95          # keep a safety margin
    n = target_n
    dx = 2.0 / n

    init_temp = 0.02
    min_temp = 1e-6
    max_delta = 0.2                     # maximal amount transferred in one move
    iter_cnt = 0
    no_improve = 0

    # ---------- main loop -------------------------------------------------
    while time.time() - start_time < max_time:
        # full correlation and current C5
        corr = _correlation_vec(h)
        cur_c5 = np.max(corr) * dx
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # shift producing the worst overlap (largest correlation entry)
        idx_max = int(np.argmax(corr))
        k_shift = idx_max - (n - 1)          # integer shift corresponding to idx

        # derivative d_i = 1 - h_{i+k}
        d = _make_derivative(h, k_shift)

        # choose a source (high contribution) and a target (low contribution)
        src_mask = h > 1e-8
        if not np.any(src_mask):
            src = random.randrange(n)
        else:
            src_candidates = np.flatnonzero(src_mask)
            src = src_candidates[np.argmax(d[src_candidates] * h[src_candidates])]

        tgt_mask = h < 1.0 - 1e-8
        if not np.any(tgt_mask):
            tgt = random.randrange(n)
        else:
            tgt_candidates = np.flatnonzero(tgt_mask)
            tgt = tgt_candidates[np.argmin(d[tgt_candidates] * (1.0 - h[tgt_candidates]))]

        # amount we may move from src to tgt while keeping box constraints
        allowance = min(h[src], 1.0 - h[tgt], max_delta)
        if allowance <= 1e-12:
            # fall back to a small random perturbation
            i, j = random.sample(range(n), 2)
            up = 1.0 - h[j]
            down = h[i]
            mx = min(up, down, max_delta)
            if mx <= 0:
                continue
            delta = mx * random.random()
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            delta = allowance * random.random()
            h_new = h.copy()
            h_new[src] -= delta
            h_new[tgt] += delta

        # evaluate new state
        new_c5 = _c5_value(h_new)

        # simulated‑annealing acceptance
        elapsed = time.time() - start_time
        temperature = max(init_temp * (1.0 - elapsed / max_time), min_temp)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            prob = math.exp(-(new_c5 - cur_c5) / temperature)
            if random.random() < prob:
                accept = True

        if accept:
            h = h_new

        # occasional restart when stuck
        if no_improve > 25000:
            h = best_h.copy()
            no_improve = 0

        # shrink maximal transfer as optimisation proceeds
        if iter_cnt % 5000 == 0 and iter_cnt > 0:
            max_delta = max(0.01, max_delta * 0.9)

        iter_cnt += 1

    # final normalisation (tiny numerical drift may appear)
    best_h = _normalize(best_h)
    best_c5 = _c5_value(best_h)

    return best_h.tolist(), float(best_c5), n
