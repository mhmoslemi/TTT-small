def project_onto_box_sum(v, target):
    n = len(v)
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(100):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

import time
import numpy as np

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x_i<=1, sum x_i = target}."""
    # Find lambda such that sum_i clip(v_i - lam,0,1) = target
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of_h(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a given step function h."""
    # full mode correlation of h with (1-h), then multiply by dx = 2/n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / h.size)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step‑function h on [0,2] with ∫h = 1
    to minimise  C5 = max_k ∫ h(x)(1‑h(x+k)) dx.

    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # -------------------------------------------------
    # initialise h
    # -------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
        n = h0.size
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        n = 80                         # a modest size, < 1k as requested
        h0 = rng.random(n)

    target_sum = n / 2.0
    cur_h = _project_box_sum(h0, target_sum)
    cur_score = _c5_of_h(cur_h)

    best_h = cur_h.copy()
    best_score = cur_score

    # -------------------------------------------------
    # optimisation parameters
    # -------------------------------------------------
    max_iters = 10_000_000               # safety upper bound
    it = 0
    stagnation = 0
    T0 = 0.015                            # initial temperature for SA
    grad_prob = 0.3                       # probability to take a gradient step

    # -------------------------------------------------
    # main loop
    # -------------------------------------------------
    while True:
        # time check
        if time.time() - start_time >= budget_s - 4.0:
            break

        # decide move type
        if rng.random() < grad_prob:
            # ---- gradient move targeting the worst lag ----
            conv = np.correlate(cur_h, cur_h, mode="full")
            idx_min = int(conv.argmin())
            lag = idx_min - (n - 1)

            # gradient = shift(cur_h, +lag) + shift(cur_h, -lag)
            grad = np.zeros_like(cur_h)
            if lag >= 0:
                grad[: n - lag] += cur_h[lag:]
                grad[lag:] += cur_h[: n - lag]
            else:
                l = -lag

                grad[: n - l] += cur_h[l:]
