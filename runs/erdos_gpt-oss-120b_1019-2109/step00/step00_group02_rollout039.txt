    lam_high = 1 - np.min(x) + 1.0

        if s < target:

            lam_high = lam_mid
    # return final y
    return np.clip(x + lam_mid, 0.0, 1.0)

def project_onto_sum_and_bounds(x, target):
   # x is any real vector.
   # Want y = clip(x + lam, 0,1) such that sum(y) = target.
   # Using binary search on lam.
   n = len(x)

   low = -np.max(x) - 1.0
   high = 1 - np.min(x) + 1.0
   for _ in range(50):
       lam = (low + high) / 2.0
       y = np.clip(x + lam, 0.0, 1.0)
       s = y.sum()
       if s < target:
           low = lam
       else:
           high = lam
   y = np.clip(x + (low + high)/2.0, 0.0, 1.0)

import numpy as np
import time

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------

def _project_to_bounds_sum(x, target):
    """
    Project vector x onto the set {0 <= y_i <= 1, sum(y) = target}
    using a simple bisection on a common shift λ:
        y_i = clip(x_i + λ, 0, 1)
    """
    lo = -np.max(x) - 1.0      # ensures all clipped to 0
    hi = 1.0 - np.min(x) + 1.0  # ensures all clipped to 1
    for _ in range(50):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of_h(h):
    """
    Compute the objective C5 for a step function represented by h.
    Uses the same routine that the harness uses (np.correlate with mode='full')
    unless a higher‑level evaluator is provided.
    """
    if 'evaluate_erdos_solution' in globals():
        return float(evaluate_erdos_solution(np.asarray(h)))
    # fallback
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return float(corr.max() * (2.0 / n))


def _gradient_of_shift(h, shift):
    """
    Subgradient of the correlation term M_k = Σ h_i (1 - h_{i+shift})
    with respect to h, for the shift that currently attains the maximum.
    """
    n = len(h)
    g = np.zeros(n, dtype=float)
    if shift >= 0:
        s = shift
        g[: n - s] += 1.0 - h[s:]          # (1 - h_{i+shift}) part
        g[s:]       -= h[: n - s]          # -h_{i-shift} part
    else:
        s = -shift
        g[s:]       += 1.0 - h[: n - s]    # (1 - h_{i-shift})
        g[: n - s]  -= h[s:]               # -h_{i+shift}
    return g


def _random_feasible(n, target):
    """Generate a random vector in [0,1]^n with prescribed sum."""
    x = np.random.rand(n)
    return _project_to_bounds_sum(x, target)

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points."""
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and an initial feasible vector
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.size
        target_sum = n / 2.0
        # guarantee feasibility (tiny numerical drift may appear)
        h0 = _project_to_bounds_sum(h0, target_sum)
    else:
        n = 80                     # a modest size when no parent is provided
        target_sum = n / 2.0
        h0 = _random_feasible(n, target_sum)

    best_h = h0.copy()
    best_c5 = _c5_of_h(best_h)

    # ------------------------------------------------------------------
    # Main search – a mixture of gradient descent on the worst shift
    # and occasional random restarts.
    # ------------------------------------------------------------------
    # parameters controlling the inner optimisation
    max_inner_iters = 4000
    step_candidates = (0.2, 0.1, 0.05, 0.02, 0.01, 0.005)

    while time.time() - start_time < budget_s - 2.0:
        # occasionally start from the best known solution, otherwise from random
        if np.random.rand() < 0.3:
            h = best_h.copy()
        else:
            h = _random_feasible(n, target_sum)

        cur_c5 = _c5_of_h(h)
        no_improve = 0

        for it in range(max_inner_iters):
            # time guard
            if time.time() - start_time > budget_s - 1.0:
                break

            # locate the shift that currently maximises the correlation term
            corr_full = np.correlate(h, 1 - h, mode='full')
            max_idx = int(corr_full.argmax())
            shift = max_idx - (n - 1)

            # compute subgradient for that shift
            g = _gradient_of_shift(h, shift)
            if np.linalg.norm(g) < 1e-12:          # flat region – poke a little
                h = _project_to_bounds_sum(h + (np.random.rand(n) - 0.5) * 0.01,
                                            target_sum)
                cur_c5 = _c5_of_h(h)
                continue

            # normalise the direction so that the largest coordinate change is 1
            g_norm = g / (np.max(np.abs(g)) + 1e-12)

            improved = False
            for alpha in step_candidates:
                h_new = _project_to_bounds_sum(h - alpha * g_norm, target_sum)
                new_c5 = _c5_of_h(h_new)
                if new_c5 < cur_c5 - 1e-9:
                    h = h_new
                    cur_c5 = new_c5
                    improved = True
                    no_improve = 0
                    if new_c5 < best_c5 - 1e-9:
                        best_c5 = new_c5
                        best_h = h.copy()
                    break

            if not improved:
                # small random jitter in case we are stuck
                noise = (np.random.rand(n) - 0.5) * 0.01
                h_new = _project_to_bounds_sum(h + noise, target_sum)
                new_c5 = _c5_of_h(h_new)
                if new_c5 < cur_c5:
                    h = h_new
                    cur_c5 = new_c5
                    no_improve = 0
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h.copy()
                else:
                    no_improve += 1
                    if no_improve > 80:
                        break   # give up on this start

    # ------------------------------------------------------------------
    # Final polishing from the best candidate found
    # ------------------------------------------------------------------
    h = best_h.copy()
    cur_c5 = best_c5
    for _ in range(800):
        if time.time() - start_time > budget_s - 0.5:
            break
        corr_full = np.correlate(h, 1 - h, mode='full')
        shift = int(corr_full.argmax()) - (n - 1)
        g = _gradient_of_shift(h, shift)
        if np.linalg.norm(g) < 1e-12:
            break
        g_norm = g / (np.max(np.abs(g)) + 1e-12)
        # progressively smaller steps
        for alpha in (0.02, 0.01, 0.005, 0.002):
            h_new = _project_to_bounds_sum(h - alpha * g_norm, target_sum)
            new_c5 = _c5_of_h(h_new)
            if new_c5 < cur_c5:
                h = h_new
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                break

    # Ensure the returned vector respects the constraints (tiny numerical adjustments)
    best_h = _project_to_bounds_sum(np.asarray(best_h, dtype=float), target_sum)

    return best_h.tolist(), float(best_c5), int(n)
