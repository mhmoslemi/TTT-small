import numpy as np, time, random, math

def compute_c5(h):
    n=len(h)
    dx=2.0/n
    corr=np.correlate(h, 1-h, mode='full')
    return np.max(corr)*dx

def random_binary(n):
    # half ones (n even)
    half=n//2
    h=np.zeros(n, dtype=np.float64)
    h[:half]=1
    np.random.shuffle(h)
    return h

def hill_climb(h, max_iter=1000):
    n=len(h)
    best_h=h.copy()
    best_c5=compute_c5(h)
    current_h=h.copy()
    current_c5=best_c5
    for it in range(max_iter):
        # random swap 
        idx1 = np.random.randint(0,n)
        idx2 = np.random.randint(0,n)
        if current_h[idx1]==current_h[idx2]:
            continue
        # swap
        new_h=current_h.copy()
        new_h[idx1], new_h[idx2] = new_h[idx2], new_h[idx1]
        new_c5=compute_c5(new_h)
        if new_c5 < current_c5 - 1e-12:
            current_h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_h = new_h.copy()
                best_c5 = new_c5
        # else maybe accept sometimes? Not needed for hill climb
    return best_h, best_c5
```

def simulated_annealing(h, max_steps, time_limit):
    n = len(h)
    cur_h = h.copy()
    cur_max = max_mismatch_count(cur_h)
    best_h = cur_h.copy()
    best_max = cur_max

    for step in range(max_steps):
        # check time limit
        if time.time() > time_limit:
            break
        # temperature decreasing
        T = max(0.01, (max_steps - step) / max_steps)  # simple schedule
        # random swap
        # find random 1 and random 0
        i1 = np.random.randint(n)
        if cur_h[i1] == 1:
            # find zero
            i0 = np.random.randint(n)
            while cur_h[i0] != 0:
                i0 = np.random.randint(n)
        else:
            # cur_h[i1]==0, find one
            i0 = np.random.randint(n)
            while cur_h[i0] != 1:
                i0 = np.random.randint(n)
        # swap
        new_h = cur_h.copy()
        new_h[i1], new_h[i0] = new_h[i0], new_h[i1]
        new_max = max_mismatch_count(new_h)
        delta = new_max - cur_max
        if delta < 0 or np.random.rand() < math.exp(-delta / T):
            cur_h = new_h
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = new_h.copy()
                # early exit if best_max very low
                if best_max * 2 / n < 0.3800:
                    break
    return best_h, best_max

import time, math, random
import numpy as np

def _max_mismatch_count(h: np.ndarray) -> int:
    """Return max_{k} sum_i h[i] * (1-h[i+k]) using linear correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return int(corr.max())

def _simulated_annealing(start_h: np.ndarray, max_steps: int, time_limit: float):
    """Simple SA that swaps a 1 with a 0, keeps the sum constraint."""
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_val = _max_mismatch_count(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    for step in range(max_steps):
        if time.time() > time_limit:
            break
        # linear temperature schedule
        T = max(0.001, (max_steps - step) / max_steps)

        # pick a random 1 and a random 0 to swap
        i1 = np.random.randint(n)
        if cur_h[i1] == 1.0:
            i0 = np.random.randint(n)
            while cur_h[i0] != 0.0:
                i0 = np.random.randint(n)
        else:
            i0 = np.random.randint(n)
            while cur_h[i0] != 1.0:
                i0 = np.random.randint(n)

        # propose swap
        new_h = cur_h.copy()
        new_h[i1], new_h[i0] = new_h[i0], new_h[i1]
        new_val = _max_mismatch_count(new_h)

        delta = new_val - cur_val
        if delta <= 0 or random.random() < math.exp(-delta / T):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
                # early exit if already very good
                if best_val * 2.0 / n < 0.3800:
                    break
    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) with a low C5 value."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    time_allow = lambda: time.time() - start_time < budget_s * 0.96

    # ------------------------------------------------------------------
    # candidate lengths (only even lengths => integer half‑sum)
    cand_ns = [40, 44, 48, 50, 52, 54, 56, 58, 60, 62, 64, 66, 68, 70, 72, 80, 90, 100]

    # If a parent construction is given, try its length first
    init_h_global = globals().get("initial_h_values")
    if isinstance(init_h_global, (list, np.ndarray)):
        init_h = np.asarray(init_h_global, dtype=np.float64)
        init_len = init_h.shape[0]
        # Ensure the sum constraint holds (it should for the provided record)
        if not np.isclose(init_h.sum() * 2.0 / init_len, 1.0):
            # If the provided vector is fractional, we keep it for a warm start
            pass
        if init_len % 2 == 0 and init_len not in cand_ns:
            cand_ns = [init_len] + cand_ns
    else:
        init_h = None

    best_h = None
    best_c5 = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    for n in cand_ns:
        half = n // 2
        # number of independent starts for this n (more for smaller n)
        max_restarts = 120 if n <= 60 else 80

        for _ in range(max_restarts):
            if not time_allow():
                break

            # ------------------------------------------------------------------
            # initialise a feasible vector
            if init_h is not None and init_h.shape[0] == n:
                h0 = init_h.copy()
                # enforce exact half‑ones if it is integer‑valued; otherwise keep as is
                if np.all(np.isin(h0, [0.0, 1.0])):
                    # already good
                    pass
                else:
                    # fallback to binary random start
                    h0 = np.concatenate((np.ones(half), np.zeros(n - half)))
                    np.random.shuffle(h0)
            else:
                h0 = np.concatenate((np.ones(half), np.zeros(n - half)))
                np.random.shuffle(h0)

            # ------------------------------------------------------------------
            # Run a modest SA on this start
            limit = start_time + budget_s * 0.96
            h_loc, max_cnt = _simulated_annealing(h0, max_steps=20000, time_limit=limit)

            # Optional short greedy improvement pass
            improved = True
            while improved and time_allow():
                improved = False
                # try a few random swaps looking for strict improvement
                for _ in range(500):
                    i, j = np.random.randint(n, size=2)
                    if h_loc[i] == h_loc[j]:
                        continue
                    cand = h_loc.copy()
                    cand[i], cand[j] = cand[j], cand[i]
                    cand_max = _max_mismatch_count(cand)
                    if cand_max < max_cnt:
                        h_loc = cand
                        max_cnt = cand_max
                        improved = True
                        break

            c5_val = max_cnt * 2.0 / n
            if c5_val < best_c5 - 1e-12:
                best_c5 = c5_val
                best_h = h_loc.copy()
                best_n = n
                if best_c5 < 0.3808:
                    # good enough – exit early
                    break
        if best_c5 < 0.3808 or not time_allow():
            break

    # ------------------------------------------------------------------
    # Fallback if nothing was found
    if best_h is None:

        n = 40
        best_h = np.concatenate((np.ones(n // 2), np.zeros(n // 2)), dtype=np.float64)
        best_n = n
        best_c5 = _max_mismatch_count(best_h) * 2.0 / n

    return (best_h, float(best_c5), int(best_n))
