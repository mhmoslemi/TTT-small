def hill_climb_anneal(N, max_iters, T0):
    # returns best_h, best_val
    # generate random balanced vector
    h = random_balanced(N)
    # pre-select indices arrays
    one_idx = list(np.where(h == 1)[0])
    zero_idx = list(np.where(h == 0)[0])
    cur_val = evaluate_h(h)
    best_h = h.copy()
    best_val = cur_val
    for it in range(max_iters):
        # temperature linearly decay
        T = T0 * (1 - it / max_iters)
        # propose swap
        i_idx = np.random.randint(len(one_idx))
        j_idx = np.random.randint(len(zero_idx))
        i = one_idx[i_idx]
        j = zero_idx[j_idx]
        # swap
        h[i], h[j] = h[j], h[i]
        # update indices lists
        # after swap, i becomes zero, j becomes one
        one_idx[i_idx] = j
        zero_idx[j_idx] = i
        new_val = evaluate_h(h)
        diff = new_val - cur_val
        if diff <= 0 or np.random.rand() < np.exp(-diff / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
            # revert indices
            one_idx[i_idx] = i
            zero_idx[j_idx] = j
        # optionally break early if best_val is low enough.
        if best_val <= target_c5: break
    return best_h, best_val

    one_idx = [int(i) for i in np.where(h==1)[0]]
    zero_idx = [int(i) for i in np.where(h==0)[0]]
    cur_val = evaluate_h(h)
    best_h = h.copy()

    # define temp schedule
    T0 = 0.02  # may adjust
    # we can compute max_iters as min(max_iters, some large)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # temperature
        T = T0 * (1 - it / max_iters)
        # propose swap
        i_idx = np.random.randint(len(one_idx))

def hill_climb_sim_ann(h_init, max_iters, target_c5, deadline):
    # returns best_h, best_val
    # local variables
    h = h_init.copy()
    # compute indices
    one_nodes = list(np.where(h == 1)[0])
    zero_nodes = list(np.where(h == 0)[0])
    cur_val = evaluate_h(h)
    best_h = h.copy()
    best_val = cur_val

    # For temperature schedule, use T0 = 0.02
    T0 = 0.02
    # compute steps per iteration
    it = 0
    last_improve_iter = 0

    while it < max_iters and time.time() < deadline:
        # compute temperature
        T = T0 * (1 - it / max_iters)
        # propose swap
        i_idx = np.random.randint(len(one_nodes))
        j_idx = np.random.randint(len(zero_nodes))
        i = one_nodes[i_idx]
        j = zero_nodes[j_idx]

        # swap
        h[i], h[j] = h[j], h[i]
        # update indices
        one_nodes[i_idx] = j
        zero_nodes[j_idx] = i

        new_val = evaluate_h(h)
        diff = new_val - cur_val
        if diff <= 0 or np.random.rand() < np.exp(-diff / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                last_improve_iter = it
                # early exit if good enough
                if best_val <= target_c5:
                    break
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
            one_nodes[i_idx] = i
            zero_nodes[j_idx] = j

        # occasional random restart if stuck
        # if it - last_improve_iter > 5000: restart
        if it - last_improve_iter > 50000:  # maybe too long, but we can do
            # restart: random shuffle
            h = random_balanced(len(h))
            one_nodes = list(np.where(h == 1)[0])
            zero_nodes = list(np.where(h == 0)[0])
            cur_val = evaluate_h(h)
            last_improve_iter = it

        it += 1

    return best_h, best_val

import time
import numpy as np

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------

def evaluate_h(h):
    """Return C5 for a step function h given as a 0/1 array."""
    n = len(h)
    # correlation of h with (1-h), full mode gives all integer lags
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = corr.max()
    return max_corr * (2.0 / n)


def random_balanced(n):
    """Random binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.int8)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1
    return h


def thue_morse(n):
    """Thue–Morse binary sequence (0/1) of length n (requires n power of 2)."""
    return np.array([bin(i).count("1") % 2 for i in range(n)], dtype=np.int8)


def optimize_h(start_h, deadline, target=0.3808):
    """
    Simulated‑annealing style local search starting from start_h.
    Returns the best vector found and its C5 value.
    """
    n = len(start_h)
    h = start_h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    cur_val = evaluate_h(h)
    best_h = h.copy()
    best_val = cur_val

    # annealing schedule parameters
    T0 = 0.02
    it = 0
    last_improve = 0
    # very large iteration cap; real stop comes from time budget
    max_iters = 10_000_000

    while time.time() < deadline and it < max_iters:
        # linear temperature decay
        T = T0 * max(0.0, 1.0 - it / 500_000)

        # propose a swap between a 1 and a 0
        i_idx = np.random.randint(len(ones))
        j_idx = np.random.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

        # perform swap
        h[i], h[j] = h[j], h[i]
        ones[i_idx] = j
        zeros[j_idx] = i

        new_val = evaluate_h(h)
        diff = new_val - cur_val
        if diff <= 0 or np.random.rand() < np.exp(-diff / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                last_improve = it
                if best_val <= target:
                    break
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
            ones[i_idx] = i
            zeros[j_idx] = j

        # periodic random restart if stuck for a long while
        if it - last_improve > 200_000:
            h = random_balanced(n)
            ones = list(np.where(h == 1)[0])
            zeros = list(np.where(h == 0)[0])
            cur_val = evaluate_h(h)
            last_improve = it

        it += 1

    return best_h, best_val


def search_one_n(n, deadline, target=0.3808):
    """Run a few independent starts for a given length n."""
    best_h = None
    best_val = np.inf
    attempts = 3
    for att in range(attempts):
        if time.time() >= deadline:
            break
        # use Thue‑Morse as first start when possible
        if att == 0 and ((n & (n - 1)) == 0):
            start = thue_morse(n)
        else:
            start = random_balanced(n)
        h_opt, val_opt = optimize_h(start, deadline, target)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt
        if best_val <= target:
            break
    return best_h, best_val
