import numpy as np, math, time

def _project(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the hyperplane sum(v)=target
    intersected with the box [0,1]^n.
    """
    lo, hi = -2.0, 2.0          # wide enough to bracket the solution
    for _ in range(60):        # binary search for the Lagrange multiplier
        tau = (lo + hi) / 2.0
        w = np.clip(v - tau, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = tau
        else:
            hi = tau
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _c5_cost(h: np.ndarray) -> float:
    """
    Compute C5 = max_k ∫ h(x)(1-h(x+k))dx for a step function represented by h.
    In the discretised setting this equals
        max(np.correlate(h, 1-h, mode='full')) * 2/len(h)
    """
    N = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / N)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Local‑search / simulated annealing optimiser for the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

        init = np.array(initial_h_values, dtype=float)   # parent may be present
    except NameError:
        init = None

    upscale = 8                     # repeat each parent entry this many times
    if init is not None:
        h = np.repeat(init, upscale)
    else:
        N0 = 200                     # even length, half ones half zeros
        h = np.zeros(N0, dtype=float)
        h[: N0 // 2] = 1.0
        rng.shuffle(h)

    N = h.shape[0]
    target_sum = N / 2.0

    # enforce exact integral (sum) and box constraints
    cur_sum = h.sum()

    # small random perturbation to break symmetry of a pure parent copy
    eps_start = 0.02

        h = _project(h, target_sum)
