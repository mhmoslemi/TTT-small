    q = 1 - h
    n = len(h)

    # returns w satisfying 0 <= w_i <= 1, sum w = M, minimizing ||w-v||^2.
    # algorithm: start with w = np.clip(v, 0, 1). Then adjust by solving sum w = M by adding uniform shift and then projection onto box.

    # We'll implement simple iteration: while sum not close, adjust.
    # Since M is between 0 and n, reasonable.
    # Use algorithm: w = v - tau, then clip to [0,1], adjust tau via binary search.
    # Compute function f(tau) = sum clip(v - tau,0,1) - M.
    # Find tau such that f(tau)=0.
    # f(tau) decreasing in tau.
    # tau in [min(v)-1, max(v)] maybe.
    # We'll binary search tau in range [low, high].
    low = np.min(v) - 1  # when shift large positive, clips to 0.
    high = np.max(v)  # when shift large negative, clips to 1.
    for _ in range(50):
        mid = (low + high) / 2
        w = np.clip(v - mid, 0, 1)
        s = w.sum()
        if s > M:
            low = mid
        else:
            high = mid
    w = np.clip(v - (low+high)/2, 0, 1)
    # adjust small numeric error by scaling of residual?
    # Compute residual diff = M - sum(w), distribute among non-bound entries.
    diff = M - w.sum()
    if abs(diff) > 1e-8:
        # distribute diff equally among entries not at bounds
        free = (w > 0) & (w < 1)
        if np.any(free):
            w[free] += diff / free.sum()
            w = np.clip(w,0,1)
        else:
            # all at bounds, adjust by smallest error
            w += diff / len(w)
    return w

import time
import numpy as np

# ---------- Helper Functions ----------
def project_to_capped_simplex(v, target_sum, max_val=1.0, tol=1e-12):
    """
    Projection of v onto the set {w | 0<=w_i<=max_val, sum w_i = target_sum}
    using a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - max_val
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, max_val)
        s = w.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, max_val)
    # small fix for numeric drift
    diff = target_sum - w.sum()
    if abs(diff) > 1e-10:
        free = (w > 0.0) & (w < max_val)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, max_val)
    return w

def compute_c5_fft(h):
    """
    Compute C5 = max_k ∫ h(x) (1-h(x+k)) dx for a step function discretised by vector h.
    Uses linear correlation via FFT, multiplies by dx.
    """
    n = h.shape[0]
    dx = 2.0 / n
    q = 1.0 - h
    # linear correlation = convolution of h with reverse(q)
    N = 1 << ((2 * n - 1).bit_length())
    H = np.fft.rfft(h, N)
    Q = np.fft.rfft(q[::-1], N)
    conv = np.fft.irfft(H * Q, N)[: 2 * n - 1]
    return float(conv.max() * dx)

def generate_h_from_b(b, n):
    """
    Symmetric three‑step shape defined by parameter b ∈ [0.5, 1]:
    left/right intervals of length b with value v,
    central interval of length 2-2b with value 1.
    v = 1 - 1/(2*b) guarantees total integral = 1.
    """
    if b <= 0.5:
        b = 0.5
    if b >= 1.0:
        b = 1.0
    v = 1.0 - 1.0 / (2.0 * b)
    d = 2.0 - 2.0 * b  # central plateau length
    dx = 2.0 / n
    # sample at mid‑points
    xs = (np.arange(n) + 0.5) * dx
    h = np.empty(n, dtype=np.float64)
    left = xs < b
    right = xs > 2.0 - b
    middle = ~left & ~right
    h[left] = v
    h[right] = v
    h[middle] = 1.0
    # enforce exact integral (may be off by discretisation)
    target_sum = n / 2.0
    h = project_to_capped_simplex(h, target_sum)
    return h

def random_initial(n, rng):
    """Generate a random feasible h (values in [0,1], sum = n/2)."""
    v = rng.random(n)
    v = v - v.mean() + 0.5   # shift to mean 0.5
    v = np.clip(v, 0.0, 1.0)
    target_sum = n / 2.0
    v = project_to_capped_simplex(v, target_sum)
    return v

# ---------- Main Solver ----------
def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    deadline = start + budget_s - 0.05  # little safety margin
    rng = np.random.default_rng(seed)

    # ----- Determine discretisation size -----
    # Prefer to keep it modest for speed; 200‑300 points give decent accuracy.
    n = 250
    if 'initial_h_values' in globals() and initial_h_values is not None:
        # use the provided construction as a fallback start
        h_start = np.array(initial_h_values, dtype=np.float64)
        n = len(h_start)
    else:
        # start from the three‑step family (b parameter) with a moderate guess
        h_start = generate_h_from_b(0.75, n)

    # ----- Evaluate initial solution -----
    best_h = h_start.copy()
    best_c5 = compute_c5_fft(best_h)

    # ----- Simple grid search over the three‑step family (if different n) -----
    if n >= 200:  # enough points for a meaningful sweep
        for b in np.linspace(0.5, 1.0, 41):
            if time.time() >= deadline:
                break
            h = generate_h_from_b(b, n)
            c5 = compute_c5_fft(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # ----- Hill‑climbing / simulated annealing -----
    cur_h = best_h.copy()
    cur_c5 = best_c5
    T = 0.02                     # initial temperature
    decay = 0.99985              # exponential cooling
    min_T = 5e-5

    while time.time() < deadline:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        hi, hj = cur_h[i], cur_h[j]
        # feasible delta keeping both within [0,1]
        delta_low = max(-hi, hj - 1.0)
        delta_high = min(1.0 - hi, hj)
        if delta_low >= delta_high:
            continue

        # draw a delta; optionally scale by temperature to keep steps moderate
        delta = rng.uniform(delta_low, delta_high)
        # (optionally limit magnitude)
        max_step = 0.2
        delta = np.clip(delta, -max_step, max_step)
        # propose new vector
        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        new_c5 = compute_c5_fft(new_h)
        delta_c = new_c5 - cur_c5

        if delta_c < 0 or rng.random() < np.exp(-delta_c / max(T, 1e-12)):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h.copy()

        # temperature schedule
        T = max(T * decay, min_T)

        # occasional random reset to current best to avoid drift
        if rng.random() < 0.001:
            cur_h = best_h.copy()
            cur_c5 = best_c5

    # final projection to guarantee feasibility (tiny numeric drift possible)
    target_sum = n / 2.0
    best_h = project_to_capped_simplex(best_h, target_sum)

    # recompute final bound (should be unchanged or slightly better)
    final_c5 = compute_c5_fft(best_h)

    return best_h, final_c5, n
