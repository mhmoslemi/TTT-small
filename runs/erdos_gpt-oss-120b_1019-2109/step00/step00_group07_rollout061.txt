def shift_sum(h, k):
    n = len(h)
    s = np.zeros(n)
    if k >= 0:
        s[:n - k] += h[k:]
        s[k:] += h[:n - k]
    else:
        kk = -k
        s[kk:] += h[:n - kk]
        s[:n - kk] += h[kk:]
    return s

def random_perturb(h, magnitude=0.1):

def improve_one_step(h):
    n = len(h)
    p = np.correlate(h, h, mode='full') # shape 2n-1
    min_idx = p.argmin()
    k = min_idx - (n - 1)  # shift value, maybe positive or negative

    # compute s_i = h[i+k] + h[i - k] (with bounds)
    s = shift_sum(h, k)

    # indices sorted
    i_max = np.argmax(s)
    i_min = np.argmin(s)

    # Transfer amount
    max_transfer = min(h[i_min], 1.0 - h[i_max])
    if max_transfer <= 1e-12:
        # cannot transfer, maybe random perturb
        return h, None
    delta = max_transfer * 0.5 # move half
    h_new = h.copy()
    h_new[i_min] -= delta
    h_new[i_max] += delta
    # after move sum remains same, bounds satisfied (by construction)
    return h_new, k

import time
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the capped simplex {0<=x_i<=1, sum(x)=S}
def project_onto_capped_simplex(x, S):
    """Euclidean projection of x onto {0<=y_i<=1, sum(y)=S}."""
    n = x.shape[0]
    # Initial clipping
    y = np.clip(x, 0.0, 1.0)
    cur = y.sum()
    if abs(cur - S) <= 1e-12:
        return y
    # binary search for lambda
    lo = x.min() - 1.0   # makes all x - lo >= 1
    hi = x.max()         # makes all x - hi <= 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(x - lam, 0.0, 1.0)
        s = y.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    return y

# --------------------------------------------------------------
# Evaluation of the Erdős overlap constant for a candidate h
def evaluate(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

# --------------------------------------------------------------
# Shift-sum needed for gradient / hill‑climbing step
def shift_sum(h, k):
    """Return vector s where s[i] = h[i+k] + h[i-k] (out of bounds -> 0)."""
    n = h.shape[0]
    s = np.zeros(n, dtype=h.dtype)
    if k >= 0:
        # forward part
        if k < n:
            s[:n - k] += h[k:]
        # backward part
        if k < n:
            s[k:] += h[:n - k]
    else:
        kk = -k
        if kk < n:
            s[kk:] += h[:n - kk]
            s[:n - kk] += h[kk:]
    return s

# --------------------------------------------------------------
# One hill‑climbing move: improve the worst (largest) overlap
def hill_climb_one_step(h):
    """Perform a single deterministic improvement step."""
    n = h.shape[0]
    # autocorrelation of h (product sums for all shifts)
    prod = np.correlate(h, h, mode='full')
    # shift with minimal product -> maximal overlap
    idx_min = int(np.argmin(prod))
    k = idx_min - (n - 1)          # integer shift
    s = shift_sum(h, k)
    i_max = int(np.argmax(s))
    i_min = int(np.argmin(s))
    # amount we can transfer while staying in [0,1]
    transfer = min(h[i_min], 1.0 - h[i_max])
    if transfer <= 1e-12:
        # fallback: tiny random perturbation between two random indices
        i, j = np.random.choice(n, 2, replace=False)
        delta = min(h[i], 1.0 - h[j], 0.05)
        if delta > 0:
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            return h_new, k
        return h.copy(), k
    delta = transfer * 0.5          # move half of the feasible amount
    h_new = h.copy()
    h_new[i_min] -= delta
    h_new[i_max] += delta
    return h_new, k

# --------------------------------------------------------------
def optimize(initial_h, budget_s):
    """Try to lower C5 using randomized restarts + hill climbing."""
    n = initial_h.shape[0]
    S = n / 2.0
    best_h = project_onto_capped_simplex(initial_h, S)
    best_c = evaluate(best_h)

    start = time.time()
    while time.time() - start < budget_s * 0.94:
        # occasionally restart from a fresh random point
        if np.random.rand() < 0.15:
            rand = np.random.rand(n)
            cur_h = project_onto_capped_simplex(rand, S)
        else:
            cur_h = best_h.copy()
        cur_c = evaluate(cur_h)

        stalls = 0
        for _ in range(8000):
            # one deterministic improvement step
            new_h, _ = hill_climb_one_step(cur_h)
            # re‑project to avoid drift due to floating error
            if abs(new_h.sum() - S) > 1e-8:
                new_h = project_onto_capped_simplex(new_h, S)
            new_c = evaluate(new_h)

            if new_c < cur_c - 1e-9:
                cur_h, cur_c = new_h, new_c
                stalls = 0
                if cur_c < best_c - 1e-9:
                    best_h, best_c = cur_h.copy(), cur_c
                    # early exit if already very good
                    if best_c < 0.3808:
                        return best_h, best_c, n
            else:
                stalls += 1
                if stalls >= 150:
                    break
            if time.time() - start >= budget_s * 0.94:
                break
    return best_h, best_c, n
