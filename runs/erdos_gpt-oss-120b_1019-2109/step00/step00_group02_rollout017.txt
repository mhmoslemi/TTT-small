def compute_c5_fft(h):
    n = len(h)
    # Let's compute linear convolution of h and rev(1-h)
    a = h
    b = (1 - h)[::-1]  # reversed
    size = 1
    while size < n + n:
        size <<= 1
    # compute fft
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    # conv length = size; required length = n + n - 1 = 2n - 1; take first 2n-1 entries
    conv = conv[:2*n - 1]
    return conv.max() * (2.0 / n)

import numpy as np, time, random
def proj_capped_simplex(v, target):
    lo = -1e5
    hi = 1e5
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    conv = np.convolve(h, (1 - h)[::-1])
    return conv.max() * (2.0 / n)

def gradient_max_lag(h, lag):
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        L = lag
        g[:n - L] += 1 - h[L:]
        g[L:] -= h[:n - L]
    else:
        D = -lag
        g[D:] += 1 - h[:n - D]
        g[:n - D] -= h[D:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random

    else:
        n = 200
        h = np.random.rand(n)
        target_sum = n / 2.0
        h = proj_capped_simplex(h, target_sum)

    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Adam parameters
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.2
    t = 0

    start = time.time()
    max_time = budget_s * 0.88  # leave margin for finalization

    # Phase 1: Adam subgradient descent
    while time.time() - start < max_time * 0.7:

import time
import random
import numpy as np

def _proj_capped_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {0≤x≤1, sum(x)=target_sum}."""
    lo, hi = -1e5, 1e5
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a step vector h."""
    n = h.size
    conv = np.convolve(h, (1 - h)[::-1])
    return conv.max() * (2.0 / n)


def _gradient_max_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Sub‑gradient of the active lag (the lag attaining the max C5)."""
    n = h.size
    g = np.zeros_like(h)
    if lag >= 0:
        L = lag
        g[: n - L] += 1 - h[L:]
        g[L:] -= h[: n - L]
    else:
        D = -lag
        g[D:] += 1 - h[: n - D]
        g[: n - D] -= h[D:]
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a low C5 value."""
    np.random.seed(seed)
    random.seed(seed)

    # --------------------------------------------------------------------
    # 1.  Initialise a feasible vector
    # --------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = h.size
        h = np.clip(h, 0.0, 1.0)
        h = _proj_capped_simplex(h, n / 2.0)
    else:
        n = 200
        h = np.random.rand(n)
        h = _proj_capped_simplex(h, n / 2.0)

    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # --------------------------------------------------------------------
    # 2.  Adam based sub‑gradient descent (70 % of the time budget)
    # --------------------------------------------------------------------
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.2
    it = 0
    start = time.time()
    max_time = budget_s * 0.88           # keep some margin for the final stage

    while time.time() - start < max_time * 0.7:
        it += 1
        conv = np.convolve(h, (1 - h)[::-1])
        lag = int(conv.argmax() - (n - 1))
        g = _gradient_max_lag(h, lag)

        # Adam update
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = base_lr / np.sqrt(it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        h_new = h - step
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _proj_capped_simplex(h_new, target_sum)

        c5_new = _c5_value(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

    # --------------------------------------------------------------------
    # 3.  Simple random pairwise mass‑transfer local search (remaining time)
    # --------------------------------------------------------------------
    while time.time() - start < max_time:
        i, j = random.sample(range(n), 2)
        if random.random() < 0.5:

            if max_delta <= 0:

            delta = max_delta * random.random() * 0.5
            cand = h.copy()
            cand[j] -= delta
            cand[i] += delta

    best_h = _proj_capped_simplex(best_h, target_sum)
    best_c5 = _c5_value(best_h)

    return best_h.tolist(), float(best_c5), n
