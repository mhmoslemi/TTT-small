import numpy as np

# --------------------------------------------------------------------
# Helper functions

def compute_c5(h):
    """Return C5 = max correlation * dx for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def worst_shift_and_contrib(h):
    """Return (shift, contributions) for the worst (max) correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # signed integer shift
    if shift >= 0:
        i_start = 0
        i_end = n - shift
        contrib = h[i_start:i_end] * (1.0 - h[i_start + shift:i_end + shift])
    else:
        s = -shift
        i_start = s
        i_end = n
        contrib = h[i_start:i_end] * (1.0 - h[i_start - s:i_end - s])  # i_start-s == 0
    return shift, contrib, idx, corr[idx]


def propose_move(h, shift, contrib):
    """Generate a single feasible move that preserves the integral."""
    n = h.shape[0]
    # overlapping region where contribution is defined
    if shift >= 0:
        i_start, i_end = 0, n - shift
    else:
        s = -shift
        i_start, i_end = s, n
    if i_end <= i_start:
        return None, None

    # pick an index i in the overlapping region
    if np.random.rand() < 0.78:
        # target a large contribution
        rel = int(np.argmax(contrib))
        i = i_start + rel
    else:
        i = np.random.randint(i_start, i_end)

    j = i + shift                     # partner index (always in bounds)

    # two elementary move types
    if np.random.rand() < 0.5:
        # increase h[j] by taking mass from a donor elsewhere
        donors = np.where(h > 1e-12)[0]
        donors = donors[donors != j]
        if donors.size == 0:
            return None, None
        d = np.random.choice(donors)
        max_delta = min(1.0, h[d], 1.0 - h[j])
        if max_delta <= 0:
            return None, None
        delta = max_delta                # full feasible transfer
        h_new = h.copy()
        h_new[d] -= delta
        h_new[j] += delta
    else:
        # decrease h[i] by sending mass to a receiver elsewhere
        receivers = np.where(h < 1.0 - 1e-12)[0]
        receivers = receivers[receivers != i]
        if receivers.size == 0:
            return None, None
        r = np.random.choice(receivers)
        max_delta = min(1.0, h[i], 1.0 - h[r])
        if max_delta <= 0:
            return None, None
        delta = max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[r] += delta

    return h_new, (i, j, delta)


def local_search(start_h, deadline):
    """Hill‑climbing / simulated‑annealing style improvement."""
    h = start_h.copy()
    n = h.shape[0]
    target_sum = n / 2.0
    best_h = h.copy()
    best_score = compute_c5(h)
    cur_score = best_score
    it = 0
    while time.time() < deadline:
        it += 1
        shift, contrib, _, _ = worst_shift_and_contrib(h)
        cand, _ = propose_move(h, shift, contrib)
        if cand is None:
            continue
        cand_score = compute_c5(cand)

        # accept improvement, otherwise occasional uphill move
        if cand_score < cur_score - 1e-12:
            h, cur_score = cand, cand_score
            if cand_score < best_score:
                best_h, best_score = cand.copy(), cand_score
        else:
            # tiny probability to escape local minima
            if np.random.rand() < 0.001:
                h, cur_score = cand, cand_score

        # enforce exact integral (tiny drift may appear)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-9:
            free = np.where((h > 0.0) & (h < 1.0))[0]
            if free.size:
                adjustment = diff / free.size
                h[free] += adjustment
                np.clip(h, 0.0, 1.0, out=h)
            else:
                # fallback: adjust first element within bounds
                h[0] = np.clip(h[0] + diff, 0.0, 1.0)

    return best_h, best_score
