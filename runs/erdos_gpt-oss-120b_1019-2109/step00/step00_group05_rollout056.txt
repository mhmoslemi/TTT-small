def directed_move(h, worst_shift, step):
    n = len(h)
    s = worst_shift
    # g array
    g = np.zeros(n)
    if s >= 0:
        n_s = n - s
        if n_s > 0:
            g[:n_s] += h[s:]
            g[s:] += h[:n_s]
    else:
        s_abs = -s
        n_s = n - s_abs
        if n_s > 0:
            g[s_abs:] += h[:n_s]
            g[:n_s] += h[s_abs:]
    donor_candidates = np.where(h > 1e-12)[0]
    receiver_candidates = np.where(h < 1 - 1e-12)[0]
    if donor_candidates.size == 0 or receiver_candidates.size == 0:
        return None
    donor = donor_candidates[np.argmin(g[donor_candidates])]
    receiver = receiver_candidates[np.argmax(g[receiver_candidates])]
    max_delta = min(h[donor], 1 - h[receiver])
    if max_delta <= 0:
        return None
    delta = min(max_delta, step)
    # optionally random factor
    delta *= np.random.rand()
    if delta <= 1e-12:
        return None
    h_new = h.copy()
    h_new[donor] -= delta
    h_new[receiver] += delta
    return h_new

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, h, mode='full') * dx
    min_O = np.min(corr)
    return float(1 - min_O)

import time, math
import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Maximum overlap integral approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1 - h) over overlapping region
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _repair(h: np.ndarray) -> np.ndarray:
    """Project h onto the feasible polytope: 0≤h≤1, sum(h)=n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # simple iterative redistribution of the residual
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:                         # need to increase some entries
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            inc = diff / idx.size
            inc = np.minimum(inc, 1.0 - h[idx])
            h[idx] += inc
        else:                                 # need to decrease some entries
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            dec = -diff / idx.size
            dec = np.minimum(dec, h[idx])
            h[idx] -= dec
    # final tiny correction on a random feasible coordinate
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        feasible = np.where((h + diff >= 0.0) & (h + diff <= 1.0))[0]
        if feasible.size:
            h[feasible[0]] += diff
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h:[0,2]->[0,1] (sampled uniformly) to minimise
    max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # start vector (use the provided one if present)
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    except NameError:                         # no parent construction supplied
        n = 70
        h0 = np.full(n, 0.5)                  # uniform start (satisfies the integral)

    h0 = _repair(h0)

    # ------------------------------------------------------------------
    # helpers
    # ------------------------------------------------------------------
    def gradient_worst_shift(h):
        """Compute the gradient of the worst‑shift objective."""
        # correlation of h with (1‑h) gives the objective values for every shift
        n = h.shape[0]
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)                 # worst shift (max overlap)
        shift = idx - (n - 1)                 # signed offset

        # mask of indices that belong to the overlapping region for this shift
        mask = np.zeros(n, dtype=bool)
        if shift >= 0:
            mask[: n - shift] = True
        else:
            mask[-shift:] = True

        # neighbor contribution h[i+shift] + h[i-shift] (zero if out of range)
        neigh = np.zeros(n, dtype=float)
        if shift >= 0:
            L = n - shift
            if L:
                neigh[:L] += h[shift:]
                neigh[shift:] += h[:L]
        else:
            s = -shift
            L = n - s
            if L:
                neigh[s:] += h[:L]
                neigh[:L] += h[s:]
        # gradient d/dh_i of C5(s) = 1 - (h_{i+shift}+h_{i-shift}) for overlapped i
        grad = np.where(mask, 1.0 - neigh, 0.0)
        return shift, grad, mask

    # ------------------------------------------------------------------
    # main optimiser ----------------------------------------------------
    # ------------------------------------------------------------------
    cur_h = h0.copy()
    cur_c5 = _c5_of(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # parameters
    step_start = 0.02               # maximal mass moved in a single proposal
    step_min = 0.0005
    temp_start = 0.02               # initial temperature for occasional uphill steps
    it = 0

    while time.time() - start < budget_s * 0.95:
        it += 1

        # identify the worst shift and its gradient
        shift, grad, mask = gradient_worst_shift(cur_h)

        # --------------------------------------------------------------
        # directed donor/receiver move (most promising)
        # --------------------------------------------------------------
        donor_cand = np.where((grad > 1e-12) & (cur_h > 1e-12))[0]
        recv_cand = np.where((grad < -1e-12) & (cur_h < 1.0 - 1e-12))[0]

        made_move = False
        if donor_cand.size and recv_cand.size and rng.random() < 0.85:
            donor = donor_cand[np.argmax(grad[donor_cand])]
            receiver = recv_cand[np.argmin(grad[recv_cand])]
            max_delta = min(cur_h[donor], 1.0 - cur_h[receiver])
            if max_delta > 0:
                step = max(step_min, step_start * (0.995 ** (it // 2000)))
                delta = min(max_delta, step) * rng.random()
                if delta > 1e-12:
                    new_h = cur_h.copy()
                    new_h[donor] -= delta
                    new_h[receiver] += delta
                    new_c5 = _c5_of(new_h)
                    # always accept improvement; otherwise probabilistic uphill move
                    if new_c5 < cur_c5 - 1e-12:
                        cur_h, cur_c5 = new_h, new_c5
                        made_move = True
                    else:
                        # simulated‑annealing acceptance
                        elapsed = time.time() - start
                        T = temp_start * max(0.0, (budget_s - elapsed) / budget_s)
                        if rng.random() < math.exp(-(new_c5 - cur_c5) / max(T, 1e-9)):
                            cur_h, cur_c5 = new_h, new_c5
                            made_move = True

        # --------------------------------------------------------------
        # fallback random swap move
        # --------------------------------------------------------------
        if not made_move and rng.random() < 0.30:
            i, j = rng.choice(n, size=2, replace=False)
            if cur_h[i] > 1e-12 and cur_h[j] < 1.0 - 1e-12:
                donor, receiver = i, j
            elif cur_h[j] > 1e-12 and cur_h[i] < 1.0 - 1e-12:
                donor, receiver = j, i
            else:
                donor = None
            if donor is not None:
                max_delta = min(cur_h[donor], 1.0 - cur_h[receiver])
                if max_delta > 0:
                    delta = max_delta * rng.random() * 0.5
                    new_h = cur_h.copy()
                    new_h[donor] -= delta
                    new_h[receiver] += delta
                    new_c5 = _c5_of(new_h)
                    if new_c5 < cur_c5 - 1e-12:
                        cur_h, cur_c5 = new_h, new_c5

        # keep the overall best
        if cur_c5 < best_c5 - 1e-12:
            best_h, best_c5 = cur_h.copy(), cur_c5

        # occasional mild perturbation to avoid stagnation
        if it % 5000 == 0:
            a, b = rng.choice(n, size=2, replace=False)
            if best_h[a] > 1e-12 and best_h[b] < 1.0 - 1e-12:
                delta = min(best_h[a], 1.0 - best_h[b]) * rng.random() * 0.3
                tmp = best_h.copy()
                tmp[a] -= delta
                tmp[b] += delta
                tmp_c5 = _c5_of(tmp)
                if tmp_c5 < best_c5:
                    best_h, best_c5 = tmp, tmp_c5
                    cur_h, cur_c5 = tmp, tmp_c5

    # ------------------------------------------------------------------
    # final safeguard: enforce feasibility exactly
    # ------------------------------------------------------------------
    best_h = _repair(best_h)
    final_c5 = _c5_of(best_h)

    return best_h, float(final_c5), n
