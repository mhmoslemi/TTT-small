def construct_h(a1,a2,b1,n):
    # compute intervals
    dx = 2.0 / n
    a3 = 1.0 - a1 - a2
    b2 = 1.0 - b1
    # boundaries of intervals
    # start at 0
    start1 = 0.0
    end1 = a1
    start2 = end1 + b1
    end2 = start2 + a2
    start3 = end2 + b2
    end3 = start3 + a3
    # ensure end3 = 2.0 (due to rounding)
    # compute centers
    idx = np.arange(n)
    centers = (idx + 0.5) * dx
    # initial assignment: inside interval -> 1 else 0
    h = np.zeros(n, dtype=np.float64)
    h[(centers >= start1) & (centers < end1)] = 1.0
    h[(centers >= start2) & (centers < end2)] = 1.0
    h[(centers >= start3) & (centers < end3)] = 1.0
    # Adjust count to N = n//2 (n even)
    target = n//2
    current = int(h.sum())
    if current != target:
        # compute distances to nearest intervals for all bins
        # For each center, compute distance to nearest interval interior (0 if inside)
        # intervals list
        intervals = [(start1,end1),(start2,end2),(start3,end3)]
        # compute distance array
        dists = np.full(n, np.inf)
        for (s,e) in intervals:
            # distance for points left of interval: s - x (if x < s)
            # points right: x - e (if x > e)
            # inside: 0
            # general vectorized:
            left = s - centers
            right = centers - e
            # for each center, compute dist:
            # if center < s: dist = s - center
            # else if center > e: dist = center - e
            # else 0
            mask_left = centers < s
            mask_right = centers > e
            dists = np.minimum(dists, np.where(mask_left, left, np.where(mask_right, right, 0.0)))
        # now adjust
        if current > target:
            # need to turn excess ones to zero; pick ones with largest distance
            ones_idx = np.where(h == 1)[0]
            # sort by dist descending
            sort_idx = ones_idx[np.argsort(-dists[ones_idx])]
            to_flip = sort_idx[:current - target]
            h[to_flip] = 0.0
        else:
            # need to turn zeros to one; pick zeros with smallest distance
            zeros_idx = np.where(h == 0)[0]
            sort_idx = zeros_idx[np.argsort(dists[zeros_idx])]
            to_flip = sort_idx[:target - current]
            h[to_flip] = 1.0
    return h

```
def hill_climb(h, current_c5, max_iter, rng, time_limit):
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    start = time.time()
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # swap
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < current_c5 - 1e-12:
            # accept
            current_c5 = new_c5
            # update ones/zeros arrays: swap indices
            # replace i1 in ones with i0, replace i0 in zeros with i1
            ones_idx = np.where(ones == i1)[0][0]
            zeros_idx = np.where(zeros == i0)[0][0]
            ones[ones_idx] = i0
            zeros[zeros_idx] = i1
        else:
            # revert
            h[i1] = 1.0
            h[i0] = 0.0
    return h, current_c5
```

import numpy as np
import time
from scipy.optimize import differential_evolution

def compute_c5(h):
    """Return max overlap C5 for binary step function h (numpy array)."""
    n = h.shape[0]
    # number of ones (should be n//2)
    N = h.sum()
    # aperiodic autocorrelation of h with itself
    auto = np.correlate(h, h, mode='full')
    min_auto = auto.min()
    # max integral of h*(1-h) = (N - min_auto)*dx
    return float((N - min_auto) * (2.0 / n))

def construct_h(a1, a2, b1, n):
    """Create binary step function with three 1-intervals of lengths a1,a2,a3,
    separated by zero gaps of lengths b1,b2. Returns numpy array of length n."""
    # enforce constraints externally
    a3 = 1.0 - a1 - a2
    b2 = 1.0 - b1
    dx = 2.0 / n
    # interval boundaries
    start1 = 0.0
    end1 = a1
    start2 = end1 + b1
    end2 = start2 + a2
    start3 = end2 + b2
    end3 = start3 + a3  # should be 2.0 (up to floating error)
    # sample centre points
    idx = np.arange(n)
    centers = (idx + 0.5) * dx
    h = np.zeros(n, dtype=np.float64)
    h[(centers >= start1) & (centers < end1)] = 1.0
    h[(centers >= start2) & (centers < end2)] = 1.0
    h[(centers >= start3) & (centers < end3)] = 1.0
    # adjust to exact half ones
    target = n // 2
    cur = int(h.sum())
    if cur != target:
        # compute distance of each bin to the nearest 1‑interval
        intervals = [(start1, end1), (start2, end2), (start3, end3)]
        dists = np.full(n, np.inf, dtype=np.float64)
        for s, e in intervals:
            left = s - centers
            right = centers - e
            mask_left = centers < s
            mask_right = centers > e
            dists = np.minimum(dists,
                               np.where(mask_left, left,
                                        np.where(mask_right, right, 0.0)))
        if cur > target:
            # turn off the farthest ones
            one_idx = np.where(h == 1)[0]
            # sort by distance descending
            order = one_idx[np.argsort(-dists[one_idx])]
            h[order[:cur - target]] = 0.0
        else:
            zero_idx = np.where(h == 0)[0]
            order = zero_idx[np.argsort(dists[zero_idx])]
            h[order[:target - cur]] = 1.0
    return h

def adjust_to_half(h):
    """Force exact half‑ones while keeping values binary."""
    n = len(h)
    target = n // 2
    cur = int(h.sum())
    if cur == target:
        return h
    # simple deterministic correction: flip farthest ones / nearest zeros
    # compute distances to the current 1‑intervals (re‑use simple metric)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if cur > target:
        # remove ones with largest index (deterministic)
        h[ones[-(cur - target):]] = 0.0
    else:
        h[zeros[:target - cur]] = 1.0
    return h

def hill_climb(h, current_c5, max_iter, rng, time_limit):
    """Perform greedy swaps to lower C5."""
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    # position maps for O(1) updates
    pos_one = np.empty(n, dtype=np.int32)
    pos_zero = np.empty(n, dtype=np.int32)
    pos_one[ones] = np.arange(len(ones))
    pos_zero[zeros] = np.arange(len(zeros))
    start = time.time()
    for _ in range(max_iter):
        if time.time() - start > time_limit:
            break
        # pick random pair
        i1 = rng.choice(ones)
        i0 = rng.choice(zeros)
        # make the swap
        h[i1] = 0.0
        h[i0] = 1.0
        new_c5 = compute_c5(h)
        if new_c5 < current_c5 - 1e-12:
            # accept
            current_c5 = new_c5
            # update index arrays
            idx1 = pos_one[i1]
            idx0 = pos_zero[i0]
            ones[idx1] = i0
            zeros[idx0] = i1
            pos_one[i0] = idx1
            pos_zero[i1] = idx0
        else:
            # revert
            h[i1] = 1.0
            h[i0] = 0.0
    return h, current_c5

def _objective_factory(n):
    """Return an objective function for DE that expects (a1,a2,b1)."""
    def obj(x):
        a1, a2, b1 = x
        # reject invalid sets early
        if a1 < 0 or a2 < 0 or a1 + a2 > 1 or b1 < 0 or b1 > 1:
            return 1.0  # worst possible C5 is 1
        h = construct_h(a1, a2, b1, n)
        return compute_c5(h)
    return obj

            best_c5 = c0
            best_n = len(init)

    # list of candidate sizes (even)
    candidate_ns = [100, 130, 160, 200, 250, 300]
    # main loop over sizes
    for n in candidate_ns:
        # leave a safety margin for final processing
        if time.time() - t0 > budget_s * 0.85:
            break
        # differential evolution on 3‑interval family
        bounds = [(0.0, 1.0), (0.0, 1.0), (0.0, 1.0)]
        obj = _objective_factory(n)
        # limited iterations for speed
        result = differential_evolution(
            obj,
            bounds,
            seed=seed,
            maxiter=18,
            popsize=9,
            polish=False,
            init='latinhypercube',
            updating='deferred',
            disp=False,
        )
        a1, a2, b1 = result.x
        h = construct_h(a1, a2, b1, n)
        c = compute_c5(h)
        # local greedy refinement
        time_remaining = budget_s - (time.time() - t0)
        max_swaps = min(4000, int(time_remaining * 5))  # generous but limited
        h, c = hill_climb(h, c, max_swaps, rng, time_limit=time_remaining)
        # final adjustment to exact half‑ones
        h = adjust_to_half(h)
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
            best_n = n

        n = 100

        h[: n // 2] = 1.0
        best_h = h
        best_c5 = compute_c5(h)
        best_n = n

    # assure constraints exactly
    best_h = adjust_to_half(best_h)
    best_c5 = compute_c5(best_h)

    # convert to Python list for output (as required)
    return (best_h.tolist(), float(best_c5), int(best_n))
