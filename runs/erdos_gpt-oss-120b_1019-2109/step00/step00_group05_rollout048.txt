import time
import numpy as np

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated by full correlation."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def max_corr_info(h: np.ndarray):
    """Return (max_corr_value, shift) where shift is integer offset of the worst shift."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    max_val = corr[max_idx]
    shift = max_idx - (len(h) - 1)
    return max_val, shift, corr

def propose_gradient_move(h: np.ndarray, shift: int, rng: np.random.Generator):
    """Move a small amount of mass from a positive‑gradient index to a negative‑gradient one."""
    n = h.shape[0]
    idx = np.arange(n)
    prev = idx - shift          # partner i‑s
    nxt = idx + shift           # partner i+s

    valid_prev = (prev >= 0) & (prev < n)
    valid_nxt = (nxt >= 0) & (nxt < n)

    term1 = np.zeros(n)
    term1[valid_prev] = 1.0 - h[prev[valid_prev]]
    term2 = np.zeros(n)
    term2[valid_nxt] = h[nxt[valid_nxt]]

    grad = term1 - term2                # d_i of the worst correlation
    src_cands = np.where((grad > 1e-12) & (h > 1e-12))[0]
    dst_cands = np.where((grad < -1e-12) & (h < 1.0 - 1e-12))[0]

    if src_cands.size == 0 or dst_cands.size == 0:
        return None

    src = src_cands[np.argmax(grad[src_cands])]
    dst = dst_cands[np.argmin(grad[dst_cands])]

    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 0:
        return None
    delta = rng.random() * max_move
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new

def propose_random_move(h: np.ndarray, rng: np.random.Generator):
    """Simple random transfer while staying in [0,1]."""
    n = h.shape[0]
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None
    # random direction
    if rng.random() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_move = min(h[src], 1.0 - h[dst])
    if max_move <= 0:
        return None
    delta = rng.random() * max_move
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[src] -= delta
    h_new[dst] += delta
    return h_new

    # increase resolution by repeating each sample (factor 2 -> 140 points)
    repeat_factor = 2
    h = np.repeat(init, repeat_factor)
    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # ensure the sum is exactly target (tiny floating error fix)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        h *= target_sum / cur_sum
        np.clip(h, 0.0, 1.0, out=h)

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ------------------------------------------------------------------
    # quick random parametric search (three‑level symmetric shape)
    # use at most ~10% of the time budget
    param_time_limit = budget_s * 0.1
    while time.time() - start_time < param_time_limit:
        a = rng.random() * 0.5                     # left zero region length fraction of half‑interval
        b_low = max(a, 0.5)
        b = b_low + rng.random() * (1.0 - b_low)   # ensure b >= 0.5
        if b - a < 1e-6:
            continue
        p = (b - 0.5) / (b - a)                     # height of middle plateau
        if not (0.0 <= p <= 1.0):
            continue
        x = np.arange(n) * dx
        left = np.where(x <= 1.0, x, 2.0 - x)       # symmetric coordinate
        h_candidate = np.where(left < a, 0.0,
                               np.where(left < b, p, 1.0))
        c5_val = compute_c5(h_candidate, dx)
        if c5_val < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_val
            cur_h = best_h.copy()
            cur_c5 = best_c5

    # ------------------------------------------------------------------
    # main stochastic local‑search loop
    max_iters_est = 2_000_000
    T_start = 1.5e-3
    prob_gradient = 0.65
    it = 0

    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:
            break
        it += 1

        # current worst shift
        _, shift, corr = max_corr_info(cur_h)
        cur_c5 = corr.max() * dx

        # propose a move
        if rng.random() < prob_gradient:
            h_new = propose_gradient_move(cur_h, shift, rng)
            if h_new is None:
                continue
        else:
            h_new = propose_random_move(cur_h, rng)
            if h_new is None:
                continue

        new_c5 = compute_c5(h_new, dx)

        # simulated‑annealing style acceptance
        if new_c5 < cur_c5 - 1e-12:
            cur_h = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_h = h_new.copy()
                best_c5 = new_c5
        else:
            temp = T_start * max(0.0, 1.0 - it / max_iters_est)
            if temp > 0.0:
                prob = np.exp(-(new_c5 - cur_c5) / temp)
                if rng.random() < prob:
                    cur_h = h_new
                    cur_c5 = new_c5

        # occasional restart from the best solution with a few random perturbations
        if it % 100_000 == 0:
            cur_h = best_h.copy()
            for _ in range(10):
                tmp = propose_random_move(cur_h, rng)
                if tmp is not None:
                    cur_h = tmp
            cur_c5 = compute_c5(cur_h, dx)

    # final sanity clip (should already be fine)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    # tiny correction to the total mass (if necessary)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        free = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if free.size > 0:
            best_h[free] += diff / free.size
            np.clip(best_h, 0.0, 1.0, out=best_h)

    final_c5 = compute_c5(best_h, dx)
    return best_h, final_c5, n
