import time

def _project_sum(h, target):
    """Adjust h in [0,1] to have sum = target by greedy fill."""
    # simple water-filling algorithm
    cur = h.sum()
    if cur < target:
        need = target - cur
        # increase smallest values first
        idx = np.argsort(h)
        for i in idx:
            room = 1.0 - h[i]
            if room <= 0:
                continue
            add = min(room, need)
            h[i] += add
            need -= add
            if need <= 1e-12:
                break
    elif cur > target:
        excess = cur - target
        idx = np.argsort(-h)
        for i in idx:
            room = h[i]
            if room <= 0:
                continue
            sub = min(room, excess)
            h[i] -= sub
            excess -= sub
            if excess <= 1e-12:
                break
    # Clip tiny errors
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _project_sum(vec, target):
    """Greedy projection of vec into [0,1]^n with sum == target."""
    cur = vec.sum()
    if cur < target:
        need = target - cur

                            if cur_max < best_val - 1e-12:
                                best_val = cur_max
                                best_h = h.copy()
                                best_O = O.copy()
            no_improve += 1
            if no_improve > 20000:
                step = max(step * 0.5, 0.001)
                no_improve = 0

    # ------------------------------------------------------------------
    # final bound and clean‑up
    # ------------------------------------------------------------------
    c5 = best_val * (2.0 / n)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum(best_h, target_sum)   # tiny numerical fix
    return (best_h.tolist(), float(c5), int(n))
