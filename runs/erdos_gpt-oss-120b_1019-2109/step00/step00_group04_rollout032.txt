def propose_move(h, max_delta):
    i = np.random.randint(n)
    j = np.random.randint(n-1)
    # ensure j != i
    while j == i:
        j = np.random.randint(n)
    # Determine max possible increase for i
    inc_limit = 1.0 - h[i]
    dec_limit = h[j]
    # direction: increase i, decrease j
    max_transfer = min(inc_limit, dec_limit, max_delta)
    if max_transfer <= 0:
        return None
    delta = np.random.uniform(0, max_transfer)
    # perform move
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, i, j, delta

    # compute initial

    best_h = h.copy()
    best_score = cur_score

    start = time.time()
    stagn = 0
    while time.time() - start < time_budget:
        # compute correlations and max shift
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = np.max(corr)
        max_shift = np.argmax(corr) - (n - 1)

        # compute mismatches sets

            right_idx = idx_range - t

        # mismatches where left 1, right 0
        mism1 = left_idx[(h[left_idx] == 1) & (h[right_idx] == 0)]

        if mism1.size >0:

            # pick any 1 in overlapping region? just choose random one
            ones = np.where(h == 1)[0]
            i = np.random.choice(ones)

        if mism2.size >0:
            j = np.random.choice(mism2)
        else:
            zeros = np.where(h == 0)[0]
            j = np.random.choice(zeros)

        # Ensure i != j; if they are same (shouldn't) skip.
        if i == j:
            continue

import numpy as np
import random
import time

def _max_corr(arr):
    """Maximum of the correlation array np.correlate(arr,1-arr,'full')."""
    return np.max(np.correlate(arr, 1 - arr, mode="full"))


def _corr_array(arr):
    """Full correlation array used to locate the worst shift."""
    return np.correlate(arr, 1 - arr, mode="full")


def _worst_shift(corr):
    """Return (max_value, shift) where shift is index relative to zero."""
    n = (len(corr) + 1) // 2          # original length of the signal
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)          # zero shift corresponds to centre
    return corr[max_idx], shift


def _mismatch_indices(h, shift):
    """
    For a given shift, return two arrays:
    - indices i where h[i]==1 and the shifted partner is 0
    - indices j where h[j]==0 and the shifted partner is 1
    All indices are expressed in the original array coordinates.
    """
    n = h.shape[0]
    if shift >= 0:
        left = h[: n - shift]
        right = h[shift:]
        mism1 = np.where((left == 1) & (right == 0))[0]               # i in left part
        mism2 = np.where((left == 0) & (right == 1))[0] + shift      # partner in right part
    else:
        s = -shift
        left = h[s:]                # corresponds to the "right side" of the original shift
        right = h[: n - s]          # corresponds to the "left side"
        mism1 = np.where((left == 1) & (right == 0))[0] + s          # i in left part (original index)
        mism2 = np.where((left == 0) & (right == 1))[0]              # j in right part (original index)
    return mism1, mism2


def _random_binary(n, half):
    """Return a binary vector of length n with exactly `half` ones."""
    arr = np.zeros(n, dtype=np.float64)
    arr[:half] = 1.0
    np.random.shuffle(arr)
    return arr


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a low value of C₅.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,) with entries in [0,1]
        c5_bound : float, the achieved bound C₅
        n_points : int, the discretisation size
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Determine problem size (n) and initialise a feasible binary vector.
    # ------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, copy=True)
        n = int(init.shape[0])
    except NameError:
        n = 200                     # fallback size when no parent is provided
    if n % 2 == 1:                 # require an even number of points
        n += 1

    half = n // 2                   # number of ones required
    dx = 2.0 / n

    # initialise from the parent if it matches the size and the constraints
    if 'init' in locals() and init.shape[0] == n:
        # force binary values while preserving the sum
        h = np.rint(init).astype(np.float64)
        cur = int(round(h.sum()))
        diff = half - cur
        if diff > 0:
            zeros = np.where(h == 0)[0]
            chosen = np.random.choice(zeros, diff, replace=False)
            h[chosen] = 1.0
        elif diff < 0:
            ones = np.where(h == 1)[0]
            chosen = np.random.choice(ones, -diff, replace=False)
            h[chosen] = 0.0
    else:
        h = _random_binary(n, half)

    best_h = h.copy()
    best_corr = _max_corr(best_h)

    # ------------------------------------------------------------------
    # Main optimisation loop (simple targeted swap hill‑climbing)
    # ------------------------------------------------------------------
    start = time.time()
    deadline = start + budget_s - 0.5            # safety margin
    stagn = 0                                     # steps without improvement

    cur_corr = best_corr
    cur_h = h.copy()

    while time.time() < deadline:
        # correlation of the current vector and the shift that is worst
        corr_arr = _corr_array(cur_h)
        cur_max, shift = _worst_shift(corr_arr)

        # indices that contribute to this worst shift
        mism1, mism2 = _mismatch_indices(cur_h, shift)

        # pick a 1 from mism1 (if any) otherwise any 1
        if mism1.size:
            i = int(np.random.choice(mism1))
        else:
            i = int(np.random.choice(np.where(cur_h == 1)[0]))

        # pick a 0 from mism2 (if any) otherwise any 0
        if mism2.size:
            j = int(np.random.choice(mism2))
        else:
            j = int(np.random.choice(np.where(cur_h == 0)[0]))

        if i == j:
            continue

        # propose the swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        new_corr = _max_corr(cur_h)

        if new_corr <= cur_corr:
            cur_corr = new_corr
            if new_corr < best_corr:
                best_corr = new_corr
                best_h = cur_h.copy()
                stagn = 0
            else:
                stagn += 1
        else:
            # reject move
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
            stagn += 1

        # occasional random restart if we are stuck
        if stagn > 5000:
            cur_h = _random_binary(n, half)
            cur_corr = _max_corr(cur_h)
            stagn = 0

    c5_bound = float(best_corr * dx)
    return best_h, c5_bound, n
