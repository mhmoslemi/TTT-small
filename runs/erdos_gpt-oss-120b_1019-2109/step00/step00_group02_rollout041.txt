def project_to_sum_and_box(v, target):
    # Clip to [0,1] initially
    w = np.clip(v, 0, 1)
    # iterative water-filling
    for _ in range(10):
        diff = target - w.sum()
        if abs(diff) < 1e-12:
            break
        # capacity for increase (if diff >0)
        if diff > 0:
            # indices where w < 1
            inc_mask = w < 1 - 1e-12
            cap = (1 - w[inc_mask]).sum()
            if cap == 0:
                break
            delta = diff * (1 - w[inc_mask]) / cap
            w[inc_mask] += delta
            w = np.clip(w, 0, 1)
        else:
            # need to decrease
            dec_mask = w > 1e-12
            cap = w[dec_mask].sum()
            if cap == 0:
                break
            delta = diff * w[dec_mask] / cap
            w[dec_mask] += delta
            w = np.clip(w, 0, 1)
    # final clip
    w = np.clip(w, 0, 1)
    return w

import numpy as np
import time

def _project_to_sum_box(v, target):
    """
    Project vector v into the box [0,1]^n while keeping the sum equal to target.
    Uses a simple binary search on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _c5_value(h):
    """Compute the C5 value for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) on the linear (non‑circular) domain
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def _random_feasible(n, rng):
    """Generate a random vector in [0,1]^n with sum exactly n/2."""
    v = rng.random(n)
    return _project_to_sum_box(v, n * 0.5)


def _gradient_step(h, shift, step):
    """Perform a small descent step aimed at reducing the worst‐shift contribution."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        g[: n - shift] = 1.0 - h[shift:]
    else:
        p = -shift
        g[p:] = 1.0 - h[: n - p]
    # move opposite to the gradient (we minimise C5)
    h_new = h - step * g
    # enforce the box constraints and the integral constraint
    return _project_to_sum_box(h_new, n * 0.5)


def _local_optimize(h, max_iter, time_limit):
    """Simple projected sub‑gradient descent with diminishing step size."""
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_val = _c5_value(h)
    start = time.time()
    step = 0.2  # initial step magnitude
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        # compute current worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = corr.argmax()
        shift = idx - (n - 1)
        # take a step
        h_new = _gradient_step(h, shift, step)
        val = _c5_value(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        # accept if improvement, otherwise keep current h
        if val <= _c5_value(h):
            h = h_new
        else:
            # occasionally shrink step if we are stuck
            step *= 0.99
        # occasional step reset to keep progress
        if it % 200 == 0 and step < 1e-4:
            step = 0.2
    return best_h, best_val


def _simulated_annealing(h, target_sum, time_budget):
    """Fast SA that preserves the sum exactly via mass‑transfer moves."""
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_val = _c5_value(h)
    cur_h = h.copy()
    cur_val = best_val
    rng = np.random.default_rng()
    start = time.time()
    # temperature schedule
    T0, Tf = 0.2, 1e-4
    it = 0
    while time.time() - start < time_budget:
        it += 1
        T = T0 * (Tf / T0) ** (it / (time_budget * 200))  # rough scaling
        # propose a mass transfer between two positions
        i, j = rng.choice(n, size=2, replace=False)
        # amount we may move while keeping bounds
        max_move = min(0.1, cur_h[i], 1.0 - cur_h[j])
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        h_new = cur_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        val = _c5_value(h_new)
        if val < cur_val or rng.random() < np.exp(-(val - cur_val) / max(T, 1e-12)):
            cur_h, cur_val = h_new, val
            if val < best_val:
                best_val, best_h = val, h_new.copy()
        # occasional big perturbation to escape plateaus
        if it % 5000 == 0:
            # re‑randomise a fraction of entries
            mask = rng.random(n) < 0.05
            cur_h[mask] = rng.random(mask.sum())
            cur_h = _project_to_sum_box(cur_h, target_sum)
            cur_val = _c5_value(cur_h)
    return best_h, best_val

            overall_best_h,
            max_iter=3000,
            time_limit=budget_s * 0.05,
        )
        overall_best_h, overall_best_val = h_final, final_val

    # safety fallback: uniform half function if nothing better was found
    if overall_best_h is None:
        n = 40
        overall_best_n = n
        overall_best_h = np.full(n, 0.5)
        overall_best_val = _c5_value(overall_best_h)

    # compute final C5 using the official evaluator if available
    try:
        evaluate_erdos_solution
        final_c5 = float(evaluate_erdos_solution(overall_best_h))
    except Exception:
        final_c5 = float(_c5_value(overall_best_h))

    return overall_best_h, final_c5, int(overall_best_n)
