import time
import math
import random
import numpy as np

def _c5_value(h):
    """Compute the Erdős C5 value for a step function h (numpy array)."""
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def _project_capped_simplex(v, total):
    """
    Project vector v onto the set {x | 0 <= x_i <= 1, sum(x) = total}.
    Uses a simple bisection on the Lagrange multiplier λ such that
    x_i = clip(v_i - λ, 0, 1) and sum(x) = total.
    """
    lo, hi = -2.0, 2.0  # bounds that surely contain the solution
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) * 0.5
    return np.clip(v - λ, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with low C₅."""
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.size
    else:
        n = 95                     # a modest size when no parent is supplied
        v = np.random.rand(n)
        h = _project_capped_simplex(v, n * 0.5)

    total_mass = n * 0.5          # ∫h = 1  →  sum(h) = n/2
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ------------------------------------------------------------------
    # simulated‑annealing phase
    # ------------------------------------------------------------------
    start_time = time.time()
    max_time = budget_s * 0.98
    T0, Tf = 0.08, 1e-5        # temperature schedule
    it = 0

    while time.time() - start_time < max_time:
        # temperature decreasing linearly with elapsed proportion
        elapsed = time.time() - start_time
        progress = elapsed / max_time
        T = T0 * (1.0 - progress) + Tf * progress

        # propose a feasible exchange move between two positions
        i, j = random.sample(range(n), 2)
        # delta added to i, subtracted from j
        delta_low = max(-h[i], h[j] - 1.0)   # most negative feasible
        delta_high = min(1.0 - h[i], h[j])   # most positive feasible
        if delta_high <= delta_low:
            continue
        # scale the proposal size with temperature (larger steps early)
        scale = max(0.01, T)                # avoid vanishing steps
        delta = random.uniform(delta_low, delta_high) * scale

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        # evaluate
        new_c5 = _c5_value(h_new)

        # accept if better, otherwise with Metropolis probability
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        if new_c5 < _c5_value(h) or random.random() < math.exp(-(new_c5 - _c5_value(h)) / max(T, 1e-12)):
            h = h_new

        it += 1

    # ------------------------------------------------------------------
    # local gradient‑descent refinement on the best candidate found
    # ------------------------------------------------------------------
    h = best_h.copy()
    for step in range(5000):
        # find the shift producing the current maximal value
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)                # signed shift value

        # construct shifted versions (zero for out‑of‑range indices)
        if shift >= 0:
            h_fwd = np.concatenate([h[shift:], np.zeros(shift)])
            h_bwd = np.concatenate([np.zeros(shift), h[:n - shift]])
        else:
            s = -shift
            h_fwd = np.concatenate([np.zeros(s), h[:n - s]])
            h_bwd = np.concatenate([h[s:], np.zeros(s)])

        # subgradient of the active term (I_ℓ = Σ h_i (1‑h_{i+ℓ}))
        grad = 1.0 - h_fwd - h_bwd

        # simple diminishing step size
        alpha = 0.12 / (1.0 + step * 0.01)

        h_candidate = h - alpha * grad
        # project back onto the feasible set
        h_candidate = _project_capped_simplex(h_candidate, total_mass)

        cand_c5 = _c5_value(h_candidate)
        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = h_candidate.copy()
        # accept if improvement; otherwise keep current h
        if cand_c5 <= _c5_value(h):
            h = h_candidate

        # early exit if we already beat the target record
        if best_c5 < 0.3808:
            break

    return (best_h.tolist(), float(best_c5), n)
