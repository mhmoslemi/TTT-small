import numpy as np

def _eval_c5(h, dx):
    """Return C5 value for step function h (numpy array)."""
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _gradient_for_shift(h, shift):
    """
    Gradient of the overlap functional for a given integer shift.
    Returns vector g where g[i] = 1 - (h[i+shift] + h[i-shift])
    with out-of-bounds values taken as 0.
    """
    n = h.size
    if shift >= 0:
        hp = np.empty_like(h)
        hm = np.empty_like(h)
        hp[: n - shift] = h[shift:]
        hp[n - shift :] = 0.0
        hm[shift:] = h[: n - shift]
        hm[:shift] = 0.0
    else:  # shift < 0
        s = -shift
        hp = np.empty_like(h)
        hm = np.empty_like(h)
        hp[s:] = h[: n - s]
        hp[:s] = 0.0
        hm[: n - s] = h[s:]
        hm[n - s :] = 0.0
    return 1.0 - (hp + hm)
