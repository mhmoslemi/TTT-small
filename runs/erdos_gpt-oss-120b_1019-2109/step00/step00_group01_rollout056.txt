import time, math
import numpy as np

def apply_delta(h: np.ndarray, corr: np.ndarray, i: int, delta: float) -> None:
    """In‑place update of h and its correlation after h[i] += delta.
    corr must be the full correlation of h with (1‑h) (mode='full').
    """
    n = h.shape[0]
    g = 1.0 - h          # complement before the change
    off = n - 1 - i
    # contribution where i is the first factor
    corr[off:off + n] += delta * g
    # contribution where i is the second factor
    start = i + n - 1
    idx = start - np.arange(n)
    corr[idx] -= delta * h
    # self‑term quadratic correction
    corr[n - 1] -= delta * delta
    # finally modify h
    h[i] += delta

def compute_c5(h: np.ndarray) -> float:
    """Exact C₅ value for a discrete h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def target_swap_pair(h: np.ndarray, shift: int, rng: np.random.Generator):
    """Return (i,j) that strictly reduces the mismatches at the given shift,
    or None if no such pair exists."""
    n = h.shape[0]
    if shift >= 0:
        idx = np.arange(0, n - shift)
        i_cand = idx[(h[idx] == 1) & (h[idx + shift] == 0)]
        j_cand = idx[(h[idx] == 0) & (h[idx + shift] == 1)]
    else:
        s = -shift
        idx = np.arange(s, n)
        i_cand = idx[(h[idx] == 1) & (h[idx - s] == 0)]
        j_cand = idx[(h[idx] == 0) & (h[idx - s] == 1)]
    if i_cand.size == 0 or j_cand.size == 0:
        return None
    i = int(rng.choice(i_cand))
    j = int(rng.choice(j_cand))
    return i, j

def random_swap_pair(h: np.ndarray, rng: np.random.Generator):
    ones = np.flatnonzero(h)
    zeros = np.flatnonzero(1.0 - h)
    i = int(rng.choice(ones))
    j = int(rng.choice(zeros))
    return i, j

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ----- initial construction -----------------------------------------
    if 'initial_h_values' in globals() and initial_h_values is not None:
        base = np.asarray(initial_h_values, dtype=np.float64)
        base_len = base.shape[0]
        repeats = 6                            # 6·47 = 282 ≤ 1000
        n = base_len * repeats
        h_cur = np.tile(base, repeats)
    else:
        n = 300
        h_cur = np.zeros(n, dtype=np.float64)
        half = n // 2
        idx = rng.choice(n, size=half, replace=False)
        h_cur[idx] = 1.0

    # ensure feasibility (exact sum = n/2, values in [0,1])
    desired_sum = n / 2.0
    cur_sum = h_cur.sum()
    if abs(cur_sum - desired_sum) > 1e-12:
        # simple redistribution to fix rounding errors
        diff = desired_sum - cur_sum
        for k in range(n):
            if diff == 0:
                break
            take = min(diff, 1.0 - h_cur[k]) if diff > 0 else max(diff, -h_cur[k])
            h_cur[k] += take
            diff -= take

    # ----- correlation and best record -----------------------------------
    corr_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    cur_c5 = np.max(corr_cur) * (2.0 / n)
    best_h = h_cur.copy()
    best_c5 = cur_c5

    # ----- optimisation loop -------------------------------------------
    no_improve = 0          # counter to trigger a restart
    max_no_improve = 20000 # after this many rejects start anew
    while True:
        if time.time() - start_time > budget_s - 0.5:
            break

        # pick shift with maximal overlap
        shift_idx = int(np.argmax(corr_cur))
        shift = shift_idx - (n - 1)

        # decide whether to attempt a guided swap
        if rng.random() < 0.3:
            pair = target_swap_pair(h_cur, shift, rng)
            if pair is None:
                i, j = random_swap_pair(h_cur, rng)
            else:
                i, j = pair
        else:
            i, j = random_swap_pair(h_cur, rng)

        # propose the swap (i:1→0, j:0→1)
        h_new = h_cur.copy()
        corr_new = corr_cur.copy()
        apply_delta(h_new, corr_new, i, -1.0)
        apply_delta(h_new, corr_new, j, +1.0)

        new_c5 = np.max(corr_new) * (2.0 / n)

        # accept only if not worse (greedy hill‑climb)
        if new_c5 <= cur_c5 + 1e-12:
            h_cur = h_new
            corr_cur = corr_new
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_cur.copy()
                # early exit if we beat the target record
                if best_c5 < 0.3808:
                    break
        else:
            no_improve += 1

        # occasional restart to escape possible plateaux
        if no_improve >= max_no_improve:
            # re‑initialise with a fresh random binary vector
            h_cur = np.zeros(n, dtype=np.float64)
            idx = rng.choice(n, size=n // 2, replace=False)
            h_cur[idx] = 1.0
            corr_cur = np.correlate(h_cur, 1.0 - h_cur, mode='full')
            cur_c5 = np.max(corr_cur) * (2.0 / n)
            no_improve = 0

    # ----- final sanity check & return ----------------------------------
    # enforce exact integral one (tiny numerical drift may appear)
    diff = desired_sum - best_h.sum()
    if abs(diff) > 1e-9:
        # adjust a non‑boundary entry
        candidates = np.where((best_h > 0) & (best_h < 1))[0]
        if candidates.size == 0:
            candidates = np.arange(n)
        best_h[candidates[0]] += diff
        best_h[candidates[0]] = min(max(best_h[candidates[0]], 0.0), 1.0)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n
