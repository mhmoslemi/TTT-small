    grad = np.zeros_like(h)
    if shift >= 0:
        # terms: for i where i+shift < N: derivative includes h[i+shift]

import time
import math
import numpy as np

# ---------- helper functions ----------

def _project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project vector v onto the box [0,1] and the hyperplane sum(v)=target_sum."""
    # binary search for a shift λ such that sum(clip(v+λ,0,1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    proj = np.clip(v + lo, 0.0, 1.0)
    # tiny adjustment for numerical drift
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        cnt = free.sum()
        if cnt:
            proj[free] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _random_feasible(N: int, target_sum: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^N with prescribed sum."""
    v = rng.random(N)  # uniform in [0,1)
    return _project_to_constraints(v, target_sum)


def _compute_c5(h: np.ndarray) -> float:
    """Exact discretised C5 for a given h."""
    N = h.shape[0]
    dx = 2.0 / N
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _gradient_of_min_corr(h: np.ndarray):
    """Return gradient of the autocorrelation at the shift giving the minimum value."""
    N = h.shape[0]
    corr = np.correlate(h, h, mode="full")
    idx = int(np.argmin(corr))
    shift = idx - (N - 1)               # shift: negative values allowed
    grad = np.zeros_like(h)

    if shift >= 0:
        L = N - shift
        # contribution from h[i+shift]
        grad[:L] += h[shift:]
        # contribution from h[i-shift] (i>=shift)
        grad[shift:] += h[:L]
    else:
        s = -shift
        L = N - s
        grad[s:] += h[:L]               # i+shift = i-s
        grad[:L] += h[s:]               # i-shift = i+s
    return grad, shift


# ---------- main optimisation routine ----------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] with ∫h = 1 that minimises
        C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    Returns (h_values, c5_bound, n_points).
    """

    # ----- initialise problem size and starting vector -----
    try:
        # use the provided construction if available
        init_vec = np.asarray(initial_h_values, dtype=float)
        N = init_vec.size
    except Exception:
        # fall back to a simple uniform start
        N = 45
        init_vec = np.full(N, 0.5, dtype=float)

    target_sum = N / 2.0                      # because ∫h = 1 and dx = 2/N
    h_start = _project_to_constraints(init_vec, target_sum)

    # keep the best solution found
    best_h = h_start.copy()
    best_c5 = _compute_c5(best_h)

    # create a few starting points (including the given one)
    start_points = [h_start]
    for _ in range(4):
