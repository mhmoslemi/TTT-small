def slsqp_opt(start):
    n = len(start)
    # ensure feasible start
    start = np.clip(start, 0.0, 1.0)
    start = project_to_bounds_sum(start, n/2)
    cons = {'type': 'eq', 'fun': lambda h: np.sum(h) - n/2}
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(lambda h: evaluate_h(h), start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol':1e-12, 'disp':False})
    h_opt = res.x
    # project to feasibility
    h_opt = np.clip(h_opt, 0.0, 1.0)
    h_opt = project_to_bounds_sum(h_opt, n/2)
    val = evaluate_h(h_opt)
    return h_opt, val
```

def local_search(h, best_val, time_limit, start_time):
    n = len(h)
    while time.time() - start_time < time_limit:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # max transferable from j to i:
        max_transfer = min(1 - h[i], h[j])
        if max_transfer <= 0:
            continue
        delta = max_transfer * rng.random() * 0.1 # small up to 10%
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # other indices unchanged, constraints satisfied
        val = evaluate_h(h_new)
        if val < best_val:
            h = h_new
            best_val = val
    return h, best_val

import time
import numpy as np
import scipy.optimize as opt

def _project_to_bounds_sum(v, target):
    """Project v onto the hyperplane sum(v)=target with box [0,1] using bisection."""
    lo, hi = -2.0, 2.0  # wide enough bounds for the Lagrange multiplier
    for _ in range(60):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _evaluate_h(h):
    """Return the discrete C5 bound for a step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _make_shape(a, n):
    """
    Create a symmetric piecewise linear + plateau shape.
    a : width of each zero-to-plateau ramp (0 <= a <= 1)
    n : number of samples on [0,2]
    """
    if a <= 1e-12:
        h = np.full(n, 0.5)  # uniform case (c = 0.5)
        return _project_to_bounds_sum(h, n / 2)

    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx          # centre of each bin
    c = 1.0 / (2.0 - a)                     # plateau height (ensures area = 1)
    h = np.empty(n)

    left = xs < a
    mid = (xs >= a) & (xs <= 2.0 - a)
    right = xs > 2.0 - a

    # left ramp: linear from 0 at x=0 to c at x=a
    h[left] = (xs[left] / a) * c
    # middle plateau
    h[mid] = c
    # right ramp: symmetric down to 0
    h[right] = ((2.0 - xs[right]) / a) * c

    # enforce exact integral constraint (sum = n/2)
    return _project_to_bounds_sum(h, n / 2)


def _slsqp_opt(start):
    """Run a short SLSQP local optimisation starting from `start`."""
    n = start.size
    cons = {"type": "eq", "fun": lambda x: np.sum(x) - n / 2.0}
    bounds = [(0.0, 1.0)] * n
    res = opt.minimize(
        _evaluate_h,
        start,
        method="SLSQP",
        bounds=bounds,
        constraints=cons,
        options={"maxiter": 250, "ftol": 1e-12, "disp": False},
    )
    h_opt = _project_to_bounds_sum(np.clip(res.x, 0.0, 1.0), n / 2.0)
    return h_opt, _evaluate_h(h_opt)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # Determine problem size. Prefer at least 80 samples for flexibility.
    try:
        init = np.array(initial_h_values, copy=True)
        n_init = init.size
    except Exception:
        init = None
        n_init = 0
    n = max(80, n_init)

    # ------------------------------------------------------------------
    # 1) basic candidates: initial (if any) and a parametric family of shapes
    # ------------------------------------------------------------------
    best_h = None
    best_val = np.inf

    if init is not None and init.size == n:
        val = _evaluate_h(init)
        best_h, best_val = init.copy(), val

    # parametric family (ramp width a in [0,1])
    best_a = 0.0
    for a in np.linspace(0.0, 1.0, 61):
        h = _make_shape(a, n)
        val = _evaluate_h(h)
        if val < best_val:
            best_val, best_h, best_a = val, h.copy(), a

    # refine around the best a
    lo_a = max(0.0, best_a - 0.2)
    hi_a = min(1.0, best_a + 0.2)
    for a in np.linspace(lo_a, hi_a, 101):
        h = _make_shape(a, n)
        val = _evaluate_h(h)
        if val < best_val:
            best_val, best_h, best_a = val, h.copy(), a

    # ------------------------------------------------------------------
    # 2) local optimisation from the current best shape
    # ------------------------------------------------------------------
    h_opt, val_opt = _slsqp_opt(best_h)
    if val_opt < best_val:
        best_h, best_val = h_opt, val_opt

    # ------------------------------------------------------------------
    # 3) many random restarts (budget‑aware)
    # ------------------------------------------------------------------
    max_restarts = 10000  # safety cap
    for _ in range(max_restarts):
        if time.time() - start_time > budget_s - 3.0:
            break
        # generate a random feasible point
        rand_vec = rng.random(n)
        start = _project_to_bounds_sum(rand_vec, n / 2.0)
        h_opt, val_opt = _slsqp_opt(start)
        if val_opt < best_val:
            best_h, best_val = h_opt, val_opt

    # ------------------------------------------------------------------
    # 4) simple pairwise transfer local search using remaining time
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 0.5:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small amount from j to i while staying in [0,1]
        max_transfer = min(1.0 - best_h[i], best_h[j])
        if max_transfer <= 0.0:
            continue
        delta = max_transfer * rng.random() * 0.05  # modest step
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        val_new = _evaluate_h(h_new)
        if val_new < best_val:
            best_h, best_val = h_new, val_new

    # final safety projection
    best_h = _project_to_bounds_sum(np.clip(best_h, 0.0, 1.0), n / 2.0)
    best_val = _evaluate_h(best_h)

    return best_h, float(best_val), n
