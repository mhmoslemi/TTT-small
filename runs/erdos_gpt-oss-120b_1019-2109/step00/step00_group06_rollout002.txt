import math

def _evaluate(h):
    """Return (c5, lag_of_max, full_corr_array)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    idx = int(np.argmax(corr))
    c5 = corr[idx] * dx
    lag = idx - (n - 1)
    return c5, lag, corr
