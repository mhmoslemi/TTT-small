def project_onto_box_simplex(v, target):
    lo = -1e3
    hi = 1e3
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(v - mid, 0, 1)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2, 0, 1)

        s = p.sum()

        else:
            hi = mid

            if new_val < best_score:
                best_score, best_h = new_val, h_new.copy()

        if time.time() - t0 > budget_s * 0.97:
            break

    return best_h, best_score, N
