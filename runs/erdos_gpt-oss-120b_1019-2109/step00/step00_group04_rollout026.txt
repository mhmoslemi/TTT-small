def local_search(h, time_budget):
    n = len(h)
    best_h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    cur_cmax = corr.max()
    cur_c5 = cur_cmax * dx
    best_c5 = cur_c5
    start = time.time()
    it = 0
    while time.time() - start < time_budget:
        # compute current correlation for lag with max
        # we need to recompute each iteration after move.
        # But we can store before move.
        # Propose move:
        i, j = np.random.choice(n, 2, replace=False)

        # Choose direction: either move from j to i or from i to j (both equally possible)
        # We'll random choose direction: if random < 0.5 then move from j -> i else i -> j.
        if np.random.rand() < 0.5:
            src, dst = j, i
        else:
            src, dst = i, j

        # compute max delta to keep dst <=1 and src >=0
        max_inc = 1.0 - h[dst]
        max_dec = h[src]
        allowed = min(max_inc, max_dec)
        if allowed <= 0:
            continue
        # step size: maybe small early large later.
        # define base eps as allowed * random factor and a schedule factor.
        # schedule factor = (remaining_time / total_time) * 0.5 + 0.001
        remaining = time_budget - (time.time() - start)
        if remaining < 0:
            remaining = 0
        scale = remaining / time_budget
        eps = allowed * (0.1 * scale + 0.001)  # small step; but need at least min size
        delta = np.random.uniform(0, eps)
        # apply
        h[dst] += delta
        h[src] -= delta
        # compute new correlation
        corr_new = np.correlate(h, 1 - h, mode='full')
        new_cmax = corr_new.max()
        new_c5 = new_cmax * dx
        # acceptance:
        # compute temperature schedule:
        temp = 0.001 + 0.01 * scale  # minimal temperature 0.001, max 0.011
        # if improvement accept
        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            cur_cmax = new_cmax
            # update best
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # accept with probability
            if np.random.rand() < np.exp(-(new_c5 - cur_c5) / temp):
                cur_c5 = new_c5
                cur_cmax = new_cmax
                # Not update best
            else:
                # revert
                h[dst] -= delta
                h[src] += delta
        it += 1
    return best_h, best_c5

import time, math
import numpy as np

def _compute_c5(h):
    """Return C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _worst_lag_info(h):
    """Return (lag, correlation vector) where the overlap is maximal."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)            # np.correlate lag offset
    return lag, corr


def _overlap_indices(n, lag):
    """Indices i (in h) and corresponding i+lag that participate in the correlation."""
    if lag >= 0:
        i = np.arange(0, n - lag, dtype=int)
        j = i + lag
    else:
        i = np.arange(-lag, n, dtype=int)
        j = i + lag
    return i, j


def _local_search(h0, time_budget):
    """Simulated‑annealing style search that keeps the sum constraint."""
    h = h0.copy()
    n = h.shape[0]
    cur_c5 = _compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break

        remaining = time_budget - elapsed
        scale = remaining / time_budget   # slowly goes to 0

        # ----- decide move -------------------------------------------------
        # 70 % of the time try a move that directly attacks the worst lag
        if np.random.rand() < 0.7:
            lag, _ = _worst_lag_info(h)
            i_idx, j_idx = _overlap_indices(n, lag)

            src_mask = (h[i_idx] > 1e-12) & (h[j_idx] < 0.5)
            dst_mask = (h[i_idx] < 1.0 - 1e-12) & (h[j_idx] > 0.5)

            src_cand = i_idx[src_mask]
            dst_cand = i_idx[dst_mask]

            if src_cand.size and dst_cand.size:
                src = np.random.choice(src_cand)
                dst = np.random.choice(dst_cand)
                max_inc = 1.0 - h[dst]
                max_dec = h[src]
                allowed = min(max_inc, max_dec)
                if allowed > 0:
                    eps = allowed * (0.5 * scale + 0.001)
                    delta = np.random.uniform(0.0, eps)
                else:
                    src = dst = None
            else:
                src = dst = None
        else:
            src = dst = None

        # fallback to a completely random transfer
        if src is None or dst is None:
            a, b = np.random.choice(n, 2, replace=False)
            if np.random.rand() < 0.5:

            else:
                src, dst = b, a
            max_inc = 1.0 - h[dst]
            max_dec = h[src]
            allowed = min(max_inc, max_dec)
            if allowed <= 0:

def _prepare_initial(h0, target_n):
    """Upsample/adjust a base vector to length target_n while keeping the integral."""
    n0 = h0.shape[0]
    if target_n == n0:
        h = h0.copy()
    else:
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h = np.interp(x_new, x_old, h0)
        # enforce sum == n/2
        target_sum = target_n * 0.5
        cur_sum = h.sum()
        if cur_sum > 0:
            h *= target_sum / cur_sum
        h = np.clip(h, 0.0, 1.0)
        # tiny fix for the exact sum
        residual = target_sum - h.sum()
        if abs(residual) > 1e-12:
            idx = np.where((h > 0.0) & (h < 1.0))[0]
            if idx.size == 0:
                idx = np.arange(target_n)
            h[idx[0]] = np.clip(h[idx[0]] + residual, 0.0, 1.0)
    # final safety clip
    h = np.clip(h, 0.0, 1.0)
    return h
