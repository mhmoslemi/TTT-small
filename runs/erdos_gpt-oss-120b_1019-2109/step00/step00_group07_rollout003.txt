def compute_corr(h):

def compute_c5(h):

    dx = 2.0 / n

def initialize_h(n):

import time
import random
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h=1 that minimizes
    C5 = max_k ∫ h(x)(1‑h(x+k))dx .
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # parameters
    n = 200                     # number of discretisation points (must be even)
    dx = 2.0 / n
    target_c5 = 0.3808          # we stop as soon as we beat this value

    # pre‑compute FFT length for correlation
    N = 1 << ((2 * n - 1).bit_length())

    def corr_fft(h: np.ndarray) -> np.ndarray:
        """full cross‑correlation of h with (1‑h) using FFT."""
        hf = np.fft.rfft(h, N)
        g_rev = (1.0 - h)[::-1]
        gf = np.fft.rfft(g_rev, N)
        conv = np.fft.irfft(hf * gf, N)[: 2 * n - 1]
        return conv

    def compute_c5(h: np.ndarray):
        """return (C5, full correlation array)."""
        c = corr_fft(h)
        return c.max() * dx, c

    # ------------------------------------------------------------------
    # initialise a binary vector (0/1) of length n with exactly n/2 ones
    try:
        # use the platform‑provided initial_h_values if present
        init = np.array(initial_h_values, dtype=float)
        if init.size != n:
            x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
        order = np.argsort(init)[::-1]
        h0 = np.zeros(n, dtype=float)
        h0[order[: n // 2]] = 1.0
    except NameError:
        # fall back on a random balanced vector
        h0 = np.zeros(n, dtype=float)
        h0[np.random.choice(n, n // 2, replace=False)] = 1.0

    # ------------------------------------------------------------------
    # local optimisation routine
    def local_search(start_h: np.ndarray, time_allowed: float):
        cur_h = start_h.copy()
        cur_c5, _ = compute_c5(cur_h)
        best_h = cur_h.copy()
        best_c5 = cur_c5
        start = time.time()
        it = 0
        while time.time() - start < time_allowed:
            it += 1
            # current full correlation and worst shift
            corr = corr_fft(cur_h)
            max_idx = int(np.argmax(corr))
            d = max_idx - (n - 1)                 # shift (positive → right)

            # indices that currently contribute to the worst loss
            if d >= 0:
                rng = np.arange(n - d)
                A = rng[(cur_h[rng] == 1) & (cur_h[rng + d] == 0)]
                B = rng[(cur_h[rng] == 0) & (cur_h[rng + d] == 1)]
            else:
                dneg = -d
                rng = np.arange(dneg, n)
                A = rng[(cur_h[rng] == 1) & (cur_h[rng - dneg] == 0)]
                B = rng[(cur_h[rng] == 0) & (cur_h[rng - dneg] == 1)]

            improved = False
            # try a targeted swap that strictly reduces the worst loss
            if A.size and B.size:
                i = int(np.random.choice(A))
                j = int(np.random.choice(B))
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
                new_c5, _ = compute_c5(cur_h)
                if new_c5 <= cur_c5:
                    cur_c5 = new_c5
                    improved = True
                else:
                    cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

            # if no targeted move, perform a random 1↔0 swap
            if not improved:
                ones = np.flatnonzero(cur_h == 1)
                zeros = np.flatnonzero(cur_h == 0)
                if ones.size and zeros.size:
                    i = int(np.random.choice(ones))
                    j = int(np.random.choice(zeros))
                    cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
                    new_c5, _ = compute_c5(cur_h)
                    if new_c5 <= cur_c5:
                        cur_c5 = new_c5
                        improved = True
                    else:
                        # occasional uphill move (small temperature)
                        T = 0.01
                        if random.random() < np.exp(-(new_c5 - cur_c5) / T):
                            cur_c5 = new_c5
                        else:
                            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

            # record best solution seen so far
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                if best_c5 <= target_c5:
                    break

            # occasional reset to the best state so far
            if it % 5000 == 0 and not improved:
                cur_h = best_h.copy()
                cur_c5 = best_c5

        return best_h, best_c5

    # ------------------------------------------------------------------
    # main optimisation loop with time budgeting
    start_time = time.time()
    remaining = budget_s - (time.time() - start_time)

    # first run from the provided (or random) start vector
    if remaining > 3.0:
        alloc = remaining * 0.8
        best_h, best_c5 = local_search(h0, alloc)
    else:
        best_h, best_c5 = h0, compute_c5(h0)[0]

    # additional random restarts while time permits
    while time.time() - start_time < budget_s - 0.5:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 2.0:
            break
        h_rand = np.zeros(n, dtype=float)
        h_rand[np.random.choice(n, n // 2, replace=False)] = 1.0
        chunk = min(5.0, remaining * 0.5)
        cand_h, cand_c5 = local_search(h_rand, chunk)
        if cand_c5 < best_c5:
            best_h, best_c5 = cand_h, cand_c5
            if best_c5 <= target_c5:
                break

    final_c5, _ = compute_c5(best_h)

    # return as required: list of floats, bound, and number of points
    return (best_h.tolist(), float(final_c5), n)
