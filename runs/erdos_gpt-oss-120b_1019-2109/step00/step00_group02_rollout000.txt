import time
import numpy as np

def _project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the set {0<=y<=1, sum(y)=target} using a 1‑D
    binary search on the shift λ such that sum(clip(x-λ,0,1))=target."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)

def _c5_bound(h: np.ndarray) -> float:
    """Compute the C5 bound for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def _smooth_step(h: np.ndarray, lr: float, beta: float) -> np.ndarray:
    """One projected gradient step on the smooth approximation of
    max_{k}∫h(1−h∘τ_k) ."""
    n = h.shape[0]
    dx = 2.0 / n
    # autocorrelation A = ∫ h·h∘τ_k
    A = np.correlate(h, h, mode='full') * dx          # length 2n‑1
    # soft‑max weights for the negative autocorrelation
    u = -beta * A
    max_u = np.max(u)
    exp_u = np.exp(u - max_u)
    p = exp_u / exp_u.sum()                           # weights, still length 2n‑1
    # convolution p * h  (centered)
    conv_full = np.convolve(p, h, mode='full')
    conv = conv_full[n - 1:n - 1 + n]                  # length n
    grad = -2.0 * beta * dx * conv                    # ∂L/∂h
    h_new = h - lr * grad
    # enforce constraints
    target = n / 2.0
    return _project_box_sum(h_new, target)

def _subgrad_step(h: np.ndarray, lr: float) -> np.ndarray:
    """One projected sub‑gradient step for the true max‑overlap objective."""
    n = h.shape[0]
    dx = 2.0 / n
    # raw overlap for all integer shifts
    r = np.correlate(h, 1.0 - h, mode='full') * dx
    idx = int(r.argmax())
    s = idx - (n - 1)                     # shift index (negative allowed)

    # compute h_{j-s}+h_{j+s} with zero padding
    if s >= 0:
        left = np.concatenate((np.zeros(s), h[:n - s]))
        right = np.concatenate((h[s:], np.zeros(s)))
    else:
        t = -s
        left = np.concatenate((h[t:], np.zeros(t)))
        right = np.concatenate((np.zeros(t), h[:n - t]))
    grad = -dx * (left + right)           # ∂r_max/∂h
    h_new = h - lr * grad                 # move opposite to gradient of r_max
    target = n / 2.0
    return _project_box_sum(h_new, target)

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a low‑C5 step function."""
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_c5 = np.inf
    best_n = None

    # detect optional parent
    init_vec = globals().get('initial_h_values')
    init_n = init_vec.shape[0] if isinstance(init_vec, np.ndarray) else None

    # candidate sample sizes (all < 1000)
    cand_n = [45, 60, 80, 100, 140, 200]
    if init_n is not None and init_n not in cand_n:
        cand_n = [init_n] + cand_n

    # main stochastic search loop
    while time.time() - start < budget_s * 0.97:
        # pick a problem size
        n = np.random.choice(cand_n)
        target_sum = n / 2.0
        dx = 2.0 / n

        # initialise h
        if init_vec is not None and n == init_n:
            h = np.array(init_vec, copy=True)
            h = _project_box_sum(h, target_sum)
        else:
            h = np.random.rand(n)
            h = _project_box_sum(h, target_sum)

        # hyper‑parameters (tuned on the fly)
        beta = 120.0
        lr_smooth = 0.02
        lr_sub = 0.5

        # inner optimisation: smooth phase
        it_smooth = int(2e3 * (200 / n)) + 500
        for it in range(it_smooth):
            if time.time() - start > budget_s * 0.97:
                break
            h = _smooth_step(h, lr_smooth, beta)
            # occasional evaluation
            if (it & 31) == 0:
                cur = _c5_bound(h)
                if cur < best_c5:
                    best_c5, best_h, best_n = cur, h.copy(), n

        # inner optimisation: sub‑gradient phase (fine tune the true max)
        it_sub = int(1e3 * (200 / n)) + 200
        for it in range(it_sub):
            if time.time() - start > budget_s * 0.97:
                break
            h = _subgrad_step(h, lr_sub)
            if (it & 31) == 0:
                cur = _c5_bound(h)
                if cur < best_c5:
                    best_c5, best_h, best_n = cur, h.copy(), n

        # final evaluation for this run
        cur = _c5_bound(h)
        if cur < best_c5:
            best_c5, best_h, best_n = cur, h.copy(), n

    # fallback in the unlikely case nothing was found
    if best_h is None:
        if init_vec is not None:
            best_h = np.array(init_vec, copy=True)
            best_n = best_h.shape[0]
            best_c5 = _c5_bound(best_h)
        else:
            best_n = 100
            best_h = np.full(best_n, 0.5)
            best_c5 = _c5_bound(best_h)

    # final projection to guarantee feasibility
    best_h = _project_box_sum(best_h, best_n / 2.0)
    best_c5 = _c5_bound(best_h)

    return best_h, float(best_c5), int(best_n)
