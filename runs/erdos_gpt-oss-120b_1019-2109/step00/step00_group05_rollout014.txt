import numpy as np, random, time, math

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def random_balanced(n):
    target = n / 2.0
    num_ones = int(math.floor(target))
    h = np.zeros(n, dtype=np.float64)
    h[:num_ones] = 1.0
    remainder = target - num_ones
    if remainder > 1e-12:
        h[num_ones] = remainder
    np.random.shuffle(h)
    return h

def hill_climb(start_h, max_no_improve, time_deadline):
    h = start_h.copy()
    cur_c5 = compute_c5(h)
    no_improve = 0
    n = len(h)
    while True:
        if time.time() > time_deadline:
            break
        # find zero and one positions (ignore fractional)
        zero_idxs = np.where(np.isclose(h, 0.0))[0]
        one_idxs = np.where(np.isclose(h, 1.0))[0]
        if zero_idxs.size == 0 or one_idxs.size == 0:
            break
        i = np.random.choice(zero_idxs)
        j = np.random.choice(one_idxs)
        # perform swap
        h[i], h[j] = 1.0, 0.0
        new_c5 = compute_c5(h)
        if new_c5 < cur_c5 - 1e-12:
            cur_c5 = new_c5
            no_improve = 0
        else:
            # revert
            h[i], h[j] = 0.0, 1.0
            no_improve += 1
            if no_improve >= max_no_improve:
                break
    return h, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math
    np.random.seed(seed)
    random.seed(seed)

    # Determine n and possibly start from initial_h_values

        n = len(init_arr)
        # Produce a binary version by taking the largest entries
        order = np.argsort(-init_arr)
        binary_start = np.zeros(n, dtype=np.float64)
        binary_start[order[:n//2]] = 1.0
        best_h = binary_start
        best_c5 = compute_c5(best_h)
    else:
        n = 70
        best_h = random_balanced(n)
        best_c5 = compute_c5(best_h)

    target = 0.3808
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # parameters for hill climbing
    max_no_improve = 2000

    # main loop: many restarts until time runs out
    while time.time() < deadline:
        # choose starting point: either random or small perturbation of best
        if random.random() < 0.3:
            start = best_h.copy()
            # apply a few random swaps to diversify
            for _ in range(random.randint(1, 3)):
                zero_idxs = np.where(np.isclose(start, 0.0))[0]
                one_idxs = np.where(np.isclose(start, 1.0))[0]
                if zero_idxs.size == 0 or one_idxs.size == 0:

                i = np.random.choice(zero_idxs)
                j = np.random.choice(one_idxs)
                start[i], start[j] = 1.0, 0.0
        else:
            start = random_balanced(n)

        # hill climb from this start
        h_opt, c5_opt = hill_climb(start, max_no_improve, deadline)

            best_h = h_opt.copy()
            if best_c5 < target:
