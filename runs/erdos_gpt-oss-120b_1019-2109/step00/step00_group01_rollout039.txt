def project_onto_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    # water-filling algorithm as described
    # iterate up to maybe 10 times.
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        diff = target_sum - s
        free = (h > 0) & (h < 1)
        n_free = free.sum()
        if n_free == 0:
            # if diff positive but all at 1 or 0, adjust by saturating
            # distribute diff proportionally to capacity
            # but may not happen.
            break
        delta = diff / n_free
        h[free] += delta
        # clip due to overshoot
        h = np.clip(h, 0.0, 1.0)
    # final clipping ensure sum close to target, adjust any small residual by adding to first free element
    s = h.sum()
    if abs(s - target_sum) > 1e-8:
        # adjust residual by adding to element not at bound or adjusting minimal amount with tolerance
        idx = np.argmax(h < 1)  # first not at 1
        if idx is not None and h[idx] < 1:
            h[idx] += target_sum - s
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time

def _project_onto_feasible(h, target_sum):
    """Clip to [0,1] and adjust to meet the sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        free = (h > 0) & (h < 1)
        nfree = int(free.sum())
        if nfree == 0:
            break
        delta = diff / nfree
        h[free] += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny residual correction
    s = h.sum()
    if abs(s - target_sum) > 1e-9:
        # distribute the rest to the first non‑saturated entry
        idx = np.where(h < 1)[0]
        if idx.size:
            h[idx[0]] += target_sum - s
            h = np.clip(h, 0.0, 1.0)
    return h

def _c5_of(h):
    """Compute the C₅ value for a discretised step function."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / N)

def _gradient_worst_shift(h):
    """Sub‑gradient of the worst shift (the one attaining the max)."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (N - 1)          # signed shift
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if s < N:
            g[: N - s] += (1.0 - h[s:])
            g[s:] -= h[: N - s]
    else:
        s = -shift
        g[s:] += (1.0 - h[: N - s])
        g[: N - s] -= h[s:]
    return g, shift

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    # problem size – a compromise between resolution and speed
    N = 400                     # must be even for the midpoint rule to be exact for p=1
    target_sum = N / 2.0
    dx = 2.0 / N
    x = (np.arange(N) + 0.5) * dx          # midpoint sample points

    best_h = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1.  A quick parametric sweep: h(x)=a*(1-|x-1|^p)  (p≥1, a chosen for area 1)
    # ------------------------------------------------------------------
    p_vals = np.linspace(1.0, 5.0, 81)      # 81 candidates, step≈0.05
    for p in p_vals:
        a = (p + 1.0) / (2.0 * p)           # scaling makes the continuous integral equal 1
        h = a * np.maximum(0.0, 1.0 - np.abs(x - 1.0) ** p)
        # tiny discretisation error – project back to the feasible set
        h = _project_onto_feasible(h, target_sum)
        val = _c5_of(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # ------------------------------------------------------------------
    # 2.  Refine the best candidate with a simple projected sub‑gradient method
    # ------------------------------------------------------------------
    start_time = time.time()
    # simple schedule for step size
    step = 0.06
    decay = 0.9997
    min_step = 0.001
    cur_h = best_h.copy()
    cur_val = best_val

    while time.time() - start_time < budget_s - 5.0:
        # sub‑gradient at the worst shift
        g, _ = _gradient_worst_shift(cur_h)
        # take a step and project back
        new_h = cur_h - step * g
        new_h = _project_onto_feasible(new_h, target_sum)
        new_val = _c5_of(new_h)

        # accept if improvement; otherwise accept with a tiny probability (annealing)
        if new_val < cur_val or np.random.rand() < np.exp((cur_val - new_val) / max(step, 1e-9)):
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_val, best_h = new_val, new_h.copy()

        # decay the step size
        step = max(step * decay, min_step)

        # occasional random shake to escape flat regions
        if np.random.rand() < 0.005:
            i, j = np.random.choice(N, 2, replace=False)
            eps = np.random.uniform(0.0, min(cur_h[i], 1.0 - cur_h[j]))
            cur_h[i] -= eps
            cur_h[j] += eps
            cur_h = _project_onto_feasible(cur_h, target_sum)

    # ------------------------------------------------------------------
    # 3.  Return the best found solution
    # ------------------------------------------------------------------
    return best_h.tolist(), float(best_val), N
