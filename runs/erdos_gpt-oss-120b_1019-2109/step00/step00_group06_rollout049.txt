        target = n / 2.0

        if abs(cur - target) > 1e-12:
            # simple scaling then clipping to fix sum
            h0 *= target / cur
            h0 = np.clip(h0, 0.0, 1.0)
            # final renormalisation if needed
            diff = target - h0.sum()
            if abs(diff) > 1e-12:
                # distribute diff uniformly, then clip again
                h0 += diff / n
                h0 = np.clip(h0, 0.0, 1.0)
