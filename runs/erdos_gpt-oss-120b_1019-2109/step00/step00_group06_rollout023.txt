def random_move(h, step):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    eps = rng.uniform(-step, step)
    hi = h[i] + eps
    hj = h[j] - eps
    if 0 <= hi <= 1 and 0 <= hj <= 1:
        new_h = h.copy()
        new_h[i] = hi
        new_h[j] = hj
        return new_h
    else:
        return None

def propose_move(h, step, rng):
    n = h.shape[0]
    i, j = rng.integers(0,n,size=2)
    while i==j:
        j = rng.integers(0,n)
    # Determine allowable epsilon range
    eps_low = max(-step, -h[i]) # cannot go below 0
    eps_high = min(step, 1 - h[i]) # cannot exceed 1
    # For j: hi_new = h[i] + eps; hj_new = h[j] - eps must be in [0,1]
    eps_low2 = max(eps_low, h[j] - 1) # Since h[j] - eps <= 1 => eps >= h[j] - 1
    eps_high2 = min(eps_high, h[j])   # Since h[j] - eps >= 0 => eps <= h[j]
    lo = max(eps_low, eps_low2)
    hi = min(eps_high, eps_high2)
    if lo >= hi:
        return None
    eps = rng.uniform(lo, hi)
    new_h = h.copy()
    new_h[i] += eps
    new_h[j] -= eps
    return new_h

import time
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] (sampled uniformly)
    that minimizes  C5 = max_k ∫ h(x)(1-h(x+k)) dx.

    Returns:
        (h_values, c5_bound, n_points)
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    try:
        # Use the provided parent construction if it exists
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        h = init.copy()
        target_sum = n / 2.0
        # Enforce the integral constraint exactly (within numerical tolerance)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            # adjust a random element safely, then spread the remainder uniformly
            idx = rng.integers(n)
            adj = np.clip(h[idx] + diff, 0.0, 1.0) - h[idx]
            h[idx] += adj
            diff2 = target_sum - h.sum()
            if abs(diff2) > 1e-12:
                h += diff2 / n
                np.clip(h, 0.0, 1.0, out=h)
    except NameError:
        # No parent; start from a random feasible point
        n = 128
        target_sum = n / 2.0
        h = rng.random(n)
        # Simple projection onto the affine slice ∑h = target_sum
        diff = target_sum - h.sum()
        h += diff / n
        np.clip(h, 0.0, 1.0, out=h)

    # ------------------------------------------------------------------
    # Helper utilities
    # ------------------------------------------------------------------
    def eval_c5(vec: np.ndarray) -> float:
        """Return C5 for a given vector."""
        conv = np.correlate(vec, 1.0 - vec, mode="full")
        return np.max(conv) * (2.0 / vec.shape[0])

    def shift(arr: np.ndarray, offset: int) -> np.ndarray:
        """Zero‑padded shift: out[i] = arr[i+offset] if in range, else 0."""
        n_ = arr.shape[0]
        out = np.zeros_like(arr)
        if offset > 0:
            out[:-offset] = arr[offset:]
        elif offset < 0:
            out[-offset:] = arr[: n_ + offset]
        else:
            out = arr.copy()
        return out

    def propose_move(vec: np.ndarray, step: float):
        """Swap a small amount ε between two coordinates, preserving the sum."""
        n_ = vec.shape[0]
        i, j = rng.integers(0, n_, size=2, endpoint=False)
        while i == j:
            j = rng.integers(0, n_)
        # feasible interval for ε
        lo_i = -vec[i]
        hi_i = 1.0 - vec[i]
        lo_j = vec[j] - 1.0
        hi_j = vec[j]
        lo = max(lo_i, lo_j, -step)
        hi = min(hi_i, hi_j, step)
        if lo >= hi:
            return None
        eps = rng.uniform(lo, hi)
        new_vec = vec.copy()
        new_vec[i] += eps
        new_vec[j] -= eps
        return new_vec

    def gradient_step(vec: np.ndarray, offset: int, step: float):
        """Take a small gradient‑type step for the worst shift."""
        sp = shift(vec, offset)
        sm = shift(vec, -offset)
        d = sp + sm - 1.0                # descent direction before projection
        d -= d.mean()                    # keep ∑d = 0  ⇒ sum unchanged
        new_vec = vec + step * d
        np.clip(new_vec, 0.0, 1.0, out=new_vec)

        # Re‑project onto the hyperplane ∑h = n/2 (tiny correction)
        diff = (n / 2.0) - new_vec.sum()
        if abs(diff) > 1e-12:
            new_vec += diff / n
            np.clip(new_vec, 0.0, 1.0, out=new_vec)
        return new_vec

    # ------------------------------------------------------------------
    # Core optimisation loop (simulated annealing + gradient steps)
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = eval_c5(best_h)

    cur_h = best_h.copy()
    cur_score = best_score

    # schedule parameters
    max_step = 0.3
    min_step = 1e-6
    max_temp = 0.005
    min_temp = 1e-6
    grad_prob = 0.6          # probability to use a gradient move

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s:
            break
        prog = elapsed / budget_s
        step = max_step * (1.0 - prog) + min_step * prog
        temp = max_temp * (1.0 - prog) + min_temp * prog

        # identify the shift that attains the current maximum
        conv = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx = int(np.argmax(conv))
        offset = idx - (n - 1)

        # propose a new candidate
        if rng.random() < grad_prob:
            new_h = gradient_step(cur_h, offset, step)
        else:
            new_h = propose_move(cur_h, step)
            if new_h is None:
                continue

        new_score = eval_c5(new_h)

        # acceptance rule (Metropolis)
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if temp > 0:
                prob = np.exp((cur_score - new_score) / temp)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_h = new_h
            cur_score = new_score
            if new_score < best_score:
                best_h = new_h.copy()
                best_score = new_score

    # ------------------------------------------------------------------
    # Final safety correction (ensure exact integral)
    # ------------------------------------------------------------------
    diff = (n / 2.0) - best_h.sum()
    if abs(diff) > 1e-12:
        best_h += diff / n
        np.clip(best_h, 0.0, 1.0, out=best_h)

    return (best_h, float(best_score), n)
