def project_to_box_simplex(v, target):
    # returns projection onto [0,1] and sum = target.
    # Use bisection on lambda.
    # We look for λ such that sum(clip(v - λ, 0, 1)) = target.
    # We find lower bound lo where sum(clip(v - lo, 0,1)) > target, and hi where sum(...) < target.
    # Actually as λ decreases, v - λ increases, leading to larger clipped values.
    # So we can start lo = -1.0, hi = 1.0 maybe not enough if values large.
    # Use bounds: lo = min(v) - 1.0 (so that v - lo >= 1 for all). hi = max(v) + 1.0 (so that v - hi <= 0 for all).
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    # Bisection:
    for _ in range(60):
        mid = (lo + hi) / 2.0
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def compute_subgradient(h):
    N = h.shape[0]
    conv = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(conv)
    shift = idx - (N - 1)
    g = np.ones(N)
    if shift >= 0:
        g[:N-shift] -= h[shift:]
        g[shift:] -= h[:N-shift]
    else:
        r = -shift
        g[r:] -= h[:N-r]
        g[:N-r] -= h[r:]
    return shift, g

import numpy as np, time, random, math

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)

def project_to_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {[0,1]^n : sum = target} using a simple bisection."""
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def subgradient_max_shift(h: np.ndarray):
    """Return the shift attaining the current max and the subgradient of that max."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)               # shift can be negative
    g = np.ones(n, dtype=float)        # derivative of Σ h_i term
    if shift >= 0:
        g[: n - shift] -= h[shift:]
        g[shift:]      -= h[: n - shift]
    else:
        r = -shift
        g[r:]   -= h[: n - r]
        g[: n - r] -= h[r:]
    return shift, g

def upsample_h(old_h: np.ndarray, n_new: int) -> np.ndarray:
    """Linearly interpolate old_h (defined on [0,2]) to n_new points."""
    n_old = old_h.shape[0]
    xs_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xs_new, xs_old, old_h)

def optimise_one(initial_h: np.ndarray, time_limit: float):
    """Gradient + small random swaps, respecting constraints."""
    n = initial_h.shape[0]
    target = n / 2.0
    h = project_to_box_simplex(initial_h, target)
    best_h = h.copy()
    best_val = compute_c5(h)

    start = time.time()
    max_iters = 20000
    step0 = 0.5

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        shift, g = subgradient_max_shift(h)
        step = step0 * (1.0 - it / max_iters) ** 1.2
        h = project_to_box_simplex(h - step * g, target)

        if (it & 9) == 0:          # check every 10 steps
            cur = compute_c5(h)
            if cur < best_val:
                best_val = cur
                best_h = h.copy()

    # a short random local search, keeping the sum fixed
    for _ in range(2000):
        if time.time() - start > time_limit:
            break
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        if np.random.rand() < 0.5:
            # move mass from i to j
            delta = np.random.rand() * min(h[i], 1.0 - h[j])
            if delta <= 0.0:
                continue
            h2 = h.copy()
            h2[i] -= delta
            h2[j] += delta
        else:
            # move mass from j to i
            delta = np.random.rand() * min(h[j], 1.0 - h[i])
            if delta <= 0.0:
                continue
            h2 = h.copy()
            h2[j] -= delta
            h2[i] += delta

        new_val = compute_c5(h2)
        cur_val = compute_c5(h)
        if new_val < cur_val or np.random.rand() < 0.001:
            h = h2
            if new_val < best_val:
                best_val, best_h = new_val, h.copy()
    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    random.seed(seed)
    np.random.seed(seed)
    t_start = time.time()

    # possible discretisations (even lengths to keep the target integer)
    n_choices = [80, 100, 150, 200, 250, 300, 350, 400, 450, 500, 600]

    # reuse the provided construction if any
    init_global = None
    if "initial_h_values" in globals():
        try:
            init_global = np.asarray(initial_h_values, dtype=float)
        except Exception:
            init_global = None

    best_h, best_val, best_n = None, float("inf"), None

    for n in n_choices:
        if time.time() - t_start > budget_s * 0.92:
            break
        target = n / 2.0

        # build a few diverse starting points
        starts = []

        # 1) up‑sample the supplied construction (if present)
        if init_global is not None:
            if init_global.size != n:
                starts.append(upsample_h(init_global, n))
            else:
                starts.append(init_global.copy())

        # 2) Sturmian word with slope φ (golden ratio)
        phi = (1.0 + 5.0 ** 0.5) / 2.0
        i = np.arange(n)
        starts.append(((i * phi) % 1.0 < 0.5).astype(float))

        # 3) a random feasible vector
        rnd = np.random.rand(n)
        starts.append(project_to_box_simplex(rnd, target))

        # optimise each start while we have time
        for h0 in starts:
            remaining = budget_s - (time.time() - t_start)
            if remaining <= 0.2:
                break
            # give each optimisation a slice of the remaining budget
            slice_time = min(remaining * 0.45, 30.0)  # at most ~30 s per trial
            h_opt, val = optimise_one(h0, slice_time)

            if val < best_val:
                best_val, best_h, best_n = val, h_opt, n

        # early stop if we already beat the target bound
        if best_val <= 0.3808:
            break

    # guarantee feasibility of the final answer
    if best_h is None:
        # fallback: simple half‑interval
        best_n = 100
        best_h = np.full(best_n, 0.5)
    best_h = project_to_box_simplex(best_h, best_n / 2.0)
    best_val = compute_c5(best_h)

    return (best_h, float(best_val), int(best_n))
