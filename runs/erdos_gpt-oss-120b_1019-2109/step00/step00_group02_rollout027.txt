import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def thue_morse_array(n, dtype=np.float64):
    seq = np.empty(n, dtype=np.int8)
    for i in range(n):
        seq[i] = bin(i).count('1') % 2
    return seq.astype(dtype)

def balance_sequence(seq, target_ones, rng):
    current = int(seq.sum())
    diff = current - target_ones
    if diff > 0:
        # too many ones -> turn diff ones to zero
        ones_idx = np.flatnonzero(seq == 1)
        choose = rng.choice(ones_idx, size=diff, replace=False)
        seq[choose] = 0
    elif diff < 0:
        zeros_idx = np.flatnonzero(seq == 0)
        choose = rng.choice(zeros_idx, size=-diff, replace=False)
        seq[choose] = 1
    return seq

def random_binary_vector(n, half, rng):

    idx = rng.choice(n, size=half, replace=False)

def move_h(cur_h, rng, max_delta):
    n = cur_h.shape[0]
    # Choose source and target
    source_idx = np.flatnonzero(cur_h > 1e-12)
    target_idx = np.flatnonzero(cur_h < 1 - 1e-12)
    if source_idx.size == 0 or target_idx.size == 0:
        return cur_h.copy()
    i = rng.choice(source_idx)
    j = rng.choice(target_idx)
    delta_max = min(cur_h[i], 1.0 - cur_h[j], max_delta)
    if delta_max <= 0:
        return cur_h.copy()
    delta = rng.random() * delta_max
    new_h = cur_h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h
```

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a step function represented by h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def random_binary_vector(n: int, ones: int, rng: np.random.Generator) -> np.ndarray:
    """Binary vector of length n with exactly `ones` entries equal to 1."""
    v = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=ones, replace=False)
    v[idx] = 1.0
    return v


def thue_morse_array(n: int, dtype=np.float64) -> np.ndarray:
    """Balanced Thue–Morse sequence (parity of popcount)."""
    seq = np.empty(n, dtype=np.int8)
    for i in range(n):
        seq[i] = bin(i).count("1") & 1
    return seq.astype(dtype)


def balance_sequence(seq: np.ndarray, target_ones: int, rng: np.random.Generator) -> np.ndarray:
    """Flip bits in seq so that exactly target_ones entries are 1."""
    cur = int(seq.sum())
    diff = cur - target_ones
    if diff > 0:
        ones_idx = np.flatnonzero(seq == 1)
        idx = rng.choice(ones_idx, size=diff, replace=False)
        seq[idx] = 0
    elif diff < 0:
        zeros_idx = np.flatnonzero(seq == 0)
        idx = rng.choice(zeros_idx, size=-diff, replace=False)
        seq[idx] = 1
    return seq


def anneal_optimize(
    n: int,
    time_budget: float,
    rng: np.random.Generator,
    global_best: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style optimization for a given length n.
    Returns (best_h, best_value).
    """
    half = n // 2
    start = time.time()
    max_temp = 0.04  # initial temperature
    best_h = None
    best_val = float("inf")

    # Prepare a few diverse starting points
    starters = []
    starters.append(random_binary_vector(n, half, rng))
    tm = thue_morse_array(n, dtype=np.float64)
    tm = balance_sequence(tm, half, rng)
    starters.append(tm)
    block = np.zeros(n, dtype=np.float64)
    block[:half] = 1.0
    starters.append(block)
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        if h0.shape[0] == n:
            starters.append(h0.copy())

    for init_h in starters:
        cur_h = init_h.copy()
        cur_val = compute_c5(cur_h)
        if cur_val < best_val:
            best_val, best_h = cur_val, cur_h.copy()
        no_improve = 0

        while time.time() - start < time_budget:
            # Choose a source (h>0) and a sink (h<1)
            src = np.flatnonzero(cur_h > 1e-12)
            dst = np.flatnonzero(cur_h < 1.0 - 1e-12)
            if src.size == 0 or dst.size == 0:
                break
            i = rng.choice(src)
            j = rng.choice(dst)
            # transfer a random amount, capped to keep bounds
            max_delta = min(cur_h[i], 1.0 - cur_h[j], 0.3)
            if max_delta <= 1e-12:
                continue
            delta = rng.random() * max_delta
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

            new_val = compute_c5(new_h)
            dval = new_val - cur_val
            elapsed = time.time() - start
            T = max_temp * max(0.0, 1.0 - elapsed / time_budget)
            accept = False
            if dval <= 0.0:
                accept = True
            elif T > 1e-12:
                if rng.random() < math.exp(-dval / T):
                    accept = True
            if accept:
                cur_h, cur_val = new_h, new_val
                no_improve = 0
                if new_val < best_val:
                    best_val, best_h = new_val, new_h.copy()
            else:
                no_improve += 1

            # occasional restart to avoid stagnation
            if no_improve >= 2000:
                cur_h = random_binary_vector(n, half, rng)
                cur_val = compute_c5(cur_h)
                no_improve = 0

    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function with a low Erdős C₅ bound."""

    best_val = float("inf")

    remaining = budget_s - (time.time() - start)

            n, t_alloc, rng, best_val

        best_h, best_val, best_n = h, compute_c5(h), n_fallback

    return best_h, best_val, best_n
