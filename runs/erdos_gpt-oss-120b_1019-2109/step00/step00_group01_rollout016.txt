import time
import numpy as np

def _project_sum(h, target, max_iter=10):
    """Project h onto [0,1] and enforce sum == target (up to tiny tolerance)."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(max_iter):
        cur = h.sum()
        diff = target - cur
        if abs(diff) < 1e-10:
            break
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
    # final tiny correction by distributing to non‑saturated entries
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-9:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if free.any():
            inc = diff / free.sum()
            h[free] += inc
            h = np.clip(h, 0.0, 1.0)
    return h

def _c5_value(h):
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx

def _propose_swap(h, rng):
    n = h.size
    i, j = rng.choice(n, 2, replace=False)
    new_h = h.copy()
    new_h[i], new_h[j] = new_h[j], new_h[i]
    return new_h

def _propose_transfer(h, max_frac, rng):
    """Move a small amount from one entry to another, preserving the total sum."""
    n = h.size
    # pick source with room to decrease and target with room to increase
    for _ in range(10):
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if h[i] > 1e-12 and h[j] < 1.0 - 1e-12:
            break
    else:
        return h.copy()
    max_amount = min(h[i], 1.0 - h[j])
    delta = max_frac * max_amount * rng.random()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    init = globals().get('initial_h_values', None)
    if init is not None:
        h = np.asarray(init, dtype=float).copy()
    else:
        # fall back: a simple Legendre‑type binary construction (p = 101 -> n = 100)
        p = 101
        n = p - 1
        residues = { (i * i) % p for i in range(1, (p - 1) // 2 + 1) }
        h = np.zeros(n, dtype=float)
        for idx in range(n):
            if (idx + 1) % p in residues:
                h[idx] = 1.0
        # if sum deviates, project
        h = _project_sum(h, n / 2)
    n = h.size
    target_sum = n / 2.0
    h = _project_sum(h, target_sum)

    cur_h = h.copy()
    cur_val = _c5_value(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    # ------------------------------------------------------------------
    # annealing / greedy loop
    # ------------------------------------------------------------------
    max_iters = 300000
    # schedule parameters
    max_frac_start = 0.3   # max fraction of transferable amount at start
    max_frac_end = 0.005   # final max fraction
    # small probability to restart from best (helps escape plateaus)
