import numpy as np, time

def compute_c5(h):
    # evaluate using np.correlate with dx scaling
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(c)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # ensure constraints:
        n = len(h)
        # clip
        h = np.clip(h, 0.0, 1.0)
        # adjust sum to n/2
        cur = h.sum()
        target = n/2.0
        if not np.isclose(cur, target):
            # rescale vs shift to maintain bounds
            # simplest: add constant to match sum and clip
            delta = target - cur
            h += delta / n
            # clip again
            h = np.clip(h, 0.0, 1.0)
            # adjust sum again by scaling factor if needed
            cur = h.sum()
            if not np.isclose(cur, target):
                # scale towards target keeping within [0,1]
                # solve scaling factor a: sum(a*(h - 0.5) + 0.5) = target -> a = (target - n*0.5)/(sum(h-0.5))
                # Only if denominator non-zero
                denom = (h - 0.5).sum()
                if abs(denom) > 1e-9:
                    a = (target - n*0.5) / denom
                    h = (h - 0.5) * a + 0.5
                    h = np.clip(h, 0.0, 1.0)
        c5 = compute_c5(h)
        return h, float(c5), n
    # fallback: generate simple pattern
    n = 80
    h = np.zeros(n)
    half = n//2
    h[:half] = 1.0
    h = np.random.permutation(h)  # random arrangement
    c5 = compute_c5(h)
    return h, float(c5), n

import time
import numpy as np

def _max_overlap(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx using discrete convolution."""
    n = h.size
    dx = 2.0 / n
    # linear convolution (non‑circular) as used by the grader
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def _propose_move(h: np.ndarray, rng: np.random.Generator) -> tuple:
    """
    Propose a feasible move that keeps the sum of h unchanged.
    Returns a new array (copy) and the indices that were altered.
    """
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None
    # we will add delta to i and subtract the same delta from j
    # delta must keep both entries inside [0,1]
    max_inc = min(1.0 - h[i], h[j])          # positive delta allowed
    max_dec = min(h[i], 1.0 - h[j])          # negative delta allowed
    if max_inc == 0.0 and max_dec == 0.0:
        return None
    delta = rng.uniform(-max_dec, max_inc)
    if abs(delta) < 1e-12:
        return None
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new, (i, j)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points).

    - h_values : np.ndarray of shape (n_points,) with values in [0,1]
    - c5_bound : float, the computed max overlap
    - n_points : int, length of the discretisation
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1. Initialise from the provided parent (if any)
    # ------------------------------------------------------------------
    try:
        # initial_h_values is supplied by the grader when a parent exists
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:          # NameError or any other problem -> fallback
        # simple balanced binary start: 50% ones, 50% zeros, shuffled
        n = 80
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        rng.shuffle(h0)

    n = h0.size
    # ensure feasibility (clip and fix sum if needed)
    h0 = np.clip(h0, 0.0, 1.0)
    target_sum = n / 2.0
    cur_sum = h0.sum()
    if not np.isclose(cur_sum, target_sum):
        # simple projection onto the hyperplane sum = target_sum
        # shift all entries by the same amount and clip again
        shift = (target_sum - cur_sum) / n
        h0 += shift
        h0 = np.clip(h0, 0.0, 1.0)
        # if still off, scale around 0.5
        cur_sum = h0.sum()
        if not np.isclose(cur_sum, target_sum):
            diff = target_sum - cur_sum
            # scale the deviation from 0.5
            dev = h0 - 0.5
            denom = dev.sum()
            if abs(denom) > 1e-12:
                a = diff / denom
                h0 = (dev * a) + 0.5
                h0 = np.clip(h0, 0.0, 1.0)

    # ------------------------------------------------------------------
    # 2. Local stochastic optimisation (hill‑climbing / simple SA)
    # ------------------------------------------------------------------
    best_h = h0
    best_val = _max_overlap(best_h)

    # parameters for the search
    max_iters = 250000                     # generous upper bound
    temperature_start = 5e-3               # for occasional uphill moves
    temperature_end = 1e-6
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:      # leave a little slack
            break
        if it >= max_iters:
            break

        # linear cooling schedule
        t = temperature_start + (temperature_end - temperature_start) * (it / max_iters)

        move = _propose_move(best_h, rng)
        if move is None:
            it += 1
            continue
        h_new, _ = move
        new_val = _max_overlap(h_new)

        # accept if better, otherwise accept with Metropolis probability
        accept = False
        if new_val < best_val:
            accept = True
        else:
            # uphill move probability
            diff = new_val - best_val
            if np.exp(-diff / max(t, 1e-12)) > rng.random():
                accept = True

        if accept:
            best_h = h_new
            best_val = new_val

        it += 1

    # ------------------------------------------------------------------
    # 3. Final check / repair (should already be feasible)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    if not np.isclose(cur_sum, target_sum):
        # tiny correction by adding a constant and re‑clipping
        correction = (target_sum - cur_sum) / n
        best_h += correction
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _max_overlap(best_h)

    return best_h, float(final_c5), n
