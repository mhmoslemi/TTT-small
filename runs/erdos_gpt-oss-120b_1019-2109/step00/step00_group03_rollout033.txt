        ones_idx.remove(i_one)

def get_mismatch_sets(h, shift):
    n = len(h)
    if shift >= 0:
        h_i = h[shift:]          # h[i] for i >= shift
        h_prev = h[:n-shift]     # h[i - shift]

        # mismatched ones A: h_i==1 & h_prev==0

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    # set n
    n = 84
    # ensures n even
    half = n // 2
    # initial h
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values)
        if init.shape[0] != n: # if length mismatched, use default random
            # ...
        else:
            # ensure binary
            h = (init >= 0.5).astype(np.int8)
            # adjust sum to half if needed
            sum_h = int(h.sum())
            diff = half - sum_h
            if diff > 0:
                zeros = np.where(h == 0)[0]
                idx = rng.choice(zeros, size=diff, replace=False)
                h[idx] = 1
            elif diff < 0:
                ones = np.where(h == 1)[0]
                idx = rng.choice(ones, size=-diff, replace=False)
                h[idx] = 0
    else:
        h = np.zeros(n, dtype=np.int8)
        ones_idx = rng.choice(n, size=half, replace=False)
        h[ones_idx] = 1

    def compute_cost(vec):
        c = np.correlate(vec, 1 - vec, mode='full')
        return int(c.max())

    best_h = h.copy()
    best_cost = compute_cost(best_h)
    current_h = best_h.copy()
    current_cost = best_cost

    # maintain list of positions
    ones = np.where(current_h == 1)[0].tolist()
    zeros = np.where(current_h == 0)[0].tolist()

    start_time = time.time()
    target_bound = 0.38085
    dx = 2.0 / n
    target_int = int(np.floor(target_bound / dx + 1e-12))

    # Simulated annealing parameters
    # Temperature schedule linear from 1.0 to 0.001
    max_iters = 10_000_000  # high but will break on time

    for iter_idx in range(max_iters):
        if time.time() - start_time > budget_s * 0.95:
            break
        # compute temperature based on progress
        t_frac = (time.time() - start_time) / (budget_s * 0.95)
        temp = max(0.001, (1.0 - t_frac))

        # random swap
        i_one_idx = rng.integers(len(ones))
        i_zero_idx = rng.integers(len(zeros))
        i_one = ones[i_one_idx]
        i_zero = zeros[i_zero_idx]

        # swap
        current_h[i_one] = 0
        current_h[i_zero] = 1

        # update lists
        ones[i_one_idx] = i_zero
        zeros[i_zero_idx] = i_one

        new_cost = compute_cost(current_h)
        delta = new_cost - current_cost
        accept = False
        if delta <= 0:
            accept = True
        else:
            # probabilistic acceptance
            if rng.random() < np.exp(-delta / temp):
                accept = True

        if accept:
            current_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = current_h.copy()
                # early exit if meets target
                if best_cost <= target_int:
                    break
        else:
            # revert swap
            current_h[i_one] = 1
            current_h[i_zero] = 0

import time
import numpy as np

def _cost(vec):
    """Return the maximal overlap count (integer) for a binary vector."""
    # correlation of h with its complement (1 - h)
    corr = np.correlate(vec, 1 - vec, mode="full")
    return int(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
