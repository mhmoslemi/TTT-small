def compute_subgradient(h, active_shifts):
    n = len(h)

        k = idx - (n - 1) # shift

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def project_capped_simplex(u, target):
    # projection onto [0,1] and sum == target
    # using bisection on lambda
    low = -np.max(u) - 1.0
    high = 1.0 - np.min(u) + 1.0
    for _ in range(60):
        lam = (low + high) * 0.5
        x = np.clip(u + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            high = lam
        else:
            low = lam
    return np.clip(u + lam, 0.0, 1.0)

def subgradient(h, active_idxs):
    n = h.shape[0]
    g = np.zeros_like(h)
    for idx in active_idxs:
        k = idx - (n - 1)
        if k >= 0:
            end = n - k
            # Add (1 - h[i+k]) for i=0..end-1
            g[:end] += (1 - h[k:])
