def initial_triangle(n):

def project_to_sum_and_box(v, S):
    # binary search for lambda such that sum(clipped(v+lambda)) = S
    lo = -np.max(v) - 1.0   # low enough: all v+lo <= 0 -> sum 0
    hi = 1.0 - np.min(v) + 1.0  # high enough: all v+hi >= 1 -> sum n
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        total = w.sum()
        if total < S:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # adjust small rounding
    diff = S - w.sum()
    if abs(diff) > 1e-9:
        # adjust a random index not at bound
        idxs = np.where((w > 0) & (w < 1))[0]
        if idxs.size > 0:
            idx = idxs[0]
            w[idx] += diff
            w[idx] = np.clip(w[idx], 0.0, 1.0)
        else:
            # fallback: modify first index
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
    return w

def triangle_shape(n):
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx  # midpoints.
    raw = 1 - np.abs(xs - 1.0)
    raw = np.clip(raw, 0.0, None)
    # project to sum
    S = n/2.0
    return project_to_sum_and_box(raw, S)

def sine_squared_shape(n):
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    raw = np.sin(np.pi * xs / 2.0)**2
    raw = np.clip(raw, 0.0, 1.0)
    S = n/2.0
    return project_to_sum_and_box(raw, S)

def uniform_shape(n):
    return np.full(n, 0.5)

import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_to_sum_and_box(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while forcing sum(v)=target_sum."""
    lo = -np.max(v) - 1.0      # guarantees all entries -> 0 after clipping
    hi = 1.0 - np.min(v) + 1.0  # guarantees all entries -> 1 after clipping
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for rounding errors
    diff = target_sum - w.sum()
    if abs(diff) > 1e-9:
        free = np.where((w > 0.0) & (w < 1.0))[0]
        if free.size:
            w[free[0]] = np.clip(w[free[0]] + diff, 0.0, 1.0)
        else:
            # fallback – adjust a random entry
            idx = np.random.randint(w.size)
            w[idx] = np.clip(w[idx] + diff, 0.0, 1.0)
    return w


def _initial_binary(n: int) -> np.ndarray:
    """Binary vector with exactly n/2 ones (rounded if n odd)."""
    target = n / 2.0
    h = np.zeros(n, dtype=float)
    ones = int(round(target))
    idx = np.random.choice(n, ones, replace=False)
    h[idx] = 1.0
    # if n is odd the sum may be off by 0.5 – project to exact sum
    return _project_to_sum_and_box(h, target)


def _local_search(start_h: np.ndarray, time_limit: float, target: float = 0.3808) -> tuple:
    """Hill‑climbing / simulated‑annealing style improvement."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    best_h = start_h.copy()
    best_c = _compute_c5(best_h)

    cur_h = start_h.copy()
    cur_c = best_c

    start = time.time()
    stagnant = 0
    while time.time() - start < time_limit:
        # correlation and worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)          # shift ∈ [-(n-1), n-1]
        # contributions for that shift
        if shift >= 0:
            rng = np.arange(n - shift)
            contrib = cur_h[rng] * (1.0 - cur_h[rng + shift])
        else:
            s = -shift
            rng = np.arange(s, n)
            contrib = cur_h[rng] * (1.0 - cur_h[rng - s])

        # pick the index that contributes most to the worst overlap
        local_max = int(np.argmax(contrib))
        i = rng[local_max]
        j = i + shift                     # the partner in the shifted complement

        # try to push mass from i towards j (i -> j)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta > 1e-12:
            delta = max_delta * np.random.rand()
            cur_h[i] -= delta
            cur_h[j] += delta
            new_c = _compute_c5(cur_h)
            if new_c <= cur_c:
                cur_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = cur_h.copy()
                    stagnant = 0
                else:
                    stagnant += 1
            else:
                # revert
                cur_h[i] += delta
                cur_h[j] -= delta
                stagnant += 1
        else:
            stagnant += 1

        # occasional random perturbation if stuck
        if stagnant > 5000:
            cur_h = _initial_binary(n)
            cur_c = _compute_c5(cur_h)
            stagnant = 0

        # early exit if we already beat the target bound
        if best_c <= target:
            break

    return best_h, best_c


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
