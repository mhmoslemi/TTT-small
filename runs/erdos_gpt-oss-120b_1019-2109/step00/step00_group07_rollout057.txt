def propose(h, sigma):
    n = len(h)
    i, j = rng.choice(n, size=2, replace=False)
    # direction: we may increase i, decrease j, or opposite.
    # random direction +/-1
    if rng.random() < 0.5:
        # increase i, decrease j
        max_inc = min(1 - h[i], h[j])
        if max_inc <= 0:
            return None
        delta = rng.uniform(0, max_inc)
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
    else:
        # increase j, decrease i
        max_inc = min(1 - h[j], h[i])
        if max_inc <= 0:
            return None
        delta = rng.uniform(0, max_inc)
        h_new = h.copy()
        h_new[j] += delta
        h_new[i] -= delta
    return h_new
```

def project(v, target):
    n = len(v)
    # Clip initially? Actually algorithm works with sorted ascending.
    # Use approach: shift = (target - sum(v)) / n; x = v + shift; then clip, then adjust iteratively.
    x = v.copy()
    shift = (target - np.sum(x)) / n
    x += shift
    # iterative clipping:
    iteration = 0
    while True:
        low_idx = x < 0
        high_idx = x > 1
        if not low_idx.any() and not high_idx.any():
            break
        # For low values (<0), set to 0 and redistribute excess shift.
        excess = -np.sum(x[low_idx])  # negative values magnitude
        if excess > 0:
            x[low_idx] = 0
            # Redistribute excess equally among remaining free variables (not at bounds)
            free = (~low_idx) & (~high_idx)
            if free.any():
                x[free] += excess / free.sum()
            else:
                # all at bounds, break
                break
        # For high values (>1)
        excess = np.sum(x[high_idx]) - 1 * high_idx.sum()
        if excess > 0:
            x[high_idx] = 1
            free = (~low_idx) & (~high_idx)
            if free.any():
                x[free] -= excess / free.sum()
            else:
                break
        iteration += 1
        if iteration > 10:
            break
    # final clip to ensure within [0,1]
    x = np.clip(x, 0, 1)
    # final adjust for sum error due to rounding
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff uniformly to non-bound values
        free = (x > 0) & (x < 1)
        if free.any():
            x[free] += diff / free.sum()
        else:
            # No free var: adjust all by diff/n? Not needed.
            x += diff / n
            x = np.clip(x, 0, 1)

def descent_step(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    s = max_idx - (n - 1)
    # compute derivative
    d = np.zeros(n)
    if s >= 0:
        # i from 0 to n-1-s
        i_range = np.arange(n - s)
        d[i_range] = 1 - h[i_range + s]
    else:
        # s negative, i from -s to n-1
        i_range = np.arange(-s, n)
        d[i_range] = 1 - h[i_range + s]
    # Determine candidate indices
    dec_mask = h > 1e-9
    inc_mask = h < 1 - 1e-9
    if not np.any(dec_mask) or not np.any(inc_mask):
        return h, max_val, False
    # Choose dec index with max d
    dec_candidates = np.where(dec_mask)[0]
    inc_candidates = np.where(inc_mask)[0]
    # Get top few
    top_k = 5
    dec_vals = d[dec_candidates]
    inc_vals = d[inc_candidates]
    # top dec: largest d
    if dec_vals.size > top_k:
        top_dec_idx = np.argpartition(-dec_vals, top_k-1)[:top_k]
    else:
        top_dec_idx = np.arange(dec_vals.size)
    # bottom inc: smallest d
    if inc_vals.size > top_k:
        top_inc_idx = np.argpartition(inc_vals, top_k-1)[:top_k]

        top_inc_idx = np.arange(inc_vals.size)

def proj_linf_simple(v, target):
    # Use iterative algorithm
    x = v.copy()
    # Clip to bounds initially
    x = np.clip(x, 0, 1)
    # Adjust sum
    while True:
        cur = np.sum(x)
        diff = target - cur
        if abs(diff) < 1e-10:
            break
        # Indices not at bounds for upward movement if diff >0, else downward movement.
        if diff > 0:

        else:
            free = x > 1e-12
        if not np.any(free):
            break
        delta = diff / np.sum(free)
        # update free entries
        x[free] += delta
        # clip again
        x = np.clip(x, 0, 1)
    return x

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def _project_to_box_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set {x | 0 <= x_i <= 1, sum x_i = target}.
    Simple bisection on the Lagrange multiplier.
    """
    lo, hi = -10.0, 10.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # final tiny fix for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def _max_corr_and_shift(h: np.ndarray):
    """Return (max_corr, shift_index) where shift = idx-(n-1)."""
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx]), idx - (h.size - 1), corr


def _descent_step(h: np.ndarray):
    """
    Perform a greedy move that tries to reduce the current maximal correlation.
    Returns (new_h, new_max_corr, improved_flag).
    """
    n = h.size
    cur_max, shift, _ = _max_corr_and_shift(h)

    # derivative of the correlation at the critical shift
    d = np.zeros(n, dtype=float)
    if shift >= 0:
        # i in [0, n-shift)
        d[: n - shift] = 1.0 - h[shift:]
    else:
        # shift < 0, i in [-shift, n)
        d[-shift :] = 1.0 - h[: n + shift]

    # indices that can be decreased / increased
    dec_mask = h > 1e-12
    inc_mask = h < 1.0 - 1e-12
    if not dec_mask.any() or not inc_mask.any():
        return h, cur_max, False

    dec_candidates = np.where(dec_mask)[0]
    inc_candidates = np.where(inc_mask)[0]

    # take a handful of extreme candidates
    top_k = 8
    # largest derivative among decrease candidates
    dec_vals = d[dec_candidates]
    if dec_vals.size > top_k:
        best_dec_idx = np.argpartition(-dec_vals, top_k - 1)[:top_k]
    else:
        best_dec_idx = np.arange(dec_vals.size)
    dec_idx = dec_candidates[best_dec_idx]

    # smallest derivative among increase candidates
    inc_vals = d[inc_candidates]
    if inc_vals.size > top_k:
        best_inc_idx = np.argpartition(inc_vals, top_k - 1)[:top_k]
    else:
        best_inc_idx = np.arange(inc_vals.size)
    inc_idx = inc_candidates[best_inc_idx]

    # try all pairings among the reduced sets
    best_improve = -1.0
    best_pair = None
    for i in dec_idx:
        for j in inc_idx:
            if i == j:
                continue
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 1e-12:
                continue
            # try the full transfer (most aggressive)
            delta = delta_max
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            new_max = np.max(np.correlate(h_new, 1 - h_new, mode="full"))
            improve = cur_max - new_max
            if improve > best_improve:
                best_improve = improve
                best_pair = (i, j, delta, new_max)

    if best_pair is not None and best_improve > 1e-12:
        i, j, delta, new_max = best_pair
        h[i] -= delta
        h[j] += delta
        return h, new_max, True

    # no improving pair found
    return h, cur_max, False


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random point in [0,1]^n with sum = n/2."""
    v = rng.random(n)
    target = n * 0.5
    return _project_to_box_and_sum(v, target)

            h = np.clip(h, 0.0, 1.0)
