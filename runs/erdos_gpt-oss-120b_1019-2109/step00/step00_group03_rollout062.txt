    # Use algorithm from "Projection onto the simplex with upper bound"
    n = len(v)
    # First, clip to [0,1]
    u = np.clip(v, 0, 1)
    # If sum(u) == s, return u
    if np.isclose(u.sum(), s):
        return u
    # If sum(u) > s: need to reduce
    if u.sum() > s:
        # Reduce u_i by delta, ensure >=0
        # Equivalent to solve find lambda >=0 such that sum_i max(u_i - lambda, 0) = s
        # Use sorting descending
        w = np.sort(u)[::-1] # descending
        cs = np.cumsum(w)
        # Find k such that (cs[k-1] - s) / k < w[k-1] and >= w[k] (?)...
        # Iterate
        k = 0
        for i in range(n):

            if t <= w[i] + 1e-12 and (i+1 == n or t >= w[i+1] - 1e-12):
                k = i+1
                lambda_val = t
                break

        if k == 0:

        p = np.maximum(u - lambda_val, 0)
        # Clip to <=1 again in case numeric overshoot
        p = np.minimum(p,1)
        return p
    else:
        # sum(u) < s: need to increase some entries <1
        # Similar: find lambda negative such that sum_i min(u_i - lambda, 1) = s where lambda <=0.
        # equivalently find lambda such that sum_i min(u_i + delta_i, 1) = s.

        # we can fill by raising low entries.

        # Let we increase each of the first k smallest entries to some level λ (≤1)
        # Equivalent: find λ >=0 such that sum_i min(u_i + λ, 1) = s

import time
import math
import numpy as np

def _project_to_box_simplex(v, target):
    """
    Project vector v onto {x | 0 <= x_i <= 1, sum(x)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1e6, 1e6
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(arr):
    """evaluate C5 = max correlation of h with (1‑h) multiplied by dx."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * dx)


def run(seed=42, budget_s=1100, **kwargs):
    """
    Hill‑climbing / simulated‑annealing search for a step function h that
    lowers the Erdős C5 constant.

    Returns (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.shape[0]
    else:
        n = 70                     # a modest default size
        h = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0           # integral constraint: sum(h) = n/2
    h = _project_to_box_simplex(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # ------------------------------------------------------------------
    # optimisation parameters
    start_time = time.time()
    max_time = budget_s * 0.95    # keep a safety margin
    step = 0.2                    # maximal amount moved in one elementary step
    max_temp = 0.05
    min_temp = 1e-5
    it = 0

    while time.time() - start_time < max_time:
        # temperature schedule (linear annealing)
        elapsed = time.time() - start_time
        frac = elapsed / max_time
        T = max_temp * (1.0 - frac) + min_temp * frac

        # pick two distinct indices and try to transfer a small amount
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = h[i], h[j]

        # we always move mass from j (which has some) to i (which can accept)
        if hi >= 1.0 or hj <= 0.0:
            continue

        delta_max = min(step, 1.0 - hi, hj)
        if delta_max <= 1e-12:
            continue
        delta = np.random.rand() * delta_max

        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        # evaluate
        c5_new = _c5_value(new_h)

        # downhill move is always accepted
        if c5_new < best_c5 - 1e-12:
            h = new_h
            best_h = new_h.copy()
            best_c5 = c5_new
        else:
            # occasional uphill acceptance (Metropolis)
            if T > 0.0 and np.random.rand() < math.exp(-(c5_new - best_c5) / T):
                h = new_h

        # re‑project occasionally to eliminate any tiny drift
        if it % 1000 == 0:
            if abs(h.sum() - target_sum) > 1e-9:
                h = _project_to_box_simplex(h, target_sum)

        it += 1

    # final safety projection
    best_h = _project_to_box_simplex(best_h, target_sum)
    best_c5 = _c5_value(best_h)

    return best_h, best_c5, n
