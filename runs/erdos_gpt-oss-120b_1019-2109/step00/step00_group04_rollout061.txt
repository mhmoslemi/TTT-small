import time
import numpy as np

def _c5_of_vector(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate does linear (zero‑padded) cross‑correlation.
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a 5‑step function h on [0,2] with ∫h=1 that minimizes
    C5 = max_k ∫ h(x)(1-h(x+k)) dx.
    Returns (list_of_h, C5_value, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # ------------------------------------------------------------
    # Parameters: use a multiple of 5 points so the step function can
    # be represented by exactly five constant blocks.
    # ------------------------------------------------------------
    n = 100                 # total number of samples (must be >=5 and %5==0)
    block_len = n // 5      # length of each block
    dx = 2.0 / n
    target_sum = n // 2     # = n/2  (because ∑h_i = n/2)

    # ------------------------------------------------------------
    # Helper to build the full vector from five block heights.
    # ------------------------------------------------------------
    def build_vector(heights):
        """heights: array-like of length 5, values in [0,1]."""
        vec = np.repeat(np.asarray(heights, dtype=float), block_len)
        # If for any reason n is not a multiple of 5, trim / pad.
        if vec.shape[0] > n:
            vec = vec[:n]
        elif vec.shape[0] < n:
            pad = np.full(n - vec.shape[0], heights[-1], dtype=float)
            vec = np.concatenate([vec, pad])
        return vec

    # ------------------------------------------------------------
    # Start from a reasonable analytic guess – a discretised Hann window.
    # ------------------------------------------------------------
    mids = (np.arange(5) + 0.5) * (2.0 / 5)      # mid‑points of the five blocks
    hann_heights = 0.5 * (1.0 - np.cos(np.pi * mids / 2.0))
    # Normalise to satisfy the exact integral constraint.
    scale = (target_sum / (block_len * hann_heights.sum()))
    hann_heights *= scale
    # Clip just in case numerical noise pushes values marginally out.
    hann_heights = np.clip(hann_heights, 0.0, 1.0)

    best_vec = build_vector(hann_heights)
    best_c5 = _c5_of_vector(best_vec)

    # If a parent construction exists, evaluate it as well.
    try:
        init_vec = np.asarray(initial_h_values, dtype=float)
        if init_vec.ndim == 1 and init_vec.size > 0:
            # enforce the correct length (pad/trim) and integral.
            if init_vec.size != n:
                # resample by simple linear interpolation to our grid size.
                x_old = np.linspace(0.0, 2.0, init_vec.size, endpoint=False)
                x_new = np.linspace(0.0, 2.0, n, endpoint=False)
                init_vec = np.interp(x_new, x_old, init_vec)
            # renormalise to exact integral.
            cur_sum = init_vec.sum() * dx
            if cur_sum != 0:
                init_vec *= (1.0 / cur_sum)
            init_c5 = _c5_of_vector(init_vec)
            if init_c5 < best_c5:
                best_c5 = init_c5
                best_vec = init_vec.copy()
    except Exception:
        pass  # no parent data available

    # ------------------------------------------------------------
    # Random search over 5‑step heights (respecting the area constraint).
    # ------------------------------------------------------------
    max_iter = 200000
    elapsed = lambda: time.time() - start_time
    while elapsed() < budget_s * 0.85:
        # Dirichlet guarantees positivity and sum‑to‑one; scale to sum=2.5
        heights = np.random.dirichlet(np.ones(5)) * (target_sum / block_len)
        if np.any(heights > 1.0):
            continue
        vec = build_vector(heights)
        c5 = _c5_of_vector(vec)
        if c5 < best_c5:
            best_c5 = c5
            best_vec = vec.copy()
            # small early stop if we beat the target comfortably
            if best_c5 <= 0.3808:
                break
        if (max_iter := max_iter - 1) <= 0:
            break

    # ------------------------------------------------------------
    # Local refinement: simple simulated‑annealing on the five heights.
    # ------------------------------------------------------------
    # Extract the current block heights (average of each block).
    cur_heights = np.array([np.mean(best_vec[i * block_len:(i + 1) * block_len])
                            for i in range(5)])

    # Temperature schedule.
    T = 0.1
    T_min = 1e-5
    decay = 0.9995

    while elapsed() < budget_s * 0.99:
        # pick two distinct blocks and propose a transfer of mass.
        i, j = np.random.choice(5, size=2, replace=False)
        # max transfer limited to keep both in [0,1] and not too large.
        max_transfer = min(0.05, 1.0 - cur_heights[i], cur_heights[j])
        min_transfer = max(-0.05, -cur_heights[i], cur_heights[j] - 1.0)
        if max_transfer <= min_transfer:
            continue
        delta = np.random.uniform(min_transfer, max_transfer)
        new_heights = cur_heights.copy()
        new_heights[i] += delta
        new_heights[j] -= delta
        if np.any(new_heights < 0.0) or np.any(new_heights > 1.0):
            continue
        # Enforce the exact sum constraint by a tiny correction on a random block.
        sum_diff = (target_sum / block_len) - new_heights.sum()
        if abs(sum_diff) > 1e-12:
            k = np.random.randint(5)
            new_heights[k] += sum_diff
            if not (0.0 <= new_heights[k] <= 1.0):
                continue
        vec = build_vector(new_heights)
        c5 = _c5_of_vector(vec)
        improve = c5 < best_c5
        # Metropolis acceptance.
        if improve or np.exp((best_c5 - c5) / max(T, 1e-12)) > np.random.rand():
            cur_heights = new_heights
            if improve:
                best_c5 = c5
                best_vec = vec.copy()
        if T > T_min:
            T *= decay

    # ------------------------------------------------------------
    # Final sanity check: enforce the integral exactly (tiny rounding).
    # ------------------------------------------------------------
    cur_sum = best_vec.sum() * dx
    if abs(cur_sum - 1.0) > 1e-12:
        best_vec *= (1.0 / cur_sum)

    return best_vec.tolist(), float(best_c5), n
