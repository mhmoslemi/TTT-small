    return np.max(corr) * (2.0 / len(h))

def propose_move(h, epsilon):
    n = len(h)
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    # Determine max delta
    # We'll adjust h[i] up/down and h[j] opposite.
    # Choose direction: we have to maintain sum unchanged.
    # Random delta sign:
    # Actually we can compute delta_range based on bounds:
    # If we add delta to i and subtract from j, delta can be between - (h[i]) ... (1 - h[i])? Wait adding to i may be limited upward, subtracting from j limited downward.
    # We'll choose sign uniformly: either add to i, subtract from j; or opposite.
    # To keep symmetrical, choose direction sign = 1 or -1.
    direction = np.random.choice([-1, 1])
    # If direction = 1: h[i] will increase, h[j] will decrease.
    # max increase for i = min(epsilon, 1 - h[i])
    # max decrease for j = min(epsilon, h[j])
    max_inc_i = min(epsilon, 1 - h[i])
    max_dec_j = min(epsilon, h[j])
    max_delta = min(max_inc_i, max_dec_j)
    # For direction = -1: i decreases, j increases.
    max_dec_i = min(epsilon, h[i])
    max_inc_j = min(epsilon, 1 - h[j])
    max_delta_neg = min(max_dec_i, max_inc_j)
    if direction == 1:
        delta = np.random.uniform(0, max_delta)
    else:
        delta = -np.random.uniform(0, max_delta_neg)
    # Apply
    new_h = np.copy(h)
    new_h[i] += delta
    new_h[j] -= delta
    # No need to clip because we limited delta.
    return new_h

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for a given step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def project_to_sum(v: np.ndarray, target_sum: float, lower: float = 0.0, upper: float = 1.0) -> np.ndarray:
    """
    Project vector v onto the intersection of the hyperplane sum(v)=target_sum
    and the box [lower,upper]^n using water‑filling (binary search for λ).
    The solution has the form clip(v+λ, lower, upper).
    """
    lo, hi = -2.0, 2.0  # wide enough to push everything to the bounds
    for _ in range(60):   # enough for double precision
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, lower, upper)
        s = w.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + lo, lower, upper)


def random_feasible_vector(n: int, target_sum: float) -> np.ndarray:
    """Generate a random vector in [0,1]^n whose entries sum to target_sum."""
    v = np.random.rand(n)
    return project_to_sum(v, target_sum)


def propose_move(h: np.ndarray, epsilon: float) -> np.ndarray:
    """
    Produce a new feasible h by transferring a small amount of mass
    (≤ epsilon) from one coordinate to another, preserving the total sum.
    """
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)

    if np.random.rand() < 0.5:
        # increase i, decrease j
        max_inc_i = min(epsilon, 1.0 - h[i])
        max_dec_j = min(epsilon, h[j])
        delta = min(max_inc_i, max_dec_j)
        if delta <= 0.0:
            return h
        delta = np.random.uniform(0.0, delta)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
    else:
        # decrease i, increase j
        max_dec_i = min(epsilon, h[i])
        max_inc_j = min(epsilon, 1.0 - h[j])
        delta = min(max_dec_i, max_inc_j)
        if delta <= 0.0:
            return h
        delta = np.random.uniform(0.0, delta)
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    return new_h


def gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the overlap term for a fixed integer shift k:
    f_k(h) = Σ_i h_i·(1 - h_{i+k})   (indices out of range are ignored).
    """
    n = h.shape[0]
    grad = np.zeros_like(h)

    if k >= 0:
        if k < n:
            idx = np.arange(0, n - k)                # i where i+k is inside
            grad[idx] += 1.0 - h[idx + k]
        if k <= n - 1:
            idx2 = np.arange(k, n)                   # i where i-k is inside
            grad[idx2] -= h[idx2 - k]
    else:
        d = -k
        idx = np.arange(d, n)                         # i where i-d is inside
        grad[idx] += 1.0 - h[idx - d]
        idx2 = np.arange(0, n - d)                    # i where i+d is inside
        grad[idx2] -= h[idx2 + d]
    return grad


# -------------------------------------------------------------
# Main entry point required by the harness
# -------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a small C₅ value."""
    np.random.seed(seed)

    # -----------------------------------------------------------------
    # Obtain an initial feasible construction (provided by the harness)
    # -----------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        # No parent supplied – fall back to a random feasible vector
        n_points = 45
        target = n_points / 2.0
        h0 = random_feasible_vector(n_points, target)

    n = h0.shape[0]
    target_sum = n / 2.0

    # Ensure the starting point respects the constraints exactly
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_to_sum(h0, target_sum)

    best_h = h0.copy()
    best_score = compute_c5(best_h)

    start = time.time()
    # -------------------------------------------------------------
    # Parameters for the stochastic search
    # -------------------------------------------------------------
    eps = 0.15               # magnitude of random two‑point moves
    max_temp = 0.02          # temperature for simulated‑annealing acceptance
    inner_iters = 5000       # steps per inner optimisation run

    while time.time() - start < budget_s * 0.95:
        # Occasionally start from a fresh random point to escape basins
        if np.random.rand() < 0.1:
            cur_h = random_feasible_vector(n, target_sum)
        else:
            cur_h = best_h.copy()

        cur_score = compute_c5(cur_h)

        for it in range(inner_iters):
            if time.time() - start >= budget_s * 0.95:
                break

            # ---------------------------------------------------------
            # Choose a move – gradient‑guided or random
            # ---------------------------------------------------------
            if np.random.rand() < 0.35:
                # Gradient step for the currently worst shift
                corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
                max_idx = int(np.argmax(corr))
                k_shift = max_idx - (n - 1)

                grad = gradient_for_shift(cur_h, k_shift)
                # Step size decays like 1/sqrt(iter) to aid convergence
                alpha = 0.5 / math.sqrt(it + 1.0)
                cand = project_to_sum(cur_h - alpha * grad, target_sum)
                cand_score = compute_c5(cand)
            else:
                # Simple random redistribution move
                cand = propose_move(cur_h, eps)
                cand_score = compute_c5(cand)

            # ---------------------------------------------------------
            # Acceptance rule (hill‑climbing + occasional uphill moves)
            # ---------------------------------------------------------
            if cand_score < cur_score:
                cur_h, cur_score = cand, cand_score
                if cand_score < best_score:
                    best_h, best_score = cand.copy(), cand_score
            else:
                # Simulated‑annealing probability
                temperature = max_temp * (1.0 - it / inner_iters)
                prob = math.exp(-(cand_score - cur_score) / (temperature + 1e-12))
                if np.random.rand() < prob:
                    cur_h, cur_score = cand, cand_score

        # slowly shrink the random‑move magnitude
        eps = max(0.001, eps * 0.995)

    # Final projection to guarantee feasibility (tiny numerical drift may occur)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum(best_h, target_sum)

    final_score = compute_c5(best_h)

    return best_h, float(final_score), int(n)
