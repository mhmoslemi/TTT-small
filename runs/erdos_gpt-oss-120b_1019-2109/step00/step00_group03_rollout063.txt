def compute_c5(h):
    n = len(h)
    max_shift = n // 5
    weight = h.sum()
    dx = 2.0 / n
    # compute max C across shifts s in -max_shift..max_shift
    max_val = -np.inf
    for s in range(-max_shift, max_shift + 1):
        if s >= 0:
            overlap = np.dot(h[:n-s], h[s:])
        else:
            s_abs = -s
            overlap = np.dot(h[s_abs:], h[:n-s_abs])
        C = (dx) * (weight - overlap)
        if C > max_val:
            max_val = C
    return max_val

import time
import math
import random
import numpy as np

def _project_to_simplex(v, target_sum):
    """
    Project vector v onto the set {x : 0 <= x_i <= 1, sum x_i = target_sum}.
    Uses a simple binary search for the Lagrange multiplier.
    """
    lo = -np.max(v) - 2.0
    hi = 1.0 - np.min(v) + 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _compute_c5(h, dx, max_shift):
    """
    Compute the C5 value using the same definition that the evaluation
    harness uses, but restricted to shifts |s| <= max_shift.
    """
    n = h.size
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # indices corresponding to shifts -max_shift .. max_shift
    centre = n - 1
    start = centre - max_shift
    end = centre + max_shift
    vals = corr[start:end + 1] * dx
    return float(vals.max())

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimizes the
    Erdős C5 overlap constant.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # number of sample points -------------------------------------------------
    if 'initial_h_values' in globals():
        n = len(initial_h_values)
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        n = 70
        h = np.full(n, 0.5, dtype=np.float64)          # uniform start

    # constraints -------------------------------------------------------------
    target_sum = n / 2.0                     # integral must equal 1
    dx = 2.0 / n
    max_shift = n // 5                       # shift range for C5

    # enforce constraints on the initial vector
    h = np.clip(h, 0.0, 1.0)
    h = _project_to_simplex(h, target_sum)

    # helper to evaluate the objective ---------------------------------------
    def current_c5(vec):
        return _compute_c5(vec, dx, max_shift)

    # initialise best solution ------------------------------------------------
    best_h = h.copy()
    best_c = current_c5(best_h)

    # annealing parameters ----------------------------------------------------
    total_iters = 2_000_000
    T_start = 0.06
    T_end = 5e-4

    # main loop ---------------------------------------------------------------
    start_time = time.time()
    for it in range(total_iters):
        # time check
        if time.time() - start_time > budget_s * 0.95:
            break

        # ----- propose a simple transfer move (keeps the sum exact) ----------
        a, b = random.randrange(n), random.randrange(n)
        while a == b:
            b = random.randrange(n)

        # choose direction randomly
        if random.random() < 0.5:
            donor, rec = a, b
        else:
            donor, rec = b, a

        max_transfer = min(h[donor], 1.0 - h[rec])
        if max_transfer < 1e-12:
            continue
        delta = max_transfer * random.random()

        new_h = h.copy()
        new_h[donor] -= delta
        new_h[rec]   += delta

        # evaluate
        new_c = current_c5(new_h)
        delta_c = new_c - best_c

        # temperature schedule (linear cooling)
        T = T_start + (T_end - T_start) * (it / total_iters)

        # Metropolis acceptance
        if delta_c < 0 or random.random() < math.exp(-delta_c / (T + 1e-12)):
            h = new_h
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()

        # occasional diversification
        if (it + 1) % 250_000 == 0:
            # small jitter around the current best
            noise = np.random.uniform(-0.03, 0.03, size=n)
            trial = best_h + noise
            trial = np.clip(trial, 0.0, 1.0)
            trial = _project_to_simplex(trial, target_sum)
            h = trial
            best_c = current_c5(best_h)   # recompute just in case

    # final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_simplex(best_h, target_sum)

    final_c5 = current_c5(best_h)
    return best_h, final_c5, n
