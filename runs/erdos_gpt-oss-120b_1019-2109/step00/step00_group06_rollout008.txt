import numpy as np
import time
import math

def _compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx


def _project_box_sum(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 50) -> np.ndarray:
    """
    Project vector `v` onto the set {x | 0 <= x_i <= 1, sum(x) == target}
    using a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0   # low enough ⇒ all entries clipped to 1
    hi = np.max(v) + 1.0    # high enough ⇒ all entries clipped to 0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _gradient_step(h: np.ndarray, target_sum: float, rng: np.random.Generator,
                   max_time: float, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a limited number of gradient‐descent / simulated‑annealing steps
    starting from `h`.  Returns the best vector encountered and its C5 value.
    """
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    start = time.time()
    it = 0
    # allow up to ~3000 effective iterations per run (adjusted by time)
    while it < 3000 and (time.time() - start) < max_time and (time.time() < deadline):
        it += 1

        # correlation and worst shift
        conv = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx = int(np.argmax(conv))
        shift = idx - (n - 1)          # integer lag giving the max

        # forward / backward copies for this shift
        forward = np.zeros_like(cur_h)
        backward = np.zeros_like(cur_h)
        if shift >= 0:
            forward[: n - shift] = cur_h[shift:]
            backward[shift:] = cur_h[: n - shift]
        else:
            s = -shift
            forward[s:] = cur_h[: n - s]
            backward[: n - s] = cur_h[s:]

        grad = (1.0 - forward) - backward   # sub‑gradient of the worst term

        # step size schedule – linear decay
        step = 0.3 * (1.0 - it / 3000.0) + 0.001

        # propose a new point
        cand = cur_h - step * grad
        cand = _project_box_sum(cand, target_sum)

        cand_c5 = _compute_c5(cand)

        # deterministic improvement
        if cand_c5 < cur_c5:
            cur_h, cur_c5 = cand, cand_c5
            if cand_c5 < best_c5:
                best_h, best_c5 = cand.copy(), cand_c5
        else:
            # occasional simulated‑annealing acceptance
            elapsed = time.time() - start
            temp = max(1e-4, 0.08 * (1.0 - elapsed / max_time))
            if rng.random() < math.exp(-(cand_c5 - cur_c5) / temp):
                cur_h, cur_c5 = cand, cand_c5

        # sporadic random perturbation to escape plateaus
        if it % 200 == 0 and rng.random() < 0.07:
            pert = rng.normal(scale=0.025, size=n)
            cur_h = _project_box_sum(best_h + pert, target_sum)
            cur_c5 = _compute_c5(cur_h)

    return best_h, best_c5

        best_overall_val = _compute_c5(best_overall_h)
        best_overall_n = n

    # final small sanity projection (guards against tiny drift)
    best_overall_h = _project_box_sum(best_overall_h, best_overall_n / 2.0)

    return best_overall_h, best_overall_val, int(best_overall_n)
