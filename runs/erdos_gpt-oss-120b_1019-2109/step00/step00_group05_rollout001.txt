def local_anneal(h_initial, time_budget):
    start = time.time()
    n = len(h_initial)
    h = h_initial.copy()
    best_h = h.copy()
    best_val = compute_C5(h)
    cur_val = best_val
    rng = np.random.default_rng(seed+1)
    T0 = best_val
    iteration = 0
    while time.time() - start < time_budget:
        iteration += 1
        # temperature schedule
        T = T0 * (0.995 ** (iteration/1000))
        # propose move
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # Ensure there is room to transfer from i to j
        hi = h[i]
        hj = h[j]
        # direction: move from i to j if hi > hj? Provide two possible directions? Actually we may move from i (source) to j (sink) based on hi >0 and hj<1.
        # Let's ensure we move from a higher to lower to not exceed bounds.
        if hi <= 0 or hj >= 1:
            continue
        max_delta = min(hi, 1 - hj)
        if max_delta <= 0:
            continue
        delta = rng.random() * max_delta
        # modify
        h[i] -= delta
        h[j] += delta
        # compute new C5
        new_val = compute_C5(h)
        # acceptance
        if new_val < cur_val - 1e-12 or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-9)):
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    return best_h, best_val

def compute_C5(h):
    n = h.shape[0]
    # compute correlation using FFT:
    # Actually correlation h with (1-h)
    # Let b = 1 - h
    # correlation full = np.convolve(h, b[::-1], mode='full')
    # using FFT:
    a = h
    b = (1 - h)[::-1]  # reversed
    nfft = 2**int(np.ceil(np.log2(2*n-1)))
    Fa = np.fft.rfft(a, n=nfft)
    Fb = np.fft.rfft(b, n=nfft)
    conv = np.fft.irfft(Fa * Fb, n=nfft)
    # Take the relevant indices: full length 2n-1 part at positions n-1 to n-1+2n-2?
    # Actually convolution length n+n-1 = 2n-1. Since conv length = nfft, we need to pick first 2n-1 values.
    corr = conv[:2*n - 1]
    # But the cross-correlation defined by np.correlate is equivalently convolution with reversed order.
    # So corr should match np.correlate.
    max_val = np.max(corr) * (2.0 / n)
    return float(max_val)

def greedy_swap_binary(h):
    n = len(h)
    changed = False
    while True:
        O_arr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(O_arr)
        max_val = O_arr[max_idx]
        if max_val <= 0: break # optimal
        # compute lag offset
        lag = max_idx - (n - 1)
        if lag == 0:
            break # O zero, done
        if lag > 0:
            i_range = np.arange(0, n - lag)
            j_range = i_range + lag
        else:
            d = -lag
            i_range = np.arange(d, n)
            j_range = i_range - d
        hi = h[i_range]
        hj = h[j_range]
        # mismatched type A: hi=1, hj=0
        mask_A = (hi == 1) & (hj == 0)
        # mismatched type B: hi=0, hj=1
        mask_B = (hi == 0) & (hj == 1)
        # indices in i_range fulfilling each mask
        if mask_A.any() and mask_B.any():
            # choose first
            i_idx = i_range[np.where(mask_A)[0][0]]
            j_idx = i_range[np.where(mask_B)[0][0]]   # note j_idx is index from i_range

            # For B, hi=0 at index i_range[pos], hj=1 at that index, we want to swap hi (0) with some other position where hi=1 (maybe at a different index).

            iA = i_idx
            iB = i_range[np.where(mask_B)[0][0]]
            # now swap h[iA] and h[iB]

def compute_C5(h):

def greedy_binary_opt(h):
    n = len(h)
    # Ensure h is binary 0/1 floats
    h = np.clip(np.rint(h), 0, 1)  # binarize
    improved = True
    while improved:
        O_arr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(O_arr)
        max_val = O_arr[max_idx]
        # find lag
        lag = max_idx - (n - 1)
        if max_val == 0:
            break
        # indices of overlapping region
        if lag > 0:
            i_range = np.arange(0, n - lag)
            j_range = i_range + lag
        elif lag < 0:
            d = -lag
            i_range = np.arange(d, n)
            j_range = i_range - d
        else:
            break
        # mismatches
        hi = h[i_range]
        hj = h[j_range]
        mask_A = (hi == 1) & (hj == 0)  # hi mismatched
        mask_B = (hi == 0) & (hj == 1)  # opposite mismatched
        if np.any(mask_A) and np.any(mask_B):
            # pick any
            iA = i_range[np.where(mask_A)[0][0]]
            iB = i_range[np.where(mask_B)[0][0]]
            # swap h[iA] and h[iB]
            h[iA], h[iB] = h[iB], h[iA]
            improved = True
        else:
            # no direct swap reducing this lag -> break
            improved = False
    return h

import time
import numpy as np

def _c5_value(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for a discretised step function."""
    n = h.shape[0]
    # correlation (full) of h with 1-h, then multiply by dx=2/n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _greedy_binary_opt(h):
    """
    Greedy pair‑swap optimiser for a binary (0/1) step function.
    Repeatedly fixes the worst lag by swapping a 1 that mismatches
    with a 0 that mismatches in the opposite direction.
    """
    n = h.shape[0]
    h = np.rint(h).astype(float)          # force exact 0/1
    improved = True
    while improved:
        # current overlap profile
        ov = np.correlate(h, 1.0 - h, mode="full")
        k = int(np.argmax(ov) - (n - 1))   # lag attaining the maximum
        if ov.max() == 0:
            break                           # perfect – cannot improve
        # indices participating in this lag
        if k > 0:
            i_idx = np.arange(0, n - k)
            j_idx = i_idx + k
        elif k < 0:
            d = -k
            i_idx = np.arange(d, n)
            j_idx = i_idx - d
        else:                               # k==0 gives zero overlap
            break
        hi = h[i_idx]
        hj = h[j_idx]
        # mismatches of the two opposite kinds
        mask_a = (hi == 1) & (hj == 0)      # 1‑0 pattern
        mask_b = (hi == 0) & (hj == 1)      # 0‑1 pattern
        if mask_a.any() and mask_b.any():
            # pick the first occurrence of each type
            i_a = i_idx[np.where(mask_a)[0][0]]
            i_b = i_idx[np.where(mask_b)[0][0]]
            # swap the two values – this reduces the worst overlap by 2
            h[i_a], h[i_b] = h[i_b], h[i_a]
            improved = True
        else:
            improved = False
    return h
