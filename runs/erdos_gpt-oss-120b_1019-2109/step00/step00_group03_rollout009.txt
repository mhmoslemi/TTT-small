import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k))dx for the discretised array h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation gives Σ_i h[i] * (1-h[i+k])
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h so that Σh = target while keeping 0≤h≤1."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # try to add/subtract uniformly from non‑saturated entries
    free = np.where((h > 0) & (h < 1))[0]
    if free.size == 0:
        # fallback: adjust a single coordinate that can absorb the whole diff
        cand = np.where((h + diff >= 0) & (h + diff <= 1))[0]
        if cand.size:
            h[cand[0]] += diff
        return h
    inc = diff / free.size
    h[free] += inc
    # clip and re‑balance if we overshoot a little
    np.clip(h, 0.0, 1.0, out=h)
    # recursive correction for any remaining tiny error
    if abs(h.sum() - target) > 1e-10:
        return _repair_sum(h, target)
    return h

    if best_c5 <= target_c5:
        # already good enough – report it.
        try:
            best_c5 = float(evaluate_erdos_solution(best_h))
        except NameError:
            pass
        return best_h.astype(float), best_c5, n

    # ------------------------------------------------------------------
    # 3. Simulated annealing parameters
    # ------------------------------------------------------------------
    start_temp = 0.05          # initial temperature
    end_temp = 1e-5            # final temperature
    max_time = budget_s - 0.05  # keep a tiny margin
    start_time = time.time()
    indices = np.arange(n)

    # ------------------------------------------------------------------
    # 4. Main annealing loop
    # ------------------------------------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break
        # linear cooling schedule
        temp = start_temp + (end_temp - start_temp) * (elapsed / budget_s)

        # pick two distinct positions – we will move mass from donor to receiver
        i = random.randrange(n)
        j = random.randrange(n - 1)
        if j >= i:
            j += 1

        hi, hj = h[i], h[j]

        # decide direction of transfer (i→j or j→i)
        if hi > 0 and hj < 1:
            donor, receiver = i, j
        elif hj > 0 and hi < 1:
            donor, receiver = j, i
            hi, hj = hj, hi
        else:
            continue

        delta_max = min(hi, 1.0 - hj)
        if delta_max <= 0:
            continue
        # pick a random amount; bias towards smaller steps a little
        delta = random.random() * delta_max
        if delta == 0.0:
            continue

        # apply the move
        h[donor] -= delta
        h[receiver] += delta

        new_c5 = _compute_c5(h)
        dC = new_c5 - cur_c5

        if dC < 0 or random.random() < math.exp(-dC / max(temp, 1e-12)):
            # accept
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                if best_c5 <= target_c5:
                    break
        else:
            # reject – revert
            h[donor] += delta
            h[receiver] -= delta

    # ------------------------------------------------------------------
    # 5. Final tidy‑up (enforce constraints exactly)
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_sum(best_h, target_sum)

    # final evaluation
    final_c5 = _compute_c5(best_h)
    try:
        final_c5 = float(evaluate_erdos_solution(best_h))
    except NameError:
        pass

    return best_h.astype(float), final_c5, n
