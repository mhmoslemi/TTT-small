def _neighbor_sum(h: np.ndarray, shift: int) -> np.ndarray:
    """Return h[i+shift] + h[i-shift] with out‑of‑bounds treated as 0."""
    n = h.size
    if shift >= 0:
        left = np.concatenate((np.zeros(shift), h[: n - shift]))
        right = np.concatenate((h[shift:], np.zeros(shift)))
    else:
        d = -shift
        left = np.concatenate((h[d:], np.zeros(d)))
        right = np.concatenate((np.zeros(d), h[: n - d]))
    return left + right


def _make_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust so that sum(h) == target."""
    # simple projection onto the box and the affine hyperplane
    n = h.size
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # distribute the remaining mass uniformly respecting the bounds
    # iterate a few times – convergence is fast for the small n we use.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        delta = diff / n
        # apply delta, but keep within [0,1]
        add = np.minimum(1.0 - h, np.maximum(0.0, delta))
        sub = np.minimum(h, np.maximum(0.0, -delta))
        h += add - sub
        diff = target - h.sum()
    # final safety clip
    np.clip(h, 0.0, 1.0, out=h)
    return h


def _initial_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Construct a feasible start vector of length n."""
    try:

            old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a resolution.

    # ------------------------------------------------------------------
    n = 120                     # can be changed – 120 is fast and accurate
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Helper to compute the objective (max overlap integral).
    # ------------------------------------------------------------------
    def max_overlap(arr: np.ndarray) -> float:
        return np.max(np.correlate(arr, 1.0 - arr, mode="full") * dx)

    # ------------------------------------------------------------------
    # Initialise the best known solution.
    # ------------------------------------------------------------------
    h_best = _initial_h(n, rng)
    best_c = max_overlap(h_best)

    max_time = budget_s * 0.95   # keep a small safety margin

    random_batch = 8            # number of random proposals when gradient move stalls

    while time.time() - start < max_time:
