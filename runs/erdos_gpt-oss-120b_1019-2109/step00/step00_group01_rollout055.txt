import numpy as np, time, random
def compute_c5_numer(h):
    return int(np.correlate(h, 1-h, mode='full').max())
def compute_c5(h):
    return (2.0 * compute_c5_numer(h)) / len(h)

def hill_climb(h, max_no_improve=2000, time_limit=None):
    n = len(h)
    current = h.copy()
    current_val = compute_c5_numer(current)
    best = current.copy()
    best_val = current_val
    # maintain indices sets for ones and zeros
    ones = np.flatnonzero(current)
    zeros = np.flatnonzero(1-current)
    # random generator rng
    rng = np.random.default_rng()
    no_improve = 0
    start = time.time()
    while no_improve < max_no_improve:
        if time_limit and time.time() - start > time_limit:
            break
        i_one = rng.choice(ones)
        i_zero = rng.choice(zeros)
        # swap
        current[i_one] = 0
        current[i_zero] = 1
        new_val = compute_c5_numer(current)
        if new_val <= current_val:
            # accept
            current_val = new_val
            # update indices
            ones = ones[ones != i_one]
            zeros = zeros[zeros != i_zero]
            # add swapped positions
            ones = np.append(ones, i_zero)
            zeros = np.append(zeros, i_one)
            if new_val < best_val:
                best_val = new_val
                best = current.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert
            current[i_one] = 1
            current[i_zero] = 0
            no_improve += 1
    return best, best_val
```

import numpy as np
import time
import math

def _c5_numer(h: np.ndarray) -> float:
    """Return the maximal overlap count (before scaling by dx)."""
    # np.correlate works with float/int arrays; result is float when h is float
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))

def _project_box_sum(v: np.ndarray, target: float, tol: float = 1e-10, max_iter: int = 60) -> np.ndarray:
    """
    Projection of v onto {x in [0,1]^n : sum(x) = target}.
    Solved by bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0  # wide enough interval for the multiplier
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        p = np.clip(v - mid, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None
    best_val = np.inf
    best_n = None
    TARGET = 0.3808

    # ----------------------------------------------------------------------
    # initialise from parent if available
    init = kwargs.get("initial_h_values")
    if init is not None:
        init_arr = np.asarray(init, dtype=float)
        n0 = init_arr.shape[0]
        init_arr = _project_box_sum(init_arr, n0 / 2.0)
        val0 = _c5_numer(init_arr)
        best_h, best_val, best_n = init_arr.copy(), val0, n0

    # candidate sizes (even numbers, modest)
    candidates = [200, 250, 300, 350, 400, 450, 500]
    if best_n is not None:
        candidates = [best_n] + candidates

    # ----------------------------------------------------------------------
    for n in candidates:
        if time.time() - start > 0.95 * budget_s:
            break
        # if we already beat the target we can stop
        if best_h is not None and (2.0 * best_val / best_n) <= TARGET:
            break

        target_sum = n / 2.0
        # ------------------------------------------------------------------
        # Several independent SA runs for each n
        max_restarts = 20
        for _ in range(max_restarts):
            if time.time() - start > 0.95 * budget_s:
                break

            # start point: either refined parent (first restart) or random
            if best_h is not None and n == best_n and _.__int__() == 0:
                h = best_h.copy()
            else:
                h = rng.random(n)
                h = _project_box_sum(h, target_sum)

            cur_val = _c5_numer(h)

            # remember if this start already beats global best
            if cur_val < best_val:
                best_h, best_val, best_n = h.copy(), cur_val, n

            # simulated‑annealing schedule
            # number of iterations is tuned dynamically by remaining time
            elapsed = time.time() - start
            est_per_iter = 3e-4                     # rough seconds per evaluation
            remaining = max(0.0, budget_s - elapsed)
            max_iter = int(min(60000, remaining / est_per_iter * 0.5))
            if max_iter < 4000:
                max_iter = int(remaining / est_per_iter * 0.9)
                max_iter = max(1000, max_iter)

            T0 = max(1.0, cur_val * 0.5)           # initial temperature

            for step in range(max_iter):
                if time.time() - start > 0.95 * budget_s:
                    break

                # linear cooling
                T = T0 * (1.0 - step / max_iter)

                # pick two distinct indices
                i = rng.integers(n)
                j = rng.integers(n - 1)
                if j >= i:
                    j += 1

                # direction: move mass from i to j or opposite
                forward = rng.random() < 0.5
                if forward:
                    max_eps = min(h[i], 1.0 - h[j])
                else:
                    max_eps = min(h[j], 1.0 - h[i])
                if max_eps <= 0.0:

                eps = rng.random() * max_eps

                # apply move
                if forward:

                    h[j] += eps
                else:

                new_val = _c5_numer(h)

                if accept:

                        # early termination if already under target bound
