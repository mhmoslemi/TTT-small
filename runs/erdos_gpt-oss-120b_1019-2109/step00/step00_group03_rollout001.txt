import time, random, math
import numpy as np

def _c5_value(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral for a given step vector."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _make_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] and the hyperplane sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    if diff > 0:  # need to increase the sum
        for i in np.argsort(v):
            inc = min(diff, 1.0 - v[i])
            if inc > 0:
                v[i] += inc
                diff -= inc
                if diff <= 1e-12:
                    break
    else:  # need to decrease the sum
        need = -diff
        for i in np.argsort(-v):
            dec = min(need, v[i])
            if dec > 0:
                v[i] -= dec
                need -= dec
                if need <= 1e-12:
                    break
    # tiny residual correction
    residual = target - v.sum()
    if abs(residual) > 1e-9:
        v[0] = np.clip(v[0] + residual, 0.0, 1.0)
    return v


def _random_binary(n: int, ones: int) -> np.ndarray:
    """Uniform random binary vector with exactly `ones` entries equal to 1."""
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, ones, replace=False)
    h[idx] = 1.0
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Heuristic search for a low‑overlap step function."""
    random.seed(seed)
    np.random.seed(seed ^ 0xDEADBEEF)

    # ---- initialise problem size and a feasible start vector ----
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
        target = n // 2  # because ∫h = 1 ⇒ sum(h) = n/2
        h = _make_feasible(init.copy(), target)
    else:
        n = 70
        target = n // 2
        h = _random_binary(n, target)

    dx = 2.0 / n

    cur_cost = _c5_value(h, dx)
    best_h = h.copy()
    best_cost = cur_cost

    start_time = time.time()
    T0 = 0.01                     # initial temperature
    prob_transfer = 0.7           # probability of using a transfer move
    iter_cnt = 0

        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.99:
            break

        # linear cooling schedule
        T = T0 * (1.0 - elapsed / budget_s)
        T = max(T, 1e-9)

        if random.random() < prob_transfer:
            # transfer a mass δ from j to i (keeps the total sum)
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]

            h_new = h.copy()
            h_new[i] = hi + delta
            h_new[j] = hj - delta
        else:
            # simple swap (useful for binary structure)
            i, j = random.sample(range(n), 2)
            h_new = h.copy()
            h_new[i], h_new[j] = h[j], h[i]

        new_cost = _c5_value(h_new, dx)
        diff = new_cost - cur_cost

        # ----- Metropolis acceptance -----
        if diff <= 0.0 or random.random() < math.exp(-diff / T):
            h = h_new
            cur_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = h.copy()

        # occasional random restart to escape stagnation
        if random.random() < 0.001:
            h = _random_binary(n, target)
            cur_cost = _c5_value(h, dx)

        iter_cnt += 1

    # final safety clipping (tiny numeric drift)
    best_h = np.clip(best_h, 0.0, 1.0)
    return (best_h.astype(float), float(best_cost), int(n))
