    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h, its C5 bound and the number of points.

    The routine starts from the supplied ``initial_h_values`` (if any) and
    performs a simple simulated‑annealing hill‑climb that respects the
    constraints sum(h)=n/2 and 0≤h≤1.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1. Obtain an initial feasible vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        # fallback: 45 points with constant value 0.5 (satisfies the constraints)
        n = 45
        h0 = np.full(n, 0.5, dtype=np.float64)

    n = h0.shape[0]
    target_sum = n / 2.0

    # Clip to [0,1] and force the exact sum while staying in the box.
    h0 = np.clip(h0, 0.0, 1.0)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Simple proportional scaling followed by a tiny redistribution of the residue.
        if cur_sum > 0:
            h0 *= target_sum / cur_sum
        h0 = np.clip(h0, 0.0, 1.0)
        residue = target_sum - h0.sum()
        if abs(residue) > 1e-12:
            free = np.where((h0 > 0.0) & (h0 < 1.0))[0]
            if free.size == 0:
                # all entries are at a bound; push a random free index a tiny amount
                free = np.arange(n)
            inc = residue / free.size
            h0[free] += inc
            h0 = np.clip(h0, 0.0, 1.0)

    # ------------------------------------------------------------------
    # 2. Evaluation function
    # ------------------------------------------------------------------
    _eval = globals().get("evaluate_erdos_solution", None)

    def _c5(h):
        return float(_eval(h)) if _eval is not None else _c5_corr(h)

    best_h = h0.copy()
    best_c = _c5(best_h)

    # ------------------------------------------------------------------
    # 3. Simulated annealing / hill climbing
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_c = best_c

    # temperature schedule
    T_start = 2.0e-2
    T_end = 1.0e-5

    # keep a counter for occasional restarts (not essential but cheap)
    it = 0
    # allocate a small safety margin so we return before the hard timeout
    time_limit = budget_s - 0.3

    while time.time() - start < time_limit:
        # linear temperature decay
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T_start * (1.0 - frac) + T_end * frac

        # propose moving mass δ from index j to index i
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        # feasible interval for δ so that both entries stay in [0,1]
        lower = -min(cur_h[i], 1.0 - cur_h[j])
        upper = min(1.0 - cur_h[i], cur_h[j])
        if upper <= lower:
            continue

        # draw a uniform δ in the feasible interval
        delta = rng.uniform(lower, upper)

        cand_h = cur_h.copy()
        cand_h[i] += delta
        cand_h[j] -= delta

        cand_c = _c5(cand_h)

        # Metropolis acceptance
        if cand_c < cur_c or rng.random() < math.exp(-(cand_c - cur_c) / (T + 1e-12)):
            cur_h, cur_c = cand_h, cand_c
            if cand_c < best_c:
                best_h, best_c = cand_h.copy(), cand_c

        # occasional random restart to avoid dead ends
        it += 1
        if it % 25000 == 0:
            # re‑initialise a fresh feasible vector
            rand = rng.random(n)
            rand = np.clip(rand, 0.0, 1.0)
            rand *= target_sum / rand.sum()
            # enforce the box again
            rand = np.clip(rand, 0.0, 1.0)
            # adjust any leftover mass
            residue = target_sum - rand.sum()
            if abs(residue) > 1e-12:
                free = np.where((rand > 0.0) & (rand < 1.0))[0]
                if free.size:
                    inc = residue / free.size
                    rand[free] += inc
                    rand = np.clip(rand, 0.0, 1.0)
            cur_h, cur_c = rand, _c5(rand)

    return best_h, float(best_c), int(n)
