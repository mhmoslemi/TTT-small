Define compute_c5:

def compute_c5(h):
    n = len(h)
    # compute autocorrelation using FFT
    L = 2*n
    H = np.fft.fft(h, L)
    corr = np.real(np.fft.ifft(H * np.conj(H)))[:2*n-1]
    min_corr = np.min(corr)
    c5 = 1 - (2*min_corr)/n

def compute_c5(h):
    n = h.shape[0]
    # Use next power of two >= 2*n
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.fft(h, L)
    corr = np.real(np.fft.ifft(H * np.conj(H)))[:2*n-1]
    min_corr = np.min(corr)
    # Ensure min_corr is not slightly negative due to rounding:
    if min_corr < 0:
        min_corr = 0.0
    c5 = 1.0 - (2.0 * min_corr) / n
    return c5

def improve_by_gradient(h, target_sum, step_max):

    n = len(h)

    # compute corr
    L = 1 << ((2*n-1).bit_length())
    H = np.fft.rfft(h, L)
    corr = np.fft.irfft(np.abs(H)**2, n=L)[:2*n-1]
    # find min shift
    p_idx = np.argmin(corr)
    p = p_idx - (n-1)

    # compute gradient contributions as described
    g = np.zeros(n)
    if p > 0:
        g[:n-p] += h[p:]
        g[p:] += h[:n-p]
    elif p < 0:
        p_abs = -p
        g[:n-p_abs] += h[p_abs:]
        g[p_abs:] += h[:n-p_abs]
    else:
        g = 2 * h

    # indices sorted
    # choose i_src where g is largest (i.e., we want to increase h where g high) but we need to decrease elsewhere.
    # Actually we want to move mass from low g positions to high g positions.
    i_dst = np.argmax(g)  # high contribution
    i_src = np.argmin(g)  # low contribution

    # Ensure feasible: max transfer amount = min(step_max, h[i_src], 1 - h[i_dst])
    max_eps = min(step_max, h[i_src], 1 - h[i_dst])
    if max_eps <= 1e-12:
        return h, corr
    eps = max_eps * np.random.rand() # random fraction: maybe 0.5*max

    # apply move
    h_new = h.copy()
    h_new[i_src] -= eps
    h_new[i_dst] += eps

    return h_new, corr

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time

    rng = np.random.default_rng(seed)

    # time
    t_start = time.time()
    time_limit = budget_s * 0.95

    # Determine number of points
    n0 = None
    if 'initial_h_values' in globals():
        init_h = globals()['initial_h_values']
        n0 = len(init_h)
    else:
        init_h = None

    # choose n = max(200, n0 if available else 200)
    n = max(200, n0 if n0 else 200)
    if n > 800:
        n = 800 # cap

    # Create initial h
    if init_h is not None:
        # interpolate
        x0 = np.linspace(0, 2, n0, endpoint=False)
        x = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x, x0, init_h)
    else:
        # random 0-1 half ones
        h = np.zeros(n)
        ones_idx = rng.choice(n, n//2, replace=False)
        h[ones_idx] = 1.0

    # projection onto capped simplex
    target_sum = n/2
    h = project_onto_box_simplex(h, target_sum)

    # compute current best
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_h = h.copy()
    curr_c5 = best_c5

    # parameters
    T = 0.02
    T_decay = 0.9995
    step_max = 0.5  # max transfer amount per move
    min_T = 1e-6
    iteration = 0

    # Precompute L_fft
    L_fft = 1 << ((2*n - 1).bit_length())

    # define function to compute C5 and correlation quickly using precomputed L
    def compute_corr_and_c5(vec):
        H = np.fft.rfft(vec, L_fft)
        corr = np.fft.irfft(np.abs(H)**2, n=L_fft)[:2*n-1]
        min_corr = np.min(corr)
        c5 = 1.0 - (2.0 * min_corr) / n
        return c5, corr

    # initial correlation
    curr_c5, curr_corr = compute_corr_and_c5(curr_h)

    while time.time() - t_start < time_limit:
        iteration += 1
        # decide move type
        if rng.random() < 0.7:
            # gradient move based on minimal shift
            p_idx = np.argmin(curr_corr)
            p = p_idx - (n-1)
            # compute gradient contributions
            g = np.zeros(n)
            if p > 0:
                g[:n-p] += curr_h[p:]
                g[p:] += curr_h[:n-p]
            elif p < 0:
                pp = -p
                g[:n-pp] += curr_h[pp:]
                g[pp:] += curr_h[:n-pp]
            else:
                g = 2 * curr_h
            # src/dst based on g
            i_dst = np.argmax(g)
            i_src = np.argmin(g)
            # compute feasible epsilon
            max_eps = min(step_max, curr_h[i_src], 1.0 - curr_h[i_dst])
            if max_eps > 1e-12:
                eps = max_eps * rng.random()
                candidate_h = curr_h.copy()
                candidate_h[i_src] -= eps
                candidate_h[i_dst] += eps
            else:
                # fallback random move
                i_src, i_dst = rng.choice(n, 2, replace=False)
                max_eps = min(step_max, curr_h[i_src], 1.0 - curr_h[i_dst])
                if max_eps <= 0:

                eps = max_eps * rng.random()
                candidate_h = curr_h.copy()
                candidate_h[i_src] -= eps
                candidate_h[i_dst] += eps
        else:
            # random transfer move
            i_src, i_dst = rng.choice(n, 2, replace=False)
            max_eps = min(step_max, curr_h[i_src], 1.0 - curr_h[i_dst])

    best_h = project_onto_box_simplex(best_h, target_sum)

def compute_corr_and_c5(vec):
    n = len(vec)
    H = np.fft.rfft(vec, L_fft)
    corr = np.fft.irfft(np.abs(H)**2, n=L_fft)[:2*n-1]
    min_corr = np.min(corr)
    c5 = 1.0 - (2.0 * min_corr) / n
    return c5, corr

    n = h.shape[0]
    H = np.fft.rfft(h, n=L_fft)
    corr = np.fft.irfft(np.abs(H) ** 2, n=L_fft)[: 2 * n - 1]
    min_corr = np.min(corr)
    return 1.0 - (2.0 * min_corr) / n, corr

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()
    limit = budget_s * 0.95

    # ---- choose resolution -------------------------------------------------
    init = kwargs.get("initial_h_values", None)
    n0 = len(init) if init is not None else 0
    n = max(200, n0)
    if n > 800:
        n = 800
    target_sum = n / 2.0
    L_fft = 1 << ((2 * n - 1).bit_length())  # power of two >= 2n-1

    # ---- initial function --------------------------------------------------
    if init is not None:
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x, x0, init)
    else:
        theta = np.pi * np.arange(n) / (n - 1)
        h = np.cos(theta) ** 2                     # symmetric cosine^2 start
    h = _project_onto_box_simplex(h, target_sum)

    best_h = h.copy()
    best_c5, best_corr = _c5_from_h(best_h, L_fft)

    cur_h = h.copy()
    cur_c5, cur_corr = best_c5, best_corr

    # ---- annealing parameters ---------------------------------------------
    T = 0.02
    decay = 0.9995
    min_T = 1e-6
    step_max = 0.5
    prob_grad = 0.7
    it = 0

    while time.time() - start < limit:
        it += 1

        # ---- decide move ---------------------------------------------------
        if rng.random() < prob_grad:
            p_idx = np.argmin(cur_corr)
            p = p_idx - (n - 1)

            # gradient for the shift that gives minimal self–overlap
            g = np.zeros(n)
            if p > 0:
                g[: n - p] += cur_h[p:]
                g[p:] += cur_h[: n - p]
            elif p < 0:
                pp = -p
                g[: n - pp] += cur_h[pp:]
                g[pp:] += cur_h[: n - pp]
            else:
                g = 2.0 * cur_h

            src_cand = np.where(cur_h > 1e-12)[0]
            dst_cand = np.where(cur_h < 1.0 - 1e-12)[0]
            if src_cand.size == 0 or dst_cand.size == 0:
                continue
            i_src = src_cand[np.argmin(g[src_cand])]
            i_dst = dst_cand[np.argmax(g[dst_cand])]
            max_eps = min(step_max, cur_h[i_src], 1.0 - cur_h[i_dst])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random()
            cand = cur_h.copy()
            cand[i_src] -= eps
            cand[i_dst] += eps
        else:
            i_src, i_dst = rng.integers(n, size=2)
            while i_dst == i_src:
                i_dst = rng.integers(n)
            max_eps = min(step_max, cur_h[i_src], 1.0 - cur_h[i_dst])
            if max_eps <= 0:
                continue
            eps = max_eps * rng.random()
            cand = cur_h.copy()
            cand[i_src] -= eps
            cand[i_dst] += eps

        # ---- evaluate -------------------------------------------------------
        cand_c5, cand_corr = _c5_from_h(cand, L_fft)
        delta = cand_c5 - cur_c5
        if delta < 0 or (T > 0 and np.exp(-delta / T) > rng.random()):
            cur_h, cur_c5, cur_corr = cand, cand_c5, cand_corr
            if cand_c5 < best_c5:
                best_c5, best_h = cand_c5, cand.copy()

        T = max(T * decay, min_T)

        # ---- occasional re‑projection ---------------------------------------
        if it % 2000 == 0:
            s = cur_h.sum()
            if abs(s - target_sum) > 1e-7:
                cur_h += (target_sum - s) / n
                cur_h = np.clip(cur_h, 0.0, 1.0)
                cur_h = _project_onto_box_simplex(cur_h, target_sum)
                cur_c5, cur_corr = _c5_from_h(cur_h, L_fft)
                if cur_c5 < best_c5:
                    best_c5, best_h = cur_c5, cur_h.copy()

    best_h = _project_onto_box_simplex(best_h, target_sum)
    best_c5, _ = _c5_from_h(best_h, L_fft)
    return best_h, best_c5, n
