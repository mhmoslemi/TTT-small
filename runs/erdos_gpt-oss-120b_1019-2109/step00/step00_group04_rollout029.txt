        n = 84
        # start with half ones half zeros random
        h = np.zeros(n)
        half = n//2
        idx = np.random.choice(n, half, replace=False)
        h[idx] = 1.0
        # adjust sum exactly: there may be leftover half if n odd; ensure sum = n/2 exactly.
        h = h * 1.0 # already sum=half
    # Ensure sum exactly n/2
    target_sum = n/2
    current_sum = h.sum()
    if abs(current_sum - target_sum) > 1e-9:
        # adjust
        diff = target_sum - current_sum
        # add diff to first element clamped
        # but diff should be integer since h values are multiples of 0.5? but we can just adjust
        h[0] = np.clip(h[0] + diff, 0, 1)
    # compute best bound
    dx = 2.0 / n
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    current_h = h.copy()
    current_val = best_val
    # Simulated annealing
    T0 = 0.1
    # approximate maximum possible bound? Could be up to 0.5.
    while time.time() - start < budget_s - 0.5:
        # schedule temperature decreasing linearly
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / budget_s)
        # propose move
        i, j = np.random.choice(n, 2, replace=False)
        # choose direction
        if random.random() < 0.5:
            # increase i, decrease j
            max_inc = min(1.0 - current_h[i], current_h[j])
            if max_inc <= 0:
                continue
            delta = random.random() * max_inc
            new_h = current_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            # increase j, decrease i
            max_inc = min(1.0 - current_h[j], current_h[i])
            if max_inc <= 0:
                continue
            delta = random.random() * max_inc
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        new_val = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx
        delta_val = new_val - current_val
        if delta_val < 0 or random.random() < math.exp(-delta_val / (T + 1e-12)):
            current_h = new_h
            current_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    # End loop
    return best_h, best_val, n

def _evaluate_raw(h):
    """Return the unscaled maximal overlap sum Σ h[i]*(1-h[i+k])."""
    return np.max(np.correlate(h, 1.0 - h, mode="full"))

def _max_lag_info(h):
    """Return (max_raw, lag, start_index, length) for the current worst lag."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_raw = float(corr.max())
    max_idx = int(corr.argmax())
    N = h.shape[0]
    lag = max_idx - (N - 1)          # signed shift (can be negative)
    if lag >= 0:
        length = N - lag
        start = 0
    else:
        length = N + lag            # lag is negative
        start = -lag
    return max_raw, lag, start, length

            i = start + i_local
            j = i + lag

        # Ensure indices are in range
        if j < 0 or j >= N:
            # cannot use this lag directly; perform a small random move
            i, j = rng.choice(N, size=2, replace=False)

        # Attempt to transfer mass from i (large contribution) to j (low h)
        if current[i] > 0.0 and current[j] < 1.0:
            delta = min(current[i], 1.0 - current[j])   # move as much as possible
            if delta > 1e-12:
                # Apply move
                current[i] -= delta
                current[j] += delta

                new_raw = _evaluate_raw(current)
                if new_raw <= cur_raw + 1e-12:
                    cur_raw = new_raw
                    # Accept move; maybe update best
                    cur_val = cur_raw * (2.0 / N)
                    if cur_val < best_val - 1e-12:
                        best_val = cur_val
                        best_raw = cur_raw
                        best_h = current.copy()
                        no_improve = 0
                        # early success?
                        if best_val <= 0.3808:
                            return best_h.tolist(), best_val, N
                else:
                    # reject – revert
                    current[i] += delta
                    current[j] -= delta
                    no_improve += 1
            else:
                no_improve += 1
        else:
            # No direct transfer possible; do a tiny random shuffle
            i, j = rng.choice(N, size=2, replace=False)
            # random direction
            if rng.random() < 0.5:
                max_inc = min(1.0 - current[i], current[j])
                if max_inc > 0:
                    delta = max_inc * rng.random() * 0.3
                    current[i] += delta
                    current[j] -= delta
            else:
                max_inc = min(1.0 - current[j], current[i])
                if max_inc > 0:
                    delta = max_inc * rng.random() * 0.3
                    current[i] -= delta
                    current[j] += delta
            # recompute raw max after this tiny move
            cur_raw = _evaluate_raw(current)
            cur_val = cur_raw * (2.0 / N)
            if cur_val < best_val:
                best_val = cur_val
                best_h = current.copy()
                no_improve = 0
            else:
                no_improve += 1

        # Periodic restart to escape possible stagnation
        if no_improve >= RESTART_AFTER:
            # keep the best so far, start a fresh randomised pattern
            current = _init_pattern(N, rng)
            cur_raw = _evaluate_raw(current)
            no_improve = 0

    # ------------------------------------------------------------------ #
    # 4. Return best result found within the budget -------------------- #
    final_val = best_raw * (2.0 / N)
    return best_h.tolist(), final_val, N
