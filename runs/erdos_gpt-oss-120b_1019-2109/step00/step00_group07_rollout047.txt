import time, math, random
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    # --------------------------------------------------------------
    # Helper to compute the C5 value for a given step function h.
    # --------------------------------------------------------------
    def compute_c5(arr, dx):
        return np.max(np.correlate(arr, 1.0 - arr, mode="full")) * dx

    # --------------------------------------------------------------
    # Initialise random generators.
    # --------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Determine problem size and starting vector.
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n = h0.size
    else:
        # fallback: 80 points, half ones.
        n = 80
        h0 = np.zeros(n, dtype=np.float64)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

    # ensure basic feasibility
    h0 = np.clip(h0, 0.0, 1.0)
    target_sum = n / 2.0
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple affine scaling then clipping to restore sum
        h0 = h0 * (target_sum / cur_sum)
        h0 = np.clip(h0, 0.0, 1.0)
        # tiny residual correction
        diff = target_sum - h0.sum()
        if abs(diff) > 1e-12:
            h0 += diff / n
            h0 = np.clip(h0, 0.0, 1.0)

    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)

    # --------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop.
    # --------------------------------------------------------------
    start_time = time.time()
    max_time = budget_s * 0.95  # leave a small margin

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # Parameters for the search
    T0 = 0.02                       # initial temperature (for occasional uphill moves)
    step_start = 0.4                # maximal transfer size at the beginning
    stagnation_limit = 20000        # steps without improvement before a restart
    steps_since_best = 0

    # Main loop
    while True:
        # time check (do it every 1000 iterations to reduce overhead)
        if (steps_since_best & 1023) == 0:  # every 1024 steps
            if time.time() - start_time > max_time:
                break

        # schedule for temperature and step size
        elapsed = time.time() - start_time
        frac = elapsed / max_time
        T = max(T0 * (1.0 - frac), 1e-9)
        step_max = max(step_start * (1.0 - frac), 1e-4)

        # propose a transfer between two distinct indices
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random direction: move from i to j or converse
        if np.random.rand() < 0.5:
            donor, acceptor = i, j
        else:
            donor, acceptor = j, i

        max_delta = min(cur_h[donor], 1.0 - cur_h[acceptor])
        if max_delta <= 0.0:
            continue
        delta = np.random.rand() * min(step_max, max_delta)

        new_h = cur_h.copy()
        new_h[donor] -= delta
        new_h[acceptor] += delta

        new_c5 = compute_c5(new_h, dx)

        # accept if improvement, otherwise probabilistically (annealing)
        if new_c5 < cur_c5 - 1e-12:
            accept = True
        else:
            accept = np.random.rand() < math.exp(-(new_c5 - cur_c5) / T)

        if accept:
            cur_h = new_h
            cur_c5 = new_c5
            steps_since_best += 1
            if new_c5 < best_c5 - 1e-12:
                best_h = new_h.copy()
                best_c5 = new_c5
                steps_since_best = 0
        else:
            steps_since_best += 1

        # restart if stagnating for a long time
        if steps_since_best > stagnation_limit:
            # restart from the best found so far with a tiny random perturbation
            cur_h = best_h.copy()
            # apply a few tiny swaps to shake out of possible flat region
            for _ in range(5):
                a, b = np.random.randint(n), np.random.randint(n - 1)
                if b >= a:
                    b += 1
                eps = step_start * 0.01 * np.random.rand()
                max_eps = min(cur_h[a], 1.0 - cur_h[b])
                eps = min(eps, max_eps)
                cur_h[a] -= eps
                cur_h[b] += eps
            cur_c5 = compute_c5(cur_h, dx)
            steps_since_best = 0

    # Final safety projection (clip and tiny sum correction)
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_diff = target_sum - best_h.sum()
    if abs(sum_diff) > 1e-12:
        best_h += sum_diff / n
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h, dx)

    return best_h.tolist(), float(final_c5), n
