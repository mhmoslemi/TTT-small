            continue

def project_to_sum(v, target):
    # Clip already within [0,1] assumed.
    # Solve for lambda such that sum(np.clip(v - lam, 0, 1)) = target
    # Use binary search over lam.
    lo, hi = -1.0, 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    return proj

import time, math
import numpy as np

def _project_to_sum(v: np.ndarray) -> np.ndarray:
    """Project onto the hyperplane sum(v)=len(v)/2 with 0<=v<=1 via binary search."""
    n = v.shape[0]
    target = n * 0.5
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """Return the C5 value for a step vector h."""
    n = h.shape[0]
    # correlation of h with (1-h); mode='full' gives the required range
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _neighbor_move(h: np.ndarray) -> np.ndarray:
    """Return a new vector obtained by moving a random amount from one entry to another."""
    n = h.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        j = np.random.randint(n)
    # decide direction of transfer
    if np.random.rand() < 0.5:
        # i -> j
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            return h
        delta = np.random.rand() * max_delta
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        # j -> i
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0.0:
            return h
        delta = np.random.rand() * max_delta
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
    return new_h


def _simulated_annealing(start_h: np.ndarray, time_budget: float) -> (np.ndarray, float):
    """Simple SA that tries to minimise C5."""
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_val = _c5_of(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    while time.time() - start < time_budget:
        # generate neighbour
        new_h = _neighbor_move(cur_h)
        new_val = _c5_of(new_h)

        # linear temperature schedule
        elapsed = time.time() - start
        frac = elapsed / time_budget
        temp = (0.2 * (1.0 - frac)) + 0.001  # from ~0.2 down to 0.001

        # acceptance test
        if new_val < cur_val or np.random.rand() < math.exp((cur_val - new_val) / temp):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
    # final projection to enforce the exact sum (tiny drift may appear)
    best_h = _project_to_sum(best_h)
    best_val = _c5_of(best_h)
    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimise the Erdős C5 bound.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : list of floats – the step function values on a uniform grid
        c5_bound : float – the achieved C5 value
        n_points : int – length of the grid
    """
    np.random.seed(seed)
    total_start = time.time()

    # initialise from the provided parent if present, otherwise build a symmetric U‑shape
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
        # ensure feasibility (tiny numerical errors may exist)
        h0 = _project_to_sum(h0)
    else:
        n = 100                     # a modest size when no parent is supplied
        t = np.arange(n, dtype=float) / (n - 1) - 0.5
        h0 = 0.25 + 3.0 * t ** 2    # edges =1, centre =0.25, average =0.5
        h0 = _project_to_sum(h0)

    # initial score (for possible early exit)
    best_h, best_val = h0, _c5_of(h0)

    # allocate most of the budget to SA; keep a small safety margin
    remaining = budget_s - (time.time() - total_start) - 0.5
    if remaining > 0.1:
        best_h, best_val = _simulated_annealing(best_h, remaining)

    # final check – project once more to be safe
    best_h = _project_to_sum(best_h)
    best_val = _c5_of(best_h)

    return best_h.tolist(), float(best_val), int(best_h.shape[0])
