def improve_worst(h, dx, max_iter=5000):
    n = len(h)
    for _ in range(max_iter):
        # compute correlation and find worst lag
        corr = np.correlate(h, 1 - h, mode='full') * dx
        worst_idx = np.argmax(corr)
        worst_lag = worst_idx - (n-1)
        # compute gradient g
        g = np.zeros(n)
        L = worst_lag
        # if shift positive (L>=0):
        # for i in 0..n-1-L-1, term h[i]h[i+L]
        # gradient derived earlier: for i with partner j = i+L: g[i] += h[j]; g[j] += h[i]
        if L >= 0:
            # for i 0..n-1-L
            i_range = slice(0, n-L)
            j_range = slice(L, n)
            hi = h[i_range]
            hj = h[j_range]
            g[i_range] += hj
            g[j_range] += hi
        else:
            # negative shift: L < 0: treat similarly with L' = -L
            L_pos = -L
            i_range = slice(0, n-L_pos)
            j_range = slice(L_pos, n)
            hi = h[i_range]
            hj = h[j_range]
            # shifting opposite
            g[i_range] += hj
            g[j_range] += hi
        # identify src and dst
        src = np.argmin(g)  # minimal gradient
        dst = np.argmax(g)  # maximal gradient
        if g[dst] <= g[src] + 1e-12:
            break  # no improvement possible
        # max transfer
        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer <= 1e-12:
            break
        # step size parameter: dynamic maybe proportional to difference
        delta = max_transfer * 0.5  # move half the possible amount
        # apply move
        h[src] -= delta
        h[dst] += delta
        # recompute cost maybe to check improvement:
        new_corr = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        # if not improved break
        if new_corr >= corr[worst_idx] - 1e-12: # not decreasing max correlation
            # revert
            h[src] += delta
            h[dst] -= delta

    return h

        # choose a 1 index and a 0 index
        ones = np.where(b==1)[0]
        zeros = np.where(b==0)[0]
        if len(ones)==0 or len(zeros)==0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        b[i], b[j] = 0, 1
        new_cost = np.max(np.correlate(b, 1-b, mode='full') * dx)
        # acceptance

                best_b = b.copy()

            # revert
            b[i], b[j] = 1, 0
        temp *= alpha
        it += 1
        # early exit if best cost < target
        if best_cost <= 0.3808:
            break
    return best_b, best_cost

import time, math
import numpy as np
import cvxpy as cp

def _evaluate(h):
    """Compute the C5 value for a vector h."""
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' matches the harness
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def _solve_mip(n, time_budget):
    """Solve the binary max‑min overlap problem with a MIP."""
    p = n // 2                     # exactly half ones
    b = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True, nonneg=True)

    constraints = [
        cp.sum(b) == p,
        b[0] == 1                  # break symmetry (both a solution and its complement are feasible)
    ]

    # product variables w_{i,j}=b_i*b_j for i<j
    w = {}
    for i in range(n):
        for j in range(i + 1, n):
            w[(i, j)] = cp.Variable(boolean=True)
            constraints += [
                w[(i, j)] <= b[i],
                w[(i, j)] <= b[j],
                w[(i, j)] >= b[i] + b[j] - 1
            ]

    # for each lag enforce at least t overlapping ones
    for L in range(1, n):
        overlap = cp.sum([w[(i, i + L)] for i in range(n - L)])
        constraints.append(overlap >= t)

    prob = cp.Problem(cp.Maximize(t), constraints)

    # give the solver a time limit (seconds)
    remaining = max(5, int(time_budget))
    try:
        # CBC usually available; pass the seconds limit
        prob.solve(solver=cp.CBC, cbc={"seconds": remaining},
                   warm_start=True, verbose=False)
    except Exception:
        try:
            # GLPK_MI as a secondary option
            prob.solve(solver=cp.GLPK_MI,
                       glpk={"msg_lev": "GLP_MSG_OFF",
                             "tm_lim": remaining * 1000},
                       warm_start=True, verbose=False)
        except Exception:
            return None, None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE):
        return None, None

    b_val = np.rint(b.value).astype(int)          # ensure exact 0/1
    t_val = int(round(t.value))
    return b_val, t_val


def _solve_sa(n, time_budget):
    """Binary local search / simulated annealing fallback."""
    p = n // 2
    rng = np.random.default_rng(0)
    # start from a random balanced binary vector
    b = np.zeros(n, dtype=int)
    ones = rng.choice(n, size=p, replace=False)
    b[ones] = 1

    dx = 2.0 / n
    cur = _evaluate(b.astype(float))
    best_b, best = b.copy(), cur
    start = time.time()
    temp = 0.02
    alpha = 0.99999
    it = 0
    while time.time() - start < time_budget - 5 and it < 5_000_000:
        # pick a 1 and a 0 to swap
        idx_one = rng.choice(np.flatnonzero(b))
        idx_zero = rng.choice(np.flatnonzero(1 - b))
        b[idx_one], b[idx_zero] = 0, 1
        new = _evaluate(b.astype(float))

        if new < cur or rng.random() < math.exp(-(new - cur) / max(temp, 1e-12)):
            cur = new
            if new < best:
                best, best_b = new, b.copy()
        else:
            # revert swap
            b[idx_one], b[idx_zero] = 1, 0

        temp *= alpha
        it += 1
        if best <= 0.3808:          # early stop once we meet the target
            break
    return best_b, best

        # derived bound from the integer programme
        dx = 2.0 / n
        bound = 1.0 - t_val * dx
        # sanity – recompute via correlation
        bound = _evaluate(b_sol.astype(float))
        h_vals = b_sol.astype(float).tolist()
        return h_vals, bound, n

    # fallback: simulated annealing on binary vectors
    time_left = budget_s - (time.time() - time_start)
    b_best, best_val = _solve_sa(n, time_left)

    h_vals = b_best.astype(float).tolist()
    return h_vals, best_val, n
```
