Pseudo algorithm for water-filling projection:

def project_onto_hyperplane_box(v, target):
    w = np.clip(v, 0.0, 1.0)
    # Use while loop to adjust sum.
    for _ in range(100): # at most 100 iterations
        s = w.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Find free indices not at bounds
        free = (w > 0) & (w < 1)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust further; maybe target is infeasible -> but should not happen.
            break
        delta = diff / n_free
        w[free] += delta
        # clip again
        w = np.clip(w, 0.0, 1.0)
    return w

This algorithm converges to the projection onto intersection of box and hyperplane (the water-filling algorithm). It works because after each adjustment, those that become out of bounds will be clipped, reducing free set.

    w = np.clip(v, 0.0, 1.0)
    # water-filling
    # For small n, iterative approach fine
    for _ in range(100):
        s = w.sum()

        if abs(diff) < 1e-12:
            break

        free = (w > 0) & (w < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # cannot adjust further; break
            break
        delta = diff / n_free
        w[free] += delta
        # Clip out-of-range
        w = np.clip(w, 0.0, 1.0)
    # final correction: due to rounding we may have tiny off; adjust
    # If sum still off, distribute difference on any index within bounds:
    s = w.sum()
    while abs(s - target_sum) > 1e-10:

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random, math

    # ------------------------------------------------------------
    # Helper: projection onto 0‑1 box and fixed sum (hyperplane)
    # ------------------------------------------------------------
    def project_onto_hyperplane_box(v, target):
        """Project vector v onto {0<=x_i<=1, sum x_i = target}."""
        w = np.clip(v, 0.0, 1.0)
        for _ in range(100):
            s = w.sum()
            diff = target - s
            if abs(diff) < 1e-12:
                break
            free = (w > 0.0) & (w < 1.0)
            n_free = int(free.sum())
            if n_free == 0:
                break
            delta = diff / n_free
            w[free] += delta
            w = np.clip(w, 0.0, 1.0)
        # tiny correction if still off (rare)
        s = w.sum()
        if abs(s - target) > 1e-9:
            diff = target - s
            if diff > 0:
                idx = np.where(w < 1.0 - 1e-12)[0]
                if idx.size:
                    inc = min(diff, 1.0 - w[idx[0]])
                    w[idx[0]] += inc
            else:
                idx = np.where(w > 1e-12)[0]
                if idx.size:
                    dec = min(-diff, w[idx[0]])
                    w[idx[0]] -= dec
        return w

    # ------------------------------------------------------------
    # Helper: compute C5 value for a candidate vector
    # ------------------------------------------------------------
    def compute_c5(h):
        n = h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return np.max(corr) * dx

    # ------------------------------------------------------------
    # Initialise problem size and feasible starting point
    # ------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)

    try:
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:                     # no supplied construction
        n = 50
        h0 = np.zeros(n)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)
    n = h0.shape[0]
    target_sum = n / 2.0
    # enforce constraints precisely
    h0 = project_onto_hyperplane_box(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # ------------------------------------------------------------
    # Simulated‑Annealing core
    # ------------------------------------------------------------
    def random_feasible():
        v = np.random.rand(n)
        return project_onto_hyperplane_box(v, target_sum)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    start = time.time()
    # schedule: linear cooling from T0 to near zero
    T0 = 0.015                     # initial temperature (tuned empirically)
    goal = 0.3808                  # target we would like to reach

    # main loop
    while True:
        now = time.time()
        if now - start > budget_s * 0.94:   # keep a margin for finalisation
            break

        # occasional random restart to escape deep basins
        if random.random() < 0.001:
            cur_h = random_feasible()
            cur_c5 = compute_c5(cur_h)

        # propose a feasible 2‑point adjustment (preserves the sum)
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = cur_h[i], cur_h[j]

        # feasible delta interval so that both remain in [0,1]
        max_inc_i = 1.0 - hi
        max_dec_i = hi
        max_inc_j = 1.0 - hj
        max_dec_j = hj

        delta_max = min(max_inc_i, max_dec_j)
        delta_min = -min(max_dec_i, max_inc_j)

        if delta_max <= delta_min:
            continue

        delta = random.uniform(delta_min, delta_max)

        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        new_c5 = compute_c5(new_h)

        # temperature based acceptance
        elapsed = now - start
        progress = elapsed / budget_s
        T = T0 * (1.0 - progress) + 1e-9   # never zero

        if new_c5 < cur_c5 or random.random() < math.exp(-(new_c5 - cur_c5) / T):
            cur_h, cur_c5 = new_h, new_c5
            # record best ever seen
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
                if best_c5 <= goal:      # satisfactory solution found
                    break

    # ------------------------------------------------------------
    # Post‑processing: short gradient‑descent polishing
    # ------------------------------------------------------------
    def polish(h, iters=4000):
        h = h.copy()
        for it in range(iters):
            n = h.shape[0]
            dx = 2.0 / n
            corr = np.correlate(h, 1.0 - h, mode='full')
            idx = np.argmax(corr)

            neigh = np.zeros_like(h)
            if off != 0:

                    neigh[off:] += h[: n - off]

        best_h = polish(best_h, iters=3000)
        best_c5 = compute_c5(best_h)

        pass

    # Final safety projection (ensures constraints are exact)
    best_h = project_onto_hyperplane_box(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    # Return as required
    return (best_h.tolist(), float(final_c5), n)
