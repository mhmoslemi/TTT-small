def solve_minimax_milp(N, time_limit=None):
    import numpy as np, cvxpy as cp, time

    h = cp.Variable(N, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == N/2, t >= 0]

    # For each lag L
    for L in range(1, N):
        m = N - L
        u = cp.Variable(m, boolean=True)
        constraints.append(u <= h[:m])
        constraints.append(u <= h[L:])
        constraints.append(u >= h[:m] + h[L:] - 1)
        # correlations
        expr_pos = cp.sum(h[:m] - u)
        expr_neg = cp.sum(h[L:] - u)
        constraints.append((2.0/N) * expr_pos <= t)
        constraints.append((2.0/N) * expr_neg <= t)
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    # Solve
    try:
        if time_limit is not None:
            prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
        else:
            prob.solve(solver=cp.CBC, verbose=False)
    except Exception:

        try:
            if time_limit is not None:
                prob.solve(solver=cp.GLPK_MI, tm_lim=int(time_limit*1000), verbose=False)
            else:
                prob.solve(solver=cp.GLPK_MI, verbose=False)
        except Exception:
            return None, None
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None
    h_val = np.rint(h.value).astype(float)  # round
    # Ensure sum correct; if small discrepancy adjust
    cur_sum = h_val.sum()

        one_idx = np.where(h_val == 1)[0]

            if len(zero_idx) >= diff:
                h_val[zero_idx[:diff]] = 1

    t_val = t.value

import time, random, math
import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the Erdős overlap constant C5 for a discrete step function.
# ----------------------------------------------------------------------
def _compute_c5(h):
    """Return max_x h(x)*(1-h(x+k)) dx using the linear correlation definition."""
    h = np.asarray(h, dtype=float)
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / h.shape[0]
    return float(np.max(corr) * dx)


# ----------------------------------------------------------------------
# MIP solver selection utilities (cvxpy with a mixed‑integer backend)
# ----------------------------------------------------------------------
def _available_mip_solver():
    try:
        import cvxpy as cp
    except Exception:
        return None, None
    for name in ("CBC", "GLPK_MI", "SCIP", "ECOS_BB", "GUROBI", "MOSEK"):
        if name in cp.installed_solvers():
            return cp, name
    return None, None


# ----------------------------------------------------------------------
# Solve the exact minimax binary problem for a given even N.
# Returns (t_opt, h_opt) where t_opt is the optimal C5 and h_opt a binary array.
# ----------------------------------------------------------------------
def _solve_minimax_mip(N, time_limit=None):
    cp, solver_name = _available_mip_solver()
    if cp is None:
        return None, None

    # binary variables for the step function
    h = cp.Variable(N, boolean=True)
    t = cp.Variable()

    constraints = [
        cp.sum(h) == N / 2,   # half of the points are 1
        t >= 0
    ]

    # product variables u[i,L] = h[i] * h[i+L] for each lag L >= 1
    for L in range(1, N):
        m = N - L
        u = cp.Variable(m, boolean=True)

        # linearization of the product
        constraints.append(u <= h[:m])
        constraints.append(u <= h[L:])
        constraints.append(u >= h[:m] + h[L:] - 1)

        # positive lag correlation   sum_i (h[i] - u[i])
        expr_pos = cp.sum(h[:m] - u)
        # negative lag correlation   sum_i (h[i+L] - u[i])
        expr_neg = cp.sum(h[L:] - u)

        factor = 2.0 / N
        constraints.append(factor * expr_pos <= t)
        constraints.append(factor * expr_neg <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # Solver‑specific time‑limit handling (if supported)
    solve_kwargs = {"solver": solver_name, "verbose": False}
    if time_limit is not None:
        if solver_name == "CBC":
            solve_kwargs["maxSeconds"] = time_limit
        elif solver_name == "GLPK_MI":
            solve_kwargs["tm_lim"] = int(time_limit * 1000)   # milliseconds
        elif solver_name == "SCIP":
            solve_kwargs["max_seconds"] = time_limit
        # ECOS_BB etc. ignore the limit – they are fast for these sizes

    try:
        prob.solve(**solve_kwargs)
    except Exception:
        # Fallback: try without explicit limits
        try:
            prob.solve(solver=solver_name, verbose=False)
        except Exception:
            return None, None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    # Extract a feasible binary vector (rounding tolerances may occur)
    h_val = np.rint(np.array(h.value, dtype=float))
    # Repair any possible rounding drift in the total weight
    target = N // 2
    diff = int(target - h_val.sum())
    if diff != 0:
        if diff > 0:
            zeros = np.where(h_val == 0)[0]
            h_val[zeros[:diff]] = 1
        else:
            ones = np.where(h_val == 1)[0]
            h_val[ones[: -diff]] = 0

    t_val = float(t.value) if t.value is not None else None
    return t_val, h_val

def _local_search(N, time_allowed, rng):
    """Return a binary vector with half ones and its C5 discovered by hill climbing."""
    # start from a random balanced vector
    h = np.zeros(N, dtype=int)
    h[: N // 2] = 1
    rng.shuffle(h)

    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    start = time.time()
    # pre‑allocate index lists for speed
    while time.time() - start < time_allowed:
        # pick a random 1 and a random 0 to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if ones.size == 0 or zeros.size == 0:

        i = rng.choice(ones)
        j = rng.choice(zeros)
        h[i], h[j] = 0, 1
        new_c5 = _compute_c5(h)

        if new_c5 + 1e-12 < best_c5:

        if rng.random() < 0.001:
