import numpy as np, time

def project_box_simplex(v, total):
    lo = -np.max(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    return x

def compute_overlap_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_raw = np.max(corr)
    return corr, max_raw

import numpy as np
import time

def project_onto_box_simplex(v, total):
    lo = -np.max(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    return x

def compute_corr_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_raw = np.max(corr)
    return corr, max_raw

def gradient_from_shift(h, shift):
    n = h.shape[0]
    s = abs(shift)
    if s == 0:
        return 1.0 - 2.0 * h
    m = n - s
    g = np.ones_like(h)
    g[:m] -= h[s:]
    g[s:] -= h[:m]
    return g

def random_binary(n, total, rng):
    h = np.zeros(n, dtype=float)
    # total may be non-integer if n odd; ensure integer count
    count = int(round(total))
    idx = rng.choice(n, size=count, replace=False)
    h[idx] = 1.0
    return h

def init_from_initial(init_arr):
    n = init_arr.shape[0]
    idx = np.argsort(-init_arr)
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[idx[:half]] = 1.0
    return h

def descent(start_h, total, dx, rng, time_deadline, global_best):
    h = start_h.copy()
    corr, max_raw = compute_corr_max(h)
    cur_val = max_raw * dx
    best_h_local = h.copy()
    best_val_local = cur_val
    it = 0
    n = h.shape[0]
    while time.time() < time_deadline and it < 2000:
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        g = gradient_from_shift(h, shift)
        lr = 0.8 / (1.0 + it * 0.02)
        trial = project_onto_box_simplex(h - lr * g, total)
        corr2, max_raw2 = compute_corr_max(trial)
        val2 = max_raw2 * dx
        # accept if improvement
        if val2 < cur_val - 1e-12:
            h = trial
            corr = corr2
            cur_val = val2
            if val2 < best_val_local:
                best_val_local = val2
                best_h_local = trial.copy()
        else:
            # occasional uphill move
            T = 0.01 * (1.0 - it / 2000.0)
            if T > 0 and rng.random() < np.exp(-(val2 - cur_val) / T):
                h = trial
                corr = corr2
                cur_val = val2
        # random local perturbation every 200 steps
        if it % 200 == 199:
            i, j = rng.choice(n, size=2, replace=False)
            up = 1.0 - h[i]
            down = h[j]
            eps = min(up, down, rng.random() * 0.15)
            if eps > 0:
                tmph = h.copy()
                tmph[i] += eps
                tmph[j] -= eps
                tmph = project_onto_box_simplex(tmph, total)
                corr3, max_raw3 = compute_corr_max(tmph)
                val3 = max_raw3 * dx
                if val3 < cur_val - 1e-12:
                    h = tmph
                    corr = corr3
                    cur_val = val3
                    if val3 < best_val_local:
                        best_val_local = val3
                        best_h_local = tmph.copy()
        it += 1
        # early stop if we beat global best significantly
        if best_val_local < global_best - 1e-6:
            # still continue search for even better
            pass
    return best_h_local, best_val_local

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # Determine problem size

import numpy as np
import time

def project_onto_box_simplex(v, total):
    """Project v onto the set {0≤x≤1, sum x = total}."""
    lo = -np.max(v) - 1.0
    hi = np.max(v)
    for _ in range(60):                      # bisection for λ
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    return x

def compute_corr_max(h):
    """Return (full correlation vector, its maximal raw value)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr, np.max(corr)

def gradient_from_shift(h, shift):
    """∂/∂h of O(k) for the shift producing the current maximum."""
    n = h.shape[0]
    s = abs(shift)
    if s == 0:
        return 1.0 - 2.0 * h
    m = n - s
    g = np.ones_like(h)
    g[:m] -= h[s:]          # term -h[i+shift]
    g[s:] -= h[:m]          # term -h[i-shift]
    return g

def random_binary(n, total, rng):
    """Binary feasible vector with sum=total (total is integer for even n)."""
    h = np.zeros(n, dtype=float)
    cnt = int(round(total))
    idx = rng.choice(n, size=cnt, replace=False)
    h[idx] = 1.0
    return h

def init_from_initial(arr):
    """Create a binary feasible vector putting mass on the largest entries."""
    n = arr.shape[0]
    idx = np.argsort(-arr)
    h = np.zeros(n, dtype=float)
    half = n // 2
    h[idx[:half]] = 1.0
    return h

def descent(start_h, total, dx, rng, time_deadline, global_best):
    """Projected gradient descent with occasional random perturbations."""
    h = start_h.copy()
    corr, max_raw = compute_corr_max(h)
    cur_val = max_raw * dx
    best_h = h.copy()
    best_val = cur_val
    n = h.shape[0]
    it = 0
    while time.time() < time_deadline and it < 2000:
        idx_max = np.argmax(corr)
        shift = idx_max - (n - 1)
        g = gradient_from_shift(h, shift)
        lr = 0.8 / (1.0 + it * 0.02)
        trial = project_onto_box_simplex(h - lr * g, total)
        corr2, max_raw2 = compute_corr_max(trial)
        val2 = max_raw2 * dx
        if val2 < cur_val - 1e-12:
            h, corr, cur_val = trial, corr2, val2
            if val2 < best_val:
                best_val, best_h = val2, trial.copy()
        else:
            # occasional uphill acceptance
            T = 0.01 * (1.0 - it / 2000.0)
            if T > 0 and rng.random() < np.exp(-(val2 - cur_val) / T):
                h, corr, cur_val = trial, corr2, val2
        # random small transfer every ~200 steps
        if it % 200 == 199:
            i, j = rng.choice(n, size=2, replace=False)
            up = 1.0 - h[i]
            down = h[j]
            eps = min(up, down, rng.random() * 0.15)
            if eps > 0:
                tmph = h.copy()
                tmph[i] += eps
                tmph[j] -= eps
                tmph = project_onto_box_simplex(tmph, total)
                corr3, max_raw3 = compute_corr_max(tmph)
                val3 = max_raw3 * dx
                if val3 < cur_val - 1e-12:
                    h, corr, cur_val = tmph, corr3, val3
                    if val3 < best_val:
                        best_val, best_h = val3, tmph.copy()
        it += 1
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- Problem size -------------------------------------------------
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        if n % 2:
            n -= 1
            init_arr = init_arr[:n]
    except Exception:
        n = 200
        init_arr = None

    total = n / 2.0          # required sum of h
    dx = 2.0 / n

    best_h = None
    best_val = float('inf')

    # ---- Use the provided construction as first start -----------------
    if init_arr is not None:
        h0 = init_from_initial(init_arr)
        h_opt, val_opt = descent(
            h0, total, dx, rng,
            start_time + budget_s * 0.9,
            best_val
        )
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()

    # ---- Random restarts ------------------------------------------------
    while time.time() - start_time < budget_s * 0.9:
        h_start = random_binary(n, total, rng)
        h_opt, val_opt = descent(
            h_start, total, dx, rng,
            start_time + budget_s * 0.9,
            best_val
        )
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()
        if best_val <= 0.3808:
            break

    # ---- Fallback (should never happen) --------------------------------
    if best_h is None:
        best_h = random_binary(n, total, rng)
        _, max_raw = compute_corr_max(best_h)
        best_val = max_raw * dx

    return (best_h.tolist(), float(best_val), n)
