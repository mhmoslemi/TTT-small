import numpy as np
import random
import time
import math

def _project_bounds_sum(v, target):
    """
    Project vector v onto the box [0,1]^n and the hyperplane sum(v)=target.
    Uses a simple water‑filling (binary search on the Lagrange multiplier).
    """
    lower, upper = 0.0, 1.0
    lo, hi = -10.0, 10.0  # wide enough for binary search
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, lower, upper)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, lower, upper)
    # tiny correction for numerical drift
    diff = proj.sum() - target
    if abs(diff) > 1e-10:
        # distribute the small error uniformly over non‑clipped entries
        free = (proj > lower) & (proj < upper)
        if free.any():
            proj[free] -= diff / free.sum()
            proj = np.clip(proj, lower, upper)
    return proj


def _max_overlap(h):
    """Exact C5 value for a step vector h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def _subgradient(h, shift):
    """
    Sub‑gradient of the max‑overlap functional at the worst shift.
    shift is an integer in [-(N-1), N-1] (as used by np.correlate).
    """
    N = h.shape[0]
    g = np.ones_like(h)
    if shift >= 0:
        k = shift
        # term (1 - h_{i+k})
        if k < N:
            g[: N - k] -= h[k:]
        # term -h_{i-k}
        if k > 0:
            g[k:] -= h[: N - k]
    else:
        d = -shift
        # (1 - h_{i-d})
        if d < N:
            g[d:] -= h[: N - d]
        # -h_{i+d}
        if d > 0:
            g[: N - d] -= h[d:]
    return g


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (sampled on a uniform grid)
    that minimizes   C5 = max_k ∫ h(x)(1‑h(x+k))dx .
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # length of the vector – use the length of the parent construction if present
    try:
        init = np.array(initial_h_values, dtype=float)
        N = init.shape[0]
    except NameError:
        init = None
        N = 42                     # a modest size that keeps everything fast
    # ensure N is even (required for the exact sum constraint)
    if N % 2:
        N += 1
    target_sum = N / 2.0           # because ∫h = 1 ⇒ sum(h) = N/2

    # initialise h
    if init is not None and init.shape[0] == N:
        h = init.copy()
        h = _project_bounds_sum(h, target_sum)
    else:
        # random start inside the feasible polytope
        h = np.random.rand(N)
        h = _project_bounds_sum(h, target_sum)

    best_h = h.copy()
    best_c = _max_overlap(best_h)

    # parameters for simulated‑annealing‑style search
    max_iters = 600_000            # enough iterations for the budget
    T0, T_end = 0.06, 5e-6        # temperature schedule
    step0 = 0.28                   # initial step length for gradient move

    for it in range(max_iters):
        # stop when we approach the time limit
        if time.time() - start_time > budget_s * 0.95:
            break

        # current objective and worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        cur_c = corr[idx] * (2.0 / N)
        shift = idx - (N - 1)

        # compute a sub‑gradient for the worst shift
        g = _subgradient(h, shift)

        # temperature and step size (linear schedules)
        frac = it / max_iters
        T = T0 * (1.0 - frac) + T_end * frac
        step = step0 * (1.0 - frac) + 1e-5 * frac

        # proposed move: gradient descent step with projection
        cand = h - step * g
        cand = _project_bounds_sum(cand, target_sum)

        cand_c = _max_overlap(cand)

        # accept if improvement or with Metropolis probability
        if cand_c < cur_c or random.random() < math.exp(-(cand_c - cur_c) / (T + 1e-12)):
            h = cand
            cur_c = cand_c
            if cand_c < best_c:
                best_c = cand_c
                best_h = cand.copy()

        # occasional random perturbation to avoid stagnation
        if (it + 1) % 8000 == 0:
            # pick two different indices and exchange a small amount of mass
            i, j = random.sample(range(N), 2)
            delta_max = min(1.0 - h[i], h[j])
            delta_min = -min(h[i], 1.0 - h[j])
            if delta_max > delta_min:
                delta = random.uniform(delta_min, delta_max) * 0.5
                pert = h.copy()
                pert[i] += delta
                pert[j] -= delta
                pert = _project_bounds_sum(pert, target_sum)
                pert_c = _max_overlap(pert)
                if pert_c < best_c:
                    best_c, best_h = pert_c, pert.copy()
                # accept this perturbation with a small probability
                if pert_c < cur_c or random.random() < 0.02:
                    h = pert

    # final projection and exact bound
    best_h = _project_bounds_sum(best_h, target_sum)
    final_c = _max_overlap(best_h)

    return best_h, float(final_c), N
