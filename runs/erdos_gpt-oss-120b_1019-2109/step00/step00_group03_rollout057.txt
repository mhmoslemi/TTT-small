import time
import numpy as np

def _project_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the set {0<=y<=1, sum(y)=target}."""
    lo = np.min(x) - 1.0          # guarantees all clip to 1 -> sum=n
    hi = np.max(x)                # guarantees all clip to 0 -> sum=0
    for _ in range(60):           # enough for double precision
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _correlation_counts(h: np.ndarray) -> np.ndarray:
    """Linear cross‑correlation of h with (1‑h) (no dx scaling)."""
    n = h.shape[0]
    # next power of two ≥ 2*n‑1
    P = 1 << ((2 * n - 1).bit_length())
    H = np.fft.fft(h, P)
    G = np.fft.fft(1.0 - h, P)
    corr = np.fft.ifft(H * np.conj(G)).real[: 2 * n - 1]
    # Numerical noise may produce tiny negative values
    return np.maximum(corr, 0.0)


def _max_overlap_and_shift(h: np.ndarray):
    """Return (max_correlation, shift_k) for the current h."""
    corr = _correlation_counts(h)
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    k = idx - (h.shape[0] - 1)   # lag corresponding to idx
    return max_val, k

    Optimize a step function for the Erdős minimum‑overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)

    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    else:
        n = 200                         # reasonable resolution if no parent
        x = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        sigma = n / 6.0
        h = np.exp(-((x - centre) ** 2) / (2.0 * sigma * sigma))
        h = np.clip(h, 0.0, 1.0)

    target_sum = n / 2.0                # because ∫h = 1 with dx = 2/n
    h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_val, _ = _max_overlap_and_shift(best_h)
    best_c = best_val * (2.0 / n)      # convert to integral value

    start = time.time()
    # Sub‑gradient descent parameters
    lr = 0.5
    min_lr = 1e-5
    max_lr = 2.0

    while time.time() - start < budget_s * 0.94:

        cur_c = cur_val * (2.0 / n)

            h = _project_box_sum(best_h + np.random.normal(scale=0.05, size=n), target_sum)
            lr = 0.5

    # Final evaluation of the best solution found
    final_val, _ = _max_overlap_and_shift(best_h)
    final_c = final_val * (2.0 / n)

    return best_h, final_c, n
