def improve_step(h, s_max, step_size):
    grad = compute_grad_worst_shift(h, s_max)
    i_decrease = np.argmax(grad)  # most positive gradient => increase h will increase C, so we want to decrease h there
    i_increase = np.argmin(grad)  # most negative gradient => increase h will decrease C
    # ensure not same
    if i_decrease == i_increase:
        return False
    # amount we can transfer
    max_decrease = h[i_decrease]  # can't go below 0
    max_increase = 1 - h[i_increase]  # can't exceed 1
    delta = min(step_size, max_decrease, max_increase)
    if delta <= 1e-12:
        return False
    h[i_decrease] -= delta
    h[i_increase] += delta
    return True

def _project(v, target_sum):
    """Clip to [0,1] and adjust to hit the exact target sum."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target_sum) < 1e-12:
        return v
    # Simple redistribution: move excess/deficit uniformly on free entries.
    if cur > target_sum:                     # need to reduce
        excess = cur - target_sum
        free = np.where(v > 0)[0]
        while excess > 1e-12 and free.size:
            red = excess / free.size
            actual = np.minimum(red, v[free])
            v[free] -= actual
            excess -= actual.sum()
            free = np.where(v > 1e-12)[0]
    else:                                    # need to increase
        deficit = target_sum - cur
        free = np.where(v < 1)[0]
        while deficit > 1e-12 and free.size:
            inc = deficit / free.size
            actual = np.minimum(inc, 1 - v[free])
            v[free] += actual
            deficit -= actual.sum()
            free = np.where(v < 1 - 1e-12)[0]
    return np.clip(v, 0.0, 1.0)


def _c5_value(h):
    """Maximum overlap integral for a discrete step function."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max() * dx


def _corr_full(h):
    """Full correlation array (not scaled)."""
    return np.correlate(h, 1 - h, mode="full")


def _grad_worst_shift(h, shift):
    """Gradient of C(shift) = Σ h[i] (1‑h[i+shift]) w.r.t. each h[i]."""
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        # term for i in [0, n‑1‑shift]
        g[: n - shift] += 1 - h[shift:]          # d/dh[i]
        g[shift:] -= h[: n - shift]              # d/dh[i+shift]
    else:
        s = -shift
        g[s:] += 1 - h[: n - s]                  # d/dh[i]
        g[: n - s] -= h[s:]                      # d/dh[i-s]
    return g

            new_j = h[j] - delta
            if 0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0:
                h_new = h.copy()
                h_new[i] = new_i
                h_new[j] = new_j
                new_c5 = _c5_value(h_new)
                if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / max(t, 1e-12)):
                    h = h_new
                    cur_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_new.copy()
        it += 1

    # Final tidy‑up (ensure exact constraints)
    best_h = _project(best_h, target_sum)
    best_c5 = _c5_value(best_h)

    # Return as required by the harness
    return (best_h.tolist(), float(best_c5), n)
