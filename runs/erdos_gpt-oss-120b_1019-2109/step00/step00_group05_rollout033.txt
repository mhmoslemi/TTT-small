import time

def compute_c5(arr: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx approximated by discrete correlation."""
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    dx = 2.0 / arr.size
    return float(np.max(corr) * dx)


def neighbor_sum_array(h: np.ndarray, k: int) -> np.ndarray:
    """For each i compute h[i+k] + h[i‑k] (out‑of‑range terms are 0)."""
    n = h.shape[0]
    ns = np.zeros(n, dtype=np.float64)
    if k >= 0:
        if k < n:
            ns[: n - k] = h[k:]               # h[i+k]
            ns[k:] += h[: n - k]               # h[i‑k]
    else:
        kk = -k
        if kk < n:
            ns[kk:] = h[: n - kk]              # h[i+k] (k negative)
            ns[: n - kk] += h[kk:]             # h[i‑k] (k negative)
    return ns


def project_sum(arr: np.ndarray, target: float) -> np.ndarray:

        arr[idx] = np.clip(arr[idx], 0.0, 1.0)

    return arr
