def project_box_simplex(v, S):
    # Clip after shifting by lambda.
    # alpha low and high.
    low = -np.max(v)  # ensure v+low <= 0 for all

    for _ in range(50):
        mid = (low + high) / 2
        w = np.clip(v + mid, 0, 1)
        s = w.sum()
        if s < S:
            low = mid
        else:
            high = mid
    w = np.clip(v + (low+high)/2, 0, 1)

import time
import math
import numpy as np

def _project_to_binary_sum_one(vec, target):
    """Convert a binary vector to have exactly `target` ones, keeping the rest zeros."""
    cur = int(vec.sum())
    if cur > target:
        excess = cur - target
        idx = np.where(vec == 1)[0]
        if excess > 0:
            sel = np.random.choice(idx, excess, replace=False)
            vec[sel] = 0
    elif cur < target:
        deficit = target - cur
        idx = np.where(vec == 0)[0]
        if deficit > 0:
            sel = np.random.choice(idx, deficit, replace=False)
            vec[sel] = 1
    return vec

def _initial_binary(n, rng):
    """Random binary vector with exactly n//2 ones."""
    h = np.zeros(n, dtype=np.int8)
    h[: n // 2] = 1
    rng.shuffle(h)
    return h

def _a_for_shift(h, s):
    """For shift s (can be negative) compute a_i = h_{i-s}+h_{i+s} with zero padding."""
    n = h.shape[0]
    a = np.zeros(n, dtype=np.int32)
    if s >= 0:
        a[s:] += h[: n - s]
        a[: n - s] += h[s:]
    else:
        ss = -s
        a[: n - ss] += h[ss:]
        a[ss:] += h[: n - ss]
    return a

def _max_corr_count(h):
    """Return the integer count max_{k} ∫ h(x)(1−h(x+k)) dx before scaling."""
    # np.correlate works with int8, returns float; cast to int for safety
    c = np.correlate(h, 1 - h, mode="full")
    return int(c.max())
