def solve_binary_opt(N, max_mismatch):
    import numpy as np, cvxpy as cp, time
    h = cp.Variable(N, boolean=True)
    w = cp.Variable((N,N), boolean=True)
    constraints = [cp.sum(h) == N // 2]
    # product constraints
    for i in range(N):
        for j in range(N):
            constraints.append(w[i,j] <= h[i])
            constraints.append(w[i,j] <= h[j])
            constraints.append(w[i,j] >= h[i] + h[j] - 1)
    # shift constraints
    for k in range(-N+1, N):
        if k == 0:
            continue
        i_start = max(0, -k)
        i_end = min(N-1, N-1 - k)
        exprs = [h[i] - w[i, i+k] for i in range(i_start, i_end+1)]
        constraints.append(cp.sum(exprs) <= max_mismatch)
    # objective dummy
    prob = cp.Problem(cp.Minimize(0), constraints)
    prob.solve(solver=cp.CBC, verbose=False)
    return h.value, prob.status
```

def solve_binary_opt(N, max_mismatch, time_limit=10.0):
    import cvxpy as cp, numpy as np, time
    h = cp.Variable(N, boolean=True)
    w = cp.Variable((N,N), boolean=True)
    constraints = [cp.sum(h) == N // 2]
    # product constraints
    for i in range(N):
        for j in range(N):
            constraints.append(w[i,j] <= h[i])
            constraints.append(w[i,j] <= h[j])
            constraints.append(w[i,j] >= h[i] + h[j] - 1)
    # shift constraints
    for k in range(-N+1, N):
        if k == 0:
            continue
        i_start = max(0, -k)
        i_end = min(N-1, N-1 - k)
        exprs = []
        for i in range(i_start, i_end+1):
            exprs.append(h[i] - w[i, i+k])
        constraints.append(cp.sum(exprs) <= max_mismatch)
    prob = cp.Problem(cp.Minimize(0), constraints)
    # Use time limit
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, warm_start=False, verbose=False)
    except TypeError:
        # maybe different argument name
        prob.solve(solver=cp.CBC, msg=False, maxClique=False)
    return h.value, prob.status
```

def solve_binary_feasible(N, max_mismatch, time_limit=60):
    import cvxpy as cp, numpy as np
    h = cp.Variable(N, boolean=True)
    w = cp.Variable((N,N), boolean=True)
    constraints = [cp.sum(h) == N // 2]
    # product constraints
    for i in range(N):
        for j in range(N):
            constraints.append(w[i,j] <= h[i])
            constraints.append(w[i,j] <= h[j])
            constraints.append(w[i,j] >= h[i] + h[j] - 1)

    for k in range(-N+1, N):
        if k == 0:

        i_start = max(0, -k)
        i_end = min(N-1, N-1 - k)

        constraints.append(cp.sum(expr) <= max_mismatch)

        # fallback without time limit
        prob.solve(solver=cp.CBC, verbose=False)
    if prob.status.lower() in ('optimal', 'optimal_inaccurate'):
        return h.value, prob.status
    else:
        # maybe solution but infeasible
        if h.value is not None:

import time
import numpy as np

def _max_mismatch(h):
    """Return the maximum overlap count  Σ h[i]*(1-h[i+k]) over all integer shifts."""
    # h is a 0/1 NumPy array
    corr = np.correlate(h, 1 - h, mode="full")
    return int(corr.max())

def _c5_from_mismatch(mismatch, n_points):
    """Convert a mismatch count to the C5 value."""
    dx = 2.0 / n_points
    return mismatch * dx

def _solve_binary_feasible(n, max_mismatch, time_limit):
    """
    Try to find a binary vector of length n with exactly n//2 ones
    such that for every shift k the mismatch count is ≤ max_mismatch.
    Returns (h_values, status) where h_values is a NumPy array of floats
    (may contain small rounding errors) and status is the cvxpy status string.
    """
    import cvxpy as cp

    # binary variables for the set
    h = cp.Variable(n, boolean=True)

    # auxiliary variables for products w[i,j] = h[i] * h[j]
    w = cp.Variable((n, n), boolean=True)

    constraints = [cp.sum(h) == n // 2]

    # linearisation of the product
    for i in range(n):
        hi = h[i]
        for j in range(n):
            constraints.append(w[i, j] <= hi)
            constraints.append(w[i, j] <= h[j])
            constraints.append(w[i, j] >= hi + h[j] - 1)

    # mismatch constraints for every non‑zero shift
    for k in range(-n + 1, n):
        if k == 0:
            continue
        i_start = max(0, -k)
        i_end = min(n - 1, n - 1 - k)
        expr = []
        for i in range(i_start, i_end + 1):
            expr.append(h[i] - w[i, i + k])
        constraints.append(cp.sum(expr) <= max_mismatch)

    # dummy objective – feasibility problem
    prob = cp.Problem(cp.Minimize(0), constraints)

    # solve with a time limit (solver‑specific keyword)
    solve_kwargs = dict(solver=cp.CBC, verbose=False)
    # most cvxpy back‑ends accept max_seconds; ignore if unsupported
    try:
        solve_kwargs["max_seconds"] = time_limit
    except Exception:
        pass

    prob.solve(**solve_kwargs)

    # status handling
    status = prob.status
    if h.value is None:
        return None, status
    return np.array(h.value).reshape(-1), status

    mismatch_target = int(np.floor(target_c5 * n_points / 2 + 1e-9))

    # ------------------------------------------------------------
    # First try a direct MILP feasibility search
    # ------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:  # give the solver a reasonable slice of time
        try:
            h_candidate, status = _solve_binary_feasible(
                n_points, mismatch_target, time_limit=min(remaining, 300)
            )
            if h_candidate is not None and status.lower().startswith("optimal"):
                # round to genuine 0/1 values
                h_bin = (h_candidate > 0.5).astype(float)
                # sanity check (occasionally solver may return slight violations)
                if abs(h_bin.sum() - n_points / 2) < 1e-6:
                    m = _max_mismatch(h_bin)
                    c5_val = _c5_from_mismatch(m, n_points)
                    if c5_val <= target_c5 + 1e-9:
                        return (h_bin.tolist(), float(c5_val), n_points)
        except Exception:
            pass  # fall back to heuristic if MILP fails

    # ------------------------------------------------------------
    # Heuristic local‑search (random restarts + greedy swaps)
    # ------------------------------------------------------------
    best_h = None
    best_score = np.iinfo(np.int32).max
    dx = 2.0 / n_points

    # Helper to evaluate a candidate quickly
    def evaluate(vec):
        return _max_mismatch(vec)

    # Main search loop – stop a bit before the deadline
    while time.time() - start_time < budget_s * 0.93:
        # random start respecting the exact ½ density
        h = np.zeros(n_points, dtype=int)
        ones_idx = rng.choice(n_points, n_points // 2, replace=False)
        h[ones_idx] = 1
        cur_score = evaluate(h)

        # greedy improvement phase
        improved = True
        while improved and time.time() - start_time < budget_s * 0.92:
            improved = False
            # limit the number of random attempts per improvement round
            for _ in range(200):
                i_one = rng.choice(np.where(h == 1)[0])
                i_zero = rng.choice(np.where(h == 0)[0])
                h_new = h.copy()
                h_new[i_one] = 0
                h_new[i_zero] = 1
                new_score = evaluate(h_new)
                if new_score < cur_score:
                    h = h_new
                    cur_score = new_score
                    improved = True
                    break

        if cur_score < best_score:
            best_score = cur_score
            best_h = h.copy()
            if best_score <= mismatch_target:
                break  # reached the desired bound

    # If we never found any candidate (extremely unlikely), fall back to init or zeros
    if best_h is None:
        if init is not None:
            best_h = init.astype(float)
            best_score = evaluate(best_h)
        else:
            best_h = np.ones(n_points // 2, dtype=float)
            best_h = np.concatenate([best_h, np.zeros(n_points // 2, dtype=float)])
            best_score = evaluate(best_h)

    # Final C5 value
    final_c5 = _c5_from_mismatch(best_score, n_points)
    return (best_h.astype(float).tolist(), float(final_c5), n_points)
