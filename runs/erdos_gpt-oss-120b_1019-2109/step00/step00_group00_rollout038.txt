import numpy as np, time

def compute_c5(h):
    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    return conv.max() * (2.0 / n)

def subgradient_max(h):
    n = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    max_val = conv.max()
    # tolerance
    idxs = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    for idx in idxs:
        s = idx - (n - 1)
        if s >= 0:
            grad[:n - s] += 1 - h[s:]
            grad[s:] -= h[:n - s]
        else:
            s_abs = -s
            grad[s_abs:] += 1 - h[:n - s_abs]
            grad[:n - s_abs] -= h[s_abs:]
    return grad, max_val

def project_to_bounds_sum(v, target, tol=1e-9, max_iter=60):
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        a = np.clip(v + lam, 0.0, 1.0)
        s = a.sum()
        if abs(s - target) < tol:
            return a
        if s > target:
            hi = lam
        else:
            lo = lam
    a = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    return a

def improve_h(h0, rng, max_iter, step0):
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_val = compute_c5(h0)
    h = h0.copy()
    cur_val = best_val
    for it in range(max_iter):
        grad, max_val = subgradient_max(h)
        # step schedule
        step = step0 * (1.0 - it / max_iter) # decreasing

            cur_val = new_val
        # Keep global best
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()
    return best_h, best_val

import time, math, numpy as np

def _c5_of_h(h):
    """return max_k ∫ h(x)*(1‑h(x+k)) dx  (discrete version)"""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * (2.0 / n)

def _subgrad_max(h):
    """subgradient of the max‑correlation term and the current max value."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    mx = conv.max()
    # indices where the max is attained (within tolerance)
    idxs = np.where(conv >= mx - 1e-12)[0]
    g = np.zeros_like(h)
    for idx in idxs:
        s = idx - (n - 1)          # integer shift
        if s >= 0:
            # i in [0, n‑s‑1]
            g[:n - s] += 1.0 - h[s:]
            g[s:]    -= h[:n - s]
        else:
            s = -s
            # i in [s, n‑1]
            g[s:]    += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    return g, mx

def _project_to_sum_and_bounds(v, target, tol=1e-10, maxit=60):
    """
    Project vector v onto the set {x | sum(x)=target, 0≤x_i≤1}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0          # ensures v+λ ≤ 0
    hi = 1.0 - np.min(v) + 1.0     # ensures v+λ ≥ 1
    for _ in range(maxit):
        lam = (lo + hi) * 0.5
        a = np.clip(v + lam, 0.0, 1.0)
        s = a.sum()
        if abs(s - target) <= tol:
            return a
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    #  problem size – a modest number of points gives a good speed/quality
    # ------------------------------------------------------------------
    n_points = 300                      # can be changed, must be ≤1000
    target_sum = n_points / 2.0         # because ∫h = 1 and length = 2

    # ------------------------------------------------------------------
    #  initial construction
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        src = np.asarray(initial_h_values, dtype=np.float64)
        xs_src = np.linspace(0.0, 2.0, src.size)
        xs = np.linspace(0.0, 2.0, n_points)
        h = np.interp(xs, xs_src, src)
        h = _project_to_sum_and_bounds(h, target_sum)
    else:
        # smooth sinusoidal start (∫ sin² = 1, max = 1)
        xs = np.linspace(0.0, 2.0, n_points, endpoint=False)
        h = np.sin(np.pi * xs / 2.0) ** 2
        h = _project_to_sum_and_bounds(h, target_sum)

    best_h = h.copy()
    best_val = _c5_of_h(best_h)

    # ------------------------------------------------------------------
    #  optimisation – sub‑gradient descent with projection.
    # ------------------------------------------------------------------
    # parameters
    max_total_iters = 200000
    step0 = 0.5            # initial step magnitude
    decay = 0.999          # multiplicative decay per iteration
    temp = 0.02            # small probability for uphill moves

    cur_h = h.copy()
    cur_val = best_val

    it = 0
    while it < max_total_iters and (time.time() - start) < (budget_s - 2):
        grad, cur_mx = _subgrad_max(cur_h)
        # decreasing step size
        step = step0 * (decay ** it)
        cand = cur_h - step * grad
        cand = _project_to_sum_and_bounds(cand, target_sum)

        cand_val = _c5_of_h(cand)

        # accept if improvement, otherwise occasional uphill moves
        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / temp):
            cur_h = cand
            cur_val = cand_val

        if cand_val < best_val:
            best_val = cand_val
            best_h = cand.copy()
            # early exit if we already beat the target record
            if best_val <= 0.3808:

        # occasional random kick to escape plateaus
        if it % 5000 == 0 and it > 0:

            cur_h = _project_to_sum_and_bounds(cur_h + kick, target_sum)
            cur_val = _c5_of_h(cur_h)

        it += 1

    # ------------------------------------------------------------------
    #  output
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_val), n_points)
