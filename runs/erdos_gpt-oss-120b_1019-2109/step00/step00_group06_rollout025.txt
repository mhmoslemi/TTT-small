    corr = np.correlate(h, 1 - h, mode='full')

        g[:ls] = 1 - h[shift:]
        g[shift:] -= h[:ls]
    else:
        s = -shift
        ls = n - s
        g[s:] = 1 - h[:ls]
        g[:ls] -= h[s:]
    return shift, g

def enforce_sum(h, target_sum):

    h = np.clip(h, 0.0, 1.0)

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return dx * float(np.max(corr))


def worst_shift_and_gradient(h: np.ndarray):
    """Find the shift giving the maximal overlap and its gradient."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # zero shift corresponds to index n‑1
    g = np.zeros_like(h)
    if shift >= 0:
        ls = n - shift
        g[:ls] = 1 - h[shift:]
        g[shift:] -= h[:ls]
    else:
        s = -shift
        ls = n - s
        g[s:] = 1 - h[:ls]
        g[:ls] -= h[s:]
    return shift, g


def enforce_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust so that sum(h) == target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - float(h.sum())
    if abs(diff) <= 1e-12:
        return h
    # try to fix using the element with most headroom
    if diff > 0:
        idx = int(np.argmax(1 - h))
        add = min(diff, 1 - h[idx])
        h[idx] += add
        diff -= add
    else:
        idx = int(np.argmax(h))
        sub = min(-diff, h[idx])
        h[idx] -= sub
        diff += sub
    # distribute any remaining tiny error uniformly
    if abs(diff) > 1e-12:
        h += diff / h.size
        h = np.clip(h, 0.0, 1.0)
        residual = target - float(h.sum())
        if abs(residual) > 1e-12:
            if residual > 0:
                idx = int(np.argmax(1 - h))
                h[idx] = min(1.0, h[idx] + residual)
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] + residual)
    return h


def random_feasible(n: int, target: float, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector with values in [0,1] and sum == target."""
    h = np.zeros(n)
    int_part = int(math.floor(target))
    frac = target - int_part
    perm = rng.permutation(n)
    if int_part > 0:
        h[perm[:int_part]] = 1.0
    if frac > 1e-12:
        h[perm[int_part]] = frac
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ---------------------------------------------------------
    try:
        init_h = np.asarray(initial_h_values, dtype=float).copy()
        n = int(init_h.shape[0])
    except Exception:
        n = 200                     # fallback size (even → integer target)
        init_h = random_feasible(n, n / 2.0, rng)

    target_sum = n / 2.0
    # make sure the starting vector is feasible
    cur_sum = float(init_h.sum())
    if abs(cur_sum - target_sum) > 1e-9:
        init_h = init_h * (target_sum / cur_sum)
        init_h = enforce_sum(init_h, target_sum)

    cur_h = init_h.copy()
    cur_c5 = compute_c5(cur_h)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ---- simulated‑annealing / hill climbing parameters --------------------
    total_iters = 180_000
    T_start = 0.02
    T_end = 1e-6
    prob_grad = 0.12               # chance of taking a gradient step
    delta_max = 0.4                # maximal mass moved in a transfer step

    iteration = 0
    while True:
        # budget check
        if time.time() - start > budget_s * 0.96:
            break
        if iteration >= total_iters:
            break

        # temperature schedule (geometric decay)
        frac = iteration / total_iters
        T = T_start * (T_end / T_start) ** frac

        if rng.random() < prob_grad:
            # ---- gradient descent on the worst shift -----------------------
            _, g = worst_shift_and_gradient(cur_h)
            step = 0.05 * (0.9 ** (iteration // 10000))
            new_h = cur_h - step * g
            new_h = enforce_sum(new_h, target_sum)
        else:
            # ---- random mass transfer ------------------------------------
            i = rng.integers(n)
            if cur_h[i] <= 1e-12:
                iteration += 1
                continue
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            max_delta = min(delta_max, cur_h[i], 1 - cur_h[j])
            if max_delta <= 0:
                iteration += 1
                continue
            delta = rng.random() * max_delta
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_c5 = compute_c5(new_h)
        delta_val = new_c5 - cur_c5

        # Metropolis acceptance
        if delta_val <= 0 or rng.random() < math.exp(-delta_val / max(T, 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5

        # keep the global best
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()

        # occasional random restart to diversify search
        if iteration and iteration % 25000 == 0:
            if rng.random() < 0.5:
                cur_h = best_h.copy()
                cur_c5 = best_c5
            else:
                cur_h = random_feasible(n, target_sum, rng)
                cur_c5 = compute_c5(cur_h)

        iteration += 1

    # final post‑processing
    best_h = enforce_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return (best_h.tolist(), float(final_c5), n)
