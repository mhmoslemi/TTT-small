def compute_gradient_min_overlap(h, selected_lags):
    n = len(h)
    grad = np.zeros_like(h)
    for L in selected_lags:
        if L == 0:
            grad += 2 * h
        else:
            grad[:n - L] += h[L:]
            grad[L:] += h[:n - L]
    grad /= len(selected_lags)
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time
    np.random.seed(seed)
    start = time.time()
    # Determine n_points
    n = 200
    total = n/2.0
    # Initial vector
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n0 = len(init)
        if n0 != n:
            # interpolate
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            v = np.interp(x_new, x0, init)
        else:
            v = init.copy()
    else:
        v = np.random.rand(n)
    # Project onto feasible
    h = project_to_sum_and_bounds(v, total)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # main loop
    iter_count = 0
    eps_lag = 1e-12
    while time.time() - start < budget_s * 0.95:
        # compute self-correlation
        corr_hh = np.correlate(h, h, mode='full')
        mid = len(corr_hh)//2
        corr_pos = corr_hh[mid:] # lags 0..n-1
        min_val = corr_pos.min()
        # tolerance for selecting near-min
        # dot: tolerance relative to max?
        # Use absolute tolerance
        sel = np.where(corr_pos <= min_val + eps_lag)[0]
        if sel.size == 0:
            sel = np.array([np.argmin(corr_pos)])
        # compute gradient
        grad = compute_gradient_min_overlap(h, sel)
        # step size schedule
        alpha = 0.5 / (1.0 + 0.001 * iter_count) # decays slowly
        # optional random perturbation
        if iter_count % 500 == 0 and iter_count > 0:
            # Add small noise
            noise_scale = max(0.01, 0.2/(np.sqrt(iter_count))) # decreasing
            h = h + np.random.normal(scale=noise_scale, size=n)

            # Ensure sum
            h = project_to_sum_and_bounds(h, total)

        h_new = h + alpha * grad

        h_new = project_to_sum_and_bounds(h_new, total)
        # compute C5
        c5 = compute_c5(h_new)

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        # Update current
        h = h_new
        iter_count += 1
        # check stagnation: if α becomes very small break? maybe continue.

import numpy as np
import time

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_sum_and_bounds(v: np.ndarray, total: float) -> np.ndarray:
    """
    Project vector v onto the feasible set:
        0 ≤ h[i] ≤ 1  and  sum(h) = total.
    Uses a bisection on the Lagrange multiplier λ for the hyperplane.
    """
    lo = -np.max(v) - 10.0
    hi = np.max(v) + 10.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    return h

def _gradient_min_overlap(h: np.ndarray, lags: np.ndarray) -> np.ndarray:
    """Gradient of the overlap sum ∑ h_i h_{i+lag} for a set of lags."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    for L in lags:
        if L == 0:
            grad += 2.0 * h
        else:
            grad[: n - L] += h[L:]
            grad[L:] += h[: n - L]
    grad /= float(lags.size)
    return grad

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a low Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Choose the discretisation size
    # ------------------------------------------------------------------
    n = 200                     # number of points (feel free to change)
    half_total = n / 2.0        # required integral = 1  →  sum(h)=n/2

    # ------------------------------------------------------------------
    # 2. Initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.size
        if n0 != n:
            # linear interpolation to the new grid
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            v = np.interp(x_new, x_old, init)
        else:
            v = init.copy()
    else:
        v = np.random.rand(n)

    h = _project_to_sum_and_bounds(v, half_total)
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (projected sub‑gradient ascent)
    # ------------------------------------------------------------------
    iteration = 0
    eps_lag = 1e-12
    # schedule parameters
    step0 = 0.5
    decay = 0.001

    while time.time() - start_time < budget_s * 0.95:
        # autocorrelation of h with itself
        corr_self = np.correlate(h, h, mode="full")
        mid = corr_self.size // 2
        corr_pos = corr_self[mid:]               # lags 0 … n‑1
        min_val = corr_pos.min()

        # lags that are (practically) minimal
        sel_lags = np.where(corr_pos <= min_val + eps_lag)[0]
        if sel_lags.size == 0:                   # safety fallback
            sel_lags = np.array([np.argmin(corr_pos)], dtype=int)

        # gradient that lifts the minimal overlaps
        grad = _gradient_min_overlap(h, sel_lags)

        # simple step‑size schedule
        alpha = step0 / (1.0 + decay * iteration)

        # occasional random perturbation to escape flat regions
        if iteration % 500 == 0 and iteration > 0:
            noise = np.random.normal(scale=0.05, size=n)
            h = np.clip(h + noise, 0.0, 1.0)
            h = _project_to_sum_and_bounds(h, half_total)

        # projected ascent step
        h_new = h + alpha * grad
        h_new = _project_to_sum_and_bounds(h_new, half_total)

        # evaluate C5
        cur_c5 = _compute_c5(h_new)

        # keep the best ever seen
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h_new.copy()

        h = h_new
        iteration += 1

    # ------------------------------------------------------------------
    # 4. Optional binary rounding (keeps the integral exact)
    # ------------------------------------------------------------------
    # rounding may occasionally give a still‑better C5
    flat = best_h.copy()
    k = int(round(half_total))
    idx = np.argsort(-flat)                # descending order
    bin_vec = np.zeros_like(flat)
    bin_vec[idx[:k]] = 1.0
    bin_c5 = _compute_c5(bin_vec)
    if bin_c5 < best_c5:
        best_c5 = bin_c5
        best_h = bin_vec.copy()

    # ------------------------------------------------------------------
    # 5. Return result
    # ------------------------------------------------------------------
    return (best_h, float(best_c5), n)
