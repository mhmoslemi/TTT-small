def project_box_sum(h, target, low=0.0, high=1.0, max_iter=100):
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        mid = (lo + hi)/2
        h_proj = np.clip(h + mid, low, high)
        s = h_proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h_proj = np.clip(h + (lo+hi)/2, low, high)
    # due to numeric, adjust any small diff:
    diff = target - h_proj.sum()
    if abs(diff) > 1e-8:
        # distribute diff among non-clipped entries (if any)
        # Find indices where low < h_proj < high
        mask = (h_proj > low+1e-12) & (h_proj < high-1e-12)
        if mask.any():
            h_proj[mask] += diff / mask.sum()
            h_proj = np.clip(h_proj, low, high)
    return h_proj
```

    # binary search lambda
    lo, hi = -max(abs(x).max() - low, high - x.min()), max(abs(x).max() - high, x.max() - low)

        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, low, high)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(x + (lo + hi) / 2.0, low, high)
    # final adjustment if sum not accurate due to clipping at extremes (should be accurate)

    diff = target - proj.sum()

            proj = np.clip(proj, low, high)
        else:
            # all at bounds, adjust by scaling
            proj = proj * (target / proj.sum())
    return proj

import numpy as np, time, math

def compute_c5_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def compute_c5(h):
    try:
        return evaluate_erdos_solution(h)
    except NameError:
        return compute_c5_vec(h)

def project_box_sum(x, target, low=0.0, high=1.0, max_iter=60):
    lo = -10.0
    hi = 10.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        proj = np.clip(x + mid, low, high)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(x + (lo + hi) / 2.0, low, high)
    # small correction
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > low + 1e-10) & (proj < high - 1e-10)
        if mask.any():
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, low, high)
        else:
            # all at bounds, normalize
            proj = proj * (target / proj.sum())
    return proj

def gradient_step(h):
    n = len(h)
    corr = np.correlate(h, h, mode='full')
    idx = int(np.argmin(corr))
    shift = idx - (n - 1)
    grad = np.zeros_like(h)
    if shift >= 0:
        k = shift
        grad[:n - k] += h[k:]
        grad[k:] += h[:n - k]
    else:
        k = -shift
        grad[k:] += h[:n - k]
        grad[:n - k] += h[k:]
    return grad

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    # Determine base size
    try:

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    max_iter = 200000
    # parameters
    max_temp = 0.2
    for it in range(max_iter):
        elapsed = time.time() - start
        if elapsed >= budget_s * 0.95:
            break
        # temperature schedule
        temp = max_temp * (1 - it / max_iter)
        if np.random.rand() < 0.7:
            # gradient based move
            grad = gradient_step(h)
            # scale step size adaptively
            step_scale = 0.05 * (1 + 0.5 * np.random.randn())
            if step_scale < 0:
                step_scale = -step_scale
            cand = h + step_scale * grad
        else:
            # random perturbation
            sigma = 0.15 * (1 - it / max_iter)
            cand = h + sigma * np.random.randn(n)
        # project to feasible set
        cand = project_box_sum(cand, target, 0.0, 1.0)
        cand_c5 = compute_c5(cand)

                best_h = cand.copy()

            rand = project_box_sum(rand, target, 0.0, 1.0)

            if rand_c5 < best_c5:

                best_c5 = rand_c5

import time, math
import numpy as np

def _c5_value(h):
    """Return C5 for a step vector h."""
    try:
        # use the harness provided evaluator if present
        return evaluate_erdos_solution(h)
    except NameError:
        n = len(h)
        dx = 2.0 / n
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx


def _project_box_sum(x, target, low=0.0, high=1.0, iters=60):
    """
    Project vector x onto the feasible set:
        low <= y_i <= high  and  sum(y) = target
    via a simple bisection on a common shift λ.
    """
    lo, hi = -1e4, 1e4
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, low, high)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) * 0.5, low, high)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        mask = (y > low + 1e-12) & (y < high - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
            y = np.clip(y, low, high)
        else:
            # all entries on a bound – just scale (should be rare)
            y = y * (target / y.sum())
    return y


def _gradient_min_corr(h):
    """Gradient that increases the minimal autocorrelation of h."""
    n = len(h)
    corr = np.correlate(h, h, mode='full')
    idx = int(np.argmin(corr))
    shift = idx - (n - 1)          # signed shift achieving min correlation
    grad = np.zeros_like(h)
    if shift >= 0:
        k = shift
        grad[: n - k] += h[k:]
        grad[k:]      += h[: n - k]
    else:
        k = -shift
        grad[k:]      += h[: n - k]
        grad[: n - k] += h[k:]
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] (discretised) to minimise
    C5 = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)

    # initialise -------------------------------------------------------------
    try:
        # parent construction may be supplied
        h0 = np.array(initial_h_values, dtype=float)
        n = len(h0)
    except NameError:
        # no parent – use a modest even size for easier sum handling
        n = 120
        h0 = np.full(n, 0.5, dtype=float)

    target = n / 2.0                         # because ∫h = 1  ⇒ Σh·dx = 1
    h = _project_box_sum(h0, target)        # ensure feasibility
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    start = time.time()
    max_iter = 200000
    max_temp = 0.20                           # initial simulated‑annealing temperature

    for it in range(max_iter):
        # time guard
        if (time.time() - start) >= budget_s * 0.95:
            break

        # temperature schedule (linear cooling)
        temp = max_temp * (1.0 - it / max_iter)

        # propose a new candidate ------------------------------------------------
        if np.random.rand() < 0.7:
            # gradient step targeting the worst shift
            grad = _gradient_min_corr(h)
            # randomised step length, modest magnitude
            step = 0.05 * (1.0 + 0.3 * np.random.randn())
            if step < 0:
                step = -step
            cand = h + step * grad
        else:
            # occasional random jitter to escape plateaus
            sigma = 0.12 * (1.0 - it / max_iter)
            cand = h + sigma * np.random.randn(n)

        cand = _project_box_sum(cand, target)

        cand_c5 = _c5_value(cand)

        # simulated‑annealing acceptance (compare with global best)
        if cand_c5 < best_c5 or np.random.rand() < math.exp(-(cand_c5 - best_c5) / max(temp, 1e-12)):
            h = cand
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand.copy()

        # occasional full random restart (keeps the search diverse)
        if (it + 1) % 20000 == 0:

            rnd_c5 = _c5_value(rnd)
            if rnd_c5 < best_c5:
                best_h = rnd.copy()
                best_c5 = rnd_c5
                h = rnd.copy()

    # final evaluation ---------------------------------------------------------
    final_c5 = _c5_value(best_h)
    return (best_h.tolist(), float(final_c5), n)
