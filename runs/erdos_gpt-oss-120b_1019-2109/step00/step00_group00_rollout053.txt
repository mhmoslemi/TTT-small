def gradient_for_shift(h, s):

    g = np.zeros_like(h)
    # first term: (1 - h[i+s]) for i such that i+s < n
    if s >= 0:

        g[i] += 1 - h[i + s]
        # second term: -h[i - s] for i such that i - s >=0 => i >= s
        i2 = np.arange(s, n)
        g[i2] -= h[i2 - s]

def project_onto_constraints(v, target):
    # Use water-filling with bounds 0 and 1
    # Returns h satisfying sum = target, 0<=h<=1.
    n = v.shape[0]
    # Clip initial values to [0,1] for bounds determination.
    lo = -1e5
    hi = 1e5
    for _ in range(60):
        mid = (lo + hi)/2
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # Adjust possible small rounding error
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # distribute diff among non-bound elements
        free = (h > 0) & (h < 1)
        cnt = free.sum()
        if cnt > 0:
            # adjust each free element equally
            adjust = diff / cnt
            h[free] += adjust
            # Clip again
            h = np.clip(h, 0, 1)
        else:
            # all at bounds; adjust smallest distance to bounds
            # fallback: adjust by small increments
            h += diff / n
            h = np.clip(h, 0, 1)
    return h

import time
import numpy as np

def _project_onto_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {h | 0<=h<=1, sum(h)=target_sum}."""
    # Use water-filling: find lambda so that sum(clip(v-lambda,0,1)) = target_sum
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # Small residual correction (unlikely needed)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        cnt = int(free.sum())
        if cnt:
            proj[free] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
        else:
            proj += diff / proj.size
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def _compute_c5(h: np.ndarray) -> float:
    """Compute the C5 value for a given discretised h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Construct a step function achieving C5 = 3/8 = 0.375."""
    # The construction is h(x) = (1 + cos(pi x)) / 2 on [0,2].
    # We sample it at mid‑points of a uniform grid.
    n = 200                     # even number gives exact integral constraint
    dx = 2.0 / n
    x_mid = (np.arange(n) + 0.5) * dx
    h = 0.5 + 0.5 * np.cos(np.pi * x_mid)   # values lie in [0,1]

    # Ensure the exact linear constraints (numerical round‑off can be removed)
    target = n / 2.0
    h = _project_onto_constraints(h, target)

    c5 = _compute_c5(h)

    return h.tolist(), c5, n
