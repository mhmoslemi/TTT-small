import time, math
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretized) that minimizes
    C5 = max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        n_points = h0.size
        h = h0.copy()
    else:
        n_points = 45
        h = np.zeros(n_points, dtype=np.float64)
        half = n_points // 2
        h[:half] = 1.0
        if n_points % 2:
            h[half] = 0.5
        rng.shuffle(h)

    target_sum = n_points / 2.0
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Adjust a random entry to hit the exact total
        for _ in range(100):
            idx = rng.randint(n_points)
            new_val = h[idx] + (target_sum - cur_sum)
            if 0.0 <= new_val <= 1.0:
                h[idx] = new_val
                break
        else:
            # fallback: uniform correction
            correction = (target_sum - cur_sum) / n_points
            h = np.clip(h + correction, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)

    dx = 2.0 / n_points

    def evaluate(arr):
        """Return C5 for a given array."""
        conv = np.correlate(arr, 1.0 - arr, mode='full') * dx
        return conv.max()

    # ------------------------------------------------------------------
    # Simulated annealing / hill climbing
    # ------------------------------------------------------------------
    current_h = h.copy()
    current_c5 = evaluate(current_h)

    best_h = current_h.copy()
    best_c5 = current_c5

    start_time = time.time()
    max_step = 0.2          # maximum mass transferred in one move
    T0 = max(0.001, best_c5 * 0.02)   # initial temperature (scale with current value)

    it = 0
    while True:
        now = time.time()
        if now - start_time > budget_s - 0.2:
            break
        it += 1

        # gradually shrink step size
        if it % 20000 == 0:
            max_step = max(0.005, max_step * 0.9)

        # pick two distinct indices
        i, j = rng.choice(n_points, size=2, replace=False)

        # decide direction of transfer (donor -> receiver)
        if rng.rand() < 0.5:
            donor, receiver = i, j
        else:
            donor, receiver = j, i

        # maximal transferrable epsilon respecting bounds and max_step
        eps_upper = min(max_step, current_h[donor], 1.0 - current_h[receiver])
        if eps_upper <= 0.0:
            continue
        eps = rng.rand() * eps_upper

        # propose new state
        new_h = current_h.copy()
        new_h[donor] -= eps
        new_h[receiver] += eps

        # evaluate
        new_c5 = evaluate(new_h)

        # temperature schedule (linear decay)
        elapsed = now - start_time
        T = T0 * (1.0 - elapsed / (budget_s - 0.2))
        T = max(T, 1e-12)

        # acceptance criterion (Metropolis)
        if new_c5 <= current_c5 or rng.rand() < math.exp(-(new_c5 - current_c5) / T):
            current_h = new_h
            current_c5 = new_c5
            # update best if improved
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()

    # final verification / cleanup
    final_c5 = evaluate(best_h)
    # enforce exact mass (tiny drift possible)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        # adjust the first element that can absorb the difference without leaving bounds
        for idx in range(n_points):
            cand = best_h[idx] + diff
            if 0.0 <= cand <= 1.0:
                best_h[idx] = cand
                break
    best_h = np.clip(best_h, 0.0, 1.0)

    return (best_h, float(final_c5), n_points)
