import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Construct a simple triangular shape h(x) = min(x, 2-x) on [0,2],
    sampled uniformly with a midpoint rule.  This satisfies
        ∫_0^2 h = 1   and   0 ≤ h ≤ 1,
    and yields a very low overlap constant (≈1/3).

    Returns:
        h_values : np.ndarray  – sampled values of h
        c5_bound : float       – computed max overlap integral
        n_points : int         – number of samples
    """
    # Number of samples – choose a modest size for good accuracy but low cost.
    n_points = 401                     # any odd number works well (≈1/3 bound)
    dx = 2.0 / n_points

    # Midpoint positions of the subintervals on [0,2]
    x = (np.arange(n_points) + 0.5) * dx

    # Triangular (tent) shape: h(x) = min(x, 2-x), reaches 1 at x=1
    h = np.minimum(x, 2.0 - x)

    # Clip just in case of round‑off; ensures 0 ≤ h ≤ 1.
    np.clip(h, 0.0, 1.0, out=h)

    # The midpoint rule integrates linear pieces exactly, so the sum constraint
    # holds to machine precision.  Adjust tiny drift if it ever appears.
    target_sum = n_points / 2.0
    current_sum = h.sum()
    if not np.isclose(current_sum, target_sum, atol=1e-12):
        # Uniformly distribute the discrepancy while keeping bounds.
        correction = (target_sum - current_sum) / n_points
        h += correction
        np.clip(h, 0.0, 1.0, out=h)

    # Compute the overlap constant C5 = max_k ∫ h(x) (1-h(x+k)) dx
    # Using the discrete definition with midpoint rule.
    c5_bound = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    return h, float(c5_bound), n_points
