import numpy as np
import time
import math
import random

def _c5_value(h):
    """return the overlap bound C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) using zero padding (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_capped_simplex(v, target_sum, ub=1.0, tol=1e-12, max_iter=200):
    """
    Project vector v onto the set {x: 0 <= x_i <= ub, sum(x)=target_sum}.
    This is the classic capped‑simplex projection via a Lagrange multiplier.
    """
    v = np.asarray(v, dtype=np.float64)

    # initial bounds for the multiplier lambda
    lo = np.min(v) - ub
    hi = np.max(v)

    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, ub)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break

    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, ub)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-10:
        # distribute the tiny remainder over non‑binding coordinates
        free = (x > 0.0) & (x < ub)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, ub)
        else:
            # fallback: uniform correction (should be negligible)
            x += diff / len(x)
            x = np.clip(x, 0.0, ub)
    return x


def _grad_step(h, step, target_sum):
    """
    Perform a single gradient‑descent step on the max‑overlap objective.
    Returns a new feasible vector and the resulting C5 value.
    """
    n = h.shape[0]
    # correlation to find the active shift
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)                 # integer shift (could be negative)

    g = np.zeros_like(h)

    if shift >= 0:
        # forward part: (1 - h[i+shift]) for i where i+shift < n
        g[: n - shift] += 1.0 - h[shift:]
        # backward part: -h[i-shift] for i where i-shift >= 0
        g[shift:] -= h[: n - shift]
    else:
        t = -shift
        # forward in the opposite direction
        g[t:] += 1.0 - h[: n - t]
        g[: n - t] -= h[t:]

    # gradient descent move   (note the minus sign because we minimise the max)
    h_raw = h - step * g
    h_proj = _project_capped_simplex(h_raw, target_sum, ub=1.0)
    return h_proj, _c5_value(h_proj)


def _local_hill_climb(start_h, target_sum, time_limit, start_time):
    """
    Simple greedy hill climbing: repeatedly transfer a small amount of mass
    between two coordinates while keeping the sum fixed.
    """
    n = start_h.shape[0]
    cur = start_h.copy()
    cur_val = _c5_value(cur)
    best = cur.copy()
    best_val = cur_val

    # schedule: step size decays linearly with remaining time
    while time.time() - start_time < time_limit:
        # linear decay
        remaining = max(0.0, time_limit - (time.time() - start_time))
        proportion = remaining / time_limit
        max_transfer = 0.08 * proportion + 1e-5   # very small at the end

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # direction: from i -> j
        max_delta = min(max_transfer, cur[i], 1.0 - cur[j])
        if max_delta <= 0.0:
            # try opposite direction
            max_delta = min(max_transfer, cur[j], 1.0 - cur[i])
            if max_delta <= 0.0:
                continue
            i, j = j, i

        delta = np.random.rand() * max_delta
        proposal = cur.copy()
        proposal[i] -= delta
        proposal[j] += delta

        val = _c5_value(proposal)
        if val < cur_val:
            cur = proposal
            cur_val = val
            if val < best_val:
                best = proposal
                best_val = val
        # occasional random restart to escape flat regions
        if np.random.rand() < 0.0008:
            rnd = np.random.rand(n)
            cur = _project_capped_simplex(rnd, target_sum, ub=1.0)
            cur_val = _c5_value(cur)

    return best, best_val


def _refine_with_gradient(h, target_sum, time_limit, start_time):
    """
    Apply a short gradient‑descent refinement after hill climbing.
    """
    n = h.shape[0]
    cur = h.copy()
    cur_val = _c5_value(cur)
    best = cur.copy()
    best_val = cur_val

    step = 0.6                 # initial step size
    decay = 0.9997

    while time.time() - start_time < time_limit:
        new, new_val = _grad_step(cur, step, target_sum)
        if new_val < cur_val:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best, best_val = new, new_val
        else:
            # reject – keep cur
            pass
        step *= decay
    return best, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimises
    max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # Determine the discretisation size
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        n = int(init.shape[0])
        target_sum = n / 2.0
        # Make sure the initial vector is feasible
        h0 = _project_capped_simplex(init, target_sum, ub=1.0)
    except NameError:
        # no parent provided – start from a modest grid
        n = 200
        target_sum = n / 2.0
        rnd = np.random.rand(n)
        h0 = _project_capped_simplex(rnd, target_sum, ub=1.0)

    best_h = h0.copy()
    best_val = _c5_value(best_h)

    start = time.time()
    time_limit = budget_s - 0.3          # small safety margin

    # Main loop: a mixture of random restarts + hill climbing + gradient refinement
    # Allocate most of the budget to cheap hill‑climbing, reserve the last ~5%
    # for a final gradient pass.
    gradient_phase = max(0.0, time_limit * 0.07)   # 7 % for gradient refinement
    hill_time = time_limit - gradient_phase

    while time.time() - start < hill_time:
        # random restart
        rnd = np.random.rand(n)
        cur = _project_capped_simplex(rnd, target_sum, ub=1.0)
        # short hill climbing from this start (few seconds)
        # each restart gets at most 0.6 s (adjustable)
        remaining = hill_time - (time.time() - start)
        # if remaining is tiny, break
        if remaining < 0.5:
            break
        # allow each restart ~0.5 s of hill climbing
        t_alloc = min(0.55, remaining)
        candidate, cand_val = _local_hill_climb(cur, target_sum, t_alloc, time.time())
        if cand_val < best_val:
            best_h, best_val = candidate.copy(), cand_val

    # Gradient refinement on the best found solution
    if time.time() - start < time_limit:
        remaining = time_limit - (time.time() - start)
        best_h, best_val = _refine_with_gradient(best_h, target_sum, remaining, time.time())

    # Final safety projection (numerical drift)
    best_h = _project_capped_simplex(best_h, target_sum, ub=1.0)
    final_val = _c5_value(best_h)

    return best_h, final_val, n
