def compute_corr(h):
    # Scale correlation (without dx)
    return np.correlate(h, 1 - h, mode='full')

def max_mismatch_and_lag(h):
    corr = compute_corr(h)
    idx = np.argmax(corr)
    max_val = corr[idx]
    lag = idx - (len(h) - 1)
    return max_val, lag, corr

def compute_gradient(h, lag):
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        i = np.arange(N - lag)
        j = i + lag
    else:
        L = -lag
        i = np.arange(L, N)
        j = i - L
    # contributions:
    grad[i] += (1 - h[j])
    grad[j] -= h[i]
    return grad

def project_to_constraints(y, target_sum):
    # Find λ such that sum(clip(y+λ,0,1))=target_sum
    lo = -2.0
    hi = 2.0
    # note monotonic: sum increases with λ
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(y + lo, 0.0, 1.0)

import time
import numpy as np

# ----- helper utilities ---------------------------------------------------- #
def _project_to_capped_simplex(v, target_sum, eps=1e-12):
    """
    Euclidean projection of vector v onto the set
        {x : 0 <= x_i <= 1 , sum(x_i) = target_sum }.
    Uses simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0          # bounds wide enough for any feasible shift
    for _ in range(30):        # 30 iterations give ~1e-9 precision
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(v + lo, 0.0, 1.0)


def _c5_value(h, dx):
    """return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _gradient_for_lag(h, lag):
    """gradient of the mismatch sum for a given lag (sub‑gradient of max)."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if lag >= 0:
        i = np.arange(n - lag)
        j = i + lag
    else:
        L = -lag
        i = np.arange(L, n)
        j = i - L
    grad[i] += 1.0 - h[j]
    grad[j] -= h[i]
    return grad


# ----- main optimisation routine ------------------------------------------ #
def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()

    # ----- choose discretisation size -------------------------------------- #
    # Prefer a finer grid (more freedom) while staying small enough to be fast.
    N = 140                     # even number, 2 / N = dx
    target_sum = N // 2        # required sum of h values
    dx = 2.0 / N

    # ----- initialise from parent if available ------------------------------ #
    if "initial_h_values" in globals():
        small = np.asarray(initial_h_values, dtype=float)
        # interpolate to the chosen grid size
        xs_old = np.linspace(0.0, 2.0, small.shape[0], endpoint=False)
        xs_new = np.linspace(0.0, 2.0, N, endpoint=False)
        h = np.interp(xs_new, xs_old, small)
    else:
        h = np.zeros(N)

    # project to feasible set (box + sum)
    h = _project_to_capped_simplex(h, target_sum)

    # ----- keep best found ------------------------------------------------- #
    best_h = h.copy()
    best_val = _c5_value(best_h, dx)

    # ----- optimisation parameters ------------------------------------------ #
    step0 = 0.6                # initial learning‑rate magnitude
    restart_iters = 5000       # iterations per random restart
    max_total_iters = 200000   # overall iteration budget
    used_iters = 0
    top_k = 3                  # number of lags to blend in the sub‑gradient
    target_val = 0.3808        # we stop as soon as we beat this

    # ----- main optimisation loop ------------------------------------------ #
    while time.time() - t_start < budget_s * 0.95 and used_iters < max_total_iters:
        # ---- random restart ------------------------------------------------ #
        h = np.zeros(N)
        idx = rng.choice(N, target_sum, replace=False)
        h[idx] = 1.0
        # small random jitter to escape binary corners
        h += rng.uniform(-0.12, 0.12, size=N)
        h = np.clip(h, 0.0, 1.0)
        h = _project_to_capped_simplex(h, target_sum)

        cur_val = _c5_value(h, dx)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            if best_val <= target_val:
                return (best_h, best_val, N)

        # ---- iterative refinement ------------------------------------------- #
        for it in range(restart_iters):
            used_iters += 1
            # correlation for all lags
            corr = np.correlate(h, 1.0 - h, mode="full")
            # pick up to top_k lags with largest mismatches
            if top_k >= corr.size:
                lag_indices = np.arange(corr.size)
            else:
                lag_indices = np.argpartition(-corr, top_k)[:top_k]
            lags = lag_indices - (N - 1)

            # accumulate gradients from the selected lags
            grad = np.zeros_like(h)
            for lag in lags:
                grad += _gradient_for_lag(h, lag)
            grad /= len(lags)

            # adaptive step size
            step = step0 / (1.0 + it * 0.001)

            # gradient step + occasional random perturbation
            y = h - step * grad
            if rng.rand() < 0.02:
                y += rng.normal(scale=0.07, size=N)

            # project back to feasible region
            h = _project_to_capped_simplex(y, target_sum)

            # evaluate
            cur_val = _c5_value(h, dx)
            if cur_val < best_val - 1e-9:
                best_val = cur_val
                best_h = h.copy()
                if best_val <= target_val:
                    return (best_h, best_val, N)

        # end of one restart – loop continues until time/iteration limit

    # final return (best found within the budget)
    final_val = _c5_value(best_h, dx)
    return (best_h, final_val, N)
