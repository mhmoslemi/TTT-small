import time
import random
import numpy as np

def _compute_c5(h, dx):
    """Return max overlap C5 for a step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr * dx).max()

def _repair(h, target):
    """
    Clip values to [0,1] and adjust the total sum to exactly `target`
    (preserving bounds).  Modifies h in‑place and also returns it.
    """
    np.clip(h, 0.0, 1.0, out=h)
    total = h.sum()
    diff = target - total
    if abs(diff) < 1e-12:
        return h
    # indices not at a bound
    free = (h > 1e-12) & (h < 1 - 1e-12)
    # if everything is saturated we scale the whole vector
    if not np.any(free):
        h *= target / total
        np.clip(h, 0.0, 1.0, out=h)
        return h
    # distribute the remaining mass uniformly among free entries
    while abs(diff) > 1e-12:
        idx = np.where(free)[0]
        step = diff / len(idx)
        # apply step, allowing temporary overshoot which we later clip
        h[idx] += step
        np.clip(h, 0.0, 1.0, out=h)
        total = h.sum()
        diff = target - total
        free = (h > 1e-12) & (h < 1 - 1e-12)
        if not np.any(free):
            # tiny adjustment left – scale everything proportionally
            h *= target / total
            break
    return h

    try:
        # existing construction, if provided by the harness
        h = np.asarray(initial_h_values, dtype=float).copy()
    except NameError:
        # fall back to a simple “V‑shape” (|x‑1|)
        n = 70
        dx = 2.0 / n
        x = np.arange(n, dtype=float) * dx
        h = np.abs(x - 1.0)
    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0
    h = _repair(h, target_sum)

    cur_val = _compute_c5(h, dx)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    max_iters = 5_000_000          # generous upper bound; loop stops by time

    eps = 1e-12
    # linear annealing schedule
    T0 = 0.010                     # initial temperature

    it = 0
    while True:
        it += 1
        # stop before time runs out
        if time.time() - start > budget_s * 0.95:
            break

        # temperature for simulated annealing
        T = T0 * max(0.0, 1.0 - it / max_iters)

        # ------------------------------------------------------------------
        # 1) locate the worst shift (largest overlap)
        # ------------------------------------------------------------------
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_vals = corr * dx
        cur_val = c5_vals.max()
        worst_idx = np.argmax(c5_vals)
        lag = worst_idx - (n - 1)               # integer shift: h[i] vs h[i+lag]

        # ------------------------------------------------------------------
        # 2) try a directed transfer that reduces the contribution of this lag
        # ------------------------------------------------------------------
        moved = False
        if lag != 0:
            i_start = max(0, -lag)
            i_end   = min(n, n - lag)
            idxs = np.arange(i_start, i_end, dtype=int)

            hi = h[idxs]
            hj = h[idxs + lag]
            mask = (hi > eps) & (hj < 1.0 - eps)
            if np.any(mask):
                # choose a random eligible pair
                eligible = np.where(mask)[0]
                sel = np.random.choice(eligible)
                i = idxs[sel]
                j = i + lag
                # allowed mass to move from i to j
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta > eps:
                    delta = max_delta * np.random.rand() * 0.5  # up to half of possible
                    h_new = h.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
                    new_val = _compute_c5(h_new, dx)

                    # acceptance test
                    if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / (T + 1e-15)):
                        h = h_new
                        cur_val = new_val
                        moved = True
                        if new_val < best_val:
                            best_val = new_val
                            best_h = h_new.copy()
        # ------------------------------------------------------------------
        # 3) if directed move impossible or rejected, perform a random swap
        # ------------------------------------------------------------------
        if not moved:
            i, j = np.random.choice(n, 2, replace=False)
            hi, hj = h[i], h[j]
            # delta transferred from i to j (negative delta moves opposite direction)
            d_low  = max(-hi,     hj - 1.0)
            d_high = min(1.0 - hi, hj)
            if d_low < d_high:
                delta = np.random.uniform(d_low, d_high)
                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta
                new_val = _compute_c5(h_new, dx)

                if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / (T + 1e-15)):
                    h = h_new
                    cur_val = new_val
                    if new_val < best_val:
                        best_val = new_val
                        best_h = h_new.copy()

        # occasionally shake the system to avoid stagnation
        if it % 200000 == 0:
            # small random perturbation preserving the total mass
            idx = np.random.choice(n, 2, replace=False)
            i, j = idx[0], idx[1]
            hi, hj = h[i], h[j]
            d_low  = max(-hi,     hj - 1.0)
            d_high = min(1.0 - hi, hj)
            if d_low < d_high:
                delta = np.random.uniform(d_low, d_high)
                h[i] += delta
                h[j] -= delta
                # re‑repair possible tiny drift
                h = _repair(h, target_sum)

    # final clean‑up
    best_h = _repair(best_h, target_sum)
    final_val = _compute_c5(best_h, dx)

    return (best_h, float(final_val), n)
