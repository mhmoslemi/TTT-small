def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math

    # Retrieve initial construction if present
    init = globals().get('initial_h_values', None)
    # Determine n_points
    if init is not None:
        h_init = np.asarray(init, dtype=float)
        n_points = h_init.shape[0]
    else:
        n_points = 70
        h_init = None

    # Ensure even length; weight = n_points // 2
    weight = n_points // 2
    dx = 2.0 / n_points

    # Create initial binary solution
    if h_init is not None and h_init.size == n_points:
        # Convert to binary by threshold 0.5; then adjust to exact weight
        binary = (h_init > 0.5).astype(int)
        cur_sum = binary.sum()
        if cur_sum != weight:
            # Adjust using fractional parts
            fractions = h_init - binary
            # If need to add ones:
            if cur_sum < weight:
                # choose zeros to turn into ones based on largest fractions
                zero_idx = np.where(binary == 0)[0]
                # sort zero indices by fractions descending
                order = np.argsort(fractions[zero_idx])[::-1]
                to_add = weight - cur_sum
                chosen = zero_idx[order[:to_add]]
                binary[chosen] = 1
            else:
                # need to remove ones
                one_idx = np.where(binary == 1)[0]
                order = np.argsort(fractions[one_idx])  # smallest fractions first (i.e., closest to 0)
                to_remove = cur_sum - weight
                chosen = one_idx[order[:to_remove]]
                binary[chosen] = 0
        h = binary.astype(int)
    else:
        # random binary with weight
        h = np.zeros(n_points, dtype=int)
        h[:weight] = 1
        np.random.shuffle(h)

    # Helper to compute max cross pairs (C as integer)
    def max_cross(arr):
        # arr is int array 0/1
        # returns integer max of correlation
        corr = np.correlate(arr, 1 - arr, mode='full')
        # corr array dtype int64 maybe
        return int(corr.max())

    # Evaluate current solution
    cur_max = max_cross(h)
    best_h = h.copy()
    best_max = cur_max

    # Setup random
    random.seed(seed)
    np.random.seed(seed + 1)

    start = time.time()
    end_time = start + budget_s - 0.1

    # SA parameters
    # Use linear cooling with T0 related to weight maybe
    T0 = 2.0  # initial temperature
    T_end = 0.001
    max_iters = 2500000  # 2.5 million iterations limit

    n = n_points
    w = weight

    # Precompute indices list maybe not needed

    # Simulated annealing loop
    it = 0
    while True:
        now = time.time()
        if now >= end_time:
            break
        if it >= max_iters:
            break

        # select random 1 and 0 indices
        # choose i=randint(0,n-1) until h[i]==1, similarly j for 0.
        i = random.randrange(n)
        while h[i] == 0:
            i = random.randrange(n)
        j = random.randrange(n)
        while h[j] == 1:
            j = random.randrange(n)

        # swap
        h[i], h[j] = 0, 1

        new_max = max_cross(h)

        # temperature schedule
        frac = it / max_iters
        T = T0 * (1 - frac) + T_end * frac

        accept = False
        if new_max <= cur_max:
            accept = True

            prob = math.exp(-(new_max - cur_max) / T) if T > 1e-12 else 0.0
            if random.random() < prob:
                accept = True
        if accept:
            cur_max = new_max

def improve_by_gradient(h):
    while True:
        # compute max C and its shift k_max
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        # compute lag from index: lag = max_idx - (n-1)
        k = max_idx - (n-1)
        # compute gradient g_i = 1 - (h_{i+k} if i+k < n else 0) - (h_{i-k} if i-k >=0 else 0)
        g = np.ones_like(h)
        # subtract contributions
        # for i where i+k < n
        idx1 = np.arange(0, n - k)
        g[idx1] -= h[idx1 + k]
        # for i where i - k >= 0
        if k > 0:
            idx2 = np.arange(k, n)
            g[idx2] -= h[idx2 - k]
        # for k negative?
        # Actually if k negative, we computed shift as negative value. Use same formula with k negative.
        # For gradient formula we can handle both signs similarly.

        # We want to find i where g_i > 0 (decrease h_i) and j where g_j < 0 (increase h_j)
        pos = np.where((g > 1e-8) & (h > 1e-8))[0]
        neg = np.where((g < -1e-8) & (h < 1 - 1e-8))[0]
        if len(pos) == 0 or len(neg) == 0:
            break
        # Choose i = argmax g_i, j = argmin g_i
        i = pos[np.argmax(g[pos])]
        j = neg[np.argmin(g[neg])]
        # maximum amount we can move is min(h[i], 1 - h[j])
        max_delta = min(h[i], 1 - h[j])
        # also may want to limit delta to something small, else big jump may overshoot
        delta = max_delta * 0.5  # move half capacity
        if delta <= 1e-8:
            break
        # apply move
        h[i] -= delta
        h[j] += delta
        # recompute max_corr and continue loop
        new_corr = np.correlate(h, 1 - h, mode='full')
        new_max = new_corr.max()
        if new_max + 1e-9 < np.max(corr):
            # improvement made, continue
            continue
        else:
            # no improvement, revert and break
            h[i] += delta
            h[j] -= delta
            break
    return h

import numpy as np
import random
import time
import math

def run(seed=42, budget_s=1100, **kwargs):
    # ------------------------------------------------------------
    # Setup
    # ------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed + 1)

    # Input h (if provided)
    init = globals().get("initial_h_values", None)

    # Determine problem size
    if init is not None:
        h0 = np.asarray(init, dtype=float)
        n = h0.shape[0]
    else:
        n = 70
        h0 = None

    # Required total mass
    w = n // 2                     # because ∫h = 1  ->  sum(h) = n/2
    dx = 2.0 / n

    # ------------------------------------------------------------
    # Helper utilities
    # ------------------------------------------------------------
    def max_overlap(arr):
        """max_k ∫ h(x)(1‑h(x+k))dx  on the discrete grid (dx already omitted)"""
        return float(np.correlate(arr, 1.0 - arr, mode="full").max())

    def legendre_binary(N):
        """binary Legendre pattern for N (requires N+1 prime)"""
        p = N + 1
        # simple primality check
        if p < 2:
            return None
        for d in range(2, int(math.isqrt(p)) + 1):
            if p % d == 0:
                return None
        out = np.zeros(N, dtype=int)
        exp = (p - 1) // 2
        for i in range(1, N + 1):
            if pow(i, exp, p) == 1:
                out[i - 1] = 1
        return out

    def random_binary():
        """random 0/1 vector with exactly w ones"""
        a = np.zeros(n, dtype=int)
        a[:w] = 1
        np.random.shuffle(a)
        return a.astype(float)

    # ------------------------------------------------------------
    # Initial solution
    # ------------------------------------------------------------
    if h0 is not None:
        # Use the provided vector, project to feasible box and mass
        h = np.clip(h0, 0.0, 1.0)
        total = h.sum()
        if abs(total - w) > 1e-12:
            # scale to correct mass, then re‑clip
            if total > 0:
                h *= (w / total)
                h = np.clip(h, 0.0, 1.0)
            # final tiny correction to hit the exact sum
            diff = w - h.sum()
            if abs(diff) > 1e-12:
                # add diff to a random non‑saturated entry
                cand = np.where((h > 0.0) & (h < 1.0))[0]
                if cand.size == 0:
                    cand = np.arange(n)
                idx = random.choice(cand.tolist())
                h[idx] = max(0.0, min(1.0, h[idx] + diff))
    else:
        # try Legendre pattern, otherwise random binary
        leg = legendre_binary(n)
        if leg is not None:
            h = leg.astype(float)
        else:
            h = random_binary()

    # ------------------------------------------------------------
    # Simulated Annealing on continuous variables
    # ------------------------------------------------------------
    best_h = h.copy()
    best_val = max_overlap(best_h)

    start_time = time.time()
    end_time = start_time + budget_s - 0.1   # tiny safety margin

    # annealing schedule parameters
    T_start = 0.5
    T_end = 1e-5
    max_iters = 3_000_000          # generous upper bound; time will stop earlier

    it = 0
    while it < max_iters and time.time() < end_time:
        # pick i (source) with h[i] > 0, j (dest) with h[j] < 1
        i = random.randrange(n)
        while h[i] <= 1e-12:
            i = random.randrange(n)
        j = random.randrange(n)
        while h[j] >= 1.0 - 1e-12:
            j = random.randrange(n)

        # permissible transfer amount
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            it += 1
            continue
        # step size linked to temperature (larger early)
        delta = max_delta * random.random() * (T_start if it == 0 else 1.0)
        if delta <= 0.0:
            it += 1
            continue

        # propose move
        h[i] -= delta
        h[j] += delta

        new_val = max_overlap(h)

        # temperature (linear cooling)
        frac = it / max_iters
        T = T_start * (1.0 - frac) + T_end * frac

        # acceptance rule
        accept = False
        if new_val <= best_val:
            accept = True
        else:
            # Metropolis criterion
            prob = math.exp(-(new_val - best_val) / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
            # keep the move (h already updated)
        else:
            # revert
            h[i] += delta
            h[j] -= delta

        it += 1

    # ------------------------------------------------------------
    # Gradient‑based local refinement (focus on worst shift)
    # ------------------------------------------------------------
    def improve_by_gradient(vec):
        """steepest local improvement targeting the currently worst shift"""
        while True:
            # current worst shift
            corr = np.correlate(vec, 1.0 - vec, mode="full")
            k_index = np.argmax(corr)
            k = k_index - (n - 1)          # lag value (can be negative)

            # gradient of C_k wrt each coordinate
            grad = np.ones_like(vec)

            if k >= 0:
                # forward part
                grad[: n - k] -= vec[k:]
                # backward part
                grad[k:] -= vec[: n - k]
            else:
                # k negative -> treat symmetrically
                kk = -k
                grad[kk:] -= vec[: n - kk]
                grad[: n - kk] -= vec[kk:]

            # indices where decreasing reduces C_k and increasing reduces C_k
            dec_candidates = np.where((grad > 1e-12) & (vec > 1e-12))[0]
            inc_candidates = np.where((grad < -1e-12) & (vec < 1.0 - 1e-12))[0]

            if dec_candidates.size == 0 or inc_candidates.size == 0:
                break

            i = dec_candidates[np.argmax(grad[dec_candidates])]
            j = inc_candidates[np.argmin(grad[inc_candidates])]

            max_delta = min(vec[i], 1.0 - vec[j])
            if max_delta <= 1e-12:
                break
            delta = max_delta * 0.5   # cautious step

            # apply tentative move
            vec[i] -= delta
            vec[j] += delta
            new_val = max_overlap(vec)

            # accept only if improvement
            if new_val + 1e-12 < corr.max():
                # keep changes
                continue
            else:
                # revert and stop
                vec[i] += delta
                vec[j] -= delta
                break
        return vec

    best_h = improve_by_gradient(best_h.copy())
    best_val = max_overlap(best_h)

    # ------------------------------------------------------------
    # Final sanity checks & return
    # ------------------------------------------------------------
    # Clip tiny numerical overshoots and enforce exact mass
    best_h = np.clip(best_h, 0.0, 1.0)
    mass_err = w - best_h.sum()
    if abs(mass_err) > 1e-8:
        # distribute the tiny error to a non‑saturated entry
        idx = np.where((best_h > 1e-12) & (best_h < 1.0 - 1e-12))[0]
        if idx.size == 0:
            idx = np.arange(n)
        best_h[idx[0]] += mass_err
        best_h = np.clip(best_h, 0.0, 1.0)

    c5_bound = best_val * dx
    return (best_h.astype(float), float(c5_bound), n)
