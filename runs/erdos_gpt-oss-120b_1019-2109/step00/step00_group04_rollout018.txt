def random_h(n):
    target = n / 2.0
    # start with zeros
    h = np.zeros(n)
    # fill random indices with random capacities
    indices = list(range(n))
    random.shuffle(indices)
    remaining = target
    for i in indices:
        if remaining <= 0:
            break
        # capacity max 1
        max_add = min(1.0, remaining)
        # assign random amount up to max_add
        value = random.random() * max_add
        h[i] = value
        remaining -= value
    # If leftover (unlikely due to random small values), distribute evenly
    if remaining > 1e-12:
        # add equally to all entries (or random)
        for i in range(n):
            add = min(1 - h[i], remaining)
            h[i] += add
            remaining -= add
            if remaining <= 1e-12:
                break
    # Ensure sum = target (perhaps small diff)
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # adjust first entry within bounds
        for i in range(n):
            avail = 1 - h[i] if diff > 0 else h[i]
            inc = min(abs(diff), avail)
            if diff > 0:

import time

import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k)) dx approximated on the discrete grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def random_h(n: int) -> np.ndarray:
    """Generate a random feasible vector of length n (values in [0,1]) with sum=n/2."""
    target = n / 2.0
    h = np.zeros(n, dtype=float)
    remaining = target
    idxs = list(range(n))
    random.shuffle(idxs)
    for i in idxs:
        if remaining <= 0:
            break
        max_add = min(1.0, remaining)
        add = random.random() * max_add
        h[i] = add
        remaining -= add
    # distribute any tiny remainder
    if abs(remaining) > 1e-12:
        for i in range(n):
            cap = 1.0 - h[i] if remaining > 0 else h[i]
            if cap <= 0:
                continue
            inc = min(abs(remaining), cap)
            if remaining > 0:
                h[i] += inc
                remaining -= inc
            else:
                h[i] -= inc
                remaining += inc
            if abs(remaining) < 1e-12:
                break
    return h

def upscale_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate old_h to a larger grid and enforce the integral constraint."""
    old_n = old_h.shape[0]
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    target = new_n / 2.0
    diff = target - new_h.sum()
    attempts = 0
    # simple redistribution of the leftover mass
    while abs(diff) > 1e-12 and attempts < 10000:
        idx = random.randrange(new_n)
        if diff > 0:
            cap = 1.0 - new_h[idx]
            if cap <= 0:
                attempts += 1
                continue
            inc = min(diff, cap)
            new_h[idx] += inc
            diff -= inc
        else:
            cap = new_h[idx]
            if cap <= 0:
                attempts += 1
                continue
            dec = min(-diff, cap)
            new_h[idx] -= dec
            diff += dec
        attempts += 1

    # final tiny correction if needed
    cur = new_h.sum()

    """
    Simulated‑annealing style hill climbing that respects the sum constraint.
    Returns the best vector found and its C5 value.
    """
    n = initial_h.shape[0]
    cur_h = initial_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5
    start = time.time()
    no_improve = 0

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = elapsed / time_limit
        # temperature decreases linearly, start ≈0.1
        T = 0.1 * (1.0 - frac) + 1e-9

        i, j = random.sample(range(n), 2)
        low = max(-cur_h[i], cur_h[j] - 1.0)
        high = min(1.0 - cur_h[i], cur_h[j])
        if low >= high:
            continue
        delta = random.uniform(low, high)

        # apply move
        cur_h[i] += delta
        cur_h[j] -= delta
        new_c5 = compute_c5(cur_h)
        dE = new_c5 - cur_c5

        accept = False
        if dE <= 0:
            accept = True
        else:
            prob = math.exp(-dE / T) if T > 0 else 0.0
            if random.random() < prob:
                accept = True

        if accept:
            cur_c5 = new_c5
            no_improve = 0
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
        else:

            cur_h[i] -= delta
            cur_h[j] += delta

        # occasional restart around the current best
