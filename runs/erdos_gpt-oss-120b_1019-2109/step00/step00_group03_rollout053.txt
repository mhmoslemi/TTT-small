import time
import math
import numpy as np

def _cost(h, dx):
    """Return C5 bound for the sampled step function."""
    # max correlation of h with (1-h) over all integer shifts
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _gradient(h, shift):
    """Gradient of the correlation for a given integer shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        L = n - shift
        i = np.arange(L, dtype=int)
        g[i] += 1.0 - h[i + shift]        # ∂/∂h_i
        g[i + shift] += -h[i]             # ∂/∂h_{i+shift}
    else:
        s = -shift
        L = n - s
        i = np.arange(L, dtype=int)
        g[i + s] += 1.0 - h[i]            # ∂/∂h_{i+s}
        g[i] += -h[i + s]                 # ∂/∂h_i
    return g


def run(seed=42, budget_s=1100, **kwargs):

    # ------------------------------------------------------------------

    if "initial_h_values" in globals():

        if cur != 0:
