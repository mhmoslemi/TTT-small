import time
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x) (1‑h(x+k)) dx for the discretized step function."""
    n = h.size
    dx = 2.0 / n
    # np.correlate computes Σ_i h[i] * (1‑h)[i+lag]
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _repair_sum(h: np.ndarray, target_sum: float, tol: float = 1e-12) -> np.ndarray:
    """
    Project h onto the simplex {0≤h≤1, Σh = target_sum}.
    Simple iterative proportional‑adjustment; fast for our dimensions.
    """
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        cur = h.sum()
        diff = target_sum - cur
        if abs(diff) <= tol:
            break
        if diff > 0:
            # increase where we have room
            cap = 1.0 - h
            total = cap.sum()
            if total == 0.0:
                break
            inc = diff * cap / total
            h += inc
        else:
            # decrease where we have mass
            cap = h
            total = cap.sum()
            if total == 0.0:
                break
            dec = (-diff) * cap / total
            h -= dec
        h = np.clip(h, 0.0, 1.0)
    return h

                delta = max_delta * rng.random() * step_frac
                cand = h.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        else:
            max_delta = min(h[j], 1.0 - h[i])
            if max_delta > 1e-12:
                delta = max_delta * rng.random() * step_frac
                cand = h.copy()
                cand[j] -= delta
                cand[i] += delta
            else:
                continue

        cand_c5 = _compute_c5(cand)
        if cand_c5 < cur_c5 - 1e-12:
            h = cand
            continue
        temp = T_start * max(0.0, 1.0 - elapsed / budget_s)
        if temp > 0.0 and rng.random() < np.exp(-(cand_c5 - cur_c5) / temp):
            h = cand
            continue

        # ------------------------------------------------------------------
        # occasional restart when stuck
        # ------------------------------------------------------------------
        if it_without_improve > stagnation_limit:
            # perturb the current best a little and continue searching
            perturb = rng.normal(scale=0.05, size=n)
            h = best_h + perturb
            h = np.clip(h, 0.0, 1.0)
            h = _repair_sum(h, n / 2)
            it_without_improve = 0

    # final projection of the best candidate
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _repair_sum(best_h, n / 2)
    final_c5 = _compute_c5(best_h)

    return (best_h.tolist(), float(final_c5), n)
