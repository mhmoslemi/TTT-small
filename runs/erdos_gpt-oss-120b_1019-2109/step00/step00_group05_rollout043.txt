def make_trapezoid(s, n):
    # s in [0,1], plateau length p = 1 - s, slope length s
    # center at 1 (midpoint)
    dx = 2.0 / n
    x = np.arange(n) * dx + dx/2  # midpoints?
    p = 1.0 - s
    half_p = p / 2.0
    # slope length s (same as slope_width)
    # For points with |x-1| <= half_p: h=1
    # else if |x-1| <= half_p + s: h = 1 - (|x-1| - half_p)/s
    # else: h = 0
    d = np.abs(x - 1.0)
    h = np.zeros_like(x)
    in_plateau = d <= half_p
    h[in_plateau] = 1.0
    in_slope = (d > half_p) & (d <= half_p + s)
    h[in_slope] = 1.0 - (d[in_slope] - half_p) / s
    # h values may be slightly out of [0,1] due to numeric error; clip
    h = np.clip(h, 0, 1)
    # adjust sum to target (n/2). The area under h for continuous shape equals 1 (by construction)
    # However due to discretization may be slightly off; we can project onto capped simplex to ensure sum = n/2.
    target = n/2
    h = proj_capped_simplex(h, target)
    return h

def hill_climb_binary(start_h, max_iters, rng):
    h = start_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    n = len(h)
    for it in range(max_iters):
        # compute conv
        conv = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(conv)
        c5_curr = conv[max_idx] * dx
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()
            if best_c5 <= 0.3808:  # early stop
                break
        L = max_idx - (n - 1)
        if L >= 0:
            i_range = np.arange(0, n - L)
        else:
            i_range = np.arange(-L, n)
        # indices i where h[i]=1 and h[i+L]=0
        mism_i = i_range[(h[i_range] > 0.5) & (h[i_range + L] < 0.5)]
        if mism_i.size > 0:
            i = rng.choice(mism_i)
            zeros = np.where(h < 0.5)[0]
            # ensure zeros exclude i (should)
            if zeros.size > 1:
                j = rng.choice(zeros)
                # propose swap
                h_new = h.copy()
                h_new[i], h_new[j] = h_new[j], h_new[i]
                # compute new c5
                conv_new = np.correlate(h_new, 1.0 - h_new, mode='full')
                c5_new = np.max(conv_new) * dx
                if c5_new < best_c5:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    continue
        # random swap fallback
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]

            i = rng.choice(ones)

def simulated_annealing(start_h, max_steps, rng):
    h = start_h.copy()
    c5_curr = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_curr
    # Temperature schedule: start T = 1.0, decay to 0.001 over steps
    T0 = 1.0
    T_end = 1e-3
    for step in range(max_steps):
        t = step / max_steps
        T = T0 * (1 - t) + T_end * t  # linear decay
        # propose swap
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i], h_new[j] = 0.0, 1.0
        c5_new = compute_c5(h_new)
        delta = c5_new - c5_curr
        if delta < 0 or rng.random() < math.exp(-delta / T):
            h = h_new
            c5_curr = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    return best_h, best_c5

        lo = -1.0
        hi = 2.0

def proj_capped_simplex(v, target):
    # binary search as earlier
    lo=-1.0
    hi=2.0
    for _ in range(60):
        lam=(lo+hi)/2
        x=np.clip(v - lam, 0.0, 1.0)
        s=x.sum()
        if s > target:
            lo=lam
        else:

    return np.clip(v - (lo+hi)/2, 0.0, 1.0)

def gradient_opt(h0, n, target, dx, steps=2000, lr=0.2, rng=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for step in range(steps):
        # compute self-correlation
        corr = np.correlate(h, h, mode='full')
        # find minimal correlation (worst overlap)
        min_idx = np.argmin(corr)
        L = min_idx - (n - 1)
        # gradient for min correlation shift
        g = np.zeros_like(h)
        if L >= 0:
            g[:n-L] += h[L:]
            g[L:] += h[:n-L]
        else:

def simulated_annealing(start_h, n, target, dx, steps, rng):
    h = start_h.copy()
    c5_curr = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5_curr
    T0 = 0.5
    T_end = 1e-4
    for step in range(steps):
        t = step / steps
        T = T0 * (1 - t) + T_end * t
        # choose random 1 and 0 indices
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i], h_new[j] = 0.0, 1.0
        c5_new = compute_c5(h_new, dx)
        delta = c5_new - c5_curr
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            h = h_new
            c5_curr = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    return best_h, best_c5

import numpy as np
import time
import math

# ---------- helper functions (top level) ----------
def proj_capped_simplex(v, target):
    """Projection of v onto the box [0,1]^n with sum equal to target."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def compute_c5(h, dx):
    """Maximum overlap integral for a step vector h."""
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx


def random_binary(n, target, rng):
    h = np.zeros(n, dtype=float)
    idx = rng.choice(n, int(target), replace=False)
    h[idx] = 1.0
    return h


def random_continuous(n, target, rng):
    v = rng.random(n)
    return proj_capped_simplex(v, target)


def simulated_annealing(start_h, n, target, dx, steps, rng):
    """Simple SA on binary vectors (swap 1/0)."""
    h = start_h.copy()
    cur = compute_c5(h, dx)
    best_h, best_val = h.copy(), cur
    T0, Tend = 0.5, 1e-4
    for t in range(steps):
        T = T0 * (1 - t / steps) + Tend * (t / steps)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)
        h_new = h.copy()
        h_new[i], h_new[j] = 0.0, 1.0
        new = compute_c5(h_new, dx)
        delta = new - cur
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            h, cur = h_new, new
            if new < best_val:
                best_h, best_val = h_new.copy(), new
                if best_val <= 0.3808:
                    break
    return best_h, best_val


def gradient_descend_max(start_h, n, target, dx, steps, lr_start, rng):
    """Gradient descent on the worst shift (sub‑gradient of max)."""
    h = start_h.copy()
    best_h, best_val = h.copy(), compute_c5(h, dx)
    lr = lr_start
    for _ in range(steps):
        conv = np.correlate(h, 1.0 - h, mode="full")
        vals = conv * dx
        idx = int(np.argmax(vals))
        cur = vals[idx]
        if cur < best_val:
            best_h, best_val = h.copy(), cur
            if best_val <= 0.3808:
                break
        L = idx - (n - 1)          # shift (positive => right)
        g = np.zeros_like(h)

        if L >= 0:
            rng_idx = np.arange(0, n - L)
            g[rng_idx] = 1.0 - h[rng_idx + L]
            mask = rng_idx >= L
            g[rng_idx[mask]] -= h[rng_idx[mask] - L]
        else:                       # L < 0
            Lneg = -L
            rng_idx = np.arange(Lneg, n)
            g[rng_idx] = 1.0 - h[rng_idx - Lneg]
            mask = rng_idx <= n - Lneg - 1
            g[rng_idx[mask]] -= h[rng_idx[mask] + Lneg]

        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = proj_capped_simplex(h_new, target)

        new = compute_c5(h_new, dx)
        if new < cur:
            h, lr = h_new, min(lr * 1.02, 0.5)
        else:
            lr *= 0.5
            if lr < 1e-6:
                break
    return best_h, best_val


def make_trapezoid(w_frac, n, target):
    """Trapezoidal shape: plateau of height 1 with linear slopes of total width w_frac."""
    dx = 2.0 / n
    w = w_frac * dx                     # width of each slope (in physical units)
    p = 1.0 - w                         # plateau length (since total area =1)
    half_p = p / 2.0
    x = (np.arange(n) + 0.5) * dx
    d = np.abs(x - 1.0)
    h = np.zeros_like(x)
    inside = d <= half_p
    h[inside] = 1.0
    slope = (~inside) & (d <= half_p + w)
    h[slope] = 1.0 - (d[slope] - half_p) / w
    h = np.clip(h, 0.0, 1.0)
    # enforce exact integral
    h = proj_capped_simplex(h, target)
    return h
