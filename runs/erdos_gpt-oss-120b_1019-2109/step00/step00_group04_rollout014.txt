import time
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (sampled uniformly) that
    minimizes   C5 = max_k ∫ h(x)*(1-h(x+k)) dx .
    The discrete approximation used is
        C5 ≈ max(np.correlate(h, 1-h, mode='full')) * dx,
    where dx = 2/n.
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # -------------------------------------------------------------
    # Helper utilities
    # -------------------------------------------------------------
    def project_to_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
        """Project v onto {0≤x_i≤1 , sum x_i = target_sum}."""
        lo, hi = -1.0, 1.0
        for _ in range(60):
            lam = (lo + hi) * 0.5
            x = np.clip(v + lam, 0.0, 1.0)
            if x.sum() > target_sum:
                hi = lam
            else:
                lo = lam
        return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    def c5_value(h: np.ndarray, dx: float) -> float:
        """Return the discrete C5 value for a given h."""
        # correlation of h with 1-h (zero padding)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return corr.max() * dx

    # -------------------------------------------------------------
    # Parameters
    # -------------------------------------------------------------
    n = 200                     # number of sample points (kept < 1000)
    dx = 2.0 / n
    target_sum = n / 2.0        # because ∫h = 1 ⇒ sum h = n/2

    # -------------------------------------------------------------
    # Initial constructions
    # -------------------------------------------------------------
    x = np.linspace(dx / 2, 2.0 - dx / 2, n)   # centre points of each bin

    def make_tent():
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
        return project_to_simplex(h, target_sum)

    def make_parabola():
        a = 0.75                                 # half‑width so that area = 1
        mask = np.abs(x - 1.0) <= a
        t = x[mask] - 1.0
        h = np.zeros_like(x)
        h[mask] = 1.0 - (t / a) ** 2
        return project_to_simplex(h, target_sum)

    def make_sine():
        h = np.sin(np.pi * x / 2.0)               # max = 1, integral = 4/π
        scale = (np.pi / 4.0)                     # makes integral exactly 1
        h = h * scale
        return project_to_simplex(h, target_sum)

    # Pick a starting shape (try a few and keep the best)
    starters = [make_tent(), make_parabola(), make_sine()]
    # also add a random feasible vector
    rand = project_to_simplex(rng.random(n), target_sum)
    starters.append(rand)

    best_h = None
    best_val = np.inf
    for h0 in starters:
        val = c5_value(h0, dx)
        if val < best_val:
            best_val = val
            best_h = h0.copy()

    # -------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing loop
    # -------------------------------------------------------------
    cur_h = best_h.copy()
    cur_val = best_val

    max_iters = 2_000_000
    # initial maximal mass transfer per move
    step = 0.5
    # temperature schedule parameters
    T0 = 0.02
    for it in range(max_iters):
        # time check
        if time.time() - start > budget_s - 0.3:
            break

        # linear temperature decay
        T = T0 * (1.0 - it / max_iters)

        # propose a feasible move that preserves the sum exactly
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        if rng.random() < 0.5:
            # move mass from j to i
            max_delta = min(step, cur_h[j], 1.0 - cur_h[i])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            # move mass from i to j
            max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_val = c5_value(new_h, dx)
        diff = new_val - cur_val

        # accept if improvement or with Metropolis probability
        if diff < 0.0 or rng.random() < np.exp(-diff / (T + 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()

        # gradually shrink the step size
        if (it + 1) % 50000 == 0:
            step = max(0.01, step * 0.9)

    # -------------------------------------------------------------
    # Final cleanup – make absolutely sure constraints are met
    # -------------------------------------------------------------
    best_h = project_to_simplex(best_h, target_sum)
    best_val = c5_value(best_h, dx)

    # Compare with the provided initial_h_values (if any) and keep the best
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        # If the length differs, evaluate as‑is (the harness will accept any n)
        init_val = c5_value(init, 2.0 / init.size)
        if init_val < best_val:
            best_h = init.copy()
            best_val = float(init_val)
            n = init.size

    return best_h, float(best_val), n
