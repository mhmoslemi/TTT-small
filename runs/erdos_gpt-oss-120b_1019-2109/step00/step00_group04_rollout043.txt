import numpy as np
import random
import time
import math

def _evaluate(h: np.ndarray) -> float:
    """Return the C5 bound for a given step function h."""
    n = h.shape[0]
    # correlation of h with (1-h) using linear (non‑circular) convolution.
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))

def _make_shape_p(N: int, p: float) -> np.ndarray:
    """
    Build a symmetric shape  h(x)=min(1, c*|x-1|**p)  with integral 1.
    The scaling constant c is chosen analytically so that the continuous
    integral equals 1, then the discrete array is corrected to satisfy the
    exact sum constraint.
    """
    dx = 2.0 / N
    xs = np.arange(N) * dx                      # left‑hand points on [0,2)
    u = np.abs(xs - 1.0)                        # distance to the centre
    # analytical scaling: c = (2p/(p+1))**p   (derived in the analysis)
    c = (2.0 * p / (p + 1.0)) ** p
    h = c * (u ** p)
    np.minimum(h, 1.0, out=h)                   # clip at 1

    # enforce the exact discrete sum N/2  (integral == 1)
    target = N * 0.5
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the correction uniformly; it is tiny for reasonable N
        h += diff / N
        np.clip(h, 0.0, 1.0, out=h)
        # a second tiny adjustment in case clipping removed some mass
        diff2 = target - h.sum()
        if abs(diff2) > 1e-12:
            h += diff2 / N
            np.clip(h, 0.0, 1.0, out=h)
    return h

    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # ------------------------------------------------------------
    # Choose a working resolution.  Use a larger N for better
    # approximation, but keep it <= 1000 as required.
    # If the harness provides an initial solution, keep its length;
    # otherwise use 400 points (a good trade‑off).
    # ------------------------------------------------------------
    try:

        N = int(init.size)
        # normalise just in case the supplied vector does not hit the exact sum
        target_sum = N * 0.5
        diff = target_sum - init.sum()
        if abs(diff) > 1e-12:
            init += diff / N
            np.clip(init, 0.0, 1.0, out=init)
        best_h = init.copy()
        best_c = _evaluate(best_h)
    except NameError:
        N = 400
        best_h = None
        best_c = float("inf")

    #    where p >= 1 (p=1 gives the linear V‑shape).

    p_vals = np.linspace(1.0, 12.0, 121)   # step 0.1
    for p in p_vals:
        h = _make_shape_p(N, p)
        c_val = _evaluate(h)
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        # early exit if we are already better than the target
        if best_c <= 0.3808:
            break

    # ------------------------------------------------------------
    # 2) Simple local optimisation around the best candidate.
    #    Moves consist of transferring a tiny amount of mass from one
    #    coordinate to another while keeping the total sum fixed.
    # ------------------------------------------------------------
    if best_h is None:
        # fallback: uniform 0.5 (should never happen)
        best_h = np.full(N, 0.5, dtype=float)
        best_c = _evaluate(best_h)

    cur_h = best_h.copy()
    cur_c = best_c

    # pre‑compute which indices can give/take mass

        """Return (i,j,delta) describing a feasible transfer."""
