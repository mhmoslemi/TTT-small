def _evaluate(h):
    """Wrapper for the pre‑imported evaluator."""
    return evaluate_erdos_solution(h)


def _random_binary(n, weight):
    """Return a random binary vector of length n with exactly `weight` ones."""
    h = np.zeros(n, dtype=np.float64)
    ones = np.random.choice(n, weight, replace=False)
    h[ones] = 1.0
    return h


def _max_shift_and_corr(h):
    """Return the shift (as an integer index offset) that gives the maximal
    correlation of h with 1‑h and the corresponding correlation value."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (len(h) - 1)            # 0 corresponds to zero shift
    return shift, corr[idx]


def _mismatched_sets(h, shift):
    """For a given shift, return two index arrays:
       A = positions i where h[i]==1 and h[i+shift]==0
       B = positions i where h[i]==0 and h[i+shift]==1
       (indices are taken from the original unshifted array)."""
    n = len(h)
    if shift == 0:
        return np.array([], dtype=int), np.array([], dtype=int)
    if shift > 0:
        i = np.arange(n - shift)
        a = i[(h[i] == 1) & (h[i + shift] == 0)]
        b = i[(h[i] == 0) & (h[i + shift] == 1)]
    else:                                   # negative shift
        s = -shift
        i = np.arange(s, n)
        a = i[(h[i] == 1) & (h[i - s] == 0)]
        b = i[(h[i] == 0) & (h[i - s] == 1)]
    return a, b


def _targeted_swap(h):
    """Perform a swap aimed at decreasing the current worst‑case overlap.
    Returns True if a swap was made (regardless of improvement)."""
    shift, _ = _max_shift_and_corr(h)
    if shift == 0:
        return False
    a, b = _mismatched_sets(h, shift)
    if a.size == 0 or b.size == 0:
        return False
    i = int(np.random.choice(a))
    j = int(np.random.choice(b))
    h[i], h[j] = h[j], h[i]
    return True

    Returns (h_values, c5_bound, n_points).

    np.random.seed(seed)

    if n % 2 == 1:                # the integral constraint prefers an even n
        n += 1

    target_sum = n // 2           # number of ones for a binary representation

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    best_h = None
    best_val = float("inf")

    if init_vec is not None:
