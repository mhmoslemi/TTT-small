def project_onto_box_simplex(v, t):
    # find lambda such that sum(clip(v - lambda, 0, 1)) = t
    lo = -1.0
    hi = 1.0
    # more extreme bounds: we can compute min(v)-1 and max(v) maybe.
    lo = min(v) - 1.0  # ensure v - lo >= 1 maybe but we want sum > t
    hi = max(v)  # such that v - hi <= 0 => sum 0 < t
    for _ in range(40):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > t:
            lo = lam
        else:
            hi = lam

def gradient_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full') * dx
    lse = logsumexp(beta * conv)
    weights = np.exp(beta * conv - lse)  # shape 2n-1
    G = np.zeros_like(h)
    center = n - 1

        w = weights[idx]
        shift = idx - center
        if shift >= 0:

            # first term
            G[:n - k] += dx * w * (1 - h[k:])
            # second term
            G[k:] -= dx * w * h[:n - k]

            # first term for negative shift
            G[s:] += dx * w * (1 - h[:n - s])
            # second term
            G[:n - s] -= dx * w * h[s:]
    return G, conv, weights

import time
import numpy as np
from scipy.special import logsumexp

def _project_to_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x | 0<=x<=1, sum(x)=target}."""
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _gradient_smooth_max(h: np.ndarray, beta: float):
    """Return gradient of (1/beta) logsumexp(beta*correlation) w.r.t. h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation for all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # softmax weights
    lse = logsumexp(beta * corr)
    w = np.exp(beta * corr - lse)  # sum(w)=1
    grad = np.zeros_like(h)
    center = n - 1
    for idx in range(2 * n - 1):
        weight = w[idx]
        shift = idx - center
        if shift >= 0:
            k = shift
            # contribution from (1 - h[i+shift])
            grad[: n - k] += dx * weight * (1.0 - h[k:])
            # contribution from -h[i]
            grad[k:] -= dx * weight * h[: n - k]
        else:
            s = -shift
            grad[s:] += dx * weight * (1.0 - h[: n - s])
            grad[: n - s] -= dx * weight * h[s:]
    return grad, corr


def _c5_value(h: np.ndarray) -> float:
    """Exact C5 for the current discretisation."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a (h_values, c5_bound, n_points) tuple.
    h_values is a NumPy 1‑D array satisfying the constraints.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, budget_s - 0.5)  # tiny safety margin

    # ------------------------------------------------------------------
    # 1. Choose discretisation size and initialise h
    # ------------------------------------------------------------------
    try:
        init = initial_h_values  # provided by the harness, may be missing
        init_len = len(init)
        n = min(800, max(150, init_len * 4))          # moderate size
        # linear up‑sampling from the coarse parent
        x_coarse = np.linspace(0.0, 2.0, init_len, endpoint=False) + 1.0 / init_len
        x_fine = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        h = np.interp(x_fine, x_coarse, init)
    except Exception:
        # no parent supplied – start from a random feasible point
        n = 200
        h = rng.random(n)

    h = np.clip(h, 0.0, 1.0)
    target_sum = n / 2.0
    h = _project_to_sum_box(h, target_sum)

    # ------------------------------------------------------------------
    # 2. Optimisation parameters (Adam + continuation on beta)
    # ------------------------------------------------------------------
    lr = 0.05
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)

    beta = 2.0          # start with a very smooth max
    beta_target = 200.0
    # multiplicative increase, ~2000 steps to reach target
    beta_mul = (beta_target / beta) ** (1.0 / 2000.0)

    lambda_pen = 0.2    # penalty for violating the integral constraint
    best_h = h.copy()
    best_c5 = _c5_value(h)

    iter_count = 0
    no_improve = 0

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        grad, _ = _gradient_smooth_max(h, beta)
        # quadratic penalty for the sum constraint (∑h = target_sum)
        sum_diff = h.sum() - target_sum
        grad += 2.0 * lambda_pen * sum_diff

        iter_count += 1
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iter_count)
        v_hat = v / (1.0 - beta2 ** iter_count)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)

        # enforce box and integral constraints
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_to_sum_box(h_new, target_sum)

        c5_now = _c5_value(h_new)
        if c5_now + 1e-9 < best_c5:
            best_c5 = c5_now
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        h = h_new

        # gradually sharpen the max
        if beta < beta_target:
            beta = min(beta_target, beta * beta_mul)

        # occasional exploration if stalling
        if no_improve >= 500:
            # random perturbation around the best found point
            pert = best_h + rng.normal(scale=0.03, size=n)
            pert = np.clip(pert, 0.0, 1.0)
            pert = _project_to_sum_box(pert, target_sum)
            h = pert
            no_improve = 0
            # reset Adam moments for a fresh start
            m.fill(0.0)
            v.fill(0.0)

        # slowly decay learning rate
        if iter_count % 800 == 0:
            lr = max(lr * 0.85, 0.003)

    # ------------------------------------------------------------------
    # 4. Final local greedy refinement (single‑coordinate adjustments)
    # ------------------------------------------------------------------
    refine_iter = 0
    while time.time() < deadline and refine_iter < 3000:
        i = rng.integers(n)
        delta = rng.uniform(-0.06, 0.06)
        cand = best_h.copy()
        cand[i] = np.clip(cand[i] + delta, 0.0, 1.0)
        cand = _project_to_sum_box(cand, target_sum)
        c5_cand = _c5_value(cand)
        if c5_cand + 1e-10 < best_c5:
            best_c5 = c5_cand
            best_h = cand
        refine_iter += 1

    # ------------------------------------------------------------------
    # 5. Return the best solution discovered
    # ------------------------------------------------------------------
    return best_h, float(best_c5), n
