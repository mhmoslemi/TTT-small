import time
import numpy as np

def _max_overlap(h: np.ndarray) -> float:
    """Return C5 value for a binary (or [0,1]) step function h on [0,2]."""
    n = h.shape[0]
    # linear convolution h * rev(1-h)
    rev = (1.0 - h)[::-1]
    nfft = 1 << ((2 * n - 1).bit_length())
    fh = np.fft.rfft(h, nfft)
    fr = np.fft.rfft(rev, nfft)
    conv = np.fft.irfft(fh * fr, nfft)[: 2 * n - 1]
    return float(conv.max() * (2.0 / n))

def _binary_random(n: int, rng: np.random.RandomState) -> np.ndarray:
    """Random binary vector of length n with exactly n/2 ones."""
    h = np.zeros(n, dtype=np.float64)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def _initial_binary(n: int, rng: np.random.RandomState) -> np.ndarray:
    """Create a binary start from the global initial_h_values if present."""
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=np.float64)
        # interpolate to length n (linear)
        x_old = np.linspace(0.0, 2.0, init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(x_new, x_old, init)
        # keep the n/2 largest values as 1's
        idx = np.argpartition(interp, -n // 2)[-n // 2 :]
        h = np.zeros(n, dtype=np.float64)
        h[idx] = 1.0
        return h
    # fallback: random binary
    return _binary_random(n, rng)

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a binary step function minimizing the Erdős overlap constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # Choose problem size; larger n gives finer resolution.
    # Keep it modest for speed but enough to improve the bound.
    n = 200
    if budget_s < 300:
        n = 100
    if n % 2:
        n += 1

    # Best solution seen so far
    best_h = _initial_binary(n, rng)
    best_val = _max_overlap(best_h)

    # Helper for time check
    def time_left():
        return budget_s - (time.time() - start_time)

    # Main optimisation loop – many random restarts with simple SA hill‑climb
    while time_left() > 5.0:  # keep a small safety margin
        # start from a fresh random binary vector (occasionally from best)
        if rng.rand() < 0.15 and best_val < 0.5:
            h = best_h.copy()
        else:
            h = _binary_random(n, rng)

        cur_val = _max_overlap(h)

        # Simulated‑annealing parameters
        T = 0.010  # initial temperature
        cooling = 0.9995
        no_improve = 0
        max_moves = 20000

        for _ in range(max_moves):
            if time_left() < 0.5:
                break

            # propose a swap of a 1 and a 0
            i = rng.randint(n)
            while h[i] < 0.5:          # need a 1 at i
                i = rng.randint(n)
            j = rng.randint(n)
            while h[j] > 0.5:          # need a 0 at j
                j = rng.randint(n)

            # Perform swap
            h[i] = 0.0
            h[j] = 1.0
            new_val = _max_overlap(h)
            delta = new_val - cur_val

            accept = False
            if delta < 0:
                accept = True
            else:
                # Metropolis criterion
                if rng.rand() < np.exp(-delta / T):
                    accept = True

            if accept:
                cur_val = new_val
                no_improve = 0
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.copy()
                    # early exit if we beat the target bound
                    if best_val <= 0.3808:
                        break
            else:
                # revert swap
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1

            # temperature schedule
            T *= cooling
            if no_improve > 5000:  # stagnation – break early
                break

        # optionally stop if we already achieved a very good bound
        if best_val <= 0.3808:
            break

    # Final evaluation (ensuring scaling is correct)
    final_val = _max_overlap(best_h)

    # Return list of floats as required by the harness
    return (best_h.tolist(), final_val, n)
