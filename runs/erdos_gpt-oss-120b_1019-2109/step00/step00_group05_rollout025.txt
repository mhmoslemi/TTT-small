import time
import math
import numpy as np

def _max_corr(h: np.ndarray) -> int:
    """
    Return the maximum value of np.correlate(h, 1 - h, mode='full').
    h is a 0/1 integer array.
    """
    corr = np.correlate(h, 1 - h, mode='full')
    return int(corr.max())

def _compute_c5(max_corr: int, n: int) -> float:
    """Convert max correlation count to the C5 value."""
    dx = 2.0 / n
    return max_corr * dx

def _random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random binary vector with exactly n/2 ones."""
    arr = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, n // 2, replace=False)
    arr[ones] = 1
    return arr

def _guided_swap(h: np.ndarray, shift: int, rng: np.random.Generator):
    """
    Perform a guided swap for the given shift.
    Returns indices (i, j) to be swapped, or (None, None) if not possible.
    """
    n = h.size
    if shift >= 0:
        length = n - shift
        a = h[:length]          # left part
        b = h[shift:]           # right part
        # positions where we have a "10" pair (mismatch contributing to max)
        A = np.where((a == 1) & (b == 0))[0]
        # positions where we have a "01" pair (the opposite)
        B = np.where((a == 0) & (b == 1))[0]
        if A.size and B.size:
            i = rng.choice(A)            # index in the left part
            j = rng.choice(B)            # index in the left part -> same offset
            return i, j
    else:  # shift < 0
        k = -shift
        length = n - k
        a = h[k:]                     # right part (indices >= k)
        b = h[:length]                # left part (indices < n-k)
        A = np.where((a == 1) & (b == 0))[0] + k  # map back to original indices
        B = np.where((a == 0) & (b == 1))[0] + k
        if A.size and B.size:
            i = rng.choice(A)
            j = rng.choice(B)
            return i, j
    return None, None

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points)
    where h_values is a 1‑D numpy array of values in [0,1],
    c5_bound is the achieved overlap constant, and
    n_points is the length of the discretisation.
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # initialise the problem size and starting vector
    # ----------------------------------------------------------------------
    try:
        # initial_h_values is provided by the harness
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        # ensure we end up with a binary vector having exactly n/2 ones
        target_ones = n // 2
        # pick the largest entries as ones (preserves any hint present)
        order = np.argsort(-init)
        h = np.zeros(n, dtype=np.int8)
        h[order[:target_ones]] = 1
    except NameError:
        # fallback: use a modest size when no parent vector exists
        n = 100
        h = _random_binary(n, rng)

    # current state
    cur_max = _max_corr(h)

    # best solution seen so far
    best_h = h.copy()
    best_max = cur_max
    best_c5 = _compute_c5(best_max, n)

    # ----------------------------------------------------------------------
    # optimisation parameters
    # ----------------------------------------------------------------------
    guided_prob = 0.6          # chance to try a guided swap
    restart_thresh = 30000     # iterations without improvement before random restart
    no_improve = 0
    iteration = 0
    # linear temperature schedule from start_temp down to near zero
    start_temp = 5.0

    # target value (optional early stopping)
    target_c5 = 0.3808
    target_max = math.floor(target_c5 * n / 2)

    while time.time() - start_time < budget_s * 0.97:

        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        temp = max(0.001, start_temp * (1.0 - frac))

        corr = np.correlate(h, 1 - h, mode='full')

        i_swap = j_swap = None

        if i_swap is None:

            zeros_idx = np.flatnonzero(1 - h)
            if ones_idx.size == 0 or zeros_idx.size == 0:

            i_swap = rng.choice(ones_idx)
