import time, random, numpy as np

# ---------- Helper functions -------------------------------------------------
def evaluate(h, dx):
    """Return max correlation value (C5) and the full correlation array."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr


def gradient_for_lag(h, lag):
    """Gradient of the correlation at a given lag (integer, may be negative)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += 1.0 - h[lag:]
            g[lag:] -= h[:m]
    else:
        k = -lag
        m = n - k
        if m > 0:
            g[k:] += 1.0 - h[:m]
            g[:m] -= h[k:]
    return g


def combined_gradient(h, lags):
    """Average gradient over all active lags."""
    g = np.zeros_like(h)
    for lag in lags:
        g += gradient_for_lag(h, lag)
    return g / len(lags)


def project_onto_box_sum(x, target_sum, eps=1e-12):
    """
    Euclidean projection of x onto the set:
        { y | 0 <= y_i <= 1, sum y_i = target_sum }.
    """
    lo, hi = -2.0, 2.0  # wide enough for any lambda
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - lo, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target_sum - y.sum()
    if abs(diff) > eps:
        free = (y > 0.0) & (y < 1.0)
        cnt = free.sum()
        if cnt > 0:
            y[free] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
    return y


def start_from_initial_or_random(n, S, init_vec=None):
    """Return a feasible start vector of length n with sum S."""
    if init_vec is not None and len(init_vec) == n:
        h = np.array(init_vec, dtype=float)
        h = project_onto_box_sum(h, S)
        return h
    # random start then projection
    h = np.random.rand(n)
    return project_onto_box_sum(h, S)


def upsample_vector(old, new_len):
    """Linear interpolation of a vector defined on a uniform grid."""
    old_len = len(old)
    x_old = np.linspace(0.0, 2.0, old_len, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_len, endpoint=False)
    return np.interp(x_new, x_old, old)
