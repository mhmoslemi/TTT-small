import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the capped simplex {0<=x_i<=1, sum x = target}
def _project_capped_simplex(v, target):
    n = v.shape[0]
    lo = np.min(v) - 1.0  # ensures all v - lo >= 1
    hi = np.max(v)        # ensures all v - hi <= 0
    for _ in range(60):   # binary search, enough for double precision
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return proj

# ------------------------------------------------------------
# Gradient of the overlap for a given lag (shift)
def _gradient_overlap(h, lag):
    N = h.shape[0]
    g = np.zeros(N, dtype=np.float64)
    if lag >= 0:
        L = lag
        I = N - L                      # length of overlapping region
        g[:I] += 1.0                   # the sum_i h_i term
        g[:I] -= h[L:]                 # -h_{i+lag}
        g[L:] -= h[:I]                 # -h_{i-lag}
    else:
        L = -lag
        I = N - L
        g[L:] += 1.0
        g[L:] -= h[:I]
        g[:I] -= h[L:]
    return g

# ------------------------------------------------------------
# Evaluate C5 for a given h (uses the pre‑imported evaluator if present)
def _c5_value(h):
    # fast local evaluation – same formula as the harness
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

# ------------------------------------------------------------
# Build a family of smooth shapes h(x) = min( A*(1-|x-1|^p), 1 )
def _shape_h(N, p):
    x = (np.arange(N) + 0.5) * (2.0 / N)   # mid‑points
    u = np.abs(x - 1.0)                    # distance from centre
    A = (p + 1.0) / (2.0 * p)              # scaling so that integral = 1 (ignoring clipping)
    h = A * (1.0 - u ** p)
    h = np.clip(h, 0.0, 1.0)
    # adjust to exact integral constraint
    target = N / 2.0
    h = _project_capped_simplex(h, target)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_limit = budget_s

    # ------------------------------------------------------------------
    # Choose a working resolution (small enough for speed, large enough for quality)
    N = 200                     # can be changed – 200 gives a good trade‑off
    target_sum = N / 2.0
    dx = 2.0 / N

    # ------------------------------------------------------------------
    # Start from a collection of candidate constructions
    candidates = []

    # (a) uniform 0.5 (always feasible)
    candidates.append(np.full(N, 0.5, dtype=np.float64))

    # (b) linear decreasing from 1 to 0 (area already 1)
    lin = np.linspace(1.0, 0.0, N, dtype=np.float64)
    candidates.append(lin)

    # (c) symmetric triangle (peak 1)
    x = (np.arange(N) + 0.5) * dx
    tri = np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)
    candidates.append(_project_capped_simplex(tri, target_sum))

    # (d) quadratic parabola (p=2)
    candidates.append(_shape_h(N, p=2.0))

    # (e) a range of p‑shapes (including clipped versions)
    p_vals = np.linspace(0.6, 2.5, 19)   # many possibilities, cheap to evaluate
    for p in p_vals:
        candidates.append(_shape_h(N, p))

    # (f) if the harness gave us a parent, use it (projected to our N)
    init_parent = globals().get('initial_h_values')
    if init_parent is not None:

            init_parent = parent_resampled

    # ------------------------------------------------------------------
    # Quick evaluation of all candidates, keep the best
    best_h = None
    best_val = np.inf
    for h in candidates:
        val = _c5_value(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # ------------------------------------------------------------------
    # Local optimisation: projected sub‑gradient descent on the worst shift
    max_iters = 12000
    for it in range(max_iters):

        # evaluate current worst shift

            delta = rng.normal(scale=0.02, size=N)

            pert = np.clip(pert, 0.0, 1.0)

        cand[i] -= delta

            best_val = val
            best_h = cand

    # ------------------------------------------------------------------
    # Ensure feasibility one last time
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_capped_simplex(best_h, target_sum)

    final_val = _c5_value(best_h)

    return best_h, float(final_val), N
