def improve_step(h):
    N = len(h)
    # compute correlation
    G = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(G)
    d = max_idx - (N - 1)  # offset shift unrolled; can be negative
    # compute derivative contributions for each index
    p = np.zeros(N)
    if d >= 0:
        # overlapping pairs: i from 0..N-1-d
        p[:N-d] += h[d:]   # h[i+d]
        p[d:] += h[:N-d]   # h[i-d]; for i>=d
    else:
        d = -d
        # shift negative, treat similarly
        p[:N-d] += h[d:]   # i+d
        p[d:] += h[:N-d]
    # now choose i_low,i_high
    # low among positions where h[i] > 0 (so can reduce)
    low_candidates = np.where(h > 1e-12)[0]
    high_candidates = np.where(h < 1 - 1e-12)[0]
    if low_candidates.size == 0 or high_candidates.size == 0:
        return h, max(G) * 2.0 / N  # no moves possible
    # indices with minimal p among low, maximal p among high
    i_low = low_candidates[np.argmin(p[low_candidates])]
    i_high = high_candidates[np.argmax(p[high_candidates])]
    if p[i_high] <= p[i_low] + 1e-12:
        # cannot improve
        return h, max(G) * 2.0 / N
    # compute delta
    max_delta = min(h[i_low], 1 - h[i_high])
    delta = max_delta * 0.5  # half max transfer
    # update
    h[i_low] -= delta
    h[i_high] += delta
    # compute new objective
    new_val = np.max(np.correlate(h, 1 - h, mode='full')) * 2.0 / N
    return h, new_val

import time
import numpy as np

def _c5_value(h):
    """Return C5 for a given step vector h."""
    n = h.shape[0]
    # correlation of h with (1-h), full mode
    corr = np.correlate(h, 1.0 - h, mode="full")
    # dx = length / n = 2 / n
    return np.max(corr) * (2.0 / n)


def _improve(h, budget):
    """Hill‑climb / simulated‑annealing style improvement."""
    start = time.time()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_value(best_h)
    cur_h = h.copy()
    cur_val = best_val

    # ---------- deterministic descent on the worst shift ----------
    while time.time() - start < budget * 0.6:
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx = int(np.argmax(corr))
        d = idx - (n - 1)                     # shift (positive => right)

        # derivative of the worst term G_d = Σ h[i]*(1-h[i+d])
        g = np.zeros(n, dtype=float)
        if d >= 0:
            length = n - d
            if length > 0:
                g[:length] = 1.0 - cur_h[d:]
        else:
            k = -d
            length = n - k
            if length > 0:
                g[k:] = 1.0 - cur_h[:length]

        # eligible indices (must belong to the overlapping region)
        mask_low = (cur_h > 1e-12) & (g > 0.0)      # we can take mass *away* here
        mask_high = (cur_h < 1.0 - 1e-12) & (g > 0.0)  # we can add mass *here*
        if not np.any(mask_low) or not np.any(mask_high):
            break

        low_ids = np.where(mask_low)[0]
        high_ids = np.where(mask_high)[0]
        i_low = low_ids[np.argmax(g[low_ids])]
        i_high = high_ids[np.argmin(g[high_ids])]

        if g[i_low] <= g[i_high] + 1e-12:
            break

        max_move = min(cur_h[i_low], 1.0 - cur_h[i_high])
        if max_move <= 1e-12:
            break
        delta = 0.5 * max_move          # conservative step
        cur_h[i_low] -= delta
        cur_h[i_high] += delta

        new_val = _c5_value(cur_h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert – no improvement
            cur_h[i_low] += delta
            cur_h[i_high] -= delta
            break

    # ---------- random local moves for the remaining time ----------
    while time.time() - start < budget:
        i, j = np.random.choice(n, 2, replace=False)
        max_move = min(cur_h[i], 1.0 - cur_h[j])
        if max_move <= 1e-12:
            continue
        delta = max_move * np.random.rand()
        cur_h[i] -= delta
        cur_h[j] += delta

        new_val = _c5_value(cur_h)
        if new_val < best_val:
            best_val = new_val
            best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta

    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)

    # ----- construct an initial feasible step function -----
    if "initial_h_values" in globals():
        base = initial_h_values.astype(float)
        rep = 4                                    # 4× more points, still <1k
        h0 = np.repeat(base, rep)
    else:
        # fallback: random feasible vector of 84 points
        n = 84
        while True:
            h0 = np.random.rand(n)
            h0 = h0 / h0.sum() * (n / 2.0)          # enforce integral = 1
            if np.all(h0 <= 1.0):

    n = h0.shape[0]
