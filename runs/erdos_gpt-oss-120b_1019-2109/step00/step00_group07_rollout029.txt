import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # Determine n_points
    if 'initial_h_values' in globals() and isinstance(initial_h_values, np.ndarray):
        n = len(initial_h_values)
    else:
        n = 80  # default
    # enforce n even because weight = n/2 must be integer
    if n % 2 != 0:
        n += 1
    w = n // 2
    dx = 2.0 / n
    
    # Setup MILP
    h = cp.Variable(n, boolean=True)
    # Create list of pairs (i,j) with i < j and (j - i) = s for s in 1..n-1
    # We'll map each pair to index in u list.
    # Could store matrix of shape (n,n) with -1 for unused.
    # But for linear constraints we need to refer to each pair.
    # Implementation: create a dictionary mapping (i,j) to variable u_ij.
    # But building many cp.Variable objects individually may be too many; we can define a vector of size num_pairs and then slice.
    # We'll create a vector u of size num_pairs and fill u for each pair.
    pair_indices = []
    pair_i = []
    pair_j = []
    # We'll also we need to construct sum of u for each shift s.
    # Use list of lists to hold slice indices for each s.
    u_s_indices = [ [] for _ in range(n) ] # s from 0..n-1 (unused s0)
    num_pairs = 0
    for i in range(n):
        for j in range(i+1, n):
            pair_i.append(i)
            pair_j.append(j)
            # shift s = j - i
            s = j - i
            u_s_indices[s].append(num_pairs)
            num_pairs += 1
    # total pairs = n*(n-1)//2
    # create u vector binary
    u = cp.Variable(num_pairs, boolean=True)
    # constraints linking u and h
    constraints = []
    # For each pair, add constraints:
    # u[k] <= h[i]
    # u[k] <= h[j]
    # u[k] >= h[i] + h[j] - 1
    # We'll use vectorized loops perhaps.
    for idx, i in enumerate(pair_i):
        j = pair_j[idx]
        ui = u[idx]
        constraints += [ui <= h[i],
                        ui <= h[j],
                        ui >= h[i] + h[j] - 1]
    # sum of h equals w
    constraints.append(cp.sum(h) == w)
    # variable t (int)
    t = cp.Variable(integer=True)
    constraints.append(t >= 0)
    # For each shift s = 0..n-1 constraints
    # shift 0: uncovered =0 automatically, but we could skip (it will be satisfied trivially)
    # Actually for s=0: j-i=0 not allowed cause pairs i=j not in u. uncover pos = sum_{i=0}^{n-1} h_i - sum h_i =0. So we can skip.
    # For s>0:
    for s in range(1, n):
        # sum left: indices i = 0..n-1-s
        left_indices = list(range(0, n - s))
        right_indices = list(range(s, n))
        # expression for sum_left
        sum_left = cp.sum(h[left_indices])
        sum_right = cp.sum(h[right_indices])
        # sum of u for this shift
        u_idx_list = u_s_indices[s]
        if len(u_idx_list) > 0:
            sum_overlap = cp.sum(u[u_idx_list])
        else:
            sum_overlap = 0
        constraints.append(sum_left - sum_overlap <= t)
        constraints.append(sum_right - sum_overlap <= t)
    # Objective: minimize t
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    # Solve with preferred solver
    try:
        prob.solve(solver=cp.CBC, warm_start=True, time_limit=int(budget_s - (time.time() - start)))
    except Exception as e:
        # fallback to ECOS_BB
        prob.solve(solver=cp.ECOS_BB, max_iters=200000)
    # Retrieve solution
    if h.value is None:
        # if fails fallback to heuristic random
        # Build random solution
        h_arr = np.zeros(n)
        ones = np.random.choice(n, w, replace=False)
        h_arr[ones] = 1.0
        c5 = np.max(np.correlate(h_arr, 1 - h_arr, mode="full")) * dx
        return h_arr, float(c5), n
    h_arr = np.round(h.value).astype(float)  # ensure 0/1
    # ensure sum exactly w, if minor numerical errors adjust
    # compute c5 bound
    corr = np.correlate(h_arr, 1 - h_arr, mode='full')
    c5 = np.max(corr) * dx
    # If we have time left, we could attempt local improvement from this solution to maybe reduce bound further.
    # Let's implement simple hill climbing limited iter count within remaining budget.
    elapsed = time.time() - start
    time_left = budget_s - elapsed
    # small local search
    if time_left > 5:
        best_h = h_arr.copy()
        best_score = c5
        # Precompute list of indices with 1 and 0
        ones_idx = np.where(best_h > 0.5)[0]
        zeros_idx = np.where(best_h < 0.5)[0]
        # We'll run a few thousand random swaps
        max_swaps = int(1e5)
        for it in range(max_swaps):
            # random swap
            i = np.random.choice(ones_idx)
            j = np.random.choice(zeros_idx)
            candidate = best_h.copy()
            candidate[i] = 0.0
            candidate[j] = 1.0
            # compute c5
            corr = np.correlate(candidate, 1 - candidate, mode='full')
            score = np.max(corr) * dx
            if score < best_score - 1e-9:
                best_h = candidate
                best_score = score
                # update index sets
                ones_idx = np.where(best_h > 0.5)[0]
                zeros_idx = np.where(best_h < 0.5)[0]
                # optional break early if best_score is already <= target
                if best_score <= 0.3808:
                    break
            # check time
            if it % 1000 == 0 and (time.time() - start) > budget_s - 0.5:
                break
        c5 = best_score
        h_arr = best_h
    return h_arr, float(c5), n

import time
import numpy as np
import cvxpy as cp

def _c5_bound(h):
    """Return C5 bound for a binary step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def run(seed=42, budget_s=1100, **kwargs):
    """Find a binary step function h on [0,2] with integral 1
    that minimizes max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points)."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Determine n_points (use the provided initial vector if any)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals() and isinstance(initial_h_values, np.ndarray):
        n = int(initial_h_values.shape[0])
    else:
        n = 80                     # reasonable default
    if n % 2 != 0:                # need integer weight w = n/2
        n += 1
    w = n // 2                    # total number of ones
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Build MILP: binary vector h (size n), auxiliary u_ij = h_i * h_j
    # Minimise t = max uncovered count over all shifts (both signs)
    # ------------------------------------------------------------------
    # binary variables for the set
    h_var = cp.Variable(n, boolean=True)

    # list all unordered pairs (i,j) with i<j
    pair_i = []
    pair_j = []
    pairs_by_shift = [[] for _ in range(n)]   # indices of pairs where j-i = s
    for i in range(n):
        for j in range(i + 1, n):
            pair_i.append(i)
            pair_j.append(j)
            s = j - i
            pairs_by_shift[s].append(len(pair_i) - 1)

    p = len(pair_i)          # number of (i,j) pairs
    u_var = cp.Variable(p, boolean=True)

    constraints = []

    # link u_ij to h_i, h_j  (standard linearisation of product)
    for idx, i in enumerate(pair_i):
        j = pair_j[idx]
        ui = u_var[idx]
        constraints += [
            ui <= h_var[i],
            ui <= h_var[j],
            ui >= h_var[i] + h_var[j] - 1,
        ]

    # exact total mass = 1  (i.e. w ones)
    constraints.append(cp.sum(h_var) == w)

    # auxiliary integer variable t (max uncovered count)
    t_var = cp.Variable(integer=True)
    constraints.append(t_var >= 0)

    # constraints for each shift s>0 (both directions)
    for s in range(1, n):
        # left side: indices 0 .. n-1-s
        left_idx = np.arange(0, n - s)
        right_idx = np.arange(s, n)

        sum_left = cp.sum(cp.hstack([h_var[i] for i in left_idx]))
        sum_right = cp.sum(cp.hstack([h_var[i] for i in right_idx]))

        u_idxs = pairs_by_shift[s]
        if u_idxs:
            ov = cp.sum(cp.hstack([u_var[k] for k in u_idxs]))
        else:
            ov = 0

        constraints.append(sum_left - ov <= t_var)   # positive shift
        constraints.append(sum_right - ov <= t_var)  # negative shift

    # objective: minimise t
    prob = cp.Problem(cp.Minimize(t_var), constraints)

    # try a fast MILP solver, fall back if unavailable
    solved = False
    for solver in (cp.CBC, cp.GLPK_MI, cp.ECOS_BB):
        try:
            prob.solve(solver=solver, warm_start=True, verbose=False)
            if prob.status in ("optimal", "optimal_inaccurate"):
                solved = True
                break
        except Exception:
            continue

    # ------------------------------------------------------------------
    # If MILP succeeded, extract solution, otherwise fall back to random.
    # ------------------------------------------------------------------
    if solved and h_var.value is not None:
        h_sol = np.rint(h_var.value).astype(int)
        # repair any tiny integrality drift
        cur = h_sol.sum()
        if cur != w:
            diff = w - cur
            if diff > 0:
                zeros = np.where(h_sol == 0)[0]
                flip = rng.choice(zeros, size=diff, replace=False)
                h_sol[flip] = 1
            else:
                ones = np.where(h_sol == 1)[0]
                flip = rng.choice(ones, size=-diff, replace=False)
                h_sol[flip] = 0
    else:
        # simple random feasible start
        h_sol = np.zeros(n, dtype=int)
        ones = rng.choice(n, size=w, replace=False)
        h_sol[ones] = 1

    best_h = h_sol.astype(float)
    best_score = _c5_bound(best_h)

    # ------------------------------------------------------------------
    # Small local search (swap a 1 with a 0) to possibly improve the bound
    # ------------------------------------------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 10.0:                     # enough time for a quick refinement
        max_swaps = int(5e5)
        ones_idx = np.flatnonzero(best_h > 0.5)
        zeros_idx = np.flatnonzero(best_h < 0.5)
        for it in range(max_swaps):
            if it % 1000 == 0:
                if time.time() - start > budget_s - 0.5:
                    break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            cand = best_h.copy()
            cand[i] = 0.0
            cand[j] = 1.0
            sc = _c5_bound(cand)
            if sc + 1e-10 < best_score:
                best_h = cand
                best_score = sc
                # update index lists incrementally
                ones_idx = np.flatnonzero(best_h > 0.5)
                zeros_idx = np.flatnonzero(best_h < 0.5)
                if best_score <= 0.3808:
                    break

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return best_h, float(best_score), n
