import numpy as np, cvxpy as cp, time

def solve_ilp(n):
    M = n // 2
    h = cp.Variable(n, boolean=True)
    t = cp.Variable(integer=True)
    constraints = [cp.sum(h) == M, t >= 0]
    shift_max = n // 5  # floor

    for s in range(1, shift_max + 1):
        w = cp.Variable(n-s, boolean=True)
        # w <= h[i] and h[i+s]
        constraints += [w <= h[:n-s], w <= h[s:], w >= h[:n-s] + h[s:] - 1]
        constraints += [cp.sum(w) >= t]

    # Maximize t
    prob = cp.Problem(cp.Maximize(t), constraints)
    prob.solve(solver=cp.CBC, verbose=False, max_iters=200000)    
    return h.value, t.value
```

def solve_min_max_mismatch(n):
    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True)
    constraints = [cp.sum(h) == M, T >= 0]
    for s in range(1, n):
        # define w
        length = n - s
        w = cp.Variable(length, boolean=True)
        # constraints linking w to h
        constraints += [w <= h[:length],
                        w <= h[s:],
                        w >= h[:length] + h[s:] - 1]
        # I_s constraints
        O_s = cp.sum(h[:length])
        J_s = cp.sum(w)
        constraints += [O_s - J_s <= T]
    # objective
    prob = cp.Problem(cp.Minimize(T), constraints)
    prob.solve(solver=cp.CBC, maxSeconds=... )  # maybe set max time
    # retrieve h, T
    h_val = np.round(h.value).astype(int)  # ensure binary
    T_val = int(round(T.value))
    return h_val, T_val, prob.status
```

def solve_for_n(n, time_limit):
    import cvxpy as cp, numpy as np, time

    M = n // 2
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True)
    constraints = [cp.sum(h) == M, T >= 0]
    for s in range(1, n):
        length = n - s
        w = cp.Variable(length, boolean=True)
        constraints += [
            w <= h[:length],
            w <= h[s:],
            w >= h[:length] + h[s:] - 1
        ]
        O = cp.sum(h[:length])
        J = cp.sum(w)
        constraints += [O - J <= T]
    prob = cp.Problem(cp.Minimize(T), constraints)
    # solve within time limit
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit)
    except Exception:
        prob.solve(solver=cp.CBC)  # fallback
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None, None
    h_val = np.rint(h.value).astype(int)
    T_val = int(round(T.value))
    c5 = T_val * 2.0 / n
    return h_val, c5, n

import time
import numpy as np

def _solve_min_mismatch(n, time_limit):
    """Solve the integer program for given n.
    Returns (h_vector, T) where T = max_{k} ∫ h(x)(1-h(x+k))dx / dx.
    """
    import cvxpy as cp

    m = n // 2  # number of 1's required
    h = cp.Variable(n, boolean=True)
    T = cp.Variable(integer=True)

    constraints = [cp.sum(h) == m, T >= 0]

    # For each shift s (1 … n‑1) introduce auxiliary variables w_{i,s}=h_i*h_{i+s}
    for s in range(1, n):
        length = n - s
        w = cp.Variable(length, boolean=True)

        constraints += [
            w <= h[:length],
            w <= h[s:],
            w >= h[:length] + h[s:] - 1,
            cp.sum(h[:length]) - cp.sum(w) <= T,   # mismatch count for this shift
        ]

    prob = cp.Problem(cp.Minimize(T), constraints)
    # give CBC a modest time budget; if it finishes early that's fine
    try:
        prob.solve(solver=cp.CBC, max_seconds=time_limit, verbose=False)
    except Exception:
        prob.solve(solver=cp.CBC, verbose=False)

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    h_val = np.rint(h.value).astype(float)  # binary vector as float
    T_val = int(round(T.value))
    return h_val, T_val


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    TARGET = 0.3808                          # desired upper bound

    # Try a few even lengths greater than the provided 42.
    # 44 is the smallest that can already beat the target if T=8.
    for n in (44, 46, 48, 50, 52, 54, 56, 58, 60):
        if time.time() - start > budget_s - 5:
            break
        # Allow up to 30 s per ILP solve (or remaining budget, whichever is smaller)
        remaining = budget_s - (time.time() - start)

        h, T = _solve_min_mismatch(n, ilp_time)
        if h is None:

        # compute C5 from T (dx = 2/n)
        c5 = T * 2.0 / n
        # verify with the official evaluator (it also applies the dx factor)
        c5_check = evaluate_erdos_solution(h)
        if c5_check < TARGET:

    # If ILP did not succeed, fall back to a local‑search heuristic.
    # Use the initial construction if available, otherwise a random one.
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        n = len(h)

        h[idx] = 1.0

    dx = 2.0 / n
    def c5_of(vec):
        return np.max(np.correlate(vec, 1.0 - vec, mode='full') * dx)

    best_h = h.copy()
    best_c5 = c5_of(best_h)

    # simple simulated‑annealing / hill‑climbing
    ones = list(np.where(best_h == 1)[0])
    zeros = list(np.where(best_h == 0)[0])
    cur_h = best_h.copy()
    cur_c5 = best_c5
    it = 0
