def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n
    n = len(initial_h_values) if 'initial_h_values' in globals() else 200
    # Ensure n even; if odd, adjust weight and incorporate fractional? We'll assume even.

    target_c5 = 0.3808

    # function to compute max overlap
    def max_overlap(vec):
        return int(np.correlate(vec, 1 - vec, mode='full').max())
    
    # local improvement
    def improve(vec):
        # greedy reduction
        while True:
            corr = np.correlate(vec, 1 - vec, mode='full')
            max_val = int(corr.max())
            # find first shift with max
            idx = int(np.argmax(corr))
            k = idx - (len(vec) - 1)
            if k >= 0:
                # positions i where vec[i]==1 and vec[i+k]==0
                ones_to_zero = np.where((vec[:n - k] == 1) & (vec[k:] == 0))[0]
                zeros_to_one = np.where((vec[:n - k] == 0) & (vec[k:] == 1))[0]
            else:
                pos = -k
                ones_to_zero = np.where((vec[pos:] == 1) & (vec[:n - pos] == 0))[0] + pos
                zeros_to_one = np.where((vec[pos:] == 0) & (vec[:n - pos] == 1))[0]
            if ones_to_zero.size == 0 or zeros_to_one.size == 0:
                break
            i = rng.choice(ones_to_zero)
            j = rng.choice(zeros_to_one)
            # swap
            vec[i], vec[j] = vec[j], vec[i]
        return vec, max_overlap(vec)
    
    best_vec = None
    best_val = n  # initialize high
    # initial vector from initial_h if exists (convert to binary)
    if 'initial_h_values' in globals():

        # threshold to binary
        bin_vec = (init > 0.5).astype(np.int8)
        # adjust number of ones if needed
        ones_needed = n // 2
        cur_ones = bin_vec.sum()
        if cur_ones != ones_needed:
            diff = ones_needed - cur_ones
            # if diff >0 need to turn some zeros to ones
            zeros_idx = np.where(bin_vec == 0)[0]
            if diff > 0:
                selected = rng.choice(zeros_idx, size=diff, replace=False)
                bin_vec[selected] = 1
            else:
                ones_idx = np.where(bin_vec == 1)[0]
                selected = rng.choice(ones_idx, size=-diff, replace=False)
                bin_vec[selected] = 0

        best_vec = vec.copy()

import numpy as np
import time
import math

def max_overlap(vec: np.ndarray) -> int:
    """Maximum of the overlap integral (without the dx factor)."""
    # vec is binary (0/1) or any values in [0,1]
    return int(np.correlate(vec, 1 - vec, mode="full").max())


def greedy_improve(vec: np.ndarray, rng: np.random.Generator) -> tuple:
    """
    Greedy local optimisation for a binary vector.
    Tries to reduce the maximum overlap by swapping 1/0 pairs that directly
    contribute to the current worst shift.
    Returns the improved vector and its max overlap.
    """
    vec = vec.copy()
    n = len(vec)

    while True:
        corr = np.correlate(vec, 1 - vec, mode="full")
        max_val = int(corr.max())
        if max_val == 0:                     # perfect – cannot improve further
            break

        idx = int(np.argmax(corr))           # first shift attaining the max
        k = idx - (n - 1)                    # shift (positive, zero or negative)

        if k >= 0:
            # indices i where (1 at i, 0 at i+k)
            ones_to_zero = np.where((vec[: n - k] == 1) & (vec[k:] == 0))[0]
            # indices i where (0 at i, 1 at i+k) – opposite direction
            zeros_to_one = np.where((vec[: n - k] == 0) & (vec[k:] == 1))[0]
        else:
            pos = -k
            # (1 at i, 0 at i-pos)
            ones_to_zero = np.where((vec[pos:] == 1) & (vec[: n - pos] == 0))[0] + pos
            # (0 at i, 1 at i-pos)
            zeros_to_one = np.where((vec[pos:] == 0) & (vec[: n - pos] == 1))[0]

        if ones_to_zero.size and zeros_to_one.size:
            i = rng.choice(ones_to_zero)
            j = rng.choice(zeros_to_one)
            vec[i], vec[j] = vec[j], vec[i]   # swap, guaranteed to lower the max
            continue

        # No direct improving swap for this shift – try a few random swaps
        improved = False
        for _ in range(30):
            ones_idx = np.flatnonzero(vec)
            zeros_idx = np.flatnonzero(1 - vec)
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            vec[i], vec[j] = vec[j], vec[i]
            new_max = max_overlap(vec)
            if new_max < max_val:
                improved = True
                break
            vec[i], vec[j] = vec[j], vec[i]   # revert
        if not improved:
            break

    return vec, max_overlap(vec)

    start_time = time.time()

    # ------------------------------------------------------------------ #

    except NameError:

        n += 1

    # Helper to create a random balanced binary vector.
    def random_binary():
        v = np.zeros(n, dtype=np.int8)
        v[rng.choice(n, size=n // 2, replace=False)] = 1
        return v

    # ------------------------------------------------------------------ #
    # Initialise best solution.
    best_vec = None
    best_val = n                                 # a value larger than any possible max

    # ------------------------------------------------------------------ #
    # If a parent construction is available, start from it (converted to binary).
    if "initial_h_values" in globals():
        init = np.array(initial_h_values, copy=False)
        bin_vec = (init > 0.5).astype(np.int8)   # simple threshold to 0/1
        # Adjust number of ones to exactly n/2
        needed = n // 2
        cur = int(bin_vec.sum())
        if cur != needed:

            if diff > 0:

    # Final bound computation and return.
    final_c5 = best_val * 2.0 / n
    return best_vec.astype(float), float(final_c5), n
