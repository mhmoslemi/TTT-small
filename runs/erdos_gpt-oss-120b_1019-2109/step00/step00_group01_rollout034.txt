def project_to_feasible(x, target_sum):
    """Project vector x onto the set {0 <= y <= 1, sum(y) = target_sum}."""
    # Clip may be needed after shifting? We'll directly find tau such that sum(clipped) = target_sum.
    low = np.min(x) - 1.0  # maybe -1 to ensure sum > target for low
    high = np.max(x)       # ensures sum <= target?
    # However if target_sum > n, but target_sum = n/2 ≤ n, safe.
    for _ in range(50):
        tau = (low + high) / 2.0
        y = np.clip(x - tau, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            low = tau
        else:
            high = tau
    y = np.clip(x - (low+high)/2.0, 0.0, 1.0)
    # final slight correction for numerical errors
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        # distribute diff among entries not at bounds
        # find indices where 0<y<1 with capacity for adding diff
        free = (y > 1e-12) & (y < 1 - 1e-12)
        # if none, just adjust a small number of entries arbitrarily
        if free.any():
            # Distribute equally
            add = diff / free.sum()
            y[free] += add
            # after addition may violate bounds, clip again
            y = np.clip(y, 0.0, 1.0)
        else:
            # fallback scaling
            y = np.clip(y + diff / len(y), 0.0, 1.0)
    return y

import time
import math
import numpy as np

def _project_to_simplex(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project x onto {0 ≤ y ≤ 1 , sum(y)=target_sum}."""
    # Clip to bounds first
    y = np.clip(x, 0.0, 1.0)
    # Bisection on a scalar shift τ so that sum(clip(y-τ,0,1)) = target_sum
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(60):
        tau = (lo + hi) * 0.5
        z = np.clip(y - tau, 0.0, 1.0)
        s = z.sum()
        if s > target_sum:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) * 0.5
    z = np.clip(y - tau, 0.0, 1.0)

    # Small residual correction (rare)
    diff = target_sum - z.sum()
    if abs(diff) > 1e-12:
        free = (z > 1e-12) & (z < 1.0 - 1e-12)
        if free.any():
            z[free] += diff / free.sum()
            z = np.clip(z, 0.0, 1.0)
        else:
            # All variables are at a bound – distribute uniformly (should hardly happen)
            z += diff / len(z)
            z = np.clip(z, 0.0, 1.0)
    return z


def _c5_value(h: np.ndarray) -> float:
    """Return C₅ for a step vector h (using the evaluation rule of the harness)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _gradient_worst_shift(h: np.ndarray) -> (np.ndarray, int):
    """
    Compute sub‑gradient of the max‑overlap objective for the worst shift.
    Returns (grad, shift) where shift is the integer displacement (negative allowed).
    """
    n = h.size
    # full autocorrelation of h with itself
    conv = np.correlate(h, h, mode="full")
    S = n / 2.0
    f_vals = S - conv                      # f_d = S - conv_d
    idx = int(np.argmax(f_vals))           # index in full array
    d = idx - (n - 1)                      # actual shift (can be negative)

    # gradient of f_d = 1 - (h_{i+d}+h_{i-d})
    grad = np.ones_like(h)
    if d >= 0:
        grad[: n - d] -= h[d:]
        grad[d:]     -= h[: n - d]
    else:
        d = -d
        grad[d:]     -= h[: n - d]
        grad[: n - d] -= h[d:]
    return grad, d


def _improve_h(h0: np.ndarray, budget: float) -> np.ndarray:
    """Simple projected sub‑gradient descent trying to lower the C₅ value."""
    n = h0.size
    target_sum = n / 2.0
    h = h0.copy()
    best = h.copy()
    best_val = _c5_value(best)

    start = time.time()
    max_iters = 5000
    lr0 = 0.5

    for it in range(max_iters):
        # time guard
        if time.time() - start > budget * 0.94:
            break

        grad, _ = _gradient_worst_shift(h)
        lr = lr0 / math.sqrt(it + 1.0)
        h = _project_to_simplex(h - lr * grad, target_sum)

        # occasional random jitter to escape tiny plateaus
        if it % 400 == 0 and it > 0:
            jitter = np.random.normal(scale=0.02, size=n)
            h = _project_to_simplex(h + jitter, target_sum)

        cur = _c5_value(h)
        if cur < best_val:
            best_val = cur
            best = h.copy()
            # early exit if already below the known record
            if best_val <= 0.38085:
                break

    return best


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function for the Erdős minimum‑overlap problem.

    The return value is a tuple (h_values, c5_bound, n_points) where
        h_values – list of step heights on the uniform grid,
        c5_bound – the achieved upper bound for C₅,
        n_points – number of grid points (length of h_values).
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64).copy()
    else:
        # a modest default: a smooth hat shape that respects the integral
        n = 200
        x = (np.arange(n) + 0.5) * (2.0 / n)          # mid‑points in [0,2]
        h0 = np.maximum(0.0, 1.0 - np.abs(x - 1.0))   # triangular hat
        # scale to exact integral 1
        target = n / 2.0
        h0 = _project_to_simplex(h0, target)

    n = h0.size
    target_sum = n / 2.0
    h0 = _project_to_simplex(h0, target_sum)

    # ------------------------------------------------------------------
    # optimisation loop (may run several restarts)
    # ------------------------------------------------------------------
    remaining = budget_s
    start_time = time.time()
    best_h = h0
    best_val = _c5_value(best_h)

    # first attempt from the given start
    best_h = _improve_h(best_h, remaining)
    best_val = _c5_value(best_h)

    # a quick secondary random restart if time permits
    if time.time() - start_time < budget_s * 0.6:
        # random feasible vector (uniform Dirichlet then clip)
        rnd = np.random.rand(n)
        rnd = _project_to_simplex(rnd, target_sum)
        cand = _improve_h(rnd, budget_s - (time.time() - start_time))
        cand_val = _c5_value(cand)
        if cand_val < best_val:
            best_h, best_val = cand, cand_val

    # ------------------------------------------------------------------
    # final result
    # ------------------------------------------------------------------
    final_c5 = _c5_value(best_h)
    return (best_h.tolist(), float(final_c5), int(n))
