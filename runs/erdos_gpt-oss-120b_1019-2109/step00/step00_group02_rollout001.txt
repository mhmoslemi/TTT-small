def project_capped_simplex(v, target):
    # Project v onto the box [0,1] with sum equal to target.
    # Finds alpha such that sum_i clip(v_i+alpha,0,1) = target.
    low = -np.max(v) - 1.0   # guarantee all v+low <= 0
    high = 1.0 - np.min(v) + 1.0  # guarantee all v+high >= 1
    for _ in range(60):

        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, 0.0, 1.0)

    # Returns the overlap constant for the discrete step function h.
    # C5 = max_k ∫ h(x)(1-h(x+k))dx ≈ max(convolution) * dx
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

        s = -shift
        if s < N:
            grad[s:] += h[:N-s]
            grad[:N-s] += h[s:]
    return grad

import numpy as np
import time

def project_capped_simplex(v, target):
    """
    Euclidean projection of vector v onto the set:
        {x | 0 <= x_i <= 1, sum(x_i) = target }.
    Finds a scalar α such that x_i = clip(v_i + α, 0, 1) and sum(x)=target.
    """
    low = -np.max(v) - 1.0          # guarantees all v+low <= 0 → sum = 0
    high = 1.0 - np.min(v) + 1.0   # guarantees all v+high >= 1 → sum = len(v)
    for _ in range(60):            # enough for double precision
        mid = (low + high) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, 0.0, 1.0)

def evaluate_h(h, dx):
    """
    Compute the Erdős overlap bound for a discrete step function h.
    C5 = max_k ∫ h(x) (1 - h(x+k)) dx ≈ max(correlation) * dx.
    """
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def grad_min_corr(h, shift):
    """
    Gradient of the autocorrelation at the shift giving the minimum value.
    For a given integer shift (positive or negative) the gradient w.r.t. h is:
        ∂/∂h_i  Σ h_i h_{i+shift} = h_{i+shift} + h_{i-shift}
    with out‑of‑range indices ignored.
    """
    N = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        s = shift
        if s < N:
            grad[:N - s] += h[s:]
            grad[s:] += h[:N - s]
    else:
        s = -shift
        if s < N:
            grad[s:] += h[:N - s]
            grad[:N - s] += h[s:]
    return grad
