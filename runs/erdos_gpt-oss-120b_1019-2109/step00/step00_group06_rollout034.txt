import time
import math
import numpy as np

def _delta_c(h, a, b):
    """Compute correlation change when swapping a (value 1) and b (value 0)
    for binary vector h."""
    n = h.size
    m = 2 * n - 1
    off = n - 1
    delta = np.zeros(m, dtype=np.int32)

    # -h[a+s]   for all s  -> slice covering a..a+(n-1)
    delta[off - a:off - a + n] -= h
    # +h[b+s]
    delta[off - b:off - b + n] += h
    # -h[a-s]   -> reversed slice
    delta[off + a - (n - 1):off + a + 1] -= h[::-1]
    # +h[b-s]
    delta[off + b - (n - 1):off + b + 1] += h[::-1]

    # X3 term
    delta[off] += 2
    d = b - a
    delta[off + d] -= 1
    delta[off - d] -= 1
    return delta

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # --- parameters ---------------------------------------------------------
    n = 200                     # even number of points (dx = 2/n)
    target = n // 2             # sum of h must be exactly target
    max_no_improve = 5000       # iterations without improvement before restart
    # -----------------------------------------------------------------------

    # initial random binary vector with exactly target ones
    h = np.zeros(n, dtype=np.int8)
    h[rng.choice(n, target, replace=False)] = 1

    # correlation of h with itself (full mode)
    c = np.correlate(h, h, mode='full')
    min_c = int(c.min())
    best_h = h.copy()
    best_min_c = min_c

    # maintain lists of positions of ones and zeros for fast sampling
    ones = list(np.flatnonzero(h == 1))
    zeros = list(np.flatnonzero(h == 0))

    offset = n - 1
    attempts = 0

    while True:
        if time.time() - start > budget_s * 0.96:
            break

        # worst shift (where correlation is minimal)
        min_idx = int(np.argmin(c))
        shift = min_idx - offset   # signed shift value

        # try a targeted swap that directly improves this worst shift
        improved = False
        if shift >= 0:
            length = n - shift
            h1 = h[:length]          # h[i]
            h2 = h[shift:]           # h[i+shift]
            mask_one = (h1 == 1) & (h2 == 0)   # i where (1,0)
            mask_zero = (h1 == 0) & (h2 == 1)  # i where (0,1)
            if mask_one.any() and mask_zero.any():
                i = rng.choice(np.flatnonzero(mask_one))
                j = rng.choice(np.flatnonzero(mask_zero))
                a = j + shift          # position of a 1
                b = i + shift          # position of a 0
        else:
            s = -shift
            length = n - s
            h1 = h[s:]               # h[i]
            h2 = h[:length]          # h[i-s]
            mask_one = (h1 == 1) & (h2 == 0)   # i where (1,0)
            mask_zero = (h1 == 0) & (h2 == 1)  # i where (0,1)
            if mask_one.any() and mask_zero.any():
                i = rng.choice(np.flatnonzero(mask_one)) + s
                j = rng.choice(np.flatnonzero(mask_zero)) + s
                a = j - s              # 1 moves from here
                b = i - s              # 0 moves here

        # if targeted move not available, fall back to a random swap
        if not ('a' in locals()):
            a = rng.choice(ones)
            b = rng.choice(zeros)

        # compute the effect of swapping a (1) and b (0)
        delta = _delta_c(h, a, b)
        new_c = c + delta
        new_min_c = int(new_c.min())

        # accept only if we improve the minimal correlation
        if new_min_c > min_c:
            # perform the swap
            h[a] = 0
            h[b] = 1
            # update ones/zeros lists
            ones.remove(a)
            zeros.append(a)
            zeros.remove(b)
            ones.append(b)
            # commit new correlation
            c = new_c
            min_c = new_min_c
            attempts = 0
            # record best solution
            if min_c > best_min_c:
                best_min_c = min_c
                best_h = h.copy()
        else:
            attempts += 1
            # if we have been unsuccessful for a while, restart
            if attempts >= max_no_improve:
                # random restart
                h = np.zeros(n, dtype=np.int8)
                h[rng.choice(n, target, replace=False)] = 1
                c = np.correlate(h, h, mode='full')
                min_c = int(c.min())
                ones = list(np.flatnonzero(h == 1))
                zeros = list(np.flatnonzero(h == 0))
                attempts = 0

        # clean up temporary locals for next iteration
        if 'a' in locals():
            del a, b
        # safety break if best possible is reached (cannot exceed target)
        if best_min_c == target:
            break

    dx = 2.0 / n
    c5 = dx * (target - best_min_c)
    return best_h.astype(float), float(c5), n
