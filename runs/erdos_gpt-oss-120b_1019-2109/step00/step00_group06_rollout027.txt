import numpy as np, time, math
def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    cor = np.correlate(h, 1 - h, mode='full')
    return np.max(cor) * dx

def build_h(n, b,c,g0,g1):
    a = 1.0 - 2*b - 2*c
    g2 = 0.5 - g0 - g1
    # Clip to avoid negative due to numeric
    a = max(a, 0.0)
    g2 = max(g2, 0.0)
    h = np.zeros(n, dtype=float)
    dx = 2.0 / n
    pos = 0.0
    # left outer zero gap g0
    pos += g0
    # outer left interval c
    start = pos
    pos += c

    # inner left zero g1
    pos += g1
    # inner left interval b
    start = pos
    pos += b
    fill_interval(h, start, pos, dx)
    # middle zero g2
    pos += g2
    # central interval a
    start = pos
    pos += a
    fill_interval(h, start, pos, dx)
    # middle zero g2
    pos += g2
    # inner right interval b

    pos += b

def make_feasible_params(b,c,g0,g1):
    # enforce b,c,g0,g1 >=0 and constraints
    b = max(b,0.0)
    c = max(c,0.0)
    g0 = max(g0,0.0)
    g1 = max(g1,0.0)
    # constraint for ones
    s = b + c
    if s > 0.5:
        scale = 0.5 / s
        b *= scale
        c *= scale
    # constraint for gaps
    t = g0 + g1
    if t > 0.5:
        scale = 0.5 / t
        g0 *= scale
        g1 *= scale
    return b,c,g0,g1

def project_to_constraints(v, target):
    import numpy as np
    # We need to find lambda such that sum(clipped(v+lam,a=0,b=1)) = target
    # Use binary search
    low = -np.max(v)  # ensures all v+low <= 0
    high = 1.0 - np.min(v)  # ensures all v+high >=1
    for _ in range(60):
        mid = (low + high) / 2.0
        clipped = np.clip(v + mid, 0.0, 1.0)
        s = clipped.sum()
        if s < target:
            low = mid
        else:
            high = mid
    lam = (low + high) / 2.0
    return np.clip(v + lam, 0.0, 1.0)

import time
import math
import numpy as np

# ---------- helpers -------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h), zero‑padded (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_to_constraints(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project a vector onto the convex set
        { x | 0 ≤ x_i ≤ 1,  sum_i x_i = target }.
    Uses a simple 1‑D bisection for the Lagrange multiplier.
    """
    low = -np.max(v)                     # makes all entries ≤0 after shift
    high = 1.0 - np.min(v)                # makes all entries ≥1 after shift
    for _ in range(60):
        mid = (low + high) * 0.5
        clipped = np.clip(v + mid, 0.0, 1.0)
        s = clipped.sum()
        if s < target:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def build_h_int(n: int, c: int, b: int, g0: int, g1: int) -> np.ndarray:
    """
    Construct a binary step function (0/1) with the symmetric 5‑interval pattern
    described in the analysis.
    Parameters are integer counts of samples (not physical lengths).
    """
    half = n // 2                     # total number of ones / zeros
    a = half - 2 * c - 2 * b          # central block length (≥0)
    g2 = half - g0 - g1               # middle zero block length (≥0)

    h = np.zeros(n, dtype=float)
    pos = 0

    pos += g0                         # left outer zeros
    h[pos:pos + c] = 1.0              # outer left ones
    pos += c
    pos += g1                         # left inner zeros
    h[pos:pos + b] = 1.0              # inner left ones
    pos += b
    pos += g2                         # left middle zeros
    h[pos:pos + a] = 1.0              # central ones
    pos += a
    pos += g2                         # right middle zeros
    h[pos:pos + b] = 1.0              # inner right ones
    pos += b
    pos += g1                         # right inner zeros
    h[pos:pos + c] = 1.0              # outer right ones
    pos += c
    # remaining right outer zeros (g0) are implicit
    return h


def random_feasible_params(rng: np.random.Generator, half: int):
    """Sample a feasible (c,b,g0,g1) integer tuple."""
    # choose (c,b) with c+b ≤ half/2
    max_sum = half // 2
    t = rng.integers(0, max_sum + 1)
    c = rng.integers(0, t + 1)
    b = t - c
    # choose (g0,g1) similarly
    t2 = rng.integers(0, max_sum + 1)
    g0 = rng.integers(0, t2 + 1)
    g1 = t2 - g0
    return int(c), int(b), int(g0), int(g1)


def propose_int(params, rng: np.random.Generator, half: int):
    """
    Produce a neighboring integer state by adding a small random delta.
    Returns None if the proposal violates the linear constraints.
    """
    c, b, g0, g1 = params
    max_sum = half // 2

    var = rng.integers(0, 4)               # which component to change
    delta = rng.integers(-5, 6)            # step size (may be zero)
    if delta == 0:
        delta = 1

    if var == 0:                           # modify c
        nc = c + delta
        nb = b
        if 0 <= nc <= max_sum and nc + nb <= max_sum:
            return (nc, nb, g0, g1)
    elif var == 1:                         # modify b
        nb = b + delta
        nc = c
        if 0 <= nb <= max_sum and nc + nb <= max_sum:
            return (nc, nb, g0, g1)
    elif var == 2:                         # modify g0
        ng0 = g0 + delta
        ng1 = g1
        if 0 <= ng0 <= max_sum and ng0 + ng1 <= max_sum:
            return (c, b, ng0, ng1)
    else:                                  # modify g1
        ng1 = g1 + delta
        ng0 = g0
        if 0 <= ng1 <= max_sum and ng0 + ng1 <= max_sum:
            return (c, b, ng0, ng1)
    return None


def gradient_descent(h: np.ndarray, steps: int, rng: np.random.Generator):
    """
    Small deterministic fine‑tuning of a binary/continuous step function.
    Uses the sub‑gradient of the maximal correlation entry.
    """
    n = h.shape[0]
    target = n / 2

    for it in range(steps):
        # correlation vector
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)               # signed shift producing the max

        grad = np.zeros_like(h)

        if shift >= 0:
            grad[: n - shift] += 1.0 - h[shift:]
            grad[shift:]       -= h[: n - shift]
        else:
            s = -shift
            grad[s:]           += 1.0 - h[: n - s]
            grad[: n - s]      -= h[s:]

        # step size schedule (larger at start, smaller later)
        eps = 0.5 * (1.0 - it / steps) * (2.0 / n)
        h_new = h - eps * grad
        # project back to the feasible set
        h = project_to_constraints(h_new, target)

    return h

    start = time.time()

    # choose a modest sample size (must be even)

        h0 = project_to_constraints(h0, half)
        best_h = h0.copy()

    # -----------------------------------------------------------------
    # Simulated annealing on the 5‑interval integer parametrisation
    cur_params = random_feasible_params(rng, half)
    cur_h = build_h_int(n, *cur_params)
    cur_score = compute_c5(cur_h)
    if cur_score < best_score:
        best_score, best_h = cur_score, cur_h.copy()

    # annealing schedule
    max_iters = 300000
    T0 = 0.12
    T_min = 1e-6

    it = 0
    while it < max_iters and time.time() - start < budget_s * 0.95:
        frac = it / max_iters
        T = T0 * ((T_min / T0) ** frac)        # exponential cooling

        cand_params = propose_int(cur_params, rng, half)
        if cand_params is None:
            it += 1
            continue

        cand_h = build_h_int(n, *cand_params)
        cand_score = compute_c5(cand_h)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < math.exp((cur_score - cand_score) / (T + 1e-12)):
            cur_params, cur_h, cur_score = cand_params, cand_h, cand_score

    best_h = gradient_descent(best_h.copy(), steps=2000, rng=rng)
    best_score = compute_c5(best_h)

    best_h = project_to_constraints(best_h, half)
    best_score = compute_c5(best_h)
