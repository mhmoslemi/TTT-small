def compute_max_corr(h):
    n = len(h)
    a = h.astype(float)
    b = 1 - a
    # Prepare length for linear convolution = 2*n
    size = 1
    while size < 2*n:
        size <<= 1
    A = np.fft.rfft(np.concatenate([a, np.zeros(size - n)]), n=size)
    B_rev = np.fft.rfft(np.concatenate([b[::-1], np.zeros(size - n)]), n=size)
    conv = np.fft.irfft(A * B_rev, n=size)
    # conv length = size; but actual linear conv length = n + n -1 = 2*n -1.
    conv = conv[:2*n-1]  # take first required length
    corr = conv[::-1]  # reverse
    max_corr = np.max(corr) * (2.0 / n)  # multiply dx = 2/n
    return max_corr

def max_corr(h):
    n = len(h)
    a = h.astype(np.float64)
    b = 1.0 - a
    size = 1 << ((2*n - 1).bit_length())
    # but a safe factor: size = 1
    # Use np.fft.rfft for real input
    # pad a and b_rev to size
    A = np.fft.rfft(np.concatenate([a, np.zeros(size - n)]))
    B_rev = np.fft.rfft(np.concatenate([b[::-1], np.zeros(size - n)]))
    conv = np.fft.irfft(A * B_rev, n=size)
    conv = conv[:2*n-1]
    corr = conv[::-1]   # full correlation
    max_corr = np.max(corr) * (2.0 / n)
    return max_corr

def shift_array(h, lag):
    n = len(h)
    if lag >= 0:
        shifted = np.empty_like(h)
        shifted[:n-lag] = h[lag:]
        shifted[n-lag:] = 0.0
    else:
        l = -lag
        shifted = np.empty_like(h)
        shifted[l:] = h[:n-l]
        shifted[:l] = 0.0
    return shifted

def compute_full_corr(h):
    n = len(h)
    a = h.astype(np.float64)
    b = 1.0 - a
    size = 1 << ((2*n-1).bit_length())
    A = np.fft.rfft(np.concatenate([a, np.zeros(size - n)]))
    B_rev = np.fft.rfft(np.concatenate([b[::-1], np.zeros(size - n)]))
    conv = np.fft.irfft(A * B_rev, n=size)
    conv = conv[:2*n-1]
    corr = conv[::-1]
    return corr

def compute_c5(h):
    corr = compute_full_corr(h)
    return np.max(corr) * (2.0 / len(h))

def simple_projection(h, target_sum):
    h = np.clip(h, 0, 1)
    diff = target_sum - h.sum()
    # iterate up to e.g., 10 times
    for _ in range(10):
        if abs(diff) < 1e-8:
            break
        # indices eligible for increase if diff>0 (not at 1) else decrease if diff<0 (not at 0)
        if diff > 0:
            free = (h < 1 - 1e-12)
        else:
            free = (h > 1e-12)
        nfree = free.sum()
        if nfree == 0:
            break
        delta = diff / nfree
        # update
        h[free] += delta
        # clip again
        h = np.clip(h, 0, 1)
        diff = target_sum - h.sum()
    # final clip for safety
    h = np.clip(h, 0, 1)
    return h

Define project_sum_box(v, target_sum):
    """Project onto [0,1] with given sum."""
    # Clip to bounds
    v = np.clip(v, 0, 1)
    # compute discrepancy
    diff = target_sum - v.sum()
    # iterative water-filling
    # We'll loop up to 20 times.
    for _ in range(20):
        if abs(diff) < 1e-8:
            break
        if diff > 0:
            free = (v < 1 - 1e-12)
        else:
            free = (v > 1e-12)
        nfree = free.sum()
        if nfree == 0:
            break
        delta = diff / nfree
        # Apply delta but ensuring bounds
        if diff > 0:
            # increase
            v[free] += delta
            v = np.clip(v, 0, 1)
        else:
            # decrease
            v[free] += delta  # delta negative
            v = np.clip(v, 0, 1)
        diff = target_sum - v.sum()
    return v

import numpy as np
import time
import random

# ---------- helper functions -------------------------------------------------

def project_sum_box(v, target):
    """
    Project vector v onto the set {x | sum(x)=target, 0<=x<=1}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0  # bounds for lambda; sufficient for values in [0,1]
    for _ in range(40):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lo, 0.0, 1.0)


def compute_full_corr(h):
    """
    Linear correlation of h with (1-h) using FFT.
    Returns the full correlation array of length 2*len(h)-1
    (lags from -(n-1) … +(n-1)).
    """
    n = h.shape[0]
    a = h.astype(np.float64)
    b = 1.0 - a
    size = 1 << ((2 * n - 1).bit_length())    # next power of two
    A = np.fft.rfft(np.concatenate([a, np.zeros(size - n)]))
    B = np.fft.rfft(np.concatenate([b[::-1], np.zeros(size - n)]))
    conv = np.fft.irfft(A * B, n=size)
    conv = conv[: 2 * n - 1]                 # keep linear part
    return conv[::-1]                        # reverse to get correlation


def c5_from_corr(corr, n):
    """Convert raw correlation (sum) to the integral bound C5."""
    return float(corr.max() * (2.0 / n))


def init_h(n):
    """
    Produce an initial feasible vector of length n.
    Uses the provided initial_h_values if present; otherwise a random binary vector.
    The result satisfies 0<=h<=1 and sum(h)=n/2.
    """
    target = n / 2.0
    if 'initial_h_values' in globals():
        init = initial_h_values
        k = init.shape[0]
        # Interpolate the parent onto the new grid
        x_old = np.linspace(0, 2, k, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
        h = np.clip(h, 0.0, 1.0)
        h = project_sum_box(h, target)
        return h
    # fallback: random binary
    h = np.zeros(n)
    h[: n // 2] = 1.0
    np.random.shuffle(h)
    return h


# ---------- main optimization ------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # Choose a moderate resolution; larger n gives more flexibility.
    n = 200
    if n % 2:
        n += 1

    # initial feasible h (real‑valued)
    h_cur = init_h(n)
    corr_cur = compute_full_corr(h_cur)
    val_cur = c5_from_corr(corr_cur, n)

    best_h = h_cur.copy()
    best_val = val_cur

    # simulated‑annealing parameters
    T = 0.02                # initial temperature
    decay = 0.9996          # exponential cooling
    it = 0
    no_improve = 0
    max_no_improve = 20000

    while time.time() - start_time < budget_s * 0.95:
        # propose a fractional swap i<->j that keeps the sum unchanged
        i, j = np.random.choice(n, 2, replace=False)
        # random direction of transfer
        if random.random() < 0.5:
            # move from j to i
            max_inc = min(1.0 - h_cur[i], h_cur[j])
            if max_inc <= 1e-12:
                continue
            delta = random.random() * max_inc
            h_new = h_cur.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            # move from i to j
            max_inc = min(1.0 - h_cur[j], h_cur[i])
            if max_inc <= 1e-12:
                continue
            delta = random.random() * max_inc
            h_new = h_cur.copy()
            h_new[j] += delta
            h_new[i] -= delta

        # evaluate
        corr_new = compute_full_corr(h_new)
        val_new = c5_from_corr(corr_new, n)

        delta_val = val_new - val_cur
        if delta_val < -1e-12 or random.random() < np.exp(-delta_val / (T + 1e-12)):
            h_cur = h_new
            corr_cur = corr_new
            val_cur = val_new
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # temperature schedule
        T *= decay
        it += 1

        # occasional restart from the best found so far
        if it % 5000 == 0:

            T = 0.02
            no_improve = 0

    # final bound recomputed for safety

    final_bound = c5_from_corr(final_corr, n)

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_sum_box(best_h, n / 2.0)

    return best_h.tolist(), float(final_bound), n
