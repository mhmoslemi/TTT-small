import numpy as np, time, random

def project_sum_bounds(v, target):
    # binary search for lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo+hi)*0.5, 0.0, 1.0)

def compute_cost(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c.max() * dx, c

def compute_gradient(h, lag):
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if lag >= 0:
        g[:n-lag] -= h[lag:]
        g[lag:] -= h[:n-lag]
    else:
        k = -lag
        g[k:] -= h[:n-k]
        g[:n-k] -= h[k:]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    random.seed(seed)
    np.random.seed(seed)

    # compute initial cost
    cost_val, _ = compute_cost(h)
    best_h = h.copy()

    start = time.time()
    # parameters
    max_iters = 500000
    iter_no_improve = 0

    for it in range(max_iters):
        if time.time() - start > budget_s * 0.95:
            break
        # compute current cost and lag
        cost_val, corr = compute_cost(h)
        # gradient descent step on worst lag
        idx = int(np.argmax(corr))
        lag = idx - (n - 1)
        grad = compute_gradient(h, lag)

        # step size adapt
        # Use step proportional to current cost
        step = 0.5 * (cost_val / (np.linalg.norm(grad) + 1e-12))
        # ensure step not too large
        if step > 0.5:
            step = 0.5
        # try line search
        improved = False
        for _ in range(5):

            # projection
            h_new = project_sum_bounds(h_new, target_sum)
            new_cost, _ = compute_cost(h_new)
            if new_cost < cost_val - 1e-12:

                cost_val = new_cost
                improved = True
                if new_cost < best_cost - 1e-12:

import time
import random
import numpy as np

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the simplex {0<=x<=1, sum(x)=target}."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _cost_and_corr(h: np.ndarray):
    """Return C5 (max overlap) and the full correlation array."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx, corr


def _gradient_max_lag(h: np.ndarray, lag: int):
    """Sub‑gradient of the active lag (the one attaining the max)."""
    n = h.shape[0]
    g = np.ones_like(h)
    if lag >= 0:
        g[: n - lag] -= h[lag:]
        g[lag:] -= h[: n - lag]
    else:
        k = -lag
        g[k:] -= h[: n - k]
        g[: n - k] -= h[k:]
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # -----------------------------------------------------------------
    # initialise h
    # -----------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)  # provided by the harness
        n = h0.shape[0]
    except Exception:
        n = 70
        h0 = np.random.rand(n)
    target_sum = n / 2.0
    h = _project_to_sum(np.clip(h0, 0.0, 1.0), target_sum)

    best_h = h.copy()
    best_val, _ = _cost_and_corr(best_h)

    start = time.time()
    it = 0
    max_iters = 300_000  # plenty for the time budget
    while it < max_iters and (time.time() - start) < budget_s * 0.94:
        it += 1

        # current cost and the lag attaining the maximum
        cur_val, cur_corr = _cost_and_corr(h)
        idx = int(np.argmax(cur_corr))
        lag = idx - (n - 1)

        # gradient descent step on the worst lag
        grad = _gradient_max_lag(h, lag)
        norm_g = np.linalg.norm(grad) + 1e-12
        step = 0.5 * cur_val / norm_g
        step = min(step, 0.5)

        improved = False
        for _ in range(5):
            cand = h - step * grad
            cand = _project_to_sum(cand, target_sum)
            cand_val, _ = _cost_and_corr(cand)
            if cand_val < cur_val - 1e-12:
                h = cand
                cur_val = cand_val
                improved = True
                if cand_val < best_val - 1e-12:
                    best_val = cand_val
                    best_h = cand.copy()
                break
            step *= 0.5

        # occasional random perturbation to escape plateaus
        if not improved and random.random() < 0.07:
            i = random.randrange(n)
            j = random.randrange(n - 1)
            if j >= i:
                j += 1
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta > 1e-9:
                delta = max_delta * random.random() * 0.3
                cand = h.copy()
                cand[i] -= delta
                cand[j] += delta
                cand_val, _ = _cost_and_corr(cand)
                if cand_val < cur_val - 1e-12:
                    h = cand
                    cur_val = cand_val
                    if cand_val < best_val - 1e-12:
                        best_val = cand_val
                        best_h = cand.copy()

        # if we have stagnated for a while, restart near the best solution
        if it % 20000 == 0 and not improved:
            h = best_h + np.random.randn(n) * 0.015
            h = np.clip(h, 0.0, 1.0)
            h = _project_to_sum(h, target_sum)

    # final polishing
    best_h = _project_to_sum(np.clip(best_h, 0.0, 1.0), target_sum)
    final_val, _ = _cost_and_corr(best_h)

    # if the harness supplies a reference evaluator, use it for the reported value
    try:
        final_val = float(evaluate_erdos_solution(best_h))
    except Exception:
        pass

    return best_h.tolist(), final_val, n
