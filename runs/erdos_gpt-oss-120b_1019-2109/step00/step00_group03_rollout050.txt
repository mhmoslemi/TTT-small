import numpy as np
import time
import math

def _max_corr_count(h: np.ndarray) -> int:
    """Maximum overlap count for a binary step function."""
    # np.correlate returns a float array; values are integers for binary h.
    return int(round(np.max(np.correlate(h, 1.0 - h, mode="full"))))


def _greedy_search(
    h: np.ndarray,
    N: int,
    half: int,
    target_max: int,
    start: float,
    budget: float,
) -> tuple[np.ndarray, int]:
    """Iteratively apply the best 1↔0 swap that reduces the maximum overlap."""
    cur_max = _max_corr_count(h)
    while True:
        # time guard
        if time.time() - start > budget - 0.1:
            break

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        best_swap = None
        best_val = cur_max

        # examine every possible swap
        for i1 in ones:
            for i0 in zeros:
                # perform temporary swap
                h[i1] = 0.0
                h[i0] = 1.0
                new_val = _max_corr_count(h)
                if new_val < best_val:
                    best_val = new_val
                    best_swap = (i1, i0)
                # revert
                h[i0] = 0.0
                h[i1] = 1.0

        if best_swap is None:
            break  # local optimum reached

        # apply the improving swap
        i1, i0 = best_swap
        h[i1] = 0.0
        h[i0] = 1.0
        cur_max = best_val

        # early exit if we already hit the target
        if cur_max <= target_max:
            break

    return h, cur_max


def _random_binary(N: int, half: int) -> np.ndarray:
    """Create a random binary vector with exactly `half` ones."""
    h = np.zeros(N, dtype=np.float64)
    idx = np.random.choice(N, half, replace=False)
    h[idx] = 1.0
    return h


def _paired_binary(N: int, half: int) -> np.ndarray:
    """
    Construct a vector where most ones appear in mirrored pairs (i,i+N/2)
    except for one single extra 1 to achieve the required total.
    """
    half_len = N // 2
    pair_cnt = (half - 1) // 2          # number of mirrored pairs (both 1)
    pair_idx = np.random.choice(half_len, pair_cnt, replace=False)

    h = np.zeros(N, dtype=np.float64)
    for i in pair_idx:
        h[i] = 1.0
        h[i + half_len] = 1.0

    # currently 2*pair_cnt ones = half-1; add the final single 1
    zeros = np.where(h == 0)[0]
    extra = np.random.choice(zeros)
    h[extra] = 1.0
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised into n points)
    with integral 1 that minimises the Erdős overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    # ----------------- parameters -----------------
    N = 70                     # number of sample points (must be even)
    half = N // 2              # required number of ones
    dx = 2.0 / N
    # target max count corresponding to C5 < 0.3808
    _target = int(math.floor(0.3808 / dx + 1e-9))

    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_val = N + 1          # initialise with a large value

    # ---- optional use of a provided initial construction ----
    try:
        init_arr = np.array(initial_h_values, dtype=np.float64)
        if init_arr.shape[0] == N:
            # convert to a binary vector with exactly `half` ones
            idx = np.argpartition(init_arr, -half)[-half:]
            h0 = np.zeros(N, dtype=np.float64)
            h0[idx] = 1.0
            h0, val = _greedy_search(h0, N, half, _target, start_time, budget_s)
            if val < best_val:
                best_val = val
                best_h = h0.copy()
    except Exception:
        # either `initial_h_values` does not exist or is malformed
        pass

    # ----------------- main stochastic search -----------------
    while time.time() - start_time < budget_s - 0.2:
        # pick a random initial vector (pure random or paired)
        if np.random.rand() < 0.5:
            cur = _random_binary(N, half)
        else:
            cur = _paired_binary(N, half)

        cur, val = _greedy_search(cur, N, half, _target, start_time, budget_s)

        if val < best_val:
            best_val = val
            best_h = cur.copy()
            if best_val <= _target:
                break   # we reached the desired bound

    # If nothing succeeded (unlikely), fall back to a simple random vector
    if best_h is None:
        best_h = _random_binary(N, half)
        best_val = _max_corr_count(best_h)

    c5_bound = best_val * dx
    return best_h, c5_bound, N
