import numpy as np
import time
import math

def _compute_c5(h):
    """Return the overlap bound C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode="full") * dx
    return float(conv.max())

def _mismatch_pairs(h, k):
    """
    For a given shift k (integer, -n+1 .. n-1) return two index arrays:
    - idx_i: positions i where h[i]=1 and h[i+k]=0  (mismatch contributing to max)
    - idx_j: positions i where h[i]=0 and h[i+k]=1  (a zero that would match a one after swap)
    """
    n = h.shape[0]
    if k >= 0:
        i_range = np.arange(n - k)
        mism_i = (h[i_range] == 1) & (h[i_range + k] == 0)
        mism_j = (h[i_range] == 0) & (h[i_range + k] == 1)
        idx_i = i_range[mism_i]
        idx_j = i_range[mism_j]
    else:
        k_abs = -k
        i_range = np.arange(k_abs, n)
        mism_i = (h[i_range] == 1) & (h[i_range - k_abs] == 0)
        mism_j = (h[i_range] == 0) & (h[i_range - k_abs] == 1)
        idx_i = i_range[mism_i]
        idx_j = i_range[mism_j]
    return idx_i, idx_j

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with values in [0,1] and integral 1
    that minimizes C5 = max_k ∫ h(x)*(1−h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    def _now(): return time.time() - start_time

    best_h = None
    best_c5 = np.inf

    # consider any provided initial construction
    if 'initial_h_values' in globals():
        try:
            h0 = np.asarray(initial_h_values, dtype=float)
            c5_0 = _compute_c5(h0)
            best_h = h0.copy()
            best_c5 = c5_0
        except Exception:
            pass  # ignore malformed initial data

    # target C5 below 0.3808
    TARGET = 0.3808

    # main search parameters
    n = 100                     # even number, half ones gives exact integral
    half = n // 2
    dx = 2.0 / n

    # helper to evaluate quickly
    def _max_mismatch(vec):
        return int(np.correlate(vec, 1 - vec, mode="full").max())

    # main optimisation loop – restart many times while time permits
    while _now() < budget_s * 0.94:
        # random binary start with exactly half ones
        h = np.zeros(n, dtype=np.int8)
        ones = rng.choice(n, size=half, replace=False)
        h[ones] = 1

        # compute initial raw max mismatch (integer)
        M_raw = np.correlate(h, 1 - h, mode="full")
        cur_max = int(M_raw.max())

        # local improvement loop
        no_improve = 0
        # bound on inner iterations – generous, actual stop driven by time
        for _ in range(200000):
            if _now() >= budget_s * 0.94:
                break

            # record any global improvement
            cur_c5 = cur_max * dx
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.astype(float).copy()
                no_improve = 0
                if best_c5 < TARGET:
                    # excellent – finish early
                    break

            # pick a shift attaining the current maximum
            max_idxs = np.flatnonzero(M_raw == cur_max)
            s_idx = rng.choice(max_idxs)
            k = s_idx - (n - 1)

            # try a directed swap that reduces mismatches for this shift
            idx_i, idx_j = _mismatch_pairs(h, k)
            if idx_i.size and idx_j.size:
                i = rng.choice(idx_i)
                j = rng.choice(idx_j)
                # swap 1 at i with 0 at j
                h[i] = 0
                h[j] = 1
                M_raw = np.correlate(h, 1 - h, mode="full")
                cur_max = int(M_raw.max())
                continue

            # fallback: random swap of any 1 and 0, accept only if not worsening
            ones_idx = np.flatnonzero(h == 1)
            zeros_idx = np.flatnonzero(h == 0)
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)
            h[i] = 0
            h[j] = 1
            new_M = np.correlate(h, 1 - h, mode="full")
            new_max = int(new_M.max())
            if new_max <= cur_max:
                M_raw = new_M
                cur_max = new_max
                no_improve = 0
            else:
                # occasional uphill acceptance
                if rng.random() < 0.001:
                    M_raw = new_M
                    cur_max = new_max
                    no_improve = 0
                else:
                    # revert swap
                    h[i] = 1
                    h[j] = 0
                    no_improve += 1

            if no_improve > 3000:
                # give up on this start
                break

        if best_c5 < TARGET:
            break

    # final safety: if nothing found, fall back to simple uniform half‑ones
    if best_h is None:
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, size=half, replace=False)] = 1
        best_h = h.astype(float)
        best_c5 = _compute_c5(best_h)

    # ensure the reported bound matches a fresh evaluation
    best_c5 = _compute_c5(best_h)

    return best_h, float(best_c5), int(best_h.shape[0])
