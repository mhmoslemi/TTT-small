def hillclimb_binary(n, weight, max_iter, time_limit):
    # initialize binary random
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, weight, replace=False)
    h[ones] = 1.0
    best_h = h.copy()
    best_val = evaluate(h)
    # precompute indices for ones and zeros
    # In each iteration:
    for it in range(max_iter):
        # time check
        if time.time() - start > time_limit:
            break
        # choose random 1 to 0 swap
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        val = evaluate(h)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
        else:
            # revert back
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_val
```

def local_refine(h, steps, time_limit):
    n = len(h)
    best_val = evaluate(h)
    best_h = h.copy()
    for _ in range(steps):
        if time.time() > time_limit:
            break
        # choose two indices p with h[p] < 1 and q with h[q] > 0
        p_candidates = np.where(best_h < 1)[0]
        q_candidates = np.where(best_h > 0)[0]
        if len(p_candidates) == 0 or len(q_candidates) == 0:
            break
        p = np.random.choice(p_candidates)
        q = np.random.choice(q_candidates)
        # choose delta: limited by bounds and sum constraint; we want to increase p and decrease q
        max_delta = min(1 - best_h[p], best_h[q])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * 0.5  # up to half max
        new_h = best_h.copy()
        new_h[p] += delta
        new_h[q] -= delta
        val = evaluate(new_h)
        if val < best_val:
            best_val = val
            best_h = new_h
    return best_h, best_val
```

def simulated_anneal(h0, time_limit):
    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_val = evaluate(h0)
    cur_h = h0.copy()
    cur_val = best_val
    start = time.time()
    # Precompute indices where h not at bounds
    # Generate lists for each iteration
    T0 = 0.1  # initial temperature relative to typical objective scale
    # Use schedule: T = T0 * (1 - fraction_elapsed) maybe
    while time.time() - start < time_limit:
        # fraction of elapsed relative to time_limit
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac)  # linearly decreasing to 0
        # propose change
        # pick i where cur_h[i] > 1e-12, j where cur_h[j] < 1 - 1e-12
        # get random selections
        i_candidates = np.where(cur_h > 1e-8)[0]
        j_candidates = np.where(cur_h < 1 - 1e-8)[0]
        if len(i_candidates) == 0 or len(j_candidates) == 0:
            break
        i = np.random.choice(i_candidates)
        j = np.random.choice(j_candidates)
        max_delta = min(cur_h[i], 1 - cur_h[j])
        if max_delta <= 0:
            continue
        delta = max_delta * np.random.rand() * 0.5  # up to half allowed
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = evaluate(new_h)
        dval = new_val - cur_val
        if dval < 0 or np.random.rand() < np.exp(-dval / max(T, 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
    return best_h, best_val

    best_h = cur_h.copy()

        if elapsed >= time_limit:
            break
        # Temperature schedule: start high then gradually drop
        # Set initial temperature using typical variation ~ (0.1)
        frac = elapsed / time_limit
        T = max(0.01, 0.2 * (1 - frac))   # linear cooling from 0.2 to 0.01
        # Determine candidates
        over_idxs = np.where(cur_h > 1e-12)[0]
        under_idxs = np.where(cur_h < 1 - 1e-12)[0]
        if over_idxs.size == 0 or under_idxs.size == 0:
            break
        i = rng.choice(over_idxs)
        j = rng.choice(under_idxs)
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        # delta scaling: bigger steps early, smaller later
        # Scale delta by (0.5 * max_delta * (1 - frac) + 0.001)

        new_h[j] += delta
        new_val = evaluate(new_h)

import time

def _evaluate(h: np.ndarray) -> float:
    """Return C5 for a step function represented by array h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h), zero‑padded, mode='full'
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _simulated_anneal(h0: np.ndarray, time_limit: float, rng: np.random.Generator) -> tuple:
    """Simple SA that preserves the integral exactly (by transferring mass)."""
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_val = _evaluate(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    # initial temperature; a few percent of a typical C5 value
    T0 = 0.15
    while True:
        elapsed = time.time() - start
        if elapsed >= time_limit:
            break
        # linear cooling schedule
        frac = elapsed / time_limit
        T = max(1e-4, T0 * (1.0 - frac))
        # pick donor (value>0) and receiver (value<1)
        donors = np.where(cur_h > 1e-12)[0]
        receivers = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = rng.choice(donors)
        j = rng.choice(receivers)
        # maximum transferable mass without leaving bounds
        max_delta = min(cur_h[i], 1.0 - cur_h[j])
        if max_delta <= 0:
            continue
        # step size shrinks as the algorithm proceeds
        step_factor = max(0.5 * (1.0 - frac), 0.02)
        delta = max_delta * rng.random() * step_factor
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = _evaluate(new_h)
        dval = new_val - cur_val
