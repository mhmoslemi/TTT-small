def gradient_of_shift(h, shift):
    n = len(h)
    # need vector of length n
    # compute (1 - h[i+shift]) for i+shift in range else 0
    # compute -h[i-shift] for i-shift in range else 0.
    g = np.zeros_like(h)
    if shift >= 0:
        # (1 - h[i+shift]) term: for i = 0..n-1-shift
        g[:n-shift] += 1 - h[shift:]
        # -h[i - shift] term: for i = shift..n-1
        g[shift:] -= h[:n-shift]
    else:
        s = -shift
        # shift negative: i+shift = i - s
        # (1 - h[i+shift]) term: for i = s..n-1
        g[s:] += 1 - h[:n-s]
        # -h[i-shift] term: -h[i + s] term (since i - (-s) = i + s)
        g[:n-s] -= h[s:]
    return g

import numpy as np, time, math

def project_onto_capped_simplex(v, S):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        total = proj.sum()
        if total > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

def compute_max_corr_int(h):
    # h is 1D array of floats (or ints)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return int(corr.max())

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def gradient_shift_vec(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        # term (1 - h[i+s]) for i from 0 to n-s-1
        g[:n-s] += 1.0 - h[s:]
        # term -h[i-s] for i from s to n-1
        g[s:] -= h[:n-s]
    else:
        s = -shift
        # shift negative:
        # (1 - h[i-s]) term for i from s to n-1
        g[s:] += 1.0 - h[:n-s]
        # -h[i+s] term for i from 0 to n-s-1
        g[:n-s] -= h[s:]
    return g

def gradient_descent(h, S, time_limit, start_time):
    cur = h.copy()
    best = cur.copy()
    best_val = compute_c5(cur)
    n = len(cur)
    iter = 0
    # initial step size
    alpha0 = 0.8
    while True:
        t = time.time()
        if t - start_time > time_limit - 0.5:
            break
        # compute correlation
        corr = np.correlate(cur, 1 - cur, mode='full')
        idx_max = int(np.argmax(corr))
        shift = idx_max - (n - 1)
        # step size schedule (e.g., alpha = alpha0 / sqrt(iter+1))
        alpha = alpha0 / math.sqrt(iter + 1)
        g = gradient_shift_vec(cur, shift)
        cur = cur - alpha * g
        # project
        cur = np.clip(cur, 0.0, 1.0)
        cur = project_onto_capped_simplex(cur, S)
        cur_val = compute_c5(cur)
        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best = cur.copy()
        iter += 1
    return best, best_val

import numpy as np, time, math

def _project_capped_simplex(v, S):
    """Project v onto {0≤x≤1, sum x = S}."""
    lo = np.min(v) - 1.0          # low enough so that v-lo≥1 everywhere
    hi = np.max(v)                # high enough so that v-hi ≤0 everywhere
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _c5_value(h):
    """C5 for a concrete h (float vector)."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _c5_int_count(h):
    """Return the integer correlation count (without dx scaling)."""
    return int(np.max(np.correlate(h, 1 - h, mode="full")))


def _grad_shift(h, shift):
    """Gradient of C(shift) w.r.t. h (sub‑gradient of the max)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        g[: n - s] += 1.0 - h[s:]          # (1 - h[i+shift])
        g[s:]      -= h[: n - s]          # - h[i-shift]
    else:
        s = -shift
        g[s:]      += 1.0 - h[: n - s]    # (1 - h[i-shift])
        g[: n - s] -= h[s:]               # -h[i+shift]
    return g


def _gradient_descent(start, S, time_budget):
    """Projected sub‑gradient descent on max‑C5."""
    n = start.shape[0]
    cur = start.copy()
    best = cur.copy()
    best_val = _c5_value(best)
    start_time = time.time()
    it = 0
    alpha0 = 0.8
    while time.time() - start_time < time_budget - 0.1:
        corr = np.correlate(cur, 1 - cur, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        # step size decays like 1/sqrt(it)
        alpha = alpha0 / math.sqrt(it + 1.0)
        g = _grad_shift(cur, shift)
        cur = cur - alpha * g
        cur = np.clip(cur, 0.0, 1.0)
        cur = _project_capped_simplex(cur, S)
        val = _c5_value(cur)
        if val + 1e-12 < best_val:
            best_val = val
            best = cur.copy()
        it += 1
    return best, best_val


def _binary_sa(n, S, time_budget):
    """Simple simulated‑annealing on binary {0,1} vectors."""
    h = np.zeros(n, dtype=np.int8)
    h[np.random.choice(n, S, replace=False)] = 1
    cur_val = _c5_int_count(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    it = 0
    T0 = 200.0
    while time.time() - start < time_budget - 0.1:
        # pick two different indices with opposite values
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        # swap
        h[i], h[j] = h[j], h[i]
        new_val = _c5_int_count(h)
        delta = new_val - cur_val
        T = T0 * (0.9995 ** it)
        accept = delta <= 0 or np.random.rand() < math.exp(-delta / max(T, 1e-12))
        if accept:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        it += 1
    # scale to C5 value
    return best_h.astype(float), best_val * (2.0 / n)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # decide dimension – use 200 points unless a different size is forced
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        n = max(200, n0)               # at least 200 for good resolution
    else:
        n = 200
    S = n / 2.0                        # required sum of h
    dx = 2.0 / n

    # ----- Phase 1 : binary simulated annealing (if we have time) -----
    time_left = budget_s - (time.time() - start_time)
    best_h = None
    best_val = float("inf")

    if time_left > 30.0:               # give SA at least a short window
        sa_time = min(0.4 * time_left, 30.0)
        h_bin, val_bin = _binary_sa(n, int(S), sa_time)
        best_h = h_bin
        best_val = val_bin
        time_left = budget_s - (time.time() - start_time)

    # ----- Phase 2 : continuous projected gradient descent restarts -----
    while time.time() - start_time < budget_s - 0.5:
        # start from a random feasible point (or from previous best)
        if best_h is not None and np.random.rand() < 0.3:
            start_vec = best_h.copy()
        else:
            start_vec = np.random.rand(n)
        start_vec = _project_capped_simplex(start_vec, S)

        # allocate a small chunk of the remaining budget to this run
        remaining = budget_s - (time.time() - start_time)
        run_budget = min(5.0, remaining - 0.2)
        if run_budget <= 0:
            break
        h_opt, val_opt = _gradient_descent(start_vec, S, run_budget)
        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()

    # final projection and evaluation
    if best_h is None:
        # fallback: uniform half‑value
        best_h = np.full(n, 0.5)
    best_h = _project_capped_simplex(best_h, S)
    final_c5 = _c5_value(best_h)

    return best_h, final_c5, n
