def random_feasible(n):

    remaining = target
    # fill randomly
    indices = np.arange(n)
    np.random.shuffle(indices)
    for i in indices:
        if remaining <= 0:
            break
        max_add = min(1.0, remaining)
        # sample uniform between 0 and max_add
        add = np.random.rand() * max_add
        h[i] = add
        remaining -= add
    # If some remaining due to micro rounding (should be zero)

        for i in idxs:
            if remaining <= 1e-12:
                break
            max_add = min(1.0, remaining)
            add = np.random.rand() * max_add
            h[i] = add
            remaining -= add
        # Tiny residual fix
        cur = h.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            mask = (h > 1e-12) & (h < 1 - 1e-12)
            mids = np.where(mask)[0]
            if mids.size:
                inc = diff / mids.size
                h[mids] = np.clip(h[mids] + inc, 0.0, 1.0)
            else:
                h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        # final clip and tiny renorm
        h = np.clip(h, 0.0, 1.0)
        cur = h.sum()
        if abs(cur - target) > 1e-9:
            h = h * (target / cur)
            h = np.clip(h, 0.0, 1.0)
        return h

    def upsample_initial_to_n(target_n):
        """Interpolate the global initial_h_values to a different length."""
        init = np.asarray(initial_h_values, dtype=np.float64)
        m = init.shape[0]
        x_init = np.linspace(0.0, 2.0, m, endpoint=False)
        x_tgt = np.linspace(0.0, 2.0, target_n, endpoint=False)
        h = np.interp(x_tgt, x_init, init)
        target_sum = target_n / 2.0
        cur_sum = h.sum()
        diff = target_sum - cur_sum
        h += diff / target_n
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            mask = (h > 0) & (h < 1)
            if mask.any():
                scale = target_sum / cur_sum
                h[mask] = np.clip(h[mask] * scale, 0.0, 1.0)
            cur_sum = h.sum()
            diff2 = target_sum - cur_sum
            if abs(diff2) > 1e-12:
                # distribute residual uniformly over non‑boundary entries
                interior = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
                if interior.size:
                    inc = diff2 / interior.size
                    h[interior] = np.clip(h[interior] + inc, 0.0, 1.0)
                else:
                    for i in range(target_n):
                        if diff2 > 0 and h[i] < 1:
                            add = min(diff2, 1 - h[i]); h[i] += add; diff2 -= add
                        elif diff2 < 0 and h[i] > 0:
                            sub = min(-diff2, h[i]); h[i] -= sub; diff2 += sub
                        if abs(diff2) < 1e-12:
                            break
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) > 1e-8:
            h = h * (target_sum / cur_sum)
            h = np.clip(h, 0.0, 1.0)
        return h

    def slsqp_opt(start_h):
        """Run a brief SLSQP optimisation from start_h."""
        n = start_h.shape[0]
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n
        eq = {'type': 'eq', 'fun': lambda x: np.sum(x) - target}
        # small maxiter to stay fast
        opts = {'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        try:
            res = minimize(lambda x: compute_c5(x), start_h, method='SLSQP',
                           bounds=bounds, constraints=[eq], options=opts)
            x = np.clip(res.x, 0.0, 1.0)
            # enforce sum (tiny numerical drift)
            cur = x.sum()
            diff = target - cur
            if abs(diff) > 1e-9:
                for _ in range(100):
                    i = np.random.randint(n)
                    if diff > 0 and x[i] < 1:
                        inc = min(diff, 1 - x[i]); x[i] += inc; diff -= inc
                    elif diff < 0 and x[i] > 0:
                        dec = min(-diff, x[i]); x[i] -= dec; diff += dec
                    if abs(diff) < 1e-12:
                        break
            x = np.clip(x, 0.0, 1.0)
            return x, compute_c5(x)
        except Exception:
            return None, None

    def local_refine(h, max_iters=2000, start_temp=0.05):
        """Simple simulated‑annealing refinement preserving the sum."""
        n = len(h)
        target = n / 2.0
        best = h.copy()
        best_val = compute_c5(best)
        cur = h.copy()
        cur_val = best_val
        for it in range(max_iters):
            # abort if out of time
            if time.time() - start_time > budget_s * 0.95:
                break
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # shrink step size linearly
            step = 0.1 * (1.0 - it / max_iters)
            # delta added to i, subtracted from j
            lower = max(-cur[i], cur[j] - 1.0, -step)
            upper = min(1.0 - cur[i], cur[j], step)
            if lower >= upper:
                continue
            delta = np.random.uniform(lower, upper)
            cur[i] += delta
            cur[j] -= delta
            val = compute_c5(cur)
            temp = start_temp * (1.0 - it / max_iters)
            accept = False
            if val < cur_val:
                accept = True
            elif temp > 1e-12:
                if np.random.rand() < math.exp(-(val - cur_val) / temp):
                    accept = True
            if accept:
                cur_val = val
                if val < best_val:
                    best_val = val
                    best = cur.copy()
            else:
                # revert move
                cur[i] -= delta
                cur[j] += delta
        return best, best_val

    # ---------- optimisation driver ----------
    best_h = None
    best_val = np.inf

    # candidate sample sizes (include the known 42)
    candidates = [42, 60, 84, 120, 150]

    for n in candidates:
        if time.time() - start_time > budget_s * 0.95:
            break
        for attempt in range(3):
            if time.time() - start_time > budget_s * 0.95:
                break
            # pick a starting vector
            if attempt == 0 and 'initial_h_values' in globals():
                if len(initial_h_values) == n:
                    h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
                    # project onto exact sum if needed
                    diff = n / 2.0 - h0.sum()
                    if abs(diff) > 1e-9:
                        h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)
                else:
                    try:
                        h0 = upsample_initial_to_n(n)
                    except Exception:
                        h0 = random_feasible(n)
            else:
                h0 = random_feasible(n)

            # first a quick SLSQP run
            x, val = slsqp_opt(h0)
            if x is not None and val is not None:
                if val < best_val:
                    best_val = val
                    best_h = x.copy()
                # polish with local refinement
                iters = max(500, int(2500 * 42 / n))
                refined, rval = local_refine(x, max_iters=iters, start_temp=0.04)
                if rval < best_val:
                    best_val = rval
                    best_h = refined.copy()

    # additional pure‑random hill‑climbs if time permits
    while time.time() - start_time < budget_s * 0.9:
        n = 120  # a comfortable size for extra searches
        h0 = random_feasible(n)
        refined, rval = local_refine(h0, max_iters=2000, start_temp=0.05)
        if rval < best_val:
            best_val = rval
            best_h = refined.copy()

    # fallback if nothing succeeded
    if best_h is None:
        n = 60
        best_h = random_feasible(n)
        best_val = compute_c5(best_h)

    # final sanity: enforce constraints exactly
    n = len(best_h)
    target_sum = n / 2.0
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-9:
        # distribute over non‑boundary entries
        mask = (best_h > 1e-12) & (best_h < 1 - 1e-12)
        idxs = np.where(mask)[0]
        if idxs.size:
            inc = diff / idxs.size
            best_h[idxs] = np.clip(best_h[idxs] + inc, 0.0, 1.0)
        else:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_val = compute_c5(best_h)

    return (best_h.tolist(), float(best_val), n)
