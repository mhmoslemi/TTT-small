    def repair(h):
        np.clip(h, 0., 1., out=h)
        total = h.sum()
        target = n / 2
        if abs(total - target) > 1e-12:
            # adjust proportionally
            diff = target - total
            # Compute capacity for positive/negative adjustments
            if diff > 0:
                # need to increase sum: can increase non-1 entries
                cap = (1 - h).sum()
                if cap > 0:
                    # scale adjustment across each entry by fraction of its capacity
                    # Add diff * ((1 - h) / cap)
                    h += diff * ((1 - h) / cap)
                else:
                    # nothing to increase, fallback
                    pass
            else:
                # need to decrease sum: reduce non-zero entries
                cap = h.sum()
                if cap > 0:
                    h += diff * (h / cap)  # diff negative
                else:
                    pass
            # clip again to ensure bounds

        return h

import numpy as np
import time
import math

def _repair_vec(v, target_sum):
    """Clamp to [0,1] and adjust linearly to hit target_sum."""
    np.clip(v, 0.0, 1.0, out=v)
    cur = v.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return v
    # distribute the missing/excess mass proportionally to the available capacity
    if diff > 0:  # need to add mass
        cap = (1.0 - v).sum()
        if cap > 0:
            v += diff * (1.0 - v) / cap
    else:        # need to remove mass
        cap = v.sum()
        if cap > 0:
            v += diff * v / cap
    np.clip(v, 0.0, 1.0, out=v)
    return v

def _cost(h, dx):
    """Return the objective C5 for a given step function."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr.max()

def _propose_swap(h, rng):
    """Transfer a random amount from a larger entry to a smaller one."""
    n = h.shape[0]
    i, j = rng.choice(n, 2, replace=False)
    # decide direction so we always move from larger to smaller
    if h[i] > h[j]:
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            return h.copy()
        delta = rng.rand() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0:
            return h.copy()
        delta = rng.rand() * max_delta
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
    return h_new
