def make_shape_q(q, n):
    x = (np.arange(n) + 0.5) * (2.0 / n) # midpoints
    s = (q + 1) / (2 * q)  # scaling factor to get area 1
    h = s * (1 - np.abs(x - 1) ** q)
    # ensure within bounds
    h = np.clip(h, 0, 1)
    # repair sum
    target = n / 2
    sum_h = h.sum()
    if sum_h != target:
        # scale by target/sum_h (should be close to 1)
        # but scaling may exceed 1; do iterative
        diff = target - sum_h
        # distribute diff to non-saturated entries
        # find indices where h not at 0 or 1 to adjust
        for i in range(n):
            if 0 < h[i] < 1:
                # adjust this element by diff
                new_val = h[i] + diff
                if new_val < 0:
                    diff += h[i]  # reduce diff by amount we can add (h[i]-0)
                    h[i] = 0
                elif new_val > 1:
                    diff -= (1 - h[i])
                    h[i] = 1
                else:
                    h[i] = new_val
                    diff = 0
                    break
        # if still diff leftover, adjust uniform small up to tolerance
        if abs(diff) > 1e-9:
            h = h + diff / n
            h = np.clip(h, 0, 1)
    return h
```

import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral using the discrete correlation definition."""
    n = h.size
    # correlation of h with (1-h), zero‑padded (mode='full')
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def repair(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce the exact integral constraint sum(h)=n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # try to fix the sum by adjusting a single entry that can absorb the diff
    for _ in range(200):
        i = np.random.randint(n)
        new_val = h[i] + diff
        if 0.0 <= new_val <= 1.0:
            h[i] = new_val
            return h
    # fallback: distribute uniformly over non‑boundary entries
    mask = (h > 0.0) & (h < 1.0)
    cnt = int(mask.sum())
    if cnt > 0:
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    else:
        # extreme case: all entries are 0 or 1 – adjust the first entry
        h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def shape_w(w: float, n: int) -> np.ndarray:
    """Symmetric piecewise‑linear shape with a flat top of width w."""
    L = 1.0 - w / 2.0                     # length of each sloping side
    h_max = 1.0 / (1.0 + w / 2.0)         # plateau height (ensures integral=1)
    x = (np.arange(n) + 0.5) * (2.0 / n)   # mid‑point positions
    if L <= 0.0:
        h = np.full(n, h_max)
    else:

        right = x >= 2.0 - L

        h = np.empty(n, dtype=float)

        h[left] = (h_max / L) * x[left]
        # right slope: h_max → 0 over [2‑L, 2]
        h[right] = (h_max / L) * (2.0 - x[right])
        # flat top
        h[middle] = h_max
    return repair(h)

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_val = compute_c5(new_h)
        diff = new_val - cur_val
        if diff < 0.0 or np.random.rand() < math.exp(-diff / (T + 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()

        # occasional global perturbation to escape plateaus
        if it % 10000 == 0 and np.random.rand() < 0.02:
            rnd = np.random.rand(n)
            rnd = rnd / rnd.mean() * 0.5
            rnd = repair(rnd)
            rnd_val = compute_c5(rnd)
            if rnd_val < best_val:
                best_val = rnd_val
                best_h = rnd.copy()
                cur_h = rnd.copy()
                cur_val = rnd_val

    # Final repair and evaluation
    best_h = repair(best_h)
    best_val = compute_c5(best_h)

    # Return as list to avoid possible serialization issues
    return (best_h.tolist(), best_val, n)
