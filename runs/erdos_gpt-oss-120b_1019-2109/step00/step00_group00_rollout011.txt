def evaluate_global(h):

def objective_global(h):

def gradient_global(h):
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    offset = idx - (_n - 1)
    grad = np.zeros_like(h)
    if offset >= 0:
        grad[:_n - offset] += (1 - h[offset:])
        grad[offset:] -= h[:_n - offset]
    else:
        off = -offset
        grad[off:] += (1 - h[:_n - off])
        grad[:_n - off] -= h[off:]
    grad *= _dx
    return grad

    return np.sum(h) - _target

def refine_h(h, time_end):
    h_cur = h.copy()
    c_cur = evaluate_global(h_cur)
    while time.time() < time_end:
        # maybe try small random adjustments
        i, j = np.random.choice(_n, 2, replace=False)

        neg = min(h_cur[j], 1 - h_cur[i])

import time
import numpy as np

# global placeholders filled in `run`
_n = None          # number of discretization points
_dx = None         # step size 2/_n
_target = None     # required sum of h values (=_n/2)


def evaluate_global(h):
    """Return C5 for the full vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * _dx


def objective_global(h):
    """Objective for the optimizer: C5."""
    return evaluate_global(h)


def gradient_global(h):
    """Sub‑gradient of the max correlation term (scaled by dx)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    offset = idx - (_n - 1)          # shift corresponding to maximal entry
    grad = np.zeros_like(h)

    if offset >= 0:
        # contributions where h[i] multiplies (1‑h[i+offset])
        grad[: _n - offset] += (1.0 - h[offset:])
        # contributions where h[i] appears as the second factor
        grad[offset:] -= h[: _n - offset]
    else:
        off = -offset
        grad[off:] += (1.0 - h[: _n - off])
        grad[: _n - off] -= h[off:]

    grad *= _dx
    return grad


def constraint_sum(h):
    """Equality constraint: total integral = 1."""
    return np.sum(h) - _target


def jac_eq_global(h):
    """Jacobian of the equality constraint."""
    return np.ones_like(h)


def project_onto_bounds_sum(v, target):
    """
    Project vector v onto the box [0,1]^n while enforcing sum(v)=target.
    Finds lambda such that sum(clip(v‑lambda,0,1)) = target.
    """
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return proj


def hill_climb(h, deadline):
    """
    Simple pairwise adjustment search that respects the sum constraint.
    """
    h_cur = h.copy()
    c_cur = evaluate_global(h_cur)
    n = _n
    while time.time() < deadline:
        i = np.random.randint(n)
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # feasible move bounds
        pos = min(h_cur[i], 1.0 - h_cur[j])
        neg = min(h_cur[j], 1.0 - h_cur[i])
        if pos + neg <= 0.0:
            continue
        eps = (np.random.rand() * (pos + neg) - neg) * 0.5  # shrink step a bit
        if abs(eps) < 1e-12:
            continue
        h_new = h_cur.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # no need to re‑project – bounds already respected
        c_new = evaluate_global(h_new)
        if c_new < c_cur:
            h_cur, c_cur = h_new, c_new
    return h_cur, c_cur


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to lower the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    global _n, _dx, _target
    np.random.seed(seed)
    start = time.time()

    # initial vector (if provided) otherwise uniform
    try:
        init_h = np.array(initial_h_values, dtype=float)  # supplied by the harness
        n = init_h.size
    except NameError:
        n = 45
        init_h = np.full(n, 0.5)

    _n, _dx, _target = n, 2.0 / n, n / 2.0
    bounds = [(0.0, 1.0)] * n
    eq_cons = {'type': 'eq', 'fun': constraint_sum, 'jac': jac_eq_global}

    best_h = init_h.copy()
    best_c = evaluate_global(best_h)

    # try a few SLSQP runs (using the analytical gradient)
    max_iters_per_opt = 200
    attempt = 0
    try:
        import scipy.optimize as opt
        have_opt = True
    except Exception:               # pragma: no cover
        have_opt = False

    while time.time() - start < budget_s * 0.85:
        attempt += 1
        if attempt == 1:
            x0 = init_h.copy()
        else:
            rnd = np.random.rand(n)
            x0 = project_onto_bounds_sum(rnd, _target)

        if have_opt:
            res = opt.minimize(objective_global, x0, method='SLSQP',
                               jac=gradient_global, bounds=bounds,
                               constraints=[eq_cons],
                               options={'maxiter': max_iters_per_opt,
                                        'ftol': 1e-12,
                                        'disp': False})
            cand_h = res.x if res.success else x0
        else:
            cand_h = x0

        cand_c = evaluate_global(cand_h)
        if cand_c < best_c:
            best_c, best_h = cand_c, cand_h.copy()
            if best_c < 0.3808:
                break

    # final refinement with cheap hill‑climbing
    deadline = start + budget_s * 0.95
    best_h, best_c = hill_climb(best_h, deadline)

    # final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_bounds_sum(best_h, _target)
    best_c = evaluate_global(best_h)

    return best_h, float(best_c), n
