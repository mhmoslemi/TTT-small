    target = len(h) / 2.0

        # Simple: add diff/len to all entries then reclip, then adjust again.
        diff = target - s
        h += diff / len(h)

        h = np.clip(h, 0.0, 1.0)

import numpy as np
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős overlap constant for a given discretized h."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1 - h, mode="full") * (2.0 / n))


def project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the box [0,1] with the additional
    linear equality sum(v) = target.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def make_piecewise_h(a: float, b: float, n: int) -> np.ndarray:
    """Create a symmetric 5‑block step function with block values a,b,c,b,a."""
    m = n // 5
    c = 2.5 - 2 * a - 2 * b
    # ensure numerical bounds
    c = max(0.0, min(1.0, c))
    blocks = [np.full(m, a), np.full(m, b), np.full(m, c), np.full(m, b), np.full(m, a)]
    return np.concatenate(blocks)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- parameters ---------------------------------------------------------
    n = 250                     # total sample points (multiple of 5)
    m = n // 5                  # block length
    step_grid = 0.02            # grid resolution for initial search
    max_time = budget_s * 0.95  # keep a safety margin
    init_temp = 0.0015          # simulated‑annealing temperature start
    # -------------------------------------------------------------------------

    # ----- coarse grid search over 5‑block shapes ----------------------------
    best_h = None
    best_c = math.inf
    a_vals = np.arange(0.0, 1.0 + 1e-12, step_grid)
    for a in a_vals:
        # feasible b range from the constraints 0≤c≤1
        b_min = max(0.0, 0.75 - a)
        b_max = min(1.0, 1.25 - a)
        if b_min > b_max:
            continue
        b_vals = np.arange(b_min, b_max + 1e-12, step_grid)
        for b in b_vals:
            h = make_piecewise_h(a, b, n)
            # enforce exact integral (tiny numeric drift)
            h = project_sum_box(h, n / 2.0)
            c5 = compute_c5(h)
            if c5 < best_c:
                best_c = c5
                best_h = h.copy()
    # -------------------------------------------------------------------------

    # If the grid already beats the target, return immediately
    if best_c < 0.3808:
        return best_h.tolist(), float(best_c), n

    # ----------- local search (simulated annealing) -------------------------
    cur_h = best_h.copy()
    cur_c = best_c
    last_improve = start_time
    while time.time() - start_time < max_time:
        # correlation and worst lag
        corr_full = np.correlate(cur_h, 1 - cur_h, mode="full")
        idx_max = int(np.argmax(corr_full))
        lag = idx_max - (n - 1)          # signed lag
        # gradient contribution for the worst lag
        d = abs(lag)
        g = np.zeros(n)
        if d > 0:
            g[: n - d] += cur_h[d:]
            g[d:] += cur_h[: n - d]
        else:
            g = 2.0 * cur_h

        # choose donor (small g, positive mass) and recipient (large g, capacity)
        donors = np.where(cur_h > 1e-12)[0]
        recips = np.where(cur_h < 1.0 - 1e-12)[0]
        if donors.size == 0 or recips.size == 0:
            break
        donor = donors[int(np.argmin(g[donors]))]
        recip = recips[int(np.argmax(g[recips]))]

        max_transfer = min(cur_h[donor], 1.0 - cur_h[recip])
        if max_transfer < 1e-12:
            # fallback to a random pair
            i, j = rng.choice(n, size=2, replace=False)
            max_transfer = min(cur_h[i], 1.0 - cur_h[j])
            if max_transfer < 1e-12:
                continue
            donor, recip = i, j

        eps = max_transfer * rng.uniform(0.1, 0.5)
        new_h = cur_h.copy()
        new_h[donor] -= eps
        new_h[recip] += eps

        new_c = compute_c5(new_h)

        # temperature based acceptance
        elapsed = time.time() - start_time
        temp = init_temp * (1.0 - elapsed / max_time)
        accept = False
        if new_c < cur_c:
            accept = True
        elif temp > 0:
            prob = math.exp(-(new_c - cur_c) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = cur_h.copy()
                last_improve = time.time()

        # occasional random perturbation to escape plateaus
        if rng.random() < 0.02:
            i, j = rng.choice(n, size=2, replace=False)
            mx = min(cur_h[i], 1.0 - cur_h[j])
            if mx > 1e-12:
                eps = mx * rng.uniform(0.1, 0.5)
                h2 = cur_h.copy()
                h2[i] -= eps
                h2[j] += eps
                c2 = compute_c5(h2)
                if c2 < cur_c or (temp > 0 and rng.random() < math.exp(-(c2 - cur_c) / temp)):
                    cur_h = h2
                    cur_c = c2
                    if c2 < best_c:
                        best_c = c2
                        best_h = cur_h.copy()
                        last_improve = time.time()

        # if no improvement for a while, inject a small random shake
        if time.time() - last_improve > 30.0:
            noise = rng.uniform(-0.02, 0.02, size=n)
            cur_h = np.clip(cur_h + noise, 0.0, 1.0)
            cur_h = project_sum_box(cur_h, n / 2.0)
            cur_c = compute_c5(cur_h)
            last_improve = time.time()

    # final projection to satisfy constraints exactly
    best_h = project_sum_box(best_h, n / 2.0)
    final_c = compute_c5(best_h)

    return best_h.tolist(), float(final_c), n
