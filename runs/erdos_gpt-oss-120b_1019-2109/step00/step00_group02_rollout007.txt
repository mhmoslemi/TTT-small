import time, math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Integral approximation of max_k ∫ h(x)(1-h(x+k)) dx."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def _max_shift_info(h: np.ndarray):
    """Return (max_value, shift_index, full_correlation_vector)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    max_val = corr[idx] * (2.0 / n)
    shift = idx - (n - 1)
    return max_val, shift, corr


def _targeted_step(h: np.ndarray, shift: int, max_delta: float, rng: np.random.Generator):
    """Move mass from a large‑product pair (i,i+shift) to reduce the max term."""
    n = h.size
    if shift > 0:
        hi = h[: n - shift]
        hj = h[shift:]
    elif shift < 0:
        hi = h[-shift :]
        hj = h[: n + shift]
    else:  # shift == 0, treat as moving from max hi*(1-hi) to a min one
        term = h * (1.0 - h)
        i = int(np.argmax(term))
        # find a position with smallest term to receive mass
        j = int(np.argmin(term))
        delta = min(max_delta, h[i], 1.0 - h[j])
        if delta <= 0:
            return None, None
        nh = h.copy()
        nh[i] -= delta
        nh[j] += delta
        return nh, delta

    # products for this shift
    prod = hi * (1.0 - hj)
    i_local = int(np.argmax(prod))
    # map back to original indices
    if shift > 0:
        i = i_local
        j = i + shift
    else:
        i = i_local - shift
        j = i + shift  # = i_local
    delta = min(max_delta, h[i], 1.0 - h[j])
    if delta <= 0:
        return None, None
    nh = h.copy()
    nh[i] -= delta
    nh[j] += delta
    return nh, delta


def _random_transfer(h: np.ndarray, max_delta: float, rng: np.random.Generator):
    """Transfer a random amount between two distinct positions."""
    n = h.size
    i, j = rng.choice(n, size=2, replace=False)
    delta = min(max_delta, h[i], 1.0 - h[j])
    if delta <= 0:
        return None, None
    delta *= rng.random()
    if delta <= 0:
        return None, None
    nh = h.copy()
    nh[i] -= delta
    nh[j] += delta
    return nh, delta
