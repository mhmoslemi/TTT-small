def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random, math

    # ---------------------------------------------------------
    # Helpers
    # ---------------------------------------------------------
    def compute_c5(h, dx):
        """Maximum overlap integral for a given step function."""
        # full convolution of h with (1-h)
        return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    def gradient_wrt_shift(h, k, dx):
        """Gradient of the overlap for shift k (the worst shift)."""
        n = h.shape[0]
        g = np.zeros_like(h)
        if k >= 0:
            # contribution from h[i] term
            i1 = np.arange(0, n - k)
            g[i1] += (1.0 - h[i1 + k])
            # contribution from h[i+k] term
            i2 = np.arange(k, n)
            g[i2] -= h[i2 - k]
        else:
            kk = -k
            i1 = np.arange(kk, n)
            g[i1] += (1.0 - h[i1 - kk])
            i2 = np.arange(0, n - kk)
            g[i2] -= h[i2 + kk]
        return g * dx

    # ---------------------------------------------------------
    # Initialise problem data
    # ---------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    # use provided initial solution if present
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = h0.shape[0]
    except NameError:
        # fall back to a simple binary construction (half 1's)
        n = 70
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

    target_sum = n / 2.0
    # Repair any tiny violation of the integral constraint
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # simple proportional scaling followed by a single correction
        h0 = np.clip(h0, 0.0, 1.0)
        cur_sum = h0.sum()
        if cur_sum > 0:
            factor = target_sum / cur_sum
            h0 = np.clip(h0 * factor, 0.0, 1.0)
        # final tiny fix
        diff = h0.sum() - target_sum
        if abs(diff) > 1e-12:
            idx = np.argmin(h0) if diff > 0 else np.argmax(h0)
            h0[idx] = min(max(h0[idx] - diff, 0.0), 1.0)

    dx = 2.0 / n

    # ---------------------------------------------------------
    # Main optimisation loop (simulated‑annealing + guided moves)
    # ---------------------------------------------------------
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, dx)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # configuration
    random_move_prob = 0.15          # probability to do a blind swap
    init_temp = 0.02                # initial temperature for SA
    final_temp = 0.001              # final temperature
    max_no_improve = 20000          # iterations before a restart
    last_improve = 0
    it = 0
    total_budget = budget_s * 0.98   # a small safety margin

    while True:
        elapsed = time.time() - start_time
        if elapsed >= total_budget:
            break
        # linear temperature schedule
        frac = elapsed / total_budget
        temp = init_temp * (1.0 - frac) + final_temp * frac

        # ------------------- propose a new candidate -------------------
        if np.random.rand() < random_move_prob:
            # blind exchange of mass between two coordinates
            i, j = np.random.choice(n, 2, replace=False)
            if np.random.rand() < 0.5:
                cap = min(1.0 - cur_h[i], cur_h[j])
                if cap <= 0.0:
                    continue
                delta = np.random.rand() * cap
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
            else:
                cap = min(cur_h[i], 1.0 - cur_h[j])
                if cap <= 0.0:
                    continue
                delta = np.random.rand() * cap
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta
        else:
            # guided move based on the current worst shift
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            idx_max = np.argmax(corr)
            shift_k = idx_max - (n - 1)          # shift value corresponding to max overlap
            grad = gradient_wrt_shift(cur_h, shift_k, dx)

            # indices that can decrease / increase while improving the worst shift
            dec_mask = (grad > 1e-12) & (cur_h > 1e-12)
            inc_mask = (grad < -1e-12) & (cur_h < 1.0 - 1e-12)

            if not dec_mask.any() or not inc_mask.any():
                # fallback to a blind move if gradient offers nothing
                continue

            dec_idx = np.argmax(grad * dec_mask)          # largest positive gradient
            inc_idx = np.argmin(grad * inc_mask)          # most negative gradient

            # maximum admissible transfer
            max_dec = cur_h[dec_idx]
            max_inc = 1.0 - cur_h[inc_idx]
            delta = min(max_dec, max_inc)

            # start with larger steps and shrink as time passes
            delta *= 0.5 * (1.0 - frac) + 0.1 * frac
            if delta <= 0.0:
                continue
            new_h = cur_h.copy()
            new_h[dec_idx] -= delta
            new_h[inc_idx] += delta

        # ------------------- evaluate -------------------
        new_c5 = compute_c5(new_h, dx)

        # ------------------- accept / reject -------------------
        improved = new_c5 < cur_c5 - 1e-12
        if improved or (temp > 0.0 and np.random.rand() < math.exp(-(new_c5 - cur_c5) / temp)):
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_h = new_h.copy()
                best_c5 = new_c5
                last_improve = it

        # ------------------- restart if stuck -------------------
        if it - last_improve > max_no_improve:
            if np.random.rand() < 0.5:
                cur_h = best_h.copy()
                cur_c5 = best_c5
            else:
                cur_h = np.zeros(n, dtype=float)
                cur_h[: n // 2] = 1.0
                np.random.shuffle(cur_h)
                cur_c5 = compute_c5(cur_h, dx)
                if cur_c5 < best_c5:
                    best_h = cur_h.copy()
                    best_c5 = cur_c5
            last_improve = it

        it += 1

    # ---------------------------------------------------------
    # Final tidy‑up (clip, enforce exact integral)
    # ---------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    total = best_h.sum()
    diff = total - target_sum
    if abs(diff) > 1e-9:
        # adjust a random index while staying inside [0,1]
        idx = np.random.randint(n)
        adj = best_h[idx] - diff
        adj = min(max(adj, 0.0), 1.0)
        best_h[idx] = adj

    final_c5 = compute_c5(best_h, dx)

    # Return a plain list for the harness, together with the bound and n
    return (best_h.tolist(), float(final_c5), int(n))
