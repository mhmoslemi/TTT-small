def project_to_feasible(v, target_sum):
    lo = -1.0
    hi = 2.0
    # ensure lo bound sum>target, hi bound sum<target
    # Expand bounds until condition satisfied.
    while np.sum(np.clip(v - lo, 0, 1)) < target_sum:
        lo *= 2
    while np.sum(np.clip(v - hi, 0, 1)) > target_sum:
        hi /= 2
    for _ in range(50):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - ((lo+hi)/2), 0, 1)

def project_to_sum_box(v, target_sum):
    # via lambda method
    lo = -1.0
    hi = 2.0
    # Expand bounds until f(lo) >= target_sum and f(hi) <= target_sum
    # but f(lo) = sum_clipped(v - lo) where lo very negative => all clipped to 1 => sum = n
    # f(hi) = sum_clipped(v - hi) where hi large => all clipped to 0 => sum = 0 
    # So any lo, hi with lo < hi works; we can simply set lo=-1e5, hi=1e5.
    lo = -1e5
    hi = 1e5
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - ((lo+hi)/2), 0, 1)

import numpy as np
import time, math, random

def _project_to_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x | 0<=x_i<=1, sum x_i = target}."""
    lo, hi = -1e5, 1e5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(vec: np.ndarray) -> float:
    """Compute the Erdős overlap value for a discretised step function."""
    n = vec.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(vec, 1.0 - vec, mode="full")) * dx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C₅ bound, and the number of samples."""
    np.random.seed(seed)
    start = time.time()

    # ----- initialise ---------------------------------------------------------
    try:
        h = initial_h_values.astype(float, copy=True)  # type: ignore
        n = h.size
    except NameError:
        n = 81
        v = np.random.rand(n)
        h = _project_to_sum_box(v, n / 2.0)

    target_sum = n / 2.0
    cur_val = _c5_value(h)
    best_h = h.copy()
    best_val = cur_val

    # ----- optimisation parameters -------------------------------------------
    step = 0.5                # initial step size for sub‑gradient updates
    step_decay = 0.9995       # geometric decay
    T0, T_end = 0.02, 1e-6    # simulated‑annealing temperature schedule
    no_improve = 0            # counter for stagnation

    idx = np.arange(n)

    # ----- main loop ---------------------------------------------------------
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break

        # correlation and the shift attaining the maximum
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)            # signed offset

        # gradient of the active term (sub‑gradient of the max)
        plus = idx + shift
        minus = idx - shift
        term1 = np.where((plus >= 0) & (plus < n), 1.0 - h[plus], 0.0)
        term2 = np.where((minus >= 0) & (minus < n), h[minus], 0.0)
        grad = term1 - term2                 # ∂C₅/∂h

        # sub‑gradient descent step (move opposite to the gradient)
        cand = h - step * grad
        cand = _project_to_sum_box(cand, target_sum)

        cand_val = _c5_value(cand)

        # temperature for occasional uphill moves
        t_frac = elapsed / (budget_s * 0.95)
        T = T0 * (1.0 - t_frac) + T_end * t_frac

        if cand_val < cur_val or random.random() < math.exp(-(cand_val - cur_val) / (T + 1e-12)):
            h = cand
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_h = cand.copy()
                best_val = cand_val
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        step *= step_decay

        # occasional random perturbation if stuck
        if no_improve > 3000:
            noise = np.random.normal(scale=0.04, size=n)
            h = _project_to_sum_box(best_h + noise, target_sum)
            cur_val = _c5_value(h)
            no_improve = 0

    # final clean‑up (ensure exact feasibility)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum_box(best_h, target_sum)
    final_val = _c5_value(best_h)

    return best_h, float(final_val), n
