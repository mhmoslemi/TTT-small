import time
import math
import random
import numpy as np

# ------------------------------------------------------------
# Helper: evaluate the C5 value of a step function h
# ------------------------------------------------------------
def _evaluate_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx approximated by discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # convolution (full) gives correlation for all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


# ------------------------------------------------------------
# Helper: projection onto the box [0,1]^n with sum constraint ∑h = target
# ------------------------------------------------------------
def _project_to_feasible(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto {0≤h_i≤1, Σh_i = target} by bisection on a common shift."""
    lo, hi = -1.0, 1.0          # bounds for the Lagrange multiplier λ
    for _ in range(60):        # enough for double precision
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:         # need larger λ to reduce the sum
            lo = lam
        else:                  # need smaller λ to increase the sum
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)

    # tiny correction if numerical error prevents exact equality
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        idx = np.where(free)[0]
        if idx.size:
            inc = diff / idx.size
            x[idx] = np.clip(x[idx] + inc, 0.0, 1.0)
        else:
            # fall back: adjust the first element that is not at a bound
            i = np.argmin(np.abs(v - lam))
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x


# ------------------------------------------------------------
# Helper: gradient of the active (worst) overlap term
# ------------------------------------------------------------
def _gradient_of_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Compute ∇ O_shift where O_shift = Σ h_i (1 - h_{i+shift}) * dx."""
    n = h.shape[0]
    dx = 2.0 / n
    arr1 = np.ones(n)                     # 1 - h_{i+shift} (or 1 if out of bounds)
    arr2 = np.zeros(n)                    # h_{i-shift}  (or 0 if out of bounds)

    if shift >= 0:
        # indices where i+shift is inside
        arr1[: n - shift] = 1.0 - h[shift:]
        # indices where i-shift is inside
        arr2[shift:] = h[: n - shift]
    else:
        s = -shift
        arr1[s:] = 1.0 - h[: n - s]
        arr2[: n - s] = h[s:]

    return dx * (arr1 - arr2)


# ------------------------------------------------------------
# Main entry point required by the evaluation harness
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimizes
    C5 = max_k ∫ h(x) (1-h(x+k)) dx.
    Returns (list_of_h, C5_value, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise problem size and a feasible starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    else:
        n = 95                     # fallback size
        h = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0           # because ∫ h = 1  ⇒  Σ h = n/2

    # ensure feasibility (might be already, but safe)
    h = _project_to_feasible(h, target_sum)

    best_h = h.copy()
    best_c5 = _evaluate_c5(best_h)

    # ------------------------------------------------------------------
    # Parameters for the stochastic optimisation
    # ------------------------------------------------------------------
    start_time = time.time()
    end_time = start_time + budget_s * 0.97   # small safety margin
    cur_h = h.copy()
    cur_c5 = best_c5

    # step size for gradient moves (scaled later)
    eta = 0.3
    # base temperature for accepting worse solutions
    T0 = 0.03
    # counters for adaptive restarts
    it_without_improve = 0
    max_no_improve = 6000      # restart after this many iterations

    while time.time() < end_time:
        # --------------------------------------------------------------
        # Determine current worst shift and its gradient
        # --------------------------------------------------------------
        n = cur_h.shape[0]
        dx = 2.0 / n
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full") * dx
        idx_max = int(np.argmax(corr))
        cur_c5 = float(corr[idx_max])
        shift = idx_max - (n - 1)

        # --------------------------------------------------------------
        # Choose a candidate move: gradient step (60%) or random mass transfer (40%)
        # --------------------------------------------------------------
        if random.random() < 0.6:          # gradient descent move
            grad = _gradient_of_shift(cur_h, shift)
            cand = _project_to_feasible(cur_h - eta * grad, target_sum)
        else:                               # random redistribution move
            # find indices that can send / receive mass
            give_idx = np.where(cur_h > 1e-8)[0]
            take_idx = np.where(cur_h < 1.0 - 1e-8)[0]
            if give_idx.size == 0 or take_idx.size == 0:
                continue
            i = int(np.random.choice(give_idx))
            j = int(np.random.choice(take_idx))
            max_delta = min(cur_h[i], 1.0 - cur_h[j])
            # wander delta in a diminishing range
            delta = max_delta * random.random() * 0.5
            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta
            # due to floating noise, re‑project very rarely
            if random.random() < 0.01:
                cand = _project_to_feasible(cand, target_sum)

        cand_c5 = _evaluate_c5(cand)

        # --------------------------------------------------------------
        # Acceptance rule (simulated annealing style)
        # --------------------------------------------------------------
        elapsed = time.time() - start_time
        frac = min(1.0, elapsed / budget_s)
        T = T0 * (1.0 - frac) + 1e-8   # temperature decays to nearly zero

        accept = False
        if cand_c5 < cur_c5 - 1e-12:
            accept = True
        else:
            # probabilistic acceptance of worse moves
            prob = math.exp(-(cand_c5 - cur_c5) / T)
            if random.random() < prob:
                accept = True

        if accept:
            cur_h = cand
            cur_c5 = cand_c5
            it_without_improve = 0
            if cand_c5 < best_c5 - 1e-12:
                best_h = cand.copy()
                best_c5 = cand_c5
                # tighten step size when we improve
                eta = min(eta * 1.1, 1.0)
        else:
            it_without_improve += 1
            # slightly shrink step size on rejection of gradient moves
            eta = max(eta * 0.9, 1e-4)

        # --------------------------------------------------------------
        # Restart if stuck for a long time
        # --------------------------------------------------------------
        if it_without_improve >= max_no_improve:
            # fresh random feasible point
            rnd = np.random.rand(n)
            cur_h = _project_to_feasible(rnd, target_sum)
            cur_c5 = _evaluate_c5(cur_h)
            it_without_improve = 0
            eta = 0.3

        # early exit if we already beat the target bound
        if best_c5 <= 0.3808:
            break

    # --------------------------------------------------------------
    # Prepare output
    # --------------------------------------------------------------
    # final verification with the provided evaluator (if available)
    try:
        c5_bound = float(evaluate_erdos_solution(best_h.tolist()))
    except Exception:
        c5_bound = _evaluate_c5(best_h)

    return (best_h.tolist(), c5_bound, n)
