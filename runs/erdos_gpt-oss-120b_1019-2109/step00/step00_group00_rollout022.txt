    low = -np.max(v) - 1.0
    high = 1.0 - np.min(v) + 1.0
    for _ in range(60):

        w = np.clip(v + mid, 0, 1)
        s = w.sum()
        if s < target:
            low = mid
        else:
            high = mid

    # Find λ such that sum(clip(v+λ,0,1)) = target.
    low = -np.max(v) - 1.0
    high = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (low + high) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            low = mid
        else:
            high = mid
    return np.clip(v + (low + high) * 0.5, 0.0, 1.0)


def _c5_of(h):
    """Compute the C5 value for a step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _gradient_of_max_lag(h):
    """Return gradient of the current worst lag, the lag index and the raw correlation."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)          # integer shift giving the max correlation

    if lag >= 0:
        forward = np.empty(n)
        forward[: n - lag] = h[lag:]
        forward[n - lag :] = 0.0
        backward = np.empty(n)
        backward[lag:] = h[: n - lag]
        backward[: lag] = 0.0
    else:
        l = -lag
        forward = np.empty(n)
        forward[l:] = h[: n - l]
        forward[: l] = 0.0
        backward = np.empty(n)
        backward[: n - l] = h[l:]
        backward[n - l :] = 0.0

    grad = 1.0 - forward - backward
    return grad, lag, float(corr[idx] * (2.0 / n))


def run(seed=42, budget_s=1100, **_):
    """Return a step function improving the known record for C5."""
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise a feasible h (use the supplied one if available)
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.array(init, dtype=np.float64)
        n = h.size
    else:
        # a simple triangular shape when no parent is given
        n = 200
        dx = 2.0 / n
        x = np.arange(n) * dx + dx * 0.5
        h = np.where(x <= 1.0, x, 2.0 - x)
    target_sum = n / 2.0                     # ∑h must equal n/2
    # enforce the constraints once (projection)
    h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_of(best_h)

    start = time.time()
    # parameters for the simple projected‑gradient search
    step_start = 0.5
    step_factor = 0.9999                     # slowly shrink step size

    # ------------------------------------------------------------------
    # main optimisation loop (time‑bounded)
    # ------------------------------------------------------------------
    while time.time() - start < budget_s - 0.3:
        grad, lag, cur_c5 = _gradient_of_max_lag(h)

        # line‑search on a fixed geometric schedule
        step = step_start
        improved = False
        while step > 1e-6:
            v = h - step * grad
            proj = _project_box_sum(v, target_sum)
            new_c5 = _c5_of(proj)
            if new_c5 < best_c5 - 1e-12:
                h = proj
                best_h = proj.copy()
                best_c5 = new_c5
                improved = True
                break
            step *= 0.5

        if not improved:
            # occasional random two‑index exchange (keeps the sum exact)
            i, j = np.random.randint(n, size=2)
            if i != j:
                delta = (np.random.rand() - 0.5) * 0.2    # modest move
                hi, hj = h[i], h[j]
                max_delta = min(hi, 1.0 - hj)
                min_delta = -min(hj, 1.0 - hi)
                delta = np.clip(delta, min_delta, max_delta)
                if abs(delta) > 1e-12:
                    h_new = h.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
                    new_c5 = _c5_of(h_new)
                    if new_c5 < best_c5:
                        h = h_new
                        best_h = h_new.copy()
                        best_c5 = new_c5

        # gradually reduce the step size to help convergence
        step_start *= step_factor

        # early exit if we already beat the target bound
        if best_c5 <= 0.3808:
            break

    # final safety projection (ensures feasibility)
    best_h = _project_box_sum(best_h, target_sum)
    final_c5 = _c5_of(best_h)

    return best_h.tolist(), final_c5, n
