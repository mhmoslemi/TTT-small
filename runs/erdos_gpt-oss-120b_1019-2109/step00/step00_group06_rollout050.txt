```
def project_to_feasible(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust sum
    s = h.sum()
    if np.abs(s - target_sum) < 1e-12:
        return h
    # Compute diff
    diff = target_sum - s
    # Determine improvable indices: where (diff > 0) need increase (h < 1), else need decrease (h > 0)
    if diff > 0:
        free = 1.0 - h
        total_free = free.sum()
        if total_free > 0:
            inc = diff * free / total_free  # proportionally distribute
            h = h + inc
        else:
            # cannot increase, fallback: do nothing
            pass
    else:
        free = h
        total_free = free.sum()
        if total_free > 0:
            dec = (-diff) * free / total_free
            h = h - dec
    # Clip again
    h = np.clip(h, 0.0, 1.0)
    # Might still not sum exactly due to floating rounding; adjust small residual by adding to first index not at bound
    s = h.sum()
    if np.abs(s - target_sum) > 1e-10:
        # fix residual by adjusting one free variable
        res = target_sum - s
        # Find idx where 0 < h < 1
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size > 0:
            h[idx[0]] += res
            h = np.clip(h, 0.0, 1.0)
        else:
            # otherwise adjust a single element within bounds (maybe at 0 or 1)
            # can force saturate if needed
            h[0] = np.clip(h[0] + res, 0.0, 1.0)
    return h
```

import time
import numpy as np

def shift(arr, k):
    """Shift arr by k positions with zero padding (no wrap)."""
    n = arr.shape[0]
    out = np.zeros_like(arr)
    if k >= 0:
        out[: n - k] = arr[k:]
    else:
        out[-k :] = arr[: n + k]
    return out


def project_to_feasible(x, target):
    """Clip to [0,1] and adjust total sum to the target."""
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    if abs(s - target) < 1e-12:
        return x
    diff = target - s
    if diff > 0:  # need to increase some entries
        cap = 1.0 - x
        tot = cap.sum()
        if tot > 0:
            inc = diff * cap / tot
            x = x + inc
        else:
            x = x + diff / len(x)
    else:          # need to decrease some entries
        cap = x
        tot = cap.sum()
        if tot > 0:
            dec = (-diff) * cap / tot
            x = x - dec
        else:
            x = x + diff / len(x)
    x = np.clip(x, 0.0, 1.0)

    # small residual fix
    s = x.sum()
    residual = target - s
    if abs(residual) > 1e-12:
        free = np.where((x > 0) & (x < 1))[0]
        if free.size:
            idx = free[0]
            x[idx] = np.clip(x[idx] + residual, 0.0, 1.0)
        else:
            idx = 0
            x[idx] = np.clip(x[idx] + residual, 0.0, 1.0)
    return x


def evaluate_c5(h):
    """Exact C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def gradient_step(h, lag, lr, target):

    plus = shift(h, lag)
    minus = shift(h, -lag)
    d = plus + minus - 1.0               # direction = -∇O_k

        if ones.size == 0 or zeros.size == 0:
            break
        i1 = np.random.choice(ones)
        i0 = np.random.choice(zeros)
        cur_h[i1], cur_h[i0] = cur_h[i0], cur_h[i1]   # swap
        new_val = evaluate_c5(cur_h)
        if new_val < cur_val:
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i1], cur_h[i0] = cur_h[i0], cur_h[i1]

    # final projection to guarantee feasibility
    best_h = project_to_feasible(best_h, best_h.size / 2.0)
    final_c5 = evaluate_c5(best_h)

    return best_h.tolist(), float(final_c5), int(best_h.size)
