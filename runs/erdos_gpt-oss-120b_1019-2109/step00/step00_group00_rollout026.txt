def compute_c5(arr):

def anneal(arr, initial_val, time_limit):
   best_arr = arr.copy()
   best_val = initial_val
   curr_arr = arr.copy()
   curr_val = initial_val
   T0 = 1.0
   T_end = 1e-4
   it = 0
   while True:
       # check time
       if time.time() - start > time_limit:
           break
       it += 1
       # linear schedule
       T = T0 * (1 - it / max_iters) if it < max_iters else T_end
       # propose a swap
       # pick random index of 1 and 0
       ones = np.flatnonzero(curr_arr)
       zeros = np.flatnonzero(1 - curr_arr)
       if len(ones) == 0 or len(zeros) == 0:
            break
       i = random.choice(ones)
       j = random.choice(zeros)
       # swap
       curr_arr[i], curr_arr[j] = 0.0, 1.0
       new_val = compute_c5(curr_arr)
       delta = new_val - curr_val
       accept = False
       if delta <= 0:
           accept = True
       else:
           prob = math.exp(-delta / max(T,1e-12))
           if random.random() < prob:
               accept = True
       if accept:
           curr_val = new_val
           if new_val < best_val - 1e-12:
               best_val = new_val
               best_arr = curr_arr.copy()
       else:
           # revert swap
           curr_arr[i], curr_arr[j] = 1.0, 0.0
   return best_arr, best_val

import numpy as np, random, time, math

def compute_c5(arr, dx):
    return np.max(np.correlate(arr, 1 - arr, mode='full')) * dx

def project_to_feasible(y, target):
    x = np.clip(y, 0.0, 1.0)
    # compute need to sum to target
    diff = target - np.sum(x)
    # tolerance small
    max_iter = 100
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        # identify indices that can be moved
        if diff > 0:
            # can increase values not at upper bound
            free = np.where(x < 1 - 1e-12)[0]
            if free.size == 0:
                break
            incr = diff / free.size
            # apply but keep within bounds
            incr = np.minimum(incr, 1 - x[free])
            x[free] += incr
        else:
            free = np.where(x > 1e-12)[0]
            if free.size == 0:
                break
            decr = diff / free.size  # diff negative
            # ensure we don't go below 0
            decr = np.maximum(decr, -x[free])
            x[free] += decr
        diff = target - np.sum(x)
    return x

def gradient_active_shift(h, k):
    n = len(h)
    g = np.ones_like(h)
    if k >= 0:
        # subtract h[i+k]
        g[:n - k] -= h[k:]
        # subtract h[i - k]
        g[k:] -= h[:n - k]
    else:
        kk = -k
        # subtract h[i - kk] (i>=kk)
        g[kk:] -= h[:n - kk]
        # subtract h[i + kk] (i < n-kk)
        g[:n - kk] -= h[kk:]
    return g

import math
import random
import time
import numpy as np

def _c5_value(h, dx):
    """Maximum overlap integral for a given step function."""
    # correlation h • (1‑h) over all integer shifts
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _project_simplex_box(v, target):
    """
    Project vector v onto the set {0 ≤ x_i ≤ 1, Σx_i = target}.
    Simple water‑filling algorithm.
    """
    x = np.clip(v, 0.0, 1.0)
    diff = target - x.sum()
    # Iterate a few times – convergence is fast because diff is tiny in our use‑case.
    for _ in range(100):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(x < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            inc = np.minimum(inc, 1.0 - x[free])
            x[free] += inc
        else:
            free = np.where(x > 1e-12)[0]
            if free.size == 0:
                break
            dec = diff / free.size               # diff < 0
            dec = np.maximum(dec, -x[free])
            x[free] += dec
        diff = target - x.sum()
    # final clipping for safety
    return np.clip(x, 0.0, 1.0)


def _gradient_active_shift(h, k):
    """
    Sub‑gradient of the active cost C(k) = Σ h_i (1‑h_{i+k})  (without the dx factor).
    Handles out‑of‑range indices by treating missing h as 0.
    """
    n = h.size
    g = np.ones_like(h)
    if k >= 0:
        # subtract h[i+k] where it exists
        g[: n - k] -= h[k:]
        # subtract h[i‑k] where it exists
        g[k:] -= h[: n - k]
    else:
        kk = -k
        # subtract h[i‑kk] where it exists
        g[kk:] -= h[: n - kk]
        # subtract h[i+kk] where it exists
        g[: n - kk] -= h[kk:]
    return g

    """Return an improved step function for the Erdős C₅ problem.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values – numpy array with values in [0,1] and Σh·dx = 1,
        c5_bound – the achieved upper bound,
        n_points – length of the discretisation (|h|).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise h – use the provided construction if present.
    # ------------------------------------------------------------------

    except NameError:            # fallback: simple balanced binary pattern
        n = 80                     # even size gives exact half measure
        h = np.zeros(n, dtype=np.float64)
        h[: n // 2] = 1.0

    dx = 2.0 / n
    target_sum = n / 2.0                     # because ∫h = 1  ⇒ Σh = n/2
    # Ensure feasibility (might be off by a tiny rounding error)
    h = _project_simplex_box(h, target_sum)

    best_h = h.copy()
    best_val = _c5_value(best_h, dx)

    # ------------------------------------------------------------------
    # Sub‑gradient descent with projection.
    # ------------------------------------------------------------------
    max_iters = 25000
    start_time = time.time()
    time_limit = budget_s * 0.90                 # keep a safety margin

    cur_h = best_h.copy()
    cur_val = best_val

    # step‑size schedule (large → small)
    alpha_start = 0.6

    for it in range(1, max_iters + 1):

    return (best_h, float(best_val), n)
