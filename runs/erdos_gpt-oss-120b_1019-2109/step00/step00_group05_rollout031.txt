def repair(h):
    # Clip values to [0,1]
    h = np.clip(h, 0.0, 1.0)
    target = len(h) / 2.0
    s = h.sum()
    if abs(s - target) < 1e-9:
        return h
    # If sum too low, add mass to non-full entries
    # If sum too high, subtract from non-zero entries
    # We'll iterate distributing diff proportionally.
    # Use while loop
    diff = target - s
    # repeat at most n times
    for _ in range(len(h)):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add mass
            # compute capacity for each index to increase: 1-h_i
            caps = 1.0 - h
            total_cap = caps.sum()
            if total_cap == 0:
                break
            # add proportionally
            delta = diff * caps / total_cap
            # ensure not overshoot per index:
            delta = np.minimum(delta, caps)
            h += delta
            diff = target - h.sum()
        else: # diff < 0
            caps = h # amount can decrease
            total_cap = caps.sum()
            if total_cap == 0:
                break
            delta = diff * caps / total_cap
            # diff negative, caps positive: delta negative
            # limit each delta not exceed -caps (cannot go below zero)
            delta = np.maximum(delta, -caps)
            h += delta
            diff = target - h.sum()
    # final clip
    h = np.clip(h, 0.0, 1.0)
    # final adjust small rounding error by adding diff to first non-saturated index
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        # find index where we can add/subtract
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def gradient_for_shift(h, shift):

    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        # contributions from forward product
        # term1: (1 - h[i+s]) for i =0 .. n-1-s

        g[i] += (1 - h[i + s])

    new_h[inc_idx] += delta
    new_h[dec_idx] -= delta

import time, math, random
import numpy as np

def _compute_c5(h, dx):
    """C5 = max_k ∫ h(x) (1-h(x+k)) dx  (discrete version)"""
    # full correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _repair(h, target):
    """Project h onto [0,1] with exact sum = target."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    # iterative redistribution among free entries
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = np.where(h < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            delta = np.minimum(inc, 1.0 - h[free])
            h[free] += delta
        else:
            free = np.where(h > 1e-12)[0]
            if free.size == 0:
                break
            dec = diff / free.size          # negative
            delta = np.maximum(dec, -h[free])
            h[free] += delta
        diff = target - h.sum()
    # tiny residual fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for i in range(len(h)):
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(h, 0.0, 1.0)

def _gradient_for_shift(h, shift):
    """Sub‑gradient of the correlation term corresponding to given shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        s = shift
        i1 = np.arange(0, n - s)
        g[i1] += 1.0 - h[i1 + s]
        i2 = np.arange(s, n)
        g[i2] -= h[i2 - s]
    else:
        s = -shift
        i1 = np.arange(s, n)
        g[i1] += 1.0 - h[i1 - s]
        i2 = np.arange(0, n - s)
        g[i2] -= h[i2 + s]
    return g

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the C5 bound."""
    np.random.seed(seed)
    random.seed(seed)

    # ----- initialise -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)   # provided by harness
    except Exception:
        # fallback: a sinusoidal start (mean 0.5, respects bounds)
        n = kwargs.get("n_points", 70)
        dx = 2.0 / n
        xs = (np.arange(n) + 0.5) * dx
        h0 = np.sin(np.pi * xs / 2.0) ** 2          # peaks at 1, zeros at ends
    n = h0.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # make feasible
    cur_h = _repair(h0.copy(), target_sum)
    cur_c5 = _compute_c5(cur_h, dx)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ----- optimisation ------------------------------------------------
    start_time = time.time()
    time_limit = budget_s * 0.95

    max_iters = 2_000_000
    it = 0
    # simple linear temperature for stochastic acceptance & step scaling
    T_start = 0.02
    T_end = 1e-6

    while time.time() - start_time < time_limit and it < max_iters:
        it += 1
        # temperature (linear decay)
        T = T_start + (T_end - T_start) * (it / max_iters)

        # ----- gradient‑driven move -----
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        g = _gradient_for_shift(cur_h, shift)
        inc_idx = int(np.argmin(g))   # most negative   → increase h
        dec_idx = int(np.argmax(g))   # most positive   → decrease h

        cap_inc = 1.0 - cur_h[inc_idx]
        cap_dec = cur_h[dec_idx]
        max_delta = min(cap_inc, cap_dec)

        if max_delta > 0:
            # delta magnitude shrinks with temperature
            delta = max_delta * np.random.rand() * (T / T_start)
            new_h = cur_h.copy()
            new_h[inc_idx] += delta
            new_h[dec_idx] -= delta
            new_h = _repair(new_h, target_sum)
            new_c5 = _compute_c5(new_h, dx)

            if new_c5 < cur_c5 or (T > 0 and random.random() < math.exp(-(new_c5 - cur_c5) / T)):
                cur_h, cur_c5 = new_h, new_c5
                if new_c5 < best_c5:
                    best_h, best_c5 = new_h.copy(), new_c5

        # ----- occasional random exchange (to escape plateaus) -----
        if it % 300 == 0:
            i = np.random.randint(n)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1
            cap_inc = 1.0 - cur_h[i]
            cap_dec = cur_h[j]
            if cap_inc > 0 and cap_dec > 0:
                delta = min(cap_inc, cap_dec) * np.random.rand() * 0.1
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
                new_h = _repair(new_h, target_sum)
                new_c5 = _compute_c5(new_h, dx)
                if new_c5 < cur_c5:
                    cur_h, cur_c5 = new_h, new_c5
                    if new_c5 < best_c5:
                        best_h, best_c5 = new_h.copy(), new_c5

        # ----- occasional restart from the best with a small perturbation -----
        if it % 50000 == 0:
            cur_h = best_h + (np.random.rand(n) - 0.5) * 0.03
            cur_h = _repair(cur_h, target_sum)
            cur_c5 = _compute_c5(cur_h, dx)

    # final safety repair
    best_h = _repair(best_h, target_sum)
    best_c5 = _compute_c5(best_h, dx)

    return (best_h, float(best_c5), n)
