import time
import numpy as np

# --------------------------------------------------------------
# Helper functions
# --------------------------------------------------------------

def _project_to_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the feasible set:
        0 <= h_i <= 1   and   sum(h) = target_sum
    The projection is done by finding λ such that
        h_i = clip(v_i + λ, 0, 1)  and  sum(h) = target_sum.
    """
    lo, hi = -2.0, 2.0                     # bounds wide enough for λ
    for _ in range(60):                   # enough for double precision
        mid = (lo + hi) * 0.5
        h = np.clip(v + mid, 0.0, 1.0)
        s = h.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_and_idx(h: np.ndarray, dx: float):
    """
    Compute C5 = max_k ∫ h(x)(1-h(x+k))dx and the index of the maximizing shift.
    """
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return float(corr[idx] * dx), idx


# --------------------------------------------------------------
# Main optimisation routine
# --------------------------------------------------------------

def run(seed: int = 42, budget_s: float = 1100, **_):
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # Initialise a feasible h
    # -----------------------------------------------------------------
    try:
        # use parent construction if available
        h = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        # no parent – build a simple start (flat 0.5)
        n = 200
        h = np.full(n, 0.5, dtype=np.float64)

    n = h.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility
    h = np.clip(h, 0.0, 1.0)
    if abs(h.sum() - target_sum) > 1e-12:
        h = _project_to_constraints(h, target_sum)

    best_h = h.copy()
    best_c5, _ = _c5_and_idx(best_h, dx)

    start_time = time.time()
    time_limit = budget_s

    # -----------------------------------------------------------------
    # Phase 1 – deterministic gradient descent on the worst shift
    # -----------------------------------------------------------------
    # number of gradient steps is limited by remaining time
    while time.time() - start_time < 0.35 * time_limit:
        cur_c5, idx = _c5_and_idx(best_h, dx)
        shift = idx - (n - 1)                       # shift corresponding to max value

        # gradient of the active term D(k) = Σ h_i (1 - h_{i+shift})
        g = np.ones_like(best_h)                    # ∂/∂h_i of Σ h_i part
        if shift >= 0:
            k = shift
            if k < n:
                g[: n - k] -= best_h[k:]            # -h_{i+shift}
                g[k:]      -= best_h[: n - k]        # -h_{i-shift}
        else:
            k = -shift
            if k < n:
                g[k:]      -= best_h[: n - k]        # -h_{i+shift}
                g[: n - k] -= best_h[k:]            # -h_{i-shift}

        # step size schedule (decrease geometrically)
        step = 0.4 * (0.97 ** (cur_c5 * 200)) * dx   # heuristic
        new_h = best_h - step * g
        new_h = _project_to_constraints(new_h, target_sum)

        new_c5, _ = _c5_and_idx(new_h, dx)
        if new_c5 < best_c5 - 1e-9:
            best_h, best_c5 = new_h, new_c5
        else:
            # no progress – break after a few stagnations
            if step < 1e-6:
                break

    # -----------------------------------------------------------------
    # Phase 2 – stochastic local search (pairwise mass transfers)
    # -----------------------------------------------------------------
    while time.time() - start_time < time_limit - 0.5:
        i, j = rng.integers(0, n, size=2)
        if i == j:

        hi, hj = best_h[i], best_h[j]

        # feasible interval for transfer δ from i to j (i gains, j loses)
        lower = max(-hi, -(1.0 - hj))
        upper = min(1.0 - hi, hj)
        if lower >= upper:

        delta = rng.uniform(lower, upper)
        new_h = best_h.copy()
        new_h[i] += delta
        new_h[j] -= delta   # sum preserved exactly

        new_c5, _ = _c5_and_idx(new_h, dx)
        if new_c5 < best_c5 - 1e-9:
            best_h, best_c5 = new_h, new_c5

    # -----------------------------------------------------------------
    # Return the best construction found
    # -----------------------------------------------------------------
    return best_h, best_c5, n
