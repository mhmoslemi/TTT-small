def perturb(h, max_step):

    i, j = np.random.choice(n, 2, replace=False)
    # direction: delta added to i, subtracted from j
    # allowable delta:
    lo = max(-max_step, -h[i], h[j] - 1)

We can solve via iterative method: apply projection onto sum constraint ignoring bounds: shift by lambda to meet sum (v_i - lambda). Then clip bounds, then re-apply sum constraint for remaining free variables. This is similar to "projection onto a capped simplex". There's known algorithm but we can implement a simple iterative approach:

- Initialize: set x = v copy.
- While True:
   - Let free = (x > 0) & (x < 1) (i.e., not at bounds). Let m = number of free items.
   - Compute sum_free = x[free].sum()
   - Compute sum_fixed = x[~free].sum()
   - Desired sum for free entries = target - sum_fixed.
   - If m == 0: break.
   - Compute avg = desired_sum / m.
   - For i in free:
      - if avg < 0: set x[i] = 0 (move to lower bound)
      - elif avg > 1: set x[i] = 1 (move to upper bound)
      - else: set x[i] = avg.
   - After updating, recompute free set (some may have hit bounds).
   - Loop until convergence.

This is correct because the optimal distribution for free variables is uniform (since objective is squared distance and constraints symmetrical). However, due to v_i may have different start values; but the projection onto simplex with bounds leads to equal Lagrange multiplier for all free variables, thus they are set to the same value (clipped). So this algorithm works.

But careful: The algorithm may produce a vector not exactly minimal distance if v has varying original values. The general solution for projection onto a capped simplex indeed has the property that for free variables, they are projected onto a common threshold value (like lambda) that ensures sum constraint, after clipping. This algorithm converges after at most n+1 iterations.

def project_onto_capped_simplex(v, target):
    n = len(v)
    # ensure feasible target <= n and >=0
    if target < 0 or target > n:
        raise ValueError
    # start with clipping to [0,1]
    x = np.clip(v, 0, 1).astype(np.float64)
    # if sum already target within tolerance:
    total = x.sum()
    if abs(total - target) < 1e-12:
        return x
    # iterative algorithm
    # initialize mask for free variables
    # We'll use while loop max n times.
    for _ in range(n):
        # Compute the set of indices that are not at bounds.
        free = (x > 0) & (x < 1)
        m = free.sum()
        if m == 0:
            break
        sum_fixed = x[~free].sum()
        # desired sum for free vars
        target_free = target - sum_fixed
        # compute new value for free variables
        new_val = target_free / m
        # clip new_val to [0,1] but we know free indices are not at bounds, so maybe new_val will be within 0..1; if not, we need to set those indices to bound and repeat
        if new_val < 0:
            # set free ones to zero (lower bound)
            x[free] = 0.0
        elif new_val > 1:
            x[free] = 1.0
        else:
            x[free] = new_val
        # after setting values, loop again to adjust bound violations (if any)
        # loop will converge
        if abs(x.sum() - target) < 1e-12:
            break
    # final clip and ensure sum
    # due to rounding errors adjust tiny differences by distributing among non-bound elements
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff to free variables if any
        free = (x > 0) & (x < 1)
        if free.any():
            # add diff uniformly
            x[free] += diff / free.sum()
            # clip again
            x = np.clip(x, 0, 1)
        else:
            # if no free variables, adjust a random variable within bounds if possible
            idx = np.random.choice(n)
            x[idx] = np.clip(x[idx] + diff, 0, 1)
    # final ensure sum close
    # may still have slight error but okay
    return x

def propose_targeted_move(h, max_step, shift, rng):
    n = len(h)
    if shift == 0:
        return None
    # choose i such that i+shift is in bounds
    if shift >= n:
        return None
    # feasible indices:
    idx_candidates = np.arange(n - shift)
    # filter those where both positions not at upper bound
    feasible_mask = (h[idx_candidates] < 1 - 1e-12) & (h[idx_candidates + shift] < 1 - 1e-12)
    if not np.any(feasible_mask):
        # fallback
        return None
    i = rng.choice(idx_candidates[feasible_mask])
    # choose j to decrease such that h[j] > 0 and j not equal to i or i+shift
    # choose from indices where h > 0 and not i,i+shift
    dec_candidates = np.where((h > 1e-12) & (np.arange(n) != i) & (np.arange(n) != i + shift))[0]
    if len(dec_candidates) == 0:
        return None
    j = rng.choice(dec_candidates)
    # compute allowed delta:
    delta_max = min(max_step,
                    1 - h[i],
                    1 - h[i+shift],
                    h[j] / 2.0)
    if delta_max <= 0:
        return None
    delta = rng.uniform(0, delta_max)
    new_h = h.copy()
    new_h[i] += delta
    new_h[i+shift] += delta
    new_h[j] -= 2 * delta
    # Ensure bound after operation (should be okay, but clip to be safe)
    new_h = np.clip(new_h, 0, 1)
    # Reproject sum? sum is unchanged: +delta + delta - 2*delta =0.
    return new_h

    best_h = adjust_sum_to_target(best_h, target_sum, rng)
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)

    return (best_h.tolist(), float(final_c5), int(n))
