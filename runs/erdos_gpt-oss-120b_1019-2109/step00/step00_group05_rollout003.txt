import time
import numpy as np

def _cost(h):
    """Return C5 for a step function given by array h."""
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))


def _mismatch_pairs(h, d):
    """Indices i where h[i]=1,h[i+d]=0 and indices j where h[j]=0,h[j+d]=1."""
    n = len(h)
    idx = np.arange(n)
    mask = (idx + d >= 0) & (idx + d < n)
    i = idx[mask]
    j = i + d
    hi = h[i]
    hj = h[j]
    A = i[(hi == 1) & (hj == 0)]
    B = i[(hi == 0) & (hj == 1)]
    return A, B


def _random_binary(n):
    """Return a binary (or half‑binary) array of length n with sum=n/2."""
    half = n // 2
    h = np.zeros(n, dtype=float)
    h[np.random.choice(n, half, replace=False)] = 1.0
    if n % 2:                     # odd length → one 0.5 entry
        rest = np.setdiff1d(np.arange(n), np.where(h == 1)[0])
        h[np.random.choice(rest)] = 0.5
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Search for a low‑overlap step function."""
    np.random.seed(seed)
    start = time.time()

    # ----- initial vector -------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.shape[0]
        half = n // 2
        h0 = np.zeros(n, dtype=float)
        top = np.argsort(-init)[:half]
        h0[top] = 1.0
        if n % 2:
            nxt = np.argsort(-init)[half]
            h0[nxt] = 0.5
    else:
        n = 70
        h0 = _random_binary(n)

    cur = h0.copy()
    cur_cost = _cost(cur)
    best = cur.copy()
    best_cost = cur_cost

    no_imp = 0                     # steps without improvement

    # ----- main stochastic descent ----------------------------------------
    while time.time() - start < budget_s * 0.95:
        # identify the worst shift
        corr = np.correlate(cur, 1 - cur, mode='full')
        d = int(np.argmax(corr) - (n - 1))

        A, B = _mismatch_pairs(cur, d)

        # try a targeted swap that fixes the worst shift
        if A.size and B.size and np.random.rand() < 0.8:
            i = np.random.choice(A)
            j = np.random.choice(B)
        else:

            ones = np.flatnonzero(cur == 1)
            zeros = np.flatnonzero(cur == 0)
            if ones.size == 0 or zeros.size == 0:
                cur = _random_binary(n)
                cur_cost = _cost(cur)
                no_imp = 0
                continue
            i = np.random.choice(ones)
            j = np.random.choice(zeros)

            cur_cost = cand_cost
            if cur_cost < best_cost - 1e-12:
                best = cur.copy()
                best_cost = cur_cost
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        # restart if stuck
        if no_imp > 5000:
            cur = _random_binary(n)
            cur_cost = _cost(cur)
            no_imp = 0

        # occasional random jump for exploration

                    improved = True
