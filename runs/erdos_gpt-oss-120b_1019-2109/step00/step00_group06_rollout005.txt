import numpy as np

def project_to_box_and_sum(u, target):
    # binary search for lambda
    lo = -1e6
    hi = 1e6
    for _ in range(50):
        mid = (lo + hi) / 2.0
        v = np.clip(u - mid, 0.0, 1.0)
        s = v.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(u - (lo+hi)/2.0, 0.0, 1.0)

def evaluate_c5(h):
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))

def optimize(initial_h, max_time):
    n = len(initial_h)
    dx = 2.0 / n
    target_sum = n / 2.0
    h = np.clip(initial_h, 0.0, 1.0)
    # project sum
    h = project_to_box_and_sum(h, target_sum)

    best_h = h.copy()
    best_c = evaluate_c5(h)
    start_time = time.time()

    gamma = 1.0
    lr = 0.05

    iter = 0
    while time.time() - start_time < max_time:
        # compute correlation
        f = np.correlate(h, 1 - h, mode='full') * dx
        C = f.max()
        if C < best_c - 1e-8:
            best_c = C
            best_h = h.copy()
        # compute softmax weights
        max_f = f.max()
        # For stability:
        exp_vals = np.exp(gamma * (f - max_f))
        w = exp_vals / exp_vals.sum()
        # gradient
        offset = n - 1
        conv1 = np.convolve(w, h, mode='full')
        # reversed w
        w_rev = w[::-1]
        conv2 = np.convolve(w_rev, h, mode='full')
        grad = -dx * (conv1[offset: offset + n] + conv2[offset: offset + n])
        # step
        u = h - lr * grad
        # projection
        h = project_to_box_and_sum(u, target_sum)
        # schedule
        iter += 1
        # optionally adjust lr or gamma
        if iter % 5000 == 0:
            # decrease lr
            lr *= 0.9
            gamma = min(gamma * 1.2, 50.0)
    return best_h, best_c
```

def project_to_sum_box(u, target, tol=1e-8):
    lo = -np.max(u) - 5.0
    hi = np.max(u) + 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        v = np.clip(u - mid, 0.0, 1.0)
        s = v.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    v = np.clip(u - (lo+hi)/2.0, 0.0, 1.0)
    # adjust possible tiny sum error due to tolerance.
    diff = v.sum() - target

        interior = (v > 1e-12) & (v < 1-1e-12)
        if interior.any():
            v[interior] -= diff / interior.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

import numpy as np
import time

def _project_to_sum_box(u, target, tol=1e-9, max_iter=60):
    """Project vector u onto the box [0,1]^n with sum = target."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        v = np.clip(u - mid, 0.0, 1.0)
        s = v.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    v = np.clip(u - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny adjustment if rounding error remains
    diff = v.sum() - target
    if abs(diff) > tol:
        interior = (v > 0.0) & (v < 1.0)
        if interior.any():
            v[interior] -= diff / interior.sum()
            v = np.clip(v, 0.0, 1.0)
    return v

def _c5_value(h, dx):
    """Return C_5 for vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def _optimize(start_h, time_limit):
    """Projected gradient descent on a smooth approximation of max overlap."""
    n = start_h.shape[0]
    target = n * 0.5
    dx = 2.0 / n
    h = _project_to_sum_box(start_h.astype(float).copy(), target)
    best_h = h.copy()
    best_c = _c5_value(h, dx)

    start = time.time()
    lr = 0.25               # base step size
    gamma = 1.0             # softmax temperature
    it = 0
    stagn = 0

    while time.time() - start < time_limit:
        # correlation and current C5
        f = np.correlate(h, 1.0 - h, mode="full") * dx
        cur_c = f.max()
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = h.copy()
            stagn = 0
        else:
            stagn += 1

        # soft‑max weights approximating the max
        m = f.max()
        w = np.exp(gamma * (f - m))
        w /= w.sum()

        # gradient of the smoothed objective
        offset = n - 1
        conv1 = np.convolve(w, h, mode="full")
        conv2 = np.convolve(w[::-1], h, mode="full")
        grad = -dx * (conv1[offset:offset + n] + conv2[offset:offset + n])

        # decreasing step size
        step = lr / (1.0 + it * 0.001)
        h = _project_to_sum_box(h - step * grad, target)

        # schedule updates
        if it % 2000 == 0 and it > 0:
            gamma = min(gamma * 1.2, 40.0)
            lr *= 0.92
        if stagn >= 5000:
            # shake to escape flat regions
            h = _project_to_sum_box(h + np.random.randn(n) * 0.02, target)
            stagn = 0

        it += 1

    return best_h, best_c

    start_all = time.time()

    # ------------------------------------------------------------------
