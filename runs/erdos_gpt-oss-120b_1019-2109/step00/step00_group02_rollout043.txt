def smooth_max_and_gradient(h, tau):
    n = h.shape[0]
    dx = 2.0 / n
    C = np.correlate(h, 1 - h, mode='full')
    C = C * dx
    maxC = C.max()
    # compute weights
    w = np.exp((C - maxC) / tau)
    w_sum = w.sum()
    w = w / w_sum
    # compute gradient
    grad = np.zeros_like(h)
    # list of shifts s where index in C corresponds to shift = i - (n - 1)
    shifts = np.arange(-n+1, n)
    for idx, s in enumerate(shifts):
        weight = w[idx]
        if weight == 0:
            continue
        # compute forward range
        i_a_start = max(0, -s)
        i_a_end = min(n, n - s)
        if i_a_start < i_a_end:
            grad[i_a_start:i_a_end] += weight * dx * (1 - h[i_a_start + s : i_a_end + s])
        # compute backward range
        i_b_start = max(0, s)
        i_b_end = min(n, n + s)
        if i_b_start < i_b_end:
            grad[i_b_start:i_b_end] -= weight * dx * h[i_b_start - s : i_b_end - s]
    # compute smooth max value: f = maxC + tau * np.log(w_sum)
    f = maxC + tau * np.log(w_sum)
    return f, grad
```

def project_to_sum_and_box(v, target_sum):
    lo = -1.0
    hi = 1.0
    # Actually need guarantee that sum at lo >= target_sum and at hi <= target_sum
    # compute sums at extremes:
    # lo: v - lo -> big value -> clipped to 1 -> sum = n
    # hi: v - hi -> small value -> clipped to 0 -> sum = 0
    # So bracketing works.
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x

def smooth_max_compute(h, tau):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    C = corr * dx
    maxC = C.max()
    exp_weights = np.exp((C - maxC) / tau)
    sumw = exp_weights.sum()
    w = exp_weights / sumw
    # compute gradient:
    grad = np.zeros_like(h)
    # shift indices from -(n-1) to n-1
    shifts = np.arange(-n+1, n)
    for idx, s in enumerate(shifts):
        weight = w[idx]
        if weight == 0:
            continue
        # forward part
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_start < i_end:
            grad[i_start:i_end] += weight * dx * (1 - h[i_start + s : i_end + s])
        # backward part
        j_start = max(0, s)
        j_end = min(n, n + s)
        if j_start < j_end:
            grad[j_start:j_end] -= weight * dx * h[j_start - s : j_end - s]
    # smooth objective value (optional)
    smooth_val = maxC + tau * np.log(sumw)
    return smooth_val, grad

    for n in extra_ns:
        if n not in candidate_ns:
            candidate_ns.append(n)

    time_per_n = budget_s / max(len(candidate_ns), 1)

    for n in candidate_ns:
        if time.time() - start_time > budget_s - 0.5:
            break

        target_sum = n / 2.0

        # ---------- initialise h ----------
        if init_arr is not None and n == init_arr.shape[0]:
            h = init_arr.copy()
            h = np.clip(h, 0.0, 1.0)
            h = _project_to_sum_and_box(h, target_sum)
        else:
            x = np.arange(n)
            # cosine‑shaped start (high at the ends, low in the middle)
            h = 0.5 + 0.5 * np.cos(np.pi * x / (n - 1))
            h += rng.uniform(-0.1, 0.1, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = _project_to_sum_and_box(h, target_sum)

        cur_val = _cmax(h)
        if cur_val < best_val:
            best_val, best_h, best_n = cur_val, h.copy(), n

        # ---------- gradient‑descent phase ----------
        tau = 0.5
        step0 = 0.4
        iter_cnt = 0
        phase_start = time.time()
        while (time.time() - start_time < budget_s - 0.5 and
               time.time() - phase_start < time_per_n):
            smooth_val, grad = _smooth_max_and_grad(h, tau)
            step = step0 / (1.0 + iter_cnt * 0.001)
            h_new = h - step * grad
            h_new = _project_to_sum_and_box(h_new, target_sum)
            new_val = _cmax(h_new)

            if new_val < cur_val - 1e-9:
                h, cur_val = h_new, new_val
                if new_val < best_val:
                    best_val, best_h, best_n = new_val, h.copy(), n
            else:
                # occasional random jiggle to escape flat regions
                if rng.random() < 0.005:
                    h = np.clip(h + rng.uniform(-0.05, 0.05, size=n), 0.0, 1.0)
                    h = _project_to_sum_and_box(h, target_sum)
                    cur_val = _cmax(h)

            if iter_cnt % 100 == 0 and tau > 0.02:
                tau *= 0.95
            iter_cnt += 1

        # ---------- random transfer fine‑tuning ----------
        for _ in range(2000):
            if time.time() - start_time > budget_s - 0.5:
                break
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            max_eps = min(0.1, h[i], 1.0 - h[j])
            if max_eps <= 0.0:
                continue
            eps = rng.uniform(0.0, max_eps)
            cand = h.copy()
            cand[i] -= eps
            cand[j] += eps
            cand_val = _cmax(cand)
            if cand_val < cur_val - 1e-9:
                h, cur_val = cand, cand_val
                if cand_val < best_val:
                    best_val, best_h, best_n = cand_val, cand.copy(), n

    # fallback in the extremely unlikely event that nothing was found
    if best_h is None:
        best_h = np.array([0.5, 0.5])
        best_val = _cmax(best_h)
        best_n = best_h.shape[0]

    return best_h, best_val, best_n
