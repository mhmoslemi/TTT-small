import math
import random
import time

import numpy as np


def _compute_c5(h: np.ndarray) -> float:
    """Return the Erdős overlap constant for the step function h."""
    n = h.shape[0]
    # correlation of h with (1‑h)
    conv = np.correlate(h, 1.0 - h, mode="full")
    return (2.0 / n) * float(conv.max())


def _min_corr_shift(h: np.ndarray) -> int:
    """Shift (as an integer offset) that gives the minimal autocorrelation."""
    corr = np.correlate(h, h, mode="full")
    idx = int(np.argmin(corr))
    return idx - (h.shape[0] - 1)


def _gradient_min_corr(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the autocorrelation at the given shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        a = np.arange(shift, n)
        b = np.arange(0, n - shift)
    else:
        s = -shift
        a = np.arange(0, n - s)
        b = np.arange(s, n)
    # each pair (a[i], b[i]) contributes h[b[i]] to grad[a[i]] and h[a[i]] to grad[b[i]]
    g[a] += h[b]
    g[b] += h[a]
    return g


def _project_bounds_sum(x: np.ndarray, target_sum: float, lower: float = 0.0, upper: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of x onto the set
        {z | lower ≤ z_i ≤ upper,  Σ z_i = target_sum }.
    """
    # binary search for λ such that sum clip(x+λ) = target_sum
    lo = -np.max(x - lower)
    hi = np.max(upper - x)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        z = np.clip(x + mid, lower, upper)
        s = z.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    z = np.clip(x + (lo + hi) / 2.0, lower, upper)

    # tiny correction for round‑off
    diff = target_sum - z.sum()
    if abs(diff) > 1e-9:
        free = (z > lower + 1e-12) & (z < upper - 1e-12)
        if free.any():
            z[free] += diff / free.sum()
            z = np.clip(z, lower, upper)
    return z


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős C₅ constant.

    Returns
    -------
    h_values : list of float
        Optimised values of h on the chosen grid.
    c5_bound : float
        The achieved upper bound C₅.
    n_points : int
        Number of discretisation points.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Initialise a feasible h
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h = np.array(init, dtype=np.float64)
        n = h.shape[0]
        h = np.clip(h, 0.0, 1.0)
    else:
        # a V‑shaped starter (high at the ends, low in the middle)
        n = 84  # modest size: gives flexibility without heavy cost
        idx = np.arange(n, dtype=np.float64)
        h = 2.0 * np.abs(idx / (n - 1.0) - 0.5)   # values in [0,1]
    target_sum = n / 2.0
    h = _project_bounds_sum(h, target_sum)

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing / projected‑gradient search
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c = _compute_c5(best_h)

    cur_h = h.copy()
    cur_c = _compute_c5(cur_h)

    start_time = time.time()
    max_delta = 0.15          # maximal change in a pairwise update
    temp0 = 0.05              # initial temperature for SA
    temp_min = 1e-6

    while True:
        if time.time() - start_time > budget_s - 0.2:
            break

        elapsed = time.time() - start_time
        frac = elapsed / budget_s

        # linear cooling for temperature and step magnitude
        temperature = max(temp_min, temp0 * (1.0 - frac))
        delta = max_delta * (1.0 - frac)

        # probability of using a gradient step (increases as we cool)
        p_gradient = 0.2 + 0.8 * frac
        if random.random() < p_gradient:
            # ------------ gradient move (increase the minimal autocorrelation) ----------
            shift = _min_corr_shift(cur_h)
            grad = _gradient_min_corr(cur_h, shift)
            # move a little in the ascent direction
            h_candidate = _project_bounds_sum(cur_h + delta * grad, target_sum)
        else:
            # ------------ random pair adjustment -----------------------------------------
            i, j = random.sample(range(n), 2)
            # Pick a delta that respects bounds on both entries
            lo = max(-cur_h[i], cur_h[j] - 1.0, -delta)
            hi = min(1.0 - cur_h[i], cur_h[j], delta)
            if lo >= hi:
                continue
            d = random.uniform(lo, hi)
            h_candidate = cur_h.copy()
            h_candidate[i] += d
            h_candidate[j] -= d
            # minor clipping to eradicate round‑off excess
            h_candidate[i] = min(max(h_candidate[i], 0.0), 1.0)
            h_candidate[j] = min(max(h_candidate[j], 0.0), 1.0)

        cand_c = _compute_c5(h_candidate)

        # ----- Metropolis acceptance -------------------------------------------------
        if cand_c < cur_c or random.random() < math.exp(-(cand_c - cur_c) / max(temperature, 1e-12)):
            cur_h, cur_c = h_candidate, cand_c

        # ----- Keep the best ever found ---------------------------------------------
        if cand_c < best_c:
            best_c, best_h = cand_c, h_candidate.copy()

    # final projection (in case of tiny drift) and evaluation
    best_h = _project_bounds_sum(best_h, target_sum)
    final_c = _compute_c5(best_h)

    return best_h.tolist(), float(final_c), n
