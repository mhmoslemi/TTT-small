import numpy as np, time, random

def compute_c5(h, dx):
    cross = np.correlate(h, 1 - h, mode='full') * dx
    return cross.max()

def move_on_worst_shift(h, shift, step):
    n = len(h)
    # compute contributions vector cont
    if shift >= 0:
        i = np.arange(0, n - shift)
        j = i + shift
    else:
        i = np.arange(-shift, n)
        j = i + shift
    cont = h[i] * (1 - h[j])
    # pick index with largest contribution
    idx = np.argmax(cont)
    donor = i[idx]
    recipient = j[idx]
    # compute allowable delta
    max_delta = min(step, h[donor], 1 - h[recipient])
    if max_delta <= 0:
        return None, 0
    # choose delta maybe random fraction of max_delta to explore
    delta = max_delta * np.random.rand()  # uniform in [0, max_delta]
    if delta <= 0:
        return None, 0
    # apply
    h_new = h.copy()
    h_new[donor] -= delta
    h_new[recipient] += delta
    # ensure still within bounds (should be)
    return h_new, delta

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    # define time left function
    def time_left():
        return budget_s - (time.time() - start)
    
    # Determine n_points

    try:
        base = initial_h_values
    except NameError:
        base = None

    # Decide on n
    if base is not None:
        n0 = len(base)
    else:
        n0 = 0

    # Choose n as max(80, n0) maybe 200
    n = max(n0, 200)  # artificially 200 for richer search
    # If base is available and we need to upsample
    if base is not None:
        if n0 != n:
            # upsample via linear interpolation
            x_old = np.linspace(0, 2, n0, endpoint=False)
            x_new = np.linspace(0, 2, n, endpoint=False)
            h = np.interp(x_new, x_old, base)
        else:
            h = np.array(base, dtype=np.float64)
    else:
        # random initialization
        h = np.random.rand(n)
    
    # Project onto constraints
    target_sum = n / 2.0  # because integral =1 -> sum = n/2
    # Clip to [0,1] first
    h = np.clip(h, 0, 1)
    # Project using lambda method
    def project(v):
        # find lambda s.t. sum clip(v - lam,0,1) = target_sum
        # Use binary search
        low = -2.0
        high = 2.0
        for _ in range(60):
            lam = (low + high) / 2.0
            proj = np.clip(v - lam, 0, 1)
            s = proj.sum()
            if s > target_sum:
                low = lam
            else:
                high = lam
        proj = np.clip(v - (low + high) / 2.0, 0, 1)
        # adjust small diff due to rounding
        diff = target_sum - proj.sum()
        if abs(diff) > 1e-12:
            # distribute among non-bound entries
            free = (proj > 0) & (proj < 1)
            free_idx = np.where(free)[0]
            if free_idx.size > 0:
                proj[free_idx] += diff / free_idx.size
                proj = np.clip(proj, 0, 1)
            else:
                # fallback: adjust first element within bounds
                i = np.argmin(proj) if diff > 0 else np.argmax(proj)
                proj[i] = np.clip(proj[i] + diff, 0, 1)
        return proj

    h = project(h)
    # Ensure sum exactly target
    # may need a small correction
    # h = project(h) ensures it.

    dx = 2.0 / n

    # compute initial C5
    cross = np.correlate(h, 1 - h, mode='full') * dx
    best_val = cross.max()
    best_h = h.copy()
    best_cross = cross.copy()
    # Keep track of iteration counts
    iter_count = 0
    no_improve = 0
    # pre-define step schedule: start large, gradually shrink

        # Use linear decay

        max_step = 0.2
        min_step = 0.001
        decay = min(iter_count / 200000, 1.0)  # after 200k iterations fully decayed
        step = max_step * (1 - decay) + min_step * decay
        
        # compute cross and worst shift
        cross = np.correlate(h, 1 - h, mode='full') * dx
        cur_max = cross.max()
        # Find shift(s) where max occurs (first)
        # Use argmax
        idx = np.argmax(cross)
        shift = idx - (n - 1)  # because length 2n-1; center index n-1 corresponds to shift 0

        if shift >= 0:

            j = i + shift

            i = np.arange(-shift, n)
            j = i + shift

    best_h = np.clip(best_h, 0, 1)

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the box [0,1] with exact sum constraint
# ------------------------------------------------------------
def _project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto {x | 0≤x≤1, sum(x)=target}.
    The solution has the form x_i = clip(v_i - λ, 0, 1) for a scalar λ.
    """
    low, high = -2.0, 2.0          # safe interval for λ (covers all possibilities)
    for _ in range(60):           # binary search for λ
        lam = (low + high) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    proj = np.clip(v - lam, 0.0, 1.0)

    # tiny numerical correction to hit the target sum exactly
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        idx = np.where(free)[0]
        if idx.size:
            proj[idx] += diff / idx.size
            proj = np.clip(proj, 0.0, 1.0)
        else:                       # all at bounds – adjust a single entry safely
            i = 0 if diff > 0 else -1
            proj[i] = np.clip(proj[i] + diff, 0.0, 1.0)
    return proj


# ------------------------------------------------------------
# Core evaluation (same formula used by the harness)
# ------------------------------------------------------------
def _c5_value(h: np.ndarray, dx: float) -> float:
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    np.random.seed(seed)

    # --------------------------------------------------------------------
    # 1. Determine the discretisation size and initialise h
    # --------------------------------------------------------------------
    try:
        base = initial_h_values          # provided by the harness, may be None
    except NameError:
        base = None

    if base is not None:
        n0 = len(base)
    else:
        n0 = 0

    # we aim for a reasonably fine grid but still cheap per iteration
    n = max(200, n0)                     # n points over [0,2]
    dx = 2.0 / n
    target_sum = n / 2.0                 # because ∫h = 1 ⇒ Σh·dx = 1

    if base is not None:
        if n0 == n:
            h = np.asarray(base, dtype=np.float64).copy()
        else:
            # linearly interpolate the parent to the new grid
            x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            h = np.interp(x_new, x_old, base).astype(np.float64)
    else:
        # random start respecting the box; projection will fix the sum
        h = np.random.rand(n)

    # enforce feasibility
    h = np.clip(h, 0.0, 1.0)
    h = _project_sum_box(h, target_sum)

    # --------------------------------------------------------------------
    # 2. Optimisation parameters
    # --------------------------------------------------------------------
    start_time = time.time()
    def time_left():
        return budget_s - (time.time() - start_time)

    # soft‑max temperature (β) – we start smooth and anneal it
    beta = 1.0
    beta_max = 40.0
    beta_inc = (beta_max / beta_max) * 0.002  # tiny increase each iteration

    # learning‑rate schedule (will be reduced if a step does not improve)
    base_lr = 0.12

    # bookkeeping
    best_h = h.copy()
    best_val = _c5_value(h, dx)
    no_improve = 0
    it = 0

    # --------------------------------------------------------------------
    # 3. Main loop – projected gradient descent on a smoothed max objective
    # --------------------------------------------------------------------
    while time_left() > 0.5:
        # ----- compute current overlap values (unscaled) -----
        cross = np.correlate(h, 1.0 - h, mode='full') * dx   # length 2n-1
        cur_max = cross.max()

        # ----- soft‑max weights for the current β -----
        # shift by the max for numerical stability
        shifted = cross - cross.max()
        w = np.exp(beta * shifted)
        w /= w.sum()                                        # shape (2n‑1,)

        # ----- convolutions needed for the gradient -----
        # B_j = Σ_k w_k h_{j‑k}
        conv_b = np.convolve(w, h, mode='full')

        grad = dx * (1.0 - A - B)                           # shape (n,)

            val_tmp = _c5_value(h_tmp, dx)
            if val_tmp < cur_max:
                h = h_tmp
                cur_max = val_tmp
                if val_tmp < best_val:
                    best_val = val_tmp
                    best_h = h.copy()
            no_improve = 0

    # --------------------------------------------------------------------
    # 4. Final sanity projection and C5 evaluation
    # --------------------------------------------------------------------
    best_h = _project_sum_box(best_h, target_sum)
    final_c5 = _c5_value(best_h, dx)

    # Return a plain Python list for portability
    return (best_h.tolist(), float(final_c5), n)
