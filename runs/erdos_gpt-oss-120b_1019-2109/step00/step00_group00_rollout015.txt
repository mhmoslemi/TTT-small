def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx, corr

def local_search(h, dx, best_val, start_time, budget_s):
    n = len(h)
    ones = set(np.flatnonzero(h == 1))
    zeros = set(np.flatnonzero(h == 0))
    best_h = h.copy()
    cur_val = best_val
    cur_corr = None # compute as needed inside loop
    # We'll compute cur_corr initially for guidance.
    cur_val, cur_corr = compute_c5(h, dx)
    # while time left
    iter_no = 0
    last_improve_iter = 0
    while time.time() - start_time < budget_s - 0.5:
        # choose shift with max
        max_idx = np.argmax(cur_corr)
        shift = max_idx - (n - 1)
        # produce list of mismatched i for that shift
        # we can compute vectorized: for shift >=0: i range(0,n-shift)
        # mismatched mask = (h[i]==1) & (h[i+shift]==0)
        # for shift <0: j = i+shift; i range(-shift,n)
        if shift >= 0:
            i_range = np.arange(0, n - shift)
            mask = (h[i_range] == 1) & (h[i_range + shift] == 0)
            contrib = i_range[mask]
        else:
            offset = -shift
            i_range = np.arange(offset, n)
            mask = (h[i_range] == 1) & (h[i_range + shift] == 0)  # shifted index smaller
            contrib = i_range[mask]
        if contrib.size == 0:
            # fallback: random pair
            i = random.choice(list(ones))
        else:
            i = int(random.choice(contrib))
        # pick j random from zeros (excluding i)
        # ensure j != i (should be zero anyway)
        j = random.choice(list(zeros))
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new correlation
        new_val, new_corr = compute_c5(h, dx)
        # Accept if new_val <= cur_val+ small tolerance else revert
        if new_val <= cur_val:
            # keep
            cur_val = new_val
            cur_corr = new_corr
            # update sets
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            # update best
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                last_improve_iter = iter_no
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
        iter_no += 1
        # If no improvement for many iterations, optionally do random restart or random swap to escape.
        if iter_no - last_improve_iter > 20000:
            # random restart: generate new random binary
            h = np.zeros(n, dtype=np.float64)
            h[:n//2] = 1.0
            np.random.shuffle(h)
            ones = set(np.flatnonzero(h == 1))
            zeros = set(np.flatnonzero(h == 0))
            cur_val, cur_corr = compute_c5(h, dx)
            last_improve_iter = iter_no
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random, math
    start_time = time.time()
    random.seed(seed)
    np.random.seed(seed)
    n = 200  # could be adjusted based on remaining time
    if n % 2 != 0: n += 1
    dx = 2.0 / n

    # Helper eval
    def compute_c5(arr):
        corr = np.correlate(arr, 1 - arr, mode='full')
        return np.max(corr) * dx, corr

    best_h = None
    best_val = float('inf')

    # if initial_h_values exist and length matches n, use as candidate
    try:
        init = initial_h_values
        if len(init) == n:
            cand = np.clip(init, 0, 1).astype(np.float64)
            # Ensure binary: threshold?
            cand_bin = (cand >= 0.5).astype(np.float64)
            # ensure sum correct
            ones_needed = n // 2
            # If sum mismatch adjust
            current_sum = int(cand_bin.sum())
            if current_sum != ones_needed:
                # adjust by setting top values accordingly
                idx_sorted = np.argsort(-cand)
                cand_bin[:] = 0
                cand_bin[idx_sorted[:ones_needed]] = 1
            val, _ = compute_c5(cand_bin)
            if val < best_val:
                best_val = val
                best_h = cand_bin.copy()
    except NameError:
        pass

    # deterministic rotation candidates
    alphas = np.linspace(0.01, 0.99, 300)  # many samples
    # also add some special alphas
    special = [ (np.sqrt(2) - 1), (np.sqrt(3) - 1), (np.sqrt(5) - 2), (np.sqrt(5)-1)/2, 0.3819660112501051 ]  # golden ratio conjugate? phi-1?
    alphas = np.concatenate([alphas, np.array(special)])
    # remove duplicates (approx)
    alphas = np.unique(alphas)

    # ensure not too many - still fine
    for alpha in alphas:
        # quick time check
        if time.time() - start_time > budget_s * 0.2:
            break
        # generate vector
        idx = np.argsort((np.arange(n) * alpha) % 1)
        cand = np.zeros(n, dtype=np.float64)
        cand[idx[:n//2]] = 1.0
        val, _ = compute_c5(cand)
        if val < best_val:
            best_val = val
            best_h = cand.copy()
    # If still no candidate (should have), fallback random binary
    if best_h is None:
        best_h = np.zeros(n, dtype=np.float64)
        best_h[:n//2] = 1.0
        np.random.shuffle(best_h)
        best_val, _ = compute_c5(best_h)

    # Local search with time budget remaining
    # define local search function
    def local_search(h, cur_val, time_limit):
        n = len(h)
        ones = set(np.flatnonzero(h == 1))
        zeros = set(np.flatnonzero(h == 0))
        best_local_h = h.copy()
        best_local_val = cur_val
        cur_h = h
        cur_corr_val, cur_corr = compute_c5(cur_h)
        iter_no = 0
        last_improve = 0
        while time.time() < time_limit - 0.5:
            # find worst shift and mismatches
            max_idx = np.argmax(cur_corr)
            shift = max_idx - (n - 1)
            # find contributors (positions where h=1 and shifted index has 0)
            if shift >= 0:
                i_rng = np.arange(0, n - shift)
                mask = (cur_h[i_rng] == 1) & (cur_h[i_rng + shift] == 0)
                contrib = i_rng[mask]
            else:
                offset = -shift
                i_rng = np.arange(offset, n)
                mask = (cur_h[i_rng] == 1) & (cur_h[i_rng + shift] == 0)
                contrib = i_rng[mask]
            if contrib.size > 0:
                i = int(random.choice(contrib))
            else:
                # fallback random 1 index
                i = random.choice(list(ones))
            # choose random zero index
            j = random.choice(list(zeros))
            # try swap
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            new_val, new_corr = compute_c5(cur_h)
            # Accept if not worse (deterministic)
            if new_val <= cur_corr_val:
                # accept
                cur_corr_val, cur_corr = new_val, new_corr
                ones.remove(i)
                zeros.add(i)
                zeros.remove(j)
                ones.add(j)
                if new_val < best_local_val:
                    best_local_val = new_val
                    best_local_h = cur_h.copy()
                    last_improve = iter_no
            else:
                # revert
                cur_h[i] = 1.0
                cur_h[j] = 0.0
            iter_no += 1
            # if stagnated, random restart
            if iter_no - last_improve > 20000:
                # random restart
                cur_h = np.zeros(n, dtype=np.float64)
                cur_h[:n//2] = 1.0
                np.random.shuffle(cur_h)
                ones = set(np.flatnonzero(cur_h == 1))
                zeros = set(np.flatnonzero(cur_h == 0))
                cur_corr_val, cur_corr = compute_c5(cur_h)
                last_improve = iter_no
        return best_local_h, best_local_val

    # compute remaining time
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 2.0:
        improved_h, improved_val = local_search(best_h.copy(), best_val, time.time() + remaining_time)
        if improved_val < best_val:
            best_val = improved_val
            best_h = improved_h.copy()

    final_val, _ = compute_c5(best_h)
    # Ensure final val consistent with best_val
    if final_val < best_val:
        best_val = final_val

    return (best_h, float(best_val), n)

import numpy as np, time, random, math

def _c5_val(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx, corr

def _construct_beats(alpha, n):
    # binary vector: ones on the smallest n/2 fractional parts of i*alpha (mod 1)
    idx = np.argsort((np.arange(n) * alpha) % 1.0)
    h = np.zeros(n, dtype=np.float64)
    h[idx[:n // 2]] = 1.0
    return h

def _local_search(h, cur_val, stop_time, dx):
    n = len(h)
    ones = set(np.flatnonzero(h == 1))
    zeros = set(np.flatnonzero(h == 0))
    best_h = h.copy()
    best_val = cur_val
    cur_h = h.copy()
    cur_val, cur_corr = _c5_val(cur_h, dx)
    it = 0
    last_improve = 0
    while time.time() < stop_time - 0.5:
        # worst shift
        max_idx = np.argmax(cur_corr)
        shift = max_idx - (n - 1)
        # mismatching positions for that shift
        if shift >= 0:
            rng = np.arange(0, n - shift)
            mask = (cur_h[rng] == 1) & (cur_h[rng + shift] == 0)
            contrib = rng[mask]
        else:
            off = -shift
            rng = np.arange(off, n)
            mask = (cur_h[rng] == 1) & (cur_h[rng + shift] == 0)
            contrib = rng[mask]
        if contrib.size:
            i = int(random.choice(contrib))
        else:
            i = random.choice(tuple(ones))
        j = random.choice(tuple(zeros))
        # try swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_val, new_corr = _c5_val(cur_h, dx)
        if new_val <= cur_val:
            # accept
            cur_val, cur_corr = new_val, new_corr
            ones.remove(i)
            zeros.add(i)
            zeros.remove(j)
            ones.add(j)
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
                last_improve = it
        else:
            # revert
            cur_h[i] = 1.0
            cur_h[j] = 0.0
        it += 1
        # stagnation -> random restart
        if it - last_improve > 20000:
            cur_h = np.zeros(n, dtype=np.float64)
            cur_h[:n // 2] = 1.0
            np.random.shuffle(cur_h)
            ones = set(np.flatnonzero(cur_h == 1))
            zeros = set(np.flatnonzero(cur_h == 0))
            cur_val, cur_corr = _c5_val(cur_h, dx)
            last_improve = it
    return best_h, best_val

def _continuous_refine(h, cur_val, stop_time, dx):
    n = len(h)
    best_h = h.copy()
    best_val = cur_val
    cur_h = h.copy()
    cur_val = cur_val
    while time.time() < stop_time - 0.5:
        i, j = np.random.choice(n, 2, replace=False)
        inc = 1.0 - cur_h[i]
        dec = cur_h[j]
        if inc <= 0 or dec <= 0:
            continue
        delta = random.random() * min(inc, dec, 0.02)
        cur_h[i] += delta
        cur_h[j] -= delta
        new_val, _ = _c5_val(cur_h, dx)
        if new_val <= cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            cur_h[i] -= delta
            cur_h[j] += delta
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # Choose a modest grid size; binary vectors give exact integral.
    n = 200
    if n % 2: n += 1
    dx = 2.0 / n

    best_h = None
    best_val = float('inf')

    # Use any supplied initial vector if it matches our size.
    try:
        init = initial_h_values
        if len(init) == n:
            cand = np.clip(init, 0, 1).astype(np.float64)

            need = n // 2
            idx = np.argsort(-cand)
            cand[:] = 0.0
            cand[idx[:need]] = 1.0
            val, _ = _c5_val(cand, dx)
            if val < best_val:
                best_val, best_h = val, cand.copy()
    except NameError:
        pass

    # Scan a set of rotation parameters (Beatty constructions).
    alphas = np.concatenate([
        np.linspace(0.01, 0.99, 250),
        np.array([np.sqrt(2) - 1, np.sqrt(3) - 1,
                  (np.sqrt(5) - 1) / 2, 0.3819660112501051])
    ])
    alphas = np.unique(alphas)

    for a in alphas:
        if time.time() - start > budget_s * 0.2:
            break
        cand = _construct_beats(a, n)
        val, _ = _c5_val(cand, dx)
        if val < best_val:
            best_val, best_h = val, cand.copy()

    # If nothing yet (unlikely) start from a random binary.
    if best_h is None:
        best_h = np.zeros(n, dtype=np.float64)
        best_h[:n // 2] = 1.0
        np.random.shuffle(best_h)
        best_val, _ = _c5_val(best_h, dx)

    # Main local search (binary swaps).
    time_left = budget_s - (time.time() - start)
    if time_left > 5.0:
        stop = time.time() + time_left * 0.7
        best_h, best_val = _local_search(best_h, best_val, stop, dx)

    # Fine‑grained continuous refinement (small transfers).
    time_left = budget_s - (time.time() - start)
    if time_left > 2.0:
        stop = time.time() + time_left * 0.9
        best_h, best_val = _continuous_refine(best_h, best_val, stop, dx)

    # Final safety check.
    final_val, _ = _c5_val(best_h, dx)
    if final_val < best_val:
        best_val = final_val

    return (best_h, float(best_val), n)
