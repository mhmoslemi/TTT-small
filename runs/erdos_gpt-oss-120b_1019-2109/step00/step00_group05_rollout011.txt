def repair(h, target):
    # Clip to [0,1] (with tolerance)
    h = np.clip(h,0,1)
    total = h.sum()
    diff = target - total
    if abs(diff) > 1e-9:
        # Choose index where adjusting won't breach bounds
        # If diff > 0 (need to increase some entries), find indices < 1
        # If diff < 0 (need to decrease), find indices > 0
        candidates = np.where((diff > 0) & (h < 1 - 1e-12) | (diff < 0) & (h > 1e-12))[0]
        if candidates.size == 0:
            # This shouldn't happen unless all at bounds; fallback adjust any index
            idx = 0
        else:
            idx = np.random.choice(candidates)
        h[idx] += diff
        h[idx] = np.clip(h[idx],0,1) # just in case tiny overshoot
    return h

def transfer_move(h, target):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # Decide direction: move mass from i to j (i decrease)
    # compute maximum delta
    max_delta = min(h[i], 1 - h[j])
    # also possible to move from j to i: max_delta_rev = min(h[j], 1 - h[i])
    # Randomly choose direction:
    if np.random.rand() < 0.5:
        # use i->j
        if max_delta <= 0:
            # no move possible, maybe choose other pair or treat as skip
            return None
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:
        # j->i
        max_delta_rev = min(h[j], 1 - h[i])
        if max_delta_rev <= 0:
            return None
        delta = np.random.rand() * max_delta_rev
        h_new = h.copy()
        h_new[j] -= delta
        h_new[i] += delta
    return h_new

import numpy as np
import time
import math
import random

def compute_c5(h):
    """Return the C5 value for a given step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / n
    return max_corr * dx

def repair(h, target):
    """Clip to [0,1] and adjust the sum to exactly `target`."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    diff = target - total
    if abs(diff) > 1e-12:
        if diff > 0:
            # need to increase some entries
            room = 1.0 - h
            capacity = room.sum()
            if capacity > 0:
                h += diff * (room / capacity)
        else:
            # need to decrease some entries
            room = h
            capacity = room.sum()
            if capacity > 0:
                h += diff * (room / capacity)   # diff is negative
        h = np.clip(h, 0.0, 1.0)
    return h

def transfer_move(h):
    """Move a random amount of mass from one coordinate to another, preserving the sum."""
    n = h.size
    i, j = np.random.randint(0, n, size=2)
    while i == j:
        j = np.random.randint(0, n)
    if np.random.rand() < 0.5:
        # i -> j
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0:
            return None
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:
        # j -> i
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0:
            return None
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[j] -= delta
        h_new[i] += delta
    return h_new

def gradient_move(h):
    """Transfer mass in a direction that decreases the current worst overlap."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    # indices in full mode that achieve the maximum (within tolerance)
    tol = 1e-12
    idxs = np.where(corr >= max_c - tol)[0]
    shifts = idxs - (n - 1)          # convert to shift values (k)
    grad = np.zeros(n, dtype=np.float64)

    for k in shifts:
        if k >= 0:
            # grad[i] += 1 - h[i+k] for i = 0..n-1-k
            grad[:n - k] += (1.0 - h[k:])
            # grad[i] -= h[i-k] for i = k..n-1
            grad[k:] -= h[:n - k]
        else:
            kk = -k
            # grad[i] += 1 - h[i+k] for i = kk..n-1
            grad[kk:] += (1.0 - h[:n - kk])
            # grad[i] -= h[i-k] for i = 0..n-1-kk
            grad[:n - kk] -= h[kk:]

    # We wish to move mass from indices with positive gradient (decrease them)
    # to indices with negative gradient (increase them).
    pos_idx = np.where(grad > 1e-9)[0]
    neg_idx = np.where(grad < -1e-9)[0]
    if pos_idx.size == 0 or neg_idx.size == 0:
        return None
    i = np.random.choice(pos_idx)
    j = np.random.choice(neg_idx)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return None
    delta = np.random.rand() * max_delta
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new

def perturb_move(h, target):
    """Add a small Gaussian perturbation to all entries and repair."""
    n = h.size
    noise = np.random.normal(scale=0.05, size=n)
    h_new = h + noise
    h_new = repair(h_new, target)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] with integral 1 that minimizes C5."""
    np.random.seed(seed)
    random.seed(seed)

    # --- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
        target = n / 2.0
        h0 = repair(h0, target)
    else:
        n = 70                       # fallback size
        target = n / 2.0
        h0 = np.zeros(n, dtype=np.float64)
        # random binary start with exact half ones
        ones = int(target)
        idx = np.arange(n)
        np.random.shuffle(idx)
        h0[idx[:ones]] = 1.0

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    cur_h = h0.copy()
    cur_c = best_c

    # --- parameters for the search -----------------------------------------
    start_temp = 0.008
    temp = start_temp
    decay = 0.9996
    min_temp = 1e-7
    p_grad = 0.68          # probability to use gradient‑guided move
    p_random = 0.25        # probability to use a plain transfer move
    p_perturb = 0.04       # probability to add a small random perturbation
    p_invert = 0.03        # probability to flip the whole function

    start_time = time.time()
    it = 0

    while True:
        if time.time() - start_time > budget_s * 0.97:
            break
        it += 1

        r = np.random.rand()
        if r < p_grad:
            cand = gradient_move(cur_h)
            if cand is None:
                continue
        elif r < p_grad + p_random:
            cand = transfer_move(cur_h)
            if cand is None:
                continue
        elif r < p_grad + p_random + p_perturb:
            cand = perturb_move(cur_h, target)
        else:   # invert move
            cand = 1.0 - cur_h
            cand = repair(cand, target)

        cand_c = compute_c5(cand)

        if cand_c < cur_c:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(cand_c - cur_c) / max(temp, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_h = cand
            cur_c = cand_c

        if cand_c < best_c:
            best_h = cand.copy()
            best_c = cand_c

        # temperature schedule
        temp = max(temp * decay, min_temp)

        # occasional shake‑up to escape stagnation
        if it % 25000 == 0:
            # re‑inject a small random perturbation around the best found
            cur_h = perturb_move(best_h, target)
            cur_c = compute_c5(cur_h)

    # Final repair and evaluation
    best_h = repair(best_h, target)
    best_c = compute_c5(best_h)

    return (best_h.tolist(), float(best_c), int(n))
