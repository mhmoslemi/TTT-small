def simulated_annealing(a0, time_limit, rng):
    best = a0.copy()
    best_val, _ = compute_max_O(best)
    cur = a0.copy()
    cur_val = best_val
    n = len(a0)
    start = time.time()
    T0 = 1.0
    while time.time() - start < time_limit:
        # temperature schedule (linear decay)
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / time_limit)
        # propose a move: choose i with cur[i] > 0, j with cur[j] < 1
        # We also ensure i != j.
        # Choose random indices:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # propose delta in [-d, d]
        max_delta = min(cur[i], 1 - cur[j])
        if max_delta <= 0:
            continue
        # Choose delta = max_delta * rng.random() maybe limited by temperature
        delta = max_delta * rng.random() * min(1, T)  # scale by T
        # create new vector
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        # compute new max_O
        new_val, _ = compute_max_O(new)
        d_val = new_val - cur_val
        if d_val < 0 or rng.random() < np.exp(-d_val / (T + 1e-12)):
            cur = new
            cur_val = new_val
            if cur_val < best_val:
                best = cur.copy()
                best_val = cur_val
    return best, best_val

def compute_max_O(a):
    n = a.shape[0]
    total = a.sum()
    # prefix sums
    tail = np.concatenate(([0], np.cumsum(a[::-1])))  # length n+1, tail[L] = sum_{i=n-L}^{n-1} a[i]
    prefix = total - tail[:n]  # for L from 0 to n-1.
    # autocorrelation
    autocorr_full = np.correlate(a, a, mode='full')
    center = n - 1
    autocorr_shifts = autocorr_full[center:center + n]  # for L=0..n-1
    O = prefix - autocorr_shifts
    max_O = np.max(O)
    return max_O, O

def compute_max_O(a):
    n = a.shape[0]
    total = a.sum()
    # tail sum: sum of last L entries
    tail = np.concatenate(([0.0], np.cumsum(a[::-1])))
    prefix = total - tail[:n]  # shape n
    # autocorrelation via np.correlate
    ac_full = np.correlate(a, a, mode='full')
    center = n - 1
    autocorr_shifts = ac_full[center:center + n]
    O = prefix - autocorr_shifts
    max_O = O.max()
    return max_O, O

def simulated_annealing(a0, time_limit, rng):
    a = a0.copy()
    n = a.shape[0]
    S = a.sum()
    max_O_cur, _ = compute_max_O(a)
    best_a = a.copy()
    best_val = max_O_cur
    start = time.time()
    while time.time() - start < time_limit:
        # temperature schedule decreasing linearly
        elapsed = time.time() - start
        T = 1.0 - elapsed / time_limit  # from 1 to 0
        # random i, j
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        # compute max delta according to bounds
        max_delta = min(a[i], 1.0 - a[j])
        if max_delta <= 0.0:
            # try opposite direction
            max_delta = min(a[j], 1.0 - a[i])
            if max_delta <= 0.0:
                continue
            i, j = j, i
        # choose delta scaling, ensure >0
        # use random fraction * T to reduce step as time advances
        delta = max_delta * rng.random() * (0.5 + 0.5 * T)  # T influences magnitude
        # apply move
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        max_O_new, _ = compute_max_O(a_new)
        d = max_O_new - max_O_cur
        if d <= 0 or rng.random() < math.exp(-d / (T + 1e-12)):
            a = a_new
            max_O_cur = max_O_new
            if max_O_cur < best_val:
                best_val = max_O_cur
                best_a = a.copy()
        # Occasionally do random restart: not needed.
    return best_a, best_val

import time, math, numpy as np

def _compute_max_O(a):
    """Return max_O and the full O array for vector a."""
    n = a.shape[0]
    total = a.sum()
    # tail sums: sum of last L entries for L = 0..n
    tail = np.concatenate(([0.0], np.cumsum(a[::-1])))
    # prefix sums for each shift L (0..n-1)
    pref = total - tail[:n]                     # pref[L] = sum_{i=0}^{n-1-L} a_i
    # autocorrelation a·a (full)
    ac_full = np.correlate(a, a, mode='full')
    center = n - 1
    autoc = ac_full[center:center + n]          # autoc[L] = Σ a_i a_{i+L}
    O = pref - autoc
    return O.max(), O


def _project_box_sum(v, target):
    """Project vector v onto the box [0,1]^n while keeping sum(v)=target."""
    lo = -np.max(v) - 2.0
    hi = np.max(v) + 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _anneal(a0, time_limit, rng):
    """Simulated annealing with directed moves."""
    a = a0.copy()
    n = a.shape[0]

        elapsed = time.time() - start
        T = 1.0 - elapsed / time_limit          # linear schedule
        max_O, O_arr = _compute_max_O(a)
        L = int(np.argmax(O_arr))               # worst shift
        # ---- directed move ----
        if rng.random() < 0.8:
            prefix_mask = (np.arange(n) <= n - 1 - L) & (a > 1e-12)
            tail_mask = (np.arange(n) > n - 1 - L) & (a < 1.0 - 1e-12)
            if prefix_mask.any() and tail_mask.any():
                i = int(rng.choice(np.where(prefix_mask)[0]))
                j = int(rng.choice(np.where(tail_mask)[0]))
                delta_max = min(a[i], 1.0 - a[j])
                if delta_max > 0.0:

                    new_val, _ = _compute_max_O(new_a)
                    d = new_val - max_O
                    if d <= 0.0 or rng.random() < math.exp(-d / (T + 1e-12)):
                        a = new_a

                    continue
        # ---- random transfer move ----

def _optimize_for_n(n, init_a, budget, rng):
    """Run several short annealing runs within the given budget."""
    best_a = None
    best_val = float('inf')
    start = time.time()
    remaining = budget
    while time.time() - start < budget:
        # allocate a short chunk (max 8 s) for each run
        chunk = min(8.0, budget - (time.time() - start))
        # start from a fresh random vector each chunk (except first)
        if best_a is None:
            a0 = init_a.copy()
        else:
            # random perturb of best so far
            a0 = best_a.copy()
            # small random shuffle of a few entries
            for _ in range(3):
                i, j = rng.integers(n), rng.integers(n)
                if i == j:

                d = min(a0[i], 1.0 - a0[j]) * rng.random() * 0.1
                a0[i] -= d
                a0[j] += d
        a_opt, val_opt = _anneal(a0, chunk, rng)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
    return best_a, best_val
