import time
import random
import math
import numpy as np

def _compute_score(arr, dx):
    """Return (C5 float, max overlap count integer)."""
    # correlation of h with (1-h) using zero padding (full mode)
    corr = np.correlate(arr, 1 - arr, mode='full')
    max_cnt = int(corr.max())
    return max_cnt * dx, max_cnt

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # ----- initial solution -------------------------------------------------
    try:
        init = np.asarray(initial_h_values, dtype=float)
        N = init.size
    except NameError:
        init = None
        N = 80

    target_sum = N // 2                     # sum of h must be N/2
    dx = 2.0 / N

    if init is None:
        # start from a random balanced binary vector
        h = np.zeros(N, dtype=int)
        h[:target_sum] = 1
        np.random.shuffle(h)
    else:
        # make a binary version, keep the same length
        h = (init > 0.5).astype(int)
        cur = int(h.sum())
        if cur != target_sum:
            # fix the number of ones exactly
            diff = target_sum - cur
            if diff > 0:
                zeros = np.where(h == 0)[0]
                pick = np.random.choice(zeros, diff, replace=False)
                h[pick] = 1
            else:
                ones = np.where(h == 1)[0]
                pick = np.random.choice(ones, -diff, replace=False)
                h[pick] = 0

    # ----- simulated annealing / local search -------------------------------
    best_h = h.copy()
    best_score, best_cnt = _compute_score(best_h, dx)
    current_h = h.copy()
    current_score, current_cnt = best_score, best_cnt

    start = time.time()
    T0 = 1.0                      # initial temperature
    target_cnt = int(math.floor(0.38085 / dx))   # 15 for the current record
    it = 0
    no_imp_iters = 0

    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break

        # temperature linearly cools to 0
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)

        # compute the worst shift (largest overlap)
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_cnt = int(corr.max())
        max_idx = int(corr.argmax())
        k_shift = max_idx - (N - 1)          # signed shift producing max

        # choose i (a 1) preferably from a mismatched position for the worst shift
        use_targeted = (random.random() < 0.8) and (max_cnt > target_cnt)
        i = None
        if use_targeted:
            if k_shift >= 0:
                seg = current_h[:N - k_shift]
                sh = current_h[k_shift:]
                mask = (seg == 1) & (sh == 0)
                mism = np.flatnonzero(mask)
                if mism.size:
                    i = int(mism[random.randrange(mism.size)])
            else:   # negative shift
                kk = -k_shift
                seg = current_h[kk:]
                sh = current_h[:N - kk]

                mism = np.flatnonzero(mask) + kk
                if mism.size:

        if i is None:
            # fallback: random 1
            ones = np.where(current_h == 1)[0]
            i = int(ones[random.randrange(ones.size)])

        # pick a random 0 to swap with
        zeros = np.where(current_h == 0)[0]
        j = int(zeros[random.randrange(zeros.size)])

        # perform the swap
        current_h[i] = 0
        current_h[j] = 1

    return (best_h.astype(float), best_score, N)
