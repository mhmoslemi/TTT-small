    g = np.zeros_like(h)

    if s >= 0:

        g[:n-s] += h[s:]

def project_onto_simplex_box(v, target):
    lo = -10.0
    hi = 10.0
    for _ in range(50):
        mid = (lo + hi) / 2
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # adjust residual
    residual = target - w.sum()
    # distribute to non-capped entries
    if abs(residual) > 1e-12:
        idx = np.where((w > 0) & (w < 1))[0]
        if len(idx) > 0:
            w[idx] += residual / len(idx)
            w = np.clip(w, 0.0, 1.0)
    return w

def compute_c5(h):

    return np.max(corr)

def gradient_of_shift(h, s):
    n = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        g[:n-s] += h[s:]
        g[s:] += h[:n-s]
    else:
        t = -s
        g[t:] += h[:n-t]
        g[:n-t] += h[t:]
    return g

def project_to_sum_bounds(v, target):
    # binary search for lambda
    lo = -5.0
    hi = 5.0
    for _ in range(60):
        mid = (lo + hi) / 2
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    # adjust residual
    diff = target - w.sum()
    if abs(diff) > 1e-8:
        mask = (w > 0) & (w < 1)
        if np.any(mask):
            w[mask] += diff / np.sum(mask)
            w = np.clip(w, 0.0, 1.0)
    return w

def improve_h(h_initial, time_remaining, target_iterations=20000):
    n = len(h_initial)
    target_sum = n/2.0
    h = h_initial.copy()
    best_h = h.copy()

    best_minconv = None # record min conv

    # compute initial minconv
    conv = np.correlate(h, h, mode='full')
    best_minconv = conv.min()
    start = time.time()
    # step size schedule
    step = 0.1
    it = 0
    while time.time() - start < time_remaining and it < 20000:
        conv = np.correlate(h, h, mode='full')
        min_val = conv.min()
        # gather indices with min
        idxs = np.where(conv == min_val)[0]
        # compute gradient
        grad = np.zeros_like(h)
        for idx in idxs:
            s = idx - (n - 1)
            grad += gradient_of_shift(h, s)
        # if gradient zero, random perturb
        if np.linalg.norm(grad) == 0:
            grad = np.random.randn(n)
        # normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm > 0:
            grad = grad / g_norm
        # step size adapt: maybe step = 0.5 * (target_sum - sum)? but we use constant
        v = h + step * grad
        h_new = project_to_sum_bounds(v, target_sum)
        # compute new min conv
        conv_new = np.correlate(h_new, h_new, mode='full')
        min_val_new = conv_new.min()
        # compute C5
        c5_new = compute_c5(h_new)
        # Accept if min conv increased (or C5 decreased)

import numpy as np, time, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_shift(h, s):
    n = h.shape[0]
    g = np.zeros_like(h)
    if s >= 0:
        g[:n - s] += h[s:]
        g[s:] += h[:n - s]
    else:
        t = -s
        g[t:] += h[:n - t]
        g[:n - t] += h[t:]
    return g

def project_onto_sum_box(v, target):
    # binary search on lambda
    lo = -5.0
    hi = 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust residual if needed
    resid = target - w.sum()
    if abs(resid) > 1e-10:
        mask = (w > 0) & (w < 1)
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            w[mask] += resid / cnt
            w = np.clip(w, 0.0, 1.0)
    return w

def local_opt_continuous(h_start, time_budget):
    n = h_start.shape[0]
    target = n / 2.0
    h = np.clip(h_start, 0.0, 1.0)
    h = project_onto_sum_box(h, target)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.2
    t0 = time.time()
    while time.time() - t0 < time_budget:
        conv = np.correlate(h, h, mode='full')
        min_val = conv.min()
        idxs = np.where(conv == min_val)[0]
        grad = np.zeros_like(h)
        for idx in idxs:
            s = idx - (n - 1)
            grad += gradient_shift(h, s)
        norm = np.linalg.norm(grad)
        if norm == 0.0:

            grad = np.random.randn(n)
            norm = np.linalg.norm(grad)

        if c5_new < best_c5:

import time
import numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------

def compute_c5(h):
    """Return the Erdős overlap constant C5 for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate with mode='full' computes the sum of products over the
    # overlapping part of the two vectors.
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def gradient_shift(h, s):
    """Gradient of the self‑correlation sum_i h_i h_{i+s} with respect to h."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if s >= 0:
        g[: n - s] += h[s:]
        g[s:] += h[: n - s]
    else:
        t = -s
        g[t:] += h[: n - t]
        g[: n - t] += h[t:]
    return g


def project_onto_sum_box(v, target):
    """
    Euclidean projection of v onto the set
        { x | 0 <= x_i <= 1, sum x_i = target }.
    Implemented by a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction (numerical)
    resid = target - w.sum()
    if abs(resid) > 1e-10:
        mask = (w > 0.0) & (w < 1.0)
        cnt = int(mask.sum())
        if cnt:
            w[mask] += resid / cnt
            w = np.clip(w, 0.0, 1.0)
    return w


def local_opt_continuous(h_start, time_budget):
    """
    Gradient‑ascent style improvement on the *minimum* self‑correlation.
    Accept a step only if it decreases the actual C5.
    """
    n = h_start.shape[0]
    target = n / 2.0
    h = np.clip(h_start, 0.0, 1.0)
    h = project_onto_sum_box(h, target)

    curr_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5
    step = 0.2
    t0 = time.time()

    while time.time() - t0 < time_budget:
        # find the shift(s) with minimal self‑correlation
        conv = np.correlate(h, h, mode="full")
        min_val = conv.min()
        idxs = np.where(conv == min_val)[0]

        # sub‑gradient (sum of gradients of the active shifts)
        grad = np.zeros_like(h)
        for idx in idxs:
            s = idx - (n - 1)
            grad += gradient_shift(h, s)

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            grad = np.random.randn(n)
            norm = np.linalg.norm(grad)
        grad /= norm

        # propose a step and project back onto the feasible set
        v = h + step * grad
        h_new = project_onto_sum_box(v, target)

        c5_new = compute_c5(h_new)

        if c5_new < curr_c5 - 1e-12:            # accepted improvement
            h = h_new
            curr_c5 = c5_new
            step *= 1.05
        else:                                   # reject, shrink step
            step *= 0.5
            if step < 1e-6:
                step = 0.2

        # keep the best seen so far
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    return best_h, best_c5


def binary_from_continuous(h_cont):
    """Return a binary vector (0/1) with exactly n/2 ones by thresholding."""
    n = h_cont.shape[0]
    m = n // 2
    idx = np.argpartition(-h_cont, m)[:m]
    bin_vec = np.zeros(n, dtype=np.float64)
    bin_vec[idx] = 1.0
    return bin_vec


def local_opt_binary(h_start, time_budget):
    """
    Simple hill‑climbing on binary vectors: swap a 1 with a 0 if it lowers C5.
    """
    n = h_start.shape[0]
    h = h_start.copy()
    curr_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = curr_c5

    # maintain lists of positions for quick random sampling
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])

    t0 = time.time()
    while time.time() - t0 < time_budget:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # perform swap
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)

        if new_c5 < curr_c5 - 1e-12:
            # keep the improvement
            curr_c5 = new_c5
            # update index lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with ∫h=1 and 0≤h≤1
    that yields a low Erdős constant C5.
    Returns (list_of_h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    # --------------------------------------------------------
    # Choose discretisation
    # --------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
        # ensure feasibility
        h0 = project_onto_sum_box(h0, n / 2.0)
    else:
        n = 80                     # a modest size that works quickly
        h0 = np.full(n, 0.5, dtype=np.float64)   # uniform start (feasible)

    target_sum = n / 2.0
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()
    deadline = start_time + budget_s - 0.1    # tiny safety margin

    # --------------------------------------------------------
    # Repeated random restarts with local optimisation
    # --------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 2.0:
            break

        # ----- start from a fresh random point (continuous) -----
        rand_vec = np.random.rand(n)
        h_start = project_onto_sum_box(rand_vec, target_sum)

        # time slice for continuous optimisation
        slice_cont = min(1.5, remaining)
        h_opt, c5_opt = local_opt_continuous(h_start, slice_cont)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

        # ----- try a binary refinement of the continuous result -----
        h_bin = binary_from_continuous(h_opt)
        remaining = deadline - time.time()
        if remaining < 0.5:
            break
        slice_bin = min(0.5, remaining)
        h_bin_opt, c5_bin_opt = local_opt_binary(h_bin, slice_bin)

        if c5_bin_opt < best_c5:
            best_c5 = c5_bin_opt
            best_h = h_bin_opt.copy()

    # final safety projection (ensures exact constraints)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_onto_sum_box(best_h, target_sum)

    return best_h.tolist(), float(best_c5), n
