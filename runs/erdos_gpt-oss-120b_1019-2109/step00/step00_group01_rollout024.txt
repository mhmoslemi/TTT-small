    best_h = h.copy()

    start = time.time()

    max_no_improve = 3000

    while time.time() - start < time_budget * 0.95:
        # compute self correlation
        O = np.correlate(h, h, mode='full')
        dx = 2.0 / N
        C = 1 - O * dx
        # compute max (worst)
        max_idx = np.argmax(C)
        max_c = C[max_idx]
        # if current best > max_c? Actually we keep best, but also maybe no need.
        if max_c < best_val - 1e-12:
            best_val = max_c
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        # Determine shift
        tau = max_idx - (N - 1)
        # compute shifted h
        shift_idx = np.arange(N) + tau
        mask = (shift_idx >= 0) & (shift_idx < N)

def project_to_simplex_box(v, target):
    lo = np.min(v) - 1.0 # maybe lower bound for λ
    hi = np.max(v) # upper bound for λ
    # Because clip(v_i - λ,0,1): def sum_func(λ) = Σ clip(v_i - λ,0,1). As λ increases, each term decreases, sum decreases.
    # We'll binary search up to tolerance 1e-10.
    for _ in range(60):
        mid = (lo + hi)/2
        clipped = np.clip(v - mid, 0.0, 1.0)
        s = clipped.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo+hi)/2, 0.0, 1.0)
    return h

   import time
   target_sum = N/2
   np.random.seed(0)
   v = np.random.rand(N)
   h = project_to_simplex_box(v, target_sum)
   best_h = h.copy()
   best_val = evaluate(h)
   start = time.time()
   iteration = 0
   step = 0.1  # step size for gradient
   while time.time() - start < time_budget * 0.95:
       O = np.correlate(h, h, mode='full')
       dx = 2.0 / N
       C = 1 - O * dx
       max_c = C.max()
       if max_c < best_val:
           best_val = max_c
           best_h = h.copy()
       # target improvement
       # identify tau with max_c
       tau_idx = np.argmax(C)
       tau = tau_idx - (N-1)
       # compute gradient wrt h for this tau
       grad = np.zeros_like(h)
       # shift forward
       if tau >= 0:
            grad[:N-tau] += h[tau:]
            grad[tau:] += h[:N-tau]
       else:
            t = -tau
            grad[t:] += h[:N-t]
            grad[:N-t] += h[t:]
       # gradient points to increase overlap; we ascend.
       # Since we want to reduce max_c, which is 1 - O*dx, equivalently maximize O at tau. So ascend.

       iteration += 1

import time
import numpy as np

# -------------------- helper functions -------------------- #

def _project_to_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the set:
        {x | 0 <= x_i <= 1, sum x_i = target}
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0          # low enough => all clipped to 1
    hi = np.max(v)                # high enough => all clipped to 0
    for _ in range(60):           # sufficient for double precision
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _evaluate_c5(h: np.ndarray) -> float:
    """
    Compute the Erdős C5 bound for a step function represented by the array h.
    """
    dx = 2.0 / h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx


def _worst_shift(h: np.ndarray):
    """
    Return the shift (as integer offset) that gives the maximal C5 value,
    together with the current C5 value and the whole C5 vector.
    """
    N = h.shape[0]
    O = np.correlate(h, h, mode='full')          # overlap counts
    dx = 2.0 / N
    C = 1.0 - O * dx
    idx = int(C.argmax())
    tau = idx - (N - 1)                           # convert index to shift
    return tau, float(C.max()), C


def _gradient_for_shift(h: np.ndarray, tau: int) -> np.ndarray:
    """
    Gradient of the overlap O_tau = sum_i h_i * h_{i+tau}
    w.r.t. h (the direction that increases O_tau).
    """
    N = h.shape[0]
    g = np.zeros_like(h)
    if tau >= 0:
        g[:N - tau] += h[tau:]
        g[tau:] += h[:N - tau]
    else:
        t = -tau
        g[t:] += h[:N - t]
        g[:N - t] += h[t:]
    return g


def _initial_triangle(N: int) -> np.ndarray:
    """
    Piecewise linear “tent’’ function: h(x)=max(0,1-|x-1|) sampled uniformly.
    """
    dx = 2.0 / N
    xs = (np.arange(N) + 0.5) * dx          # mid‑point sampling
    h = 1.0 - np.abs(xs - 1.0)
    h[h < 0.0] = 0.0
    return h


# -------------------- main optimisation routine -------------------- #

def _optimise(N: int, time_limit: float, seed: int):
    """
    Optimise a step function with N samples within the allocated time.
    Returns best (h, C5) found.
    """
    rng = np.random.default_rng(seed)
    target_sum = N / 2.0

    # start from the tent shape, then project to the feasible set
    h = _project_to_bounds_and_sum(_initial_triangle(N), target_sum)

    best_h = h.copy()
    best_val = _evaluate_c5(best_h)

    start = time.time()
    step = 0.1               # initial step size for gradient ascent
    no_improve = 0

    while time.time() - start < time_limit:
        # Identify the worst shift (largest C5)
        tau, cur_val, _ = _worst_shift(h)

        # keep track of the best solution seen so far
        if cur_val + 1e-12 < best_val:
            best_val = cur_val
            best_h = h.copy()
            no_improve = 0
            if best_val <= 0.3808:
                break
        else:
            no_improve += 1

        # Gradient step aimed at increasing overlap for this tau
        grad = _gradient_for_shift(h, tau)

        # Propose a new point
        v_new = h + step * grad
        h_new = _project_to_bounds_and_sum(v_new, target_sum)
        new_val = _evaluate_c5(h_new)

        # Accept if we improve the worst‑case value; otherwise adapt step size
        if new_val + 1e-12 < cur_val:
            h = h_new
            step = min(step * 1.1, 1.0)          # grow a little
        else:
            # try a small random jitter before giving up on this direction
            step *= 0.5
            if step < 1e-5 or no_improve > 5000:
                # restart from a fresh random point to escape possible local minima
                rand_vec = rng.random(N)
                h = _project_to_bounds_and_sum(rand_vec, target_sum)
                step = 0.1
                no_improve = 0

    return best_h, best_val


# -------------------- required entry point -------------------- #

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Find a step function h on [0,2] that gives a low Erdős C5 bound.
    Returns (list_of_h_values, c5_bound, n_points).
    """
    # Determine a reasonable discretisation size.
    # We want an even number of points (so the integral constraint is integer‑feasible).
    # Use at least 120 samples; if an initial vector is supplied, honour its size parity.
    try:
        init = initial_h_values          # supplied by the harness (may be None)
        n0 = len(init)
    except NameError:
        init = None
        n0 = 0

    N = max(120, n0)
    if N % 2 == 1:
        N += 1

    # Allocate almost the whole budget to the optimiser (keep a tiny margin for safety)
    time_limit = budget_s * 0.95

    # Run optimiser
    best_h, best_c5 = _optimise(N, time_limit, seed)

    # Return plain Python list for compatibility with the harness
    return (best_h.tolist(), float(best_c5), int(N))
