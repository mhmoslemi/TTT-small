def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math, random
    start = time.time()
    np.random.seed(seed)
    # use initial points if available to set initial guess
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        N = len(init)
        # ensure sum constraint satisfied; if not, project
        init = project_feasible(init, N/2)
        best_h = init.copy()

    else:
        N = 84  # default even number
        best_h = None
        best_c5 = np.inf

    # Maintain best solution and iterate
    while time.time() - start < budget_s - 5:
        # random restart
        h = np.random.rand(N)
        # project onto feasible set
        h = project_feasible(h, N/2)

        # perform gradient descent for some iterations
        lr = 0.5
        beta = 1.0

            inner = inner_products(h)
            f = N/2 - inner
            # exclude shift 0 (index N-1)

            exps = np.exp(beta * (f - max_f))

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def inner_products(h: np.ndarray) -> np.ndarray:
    """Full auto‑correlation of h with itself (length 2N‑1)."""
    return np.correlate(h, h, mode="full")


def project_feasible(x: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of x onto the set
        { y | 0 ≤ y_i ≤ 1 ,  sum(y) = target_sum }.
    Simple water‑filling scheme that converges quickly for the sizes we use.
    """
    # Clip to box
    y = np.clip(x, 0.0, 1.0)

    # Iteratively adjust the free variables to meet the sum constraint
    for _ in range(100):
        cur = y.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (y > 0.0) & (y < 1.0)
        nfree = int(free.sum())
        if nfree == 0:               # cannot move any free variable – stop
            break
        delta = diff / nfree
        y[free] += delta
        y = np.clip(y, 0.0, 1.0)

    # Final tiny correction if needed
    cur = y.sum()
    if abs(cur - target_sum) > 1e-8:
        diff = target_sum - cur
        # distribute the remaining tiny error greedily
        for i in range(len(y)):
            if diff == 0.0:
                break
            if diff > 0.0 and y[i] < 1.0:
                step = min(diff, 1.0 - y[i])
                y[i] += step
                diff -= step
            elif diff < 0.0 and y[i] > 0.0:
                step = max(diff, -y[i])
                y[i] += step
                diff -= step
    return y


def smooth_max(values: np.ndarray, beta: float) -> tuple:
    """
    Returns a smooth approximation of max(values) together with the
    softmax weights (so that the gradient can be formed as a weighted sum).
    """
    vmax = values.max()
    scaled = beta * (values - vmax)
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    smooth = vmax + (1.0 / beta) * np.log(exp_scaled.sum())
    return smooth, w

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points).

    h_values : numpy array of length n_points containing the step‑function
               values (in [0,1]).
    c5_bound : the resulting upper bound C₅ obtained by the provided evaluator.
    n_points : number of discretisation points (the length of h_values).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and starting point
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        N = len(h0)
        # Ensure feasibility (the parent construction should already be feasible,
        # but we safeguard against rounding issues)
        h0 = project_feasible(h0, N / 2.0)
        best_h = h0.copy()
        best_c5 = float(evaluate_erdos_solution(best_h))
    else:
        # No parent solution – start with a modest size.
        N = 84                     # even, allows sum = N/2 integer
        best_h = None
        best_c5 = np.inf

    # ------------------------------------------------------------------
    # Parameters for the local optimiser
    # ------------------------------------------------------------------
    target_sum = N / 2.0
    max_iters_per_restart = 1500      # inner gradient steps
    beta_start = 1.0
    beta_factor = 1.5
    beta_max = 50.0
    lr_start = 0.7
    lr_decay = 0.75

    # ------------------------------------------------------------------
    # Main optimisation loop – many short restarts until budget expires
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 4.0:
        # ----- random feasible start ------------------------------------
        h = rng.random(N)
        h = project_feasible(h, target_sum)

        # optimisation hyper‑parameters for this restart
        lr = lr_start
        beta = beta_start

        # ----- gradient descent on smooth max --------------------------
        for it in range(max_iters_per_restart):
            # auto‑correlation (inner products for all shifts)
            inner = inner_products(h)               # length 2N‑1
            f_vals = target_sum - inner              # f(s) = N/2 – ⟨h,shift·h⟩

            smooth, w = smooth_max(f_vals, beta)

            # gradient of smooth‑max: ∇ = 1 – Σ_s w_s (h_{i‑s}+h_{i+s})
            # first term (the constant 1) is the same for all i

            grad = np.ones(N) - conv_fwd - conv_rev

                beta = min(beta * beta_factor, beta_max)
