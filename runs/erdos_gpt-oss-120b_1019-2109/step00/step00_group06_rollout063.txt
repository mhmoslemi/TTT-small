def compute_c5(h, dx):
    cur = np.correlate(h, 1 - h, mode='full')
    return np.max(cur) * dx

def local_search(h_init, time_limit):
    n = len(h_init)
    dx = 2.0 / n
    h = np.copy(h_init)
    best_h = np.copy(h)
    best_c5 = compute_c5(h, dx)
    start = time.time()
    # parameters
    max_iter = 100000
    temperature = 0.01
    decay = 0.9995
    min_temp = 1e-6
    step_size = 0.05  # relative delta fraction of 1
    while time.time() - start < time_limit - 0.1:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        # ensure we can transfer both directions; we may choose direction based on random.
        # Determine direction
        if np.random.rand() < 0.5:
            src, tgt = i, j
        else:
            src, tgt = j, i
        # compute feasible delta
        max_delta = min(step_size, h[src])  # can't decrease below 0
        max_delta = min(max_delta, 1 - h[tgt])  # can't exceed 1
        if max_delta <= 1e-12:
            continue
        delta = np.random.rand() * max_delta
        # transfer
        h[src] -= delta
        h[tgt] += delta
        # evaluate
        c5 = compute_c5(h, dx)
        # acceptance
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = np.copy(h)
        # Simulated annealing acceptance
        if c5 < compute_c5(best_h, dx) or np.exp((best_c5 - c5) / max(temperature, 1e-12)) > np.random.rand():
            # accept new h (it already applied)
            pass
        else:
            # revert move
            h[src] += delta
            h[tgt] -= delta
        # update temperature
        temperature = max(min_temp, temperature * decay)
        # optionally reduce step_size gradually
        if temperature < 1e-4:
            step_size = 0.02
    return best_h, best_c5

def _c5_of(h):
    """Compute the C5 value for a step function h (array of values)."""
    dx = 2.0 / h.size
    # correlation h with (1-h) using full mode (zero padding outside [0,2])
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _random_transfer(h, step_scale):

    n = h.size
    i, j = np.random.randint(0, n, size=2)

        return False
    # we will move a small amount from i to j (or opposite) while staying in [0,1]
    if np.random.rand() < 0.5:
        # i -> j
        max_move = min(h[i], 1.0 - h[j])

        h[i] += delta
    return True

def _targeted_transfer(h):

    n = h.size

        np.clip(best_h, 0.0, 1.0, out=best_h)

    # Return plain Python list for the caller, together with the bound and length
    return (best_h.tolist(), float(final_c5), n)
