def project_box_sum(v, S, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    low = -10.0
    high = 10.0

        s = w.sum()

        if s > S:
            low = lam
        else:
            high = lam
    return w

import numpy as np, time, math

def project_box_sum(v, target_sum, lower=0.0, upper=1.0, tol=1e-9, max_iter=60):
    # Find lambda such that sum_i clip(v_i - lambda, lower, upper) = target_sum
    lo = np.min(v) - upper
    hi = np.max(v) - lower
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        w = np.clip(v - lam, lower, upper)
        s = w.sum()
        if abs(s - target_sum) <= tol:
            return w
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return w

def compute_c5(h):
    n = h.shape[0]
    # correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def compute_gradient(h, k):
    n = h.shape[0]
    grad = np.ones_like(h)
    if k >= 0:
        if n - k > 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
    else:
        kk = -k
        if n - kk > 0:
            grad[kk:] -= h[:n - kk]
            grad[:n - kk] -= h[kk:]
    return grad

def random_feasible(m, target_sum):
    v = np.random.rand(m)
    return project_box_sum(v, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    # Determine dimension
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        n = init.size
        target = n / 2.0
        h = project_box_sum(init, target)
    else:
        n = 200
        target = n / 2.0
        h = random_feasible(n, target)
    # initial evaluation
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # parameters
    step = 0.5
    decay = 0.9998
    iter_count = 0
    # Main loop
    while True:
        now = time.time()
        if now - start > budget_s - 2.0:
            break
        # compute correlation and worst lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx_max = np.argmax(corr)
        max_val = corr[idx_max] * (2.0 / n)
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        k = idx_max - (n - 1)
        # gradient descent step for worst lag
        grad = compute_gradient(h, k)
        h_new = project_box_sum(h - step * grad, target)
        # occasional random perturbation
        if np.random.rand() < 0.02:
            # random pair move
            i, j = np.random.choice(n, size=2, replace=False)
            low = max(-h[i], h[j] - 1.0)
            high = min(1.0 - h[i], h[j])
            if low < high:
                delta = np.random.uniform(low, high)
                h_new[i] += delta
                h_new[j] -= delta
                h_new = project_box_sum(h_new, target)
        # accept new state
        new_c5 = compute_c5(h_new)
        if new_c5 <= cur_c5 or np.random.rand() < math.exp(-(new_c5 - cur_c5) / max(step, 1e-8)):
            h = h_new
            cur_c5 = new_c5
        # decay step size
        step *= decay
        iter_count += 1
        # if stagnation, reinitialize
        if iter_count % 5000 == 0 and cur_c5 - best_c5 > 1e-4:
            # restart from a random point
            h = random_feasible(n, target)
            cur_c5 = compute_c5(h)
            step = 0.5
    return best_h, best_c5, n

import numpy as np
import time
import math

def _project_box_sum(v, target, lo=0.0, hi=1.0, tol=1e-9, max_iter=60):
    """Projection of v onto the box [lo,hi] with sum == target."""
    lam_low = np.min(v) - hi
    lam_high = np.max(v) - lo
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        w = np.clip(v - lam, lo, hi)
        s = w.sum()
        if abs(s - target) <= tol:
            return w
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    return w

def _c5_of(h):
    """Compute C5 for a step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def _gradient(h, k):
    """Gradient of the maximal overlap term w.r.t. h (for a given lag k)."""
    n = h.shape[0]
    g = np.ones_like(h)
    if k >= 0:
        if n - k > 0:
            g[:n - k] -= h[k:]
            g[k:]      -= h[:n - k]
    else:  # k < 0
        kk = -k
        if n - kk > 0:
            g[kk:]   -= h[:n - kk]
            g[:n-kk] -= h[kk:]
    return g

def _random_feasible(m, target):
    """Generate a random vector in [0,1]^m with prescribed sum = target."""
    v = np.random.rand(m)
    return _project_box_sum(v, target)

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function h on [0,2] that minimizes C5."""
    np.random.seed(seed)
    start_time = time.time()

    # --- initialise ---------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        target_sum = n / 2.0
        h = _project_box_sum(init, target_sum)
    else:
        n = 200                     # default resolution (kept <1k)
        target_sum = n / 2.0
        h = _random_feasible(n, target_sum)

    cur_val = _c5_of(h)
    best_h = h.copy()
    best_val = cur_val

    # --- parameters ---------------------------------------------------------
    step = 0.5                     # initial gradient step magnitude
    decay = 0.9998                 # exponential decay of step
    it = 0
    stagnation = 0

    # --- main optimisation loop --------------------------------------------
    while True:
        if time.time() - start_time > budget_s - 2.0:
            break

        # current worst lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argmax(corr)
        max_val = corr[idx] * (2.0 / n)
        if max_val < best_val:
            best_val = max_val
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1

        k = idx - (n - 1)

        # gradient descent step on the worst lag
        g = _gradient(h, k)
        h_new = _project_box_sum(h - step * g, target_sum)

        # occasional random pair perturbation (helps escape plateaus)
        if np.random.rand() < 0.02:
            i, j = np.random.choice(n, size=2, replace=False)
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if lo < hi:
                delta = np.random.uniform(lo, hi)
                h_new[i] += delta
                h_new[j] -= delta
                h_new = _project_box_sum(h_new, target_sum)

        new_val = _c5_of(h_new)

        # accept if improvement or with a tiny Metropolis probability
        if new_val <= cur_val or np.random.rand() < math.exp(-(new_val - cur_val) / max(step, 1e-8)):
            h = h_new
            cur_val = new_val

        # schedule
        step *= decay
        it += 1

        # if stuck for a while, restart from a fresh random point
        if stagnation > 5000:
            h = _random_feasible(n, target_sum)
            cur_val = _c5_of(h)
            step = 0.5
            stagnation = 0

    return best_h, best_val, n
