def propose_random_move(h, rng):

        j = rng.integers(n)
    hi = h[i]
    hj = h[j]

# -*- coding: utf-8 -*-
import time, math, numpy as np

def _c5_of(h: np.ndarray) -> float:
    """return C5 = max_k ∫ h(x)*(1-h(x+k)) dx for a discrete step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _min_corr(h: np.ndarray) -> float:
    """minimum autocorrelation ∫ h(x)h(x+k) dx (used internally)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, h, mode="full") * dx
    return float(np.min(corr))


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """produce a random vector in [0,1]^n with sum = n/2."""
    # start from uniform numbers, then scale to the required sum.
    h = rng.random(n)
    target = n / 2.0
    h = h * (target / h.sum())
    # after scaling some entries may be a tiny bit >1 because of rounding;
    # clip and fix the remaining small deviation.
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = rng.integers(n)
        h[idx] = min(1.0, max(0.0, h[idx] + diff))
    return h


def _propose_random_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """move a small amount of mass from one coordinate to another."""
    n = h.shape[0]
    i, j = rng.integers(n, size=2)
    while i == j:
        j = rng.integers(n)
    hi, hj = h[i], h[j]
    # allowed delta so that both stay in [0,1]
    delta_min = max(-hi, hj - 1.0)
    delta_max = min(1.0 - hi, hj)
    if delta_max <= delta_min:
        return None
    delta = rng.uniform(delta_min, delta_max)
    if abs(delta) < 1e-14:
        return None
    h2 = h.copy()
    h2[i] = hi + delta
    h2[j] = hj - delta
    return h2


def _propose_directed_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """increase the product on the worst shift and compensate elsewhere."""
    n = h.shape[0]
    # worst shift (minimum autocorrelation)
    corr = np.correlate(h, h, mode="full") * (2.0 / n)
    idx_min = int(np.argmin(corr))
    offset = idx_min - (n - 1)          # signed shift amount

    # indices that actually overlap for this shift
    if offset >= 0:
        i_idx = np.arange(0, n - offset)
        j_idx = i_idx + offset
    else:
        off = -offset
        i_idx = np.arange(off, n)
        j_idx = i_idx - off

    # pair (i,j) with the smallest product on that shift
    prod = h[i_idx] * h[j_idx]
    min_pair = int(np.argmin(prod))
    i = int(i_idx[min_pair])
    j = int(j_idx[min_pair])

    # two highest‑value positions not equal to i or j (to donate mass)
    high = np.argsort(-h)
    donors = []
    for v in high:
        if v != i and v != j:
            donors.append(v)
        if len(donors) == 2:
            break
    if len(donors) < 2:
        return None
    p, q = donors

    # maximal feasible delta respecting box constraints
    delta_up = min(1.0 - h[i], 1.0 - h[j])
    delta_down = min(h[p], h[q])
    delta_max = min(delta_up, delta_down)
    if delta_max <= 0.0:
        return None
    # take a modest fraction of the allowed move
    delta = rng.uniform(0.0, delta_max) * 0.5
    if delta < 1e-14:
        return None

    h2 = h.copy()
    h2[i] += delta
    h2[j] += delta
    h2[p] -= delta
    h2[q] -= delta
    return h2


def _hill_climb(start_h: np.ndarray, budget: float, start_time: float) -> tuple[np.ndarray, float]:
    """simple deterministic hill climbing (improvements only)."""
    rng = np.random.default_rng(0)          # deterministic moves
    h = start_h.copy()
    best_h = h.copy()
    best_val = _c5_of(h)
    n = len(h)

    it = 0
    while True:
        if time.time() - start_time > budget * 0.97:
            break
        it += 1

        # 70 % random swap, 30 % directed move
        if rng.random() < 0.7:
            cand = _propose_random_move(h, rng)
        else:

        if cand is None:
            continue

        val = _c5_of(cand)
        if val < best_val - 1e-12:
            best_val = val
            best_h = cand.copy()
            h = cand
        # else discard

        # occasional restart from current best to avoid stagnation
        if it % 4000 == 0:
            h = best_h.copy()
    return best_h, best_val

        h0 = np.clip(h0, 0.0, 1.0)
        diff = target - h0.sum()
        if abs(diff) > 1e-12:
            idx = rng.integers(n)
            h0[idx] = min(1.0, max(0.0, h0[idx] + diff))

    # quick evaluation of the supplied candidate
    best_h = h0
    best_val = _c5_of(best_h)

    # if the supplied candidate already beats the record, return it immediately
    RECORD = 0.38085
    if best_val <= RECORD:
        return best_h.tolist(), float(best_val), n

    # ----- search phase ---------------------------------------------------
    # allocate most of the remaining budget to hill climbing
    remaining_budget = budget_s - (time.time() - start)
    if remaining_budget > 15.0:
        # try a few restarts with different random seeds
        tries = 3
        for _ in range(tries):
            # random start preserving the integral
            rand_start = _random_feasible(n, rng)
            h_opt, val_opt = _hill_climb(rand_start, remaining_budget, start)
            if val_opt < best_val:
                best_val, best_h = val_opt, h_opt
            # shrink the time budget a bit between restarts
            remaining_budget = budget_s - (time.time() - start)
            if remaining_budget <= 5.0:
                break
        # finally polish the best found solution
        h_opt, val_opt = _hill_climb(best_h, remaining_budget, start)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt

    # final sanity check
    final_val = _c5_of(best_h)
    best_val = float(final_val)

    return best_h.tolist(), best_val, len(best_h)
