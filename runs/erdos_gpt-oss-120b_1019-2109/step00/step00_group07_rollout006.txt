Complete algorithm:

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
```

Now local search:

def random_swap_heuristic(h, dx, budget, start_time):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    n = len(h)
    # Generate list of indices of ones and zeros
    ones = np.where(best_h == 1)[0]
    zeros = np.where(best_h == 0)[0]
    while time.time() - start_time < budget:
        if len(ones)==0 or len(zeros)==0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # perform swap
        h[i], h[j] = 0.0, 1.0
        # evaluate
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            # update ones and zeros sets
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
        else:
            # revert swap
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5
```

def run(seed=42, budget_s=1100, **kwargs):
    import time, random, math
    import numpy as np

    # try cvxpy for an exact binary MILP solution
    try:
        import cvxpy as cp
    except Exception:
        cp = None

    rnd = np.random.RandomState(seed)
    start = time.time()

    # ----------------------------------------------------------------------
    # Get initial data (if provided by the harness)
    # ----------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)  # reference, not literal copy
    except Exception:
        init_h = None

    if init_h is None:
        n = 80
        # simple random binary start
        h0 = np.zeros(n)
        h0[rnd.choice(n, n // 2, replace=False)] = 1.0
    else:
        n = int(init_h.shape[0])
        h0 = init_h.copy()

    dx = 2.0 / n
    target_ones = n // 2                     # because sum(h)*dx must equal 1

    best_h = None
    best_c5 = float("inf")

    # ----------------------------------------------------------------------
    # 1) Try a binary integer programming formulation (exact optimum)
    # ----------------------------------------------------------------------
    if cp is not None:
        try:
            # binary vector x[i] ∈ {0,1}
            x = cp.Variable(n, boolean=True)
            t = cp.Variable()                 # the max over shifts (scaled by dx later)

            constraints = [cp.sum(x) == target_ones, t >= 0]

            # auxiliary product variables Z_{i,s} = x[i] * x[i+s]
            for s in range(1, n):
                length = n - s
                Zs = cp.Variable(length, boolean=True)
                # Zs ≤ x[i]   and   Zs ≤ x[i+s]
                constraints.append(Zs <= x[:length])
                constraints.append(Zs <= x[s:])
                # Zs ≥ x[i] + x[i+s] – 1   (enforces AND)
                constraints.append(Zs >= x[:length] + x[s:] - 1)
                # sum_i x[i] – sum_i Z_{i,s} ≤ t   (the shift constraint)
                constraints.append(cp.sum(x[:length]) - cp.sum(Zs) <= t)

            prob = cp.Problem(cp.Minimize(t), constraints)

            # pick a MILP solver that is likely to be present
            solver_used = None
            for solver_name in ("GLPK_MI", "CBC", "SCIP", "ECOS_BB"):
                try:
                    prob.solve(solver=getattr(cp, solver_name), warm_start=True, verbose=False)
                    solver_used = solver_name
                    break
                except Exception:
                    continue

            if prob.status in ("optimal", "optimal_inaccurate") and x.value is not None:
                xv = np.array(x.value).flatten()
                # round to nearest integer (0/1) and fix any small integrality drift
                h_bin = np.rint(xv)
                h_bin = np.clip(h_bin, 0, 1)

                # Ensure exact number of ones
                cur = int(round(h_bin.sum()))
                diff = target_ones - cur
                if diff != 0:
                    resid = xv - h_bin
                    if diff > 0:
                        cand = np.where(h_bin == 0)[0]
                        order = cand[np.argsort(-resid[cand])]
                        h_bin[order[:diff]] = 1
                    else:
                        cand = np.where(h_bin == 1)[0]
                        order = cand[np.argsort(resid[cand])]
                        h_bin[order[: -diff]] = 0

                # final check (should hold)
                if int(round(h_bin.sum())) == target_ones:
                    c5_val = np.max(np.correlate(h_bin, 1 - h_bin, mode="full")) * dx
                    best_h = h_bin.astype(float)
                    best_c5 = float(c5_val)
        except Exception:
            # any problem – fall back to heuristic below
            pass

    # ----------------------------------------------------------------------
    # 2) If no exact solution yet, run a simple binary hill‑climbing search
    # ----------------------------------------------------------------------
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if best_h is None or remaining > 0.5:

        if best_h is not None:

                    cur_h[order[:diff]] = 1.0

                    order = one_idx[np.argsort(h0[one_idx])]
