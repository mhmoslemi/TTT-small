def gradient_of_c(h, k):
    n = h.shape[0]
    grad = np.ones(n)   # initialize term1 default 1
    if k >= 0:
        # term1: for i in 0..n-k-1: 1 - h[i+k]
        if k < n:
            grad[:n - k] = 1 - h[k:]
        # term2: subtract h[i - k] for i >= k
        if k > 0:
            grad[k:] -= h[:n - k]
    else:
        kk = -k
        # term1 for i >= kk: 1 - h[i - kk] (i - kk = i + k)
        if kk < n:
            grad[kk:] = 1 - h[:n - kk]
        # term2 subtract h[i + kk] for i < n - kk
        grad[:n - kk] -= h[kk:]
    return grad

def local_optimize(h0, target_sum, time_limit, start_time):
    h = h0.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_val = compute_max_c(h)
    it = 0
    # schedule 5000 iterations max
    max_iters = 5000
    while it < max_iters and time.time() - start_time < time_limit:
        # compute correlation and find max lag
        corr = np.correlate(h, 1 - h, mode='full')
        C_vals = corr * dx
        k_idx = np.argmax(C_vals)
        cur_val = C_vals[k_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # compute lag
        k = k_idx - (n - 1)
        # grad
        grad = gradient_of_c(h, k)
        # step size decreasing
        step = 0.5 * (1.0 - it / max_iters)
        # try decreasing step until improvement
        improved = False
        step_try = step
        while step_try > 1e-6:
            v = h - step_try * grad
            v = project_to_sum_box(v, target_sum)
            new_val = compute_max_c(v)
            if new_val <= cur_val + 1e-12:
                # accept
                h = v
                improved = True
                break
            step_try *= 0.5
        if not improved:
            # random perturbation
            # randomly pick two indices and exchange some mass
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            # amount to transfer
            max_transfer = min(h[i], 1 - h[j])
            if max_transfer <= 0:
                continue
            delta = rng.uniform(0, max_transfer)
            v = h.copy()
            v[i] -= delta
            v[j] += delta
            v = project_to_sum_box(v, target_sum)
            h = v
        it += 1
    # after loop, evaluate best
    final_val = compute_max_c(best_h)
    return best_h, final_val

import time
import numpy as np

def _project_to_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] with sum equal to target."""
    lo, hi = -3.0, 3.0  # wide enough bounds for the Lagrange multiplier
    for _ in range(60):
        mid = (lo + hi) / 2.0
        u = np.clip(v - mid, 0.0, 1.0)
        s = u.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _max_overlap(h: np.ndarray) -> float:
    """Compute the maximal overlap C_k for the vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _gradient_of_max(h: np.ndarray, lag: int) -> np.ndarray:
    """Gradient of the overlap C_k for a given lag (k)."""
    n = h.size
    grad = np.ones(n, dtype=float)  # term1 = 1 for all indices
    if lag >= 0:
        k = lag
        if k < n:
            grad[: n - k] = 1.0 - h[k:]
        if k > 0:
            grad[k:] -= h[: n - k]
    else:
        kk = -lag
        if kk < n:
            grad[kk:] = 1.0 - h[: n - kk]
        grad[: n - kk] -= h[kk:]
    return grad


def _local_optimize(
    h0: np.ndarray,
    target_sum: float,
    time_limit: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Simple projected‑gradient descent with occasional random perturbations."""
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_val = _max_overlap(best_h)

    max_iters = 8000
    it = 0
    while it < max_iters and (time.time() < time_limit):
        # compute current maximal lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        c_vals = corr * dx
        idx = int(np.argmax(c_vals))
        cur_val = float(c_vals[idx])
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        lag = idx - (n - 1)
        grad = _gradient_of_max(h, lag)

        # decreasing step size
        step = 0.6 * (1.0 - it / max_iters)
        improved = False
        while step > 1e-6:
            v = h - step * grad
            v = _project_to_sum_box(v, target_sum)
            new_val = _max_overlap(v)
            if new_val <= cur_val + 1e-12:
                h = v
                improved = True
                break
            step *= 0.5

        if not improved:
            # random mass‑transfer perturbation
            i, j = rng.integers(n, size=2)
            if i != j:
                max_move = min(h[i], 1.0 - h[j])
                if max_move > 1e-8:
                    delta = rng.uniform(0.0, max_move)
                    v = h.copy()
                    v[i] -= delta
                    v[j] += delta
                    h = _project_to_sum_box(v, target_sum)
        it += 1

    final_val = _max_overlap(best_h)
    return best_h, final_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step‑function lower‑bound for the Erdős C₅ constant."""
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise a feasible vector
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # default problem size – modest enough for fast evaluation
        n = 200
        h0 = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h0[ones] = 1.0

    n = h0.size
    target_sum = n / 2.0
    h0 = _project_to_sum_box(h0, target_sum)

    best_h = h0.copy()
    best_val = _max_overlap(best_h)

    # ------------------------------------------------------------------
    # optimisation loop (a few restarts, each limited by remaining time)
    total_time = budget_s * 0.95
    remainder = start_time + total_time
    restarts = 3
    for _ in range(restarts):
        if time.time() >= remainder:
            break
        # fresh start – first restart uses the supplied vector
        if _ == 0:
            h_start = h0.copy()
        else:
            # random binary start
            h_start = np.zeros(n, dtype=float)
            idx = rng.choice(n, n // 2, replace=False)
            h_start[idx] = 1.0
            h_start = _project_to_sum_box(h_start, target_sum)

        # run local optimisation within the remaining budget
        local_deadline = min(remainder, time.time() + (budget_s * 0.25))
        h_opt, val_opt = _local_optimize(h_start, target_sum, local_deadline, rng)

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
        # early exit if we already beat the target
        if best_val <= 0.3808:
            break

    # final sanity projection
    best_h = _project_to_sum_box(best_h, target_sum)
    final_c = _max_overlap(best_h)

    return (best_h.tolist(), float(final_c), n)
