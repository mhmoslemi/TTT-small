import time, random
import numpy as np
import cvxpy as cp

# Helper: compute the C5 bound for a binary (or fractional) vector h
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def solve_feasible(n, max_mismatch):
    """Find a binary vector of length n with exactly n/2 ones such that
    for every shift s (both directions) the mismatch count ≤ max_mismatch.
    Returns the vector (as numpy array) on success, or None."""
    half = n // 2
    # binary variables for the function values
    h = cp.Variable(n, boolean=True)

    # product variables y_{i,s} = h_i * h_{i+s}
    pair_idx = []                     # (i, s) pairs
    shift_to_y = {s: [] for s in range(1, n)}   # indices of y belonging to each shift
    for s in range(1, n):
        for i in range(n - s):
            pair_idx.append((i, s))
            shift_to_y[s].append(len(pair_idx) - 1)

    y = cp.Variable(len(pair_idx), boolean=True)

    constraints = []

    # sum of h must be exactly half
    constraints.append(cp.sum(h) == half)

    # linearization of the product y = h_i * h_{i+s}
    for idx, (i, s) in enumerate(pair_idx):
        constraints.append(y[idx] <= h[i])
        constraints.append(y[idx] <= h[i + s])
        constraints.append(y[idx] >= h[i] + h[i + s] - 1)

    # mismatch constraints for each shift (both directions)
    for s, y_idx in shift_to_y.items():
        ys = y[y_idx]
        # forward mismatches: sum_{i=0}^{n-s-1} h_i - sum y_{i,s}
        forward = cp.sum(h[: n - s]) - cp.sum(ys)
        constraints.append(forward <= max_mismatch)

        # backward mismatches: sum_{i=s}^{n-1} h_i - sum y_{i,s}
        backward = cp.sum(h[s:]) - cp.sum(ys)
        constraints.append(backward <= max_mismatch)

    # T (max mismatch) is just a constant max_mismatch
    prob = cp.Problem(cp.Minimize(0), constraints)

    # try a few solvers that support MILP
    for solver in (cp.CBC, cp.GLPK_MI, cp.SCIP):
        try:
            prob.solve(solver=solver, verbose=False, warm_start=True)
            if prob.status in ("optimal", "optimal_inaccurate"):
                vals = np.rint(h.value).astype(int)
                if vals.sum() == half:
                    return vals
        except Exception:
            continue
    return None


def hill_climb(initial_h, time_limit, target_c5):
    """Simple stochastic hill‑climb: swap a 1 and a 0 if it improves C5."""
    h = initial_h.copy()
    n = len(h)
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    best = compute_c5(h)
    start = time.time()

    while time.time() - start < time_limit:
        i = random.choice(tuple(ones))
        j = random.choice(tuple(zeros))
        # propose swap
        h[i], h[j] = 0, 1
        cur = compute_c5(h)
        if cur < best - 1e-7:
            best = cur
            ones.remove(i)
            ones.add(j)
            zeros.remove(j)
            zeros.add(i)
            if best <= target_c5:
                break
        else:
            # revert
            h[i], h[j] = 1, 0
    return h, best


def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    target = 0.3808

    # Try increasing even sizes until we find a feasible construction.
    # The maximal allowed mismatch for size n is floor(target * n / 2).
    for n in (60, 80, 100, 120, 140, 160, 180, 200):
        if time.time() - start_time > 0.85 * budget_s:
            break
        if n % 2 != 0:
            continue
        max_mis = int(np.floor(target * n / 2.0))
        sol = solve_feasible(n, max_mis)
        if sol is not None:
            c5 = compute_c5(sol.astype(float))
            return (sol.astype(float).tolist(), c5, n)

    # If the exact MILP approach failed, fall back to a stochastic search.
    # Use a modest size (200) to give enough resolution.
    n = 200
    half = n // 2
    max_time = budget_s - (time.time() - start_time)
    # Random initial binary vector
    h0 = np.zeros(n, dtype=int)
    h0[:half] = 1
    np.random.shuffle(h0)
    # Improve it with hill climbing while time permits
    time_for_hc = max_time * 0.9
    best_h, best_c5 = hill_climb(h0, time_for_hc, target)
    # If still not good enough, try a few random restarts
    while time.time() - start_time < budget_s * 0.98 and best_c5 > target:
        h0 = np.zeros(n, dtype=int)
        h0[:half] = 1
        np.random.shuffle(h0)
        hc_h, hc_c5 = hill_climb(h0, time_for_hc, target)
        if hc_c5 < best_c5:
            best_h, best_c5 = hc_h, hc_c5
        if best_c5 <= target:
            break

    return (best_h.astype(float).tolist(), best_c5, n)
