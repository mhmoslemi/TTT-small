def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

import numpy as np
import time
import math

def _compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)*(1-h(x+k)) dx approximated by full correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate with mode='full' gives the discrete correlation.
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _mismatch_indices(h: np.ndarray, shift: int):
    """
    For a given shift, return two arrays:
    - indices where h[i]=1 and h[i+shift]=0   (mismatched ones)
    - indices where h[i]=0 and h[i+shift]=1   (mismatched zeros)

    Works for positive and negative shift.
    """
    n = h.shape[0]
    if shift >= 0:
        a = h[: n - shift]
        b = h[shift:]
        mis_one = np.where((a == 1) & (b == 0))[0]
        mis_zero = np.where((a == 0) & (b == 1))[0]
    else:
        s = -shift
        a = h[s:]
        b = h[: n - s]
        mis_one = np.where((a == 1) & (b == 0))[0] + s
        mis_zero = np.where((a == 0) & (b == 1))[0] + s
    return mis_one, mis_zero

        n = 70  # a reasonable default size
    target_sum = n / 2.0
    dx = 2.0 / n

    # construct two possible starts: binary from ranking & continuous copy
    starts = []

    if init_h is not None:
        # continuous start – clip & adjust sum
        h_cont = np.clip(init_h, 0.0, 1.0)
        diff = target_sum - h_cont.sum()
        # distribute the remaining diff by greedy fixing
        while abs(diff) > 1e-12:
            if diff > 0:
                caps = 1.0 - h_cont
                idx = np.argmax(caps)
                delta = min(diff, caps[idx])
                h_cont[idx] += delta
                diff -= delta
            else:
                caps = h_cont
                idx = np.argmax(caps)
                delta = min(-diff, caps[idx])
                h_cont[idx] -= delta
                diff += delta
        starts.append(h_cont)

        # binary start – top half entries become 1
