def project_to_hyperplane(x, target, max_iter=50, tol=1e-9):
    lo = -1.0
    hi = 1.0
    # Expand boundaries if needed.
    while np.sum(np.clip(x + lo,0,1)) > target:
        lo *= 2
    while np.sum(np.clip(x + hi,0,1)) < target:
        hi *= 2
    for _ in range(max_iter):
        mid = (lo+hi)/2
        y = np.clip(x + mid,0,1)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo+hi)/2,0,1)

def project_to_hyperplane(x, target, max_iter=50):
    lo = -2.0
    hi = 2.0
    # Expand if not enough
    while np.sum(np.clip(x + lo, 0, 1)) > target:
        lo *= 2
    while np.sum(np.clip(x + hi, 0, 1)) < target:
        hi *= 2
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        y = np.clip(x + mid, 0, 1)
        s = np.sum(y)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) / 2, 0, 1)

import numpy as np

# Helper: project onto the hyperplane sum(x)=target with box [0,1]
def _project_hyperplane(x, target, max_iter=60, tol=1e-12):
    """Find y = clip(x + λ,0,1) such that sum(y)=target."""
    lo, hi = -2.0, 2.0
    # expand bounds until target lies inside the interval of sums
    while np.sum(np.clip(x + lo, 0, 1)) > target:
        lo *= 2.0
    while np.sum(np.clip(x + hi, 0, 1)) < target:
        hi *= 2.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0, 1)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
        if hi - lo < tol:
            break
    return np.clip(x + (lo + hi) * 0.5, 0, 1)

# Compute the Erdős C5 value for a given step vector
def _c5_value(h):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    return corr.max() * (2.0 / n)
