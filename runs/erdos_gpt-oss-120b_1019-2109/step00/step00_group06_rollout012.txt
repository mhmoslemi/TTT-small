    start = time.time()
    best_a = a.copy()
    best_val = compute_maxS(a)
    curr_a = a.copy()
    curr_val = best_val
    T0 = 0.5 * best_val  # initial temperature maybe
    # schedule T = T0 * (0.99)**i
    for iter in range(max_iter):
        if time.time() - start > time_limit: break
        # propose swap: choose random positions i,j with a[i]!=a[j]
        i = random.randrange(N)
        j = random.randrange(N)
        if curr_a[i] == curr_a[j]:
            continue
        # perform swap
        curr_a[i], curr_a[j] = curr_a[j], curr_a[i]
        # compute new maxS
        new_val = compute_maxS(curr_a)  # O(N log N)
        # acceptance
        delta = new_val - curr_val
        T = T0 * (0.9995**iter)  # decaying
        if delta <= 0 or random.random() < math.exp(-delta / (T+1e-12)):
            curr_val = new_val
            # possibly update best
            if new_val < best_val:
                best_val = new_val
                best_a = curr_a.copy()
                # optionally break early if near target
        else:
            # revert swap
            curr_a[i], curr_a[j] = curr_a[j], curr_a[i]

import time
import math
import random
import numpy as np

def _max_correlation(a):
    """Return max_{k} sum_i a_i(1-a_{i+k}) for binary a."""
    # using np.correlate which computes sum_i a_i * b_{i+k} with b = 1-a
    return np.correlate(a, 1 - a, mode='full').max()

def _propose_swap(a, shift):
    """Propose a swap (i,j) aimed at reducing the given shift's overlap.
    Returns a pair of indices (i,j)."""
    N = a.size
    s = abs(shift)
    if s == 0:
        # any random 1-0 swap
        i = int(np.random.choice(np.where(a == 1)[0]))
        j = int(np.random.choice(np.where(a == 0)[0]))
        return i, j

    if shift > 0:
        # mismatches: a[i]=1, a[i+s]=0   (type A)
        # opposite:   a[i]=0, a[i+s]=1   (type B)
        left_one = np.where((a[:N - s] == 1) & (a[s:] == 0))[0]
        left_zero = np.where((a[:N - s] == 0) & (a[s:] == 1))[0]
        if left_one.size and left_zero.size:
            i = int(np.random.choice(left_one))
            j = int(np.random.choice(left_zero) + s)
            return i, j
    else:
        # shift < 0: mismatches are opposite orientation
        # mismatches: a[i]=0, a[i+s]=1   (type A for negative shift)
        # opposite:   a[i]=1, a[i+s]=0   (type B)
        left_zero = np.where((a[:N - s] == 0) & (a[s:] == 1))[0]
        left_one = np.where((a[:N - s] == 1) & (a[s:] == 0))[0]
        if left_zero.size and left_one.size:
            i = int(np.random.choice(left_zero))
            j = int(np.random.choice(left_one) + s)
            return i, j

    # fallback: any random 1‑0 swap
    i = int(np.random.choice(np.where(a == 1)[0]))
    j = int(np.random.choice(np.where(a == 0)[0]))
    return i, j


def run(seed=42, budget_s=1100):
    """Return (h_values, c5_bound, n_points) with a low C5 bound."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------
    # Choose an even number of sample points (small enough for speed)
    N = 200                       # length of the step function
    half = N // 2                 # required number of ones

    # ------------------------------------------------------------
    # Build an initial binary pattern using the supplied initial_h_values
    try:
        init = initial_h_values                # provided by the harness
        p = np.take(init, np.arange(N) % init.shape[0])
    except NameError:
        # No parent available – use a uniform probability
        p = np.full(N, 0.5)

    # Take the half positions with largest probabilities as ones
    a = np.zeros(N, dtype=np.int8)
    order = np.argsort(-p, kind='stable')
    a[order[:half]] = 1

    # ------------------------------------------------------------
    # Simulated‑annealing / greedy improvement loop
    best_a = a.copy()
    best_val = _max_correlation(a)

    target_bound = 0.3808                     # desired C5 upper bound
    target_max = int(math.floor(target_bound * N / 2))

    # temperature parameters
    T0 = max(1.0, best_val * 0.5)
    Tmin = 1e-4
    decay = 0.9995

    cur_a = a.copy()
    cur_val = best_val
    it = 0

    while True:
        # time check
        if time.time() - start_time > budget_s * 0.97:
            break

        # compute correlation and current maximum shift
        corr = np.correlate(cur_a, 1 - cur_a, mode='full')
        cur_val = int(corr.max())
        if cur_val < best_val:
            best_val = cur_val
            best_a = cur_a.copy()
            if best_val <= target_max:
                break

        # temperature schedule (linear cooling)
        elapsed = time.time() - start_time
        T = max(Tmin, T0 * (1.0 - elapsed / budget_s))

        # propose a swap aiming at the worst shift
        shift_idx = int(corr.argmax())
        shift = shift_idx - (N - 1)          # convert index to signed shift
        i, j = _propose_swap(cur_a, shift)

        # perform tentative swap
        cur_a[i], cur_a[j] = cur_a[j], cur_a[i]
        new_val = int(np.correlate(cur_a, 1 - cur_a, mode='full').max())

        # acceptance test
        if new_val <= cur_val or random.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur_val = new_val
            # keep the new configuration
        else:
            # revert swap
            cur_a[i], cur_a[j] = cur_a[j], cur_a[i]

        # occasional random restart if stagnation is detected
        it += 1
        if it % (N * 20) == 0:
            # random reshuffle while preserving weight
            ones_idx = np.where(cur_a == 1)[0]
            zeros_idx = np.where(cur_a == 0)[0]
            np.random.shuffle(ones_idx)
            np.random.shuffle(zeros_idx)
            # swap a few random pairs to inject diversity
            for _ in range(N // 10):
                if not ones_idx.size or not zeros_idx.size:
                    break
                i = int(np.random.choice(ones_idx))
                j = int(np.random.choice(zeros_idx))
                cur_a[i], cur_a[j] = cur_a[j], cur_a[i]

    # ------------------------------------------------------------
    # Prepare final answer
    h_vals = best_a.astype(float)                # float array in [0,1]
    c5_bound = best_val * (2.0 / N)              # max correlation * dx
    n_points = N

    return h_vals, c5_bound, n_points
