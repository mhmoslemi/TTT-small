def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    rev = (1 - h)[::-1]
    N = 2*n - 1
    # compute convolution via FFT
    H = np.fft.rfft(h, n=N)
    R = np.fft.rfft(rev, n=N)
    conv = np.fft.irfft(H * R, n=N)
    # conv length N = 2*n-1
    # max conv
    max_val = np.max(conv)
    return max_val * dx

def build_shape(n, a, L):
    dx = 2.0/n
    H = 1.0/(L + a)  # height

    # compute x coordinate for each sample: midpoints: xs = (np.arange(n) + 0.5) * dx
    xs = (np.arange(n) + 0.5) * dx
    h = np.zeros(n)
    # left ramp region: xs < a
    mask = xs < a
    h[mask] = H * xs[mask] / a

    # plateau region: (xs >= a) & (xs < a+L)
    mask = (xs >= a) & (xs < a+L)
    h[mask] = H

    # right ramp region: (xs >= a+L) & (xs < a+L+a)
    mask = (xs >= a+L) & (xs < a+L+a)
    h[mask] = H * (a+L+a - xs[mask]) / a

    # else remains 0
    return h

def compute_c5(h):
    n = h.shape[0]; N = 2*n - 1
    rev = (1.0 - h)[::-1]
    m = 1 << ((N-1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:N]
    return conv.max() * (2.0 / n)

def build_shape(n, a, L):
    dx = 2.0 / n
    H = 1.0 / (L + a) if (L + a) > 0 else 0.0
    xs = (np.arange(n) + 0.5) * dx
    h = np.zeros_like(xs)
    # left ramp
    if a > 0:
        mask = xs < a
        h[mask] = H * xs[mask] / a
    # plateau
    mask = (xs >= a) & (xs < a + L)
    h[mask] = H
    # right ramp
    if a > 0:
        mask = (xs >= a + L) & (xs < a + L + a)
        h[mask] = H * (a + L + a - xs[mask]) / a
    # else ramp length zero => nothing
    # Ensure within [0,1]
    np.clip(h, 0.0, 1.0, out=h)
    # Project to sum constraint:
    target = n / 2.0
    total = h.sum()
    if abs(total - target) > 1e-10:
        # Scale to meet sum then adjust clamping
        scale = target / total if total > 0 else 0.0
        h *= scale
        # Clip again if needed
        np.clip(h, 0.0, 1.0, out=h)
        total = h.sum()
        diff = target - total
        if abs(diff) > 1e-12:
            # find an index not at bound to adjust
            idx = np.where((h > 0) & (h < 1))[0]
            if idx.size == 0:
                # fallback: adjust first element within bound
                idx = np.arange(n)
            # adjust first element
            i = idx[0]
            h[i] += diff
            # Clip again ensure bounds
            if h[i] < 0:
                h[i] = 0.0
            elif h[i] > 1:
                h[i] = 1.0
            # After adjustment sum might be off by rounding; but diff small.
    return h

def grid_search(n, time_limit):
    best_score = np.inf
    best_h = None
    best_params = None
    a_vals = np.linspace(0.001, 0.999, 100) # 0 < a < 1

        L_max = 2.0 - 2.0 * a

def refine_sa(h_initial, best_score, n, time_remaining):
    h = h_initial.copy()
    best_h = h.copy()
    best = best_score
    target = n / 2.0
    start = time.time()
    # total iterations relative to time
    max_iters = int(5e5)  # big enough
    for it in range(max_iters):
        elapsed = time.time() - start
        if elapsed > time_remaining:
            break
        # propose move
        i, j = np.random.randint(0, n, 2)
        while i == j:
            j = np.random.randint(0, n)
        # compute max possible delta
        max_delta = min(h[i], 1.0 - h[j])
        min_delta = -min(1.0 - h[i], h[j])
        if max_delta < -min_delta:
            # choose sign randomly
            delta = np.random.rand() * max_delta  # will be <=0 maybe
        else:
            delta = (np.random.rand() * (max_delta - min_delta) + min_delta)
        if abs(delta) < 1e-12:
            continue
        h[i] -= delta
        h[j] += delta
        # evaluate
        score = compute_c5(h)
        # temperature schedule
        T = 0.02 * (1 - it / max_iters)
        if score < best or np.random.rand() < np.exp(-(score - best) / (T + 1e-12)):
            # accept
            if score < best:
                best = score
                best_h = h.copy()
        else:
            # revert
            h[i] += delta
            h[j] -= delta
    return best_h, best

import time
import random
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Compute the C5 value for a given discretised step function."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    N = 2 * n - 1
    # next power of two for fast FFT
    m = 1 << ((N - 1).bit_length())
    H = np.fft.rfft(h, m)
    R = np.fft.rfft(rev, m)
    conv = np.fft.irfft(H * R, m)[:N]
    return conv.max() * (2.0 / n)

def _project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Scale and adjust a vector so that sum equals target while staying in [0,1]."""
    total = h.sum()
    if total != 0:
        scale = target / total
        h = h * scale
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny remainder to a non‑boundary entry if possible
        idx = np.where((h > 0) & (h < 1))[0]
        if idx.size == 0:
            idx = np.arange(h.size)
        i = idx[0]
        h[i] += diff
        h[i] = min(1.0, max(0.0, h[i]))
    return h

def _build_shape(n: int, a: float, L: float) -> np.ndarray:
    """Symmetric piecewise‑linear shape with ramp length a and plateau length L."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.zeros_like(xs)

    if a < 1e-12:  # pure plateau (constant height)
        if L <= 0:
            return h
        H = 1.0 / L
        left = (2.0 - L) / 2.0
        right = left + L
        mask = (xs >= left) & (xs < right)
        h[mask] = H
    else:
        H = 1.0 / (L + a)               # height chosen so that area = 1
        # left ramp
        mask = xs < a
        h[mask] = H * xs[mask] / a
        # plateau
        mask = (xs >= a) & (xs < a + L)
        h[mask] = H
        # right ramp
        mask = (xs >= a + L) & (xs < a + L + a)
        h[mask] = H * (a + L + a - xs[mask]) / a

    # enforce constraints exactly
    h = np.clip(h, 0.0, 1.0)
    h = _project_sum(h, n / 2.0)
    return h

def _grid_search(n: int, time_budget: float):
    """Coarse search over (a, L) parameters."""
    start = time.time()
    best_val = np.inf
    best_h = None
    a_vals = np.linspace(0.001, 0.999, 120)
    for a in a_vals:
        L_min = max(0.0, 1.0 - a)
        L_max = 2.0 - 2.0 * a
        if L_max < L_min:
            continue
        L_vals = np.linspace(L_min, L_max, 80)
        for L in L_vals:
            h = _build_shape(n, a, L)
            val = _c5_value(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
        if time.time() - start > time_budget:
            break
    return best_h, best_val

def _simulated_annealing(start_h: np.ndarray, start_val: float, n: int, time_budget: float):
    """Neighbourhood search that preserves the integral exactly."""
    h = start_h.copy()
    best_h = h.copy()
    best_val = start_val
    target = n / 2.0
    start = time.time()
    it = 0
    # upper bound for iterations – will be cut by time budget
    max_iters = 1_000_000
    while it < max_iters and (time.time() - start) < time_budget:
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        hi, hj = h[i], h[j]
        # move delta from i to j, respecting [0,1]
        max_delta = min(hi, 1.0 - hj)
        min_delta = -min(1.0 - hi, hj)
        if max_delta <= min_delta:
            continue
        delta = np.random.rand() * (max_delta - min_delta) + min_delta
        if abs(delta) < 1e-14:
            continue
        h[i] = hi - delta
        h[j] = hj + delta
        val = _c5_value(h)

        # simple linear cooling
        T = 0.02 * (1.0 - it / max_iters)
        accept = False
        if val < best_val:
            accept = True
        else:
            prob = np.exp(-(val - best_val) / (T + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
        it += 1
    # final projection (numerical drift may appear)
    best_h = _project_sum(best_h, target)
    return best_h, best_val
