import time, math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h."""
    n = h.size
    # raw correlation (sum of h_i * (1 - h_{i+k}) over overlapping indices)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _project_to_sum_and_bounds(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project x onto the hyperplane sum(x)=target_sum with box 0≤x≤1."""
    # Clip first, then find a shift λ such that sum(clip(x+λ,0,1)) = target_sum.
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
