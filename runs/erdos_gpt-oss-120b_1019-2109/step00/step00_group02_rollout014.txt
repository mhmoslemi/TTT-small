import numpy as np
import random
import time
import math

def _compute_c5(h):
    """Return C5 = max_x h(x)*(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _gradient_for_shift(h, k):
    """Gradient of P_k = sum_i h_i h_{i+k} w.r.t. h (for a fixed shift k)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if k >= 0:
        # term h[i] * h[i+k]
        g[: n - k] += h[k:]
        g[k:] += h[: n - k]
    else:
        d = -k
        g[d:] += h[: n - d]
        g[: n - d] += h[d:]
    return g

def _random_initial(n):
    """Binary (or with a single fractional entry) vector with sum = n/2."""
    target = n / 2.0
    h = np.zeros(n, dtype=float)
    whole = int(math.floor(target))
    if whole > 0:
        idx = np.random.choice(n, whole, replace=False)
        h[idx] = 1.0
    remainder = target - whole
    if remainder > 1e-12:
        free = np.where(h == 0.0)[0]
        i = np.random.choice(free)
        h[i] = remainder
    return h

def _improve_h(h0, time_budget, target_c5=0.3808):
    """Hill‑climbing / simulated‑annealing optimizer for a fixed n."""
    n = h0.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    cur_h = h0.copy()
    cur_c5 = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    start = time.time()
    # parameters
    T0 = 0.006          # initial temperature for Metropolis acceptance
    max_step_frac = 0.6 # initial max fraction of the allowable move
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # linear schedules
        T = T0 * max(0.0, 1.0 - elapsed / time_budget)
        step_frac = max_step_frac * max(0.0, 1.0 - elapsed / time_budget)

        # compute the shift with minimal inner product (max overlap)
        conv = np.correlate(cur_h, cur_h, mode="full")
        idx_min = int(np.argmin(conv))
        k = idx_min - (n - 1)          # shift corresponding to the minimum product
        grad = _gradient_for_shift(cur_h, k)

        # pick indices: increase where gradient large, decrease where gradient small
        inc_candidates = np.where(cur_h < 1.0 - 1e-12)[0]
        dec_candidates = np.where(cur_h > 1e-12)[0]
        if inc_candidates.size and dec_candidates.size:
            i = inc_candidates[np.argmax(grad[inc_candidates])]
            j = dec_candidates[np.argmin(grad[dec_candidates])]
            max_up = min(1.0 - cur_h[i], cur_h[j])   # amount we can move i up, j down
            if max_up < 1e-12:   # fallback to random move
                i, j = np.random.randint(0, n, size=2)
                while i == j:
                    j = np.random.randint(0, n)
                max_up = min(1.0 - cur_h[i], cur_h[j])
                if max_up < 1e-12:
                    continue
        else:
            # very unlikely, but handle degenerate case
            i, j = np.random.randint(0, n, size=2)
            while i == j:
                j = np.random.randint(0, n)
            max_up = min(1.0 - cur_h[i], cur_h[j])
            if max_up < 1e-12:
                continue

        # propose epsilon (move amount)
        eps = max_up * step_frac * random.random()
        if eps <= 0.0:
            continue

        new_h = cur_h.copy()
        new_h[i] += eps
        new_h[j] -= eps

        # enforce the box just in case of floating‑point slip
        new_h[i] = min(1.0, max(0.0, new_h[i]))
        new_h[j] = min(1.0, max(0.0, new_h[j]))

        # calculate new C5
        new_c5 = _compute_c5(new_h)

        # Metropolis acceptance
        improve = new_c5 < cur_c5 - 1e-14
        if improve:
            accept = True
        elif T > 0.0:
            accept = math.exp(-(new_c5 - cur_c5) / T) > random.random()
        else:
            accept = False

        if accept:
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_h.copy()
                if best_c5 < target_c5:
                    # early exit if target reached
                    return best_h, best_c5
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function (list), its C5 bound, and number of points."""
    random.seed(seed)
    np.random.seed(seed)

    start_total = time.time()
    # Determine starting vector
    init = globals().get("initial_h_values")
    if init is not None:
        h0 = np.array(init, dtype=float)
        n = h0.shape[0]
        # make sure the integral constraint is (exactly) satisfied
        target = n / 2.0
        diff = target - h0.sum()
        if abs(diff) > 1e-12:
            # shift the whole vector uniformly (still respects bounds)
            h0 += diff / n
            h0 = np.clip(h0, 0.0, 1.0)
            # re‑balance if needed
            cur = h0.sum()
            if abs(cur - target) > 1e-12:
                # simple water‑filling correction
                for _ in range(10):
                    excess = cur - target
                    if abs(excess) < 1e-12:
                        break
                    free_up = np.where(h0 < 1.0 - 1e-12)[0]
                    free_dn = np.where(h0 > 1e-12)[0]
                    if excess > 0 and free_up.size:
                        delta = excess / free_up.size
                        h0[free_up] -= delta
                    elif excess < 0 and free_dn.size:
                        delta = -excess / free_dn.size
                        h0[free_dn] += delta
                    h0 = np.clip(h0, 0.0, 1.0)
                    cur = h0.sum()
    else:
        # No parent solution – start with a random feasible vector.
        n = 120  # a modest size that still runs fast
        h0 = _random_initial(n)

    # Time left for the optimizer (reserve a little margin for final work)
    time_left = budget_s - (time.time() - start_total) - 0.3
    if time_left < 0.1:
        time_left = 0.1

    # Run the improvement routine
    best_h, best_c5 = _improve_h(h0, time_left, target_c5=0.3808)

    # Final projection to satisfy the sum constraint exactly
    target_sum = n / 2.0
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-10:
        best_h += diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
        # Small iterative correction if clipping broke the sum again
        for _ in range(10):
            cur_sum = best_h.sum()
            err = target_sum - cur_sum
            if abs(err) < 1e-12:
                break
            free = (best_h > 1e-12) & (best_h < 1 - 1e-12)
            cnt = np.count_nonzero(free)
            if cnt == 0:
                break
            adj = err / cnt
            best_h[free] += adj
            best_h = np.clip(best_h, 0.0, 1.0)

    best_c5 = _compute_c5(best_h)

    # Return values as required by the harness
    return best_h.tolist(), float(best_c5), int(n)
