import time, numpy as np, math

def compute_max_corr(h):
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    idx = corr.argmax()
    return max_val, idx

def generate_modular_pattern(N, a):
    # binary pattern where ((i*a) % N) < N/2
    i = np.arange(N, dtype=int)
    return ((i * a) % N) < (N // 2)   # N even ensures N/2 integer.

def mismatched_indices(h, shift):
    N = h.shape[0]
    if shift >= 0:
        i = np.arange(0, N - shift)
        j = i + shift
    else:
        s = -shift
        i = np.arange(s, N)
        j = i - s
    hi = h[i]
    hj = h[j]
    mis_one = i[(hi == 1) & (hj == 0)]
    mis_zero = i[(hi == 0) & (hj == 1)]
    return mis_one, mis_zero

def hill_climb_binary(h, budget_s, start_time):
    N = h.shape[0]
    dx = 2.0 / N
    # compute initial
    best_h = h.copy()
    best_corr, best_idx = compute_max_corr(h)
    best_c5 = best_corr * dx
    cur_corr, cur_idx = best_corr, best_idx
    cur_shift = cur_idx - (N - 1)
    rng = np.random.default_rng()
    while time.time() - start_time < budget_s - 0.1:
        # generate swap candidates
        mis_one, mis_zero = mismatched_indices(h, cur_shift)
        if mis_one.size > 0 and mis_zero.size > 0:
            i = rng.choice(mis_one)
            j = rng.choice(mis_zero)
        else:
            # random swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # perform swap
        h[i], h[j] = h[j], h[i]
        new_corr, new_idx = compute_max_corr(h)
        if new_corr <= cur_corr:
            # accept
            cur_corr, cur_idx = new_corr, new_idx
            cur_shift = cur_idx - (N - 1)
            if new_corr < best_corr:
                best_corr = new_corr
                best_h = h.copy()
                best_c5 = best_corr * dx
        else:
            # revert
            h[i], h[j] = h[j], h[i]
    return best_h, best_c5

    total = best_h.sum()

import time, math
import numpy as np

def _compute_max_corr(h):
    """Return (max correlation value, index of max) for h against (1-h)."""
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max(), int(corr.argmax())

def _modular_pattern(N, a):
    """Binary pattern of length N with density 1/2 via modular arithmetic."""
    i = np.arange(N, dtype=int)
    return ((i * a) % N) < (N // 2)

def _mismatch_indices(h, shift):
    """Indices i where h[i]==1 & h[i+shift]==0 (and opposite) in the overlap region."""
    N = h.shape[0]
    if shift >= 0:
        i = np.arange(0, N - shift, dtype=int)
        j = i + shift
    else:
        s = -shift
        i = np.arange(s, N, dtype=int)
        j = i - s
    hi = h[i]
    hj = h[j]
    mis_one = i[(hi == 1) & (hj == 0)]
    mis_zero = i[(hi == 0) & (hj == 1)]
    return mis_one, mis_zero

def _hill_climb_binary(h, start_time, end_time, max_temp=0.2):
    """Simple simulated‑annealing hill climbing for binary h."""
    N = h.shape[0]
    dx = 2.0 / N
    cur_corr, cur_idx = _compute_max_corr(h)
    cur_shift = cur_idx - (N - 1)
    best_h = h.copy()
    best_corr = cur_corr
    best_c5 = cur_corr * dx
    rng = np.random.default_rng()
    while time.time() < end_time - 0.01:
        mis_one, mis_zero = _mismatch_indices(h, cur_shift)
        if mis_one.size and mis_zero.size:
            i = rng.choice(mis_one)
            j = rng.choice(mis_zero)
        else:
            ones = np.where(h == 1)[0]

            i = rng.choice(ones)
            j = rng.choice(zeros)
