def compute_c5(h):
    N = len(h)
    nfft = 1 << (2*N-1).bit_length()
    H = np.fft.rfft(h, n=nfft)
    G = np.fft.rfft((1-h)[::-1], n=nfft)
    corr = np.fft.irfft(H*G, n=nfft)[:2*N-1]  # length 2N-1
    # Now the correlation values correspond to lag index from -N+1 to N-1.
    # Convert to max.
    max_corr = corr.max()
    return max_corr * (2.0 / N)

def _corr_array_fft(h, nfft):
    """Return full correlation array of h with (1-h) using FFT."""
    H = np.fft.rfft(h, nfft)
    Z = np.fft.rfft((1.0 - h)[::-1], nfft)

    # Core optimisation loop (simulated annealing + targeted swaps)
    it = 0
    while time.time() < deadline:
        it += 1

        # Occasionally recompute full correlation to get the worst shift.
        # We do this every iteration – cheap for N≈256.
        corr = _corr_array_fft(h, nfft)
        max_idx = int(corr.argmax())
        max_val = float(corr[max_idx])
        shift = max_idx - (N - 1)           # shift in the original definition

        # -----------------------------------------------------
        # Targeted move: try to break a (1,0) pair that contributes
        # to the current maximum.
        moved = False
        if is_binary and max_val > 0:
            if shift > 0:
                # look for i with h[i]=1, h[i+shift]=0
                mask = (h[:N - shift] > 0.5) & (h[shift:] < 0.5)
                cand = np.flatnonzero(mask)
                if cand.size:
                    i = int(rng.choice(cand))
                    j = i + shift
                    # perform a full swap (binary)
                    h[i], h[j] = h[j], h[i]
                    moved = True
            elif shift < 0:
                s = -shift
                mask = (h[s:] > 0.5) & (h[:N - s] < 0.5)
                cand = np.flatnonzero(mask)
                if cand.size:
                    i = int(rng.choice(cand)) + s
                    j = i - s
                    h[i], h[j] = h[j], h[i]
                    moved = True
            # if shift==0 we cannot improve (value always 0 for binary)

        # -----------------------------------------------------
        # If targeted move was not performed, fall back to a random
        # exchange (or fractional transfer when not binary).
        if not moved:
            # pick two distinct indices
            i, j = rng.integers(N, size=2, endpoint=False)
            if i == j:
                continue
            hi, hj = h[i], h[j]

            # maximal amount we can move from i to j without breaking bounds
            max_delta = min(hi, 1.0 - hj)
            if max_delta <= 1e-12:
                continue

            if is_binary:
                # full swap
                delta = max_delta
            else:
                # random fraction of the allowed amount
                delta = max_delta * rng.random()

            # propose change
            h[i] -= delta
            h[j] += delta

        # -----------------------------------------------------
        # Evaluate new max correlation
        new_max = _max_corr_fft(h, nfft)

        # Acceptance rule (simulated annealing)
        if new_max < cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            # temperature schedule: linear decay to 0
            elapsed = time.time() - start_time
            T0 = 0.5 * cur_max          # initial scale roughly half current max
            T = max(0.0, T0 * (1.0 - elapsed / budget_s))
            if T > 0 and np.exp(-(new_max - cur_max) / T) > rng.random():
                cur_max = new_max
                # keep the changed h (accept uphill)
                if new_max < best_max:
                    best_max = new_max
                    best_h = h.copy()
            else:
                # revert the change
                if not moved:
                    h[i] += delta
                    h[j] -= delta
                else:
                    # swap back
                    h[i], h[j] = h[j], h[i]

        # early exit if we already beat the target
        if best_max * (2.0 / N) <= target_c5:
            break

    # ---------------------------------------------------------
    # Final C5 value using the harness's evaluator if available,
    # otherwise compute directly.
    if 'evaluate_erdos_solution' in globals():
        final_c5 = float(evaluate_erdos_solution(best_h))
    else:
        final_c5 = best_max * (2.0 / N)

    return best_h, final_c5, N
