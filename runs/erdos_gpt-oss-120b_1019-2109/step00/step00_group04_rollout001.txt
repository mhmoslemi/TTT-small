import time
import numpy as np

def _compute_c5(h):
    """Return the C5 value for a given step vector h."""
    n = h.size
    dx = 2.0 / n
    # max over k of ∫ h(x) (1-h(x+k)) dx  ==  1 - min_k ∫ h(x)h(x+k)dx
    overlap = np.correlate(h, h, mode="full")
    min_overlap = overlap.min() * dx
    return float(1.0 - min_overlap)


def _gradient_for_shift(h, shift):
    """Gradient of the overlap ∑ h_i h_{i+shift} w.r.t. h, assuming zero padding."""
    n = h.size
    g = np.zeros_like(h)
    if shift > 0:
        # i + shift term
        g[: n - shift] += h[shift:]
        # i - shift term
        g[shift:] += h[: n - shift]
    elif shift < 0:
        s = -shift
        # i + shift term (i-s)
        g[s:] += h[: n - s]
        # i - shift term (i+s)
        g[: n - s] += h[s:]
    else:  # shift == 0 : derivative of sum h_i^2
        g += 2.0 * h
    return g


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] with ∫h = 1, 0≤h≤1
    to minimise max_k ∫ h(x)(1-h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1.  initialise h
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=np.float64)
        n = init_h.size
    except NameError:
        # fallback: uniform 0.5 on 42 points (integral 1)
        n = 42
        init_h = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0                     # because dx = 2/n and integral must be 1
    # ensure feasibility (clip + tiny adjustment)
    init_h = np.clip(init_h, 0.0, 1.0)
    cur_sum = init_h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # distribute the tiny discrepancy to a random feasible component
        for _ in range(10):
            idx = rng.integers(0, n)
            lo = max(-diff, -init_h[idx])
            hi = min(diff, 1.0 - init_h[idx])
            if lo <= hi:
                delta = np.clip(diff, lo, hi)
                init_h[idx] += delta
                diff -= delta
                if abs(diff) < 1e-12:
                    break

    # ------------------------------------------------------------------
    # 2.  basic helpers
    # ------------------------------------------------------------------
    def propose_random_move(h):
        """Pick two indices and transfer a feasible amount while respecting bounds."""
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)
        hi, hj = h[i], h[j]
        lower = max(-hi, hj - 1.0)
        upper = min(1.0 - hi, hj)
        if upper <= lower:
            return None, None, None  # no feasible move
        delta = rng.random() * (upper - lower) + lower
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        return new_h, i, j

    def propose_greedy_move(h, shift):
        """Move mass from the index with smallest gradient to the one with largest."""
        g = _gradient_for_shift(h, shift)
        i_max = int(np.argmax(g))
        i_min = int(np.argmin(g))
        if i_max == i_min:
            return None, None, None
        upper = min(1.0 - h[i_max], h[i_min])
        if upper <= 0:
            return None, None, None
        # take a random fraction of the feasible amount
        delta = upper * rng.random()
        new_h = h.copy()
        new_h[i_max] += delta
        new_h[i_min] -= delta
        return new_h, i_max, i_min

    # ------------------------------------------------------------------
    # 3.  optimisation loop (simulated annealing + greedy)
    # ------------------------------------------------------------------
    start_time = time.time()
    best_h = init_h.copy()
    best_c5 = _compute_c5(best_h)

    cur_c5 = best_c5

    # annealing schedule parameters
    T0 = 0.02                # initial temperature
    half_life = budget_s * 0.4   # time to halve temperature
    p_greedy = 0.6          # probability to try a directed move each iteration

    # statistics for stagnation detection
    last_improve_iter = 0
    iter_count = 0

    while True:

        if elapsed > budget_s * 0.96:
            break
        iter_count += 1

        # current temperature (exponential decay)
        temp = max(1e-8, T0 * np.exp(-elapsed / half_life))

        # decide move type
        if rng.random() < p_greedy:
            # Determine the worst shift (gives max C5)
            corr = np.correlate(cur_h, cur_h, mode="full")
            min_idx = int(corr.argmin())
            shift = min_idx - (n - 1)          # translate index to lag
            new_h, i_a, i_b = propose_greedy_move(cur_h, shift)
            if new_h is None:
                # fallback to random move
                new_h, i_a, i_b = propose_random_move(cur_h)
        else:
            new_h, i_a, i_b = propose_random_move(cur_h)

        if new_h is None:

        new_c5 = _compute_c5(new_h)
        delta = new_c5 - cur_c5

            cur_h = new_h

        # occasional gentle restart if stagnated for a long while
        if iter_count - last_improve_iter > 200000:

            pert = rng.normal(scale=0.02, size=n)
            cand = best_h + pert
            cand = np.clip(cand, 0.0, 1.0)

                cand[idx] += adj
            cur_h = cand
            cur_c5 = _compute_c5(cur_h)
            last_improve_iter = iter_count

    # ------------------------------------------------------------------
    # 4.  final sanity check & return
    # ------------------------------------------------------------------
    # final repair: ensure exact sum (tiny numerical drift)
    final_sum = best_h.sum()
    if abs(final_sum - target_sum) > 1e-10:
        diff = target_sum - final_sum
        # distribute diff to a random feasible entry
        for _ in range(10):

            hi = min(diff, 1.0 - best_h[idx])

                best_h[idx] += np.clip(diff, lo, hi)
                break

    # final bound recomputed (in case of tiny adjustments)
    final_c5 = _compute_c5(best_h)

    return best_h, final_c5, n
