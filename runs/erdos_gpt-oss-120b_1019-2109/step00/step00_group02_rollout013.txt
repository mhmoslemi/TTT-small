def project_onto_capped_simplex(v, target, ub=1.0):
    n = len(v)
    # clip to bounds after offset
    low = -np.max(v) - 1.0
    high = ub - np.min(v) + 1.0
    for _ in range(60):
        lam = (low + high) / 2.0
        y = np.clip(v + lam, 0.0, ub)
        s = y.sum()
        if s < target:
            low = lam
        else:
            high = lam
    return np.clip(v + (low+high)/2.0, 0.0, ub)

def compute_gradient(h, shift):
    n = len(h)
    g = np.zeros_like(h)
    if shift >=0:
        # term (1 - h[t+shift]) for t such that t+shift < n
        g[:n-shift] += 1 - h[shift:]
        # term -h[t-shift] for t>=shift
        g[shift:] -= h[:n-shift]
    else:
        s = -shift
        # shift negative: shift = -s
        # term (1 - h[t - s]) for t - s >=0 => t>=s
        g[s:] += 1 - h[:n-s]
        # term -h[t + s] for t + s < n => t < n-s
        g[:n-s] -= h[s:]
    return g

def project_onto_capped_simplex(v, target, ub=1.0):
    # v: numpy array
    # we find lambda such that sum_i clip(v_i + lambda, 0, ub) = target
    # binary search
    low = -np.max(v) - 1.0
    high = ub - np.min(v) + 1.0
    for _ in range(50):
        lam = (low + high) / 2.0
        y = np.clip(v + lam, 0.0, ub)
        s = y.sum()
        if s < target:
            low = lam
        else:
            high = lam
    return np.clip(v + (low + high) / 2.0, 0.0, ub)

import time, math, random
import numpy as np

def _compute_c5(h):
    """Return C5 for a step function h (discretized on [0,2])."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h), zero‑padded, scaled by dx
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _gradient(h, shift):
    """Sub‑gradient of the max‑correlation term for a given integer shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    idx = np.arange(n)
    # term (1 - h[i+shift]) where defined
    j = idx + shift
    mask = (j >= 0) & (j < n)
    g[mask] += 1.0 - h[j[mask]]
    # term -h[i-shift] where defined
    j2 = idx - shift
    mask2 = (j2 >= 0) & (j2 < n)
    g[mask2] -= h[j2[mask2]]
    return g

def _project_capped_simplex(v, target, ub=1.0, iters=60):
    """
    Euclidean projection of vector v onto {0<=x_i<=ub, sum x_i = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0
    hi = ub - np.min(v) + 1.0
    for _ in range(iters):
        lam = (lo + hi) / 2.0
        y = np.clip(v + lam, 0.0, ub)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, ub)

def _subgradient_optimize(h, max_iter, seed):
    """Run a sub‑gradient descent on h, returning the improved h."""
    random.seed(seed)
    n = h.shape[0]
    target = n / 2.0
    # step‑size schedule parameters
    base_eta = 0.6
    for it in range(max_iter):
        # correlation & locate worst shift
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        k = int(np.argmax(corr) - (n - 1))
        # gradient of the active term
        g = _gradient(h, k) * (2.0 / n)
        eta = base_eta / math.sqrt(it + 1.0)
        h = h - eta * g
        h = _project_capped_simplex(h, target, ub=1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()
    best_h = None
    best_c = float("inf")

    # ------------------------------------------------------------------
    # 1.   Evaluate any provided initial construction (if present)
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, copy=True)  # provided by the harness
        c_init = _compute_c5(init_h)
        if c_init < best_c:
            best_c, best_h = c_init, init_h.copy()
    except Exception:
        init_h = None

    # ------------------------------------------------------------------
    # 2.   Helper to generate a random feasible vector of length n
    # ------------------------------------------------------------------
    def random_feasible(n):
        v = np.random.rand(n)
        return _project_capped_simplex(v, n / 2.0, ub=1.0)

    # ------------------------------------------------------------------
    # 3.   Candidate sequence lengths (even numbers, not too large)
    # ------------------------------------------------------------------
    cand_ns = [45, 48, 56, 64, 72, 84, 96, 108]

    # ------------------------------------------------------------------
    # 4.   Main optimisation loop
    # ------------------------------------------------------------------
    outer_counter = 0
    for n in cand_ns:
        if time.time() - start > budget_s * 0.94:
            break
        target = n / 2.0
        # a few restarts per size
        restarts = 6 if n <= 64 else 4
        for r in range(restarts):
            if time.time() - start > budget_s * 0.94:
                break
            # initialise: use parent if size matches, else random
            if init_h is not None and len(init_h) == n and r == 0:
                h = init_h.copy()
                # ensure exact feasibility (projection cheap)
                h = _project_capped_simplex(h, target, ub=1.0)
            elif n in (64, 128, 256) and r == 0:
                # Thue‑Morse seed for powers of two
                seq = np.fromiter(((bin(i).count("1") & 1) for i in range(n)),
                                 dtype=float, count=n)
                h = seq  # already binary, sum = n/2 for powers of two
                h = _project_capped_simplex(h, target, ub=1.0)
            else:
                h = random_feasible(n)

            # sub‑gradient optimisation
            max_iters = 2500 if n <= 64 else 1800
            h_opt = _subgradient_optimize(h, max_iters, seed + outer_counter)
            outer_counter += 1

            # evaluate
            c_val = _compute_c5(h_opt)
            if c_val < best_c - 1e-9:
                best_c, best_h = c_val, h_opt.copy()
            # occasional early stop if we beat the target comfortably
            if best_c <= 0.3805:
                break
        if best_c <= 0.3805:
            break

    # ------------------------------------------------------------------
    # 5.   Fallback – uniform 0.5 if nothing better found (should never happen)
    # ------------------------------------------------------------------
    if best_h is None:
        n = 40
        best_h = np.full(n, 0.5)
        best_c = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # 6.   Final exact evaluation via the provided helper (if available)
    # ------------------------------------------------------------------
    try:
        best_c = float(evaluate_erdos_solution(best_h))
    except Exception:
        best_c = _compute_c5(best_h)

    return best_h, best_c, int(best_h.shape[0])
