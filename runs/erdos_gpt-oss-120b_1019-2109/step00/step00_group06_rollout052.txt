    # we can compute capacity

                break

    max_move = min(h[i], 1 - h[j])

    return new_h

def improvement_on_max_shift(h, rng, max_delta):
    n = len(h)
    cross = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(cross)
    shift = idx - (n - 1)
    # Choose random overlapping i.
    if shift >= 0:
        i = rng.integers(0, n - shift)
        j = i + shift
    else:
        i = rng.integers(-shift, n) # shift negative
        j = i + shift
    # transfer from i to j
    max_move = min(h[i], 1 - h[j])
    if max_move <= 0:
        # try swapping direction: from j to i
        max_move = min(h[j], 1 - h[i])
        if max_move <= 0:
            return h
        delta = rng.uniform(0, min(max_move, max_delta))
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
        return new_h
    else:
        delta = rng.uniform(0, min(max_move, max_delta))
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        return new_h

import time
import numpy as np


def _dx(n: int) -> float:
    """Grid spacing for n points on [0,2]."""
    return 2.0 / n


def compute_c5(h: np.ndarray) -> float:
    """C₅ = max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = _dx(n)
    # linear correlation (full) + scaling by dx
    val = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(val) * dx)


def _project_to_constraints(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce sum(h)=n/2 by a simple projection."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # iterative redistribution of the difference
    for _ in range(10):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1.0 - h
        else:
            slack = h
        total = slack.sum()
        if total == 0.0:
            break
        h += diff * (slack / total)
        h = np.clip(h, 0.0, 1.0)
    return h


def _neighbor_move(h: np.ndarray, rng: np.random.Generator, max_delta: float) -> np.ndarray:
    """Random mass transfer between two distinct indices, preserving the sum."""
    n = h.shape[0]
    new_h = h.copy()
    for _ in range(5):
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        # try moving from i to j
        max_move = min(new_h[i], 1.0 - new_h[j])
        if max_move > 0.0:
            delta = rng.uniform(0.0, min(max_move, max_delta))
            new_h[i] -= delta
            new_h[j] += delta
            return new_h
        # try opposite direction
        max_move = min(new_h[j], 1.0 - new_h[i])
        if max_move > 0.0:
            delta = rng.uniform(0.0, min(max_move, max_delta))
            new_h[j] -= delta
            new_h[i] += delta
            return new_h
    return new_h


def _improve_on_max_shift(h: np.ndarray, rng: np.random.Generator, max_delta: float) -> np.ndarray:
    """Targeted move that reduces the current worst shift."""
    n = h.shape[0]
    # correlation that yields the worst shift
    cross = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(cross))
    shift = idx - (n - 1)            # shift in index units (positive ⇒ forward)

    if shift >= 0:
        i_vals = np.arange(n - shift)
        j_vals = i_vals + shift
    else:
        i_vals = np.arange(-shift, n)
        j_vals = i_vals + shift

    # contribution of each overlapping pair to the worst shift
    terms = h[i_vals] * (1.0 - h[j_vals])
    local_max = int(np.argmax(terms))
    i = i_vals[local_max]
    j = j_vals[local_max]

    # move mass from the larger‑h index towards the smaller‑h partner
    if h[i] > h[j]:
        max_move = min(h[i], 1.0 - h[j])
        if max_move > 0.0:
            delta = rng.uniform(0.0, min(max_move, max_delta))
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
            return new_h
    else:
        max_move = min(h[j], 1.0 - h[i])
        if max_move > 0.0:
            delta = rng.uniform(0.0, min(max_move, max_delta))
            new_h = h.copy()
            new_h[j] -= delta
            new_h[i] += delta
            return new_h

    # fall back to a generic neighbor move
    return _neighbor_move(h, rng, max_delta)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function for the Erdős C₅ constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise a feasible function
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        # Upsample the supplied construction for more flexibility
        small = np.asarray(initial_h_values, dtype=np.float64)
        n_target = 300                     # chosen moderate resolution
        xs_small = np.linspace(0.0, 2.0, small.size, endpoint=False)
        xs_big = np.linspace(0.0, 2.0, n_target, endpoint=False)

        # periodic linear interpolation
        ext_x = np.concatenate((xs_small - 2.0, xs_small, xs_small + 2.0))
        ext_h = np.concatenate((small, small, small))
        h = np.interp(xs_big, ext_x, ext_h)
    else:
        # No parent: start from a simple balanced binary pattern
        n_target = 200
        h = np.zeros(n_target, dtype=np.float64)
        ones = n_target // 2
        h[:ones] = 1.0
        if n_target % 2:
            h[ones] = 0.5
        rng.shuffle(h)

    h = _project_to_constraints(h)
    n = h.shape[0]

    # ------------------------------------------------------------------
    # basic Simulated Annealing / hill climbing
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # SA parameters
    T = 0.012                # initial temperature
    T_min = 1e-6
    decay_T = 0.99993

    max_delta = 0.18
    min_delta = 0.001
    decay_delta = 0.99996

    iteration = 0
    # main loop – stop a little before the hard budget
    while True:
        if time.time() - start_time > budget_s * 0.98:
            break
        iteration += 1

        # gradually shrink step size
        max_delta = max(min_delta, max_delta * decay_delta)

        # propose a new candidate
        if rng.random() < 0.85:
            cand_h = _improve_on_max_shift(cur_h, rng, max_delta)
        else:
            cand_h = _neighbor_move(cur_h, rng, max_delta)

        cand_c5 = compute_c5(cand_h)
        diff = cand_c5 - cur_c5

        # Metropolis acceptance
        if diff < 0.0:
            accept = True
        else:
            prob = np.exp(-diff / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_h = cand_h
            cur_c5 = cand_c5
            if cur_c5 < best_c5 - 1e-12:
                best_h = cur_h.copy()
                best_c5 = cur_c5

        # temperature cooling
        T = max(T_min, T * decay_T)

        # occasional mild perturbation to escape plateaus
        if iteration % 15000 == 0:
            # start a fresh walk from the current best with a small random jitter
            jitter = rng.normal(0.0, max_delta, size=n)
            trial = np.clip(best_h + jitter, 0.0, 1.0)
            trial = _project_to_constraints(trial)
            cur_h = trial
            cur_c5 = compute_c5(cur_h)

    # final tidy‑up
    best_h = _project_to_constraints(best_h)
    best_c5 = compute_c5(best_h)

    return (best_h, float(best_c5), n)
