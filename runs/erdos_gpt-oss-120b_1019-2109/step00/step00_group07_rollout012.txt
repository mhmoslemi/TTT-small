                # try a few random swaps
                for _ in range(200):
                    i = np.random.choice(ones_idx)
                    j = np.random.choice(zeros_idx)
                    s_new = s.copy()
                    s_new[i] = 0.0
                    s_new[j] = 1.0
                    h_new = np.concatenate([s_new, s_new])
                    new_score = compute_c5(h_new)
                    inner_iters += 1
                    if new_score + 1e-12 < score:
                        s = s_new
                        score = new_score
                        improved = True
                        break

            if score < best_local_score:
                best_local_score = score
                best_local = np.concatenate([s, s])
                if best_local_score <= 0.3808:
                    return best_local, best_local_score
        return best_local, best_local_score

    # ------------------------------------------------------------------
    # Phase B – full‑length binary hill climbing
    # ------------------------------------------------------------------
    def phase_b(time_allow):
        best_local = None
        best_local_score = float('inf')
        t0 = time.time()
        while time.time() - t0 < time_allow:
            h = np.zeros(n, dtype=float)
            h[np.random.choice(n, int(target_sum), replace=False)] = 1.0
            cur_score = compute_c5(h)

            # simple swap‑based hill climbing
            for _ in range(5000):
                if time.time() - t0 >= time_allow:
                    break
                ones_idx = np.where(h == 1)[0]
                zeros_idx = np.where(h == 0)[0]
                if ones_idx.size == 0 or zeros_idx.size == 0:
                    break
                i = np.random.choice(ones_idx)
                j = np.random.choice(zeros_idx)
                h_new = h.copy()
                h_new[i] = 0.0
                h_new[j] = 1.0
                new_score = compute_c5(h_new)
                if new_score + 1e-12 < cur_score:
                    h = h_new
                    cur_score = new_score

            if cur_score < best_local_score:
                best_local_score = cur_score
                best_local = h.copy()
                if best_local_score <= 0.3808:
                    break
        return best_local, best_local_score

    # ------------------------------------------------------------------
    # Phase C – continuous simulated‑annealing style refinement
    # ------------------------------------------------------------------
    def phase_c(time_allow, start_vec):
        cur = project_feasible(start_vec.copy())
        cur_score = compute_c5(cur)
        best_loc = cur.copy()
        best_loc_score = cur_score
        t0 = time.time()
        max_iters = 2000000
        for it in range(max_iters):
            if time.time() - t0 >= time_allow:
                break
            # pick two distinct indices
            i, j = np.random.randint(0, n, size=2)
            while i == j:
                j = np.random.randint(0, n)
            hi, hj = cur[i], cur[j]

            # feasible delta that keeps both in [0,1]
            delta_min = -min(hi, 1.0 - hj)
            delta_max = min(1.0 - hi, hj)
            if delta_max <= delta_min:
                continue

            # temperature schedule (slow cooling)
            temp = 0.05 * (1.0 - it / max_iters) + 1e-5
            # propose a random delta inside the feasible interval
            delta = np.random.uniform(delta_min, delta_max) * 0.5
            cand = cur.copy()
            cand[i] = hi + delta
            cand[j] = hj - delta
            cand = np.clip(cand, 0.0, 1.0)

            # re‑project the sum (tiny correction)
            diff = cand.sum() - target_sum
            if abs(diff) > 1e-9:
                k = np.random.randint(0, n)
                while k == i or k == j:
                    k = np.random.randint(0, n)
                cand[k] = np.clip(cand[k] - diff, 0.0, 1.0)
                cand = project_feasible(cand)

            new_score = compute_c5(cand)
            if new_score < cur_score or random.random() < math.exp((cur_score - new_score) / temp):
                cur = cand
                cur_score = new_score
                if new_score < best_loc_score:
                    best_loc = cand.copy()
                    best_loc_score = new_score
                    if best_loc_score <= 0.3808:
                        break
        return best_loc, best_loc_score

    # ------------------------------------------------------------------
    # Allocate time to the three phases
    # ------------------------------------------------------------------
    total = budget_s
    # modest fractions; phases auto‑terminate early if they already beat the target
    t_a = total * 0.30
    t_b = total * 0.45
    t_c = total - (t_a + t_b)

    # Phase A
    h_a, s_a = phase_a(t_a)
    if h_a is not None and s_a < best_score:
        best_h, best_score = h_a, s_a

    # Phase B
    h_b, s_b = phase_b(t_b)
    if h_b is not None and s_b < best_score:
        best_h, best_score = h_b, s_b

    # Phase C – start from the best so far (or random if none yet)
    start_vec = best_h if best_h is not None else np.random.rand(n)
    h_c, s_c = phase_c(t_c, start_vec)
    if h_c is not None and s_c < best_score:
        best_h, best_score = h_c, s_c

    # Final safety projection and evaluation
    best_h = project_feasible(best_h)
    final_score = compute_c5(best_h)

    return (best_h, float(final_score), n)
