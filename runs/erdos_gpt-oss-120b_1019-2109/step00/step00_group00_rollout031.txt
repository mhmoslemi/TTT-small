    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    dx = 2.0 / N
    return max_corr * dx

def hill_climb(initial_h, time_limit):
    N = len(initial_h)
    target = None
    h = initial_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)
    start = time.time()

import time
import numpy as np
from scipy.optimize import differential_evolution


def _compute_c5(arr: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)(1‑h(x+k))dx  (discrete version)."""
    n = arr.shape[0]
    # correlation h with (1‑h) over full lags
    max_corr = np.max(np.correlate(arr, 1.0 - arr, mode="full"))
    return max_corr * (2.0 / n)


def _project_feasible(x: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto the box [0,1] and the hyper‑plane sum = target_sum."""
    y = np.clip(x, 0.0, 1.0)
    # simple water‑filling iteration
    for _ in range(10):
        s = y.sum()
        if abs(s - target_sum) < 1e-12:
            break
        factor = target_sum / s if s != 0 else 0.0
        y = np.clip(y * factor, 0.0, 1.0)
    # final tiny correction (adds/subtracts from a random free index)
    diff = target_sum - y.sum()
    if abs(diff) > 1e-12:
        for i in range(len(y)):
            if diff > 0 and y[i] < 1.0:
                inc = min(diff, 1.0 - y[i])
                y[i] += inc
                diff -= inc
            elif diff < 0 and y[i] > 0.0:
                dec = min(-diff, y[i])
                y[i] -= dec
                diff += dec
            if abs(diff) < 1e-12:
                break
    return y
