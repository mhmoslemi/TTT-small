def subgradient_step(h, l, alpha):
    n = len(h)
    g = np.ones_like(h)
    # shift contributions
    if l >= 0:
        # positive shift
        g[:n - l] -= h[l:]  # h_{i+l}
        g[l:] -= h[:n - l]  # h_{i-l} (since i>=l)
    else:
        # l negative: treat similarly by symmetry
        k = -l

def _compute_c5(arr, dx):
    """Return C5 for a given array using the problem's correlation definition."""
    return np.max(np.correlate(arr, 1 - arr, mode="full")) * dx

def _enforce_sum_and_bounds(x, target_sum):
    """Clip to [0,1] and then adjust to meet the exact sum constraint."""
    np.clip(x, 0.0, 1.0, out=x)
    diff = target_sum - x.sum()
    if abs(diff) < 1e-12:
        return x
    # simple redistribution of the remaining mass
    # add to entries that are not at a bound
    while abs(diff) > 1e-12:
        if diff > 0.0:
            free = np.where(x < 1.0 - 1e-12)[0]
            if free.size == 0:
                break
            inc = diff / free.size
            incs = np.minimum(inc, 1.0 - x[free])
            x[free] += incs
        else:
            free = np.where(x > 1e-12)[0]
            if free.size == 0:

def _gradient_step(h, shift, alpha, target_sum):
    """Perform one projected sub‑gradient step for the shift giving maximal overlap."""
    n = h.shape[0]
    g = np.ones_like(h)

            g[: n - shift] -= h[shift:]
            g[shift:] -= h[: n - shift]

            g[k:] -= h[: n - k]
            g[: n - k] -= h[k:]
