def projected_gradient_descent(h0, max_iters, step):

def compute_c5(h):

def projected_gd(h_init, max_iter, base_step):
    h = h_init.copy()
    best_h_local = h.copy()
    best_val_local = compute_c5(h)
    n = len(h)
    for i in range(max_iter):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        # compute max shift(s)
        idxs = np.where(corr == max_corr)[0]
        # compute shift indices
        shifts = idxs - (n - 1)
        # compute gradient sum for all
        grad_total = np.zeros_like(h)
        for s in shifts:
            grad_total += grad_for_shift(h, s)
        grad = grad_total / len(shifts)
        # step size
        step = base_step / np.sqrt(i + 1)
        h_new = h - step * grad
        # project
        h = project_capped_simplex(h_new, n/2)
        val = compute_c5(h)
        if val < best_val_local - 1e-12:
            best_val_local = val
            best_h_local = h.copy()
    return best_h_local, best_val_local

def hill_climb(h, iters, step_size):
    for _ in range(iters):
        i, j = np.random.choice(n, 2, replace=False)
        # compute allowable transfer
        delta_max = min(step_size, h[i], 1 - h[j])
        if delta_max <= 0: continue
        delta = np.random.rand() * delta_max
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        val = compute_c5(h_new)
        if val < best_val:
            h = h_new; best_val = val; best_h = h_new
    return best_h, best_val

def project_capped_simplex(v, total):
    # solve: y = clip(v + lam, 0, 1), sum(y) = total
    lo = -np.max(v) - 1.0   # ensure all v+lam <=0 -> clip to 0, sum 0 <= total
    hi = 1 - np.min(v) + 1.0 # ensure v+lam >=1 -> clip to 1, sum >= n
    # binary search 60 iterations
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if s > total:
            hi = mid
        else:
            lo = mid
    y = np.clip(v + lo, 0.0, 1.0)
    # due to numerical errors, adjust small difference:
    diff = total - y.sum()
    if abs(diff) > 1e-12:
        # distribute diff to entries not at bounds
        free = (y > 0) & (y < 1)
        if free.any():
            y[free] += diff / free.sum()
            # clip again
            y = np.clip(y, 0.0, 1.0)
        else:
            # if all at bounds, adjust first entry within bounds
            y[0] = np.clip(y[0] + diff, 0.0, 1.0)
    return y

def grad_for_shift(h, s):
    n = len(h)
    j = np.arange(n)
    idx1 = j + s
    term1 = np.ones(n)
    mask1 = (idx1 >= 0) & (idx1 < n)

    idx2 = j - s
    term2 = np.zeros(n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    term2[mask2] = h[idx2[mask2]]

def projected_gd(h0, max_iter=2000, step0=0.5):
    h = h0.copy()
    n = len(h)
    best_h = h.copy()
    best_val = compute_c5(h)
    for i in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()
        idxs = np.where(corr == max_corr)[0]
        shifts = idxs - (n - 1)
        grad_total = np.zeros_like(h)
        for s in shifts:
            grad_total += grad_for_shift(h, s)
        grad = grad_total / len(shifts)
        # step schedule
        step = step0 / np.sqrt(i + 1)
        h_new = h - step * grad

        if val < best_val:

def simulated_annealing(h0, steps=5000, temp0=0.1, temp_end=1e-6):
    h = h0.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best = cur
    n = len(h)
    for i in range(steps):
        t = temp0 * ((temp_end / temp0) ** (i / steps))
        # propose move
        i1, i2 = np.random.randint(n, size=2)
        if i1 == i2:
            continue
        # compute maximum delta that preserves bounds
        max_delta = min(h[i1], 1.0 - h[i2])
        if max_delta <= 0:
            continue
        # random delta
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[i1] -= delta
        h_new[i2] += delta
        val_new = compute_c5(h_new)

        if delta_val < 0 or np.random.rand() < np.exp(-delta_val / (t + 1e-12)):

            cur = val_new
            if val_new < best:
                best = val_new
                best_h = h_new.copy()

        w = 0.0

    h = np.empty(n)
    # left ramp
    left = xs <= a
    if a > 0:
        h[left] = p * xs[left] / a
    else:
        h[left] = 0.0
    # plateau
    plateau = (xs > a) & (xs < a + w)
    h[plateau] = p
    # right ramp
    right = xs >= a + w

        h[right] = p * (2 - xs[right]) / a

        h[right] = 0.0

    h = np.clip(h, 0.0, 1.0)
    # project
    h = project_capped_simplex(h, n/2.0)
    return h

def compute_c5(h):
    """C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def grad_for_shift(h, s):
    """Sub‑gradient of the overlap for a given integer shift s."""
    n = h.shape[0]
    j = np.arange(n)
    idx1 = j + s
    term1 = np.ones(n)
    mask1 = (idx1 >= 0) & (idx1 < n)
    term1[mask1] -= h[idx1[mask1]]

    idx2 = j - s
    term2 = np.zeros(n)
    mask2 = (idx2 >= 0) & (idx2 < n)
    term2[mask2] = h[idx2[mask2]]

    return term1 - term2


def project_capped_simplex(v, total):
    """
    Projection of v onto the set {x | 0<=x_i<=1, sum x_i = total}.
    Uses a simple 1‑D bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0          # guarantees all entries <=0 after shift
    hi = 1.0 - np.min(v) + 1.0     # guarantees all entries >=1 after shift
    for _ in range(60):
        mid = (lo + hi) * 0.5
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if s > total:
            hi = mid
        else:
            lo = mid
    y = np.clip(v + lo, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = total - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 0.0) & (y < 1.0)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
        else:
            y[0] = np.clip(y[0] + diff, 0.0, 1.0)
    return y


def projected_gd(h0, max_iter=2000, step0=0.5):
    """Projected gradient descent on the max‑overlap objective."""
    h = h0.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = compute_c5(h)

    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = corr.max()

        shifts = idxs - (n - 1)

        h_new = h - step * grad
        h = project_capped_simplex(h_new, n / 2.0)

        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    return best_h, best_val


def simulated_annealing(h0, steps=3000, temp0=0.1, temp_end=1e-6):
    """Simple SA that moves mass between two coordinates while keeping the sum."""
    h = h0.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best = cur
    n = h.shape[0]

    for k in range(steps):
        # temperature schedule (geometric)
        t = temp0 * ((temp_end / temp0) ** (k / steps))

        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            continue
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        new_val = compute_c5(h_new)
        d = new_val - cur
        if d < 0.0 or np.random.rand() < np.exp(-d / (t + 1e-12)):
            h = h_new
            cur = new_val
            if new_val < best:
                best = new_val
                best_h = h_new.copy()
    return best_h, best


def piecewise_shape(p, n):
    """
    Symmetric shape: ramp up from 0 to p over length a,
    plateau of height p and width w, ramp down.
    Parameters are chosen so that the integral equals 1.
    """
    a = 2.0 - 1.0 / p
    w = 2.0 / p - 2.0
    a = max(a, 0.0)
    w = max(w, 0.0)
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx

    h = np.empty(n)
    # left ramp
    left = xs <= a
    if a > 0.0:
        h[left] = p * xs[left] / a
    else:
        h[left] = 0.0
    # plateau
    plateau = (xs > a) & (xs < a + w)
    h[plateau] = p
    # right ramp
    right = xs >= a + w
    if a > 0.0:
        h[right] = p * (2.0 - xs[right]) / a
    else:
        h[right] = 0.0

    h = np.clip(h, 0.0, 1.0)
    h = project_capped_simplex(h, n / 2.0)
    return h


# -------------------- main entry point -------------------- #

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] that gives a small
    C5 bound. Returns (h_array, bound_value, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()

    # decide on number of discretisation points
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
        init_arr = np.clip(init_arr, 0.0, 1.0)
        init_arr = project_capped_simplex(init_arr, n / 2.0)
        starts = [init_arr]
    except Exception:
        # no parent construction – use a modest size
        n = 101
        starts = []

    # basic deterministic seeds
    const = np.full(n, 0.5, dtype=float)
    const = project_capped_simplex(const, n / 2.0)
    starts.append(const)

    # a few analytic shapes
    for p in (0.55, 0.6, 0.65, 0.7, 0.75, 0.8, 0.85, 0.9):
        starts.append(piecewise_shape(p, n))

    # random starts
    for _ in range(30):
        rnd = np.random.rand(n)
        rnd = project_capped_simplex(rnd, n / 2.0)
        starts.append(rnd)

    best_h = None
    best_val = float('inf')

    # evaluate all seeds once
    for h in starts:
        val = compute_c5(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # local optimisation from each seed
    gd_iters = 2500
    for h0 in starts:
        if time.time() - start > budget_s * 0.90:
            break
        h_opt, val_opt = projected_gd(h0, max_iter=gd_iters, step0=0.6)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()

        # a quick SA refinement
        if time.time() - start > budget_s * 0.90:
            break
        h_sa, val_sa = simulated_annealing(h_opt, steps=2000, temp0=0.05, temp_end=1e-7)
        if val_sa < best_val:
            best_val, best_h = val_sa, h_sa.copy()

    # final random hill‑climbing loops until time runs out
    while time.time() - start < budget_s * 0.97:
        # perturb the current best slightly
        h = best_h.copy()
        for _ in range(5):
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            max_delta = min(h[i], 1.0 - h[j])
            if max_delta <= 0.0:
                continue
            delta = np.random.rand() * max_delta
            h[i] -= delta
            h[j] += delta
        val = compute_c5(h)
        if val < best_val:
            best_val, best_h = val, h.copy()

    return (best_h, best_val, n)
