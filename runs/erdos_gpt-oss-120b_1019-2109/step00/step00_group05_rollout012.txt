def project_to_feasible(x):
    target = n/2
    # if sum(x) already target & x in bounds, return.
    # else binary search for lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) / 2
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x + (lo+hi)/2, 0.0, 1.0)
    # After projection, due to numeric errors sum may deviate slightly.
    # Adjust any small error by distributing among non-bound entries.
    err = target - y.sum()
    if abs(err) > 1e-12:
        # distribute err equally among free indices not at bound
        free = (y > 1e-12) & (y < 1-1e-12)
        if free.any():
            y[free] += err / free.sum()
            # clamp again to ensure bounds
            y = np.clip(y, 0.0, 1.0)
        else:
            # fallback: adjust uniformly
            y += err / len(y)
            y = np.clip(y, 0.0, 1.0)
    return y
```

    return np.max(corr) * dx

def project_to_feasible(x, target):

    lo, hi = -1.0, 1.0

        lam = (lo + hi) / 2
        y = np.clip(x + lam, 0.0, 1.0)

            lo = lam
        else:
            hi = lam

def local_search(start_h, n_iter, target, dx, rng):

def subgradient_of_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = np.max(corr)
    ks = np.flatnonzero(corr == max_val)
    grad = np.zeros(n, dtype=np.float64)
    for k in ks:
        lag = k - (n - 1)
        if lag >= 0:
            # i from 0 to n-1-lag: term1: (1 - h[i+lag]), term2: -h[i]
            grad[:n - lag] += (1 - h[lag:])  # term1 for i
            grad[lag:] += -h[:n - lag]      # term2 for i+lag? Wait careful: we need per i indexes.
            # Actually computing per index:
            # for i in 0..n-1-lag: g_i += (1 - h[i+lag])
            # for i in lag..n-1: g_i += -h[i - lag]
        else:
            lag_pos = -lag
            # lag negative: shift left; So i+lag = i - lag_pos
            # term1 for i where i - lag_pos >=0 => i >= lag_pos: (1 - h[i - lag_pos])
            # term2 for i where i + lag_pos < n => i < n - lag_pos: -h[i + lag_pos]
            # Compute.
            # For i >= lag_pos:
            grad[lag_pos:] += (1 - h[:n - lag_pos])
            # For i < n-lag_pos
            grad[:n - lag_pos] += -h[lag_pos:]
        # Loop ends
    # Average over number of ks (set of max lags)
    grad /= len(ks)
    return grad * (2.0 / n)  # multiply by dx (since objective = dx * max corr)

    # Average over ks if multiple
    grad /= len(ks)
    # Multiply by dx
    grad *= (2.0 / n)
    return grad
```

def project_to_feasible(x, target):
    # Clip by solving lambda
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Adjust any small residual
    err = target - y.sum()
    if abs(err) > 1e-12:
        # distribute to non-bound entries
        free = (y > 1e-12) & (y < 1-1e-12)
        if free.any():
            y[free] += err / np.count_nonzero(free)
        else:
            y += err / len(y)
        y = np.clip(y, 0.0, 1.0)
    return y
```

import time, math
import numpy as np

def compute_cost(h, dx):
    # max overlap integral
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_feasible(x, target):
    # Euclidean projection onto {0<=y<=1, sum(y)=target}
    lo = -np.max(x) - 1.0
    hi = 1.0 - np.min(x) + 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    err = target - y.sum()
    if abs(err) > 1e-12:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.any():
            y[free] += err / free.sum()
        else:
            y += err / y.size
        y = np.clip(y, 0.0, 1.0)
    return y

def random_feasible_vector(n, target, rng):
    x = rng.random(n)
    return project_to_feasible(x, target)

def subgradient_of_max_corr(h):
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    ks = np.flatnonzero(corr == max_val)
    grad = np.zeros_like(h)
    for k in ks:
        lag = k - (n - 1)
        if lag >= 0:
            grad[: n - lag] += 1.0 - h[lag:]
            grad[lag:] += -h[: n - lag]
        else:
            lp = -lag
            grad[lp:] += 1.0 - h[: n - lp]
            grad[: n - lp] += -h[lp:]
    grad /= len(ks)
    grad *= (2.0 / n)          # include dx
    return grad

def subgradient_descent(start_h, target, dx, max_iters=3000, step0=1.0):
    h = start_h.copy()
    cur_cost = compute_cost(h, dx)
    best_h = h.copy()
    best_cost = cur_cost
    for it in range(1, max_iters + 1):
        grad = subgradient_of_max_corr(h)
        step = step0 / math.sqrt(it)
        h_new = project_to_feasible(h - step * grad, target)
        new_cost = compute_cost(h_new, dx)
        if new_cost < cur_cost:
            h = h_new
            cur_cost = new_cost
            if new_cost < best_cost:
                best_h = h_new.copy()
                best_cost = new_cost
        # optional early stop
        if abs(cur_cost - best_cost) < 1e-12:
            break
    return best_h, best_cost

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    init_glob = globals().get('initial_h_values')
    if init_glob is not None:
        h0 = np.array(init_glob, dtype=np.float64)
        n = h0.size
    else:
        n = 70
        h0 = random_feasible_vector(n, n / 2, rng)
    target = n / 2
    dx = 2.0 / n
    h0 = project_to_feasible(h0, target)
    best_h, best_cost = subgradient_descent(h0, target, dx, max_iters=2000, step0=1.0)

    # random restarts with short descents
    while time.time() - start_time < budget_s * 0.95:
        if rng.random() < 0.4:
            start_h = random_feasible_vector(n, target, rng)
        else:
            noise = (rng.random(n) - 0.5) * 0.2
            start_h = np.clip(best_h + noise, 0.0, 1.0)
            start_h = project_to_feasible(start_h, target)
        h_opt, cost_opt = subgradient_descent(start_h, target, dx, max_iters=1500, step0=1.0)
        if cost_opt < best_cost:
            best_h, best_cost = h_opt, cost_opt
            if best_cost < 0.3808:
                break

    final_h = project_to_feasible(best_h, target)
    final_cost = compute_cost(final_h, dx)
    return final_h.tolist(), float(final_cost), n
