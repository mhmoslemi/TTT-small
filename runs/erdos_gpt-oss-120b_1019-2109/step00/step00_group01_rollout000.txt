def evaluate_c5(h):
    n = len(h)
    dx = 2.0/n
    conv = np.correlate(h, 1-h, mode='full')
    return np.max(conv) * dx

def project_to_bounds_sum(v, target):
    # binary search for shift alpha s.t. sum(clipped(v+alpha))=target
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo+hi)/2
        x = np.clip(v+mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v+(lo+hi)/2, 0.0, 1.0)

def optimize_h(initial_h, max_iter=200, seed=None):
    n = len(initial_h)
    target_sum = n/2
    # Ensure feasibility
    x0 = project_to_bounds_sum(initial_h, target_sum)
    constraints = {'type':'eq', 'fun': lambda x: np.sum(x) - target_sum}
    bounds = [(0.0,1.0)]*n
    # Use SLSQP
    res = opt.minimize(lambda x: evaluate_c5(x), x0, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter':max_iter, 'ftol':1e-12, 'disp':False})
    # Ensure feasible projection
    x_opt = project_to_bounds_sum(res.x, target_sum) if not res.success else project_to_bounds_sum(res.x, target_sum)
    val = evaluate_c5(x_opt)
    return x_opt, val, res.success

import time, math
import numpy as np
from scipy.optimize import minimize

def _proj_bounds_sum(v, target, lo=0.0, hi=1.0, tol=1e-9, it=70):
    a, b = -2.0, 2.0
    for _ in range(it):
        mid = (a + b) * 0.5
        x = np.clip(v + mid, lo, hi)
        s = x.sum()
        if s > target:
            b = mid
        else:
            a = mid
        if b - a < tol:
            break
    return np.clip(v + (a + b) * 0.5, lo, hi)

def _c5_val(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def _optimize(start, max_iter=200):
    n = start.shape[0]
    target = n * 0.5
    x0 = _proj_bounds_sum(start, target)
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target},)
    bounds = [(0.0, 1.0)] * n
    res = minimize(_c5_val, x0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    x = _proj_bounds_sum(res.x, target)
    return x, _c5_val(x)
