def project_onto_box_and_sum(a):
    # Find λ such that sum(clipped) = 1
    lo = -np.max(a) - 5.0
    hi = 1 - np.min(a) + 5.0
    for _ in range(30):

        s = proj.sum()
        if s > 1.0:
            hi = mid
        else:
            lo = mid

import time, math, numpy as np

def _proj_box_sum(a, target):
    """Project vector a onto [0,1]^n with sum == target."""
    lo = -np.max(a) - 5.0          # guarantee all entries <=0 after shift
    hi = 1.0 - np.min(a) + 5.0     # guarantee all entries >=1 after shift
    for _ in range(40):
        mid = (lo + hi) * 0.5
        x = np.clip(a + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(a + (lo + hi) * 0.5, 0.0, 1.0)

def _c5_val(h):
    """Unbiased C5 evaluation using the same formula as the harness."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full") * (2.0 / n))

def _softmax_weights(O, p):
    m = O.max()
    ex = np.exp(p * (O - m))
    s = ex.sum()
    return ex / s if s != 0 else np.full_like(O, 1.0 / O.size)

def _smooth_grad(h, w):
    """Gradient of the soft‑max of the overlap functional."""
    n = h.size
    conv1 = np.convolve(w, h, mode="full")
    conv2 = np.convolve(w[::-1], h, mode="full")
    off = n - 1
    c1 = conv1[off:off + n]
    c2 = conv2[off:off + n]
    dx = 2.0 / n
    return dx * (1.0 - (c1 + c2))

def _binary_refine(h, budget):
    """Simple hill‑climbing on a binary version of h."""
    n = h.size
    target = n // 2
    # start from the top‑target entries of h
    inds = np.argpartition(h, -target)[-target:]
    bin_h = np.zeros_like(h)
    bin_h[inds] = 1.0
    cur = _c5_val(bin_h)
    start = time.time()
    while time.time() - start < budget:
        ones = np.flatnonzero(bin_h)
        zeros = np.flatnonzero(1.0 - bin_h)
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = bin_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        val = _c5_val(cand)
        if val < cur:
            bin_h, cur = cand, val
    return bin_h, cur

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ----- problem size --------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 200                     # even, modest resolution
        h0 = np.random.rand(n)

    target_sum = n // 2            # because ∫h = 1  →  Σh·dx = 1  → Σh = n/2
    h0 = _proj_box_sum(h0 - h0.mean() + 0.5, target_sum)

    best_h = h0.copy()
    best_val = _c5_val(best_h)

    # ----- optimisation parameters ----------------------------------------
    max_iters = 2000               # per restart
    restarts = 30
    p_start, p_end = 5.0, 30.0
    step0 = float(target_sum)      # initial step magnitude

    while time.time() - start_time < budget_s - 5.0:
        # random restart
        h = _proj_box_sum(np.random.rand(n), target_sum)

        for it in range(1, max_iters + 1):
            O = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
            cur_val = O.max()
            if cur_val < best_val:
                best_val, best_h = cur_val, h.copy()
                if best_val <= 0.3808:

            # soft‑max weighting
            p = p_start + (p_end - p_start) * min(1.0, it / max_iters)
            w = _softmax_weights(O, p)

            grad = _smooth_grad(h, w)
            step = step0 / math.sqrt(it)
            h = _proj_box_sum(h - step * grad, target_sum)

            if time.time() - start_time > budget_s - 5.0:
                break

        # binary post‑processing (quick hill‑climb)
        bin_h, bin_val = _binary_refine(best_h, budget=2.0)
        if bin_val < best_val:

            if best_val <= 0.3808:
