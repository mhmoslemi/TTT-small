import time
import math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a given step function h (discrete)."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)   # dx = 2/n

def _project_sum_one_half(h: np.ndarray) -> None:
    """Adjust h in‑place so that sum(h) = len(h)/2 while staying in [0,1]."""
    n = h.size
    target = n * 0.5
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return
    # simple greedy fix using the first feasible index
    for i in range(n):
        if diff > 0 and h[i] < 1.0:
            add = min(diff, 1.0 - h[i])
            h[i] += add
            diff -= add
        elif diff < 0 and h[i] > 0.0:
            sub = min(-diff, h[i])
            h[i] -= sub
            diff += sub
        if abs(diff) < 1e-12:
            break
    # final clipping (should be already valid)
    np.clip(h, 0.0, 1.0, out=h)
