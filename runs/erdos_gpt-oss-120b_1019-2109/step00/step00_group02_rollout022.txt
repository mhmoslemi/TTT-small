def solve_for_n(n, time_limit):
    import cvxpy as cp, numpy as np, time, itertools
    # define variables
    h = cp.Variable(n, boolean=True)
    # add constraints h[0] = 1, h[1] = 0 if n > 1
    cons = [cp.sum(h) == n//2, h[0] == 1, h[1] == 0]
    M = cp.Variable()
    cons.append(M >= 0)
    # pairs
    pair_list = []
    for i in range(n):
        for j in range(i+1, n):
            pair_list.append((i, j))
    w = cp.Variable(len(pair_list), boolean=True)
    # linking constraints
    for idx, (i, j) in enumerate(pair_list):
        cons.append(w[idx] <= h[i])
        cons.append(w[idx] <= h[j])
        cons.append(w[idx] >= h[i] + h[j] - 1)
    # shift constraints
    # create mapping dict
    pair_to_idx = {pair: idx for idx, pair in enumerate(pair_list)}
    for m in range(1, n):
        # forward
        terms = []
        for i in range(n-m):
            idx = pair_to_idx[(i, i+m)]
            terms.append(h[i] - w[idx])
        cons.append(cp.sum(cp.hstack(terms)) <= M)
        # backward (negative shift)
        terms2 = []
        for i in range(n-m):
            idx = pair_to_idx[(i, i+m)]
            terms2.append(h[i+m] - w[idx])
        cons.append(cp.sum(cp.hstack(terms2)) <= M)
    # problem
    prob = cp.Problem(cp.Minimize(M), cons)
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except Exception:
        # fallback to GLPK_MI
        prob.solve(solver=cp.GLPK_MI, tm_lim=int(time_limit*1000), verbose=False)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, None

    # compute C5
    c5 = evaluate_erdos_solution(h_vals)

import time, random, numpy as np
import cvxpy as cp

# helpers ---------------------------------------------------------

def _c5_of(h):
    """Return C5 for a given step function vector."""
    # evaluate_erdos_solution is provided by the harness
    return evaluate_erdos_solution(h)


def _solve_milp(n, max_seconds):
    """Return a binary vector of length n with sum n/2 that minimizes the
    maximal overlap count.  Returns None on failure."""
    # binary decision variables for the function values
    h = cp.Variable(n, boolean=True)

    # integer decision variables for pairwise ANDs (only i<j needed)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    w = cp.Variable(len(pairs), boolean=True)
    pair_index = {p: idx for idx, p in enumerate(pairs)}

    # auxiliary variable for the maximal correlation (real is fine)
    M = cp.Variable()

    cons = [
        cp.sum(h) == n // 2,      # exactly half 1's
        h[0] == 1,                # break complement symmetry
        h[1] == 0,                # break reversal symmetry
        M >= 0
    ]

    # linearisation of w_{ij} = h_i * h_j for i<j
    for (i, j), idx in pair_index.items():
        cons.append(w[idx] <= h[i])
        cons.append(w[idx] <= h[j])
        cons.append(w[idx] >= h[i] + h[j] - 1)

    # constraints for all positive and negative shifts
    for m in range(1, n):
        idxs = [pair_index[(i, i + m)] for i in range(n - m)]

        # forward shift (+m)
        terms_fwd = [h[i] - w[idx] for i, idx in zip(range(n - m), idxs)]
        cons.append(cp.sum(terms_fwd) <= M)

        # backward shift (-m)
        terms_bwd = [h[i + m] - w[idx] for i, idx in zip(range(n - m), idxs)]
        cons.append(cp.sum(terms_bwd) <= M)

    prob = cp.Problem(cp.Minimize(M), cons)

    # try a few common MILP solvers
    for solver in (cp.CBC, cp.GLPK_MI, cp.ECOS_BB, cp.SCIP):
        try:
            prob.solve(solver=solver, max_seconds=max_seconds, verbose=False)
        except Exception:  # pragma: no cover
            continue
        if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
            break
    else:
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None

    # round to 0/1 and enforce exact half‑sum
    h_val = np.rint(h.value).astype(int)
    h_val = (h_val > 0).astype(int)

    cur = h_val.sum()
    target = n // 2
    if cur != target:
        diff = target - cur
        # flip a few bits to restore the exact count
        zeros = np.where(h_val == 0)[0]
        ones = np.where(h_val == 1)[0]
        if diff > 0:
            np.random.shuffle(zeros)
            h_val[zeros[:diff]] = 1
        elif diff < 0:
            np.random.shuffle(ones)
            h_val[ones[: -diff]] = 0
    return h_val


def _hill_climb(start_h, max_seconds):
    """Simple stochastic swap search to improve a binary vector."""
    best_h = start_h.copy()
    best_c5 = _c5_of(best_h)
    n = len(best_h)
    t0 = time.time()
    while time.time() - t0 < max_seconds:
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best_h.copy()
        cand[i] = 0
        cand[j] = 1
        c5 = _c5_of(cand)
        if c5 < best_c5 - 1e-12:
            best_h, best_c5 = cand, c5
    return best_h, best_c5

    """Return a step function (vector), its C5 bound and its length."""
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()

    best_h = None

    # initial construction, if supplied
