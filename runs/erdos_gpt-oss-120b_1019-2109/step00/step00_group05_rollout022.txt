import numpy as np, time, math

def evaluate_h(h):
    dx = 2.0 / len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def propose_move(h, rng, max_delta):
    n = len(h)
    # find sources where h > 0 (maybe > 1e-12)
    src = rng.choice(np.where(h > 1e-12)[0])
    tgt = rng.choice(np.where(h < 1 - 1e-12)[0])
    # compute allowable delta
    delta_max = min(max_delta, h[src], 1 - h[tgt])
    if delta_max <= 0:
        return h  # no move
    delta = rng.random() * delta_max
    h_new = h.copy()
    h_new[src] -= delta
    h_new[tgt] += delta
    return h_new

def random_initial(n, rng):
    # binary random arrangement with exactly n/2 ones
    m = n // 2
    h = np.zeros(n)
    indices = rng.choice(n, size=m, replace=False)
    h[indices] = 1.0
    return h

import numpy as np
import math
import time

# ---------- utility functions ----------
def eval_c5(h: np.ndarray) -> float:
    """Return the C5 bound for a given step function h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with 1-h, mode='full' gives all overlaps
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx


def repair_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h in-place so that sum(h) == target while staying in [0,1]."""
    diff = target - h.sum()
    # simple greedy fix – move mass from the largest entries to the smallest
    while abs(diff) > 1e-12:
        if diff > 0:
            # need to increase total sum
            idx = int(np.argmax(1.0 - h))          # biggest headroom
            inc = min(diff, 1.0 - h[idx])
            h[idx] += inc
            diff -= inc
        else:
            # need to decrease total sum
            idx = int(np.argmax(h))                # largest value
            dec = min(-diff, h[idx])
            h[idx] -= dec
            diff += dec
    return h


def propose_small_move(h: np.ndarray, rng: np.random.Generator, max_delta: float) -> np.ndarray:
    """Transfer a small amount of mass from one index to another."""
    n = h.size
    # source must have some mass, target must have room
    src_candidates = np.where(h > 1e-12)[0]
    tgt_candidates = np.where(h < 1.0 - 1e-12)[0]
    if src_candidates.size == 0 or tgt_candidates.size == 0:
        return h.copy()
    i = rng.choice(src_candidates)
    j = rng.choice(tgt_candidates)
    delta = min(max_delta, h[i], 1.0 - h[j])
    if delta <= 0.0:
        return h.copy()
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


def propose_large_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Replace a random entry by a fresh value and compensate elsewhere."""
    n = h.size
    i = rng.integers(n)
    new_val = rng.random()
    delta = new_val - h[i]
    if abs(delta) < 1e-15:
        return h.copy()
    # need a partner j to keep the sum unchanged
    if delta > 0:
        # take delta from some entry that is at least delta
        eligible = np.where(h > delta)[0]
        if eligible.size == 0:
            return h.copy()
        j = rng.choice(eligible)
        h_new = h.copy()
        h_new[i] = new_val
        h_new[j] -= delta
    else:
        # give -delta to an entry that can receive it
        need = -delta
        eligible = np.where(h < 1.0 - need)[0]
        if eligible.size == 0:
            return h.copy()
        j = rng.choice(eligible)
        h_new = h.copy()
        h_new[i] = new_val
        h_new[j] += need
    return h_new


def generate_sin_shape(n: int) -> np.ndarray:
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = 0.5 + 0.5 * np.sin(np.pi * x / 2.0)   # ranges between 0 and 1, mean 0.5
    return repair_sum(h, n / 2.0)


def generate_sin2_shape(n: int) -> np.ndarray:
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * x / 2.0) ** 2          # also mean 0.5, values in [0,1]
    return repair_sum(h, n / 2.0)


def generate_binary_start(n: int, rng: np.random.Generator) -> np.ndarray:
    h = np.zeros(n, dtype=float)
    ones = n // 2
    idx = rng.choice(n, size=ones, replace=False)
    h[idx] = 1.0
    return h

    while time.time() - start_time < budget_s - 0.12:

        if cand_c5 < current_c5 or rng.random() < math.exp(-(cand_c5 - current_c5) / (temp + 1e-12)):
            current_h = cand_h
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = cand_h.copy()

        # Cool down
        temp *= cooling

    # ------------------------------------------------------------------
    # Final safety repair (tiny numeric drift may appear)
    best_h = repair_sum(best_h, target_sum)

    final_c5 = eval_c5(best_h)
    # Return the result as required
    return best_h, final_c5, n
