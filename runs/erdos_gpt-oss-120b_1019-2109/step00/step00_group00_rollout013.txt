def compute_term1(h, w):
    n = len(h)
    shifts = np.arange(-(n-1), n)
    term = np.zeros(n)
    for idx, k in enumerate(shifts):
        wk = w[idx]
        if wk == 0:

            term[:n-k] += wk * h[k:]

    lo = -np.max(v) - 1.0  # not used

    # lam high (very positive) => x_i = 0 => sum =0.
    # So we can bisection over lam to achieve target_sum.
    lo = -1.0  # lam low enough that all v_i - lam >= 1
    hi = 1.0   # lam high enough that all v_i - lam <= 0

        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0, 1)
    # Ensure sum within tolerance:
    # After projection, due to clipping may not exactly equal target; adjust small residual.

def project_to_sum(v, target):
    # ensure 0 <= v <=1 after projection onto box and hyperplane
    lo = np.min(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # adjust small residual
    residual = target - x.sum()
    if abs(residual) > 1e-10:
        # distribute to non-saturated elements
        free = np.where((x > 1e-12) & (x < 1-1e-12))[0]
        if free.size > 0:
            delta = residual/ free.size
            x[free] += delta
            x = np.clip(x, 0.0, 1.0)
    return x

import time
import numpy as np

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of v onto the set
        {x | 0 <= x_i <= 1, sum(x) = target }.
    Uses a water‑filling/bisection on the Lagrange multiplier.
    """
    lo = np.min(v - 1.0)          # λ low enough → all x_i = 1
    hi = np.max(v)                # λ high enough → all x_i = 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny residual correction for numeric drift
    resid = target - x.sum()
    if abs(resid) > 1e-10:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            delta = resid / free.size
            x[free] += delta
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_value(h: np.ndarray) -> float:
    """Exact C5 for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

def _softmin_weights(matches: np.ndarray, alpha: float) -> np.ndarray:
    """
    Returns a probability vector proportional to exp(-α·matches).
    The smallest match receives the largest weight.
    """
    m_min = matches.min()
    exps = np.exp(-alpha * (matches - m_min))
    return exps / exps.sum()

def _gradient_step(h: np.ndarray, target: float, alpha: float, lr: float) -> np.ndarray:
    """
    One projected gradient descent step for the smooth min‑matches objective.
    """
    # matches = correlation of h with itself (no scaling needed for the gradient)
    matches = np.correlate(h, h, mode="full")
    w = _softmin_weights(matches, alpha)               # length 2n‑1
    # term_i = Σ_k w_k * h_{i+k}  (symmetry of w lets us use convolution)
    conv = np.convolve(w, h, mode="full")
    term = conv[h.shape[0] - 1 : h.shape[0] - 1 + h.shape[0]]
    grad = -2.0 * term                                 # ∂f/∂h
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _project_to_sum(h_new, target)
    return h_new

def _optimize(h0: np.ndarray, budget: float, start: float) -> tuple:
    """
    Local optimisation starting from h0, respecting the time budget.
    Returns the best vector seen and its C5 value.
    """
    n = h0.shape[0]
    target_sum = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(h)
    alpha = 5.0
    lr = 0.5
    itr = 0
    while True:
        now = time.time()
        if now - start > budget:
            break
        h = _gradient_step(h, target_sum, alpha, lr)
        if itr % 200 == 0:
            val = _c5_value(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
            # slowly sharpen focus and shrink step‑size
            alpha = min(alpha * 1.03, 40.0)
            lr = max(lr * 0.99, 0.01)
        itr += 1
    return best_h, best_val

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --------------------------------------------------------------
    # Choose the initial vector.
    # --------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.shape[0]
        target = n / 2.0
        h0 = _project_to_sum(h0, target)
    else:
        n = 80                     # even size, modest yet flexible
        target = n / 2.0
        h0 = np.zeros(n, dtype=float)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0

    best_h, best_val = h0.copy(), _c5_value(h0)

    # --------------------------------------------------------------
    # Multiple restarts with small random perturbations.
    # --------------------------------------------------------------
    attempts = 6
    for k in range(attempts):
        remaining = budget_s - (time.time() - start_time)
        if remaining <= 0:
            break
        # give each attempt a slice of the remaining budget
        alloc = remaining / (attempts - k + 0.5)

        # start from the currently best solution, add a tiny perturbation
        h_start = best_h.copy()
        if np.all((h_start == 0) | (h_start == 1)):
            # binary – perform a few random swaps
            swap_cnt = max(1, int(0.02 * n))
            ones = np.where(h_start == 1)[0]
            zeros = np.where(h_start == 0)[0]
            if ones.size >= swap_cnt and zeros.size >= swap_cnt:
                o = rng.choice(ones, swap_cnt, replace=False)
                z = rng.choice(zeros, swap_cnt, replace=False)
                h_start[o] = 0.0
                h_start[z] = 1.0
        else:
            # continuous – add Gaussian noise then re‑project
            h_start += rng.normal(scale=0.04, size=n)
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = _project_to_sum(h_start, target)

        # run local optimiser
        h_opt, val_opt = _optimize(h_start, alloc, start_time)
        if val_opt < best_val:
            best_val, best_h = val_opt, h_opt.copy()

    # --------------------------------------------------------------
    # Final tidy‑up: enforce constraints exactly and evaluate.
    # --------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_sum(best_h, n / 2.0)
    final_val = _c5_value(best_h)

    # In the unlikely event the optimisation failed to improve the
    # supplied construction, fall back to it (it already satisfies the
    # bound 0.38085 according to the problem statement).
    try:
        init_val = _c5_value(initial_h_values)
        if init_val < final_val:
            best_h = np.asarray(initial_h_values, dtype=float).copy()
            final_val = init_val
    except Exception:
        pass

    return (best_h, final_val, n)
