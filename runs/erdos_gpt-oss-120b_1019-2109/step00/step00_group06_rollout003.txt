def propose_targeted_swap(h):
    conv = np.correlate(h, 1-h, mode='full')
    kmax = np.argmax(conv)
    shift = kmax - (len(h)-1)
    n = len(h)
    if shift >= 0:
        start = 0
        end = n - shift
        # i indexes considered: i from start to end-1
        bad1 = [i for i in range(start, end) if h[i]==1 and h[i+shift]==0]
        bad0 = [i for i in range(start, end) if h[i]==0 and h[i+shift]==1]
    else:
        s = -shift
        start = s
        end = n
        bad1 = [i for i in range(start, end) if h[i]==1 and h[i-s]==0]
        bad0 = [i for i in range(start, end) if h[i]==0 and h[i-s]==1]
    if bad1 and bad0:
        i = random.choice(bad1)
        j = random.choice(bad0)
        # Ensure we maintain indices within bounds where mismatch lists defined.
        # Swap them
        return i, j
    else:
        # fallback random swap
        # find any 1 index and 0 index
        ones = np.where(h==1)[0]
        zeros = np.where(h==0)[0]
        if len(ones)==0 or len(zeros)==0:
            return None
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

def init_binary_h(n, init_arr=None, rng=np.random.default_rng()):
    if init_arr is not None:
        # Interpolate to length n
        old_n = len(init_arr)
        # positions along [0,2)
        old_x = np.linspace(0,2,old_n,endpoint=False)
        new_x = np.linspace(0,2,n,endpoint=False)
        interp = np.interp(new_x, old_x, init_arr)
        h = (interp > 0.5).astype(float)
    else:
        h = np.zeros(n, dtype=float)
        ones_indices = rng.choice(n, size=n//2, replace=False)
        h[ones_indices] = 1.0
    # adjust to exact n/2 ones if needed
    cur_ones = int(h.sum())
    target_ones = n // 2
    if cur_ones > target_ones:
        # turn off some ones
        idx = np.where(h==1)[0]
        off = rng.choice(idx, size=cur_ones - target_ones, replace=False)
        h[off] = 0
    elif cur_ones < target_ones:
        idx = np.where(h==0)[0]
        on = rng.choice(idx, size=target_ones - cur_ones, replace=False)
        h[on] = 1
    return h

```
def hill_climb(h, time_limit, target_c5=None):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_max = compute_conv_max_counts(best_h)[1]
    best_c5 = best_max * dx
    current_h = best_h.copy()
    current_max = best_max

    start = time.time()
    rng = np.random.default_rng()
    T = 0.05  # starting temperature for occasional acceptance
    iter = 0
    while time.time() - start < time_limit:
        # compute conv for current
        conv = np.correlate(current_h, 1-current_h, mode='full')
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        shift = max_idx - (n - 1)
        # propose target swap
        # compute mismatched indices lists for this shift
        if shift >= 0:
            start_i = 0
            end_i = n - shift
            bad1 = np.where((current_h[start_i:end_i]==1) & (current_h[start_i+shift:end_i+shift]==0))[0] + start_i
            bad0 = np.where((current_h[start_i:end_i]==0) & (current_h[start_i+shift:end_i+shift]==1))[0] + start_i
        else:
            s = -shift
            start_i = s
            end_i = n
            bad1 = np.where((current_h[start_i:end_i]==1) & (current_h[start_i-s:end_i-s]==0))[0] + start_i
            bad0 = np.where((current_h[start_i:end_i]==0) & (current_h[start_i-s:end_i-s]==1))[0] + start_i

        if bad1.size > 0 and bad0.size > 0:
            i = rng.choice(bad1)
            j = rng.choice(bad0)
        else:
            # random swap - pick random 1 and random 0
            ones_idx = np.where(current_h==1)[0]
            zeros_idx = np.where(current_h==0)[0]
            if ones_idx.size == 0 or zeros_idx.size == 0:
                break
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)

        # candidate
        cand = current_h.copy()
        cand[i], cand[j] = cand[j], cand[i]

        # evaluate
        cand_max = np.correlate(cand, 1-cand, mode='full').max()
        if cand_max < current_max or rng.random() < np.exp((current_max - cand_max) / T):
            current_h = cand
            current_max = cand_max
            if cand_max < best_max:
                best_h = cand.copy()
                best_max = cand_max
                best_c5 = best_max * dx
                # optionally break if target reached
                if target_c5 is not None and best_c5 <= target_c5:
                    break
        # schedule temperature
        iter += 1
        # simple cooling
        if iter % 1000 == 0:
            T = max(T * 0.5, 1e-5)
    return best_h, best_c5
```

def hill_climb(h_start, time_limit, target_c5=None):
    import numpy as np, time, math
    n = len(h_start)
    dx = 2.0 / n
    best_h = h_start.copy()
    best_max = np.correlate(best_h, 1 - best_h, mode='full').max()
    best_c5 = best_max * dx
    current_h = best_h.copy()
    # compute initial conv
    conv = np.correlate(current_h, 1 - current_h, mode='full')
    current_max = conv.max()
    start = time.time()
    rng = np.random.default_rng()
    no_improve = 0
    restart_limit = 5000  # iterations without improvement before restart
    iterations = 0
    # Temperature for occasional uphill moves
    T = 0.01
    while time.time() - start < time_limit:
        # shift detection
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)
        # produce indices lists
        if shift >= 0:
            i_start = 0
            i_end = n - shift
            h_i = current_h[i_start:i_end]
            h_j = current_h[i_start+shift:i_end+shift]
        else:
            s = -shift
            i_start = s
            i_end = n
            h_i = current_h[i_start:i_end]
            h_j = current_h[i_start - s:i_end - s]   # because i+shift = i - s
        # compute mismatched indices
        bad1 = np.where((h_i == 1) & (h_j == 0))[0] + i_start
        bad0 = np.where((h_i == 0) & (h_j == 1))[0] + i_start
        if bad1.size > 0 and bad0.size > 0:
            i = rng.choice(bad1)
            j = rng.choice(bad0)
        else:
            # random swap 1 and 0
            ones = np.where(current_h == 1)[0]
            zeros = np.where(current_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
        # Swap
        cand = current_h.copy()
        cand[i], cand[j] = cand[j], cand[i]
        # evaluate candidate
        cand_max = np.correlate(cand, 1 - cand, mode='full').max()
        # acceptance criterion
        accept = False
        if cand_max < current_max:
            accept = True
        else:
            # occasional uphill
            # probability decreases with T
            if rng.random() < math.exp((current_max - cand_max) / T):
                accept = True
        if accept:
            current_h = cand
            current_max = cand_max
            conv = np.correlate(current_h, 1 - current_h, mode='full')
            # update best
            if cand_max < best_max:
                best_h = cand.copy()
                best_max = cand_max
                best_c5 = best_max * dx
                no_improve = 0
                # early exit if target achieved
                if target_c5 is not None and best_c5 <= target_c5:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1
        # restart if stuck
        if no_improve >= restart_limit:
            # reinitialize randomly
            cand2 = np.zeros(n, dtype=float)
            cand2[rng.choice(n, size=n//2, replace=False)] = 1.0
            current_h = cand2
            conv = np.correlate(current_h, 1 - current_h, mode='full')
            current_max = conv.max()
            no_improve = 0
        # cooling schedule
        iterations += 1
        if iterations % 2000 == 0:
            T = max(T * 0.5, 1e-6)
    return best_h, best_c5

    """Create a binary vector of length n with exactly n/2 ones.
    If init_arr is given, interpolate it onto n points and threshold."""
    if init_arr is not None:
        old_n = init_arr.shape[0]
        if old_n != n:
            old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n, endpoint=False)
            interp = np.interp(new_x, old_x, init_arr)
            h = (interp > 0.5).astype(float)
        else:
            h = (init_arr > 0.5).astype(float)
    else:
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, size=n // 2, replace=False)
        h[ones] = 1.0

    # adjust to exact half‑ones
    cur = int(h.sum())
    target = n // 2
    if cur != target:
        if cur > target:
            idx = np.where(h == 1.0)[0]
            off = rng.choice(idx, size=cur - target, replace=False)
            h[off] = 0.0
        else:
            idx = np.where(h == 0.0)[0]
            on = rng.choice(idx, size=target - cur, replace=False)
            h[on] = 1.0
    return h


def hill_climb(h_start, time_limit, target_c5=None, rng=None):
    """Local search that swaps a 1 with a 0 to reduce the maximal overlap."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(h_start)
    dx = 2.0 / n

    best_h = h_start.copy()
    best_max = np.correlate(best_h, 1 - best_h, mode="full").max()
    best_c5 = best_max * dx

    cur_h = best_h.copy()
    conv = np.correlate(cur_h, 1 - cur_h, mode="full")
    cur_max = conv.max()

    start = time.time()
    no_improve = 0
    restart_limit = 5000
    T = 0.01
    it = 0

    while time.time() - start < time_limit:
        # shift that currently attains the maximum
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)

        # indices that contribute to the current max mismatch
        if shift >= 0:
            a0, a1 = 0, n - shift
            h_i = cur_h[a0:a1]
            h_j = cur_h[a0 + shift:a1 + shift]
        else:
            s = -shift
            a0, a1 = s, n
            h_i = cur_h[a0:a1]
            h_j = cur_h[a0 - s:a1 - s]

        bad1 = np.where((h_i == 1) & (h_j == 0))[0] + a0
        bad0 = np.where((h_i == 0) & (h_j == 1))[0] + a0

        if bad1.size and bad0.size:
            i = rng.choice(bad1)
            j = rng.choice(bad0)
        else:
            ones = np.where(cur_h == 1.0)[0]
            zeros = np.where(cur_h == 0.0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)

        cand = cur_h.copy()
        cand[i], cand[j] = cand[j], cand[i]

        cand_max = np.correlate(cand, 1 - cand, mode="full").max()

        # accept if better, otherwise with small probability
        accept = False
        if cand_max < cur_max:
            accept = True
        else:
            if rng.random() < math.exp((cur_max - cand_max) / T):
                accept = True

        if accept:
            cur_h = cand
            conv = np.correlate(cur_h, 1 - cur_h, mode="full")
            cur_max = conv.max()
            if cand_max < best_max:
                best_h = cand.copy()
                best_max = cand_max
                best_c5 = best_max * dx
                no_improve = 0
                if target_c5 is not None and best_c5 <= target_c5:
                    break
            else:
                no_improve += 1
        else:
            no_improve += 1

        it += 1
        if no_improve >= restart_limit:
            # random restart
            cur_h = np.zeros(n, dtype=float)
            cur_h[rng.choice(n, size=n // 2, replace=False)] = 1.0
            conv = np.correlate(cur_h, 1 - cur_h, mode="full")
            cur_max = conv.max()
            no_improve = 0

        if it % 2000 == 0:
            T = max(T * 0.5, 1e-6)

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C₅ bound, and the number of sample points."""
    start_total = time.time()
    rng = np.random.default_rng(seed)

    # length of discretisation (must be even)
    n = 200
    init_arr = globals().get("initial_h_values")
    if init_arr is not None:
        init_arr = np.array(init_arr, dtype=float)

    best_h = None
    best_c5 = math.inf

    # allocate most of the budget to repeated local searches
    while time.time() - start_total < budget_s * 0.95:
        time_left = budget_s - (time.time() - start_total)
        if time_left < 5.0:
            break
        # each local search gets at most 30 s or 60 % of remaining time
        local_time = min(30.0, time_left * 0.6)

        h0 = generate_initial_binary(n, init_arr, rng)
        h_local, c5_local = hill_climb(h0, local_time, target_c5=0.3808, rng=rng)

        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local
            if best_c5 <= 0.3808:
                break

    # fallback if nothing succeeded
    if best_h is None:
        best_h = generate_initial_binary(n, init_arr, rng)
        best_c5 = compute_c5(best_h)

    # final verification and return
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), n
