def project_to_feasible(v, S):
    # returns x = clamp(v - lam, 0, 1) where lam solves sum(x) = S
    # If sum after simple clipping is >S we need to reduce, else increase.
    # Use binary search for lam in [-M, M] where M large enough.
    # Actually x_i = clamp(v_i - lam, 0, 1). So monotonic decreasing in lam.
    # We search lam such that sum(x) = S.
    low = -1.0
    high = 1.0
    # Expand bounds to ensure solution lies within.
    # We can compute min(v_i-1) and max(v_i)
    # Actually lam can be up to max(v_i) because v_i - lam <=0 => clamp to 0 for all i => sum 0.
    # So we can set low = -max(v_i) (makes v_i - lam large positive -> clamp 1). Not exactly.
    # Let's just compute feasible range:
    # For lam = -inf, v_i - lam -> +inf, clamp->1 => sum=n
    # For lam = +inf, clamp->0 => sum=0.
    # So solution always exists for any S in [0,n].
    # Use low = -max(v_i) - 2 (to ensure all >1 maybe) and high = max(v_i) + 2.
    low = -2.0
    high = 2.0
    # Expand if needed.
    for _ in range(60):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > S:
            low = lam
        else:
            high = lam
    x = np.clip(v - (low + high)/2.0, 0.0, 1.0)
    return x

def local_search(h, time_budget):
    start = time.time()
    n = len(h)
    cur_h = h.copy()
    cur_val = evaluate(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    it = 0
    while time.time() - start < time_budget:
        it += 1
        # compute correlation.
        conv = np.correlate(cur_h, 1 - cur_h, mode='full')
        idx = np.argmax(conv)
        max_val = conv[idx] * (2.0/ len(cur_h))
        shift = idx - (len(cur_h) - 1)
        # try target move
        moved = False
        if shift != 0:
            # get shifted array
            if shift > 0:
                shifted = np.empty_like(cur_h)
                shifted[:n-shift] = cur_h[shift:]
                shifted[n-shift:] = 0
            else:
                s = -shift
                shifted = np.empty_like(cur_h)
                shifted[s:] = cur_h[:n-s]
                shifted[:s] = 0
            bad_indices = np.where((cur_h == 1) & (shifted == 0))[0]
            good_indices = np.where((cur_h == 0) & (shifted == 1))[0]
            if bad_indices.size > 0 and good_indices.size > 0:
                i = np.random.choice(bad_indices)
                j = np.random.choice(good_indices)
                # swap
                new_h = cur_h.copy()
                new_h[i], new_h[j] = new_h[j], new_h[i]
                new_val = evaluate(new_h)
                if new_val < cur_val:
                    cur_h, cur_val = new_h, new_val
                    moved = True
                    if new_val < best_val:
                        best_h, best_val = new_h.copy(), new_val
                        # maybe no need else
        if not moved:
            # random swap as fallback
            # choose random positions with opposite values
            idx1 = np.random.randint(n)
            idx2 = np.random.randint(n-1)
            if idx2 >= idx1:
                idx2 += 1
            if cur_h[idx1] == cur_h[idx2]:
                continue
            new_h = cur_h.copy()
            new_h[idx1], new_h[idx2] = new_h[idx2], new_h[idx1]
            new_val = evaluate(new_h)
            # accept if better or with bounded probability.
            if new_val < cur_val:
                cur_h, cur_val = new_h, new_val
                if new_val < best_val:
                    best_h, best_val = new_h.copy(), new_val
            else:
                # optional annealing
                # Use temperature schedule decreasing from 0.01 to 0.
                # compute elapsed fraction
                elapsed = time.time() - start
                frac = elapsed / time_budget
                T = 0.01 * (1 - frac)
                if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                    cur_h, cur_val = new_h, new_val
        # optionally break after many iterations without improvement
        # but not essential

    return best_h, best_val

def evaluate(h):
    """Return the C5 value for a step function h given as a NumPy array."""
    n = h.shape[0]
    dx = 2.0 / n
    # full cross‑correlation of h with (1-h); numpy correlates with zero‑padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def project_to_sum(v, target):
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane sum==target.
    The solution is x_i = clip(v_i - λ, 0, 1) for a suitable λ found by bisection.
    """
    lo, hi = -2.0, 2.0          # wide enough to force all entries to 1 or 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def shifted_view(h, shift):
    """
    Return an array `s` of the same length as h such that s[i] = h[i+shift] if in range,
    otherwise 0.  This is exactly what np.correlate expects for the zero‑padded shift.
    """
    n = h.shape[0]
    s = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            s[:n - shift] = h[shift:]
    else:
        s_abs = -shift
        if s_abs < n:
            s[s_abs:] = h[:n - s_abs]
    return s


def thue_morse(n):
    """Generate the Thue‑Morse binary sequence of length n (values 0 or 1)."""
    # parity of number of 1‑bits in i (mod 2)
    seq = np.fromiter((bin(i).count("1") & 1 for i in range(n)), dtype=np.float64)
    return seq


# ------------------------------------------------------------
# Core local‑search / simulated‑annealing routine
# ------------------------------------------------------------

def improve_sequence(init_h, time_budget, rng):
    """
    Starting from a binary array init_h (0/1) with exactly n/2 ones,
    try to minimise the C5 value using a mix of deterministic
    “bad‑good” swaps and simulated‑annealing random swaps.
    """
    n = init_h.shape[0]
    cur_h = init_h.copy()
    cur_val = evaluate(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    # temperature schedule: linear from T0 down to ~0
    T0 = 0.015

    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_budget:
            break
        it += 1

        # ------------------------------------------------------------------
        # 1) locate the shift that currently attains the maximum value
        # ------------------------------------------------------------------
        corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = int(corr_full.argmax())
        max_corr = corr_full[idx_max] * (2.0 / n)
        shift = idx_max - (n - 1)

        moved = False

        # ------------------------------------------------------------------
        # 2) try a “bad‑good” swap that directly improves this worst shift
        # ------------------------------------------------------------------
        if shift != 0:
            shifted = shifted_view(cur_h, shift)
            bad = np.where((cur_h == 1) & (shifted == 0))[0]
            good = np.where((cur_h == 0) & (shifted == 1))[0]
            if bad.size and good.size:
                i = rng.choice(bad)
                j = rng.choice(good)
                new_h = cur_h.copy()
                new_h[i], new_h[j] = 0, 1          # swap a 1 with a 0
                new_val = evaluate(new_h)
                if new_val <= cur_val:
                    cur_h, cur_val = new_h, new_val
                    moved = True
                    if new_val < best_val:
                        best_h, best_val = new_h.copy(), new_val
                # fall back to random move if not improving
        # ------------------------------------------------------------------
        # 3) fallback: random swap with simulated‑annealing acceptance
        # ------------------------------------------------------------------
        if not moved:
            # pick two distinct positions with opposite values
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            if cur_h[i] == cur_h[j]:
                # nothing to swap – skip this iteration
                continue
            new_h = cur_h.copy()
            new_h[i], new_h[j] = cur_h[j], cur_h[i]   # swap
            new_val = evaluate(new_h)

            # acceptance rule
            if new_val < cur_val:
                accept = True
            else:
                # temperature decays linearly
                frac = elapsed / time_budget
                T = T0 * (1.0 - frac)
                accept = T > 0 and rng.random() < math.exp(-(new_val - cur_val) / T)

            if accept:
                cur_h, cur_val = new_h, new_val
                if new_val < best_val:
                    best_h, best_val = new_h.copy(), new_val

    return best_h, best_val


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function h on [0,2] to minimise the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    # --------------------------------------------------------
    # Choose a starting length and construct an initial vector
    # --------------------------------------------------------
    try:
        # use the supplied parent construction if present
        init_parent = initial_h_values  # provided by the evaluation harness
        n = int(init_parent.shape[0])
        # Clip to [0,1] and project onto the feasible set (sum = n/2)
        h0 = np.clip(np.asarray(init_parent, dtype=np.float64), 0.0, 1.0)
        target_sum = n / 2.0
        if not np.isclose(h0.sum(), target_sum, atol=1e-9):
            h0 = project_to_sum(h0, target_sum)
        # ensure binary for simplicity – round while preserving total sum
        # (if it is already binary we keep it)
        if not np.all(np.logical_or(h0 == 0, h0 == 1)):
            # convert to a binary vector with the same number of ones
            ones_needed = int(round(target_sum))
            idx = rng.permutation(n)[:ones_needed]
            h0[:] = 0.0
            h0[idx] = 1.0
    except NameError:
        # No parent supplied – build a fresh candidate
        # Prefer a power‑of‑two length (nice for Thue‑Morse)
        n = 256
        h0 = thue_morse(n)
        # ensure exact half‑ones (Thue‑Morse already satisfies this for powers of two)
        # but floating rounding may produce 0/1, so cast
        h0 = h0.astype(np.float64)

    # --------------------------------------------------------
    # Run the improvement routine for the allocated time budget
    # --------------------------------------------------------
    # Reserve a small margin for final evaluation/return
    time_for_opt = budget_s * 0.94
    best_h, best_val = improve_sequence(h0, time_for_opt, rng)

    # --------------------------------------------------------
    # Return the result in the requested format
    # --------------------------------------------------------
    return best_h, float(best_val), int(n)
