def propose(h):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # Determine max increase for i and max decrease for j
    max_inc = 1.0 - h[i]
    max_dec = h[j]
    # We can increase i and decrease j.
    # choose delta <= min(max_inc, max_dec)
    max_delta = min(max_inc, max_dec)
    # also we could do opposite direction: increase j, decrease i.
    # Randomly choose direction
    if np.random.rand() < 0.5:
        # inc i, dec j
        direction = 1
    else:
        # inc j, dec i
        direction = -1
    if direction == 1:
        if max_delta <= 0:
            return None
        delta = np.random.rand() * max_delta
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
    else:
        # inc j, dec i
        max_inc2 = 1.0 - h[j]
        max_dec2 = h[i]
        max_delta2 = min(max_inc2, max_dec2)
        if max_delta2 <= 0:
            return None
        delta = np.random.rand() * max_delta2
        h_new = h.copy()
        h_new[j] += delta
        h_new[i] -= delta
    return h_new

        return v

    # We'll iterate adjusting non-bound entries proportionally.
    # Simple: while diff not small, adjust first free element.
    # Safer approach: compute mask of non-bound entries.
    mask = (v > 0.0 + 1e-12) & (v < 1.0 - 1e-12)
    free_idx = np.where(mask)[0]
    if free_idx.size == 0:
        # all at bounds, force by editing first element
        idx = 0
        v[idx] -= diff
        v[idx] = np.clip(v[idx], 0.0, 1.0)
        return v

    delta = diff / free_idx.size
    # Clip each adjustment to avoid leaving bounds
    for idx in free_idx:
        new = v[idx] - delta
        if new < 0.0:
            new = 0.0
        elif new > 1.0:
            new = 1.0
        v[idx] = new
    # Recursively fix remaining diff
    return _project_sum(v, target)

import time
import random
import math
import numpy as np

def _eval_c5(arr: np.ndarray) -> float:
    """Return the C5 value for a step function given by *arr*."""
    n = arr.size
    dx = 2.0 / n
    # correlation with zero padding (the same used by the judge)
    r = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(r) * dx)


def _project_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project *v* onto the feasible set:
        0 ≤ v[i] ≤ 1  and  sum(v) = target
    Simple water‐filling with recursion – more than enough for n≈1000.
    """
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    diff = cur - target
    if abs(diff) < 1e-12:
        return v
    # indices that are not at a bound
    free = np.where((v > 1e-12) & (v < 1.0 - 1e-12))[0]
    if free.size == 0:
        # all entries are at a bound – force a small correction on the first element
        v[0] = np.clip(v[0] - diff, 0.0, 1.0)
        return v
    # distribute the excess uniformly over the free entries
    delta = diff / free.size
    for idx in free:
        v[idx] -= delta
        if v[idx] < 0.0:
            v[idx] = 0.0
        elif v[idx] > 1.0:
            v[idx] = 1.0
    # recurse until the sum matches (typically one extra call)
    return _project_sum(v, target)
