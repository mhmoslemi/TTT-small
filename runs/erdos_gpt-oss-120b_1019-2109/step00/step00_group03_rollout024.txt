def compute_gradient_for_lag(h, L):
    N = len(h)
    g = np.empty(N)
    # term1: (1 - h[i+L]) if within bounds else 1.
    if L >= 0:
        # part where i+L < N
        g[:N-L] = 1 - h[L:]  # for i=0..N-L-1
        # part where i+L >= N
        g[N-L:] = 1.0
        # term2: h[i-L] where i - L >= 0 else 0
        # i >= L => index i-L exists
        term2 = np.zeros(N)
        term2[L:] = h[:N-L]
        # gradient = term1 - term2
        g -= term2
    else:
        # L negative: define delta = -L positive
        delta = -L
        # term1 for i+L in bounds => i >= delta
        g[delta:] = 1 - h[:N-delta]  # because i+L = i - delta, so h index = i - delta
        g[:delta] = 1.0
        # term2: h[i-L] = h[i+delta] for i+delta < N => i<N-delta
        term2 = np.zeros(N)
        term2[:N-delta] = h[delta:]
        g -= term2
    return g

import numpy as np, time, math

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def worst_lag_info(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr)
    max_val = corr[max_idx] * dx
    L = max_idx - (len(h) - 1)
    return L, max_val

def gradient_for_lag(h, L):
    N = len(h)
    g = np.empty(N)
    if L >= 0:
        # term1
        g[:N-L] = 1 - h[L:]
        g[N-L:] = 1.0
        # term2
        term2 = np.zeros(N)
        term2[L:] = h[:N-L]
        g -= term2
    else:
        delta = -L
        g[delta:] = 1 - h[:N-delta]
        g[:delta] = 1.0
        term2 = np.zeros(N)
        term2[:N-delta] = h[delta:]
        g -= term2
    return g

def optimize_h(initial_h, S, dx, time_budget):
    start = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)
    N = len(h)
    alpha = 0.1
    rng = np.random.default_rng(12345) # we could use seed but may use other
    no_improve = 0
    while time.time() - start < time_budget * 0.95:   # keep margin
        # compute worst lag and cost
        L, cur_val = worst_lag_info(h, dx)
        # compute gradient for that lag
        grad = gradient_for_lag(h, L)
        # propose step
        v = h - alpha * grad
        # project
        h_new = project_onto_capped_simplex(v, S)
        new_val = compute_c5(h_new, dx)
        # acceptance criteria
        if new_val <= cur_val:
            h = h_new
            cur_val = new_val
            # maybe increase step size, but keep bounded
            alpha = min(alpha * 1.02, 1.0)
            # record best
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                no_improve = 0
        else:
            # reduce step size
            alpha *= 0.5
            no_improve += 1

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------
def compute_c5(h, dx):
    """C₅ value for a given step vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def worst_lag_info(h, dx):
    """Return (lag, C₅) for the lag attaining the maximum overlap."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argmax(corr)
    lag = idx - (len(h) - 1)                     # shift index (can be negative)
    return lag, corr[idx] * dx


def gradient_for_lag(h, lag):
    """Gradient of the overlap term corresponding to *lag*."""
    N = len(h)
    g = np.empty(N, dtype=float)

    if lag >= 0:
        # part where i+lag is inside the array
        g[: N - lag] = 1.0 - h[lag:]
        # part where i+lag is outside → h_{i+lag}=0 ⇒ (1‑h)=1
        g[N - lag :] = 1.0
        # second contribution: -h_{i‑lag}
        term2 = np.zeros(N, dtype=float)
        term2[lag:] = h[: N - lag]
        g -= term2
    else:
        d = -lag
        # i+lag inside → i-d ≥ 0
        g[d:] = 1.0 - h[: N - d]
        g[:d] = 1.0
        term2 = np.zeros(N, dtype=float)
        term2[: N - d] = h[d:]
        g -= term2
    return g


def project_onto_capped_simplex(v, target_sum, tol=1e-9, max_iter=60):
    """
    Euclidean projection of *v* onto the set
        { w | 0 ≤ w_i ≤ 1 , sum w_i = target_sum }.
    """
    lo = -np.max(v) - 2.0          # guarantees v+lo ≤ 0
    hi = 1.0 - np.min(v) + 2.0     # guarantees v+hi ≥ 1

    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
        if hi - lo < tol:
            break
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    # final tiny adjustment in case of numerical drift
    diff = target_sum - w.sum()
    if abs(diff) > 1e-10:
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w


# ------------------------------------------------------------
# Core optimizer (projected sub‑gradient on the worst lag)
# ------------------------------------------------------------
def optimise_h(initial_h, target_sum, dx, budget):
    start = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h, dx)

    N = len(h)
    rng = np.random.default_rng(123456)

    # step‑size schedule
    step = 0.5
    no_improve = 0

    while time.time() - start < budget * 0.94:
        # worst shift and its gradient
        lag, cur_val = worst_lag_info(h, dx)
        grad = gradient_for_lag(h, lag)

        # trial move
        trial = h - step * grad
        trial = project_onto_capped_simplex(trial, target_sum)

        trial_val = compute_c5(trial, dx)

        # accept if we improve the current worst‑lag value
        if trial_val <= cur_val:
            h = trial
            if trial_val < best_val:
                best_val = trial_val
                best_h = h.copy()
                no_improve = 0
            step = min(step * 1.02, 2.0)
        else:
            step *= 0.5
            no_improve += 1

        # occasional random shake‑up to escape plateaus
        if no_improve >= 2000 or rng.random() < 0.001:
            noise = (rng.random(N) - 0.5) * 0.04
            shaken = np.clip(h + noise, 0.0, 1.0)
            shaken = project_onto_capped_simplex(shaken, target_sum)
            shaken_val = compute_c5(shaken, dx)
            if shaken_val < best_val:
                best_val = shaken_val
                best_h = shaken.copy()
                h = shaken
                step = 0.5
            no_improve = 0

    # --------------------------------------------------------
    # Final local refinement by tiny pairwise swaps
    # --------------------------------------------------------
    refine_iters = 5000
    h = best_h.copy()
    for _ in range(refine_iters):
        i, j = rng.integers(0, N, size=2)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 0.02
        hi = h[i] - delta
        hj = h[j] + delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue
        h_candidate = h.copy()
        h_candidate[i] = hi
        h_candidate[j] = hj
        # sum unchanged, only bounds need checking
        cand_val = compute_c5(h_candidate, dx)
        if cand_val < best_val:
            best_val = cand_val
            best_h = h_candidate.copy()
            h = h_candidate

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where
        h_values – list of step heights,
        c5_bound – the achieved upper bound for C₅,
        n_points – number of discretisation points.
    The routine respects the supplied time budget.
    """
    # --------------------------------------------------------------------
    # Determine the discretisation size and an initial feasible vector.
    # --------------------------------------------------------------------
    if 'initial_h_values' in globals():
        n = int(len(initial_h_values))
        h0 = np.asarray(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
    else:
        n = 200                     # a modest size that stays well below the 1 k limit
        idx = np.arange(n)
        mid = n // 2
        # a symmetric triangular start (area 1 ⇒ sum = n/2 automatically)
        h0 = np.where(idx <= mid, idx / mid, (n - idx) / mid).astype(float)

    target_sum = n / 2.0
    dx = 2.0 / n

    # ensure the mandatory sum constraint exactly (projection onto the feasible set)
    h0 = project_onto_capped_simplex(h0, target_sum)

    # --------------------------------------------------------------------
    # Run the optimisation within the remaining budget.
    # --------------------------------------------------------------------
    start = time.time()
    remaining = budget_s - (time.time() - start)
    best_h, best_val = optimise_h(h0, target_sum, dx, remaining)

    # final safety projection (numeric drift)
    best_h = project_onto_capped_simplex(best_h, target_sum)
    best_val = compute_c5(best_h, dx)

    return (best_h.tolist(), float(best_val), n)
