import numpy as np, time, random, math

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c.max()

```python
def hill_climb(h, max_iter, time_limit, start_time):
    n = len(h)
    best = h.copy()
    best_score = compute_c5(best)
    # Precompute indices of ones and zeros
    one_idx = np.where(best == 1)[0]
    zero_idx = np.where(best == 0)[0]
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        # random swap
        i = random.choice(one_idx)
        j = random.choice(zero_idx)
        # swap
        best[i], best[j] = 0, 1
        new_score = compute_c5(best)
        if new_score < best_score - 1e-12:
            best_score = new_score
            # Update indices
            one_idx = np.where(best == 1)[0]
            zero_idx = np.where(best == 0)[0]
        else:
            # swap back
            best[i], best[j] = 1, 0
    return best, best_score
```

                h[free] += diff / free.size

import time
import numpy as np

def _c5_of(h):
    """Return the C5 value for a given 1‑D array h."""
    n = h.size
    dx = 2.0 / n
    # correlation of h with (1-h), zero‑padded outside the interval
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())

def _triangular_h(n):
    """Construct a piecewise‑linear triangular step function with exactly n points."""
    dx = 2.0 / n
    # mid‑point of each sub‑interval
    xs = (np.arange(n) + 0.5) * dx          # values in (0,2)
    h = np.where(xs <= 1.0, xs, 2.0 - xs)   # h(x)=x for x≤1, h(x)=2−x otherwise
    # The midpoint rule integrates a linear function exactly,
    # therefore sum(h) should already be n/2 (up to round‑off).
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny correction over the interior points (where 0<h<1)
        interior = np.where((h > 0) & (h < 1))[0]
        if interior.size:
            h[interior] += diff / interior.size
        else:
            h[0] += diff
        np.clip(h, 0.0, 1.0, out=h)
    return h
