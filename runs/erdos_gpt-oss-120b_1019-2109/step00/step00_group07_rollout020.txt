def project_box_sum(y, target):
    # Box [0,1] and sum = target.
    # Use bisection on λ.
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo+hi)/2.0, 0.0, 1.0)
    # due to clipping, sum may not be exactly target but near. Enforce slight correction:
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # distribute diff among non-boundary entries
        free_idx = (x > 0) & (x < 1)
        if free_idx.any():
            # equally distribute
            adjustment = diff / free_idx.sum()
            x[free_idx] += adjustment
            # clip again
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: raise or just scale all values.
            x = np.clip(x + diff / len(x), 0.0, 1.0)
    return x

import time, math
import numpy as np

def _project_box_sum(y, target):
    """Project y onto the box [0,1]^n while enforcing sum(y)=target."""
    lo = -np.max(y) - 1.0          # makes y+lo <= 0  ⇒ all zeros
    hi = 1.0 - np.min(y) + 1.0     # makes y+hi >= 1  ⇒ all ones
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(y + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if rounding left a residual
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x = np.clip(x + diff / len(x), 0.0, 1.0)
    return x


def _c5_value(h):
    """Compute the C5 bound for a discrete step function h."""
    dx = 2.0 / len(h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _gradient_for_shift(h, shift):
    """Sub‑gradient of the worst‑shift term at current h."""
    n = len(h)
    g = np.ones(n)
    if shift >= 0:
        g[: n - shift] -= h[shift:]
        g[shift:]     -= h[: n - shift]
    else:
        s = -shift
        g[s:]         -= h[: n - s]
        g[: n - s]    -= h[s:]
    return g
