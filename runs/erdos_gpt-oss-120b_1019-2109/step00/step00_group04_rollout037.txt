def compute_c5(h):
    n = h.shape[0]
    hp = 1 - h
    corr = np.correlate(h, hp, mode='full')
    max_val = np.max(corr) * (2.0 / n)
    return max_val

def solve_binary_milp(n, time_limit=100):
    import cvxpy as cp, numpy as np
    x = cp.Variable(n, boolean=True)
    # build y list
    pairs = []
    for i in range(n):
        for j in range(i+1, n):
            pairs.append((i,j))
    m = len(pairs)
    y = cp.Variable(m, boolean=True)
    # mapping from pair index
    pair_dict = {pair: idx for idx, pair in enumerate(pairs)}
    # constraints
    constraints = [cp.sum(x) == n//2]
    # linking constraints
    for idx, (i,j) in enumerate(pairs):
        constraints.append(y[idx] <= x[i])
        constraints.append(y[idx] <= x[j])
        constraints.append(y[idx] >= x[i] + x[j] - 1)
    t = cp.Variable(integer=True)
    constraints.append(t >= 0)
    # for each shift
    for k in range(1, n):
        # forward
        idx_range = range(0, n - k)
        forward_expr = cp.sum([x[i] for i in idx_range]) - cp.sum([y[pair_dict[(i,i+k)]] for i in idx_range])
        constraints.append(forward_expr <= t)
        # backward
        backward_expr = cp.sum([x[i + k] for i in idx_range]) - cp.sum([y[pair_dict[(i,i+k)]] for i in idx_range])
        constraints.append(backward_expr <= t)
    # objective
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    # solve with time limit
    prob.solve(solver=cp.CBC, mip_gap=0.01, timeLimit=time_limit)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        raise RuntimeError("MILP not solved")
    xval = np.round(x.value).astype(float)  # ensure 0/1
    tval = int(round(t.value))
    return xval, tval

def refine_continuous(h0, n, remaining_time):
    import numpy as np, time
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    step0 = 1.0
    iter = 0
    while time.time() - start < remaining_time and iter < 5000:
        # compute correlation and lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        lag = max_idx - (n - 1)
        # compute gradient
        g = np.zeros_like(h)
        if lag >= 0:
            L = n - lag
            g[:L] += 1 - h[lag:]    # B shifted left
            g[lag:] += -h[:L]       # -A shifted right
        else:
            k = -lag
            L = n - k
            g[k:] += 1 - h[:L]      # B from left side
            g[:L] += -h[k:]         # -A from right side

        if c5 < best_c5 - 1e-9:

            best_h = h.copy()
        iter += 1

import time
import numpy as np
import cvxpy as cp

def compute_c5(h: np.ndarray) -> float:
    """C5 bound using the same evaluator as the harness."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def project_onto_sum_and_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while forcing sum(v)=target_sum."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def gradient_of_max_corr(h: np.ndarray) -> np.ndarray:
    """Gradient of the correlation term that attains the current max."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    lag = max_idx - (n - 1)               # lag = shift amount
    g = np.zeros_like(h)

    if lag >= 0:
        L = n - lag
        # derivative w.r.t. h_i for i in [0, L)
        g[:L] += 1.0 - h[lag:]            # (1 - h_{i+lag})
        # derivative w.r.t. h_i for i in [lag, n)
        g[lag:] += -h[:L]                 # -h_{i-lag}
    else:
        k = -lag
        L = n - k
        g[k:] += 1.0 - h[:L]              # (1 - h_{i-k}) on the right side
        g[:L] += -h[k:]                   # -h_{i+k} on the left side
    return g


def refine_continuous(start_h: np.ndarray, remaining_time: float) -> tuple:
    """Sub‑gradient descent on the max‑correlation objective."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    h = start_h.copy()
    best_h = h.copy()
    best_val = compute_c5(h)

    start = time.time()
    it = 0
    while time.time() - start < remaining_time and it < 5000:
        g = gradient_of_max_corr(h)
        step = 0.6 / np.sqrt(it + 1.0)
        h = h - step * g
        h = project_onto_sum_and_bounds(h, target_sum)

        val = compute_c5(h)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
        it += 1
    return best_h, best_val


def solve_binary_milp(n: int, max_seconds: float = 60.0) -> tuple:
    """Binary {0,1} pattern with minimal max overlap (exact MILP)."""
    # variable for positions
    x = cp.Variable(n, boolean=True)

    # all unordered pairs (i<j)
    pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
    m = len(pairs)
    y = cp.Variable(m, boolean=True)
    pair_index = {p: idx for idx, p in enumerate(pairs)}

    constraints = [cp.sum(x) == n // 2]

    # linking y = x_i * x_j
    for idx, (i, j) in enumerate(pairs):
        constraints.append(y[idx] <= x[i])
        constraints.append(y[idx] <= x[j])
        constraints.append(y[idx] >= x[i] + x[j] - 1)

    t = cp.Variable(integer=True)
    constraints.append(t >= 0)

    # forward and backward shift constraints
    for k in range(1, n):
        L = n - k
        idx_range = range(L)
        y_idx = [pair_index[(i, i + k)] for i in idx_range]

        forward = cp.sum([x[i] for i in idx_range]) - cp.sum([y[j] for j in y_idx])
        backward = cp.sum([x[i + k] for i in idx_range]) - cp.sum([y[j] for j in y_idx])
        constraints.append(forward <= t)
        constraints.append(backward <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # try CBC, fall back to GLPK_MI if unavailable
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_iters=100000)
    except Exception:
        prob.solve(solver=cp.GLPK_MI, verbose=False)

    if prob.status not in ("optimal", "optimal_inaccurate"):
        raise RuntimeError("MILP did not solve to optimality")

    x_val = np.round(x.value).astype(float)
    t_val = int(round(t.value))
    return x_val, t_val


def resize_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a step function by linear interpolation."""
    old_n = old_h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # enforce exact integral after interpolation
    target_sum = new_n / 2.0
    new_h = project_onto_sum_and_bounds(new_h, target_sum)
    return new_h


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_val = float("inf")
    best_n = None

    # list of problem sizes to attempt (small for speed)
    sizes = [42, 48, 54, 60]

    # if a parent construction exists, use its size as a starting point
    try:
        parent = np.array(initial_h_values, dtype=float)
        parent_n = parent.shape[0]
        if parent_n not in sizes:
            sizes.insert(0, parent_n)
    except Exception:
        parent = None

    # iterate over chosen sizes
    for n in sizes:
        if time.time() - start_time > budget_s - 5:
            break

        # obtain an initial vector
        if parent is not None and parent.shape[0] == n:
            h0 = parent.copy()
        elif parent is not None:
            h0 = resize_h(parent, n)
        else:
            # smooth cosine start (average 0.5, integral 1)
            x = np.linspace(0.0, 2.0, n, endpoint=False)
            h0 = 0.5 * (1.0 + np.cos(np.pi * x))
            h0 = project_onto_sum_and_bounds(h0, n / 2.0)

        # --- Phase 1: integer MILP (if n is modest) -----------------
        if n <= 54:
            try:
                milp_h, _ = solve_binary_milp(n)
                val = compute_c5(milp_h)
                if val < best_val:
                    best_val, best_h, best_n = val, milp_h, n
                # refine from MILP solution (continuous)
                time_left = budget_s - (time.time() - start_time) - 2
                if time_left > 1.0:
                    refined_h, refined_val = refine_continuous(milp_h, time_left)
                    if refined_val < best_val:
                        best_val, best_h, best_n = refined_val, refined_h, n
            except Exception:
                # MILP failed – fall back to purely continuous search
                pass

        # --- Phase 2: continuous refinement from smooth start -------
        time_left = budget_s - (time.time() - start_time) - 2
        if time_left > 1.0:
            refined_h, refined_val = refine_continuous(h0, time_left)
            if refined_val < best_val:
                best_val, best_h, best_n = refined_val, refined_h, n

    # fallback: if nothing succeeded, fall back to a constant 0.5 function
    if best_h is None:
        n = 40
        best_n = n
        best_h = np.full(n, 0.5)
        best_val = compute_c5(best_h)

    return best_h, best_val, best_n
