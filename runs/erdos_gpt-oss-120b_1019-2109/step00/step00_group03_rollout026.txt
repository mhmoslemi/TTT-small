def project_onto_capped_simplex(y, m):
    # y: np.ndarray float
    # returns x such that min=0, max=1, sum=m.
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > m:
            lo = lam
        else:
            hi = lam
    x = np.clip(y - (lo+hi)/2.0, 0.0, 1.0)
    # adjust for any residual numerical error by scaling?
    # Ensure sum exactly m: after clipping, sum may be slightly off; we can correct by adjusting the largest free entries.
    diff = m - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff to entries not at bounds
        free = (x>0) & (x<1)
        if free.any():
            # add diff equally among free entries
            inc = diff / free.sum()
            x[free] += inc
            # after inc, still may exceed bounds slightly; clip again
            x = np.clip(x,0,1)
    # final scaling may still not sum to m due to rounding; adjust
    # Simple iterative fix:
    diff = m - x.sum()
    if abs(diff) > 1e-9:
        # adjust by adding/subtracting to a random free entry
        # choose index not at bound
        indices = np.where((x > 1e-12) & (x < 1-1e-12))[0]
        if len(indices) == 0:
            # if all at bounds, adjust by distributing to any
            indices = np.arange(len(x))
        # add diff to first index (or distribute)
        i = indices[0]
        x[i] = np.clip(x[i] + diff, 0, 1)
    # final check
    return x

import numpy as np

def project_onto_capped_simplex(v, target):
    # binary search for lambda
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # adjust small residual
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # distribute residual to entries not at bound
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if mask.any():
            inc = diff / mask.sum()
            x[mask] += inc
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback: adjust first element
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

import numpy as np

def _project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the set
        {x | 0 <= x_i <= 1, sum(x) = target}.
    Uses a simple binary search for the Lagrange multiplier λ.
    """
    lo = np.min(v) - 1.0          # if λ <= lo, all entries become 1
    hi = np.max(v)                # if λ >= hi, all entries become 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)

    # tiny correction for possible numeric drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            inc = diff / mask.sum()
            x[mask] += inc
            x = np.clip(x, 0.0, 1.0)
        else:
            # fallback – adjust the first element
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Construct a linear ramp that yields C₅ ≈ 0.328 (< 0.3808)."""
    n = 70                              # number of discretisation points
    # linear ramp: h[i] = i/(n-1)  (values from 0 to 1 inclusive)
