def random_feasible(n, target_sum, rng):
    # generate random vector in [0,1] with sum target using Dirichlet and scaling then clipping iterative
    # simpler: start with zeros, fill random indices with random fractions
    # We can sample uniform random numbers in [0,1] for each entry, then scale to target sum and clip.
    v = rng.random(n)
    # scale v so sum = target_sum
    v = v * (target_sum / v.sum())
    # clip may break sum; adjust using residual approach:
    # if any >1, set to 1 and redistribute excess.
    excess = v - 1
    if np.any(v > 1):
        # reduce those >1 to 1
        over = v > 1
        excess_amount = np.sum(v[over] - 1)
        v[over] = 1.0
        # distribute excess proportionally among underfilled entries
        under = v < 1
        if np.any(under):
            v[under] += excess_amount * (v[under] / np.sum(v[under]))
    # after this, sum may deviate; fix small residual by adding to random index.
    total = v.sum()
    diff = target_sum - total
    # adjust diff to ensure boundaries
    i = rng.integers(n)
    # ensure we can add diff within [0,1]
    v[i] = np.clip(v[i] + diff, 0, 1)
    # final adjustment to meet sum exactly: we skip.
    return v

    rng = np.random.default_rng(seed)
    start = time.time()
    # obtain initial h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:

        diff = target_sum - cur_sum

            change = min(abs(diff), cap)
            if diff > 0:
                h0[idx] += change
                diff -= change
            else:
                h0[idx] -= change
                diff += change
            if abs(diff) < 1e-12:
                break

    h0 = np.clip(h0,0,1)

                if time.time() - start > budget_s * 0.9:
                    raise StopIteration

                if res.success:

                diff += sub

import time
import math
import numpy as np

def _evaluate_h(h: np.ndarray) -> float:
    """Return the C5 value for a step function h (discretized)."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h); mode='full' matches the harness definition
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points."""
    rng = np.random.default_rng(seed)
    start = time.time()

    # ------------------------------------------------------------------ #
    #   initialise h --------------------------------------------------- #
    # ------------------------------------------------------------------ #
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
    else:
        # fallback: a half‑filled binary vector of length 42
        n = 42
        h = np.zeros(n, dtype=float)
        ones = rng.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    n = h.shape[0]
    target_sum = n / 2.0  # because ∫h = 1 ⇒ sum(h) = n/2

    # enforce the linear equality exactly (tiny adjustments)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) > 1e-12:
        # distribute the needed mass greedily
        for i in range(n):
            if diff > 0:
                cap = 1.0 - h[i]
                add = min(cap, diff)
                h[i] += add
                diff -= add
            else:
                cap = h[i]
                sub = min(cap, -diff)
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    h = np.clip(h, 0.0, 1.0)

    best_h = h.copy()
    best_C = _evaluate_h(best_h)

    # ------------------------------------------------------------------ #
    #   optional local optimisation with SLSQP -------------------------- #
    # ------------------------------------------------------------------ #
    time_left = budget_s - (time.time() - start)
    if time_left > 15.0:
        try:
            from scipy.optimize import minimize

            bounds = [(0.0, 1.0) for _ in range(n)]
            eq_con = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}

            def obj(x):
                # no need for the dx factor here – monotone
                return np.max(np.correlate(x, 1.0 - x, mode="full"))

            res = minimize(
                obj,
                best_h,
                method="SLSQP",
                bounds=bounds,
                constraints=[eq_con],
                options={"maxiter": 800, "ftol": 1e-9},
            )
            if res.success:
                cand = np.clip(res.x, 0.0, 1.0)
                # tiny correction of the equality constraint
                s = cand.sum()
                if abs(s - target_sum) > 1e-9:
                    cand *= target_sum / s
                cand = np.clip(cand, 0.0, 1.0)
                Cc = _evaluate_h(cand)
                if Cc < best_C:
                    best_h, best_C = cand, Cc
        except Exception:
            pass  # SciPy may be unavailable – continue with heuristic search

    # ------------------------------------------------------------------ #
    #   heuristic refinement (targeted + annealing) -------------------- #
    # ------------------------------------------------------------------ #
    cur_h = best_h.copy()
    cur_C = best_C

    # initial temperature proportional to the current bound
    T0 = best_C * 0.5
    total_time = budget_s

    # helper for a cheap sum‑correction at the end of a move
    def _repair(arr):
        """Make sure sum(arr) == target_sum while staying in [0,1]."""
        diff = target_sum - arr.sum()
        if abs(diff) < 1e-12:
            return arr
        # distribute a tiny correction to a random feasible coordinate
        for idx in rng.permutation(n):
            if diff > 0:
                cap = 1.0 - arr[idx]
                add = min(cap, diff)
                arr[idx] += add
                diff -= add
            else:
                cap = arr[idx]
                sub = min(cap, -diff)
                arr[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
        return arr

    while True:
        elapsed = time.time() - start
        if elapsed > total_time * 0.97:
            break

        # correlation and the shift giving the current maximum
        corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # contribution of each index to that maximal correlation
        idx = np.arange(n)
        partner = idx + shift
        inside = (partner >= 0) & (partner < n)
        term = np.empty(n, dtype=float)
        term[inside] = cur_h[idx[inside]] * (1.0 - cur_h[partner[inside]])
        term[~inside] = cur_h[idx[~inside]]

        src = int(np.argmax(term))
        dst = int(np.argmin(term))
        if src == dst:
            # fallback to a random pair
            src, dst = rng.integers(n, size=2)
            while src == dst:
                dst = rng.integers(n)

        # maximal feasible transfer from src to dst
        delta_max = min(cur_h[src], 1.0 - cur_h[dst])
        if delta_max <= 0.0:
            # try the opposite direction
            delta_max = min(cur_h[dst], 1.0 - cur_h[src])
            if delta_max > 0.0:
                src, dst = dst, src
            else:
                # give up this iteration
                continue

        # temperature schedule (linear cooling)
        T = T0 * max(0.0, 1.0 - elapsed / (total_time * 0.97))

        # choose a random fraction of the allowed delta
        delta = delta_max * rng.random()

        # propose move
        new_h = cur_h.copy()
        new_h[src] -= delta
        new_h[dst] += delta

        new_C = _evaluate_h(new_h)

        # Metropolis acceptance
        accept = False
        if new_C < cur_C:
            accept = True
        else:
            if T > 0.0:
                prob = math.exp(-(new_C - cur_C) / (T + 1e-15))
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_h = new_h
            cur_C = new_C
            if new_C < best_C:
                best_h = new_h.copy()
                best_C = new_C

        # occasional random perturbation to escape possible plateaus
        if rng.random() < 0.001:
            i, j = rng.integers(n, size=2)
            if i != j:
                delta_max2 = min(cur_h[i], 1.0 - cur_h[j])
                if delta_max2 > 0.0:
                    d2 = delta_max2 * rng.random()
                    cur_h[i] -= d2
                    cur_h[j] += d2
                    cur_C = _evaluate_h(cur_h)
                    if cur_C < best_C:
                        best_h = cur_h.copy()
                        best_C = cur_C

    # final repair to guarantee the integral exactly equals 1
    best_h = _repair(best_h)
    best_C = _evaluate_h(best_h)

    return best_h.tolist(), float(best_C), n
