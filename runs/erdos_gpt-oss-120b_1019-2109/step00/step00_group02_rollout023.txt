def _project_to_bounds_and_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] with sum(v)=target using bisection."""
    lo, hi = -1e6, 1e6
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid

def _c5_and_shifts(h: np.ndarray):
    """Return C₅ and the list of integer shifts attaining the maximum."""
    n = h.shape[0]
    c = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(c)
    # shift indices relative to zero shift (center of the full correlation)
    shifts = np.where(c == max_c)[0] - (n - 1)
    return max_c * (2.0 / n), shifts
