def compute_c5(h):

def max_shift_and_value(h):
    n = len(h)

def project_to_box_simplex(v, target_sum):
    # projection onto [0,1]^n with sum = target_sum
    # Using bisection for lambda.
    lo = -1.0  # lower bound
    hi = 2.0   # upper bound (ensures v - hi <= -1, clamp to 0)

        s = x.sum()
        if s > target_sum:

        else:

    return x

import time
import numpy as np

def _proj_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the set {x | 0 <= x_i <= 1, sum x_i = target}."""
    # dual variable lambda searched by bisection
    lo, hi = -1.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # final tiny correction (rare)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-12) & (x < 1 - 1e-12))[0]
        if free.size:
            add = diff / free.size
            x[free] = np.clip(x[free] + add, 0.0, 1.0)
    return x


def _corr_vector(h: np.ndarray) -> np.ndarray:
    """Full linear correlation of h with (1-h)."""
    return np.correlate(h, 1.0 - h, mode="full")


def _c5_value(h: np.ndarray) -> float:
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(_corr_vector(h)) * dx


def _max_shift_and_value(h: np.ndarray):
    """Return shift (int) giving maximal correlation and the raw correlation."""
    n = h.shape[0]
    corr = _corr_vector(h)
    idx = int(np.argmax(corr))
    max_val = float(corr[idx])
    shift = idx - (n - 1)  # zero shift corresponds to idx = n-1
    return shift, max_val


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the (unscaled) correlation at the given shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift > 0:
        s = shift
        g[: n - s] += 1.0 - h[s:]
        g[s:] -= h[: n - s]
    elif shift < 0:
        s = -shift
        g[s:] += 1.0 - h[: n - s]
        g[: n - s] -= h[s:]
    # shift == 0 gives zero gradient (correlation is zero for binary h)
    return g


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # obtain an initial feasible h
    try:
        init = np.array(initial_h_values, dtype=np.float64)  # pre‑imported if present
    except Exception:
        init = None
    if init is None:
        # no parent solution: build a fresh binary vector of even length
        n = 100
        h = np.zeros(n, dtype=np.float64)
        h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    else:
        h = init.copy()
        n = h.shape[0]

    target_sum = n / 2.0
    h = _proj_box_simplex(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    # algorithm hyper‑parameters
    eta = 0.6               # initial step size
    decay = 0.9995          # exponential decay of step size
    min_eta = 5e-5
    temp_start = 0.02       # for occasional uphill moves
    last_improve = 0
    iteration = 0

    while time.time() - start < budget_s - 0.3:
        iteration += 1

        shift, _ = _max_shift_and_value(h)
        # avoid pathological zero‑shift dominance
        if shift == 0:
            # small random kick
            h = _proj_box_simplex(h + rng.normal(scale=0.02, size=n), target_sum)
            continue

        g = _gradient_for_shift(h, shift)
        h_new = h - eta * g
        h_new = _proj_box_simplex(h_new, target_sum)

        c5_new = _c5_value(h_new)

        # acceptance rule (pure descent with tiny uphill probability)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improve = iteration
        else:
            # probabilistic uphill acceptance
            elapsed = time.time() - start
            temp = max(0.0, temp_start * (1.0 - elapsed / budget_s))
            if temp > 0 and rng.random() < np.exp(-(c5_new - _c5_value(h)) / temp):
                h = h_new
                continue

        # always move to the new point if it improves the current value
        if c5_new < _c5_value(h):
            h = h_new
        else:
            # occasionally accept a non‑improving move to escape plateaus
            if rng.random() < 0.005:
                h = h_new

        eta = max(min_eta, eta * decay)

        # restart if stuck for a long time
        if iteration - last_improve > 4000:
            # random perturbation preserving the sum
            pert = rng.normal(scale=0.05, size=n)
            h = _proj_box_simplex(h + pert, target_sum)
            last_improve = iteration

    return best_h, best_c5, n
