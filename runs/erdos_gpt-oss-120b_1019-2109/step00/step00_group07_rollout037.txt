def compute_gradient_for_shift(h, k):
    # h shape (n,)
    # compute term1 = 1 - h shifted left by k (i+k) where out-of-bounds yields 1.
    # term2 = h shifted right by k (i-k) where out-of-bounds yields 0.
    term1 = np.ones_like(h)
    term2 = np.zeros_like(h)
    if k >= 0:
        # i + k within bounds for i from 0 to n-1-k
        term1[:n-k] = 1 - h[k:]  # because i + k index = i + k
        # term2: i - k >= 0 for i >= k
        term2[k:] = h[:n-k]
    else:
        # k negative: shift left to right reversed
        kk = -k
        # i + k = i - kk, within bounds for i >= kk
        term1[kk:] = 1 - h[:n-kk]
        # term2: i - k = i + kk, within bounds for i <= n-1-kk
        term2[:n-kk] = h[kk:]
    g = term1 - term2
    return g

    Returns
