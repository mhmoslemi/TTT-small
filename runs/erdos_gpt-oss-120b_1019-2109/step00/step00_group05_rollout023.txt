    n = len(h)
    # Find indices where h not at bounds
    can_increase = np.where(h < 1)[0]
    can_decrease = np.where(h > 0)[0]
    if len(can_increase)==0 or len(can_decrease)==0:
        return h.copy()
    i = np.random.choice(can_increase)
    j = np.random.choice(can_decrease)
    max_inc = 1 - h[i]
    max_dec = h[j]
    max_transfer = min(max_inc, max_dec)
    if max_transfer <= 0:
        return h.copy()  # no move possible
    delta = max_transfer * np.random.rand()  # uniform between 0 and max_transfer
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

import time, math, random
import numpy as np

def _project_onto_bounds_and_sum(v, target):
    """Project v to satisfy 0<=v<=1 and sum(v)=target."""
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    w = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        free = (w > 0.0) & (w < 1.0)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w

def _cost_and_raw(h):
    """Return (C5, raw_max) for the current vector."""
    n = h.shape[0]
    half = n // 2
    conv = np.correlate(h, 1.0 - h, mode="full")
    mid = n - 1
    vals = conv[mid - half : mid + half + 1]
    raw = vals.max()
    return raw * (2.0 / n), raw

def _partner_sum(h, k):
    """Zero‑padded partner sum for a given shift k (non‑negative)."""
    n = h.shape[0]
    if k == 0:
        return 2.0 * h
    left = np.empty(n)
    left[:k] = 0.0
    left[k:] = h[:-k]
    right = np.empty(n)
    right[-k:] = 0.0
    right[:-k] = h[k:]
    return left + right

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # ------------------------------------------------------------------
    # initialise vector (try to use the provided construction if any)
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=np.float64)
        scale = 2                                    # double the resolution
        h = np.repeat(base, scale)                   # respects sum automatically
        n = h.shape[0]
    else:
        n = 140                                      # modest size, good resolution
        h = rng.random(n)
        n = h.shape[0]

    target_sum = n / 2.0
    h = _project_onto_bounds_and_sum(h, target_sum)

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    half = n // 2
    dx = 2.0 / n
    T0 = 0.07                                      # initial temperature
    start = time.time()
    deadline = start + budget_s * 0.94             # keep a safety margin
    best_h = h.copy()
    best_cost, _ = _cost_and_raw(best_h)

    # main simulated‑annealing loop
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = min(1.0, elapsed / (budget_s * 0.94))
        T = T0 * (1.0 - progress)

        # worst shift (max overlap)
        conv = np.correlate(h, 1.0 - h, mode="full")
        mid = n - 1
        vals = conv[mid - half : mid + half + 1]
        max_idx = int(np.argmax(vals))
        max_raw = vals[max_idx]
        cur_cost = max_raw * dx
        shift = max_idx - half
        k = abs(shift)

        # decide move: 10 % random, otherwise targeted
        if rng.random() < 0.10:
            inc = np.where(h < 1.0)[0]
            dec = np.where(h > 0.0)[0]
            if inc.size == 0 or dec.size == 0:
                continue
            i = rng.choice(inc)
            j = rng.choice(dec)
        else:
            partner = _partner_sum(h, k)
            inc_mask = h < 1.0 - 1e-12
            dec_mask = h > 1e-12
            if inc_mask.any():
                tmp = partner.copy()
                tmp[~inc_mask] = -1.0
                i = int(np.argmax(tmp))
            else:
                i = rng.integers(n)
            if dec_mask.any():
                tmp = partner.copy()
                tmp[~dec_mask] = 1e9
                j = int(np.argmin(tmp))
            else:
                j = rng.integers(n)

        if i == j:
            continue
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        delta = min(max_inc, max_dec) * rng.random()
        if delta <= 1e-12:
            continue

        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta

        new_cost, _ = _cost_and_raw(h_new)

        # acceptance test
        if new_cost < cur_cost or rng.random() < math.exp(-(new_cost - cur_cost) / max(T, 1e-12)):
            h = h_new
            cur_cost = new_cost
            if new_cost < best_cost:
                best_h = h_new.copy()
                best_cost = new_cost

    # final projection (tiny numerical drift may appear)
    best_h = _project_onto_bounds_and_sum(best_h, target_sum)
    best_cost, _ = _cost_and_raw(best_h)
