def gradient_shift(h, s):
    n = len(h)
    # term1: 1 - h[i+s] where i+s in range else 1
    # create shifted version with padding zeros:
    term1 = np.ones_like(h)
    term2 = np.zeros_like(h)
    if s >= 0:
        term1[:n-s] = 1 - h[s:]      # i from 0 to n-s-1: i+s = s..n-1
        term2[s:] = h[:n-s]          # i-s out of range for i < s -> term2 stays 0
    else:
        s_abs = -s
        term1[s_abs:] = 1 - h[:n-s_abs]
        term2[:n-s_abs] = h[s_abs:]
    diff = term1 - term2
    return diff

    if s != 0:
        h = h * (target_sum / s)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # distribute the tiny deficit / surplus uniformly over free entries
        free = (h > 0) & (h < 1)
        cnt = free.sum()
        if cnt:
            h[free] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    # final safeguard
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # adjust first non‑boundary element
        for i in range(n):
            if 0 < h[i] < 1:
                h[i] += target_sum - cur_sum
                h[i] = min(1.0, max(0.0, h[i]))
                break

    dx = 2.0 / n

    # utility
    def c5_val(vec):
        return np.max(np.correlate(vec, 1 - vec, mode="full")) * dx

    cur_val = c5_val(h)
    best_h = h.copy()
    best_val = cur_val

    # ---- main search loop (simulated‑annealing style) ----
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= budget_s - 0.2:
            break

        # temperature linearly decays from 0.05 to near zero
        T = max(1e-6, 0.05 * (1.0 - elapsed / budget_s))

        # correlation & worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        cur_val = corr.max() * dx
        idx_max = corr.argmax()
        shift = idx_max - (n - 1)          # shift corresponding to the max

        # gradient‑like direction for this worst shift
        term1 = np.ones(n)
        term2 = np.zeros(n)
        if shift >= 0:
            if shift < n:
                term1[: n - shift] = 1.0 - h[shift:]
                term2[shift:] = h[: n - shift]
        else:
            t = -shift
            if t < n:
                term1[t:] = 1.0 - h[: n - t]
                term2[: n - t] = h[t:]

        diff_arr = term1 - term2
        i_up = int(np.argmax(diff_arr))           # decreasing this index helps
        i_down = int(np.argmin(diff_arr))         # increasing this index helps

        use_direct = diff_arr[i_up] > 0 and diff_arr[i_down] < 0
        if use_direct:
            max_eps = min(h[i_up], 1.0 - h[i_down])
        else:
            # random pair move when no clear direction
            i, j = random.sample(range(n), 2)
            if random.random() < 0.5:
                i_up, i_down = i, j
            else:
                i_up, i_down = j, i
            max_eps = min(h[i_up], 1.0 - h[i_down])

        if max_eps <= 0:
            continue

        # step size shrinks with temperature
        step_factor = (T / 0.05) + 0.1
        eps = max_eps * random.random() * step_factor
        eps = max(eps, max_eps * 1e-8)   # avoid zero

        h_new = h.copy()
        h_new[i_up] -= eps
        h_new[i_down] += eps

        new_val = np.max(np.correlate(h_new, 1 - h_new, mode="full")) * dx
        delta = new_val - cur_val

        if delta <= 0 or random.random() < math.exp(-delta / (T + 1e-12)):
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()

    # ---- final projection (tiny numeric fixes) ----
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_diff = target_sum - best_h.sum()
    if abs(sum_diff) > 1e-12:
        free = (best_h > 0) & (best_h < 1)
        cnt = free.sum()
        if cnt:
            best_h[free] += sum_diff / cnt
            best_h = np.clip(best_h, 0.0, 1.0)

    best_val = c5_val(best_h)
    return (best_h.tolist(), float(best_val), n)
