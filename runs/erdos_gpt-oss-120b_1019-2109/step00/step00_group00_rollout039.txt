# Helper to evaluate the Erdős overlap constant for a discretized h
def compute_c5(h):
    """Return max_k ∫ h(x)(1‑h(x+k))dx for the discretised step function."""
    n = len(h)
    # correlation of h with (1‑h) using the same definition as the evaluator
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * (2.0 / n))

    Parameters
    ----------
    a  : height of the two middle steps (0<a<1)
    L1 : length of each a‑step (left and right, same)
    n  : number of discretisation points (must be even)

    Returns
    -------
    h : ndarray of shape (n,) with values in [0,1] and sum(h)=n/2
    """

    if L0 < -1e-12:
        return None

    center_end = left_a_end + L2
    right_a_end = center_end + L1

    # fill the three non‑zero tiers
    h[(pos >= left_zero_end) & (pos < left_a_end)] = a
    h[(pos >= left_a_end) & (pos < center_end)] = 1.0
    h[(pos >= center_end) & (pos < right_a_end)] = a

    # enforce the exact integral constraint (∑h = n/2)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # look for a mutable entry (not 0 or 1)
        cand = np.where((h > 0.0) & (h < 1.0))[0]
        if cand.size == 0:
            cand = np.arange(n)
        idx = cand[0]
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

    return h

# Exhaustive grid search over the 5‑step family
def search_5step(n, time_deadline):
    best_h = None
    best_c5 = float("inf")
    best_par = None

    # coarse grid
    for a in np.linspace(0.05, 0.95, 19):
        if time.time() > time_deadline:
            break
        # admissible L1‑range for this a
        if a <= 0.0 or a >= 1.0:
            continue
        L1_max_1 = 0.5 / (1.0 - a)          # from L0 ≥ 0
        L1_max_2 = 1.0 / (2.0 * a)          # from L2 ≥ 0
        L1_max = min(L1_max_1, L1_max_2)
        if L1_max <= 0.0:
            continue
        for L1 in np.linspace(0.0, L1_max, 21):
            h = build_h_5step(a, L1, n)
            if h is None:
                continue
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h
                best_par = (a, L1)

    # refined search around the current best
    if best_par is not None:
        a0, L1_0 = best_par
        for a in np.linspace(max(0.0, a0 - 0.02), min(1.0, a0 + 0.02), 81):
            if time.time() > time_deadline:
                break
            if a <= 0.0 or a >= 1.0:
                continue
            L1_max_1 = 0.5 / (1.0 - a)
            L1_max_2 = 1.0 / (2.0 * a)
            L1_max = min(L1_max_1, L1_max_2)
            if L1_max <= 0.0:
                continue
            lo = max(0.0, L1_0 - 0.02)
            hi = min(L1_max, L1_0 + 0.02)
            for L1 in np.linspace(lo, hi, 81):
                h = build_h_5step(a, L1, n)
                if h is None:
                    continue
                c = compute_c5(h)
                if c < best_c5:
                    best_c5 = c
                    best_h = h
                    best_par = (a, L1)

    return best_h, best_c5

# Simple hill‑climbing on the full discretised vector
def hill_climb(h_start, c_start, n, deadline):
    """Pairwise adjustments that keep the total mass fixed."""
    best_h = h_start.copy()
    best_c = c_start
    cur_h = h_start.copy()
    cur_c = c_start

    while time.time() < deadline:
        # a small batch of random proposals
        for _ in range(300):
            if time.time() >= deadline:
                break
            i, j = random.sample(range(n), 2)
            hi, hj = cur_h[i], cur_h[j]

            # maximum admissible transfer from j → i
            inc = min(1.0 - hi, hj)
            # maximum admissible transfer from i → j
            dec = min(hi, 1.0 - hj)

            if inc <= 0.0 and dec <= 0.0:
                continue
            if random.random() < 0.5 and inc > 0.0:
                delta = random.random() * inc
            elif dec > 0.0:
                delta = -random.random() * dec
            else:
                continue

            new_h = cur_h.copy()
            new_h[i] = hi + delta
            new_h[j] = hj - delta
            new_c = compute_c5(new_h)

            if new_c + 1e-12 < cur_c:
                cur_h, cur_c = new_h, new_c
                if new_c + 1e-12 < best_c:
                    best_h, best_c = new_h.copy(), new_c
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start = time.time()
    # keep a safety margin for the harness
    deadline = start + budget_s * 0.95

    # choose a modest resolution (even length is required)
    n = 200

    # --------------------------------------------------------------------
    # 1) start from a provided construction, if any
    best_h = None
    best_c = float("inf")
    try:
        init = np.array(initial_h_values, dtype=np.float64)
        if init.ndim == 1 and init.size > 1:
            # enforce the mass constraint without breaking bounds
            target = init.size / 2.0
            diff = target - init.sum()
            if abs(diff) > 1e-12:
                cand = np.where((init > 0.0) & (init < 1.0))[0]
                if cand.size:
                    init[cand[0]] += diff
                else:
                    init[0] = np.clip(init[0] + diff, 0.0, 1.0)
            c0 = compute_c5(init)
            best_h, best_c = init, c0
    except NameError:
        pass  # initial_h_values not supplied

    # --------------------------------------------------------------------
    # 2) search the 5‑step family (which contains the known optimal shape)
    h_step, c_step = search_5step(n, deadline)
    if c_step < best_c:
        best_h, best_c = h_step, c_step

    # --------------------------------------------------------------------
    # 3) local refinement by pairwise hill climbing (if time permits)
    if time.time() < deadline - 5.0:
        clim_deadline = time.time() + (deadline - time.time()) * 0.6
        best_h, best_c = hill_climb(best_h, best_c, n, clim_deadline)

    # --------------------------------------------------------------------
    # final sanity check – enforce exact mass constraint once more
    target = n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-9:
        mutable = np.where((best_h > 0.0) & (best_h < 1.0))[0]
        if mutable.size:
            best_h[mutable[0]] += diff
        else:
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c = compute_c5(best_h)
    return (best_h, final_c, n)
