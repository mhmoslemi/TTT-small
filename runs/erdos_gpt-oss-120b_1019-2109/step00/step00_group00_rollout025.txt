import numpy as np, time, random, math
from scipy.optimize import minimize (maybe optional)

def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def random_binary_h(n):
    h = np.zeros(n)
    half = n // 2
    idx = np.random.choice(n, half, replace=False)
    h[idx] = 1.0
    if n % 2 == 1:
        # need additional 0.5 at random position
        extra_idx = np.random.choice([i for i in range(n) if h[i]==0])
        h[extra_idx] = 0.5
    return h

def v_shape_h(n):
    positions = np.arange(n)
    mid = (n - 1) / 2.0
    dist = np.abs(positions - mid)
    max_dist = mid
    h = dist / max_dist
    # scale to meet sum n/2
    target_sum = n / 2.0
    factor = target_sum / h.sum()
    h = h * factor
    # Clip within [0,1] (should be)
    h = np.clip(h, 0.0, 1.0)
    # after clipping may adjust sum; simple correction: subtract excess uniformly or scale again.
    # We'll ensure sum close-target: if sum deviates, adjust proportionally while staying within bounds.
    # final scaling may be okay, but we can also enforce exactly by adding small correction to middle.
    # For robustness, after scaling we check sum and adjust the middle element.
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        diff = target_sum - cur_sum
        # adjust the middle (or any non-boundary) by diff
        # Find index near middle where 0< h <1
        idx = int(round(mid))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
        # If still mismatch due to clipping, adjust another index
        # Use while loop
        cur_sum = h.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            # normalize whole vector while keeping bounds? simple approach: distribute diff equally among all interior points
            interior = np.where((h > 0) & (h < 1))[0]
            if interior.size > 0:
                add = diff / interior.size
                h[interior] = np.clip(h[interior] + add, 0, 1)
    return h

def hill_climb(h_initial, time_limit):
    h = h_initial.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    n = len(h)
    start = time.time()
    # parameters
    max_iters = int(1e7)
    temp_start = 0.05
    temp_end = 0.0001
    decay = 0.9995
    temp = temp_start
    iter = 0
    while time.time() - start < time_limit:
        # propose move
        # randomly pick source (j) with h[j] > 0 and sink (i) with h[i] < 1
        # pick j from indices with >eps
        possible_sources = np.where(h > 1e-12)[0]
        possible_sinks = np.where(h < 1 - 1e-12)[0]
        if possible_sources.size == 0 or possible_sinks.size == 0:
            break
        j = np.random.choice(possible_sources)
        i = np.random.choice(possible_sinks)
        max_inc = 1.0 - h[i]
        max_dec = h[j]
        delta_max = min(max_inc, max_dec)
        if delta_max <= 0:
            continue
        # step size as random fraction
        delta = delta_max * np.random.rand()
        # make small min step
        if delta < 1e-6:
            delta = delta_max * 0.5
        # apply
        h[i] += delta
        h[j] -= delta
        # compute new c5
        new_c5 = compute_c5(h)
        # acceptance
        if new_c5 < cur_c5 or np.random.rand() < math.exp(-(new_c5 - cur_c5) / max(temp, 1e-12)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta
        # temperature schedule
        temp = max(temp_end, temp * decay)
        iter += 1
        # optional break on convergence
        if iter % 10000 == 0:
            # maybe adjust step size etc.
            pass
    return best_h, best_c5

            break

        # fallback
        best_h = np.full(n_points, 0.5)
        best_c5 = compute_c5(best_h)

    target_sum = n_points / 2.0
    cur_sum = best_h.sum()

        interior = np.where((best_h > 0) & (best_h < 1))[0]

            best_h = np.clip(best_h, 0, 1)

    final_c5 = compute_c5(best_h)

    return (best_h, final_c5, n_points)

import time
import math
import random
import numpy as np

def _c5_of(h):
    """C5 value for a given step function h."""
    n = h.shape[0]
    # correlation h with (1-h), zero‑padded (np.correlate does exactly that)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _v_shape(n):
    """V‑shaped start: |i-mid|, scaled to satisfy the integral constraint."""
    mid = (n - 1) / 2.0
    dist = np.abs(np.arange(n) - mid)
    h = dist / mid                     # ends = 1, centre = 0
    target = n / 2.0
    scale = target / h.sum()
    h *= scale
    # tiny correction to hit the sum exactly (only needed when rounding messes up)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # adjust the centre point (or the nearest non‑bound point)
        centre = int(round(mid))
        h[centre] = np.clip(h[centre] + diff, 0.0, 1.0)
    return h

def _random_binary(n):
    """Random binary feasible vector (exactly n/2 ones). n must be even."""
    h = np.zeros(n)
    ones = np.random.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def _hill_climb(start_h, time_budget):
    """Simple simulated‑annealing hill‑climber that preserves sum(h)=n/2."""
    h = start_h.astype(float).copy()
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_of(h)
    cur_val = best_val

    start = time.time()
    # annealing schedule
    temp = 0.04
    decay = 0.9995
    min_temp = 1e-5

    # pre‑compute indices that can give a move (avoid scanning each time)
    while time.time() - start < time_budget:
        # pick a donor (h>0) and a receiver (h<1)
        donors = np.where(h > 1e-12)[0]
        receivers = np.where(h < 1.0 - 1e-12)[0]
        if donors.size == 0 or receivers.size == 0:
            break
        i = random.choice(receivers)   # will receive mass
        j = random.choice(donors)      # will give mass
        # maximal transferable amount
        delta_max = min(1.0 - h[i], h[j])
        if delta_max <= 0:
            continue
        # propose a random fraction of the allowed transfer
        delta = delta_max * random.random()
        # very small steps are useless
        if delta < 1e-6:
            delta = delta_max * 0.5

        h[i] += delta
        h[j] -= delta
        new_val = _c5_of(h)

        # accept if better, otherwise Metropolis criterion
        if new_val < cur_val or random.random() < math.exp(-(new_val - cur_val) / max(temp, 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] -= delta
            h[j] += delta

        # cool down
        temp = max(min_temp, temp * decay)

    return best_h, best_val

    Find a step function h on [0,2] with ∫h=1 that minimizes
    C5 = max_k ∫ h(x)*(1-h(x+k))dx.

    """
    random.seed(seed)
    np.random.seed(seed)

    # ---------- choose discretisation ----------
    n_points = 60                     # even, gives ∑h = n/2 = 30

    candidates.append(np.full(n_points, 0.5))

    # several random binary starts
    for _ in range(5):
        candidates.append(_random_binary(n_points))

    # if a parent construction exists, adapt it

                xs_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
                h_adj = np.interp(xs_new, xs_old, init)
                # enforce the integral constraint
                h_adj *= (target_sum / h_adj.sum())
                h_adj = np.clip(h_adj, 0.0, 1.0)

    # final evaluation
    final_c5 = _c5_of(best_h)

    return (best_h, float(final_c5), n_points)
