def random_feasible(N, target):
    # generate random numbers uniform [0,1]
    h = np.random.rand(N)
    # compute sum
    s = h.sum()
    # scale to target sum
    h *= target / s
    # after scaling some may exceed 1; clip and redistribute.
    # We'll handle by iterative approach:
    for _ in range(10):
        over = h > 1
        if over.any():
            excess = h[over] - 1
            h[over] = 1
            # redistribute excess to underful entries
            under = ~over
            if not under.any():
                break
            h[under] += excess.sum() / under.sum()
        under = h < 0
        if under.any():
            deficit = -h[under]
            h[under] = 0
            over = ~under
            if not over.any():
                break
            h[over] += deficit.sum() / over.sum()
    # after adjustments, may still slightly deviate; final rescale
    h = np.clip(h, 0, 1)
    cur = h.sum()
    if cur != target:
        # final small adjust: distribute diff equally among indices not at boundaries
        diff = target - cur
        # find indices with room
        if diff > 0:
            idx = np.where(h < 1)[0]
        else:
            idx = np.where(h > 0)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
        # clip again
        h = np.clip(h, 0, 1)
        # final adjust by scaling small amount
        s = h.sum()
        if abs(s - target) > 1e-8:
            h *= target / s
            h = np.clip(h, 0, 1)
    return h

def project_to_feasible(h):
    # ensures sum = target and 0<=h<=1
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if s == target_sum:
        return h
    # We'll use iterative water-filling:
    # While sum != target:
    #   diff = target - s
    #   free = indices where (diff >0 and h<1) or (diff <0 and h>0)
    #   adjust each free by diff / len(free)
    #   then clip again.
    # Repeat a few times.
    max_iter = 10
    for _ in range(max_iter):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = (h < 1)
        else:
            free = (h > 0)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = diff / n_free
        h[free] += delta
        # ensure no bound violation after adjustment
        h = np.clip(h, 0.0, 1.0)
    # final scaling adjustment if needed
    s = h.sum()
    if abs(s - target_sum) > 1e-10:
        # small uniform scaling to fix small error
        h *= target_sum / s
        h = np.clip(h, 0.0, 1.0)
    return h

    # ------------------------------------------------------------------
    best_h = project_to_feasible(best_h)

    # ------------------------------------------------------------------
    # Final evaluation (use the pre‑imported helper if available)
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_erdos_solution(best_h)
    except Exception:
        final_val = c5_and_corr(best_h)[0]

    return best_h, float(final_val), N
