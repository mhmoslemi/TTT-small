def project_to_constraints(h):
    n = h.shape[0]
    target = n / 2.0
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust sum to target using water-filling
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    # We'll adjust only elements not at bound in the direction needed.
    # Use while loop
    # To avoid infinite loops due to floating errors, break after many loops.
    max_iter = 1000
    for _ in range(max_iter):
        diff = target - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to increase sum: can increase elements with h < 1
            mask = h < 1 - 1e-12
        else:
            mask = h > 1e-12
        eligible = np.where(mask)[0]
        if eligible.size == 0:
            # can't adjust
            break
        # compute amount per eligible
        delta = diff / eligible.size
        # apply but keep within bounds
        for idx in eligible:
            if diff > 0:
                inc = min(delta, 1.0 - h[idx])
                h[idx] += inc
                cur += inc
                diff -= inc
            else:
                dec = min(-delta, h[idx])
                h[idx] -= dec
                cur -= dec
                diff += dec
        # loop again if still diff not zero
    # final clip and ensure sum
    h = np.clip(h, 0.0, 1.0)
    # final adjust any slight discrepancy by adding small epsilon to random element
    cur = h.sum()
    if abs(cur - target) > 1e-10:
        diff = target - cur
        # apply to random element within bounds
        idxs = np.arange(n)
        np.random.shuffle(idxs)
        for idx in idxs:
            if diff > 0 and h[idx] < 1:
                add = min(diff, 1 - h[idx])
                h[idx] += add
                diff -= add
            elif diff < 0 and h[idx] > 0:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

import time, math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate implements the a·b cross‑correlation with zero padding
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_constraints(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Project h onto [0,1]^n with Σh = target_sum (water‑filling)."""
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target_sum) < 1e-12:
        return h

    # iterative water‑filling
    for _ in range(200):
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        idx = np.where(mask)[0]
        if idx.size == 0:
            break
        delta = diff / idx.size
        # apply delta respecting bounds
        for i in idx:
            if diff > 0:
                inc = min(delta, 1.0 - h[i])
                h[i] += inc
                cur += inc
                diff -= inc
            else:
                dec = min(-delta, h[i])
                h[i] -= dec
                cur -= dec
                diff += dec
    # final tiny correction if needed
    cur = h.sum()
    if abs(cur - target_sum) > 1e-10:
        diff = target_sum - cur
        order = np.arange(h.size)
        np.random.shuffle(order)
        for i in order:
            if diff > 0 and h[i] < 1.0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            elif diff < 0 and h[i] > 0.0:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

        if not made_move:
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1  # ensure i != j
            # random direction of transfer
            if rng.random() < 0.5:
                inc_cap = 1.0 - cur_h[i]
                dec_cap = cur_h[j]
                cap = min(inc_cap, dec_cap, max_delta)
                if cap < 1e-12:
                    continue
                delta = rng.random() * cap
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
            else:
                inc_cap = 1.0 - cur_h[j]
                dec_cap = cur_h[i]
                cap = min(inc_cap, dec_cap, max_delta)
                if cap < 1e-12:
                    continue
                delta = rng.random() * cap
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta

        # ----------------------------------------------------------------
        # evaluate the proposal
        new_val = _compute_c5(new_h)

        # accept if better, otherwise with SA probability
        if new_val < cur_val - 1e-14:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val - 1e-14:
                best_h, best_val = new_h.copy(), new_val
        else:
            # Metropolis criterion
            prob = math.exp(-(new_val - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                cur_h, cur_val = new_h, new_val

        # slowly shrink the maximal transfer magnitude
        if it % 5000 == 0:
            max_delta = max(max_delta * 0.92, 1e-4)

        # occasional re‑projection to keep numerical sum exact
        if it % 20000 == 0:
            if abs(cur_h.sum() - n / 2.0) > 1e-9:
                cur_h = _project_to_constraints(cur_h, n / 2.0)
                cur_val = _compute_c5(cur_h)
            if abs(best_h.sum() - n / 2.0) > 1e-9:
                best_h = _project_to_constraints(best_h, n / 2.0)
                best_val = _compute_c5(best_h)

    # final safety projection
    best_h = _project_to_constraints(best_h, n / 2.0)
    best_val = _compute_c5(best_h)

    return (best_h.tolist(), float(best_val), int(n))
