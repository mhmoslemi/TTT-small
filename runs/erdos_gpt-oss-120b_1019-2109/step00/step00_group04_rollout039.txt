def project_simplex(v, S):
    # Clip to [0,1]
    v = np.clip(v, 0, 1)
    # Compute excess sum
    while True:
        total = v.sum()
        if abs(total - S) < 1e-12:
            break
        diff = S - total
        # Determine indices that can be increased (if diff >0) or decreased (if diff<0)
        if diff > 0:
            # increase non-full elements
            free = np.where(v < 1 - 1e-12)[0]
            if len(free) == 0: break
            inc = diff / len(free)
            # ensure not exceed 1
            inc = np.minimum(inc, 1 - v[free])
            v[free] += inc
        else:
            free = np.where(v > 1e-12)[0]
            if len(free) == 0: break
            dec = diff / len(free)  # diff negative
            # ensure not go below 0
            dec = np.maximum(dec, -v[free])
            v[free] += dec
    return v

import numpy as np, time, math

def max_correlation(h):
    n = h.shape[0]
    N = 1 << ((2 * n - 1).bit_length())
    ha = np.fft.rfft(np.concatenate([h, np.zeros(N - n)]))
    hb = np.fft.rfft(np.concatenate([1 - h, np.zeros(N - n)]))
    c = np.fft.irfft(ha * np.conj(hb), n=N)
    c = c[:2 * n - 1]
    return np.max(c) * (2.0 / n)

def project_to_sum_bounds(v, target):
    # v already clipped 0..1
    # enforce sum target by water-filling
    # Use while loop
    eps = 1e-12
    while True:
        total = v.sum()
        diff = target - total
        if abs(diff) < eps:
            break
        if diff > 0:
            free = np.where(v < 1 - eps)[0]
            if len(free) == 0:
                break
            inc = diff / len(free)
            inc = np.minimum(inc, 1 - v[free])
            v[free] += inc
        else:
            free = np.where(v > eps)[0]
            if len(free) == 0:
                break
            dec = diff / len(free)  # diff negative
            dec = np.maximum(dec, -v[free])
            v[free] += dec
    return v

def random_feasible(m, total):
    # generate random vector in [0,1] sum = total
    # use Dirichlet-like random then project
    v = np.random.rand(m)
    v = project_to_sum_bounds(v, total)
    return v

def run(seed=42, budget_s=1100):
    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    # parameters
    n = 200
    m = 10
    L = n // m
    target_sum = n / 2.0
    block_sum_target = target_sum / L  # sum of block heights needed
    # initialize best with uniform
    v_best = np.full(m, 0.5)
    h_best = np.repeat(v_best, L)
    best_val = max_correlation(h_best)
    # global best
    best_h = h_best.copy()
    best_c = best_val
    # annealing parameters
    step = 0.2
    temp = 0.01
    # compute number of iterations based on time budget
    max_iters = 500000
    it = 0
    v = v_best.copy()
    while time.time() - start < budget_s - 5 and it < max_iters:
        # propose random change
        i, j = np.random.choice(m, 2, replace=False)
        # decide direction
        sign = 1 if np.random.rand() < 0.5 else -1
        if sign > 0:
            max_inc = min(step, 1 - v[i], v[j])
            if max_inc <= 0:
                continue
            delta = np.random.rand() * max_inc
        else:
            max_dec = min(step, v[i], 1 - v[j])
            if max_dec <= 0:
                continue
            delta = -np.random.rand() * max_dec
        v_new = v.copy()
        v_new[i] += delta
        v_new[j] -= delta
        # optional tiny random perturb to avoid stuck
        # compute h and value
        h_new = np.repeat(v_new, L)
        val_new = max_correlation(h_new)
        # acceptance
        if val_new < best_c or (temp > 1e-12 and np.random.rand() < math.exp(-(val_new - best_c) / temp)):
            v = v_new
            if val_new < best_c:
                best_c = val_new
                best_h = h_new.copy()
        # schedule
        it += 1
        # reduce step and temp gradually
        if it % 1000 == 0:
            step = max(1e-4, step * 0.995)
            temp = max(1e-6, temp * 0.995)
    # final check
    # ensure constraints
    # compute final c
    final_c = max_correlation(best_h)
    # return
    return (best_h, final_c, n)

import numpy as np, time, math

def max_corr(h):
    n = h.shape[0]
    N = 1 << ((2*n - 1).bit_length())
    ha = np.fft.rfft(np.concatenate([h, np.zeros(N - n)]))
    hb = np.fft.rfft(np.concatenate([1 - h, np.zeros(N - n)]))
    c = np.fft.irfft(ha * np.conj(hb), n=N)
    c = c[:2*n-1]
    return np.max(c) * (2.0 / n)

import time, math
import numpy as np

def _max_corr(h: np.ndarray) -> float:
    """Return C5 for a step vector h using the same formula as the evaluator."""
    n = h.shape[0]
    # size for linear (zero‑padded) convolution
    N = 1 << ((2 * n - 1).bit_length())
    ha = np.fft.rfft(np.concatenate([h, np.zeros(N - n)]))
    hb = np.fft.rfft(np.concatenate([1 - h, np.zeros(N - n)]))
    c = np.fft.irfft(ha * np.conj(hb), n=N)
    c = c[: 2 * n - 1]                     # linear correlation output
    return float(c.max() * (2.0 / n))      # scale by dx = 2/n


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Stochastic local optimisation for the Erdős minimum overlap constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # ------- parameters -------------------------------------------------
    n = 200                     # number of discretisation points (multiple of 5)
    target_sum = n / 2.0        # integral constraint (∑h_i = n/2)

    # initial feasible vector – linear decreasing from 1 to 0
    h = np.linspace(1.0, 0.0, n)
    cur_val = _max_corr(h)

    best_h = h.copy()
    best_val = cur_val

    # simulated‑annealing style parameters
    step = 0.2                  # maximal amount moved between two positions
    temp = 0.01                 # temperature for uphill moves
    it = 0
    max_iters = 5_000_000

    while time.time() - start < budget_s - 5 and it < max_iters:
        # pick two distinct coordinates
        i, j = np.random.choice(n, 2, replace=False)

        if np.random.rand() < 0.5:
            # try to increase h[i], decrease h[j]
            max_inc = min(step, 1.0 - h[i], h[j])
            if max_inc <= 0.0:
                continue
            delta = np.random.rand() * max_inc
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
        else:
            # try to decrease h[i], increase h[j]
            max_dec = min(step, h[i], 1.0 - h[j])
            if max_dec <= 0.0:
                continue
            delta = np.random.rand() * max_dec
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta

        # evaluate candidate
        new_val = _max_corr(h_new)

        # acceptance rule
        accept = False
        if new_val < cur_val:
            accept = True
        elif temp > 1e-12:
            if np.random.rand() < math.exp(-(new_val - cur_val) / temp):
                accept = True

        if accept:
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()

        # schedule
        it += 1
        if it % 1000 == 0:
            step = max(1e-4, step * 0.99)
            temp = max(1e-6, temp * 0.995)

    # final sanity check – enforce the exact integral (tiny numerical drift)
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-9:
        # distribute the small difference uniformly while staying in bounds
        mask = (best_h > 0) & (best_h < 1)
        if mask.any():
            adjust = diff / mask.sum()
            best_h[mask] += adjust
            # clip just in case
            np.clip(best_h, 0.0, 1.0, out=best_h)

    final_c5 = _max_corr(best_h)
    return best_h, final_c5, n
