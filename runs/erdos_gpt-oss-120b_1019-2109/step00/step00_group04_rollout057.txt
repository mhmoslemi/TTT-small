    corr = np.correlate(h, 1 - h, mode='full')
    raw_max = np.max(corr)

        i_range = slice(0, n-shift)
        g[i_range] -= h[shift:] # term1
        # i-shift in bounds for i=shift..n-1
        j_range = slice(shift, n)
        g[j_range] -= h[:n-shift] # term2
    else:
        s = -shift
        # i+shift valid for i=s..n-1
        i_range = slice(s, n)
        g[i_range] -= h[:n-s] # term1: h[i+shift] = h[i-s]
        # i-shift valid for i=0..n-s-1
        j_range = slice(0, n-s)
        g[j_range] -= h[s:] # term2: h[i-shift] = h[i+s]
    # pick i_min with smallest g (most negative) and i_max with largest g (most positive)
    i_min = int(np.argmin(g))

import time
import math
import numpy as np

def _raw_max_corr(h):
    """Maximum of the (unscaled) correlation of h with 1-h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full"))


def _gradient_step(h, shift):
    """Return a new h obtained by moving one unit of mass from the
    index with largest positive gradient to the one with most negative
    gradient for the given shift.  Returns None if no feasible move."""
    n = h.size
    g = np.ones(n, dtype=np.float64)               # dR/dh_i = 1 - (h[i+shift]+h[i-shift])
    if shift >= 0:
        if shift < n:
            g[: n - shift] -= h[shift:]           # term h[i+shift]
            g[shift:] -= h[: n - shift]           # term h[i-shift]
    else:
        s = -shift
        if s < n:
            g[s:] -= h[: n - s]                   # term h[i+shift]
            g[: n - s] -= h[s:]                   # term h[i-shift]
    i_inc = int(np.argmin(g))                      # most negative gradient → increase h[i]
    i_dec = int(np.argmax(g))                      # most positive gradient → decrease h[i]
    # feasible delta respecting box constraints
    delta_up = 1.0 - h[i_inc]
    delta_down = h[i_dec]
    delta = min(delta_up, delta_down)
    if delta <= 0.0:
        return None
    h_new = h.copy()
    h_new[i_inc] += delta
    h_new[i_dec] -= delta
    return h_new


def _random_pair_move(h, rng):
    """Propose a random feasible move that keeps the total sum unchanged."""
    n = h.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None
    hi, hj = h[i], h[j]
    low = max(-hi, hj - 1.0)          # allowed lower bound for delta
    high = min(1.0 - hi, hj)          # allowed upper bound for delta
    if low >= high:
        return None
    delta = low + rng.random() * (high - low)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new


def _local_search(h_start, rng, time_budget):
    """Hill‑climbing / simulated‑annealing style search."""
    n = h_start.size
    target_sum = n / 2.0
    h = h_start.copy()
    cur_raw = _raw_max_corr(h)
    best_h = h.copy()
    best_raw = cur_raw
    start = time.time()
    start_temp = 0.5                     # initial temperature for SA moves

    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break

        # current correlation & shift of the current worst overlap
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        # decide deterministic gradient move vs random move
        if rng.random() < 0.6:
            h_new = _gradient_step(h, shift)
        else:
            h_new = _random_pair_move(h, rng)

        if h_new is None:
            continue

        new_raw = _raw_max_corr(h_new)

        # annealed acceptance
        remaining = time_budget - elapsed
        temp = start_temp * (remaining / time_budget) if time_budget > 0 else 0.0
        improve = new_raw < cur_raw - 1e-12
        accept = improve or (temp > 0.0 and rng.random() < math.exp((cur_raw - new_raw) / temp))
        if accept:
            h = h_new
            cur_raw = new_raw
            if new_raw < best_raw - 1e-12:
                best_raw = new_raw
                best_h = h.copy()

    # final sum correction (tiny drift may appear)
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        for idx in rng.permutation(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break
    np.clip(best_h, 0.0, 1.0, out=best_h)
    best_raw = _raw_max_corr(best_h)
    return best_h, best_raw
