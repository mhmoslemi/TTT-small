import numpy as np, time, math

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Get an initial vector if the harness provides one
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=np.float64)
        N = init_h.size
    else:
        N = 80                     # fallback size (must be even for half‑mass)
        init_h = np.full(N, 0.5, dtype=np.float64)

    dx = 2.0 / N                     # sampling step

    # ------------------------------------------------------------------
    # Helper: unscaled maximal overlap = max_k ∑ h[i](1‑h[i+k])
    # ------------------------------------------------------------------
    def raw_max_overlap(h):
        return np.max(np.correlate(h, 1.0 - h, mode='full'))

    # ------------------------------------------------------------------
    # Keep the best solution seen so far
    # ------------------------------------------------------------------
    best_h = init_h.copy()
    best_raw = raw_max_overlap(best_h)

    # ------------------------------------------------------------------
    # Build a handful of starting points (the given one + a few random ones)
    # ------------------------------------------------------------------
    starts = [init_h]
    if not np.allclose(init_h, 0.5):
        starts.append(np.full(N, 0.5, dtype=np.float64))

    for _ in range(2):
        v = rng.random(N)
        v = v / v.sum() * (N / 2.0)          # enforce the integral constraint in expectation
        v = np.clip(v, 0.0, 1.0)
        # fix the tiny sum error that clipping may have created
        diff = v.sum() - N / 2.0
        if abs(diff) > 1e-12:
            idx = rng.integers(N)
            if diff > 0:
                dec = min(diff, v[idx])
                v[idx] -= dec
            else:
                inc = min(-diff, 1.0 - v[idx])
                v[idx] += inc
        starts.append(v)

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    start_time   = time.time()
    time_limit   = budget_s * 0.95            # a safety margin
    max_iters    = 5_000_000
    init_temp    = 0.02
    final_temp   = 1e-5
    max_delta    = 0.4                       # maximal raw move before scaling

    total_iters = 0

    for start_vec in starts:
        if time.time() - start_time > time_limit:
            break
        h = start_vec.copy()
        cur_raw = raw_max_overlap(h)

        while time.time() - start_time < time_limit:
            # Linear cooling schedule
            temp = init_temp * max(0.0, 1.0 - total_iters / max_iters) + final_temp

            # Pick two distinct indices
            i = rng.integers(N)
            j = rng.integers(N - 1)
            if j >= i:
                j += 1

            # Propose a change whose magnitude shrinks with the temperature
            delta = rng.uniform(-max_delta, max_delta) * (temp / init_temp)

            # Keep the box constraints [0,1]
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if lo > hi:
                continue
            delta = max(min(delta, hi), lo)
            if abs(delta) < 1e-12:
                continue

            # Apply the move (sum stays constant)
            h[i] += delta
            h[j] -= delta

            new_raw = raw_max_overlap(h)
            d = new_raw - cur_raw

            accept = False
            if d <= 0:
                accept = True
            else:
                if temp > 0 and rng.random() < math.exp(-d / temp):
                    accept = True

            if accept:
                cur_raw = new_raw
                if new_raw < best_raw:
                    best_raw = new_raw
                    best_h = h.copy()
            else:
                # revert
                h[i] -= delta
                h[j] += delta

            total_iters += 1

    # ------------------------------------------------------------------
    # Greedy gradient refinement on the best vector found
    # ------------------------------------------------------------------
    while time.time() - start_time < time_limit:
        cur_raw = raw_max_overlap(best_h)
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (N - 1)                     # actual lag

        # gradient of the worst‑overlap shift
        g = np.ones(N)
        if shift >= 0:
            g[:N - shift] -= best_h[shift:]
            g[shift:]     -= best_h[:N - shift]
        else:
            kk = -shift
            g[kk:]        -= best_h[:N - kk]
            g[:N - kk]    -= best_h[kk:]

        pos = np.where((g > 1e-8) & (best_h > 1e-8))[0]
        neg = np.where((g < -1e-8) & (best_h < 1 - 1e-8))[0]
        if pos.size == 0 or neg.size == 0:
            break

        i = pos[g[pos].argmax()]                # largest positive gradient
        j = neg[g[neg].argmin()]                # most negative gradient

        step = min(g[i], -g[j], best_h[i], 1 - best_h[j]) * 0.5
        if step < 1e-12:
            break

        # trial move
        best_h[i] -= step
        best_h[j] += step
        new_raw = raw_max_overlap(best_h)
        if new_raw < cur_raw:
            continue               # keep the improvement
        else:
            # revert and stop the refinement
            best_h[i] += step
            best_h[j] -= step
            break

    # ------------------------------------------------------------------
    # Enforce the integral constraint exactly (sum = N/2)
    # ------------------------------------------------------------------
    target = N / 2.0
    cur_sum = best_h.sum()
    diff = cur_sum - target
    if abs(diff) > 1e-9:
        if diff > 0:
            cand = np.where(best_h > 0)[0]
            if cand.size:
                idx = rng.choice(cand)
                best_h[idx] = max(0.0, best_h[idx] - diff)
        else:
            cand = np.where(best_h < 1)[0]
            if cand.size:
                idx = rng.choice(cand)
                best_h[idx] = min(1.0, best_h[idx] - diff)   # diff negative ⇒ increase
        # If a tiny residue remains, distribute it
        cur_sum = best_h.sum()
        diff = target - cur_sum
        if abs(diff) > 1e-12:
            for i in range(N):
                if diff == 0:
                    break
                if diff > 0 and best_h[i] < 1.0:
                    add = min(diff, 1.0 - best_h[i])
                    best_h[i] += add
                    diff -= add
                elif diff < 0 and best_h[i] > 0.0:
                    sub = min(-diff, best_h[i])
                    best_h[i] -= sub
                    diff += sub

    final_C5 = raw_max_overlap(best_h) * dx
    return best_h.tolist(), float(final_C5), int(N)
