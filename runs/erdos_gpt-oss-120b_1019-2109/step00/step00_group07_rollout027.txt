def project_to_constraints(x, target):
    # Clip to range
    x = np.clip(x, 0.0, 1.0)
    # adjust sum using water-filling
    # This algorithm iterates at most few times.
    for _ in range(10):
        s = np.sum(x)
        diff = target - s
        if abs(diff) < 1e-12:
            break
        # Determine free indices where x not at bound
        free = (x > 0) & (x < 1)
        n_free = np.count_nonzero(free)
        if n_free == 0:
            # No free variables, break
            break
        delta = diff / n_free
        # Update free indices
        x[free] += delta
        # Clip again
        x = np.clip(x, 0.0, 1.0)
    # If after loop sum not correct due to rounding, adjust by small epsilon on free ones.
    # (Simplify) no further.
    return x

import time
import numpy as np

def _target_sum(n: int) -> float:
    """Desired sum of h values (integral = 1 over [0,2])."""
    return n / 2.0

def _project_to_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Clamp to [0,1] and adjust to meet the exact sum `target`."""
    x = np.clip(x, 0.0, 1.0)
    # Water‑filling style adjustments.
    for _ in range(10):
        s = float(x.sum())
        diff = target - s
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        nfree = int(free.sum())
        if nfree == 0:
            # No free variables – push the most flexible entry.
            if diff > 0:
                i = int(np.argmax(1.0 - x))
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            else:
                i = int(np.argmin(x))
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            continue
        delta = diff / nfree
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # Final tiny correction if needed.
    s = float(x.sum())
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0:
            cand = np.where(x < 1.0 - 1e-12)[0]
        else:
            cand = np.where(x > 1e-12)[0]
        if cand.size:
            i = int(cand[0])
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

def _c5_score(h: np.ndarray) -> float:
    """Return C₅ for a given h (uses the provided evaluator when possible)."""
    try:
        # Provided by the harness.
        return evaluate_erdos_solution(h)
    except Exception:
        # Fallback – manual computation.
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(corr.max() * (2.0 / h.size))

def _worst_shift(h: np.ndarray):
    """Find the shift k with maximum overlap and return (k, max_corr)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    k = idx - (h.size - 1)         # shift ranging from -(n-1) … n-1
    return k, float(corr[idx])

def _gradient_for_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Gradient of Σ_i h_i h_{i+k} (the quantity we wish to raise) w.r.t. h."""
    n = h.size
    g = np.zeros_like(h)
    if k >= 0:
        g[: n - k] += h[k:]          # h_{j+k} term
        g[k:] += h[: n - k]          # h_{j-k} term
    else:
        kk = -k
        g[kk:] += h[: n - kk]        # h_{j-k}
        g[: n - kk] += h[kk:]        # h_{j+k}
    return g

def _pair_transfer_move(h: np.ndarray) -> np.ndarray:
    """Generate a feasible neighbour by moving mass between two coordinates."""
    n = h.size
    i, j = np.random.choice(n, size=2, replace=False)
    hi, hj = h[i], h[j]
    # Transfer δ from i to j: new_i = hi - δ, new_j = hj + δ
    lo = max(-hi, hj - 1.0)          # lower bound for δ
    hi_ = min(hi, 1.0 - hj)          # upper bound for δ
    if lo >= hi_:
        # no feasible transfer; fall back to a tiny random jitter.
        cand = h + np.random.uniform(-0.001, 0.001, size=n)
        return _project_to_sum(cand, _target_sum(n))
    delta = np.random.uniform(lo, hi_)
    cand = h.copy()
    cand[i] -= delta
    cand[j] += delta
    return cand

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) improving the Erdős C₅ bound."""
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Problem size and initial feasible point.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        n = int(len(initial_h_values))
        h_cur = np.array(initial_h_values, dtype=np.float64)
    else:
        n = 80
        # start from a Dirichlet sample (fairly spread).
        h_cur = np.random.dirichlet(np.ones(n) * 0.5) * _target_sum(n)

    h_cur = _project_to_sum(h_cur, _target_sum(n))
    cur_score = _c5_score(h_cur)
    best_h = h_cur.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # 2. Simulated annealing parameters.
    # ------------------------------------------------------------------
    T0 = 0.05                       # initial temperature
    iter_cnt = 0

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.05:
            break
        # Temperature linearly cooled to near zero.
        T = max(T0 * (1.0 - elapsed / budget_s), 1e-8)

        # ------------------------------------------------------------------
        # 3. Propose a candidate.
        # ------------------------------------------------------------------
        if np.random.rand() < 0.05:
            # a global Gaussian perturbation (helps escape plateaus)
            cand = h_cur + np.random.normal(scale=0.05, size=n)
        else:
            # a simple pair‑transfer move (preserves the sum exactly)
            cand = _pair_transfer_move(h_cur)

        cand = _project_to_sum(cand, _target_sum(n))
        cand_score = _c5_score(cand)

        # ------------------------------------------------------------------
        # 4. Acceptance test (Metropolis).
        # ------------------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # uphill move accepted with probability exp(-Δ/T)
            prob = np.exp((cur_score - cand_score) / T)
            accept = np.random.rand() < prob

        if accept:
            h_cur = cand
            cur_score = cand_score
            if cand_score < best_score:
                best_h = cand.copy()
                best_score = cand_score
                # Early exit if we have already beaten the target bound.
                if best_score <= 0.3808:
                    break

        # ------------------------------------------------------------------
        # 5. Occasionally try a move directed by the worst shift.
        # ------------------------------------------------------------------
        if np.random.rand() < 0.02:
            k_worst, _ = _worst_shift(h_cur)
            g = _gradient_for_shift(h_cur, k_worst)
            # Small step in the gradient direction.
            step = 0.01
            cand2 = _project_to_sum(h_cur + step * g, _target_sum(n))
            cand2_score = _c5_score(cand2)
            if cand2_score < cur_score or np.random.rand() < np.exp((cur_score - cand2_score) / T):
                h_cur = cand2
                cur_score = cand2_score
                if cand2_score < best_score:
                    best_h = cand2.copy()
                    best_score = cand2_score
                    if best_score <= 0.3808:
                        break

        # ------------------------------------------------------------------
        # 6. Rare full restarts to explore distant regions.
        # ------------------------------------------------------------------
        if np.random.rand() < 0.001:
            h_cur = np.random.dirichlet(np.ones(n) * 0.3) * _target_sum(n)
            h_cur = _project_to_sum(h_cur, _target_sum(n))
            cur_score = _c5_score(h_cur)

        iter_cnt += 1

    # ------------------------------------------------------------------
    # 7. Final sanitisation and return.
    # ------------------------------------------------------------------
    best_h = _project_to_sum(best_h, _target_sum(n))
    best_score = _c5_score(best_h)
    return best_h, best_score, n
