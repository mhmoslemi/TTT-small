import time
import random
import numpy as np

def run(seed: int = 42, budget_s: int = 1100):
    rng = np.random.default_rng(seed)
    random.seed(seed)
    start = time.time()

    # ----- problem size -------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 120                     # fallback size
        h0 = rng.random(n)

    target_sum = n / 2.0
    dx = 2.0 / n

    # ----- projection onto box [0,1] with fixed sum --------------------
    def project(x: np.ndarray) -> np.ndarray:
        lo, hi = -2.0, 2.0
        for _ in range(60):        # bisection for lambda
            mid = (lo + hi) * 0.5
            xm = np.clip(x + mid, 0.0, 1.0)
            s = xm.sum()
            if s > target_sum:
                hi = mid
            else:
                lo = mid
        return np.clip(x + lo, 0.0, 1.0)

    h = project(h0.copy())

    # ----- evaluation ----------------------------------------------------
    def eval_c5(vec: np.ndarray) -> float:
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return float(corr.max() * dx)

    best_h = h.copy()
    best_val = eval_c5(best_h)

    # ----- gradient for the worst shift --------------------------------
    def max_shift_and_grad(v: np.ndarray):
        corr = np.correlate(v, 1.0 - v, mode='full')
        idx = int(corr.argmax())
        shift = idx - (n - 1)                # actual shift value
        grad = np.zeros_like(v)

        if shift >= 0:
            k = shift
            grad[:n - k] += (1.0 - v[k:])
            grad[k:]      -= v[:n - k]
        else:
            s = -shift
            grad[s:]      += (1.0 - v[:n - s])
            grad[:n - s]  -= v[s:]
        return shift, grad, corr[idx]

    # ----- main optimisation loop ---------------------------------------
    lr = 0.5
    it = 0
    last_improve = 0

    while time.time() - start < budget_s * 0.95:
        _, grad, _ = max_shift_and_grad(h)
        h_new = project(h - lr * grad)
        val_new = eval_c5(h_new)

        if val_new < best_val - 1e-12:
            best_val = val_new
            best_h = h_new.copy()
            last_improve = it

        h = h_new
        it += 1

        # learning‑rate schedule
        if it % 200 == 0:
            lr = max(lr * 0.7, 1e-4)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            h = project(best_h + (rng.random(n) - 0.5) * 0.05)
            lr = 0.5

        # restart if stuck for a long time
        if it - last_improve > 3000:
            h = project(best_h + (rng.random(n) - 0.5) * 0.1)
            lr = 0.5
            last_improve = it

        # tiny pairwise swaps every few thousand iterations
        if it % 1000 == 0:
            for _ in range(20):
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                # propose a small transfer delta from i to j
                delta = rng.uniform(-0.04, 0.04)
                delta = max(-h[i], delta)          # keep i within [0,1]
                delta = min(1.0 - h[i], delta)
                delta_j = -delta
                if h[j] + delta_j < 0.0:
                    delta = -h[j]
                    delta_j = -delta

                    delta = 1.0 - h[j]
