import numpy as np, random, time, math

def _correlation_max(h):
    """Return max correlation value (integer count) and its lag."""
    corr = np.correlate(h, 1 - h, mode='full')
    max_idx = int(corr.argmax())
    max_val = int(corr[max_idx])
    lag = max_idx - (len(h) - 1)
    return max_val, lag

def _random_binary(n, ones):
    """Generate a binary vector of length n with exactly `ones` ones."""
    v = np.zeros(n, dtype=int)
    v[:ones] = 1
    np.random.shuffle(v)
    return v

def _repair_to_target(v, target):
    """Adjust binary vector v so that sum(v) == target."""
    cur = int(v.sum())
    if cur == target:
        return v
    idx_one = np.where(v == 1)[0]
    idx_zero = np.where(v == 0)[0]
    if cur > target:
        # turn some ones to zero
        to_zero = cur - target
        np.random.shuffle(idx_one)
        v[idx_one[:to_zero]] = 0
    else:
        # turn some zeros to one
        to_one = target - cur
        np.random.shuffle(idx_zero)
        v[idx_zero[:to_one]] = 1
    return v
