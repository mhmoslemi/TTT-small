def project_feasible(x):
    # enforce box [0,1]
    y = np.clip(x, 0.0, 1.0)
    total = np.sum(y)
    S = len(y) / 2.0
    diff = S - total
    # We'll adjust iteratively
    # Use simple water-filling: while diff>eps:
    #   compute free indices where (y < 1 if diff>0) or (y > 0 if diff<0)
    #   adjust uniformly by diff/num_free, clip again
    # But for simplicity iter 10 times:
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = None
        if diff > 0:
            free = y < 1.0 - 1e-12
        else:
            free = y > 1e-12
        n_free = np.count_nonzero(free)
        if n_free == 0:
            break
        adjust = diff / n_free
        y[free] += adjust
        # clip after adjustment to maintain bounds
        y = np.clip(y, 0.0, 1.0)
        diff = S - np.sum(y)
    # final clip again
    y = np.clip(y, 0.0, 1.0)
    # final adjust tiny discrepancy by scaling (optional)
    # Compute final diff
    diff = S - np.sum(y)
    if abs(diff) > 1e-9:
        # proportionally distribute to all entries not at bound
        # simple approach: add diff/len(y) to all, clipping iteratively.
        y += diff/len(y)
        y = np.clip(y, 0.0, 1.0)
        # may still not be exact but close.
    return y

def gradient_step(h, alpha):
    n = len(h)
    # compute correlation and find shift index of maximum
    corr = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(corr)
    shift = idx - (n - 1)
    i = np.arange(n)
    up_idx = i + shift
    down_idx = i - shift
    term1 = np.where((up_idx >= 0) & (up_idx < n), 1 - h[up_idx], 1.0)
    term2 = np.where((down_idx >= 0) & (down_idx < n), h[down_idx], 0.0)
    grad = term1 - term2
    # Step along negative gradient (to reduce C) which is -grad
    h_new = h - alpha * grad
    # Clip to [0,1] and enforce sum via project
    h_new = project_feasible(h_new)
    return h_new

def project_feasible(x):

    y = np.clip(x, 0.0, 1.0)
    n = y.shape[0]
    S = n / 2.0

        diff = S - total
        if abs(diff) < 1e-12:

import numpy as np, time, math

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for a discretised h."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate gives the unscaled sum; multiply by dx afterwards.
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def project_feasible(x):
    """
    Project a vector onto the feasible set:
    0 ≤ x[i] ≤ 1 and sum(x) = n/2  (where n = len(x)).
    Uses a simple water‑filling iteration.
    """
    y = np.clip(x, 0.0, 1.0)
    n = y.size
    target = n / 2.0
    for _ in range(20):
        tot = y.sum()
        diff = target - tot
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = y < 1.0 - 1e-12
        else:
            mask = y > 1e-12
        free = np.count_nonzero(mask)
        if free == 0:
            break
        adjust = diff / free
        y[mask] += adjust
        y = np.clip(y, 0.0, 1.0)
    # Tiny residual correction (very unlikely to be needed)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        y += diff / n
        y = np.clip(y, 0.0, 1.0)
    return y


def propose_move(h):
    """
    Produce a neighbour of h by moving a small amount of mass from one
    index to another while keeping the total sum unchanged.
    """
    n = h.size
    for _ in range(10):
        i, j = np.random.choice(n, 2, replace=False)
        if h[i] > 1e-12 and h[j] < 1.0 - 1e-12:
            break
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h.copy()
    delta = max_delta * np.random.rand() * 0.5  # up to half of the allowable amount
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new


def gradient_step(h, alpha):
    """
    Take a small step opposite to the gradient of the dominant shift.
    The gradient of C_k (without the dx factor) at the worst shift m is:
        grad[i] = (1 - h[i+m]) - h[i-m]
    (indices out of range are treated as 0 for h).
    """
    n = h.size
    corr = np.correlate(h, 1 - h, mode="full")
    idx = np.argmax(corr)
    shift = idx - (n - 1)          # signed shift that attains the max
    i = np.arange(n)
    up = i + shift
    down = i - shift
    term1 = np.where((up >= 0) & (up < n), 1.0 - h[up], 1.0)
    term2 = np.where((down >= 0) & (down < n), h[down], 0.0)
    grad = term1 - term2
    h_new = h - alpha * grad
    return project_feasible(h_new)

        if n % 2:
            n += 1
        cur = np.zeros(n, dtype=float)
        perm = np.random.permutation(n)
        cur[perm[: n // 2]] = 1.0
        cur = project_feasible(cur)

    # ------------------------------------------------------------------
    # basic bookkeeping
    cur_val = compute_c5(cur)
    best_h = cur.copy()
    best_val = cur_val

    iter_cnt = 0
    grad_every = 500                     # perform a gradient step every this many iterations

    # ------------------------------------------------------------------
    # main stochastic search loop
    while True:
        now = time.time()
        if now - start > budget_s - 0.5:     # leave a small safety margin
            break

        # simple linear cooling based on remaining time
        remaining = max(0.0, budget_s - (now - start))
        T = max(1e-5, (remaining / budget_s) * 0.02)

        # generate a candidate
        if iter_cnt % grad_every == 0:
            cand = gradient_step(cur, alpha=0.1)
        else:
            cand = propose_move(cur)

        cand_val = compute_c5(cand)

        # Metropolis acceptance
        if cand_val < cur_val or np.random.rand() < math.exp(-(cand_val - cur_val) / T):
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_h = cand.copy()
                best_val = cand_val

        # occasional diversification: re‑seed from the current best
        if iter_cnt % 5000 == 0 and iter_cnt > 0:
            mask = np.random.rand(n) < 0.1
            rnd = np.random.rand(mask.sum())
            tmp = best_h.copy()
            tmp[mask] = rnd
            cur = project_feasible(tmp)
            cur_val = compute_c5(cur)

        iter_cnt += 1

    # ------------------------------------------------------------------
    final_val = compute_c5(best_h)
    return best_h.tolist(), float(final_val), int(n)
