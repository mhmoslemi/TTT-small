    """Project x onto box [0,1] and sum=target."""
    n = x.shape[0]
    # First clip to [0,1]
    x = np.clip(x, 0, 1)
    # Compute sum
    s = x.sum()
    # The sum must lie within [0, n] because each bound is [0,1]
    # Since target is in [0,n], feasible.
    # Use while loop adjust with waterfilling
    # We'll treat diff = target - s.
    # We'll adjust selected free indices.

    while abs(s - target) > tol and iteration < max_iter:

def _project_to_sum_box(x: np.ndarray, target: float) -> np.ndarray:
    """Clip x to [0,1] and adjust to make sum exactly target (box + linear equality)."""
    x = np.clip(x, 0.0, 1.0)
    s = x.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return x
    # increase if diff > 0
    if diff > 0:
        inc_idxs = np.where(x < 1.0)[0]
        for i in inc_idxs:
            add = min(diff, 1.0 - x[i])
            if add <= 0:
                continue
            x[i] += add
            diff -= add
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        dec_idxs = np.where(x > 0.0)[0]
        for i in dec_idxs:
            sub = min(diff, x[i])
            if sub <= 0:
                continue
            x[i] -= sub
            diff -= sub
            if diff <= 1e-12:
                break
    return x


def _compute_c5(h: np.ndarray) -> float:

    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx

            new_score = _compute_c5(neighbor)
            if new_score < cur_score:
                cur_h, cur_score = neighbor, new_score
                if new_score < best_score:
                    best_score = float(new_score)
                    best_h = neighbor.copy()

            if (it + 1) % 500 == 0 and step > 0.01:
                step *= 0.9

    # ------------------------------------------------------------------
    # Final bound (use the provided evaluator if it exists)
    # ------------------------------------------------------------------
    try:
        c5_bound = evaluate_erdos_solution(best_h)  # type: ignore[name-defined]
    except Exception:
        c5_bound = _compute_c5(best_h)

    return (best_h, c5_bound, n)
