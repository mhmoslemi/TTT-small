        if s < target:
            lo = mid
        else:

import numpy as np, time, math

def _compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral C5 for step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)


def _max_overlap_shift(h: np.ndarray):
    """Return (shift, value) where shift gives the maximal overlap C5."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    shift = idx - (n - 1)          # true integer shift
    val = conv[idx]
    return shift, float(val)


def _project_to_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the set {0 <= x_i <= 1, sum x_i = target}
    using a simple bisection on a uniform shift lambda:
        x_i = clip(v_i + λ, 0, 1)
    """
    lo = -np.max(v) - 1.0          # forces all entries to 0
    hi = 1.0 - np.min(v) + 1.0     # forces all entries to 1
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    """Create a feasible start vector of length n."""
    target = n / 2.0
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n0 = h0.size
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x1 = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x1, x0, h0)
        h = _project_to_bounds(h, target)
    else:
        rnd = np.random.rand(n)
        h = _project_to_bounds(rnd, target)
    return h


def _local_search(h0: np.ndarray, sec: float):
    """Simple simulated‑annealing / hill‑climbing on the worst shift."""
    start = time.time()
    n = h0.size
    target = n / 2.0
    cur = h0.copy()
    cur_val = _compute_c5(cur)
    best = cur.copy()
    best_val = cur_val
    it = 0

    while time.time() - start < sec:
        it += 1
        # decay factor for step size (smaller toward the end)
        decay = 1.0 - (time.time() - start) / sec
        step_scale = max(0.02, decay) * 0.2

        # shift that currently gives the maximal C5 (i.e. minimal self‑overlap)
        shift, _ = _max_overlap_shift(cur)

        # pick a pair linked by this shift, fall back to any pair if shift==0
        if shift != 0:
            if shift > 0:
                i = np.random.randint(0, n - shift)
                j = i + shift
            else:
                i = np.random.randint(-shift, n)
                j = i + shift
        else:
            i, j = np.random.choice(n, 2, replace=False)

        # move mass from the larger entry toward the smaller one
        if cur[i] > cur[j]:
            max_eps = min(cur[i], 1.0 - cur[j])
            if max_eps <= 1e-12:
                continue
            eps = max_eps * np.random.rand() * step_scale
            new = cur.copy()
            new[i] -= eps
            new[j] += eps
        elif cur[j] > cur[i]:
            max_eps = min(cur[j], 1.0 - cur[i])
            if max_eps <= 1e-12:
                continue
            eps = max_eps * np.random.rand() * step_scale
            new = cur.copy()
            new[j] -= eps
            new[i] += eps
        else:
            # equal values – just pick a random feasible transfer
            max_eps = min(cur[i], 1.0 - cur[j])
            if max_eps <= 1e-12:
                continue
            eps = max_eps * np.random.rand() * step_scale
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _compute_c5(new)

        # accept if improvement, otherwise with a tiny uphill probability
        if new_val < cur_val:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best, best_val = new.copy(), new_val
        else:
            if np.random.rand() < 0.001:
                cur, cur_val = new, new_val

        # every so often re‑centre the search around the best solution
        if it % 20000 == 0:
            if cur_val > best_val * 1.0001:
                cur, cur_val = best.copy(), best_val
            else:
                # small random perturbation of the best as a fresh start
                cur = best + 0.02 * np.random.randn(n)
                cur = np.clip(cur, 0.0, 1.0)
                cur = _project_to_bounds(cur, target)
                cur_val = _compute_c5(cur)

    return best, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) meeting the Erdos C5 constraints."""
    np.random.seed(seed)
    start_total = time.time()

    # choose a resolution – a few hundred points balances speed & quality
    base_n = int(globals().get("initial_h_values", np.zeros(0)).size)
    n = max(200, base_n * 4)          # at least 200 points, larger if a parent exists
    h0 = _initial_h(n)

    # allocate the bulk of the budget to the local search
    time_used = time.time() - start_total
    remaining = budget_s - time_used
    search_time = max(0.0, remaining * 0.9)

    best_h, _ = _local_search(h0, search_time)

    # final bound (computed exactly as the evaluator will)
    c5 = _compute_c5(best_h)

    return best_h.tolist(), float(c5), n
