import time, numpy as np
def compute_c5(h):
    n = h.size
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def min_self_corr(h):
    return np.min(np.correlate(h, h, mode='full'))

def improve_h(h, time_budget):
    start = time.time()
    n = h.size
    target_sum = n / 2
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_min_corr = min_self_corr(best_h)
    step = 0.02
    # ensure sum exactly target: we maintain via transfers
    while time.time() - start < time_budget:
        # compute self-correlation
        corr_self = np.correlate(h, h, mode='full')
        min_idx = np.argmin(corr_self)
        L = min_idx - (n - 1)
        # compute gradient contributions
        g = np.zeros_like(h)
        if L >= 0:
            g[L:] += h[:n-L]
            g[:n-L] += h[L:]
        else:
            p = -L
            g[:n-p] += h[p:]
            g[p:] += h[:n-p]
        # find indices to transfer
        inc_mask = h < 1 - 1e-12
        dec_mask = h > 1e-12
        if not inc_mask.any() or not dec_mask.any():
            break
        # index with max g among inc possible
        i_up = np.argmax(g * inc_mask) # multiplies by False = 0
        i_down = np.argmin(g + (1 - dec_mask) * 1e6) # large penalty if not dec
        # ensure i_up and i_down are distinct
        if i_up == i_down:
            # adjust random
            choices = np.where(inc_mask & dec_mask)[0]
            if choices.size >= 2:
                i_up, i_down = np.random.choice(choices, size=2, replace=False)
            else:
                # random pair
                i_up, i_down = np.random.choice(np.arange(n), size=2, replace=False)
        max_up = 1.0 - h[i_up]
        max_down = h[i_down]
        delta = min(step, max_up, max_down)
        if delta <= 1e-12:
            # reduce step
            step *= 0.5
            continue
        # apply
        h[i_up] += delta
        h[i_down] -= delta
        # compute new min self-corr
        new_min_corr = np.min(np.correlate(h, h, mode='full'))
        if new_min_corr + 1e-12 < best_min_corr:
            # revert move
            h[i_up] -= delta
            h[i_down] += delta
            step *= 0.5
            continue
        else:
            # accept move
            # update best if better
            if new_min_corr > best_min_corr:
                best_min_corr = new_min_corr
                best_h = h.copy()
                best_c5 = compute_c5(best_h)
            # maybe increase step gradually
            step = min(step * 1.005, 0.05)
        # occasionally random jitter
        if np.random.rand() < 0.001:
            # random pair small delta
            i,j = np.random.choice(n, 2, replace=False)
            max_delta = min(0.01, 1-h[i], h[j])
            if max_delta > 0:
                delta = np.random.rand() * max_delta
                h[i] += delta
                h[j] -= delta
    return best_h, best_c5

def project_to_constraints(h, target):
    # clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust sum to target by adding or subtracting from entries not at bounds
    s = h.sum()
    if abs(s - target) < 1e-12:
        return h
    # compute difference
    diff = target - s
    # We'll distribute diff equally across non-bound entries
    while abs(diff) > 1e-12:
        # find indices not at bound (0 if diff>0 else 1)
        if diff > 0:
            mask = h < 1 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        # distribute diff equally across those indices
        num = mask.sum()
        inc = diff / num
        # cap inc to not violate bounds
        # for each idx, compute max possible change
        # but we can just add inc then clip again, to iterative adjust
        h[mask] += inc
        # clip again
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target - s
    return h

            dec_indices = np.where(dec_mask)[0]

def _compute_c5(h: np.ndarray) -> float:
    """Calculate the Erdős overlap bound for a step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _min_self_corr(h: np.ndarray) -> float:
    """Minimum self‑correlation (overlap) of h with itself."""
    return float(np.min(np.correlate(h, h, mode="full")))

        else:
            # fall back to any index that can absorb the remainder
            i = np.argmin(v) if remaining > 0 else np.argmax(v)
            v[i] = np.clip(v[i] + remaining, 0.0, 1.0)
    return v


def _improve_h(h: np.ndarray, time_limit: float) -> tuple:
    """Hill‑climb to increase the minimal self‑overlap."""
    start = time.time()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c5 = _compute_c5(best_h)
    best_min = _min_self_corr(best_h)

    # initial step size (amount transferred per move)
    step = 0.02

    it = 0
    while time.time() - start < time_limit:
        it += 1
        # current self‑correlation and worst lag
        corr_self = np.correlate(h, h, mode="full")
        min_val = corr_self.min()
        min_pos = np.where(corr_self == min_val)[0]
        lag_idx = np.random.choice(min_pos)
        L = lag_idx - (n - 1)

        # gradient of the overlap sum for this lag
        g = np.zeros_like(h)
        if L >= 0:
            if L < n:
                g[L:] += h[: n - L]
                g[: n - L] += h[L:]
        else:
            p = -L
            if p < n:
                g[: n - p] += h[p:]
                g[p:] += h[: n - p]

        inc_mask = h < 1.0 - 1e-12
        dec_mask = h > 1e-12
        if not inc_mask.any() or not dec_mask.any():
            break

        g_up = g.copy()
        g_up[~inc_mask] = -np.inf
        i_up = int(np.argmax(g_up))

        g_down = g.copy()
        g_down[~dec_mask] = np.inf
        i_down = int(np.argmin(g_down))

        if i_up == i_down:
            # pick a random admissible pair if degenerate
            inc_idx = np.where(inc_mask)[0]
            dec_idx = np.where(dec_mask)[0]
            if inc_idx.size == 0 or dec_idx.size == 0:
                break
            i_up = int(np.random.choice(inc_idx))
            i_down = int(np.random.choice(dec_idx))
            if i_up == i_down:
                continue

        delta = min(step, 1.0 - h[i_up], h[i_down])
        if delta <= 1e-12:
            step *= 0.5
            continue

        # apply transfer
        h[i_up] += delta
        h[i_down] -= delta

        new_min = _min_self_corr(h)
        if new_min + 1e-12 < best_min:
            # reject – rollback
            h[i_up] -= delta
            h[i_down] += delta
            step *= 0.5
            continue

        # accept
        if new_min > best_min + 1e-12:
            best_min = new_min
            best_h = h.copy()
            best_c5 = _compute_c5(best_h)

        # adapt step size slowly
        step = min(step * 1.001, 0.05)

        # occasional random jitter to avoid stagnation
        if it % 500 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            cap = min(0.01, 1.0 - h[i], h[j])
            if cap > 0:
                d = np.random.rand() * cap
                h[i] += d
                h[j] -= d

    # final projection (should already be feasible)
    best_h = _project_to_constraints(best_h, target)
    return best_h, _compute_c5(best_h)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function giving a low Erdős C₅ bound."""
    np.random.seed(seed)
    start_all = time.time()

    # Determine problem size.
    try:
        init = np.asarray(initial_h_values, dtype=float)
        n = init.size
        h_start = _project_to_constraints(init.copy(), n / 2.0)
    except NameError:
        # No parent supplied – use a simple binary start.
        n = 84  # an even number giving decent resolution
        h_start = np.zeros(n, dtype=float)
        h_start[: n // 2] = 1.0
        np.random.shuffle(h_start)

    target_sum = n / 2.0
    best_h = None
    best_c5 = np.inf

    # First try the supplied (or binary) start.
    time_left = budget_s - (time.time() - start_all)
    if time_left > 5:
        h_opt, c5 = _improve_h(h_start, time_left * 0.5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Perform a few random restarts.
    while time.time() - start_all < budget_s - 0.5:
        remaining = budget_s - (time.time() - start_all)
        if remaining < 5:
            break
        # Random binary seed.
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)
        # Slight random perturbation to break symmetry.
        perturb = np.random.rand(n) * 0.1
        h0 = np.clip(h0 + perturb, 0.0, 1.0)
        h0 = _project_to_constraints(h0, target_sum)

        attempt_time = min(remaining * 0.3, 30.0)
        h_opt, c5 = _improve_h(h0, attempt_time)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    if best_h is None:
        best_h = h_start.copy()
        best_c5 = _compute_c5(best_h)

    # Ensure final feasibility.
    best_h = _project_to_constraints(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h, float(final_c5), int(n)
