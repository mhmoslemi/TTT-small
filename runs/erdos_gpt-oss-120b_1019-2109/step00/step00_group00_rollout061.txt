    best_h_local = h.copy()
    best_val_local = max_overlap(h)

import time, math, numpy as np
from scipy.optimize import minimize

# ----- objective (Lp‑norm of the correlation) -----
def _corr_vec(h):
    """full correlation of h with (1‑h)"""
    return np.correlate(h, 1.0 - h, mode='full')

def _lp_obj(h, N, p):
    """Lp‑norm of the correlation (smooth surrogate of max)"""
    c = _corr_vec(h)
    return (np.sum(c ** p)) ** (1.0 / p)

def _lp_grad(h, N, p):
    """gradient of the Lp surrogate"""
    c = _corr_vec(h)
    cp = c ** p
    F = np.sum(cp)
    f = F ** (1.0 / p)

    # weights for each shift
    if f == 0.0:
        w = np.zeros_like(c)
    else:
        # w = (c / f) ** (p‑1) ; zero where c==0
        with np.errstate(divide='ignore', invalid='ignore'):
            w = (c / f) ** (p - 1)
            w = np.where(c <= 0.0, 0.0, w)

    grad = np.zeros(N, dtype=float)
    # shifts run from -(N‑1) … N‑1
    for idx, sh in enumerate(range(-N + 1, N)):
        ws = w[idx]
        if ws == 0.0:
            continue
        if sh >= 0:
            end = N - sh
            if end:
                grad[:end] += ws * (1.0 - h[sh:])          # (1‑h_{j+ℓ})
                grad[sh:]   -= ws * h[:end]                # –h_{j‑ℓ}
        else:
            s = -sh
            end = N - s
            if end:
                grad[s:]    += ws * (1.0 - h[:end])        # (1‑h_{j+ℓ})
                grad[:end]  -= ws * h[s:]                  # –h_{j‑ℓ}
    return grad

# ----- equality constraint (integral = 1) -----
def _eq_sum(h, target):
    return np.sum(h) - target

def _eq_sum_jac(_h, _target):
    return np.ones_like(_h)

# ----- helper to build a start vector ---------------------------------
def _binary_start(N):
    """random binary vector with exactly N/2 ones"""
    h = np.zeros(N, dtype=float)
    ones = np.random.choice(N, N // 2, replace=False)
    h[ones] = 1.0
    return h

def _repeat_half_start(N):
    """pattern of length N/2 repeated twice – often a good seed"""
    half = np.zeros(N // 2, dtype=float)
    idx = np.random.choice(N // 2, N // 4, replace=False)
    half[idx] = 1.0
    return np.concatenate([half, half])

def _upscaled_initial(N):
    """use the provided initial_h_values (if present) and turn it into a binary start"""
    global initial_h_values
    if 'initial_h_values' not in globals():
        return None
    init = np.asarray(initial_h_values, dtype=float)
    L = init.size
    # linear interpolation onto N points
    x_old = np.linspace(0.0, 2.0, L, endpoint=False)
    x_new = np.linspace(0.0, 2.0, N, endpoint=False)
    up = np.interp(x_new, x_old, init)
    # keep the N/2 largest values as ones
    h = np.zeros(N, dtype=float)
    h[np.argsort(-up)[: N // 2]] = 1.0
    return h

# ----- main entry point -------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # choose discretisation (even for binary convenience)
    N = 100                     # number of sample points (dx = 2/N)
    p = 20                      # exponent for the Lp‑norm surrogate
    target_sum = N / 2.0
    max_iter_opt = 180          # SLSQP iterations per restart
    best_h = None
    best_val = math.inf

    # constraint dictionary for SLSQP (allows passing the target via args)
    cons = [{
        'type': 'eq',
        'fun': _eq_sum,
        'jac': _eq_sum_jac,
        'args': (target_sum,)
    }]

    # bounds for every coordinate
    bounds = [(0.0, 1.0) for _ in range(N)]

    # ---- a small pool of initial seeds ---------------------------------
    seeds = []
    # 1) upscaled problem specific start (if available)
    up = _upscaled_initial(N)
    if up is not None:
        seeds.append(up)
    # 2) nice symmetric binary start (repeat‑half)
    seeds.append(_repeat_half_start(N))
    # 3) plain random binary start
    seeds.append(_binary_start(N))

    # ---- optimisation loop (time‑bounded) -------------------------------
    # we keep trying different seeds until the budget is exhausted
    while time.time() - start_time < budget_s * 0.95:
        # pick a seed (cycle through the list)
        h0 = seeds[len(seeds) % len(seeds)].copy()
        seeds.append(h0)   # rotate

        # SLSQP minimise the Lp‑norm surrogate
        try:
            res = minimize(
                fun=_lp_obj,
                x0=h0,
                args=(N, p),
                jac=_lp_grad,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_iter_opt, 'ftol': 1e-9, 'disp': False}
            )
        except Exception:
            continue

        if not res.success:
            # fallback to the raw result if it is feasible
            h_opt = np.clip(res.x, 0.0, 1.0)
        else:
            h_opt = np.clip(res.x, 0.0, 1.0)

        # enforce the integral exactly (tiny numeric drift)
        cur = np.sum(h_opt)
        if abs(cur - target_sum) > 1e-6:
            # simple projection onto the hyper‑plane while staying in [0,1]
            diff = target_sum - cur
            # distribute the correction uniformly, then clip
            h_opt += diff / N
            np.clip(h_opt, 0.0, 1.0, out=h_opt)
            # re‑normalise if needed
            cur = np.sum(h_opt)
            if abs(cur - target_sum) > 1e-6:
                # final tiny adjustment on a random coordinate
                i = np.random.randint(N)
                h_opt[i] += target_sum - cur
                h_opt[i] = min(1.0, max(0.0, h_opt[i]))

        # evaluate the true objective (max overlap)
        cor = _corr_vec(h_opt)
        bound = np.max(cor) * (2.0 / N)

        if bound < best_val:
            best_val = bound
            best_h = h_opt.copy()

        # early exit if we already beat the target record
        if best_val <= 0.3808:
            break

    # ----- fallback -------------------------------------------------------
    if best_h is None:
        # trivial feasible (uniform) solution – never happens in practice
        best_h = np.full(N, 0.5)
        best_val = np.max(_corr_vec(best_h)) * (2.0 / N)

    return (best_h, float(best_val), N)
