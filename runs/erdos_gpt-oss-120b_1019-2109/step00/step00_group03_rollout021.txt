    C = np.correlate(h, 1 - h, mode='full')
    # compute weights
    maxalphaC = np.max(alpha * C)
    exp_vals = np.exp(alpha * C - maxalphaC)
    sumexp = exp_vals.sum()
    w = exp_vals / sumexp

    grad = np.zeros_like(h)

    # zero shift
    w0 = w[n-1]
    grad += w0 * (1 - 2 * h)

    for k in range(1, n):

        w_minus = w[n-1 - k]

def _project_onto_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy sum(v)=target."""
    low, high = -5.0, 5.0
    for _ in range(60):
        mid = (low + high) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            low = mid
        else:
            high = mid
    h = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    # tiny residual fix
    resid = target - h.sum()
    if abs(resid) > 1e-12:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            h[free] += resid / free.sum()
        else:
            idx = np.argmax(h)
            h[idx] = np.clip(h[idx] + resid, 0.0, 1.0)
        h = np.clip(h, 0.0, 1.0)
    return h


def _max_corr(h: np.ndarray, dx: float) -> float:
    """Exact objective: max_k ∫ h(x)(1‑h(x+k)) dx approximated by correlation."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def _smooth_max_and_grad(h: np.ndarray, alpha: float):
    """
    Smooth approximation of the max and its gradient.
    Returns (value, gradient) w.r.t. h.
    """
    n = h.shape[0]
    C = np.correlate(h, 1.0 - h, mode="full")          # length 2n‑1
    aC = alpha * C
    m = aC.max()
    exp_vals = np.exp(aC - m)                         # stable softmax
    sumexp = exp_vals.sum()
    w = exp_vals / sumexp                             # weights, sum to 1

    # gradient of the (non‑scaled) smooth max
    grad = np.zeros_like(h)
    w0 = w[n - 1]
    grad += w0 * (1.0 - 2.0 * h)                      # shift 0

    for k in range(1, n):
        w_plus = w[n - 1 + k]
        w_minus = w[n - 1 - k]

        # +k shift contributions
        grad[: n - k] += w_plus * (1.0 - h[k:])
        grad[k:]       -= w_plus * h[: n - k]

        # –k shift contributions
        grad[: n - k] -= w_minus * h[k:]
        grad[k:]       += w_minus * (1.0 - h[: n - k])

    # smooth max value = (log sum exp(alpha*C))/alpha
    smooth_val = (m + np.log(sumexp)) / alpha
    return smooth_val, grad

    n = h0.shape[0]
    dx = 2.0 / n
    target_sum = n * 0.5                                 # ∫ h = 1  ⇒ Σh = n/2
    h0 = _project_onto_bounds_sum(h0, target_sum)

    best_h = h0.copy()
    best_val = _max_corr(best_h, dx)

    # ---- optimisation parameters -------------------------------------------
    alphas = [5.0, 12.0, 25.0, 50.0]                     # gradually sharpen
    bounds = [(0.0, 1.0)] * n
    eq_con = {
        "type": "eq",
        "fun": lambda x: np.sum(x) - target_sum,
        "jac": lambda x: np.ones_like(x),
    }

    start = time.time()
    elapsed = 0.0
    x = h0.copy()

    # ---- smooth‑max SLSQP optimisation --------------------------------------
    for alpha in alphas:
        if elapsed > budget_s - 5.0:
            break

        def fun(v):
            val, _ = _smooth_max_and_grad(v, alpha)
            return val

        def jac(v):
            _, g = _smooth_max_and_grad(v, alpha)
            return g

        try:
            res = minimize(
                fun=fun,
                x0=x,
                jac=jac,
                bounds=bounds,
                constraints=[eq_con],
                method="SLSQP",
                options={"maxiter": 1500, "ftol": 1e-9},
            )
        except Exception:
            break

        x = np.clip(res.x, 0.0, 1.0)
        x = _project_onto_bounds_sum(x, target_sum)

        cur_val = _max_corr(x, dx)
        if cur_val < best_val:
            best_val = cur_val
            best_h = x.copy()

        elapsed = time.time() - start

    # ---- cheap random local search for final polishing -----------------------
    while time.time() - start < budget_s - 2.0:
        i, j = np.random.choice(n, 2, replace=False)
        # amount we can shift from i to j while staying feasible
        delta_max = min(best_h[i], 1.0 - best_h[j])
        if delta_max <= 0.0:
            continue
        # pick a random fraction of the available amount
        delta = delta_max * (np.random.rand() * 0.3 + 0.1)   # 10‑40 % of max
        h_new = best_h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        val_new = _max_corr(h_new, dx)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new

    # final projection (safety)
    best_h = _project_onto_bounds_sum(best_h, target_sum)

    return best_h, best_val, n
