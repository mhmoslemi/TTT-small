import time, math, random
import numpy as np

def _project_to_sum_box(v, target, tol=1e-9, maxit=60):
    """Project v onto the set {x: sum(x)=target, 0<=x<=1}."""
    lo, hi = -2.0, 2.0
    # enlarge bounds until they bracket the solution
    while np.clip(v + lo, 0, 1).sum() > target:
        lo *= 2
    while np.clip(v + hi, 0, 1).sum() < target:
        hi *= 2
    for _ in range(maxit):
        mid = (lo + hi) * 0.5
        y = np.clip(v + mid, 0, 1)
        s = y.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) <= tol:
            break
    return np.clip(v + (lo + hi) * 0.5, 0, 1)


def _upsample(parent, N):
    """Linear interpolation of parent (array) to length N, then enforce sum."""
    m = len(parent)
    if m == N:
        return parent.copy()
    old_x = np.linspace(0.0, 2.0, m, endpoint=False)
    new_x = np.linspace(0.0, 2.0, N, endpoint=False)
    up = np.interp(new_x, old_x, parent)
    up = _project_to_sum_box(up, N / 2.0)
    return up


def _random_move(h, max_step):
    """Pick two indices i,j and transfer a small amount while keeping bounds."""
    n = h.shape[0]
    for _ in range(10):
        i, j = np.random.choice(n, 2, replace=False)
        # delta added to i, subtracted from j
        lo = max(-h[i], h[j] - 1.0, -max_step)
        hi = min(1.0 - h[i], h[j], max_step)
        if hi <= lo:
            continue
        delta = np.random.uniform(lo, hi)
        cand = h.copy()
        cand[i] += delta
        cand[j] -= delta
        return cand
    return None


def _directed_move(h, shift, max_step):
    """Transfer mass from a mismatched 1-position to its partner 0-position."""
    n = h.shape[0]
    if shift == 0:

    # indices where i+shift is inside array

    n = h.shape[0]

    conv = np.correlate(h, 1.0 - h, mode="full")

            # directed move using the worst shift of the *current* function
            shift = np.argmax(cur_conv) - (n_points - 1)
            cand = _directed_move(h, shift, max_step)
        else:
            cand = _random_move(h, max_step)

        if cand is None:
            continue

        cand_val, cand_conv = _c5_value(cand)
        delta = cand_val - cur_val

        if delta < 0 or np.random.rand() < math.exp(-delta / T):
            h = cand
            cur_val = cand_val
            cur_conv = cand_conv
            if cand_val < best_val:
                best_val = cand_val
                best_h = cand.copy()

    # final projection to guarantee feasibility
    best_h = _project_to_sum_box(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)

    final_val = _c5_value(best_h)[0]
    return best_h, final_val, n_points
