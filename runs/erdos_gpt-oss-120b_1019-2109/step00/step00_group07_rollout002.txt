import numpy as np
import time
import random

def _compute_c5(h, dx):
    """Maximum of the overlap integral for a given h."""
    # correlation of h with (1-h), full mode → includes all integer shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def _project_to_sum_one_half(h, target_sum, eps=1e-12, max_iter=100):
    """Adjust h ∈ [0,1] so that its sum equals target_sum."""
    # simple iterative projection: distribute excess/deficit uniformly.
    for _ in range(max_iter):
        cur = h.sum()
        diff = cur - target_sum
        if abs(diff) <= eps:
            break
        if diff > 0:
            # need to reduce total mass
            mask = h > 0
            if not np.any(mask):
                break
            reduction = diff / mask.sum()
            # ensure we don't go below 0
            reduction = min(reduction, h[mask].min())
            h[mask] -= reduction
        else:
            # need to increase total mass
            mask = h < 1
            if not np.any(mask):
                break
            addition = -diff / mask.sum()
            addition = min(addition, 1 - h[mask].max())
            h[mask] += addition
        # clip to stay inside bounds (numerical safety)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _random_binary_solution(n, target_sum):
    """Generate a binary feasible vector of length n with exactly target_sum ones."""
    h = np.zeros(n, dtype=float)
    idx = np.random.permutation(n)[:int(target_sum)]
    h[idx] = 1.0
    return h

def _propose_move(h, eps=1e-12):
    """Return a new h obtained by moving a small amount of mass between two coordinates."""
    n = h.shape[0]
    i, j = np.random.randint(n, size=2)
    while i == j:
        i, j = np.random.randint(n, size=2)

    # random direction: from i → j or from j → i
    if np.random.rand() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i

    # max admissible transfer while staying in [0,1]
    max_delta = min(h[src], 1.0 - h[dst])
    if max_delta <= eps:
        return None
    # choose a random portion of the feasible delta (up to 50% of max)
    delta = np.random.rand() * max_delta * 0.5
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize the step function for the Erdős minimum overlap problem.
    Returns a tuple (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ----------------------------------------------------------------------
    # 1. Load or construct an initial feasible vector
    # ----------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        # fallback: simple contiguous block of 1's (always feasible)
        n_fallback = 80
        h0 = np.zeros(n_fallback, dtype=float)
        h0[: n_fallback // 2] = 1.0

    n = h0.shape[0]
    dx = 2.0 / n
    target_sum = n / 2.0

    # Ensure feasibility (bounds and exact sum)
    np.clip(h0, 0.0, 1.0, out=h0)
    h0 = _project_to_sum_one_half(h0, target_sum)

    # ----------------------------------------------------------------------
    # 2. Baseline evaluation
    # ----------------------------------------------------------------------
    best_h = h0.copy()
    best_c5 = _compute_c5(best_h, dx)

    # keep a mutable current state for hill‑climbing
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ----------------------------------------------------------------------
    # 3. Main stochastic local search
    # ----------------------------------------------------------------------
    start_time = time.time()
    time_limit = budget_s - 0.3   # tiny safety margin

    # parameters controlling occasional restarts
    restart_prob = 0.001
    iteration = 0

        if cand_h is None:

        cand_c5 = _compute_c5(cand_h, dx)

        if cand_c5 <= cur_c5 + 1e-12:
