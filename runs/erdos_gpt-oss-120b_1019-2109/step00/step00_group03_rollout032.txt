import time, numpy as np, cvxpy as cp, random

def evaluate_corr(h, dx):
    corr = np.correlate(h.astype(np.float64), (1-h).astype(np.float64), mode='full')
    max_val = int(np.max(corr))
    return max_val, max_val * dx

def evaluate_max_corr(h):
    return np.max(np.correlate(h, 1-h, mode='full'))

def hill_climb(h, cur_max, dx, time_limit):
    n = len(h)
    indices_one = np.where(h==1)[0].tolist()
    indices_zero = np.where(h==0)[0].tolist()
    random.shuffle(indices_one)
    random.shuffle(indices_zero)
    start = time.time()
    best_h = h.copy()
    best_val = cur_max
    while time.time() - start < time_limit:
        i = random.choice(indices_one)
        j = random.choice(indices_zero)
        # swap
        h[i] = 0
        h[j] = 1
        # compute new max
        new_val = evaluate_max_corr(h)
        if new_val <= cur_max:
            # accept
            cur_max = new_val
            # update lists
            indices_one.remove(i)
            indices_one.append(j)
            indices_zero.remove(j)
            indices_zero.append(i)
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                # optional early exit if best_val small enough (like 13)
                if best_val <= 13:
                    break
        else:
            # revert
            h[i] = 1
            h[j] = 0
    return best_h, best_val

def solve_mip(n, target, time_limit):
    # build model as described
    import cvxpy as cp
    x = cp.Variable(n, boolean=True)
    Y = cp.Variable((n, n), boolean=True)
    constraints = []
    # sum constraint
    constraints.append(cp.sum(x) == target)
    # linearization
    for i in range(n):
        constraints.append(Y[i,i] == x[i])  # optional
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(Y[i,j] <= x[i])
            constraints.append(Y[i,j] <= x[j])
            constraints.append(Y[i,j] >= x[i] + x[j] - 1)
    # t variable
    t = cp.Variable()
    constraints.append(t >= 0)
    # for shifts k
    for k in range(-n+1, n):
        if k == 0:
            # correlation zero automatically (since sum x - sum y on same indices equals 0)
            continue
        if k > 0:
            region_sum = cp.sum(x[:n - k])
            sum_y = cp.sum([Y[i, i + k] for i in range(n - k)])
        else:
            m = -k
            region_sum = cp.sum(x[m:])
            sum_y = cp.sum([Y[i - m, i] for i in range(m, n)])
        corr = region_sum - sum_y
        constraints.append(t >= corr)
    # objective
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    # Solve with time limit
    solvers = [s for s in cp.installed_solvers() if s in ('CBC', 'GLPK_MI', 'ECOS_BB')]
    if not solvers:
        raise RuntimeError('No suitable MILP solver')
    solver = solvers[0]
    if solver == 'CBC':
        prob.solve(solver=solver, verbose=False, warm_start=True, timelimit=time_limit)
    elif solver == 'GLPK_MI':
        prob.solve(solver=solver, verbose=False, warm_start=True, glpk={'tm_lim': int(time_limit*1000)})
    elif solver == 'ECOS_BB':
        prob.solve(solver=solver, verbose=False, warm_start=True, max_iters=int(time_limit*100))
    else:
        prob.solve(solver=solver, verbose=False, warm_start=True)
    # Check status
    if prob.status not in ('optimal', 'optimal_inaccurate'):
        # May return best feasible
        pass
    x_val = np.rint(x.value).astype(int)  # round
    # Ensure exact target sum
    cur_sum = x_val.sum()
    if cur_sum != target:
        # adjust heuristically
        diff = target - cur_sum
        if diff > 0:
            zeros = np.where(x_val == 0)[0]
            flip = np.random.choice(zeros, diff, replace=False)
            x_val[flip] = 1
        elif diff < 0:
            ones = np.where(x_val == 1)[0]
            flip = np.random.choice(ones, -diff, replace=False)
            x_val[flip] = 0
    max_val = evaluate_max_corr(x_val)
    return x_val, max_val

import time, random, numpy as np, cvxpy as cp

def _max_corr(h):
    # h is binary (0/1) vector
    return int(np.max(np.correlate(h, 1 - h, mode='full')))

def _hill_climb(h, cur_max, timelimit):
    """simple swap hill‑climb, works on binary vectors"""
    n = len(h)
    idx_one = list(np.where(h == 1)[0])
    idx_zero = list(np.where(h == 0)[0])
    best_h, best_val = h.copy(), cur_max
    start = time.time()
    while time.time() - start < timelimit:
        i = random.choice(idx_one)
        j = random.choice(idx_zero)
        h[i] = 0
        h[j] = 1
        new_val = _max_corr(h)
        if new_val <= cur_max:
            idx_one.remove(i)
            idx_one.append(j)
            idx_zero.remove(j)
            idx_zero.append(i)
            cur_max = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
                if best_val <= 13:        # 13 → C5≈0.3714, already beating target
                    break
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_val

def _solve_mip(n, target, timelimit):
    """exact MILP formulation (binary variables only)"""
    x = cp.Variable(n, boolean=True)            # h_i
    Y = cp.Variable((n, n), boolean=True)      # y_{i,j}=x_i·x_j
    cons = [cp.sum(x) == target]

    # diagonal consistency (optional)
    for i in range(n):
        cons.append(Y[i, i] == x[i])

    # product linearisation for i<j
    for i in range(n):
        for j in range(i + 1, n):
            cons.append(Y[i, j] <= x[i])
            cons.append(Y[i, j] <= x[j])
            cons.append(Y[i, j] >= x[i] + x[j] - 1)

    t = cp.Variable()          # max correlation to minimise
    cons.append(t >= 0)

    # constraints for every shift k
    for k in range(-n + 1, n):
        if k == 0:
            continue
        if k > 0:
            region = cp.sum(x[: n - k])
            overlap = cp.sum([Y[i, i + k] for i in range(n - k)])
        else:
            m = -k
            region = cp.sum(x[m:])
            overlap = cp.sum([Y[i - m, i] for i in range(m, n)])
        cons.append(t >= region - overlap)

    prob = cp.Problem(cp.Minimize(t), cons)
    installed = cp.installed_solvers()
    if 'GLPK_MI' in installed:
        prob.solve(solver='GLPK_MI',
                   glpk={'tm_lim': int(timelimit * 1000)},
                   warm_start=True, verbose=False)
    elif 'CBC' in installed:

                   cbc={'seconds': timelimit},

    elif 'ECOS_BB' in installed:

                   max_iters=int(timelimit * 100),
                   warm_start=True, verbose=False)
    else:
        prob.solve()

        raise RuntimeError('MILP solver failed')

        else:
            ones = np.where(xv == 1)[0]
            np.random.shuffle(ones)
            xv[ones[: -diff]] = 0
