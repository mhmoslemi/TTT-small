    def compute_C(h):
        n = len(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n)

import numpy as np, time, random

def hill_climb_binary(n, init_h=None, max_steps=5000):
    if init_h is not None and len(init_h)==n:
        h = np.array(init_h, dtype=np.uint8)
        # ensure sum == n/2: If not, we could adjust by randomly flipping positions.
        # but assume it's fine.
    else:
        h = np.zeros(n, dtype=np.uint8)
        ones = np.random.choice(n, n//2, replace=False)
        h[ones] = 1

    current_corr = np.max(np.correlate(h, 1 - h, mode='full'))
    current_C = current_corr * (2.0 / n)

    best_h = h.copy()
    best_C = current_C

    for step in range(max_steps):
        # random swap 1 and 0
        ones_idx = np.where(h==1)[0]
        zeros_idx = np.where(h==0)[0]
        i = random.choice(ones_idx)
        j = random.choice(zeros_idx)
        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1
        new_corr = np.max(np.correlate(h_new, 1 - h_new, mode='full'))
        new_C = new_corr * (2.0 / n)
        # Accept if improved
        if new_C < current_C:
            h = h_new
            current_C = new_C
            if new_C < best_C:
                best_C = new_C
                best_h = h.copy()
        else:
            # maybe accept with low probability
            if random.random() < 0.001:
                h = h_new
                current_C = new_C
    return best_h, best_C

def improve_once(h):
    n = len(h)

    idx = int(np.argmax(corr))
    shift = idx - (n - 1)

    # find mismatched positions (i where h[i]=1, h[i+shift]=0)
    if shift >=0:
        i_range = np.arange(n - shift)
        first = h[i_range]
        second = h[i_range + shift]
        mask = (first == 1) & (second == 0)
        mismatched_i = i_range[mask]
