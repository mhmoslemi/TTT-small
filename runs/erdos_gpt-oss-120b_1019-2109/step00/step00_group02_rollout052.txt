import time, random, numpy as np

def _c5_value(h: np.ndarray):
    """Return C5 for a step vector h."""
    n = h.shape[0]
    # Correlation of h with (1-h) in linear (zero‑padded) mode
    corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
    return float(corr.max() * (2.0 / n)), corr


def _local_opt_binary(h: np.ndarray, max_iter=3000, patience=300):
    """Greedy improvement for binary h (0/1) using swaps that reduce the worst lag."""
    n = h.shape[0]
    best_c, _ = _c5_value(h)
    no_improve = 0
    for _ in range(max_iter):
        corr = np.correlate(h.astype(np.float64), 1.0 - h.astype(np.float64), mode="full")
        max_val = corr.max()
        if max_val == 0:
            break
        s = int(corr.argmax() - (n - 1))          # shift giving the current max
        if s >= 0:
            # overlapping region length = n-s
            a = h[: n - s]
            b = h[s:]
            mis1 = np.where((a == 1) & (b == 0))[0]   # 1 -> 0
            mis0 = np.where((a == 0) & (b == 1))[0]   # 0 -> 1
            if mis1.size == 0 or mis0.size == 0:
                break
            i_local = mis1[random.randrange(mis1.size)]
            j_local = mis0[random.randrange(mis0.size)]
            i, j = i_local, j_local + s
        else:                       # s < 0
            t = -s
            a = h[t:]
            b = h[: n - t]
            mis1 = np.where((a == 1) & (b == 0))[0]
            mis0 = np.where((a == 0) & (b == 1))[0]
            if mis1.size == 0 or mis0.size == 0:
                break
            i_local = mis1[random.randrange(mis1.size)]
            j_local = mis0[random.randrange(mis0.size)]
            i, j = i_local + t, j_local + t
        # perform the swap
        h[i], h[j] = h[j], h[i]
        new_c, _ = _c5_value(h)
        if new_c < best_c - 1e-12:

            no_improve = 0
        else:
            # revert
            h[i], h[j] = h[j], h[i]
            no_improve += 1
