    a = (2 - L) / 2.0
    H = 1.0 / (1.0 + L/2.0)  # ensures area =1

    mask1 = xs < a

    # region 2: a <= x <= a+L
    mask2 = (xs >= a) & (xs <= a+L)
    h[mask2] = H
    # region 3: xs > a+L
    mask3 = xs > a+L
    h[mask3] = H - (H / a) * (xs[mask3] - (a + L))
    # ensure bounds [0,1]

    v = v / v.sum() * w

        excess_amount = excess[over_idx]

        v[over_idx] = 1.0

import time
import numpy as np

def _evaluate(h, dx):
    """Return max overlap integral C5 for a given step function h."""
    # correlation of h with (1-h); mode='full' gives all shifts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _shape_trapezoidal(n, L):
    """
    Construct a symmetric trapezoidal (or triangular) shape on [0,2] with
    plateau length L (0 ≤ L ≤ 2). The continuous shape has integral 1.
    """
    # avoid division by zero
    if L >= 2.0 - 1e-12:
        return np.full(n, 0.5)                     # uniform (limit case)
    a = (2.0 - L) / 2.0                            # slope length on each side
    H = 1.0 / (1.0 + L / 2.0)                      # plateau height (ensures area 1)
    xs = (np.arange(n) + 0.5) * (2.0 / n)          # mid‑point sample positions
    h = np.empty(n, dtype=float)

    # left ramp
    mask1 = xs < a
    h[mask1] = (H / a) * xs[mask1]

    # plateau
    mask2 = (xs >= a) & (xs <= a + L)
    h[mask2] = H

    # right ramp
    mask3 = xs > a + L
    h[mask3] = H - (H / a) * (xs[mask3] - (a + L))

    # clip to [0,1] (tiny numerical overshoots can happen)
    h = np.clip(h, 0.0, 1.0)

    # enforce exact sum = n/2 (integral = 1)
    w = n / 2.0
    diff = w - h.sum()
    if abs(diff) > 1e-12:
        # adjust a single entry that is not at a bound
        if diff > 0:
            idx = np.argmin(h)                      # smallest value
            delta = min(diff, 1.0 - h[idx])
            h[idx] += delta
        else:
            idx = np.argmax(h)                      # largest value
            delta = min(-diff, h[idx])
            h[idx] -= delta
    return h

def _transfer_move(h, shift, delta):
    """
    Transfer a mass `delta` from index i to i+shift (if feasible).
    Returns a new array or None if no feasible pair exists.
    """
    n = h.shape[0]
    start = max(0, -shift)
    end = min(n, n - shift)
    if end <= start:
        return None
    hi = h[start:end]
    hj = h[start + shift : end + shift]
    cand = np.where((hi > 1e-10) & (hj < 1.0 - 1e-10))[0]
    if cand.size == 0:
        return None
    i = cand[np.random.randint(cand.size)] + start
    j = i + shift
    delta = min(delta, h[i], 1.0 - h[j])
    if delta <= 0:
        return None
    h_new = h.copy()
    h_new[i] -= delta
    h_new[j] += delta
    return h_new
