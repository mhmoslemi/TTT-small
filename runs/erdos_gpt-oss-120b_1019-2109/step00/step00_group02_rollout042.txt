def solve_binary_milp(N, time_limit):
    M = N//2
    h = cp.Variable(N, boolean=True)
    lam = cp.Variable()
    constraints = [cp.sum(h) == M,
                   lam <= M]
    w_by_t = []
    for t in range(N):
        w_t = []
        for i in range(N - t):
            w = cp.Variable(boolean=True)
            constraints.append(w <= h[i])
            constraints.append(w <= h[i + t])
            constraints.append(w >= h[i] + h[i + t] - 1)
            w_t.append(w)
        constraints.append(cp.sum(w_t) >= lam)
        w_by_t.append(w_t)
    prob = cp.Problem(cp.Maximize(lam), constraints)
    try:
        prob.solve(solver=cp.CBC, timeLimit=int(time_limit))
    except Exception as e:
        # fallback to None
        return None, None
    if prob.status not in ["optimal","optimal_inaccurate"]:
        return None, None
    h_val = np.round(np.array(h.value)).astype(float)
    lam_val = lam.value
    return h_val, lam_val

def project_onto_capped_simplex(v, target):
    # Uses bisection to find tau such that sum(clip(v - tau, 0,1)) = target
    lower = np.min(v) - 1   # lower bound for tau
    upper = np.max(v)       # upper bound for tau
    for _ in range(100):
        tau = (lower + upper) / 2.0
        proj = np.clip(v - tau, 0, 1)
        s = proj.sum()
        if np.abs(s - target) < 1e-9:
            break
        if s > target:
            lower = tau
        else:
            upper = tau
    return proj

def hill_climb_binary(N, time_limit, start_h=None):
    M = N // 2
    if start_h is None:
        # random binary with M ones, ensure edges 0 and N-1 are 1 to help large shifts
        h = np.zeros(N, dtype=float)
        # set edges
        h[0] = 1
        h[-1] = 1
        remaining = M - 2
        idx = np.random.choice(np.arange(1, N-1), remaining, replace=False)
        h[idx] = 1
    else:
        # start_h may be fractional; convert to binary by threshold
        h = (start_h >= 0.5).astype(float)
        # adjust to exactly M ones:
        # if sum > M, randomly flip some ones to zero; if sum < M, flip some zeros.
        while h.sum() > M:
            ones = np.where(h==1)[0]
            i = np.random.choice(ones)
            h[i] = 0
        while h.sum() < M:
            zeros = np.where(h==0)[0]
            i = np.random.choice(zeros)
            h[i] = 1

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start_time = time.time()
    # precompute indices of ones and zeros for faster swaps
    ones = list(np.where(h==1)[0])
    zeros = list(np.where(h==0)[0])
    # iteration loop
    while time.time() - start_time < time_limit:
        # pick random 1 and 0
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        # compute c5
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-8:
            best_c5 = c5
            best_h = h.copy()
            # update ones/zeros sets accordingly

            ones.remove(i)
            ones.append(j)

            zeros.append(i)
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        # maybe occasionally perturb (swap two pairs) to escape local minima
    return best_h, best_c5

def gradient_from_weights(h, w):
    N = len(h)
    grad = np.ones(N)  # sum of w =1
    # w corresponds to t values from -(N-1) .. N-1
    t_vals = np.arange(-N+1, N)  # same order as autocorr/ O arr
    for w_t, t in zip(w, t_vals):
        if w_t == 0:
            continue
        if t >= 0:
            i_start = 0
            i_end = N - t
            if i_end > i_start:
                grad[i_start:i_end] -= w_t * h[t:i_end + t]  # h[i+t]
            # for i-t term
            i_start2 = t
            i_end2 = N
            if i_end2 > i_start2:
                grad[i_start2:i_end2] -= w_t * h[i_start2 - t:i_end2 - t]  # h[i-t] = h[i - t] = h[i + (-t)]
        else:
            t_abs = -t
            # i_start = t_abs
            i_start = t_abs
            i_end = N
            if i_end > i_start:
                grad[i_start:i_end] -= w_t * h[i_start - t_abs : i_end - t_abs]  # h[i + t] where t negative -> h[i - t_abs]
            # i-t term: i - t = i + t_abs
            i_start2 = 0
            i_end2 = N - t_abs
            if i_end2 > i_start2:
                grad[i_start2:i_end2] -= w_t * h[t_abs:i_end2 + t_abs]  # h[i - t] = h[i + t_abs]
    return grad

import numpy as np, time

def compute_autocorr(h):
    N = h.shape[0]
    m = 2*N - 1
    H = np.fft.rfft(h, n=m)
    auto = np.fft.irfft(H * np.conj(H), n=m)
    return auto

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    auto = compute_autocorr(h)
    O = 1 - auto * dx
    return float(np.max(O))

def project_onto_capped_simplex(v, target, max_iter=100):
    # binary search for lambda such that sum clip(v - lambda, 0, 1) = target
    lo = -2.0
    hi = 2.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if abs(s - target) < 1e-9:
            return proj
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def hill_climb_binary(N, time_limit, start_h=None):
    M = N // 2
    rng = np.random.default_rng()
    # start with fixed edges 1
    h = np.zeros(N, dtype=float)
    h[0] = 1.0
    h[-1] = 1.0
    remaining = M - 2
    if remaining < 0:
        # N=2 case, adjust
        remaining = M
        h[:] = 0.0
    idx = rng.choice(np.arange(1, N-1), size=remaining, replace=False)
    h[idx] = 1.0
    # if start_h provided, incorporate rounding
    if start_h is not None:
        # round start_h to binary but keep sum M
        binary = (start_h >= 0.5).astype(float)
        # ensure edges are 1
        binary[0] = 1.0
        binary[-1] = 1.0
        # adjust count
        cur = int(binary.sum())
        if cur > M:
            # turn some ones to zero
            ones = np.where(binary == 1)[0]
            # exclude edges
            ones = ones[(ones != 0) & (ones != N-1)]
            to_flip = rng.choice(ones, size=cur - M, replace=False)
            binary[to_flip] = 0.0
        elif cur < M:
            zeros = np.where(binary == 0)[0]
            zeros = zeros[(zeros != 0) & (zeros != N-1)]
            to_flip = rng.choice(zeros, size=M - cur, replace=False)
            binary[to_flip] = 1.0
        h = binary.astype(float)

    best_h = h.copy()
    best_c5 = compute_c5(h)

    start = time.time()
    # maintain list of indices of ones and zeros (excluding edges)
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_set = set(np.where(h == 0)[0].tolist())
    # keep edges always in ones_set
    ones_set.discard(0)
    ones_set.discard(N-1)
    # zeros_set also remove edges
    zeros_set.discard(0)
    zeros_set.discard(N-1)

    while time.time() - start < time_limit:
        if not ones_set or not zeros_set:
            break
        i = rng.choice(list(ones_set))
        j = rng.choice(list(zeros_set))
        # swap
        h[i], h[j] = h[j], h[i]
        # evaluate
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            # accept swap
            ones_set.remove(i)
            ones_set.add(j)
            zeros_set.remove(j)
            zeros_set.add(i)
        else:
            # revert
            h[i], h[j] = h[j], h[i]
        # occasionally random restart if stagnates?
        # Not needed for simple.
    return best_h, best_c5

def gradient_from_weights(h, w):
    N = len(h)
    grad = np.ones(N)
    t_vals = np.arange(-N+1, N)
    for weight, t in zip(w, t_vals):
        if weight == 0.0:
            continue
        if t >= 0:
            # i from 0 to N - t - 1
            if N - t > 0:
                grad[0:N - t] -= weight * h[t:N]
                grad[t:N] -= weight * h[0:N - t]
        else:
            k = -t
            if N - k > 0:
                grad[k:N] -= weight * h[0:N - k]
                grad[0:N - k] -= weight * h[k:N]
    return grad

def continuous_optimize(h0, time_budget):
    N = len(h0)
    target = N / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    beta = 1.0
    step = 0.05
    while time.time() - start < time_budget:
        # compute O array
        N = h.shape[0]
        dx = 2.0 / N
        auto = compute_autocorr(h)
        O = 1 - auto * dx
        # softmax weights
        max_O = O.max()
        exp_vals = np.exp(beta * (O - max_O))
        w = exp_vals / exp_vals.sum()
        # gradient
        grad = gradient_from_weights(h, w)
        # step
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_onto_capped_simplex(h_new, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        # accept if improvement else revert
        if c5_new <= compute_c5(h):
            h = h_new
        # adjust parameters
        beta = min(200.0, beta * 1.02)
        step = max(1e-4, step * 0.999)
    return best_h, best_c5

import time
import numpy as np

# ---------- utilities ----------

def _autocorr_full(h: np.ndarray) -> np.ndarray:
    """Linear autocorrelation of a real vector (full mode)."""
    n = h.shape[0]
    m = 2 * n - 1
    H = np.fft.rfft(h, n=m)
    auto = np.fft.irfft(H * np.conj(H), n=m)
    return auto  # length 2n‑1, lag –(n‑1)…+(n‑1)


def _c5_value(h: np.ndarray) -> float:
    """C₅ = max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    auto = _autocorr_full(h)
    O = 1.0 - auto * dx
    return float(np.max(O))


def _project_capped_simplex(v: np.ndarray, target: float, iters: int = 80) -> np.ndarray:
    """
    Projection onto {0≤x≤1, sum(x)=target}.
    Solves x = clip(v‑λ,0,1) via bisection on λ.
    """
    lo, hi = -2.0, 2.0
    for _ in range(iters):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < 1e-12:
            return x
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


# ---------- binary hill‑climbing ----------

def _hill_climb_binary(N: int, timelimit: float, rng: np.random.Generator) -> tuple:
    """Binary (0/1) search for a low C₅ value."""
    M = N // 2                       # required number of ones
    # initialise – edges forced to 1 (helps large shifts)
    h = np.zeros(N, dtype=float)
    h[0] = h[-1] = 1.0
    remaining = max(0, M - 2)
    idx = rng.choice(np.arange(1, N - 1), size=remaining, replace=False)
    h[idx] = 1.0

    best_h, best_c5 = h.copy(), _c5_value(h)

    # keep mutable index sets without the fixed edges
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    ones.discard(0); ones.discard(N - 1)
    zeros.discard(0); zeros.discard(N - 1)

    start = time.time()
    while time.time() - start < timelimit:
        if not ones or not zeros:
            break
        i = rng.choice(list(ones))
        j = rng.choice(list(zeros))
        h[i], h[j] = h[j], h[i]                     # proposed swap
        c5 = _c5_value(h)
        if c5 < best_c5 - 1e-12:
            best_c5, best_h = c5, h.copy()
            # accept swap permanently
            ones.remove(i); ones.add(j)
            zeros.remove(j); zeros.add(i)
        else:
            h[i], h[j] = h[j], h[i]                 # revert
    return best_h, best_c5


# ---------- continuous optimisation (soft‑max surrogate) ----------

def _grad_from_weights(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """∇ (soft‑max approximation of max O) for a given weight vector w."""
    N = h.shape[0]
    grad = np.ones(N)                # Σ_t w_t * 1   (since Σ w_t = 1)
    t_vals = np.arange(-N + 1, N)    # order matches autocorrelation output
    for wt, t in zip(w, t_vals):
        if wt == 0.0:
            continue
        if t >= 0:
            if N - t > 0:
                grad[:N - t] -= wt * h[t:]          # –h_{i+t}
                grad[t:]     -= wt * h[:N - t]      # –h_{i‑t}
        else:
            k = -t
            if N - k > 0:
                grad[k:]      -= wt * h[:N - k]     # –h_{i+t} (t negative)
                grad[:N - k]  -= wt * h[k:]         # –h_{i‑t}
    return grad


def _continuous_optimize(h0: np.ndarray, timelimit: float) -> tuple:
    """Projected gradient descent on the smooth max‑O surrogate."""
    N = h0.shape[0]
    target = N / 2.0
    h = h0.copy()
    best_h, best_c5 = h.copy(), _c5_value(h)

    start = time.time()
    beta = 1.0          # soft‑max temperature (grows → sharper)
    step = 0.05         # learning rate
    while time.time() - start < timelimit:
        dx = 2.0 / N
        auto = _autocorr_full(h)
        O = 1.0 - auto * dx               # vector of O(t) values
        # soft‑max weights
        mO = O.max()
        exp_vals = np.exp(beta * (O - mO))
        w = exp_vals / exp_vals.sum()

        # gradient and projection
        grad = _grad_from_weights(h, w)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_capped_simplex(h_new, target)

        c5_new = _c5_value(h_new)
        if c5_new < best_c5:
            best_c5, best_h = c5_new, h_new.copy()
        # simple acceptance rule
        if c5_new <= _c5_value(h):
            h = h_new
        # schedule
        beta = min(200.0, beta * 1.02)
        step = max(1e-4, step * 0.999)
    return best_h, best_c5
