```python
def project_onto_capped_simplex(y, S):
    # y:numpy array
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        clipped = np.clip(y - mid, 0, 1)
        total = clipped.sum()
        if total > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2.0, 0, 1)
```

```python
def project_to_capped_simplex(y, S):
    lo = np.min(y) - 1.0
    hi = np.max(y)
```

def project_to_capped_simplex(y, S, max_iter=50):
    lo = np.min(y) - 1.0
    hi = np.max(y)

import numpy as np, time, math

def project_to_capped_simplex(y, S, max_iter=50):
    """Euclidean projection onto {0<=x_i<=1, sum x_i = S}."""
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        clipped = np.clip(y - mid, 0.0, 1.0)
        total = clipped.sum()
        if total > S:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def autocorr_full(x):
    """Linear (zero‑padded) autocorrelation of x with itself."""
    n = x.shape[0]
    size = 1 << ((2 * n - 1) - 1).bit_length()      # next power of two
    fx = np.fft.rfft(x, size)
    fr = np.fft.rfft(x[::-1], size)
    corr = np.fft.irfft(fx * fr, size)
    return corr[: 2 * n - 1]


def evaluate_c5(h):
    """Exact C5 value for a feasible h."""
    n = h.shape[0]
    dx = 2.0 / n
    S = h.sum()
    corr = autocorr_full(h)
    return (S - corr.min()) * dx


def optimize_h(initial_h, max_time):
    """Projected gradient ascent on the smooth‐min of the autocorrelation."""
    n = initial_h.shape[0]
    S = n / 2.0
    h = project_to_capped_simplex(initial_h.copy(), S)

    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    start = time.time()
    # schedule parameters
    beta_start, beta_end = 0.5, 25.0
    step_start, step_end = 0.25, 0.001
    target_iters = 8000

    for it in range(target_iters):
        if time.time() - start > max_time:
            break

        # --- gradient based on a smoothed min of autocorrelation ---
        corr = autocorr_full(h)
        beta = beta_start + (beta_end - beta_start) * (it / target_iters)
        cmin = corr.min()
        w = np.exp(-beta * (corr - cmin))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # gradient = 2 * (w * h) convolution (linear, zero‑padded)
        conv_full = np.convolve(w, h, mode='full')
        grad = 2.0 * conv_full[n - 1 : n - 1 + n]

        # step size linearly decreasing
        step = step_start + (step_end - step_start) * (it / target_iters)

        # ascent step and projection back onto feasibility set
        h_new = project_to_capped_simplex(h + step * grad, S)

        # evaluate new C5
        c5 = evaluate_c5(h_new)

        # keep global best
        if c5 < best_c5 - 1e-9:
            best_c5 = c5
            best_h = h_new.copy()

        # simple simulated‑annealing acceptance to escape shallow local minima
        elapsed = time.time() - start
        T = max(0.0, (max_time - elapsed) / max_time)   # linear temperature
        if c5 <= best_c5 + 1e-6 or np.random.rand() < math.exp(-(c5 - best_c5) / max(T, 1e-6)):
            h = h_new

    return best_h, best_c5
