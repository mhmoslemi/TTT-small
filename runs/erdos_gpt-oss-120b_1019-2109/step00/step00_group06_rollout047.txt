def project_to_feasible(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Compute total
    total = h.sum()
    # Want total = target_sum
    # Use simple iterative fix: if total < target_sum, add to entries <1.
    # if total > target_sum, subtract from entries >0.
    # We'll compute deficiency = target_sum - total.
    # distribute among non-bound entries uniformly.
    # We'll loop for up to maybe 10 passes.
    for _ in range(10):
        if abs(total - target_sum) < 1e-12:
            break
        if total < target_sum:
            # need to increase
            free = np.where(h < 1.0)[0]
            if len(free) == 0:
                break
            incr = min(target_sum - total, (1.0 - h[free]).sum())
            # distribute equally
            add = incr / len(free)
            h[free] += add
        else: # total > target_sum
            free = np.where(h > 0.0)[0]
            if len(free) == 0:
                break
            decr = min(total - target_sum, h[free].sum())
            sub = decr / len(free)
            h[free] -= sub
        total = h.sum()
    # Final clip again
    h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time
import math

def _compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)*(1-h(x+k))dx  (discrete version)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_feasible(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the exact linear sum."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-12:
            break
        if diff > 0:

            dec = min(-diff, h[free].sum())

            h[free] -= sub
