            hi = mid

import time
import numpy as np

def _project_boxed_simplex(v, target):
    """
    Euclidean projection of v onto the set {x | 0<=x_i<=1, sum x_i = target }.
    Uses a simple binary search on the offset λ such that x_i = clip(v_i+λ,0,1)
    satisfies the equality.
    """
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):                     # enough for double precision
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + mid, 0.0, 1.0)

def _corr_max(h):
    """return max correlation value (not scaled by dx) and the shift index."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    return corr[idx], idx

def _gradient_shift(h, shift):
    """
    Gradient of the correlation for a given integer shift (as in np.correlate)
    with respect to h.  The gradient respects the zero‑padding used by
    np.correlate.
    """
    N = h.size
    g = np.zeros(N, dtype=float)
    if shift >= 0:
        # overlapping region i = 0 .. N-1-shift
        end = N - shift
        # + (1 - h[i+shift])
        g[:end] += 1.0 - h[shift:]
        # - h[i] contributes to position i+shift
        g[shift:] -= h[:end]
    else:
        s = -shift
        # overlapping region i = s .. N-1
        start = s
        # + (1 - h[i-s])
        g[start:] += 1.0 - h[:N - s]
        # - h[i] contributes to position i-s
        g[:N - s] -= h[start:]
    return g

def _evaluate(h):
    """scaled max correlation (the C5 value)."""
    N = h.size
    mx, _ = _corr_max(h)
    return mx * (2.0 / N)

def _random_feasible(N, rng):
    """draw a random point in the bounded simplex."""
    v = rng.random(N)
    return _project_boxed_simplex(v, N / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise --------------------------------------------------------
    # ------------------------------------------------------------------
    try:
        h = np.asarray(initial_h_values, dtype=float)  # provided by the harness
    except NameError:
        # fall back: 60 points, uniform start
        N = 60
        h = np.full(N, 0.5, dtype=float)
    else:
        N = h.size

    # make sure the sum constraint holds (tiny numeric errors may appear)
    h = _project_boxed_simplex(h, N / 2.0)

    best_h = h.copy()
    best_val = _evaluate(best_h)

    # parameters --------------------------------------------------------
    step0 = 0.6            # initial step magnitude
    no_improve = 0        # counter for stagnation
    ITER_LIMIT = 200000   # safety bound

    it = 0
    while it < ITER_LIMIT and (time.time() - start_time) < budget_s * 0.95:
        # ------------------------------------------------------------------
        # current worst shift ------------------------------------------------
        # ------------------------------------------------------------------
        max_corr_raw, idx = _corr_max(h)
        max_corr = max_corr_raw * (2.0 / N)
        shift = idx - (N - 1)           # integer shift used by np.correlate
        # gradient of that worst term
        g = _gradient_shift(h, shift)
        # make it sum‑zero (preserve the integral when we move)
        d = -(g - g.mean())

        # ------------------------------------------------------------------
        # line search -------------------------------------------------------
        # ------------------------------------------------------------------
        step = step0 / np.sqrt(it + 1.0)
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.1):
            h_candidate = _project_boxed_simplex(h + step * factor * d, N / 2.0)
            cand_val = _evaluate(h_candidate)
            if cand_val + 1e-12 < max_corr:          # strict improvement
                h = h_candidate
                max_corr = cand_val
                improved = True
                break

        # ------------------------------------------------------------------
        # fallback random tweak ---------------------------------------------
        # ------------------------------------------------------------------
        if not improved:
            # two‑index exchange keeping the sum exact
            i, j = rng.choice(N, size=2, replace=False)
            low = max(-h[i], h[j] - 1.0)
            high = min(1.0 - h[i], h[j])
            if low < high:
                delta = low + (high - low) * rng.rand()
                h2 = h.copy()
                h2[i] += delta
                h2[j] -= delta
                cand_val = _evaluate(h2)
                if cand_val + 1e-12 < max_corr:
                    h = h2
                    max_corr = cand_val
                    improved = True

        # ------------------------------------------------------------------
        # bookkeeping --------------------------------------------------------
        # ------------------------------------------------------------------
        if max_corr < best_val - 1e-15:
            best_val = max_corr
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # occasional restart if stuck for a while
        if no_improve > 3000:
            h = _random_feasible(N, rng)
            no_improve = 0

        it += 1

    # final evaluation (scaled)
    final_c5 = _evaluate(best_h)

    # ensure constraints are satisfied to the required tolerance
    assert abs(best_h.sum() - N / 2.0) < 1e-9
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)

    return best_h.tolist(), float(final_c5), int(N)
