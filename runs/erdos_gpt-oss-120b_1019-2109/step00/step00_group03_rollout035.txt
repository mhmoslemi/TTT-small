def grad_smooth_raw(h, w):
    N = h.size
    grad = np.ones(N)  # base from sum_i w_i * 1 = 1
    # subtract contributions
    # iterate over shift d with weight w_d
    for shift_idx, weight in enumerate(w):
        d = shift_idx - (N - 1)
        if d == 0:
            grad -= 2 * weight * h
        elif d > 0:
            grad[:N-d] -= weight * h[d:]
            grad[d:]   -= weight * h[:N-d]
        else:  # d < 0
            dd = -d
            grad[dd:]  -= weight * h[:N-dd]
            grad[:N-dd] -= weight * h[dd:]
    return grad
```

def gradient_phase(h, epsilon, lr, max_iter, target_sum, remaining_time):
    start_phase = time.time()
    for it in range(max_iter):
        if time.time() - start_phase > remaining_time:
            break
        raw = np.correlate(h, 1 - h, mode='full')
        # compute max raw for record as we may also track.
        # compute weights
        scaled = raw / epsilon
        max_s = np.max(scaled)

def directed_search(h, target_sum, dx, best_val, best_h, time_limit):
    while time.time() < time_limit:
        # compute product correlation
        prod = np.correlate(h, h, mode='full')
        center = n-1
        min_shift_idx = np.argmin(prod[center+1:]) + 1 # shift >=1
        d = min_shift_idx
        # compute neighbor sums
        left = np.concatenate((np.zeros(d), h[:-d]))
        right = np.concatenate((h[d:], np.zeros(d)))
        nsum = left + right
        # choose src and tgt
        src_candidates = np.where(h > 1e-12)[0]
        if src_candidates.size == 0:
            break
        src = src_candidates[np.argmin(nsum[src_candidates])]
        tgt_candidates = np.where(h < 1 - 1e-12)[0]
        if tgt_candidates.size == 0:
            break
        tgt = tgt_candidates[np.argmax(nsum[tgt_candidates])]
        max_delta = min(0.05, h[src], 1 - h[tgt])  # step size small
        if max_delta < 1e-9:
            # random fallback
            i,j = rng.choice(n,2,replace=False)
            max_delta = min(0.05, h[i], 1 - h[j])
            src, tgt = i,j
        delta = rng.random() * max_delta
        h_new = h.copy()
        h_new[src] -= delta
        h_new[tgt] += delta
        val = compute_c5(h_new, dx)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        # accept move only if improves else revert
        if val <= compute_c5(h, dx) + 1e-12:
            h = h_new
        else:
            # optionally accept with small probability
            if rng.random() < 0.001:
                h = h_new
        # break if time near end
    return best_h, best_val
```

def run(seed=42, budget_s=1100):

    rng = np.random.default_rng(seed)
    start = time.time()
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 140
        i = np.arange(n)
        h = 0.5 + 0.5 * np.cos(np.pi * i / (n/2))
    target_sum = n/2
    dx = 2.0 / n

    h = np.clip(h, 0.0, 1.0)
    # enforce sum exactly
    h = project_to_sum(h, target_sum)

    best_h = h.copy()
    best_val = compute_max_overlap(h, dx)

    # parameters
    step = 0.1
    temperature = 0.01

    # main loop
    iter = 0
    while time.time() - start < budget_s * 0.95:
        # compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        d = max_idx - (n - 1)
        # compute gradient for shift d
        grad = np.zeros(n)
        for i in range(n):
            j = i + d
            if 0 <= j < n:
                grad[i] += 1 - h[j]
                grad[j] -= h[i]
        # source: positive grad, can decrease
        src_candidates = np.where((h > 1e-12) & (grad > 0))[0]
        tgt_candidates = np.where((h < 1 - 1e-12) & (grad < 0))[0]
        if src_candidates.size > 0 and tgt_candidates.size > 0:
            src = src_candidates[np.argmax(grad[src_candidates])]
            tgt = tgt_candidates[np.argmin(grad[tgt_candidates])]
            max_delta = min(step, h[src], 1 - h[tgt])
            delta = rng.random() * max_delta
            h_new = h.copy()
            h_new[src] -= delta
            h_new[tgt] += delta
        else:
            # random move
            i, j = rng.choice(n, size=2, replace=False)
            max_delta = min(step, h[i], 1 - h[j])
            if max_delta <= 1e-12:
                continue
            delta = rng.random() * max_delta
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        # evaluate

        accept = False

            best_val = new_val

    final_val = compute_max_overlap(best_h, dx)

import time, math
import numpy as np

def _project_to_sum(x: np.ndarray, target: float) -> np.ndarray:
    """Project x onto the box [0,1] while forcing sum(x)=target."""
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)


def _max_overlap(h: np.ndarray, dx: float) -> float:
    """C5 value for a given step function h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _worst_shift(h: np.ndarray):
    """Return (shift, correlation array) of the worst (maximum) overlap."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (h.size - 1)          # integer shift producing the max
    return shift, corr


def _grad_worst_shift(h: np.ndarray, d: int) -> np.ndarray:
    """Gradient of the overlap corresponding to shift d."""
    n = h.size
    g = np.zeros(n, dtype=float)
    # contribution from each i where i+d is valid
    for i in range(n):
        j = i + d
        if 0 <= j < n:
            g[i] += 1.0 - h[j]   # derivative w.r.t. h_i
            g[j] -= h[i]        # derivative w.r.t. h_j
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- initialise ---------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 140                     # a moderate size – still fast to evaluate
        i = np.arange(n, dtype=float)
        # cosine^2 shape, automatically satisfies the integral constraint
        h = 0.5 + 0.5 * np.cos(np.pi * i / (n / 2))
    target_sum = n / 2.0
    h = _project_to_sum(h, target_sum)

    dx = 2.0 / n
    best_h = h.copy()
    best_val = _max_overlap(best_h, dx)

    # ---- optimisation parameters -------------------------------------------
    step = 0.12                     # maximal mass moved per elementary step
    temperature = 0.02             # annealing temperature
    it = 0

    # ---- main loop ---------------------------------------------------------
    while time.time() - start < budget_s * 0.94:
        cur_val = _max_overlap(h, dx)
        d, _ = _worst_shift(h)          # worst shift
        grad = _grad_worst_shift(h, d)

        # pick a source (positive gradient, can be decreased) and a target (negative)
        src_mask = (h > 1e-12) & (grad > 0)
        tgt_mask = (h < 1.0 - 1e-12) & (grad < 0)

        if src_mask.any() and tgt_mask.any():
            src = int(np.argmax(grad * src_mask))
            tgt = int(np.argmin(grad * tgt_mask))
            max_delta = min(step, h[src], 1.0 - h[tgt])
            if max_delta <= 1e-12:
                src = tgt = None
        else:
            src = tgt = None

        if src is None:
            # random feasible pair
            i1, i2 = rng.choice(n, 2, replace=False)
            max_delta = min(step, h[i1], 1.0 - h[i2])
            if max_delta <= 1e-12:
                continue
            src, tgt = i1, i2

        delta = rng.random() * max_delta
        h_new = h.copy()
        h_new[src] -= delta
        h_new[tgt] += delta

        new_val = _max_overlap(h_new, dx)

        # acceptance test (deterministic improvement or annealed)
        accept = False
        if new_val <= cur_val + 1e-12:
            accept = True
        else:
            if temperature > 0:
                prob = math.exp(-(new_val - cur_val) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            h = h_new

        # record best ever
        if new_val < best_val - 1e-12:
            best_val = new_val
            best_h = h_new.copy()

        # schedule updates
        it += 1
        if it % 600 == 0:
            temperature = max(temperature * 0.90, 1e-5)
            step = max(step * 0.92, 0.001)

    # final tidy‑up
    best_h = _project_to_sum(best_h, target_sum)
    final_val = _max_overlap(best_h, dx)

    return best_h.tolist(), float(final_val), int(n)
