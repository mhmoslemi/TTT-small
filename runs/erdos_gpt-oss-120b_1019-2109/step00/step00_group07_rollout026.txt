def simulated_annealing(h, time_limit):
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    best = h.copy(); best_val = compute_c5(h)
    cur = h.copy(); cur_val = best_val
    # set algorithm parameters
    max_iters = 200_000
    it = 0
    while time.time() - start < time_limit and it < max_iters:
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        # Randomly determine direction.
        if np.random.rand() < 0.5:
            max_transfer = min(cur[i], 1 - cur[j])
            if max_transfer <= 0: continue
            delta = np.random.rand() * max_transfer
            new_val_candidate = cur.copy()
            new_val_candidate[i] -= delta
            new_val_candidate[j] += delta
        else:
            max_transfer = min(cur[j], 1 - cur[i])
            if max_transfer <= 0: continue
            delta = np.random.rand() * max_transfer
            new_val_candidate = cur.copy()
            new_val_candidate[i] += delta
            new_val_candidate[j] -= delta
        new_val = compute_c5(new_val_candidate)
        # Simulated annealing acceptance
        if new_val < cur_val:
            cur = new_val_candidate
            cur_val = new_val
            if new_val < best_val:
                best = new_val_candidate.copy()
                best_val = new_val
        else:
            # temperature schedule: linear decay
            T = max(0.0, 0.01 * (1 - it / max_iters))
            if T > 0 and np.random.rand() < np.exp(-(new_val - cur_val) / T):
                cur = new_val_candidate
                cur_val = new_val
        it += 1
    return best, best_val

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def fix_sum(vec, target):
    diff = target - vec.sum()
    # adjust iteratively
    # Use while loop until diff small
    # but ensure not infinite loops due to rounding
    max_iter = 10
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            slack = 1 - vec
            idx = np.argmax(slack)
            if slack[idx] <= 1e-12:
                break
            inc = min(diff, slack[idx])
            vec[idx] += inc
            diff -= inc
        else:
            slack = vec
            idx = np.argmax(slack)
            if slack[idx] <= 1e-12:
                break
            dec = min(-diff, slack[idx])
            vec[idx] -= dec
            diff += dec
    # final slight correction if still diff remains due to rounding
    if abs(diff) > 1e-12:
        # distribute evenly across all with slack
        if diff > 0:
            # add diff/n to all (still may exceed bounds but minimal)
            vec += diff / len(vec)
        else:
            vec += diff / len(vec)
        # clip to [0,1] maybe adjust
        vec = np.clip(vec, 0, 1)
        # adjust sum again recursively once
        # but ignore small errors
    return vec

def optimize_h(initial_h, time_budget):
    n = len(initial_h)
    target_sum = n / 2.0
    h = np.array(initial_h, dtype=np.float64)
    # ensure sum exactly target via fix

    best_h = h.copy()
    best_val = compute_c5(h)
    cur_val = best_val
    start = time.time()
    # schedule
    max_iters = 500_000
    for it in range(max_iters):
        if time.time() - start > time_budget:
            break
        # compute offset for max overlap
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        offset = idx_max - (n - 1)
        # gradient
        if offset != 0:
            d = abs(offset)
            grad = np.zeros_like(h)
            # use slicing

            grad[d:] += h[:n-d]

                best_val = new_val
        else:

            i = np.random.randint(0, n)

            max_up = 1 - h[i]

            if max_up > 0 and max_down > 0:

                if np.random.rand() < 0.5:

                    j = np.random.randint(0, n-1)

                    h[j] += max_add
                    # adjust diff as above
        # optionally break if best_val <= target bound 0.3808
        if best_val <= 0.3808:
            break
    return best_h, best_val

import time
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """C5 = max_k ∫ h(x)*(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation, then scale
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _fix_sum(vec: np.ndarray, target: float) -> np.ndarray:
    """Adjust vec (in‑place) so that sum(vec) == target while staying in [0,1]."""
    diff = target - vec.sum()
    # simple greedy redistribution, iterate few times at most
    for _ in range(50):
        if abs(diff) < 1e-14:
            break
        if diff > 0.0:
            slack = 1.0 - vec
            idx = int(np.argmax(slack))
            inc = min(diff, slack[idx])
            if inc <= 0.0:
                break
            vec[idx] += inc
            diff -= inc
        else:
            slack = vec
            idx = int(np.argmax(slack))
            dec = min(-diff, slack[idx])
            if dec <= 0.0:
                break
            vec[idx] -= dec
            diff += dec
    # tiny residual adjustment (should be negligible)
    if abs(diff) > 1e-12:
        vec += diff / vec.size
        np.clip(vec, 0.0, 1.0, out=vec)
    return vec


def _gradient_step(h: np.ndarray, offset: int, lr: float) -> np.ndarray:
    """Return a gradient step that increases the overlap for the given offset."""
    n = h.shape[0]
    if offset == 0:
        # push extremes to increase sum of squares
        grad = 2.0 * (h - 0.5)
    else:
        d = abs(offset)
        grad = np.zeros_like(h)
        # contribution from the forward direction
        grad[: n - d] += h[d:]
        # contribution from the backward direction
        grad[d:] += h[: n - d]
    step = lr * grad
    # limit the absolute change per coordinate
    step = np.clip(step, -0.2, 0.2)
    return step


def _optimize_once(init_h: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """Local optimisation of a single start vector."""
    n = init_h.shape[0]
    target_sum = n / 2.0
    h = init_h.copy()
    h = _fix_sum(h, target_sum)

    best_h = h.copy()
    best_val = compute_c5(h)

    cur_val = best_val
    start = time.time()
    it = 0
    max_iters = 600_000

    while it < max_iters and (time.time() - start) < time_budget:
        # find worst shift (max C5)
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr))
        offset = idx_max - (n - 1)

        # learning rate decays slowly
        lr = 0.25 * (0.9995 ** (it // 1000))

        step = _gradient_step(h, offset, lr)
        new_h = h + step
        np.clip(new_h, 0.0, 1.0, out=new_h)
        new_h = _fix_sum(new_h, target_sum)

        new_val = compute_c5(new_h)

        # simulated‑annealing style acceptance
        if new_val < cur_val:
            h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
        else:
            # temperature schedule
            T = max(0.0005, 0.08 * (1.0 - it / max_iters))
            if np.random.rand() < np.exp(-(new_val - cur_val) / T):
                h, cur_val = new_h, new_val

        # occasional random mass‑transfer to diversify search
        if (it & 511) == 0:
            i, j = np.random.randint(0, n, size=2)
            if i != j:

                    max_tr = min(h[i], 1.0 - h[j])
                else:
                    max_tr = min(h[j], 1.0 - h[i])
                if max_tr > 0.0:
                    delta = np.random.rand() * max_tr
                    if np.random.rand() < 0.5:

                    else:
                        h[i] += delta
                        h[j] -= delta
                    np.clip(h, 0.0, 1.0, out=h)
                    h = _fix_sum(h, target_sum)

        # early exit if we already beat the target bound
        if best_val <= 0.3808:
            break

        it += 1

    return best_h, best_val


def _upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a finer grid (preserving the integral)."""
    old_n = h.shape[0]
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, h, period=2.0)
    # repair the sum (tiny interpolation error)
    new_h = _fix_sum(new_h, new_n / 2.0)
    return new_h

    stage1_budget = total_budget * 0.40
