def compute_c5(h):
    n = h.size
    g = 1.0 - h
    # Use FFT for correlation h * g
    L = 1 << ((2*n-1).bit_length())

def compute_c5(h):
    n = h.shape[0]
    g = 1.0 - h
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g, n=L)
    corr = np.fft.irfft(H * np.conj(G), n=L)[:2*n - 1]
    max_corr = np.max(corr)
    return max_corr * (2.0 / n)
