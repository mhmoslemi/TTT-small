import time
import numpy as np


def _c5_value(h: np.ndarray) -> float:
    """Return C5 = max correlation * dx for a given step function."""
    n = h.shape[0]
    max_corr = np.max(np.correlate(h, 1.0 - h, mode="full"))
    return max_corr * (2.0 / n)


def _project_to_half(h: np.ndarray) -> np.ndarray:
    """Project vector onto the feasible set:
       0 ≤ h ≤ 1 and sum(h) = n/2.
    """
    n = h.shape[0]
    target = n / 2.0
    # clip to box
    np.clip(h, 0.0, 1.0, out=h)
    eps = 1e-12
    # water‑filling style projection onto the capped simplex
    while True:
        diff = target - h.sum()
        if abs(diff) <= eps:
            break
        free = (h > eps) & (h < 1.0 - eps)
        nf = free.sum()
        if nf == 0:
            # no interior point – adjust any coordinate that can move
            idx = np.where(h < 1.0 - eps)[0]
            if idx.size == 0:
                idx = np.array([0])
            i = idx[0]
            h[i] = np.clip(h[i] + diff, 0.0, 1.0)
        else:
            delta = diff / nf
            h[free] += delta
            np.clip(h, 0.0, 1.0, out=h)
    return h


def _max_corr_shift(h: np.ndarray):
    """Return (max_corr, shift, full_corr_array)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return corr[idx], idx - (n - 1), corr

    """Search for a step function with a small C5 value.

    Returns

        * n_points – length of the discretisation.
    """

    rng = np.random.default_rng(seed)

    # initialise a feasible h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
        h = _project_to_half(h)
    else:
        n = 200                     # modest size, fast to evaluate
        h = np.full(n, 0.5, dtype=np.float64)   # sum = n/2 already

    best_h = h.copy()
    best_val = _c5_value(best_h)

    # ------------------------------------------------------------------
    # parameters for the search
    # ------------------------------------------------------------------
    total_budget = budget_s * 0.95          # safety margin
    no_improve = 0

        # store best solution

            best_h = h.copy()

        # choose a move: gradient step (probability 0.75) or random swap
        # ------------------------------------------------------------------
        if rng.random() < 0.75:

            shift = k
            if shift >= 0:

                s_plus[:shift] = 0.0
