def compute_cost(h):
    n = len(h)
    rev = (1 - h)[::-1]
    L = 1 << (2 * n - 1).bit_length() # next power of 2 >= 2n-1
    # for speed: use np.fft.rfft
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)[:2*n-1]
    # max correlation:
    max_val = np.max(conv)
    dx = 2.0 / n
    return max_val * dx

def neighbor_pairwise(h, step_scale=0.1):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # compute max delta that we can add to i (decrease j)
    max_inc = min(step_scale, 1 - h[i])
    max_dec = min(step_scale, h[j])
    # delta positive: increase i, decrease j
    max_delta = min(max_inc, max_dec)
    # also negative direction: decrease i, increase j
    max_dec_i = min(step_scale, h[i])
    max_inc_j = min(step_scale, 1 - h[j])
    min_delta = -min(max_dec_i, max_inc_j) # negative bound

    delta = np.random.uniform(min_delta, max_delta)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def adjust_sum_and_bounds(h):

    h = np.clip(h, 0, 1)
    n = len(h)
    target = n / 2.0
    # Perform Euclidean projection onto the hyperplane sum = target while staying in [0,1].
    # This can be done with algorithm while loops.
    # We'll use simple iterative water-filling as defined earlier.
    # For speed we can call a function that solves this exactly via algorithm described by Duchi 2008.
    # Actually projection onto simplex with upper bound 1: we can treat as projection onto intersection of simplex sum=target and box constraints.
    # Equivalent to solving min ||x - h0||^2 s.t. sum(x)=target, 0 <= x <= 1.
    # This can be solved via a variant of sorting algorithm (like projecting onto capped simplex).
    # But we can implement naive method: Use lambda to shift h then clip.
    # We'll find lambda such that sum(clip(h0 - lambda, 0, 1)) = target.
    # Binary search for lambda.
    # Since sum of clipped is decreasing function of lambda.
    def projection_capped_simplex(v, target):
        # find lambda in [min(v-1) - something] range.
        lo = -1.0
        hi = 1.0
        # Expand bounds.
        # The function f(lambda) = sum(clip(v - lambda,0,1)) - target.
        # It is decreasing in lambda (increase lambda reduces each term).
        for _ in range(50):
            mid = (lo + hi) / 2
            proj = np.clip(v - mid, 0, 1)
            s = proj.sum()
            if s > target:
                lo = mid
            else:
                hi = mid
        return np.clip(v - (lo+hi)/2, 0, 1)
    adjusted = projection_capped_simplex(h, target)
    return adjusted

import numpy as np
import time
import math

def fast_cost(h: np.ndarray) -> float:
    """Fast evaluation of max_{k} ∫ h(x)(1−h(x+k))dx using FFT."""
    n = h.shape[0]
    rev = (1.0 - h)[::-1]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2n‑1
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)[: 2 * n - 1]
    return np.max(conv) * (2.0 / n)


def exact_cost(h: np.ndarray) -> float:
    """Exact cost using np.correlate (the reference evaluator)."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def pairwise_transfer(h: np.ndarray, step: float, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random amount ≤ step from one entry to another, preserving the sum."""
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)

    if rng.random() < 0.5:
        # move from j → i
        max_delta = min(step, 1.0 - h[i], h[j])
        if max_delta <= 0.0:
            return h
        delta = rng.random() * max_delta
    else:
        # move from i → j
        max_delta = min(step, h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            return h
        delta = -rng.random() * max_delta

    nh = h.copy()
    nh[i] += delta
    nh[j] -= delta
    return nh


def _adjust_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Project onto the hyperplane sum(h)=target while staying in [0,1]."""
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = 1.0 - h
            total = free.sum()
            if total == 0:
                break
            inc = diff / total
            h += free * inc
        else:
            free = h
            total = free.sum()
            if total == 0:
                break
            dec = -diff / total
            h -= free * dec
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
    return h

    # ----------------------------------------------------------------------

    else:

        idx = rng.choice(n, size=n // 2, replace=False)
        h0[idx] = 1.0

    # Ensure the integral constraint exactly (numeric safety)
    target_sum = n / 2.0
    h0 = _adjust_sum(h0, target_sum)

    # ----------------------------------------------------------------------
    # 2. Simulated annealing
    # ----------------------------------------------------------------------
    cur_h = h0.copy()
    cur_cost = fast_cost(cur_h)
    best_h = cur_h.copy()
    best_cost = cur_cost

    start_time = time.time()
    limit_time = start_time + budget_s * 0.97   # leave a tiny margin

    T = 0.01                # temperature
    step = 0.2              # maximal transferred mass per move
