def project_box_sum(x, target_sum):
    # We need to find lambda s.t. clamp(x - lambda, 0, 1) sums to target.
    # Use bisection.
    lo = np.min(x) - 1  # maybe -1

def propose_move(h, worst_shift, step):
    n = len(h)
    s = worst_shift - (n - 1)  # shift in terms of index difference? Wait worst_shift index in full correlation array; conv index d = worst_shift - (n-1)
    # compute shift offset
    offset = s  # can be negative as well
    # for simplicity handle positive offset only (if negative, we can swap indices)
    if offset >= 0:
        start = np.random.randint(0, n - offset)
        i = start
        j = i + offset
    else:
        # negative shift: use i offset positions
        start = np.random.randint(-offset, n)  # ensure i+offset >=0
        i = start
        j = i + offset
    # compute delta allowed to increase both
    increase_max = min(step, 1 - h[i], 1 - h[j])
    if increase_max <= 0:
        return None
    # choose reduction indices distinct from i,j
    # find two indices p,q where h > 0
    candidates = np.where(h > 0)[0]
    candidates = candidates[(candidates != i) & (candidates != j)]
    if len(candidates) < 2:
        return None
    p, q = np.random.choice(candidates, size=2, replace=False)
    # delta reduction per index (maybe same) to keep sum: since we increase two values by inc, sum increases by 2*inc, need to decrease sum by 2*inc, we can reduce p and q each by inc (if they have >= inc)
    dec_max = min(increase_max, h[p], h[q])
    delta = dec_max
    if delta <= 0:
        return None
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] += delta
    new_h[p] -= delta
    new_h[q] -= delta
    # ensure bounds
    new_h = np.clip(new_h, 0.0, 1.0)
    return new_h

def _integral_step(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation with zero padding
    vals = np.correlate(h, 1.0 - h, mode="full") * dx
    return vals.max()


def _project_sum_box(v, target):
    """
    Project vector v onto the intersection of the hyperplane sum(v)=target
    and the box 0<=v_i<=1 using a simple water‑filling bisection.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _worst_shift_info(h):
    """Return (index of worst shift, offset in samples)."""
    n = h.shape[0]
    dx = 2.0 / n
    M = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = int(np.argmax(M))
    offset = idx - (n - 1)          # integer shift (positive → right)
    return idx, offset


def _make_move(h, offset, step, rng):
    """Propose a feasible move that tries to improve the worst shift."""
    n = h.shape[0]
    if offset == 0:
        # increase two distinct positions, decrease two others
        inc = rng.choice(n, size=2, replace=False)
        i, j = inc[0], inc[1]
    else:
        # pick a pair (i, i+offset) that exists
        if offset > 0:
            i = rng.integers(0, n - offset)
        else:
            i = rng.integers(-offset, n)
        j = i + offset
    # choose two positions for decreasing mass, avoiding i and j
    cand = np.where((h > 0) & (np.arange(n) != i) & (np.arange(n) != j))[0]
    if cand.shape[0] < 2:
        return None
    dec = rng.choice(cand, size=2, replace=False)
    p, q = dec[0], dec[1]

    # maximal admissible delta
    delta = step
    delta = min(delta, 1.0 - h[i], 1.0 - h[j])   # keep under 1
    delta = min(delta, h[p], h[q])              # keep above 0
    if delta <= 0:
        return None

    nh = h.copy()
    nh[i] += delta
    nh[j] += delta
    nh[p] -= delta
    nh[q] -= delta
    # keep within bounds; sum is already preserved
    np.clip(nh, 0.0, 1.0, out=nh)
    return nh


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] that reduces the C5 bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        init_h = np.array(initial_h_values, dtype=float)   # provided by harness
        n = init_h.shape[0]
    except Exception:
        # fallback: 47 points, uniform average 0.5
        n = 47
        init_h = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    # make sure the starting vector satisfies the constraints exactly
    init_h = np.clip(init_h, 0.0, 1.0)
    init_h = _project_sum_box(init_h, target_sum)

    best_h = init_h.copy()
    best_c5 = _integral_step(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    start_time = time.time()
    end_time = start_time + budget_s * 0.98   # safety margin
    # parameters for the simulated‑annealing style search
    step = 0.12                # maximal amount moved in a single operation
    T0 = 0.08                  # initial temperature
    while time.time() < end_time:
        elapsed = time.time() - start_time
        progress = elapsed / budget_s
        temperature = max(T0 * (1.0 - progress), 1e-6)

        # location of the current worst shift
        _, off = _worst_shift_info(cur_h)

        # propose a move
        nh = _make_move(cur_h, off, step, rng)
        if nh is None or np.random.rand() < 0.07:   # occasional random jitter
            # random perturbation followed by projection
            delta = rng.normal(scale=step, size=n)
            nh = cur_h + delta
            nh = np.clip(nh, 0.0, 1.0)
            nh = _project_sum_box(nh, target_sum)

        new_c5 = _integral_step(nh)

        # acceptance test
        if new_c5 < cur_c5 - 1e-12:
            cur_h, cur_c5 = nh, new_c5
            if new_c5 < best_c5 - 1e-12:
                best_h, best_c5 = nh.copy(), new_c5
        else:
            # Metropolis criterion
            prob = math.exp(-(new_c5 - cur_c5) / temperature)
            if rng.random() < prob:
                cur_h, cur_c5 = nh, new_c5

        # occasionally shrink the step size to fine‑tune
        if progress > 0.6:
            step = max(step * 0.995, 1e-4)

    # ------------------------------------------------------------------
    # final local polishing with a gradient‑free constrained optimizer
    # ------------------------------------------------------------------
    def objective(vec):
        return _integral_step(vec)

    eq_con = {"type": "eq", "fun": lambda v: np.sum(v) - target_sum}
    bounds = [(0.0, 1.0)] * n
    polish = minimize(
        objective,
        best_h,
        method="SLSQP",
        bounds=bounds,
        constraints=[eq_con],
        options={"maxiter": 250, "ftol": 1e-12, "disp": False},
    )
    polished = _project_sum_box(np.clip(polish.x, 0.0, 1.0), target_sum)
    polished_c5 = _integral_step(polished)
    if polished_c5 < best_c5:
        best_h, best_c5 = polished, polished_c5

    return best_h, float(best_c5), n
