import numpy as np, time, math

def _project_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {0<=x<=1, sum(x)=target} using a simple bisection on a shift."""
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5(h: np.ndarray) -> float:
    """Return C5 for the current step function."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _gradient_step(h: np.ndarray, target: float, alpha: float = 0.05) -> np.ndarray:
    """One projected gradient step using the lag that currently attains C5."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    lag = np.argmax(corr) - (n - 1)

    if lag >= 0:
        fwd = np.zeros_like(h)
        fwd[: n - lag] = 1.0 - h[lag:]
        bwd = np.zeros_like(h)
        bwd[lag:] = h[: n - lag]
    else:
        k = -lag
        fwd = np.zeros_like(h)
        fwd[k:] = 1.0 - h[: n - k]
        bwd = np.zeros_like(h)
        bwd[: n - k] = h[k:]
    g = fwd - bwd
    return _project_sum(h - alpha * g, target)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)  # use the provided parent
    else:
        n = 200
        if n % 2:
            n += 1
        v = rng.random(n)
        h = _project_sum(v, n / 2.0)

    n = h.size
    target = n / 2.0
    # guarantee feasibility (in case the parent is slightly off)
    h = _project_sum(h, target)

    best_h = h.copy()
    best_val = _c5(h)

    # ------------------------------------------------------------------
    # simulated‑annealing style refinement
    # ------------------------------------------------------------------
    start = time.time()
    cur = h.copy()
    cur_val = best_val
    temp = 0.12          # initial temperature
    sigma = 0.12         # initial perturbation scale
    it = 0

    while time.time() - start < budget_s * 0.96:
        # propose a Gaussian perturbation then project back to the feasible set
        prop = cur + rng.normal(scale=sigma, size=n)
        prop = np.clip(prop, 0.0, 1.0)
        prop = _project_sum(prop, target)

        prop_val = _c5(prop)

        # deterministic acceptance if improvement, otherwise Metropolis
        if prop_val < cur_val or rng.random() < math.exp(-(prop_val - cur_val) / max(temp, 1e-12)):
            cur, cur_val = prop, prop_val
            if prop_val < best_val:
                best_val, best_h = prop_val, prop.copy()

        # cooling schedules
        temp *= 0.9996
        sigma *= 0.9997
        it += 1

        # occasional restart to diversify search
        if it % 2500 == 0:
            if rng.random() < 0.5:
                # random start preserving size
                v = rng.random(n)
                cur = _project_sum(v, target)
                cur_val = _c5(cur)
            else:
                # jump back to the current best with a modest shake
                cur = best_h.copy()
                sigma = 0.08
                temp = 0.15

    # ------------------------------------------------------------------
    # fine‑tune the best candidate with a few deterministic gradient steps
    # ------------------------------------------------------------------
    refined = best_h.copy()
    for _ in range(400):
        refined = _gradient_step(refined, target, alpha=0.03)
    refined_val = _c5(refined)
    if refined_val < best_val:
        best_val, best_h = refined_val, refined.copy()

    # final sanity check (project once more to be safe)
    best_h = _project_sum(best_h, target)
    best_val = _c5(best_h)

    return best_h.tolist(), best_val, n
