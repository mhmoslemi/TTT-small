def gradient_descent_step(h, step):
    # compute worst lag
    s = np.correlate(h, h, mode='full')
    # f_vals = T - s...
    idx_min = np.argmin(s)  # minimal autocorrelation
    lag = idx_min - (len(h) - 1)
    # gradient
    g = np.ones_like(h)
    if lag > 0:
        g[:-lag] -= h[lag:]
        g[lag:] -= h[:-lag]
    elif lag < 0:
        d = -lag
        g[d:] -= h[:-d]
        g[:-d] -= h[d:]
    else:
        g -= 2 * h
    # step
    y = h - step * g

def solve_lp_linearized(h_curr):
    N = h_curr.shape[0]
    T = N / 2.0
    # compute correlation
    s = np.correlate(h_curr, h_curr, mode='full')
    f_vals = T - s
    # compute gradient matrix for each lag
    h = h_curr
    # setup cvxpy variables
    h_var = cp.Variable(N)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == T]
    # For each lag:
    n_lags = len(f_vals)
    # shift from -(N-1) to (N-1)
    for idx in range(n_lags):
        lag = idx - (N - 1)
        # compute gradient g for this lag
        g = np.ones(N)
        if lag > 0:
            g[:N-lag] -= h[lag:]
            g[lag:] -= h[:N-lag]
        elif lag < 0:
            d = -lag
            g[d:] -= h[:N-d]
            g[:N-d] -= h[d:]
        else:
            g -= 2 * h
        # compute b = -f_vals[idx] + g @ h
        b = -f_vals[idx] + np.dot(g, h)
        # constraint: g @ h_var - t_var <= b
        constraints.append(g @ h_var - t_var <= b)
    # objective
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None
    return np.array(h_var.value).clip(0, 1), float(t_var.value)

def gradient_for_lag(h, lag):
    N = len(h)
    g = np.ones(N)
    if lag > 0:
        # shift left (i+lag)
        g[:N - lag] -= h[lag:]
        # shift right (i-lag)
        g[lag:] -= h[:N - lag]
    elif lag < 0:
        d = -lag
        g[d:] -= h[:N - d]
        g[:N - d] -= h[d:]
    else:
        g -= 2.0 * h
    return g

def solve_linearized_lp(h):
    import cvxpy as cp
    N = len(h)
    T = N / 2.0
    s = np.correlate(h, h, mode='full')
    f_vals = T - s
    h_var = cp.Variable(N)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == T]

    for idx, f_val in enumerate(f_vals):
        lag = idx - (N - 1)
        # compute gradient vector g
        g = np.ones(N)
        if lag > 0:
            g[:N - lag] -= h[lag:]
            g[lag:] -= h[:N - lag]
        elif lag < 0:
            d = -lag
            g[d:] -= h[:N - d]
            g[:N - d] -= h[d:]
        else:
            g -= 2.0 * h
        b = -f_val + np.dot(g, h)
        constraints.append(g @ h_var - t_var <= b)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    # Solve with ECOS
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        return None, None
    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None, None
    h_opt = np.array(h_var.value)
    t_opt = float(t_var.value)
    # Clip small negative due to numeric
    h_opt = np.clip(h_opt, 0.0, 1.0)
    # Ensure sum close to T
    # maybe re-project to guarantee sum exactly
    h_opt = project_capped_simplex(h_opt, T)
    return h_opt, t_opt

import time
import numpy as np
import cvxpy as cp

def _dx_from_n(n):
    return 2.0 / n

def _c5_of_h(h):
    """C5 = max correlation of h with (1-h) times dx."""
    n = h.shape[0]
    dx = _dx_from_n(n)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_capped_simplex(v, target, eps=1e-12):
    """Project v onto {x: 0≤x≤1, sum(x)=target}."""
    lo = -np.max(v) - 1.0      # forces all entries to be clipped to 1
    hi = np.max(v) + 1.0       # forces all entries to be clipped to 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if rounding left a small residual
    diff = target - x.sum()
    if abs(diff) > eps:
        free = (x > eps) & (x < 1.0 - eps)
        cnt = free.sum()
        if cnt > 0:
            x[free] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

def _worst_lag_and_value(h):
    """Return lag (as integer offset) giving the maximal overlap and its value."""
    n = h.shape[0]
    s = np.correlate(h, h, mode="full")
    total = n / 2.0
    f_vals = total - s                # f(k) = total - autocorrelation
    idx = int(np.argmax(f_vals))      # worst lag -> max f
    lag = idx - (n - 1)
    return lag, float(f_vals[idx])

def _gradient_for_lag(h, lag):
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if lag > 0:
        g[: n - lag] -= h[lag:]
        g[lag:]      -= h[: n - lag]
    elif lag < 0:
        d = -lag
        g[d:]        -= h[: n - d]
        g[: n - d]   -= h[d:]
    else:  # lag == 0
        g -= 2.0 * h
    return g

def _grad_step(h, step):
    lag, _ = _worst_lag_and_value(h)
    g = _gradient_for_lag(h, lag)
    y = h - step * g
    target = h.shape[0] / 2.0
    return _project_capped_simplex(y, target)

def _random_feasible(n, target):
    y = np.random.rand(n)
    return _project_capped_simplex(y, target)

def _solve_linearized_lp(h):
    """One convex‑concave iteration: linearize the worst‑lag function and solve."""
    n = h.shape[0]
    target = n / 2.0
    # autocorrelation and current f-values
    s = np.correlate(h, h, mode="full")
    f_vals = target - s

    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target]

    # add a linearized bound for every lag
    for idx, f_val in enumerate(f_vals):
        lag = idx - (n - 1)
        g = _gradient_for_lag(h, lag)
        # Linearization: f(lag, h) ≈ f_val + g·(h - h_cur)
        # Constraint: f_val + g·(h - h_cur) ≤ t  ⇔  g·h - t ≤ -f_val + g·h_cur
        rhs = -float(f_val) + float(g @ h)
        constraints.append(g @ h_var - t_var <= rhs)

    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:
        return None, None
    if prob.status not in ("optimal", "optimal_inaccurate"):
        return None, None
    h_new = np.asarray(h_var.value, dtype=np.float64)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _project_capped_simplex(h_new, target)
    return h_new, float(t_var.value)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ----- initial feasible vector -----
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=np.float64)
    else:
        # fallback: symmetric linear tent (will be re‑projected)
        n = 80
        x = (np.arange(n) + 0.5) * _dx_from_n(n)   # centre of each cell
        h = np.clip(1.0 - np.abs(x - 1.0), 0.0, 1.0)
    n = h.shape[0]
    target = n / 2.0
    h = _project_capped_simplex(h, target)

    best_h = h.copy()
    best_val = _c5_of_h(best_h)

    # ----- phase 1: gradient descent (40% of time) -----
    step = 0.25
    grad_iters = 0
    while time.time() - start < 0.4 * budget_s:
        h_new = _grad_step(best_h, step)
        val = _c5_of_h(h_new)
        if val < best_val:
            best_val, best_h = val, h_new.copy()
            step = min(step * 1.05, 1.0)   # accelerate when improving
        else:
            step *= 0.97                   # slowly shrink step size
        grad_iters += 1
        if grad_iters > 20000:            # safety cap
            break

    # ----- phase 2: random sampling (20% of time) -----
    while time.time() - start < 0.6 * budget_s:
        h_rand = _random_feasible(n, target)
        val = _c5_of_h(h_rand)
        if val < best_val:
            best_val, best_h = val, h_rand.copy()
        # quick break if time almost used
        if time.time() - start > 0.6 * budget_s:
            break

    # ----- phase 3: linearised LP refinement (up to 5 solves) -----
    lp_solved = 0
    while time.time() - start < 0.9 * budget_s and lp_solved < 5:
        h_lp, _ = _solve_linearized_lp(best_h)
        if h_lp is None:
            break
        val = _c5_of_h(h_lp)
        if val < best_val:
            best_val, best_h = val, h_lp.copy()
        else:
            # if no improvement, try again from a fresh random point
            best_h = _random_feasible(n, target)
        lp_solved += 1

    # ----- phase 4: tiny pairwise tweaks (remainder of time) -----
    while time.time() - start < budget_s - 0.05:
        i, j = np.random.choice(n, 2, replace=False)
        hi, hj = best_h[i], best_h[j]
        # feasible delta interval keeping both in [0,1]
        lo = max(-hi, -(1 - hj))
        hi_delta = min(1 - hi, hj)
        if lo >= hi_delta:
            continue
        delta = np.random.uniform(lo, hi_delta) * 0.4   # small step
        cand = best_h.copy()
        cand[i] += delta
        cand[j] -= delta
        val = _c5_of_h(cand)
        if val < best_val:
            best_val, best_h = val, cand

    # final safety projection (ensures constraints exactly)
    best_h = _project_capped_simplex(best_h, target)

    # return as list for compatibility
    return best_h.tolist(), best_val, n
