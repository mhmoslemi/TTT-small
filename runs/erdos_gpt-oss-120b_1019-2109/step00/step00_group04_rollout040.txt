def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, random
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # Determine initial vector length
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        N0 = len(h0)
        # Upsample to N = N0 * up_factor (choose maybe 2 or 3)
        up_factor = 2
        N = N0 * up_factor
        # repeat each value up_factor times
        h = np.repeat(h0, up_factor)
    else:
        # fallback: start with random balanced binary vector of moderate length
        N = 84
        h = np.zeros(N, dtype=np.float64)
        idx = np.random.choice(N, N//2, replace=False)
        h[idx] = 1.0

    # ensure sum constraint holds
    total_sum = h.sum()
    target_sum = N/2
    # due to upsample, total_sum may already equal target, but adjust slight if needed:
    diff = target_sum - total_sum
    if abs(diff) > 1e-12:
        # adjust a random element within bounds
        # if diff > 0, increase some element <1 by diff; else decrease some element >0
        if diff > 0:
            candidates = np.where(h < 1 - 1e-12)[0]
            i = np.random.choice(candidates)
            h[i] += diff
        else:
            candidates = np.where(h > diff + 1e-12)[0]  # ensure remain >=0
            i = np.random.choice(candidates)
            h[i] += diff  # diff negative
    # ensure bounds by clipping
    np.clip(h, 0, 1, out=h)

    best_h = h.copy()
    best_val = compute_c5(best_h)
    curr_h = h.copy()
    curr_val = best_val

    # parameters for SA
    max_iter = 10_000_000  # high but we will break on time
    step_initial = 0.05  # max delta to move
    step_final = 0.001
    T_initial = 0.01
    T_final = 1e-6

    elapsed = time.time() - start
    while elapsed < budget_s * 0.95:
        # linear schedule for step size and temperature
        frac = elapsed / (budget_s * 0.95)

import numpy as np, time, random

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx, corr

def shifted_array(h, shift):
    N = h.shape[0]
    w = np.zeros(N, dtype=h.dtype)
    if shift >= 0:
        if shift < N:
            w[:N-shift] = h[shift:]
    else:
        k = -shift
        if k < N:
            w[k:] = h[:N - k]
    return w

def binarize_vector(h):
    N = h.shape[0]
    idx = np.argsort(-h)  # descending
    new = np.zeros_like(h)
    new[idx[:N//2]] = 1.0
    return new

            delta = random.random() * dmax
            new_h = curr_h.copy()
            new_h[i] += delta
            new_h[j] -= delta

        new_val, _ = compute_c5(new_h)
        if new_val < curr_val or random.random() < np.exp(-(new_val - curr_val) / max(T, 1e-12)):
            curr_h = new_h
            curr_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val

        it += 1
        # occasional binary check (cheap, helps escape plateaus)
        if it % 2000 == 0:
            bin_h = binarize_vector(best_h)
            bin_val, _ = compute_c5(bin_h)
            if bin_val < best_val:
                best_h = bin_h
                best_val = bin_val

    # ------------------------------------------------------------------ #
    # final binary round‑up attempt
    bin_h = binarize_vector(best_h)
    bin_val, _ = compute_c5(bin_h)
    if bin_val < best_val:
        best_h = bin_h
        best_val = bin_val

    # tiny correction to satisfy the integral exactly
    total = best_h.sum()
    diff = target_sum - total
    if abs(diff) > 1e-10:
        best_h[0] += diff
        best_h[0] = max(0.0, min(1.0, best_h[0]))

    final_c5, _ = compute_c5(best_h)
    return best_h, final_c5, N
