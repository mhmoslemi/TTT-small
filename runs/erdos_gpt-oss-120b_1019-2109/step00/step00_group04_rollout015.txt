def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def improve_h(initial_h, time_budget):
    n = len(initial_h)
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
    start_time = time.time()
    iteration = 0
    temperature = 0.01
    # schedule: temperature decays over iterations
    while time.time() - start_time < time_budget:
        # propose
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # compute allowed delta
        max_transfer = min(cur_h[i], 1 - cur_h[j])   # amount to move from i to j
        min_transfer = -min(cur_h[j], 1 - cur_h[i]) # negative amount: move from j to i
        # random delta in allowed range
        delta = min_transfer + (max_transfer - min_transfer) * np.random.rand()
        # optionally restrict step size: we can also limit magnitude
        # here maybe pass directly
        # apply
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # compute C5
        new_c5 = compute_c5(new_h)
        # SA acceptance
        if new_c5 < cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / max(temperature, 1e-12)):
            cur_h = new_h
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_h = cur_h.copy()
                best_c5 = cur_c5
        # update temperature
        iteration += 1
        temperature = max(1e-6, temperature * 0.9999) # decays slowly
    return best_h, best_c5

import time, math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def fix_sum_constraints(h: np.ndarray) -> np.ndarray:
    """
    Adjust h in-place so that sum(h) = n/2 while keeping 0 ≤ h ≤ 1.
    """
    n = h.shape[0]
    target = n / 2.0
    diff = target - h.sum()
    # simple iterative repair – each step moves the whole remaining diff
    # to a randomly chosen free coordinate.
    # The loop stops when the residual diff is negligible.
    while abs(diff) > 1e-12:
        if diff > 0.0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            inc = min(diff, 1.0 - h[i])
            h[i] += inc
            diff -= inc
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            i = np.random.choice(idx)
            dec = min(-diff, h[i])
            h[i] -= dec
            diff += dec
    return h


def interpolate_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of an existing h onto a new grid of size new_n."""
    old_n = old_h.shape[0]
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(new_x, old_x, old_h)
    # enforce the integral constraint
    target = new_n / 2.0
    cur = new_h.sum()
    if cur != 0.0:
        new_h *= target / cur
    new_h = np.clip(new_h, 0.0, 1.0)
    return fix_sum_constraints(new_h)


def local_optimize(h0: np.ndarray, time_budget: float):
    """Simple simulated‑annealing on the feasible set."""
    n = h0.shape[0]
    cur_h = h0.copy()
    cur_c = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    start = time.time()
    # start temperature proportional to the current objective
    temperature = max(1e-6, cur_c * 0.08)

    while time.time() - start < time_budget:
        # propose a transfer of mass between two positions
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # amount that can be moved from i to j without leaving bounds
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        min_transfer = -min(cur_h[j], 1.0 - cur_h[i])  # moving opposite way
        if max_transfer == 0.0 and min_transfer == 0.0:
            continue

        # uniform proposal in the feasible interval
        delta = min_transfer + (max_transfer - min_transfer) * np.random.rand()

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        new_c = compute_c5(new_h)

        # Metropolis acceptance
        if new_c < cur_c or np.random.rand() < math.exp(-(new_c - cur_c) / max(temperature, 1e-12)):
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_h = cur_h.copy()
                best_c = cur_c

        # geometric cooling
        temperature *= 0.9999
        # occasional reheating to avoid freezing too early
        if temperature < 1e-9:
            temperature = max(1e-6, best_c * 0.02)

    return best_h, best_c
