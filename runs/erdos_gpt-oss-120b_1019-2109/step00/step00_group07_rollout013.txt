import numpy as np
import random
import time

def _evaluate_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx using the discretisation."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _project_to_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto {0 ≤ x_i ≤ 1, sum x_i = target_sum}."""
    lower, upper = 0.0, 1.0
    # binary search for λ such that sum clip(v+λ) = target_sum
    lam_low = lower - np.max(v)          # makes all entries ≤ lower after clipping
    lam_high = upper - np.min(v)         # makes all entries ≥ upper after clipping
    for _ in range(60):
        lam = (lam_low + lam_high) * 0.5
        w = np.clip(v + lam, lower, upper)
        s = w.sum()
        if s > target_sum:
            lam_high = lam
        else:
            lam_low = lam
    w = np.clip(v + lam_low, lower, upper)

    # tiny correction to hit the sum exactly (within floating point limits)
    diff = target_sum - w.sum()
    if abs(diff) > 1e-12:
        free = np.where((w > 0.0) & (w < 1.0))[0]
        if free.size > 0:
            # distribute the tiny remainder evenly
            add = diff / free.size
            w[free] += add
            w = np.clip(w, lower, upper)
            # safety final tweak
            diff2 = target_sum - w.sum()
            if abs(diff2) > 1e-12:
                i = free[0]
                w[i] += diff2
                w = np.clip(w, lower, upper)
    return w

def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Sub‑gradient of the term h·(1‑h_shift) for a given integer shift."""
    n = h.shape[0]
    g = np.zeros_like(h)

    if shift >= 0:
        i = np.arange(0, n - shift, dtype=int)
        j = i + shift
    else:
        s = -shift
        i = np.arange(s, n, dtype=int)
        j = i - s

    # derivative w.r.t. h_i : 1 - h_j
    g[i] += 1.0 - h[j]
    # derivative w.r.t. h_j : -h_i
    g[j] += -h[i]
    return g

                    break
        else:
            # simple annealing decision
            T = max(T0 * (1.0 - frac), 1e-7)
            prob = np.exp(-(c5_candidate - current_c5) / T)
            if random.random() < prob:
                current_h, current_c5 = h_candidate, c5_candidate

        # --------------------------------------------------------------
        # occasional random jitter to explore new regions
        if iteration % 250 == 0:
            # add a small Gaussian perturbation then re‑project
            noise = np.random.randn(n) * 0.018
            h_jitter = _project_to_feasible(current_h + noise, target_sum)
            c5_jitter = _evaluate_c5(h_jitter)
            if c5_jitter < best_c5:
                best_h, best_c5 = h_jitter.copy(), c5_jitter
                current_h, current_c5 = h_jitter.copy(), c5_jitter
            else:
                # occasionally reset to the best found so far
                if random.random() < 0.08:
                    current_h, current_c5 = best_h.copy(), best_c5

        # --------------------------------------------------------------
        # a pure random restart (very rare) – helps escape deep traps
        if iteration % 3200 == 0:
            rnd = np.random.rand(n)
            h_rand = _project_to_feasible(rnd, target_sum)
            c5_rand = _evaluate_c5(h_rand)
            if c5_rand < best_c5:
                best_h, best_c5 = h_rand.copy(), c5_rand
                current_h, current_c5 = h_rand.copy(), c5_rand

    # ------------------------------------------------------------------
    # final safety: ensure the returned vector satisfies the constraints
    best_h = _project_to_feasible(best_h, target_sum)

    return best_h, best_c5, n
