def anneal(initial_h, time_limit):
    n = len(initial_h)
    ones = list(np.where(initial_h==1)[0])
    zeros = list(np.where(initial_h==0)[0])
    h = initial_h.copy()
    cur = compute_max_mismatch(h)
    best_h, best = h.copy(), cur
    it = 0
    start = time.time()
    while time.time() - start < time_limit:
        # temperature schedule
        t_frac = (time.time() - start) / time_limit
        T = max_temp * (1 - t_frac)   # linear decay
        # propose swap
        i = random.choice(ones)
        j = random.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        # compute new cost
        new = compute_max_mismatch(h)
        # decide acceptance
        if new < cur or random.random() < math.exp(-(new - cur)/T):
            cur = new
            # update ones and zeros
            ones.remove(i); zeros.append(i)
            zeros.remove(j); ones.append(j)
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
        if cur < best:
            best, best_h = cur, h.copy()
        it += 1
    return best_h, best
```

def hill_climb(h, time_limit, max_no_improve=10000):
    n = len(h)
    cur_c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), cur_c5
    no_improve = 0
    start = time.time()
    while time.time() - start < time_limit:
        # propose swap
        i = random.randint(0, n-1)
        # find j such that h[j] != h[i]
        # we can pick j= random index until diff
        j = random.randint(0, n-1)
        while h[j] == h[i]:
            j = random.randint(0, n-1)
        # swap
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        # acceptance: if new <= cur, accept
        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = h[j], h[i]
            no_improve += 1
            if no_improve >= max_no_improve:
                # maybe random restart
                break
    return best_h, best_c5
```

def simulated_annealing(h0, time_limit):
    import numpy as np, random, math, time
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    # ensure binary
    # maintain sets of ones and zeros
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    start = time.time()
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        # temperature schedule
        fraction = elapsed / time_limit
        T = MAX_TEMP * (1.0 - fraction)
        # propose swap
        i = random.choice(tuple(ones))
        j = random.choice(tuple(zeros))
        # swap
        h[i] = 0
        h[j] = 1
        # update set
        ones.remove(i); zeros.add(i)
        zeros.remove(j); ones.add(j)
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            if T > 0:
                prob = math.exp(-delta / T)
                if random.random() < prob:
                    accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                # early exit if target reached
                if best_c5 <= TARGET:
                    break
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
            ones.add(i); zeros.remove(i)
            zeros.add(j); ones.remove(j)
        # occasional restart may improve?? keep same
    return best_h, best_c5

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # use np.correlate with mode='full'
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def hill_climb(h, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    # maintain lists
    ones = [i for i, val in enumerate(cur_h) if val > 0.5]
    zeros = [i for i, val in enumerate(cur_h) if val <= 0.5]
    start = time.time()
    while time.time() - start < time_limit:
        # Candidate sampling
        # number of candidates per iteration
        candidates = 5
        best_swap = None
        best_new_c5 = cur_c5
        for _ in range(candidates):
            i = random.choice(ones)
            j = random.choice(zeros)
            # swap temporarily
            cur_h[i], cur_h[j] = 0.0, 1.0
            new_c5 = compute_c5(cur_h)
            if new_c5 < best_new_c5:
                best_new_c5 = new_c5
                best_swap = (i, j)
            # revert
            cur_h[i], cur_h[j] = 1.0, 0.0
        if best_swap is not None and best_new_c5 < cur_c5:
            i, j = best_swap
            cur_h[i], cur_h[j] = 0.0, 1.0
            cur_c5 = best_new_c5
            # update lists

            zeros.append(i)
            zeros.remove(j)

def hill_climb(h, time_limit):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    # maintain ones, zeros lists
    ones = [i for i, val in enumerate(cur_h) if val == 1]
    zeros = [i for i, val in enumerate(cur_h) if val == 0]
    start = time.time()
    while time.time() - start < time_limit:
        # compute worst shift
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        idx_max = int(np.argmax(corr))
        d = idx_max - (n - 1)  # shift (could be negative)
        if d < 0:
            d = -d
        # collect offending positions
        offending = None
        if d < n:
            if d >= 0:
                # left indices i where h[i]=1, h[i+d]=0
                # overlapping region i in 0..n-d-1
                # we can loop to build list quickly
                off = []
                for i in range(0, n - d):
                    if cur_h[i] == 1.0 and cur_h[i + d] == 0.0:
                        off.append(i)
                offending = off
        # attempt targeted swap
        made_swap = False
        if offending:
            i = random.choice(offending)
            # choose j from zeros (any)
            j = random.choice(zeros)
            if cur_h[j] == 0.0:
                # perform swap
                cur_h[i] = 0.0
                cur_h[j] = 1.0
                # update ones/zeros
                ones.remove(i); zeros.append(i)
                zeros.remove(j); ones.append(j)
                new_c5 = compute_c5(cur_h)
                if new_c5 < cur_c5:
                    cur_c5 = new_c5
                    made_swap = True
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = cur_h.copy()
                else:
                    # revert
                    cur_h[i] = 1.0
                    cur_h[j] = 0.0
                    ones.append(i); zeros.remove(i)
                    zeros.append(j); ones.remove(j)
        if not made_swap:
            # random swap fallback
            i = random.choice(ones)
            j = random.choice(zeros)
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            # update ones zeros
            ones.remove(i); zeros.append(i)
            zeros.remove(j); ones.append(j)
            new_c5 = compute_c5(cur_h)
            if new_c5 < cur_c5:
                cur_c5 = new_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur_h.copy()

                # revert

                cur_h[j] = 0.0
                ones.append(i); zeros.remove(i)
                zeros.append(j); ones.remove(j)
        # early stopping if best_c5 <= TARGET
        if best_c5 <= TARGET:
            break
    return best_h, best_c5

import time, math, random
import numpy as np

# ---------- helpers ----------
def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx for discrete step h."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def hill_climb(start_h: np.ndarray, time_limit: float, target: float = 0.3808) -> tuple:
    """
    Simple deterministic+targeted hill‑climbing.
    Returns (best_h, best_c5) found within time_limit seconds.
    """
    n = start_h.shape[0]
    cur_h = start_h.copy()
    best_h = cur_h.copy()
    cur_c5 = compute_c5(cur_h)
    best_c5 = cur_c5

    # sets of indices for fast random sampling
    ones = set(int(i) for i, v in enumerate(cur_h) if v > 0.5)
    zeros = set(int(i) for i, v in enumerate(cur_h) if v <= 0.5)

    start = time.time()
    while time.time() - start < time_limit:
        # ---------- find worst shift ----------
        corr_full = np.correlate(cur_h, 1.0 - cur_h, mode="full")
        idx_max = int(np.argmax(corr_full))
        d = abs(idx_max - (n - 1))          # shift magnitude (non‑negative)

        # ---------- try a targeted swap ----------
        made_change = False
        if 0 < d < n:                       # there is an overlapping region
            # collect positions i where h[i]=1 and h[i+d]=0 (a contribution to the max)
            offending = [i for i in range(0, n - d)
                         if cur_h[i] == 1.0 and cur_h[i + d] == 0.0]
            if offending:
                i = random.choice(offending)          # remove a bad contribution
                j = random.choice(tuple(zeros))       # bring a 1 from elsewhere
                # perform swap
                cur_h[i] = 0.0
                cur_h[j] = 1.0
                # update sets
                ones.remove(i); zeros.add(i)
                zeros.remove(j); ones.add(j)

                new_c5 = compute_c5(cur_h)
                if new_c5 <= cur_c5:
                    cur_c5 = new_c5
                    made_change = True
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = cur_h.copy()
                else:
                    # revert
                    cur_h[i] = 1.0
                    cur_h[j] = 0.0
                    ones.add(i); zeros.remove(i)
                    zeros.add(j); ones.remove(j)

        # ---------- fallback random swap ----------
        if not made_change:
            i = random.choice(tuple(ones))
            j = random.choice(tuple(zeros))
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            ones.remove(i); zeros.add(i)
            zeros.remove(j); ones.add(j)

            new_c5 = compute_c5(cur_h)
            if new_c5 <= cur_c5:
                cur_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = cur_h.copy()
            else:
                # revert
                cur_h[i] = 1.0
                cur_h[j] = 0.0
                ones.add(i); zeros.remove(i)
                zeros.add(j); ones.remove(j)

        if best_c5 <= target:
            break

    return best_h, best_c5

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    TARGET = 0.3808                 # we try to beat this value
    TOTAL_TIME = float(budget_s)

    # pick a convenient even length (the algorithm works best with binary vectors)
    n = 200
    if n % 2:
        n += 1

    # start with a random feasible binary step function
    h0 = np.zeros(n, dtype=np.float64)
    h0[np.random.choice(n, n // 2, replace=False)] = 1.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    start_all = time.time()
    # allocate most of the budget to a series of short hill‑climbs with fresh starts
    while time.time() - start_all < TOTAL_TIME * 0.85:
        # fresh random start
        h_start = np.zeros(n, dtype=np.float64)
        h_start[np.random.choice(n, n // 2, replace=False)] = 1.0

        # give each climb a few seconds (or the remaining budget if tiny)
        remaining = TOTAL_TIME - (time.time() - start_all)
        climb_time = min(3.0, remaining * 0.5)
        if climb_time <= 0.0:
            break

        h_opt, c5_opt = hill_climb(h_start, climb_time, target=TARGET)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()
            if best_c5 <= TARGET:
                break

    # final refinement from the best found so far, using the rest of the time
    remaining = TOTAL_TIME - (time.time() - start_all)
    if remaining > 0.2:
        h_opt, c5_opt = hill_climb(best_h, remaining, target=TARGET)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()

    # ensure feasibility (floating point noise may appear)
    best_h = np.clip(best_h, 0.0, 1.0)
    # final sanity check – enforce exact half‑measure if needed
    # (tiny adjustments via rounding are safe because the bound is tolerant)
    half = n // 2
    ones = int(round(best_h.sum()))
    if ones != half:
        # flip a few entries to fix the count
        diff = half - ones
        idx_one = np.where(best_h > 0.5)[0]
        idx_zero = np.where(best_h <= 0.5)[0]
        if diff > 0:
            # need more ones
            to_one = np.random.choice(idx_zero, diff, replace=False)
            best_h[to_one] = 1.0
        elif diff < 0:
            # need fewer ones
            to_zero = np.random.choice(idx_one, -diff, replace=False)
            best_h[to_zero] = 0.0

    return best_h.tolist(), float(best_c5), int(n)
