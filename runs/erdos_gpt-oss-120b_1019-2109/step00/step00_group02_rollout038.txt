def simulated_annealing(n, time_budget):
    rng = np.random.default_rng(seed)
    # initial random binary with half ones
    h = np.zeros(n, dtype=np.float64)
    ones_idx = rng.choice(n, size=n//2, replace=False)
    h[ones_idx] = 1.0
    best_h = h.copy()
    best_val = max_overlap(h)
    current_val = best_val
    # list of indices of ones and zeros
    idx_ones = list(ones_idx)
    idx_zeros = [i for i in range(n) if h[i]==0]
    # temperature schedule: start with T0 = something like 0.1 * (best_val) or something
    start_time = time.time()
    it = 0
    while time.time() - start_time < time_budget:
        # propose swap
        i = rng.choice(idx_ones)
        j = rng.choice(idx_zeros)
        # perform swap
        h[i] = 0.0; h[j] = 1.0
        # update idx lists
        # (swap indices)
        # But to update we need to maintain sets; we can rebuild each iteration? Simpler: just recompute ones and zeros after each swap via lazy but maybe slower.
        # Actually we can maintain arrays:
        idx_ones.remove(i); idx_ones.append(j)
        idx_zeros.remove(j); idx_zeros.append(i)
        val = max_overlap(h)
        # acceptance
        if val < current_val or rng.random() < np.exp(-(val - current_val) / T):
            current_val = val

        it+=1
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    from scipy.optimize import minimize

    rng = np.random.default_rng(seed)
    start = time.time()
    # Determine n
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values, dtype=np.float64)
        n = init_h.size
    else:
        n = 45 # default maybe
        init_h = np.full(n, 0.5, dtype=np.float64)
    total_sum = n / 2.0
    dx = 2.0 / n

    def compute_c5(h):
        c = np.correlate(h, 1 - h, mode='full')
        return np.max(c) * dx

    # ensure feasibility of init_h:
    def repair(h):
        h = np.clip(h, 0.0, 1.0)
        cur_sum = h.sum()
        if abs(cur_sum - total_sum) > 1e-9:
            # adjust uniformly
            diff = total_sum - cur_sum
            # distribute diff proportionally to non-bound entries
            free = (h > 0) & (h < 1)
            if free.any():
                # add diff proportionally
                h[free] += diff / free.sum()
                h = np.clip(h, 0.0, 1.0)
                # if still off, scale
                cur_sum = h.sum()
                if abs(cur_sum - total_sum) > 1e-6:
                    h = h * (total_sum / cur_sum)
            else:
                h = h * (total_sum / cur_sum)
            h = np.clip(h, 0.0, 1.0)
        # final clamp
        h = np.clip(h, 0.0, 1.0)
        return h

    best_h = repair(init_h.copy())
    best_c = compute_c5(best_h)

    # Evaluate initial if function provided:
    try:
        from __main__ import evaluate_erdos_solution
        # maybe evaluate_erdos_solution returns same as compute_c5
        # but we compute anyway
        best_c = compute_c5(best_h)
    except Exception:
        pass

    # Gradient-based optimizer helper
    def smoothed_max_obj(h, alpha):
        # total sum known
        A = np.correlate(h, h, mode='full')
        C = total_sum - A
        maxC = C.max()
        # stability
        exp_term = np.exp(alpha * (C - maxC))
        return maxC + np.log(exp_term.sum()) / alpha

    def smoothed_max_grad(h, alpha):
        n_local = h.size
        A = np.correlate(h, h, mode='full')
        C = total_sum - A
        maxC = C.max()
        exp_term = np.exp(alpha * (C - maxC))
        Z = exp_term.sum()
        probs = exp_term / Z
        offset = n_local - 1
        term = np.zeros_like(h)
        for idx, prob in enumerate(probs):
            k = idx - offset
            if prob == 0.0:
                continue
            if k == 0:
                term += prob * (2.0 * h)
            elif k > 0:
                term[:n_local - k] += prob * h[k:]
                term[k:] += prob * h[:n_local - k]
            else:

                term[:n_local - k_abs] += prob * h[k_abs:]

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----- helper functions ----------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear (non‑circular) correlation as used by the evaluator
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def repair_h(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Project h onto the feasible set 0≤h≤1, Σh = target_sum."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12:
        free = (h > 0.0) & (h < 1.0)
        if free.any():
            # spread the discrepancy uniformly over free entries
            adj = (target_sum - s) / free.sum()
            h[free] += adj
            h = np.clip(h, 0.0, 1.0)
            s = h.sum()
            if abs(s - target_sum) > 1e-9:
                h *= target_sum / s
        else:
            h *= target_sum / s
        h = np.clip(h, 0.0, 1.0)
    return h


# ----- main optimisation routine -------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ---- initialise --------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)  # type: ignore
        n = h0.size
    except Exception:
        # fallback: a modest size that respects the “<1k samples” guideline
        n = 45
        h0 = np.full(n, 0.5, dtype=np.float64)

    target_sum = n / 2.0
    h0 = repair_h(h0, target_sum)

    best_h = h0.copy()
    best_c = compute_c5(best_h)

    # ---- simple simulated‑annealing hill‑climb ----------------------------
    cur_h = best_h.copy()
    cur_c = best_c

    # temperature schedule (very low because the objective is already small)
    T = 0.005
    cooling = 0.99994

    # number of iterations is governed by the time budget
    while time.time() - start_time < budget_s * 0.94:
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # amount we may move from i to j respecting the box constraints
        max_transfer = min(cur_h[i], 1.0 - cur_h[j])
        if max_transfer <= 0.0:
            continue

        # random step size (larger steps early, smaller later)
        delta = rng.random() * max_transfer
        # apply the move
        cur_h[i] -= delta
        cur_h[j] += delta

        # evaluate
        new_c = compute_c5(cur_h)

        # keep global best
        if new_c < best_c:
            best_c = new_c
            best_h = cur_h.copy()

        # accept if improvement or with Metropolis probability
        if new_c < cur_c or rng.random() < math.exp(-(new_c - cur_c) / T):
            cur_c = new_c
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta

        # gradual cooling
        T = max(T * cooling, 1e-9)

        # occasional reset to the global best to avoid drift
        if (time.time() - start_time) % 30 < 0.1:
            cur_h = best_h.copy()
            cur_c = best_c

    # ---- final polishing with a few SLSQP refinements --------------------
    # The quadratic nature of the problem makes a short constrained optimisation
    # useful even if we start from a good discrete point.
    def objective(x, alpha):
        A = np.correlate(x, x, mode="full")
        Cvec = target_sum - A
        mx = Cvec.max()
        # smooth approximation of max
        exp_term = np.exp(alpha * (Cvec - mx))
        return mx + np.log(exp_term.sum()) / alpha

    def gradient(x, alpha):
        nloc = x.size
        A = np.correlate(x, x, mode="full")
        Cvec = target_sum - A
        mx = Cvec.max()
        exp_term = np.exp(alpha * (Cvec - mx))
        Z = exp_term.sum()
        probs = exp_term / Z
        offset = nloc - 1
        term = np.zeros_like(x)
        for idx, p in enumerate(probs):
            k = idx - offset
            if p == 0.0:
                continue
            if k == 0:
                term += p * (2.0 * x)
            elif k > 0:
                term[: nloc - k] += p * x[k:]
                term[k:] += p * x[: nloc - k]
            else:   # k < 0
                k = -k
                term[: nloc - k] += p * x[k:]
                term[k:] += p * x[: nloc - k]
        return 1.0 - term

    # perform a short cascade of increasing smoothness
    refined = best_h.copy()
    for alpha in (5.0, 20.0, 80.0):
        cons = {"type": "eq", "fun": lambda x: np.sum(x) - target_sum}
        bounds = [(0.0, 1.0)] * n
        res = minimize(
            fun=lambda x, a=alpha: objective(x, a),
            x0=refined,
            jac=lambda x, a=alpha: gradient(x, a),
            method="SLSQP",
            bounds=bounds,
            constraints=cons,
            options={"maxiter": 180, "ftol": 1e-10},
        )
        refined = repair_h(res.x, target_sum)

    refined_c = compute_c5(refined)
    if refined_c < best_c:
        best_c = refined_c
        best_h = refined.copy()

    # ---- final sanity -------------------------------------------------------
    best_h = repair_h(best_h, target_sum)
    try:
        # the harness provides this helper; fall back to our own if unavailable
        final_c = float(evaluate_erdos_solution(best_h))  # type: ignore
    except Exception:
        final_c = compute_c5(best_h)

    return (best_h, final_c, n)
