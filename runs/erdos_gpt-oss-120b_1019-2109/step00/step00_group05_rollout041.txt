def project_onto_simplex_box(v, target):
    # find lambda such that sum_i min(1, max(0, v_i - lam)) = target
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    return x

def compute_gradient(h, shift, dx):
    n = h.shape[0]
    if shift >= 0:
        h_plus = np.concatenate([h[shift:], np.zeros(shift)])
        h_minus = np.concatenate([np.zeros(shift), h[:n-shift]])
    else:
        s = -shift
        h_plus = np.concatenate([np.zeros(s), h[:n-s]])
        h_minus = np.concatenate([h[s:], np.zeros(s)])
    return dx * (1.0 - (h_plus + h_minus))
```

def project_onto_simplex_box(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return x
```

import time
import numpy as np

# -------------------- helper functions --------------------

def _project_onto_capped_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Projection of v onto the set {x : 0<=x_i<=1, sum x_i = target}."""
    # Find λ such that sum_i clip(v_i - λ,0,1) = target.
    lo = v.min() - 1.0        # makes all entries >1 after subtraction
    hi = v.max()              # makes all entries <=0 after subtraction
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Evaluate C5 = max_k ∫ h(x)*(1-h(x+k)) dx using discrete correlation."""
    # np.correlate gives the discrete convolution (full) of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _self_corr(h: np.ndarray, dx: float) -> np.ndarray:
    """Self‑correlation of h (∫ h(x)h(x+k)dx) for all integer shifts."""
    return np.correlate(h, h, mode="full") * dx


def _gradient_for_shift(h: np.ndarray, shift: int, dx: float) -> np.ndarray:
    """Gradient of I_k = ∫ h(1-h_shift) for a fixed shift."""
    n = h.shape[0]
    if shift >= 0:
        plus = np.concatenate([h[shift:], np.zeros(shift)])
        minus = np.concatenate([np.zeros(shift), h[: n - shift]])
    else:
        s = -shift
        plus = np.concatenate([np.zeros(s), h[: n - s]])
        minus = np.concatenate([h[s:], np.zeros(s)])
    return dx * (1.0 - (plus + minus))


def _local_optimize(
    h0: np.ndarray,
    target_sum: float,
    dx: float,
    max_iter: int,
    rng: np.random.RandomState,
) -> np.ndarray:
    """Simple projected gradient descent with occasional random jumps."""
    h = h0.copy()
    best_val = _c5_value(h, dx)

    n = h.shape[0]
    no_improve = 0

    for it in range(max_iter):
        # locate the shift giving the current worst overlap (largest I)
        corr = _self_corr(h, dx)
        worst_idx = np.argmin(corr)               # minimal self‑correlation
        shift = worst_idx - (n - 1)

        # gradient of I for that shift
        grad = _gradient_for_shift(h, shift, dx)

        improved = False
        # try a few step sizes
        for step in (1.0, 0.5, 0.2, 0.1, 0.05):
            y = h - step * grad
            y = _project_onto_capped_simplex(y, target_sum)
            val = _c5_value(y, dx)
            if val < best_val - 1e-12:
                best_val = val
                h = y
                improved = True
                no_improve = 0
                break

        if not improved:
            no_improve += 1
            # occasional random perturbation to escape stagnation
            if no_improve > 30 or rng.rand() < 0.05:
                # small random transfer between two coordinates
                i, j = rng.choice(n, size=2, replace=False)
                # amount we can move without breaking bounds
                max_move = min(h[i], 1.0 - h[j])
                if max_move > 1e-6:
                    eps = rng.rand() * max_move
                    y = h.copy()
                    y[i] -= eps
                    y[j] += eps
                    y = _project_onto_capped_simplex(y, target_sum)
                    h = y
                else:
                    # full random restart
                    y = rng.rand(n)
                    y = _project_onto_capped_simplex(y, target_sum)
                    h = y
                best_val = _c5_value(h, dx)
                no_improve = 0
    return h


# -------------------- main entry point --------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h on [0,2] as a list of values,
    the achieved C5 bound, and the number of sample points.
    """
    rng = np.random.RandomState(seed)
    start_time = time.time()

    # Determine problem size
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = int(init.size)
    else:
        n = 70                      # default size if no parent solution
        init = None

    dx = 2.0 / n
    target_sum = n / 2.0

    # -----------------------------------------------------------------
    # Build an initial pool of candidates
    # -----------------------------------------------------------------
    candidates = []

    if init is not None:
        candidates.append(_project_onto_capped_simplex(init, target_sum))

    # alternating 0/1 pattern (exactly n/2 ones)
    alt = np.zeros(n)
    alt[::2] = 1.0
    candidates.append(_project_onto_capped_simplex(alt, target_sum))

    # half‑block pattern
    half = np.zeros(n)
    half[: n // 2] = 1.0
    candidates.append(_project_onto_capped_simplex(half, target_sum))

    # a simple linear “tent” shape (high at edges, low in centre)
    x = np.linspace(0.0, 2.0, n)
    tent = 1.0 - np.abs(x - 1.0)  # goes from 0 at centre to 1 at edges
    candidates.append(_project_onto_capped_simplex(tent, target_sum))

    # a few completely random starts
    for _ in range(5):
        y = rng.rand(n)
        candidates.append(_project_onto_capped_simplex(y, target_sum))

    best_h = None
    best_val = float("inf")

    # -----------------------------------------------------------------
    # Optimize each candidate while time permits
    # -----------------------------------------------------------------
    for h0 in candidates:
        if time.time() - start_time > budget_s * 0.94:
            break

        # allocate iteration budget proportionally to remaining time
        remaining = max(0.0, budget_s - (time.time() - start_time))
        max_iter = int(min(8000, max(200, remaining * 25)))

        h_opt = _local_optimize(h0, target_sum, dx, max_iter, rng)
        val = _c5_value(h_opt, dx)

        if val < best_val:
            best_val = val
            best_h = h_opt.copy()

    # If nothing succeeded, fall back to a simple feasible vector
    if best_h is None:
        best_h = np.full(n, 0.5)
        best_h = _project_onto_capped_simplex(best_h, target_sum)
        best_val = _c5_value(best_h, dx)

    # Final projection to guarantee feasibility
    best_h = _project_onto_capped_simplex(best_h, target_sum)
    best_val = _c5_value(best_h, dx)

    return (best_h.tolist(), float(best_val), n)
