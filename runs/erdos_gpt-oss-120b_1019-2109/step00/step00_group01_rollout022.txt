    # Want projection onto set {x in [0,1]^n, sum x = s}
    # Use algorithm: find lambda via binary search such that sum clip(v - lambda,0,1) = s.
    # Since function decreasing in lambda.
    # We'll compute lower and upper bounds for lambda.
    n = len(v)
    # lambda lower bound: min(v) - 1 maybe? Actually for lambda very negative, clip(v - lambda,0,1) = 1 for all entries that v - lambda > 1 => if lambda << min(v)-1 then all entries become 1, sum=n.
    # For very large lambda, all become 0.
    # So we can binary search between lo = min(v)-1 and hi = max(v) (since if lambda = max(v) then v - lambda <=0 => all 0). Ensure sum hi <= s.

    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0, 1)
        cur = np.sum(x)
        if cur > s:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo+hi)/2.0, 0, 1)

def compute_softmin_grad(h, gamma):
    corr = np.correlate(h, h, mode='full')
    z = -gamma * corr
    # stable log-sum-exp calculation
    maxz = np.max(z)
    expz = np.exp(z - maxz)
    p = expz / expz.sum()
    n = h.shape[0]
    offset = n - 1
    grad = np.zeros_like(h)
    for idx, weight in enumerate(p):
        if weight == 0:
            continue
        d = idx - offset
        if d >= 0:
            # contribution from h[i+d]
            grad[:n-d] += weight * h[d:]
            # contribution from h[i-d] (i-d = i-d)
            grad[d:] += weight * h[:n-d]
        else:
            dd = -d
            grad[dd:] += weight * h[:n-dd]
            grad[:n-dd] += weight * h[dd:]
    return grad

import time
import numpy as np

def _proj_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0≤x≤1, sum(x)=target} using binary search on λ."""
    lo = np.min(v) - 1.0          # makes all entries ->1 after clipping
    hi = np.max(v) + 1.0          # makes all entries ->0 after clipping
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _softmin_corr_grad(h: np.ndarray, gamma: float) -> np.ndarray:
    """Gradient of the soft‑min of the autocorrelation (to be maximised)."""
    n = h.shape[0]
    corr = np.correlate(h, h, mode='full')
    z = -gamma * corr
    maxz = np.max(z)
    expz = np.exp(z - maxz)
    prob = expz / expz.sum()               # weight for each shift
    offset = n - 1
    grad = np.zeros_like(h)

    # contribution of each shift
    for idx, w in enumerate(prob):
        if w == 0.0:
            continue
        d = idx - offset                    # shift amount
        if d >= 0:
            # forward part
            grad[: n - d] += w * h[d:]
            # backward part
            grad[d:] += w * h[: n - d]
        else:
            dd = -d
            grad[dd:] += w * h[: n - dd]
            grad[: n - dd] += w * h[dd:]
    return grad


def _c5_from_h(h: np.ndarray) -> float:
    """Compute C₅ = max_k ∫ h(x)(1‑h(x+k))dx using the discrete definition."""
    n = h.shape[0]
    # min autocorrelation gives the maximal non‑overlap
    min_corr = np.min(np.correlate(h, h, mode='full'))
    return 1.0 - (2.0 * min_corr) / n


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a low Erdos C₅ bound.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Initialise ---------------------------------------------------------
    try:
        # pre‑imported initial_h_values exists
        h = np.asarray(initial_h_values, dtype=float)
        n = h.shape[0]
    except NameError:
        # fall back to a modest size if no parent is supplied
        n = 200
        h = np.full(n, 0.5, dtype=float)

    target_sum = n * 0.5                     # ∑h_i = n/2
    h = _proj_box_simplex(h, target_sum)

    best_h = h.copy()
    best_val = _c5_from_h(best_h)

    # ------------------------------------------------------------------
    # Optimisation parameters -------------------------------------------
    step0 = 0.12
    gamma = 1.0
    it = 0
    # main optimisation loop -------------------------------------------------
    while True:
        if time.time() - start > budget_s * 0.97:
            break

        # gradient ascent on the soft‑min of autocorrelation
        grad = _softmin_corr_grad(h, gamma)

        # simple decreasing step‑size schedule
        step = step0 / (1.0 + it * 1e-3)

        # take a step and project back to feasibility
        h_new = _proj_box_simplex(h + step * grad, target_sum)

        # occasional tiny random jitter to escape shallow plateaus
        if it % 500 == 0 and it > 0:
            jitter = (np.random.rand(n) - 0.5) * 0.02
            h_new = _proj_box_simplex(h_new + jitter, target_sum)

        val = _c5_from_h(h_new)

        if val < best_val:
            best_val = val
            best_h = h_new.copy()

        h = h_new
        it += 1

        # slowly increase the soft‑min sharpness
        if it % 200 == 0 and gamma < 500:
            gamma *= 1.07

    # ------------------------------------------------------------------
    # Final result -------------------------------------------------------
    final_c5 = _c5_from_h(best_h)
    return best_h, final_c5, best_h.shape[0]
