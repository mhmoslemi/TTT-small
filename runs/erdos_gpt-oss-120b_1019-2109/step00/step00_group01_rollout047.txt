def compute_c5(h):
    conv = np.correlate(h, 1 - h, mode='full')
    return conv.max() * (2.0 / len(h))

def hill_climb(initial_h, budget):
    start_time = time.time()
    n = len(initial_h)
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    # precompute contributions maybe not needed
    temperature = 1.0
    decay = 0.9995
    while time.time() - start_time < budget:
        # choose two indices
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            j = np.random.randint(0, n)
        # propose move
        # bound the max epsilon
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps <= 1e-12:
            continue
        eps = np.random.rand() * max_eps  # uniform [0, max_eps]
        # apply move
        h[i] -= eps
        h[j] += eps
        # compute cost
        new_c5 = compute_c5(h)
        # decide acceptance
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        # Simulated annealing acceptance
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            # keep change
            pass
        else:
            # revert
            h[i] += eps
            h[j] -= eps
        temperature *= decay
    return best_h, best_c5

    h = h_init.copy()
    n = len(h)
    cur_c5 = compute_c5(h)
    best_h_local = h.copy()
    best_c5_local = cur_c5

    # temperature schedule
    temp = 0.1
    decay = 0.9995
    while time.time() - start < time_limit:
        # compute shift with max correlation
        conv = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)
        # decide move type
        if random.random() < 0.7:
            move = targeted_move(h, shift)
            if move is None:
                # fallback random
                move = random_move(h)
        else:
            move = random_move(h)
        # compute new cost
        new_c5 = compute_c5(h)
        # SA acceptance
        if new_c5 < cur_c5 or random.random() < np.exp(-(new_c5 - cur_c5) / temp):

            i, j, eps = move
            h[i] += eps
            h[j] -= eps

    # This yields a feasible vector exactly due to careful handling digits.

        # break if changes are negligible
        if np.max(np.abs(actual_inc)) < 1e-12:
            break
    # final clip
    a = np.clip(a, 0, 1)

import time, random, math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(conv.max() * (2.0 / h.size))

def project_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Adjust h ∈ [0,1]^n so that sum(h) = target (leaves other coords unchanged)."""
    a = h.copy()
    deficit = target - a.sum()
    # water‑filling style redistribution
    for _ in range(10):
        if abs(deficit) < 1e-12:
            break
        if deficit > 0:
            can = np.where(a < 1.0 - 1e-12)[0]
        else:
            can = np.where(a > 1e-12)[0]
        if can.size == 0:
            break
        inc = deficit / can.size
        if deficit > 0:
            max_inc = 1.0 - a[can]
            real = np.minimum(inc, max_inc)
        else:
            max_dec = a[can]
            real = np.maximum(inc, -max_dec)
        a[can] += real
        deficit = target - a.sum()
        if np.max(np.abs(real)) < 1e-12:
            break
    a = np.clip(a, 0.0, 1.0)
    # final tiny correction if needed
    diff = target - a.sum()
    if abs(diff) > 1e-9:
        for i in range(a.size):
            if diff > 0 and a[i] < 1.0:
                delta = min(diff, 1.0 - a[i])
                a[i] += delta
                diff -= delta
            elif diff < 0 and a[i] > 0.0:
                delta = min(-diff, a[i])
                a[i] -= delta
                diff += delta
            if abs(diff) < 1e-12:
                break
    return a

def upsample_vector(v: np.ndarray, n_target: int) -> np.ndarray:
    """Linear interpolation of v (defined on [0,2)) onto n_target points."""
    n0 = v.size
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h = np.interp(xt, x0, v)
    h = np.clip(h, 0.0, 1.0)
    return project_sum(h, n_target / 2.0)

def init_h_binary(n: int) -> np.ndarray:
    """Random 0–1 vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    idx = np.random.choice(n, n // 2, replace=False)
    h[idx] = 1.0
    return h

def random_move(h: np.ndarray):
    """Swap a small amount of mass from a positive entry to a negative entry."""
    pos = np.where(h > 1e-12)[0]
    neg = np.where(h < 1.0 - 1e-12)[0]
    if pos.size == 0 or neg.size == 0:
        return None
    i = int(np.random.choice(pos))
    j = int(np.random.choice(neg))
    max_eps = min(h[i], 1.0 - h[j])
    eps = max_eps * np.random.rand()
    h[i] -= eps
    h[j] += eps
    return i, j, eps

def targeted_move(h: np.ndarray, shift: int):
    """Move mass from a high‑contributing pair (i,i+shift) to a low one."""
    n = h.size
    if shift >= 0:
        start, end = 0, n - shift
    else:
        start, end = -shift, n
    if end <= start:
        return None
    idx = np.arange(start, end, dtype=int)
    contrib = h[idx] * (1.0 - h[idx + shift])
    i = int(idx[np.argmax(contrib)])
    rev = (1.0 - h[idx]) * h[idx + shift]
    j = int(idx[np.argmin(rev)])
    max_eps = min(h[i], 1.0 - h[j])
    if max_eps <= 0.0:
        return None
    eps = max_eps * (0.5 * np.random.rand())
    h[i] -= eps
    h[j] += eps
    return i, j, eps

def simulated_annealing(h0: np.ndarray, limit_sec: float):
    """Simple SA that respects the sum constraint (exactly)."""
    h = h0.copy()
    n = h.size
    cur = compute_c5(h)
    best_h = h.copy()
    best = cur
    T = 0.05
    decay = 0.9995
    start = time.time()
    while time.time() - start < limit_sec:
        conv = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(conv))
        shift = max_idx - (n - 1)
        if random.random() < 0.75:
            mv = targeted_move(h, shift)
            if mv is None:
                mv = random_move(h)
        else:
            mv = random_move(h)
        if mv is None:
            continue
        new = compute_c5(h)
        accept = False
        if new < cur:
            accept = True
        else:
            prob = math.exp(-(new - cur) / (T + 1e-12))
            if random.random() < prob:
                accept = True
        if accept:
            cur = new

                best_h = h.copy()
        else:
