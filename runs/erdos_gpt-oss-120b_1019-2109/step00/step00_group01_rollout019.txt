def project_onto_interval_sum(v, target_sum, lower=0.0, upper=1.0, max_iter=100, tol=1e-9):
    # Project onto [lower, upper] and sum constraints.
    # Use bisection on lambda.
    # The projection of v onto the capped simplex is solution of min ||x - v||^2 subject to lower <= x_i <= upper, sum x_i = target_sum.
    # Equivalent to x_i = clip(v_i - lam, lower, upper) and sum_x = target_sum; find lam.
    lo = -1e6
    hi = 1e6
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    x = np.clip(v - (lo + hi) / 2.0, lower, upper)
    # ensure sum close to target, correct rounding errors by adjusting smallest (or any) elements: compute diff and add to those not at bounds
    diff = target_sum - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff uniformly across non-bound elements, but avoid breaking bounds.
        # Simple loop: while diff not negligible:
        idx = np.where((x > lower + 1e-8) & (x < upper - 1e-8))[0]
        if len(idx) > 0:
            # adjust equally
            inc = diff / len(idx)
            x[idx] += inc
            # clip again, recursion might be needed but diff small.
            x = np.clip(x, lower, upper)
    # final clip to ensure bounds
    return x

    n = h.shape[0]
    # term A: Σ_k p_k h_{j+k}
    conv_a = np.convolve(p[::-1], h, mode='full')
    a = conv_a[n-1:n-1+n]

            x[free] += add
            x = np.clip(x, lower, upper)
        else:                           # fallback – adjust a single entry
            idx = 0
            x[idx] = np.clip(x[idx] + diff, lower, upper)
    return x


def compute_c5(h):
    """Exact evaluation used by the harness."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * dx)


def compute_gradient(h, p, dx):
    """∇_h (smooth max) using two convolutions."""
    n = h.size
    # Σ_k p_k h_{j+k}
    conv1 = np.convolve(p[::-1], h, mode='full')
    term1 = conv1[n - 1:n - 1 + n]
    # Σ_k p_k h_{j-k}
    conv2 = np.convolve(p, h, mode='full')
    term2 = conv2[n - 1:n - 1 + n]
    return -dx * (term1 + term2)


def resize_initial(old_h, new_n):
    """Linear interpolation of a given construction to a new size."""
    old_n = len(old_h)
    if old_n == new_n:
        h = np.array(old_h, dtype=float)
    else:
        x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
        h = np.interp(x_new, x_old, old_h)
    h = np.clip(h, 0.0, 1.0)
    h = project_onto_interval_sum(h, target_sum=new_n / 2)
    return h


def gradient_optimize(h0, max_iters=2000, max_time=25.0,
                     alpha_start=20.0, alpha_end=200.0,
                     step_size=5.0):
    """Smoothed‑max gradient descent with back‑tracking line search."""
    n = h0.size
    dx = 2.0 / n
    h = h0.copy()
    best_h, best_val = h.copy(), compute_c5(h)
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > max_time:
            break
        autocorr = np.correlate(h, h, mode='full')
        f = 1.0 - dx * autocorr                # f_k = overlap for each shift
        max_f = np.max(f)
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)
        w = np.exp(alpha * (f - max_f))        # stable soft‑max weights
        p = w / w.sum()
        grad = compute_gradient(h, p, dx)

        # back‑tracking line search
        step = step_size
        cur_val = compute_c5(h)
        while True:
            h_new = h - step * grad
            h_new = project_onto_interval_sum(h_new, target_sum=n / 2)
            new_val = compute_c5(h_new)
            if new_val <= cur_val or step < 1e-8:
                break
            step *= 0.5

        if new_val < cur_val:
            h, cur_val = h_new, new_val
        if cur_val < best_val:
            best_h, best_val = h.copy(), cur_val
    return best_h, best_val


def hill_climb(h0, rng, max_iters=5000, max_time=20.0, step_frac=0.2):
    """Simple pairwise swap/adjust local search."""
    n = h0.size
    h = h0.copy()
    cur_val = compute_c5(h)
    start = time.time()
    for _ in range(max_iters):
        if time.time() - start > max_time:
            break
        i, j = rng.choice(n, size=2, replace=False)
        max_up = 1.0 - h[i]
        max_down = h[j]
        if max_up <= 0 or max_down <= 0:
            continue
        max_move = min(max_up, max_down)
        delta = (rng.rand() - 0.5) * 2.0 * step_frac * max_move
        if abs(delta) < 1e-12:
            continue
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        new_val = compute_c5(h_new)
        if new_val < cur_val:
            h, cur_val = h_new, new_val
    return h, cur_val


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()
    best_val = np.inf
    best_h = None
    best_n = None

    # problem sizes – increase gradually, stop when budget runs low
    n_candidates = [80, 120, 160, 200, 240, 300]

    for n in n_candidates:
        if time.time() - start > budget_s * 0.95:
            break

        # build a few starting points
        starts = []
        if 'initial_h_values' in globals():
            starts.append(resize_initial(initial_h_values, n))
        for _ in range(2):
            rand_h = rng.rand(n)
            rand_h = project_onto_interval_sum(rand_h, target_sum=n / 2)
            starts.append(rand_h)

        for h0 in starts:
            time_left = budget_s - (time.time() - start)
            if time_left <= 0:
                break

            # split remaining time between the two phases
            t_grad = min(time_left * 0.6, 30.0)
            t_hc   = min(time_left * 0.4, 20.0)

            h_opt, val_opt = gradient_optimize(h0,
                                               max_time=t_grad)
            h_opt, val_opt = hill_climb(h_opt, rng,
                                        max_time=t_hc)

            if val_opt < best_val:
                best_val, best_h, best_n = val_opt, h_opt.copy(), n

            # early stop if record already beaten
            if best_val < 0.3808:
                break
        if best_val < 0.3808:
            break

    # safety fallback
    if best_h is None:
        best_n = 100
        best_h = np.full(best_n, 0.5)
        best_val = compute_c5(best_h)

    # final projection just in case of rounding drift
    best_h = project_onto_interval_sum(best_h, target_sum=best_n / 2)

    return best_h.tolist(), float(best_val), best_n
