import time
import numpy as np

def _project_to_sum_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of v onto the box [0,1]^n with a prescribed sum.
    Solves   min ||u - v||^2  s.t. 0 <= u_i <= 1, sum(u) = target_sum.
    """
    lo = np.min(v) - 1.0          # low enough so that all v_i - lo > 1
    hi = np.max(v) + 1.0          # high enough so that all v_i - hi < 0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        u = np.clip(v - mid, 0.0, 1.0)
        s = u.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    u = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target_sum - u.sum()
    if abs(diff) > 1e-12:
        free = (u > 0.0) & (u < 1.0)
        cnt = free.sum()
        if cnt > 0:
            u[free] += diff / cnt
            u = np.clip(u, 0.0, 1.0)
    return u


def _c5_value(h: np.ndarray) -> float:
    """Return the objective C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) = sum_i h_i - sum_i h_i*h_{i+k}
    # sum_i h_i * dx == 1  (guaranteed by constraints)
    # therefore C5 = max( 1 - dx * autocorr )
    autocorr = np.correlate(h, h, mode="full")
    O = 1.0 - autocorr * dx
    return O.max()


def _gradient(h: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max smoothed gradient of the max‑overlap objective."""
    n = h.shape[0]
    dx = 2.0 / n
    # autocorrelation
    ac = np.correlate(h, h, mode="full")
    O = 1.0 - ac * dx                     # overlap values for all integer shifts
    # soft‑max weights (focus on the largest overlaps)
    m = O.max()
    w = np.exp(alpha * (O - m))
    w /= w.sum()                         # sum_i w_i == 1
    # compute g_i = Σ_s w_s (h_{i+s}+h_{i-s})
    g = np.zeros_like(h)
    off = n - 1
    # non‑negative shifts (including zero)
    for s in range(0, n):
        weight = w[off + s]
        if weight == 0.0:
            continue
        if s == 0:
            # contribution 2 * weight * h_i
            g += 2.0 * weight * h
        else:
            # h_{i+s} term
            g[: n - s] += weight * h[s:]
            # h_{i-s} term
            g[s:] += weight * h[: n - s]
    # negative shifts
    for s in range(1, n):
        weight = w[off - s]
        if weight == 0.0:
            continue
        # shift = -s  (use same pattern as above)
        g[s:] += weight * h[: n - s]
        g[: n - s] += weight * h[s:]
    # gradient of the smoothed max‑overlap
    grad = 1.0 - g
    return grad


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize the step function h on a uniform grid to lower the Erdős
    C5 bound.  Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # -------------------------------------------------------------
    # Choose grid size (smaller is faster, but enough points to be expressive)
    # -------------------------------------------------------------
    n = 200                         # 200 points -> dx = 0.01
    target_sum = n / 2.0            # because ∫h = 1  and dx = 2/n
    dx = 2.0 / n

    # -------------------------------------------------------------
    # Initial vector
    # -------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=float).copy()
        if h.shape[0] != n:
            # resize by linear interpolation if needed
            x_old = np.linspace(0.0, 2.0, h.shape[0], endpoint=False) + dx / 2
            x_new = np.linspace(0.0, 2.0, n, endpoint=False) + dx / 2
            h = np.interp(x_new, x_old, h)
        h = np.clip(h, 0.0, 1.0)
        h = _project_to_sum_bounds(h, target_sum)
    else:
        # start from a slightly perturbed uniform function (h ≡ 0.5)
        h = 0.5 + 0.05 * (np.random.rand(n) - 0.5)
        h = np.clip(h, 0.0, 1.0)
        h = _project_to_sum_bounds(h, target_sum)

    best_h = h.copy()
    best_val = _c5_value(best_h)

    # -------------------------------------------------------------
    # Optimization parameters
    # -------------------------------------------------------------
    alpha = 5.0          # soft‑max concentration
    step = 0.05          # learning‑rate‑like step size
    stagnation = 0       # counter for iterations without improvement
    max_stagn = 2000

    while time.time() - start_time < budget_s * 0.95:
        # current objective
        cur_val = _c5_value(h)

        # keep the best solution seen
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1

        # compute gradient of the smoothed max‑overlap
        grad = _gradient(h, alpha)

        # tentative update
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _project_to_sum_bounds(h_new, target_sum)

        new_val = _c5_value(h_new)

        # simple acceptance rule (prefer improvement)
        if new_val < cur_val or np.random.rand() < 0.001:
            h = h_new
            # modestly increase step if we are making progress
            step = min(step * 1.02, 0.5)
        else:
            # otherwise shrink step
            step *= 0.5

        # adapt the soft‑max temperature slowly
        alpha = min(alpha * 1.005, 200.0)

        # occasional random restart if stuck
        if stagnation > max_stagn:
            # re‑initialize with a fresh perturbation around the current best
            h = best_h + 0.03 * (np.random.rand(n) - 0.5)
            h = np.clip(h, 0.0, 1.0)
            h = _project_to_sum_bounds(h, target_sum)
            stagnation = 0

        # safety for numerical drift
        if abs(h.sum() - target_sum) > 1e-9:
            h = _project_to_sum_bounds(h, target_sum)

    # final evaluation of the returned vector
    final_val = _c5_value(best_h)

    return best_h, final_val, n
