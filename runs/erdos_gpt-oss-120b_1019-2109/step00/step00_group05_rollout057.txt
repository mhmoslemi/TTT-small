def smooth_h_vec(h, lag, alpha):
    n = len(h)
    if lag >= 0:
        i = np.arange(n - lag)
        hi = h[i]; hj = h[i+lag]
        diff = hi - hj
        mask = diff > 0
        delta = np.zeros_like(diff)
        delta[mask] = alpha * diff[mask]
        # clip by constraints
        delta = np.minimum(delta, hi)
        delta = np.minimum(delta, 1 - hj)
        # Ensure delta only where mask
        delta = np.where(mask, delta, 0.0)
        # apply
        h[i] -= delta
        h[i+lag] += delta
    else:
        l = -lag
        i = np.arange(l, n)
        hi = h[i]; hj = h[i-l]
        diff = hi - hj
        mask = diff > 0
        delta = np.zeros_like(diff)
        delta[mask] = alpha * diff[mask]
        delta = np.minimum(delta, hi)
        delta = np.minimum(delta, 1 - hj)
        delta = np.where(mask, delta, 0.0)
        h[i] -= delta
        h[i-l] += delta
    return h

import numpy as np, random, time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1 - h, mode='full')
    max_val = conv.max() * dx
    return max_val, conv

def smooth_h_vec(h, lag, alpha):
    # modifies in place
    n = len(h)
    if lag >= 0:
        i = np.arange(n - lag)
        hi = h[i]
        hj = h[i + lag]
        diff = hi - hj
        mask = diff > 0
        delta = np.zeros_like(diff)
        delta[mask] = alpha * diff[mask]
        # ensure not exceed bounds
        delta = np.minimum(delta, hi)
        delta = np.minimum(delta, 1 - hj)
        # only apply where mask
        delta = np.where(mask, delta, 0.0)
        h[i] -= delta
        h[i + lag] += delta
    else:
        l = -lag
        i = np.arange(l, n)
        hi = h[i]
        hj = h[i - l]
        diff = hi - hj
        mask = diff > 0
        delta = np.zeros_like(diff)
        delta[mask] = alpha * diff[mask]
        delta = np.minimum(delta, hi)
        delta = np.minimum(delta, 1 - hj)
        delta = np.where(mask, delta, 0.0)
        h[i] -= delta
        h[i - l] += delta
    return h

def repair_sum(h, target_sum):
    # Ensure sum equals target, clamp values to [0,1]
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target_sum - cur
    if abs(diff) < 1e-12:
        return h
    # find indices not at bounds
    free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
    if free.size == 0:
        free = np.arange(len(h))
    adjust = diff / free.size
    h[free] += adjust
    # clip again
    h = np.clip(h, 0.0, 1.0)
    # possibly still off due to clipping; fix with small correction to first index
    cur = h.sum()
    if abs(cur - target_sum) > 1e-9:
        i = np.argmax(h) if cur < target_sum else np.argmin(h)
        # adjust within bounds
        max_inc = 1 - h[i] if cur < target_sum else h[i]
        needed = target_sum - cur
        delta = np.clip(needed, -max_inc, max_inc)  # but we need sign
        if cur < target_sum:
            delta = min(max_inc, needed)
        else:
            delta = -min(max_inc, -needed)
        h[i] += delta
    return h

# -*- coding: utf-8 -*-

import time
import random
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def compute_c5(h):
    """
    Return (C5 value, full correlation array) for a step function h.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = corr.max() * dx
    return max_val, corr


def smooth_h_vec(h, lag, alpha):
    """
    One smoothing step for a given lag.
    Moves a fraction alpha of the positive differences hi - h_{i+lag}
    from the larger entry to the smaller one.
    Operates in-place and returns h.
    """
    n = h.shape[0]
    if lag >= 0:
        i = np.arange(n - lag)
        hi = h[i]
        hj = h[i + lag]
        diff = hi - hj
        mask = diff > 0
        delta = np.zeros_like(diff)
        delta[mask] = alpha * diff[mask]
        # keep within [0,1]
        delta = np.minimum(delta, hi)
        delta = np.minimum(delta, 1.0 - hj)
        delta = np.where(mask, delta, 0.0)
        h[i] -= delta
        h[i + lag] += delta
    else:
        l = -lag
        i = np.arange(l, n)
        hi = h[i]
        hj = h[i - l]
        diff = hi - hj
        mask = diff > 0
        delta = np.zeros_like(diff)
        delta[mask] = alpha * diff[mask]
        delta = np.minimum(delta, hi)
        delta = np.minimum(delta, 1.0 - hj)
        delta = np.where(mask, delta, 0.0)
        h[i] -= delta
        h[i - l] += delta
    return h


def repair_sum(h, target):
    """
    Project h onto the feasible set:
      0 <= h_i <= 1   and   sum(h) == target
    Small violations are corrected by a simple redistribution.
    """
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h

    # indices that can still move
    free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
    if free.size == 0:
        free = np.arange(h.size)

    # distribute the deficit/excess uniformly
    h[free] += diff / free.size
    h = np.clip(h, 0.0, 1.0)

    # final tiny correction (if needed)
    cur = h.sum()
    if abs(cur - target) > 1e-8:
        # adjust a single safe index
        if cur < target:
            idx = np.argmin(h)            # most room to increase
            inc = min(target - cur, 1.0 - h[idx])
            h[idx] += inc
        else:
            idx = np.argmax(h)            # most room to decrease
            dec = min(cur - target, h[idx])
            h[idx] -= dec
    return h


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimise a step function h on [0,2] that minimises
        C5 = max_x  ∫ h(x) (1 - h(x+x)) dx
    Returns (list_of_h, C5_value, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ---- initial vector -------------------------------------------------
    try:
        # `initial_h_values` is injected by the harness when available.
        h = np.array(initial_h_values, dtype=np.float64)
        n = h.shape[0]
    except NameError:
        # fallback: construct a simple binary pattern (half ones)
        n = 100
        h = np.zeros(n, dtype=np.float64)
        ones_idx = random.sample(range(n), n // 2)
        h[ones_idx] = 1.0

    target_sum = n / 2.0
    h = repair_sum(h, target_sum)

    best_h = np.copy(h)
    best_c5, _ = compute_c5(best_h)
    current_h = np.copy(best_h)

    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # a tiny safety margin

    iteration = 0
    while time.time() < deadline:
        cur_c5, conv = compute_c5(current_h)

        # keep the global best
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = np.copy(current_h)

        # worst lag (the one attaining the current max)
        idx_max = int(np.argmax(conv))
        lag = idx_max - (n - 1)

        # annealing‑style step size
        elapsed = time.time() - start_time
        frac = elapsed / budget_s
        alpha = max(0.001, 0.45 * (1.0 - frac))

        # candidate from a deterministic smoothing move
        cand = np.copy(current_h)
        cand = smooth_h_vec(cand, lag, alpha)

        # occasional random perturbation to escape plateaus
        if random.random() < 0.07:
            i, j = random.sample(range(n), 2)
            max_move = min(cand[i], 1.0 - cand[j])
            if max_move > 0:
                delta = random.random() * max_move
                cand[i] -= delta
                cand[j] += delta

        cand = np.clip(cand, 0.0, 1.0)
        cand = repair_sum(cand, target_sum)

        cand_c5, _ = compute_c5(cand)

        # simple acceptance rule
        if cand_c5 < cur_c5:
            current_h = cand
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = np.copy(cand)
        else:
            # occasional uphill move for exploration
            if random.random() < 0.015:
                current_h = cand

        iteration += 1

    # final clean‑up
    best_h = repair_sum(best_h, target_sum)
    final_c5, _ = compute_c5(best_h)

    return best_h.tolist(), float(final_c5), n
