def generate_shape(n, l, w, target_sum):
    centre = (n - 1) / 2.0
    i = np.arange(n)
    d = np.abs(i - centre)
    # compute unscaled shape values
    f = np.zeros(n)
    # plateau
    mask_plateau = d <= l
    f[mask_plateau] = 1.0
    # tails
    mask_tail = (d > l) & (d <= l + w)
    f[mask_tail] = (l + w - d[mask_tail]) / w
    factor = f.sum()
    if factor < target_sum - 1e-12:
        return None
    a = target_sum / factor
    if a > 1.0:
        return None
    h = a * f
    return h
```

import numpy as np, time, random, math

def _c5_of(h):
    """Return C5 for a step function given its sampled values."""
    n = h.size
    S = h.sum()
    # autocorrelation (linear, zero-padded)
    ac = np.correlate(h, h, mode="full")
    min_corr = ac.min()
    return (2.0 / n) * (S - min_corr)


def _worst_shift_and_gradient(h):
    """Find shift giving minimal autocorrelation and its gradient."""
    n = h.size
    ac = np.correlate(h, h, mode="full")
    idx = ac.argmin()
    k = idx - (n - 1)                     # integer shift (may be negative)

    # gradient of the minimal autocorrelation term w.r.t. each entry of h
    g = np.zeros(n)
    # forward part: contribution from h_i * h_{i+k}
    j = np.arange(n) + k
    mask_f = (j >= 0) & (j < n)
    g[mask_f] += h[j[mask_f]]
    # backward part: contribution from h_{i-k} * h_i
    i = np.arange(n) - k
    mask_b = (i >= 0) & (i < n)
    g[mask_b] += h[i[mask_b]]
    return k, g


def _generate_plateau_shape(n, l, w, target):
    """
    Symmetric shape: plateau of half‑width `l` (inclusive) at height 1,
    linear tails of half‑width `w` down to 0.
    Returns the shape scaled to have total sum = target, or None if impossible.
    """
    centre = (n - 1) / 2.0
    i = np.arange(n)
    d = np.abs(i - centre)

    f = np.zeros(n)
    # plateau
    f[d <= l] = 1.0
    # linear tails
    mask_tail = (d > l) & (d <= l + w)
    f[mask_tail] = (l + w - d[mask_tail]) / w

    factor = f.sum()
    if factor < target - 1e-12:
        return None
    a = target / factor
    if a > 1.0 + 1e-12:
        return None
    return a * f
