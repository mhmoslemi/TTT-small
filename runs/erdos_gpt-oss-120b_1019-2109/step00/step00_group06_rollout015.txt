        maxg = g.max()
        exp_terms = np.exp(gamma * (g - maxg))   # stability
        w = exp_terms / exp_terms.sum()          # weights over all shifts

        # gradient of the smooth objective
        w_rev = w[::-1]
        term1 = np.convolve(w_rev, (1 - h), mode='full')[n - 1:n - 1 + n]
        term2 = np.convolve(w, h, mode='full')[n - 1:n - 1 + n]
        grad = dx * (term1 - term2)       # shape (n,)

        # ---- step & projection ----
        step = step0 / math.sqrt(it + 1.0)
        h_new = h - step * grad
        h_new = _project_to_sum_box(h_new, target_sum)

        # occasional random jitter to escape shallow minima
        if (it & 511) == 0 and it > 0:
            jitter = rng.normal(scale=0.02, size=n)
            h_new = _project_to_sum_box(h_new + jitter, target_sum)

        h = h_new

        # restart from the best if stagnation is detected
        if it - last_improve > 3000:
            h = best_h + rng.normal(scale=0.05, size=n)
            h = _project_to_sum_box(h, target_sum)
            last_improve = it

    # -------------------------------------------------------------
    # 5. Final evaluation (using the provided helper when available)
    # -------------------------------------------------------------
    try:
        final_val = evaluate_erdos_solution(best_h)
    except Exception:
        final_corr = np.correlate(best_h, 1 - best_h, mode='full')
        final_val = final_corr.max() * dx

    # -------------------------------------------------------------
    # 6. Tiny correction to respect the integral exactly
    # -------------------------------------------------------------
    diff = best_h.sum() - target_sum
    if abs(diff) > 1e-9:
        free = np.where((best_h > 1e-8) & (best_h < 1.0 - 1e-8))[0]
        if free.size:
            best_h[free[0]] -= diff
            best_h = np.clip(best_h, 0.0, 1.0)
        else:
            best_h = _project_to_sum_box(best_h, target_sum)

    return best_h, float(final_val), n
