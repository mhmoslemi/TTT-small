def project_onto_box_sum(v, target_sum):
    # Clip after shifting
    lo = -1e5
    hi = 1e5
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-12:
            break
    # final projection
    proj = np.clip(v - (lo+hi)/2, 0, 1)
    # Adjust slight difference
    diff = proj.sum() - target_sum
    if abs(diff) > 1e-8:
        # fix by adjusting a single element within bounds
        # Find an index not at bound
        idx = np.where((proj > 0) & (proj < 1))[0]
        if idx.size > 0:
            i = idx[0]
            proj[i] -= diff
            proj[i] = np.clip(proj[i], 0, 1)
        else:
            # all at bounds; fallback: distribute diff equally (might be small)
            proj += diff / len(proj)
            proj = np.clip(proj, 0, 1)
    return proj

def project_onto_box_sum(v, target_sum):
    # water-filling via binary search
    lo = -1e6
    hi = 1e6
    for _ in range(100):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
        if hi - lo < 1e-12:
            break
    proj = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Adjust small rounding error
    diff = proj.sum() - target_sum
    if abs(diff) > 1e-8:
        # distribute diff proportionally to entries not at bounds
        free = (proj > 0) & (proj < 1)
        count = free.sum()
        if count > 0:
            proj[free] -= diff / count
            proj = np.clip(proj, 0.0, 1.0)
        else:
            # fallback: adjust all
            proj -= diff / len(proj)
            proj = np.clip(proj, 0.0, 1.0)
    return proj

import time

def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _project_onto_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the feasible set:
        0 <= x_i <= 1  and  sum x_i = target_sum.
    Solved by a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1e6, 1e6
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for rounding errors
    diff = proj.sum() - target_sum
    if abs(diff) > 1e-8:
        free = (proj > 0.0) & (proj < 1.0)
        cnt = free.sum()
        if cnt:
            proj[free] -= diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
        else:
            proj -= diff / len(proj)
            proj = np.clip(proj, 0.0, 1.0)
    return proj
