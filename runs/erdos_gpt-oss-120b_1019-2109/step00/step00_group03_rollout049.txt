    binary_deadline = start + budget_s * 0.60
    T = 0.12                # initial temperature
    decay = 0.9997

    while time.time() < binary_deadline:
        # correlation to locate the worst lag
        cor = np.correlate(cur_h, 1 - cur_h, mode='full') * dx
        lag = int(np.argmax(cor) - (n - 1))

        # mismatched 1–0 and 0–1 positions for that lag
        idx = np.arange(n)
        if lag >= 0:
            i_cand = idx[(cur_h == 1) & (idx + lag < n) & (cur_h[idx + lag] == 0)]
            j_cand = idx[(cur_h == 0) & (idx + lag < n) & (cur_h[idx + lag] == 1)]
        else:
            L = -lag
            i_cand = idx[(cur_h == 1) & (idx - L >= 0) & (cur_h[idx - L] == 0)]
            j_cand = idx[(cur_h == 0) & (idx - L >= 0) & (cur_h[idx - L] == 1)]

        if i_cand.size and j_cand.size:
            i = i_cand[np.random.randint(i_cand.size)]
            j = j_cand[np.random.randint(j_cand.size)]
        else:
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            i = np.random.choice(ones)
            j = np.random.choice(zeros)

        # propose swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_cost = max_overlap(cur_h)

        delta = new_cost - cur_cost
        if delta <= 0 or (T > 1e-12 and np.random.rand() < math.exp(-delta / T)):
            cur_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        T = max(T * decay, 1e-12)

    # ------------------------------------------------------------------
    # 3) continuous refinement (gradient descent on the worst lag)
    # ------------------------------------------------------------------
    # start from the best binary vector found
    h_cont = best_h.astype(np.float64).copy()
    eta = 0.25                     # initial step size
    eta_decay = 0.9995
    cont_deadline = start + budget_s - 0.5   # leave a small safety margin

    while time.time() < cont_deadline:
        # full overlap vector
        ov = np.correlate(h_cont, 1 - h_cont, mode='full') * dx
        max_idx = int(np.argmax(ov))
        lag = max_idx - (n - 1)

        # gradient of the overlap at this lag (sub‑gradient of the max)
        grad = np.zeros(n, dtype=np.float64)
        if lag >= 0:
            # (1 - h[i+lag]) part
            idx1 = np.arange(n - lag)
            grad[idx1] += 1.0 - h_cont[idx1 + lag]
            # -h[i-lag] part
            idx2 = np.arange(lag, n)
            grad[idx2] -= h_cont[idx2 - lag]
        else:
            L = -lag
            idx1 = np.arange(L, n)                # i such that i-L >=0
            grad[idx1] += 1.0 - h_cont[idx1 - L]
            idx2 = np.arange(0, n - L)            # i such that i+L < n
            grad[idx2] -= h_cont[idx2 + L]

        # descent step and projection back onto feasibility set
        h_next = h_cont - eta * grad
        h_next = _project_onto_box_sum(h_next, 0.0, 1.0, target_sum)

        new_cost = max_overlap(h_next)

        if new_cost < best_cost:
            best_cost = new_cost
            best_h = h_next.copy()

        h_cont = h_next
        eta = max(eta * eta_decay, 1e-5)

    # ------------------------------------------------------------------
    # 4) final output
    # ------------------------------------------------------------------
    h_vals = best_h.astype(float).tolist()
    return (h_vals, float(best_cost), n)
