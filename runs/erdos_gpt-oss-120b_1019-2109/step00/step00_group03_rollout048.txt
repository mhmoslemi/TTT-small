def compute_c5(h):
    g = 1 - h
    L = len(h)
    # Perform convolution via FFT.
    m = 2*L - 1
    Nfft = 1 << (m-1).bit_length() # next power of 2
    H = np.fft.rfft(h, n=Nfft)
    G = np.fft.rfft(g, n=Nfft)
    conv = np.fft.irfft(H*G, n=Nfft)
    # conv length Nfft, but needed entries 0..2L-2, linear convolution.
    # We need conv[:2*L-1]
    linear = conv[:2*L-1]
    # Multiply by dx
    return np.max(linear) * (2.0 / L)  # dx = 2/L

def compute_c5(h):
    # compute correlation of h with 1-h
    g = 1 - h
    L = h.shape[0]

import time, math
import numpy as np

def _corr_and_cost(vec, n):
    """Return (full correlation of vec and 1-vec, C5 value)."""
    g = 1 - vec
    corr = np.correlate(vec, g, mode='full')
    c5 = corr.max() * (2.0 / n)   # dx = 2/n
    return corr, c5

def _fix_weight(vec, rng):
    """Force a binary vector to have exactly len(vec)//2 ones."""
    n = vec.size
    target = n // 2
    cur = int(vec.sum())
    if cur == target:
        return vec
    if cur > target:                     # too many ones -> turn some to zero
        ones = np.flatnonzero(vec)
        to_flip = cur - target
        idx = rng.choice(ones, size=to_flip, replace=False)
        vec[idx] = 0
    else:                                 # too few ones -> turn some zeros to one
        zeros = np.flatnonzero(vec == 0)
        to_flip = target - cur
        idx = rng.choice(zeros, size=to_flip, replace=False)
        vec[idx] = 1
    return vec

            i_candidates = i_range[mism]

        if i_candidates.size:
            i = rng.choice(i_candidates)
            zero_idxs = np.flatnonzero(h_cur == 0)
            # search a zero that does NOT create a new mismatch at shift k
            for _ in range(12):
                j = rng.choice(zero_idxs)
                jj = j + k
                if 0 <= jj < n and h_cur[jj] == 0:
                    # would create a new mismatch → skip
                    continue
                neighbor = h_cur.copy()
                neighbor[i] = 0
                neighbor[j] = 1
                break

        # --------------------------------------------------------------
        # If no good targeted move, perform a random balanced swap
        # --------------------------------------------------------------
        if neighbor is None:
            ones = np.flatnonzero(h_cur)
            zeros = np.flatnonzero(h_cur == 0)
            i = rng.choice(ones)
            j = rng.choice(zeros)
            neighbor = h_cur.copy()
            neighbor[i] = 0
            neighbor[j] = 1

        # --------------------------------------------------------------
        # Evaluate the candidate
        # --------------------------------------------------------------
        new_corr, new_c = _corr_and_cost(neighbor, n)

        # --------------------------------------------------------------
        # Acceptance step (simulated annealing)
        # --------------------------------------------------------------
        accept = False
        if new_c < cur_c:
            accept = True
        else:
            prob = math.exp(-(new_c - cur_c) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            h_cur = neighbor
            cur_corr, cur_c = new_corr, new_c
            if new_c < best_c:
                best_c = new_c
                best_h = neighbor.copy()
                last_improve = it

        # --------------------------------------------------------------
        # Temperature cooling
        # --------------------------------------------------------------
        it += 1
        if it % 2000 == 0:
            T = max(T * 0.92, 1e-6)

        # --------------------------------------------------------------
        # Restart if we have stagnated for a long time
        # --------------------------------------------------------------
        if it - last_improve > 80000:
            # fresh random start
            h_cur = np.zeros(n, dtype=np.int8)
            h_cur[rng.choice(n, size=n // 2, replace=False)] = 1
            cur_corr, cur_c = _corr_and_cost(h_cur, n)
            last_improve = it

    # ------------------------------------------------------------------
    # Return the best found construction
    # ------------------------------------------------------------------
    return best_h.astype(float).tolist(), float(best_c), n
