def project_onto_box_and_sum(v, total):
    # v is unconstrained after gradient step
    # we need to find x = clip(v - tau, 0, 1) such that sum(x) = total

        x = np.clip(v - tau, 0, 1)
        s = x.sum()
        if s > total:
            lo = tau
        else:
            hi = tau
    return x

    """project v onto [0,1]^n with sum(v)=total using bisection on a shift."""
    lo = np.min(v) - 1.0
    hi = np.max(v)

        proj = np.clip(v - mid, 0.0, 1.0)

    return _project_box_sum(v, n / 2.0)

def _anneal(start_h, max_iters, budget, start_time):
    """simple simulated‑annealing on two‑index mass transfers."""
    n = start_h.shape[0]
    total = n / 2.0
    cur = start_h.copy()
    cur_val = _c5_val(cur)
    best = cur.copy()
    best_val = cur_val
    init_T = 0.02
    for it in range(max_iters):
        if time.time() - start_time > budget * 0.95:
            break
        # temperature decreasing
        T = init_T * (1.0 - it / max_iters)
        # propose a transfer between two indices
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # allowable delta range keeping both entries in [0,1]
        delta_pos = min(1.0 - cur[i], cur[j])
        delta_neg = min(cur[i], 1.0 - cur[j])
        if delta_pos + delta_neg == 0.0:
            continue
        # large moves early, small moves later
        scale = 0.1 + 0.9 * (1.0 - it / max_iters)
        delta = (np.random.rand() * (delta_pos + delta_neg) - delta_neg) * scale
        if abs(delta) < 1e-12:
            continue
        trial = cur.copy()
        trial[i] += delta
        trial[j] -= delta
        trial_val = _c5_val(trial)
        dE = trial_val - cur_val
        if dE < 0.0 or np.random.rand() < math.exp(-dE / (T + 1e-12)):
            cur, cur_val = trial, trial_val
            if trial_val < best_val:
                best, best_val = trial.copy(), trial_val
        # occasional pure swap to mix large values
        if it % 500 == 0:
            a, b = np.random.randint(n, size=2)
            if a != b:
                trial2 = cur.copy()
                trial2[a], trial2[b] = trial2[b], trial2[a]
                trial2_val = _c5_val(trial2)
                dE2 = trial2_val - cur_val
                if dE2 < 0.0 or np.random.rand() < math.exp(-dE2 / (T + 1e-12)):
                    cur, cur_val = trial2, trial2_val
                    if trial2_val < best_val:
                        best, best_val = trial2.copy(), trial2_val
    return best, best_val

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    # collect any supplied initial vector
    init_vec = globals().get('initial_h_values')
    if isinstance(init_vec, (list, tuple)):
        init_vec = np.asarray(init_vec, dtype=float)

    # candidate sample sizes (must be even)
    cand_n = [45, 60, 72, 84, 96, 108, 120, 144, 168, 192]
    best_h, best_val, best_n = None, float('inf'), None

    for n in cand_n:
        if time.time() - start_time > budget_s * 0.9:
            break
        target_sum = n / 2.0
        bases = []

        # use interpolated version of the provided construction if present
        if init_vec is not None:
            if init_vec.shape[0] == n:
                bases.append(init_vec.copy())
            else:
                old_x = np.linspace(0.0, 2.0, init_vec.shape[0], endpoint=False)
                new_x = np.linspace(0.0, 2.0, n, endpoint=False)
                interp = np.interp(new_x, old_x, init_vec)
                interp = np.clip(interp, 0.0, 1.0)
                bases.append(_project_box_sum(interp, target_sum))
        # a completely random start
        bases.append(_random_feasible(n))

        # anneal each base, keep the best for this n
        for base in bases:
            # number of annealing steps roughly scaled with n
            iters = int(4e4) if n <= 120 else int(3e4)
            h_opt, val_opt = _anneal(base, iters, budget_s, start_time)
            if val_opt < best_val:
                best_h, best_val, best_n = h_opt.copy(), val_opt, n

    # final safety projection and value recomputation
    if best_h is None:
        # fallback uniform step function
        best_n = 60
        best_h = np.full(best_n, 0.5)
        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_val = _c5_val(best_h)
    else:
        best_h = _project_box_sum(best_h, best_n / 2.0)
        best_val = _c5_val(best_h)

    return best_h, best_val, best_n
