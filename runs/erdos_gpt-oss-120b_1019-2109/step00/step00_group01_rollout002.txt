def propose_move(h, max_transfer=0.1):
    n = len(h)
    i, j = np.random.choice(n, size=2, replace=False)
    # random sign
    direction = np.random.choice([-1, 1])
    if direction == 1:
        # transfer from i to j
        max_delta = min(h[i], 1 - h[j])
    else:
        # transfer from j to i
        max_delta = min(h[j], 1 - h[i])
    # limit to max_transfer
    max_delta = min(max_delta, max_transfer)
    if max_delta <= 0:
        return None  # no move possible
    delta = np.random.rand() * max_delta
    # apply
    new_h = h.copy()
    if direction == 1:
        new_h[i] -= delta
        new_h[j] += delta
    else:
        new_h[i] += delta
        new_h[j] -= delta
    return new_h

def random_feasible(n):
    target = n/2
    # start with zeros
    h = np.zeros(n)
    # assign integer part
    integer_part = int(np.floor(target))
    remainder = target - integer_part
    # set integer_part indices randomly to 1
    idx_ones = np.random.choice(n, size=integer_part, replace=False)
    h[idx_ones] = 1.0
    if remainder > 1e-12:
        # pick a remaining index not already set to 1
        remaining_idx = np.random.choice(np.setdiff1d(np.arange(n), idx_ones))
        h[remaining_idx] = remainder
    return h

def project_to_sum_cap(v, target):
    # project onto the simplex with upper bound 1.
    # Approach: water-filling algorithm (like projection onto capped simplex).
    # We'll implement following algorithm:
    # 1. Clip v to [0,1].
    # 2. If sum(v) == target: return v.
    # 3. Find lambda such that sum(clip(v - lambda, 0, 1)) = target.
    # This is similar to projection onto simplex with upper bound.
    # Solve via bisection because function is monotonic decreasing in lambda.
    # We'll implement bisection between low = -1 (so v - low = v+1 >=0) and high = 1 (so v - high <=0).
    # Or compute more precise bounds: lower bound = min(v)-1 etc.
    # Since N moderate, we can do binary search for lambda with tolerance 1e-9.
    v = np.clip(v, 0.0, 1.0)
    # Check immediate sum
    current_sum = v.sum()
    if np.isclose(current_sum, target):
        return v
    # Use bisection for lambda
    # For lambda low, high such that sum(clip(v - lambda, 0, 1)) > target for low, < target for high.
    # At lambda = -1, v - (-1) = v+1 > 1 so clip to 1, sum = n*1 = n > target (since target <= n)
    # At lambda = 2, v - 2 <= -1 => clip to 0 => sum = 0 < target.
    lo = -1.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Due to rounding we might be slightly off target; adjust by adding small delta uniformly to non-bound variables
    diff = target - proj.sum()
    if abs(diff) > 1e-10:
        # distribute diff among free entries
        free = (proj > 0) & (proj < 1)
        if free.any():
            proj[free] += diff / free.sum()
            # ensure still within bounds
            proj = np.clip(proj, 0.0, 1.0)
    # final adjustment ensure sum matches as close as possible
    return proj

def propose_move(h, max_transfer=0.5):
    n = len(h)
    # Choose two distinct indices
    i, j = np.random.choice(n, size=2, replace=False)
    # random direction: if transfer from i to j or opposite
    # compute max delta respecting bounds and also not exceed max_transfer
    # Transfer from i (source) to j (dest)
    # We'll use random direction choice.
    direction = np.random.choice([0,1])  # 0: i->j, 1: j->i
    if direction == 0:
        source = i; dest = j
    else:
        source = j; dest = i
    max_delta = min(h[source], 1 - h[dest])
    max_delta = min(max_delta, max_transfer)
    if max_delta <= 0:
        return None
    delta = np.random.rand() * max_delta
    new_h = h.copy()
    new_h[source] -= delta
    new_h[dest] += delta
    return new_h

def compute_c5(h):

    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

import time
import math
import numpy as np

def _compute_c5(h):
    """Maximum overlap integral for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _project_to_sum_cap(v, target):
    """Project vector v onto the set {0<=x<=1, sum(x)=target}."""
    v = np.clip(v, 0.0, 1.0)
    # If sum already good, return
    s = v.sum()
    if np.isclose(s, target, atol=1e-12):
        return v
    # Bisection on lambda for projection onto capped simplex
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        sm = proj.sum()
        if sm > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Tiny redistribution to hit target exactly (helps with numerical noise)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def _propose_move(h, max_transfer=0.5):
    """Transfer a small amount of mass from one index to another, preserving sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, size=2, replace=False)
    # random direction of transfer
    if np.random.rand() < 0.5:
        src, dst = i, j
    else:
        src, dst = j, i
    max_delta = min(h[src], 1.0 - h[dst])
    max_delta = min(max_delta, max_transfer)
    if max_delta <= 0.0:
        return None
    delta = np.random.rand() * max_delta
    new_h = h.copy()
    new_h[src] -= delta
    new_h[dst] += delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    total_budget = float(budget_s)

    # ------------------------------------------------------------------
    # 1. Initial solution (use the supplied one if present)
    # ------------------------------------------------------------------
    init = kwargs.get("initial_h_values", None)

        n = 150
        h0 = np.zeros(n)
        ones = int(n // 2)
        h0[np.random.choice(n, size=ones, replace=False)] = 1.0

    target_sum = n / 2.0
    h0 = _project_to_sum_cap(h0, target_sum)

    best_h = h0.copy()
    best_c5 = _compute_c5(best_h)

    cur_h = best_h.copy()
    cur_c5 = best_c5

    # ------------------------------------------------------------------
    # 2. Parameters for the annealing / projected‑gradient loop
    # ------------------------------------------------------------------
    beta_start, beta_end = 1.0, 30.0          # softness of the soft‑min
    step_start, step_end = 0.5, 0.02        # step size for gradient updates
    max_transfer = 0.5                      # for occasional swap moves
    iter_cnt = 0
    last_improve_iter = 0

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    # ------------------------------------------------------------------
    while True:
        now = time.time()
        if now - start_time > total_budget - 1.0:
            break

        frac = (now - start_time) / total_budget
        beta = beta_start + (beta_end - beta_start) * frac
        step = step_start + (step_end - step_start) * (1.0 - frac)

        dx = 2.0 / n

        shifts = np.arange(-(n - 1), n)                     # length 2n‑1

        w = np.exp(-beta * a_shift)

    best_h = _project_to_sum_cap(best_h, n / 2.0)
    best_c5 = _compute_c5(best_h)

    return best_h, float(best_c5), int(n)
