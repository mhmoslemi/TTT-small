def gradient_wrt_h(h, lag):
    n = len(h)
    grad = np.zeros_like(h)
    # term for i where i+lag < n:
    if lag >= 0:
        # i from 0 to n-1-lag
        grad[:n-lag] += 1 - h[lag:]  # (1 - h[i+lag]) when i index
        # i from lag to n-1: for i that have i-lag >=0:
        grad[lag:] -= h[:n-lag]  # -h[i-lag]
    else:
        # lag < 0 (shift left), consider k negative
        lag_abs = -lag
        # i from lag_abs to n-1: i+lag >=0 indices
        grad[lag_abs:] += 1 - h[:n-lag_abs]  # (1 - h[i+lag]) = (1 - h[i-lag_abs])
        # i from 0 to n-1+lag: i-lag = i+lag_abs
        grad[:n+lag] -= h[lag_abs:]  # -h[i-lag] = -h[i+lag_abs]
    return grad

def project_onto_bounds_and_sum(v, target, lower=0.0, upper=1.0):
    # The solution is x_i = min(upper, max(lower, v_i + λ))
    # Find λ such that sum_i x_i = target.
    # If the sum of lower bounds > target or sum of upper < target, infeasible (shouldn't happen given target within [n*lower, n*upper]).
    lo = -np.max(v) - 1.0  # ensures v+λ <= 0 for all => clamped to lower
    # Actually we can compute lo = lower - np.max(v)
    lo = lower - np.max(v) - 1.0 # enough negative
    hi = upper - np.min(v) + 1.0 # enough positive

    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo+hi)/2.0, lower, upper)
    return x

def project_onto_bounds_and_sum(v, target, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):

    lo = lower - np.max(v) - 1.0  # ensures all clipped to lower
    hi = upper - np.min(v) + 1.0  # ensures all clipped to upper
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, lower, upper)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
        if hi - lo < tol:
            break
    x = np.clip(v + (lo + hi) / 2.0, lower, upper)
    # In rare case sum not exactly target due to bound saturation, adjust by distributing remainder.
    diff = target - x.sum()
    if abs(diff) > tol:
        # distribute diff uniformly among non-saturated elements
        free = np.logical_and(x > lower + tol, x < upper - tol)
        if free.any():
            x[free] += diff / free.sum()
        else:
            # adjust all within bounds but ensure not exceed
            x += diff / len(x)
            x = np.clip(x, lower, upper)
    return x

def _project_bounds_sum(v, target, lo=0.0, hi=1.0, tol=1e-9, maxit=80):
    """Project vector v onto [lo,hi] with exact sum=target."""
    l = lo - np.max(v) - 1.0
    h = hi - np.min(v) + 1.0
    for _ in range(maxit):
        m = (l + h) * 0.5
        x = np.clip(v + m, lo, hi)
        s = x.sum()
        if s > target:
            h = m
        else:
            l = m
        if h - l < tol:
            break
    x = np.clip(v + (l + h) * 0.5, lo, hi)
    diff = target - x.sum()
    if abs(diff) > tol:
        free = (x > lo + tol) & (x < hi - tol)
        if free.any():
            x[free] += diff / free.sum()
        else:
            x += diff / len(x)
            x = np.clip(x, lo, hi)
    return x
