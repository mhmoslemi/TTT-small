def compute_c5(h):
    total = h.sum()
    # compute circular autocorrelation using FFT
    fft_h = np.fft.rfft(h) # use real FFT for speed
    # compute autocorrelation via ifft of product with conj:
    # For real sequence length n, rfft yields n//2+1 points.
    # Use irfft: autocorr = np.fft.irfft(fft_h * np.conj(fft_h), n=n)
    autocorr = np.fft.irfft(fft_h * np.conj(fft_h), n=len(h))
    # O = total - autocorr
    O = total - autocorr
    max_O = O.max()
    return max_O * (2.0 / len(h))

def compute_corr_and_O(h):
    total = h.sum()
    fft_h = np.fft.rfft(h)
    autocorr = np.fft.irfft(fft_h * np.conj(fft_h), n=len(h))
    # autocorr entries (k) for shift 0..n-1
    O = total - autocorr
    max_O = O.max()
    max_k = O.argmax()
    return O, max_O, max_k

    # clip to [0,1] after shifting by tau
    lo = np.min(v) - 1
    hi = np.max(v) + 1

def compute_c5(h):
    n = len(h)
    fft_h = np.fft.rfft(h)
    autocorr = np.fft.irfft(fft_h * np.conj(fft_h), n=n)    # real length n
    total = h.sum()
    O = total - autocorr   # vector of O_k for shift 0..n-1
    max_O = np.max(O)
    return max_O * (2.0 / n)

import time, math, numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Find a step‑function h on [0,2] with ∫h=1 that makes
       C5 = max_k ∫ h(x)*(1-h(x+k)) dx as small as possible.

       The function returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)

    # ----- problem size -------------------------------------------------
    # Prefer a moderately fine discretisation (200 points).  If a parent
    # construction exists we up‑sample it, otherwise we start from random.
    TARGET_N = 200
    if 'initial_h_values' in globals():
        old = np.asarray(initial_h_values, dtype=float)
        old_n = old.size
        # up‑sample by linear interpolation onto a uniform grid of length TARGET_N
        xs_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
        xs_new = np.linspace(0.0, 2.0, TARGET_N, endpoint=False)
        h0 = np.interp(xs_new, xs_old, old)
    else:
        h0 = np.random.rand(TARGET_N)

    n = TARGET_N
    target_sum = n / 2.0               # because ∑h = n/2  ⇔  ∫h = 1
    dx = 2.0 / n

    # ----- projection onto the capped simplex (0≤h_i≤1, Σh_i = target_sum) -----
    def proj_capped(v):
        lo = np.min(v) - 1.0
        hi = np.max(v) + 1.0
        for _ in range(50):
            tau = (lo + hi) * 0.5
            w = np.clip(v - tau, 0.0, 1.0)
            s = w.sum()
            if s > target_sum:
                lo = tau
            else:
                hi = tau
        return np.clip(v - ((lo + hi) * 0.5), 0.0, 1.0)

    # enforce feasibility for the starting point
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = proj_capped(h0)

    # ----- objective (circular correlation) ---------------------------------
    def c5_of(h):
        """C5 for a given h using circular shifts."""
        fft_h = np.fft.rfft(h)
        autocorr = np.fft.irfft(fft_h * np.conj(fft_h), n=len(h))
        total = h.sum()
        # O_k = total - autocorr_k
        max_O = np.max(total - autocorr)
        return max_O * dx

    cur_h = h0.copy()
    cur_score = c5_of(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # ----- simulated annealing ------------------------------------------------
    start_time = time.time()
    temp_start, temp_end = 0.04, 1e-5           # temperature schedule
    no_improve = 0
    max_no_improve = 20000

    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
        prog = elapsed / budget_s
        T = temp_start * (1 - prog) + temp_end * prog

        # propose a mass‑transfer between two coordinates (keeps Σh constant)
        i, j = np.random.choice(n, size=2, replace=False)
        # feasible interval for d where h_i←h_i+d, h_j←h_j−d
        d_low = max(-cur_h[i], cur_h[j] - 1.0)
        d_high = min(1.0 - cur_h[i], cur_h[j])
        if d_low < d_high:
            d = np.random.uniform(d_low, d_high)
            cand = cur_h.copy()
            cand[i] += d
            cand[j] -= d
        else:
            # fallback: a small random perturbation followed by projection
            cand = cur_h + 0.15 * np.random.randn(n)
            cand = np.clip(cand, 0.0, 1.0)
            cand = proj_capped(cand)

        cand_score = c5_of(cand)
        delta = cand_score - cur_score
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-12)):
            cur_h, cur_score = cand, cand_score
            no_improve = 0
        else:
            no_improve += 1

        if cand_score < best_score:
            best_h, best_score = cand.copy(), cand_score
            no_improve = 0

        if no_improve > max_no_improve:
            # random restart to escape a plateau
            cur_h = proj_capped(np.random.rand(n))
            cur_score = c5_of(cur_h)
            no_improve = 0

    # ----- final greedy refinement (hill‑climbing) ---------------------------
    for _ in range(5000):
        i, j = np.random.choice(n, size=2, replace=False)
        d_low = max(-best_h[i], best_h[j] - 1.0)
        d_high = min(1.0 - best_h[i], best_h[j])
        if d_low >= d_high:
            continue
        # use a reduced step size to fine‑tune
        d = np.random.uniform(d_low, d_high) * 0.4
        new_h = best_h.copy()
        new_h[i] += d
        new_h[j] -= d
        new_score = c5_of(new_h)
        if new_score < best_score:
            best_h, best_score = new_h, new_score

    # final projection for safety
    best_h = proj_capped(best_h)
    best_score = c5_of(best_h)

    return (best_h.tolist(), float(best_score), n)
