        if rng.random() < 0.0005:
            # small perturbation around best or a fresh random feasible point
            if rng.random() < 0.5 and best_h is not None:
                # perturb best
                pert = best_h.copy()
                # add small Gaussian noise then project
                noise = rng.normal(scale=0.05, size=n)
                pert += noise
                pert = project_to_feasible(pert)
                cur = pert
                cur_score = compute_c5(cur)
            else:
                cur = rng.random(n)
                cur = project_to_feasible(cur)
                cur_score = compute_c5(cur)

    # final projection to be safe
    best_h = project_to_feasible(best_h)
    final_score = compute_c5(best_h)
    return (best_h.tolist(), float(final_score), n)

import time
import math
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine problem size and possible parent solution
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=np.float64)
        n = init_arr.shape[0]
    else:
        init_arr = None
        n = 80                     # fallback size
    dx = 2.0 / n
    target_sum = n / 2.0           # integral constraint: sum(h) == target_sum

    # ----------------------------------------------------------------------
    # Helper utilities
    # ----------------------------------------------------------------------
    def compute_c5(h):
        """Maximum overlap integral for a candidate h."""
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * dx)

    def project_to_feasible(v):
        """Clip to [0,1] and enforce sum(v)=target_sum (iterative water‑filling)."""
        v = np.clip(v, 0.0, 1.0)
        # Early exit if already satisfied
        if abs(v.sum() - target_sum) < 1e-12:
            return v
        w = v.copy()
        # Scale globally then clip, iterating to satisfy the sum
        for _ in range(10):
            total = w.sum()
            if total == 0:
                w[:] = target_sum / n
                break
            factor = target_sum / total
            w *= factor
            w = np.clip(w, 0.0, 1.0)
            diff = target_sum - w.sum()
            if abs(diff) < 1e-12:
                break
            mask = (w > 0.0) & (w < 1.0)
            cnt = mask.sum()
            if cnt == 0:
                # cannot distribute further – break
                break
            w[mask] += diff / cnt
        # Final tiny rescaling to erase numeric drift
        total = w.sum()
        if total > 0:
            w *= target_sum / total
        return np.clip(w, 0.0, 1.0)

    # ----------------------------------------------------------------------
    # Initialise best solution (if a parent exists)
    # ----------------------------------------------------------------------
    best_h = None
    best_score = float('inf')
    if init_arr is not None:
        h0 = project_to_feasible(init_arr)
        s0 = compute_c5(h0)
        best_h = h0.copy()
        best_score = s0

    # ----------------------------------------------------------------------
    # Phase 1 – random binary search (only when n is even)
    # ----------------------------------------------------------------------
    def random_binary():
        """Uniform random binary vector with exactly target_sum ones."""
        arr = np.zeros(n, dtype=np.float64)
        ones = int(round(target_sum))
        idx = rng.choice(n, size=ones, replace=False)
        arr[idx] = 1.0
        return arr

    if n % 2 == 0:
        phase1_deadline = start_time + budget_s * 0.75
        while time.time() < phase1_deadline:
            cand = random_binary()
            sc = compute_c5(cand)
            if sc < best_score:
                best_score = sc
                best_h = cand.copy()

    # ----------------------------------------------------------------------
    # Phase 2 – continuous simulated annealing starting from best_h
    # ----------------------------------------------------------------------
    if best_h is None:
        # No feasible candidate yet – build a random continuous one
        cur = rng.random(n)
        cur = project_to_feasible(cur)
    else:
        cur = best_h.copy()
    cur_score = compute_c5(cur)

    # SA parameters
    T0 = 0.02                         # initial temperature
    max_fraction = 0.99               # stop slightly before the budget expires

    while True:
        now = time.time()
        if now - start_time >= budget_s * max_fraction:
            break

        # Linear cooling
        elapsed_frac = (now - start_time) / budget_s
        T = T0 * (1.0 - elapsed_frac)

        # Propose a two‑coordinate move that preserves the sum
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur[i], cur[j]

        # Feasible delta interval (add to i, subtract from j)
        low = max(-hi,      hj - 1.0)
        high = min(1.0 - hi, hj)
        if low >= high:
            continue
        delta = rng.uniform(low, high)

        new = cur.copy()
        new[i] = hi + delta
        new[j] = hj - delta

        new_score = compute_c5(new)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = new.copy()

        # Occasional diversification
        if rng.random() < 5e-4:
            if best_h is not None and rng.random() < 0.5:
                # small Gaussian perturbation around the current best
                pert = best_h + rng.normal(scale=0.07, size=n)
                cur = project_to_feasible(pert)
            else:
                cur = rng.random(n)
                cur = project_to_feasible(cur)
            cur_score = compute_c5(cur)

    # Final safety projection and evaluation
    best_h = project_to_feasible(best_h)
    final_score = compute_c5(best_h)

    return (best_h.tolist(), float(final_score), n)
