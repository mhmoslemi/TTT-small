import time
import numpy as np

def _dx(n):
    """dx for n points on [0,2]."""
    return 2.0 / n

def _c5_of(h):
    """Return the C5 value for a step function h given as a 1‑d array."""
    n = h.shape[0]
    dx = _dx(n)
    # linear cross‑correlation (same as the evaluator)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _grad_for_shift(h, k):
    """
    Gradient of the correlation term that attains the current maximum.
    Returns a vector g where  dC/dh = g .
    """
    n = h.shape[0]
    g = np.zeros_like(h)
    if k == 0:
        # not needed (C_0 = 0 for binary h) – use a harmless gradient
        g = 1.0 - 2.0 * h
        return g
    if k > 0:
        # i in [0, n-1-k] contributes the constant 1 term
        i1 = np.arange(n - k)
        g[i1] = 1.0
        # subtract h_{i+k}
        g[i1] -= h[i1 + k]
        # indices i >= k subtract h_{i-k}
        i2 = np.arange(k, n)
        g[i2] -= h[i2 - k]
    else:  # k < 0, work with its absolute value
        kk = -k
        # i in [kk, n-1] contributes the constant 1 term
        i1 = np.arange(kk, n)
        g[i1] = 1.0
        # subtract h_{i-kk}
        g[i1] -= h[i1 - kk]
        # i in [0, n-1-kk] subtract h_{i+kk}
        i2 = np.arange(n - kk)
        g[i2] -= h[i2 + kk]
    return g

def _feasible_step(h, d):
    """Maximum step (alpha) in direction d that keeps 0≤h+αd≤1."""
    # distances to bounds
    pos = d > 1e-12
    neg = d < -1e-12
    max_step = np.inf
    if np.any(pos):
        max_step = min(max_step, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        max_step = min(max_step, np.min(h[neg] / (-d[neg])))
    return max_step

def _project_sum_one(h):
    """Renormalise h so that sum(h) == len(h)/2, keeping the bounds."""
    n = h.shape[0]
    target = n / 2.0
    s = h.sum()
    if s == 0:
        # unlikely, but protect division
        h[:] = target / n
        return h
    h *= target / s

            h[free] += adj

    return h
