def max_overlap(h):
    n = h.size
    # Use FFT length = 1 << (2*n-1).bit_length()
    L = 1 << ((2*n-1).bit_length())
    h_fft = np.fft.rfft(h, n=L)
    rev = (1 - h)[::-1]
    rev_fft = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(h_fft * rev_fft, n=L)[:2*n-1]
    max_val = conv.max() * (2.0 / n)
    max_idx = conv.argmax()
    return max_val, max_idx

            if weight == 0:
                continue
            if d >= 0:

                grad[:n-d] -= weight * h[d:]
                # term -weight * h_{i-d} (i-d >=0) i >= d
                grad[d:] -= weight * h[:n-d]
            else:
                dd = -d
                grad[dd:] -= weight * h[:n-dd]
                grad[:n-dd] -= weight * h[dd:]
        # step
        h_new = h - lr * grad
        # clip and project
        h_new = np.clip(h_new, 0, 1)
        # adjust sum to target n/2
        h_new = project_onto_bounds(h_new, n/2)
        # evaluate max overlap
        new_val = max_overlap_and_offset(h_new)[0]
        # accept if improves or sometimes accept? We'll just accept always (as we project).
        if new_val > max_O + 1e-12: # objective got worse maybe due to smoothing? maybe skip
            # revert
            break
        h = h_new
    return h

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _project_to_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project v onto the intersection of the hyperplane sum(v)=target
    and the box 0 <= v <= 1 using a simple water‑filling (binary search) scheme.
    """
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _max_overlap_and_index(h: np.ndarray):
    """
    Return (max_overlap, max_index) where the overlap is computed as
        C(k) = sum_i h_i * (1 - h_{i+k}) * dx
    and max_index is the index in the full correlation array (0 … 2n‑2).
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return float(corr[idx] * dx), idx


def _partner_array(h: np.ndarray, offset: int) -> np.ndarray:
    """
    For a given offset (positive or negative) produce the array
        L_i = h_{i+offset}               if the index is inside the domain
        L_i = 0.0                         otherwise
    """
    n = h.shape[0]
    L = np.empty(n, dtype=np.float64)
    L.fill(0.0)
    if offset >= 0:
        L[: n - offset] = h[offset:]
    else:
        off = -offset
        L[off:] = h[: n - off]
    return L


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def _improve_h(h: np.ndarray, time_limit: float) -> np.ndarray:
    """
    Perform a greedy improvement that repeatedly moves mass from positions
    with small partner values to positions with large partner values for the
    currently worst shift.  The routine stops when no improving move exists or
    the time budget expires.
    """
    n = h.shape[0]
    target = n / 2.0
    cur_val, _ = _max_overlap_and_index(h)

    while True:
        # time check is performed by the caller
        max_val, max_idx = _max_overlap_and_index(h)
        if max_val < cur_val - 1e-12:
            cur_val = max_val
        else:
            # no improvement in the last move – stop
            break

        offset = max_idx - (n - 1)
        L = _partner_array(h, offset)

        eps = 1e-14
        src_mask = (h > eps)
        sink_mask = (h < 1.0 - eps)

        if not src_mask.any() or not sink_mask.any():
            break

        src_candidates = np.where(src_mask)[0]
        sink_candidates = np.where(sink_mask)[0]

        src = src_candidates[np.argmin(L[src_candidates])]
        sink = sink_candidates[np.argmax(L[sink_candidates])]

        if L[sink] <= L[src] + 1e-14:
            break

        delta = min(h[src], 1.0 - h[sink])
        if delta <= 0.0:
            break

        # try the move
        h_new = h.copy()
        h_new[src] -= delta
        h_new[sink] += delta
        new_val, _ = _max_overlap_and_index(h_new)

        if new_val < cur_val - 1e-12:
            h = h_new
            cur_val = new_val
        else:
            break

    # final projection to keep the exact sum (numerical drift)
    return _project_to_bounds(h, target)


def _random_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random vector respecting the constraints."""
    v = rng.random(n)
    return _project_to_bounds(v, n / 2.0)
