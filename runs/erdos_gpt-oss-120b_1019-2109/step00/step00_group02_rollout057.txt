def simulated_annealing_for_h(n, h_start, time_budget, rng):
    h = h_start.copy()
    cur_val = evaluate(h)
    best_h = h.copy()
    best_val = cur_val
    start = time.time()
    max_step_initial = 0.5
    init_temp = 0.02  # choose a small temperature, we can set relative to cur_val
    # set step and temperature scaling
    no_improve = 0
    while time.time() - start < time_budget:
        elapsed = time.time() - start
        frac = elapsed / time_budget
        temp = init_temp * (1 - frac) + 1e-9
        step = max_step_initial * (1 - frac) + 1e-6
        # pick two distinct indices
        i = rng.randint(n)
        j = rng.randint(n-1)
        if j >= i:
            j += 1
        # random direction
        if rng.rand() < 0.5:
            delta_max = min(step, h[i], 1 - h[j])
            if delta_max <= 0:
                continue
            delta = rng.uniform(0, delta_max)
            new_h = h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            delta_max = min(step, h[j], 1 - h[i])
            if delta_max <= 0:
                continue
            delta = rng.uniform(0, delta_max)
            new_h = h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        new_val = evaluate(new_h)
        dval = new_val - cur_val
        # acceptance
        if dval <= 0 or rng.rand() < math.exp(-dval / (temp + 1e-12)):
            h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        # random restart if stuck for many iterations
        if no_improve > 20000:
            # random reinitialize (binary as start)
            h = np.zeros(n)
            if n % 2 == 0:
                h[:n//2] = 1.0
                rng.shuffle(h)
            else:
                half = n//2
                h[:half] = 1.0
                h[half] = 0.5
                rng.shuffle(h)
            cur_val = evaluate(h)
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            no_improve = 0
    return best_h, best_val

import time, math
import numpy as np

def _eval_overlap(h):
    """Return C5 bound for a step function h (numpy 1‑D array)."""
    n = h.size
    # correlation of h with (1-h) using full mode
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * (2.0 / n)


def _binary_initial(n, rng):
    """Balanced binary (0/1) vector of length n (sum = n/2)."""
    h = np.zeros(n, dtype=float)
    if n % 2 == 0:
        h[: n // 2] = 1.0
    else:
        half = n // 2
        h[:half] = 1.0
        h[half] = 0.5           # the unavoidable 0.5 when n is odd
    rng.shuffle(h)
    return h


def _triangle_initial(n):
    """Triangular shape: h(x)=1-|x-1| on [0,2] sampled at mid‑points."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(x - 1.0)
    h = np.clip(h, 0.0, 1.0)
    return h


def _anneal_one(h_start, time_budget, rng):
    """
    Simulated‑annealing style improvement of a single start vector.
    Returns best vector and its C5 value found within time_budget seconds.
    """
    n = h_start.size
    best_h = h_start.copy()
    best_val = _eval_overlap(best_h)
    cur_h = h_start.copy()
    cur_val = best_val

    start = time.time()
    # annealing schedules (linear decay)
    INIT_TEMP = 0.02          # temperature scale (absolute, not multiplied by C)
    MAX_DELTA = 0.5           # maximal mass transferred in one move

    no_imp = 0                # counter for stagnation
    while True:
        now = time.time()
        if now - start >= time_budget:
            break
        frac = (now - start) / time_budget
        temp = INIT_TEMP * (1.0 - frac) + 1e-12
        delta_max = MAX_DELTA * (1.0 - frac) + 1e-6

        # pick two distinct indices
        i = rng.randint(n)
        j = rng.randint(n - 1)
        if j >= i:
            j += 1

        # decide direction of mass transfer
        if rng.rand() < 0.5:
            # move from i to j
            allowed = min(delta_max, cur_h[i], 1.0 - cur_h[j])
            if allowed <= 0.0:
                continue
            d = rng.uniform(0.0, allowed)
            new_h = cur_h.copy()
            new_h[i] -= d
            new_h[j] += d
        else:
            # move from j to i
            allowed = min(delta_max, cur_h[j], 1.0 - cur_h[i])
            if allowed <= 0.0:
                continue
            d = rng.uniform(0.0, allowed)
            new_h = cur_h.copy()
            new_h[i] += d
            new_h[j] -= d

        new_val = _eval_overlap(new_h)
        dval = new_val - cur_val
        if dval <= 0.0 or rng.rand() < math.exp(-dval / temp):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        # occasional full restart if we are stuck for a long time
        if no_imp > 20000:
            cur_h = _binary_initial(n, rng)
            cur_val = _eval_overlap(cur_h)
            no_imp = 0
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()

    return best_h, best_val
