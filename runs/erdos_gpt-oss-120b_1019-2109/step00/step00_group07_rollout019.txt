                best_h = h_new.copy()

                cur_val = compute_c5(h)
                stagnation = 0

import time, math
import numpy as np

def _project_capped_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project v onto {x: 0<=x<=1, sum(x)=S}."""
    lo = -np.max(v) - 1.0   # makes all entries <=0 after shift
    hi = 1.0 - np.min(v) + 1.0   # makes all entries >=1 after shift
    for _ in range(200):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


def _gradient_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the worst (max) correlation term."""
    n = h.size
    g = np.ones(n)
    if shift >= 0:
        if shift < n:
            g[: n - shift] -= h[shift:]
            g[shift:] -= h[: n - shift]
    else:
        s = -shift
        g[s:] -= h[: n - s]
        g[: n - s] -= h[s:]
    return g


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Find a step function h on [0,2] with ∫h=1, 0≤h≤1 that lowers the
    Erdős C5 constant. Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # ---- initialise -------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=float)
        n = h.size
    else:
        n = 80                     # fallback grid size
        h = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0           # ∑h = n/2  <=>  ∫ h = 1
    h = _project_capped_simplex(h, target_sum)

    cur_val = _c5_value(h)
    best_h = h.copy()
    best_val = cur_val

    # ---- optimisation parameters ------------------------------------
    it = 0
    stagn = 0
    T = 0.02               # temperature for uphill moves
    noise_scale = 0.3

    # ---- main optimisation loop ------------------------------------
    while True:
        # time budget check
        if time.time() - start > budget_s - 0.5:
            break

        # worst shift and its gradient
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)

        grad = _gradient_worst_shift(h, shift)

        # step size (decays slowly)
        eta = 0.2 * max(0.01, 1.0 - it / 200000.0)

        # stochastic proposal
        proposal = h - eta * grad + noise_scale * np.random.randn(n)
        h_new = _project_capped_simplex(proposal, target_sum)

        new_val = _c5_value(h_new)

        # Metropolis acceptance
        delta = new_val - cur_val
        if delta < 0.0 or np.random.rand() < math.exp(-delta / max(T, 1e-12)):
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()
                stagn = 0
            else:
                stagn += 1
