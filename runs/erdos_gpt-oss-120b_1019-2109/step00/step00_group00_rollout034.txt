   xs = (np.arange(N) + 0.5)*(2.0/N)
   if r <= 0:

   else:
       H = 1.0/(2 - r)
       ramp_len = r
       # slope = H / ramp_len
       slope = H / ramp_len
       # plateau start at ramp_len, end at 2 - ramp_len
       # compute h piecewise
       h = np.empty(N)
       # left ramp region
       mask_left = xs < ramp_len
       h[mask_left] = slope * xs[mask_left]
       # plateau region
       mask_plateau = (xs >= ramp_len) & (xs <= 2 - ramp_len)
       h[mask_plateau] = H
       # right ramp region
       mask_right = xs > 2 - ramp_len
       h[mask_right] = slope * (2 - xs[mask_right])
   # Ensure bounds

   # Adjust sum to target via scaling factor (should be near 1)
   target = N / 2
   sum_h = h.sum()
   factor = target / sum_h

       # Clip to 1
       h = np.minimum(h, 1.0)
       # Rescale again if needed
       sum_h = h.sum()
       factor = target / sum_h
       h *= factor
   return h

        ...

    def bound_for_r(r):

        # evaluate bound
        corr = np.correlate(h, 1 - h, mode='full')

    # minimize bound; use bounded method

    current_h = best_h.copy()
    current_score = best_score

    # SA parameters
    max_time = budget_s - 0.5
    while time.time() - start < max_time:
        elapsed = time.time() - start
        # linear temperature schedule
        T = 0.02 * (1 - elapsed / max_time)  # starting T~0.02
        # step size linear schedule
        step = 0.3 * (1 - elapsed / max_time) + 0.001

        # propose
        i = np.random.randint(N)
        j = np.random.randint(N-1)
        if j >= i: j += 1
        # random direction
        if np.random.rand() < 0.5:
            max_delta = min(step, 1 - current_h[i], current_h[j])
            if max_delta <= 0:
                continue
            delta = np.random.uniform(0, max_delta)
            new_h = current_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        else:
            max_delta = min(step, current_h[i], 1 - current_h[j])
            if max_delta <= 0:
                continue
            delta = np.random.uniform(0, max_delta)
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta

        new_score = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * dx

        if new_score < best_score:
            best_score = new_score
            best_h = new_h.copy()

        # Accept?
        if new_score < current_score or np.random.rand() < math.exp(-(new_score - current_score) / (T + 1e-12)):

import time
import math
import numpy as np
from scipy.optimize import minimize_scalar

def _make_h(r, N):
    """Create a symmetric trapezoid shape with parameter r∈[0,1].
    r = ramp length, H = height, plateau width = 2-2r.
    The continuous shape has integral 1, the discrete version is scaled
    to satisfy the exact sum constraint."""
    xs = (np.arange(N) + 0.5) * (2.0 / N)          # mid‑point sample positions
    if r <= 0.0:                                   # constant 0.5
        h = np.full(N, 0.5, dtype=float)
    elif r >= 1.0:                                 # pure triangle
        h = np.where(xs <= 1.0, xs, 2.0 - xs)
    else:
        H = 1.0 / (2.0 - r)                        # height from area condition
        slope = H / r
        left = xs < r
        right = xs > 2.0 - r
        plateau = ~left & ~right
        h = np.empty(N, dtype=float)
        h[left] = slope * xs[left]
        h[right] = slope * (2.0 - xs[right])
        h[plateau] = H
    # enforce box constraints and exact integral (within floating tolerance)
    np.clip(h, 0.0, 1.0, out=h)
    target = N / 2.0
    factor = target / h.sum()
    h *= factor
    # very small overshoot can be clipped again
    np.clip(h, 0.0, 1.0, out=h)
    # final tiny correction of the total mass
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # add the remaining mass to a point that can still move
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] = max(0.0, min(1.0, h[idx] + diff))
    return h

def _c5_of_h(h):
    """Return the Erdős overlap bound for a discretised step function."""
    N = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / N)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # number of samples – a compromise between resolution and speed
    N = 200
    dx = 2.0 / N

    # ---- 1‑parameter family (symmetric trapezoids) -----------------------
    def bound_for_r(r):
        h = _make_h(r, N)
        return _c5_of_h(h)

    # minimise the bound over r∈[0,1] (bracketed by constant and triangle)
    opt = minimize_scalar(bound_for_r, bounds=(0.0, 1.0), method="bounded",
                          options={"xatol": 1e-4, "maxiter": 300})
    r_best = opt.x
    h_best = _make_h(r_best, N)
    best_score = _c5_of_h(h_best)

    # ---- Simulated annealing refinements -------------------------------
    cur_h = h_best.copy()
    cur_score = best_score

    # simple helper to propose a feasible two‑index move preserving the sum
    def propose_move(h, max_step):
        N = h.shape[0]
        i, j = np.random.randint(N, size=2)
        while i == j:
            j = np.random.randint(N)
        # pick a direction uniformly
        if np.random.rand() < 0.5:          # increase i, decrease j
            max_delta = min(max_step, 1.0 - h[i], h[j])
            if max_delta <= 0:
                return None, None, None
            delta = np.random.uniform(0.0, max_delta)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta
            return h_new, delta, (i, j)
        else:                               # decrease i, increase j
            max_delta = min(max_step, h[i], 1.0 - h[j])
            if max_delta <= 0:
                return None, None, None
            delta = np.random.uniform(0.0, max_delta)
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
            return h_new, delta, (i, j)

    # main annealing loop
    while True:
        if time.time() - start_time > budget_s - 0.5:   # small safety margin
            break
        elapsed = time.time() - start_time
        t_frac = 1.0 - elapsed / budget_s
        # temperature and step size decay linearly
        T = 0.02 * t_frac
        step = 0.30 * t_frac + 0.001

        h_new, delta, idxs = propose_move(cur_h, step)
        if h_new is None:
            continue

        new_score = _c5_of_h(h_new)

        if new_score < best_score:
            best_score = new_score
            h_best = h_new.copy()

        # Metropolis acceptance
        if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / (T + 1e-12)):
            cur_h = h_new
            cur_score = new_score

    # final tiny mass‑fix (ensures sum exactly N/2)
    target = N / 2.0
    diff = target - h_best.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h_best)
        else:
            idx = np.argmin(h_best)
        h_best[idx] = max(0.0, min(1.0, h_best[idx] + diff))

    final_score = _c5_of_h(h_best)
    return (h_best.tolist(), float(final_score), N)
