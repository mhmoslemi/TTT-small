def project_onto_constraints(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # Adjust sum
    diff = target - v.sum()
    # tolerance
    tol = 1e-12
    max_iter = 100
    for _ in range(max_iter):
        if abs(diff) < tol:
            break
        # indices that are not at bounds
        free = (v > 0 + 1e-12) & (v < 1 - 1e-12)
        free_count = free.sum()
        if free_count == 0:
            # cannot adjust further; break
            break
        # amount to add per free variable
        add = diff / free_count
        # clip add to not exceed bounds
        # compute new values for free indices
        v_new = v.copy()
        v_new[free] = np.clip(v[free] + add, 0.0, 1.0)
        diff = target - v_new.sum()
        # update v with new values
        v = v_new
    # at end, maybe still off due to rounding; adjust a few
    if abs(v.sum() - target) > 1e-8:
        # distribute residual uniformly within bounds
        residual = target - v.sum()
        # pick random free index
        idx = np.random.choice(np.where(v < 1.0)[0])
        v[idx] += residual
        v = np.clip(v, 0.0, 1.0)
    return v

import time
import math
import random
import numpy as np

def compute_c5(h, dx):
    """Maximum overlap integral for a vector h."""
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def gradient_for_shift(h, shift):
    """
    Gradient of the overlap for a given integer shift (np.arange(-(n-1), n)).
    Returns d/dh_i of  sum_i h_i (1 - h_{i+shift})  (zero‑padded version).
    """
    n = h.shape[0]
    g = np.zeros(n, dtype=np.float64)

    # indices where i and i+shift both lie in [0, n-1]
    if shift >= 0:
        i_start = 0
        i_end = n - shift
    else:
        i_start = -shift
        i_end = n
    if i_end > i_start:
        g[i_start:i_end] += 1.0                     # contribution of sum_i h_i

    # subtract h_{i+shift}
    if shift >= 0:
        g[0:n - shift] -= h[shift:n]
    else:
        g[-shift:n] -= h[0:n + shift]

    # subtract h_{i-shift}
    if shift >= 0:
        g[shift:n] -= h[0:n - shift]
    else:
        g[0:n + shift] -= h[-shift:n]

    return g


def project_constraints(v, target_sum, eps=1e-12, max_iter=200):
    """
    Euclidean projection of v onto the set {x | 0<=x_i<=1, sum x_i = target_sum}.
    Simple water‑filling style iteration.
    """
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        diff = target_sum - s
        if abs(diff) <= eps:
            break
        # free entries are those not on a bound
        free = (v > eps) & (v < 1.0 - eps)
        cnt = free.sum()
        if cnt == 0:
            # cannot move any further; distribute residually (tiny adjustments)
            if abs(diff) > eps:
                # adjust a random non‑saturated entry
                idx = random.choice(np.where(v < 1.0)[0])
                v[idx] = min(1.0, max(0.0, v[idx] + diff))
            break
        add = diff / cnt
        # apply and re‑clip
        v[free] += add
        v = np.clip(v, 0.0, 1.0)
    # final tiny correction if needed
    if abs(v.sum() - target_sum) > 1e-9:
        # distribute the tiny residual uniformly among non‑saturated slots
        resid = target_sum - v.sum()
        free = (v > eps) & (v < 1.0 - eps)
        cnt = free.sum()
        if cnt > 0:
            v[free] += resid / cnt
            v = np.clip(v, 0.0, 1.0)
    return v
