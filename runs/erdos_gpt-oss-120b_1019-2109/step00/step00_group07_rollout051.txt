def compute_gradient(h, shift):
    n = len(h)
    grad = np.ones_like(h)
    if shift >= 0:
        # forward: h[i+shift], i from 0..n-shift-1
        grad[:n-shift] -= h[shift:]
        # backward: h[i-shift] exists for i>=shift
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        # forward: h[i - s] (since shift negative)
        grad[s:] -= h[:n-s]
        # backward: h[i + s]
        grad[:n-s] -= h[s:]
    return grad

def gradient_descent(h, steps=500, alpha=0.1):
    target_sum = len(h)/2
    for i in range(steps):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (len(h)-1)
        # gradient compute
        if shift >= 0:
            grad = np.ones_like(h)
            grad[:len(h)-shift] -= h[shift:]
            grad[shift:] -= h[:len(h)-shift]
        else:
            s = -shift
            grad = np.ones_like(h)
            grad[s:] -= h[:len(h)-s]
            grad[:len(h)-s] -= h[s:]
        # step
        h = h - alpha * grad
        h = project_onto_simplex_box(h, target_sum)
        # slowly decrease alpha
        alpha *= 0.9995
    return h

import numpy as np, time, math, random

def project_onto_box_simplex(y, target):
    lo = -np.max(y) - 1.0
    hi = np.max(y) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        v = np.clip(y - mid, 0.0, 1.0)
        s = v.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2.0, 0.0, 1.0)

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    # full correlation
    # (h*(1 - h_shift)) using np.correlate
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def gradient_descent(h, steps=2000, alpha_start=0.3):
    n = h.shape[0]
    target_sum = n / 2.0
    alpha = alpha_start
    for _ in range(steps):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        shift = max_idx - (n - 1)
        if shift >= 0:
            grad = np.ones_like(h)
            grad[:n-shift] -= h[shift:]
            grad[shift:] -= h[:n-shift]
        else:
            s = -shift
            grad = np.ones_like(h)
            grad[s:] -= h[:n-s]
            grad[:n-s] -= h[s:]
        # update
        h = h - alpha * grad
        # project
        h = project_onto_box_simplex(h, target_sum)
        alpha *= 0.9995
    return h

def random_feasible(n):
    y = np.random.rand(n)
    return project_onto_box_simplex(y, n/2.0)

def triangular_initial(n):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.maximum(0.0, 1 - np.abs(x - 1))
    # adjust sum
    target = n/2.0
    h = project_onto_box_simplex(h, target)
    return h

import numpy as np
import random
import time

# ---------- helper utilities ----------
def _project_to_simplex_box(y: np.ndarray, target_sum: float) -> np.ndarray:
    """Project y onto the set {0<=x<=1, sum(x)=target_sum}."""
    lo = -np.max(y) - 1.0          # lower bound for lambda (forces all entries to 1)
    hi = np.max(y) + 1.0           # upper bound for lambda (forces all entries to 0)
    for _ in range(60):           # binary search, enough for double precision
        lam = (lo + hi) * 0.5
        v = np.clip(y - lam, 0.0, 1.0)
        s = v.sum()
        if s > target_sum:
            lo = lam               # need larger lambda to reduce the sum
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(y - lam, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _gradient_step(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of max‑overlap term w.r.t. h for a given integer shift."""
    n = h.shape[0]
    g = np.ones_like(h)
    if shift >= 0:
        g[: n - shift] -= h[shift:]
        g[shift:]     -= h[: n - shift]
    else:
        s = -shift
        g[s:]         -= h[: n - s]
        g[: n - s]    -= h[s:]
    return g


def _gradient_descent(
    h: np.ndarray,
    steps: int = 1500,
    alpha_start: float = 0.4,
) -> np.ndarray:
    """Simple projected gradient descent on the worst shift."""
    n = h.shape[0]
    target = n / 2.0
    alpha = alpha_start
    for _ in range(steps):
        # locate worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        # take a step opposite to the sub‑gradient
        h = h - alpha * _gradient_step(h, shift)
        # enforce constraints
        h = _project_to_simplex_box(h, target)
        alpha *= 0.9995
    return h


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random vector with sum = n/2 and values in [0,1]."""
    y = np.random.rand(n)
    return _project_to_simplex_box(y, n / 2.0)


def _triangular_initial(n: int) -> np.ndarray:
    """Discretised continuous triangle h(x)=max(0,1-|x-1|)."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    return _project_to_simplex_box(h, n / 2.0)


# ---------- main optimisation ----------
def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h (as a NumPy array), its C5 bound, and n_points.
    The routine tries a few deterministic starts followed by a simple
    hill‑climbing / gradient refinement within the given time budget.
    """
    random.seed(seed)
    np.random.seed(seed)
    t0 = time.time()

    # ---- problem size -------------------------------------------------
    if "initial_h_values" in globals():
        n = int(np.shape(initial_h_values)[0])
    else:
        n = 80                     # a small, fast grid when no parent is given
    target_sum = n / 2.0

    # ---- build a few starting vectors ---------------------------------
    starters = []

    if "initial_h_values" in globals():
        h0 = np.clip(np.asarray(initial_h_values, dtype=float), 0.0, 1.0)
        h0 = _project_to_simplex_box(h0, target_sum)
        starters.append(h0)

    starters.append(_triangular_initial(n))
    starters.append(_random_feasible(n))

    best_h = None
    best_score = float("inf")

    # ---- phase 1 : deterministic gradient refinement -------------------
    for h0 in starters:
        if time.time() - t0 > 0.6 * budget_s:
            break
        h_opt = _gradient_descent(h0.copy(), steps=1800, alpha_start=0.45)
        val = _c5_value(h_opt)
        if val < best_score:
            best_score = val
            best_h = h_opt.copy()

    # ---- phase 2 : simple stochastic improvement -----------------------
    if best_h is None:
        best_h = _random_feasible(n)
        best_score = _c5_value(best_h)

    cur_h = best_h.copy()
    cur_score = best_score
    stagnant = 0

    while time.time() - t0 < 0.97 * budget_s:
        # pick two indices and move a tiny amount of mass from i to j
        pos = np.where(cur_h > 1e-9)[0]
        neg = np.where(cur_h < 1.0 - 1e-9)[0]
        if pos.size == 0 or neg.size == 0:
            cur_h = _random_feasible(n)
            cur_score = _c5_value(cur_h)
            continue

        i = np.random.choice(pos)
        j = np.random.choice(neg)
        eps = np.random.rand() * min(cur_h[i], 1.0 - cur_h[j])
        if eps <= 0:

        new_h = cur_h.copy()
        new_h[i] -= eps
        new_h[j] += eps
        new_score = _c5_value(new_h)

        if new_score + 1e-12 < cur_score:

            cur_score = new_score
            stagnant = 0

        else:

        # occasional local gradient polishing when we stall

            stagnant = 0

    return (best_h, final_c5, n)
