import numpy as np
import itertools
import time
import math

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral for a discretised step function."""
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def generate_block_patterns() -> list:
    """All 5‑point patterns containing exactly two 1’s, two 0’s and one 0.5."""
    pats = []
    idx = list(range(5))
    for ones in itertools.combinations(idx, 2):
        remaining = set(idx) - set(ones)
        for zeros in itertools.combinations(list(remaining), 2):
            half = (remaining - set(zeros)).pop()
            pat = np.zeros(5, dtype=float)
            pat[list(ones)] = 1.0
            pat[list(zeros)] = 0.0
            pat[half] = 0.5
            pats.append(pat)
    return pats


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1.  Collect a feasible starting point (if supplied) and its score.
    # ------------------------------------------------------------------
    best_h = None
    best_score = float("inf")
    if "initial_h_values" in kwargs:
        init = np.array(kwargs["initial_h_values"], dtype=float).copy()
        if init.ndim == 1:
            dx0 = 2.0 / init.shape[0]
            best_score = compute_c5(init, dx0)
            best_h = init

    # ------------------------------------------------------------------
    # 2.  Choose a problem size divisible by 5 (so the 2‑1‑0‑1‑0 pattern works).
    # ------------------------------------------------------------------
    N = kwargs.get("n_points", 250)               # default size
    N = int(N)
    if N % 5 != 0:
        N += 5 - (N % 5)                          # make it a multiple of 5
    dx = 2.0 / N
    K = N // 5                                     # number of 5‑point blocks

    # ------------------------------------------------------------------
    # 3.  Prepare the pool of 5‑point building blocks.
    # ------------------------------------------------------------------
    blocks = generate_block_patterns()            # length = 30

    # ------------------------------------------------------------------
    # 4.  Random initial construction using the blocks.
    # ------------------------------------------------------------------
    cur_h = np.empty(N, dtype=float)
    cur_block_ids = np.random.randint(len(blocks), size=K)
    for b in range(K):
        cur_h[b * 5 : (b + 1) * 5] = blocks[cur_block_ids[b]]
    cur_score = compute_c5(cur_h, dx)

    if cur_score < best_score:
        best_score = cur_score
        best_h = cur_h.copy()

    # ------------------------------------------------------------------
    # 5.  Coordinate descent on the block level.
    #    Repeatedly try to replace each block by a better one.
    # ------------------------------------------------------------------
    while time.time() - start < budget_s * 0.90:
        improved = False
        order = np.random.permutation(K)
        for b in order:
            sl = slice(b * 5, (b + 1) * 5)
            original = cur_h[sl].copy()
            best_local_score = cur_score
            best_local_pat = original
            # try every alternative block (30 in total)
            for pat in blocks:
                if np.array_equal(pat, original):
                    continue
                cur_h[sl] = pat
                new_score = compute_c5(cur_h, dx)
                if new_score < best_local_score - 1e-12:
                    best_local_score = new_score
                    best_local_pat = pat.copy()
            # commit the best found for this block

                cur_h[sl] = best_local_pat
                cur_score = best_local_score
                improved = True
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_h = cur_h.copy()

                cur_h[sl] = original
        if not improved:
            break

    # ------------------------------------------------------------------
    # 6.  Fine‑grained hill climbing: random swaps of two positions.
    # ------------------------------------------------------------------
    max_swaps = 30000
    for _ in range(max_swaps):
        if time.time() - start > budget_s * 0.98:
            break
        i, j = np.random.randint(N, size=2)
        if cur_h[i] == cur_h[j]:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_score = compute_c5(cur_h, dx)
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    # ------------------------------------------------------------------
    # 7.  Final sanity checks and return.
    # ------------------------------------------------------------------
    if best_h is None:                     # fallback – should never happen
        best_h = cur_h.copy()
        best_score = cur_score
    best_h = np.clip(best_h, 0.0, 1.0)

    # tiny correction to enforce the integral exactly (numerical safety)
    target_sum = N / 2.0
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-8:
        # adjust the first element within the bounds
        best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    final_score = compute_c5(best_h, dx)
    return best_h, float(final_score), N
