def gradient(h, tau):
    N = len(h)
    grad = np.zeros(N)
    if tau >= 0:
        # g[i+tau] defined for i in 0..N-1-tau
        grad[:N-tau] += 1 - h[tau:]   # g shifted left
        # -h[i - tau] defined for i in tau..N-1

    N = len(h)
    corrs = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corrs)
    return max_corr * (2.0 / N)

def gradient_of_shift(h, tau):
    N = len(h)
    grad = np.zeros(N)
    if tau >= 0:
        # part 1: i in 0..N-1-tau
        grad[:N - tau] += 1 - h[tau:]
        # part 2: i in tau..N-1
        grad[tau:] -= h[:N - tau]
    else:
        d = -tau
        # part 1: i in d..N-1
        grad[d:] += 1 - h[:N - d]
        # part 2: i in 0..N-1-d
        grad[:N - d] -= h[d:]
    return grad

def project_capped_simplex(v, total, max_iter=50):
    # binary search for lambda
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        x = np.clip(v - mid, 0, 1)
        s = x.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    x = np.clip(v - lam, 0, 1)
    # due to possible rounding, adjust small residual
    diff = total - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff to non-saturated positions
        free = (x > 0) & (x < 1)
        cnt = free.sum()
        if cnt > 0:
            # adjust uniformly, then clip again
            adjust = diff / cnt
            x[free] += adjust
            # clip again
            x = np.clip(x, 0, 1)
            # recompute diff iteratively; ignore minor error
    return x

import time
import numpy as np

def _cost(h):
    """Maximum overlap integral C₅ for a given step function h."""
    N = h.size
    # full correlation h * (1 - h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * (2.0 / N)   # dx = 2/N


def _gradient(h, tau):
    """Sub‑gradient of the overlap for the shift that currently attains the maximum."""
    N = h.size
    g = np.empty(N, dtype=float)
    g.fill(0.0)
    if tau >= 0:
        # (1 - h[i+tau]) term for i = 0 .. N‑tau‑1
        g[: N - tau] += 1.0 - h[tau:]
        # -h[i‑tau] term for i = tau .. N‑1
        g[tau:] -= h[: N - tau]
    else:                     # tau < 0  →  d = -tau > 0
        d = -tau
        # (1 - h[i‑d]) term for i = d .. N‑1
        g[d:] += 1.0 - h[: N - d]
        # -h[i+d] term for i = 0 .. N‑d‑1
        g[: N - d] -= h[d:]
    return g


def _project_onto_capped_simplex(v, total, max_iter=60, eps=1e-12):
    """
    Projection of v onto {x | 0 ≤ x_i ≤ 1, sum x_i = total}.
    Uses a simple binary‑search on the Lagrange multiplier λ.
    """
    lo = np.min(v) - 1.0          # low enough that clipping makes all zeros
    hi = np.max(v)                # high enough that clipping makes all ones
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = total - x.sum()
    if abs(diff) > eps:
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt > 0:
            delta = diff / cnt
            x[free] += delta
            x = np.clip(x, 0.0, 1.0)
    return x
