def weighted_shift_sum(h, w):
    n = len(h)
    u_plus = np.zeros(n)
    u_minus = np.zeros(n)
    for idx, weight in enumerate(w):
        s = idx - (n - 1)
        if s >= 0:
            # shift_plus: h[i+s]; i from 0 to n-s-1
            u_plus[:n - s] += weight * h[s:]
            # shift_minus: h[i-s] but i-s >=0 => i >= s, so for i>=s, i-s valid
            u_minus[s:] += weight * h[:n - s]
        else:
            t = -s
            # shift_plus for negative s: h[i+s] = h[i - t]; valid for i>=t
            u_plus[t:] += weight * h[:n - t]
            # shift_minus: h[i - s] = h[i + t]; valid for i < n - t
            u_minus[:n - t] += weight * h[t:]
    return u_plus, u_minus
```

def smooth_obj_and_grad(h, alpha):
    O = np.correlate(h, 1 - h, mode='full')
    # stable exp
    aO = alpha * O
    max_aO = np.max(aO)
    exp_aO = np.exp(aO - max_aO)

    return f, grad

import numpy as np
import time
from scipy.optimize import minimize

# global helpers for the optimizer (avoid closures)
_target_sum = None
_current_alpha = None


def _project_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project onto {0<=x<=1, sum(x)=target} by a simple bisection on the shift."""
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discretised step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _weighted_shift_sum(h: np.ndarray, w: np.ndarray):
    """Return the vectors sum_s w_s * h_{i+s} and sum_s w_s * h_{i-s} (zero‑padded)."""
    n = h.size
    u_plus = np.zeros(n)
    u_minus = np.zeros(n)
    shift0 = n - 1
    for idx, wt in enumerate(w):
        s = idx - shift0
        if s >= 0:
            if wt != 0.0:
                u_plus[: n - s] += wt * h[s:]
                u_minus[s:] += wt * h[: n - s]
        else:
            t = -s
            if wt != 0.0:
                u_plus[t:] += wt * h[: n - t]
                u_minus[: n - t] += wt * h[t:]
    return u_plus, u_minus


def _smooth_obj_and_grad(h: np.ndarray, alpha: float):
    """Smooth approximation of max O_s and its gradient."""
    n = h.size
    O = np.correlate(h, 1.0 - h, mode="full")
    aO = alpha * O
    mx = aO.max()
    exp_a = np.exp(aO - mx)
    Z = exp_a.sum()
    f = (np.log(Z) + mx) / alpha
    w = exp_a / Z                     # softmax weights
    u_plus, u_minus = _weighted_shift_sum(h, w)
    grad = np.ones_like(h) - u_plus - u_minus
    return f, grad


def _smooth_obj(x):
    """Objective for the current alpha (global variable)."""
    f, _ = _smooth_obj_and_grad(x, _current_alpha)
    return f


def _smooth_jac(x):
    """Gradient for the current alpha (global variable)."""
    _, g = _smooth_obj_and_grad(x, _current_alpha)
    return g


def _sum_constraint(x):
    """Equality constraint sum(x) = target (global variable)."""
    return x.sum() - _target_sum


def _optimize_continuation(h0: np.ndarray, target: float, time_budget: float):
    """Continuation SLSQP optimizer over a list of alphas."""
    global _target_sum, _current_alpha
    _target_sum = target
    best_h = h0.copy()
    best_c = compute_c5(best_h)
    start = time.time()
    alphas = [5.0, 10.0, 20.0, 40.0, 80.0]
    bounds = [(0.0, 1.0)] * h0.size
    cons = {"type": "eq", "fun": _sum_constraint}
    for a in alphas:
        if time.time() - start > time_budget:
            break
        _current_alpha = a
        res = minimize(_smooth_obj, best_h, method="SLSQP", jac=_smooth_jac,
                       bounds=bounds, constraints=cons,
                       options={"maxiter": 200, "ftol": 1e-9, "disp": False})
        if not res.success:
            # still project to be safe
            cand = _project_bounds_sum(res.x, target)
        else:
            cand = _project_bounds_sum(res.x, target)
        cval = compute_c5(cand)
        if cval < best_c:
            best_c = cval
            best_h = cand.copy()
    return best_h, best_c

        h_init = np.array(initial_h_values, dtype=float)
        n_pts = h_init.size
    else:
        n_pts = kwargs.get("n_points", 60)
        h_init = np.random.rand(n_pts)

    # If a specific n_points is demanded, resample the initial vector
    if "n_points" in kwargs and kwargs["n_points"] != n_pts:
        n_pts = int(kwargs["n_points"])
        x_old = np.linspace(0.0, 2.0, h_init.size, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_pts, endpoint=False)
        h_init = np.interp(x_new, x_old, h_init)

    target_sum = n_pts / 2.0
    h_init = _project_bounds_sum(h_init, target_sum)

    best_h = h_init.copy()
    best_c = compute_c5(best_h)

    # Time split: 80 % for the main optimisation, remainder for a quick fallback
    time_for_opt = budget_s * 0.80
    time_for_fallback = budget_s - time_for_opt

    # ------- main optimisation (continuation + random restarts) -------
    while time.time() - start_all < time_for_opt:
        # random perturbation around current best
        noise = np.random.randn(n_pts) * 0.05
        cand0 = _project_bounds_sum(best_h + noise, target_sum)
        # give each restart a tiny slice of the remaining time
        remaining = time_for_opt - (time.time() - start_all)
        h_opt, c_opt = _optimize_continuation(cand0, target_sum, remaining * 0.5)
        if c_opt < best_c:
            best_c, best_h = c_opt, h_opt.copy()
        # early exit if we already beat the target record
        if best_c <= 0.3808:
            break

    # ------- simple sub‑gradient fallback (if time left) -------
    left = budget_s - (time.time() - start_all)
    if left > 1.0:
        h = best_h.copy()
        n = n_pts
        dx = 2.0 / n
        target = n / 2.0
        lr0 = 0.5
        it = 0
        while time.time() - start_all < budget_s:
            O = np.correlate(h, 1.0 - h, mode="full")
            idx = int(O.argmax())
            shift = idx - (n - 1)
            if shift >= 0:
                plus = np.concatenate([h[shift:], np.zeros(shift)])
                minus = np.concatenate([np.zeros(shift), h[: n - shift]])
            else:
                t = -shift
                plus = np.concatenate([np.zeros(t), h[: n - t]])
                minus = np.concatenate([h[t:], np.zeros(t)])
            grad = 1.0 - plus - minus
            lr = lr0 / np.sqrt(it + 1)
            h = _project_bounds_sum(h - lr * grad, target)
            it += 1
            if it % 500 == 0:
                cnow = compute_c5(h)
                if cnow < best_c:
                    best_c, best_h = cnow, h.copy()
                if best_c <= 0.3808:
                    break

    best_h = _project_bounds_sum(best_h, target_sum)
    best_c = compute_c5(best_h)

    return best_h.tolist(), float(best_c), int(n_pts)
