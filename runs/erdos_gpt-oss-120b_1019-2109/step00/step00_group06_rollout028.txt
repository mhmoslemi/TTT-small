def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # compute correlation via np.correlate; output length 2n-1
    corr = np.correlate(h, 1 - h, mode='full')

    return np.max(corr) * dx

def hill_climb(h, max_iter, time_limit):
    n = len(h)
    S = np.sum(h)
    # ensure binary? we will enforce integer 0/1.
    # ensure sum = n/2.
    # compute current C.
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    for it in range(max_iter):
        # periodically check time
        if time.time() - start > time_limit:
            break
        # choose two indices i with h[i]=1, j with h[j]=0.
        # Use arrays of indices of ones and zeros.
        ones = np.where(h > 0.5)[0]  # binary
        zeros = np.where(h < 0.5)[0]
        # If none, break
        if len(ones)==0 or len(zeros)==0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        # compute new C
        new_c = compute_c5(h)
        # Accept if new_c <= cur_c or with some probability
        if new_c <= cur_c or np.random.rand() < np.exp((cur_c - new_c) / (0.01 + it * 0.0)):
            cur_c = new_c
            # Update best if improved
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
    return best_h, best_c

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: compute the Erdős C5 value for a candidate vector.
# Uses the pre‑imported evaluate_erdos_solution if available,
# otherwise falls back to the direct NumPy formula.
# ----------------------------------------------------------------------
def compute_c5(v):
    try:
        # The harness may provide a fast evaluator.
        return float(evaluate_erdos_solution(v))
    except Exception:
        a = np.asarray(v, dtype=float)
        n = a.size
        dx = 2.0 / n
        corr = np.correlate(a, 1 - a, mode='full')
        return float(np.max(corr) * dx)


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber for binary (0/1) step functions.
# Keeps exactly half the entries equal to 1.
# ----------------------------------------------------------------------
def hill_climb(start_vec, run_seconds, seed):
    np.random.seed(seed)
    h = np.array(start_vec, copy=True)
    n = h.size
    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    # linear temperature schedule
    T0, T1 = 0.6, 1e-4

    while time.time() - start < run_seconds:
        elapsed = time.time() - start
        prog = elapsed / run_seconds
        T = T0 * (1 - prog) + T1 * prog

        # pick a random 1 and a random 0 and swap them
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = h[j], h[i]

        new_val = compute_c5(h)

        # accept if better or with Metropolis probability
        if new_val < cur_val or np.random.rand() < math.exp((cur_val - new_val) / (T + 1e-12)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert the swap
            h[i], h[j] = h[j], h[i]

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Build an initial binary vector.
    # If the harness gave us an initial_h_values array, reuse it,
    # repeat it once to obtain an even length and force exactly
    # half the entries to be 1.
    # --------------------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        # binary mask (any value >0.5 becomes 1)
        base_bin = (base > 0.5).astype(float)

        # repeat once => even length, keeps density ≈½
        h0 = np.repeat(base_bin, 2)
        n = h0.size
        target = n // 2

        # enforce exact half‑ones constraint
        cur = int(h0.sum())
        diff = cur - target
        if diff > 0:                     # too many ones
            idx = np.where(h0 == 1)[0]
            if len(idx) >= diff:
                flip = np.random.choice(idx, diff, replace=False)
                h0[flip] = 0
        elif diff < 0:                   # too few ones
            idx = np.where(h0 == 0)[0]
            need = -diff
            if len(idx) >= need:
                flip = np.random.choice(idx, need, replace=False)
                h0[flip] = 1
    else:
        # fall back to a fresh random binary vector of moderate size
        n = 200
        h0 = np.zeros(n, dtype=float)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

    # --------------------------------------------------------------
    # Primary optimisation pass (majority of the time budget)
    # --------------------------------------------------------------
    time_left = max(0.0, budget_s - (time.time() - start_time))
    primary_time = time_left * 0.85
    best_h, best_val = hill_climb(h0, primary_time, seed + 1)

    # --------------------------------------------------------------
    # Optional extra random restarts if time permits.
    # --------------------------------------------------------------
    while True:
        now = time.time()
        remaining = budget_s - (now - start_time)
        if remaining < 8.0:
            break
        # each extra attempt gets a modest slice of the remaining budget
        attempt_time = min(20.0, remaining * 0.4)

        # generate a fresh random binary start of the same length
        tmp = np.zeros(n, dtype=float)
        tmp[: n // 2] = 1.0
        np.random.shuffle(tmp)

        cand_h, cand_val = hill_climb(tmp, attempt_time, seed + int(now))
        if cand_val < best_val:
            best_h, best_val = cand_h, cand_val

    # --------------------------------------------------------------
    # Final sanity check: enforce the exact integral constraint.
    # --------------------------------------------------------------
    n_points = best_h.size
    target_sum = n_points / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        # scale uniformly and clip – this keeps the bound valid.
        best_h = np.clip(best_h * (target_sum / cur_sum), 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return (best_h.tolist(), float(final_c5), n_points)
