def compute_c5(h):
    n = len(h)

    # use FFT over size 2*n maybe
    sz = 1 << (2*n-1).bit_length()   # next power of two

    L = 1 << ((2*n-1).bit_length())
    h_pad = np.fft.rfft(np.concatenate([h, np.zeros(L - n)]))
    g = (1 - h)[::-1]

    conv = np.fft.irfft(h_pad * g_pad, L)[:2*n-1]
    return conv.max() * (2.0 / n)

def improve_step(h):
    # compute correlation values:
    corr = np.correlate(h, 1 - h, mode='full')
    idx_max = np.argmax(corr)
    shift = idx_max - (len(h) - 1)
    n = len(h)
    if shift >= 0:
        i_range = np.arange(n - shift)
        # compute diff = h[i] - h[i+shift]
        diff = h[i_range] - h[i_range + shift]
        # find index with max |diff| positive diff
        pos_idx = np.where(diff > 1e-6)[0]
        if len(pos_idx) > 0:
            i = pos_idx[np.argmax(diff[pos_idx])]
            delta = min(h[i], 1 - h[i + shift]) * 0.5  # move half of possible
            h[i] -= delta
            h[i + shift] += delta
            return True
        else:
            # possibly negative diff: move opposite direction
            neg_idx = np.where(diff < -1e-6)[0]
            if len(neg_idx) > 0:
                i = neg_idx[np.argmin(diff[neg_idx])]  # most negative diff
                delta = min(h[i+shift], 1 - h[i]) * 0.5
                h[i] += delta
                h[i + shift] -= delta
                return True
    else:
        s = -shift
        i_range = np.arange(s, n)
        # overlapping i's: i = s..n-1, pair (i, i-s)
        diff = h[i_range] - h[i_range - s]
        # similar logic
        pos_idx = np.where(diff > 1e-6)[0]
        if len(pos_idx)>0:
            i = i_range[pos_idx[np.argmax(diff[pos_idx])]]
            delta = min(h[i], 1 - h[i - s]) * 0.5
            h[i] -= delta
            h[i - s] += delta
            return True

def _compute_c5(arr: np.ndarray) -> float:
    """Exact evaluation used by the harness."""
    n = arr.shape[0]
    # correlation of h with (1-h) (full, zero‑padded)
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(corr.max() * (2.0 / n))


def _random_move(h: np.ndarray) -> np.ndarray | None:
    """Transfer a random amount of mass from one index to another."""
    n = h.size
    i, j = np.random.randint(n, size=2)
    if i == j:
        return None
    # move from j to i
    max_delta = min(1.0 - h[i], h[j])
    if max_delta <= 0:
        # opposite direction
        max_delta = min(1.0 - h[j], h[i])
        if max_delta <= 0:
            return None
        delta = max_delta * random.random()
        h2 = h.copy()
        h2[j] += delta
        h2[i] -= delta
        return h2
    delta = max_delta * random.random()
    h2 = h.copy()
    h2[i] += delta
    h2[j] -= delta
    return h2


def _directed_move(h: np.ndarray) -> np.ndarray | None:
    """Improve the currently worst shift."""
    n = h.size
    # full correlation, locate the worst (largest) value
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)          # can be negative
    k = abs(shift)                # distance of the overlapping blocks
    if k == 0:
        return _random_move(h)

    # overlapping indices for the pair (i, i+k)
    i_range = np.arange(n - k)
    diff = h[i_range] - h[i_range + k]   # positive → move mass forward

    # try to push mass from larger to smaller
    pos = np.where(diff > 1e-12)[0]
    if pos.size:
        p = pos[np.argmax(diff[pos])]
        i = i_range[p]
        max_delta = min(h[i], 1.0 - h[i + k])
        if max_delta <= 0:
            return _random_move(h)
        delta = max_delta * random.random()
        h2 = h.copy()
        h2[i] -= delta
        h2[i + k] += delta
        return h2

    # otherwise move opposite direction
    neg = np.where(diff < -1e-12)[0]
    if neg.size:
        p = neg[np.argmin(diff[neg])]
        i = i_range[p]
        max_delta = min(h[i + k], 1.0 - h[i])
        if max_delta <= 0:
            return _random_move(h)
        delta = max_delta * random.random()
        h2 = h.copy()
        h2[i + k] -= delta
        h2[i] += delta
        return h2

    # all equal – fall back to a random move
    return _random_move(h)


def _project(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and enforce the integral constraint exactly."""
    n = h.size
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    cur = h.sum()
    if abs(cur - target) < 1e-12:
        return h
    diff = target - cur
    # simple redistribution loop – fast for the tiny residuals we have
    while abs(diff) > 1e-12:
        idx = np.random.randint(n)
        if diff > 0:
            inc = min(diff, 1.0 - h[idx])
            if inc <= 0:
                continue
            h[idx] += inc
            diff -= inc
        else:
            dec = min(-diff, h[idx])
            if dec <= 0:
                continue
            h[idx] -= dec
            diff += dec
    return h

    while True:
        elapsed = time.time() - start
        if elapsed >= limit:
            break
        # annealing temperature (linear decay)
        t_frac = elapsed / limit
        T = T0 * (1.0 - t_frac) + T_end * t_frac

        # mostly directed moves, occasional pure random moves
        if random.random() < 0.85:
            cand_h = _directed_move(cur_h)
        else:
            cand_h = _random_move(cur_h)

        if cand_h is None:
            continue

        cand_val = _compute_c5(cand_h)

        # Metropolis acceptance
        accept = False
        if cand_val < cur_val:
            accept = True
        else:
            prob = math.exp(-(cand_val - cur_val) / max(T, 1e-12))
            if random.random() < prob:
                accept = True

        if accept:
            cur_h = cand_h
            cur_val = cand_val
            if cand_val < best_val - 1e-12:
                best_val = cand_val
                best_h = cand_h.copy()

    # final projection to be absolutely safe
    best_h = _project(best_h)

    # final evaluation (use the harness helper if available)
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except NameError:
        final_c5 = _compute_c5(best_h)

    return best_h.tolist(), float(final_c5), n
