import numpy as np
import time
import math


def compute_cmax(h: np.ndarray) -> float:
    """Return the overlap constant C₅ for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def propose_move_worst_shift(h: np.ndarray, rng: np.random.RandomState):
    """
    Produce a new h by moving a little mass from the larger to the smaller entry
    of the pair that belongs to the shift giving the maximal overlap.
    Returns None if no feasible move is found.
    """
    n = h.shape[0]
    # raw correlation (without dx) – its maximum corresponds to the worst shift
    corr_raw = np.correlate(h, 1 - h, mode="full")
    worst_idx = int(np.argmax(corr_raw))
    s = worst_idx - (n - 1)          # shift value

    # indices forming the pairs for this shift
    if s >= 0:
        i_vals = np.arange(0, n - s)
        j_vals = i_vals + s
    else:
        i_vals = np.arange(-s, n)
        j_vals = i_vals + s

    if i_vals.size == 0:
        return None

    diff = h[i_vals] - h[j_vals]
    absdiff = np.abs(diff)
    if np.all(absdiff < 1e-12):
        return None

    max_idx = int(np.argmax(absdiff))
    i = int(i_vals[max_idx])
    j = int(j_vals[max_idx])
    d = diff[max_idx]

    if d > 0:
        # move from i (larger) to j (smaller)
        max_transfer = min(d, h[i], 1.0 - h[j])
        if max_transfer <= 0:
            return None
        delta = max_transfer * (0.5 + 0.5 * rng.rand())
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        # move from j to i
        max_transfer = min(-d, h[j], 1.0 - h[i])
        if max_transfer <= 0:
            return None
        delta = max_transfer * (0.5 + 0.5 * rng.rand())
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta

    return new_h


def random_pair_move(h: np.ndarray, rng: np.random.RandomState):
    """
    Transfer a random amount of mass between two arbitrary indices while
    respecting the [0,1] bounds and keeping the total sum unchanged.
    Returns None if no feasible transfer exists.
    """
    n = h.shape[0]
    i, j = rng.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    # delta added to i, subtracted from j
    lower = max(-hi, hj - 1.0)   # smallest delta keeping both in [0,1]
    upper = min(1.0 - hi, hj)    # largest such delta
    if upper <= lower:
        return None
    delta = lower + (upper - lower) * rng.rand()
    if abs(delta) < 1e-12:
        return None

    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # Clip merely for safety against rounding error
    new_h[i] = min(max(new_h[i], 0.0), 1.0)
    new_h[j] = min(max(new_h[j], 0.0), 1.0)
    return new_h
