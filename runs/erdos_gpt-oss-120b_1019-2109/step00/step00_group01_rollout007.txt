        h[i] = 0.0
        h[j] = 1.0

        new_val = np.max(np.correlate(h, 1-h, mode='full')) * dx

import time
import math
import numpy as np

def _project_capped_simplex(v, target_sum, upper=1.0, eps=1e-12, max_iter=500):
    """
    Euclidean projection of v onto the set {0 <= y <= upper, sum(y)=target_sum}.
    Simple water‑filling algorithm.
    """
    y = np.clip(v, 0.0, upper)
    for _ in range(max_iter):
        total = y.sum()
        if abs(total - target_sum) <= eps:
            break
        # variables that are not at a bound
        free = (y > eps) & (y < upper - eps)
        n_free = free.sum()
        if n_free == 0:
            break
        delta = (target_sum - total) / n_free
        y[free] += delta
        # enforce bounds again
        if np.any(y > upper):
            y[y > upper] = upper
        if np.any(y < 0):
            y[y < 0] = 0.0
    # final clipping (numerical safety)
    y = np.clip(y, 0.0, upper)
    # If the sum is slightly off, distribute the remainder uniformly on free vars
    resid = target_sum - y.sum()
    if abs(resid) > eps:
        free = (y > eps) & (y < upper - eps)
        if free.any():
            y[free] += resid / free.sum()
    return y

    """Return max_k ∫ h(x)(1‑h(x+k))dx for the current discretisation."""
    n = h.size

def _subgradient_opt(h0, max_iter, time_limit):
    """
    Sub‑gradient descent on the max‑overlap functional.
    Works with continuous h∈[0,1] and fixes the integral by projection.
    """
    h = h0.copy()
    n = h.size
    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_val, _ = _max_overlap(h)
    start = time.time()

    # diminishing step‑size schedule
    eta0 = 0.8
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break

        # compute the worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        cur_val = corr[max_idx] * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        # termination if already excellent
        if best_val <= 0.3808:
            break

        # shift corresponding to the worst lag
        L = max_idx - (n - 1)

        # build gradient of the active lag
        g = np.zeros_like(h)

        if L >= 0:
            i = np.arange(n - L)
            j = i + L
        else:
            i = np.arange(-L, n)
            j = i + L

        # first term: +1 on the overlapping indices
        g[i] += 1.0

        # second term: –h_{i+L} on the same i's (where defined)
        g[i] -= h[j]

        # third term: –h_{i} on the shifted indices
        g[j] -= h[i]

        # step size

        h_new = h - eta * g

        # project back to feasibility
        h = _project_capped_simplex(h_new, target_sum, upper=1.0)

    final_val, _ = _max_overlap(best_h)
    return best_h, final_val

        bad = i[(hi == 1.0) & (hj == 0.0)]       # 1→0 mismatches
        good = i[(hi == 0.0) & (hj == 1.0)]      # 0→1 matches

        if bad.size and good.size:
            a = rng.choice(bad)
            b = rng.choice(good)
        else:
            # fallback: random opposite‑value swap
            ones = np.flatnonzero(h == 1.0)
            zeros = np.flatnonzero(h == 0.0)
            a = rng.choice(ones)
            b = rng.choice(zeros)

        # perform swap
        h[a] = 0.0
        h[b] = 1.0

        new_val, _ = _max_overlap(h)
        if new_val <= cur_val + 1e-12:
            # keep if not worse
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[a] = 1.0
            h[b] = 0.0

        if best_val <= 0.3808:
            break

    final_val, _ = _max_overlap(best_h)
    return best_h, final_val


def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step function h on [0,2] (as a NumPy array),
    its associated C5 bound, and the number of discretisation points.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # choose a moderate resolution (even number for exact half‑mass)
    n_points = 300
    if n_points % 2 != 0:
        n_points += 1

    # ---- initial feasible vector ---------------------------------
    h = _initial_h(n_points, rng)

    # ---- phase 1 : sub‑gradient descent -------------------------
    time_for_sg = min(budget_s * 0.5, 400.0)   # leave time for later phases
    h, val = _subgradient_opt(h, max_iter=200000,
                              time_limit=time_for_sg)

    # ---- phase 2 : local swap improvement -----------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining > 1.0:
        h, val = _local_swap_improve(h, max_iter=200000,
                                     time_limit=remaining,
                                     rng=rng)

    # final evaluation
    C_max, _ = _max_overlap(h)

    # ensure feasibility one last time (tiny numerical drift)
    target_sum = n_points / 2.0
    h = _project_capped_simplex(h, target_sum, upper=1.0)

    return h, float(C_max), n_points
