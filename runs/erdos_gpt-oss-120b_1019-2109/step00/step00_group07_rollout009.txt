def refine(h, current_best_val, time_limit):
    start = time.time()
    h_cur = h.copy()
    best_val = current_best_val
    best_h = h_cur.copy()
    T0 = 0.01
    iter = 0
    while time.time() - start < time_limit:
        # compute objective & conv
        conv = np.correlate(h_cur, 1 - h_cur, mode='full')
        max_idx = np.argmax(conv)
        max_val = conv[max_idx] * dx
        # compute shift s
        s = max_idx - (n - 1)
        # compute weight w
        w = np.zeros(n)
        if s >= 0:
            w[:n - s] += h_cur[s:]
            w[s:] += h_cur[:n - s]
        else:
            s_abs = -s
            w[s_abs:] += h_cur[:n - s_abs]
            w[:n - s_abs] += h_cur[s_abs:]
        # candidate indices
        # Choose i with high w*(1 - h)
        scores_i = w * (1 - h_cur)
        # Choose j with high h*w? Actually low w*(h?), but we aim to decrease h where w is low.
        scores_j = (1 - w) * h_cur
        # sample i and j from distributions
        if scores_i.max() > 0:
            i = np.random.choice(n, p=scores_i / scores_i.sum())
        else:
            i = np.random.randint(0, n)
        if scores_j.max() > 0:
            j = np.random.choice(n, p=scores_j / scores_j.sum())
        else:
            j = np.random.randint(0, n)
        # Ensure i!=j and feasible epsilon
        if i == j:
            continue
        max_eps = min(0.1, 1 - h_cur[i], h_cur[j])
        if max_eps <= 1e-8:
            continue
        epsilon = np.random.rand() * max_eps
        h_new = h_cur.copy()
        h_new[i] += epsilon
        h_new[j] -= epsilon
        # Clip (should stay in bounds)
        # Evaluate
        val_new = compute_obj(h_new)
        # Accept if better or with prob
        if val_new < max_val:
            h_cur = h_new
            if val_new < best_val:
                best_val = val_new
                best_h = h_new.copy()

            if T > 0 and np.random.rand() < np.exp(-(val_new - max_val) / T):
                h_cur = h_new
        iter += 1
    return best_h, best_val

def refine_sequence(h, time_budget, rng):
    n = len(h)
    start = time.time()
    best_h = h.copy()
    best_val = evaluate_erdos_solution(h)  # C5
    cur_h = h.copy()
    cur_val = best_val
    # temperature schedule
    while time.time() - start < time_budget:
        # compute worst shift
        raw = np.correlate(cur_h, 1 - cur_h, mode='full')
        idx = np.argmax(raw)
        s = idx - (n - 1)
        # compute weight w
        w = np.zeros(n)
        if s >= 0:
            w[:n - s] += cur_h[s:]
            w[s:] += cur_h[:n - s]
        else:
            s_abs = -s
            w[s_abs:] += cur_h[:n - s_abs]
            w[:n - s_abs] += cur_h[s_abs:]
        # choose i and j candidates
        # scores for i: high w and not full
        scores_i = w * (1 - cur_h)
        if scores_i.sum() > 0:

import time
import numpy as np
import itertools

# The evaluation helper is provided by the harness.
# It expects a 1‑D array h and returns C₅.
# We will also use direct correlation for internal logic.

def _objective(h):
    """Return C₅ for the candidate h."""
    return evaluate_erdos_solution(h)


def _worst_shift_and_raw(h):
    """Return (shift, raw_correlation_array, max_raw) for h."""
    raw = np.correlate(h, 1 - h, mode="full")
    idx = int(np.argmax(raw))
    shift = idx - (len(h) - 1)
    return shift, raw, raw[idx]


def _refine_sequence(h, time_budget, rng):
    """Greedy / simulated‑annealing improvement of h within time_budget seconds."""
    n = h.shape[0]
    dx = 2.0 / n
    start = time.time()
    best_h = h.copy()
    raw = np.correlate(best_h, 1 - best_h, mode="full")
    best_val = raw.max() * dx
    cur_h = h.copy()

    while time.time() - start < time_budget:
        # worst shift for the current candidate
        raw = np.correlate(cur_h, 1 - cur_h, mode="full")
        idx = int(np.argmax(raw))
        max_raw = raw[idx]
        cur_val = max_raw * dx
        s = idx - (n - 1)

        # weight = sum of partner values for this shift
        w = np.zeros(n)
        if s >= 0:
            w[: n - s] += cur_h[s:]
            w[s:] += cur_h[: n - s]
        else:
            s_abs = -s
            w[s_abs:] += cur_h[: n - s_abs]
            w[: n - s_abs] += cur_h[s_abs:]

        # pick an index to increase (high partner, not saturated)
        scores_i = w * (1.0 - cur_h)
        sum_i = scores_i.sum()
        if sum_i > 0:
            i = rng.choice(n, p=scores_i / sum_i)
        else:
            i = rng.integers(0, n)

        # pick an index to decrease (low partner, not empty)
        scores_j = (1.0 - w) * cur_h
        sum_j = scores_j.sum()
        if sum_j > 0:
            j = rng.choice(n, p=scores_j / sum_j)
        else:
            j = rng.integers(0, n)

        if i == j:
            continue

        # amount to move while staying feasible
        max_eps = min(0.05, 1.0 - cur_h[i], cur_h[j])
        if max_eps <= 1e-12:
            continue
        eps = rng.random() * max_eps

        new_h = cur_h.copy()
        new_h[i] += eps
        new_h[j] -= eps

        new_raw = np.correlate(new_h, 1 - new_h, mode="full")
        new_val = new_raw.max() * dx

        if new_val < cur_val:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()
        else:
            # occasional uphill move (simulated annealing)
            elapsed = time.time() - start
            T = 0.01 * max(0.0, 1.0 - elapsed / time_budget)
            if T > 0.0 and rng.random() < np.exp(-(new_val - cur_val) / T):
                cur_h = new_h
                cur_val = new_val

    return best_h, best_val


def _enumerate_binary_patterns(n):
    """Search over binary patterns with exact half‑ones for some divisors of n."""
    best_h = None
    best_val = np.inf
    for L in (8, 10, 16):
        if n % L != 0:
            continue
        repeats = n // L
        half = L // 2
        # enumerate all vectors of length L having exactly L/2 ones
        for ones in itertools.combinations(range(L), half):
            base = np.zeros(L, dtype=np.float64)
            base[list(ones)] = 1.0
            h = np.tile(base, repeats)
            val = _objective(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) respecting the time budget."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Load or create an initial feasible vector
    # ------------------------------------------------------------------

        h_initial = np.array(initial_h_values, dtype=np.float64)

    except NameError:
        n = 80

        h_initial *= (n / 2.0) / h_initial.sum()

    target_sum = n / 2.0
    h_initial = np.clip(h_initial, 0.0, 1.0)
    s = h_initial.sum()
    if abs(s - target_sum) > 1e-9:
        # simple scaling then a small correction
        h_initial *= target_sum / s
        h_initial = np.clip(h_initial, 0.0, 1.0)
        diff = target_sum - h_initial.sum()
        if abs(diff) > 1e-9:
            free = (h_initial > 0.0) & (h_initial < 1.0)
            cnt = max(1, free.sum())
            h_initial[free] += diff / cnt
            h_initial = np.clip(h_initial, 0.0, 1.0)

    best_h = h_initial.copy()
    best_val = _objective(best_h)

    start_time = time.time()

    # ------------------------------------------------------------------
    # 2. Exhaustive search over small periodic binary patterns
    # ------------------------------------------------------------------
    h_bin, val_bin = _enumerate_binary_patterns(n)
    if h_bin is not None and val_bin < best_val:
        best_h, best_val = h_bin, val_bin

    # ------------------------------------------------------------------
    # 3. Random binary restarts with local refinement
    # ------------------------------------------------------------------
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 1.0  # tiny safety margin
    while remaining > 6.0:
        # generate a random half‑filled binary vector
        idx = rng.choice(n, size=n // 2, replace=False)
        h_rand = np.zeros(n, dtype=np.float64)
        h_rand[idx] = 1.0

        cur_h, cur_val = _refine_sequence(h_rand, t_refine, rng)
        if cur_val < best_val:
            best_h, best_val = cur_h, cur_val

        elapsed = time.time() - start_time

    # ------------------------------------------------------------------
    # 4. Final refinement of the best candidate with remaining time
    # ------------------------------------------------------------------
    final_time = max(0.0, budget_s - (time.time() - start_time) - 0.5)

    # ------------------------------------------------------------------
    # 5. Final feasibility tidy‑up and return
    # ------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)

    if abs(diff) > 1e-9:

        best_h[free] += diff / cnt
        best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = _objective(best_h)
    return best_h, float(final_c5), n
