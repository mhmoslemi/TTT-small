import time
import numpy as np

def _max_overlap(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    # np.correlate implements the discrete version of the integral.
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    #  Initialise h – use the provided construction if available.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        n = 80
        h = np.full(n, 0.5, dtype=np.float64)   # sum = n/2 automatically

    n = h.shape[0]
    target_sum = n / 2.0

    # Project onto the feasible set (sum = n/2, 0≤h≤1)
    cur = h.sum()
    if abs(cur - target_sum) > 1e-12:
        h *= target_sum / cur
        h = np.clip(h, 0.0, 1.0)
        diff = h.sum() - target_sum
        if abs(diff) > 1e-12:
            idx = rng.integers(0, n)
            h[idx] -= diff
            h[idx] = np.clip(h[idx], 0.0, 1.0)

    # ------------------------------------------------------------------
    #  Local optimisation – transfer mass between entries while keeping
    #  the total integral fixed.
    # ------------------------------------------------------------------
    cur_val = _max_overlap(h)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    step = 0.2          # maximal transferred amount per move
    idle = 0            # counts iterations without improvement

    while time.time() - start < budget_s * 0.95:
        # current worst shift (the one attaining the maximum overlap)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = int(np.argmax(corr))
        cur_val = corr[max_idx] * (2.0 / n)
        L = max_idx - (n - 1)               # integer shift producing the max

        improved = False

        # ---- directed move: fix the worst lag by moving mass from i → i+L ----
        if L >= 0:
            i_vals = np.arange(n - L)
        else:
            i_vals = np.arange(-L, n)
        j_vals = i_vals + L

        feasible = (h[i_vals] > 1e-12) & (h[j_vals] < 1 - 1e-12)
        if np.any(feasible):
            # Score = reduction of the overlap term for this pair
            scores = (h[i_vals] + (1 - h[j_vals])) * feasible
            best_candidates = np.flatnonzero(scores == scores.max())
            local = rng.choice(best_candidates)
            i, j = i_vals[local], j_vals[local]

            eps_max = min(step, h[i], 1 - h[j])
            if eps_max > 0:
                eps = rng.random() * eps_max
                if eps > 0:
                    h_new = h.copy()
                    h_new[i] -= eps
                    h_new[j] += eps
                    val = _max_overlap(h_new)
                    if val < cur_val - 1e-13:
                        h = h_new
                        cur_val = val
                        improved = True
                        if val < best_val - 1e-13:
                            best_val = val
                            best_h = h.copy()
                            idle = 0

        # ---- random transfer move (fallback) ----
        if not improved:
            a = rng.integers(0, n)
            b = rng.integers(0, n - 1)
            if b >= a:
                b += 1

            # decide direction that respects the bounds
            if h[a] > 1e-12 and h[b] < 1 - 1e-12:
                src, dst = a, b
            elif h[b] > 1e-12 and h[a] < 1 - 1e-12:
                src, dst = b, a
            else:
                src = dst = None

            if src is not None:
                eps_max = min(step, h[src], 1 - h[dst])
                if eps_max > 0:
                    eps = rng.random() * eps_max
                    h_new = h.copy()
                    h_new[src] -= eps
                    h_new[dst] += eps
                    val = _max_overlap(h_new)
                    if val < cur_val - 1e-13:
                        h = h_new
                        cur_val = val
                        improved = True
                        if val < best_val - 1e-13:
                            best_val = val
                            best_h = h.copy()
                            idle = 0

        # ---- bookkeeping ----
        if not improved:
            idle += 1
            if idle > 5000:
                step = max(step * 0.5, 1e-4)
                idle = 0

        if best_val <= 0.3808:
            break

    # ------------------------------------------------------------------
    #  Final feasibility tweak (tiny rounding errors)
    # ------------------------------------------------------------------
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-12:
        idx = rng.integers(0, n)
        best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)

    final_val = _max_overlap(best_h)
    return (best_h, float(final_val), n)
