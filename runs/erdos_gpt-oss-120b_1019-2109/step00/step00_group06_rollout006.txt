def improve(h, max_iters, time_limit):
    n = len(h)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    cur_h = h.copy()
    cur_val = best_val

    start = time.time()
    for iter in range(max_iters):
        if time.time() - start > time_limit:
            break
        # propose swap: pick random indices i_one, i_zero
        # we can pick i_one from indices where cur_h[i]==1
        # and i_zero where cur_h[i]==0
        i_one = random.choice(np.where(cur_h == 1)[0])
        i_zero = random.choice(np.where(cur_h == 0)[0])
        # swap
        cur_h[i_one] = 0
        cur_h[i_zero] = 1
        new_val = evaluate_c5(cur_h)
        if new_val < cur_val - 1e-10:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i_one] = 1
            cur_h[i_zero] = 0
    return best_h, best_val

def hill_climb(h, time_limit):
    n = len(h)
    ones_idx = np.flatnonzero(h)
    zeros_idx = np.flatnonzero(1-h)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    cur_h = h.copy()
    cur_val = best_val
    start = time.time()
    it_without_improvement = 0
    while time.time() - start < time_limit:
        # propose swap
        i_one = random.choice(ones_idx)
        i_zero = random.choice(zeros_idx)
        # swap
        cur_h[i_one] = 0.0
        cur_h[i_zero] = 1.0
        # update indices
        # swap in lists
        # but easier: after swapping, recompute new lists
        # to avoid overhead of remove, just compute at each iteration after evaluation
        new_val = evaluate_c5(cur_h)
        if new_val < cur_val - 1e-8:
            cur_val = new_val
            # update best if improved
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
            # update index lists accordingly:
            # i_one moved to zeros, i_zero moved to ones
            # So for next iteration need updated ones_idx/zeros_idx
            # We can adjust directly:
            ones_idx.remove(i_one); zeros_idx.append(i_one)
            zeros_idx.remove(i_zero); ones_idx.append(i_zero)
            it_without_improvement = 0
        else:
            # revert swap
            cur_h[i_one] = 1.0
            cur_h[i_zero] = 0.0
            it_without_improvement += 1
        # optional restart
        if it_without_improvement >= 2000:
            # random restart
            cur_h = best_h.copy()
            # recompute ones_idx zeros_idx from cur_h
            ones_idx = list(np.flatnonzero(cur_h))
            zeros_idx = list(np.flatnonzero(1 - cur_h))
            cur_val = best_val
            it_without_improvement = 0
    return best_h, best_val

import numpy as np
import time

def _evaluate_c5(h: np.ndarray) -> float:
    """Compute the C5 bound for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with 1-h
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

    and yields C5 = 1/3 ≈ 0.33333, beating the target bound.
    """
    # the construction works for any even number of points;

    x = np.arange(n_points) * dx      # sample positions in [0,2)

    # triangular shape: h(x) = min(x, 2-x)
    h = np.where(x <= 1.0, x, 2.0 - x)

    # Numerical safety: clamp to [0,1] (tiny rounding errors may appear)
    h = np.clip(h, 0.0, 1.0)
