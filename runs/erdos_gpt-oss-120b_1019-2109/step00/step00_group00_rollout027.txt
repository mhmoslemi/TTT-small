import numpy as np, time, math

def project_onto_capped_simplex(v, S):
    # binary search λ
    lo = -1.0
    hi = 1.0
    for _ in range(60):
        lam = (lo+hi)/2
        w = np.clip(v - lam, 0, 1)
        if w.sum() > S:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

def gradient_for_shift(h, s):
    n = len(h)
    g = np.zeros_like(h)
    if s >= 0:
        g[:n - s] += 1 - h[s:]
        g[s:] -= h[:n - s]
    else:
        s_abs = -s
        g[s_abs:] += 1 - h[:n - s_abs]
        g[:n - s_abs] -= h[s_abs:]
    return g

def smooth_max_gradient(h, beta):
    n = len(h)
    f = np.correlate(h, 1 - h, mode='full')  # not multiplied by dx yet; but scaling preserving ordering.
    # f vector length 2n-1
    max_f = f.max()
    # shift for numerical stability
    exp_f = np.exp(beta * (f - max_f))
    w = exp_f / exp_f.sum()
    # compute weighted gradient
    grad = np.zeros_like(h)
    # shift offset = -(n-1)
    offset = -(n - 1)
    for idx, weight in enumerate(w):
        if weight == 0:
            continue
        s = idx + offset  # shift value
        # compute gradient for this s scaled by weight
        # we could compute using slice operations as per gradient_for_shift
        if s >= 0:
            grad[:n - s] += weight * (1 - h[s:])
            grad[s:] -= weight * h[:n - s]
        else:
            s_abs = -s
            grad[s_abs:] += weight * (1 - h[:n - s_abs])
            grad[:n - s_abs] -= weight * h[s_abs:]
    return grad, f

import time
import math
import random
import numpy as np

def project_onto_capped_simplex(v: np.ndarray, S: float) -> np.ndarray:
    """Project v onto the set {0<=x_i<=1, sum x_i = S}."""
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        if w.sum() > S:
            lo = lam
        else:
            hi = lam
    return np.clip(v - ((lo + hi) * 0.5), 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function represented by h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def weighted_gradient(h: np.ndarray, beta: float):
    """Smooth‑max gradient of the overlap objective."""
    n = h.shape[0]
    f = np.correlate(h, 1.0 - h, mode="full")          # without dx, same ordering
    max_f = f.max()
    exp_f = np.exp(beta * (f - max_f))
    w = exp_f / exp_f.sum()

    grad = np.zeros_like(h)
    offset = -(n - 1)                                   # index shift corresponding to lag 0
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx + offset
        if s >= 0:
            grad[: n - s] += weight * (1.0 - h[s:])
            grad[s:]       -= weight * h[: n - s]
        else:
            s_abs = -s
            grad[s_abs:]   += weight * (1.0 - h[: n - s_abs])
            grad[: n - s_abs] -= weight * h[s_abs:]
    return grad, f


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    time_limit = budget_s - 0.2   # safety margin

    # ----------------------------------------------------------------------
    # Choose a working discretisation size.
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n0 = init.size
        # aim for ~200–250 points; pick a small integer factor
        factor = max(2, int(round(225 / n0)))
        n = n0 * factor
        h = np.repeat(init, factor)
        # the repeated vector already satisfies the integral constraint
    else:
        n = 200
        h = np.random.rand(n)

    target_sum = n / 2.0
    h = project_onto_capped_simplex(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # ----------------------------------------------------------------------
    # Optimisation parameters
    # ----------------------------------------------------------------------
    beta = 1.0
    beta_max = 60.0
    beta_step = 200          # increase beta every this many inner iterations
    lr0 = 0.5                # initial learning rate
    inner_iters = 4000       # maximum inner iterations per restart
    max_restarts = 20        # total number of random restarts

    # ----------------------------------------------------------------------
    # Main optimisation loop (multiple restarts)
    # ----------------------------------------------------------------------
    restarts = 0
    while time.time() - start_time < time_limit and restarts < max_restarts:

        if restarts > 0:
            h = np.random.rand(n)
            h = project_onto_capped_simplex(h, target_sum)

        beta = 1.0
        for it in range(inner_iters):
            now = time.time()
            if now - start_time > time_limit:
                break

            # gradient of smooth max
            grad, f_vals = weighted_gradient(h, beta)

            # learning rate schedule
            lr = lr0 / math.sqrt(it + 1.0)

    # Return results
    # ----------------------------------------------------------------------
    h_values = best_h.tolist()
    c5_bound = float(best_c5)
