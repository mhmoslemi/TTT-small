def project_onto_simplex_box(v, target):
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        lam = (lo + hi) / 2.0
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam

import numpy as np, time

def project_onto_simplex_box(v, target):
    # binary search lambda to enforce sum and box constraints
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(100):
        lam = (lo + hi) / 2.0
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo+hi)/2.0, 0.0, 1.0)

import time
import numpy as np

def _project_to_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project a vector onto the set {0 <= x_i <= 1, sum x_i = target_sum}
    using a simple bisection on the Lagrange multiplier.
    """
    lo = v.min() - 1.0          # low enough to make all entries 1 after clipping
    hi = v.max()                # high enough to make all entries 0 after clipping
    for _ in range(80):
        lam = (lo + hi) * 0.5
        y = np.clip(v - lam, 0.0, 1.0)
        s = y.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
