import time, math, random
import numpy as np

def _c5_value(h):
    """Return C5 for a step vector h."""
    n = len(h)
    # full correlation of h and (1-h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def _proj_sum_box(v, target):
    """
    Euclidean projection of vector v onto the set
        {x | 0 <= x_i <= 1, sum x_i = target }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0
    for _ in range(30):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

        g = np.zeros_like(h)
        if k > 0:
            i = np.arange(0, n - k)
            j = i + k
        else:
            s = -k
            i = np.arange(s, n)
            j = i - s
        g[i] = 1.0 - h[j]
        g[j] = -h[i]

        # diminishing step size
        step = init_step / math.sqrt(it)
        h_next = _proj_sum_box(h - step * g, target)
        val = _c5_value(h_next)

        if val < best_val - 1e-12:
            best_val = val
            best_h = h_next.copy()
        h = h_next
    return best_h, best_val


def _hill_climb_binary(h0, max_iter, deadline):
    """
    Simple hill‑climbing on binary vectors (0/1) that swaps a 1 with a 0.
    Guided moves are chosen using the currently worst shift.
    """
    n = len(h0)
    h = h0.copy()
    # enforce strict binary values
    h = np.where(h > 0.5, 1.0, 0.0)
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    cur_val = _c5_value(h)
    best_h, best_val = h.copy(), cur_val

    it = 0
    while it < max_iter and time.time() < deadline:
        it += 1
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = corr.argmax()
        shift = max_idx - (n - 1)

        # try a guided swap
        i = j = None
        if shift != 0:
            if shift > 0:
                bad_one = [idx for idx in ones if idx + shift < n and h[idx + shift] == 0.0]
                bad_zero = [idx for idx in zeros if idx + shift < n and h[idx + shift] == 1.0]
            else:
                s = -shift
                bad_one = [idx for idx in ones if idx - s >= 0 and h[idx - s] == 0.0]
                bad_zero = [idx for idx in zeros if idx - s >= 0 and h[idx - s] == 1.0]
            if bad_one and bad_zero:
                i = random.choice(bad_one)
                j = random.choice(bad_zero)

        if i is None:          # fallback random swap
            i = random.choice(ones)
            j = random.choice(zeros)

        # perform swap
        h[i], h[j] = h[j], h[i]

        new_val = _c5_value(h)
        if new_val + 1e-12 < cur_val:          # improvement
            cur_val = new_val
            # update index lists
            ones.remove(i); zeros.append(i)
            zeros.remove(j); ones.append(j)
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
        else:                                 # revert
            h[i], h[j] = h[j], h[i]

        # early exit if already excellent
        if best_val <= 0.3808:
            break
    return best_h, best_val


def _random_binary(n):
    """Return a binary vector of length n with exactly n/2 ones (or floor+0.5 if odd)."""
    if n % 2 == 0:
        ones = np.random.choice(n, n // 2, replace=False)
        h = np.zeros(n, dtype=float)
        h[ones] = 1.0
        return h
    # odd length – use a single 0.5 entry to keep the average 0.5
    m = n // 2
    h = np.zeros(n, dtype=float)
    ones = np.random.choice(n, m, replace=False)
    h[ones] = 1.0
    remaining = np.setdiff1d(np.arange(n), ones)
    h[remaining[0]] = 0.5
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] that minimizes the
    Erdős C_5 overlap integral. Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s

    # ------------------------------------------------------------------
    # Initial candidate (if supplied)
    # ------------------------------------------------------------------
    init_vec = globals().get('initial_h_values', None)
    if init_vec is not None:
        h_best = np.array(init_vec, dtype=float).copy()
        n_best = len(h_best)
        # Ensure exact sum constraint (tiny rounding errors may exist)
        target = n_best * 0.5
        diff = target - h_best.sum()
        if abs(diff) > 1e-9:
            h_best[0] = np.clip(h_best[0] + diff, 0.0, 1.0)
        c5_best = _c5_value(h_best)
    else:
        # start from a small random binary pattern
        n_best = 46
        h_best = _random_binary(n_best)
        c5_best = _c5_value(h_best)

    # ------------------------------------------------------------------
    # Parameterisation of the search
    # ------------------------------------------------------------------
    candidate_ns = [46, 48, 50, 52, 56, 60, 64, 68, 72, 80]
    # ensure the initial length is also tried
    if init_vec is not None and len(init_vec) not in candidate_ns:
        candidate_ns.insert(0, len(init_vec))

    # Main loop: keep trying until time runs out
    while time.time() < deadline - 5.0:   # leave a small safety margin
        n = random.choice(candidate_ns)

        # --------------------------------------------------------------
        # Phase 1 – binary hill climbing
        # --------------------------------------------------------------
        h0 = _random_binary(n)
        # occasionally start from the current global best (rescaled)
        if random.random() < 0.15 and n == n_best:
            h0 = h_best.copy()

        h_loc, val_loc = _hill_climb_binary(h0, max_iter=4000, deadline=deadline)

        # --------------------------------------------------------------
        # Phase 2 – continuous refinement (projected sub‑gradient)
        # --------------------------------------------------------------
        h_ref, val_ref = _subgrad_optimize(h_loc, max_iter=3000,
                                           init_step=0.5, deadline=deadline)

        # --------------------------------------------------------------
        # Record improvement
        # --------------------------------------------------------------
        if val_ref < c5_best - 1e-9:
            h_best, c5_best, n_best = h_ref, val_ref, n

        # early exit if we beat the target comfortably
        if c5_best <= 0.3808:
            break

    # Final safety projection (exact sum)
    target = n_best * 0.5
    diff = target - h_best.sum()
    if abs(diff) > 1e-9:
        h_best[0] = np.clip(h_best[0] + diff, 0.0, 1.0)

    final_c5 = _c5_value(h_best)
    return (h_best, final_c5, n_best)
