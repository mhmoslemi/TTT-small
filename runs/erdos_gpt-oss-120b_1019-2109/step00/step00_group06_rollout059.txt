def project_onto_bounds_and_sum(v, target_sum):
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

def gradient_step(h, target_sum, step_max):
    n = len(h)
    auto = np.correlate(h, h, mode='full')
    min_val = auto.min()
    # indices of near-min values
    # tolerance as 1e-8 absolute or relative
    eps = 1e-6
    idxs = np.where(auto <= min_val + eps)[0]
    # accumulate gradient
    grad = np.zeros_like(h)

        if k >= 0:

import numpy as np
import time

def _project_onto_constraints(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the box [0,1]^n and the hyperplane sum(v)=target_sum."""
    # binary search for a scalar lambda such that sum(clip(v+lambda,0,1)) == target_sum
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_from_h(h: np.ndarray) -> float:
    """Return the C5 value for a step vector h."""
    n = h.shape[0]
    auto = np.correlate(h, h, mode="full")
    min_auto = auto.min()
    return 1.0 - (2.0 / n) * min_auto


def _gradient_step(h: np.ndarray, target_sum: float, max_change: float) -> np.ndarray:
    """
    Perform one sub‑gradient step that tries to raise the minimal autocorrelation.
    max_change caps the absolute change of any coordinate.
    """
    n = h.shape[0]
    auto = np.correlate(h, h, mode="full")
    min_val = auto.min()
    # indices where the autocorrelation is (almost) minimal
    eps = 1e-10
    idxs = np.where(auto <= min_val + eps)[0]

    grad = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)          # shift corresponding to this autocorrelation entry
        if k >= 0:
            # forward part
            grad[: n - k] += h[k:]
            # backward part
            grad[k:] += h[: n - k]
        else:
            kk = -k
            grad[kk:] += h[: n - kk]
            grad[: n - kk] += h[kk:]

    # scale so that the largest absolute step does not exceed max_change
    max_abs = np.max(np.abs(grad))
    if max_abs > 0:
        scale = max_change / max_abs
    else:
        scale = 0.0
    h_new = h + scale * grad
    # enforce constraints
    return _project_onto_constraints(h_new, target_sum)


def _improve_h(initial_h: np.ndarray,
               target_sum: float,
               time_limit: float,
               start_time: float) -> (np.ndarray, float):
    """Run a simple sub‑gradient descent within the time budget."""
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = _c5_from_h(h)
    cur_c5 = best_c5
    max_change = 0.03           # initial maximal per‑coordinate move
    it = 0
    while time.time() - start_time < time_limit:
        h_next = _gradient_step(h, target_sum, max_change)
        new_c5 = _c5_from_h(h_next)

        # accept improvements always, occasional uphill moves to escape flat regions
        if new_c5 < cur_c5 or np.random.rand() < 0.001:
            h, cur_c5 = h_next, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = h_next.copy(), new_c5

        # slowly decay the step size
        if it % 1500 == 0 and it > 0:
            max_change *= 0.9
        it += 1
    return best_h, best_c5


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize the Erdős minimum overlap step function.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        # fallback: a modest size with a smooth start

        h0 = rng.random(n)
        h0 = np.clip(h0, 0.0, 1.0)
        target = n / 2.0
        h0 = _project_onto_constraints(h0, target)

    n = h0.shape[0]
    target_sum = n / 2.0
    h0 = _project_onto_constraints(h0, target_sum)

    start = time.time()
    # First phase: deterministic sub‑gradient descent
    half_budget = budget_s * 0.55
    best_h, best_c5 = _improve_h(h0, target_sum,

    while time.time() - start < budget_s * 0.95:
