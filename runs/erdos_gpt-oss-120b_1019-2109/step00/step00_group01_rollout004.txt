    best_h, best_val = None, np.inf
    for it in range(max_iters):
        # compute conv etc.
        conv = np.correlate(h, 1-h, mode='full')
        dx = 2.0 / len(h)
        max_val = conv.max() * dx
        if max_val < best_val:
            best_val = max_val
            best_h = h.copy()
        # compute gradient
        j_max = conv.argmax()
        offset = j_max - (len(h)-1)
        grad = np.zeros_like(h)

            grad[:N-d] += (1 - h[d:])

def subgradient(h):
    N = len(h)
    conv = np.correlate(h, 1 - h, mode='full')
    max_val = conv.max()
    # indices close to max within tolerance
    eps = max_val * 1e-12 + 1e-12 # careful for zero
    idxs = np.where(conv >= max_val - eps)[0]
    # compute gradient as average over these indices
    grad = np.zeros(N)
    for j in idxs:
        offset = j - (N-1)
        if offset >=0:
            d = offset
            grad[:N-d] += (1 - h[d:])
            grad[d:] -= h[:N-d]
        else:
            d = -offset
            grad[d:] += (1 - h[:N-d])
            grad[:N-d] -= h[d:]
    grad /= len(idxs)
    # The gradient of conv w.r.t h, but objective is C5 = max(conv)*dx; dx constant factor, not needed for direction.
    return grad

def optimize_h(h_init, max_iters, time_budget, start_time, best_global_val, best_global_h):
    h = h_init.copy()
    N = len(h)
    S = N/2
    lr0 = 0.5
    for it in range(max_iters):
        # compute current val
        conv = np.correlate(h, 1-h, mode='full')
        cur_val = conv.max() * (2.0/N)
        if cur_val < best_global_val:
            best_global_val = cur_val
            best_global_h = h.copy()
        # check time
        if time.time() - start_time > time_budget*0.95:  # leave margin
            break
        # compute subgradient
        max_val = conv.max()
        eps = 1e-12 + max_val*1e-12
        idxs = np.where(conv >= max_val - eps)[0]  # might include many with equal max
        grad = np.zeros(N)
        for j in idxs:
            offset = j - (N-1)
            if offset >= 0:
                d = offset
                grad[:N-d] += (1 - h[d:])
                grad[d:] -= h[:N-d]
            else:

                grad[:N-d] -= h[d:]
        grad /= len(idxs)
        # step

        h = project(h - lr*grad, S)
    return best_global_h, best_global_val

def project(v, S):
    # find λ via binary search
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(30):
        mid = (lo + hi) / 2.0
        w = np.clip(v + mid, 0.0, 1.0)
        total = w.sum()
        if total > S:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # adjust small numeric error
    # If sum deviates slightly, apply a small correction uniformly on non-bound entries
    diff = w.sum() - S
    if abs(diff) > 1e-9:
        # compute indices not at bounds
        free = (w > 1e-12) & (w < 1 - 1e-12)
        cnt = free.sum()
        if cnt > 0:
            adjustment = diff / cnt
            w[free] -= adjustment
            w = np.clip(w, 0.0, 1.0)

            # compute final sum error; but we assume it's small.
    return w

        i, j = np.random.randint(0, N, size=2)
        if i == j:

        # propose delta moving from i to j
        max_delta = min(best_h[i], 1 - best_h[j])
        if max_delta <= 0:

            continue
        delta = np.random.rand() * max_delta
        new_h = best_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        val = compute_c5(new_h)
        if val < best_val:
            best_h = new_h
            best_val = val
        steps += 1
    return best_h, best_val

import time
import numpy as np

def _project(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto {0<=x<=1, sum(x)=target_sum} by a simple binary search on the shift."""
    lo = -np.max(v) - 1.0        # forces all entries to 0
    hi = 1.0 - np.min(v) + 1.0   # forces all entries to 1
    for _ in range(30):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = w.sum() - target_sum
    if abs(diff) > 1e-10:
        free = (w > 1e-12) & (w < 1.0 - 1e-12)
        cnt = free.sum()
        if cnt:
            w[free] -= diff / cnt
            w = np.clip(w, 0.0, 1.0)
    return w


def _c5_value(h: np.ndarray) -> float:
    """C5 for a discretised step function."""
    n = h.size
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx


def _subgradient(h: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """Average sub‑gradient of the max‑correlation at the given index positions."""
    n = h.size
    grad = np.zeros_like(h)
    for j in idxs:
        off = j - (n - 1)
        if off >= 0:
            d = off
            grad[: n - d] += 1.0 - h[d:]
            grad[d:]      -= h[: n - d]
        else:
            d = -off
            grad[d:]      += 1.0 - h[: n - d]
            grad[: n - d] -= h[d:]
    grad /= idxs.size
    return grad


def _gradient_descent(h0: np.ndarray, iters: int,
                     budget: float, start: float) -> tuple:
    """Projected sub‑gradient descent with a diminishing step."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(h)
    base_lr = 0.8
    for it in range(iters):
        if time.time() - start > budget * 0.95:
            break
        conv = np.correlate(h, 1.0 - h, mode='full')
        cur_val = conv.max() * (2.0 / n)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # indices attaining (or almost attaining) the maximum
        mx = conv.max()
        eps = 1e-12
        idxs = np.where(conv >= mx - eps)[0]
        g = _subgradient(h, idxs)
        lr = base_lr / np.sqrt(it + 1.0)
        h = _project(h - lr * g, target)
    # final check
    val = _c5_value(h)
    if val < best_val:
        best_val = val
        best_h = h.copy()
    return best_h, best_val


def _local_search(h0: np.ndarray, moves: int,
                  budget: float, start: float) -> tuple:
    """Simple hill‑climbing that moves mass between two coordinates."""
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _c5_value(h)
    for _ in range(moves):
        if time.time() - start > budget * 0.95:
            break
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # move a random amount from i to j
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0.0:
            continue
        delta = np.random.rand() * delta_max
        cand = h.copy()
        cand[i] -= delta
        cand[j] += delta
        val = _c5_value(cand)
        if val < best_val:
            best_val = val
            best_h = cand
            h = cand
    return best_h, best_val


def _upsample(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid followed by projection."""
    old_n = h.size
    if new_n == old_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = _project(h_new, new_n / 2.0)
    return h_new


def _initial_h():
    """Return the supplied construction if present, otherwise a random feasible vector."""
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=float)
        n = h.size
        h = _project(h, n / 2.0)
    else:
        n = 100
        h = np.random.rand(n)
        h = _project(h, n / 2.0)
    return h


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Search for a step function with small C5."""
    np.random.seed(seed)
    start_time = time.time()
    best_h = _initial_h()
    best_val = _c5_value(best_h)

    # sequence of resolutions (powers of two, capped at 1000 points)
    n0 = best_h.size
    resolutions = [n0]
    while resolutions[-1] * 2 <= 1000:
        resolutions.append(resolutions[-1] * 2)

    for N in resolutions:
        if time.time() - start_time > budget_s * 0.95:
            break

        # generate a starting point at this resolution
        h_start = _upsample(best_h, N)

        # allocate a modest number of iterations; more for coarser grids
        base_iters = 4000
        iters = max(500, int(base_iters * (200.0 / N)))
        restarts = 3
        iters_per = max(1, iters // restarts)

        local_best_h = None
        local_best_val = np.inf

        for r in range(restarts):
            if time.time() - start_time > budget_s * 0.95:
                break
            if r == 0:
                h0 = h_start.copy()
            else:
                # add small noise before projection
                noise = np.random.randn(N) * 0.02
                h0 = _project(h_start + noise, N / 2.0)

            h_opt, val_opt = _gradient_descent(h0, iters_per,
                                               budget_s, start_time)
            # local random search
            h_opt, val_opt = _local_search(h_opt, moves=2000,
                                           budget=budget_s, start=start_time)

            if val_opt < local_best_val:

        # keep the best across resolutions
        if local_best_val < best_val:
            best_val = local_best_val
            best_h = local_best_h

    # final projection to guarantee feasibility
    best_h = _project(best_h, best_h.size / 2.0)
    final_val = _c5_value(best_h)
    return best_h, final_val, best_h.size
