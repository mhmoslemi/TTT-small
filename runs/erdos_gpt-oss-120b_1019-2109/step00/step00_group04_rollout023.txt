def project_box_sum(v, total):

        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            low = mid
        else:
            high = mid
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # adjust small rounding errors: if sum deviates, adjust a few elements
    diff = total - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff uniformly over not-at-bound entries up to epsilon
        # for simplicity just adjust first element within bounds
        # ensure not exceed boundaries

                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:

import time
import numpy as np

# ----- helper functions -----------------------------------------------------

def project_box_sum(v: np.ndarray, total: float) -> np.ndarray:
    """Project onto the set {0<=x_i<=1, sum x_i = total}."""
    low, high = -1.0, v.max() + 1.0  # bounds for the shift λ
    for _ in range(50):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            low = mid
        else:
            high = mid
    lam = (low + high) * 0.5
    x = np.clip(v - lam, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = total - x.sum()
    if abs(diff) > 1e-12:
        # distribute the remaining mass on free coordinates
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            add = diff / free.sum()
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x


def upsample_h(old: np.ndarray, n_new: int) -> np.ndarray:
    """Linear interpolation of a step function to a new resolution."""
    n_old = len(old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, old)


def evaluate(h: np.ndarray) -> float:
    """Return C5 for a given discretized function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def max_corr_shift_and_gradient(h: np.ndarray):
    """Find shift attaining the current max overlap and its gradient."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)                     # integer shift (can be negative)
    g = np.zeros_like(h)

    if shift >= 0:
        ns = n - shift
        if ns:
            g[:ns] += 1.0 - h[shift:]
            g[shift:] += -h[:ns]
    else:
        s = -shift
        ns = n - s
        if ns:
            g[s:] += 1.0 - h[:ns]
            g[:ns] += -h[s:]

    max_corr = corr[idx] * (2.0 / n)          # C5 value
    return shift, g, max_corr


# ----- core optimizer -----------------------------------------------------

def optimize_h(h0: np.ndarray, timelimit: float):
    """Local optimisation of the step function within a time budget."""
    start = time.time()
    n = h0.shape[0]
    target_sum = n / 2.0

    best_h = h0.copy()
    best_val = evaluate(best_h)

    cur_h = h0.copy()
    cur_val = best_val

    # parameters for the inner loops
    alphas = (0.12, 0.06, 0.03, 0.015, 0.0075)
    pair_move_prob = 0.005

    while time.time() - start < timelimit:
        # --- gradient descent on the current worst shift -------------------
        shift, grad, cur_val = max_corr_shift_and_gradient(cur_h)

        improved = False
        for a in alphas:
            v = cur_h - a * grad
            v = project_box_sum(v, target_sum)
            new_val = evaluate(v)
            if new_val < cur_val - 1e-9:
                cur_h, cur_val = v, new_val
                improved = True
                if new_val < best_val - 1e-9:
                    best_h, best_val = v.copy(), new_val
                break

        # --- if gradient step stagnates, try a random pair move ----------
        if not improved and np.random.rand() < pair_move_prob:
            i, j = np.random.randint(0, n, size=2)
            if i != j:
                # move mass from i to j (or opposite) while respecting bounds
                if np.random.rand() < 0.5:
                    max_inc = min(1.0 - cur_h[j], cur_h[i])

                    delta = max_inc * np.random.rand() * 0.4
                    v = cur_h.copy()
                    if np.random.rand() < 0.5:
                        v[i] -= delta
                        v[j] += delta
                    else:
                        v[i] += delta
                        v[j] -= delta
                    v = project_box_sum(v, target_sum)
                    new_val = evaluate(v)
                    if new_val < cur_val - 1e-9:
                        cur_h, cur_val = v, new_val
