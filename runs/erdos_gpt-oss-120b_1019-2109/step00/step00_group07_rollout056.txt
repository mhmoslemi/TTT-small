def project_onto_box_and_sum(v, target):
    # Input v: numpy array (any shape)
    # Output w: same shape, each entry in [0,1], sum(w)=target.
    # Use bisection.
    lo = -np.max(v) - 1  # ensure v+λ <= 0? Actually we need lower bound where all w=0 => λ <= -max(v)
    hi = 1 - np.min(v) + 1  # ensure λ large enough that all w=1
    # Actually we can compute exact extremes:
    # We need λ such that sum(clip(v+λ,0,1)) = target.
    # For λ low, all v+λ <=0, sum=0. For λ high, all >=1, sum=n.
    # So low can be -max(v) - eps ; hi = 1 - min(v) + eps.
    lo = -np.max(v) - 1.0
    hi = 1 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi)/2.0
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + (lo+hi)/2.0, 0.0, 1.0)
    # To correct any small residual due to rounding, we could adjust by distributing leftover
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        # add diff to some non-saturation entries evenly (or one entry)
        # Find indices where w not at bound
        mask = (w > 1e-12) & (w < 1-1e-12)
        if mask.any():
            # distribute diff equally
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            # fallback: adjust first element
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
    return w

import time
import numpy as np

def _project_box_sum(v, target):
    """Project v onto [0,1]^n with sum == target using a simple bisection."""
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction if numerical error remains
    diff = target - w.sum()
    if abs(diff) > 1e-12:
        mask = (w > 1e-12) & (w < 1.0 - 1e-12)
        if mask.any():
            w[mask] += diff / mask.sum()
            w = np.clip(w, 0.0, 1.0)
        else:
            w[0] = np.clip(w[0] + diff, 0.0, 1.0)
    return w

def _c5_value(h):
    """Compute the C5 value for a step function h."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c = c * (2.0 / n)          # dx = 2/n
    return float(c.max())

def _binary_from_continuous(arr):
    """Turn any array in [0,1] into a binary vector with exactly n/2 ones."""
    n = len(arr)
    k = n // 2
    idx = np.argsort(arr)[::-1]
    b = np.zeros(n, dtype=float)
    b[idx[:k]] = 1.0
    return b

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    # ----------------------------------------------------------------------
    # Determine problem size
    try:
        init = initial_h_values               # provided by the harness
        n = len(init)
    except NameError:
        # fallback size
        n = 80
        init = None

    # Ensure n is even (required for a exactly half‑filled binary vector)
    if n % 2:
        n += 1

    target_sum = n // 2                      # half the points must be 1

    # ----------------------------------------------------------------------
    # Evaluation routine (use harness version if available)
    if 'evaluate_erdos_solution' in globals():
        evaluate = evaluate_erdos_solution
    else:
        evaluate = _c5_value

    # ----------------------------------------------------------------------
    # Build an initial pool of binary candidates
    candidates = []

    # 1) binary version of the supplied construction (if any)
    if init is not None:
        b = _binary_from_continuous(init)
        candidates.append(b)

    # 2) simple deterministic patterns
    x = np.linspace(0, 2, n, endpoint=False)

    # triangle (linear) shape, then binarised
    tri = np.where(x <= 1, x, 2 - x)
    candidates.append(_binary_from_continuous(tri))

    # sin^2 shape, then binarised
    sin2 = np.sin(np.pi * x / 2) ** 2
    candidates.append(_binary_from_continuous(sin2))

    # 3) a few random binary start points
    for _ in range(5):
        rand = np.zeros(n, dtype=float)
        rand[:target_sum] = 1.0
        np.random.shuffle(rand)
        candidates.append(rand)

    # ----------------------------------------------------------------------
    # Pick the best seed among the candidates
    best_h = None
    best_val = np.inf
    for cand in candidates:
        val = evaluate(cand)
        if val < best_val:
            best_val = val
            best_h = cand.copy()

    # ----------------------------------------------------------------------
    # Simulated‑annealing / hill‑climbing on binary vectors
    # ----------------------------------------------------------------------
    # Helper to keep track of positions of ones / zeros
    def _positions(v):
        ones = np.where(v > 0.5)[0]
        zeros = np.where(v <= 0.5)[0]
        return ones, zeros

    # Parameters for the annealing schedule
    max_iters = 10_000_000                    # effectively limited by time
    iter_cnt = 0
    # start with a relatively high temperature; it cools linearly to a floor
    T0 = 0.05
    T_floor = 1e-4

    # current solution starts from the best seed found
    cur_h = best_h.copy()
    cur_val = best_val
    cur_ones, cur_zeros = _positions(cur_h)

    while True:
        if time.time() - start > budget_s - 0.2:
            break
        # linear cooling based on elapsed fraction
        elapsed = time.time() - start
        frac = elapsed / budget_s
        temp = max(T_floor, T0 * (1.0 - frac))

        # propose a swap of a random 1 with a random 0
        i = np.random.choice(cur_ones)
        j = np.random.choice(cur_zeros)

        # perform swap
        cand = cur_h.copy()
        cand[i] = 0.0
        cand[j] = 1.0

        cand_val = evaluate(cand)

        # acceptance rule
        if cand_val < cur_val or np.random.rand() < np.exp(-(cand_val - cur_val) / (temp + 1e-12)):

            cur_h = cand
            cur_val = cand_val
            cur_ones, cur_zeros = _positions(cur_h)

            if cand_val < best_val:

                best_h = cand.copy()

        # occasional random restart to diversify search

        if iter_cnt % 20000 == 0:

            fresh[:target_sum] = 1.0
            np.random.shuffle(fresh)
            cur_h = fresh
            cur_val = evaluate(cur_h)
            cur_ones, cur_zeros = _positions(cur_h)

    # ----------------------------------------------------------------------
    # Return the best binary vector found
    final_val = evaluate(best_h)
    return best_h, float(final_val), n
