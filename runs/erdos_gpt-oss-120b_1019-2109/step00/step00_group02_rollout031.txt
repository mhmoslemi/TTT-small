import time, math
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ---------- helper ---------- #
    def compute_c5(h):
        """C5 for a step function given as a numpy array."""
        n = h.size
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(np.max(corr) * dx)

    def adjust_sum(vec, target):
        """Force integer sum = target (exact) by flipping a few entries."""
        cur = int(round(vec.sum()))
        diff = target - cur
        if diff > 0:
            zeros = np.where(vec < 0.5)[0]
            if zeros.size >= diff:
                idx = rng.choice(zeros, size=diff, replace=False)
                vec[idx] = 1.0
        elif diff < 0:
            ones = np.where(vec > 0.5)[0]
            if ones.size >= -diff:
                idx = rng.choice(ones, size=-diff, replace=False)
                vec[idx] = 0.0
        return vec

    # ---------- initial candidate ---------- #
    best_h = None
    best_c5 = float('inf')
    best_n = None

    # use supplied construction if any
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        c5 = compute_c5(init)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, init.copy(), init.size

    # ---------- binary search ---------- #
    def binary_search(N, max_iters):
        nonlocal best_c5, best_h, best_n
        # enforce even length for a pure 0/1 vector
        if N % 2:
            N += 1
        target_sum = N // 2

        # random start with exact sum
        vec = np.zeros(N, dtype=np.float64)
        vec[:target_sum] = 1.0
        rng.shuffle(vec)

        curr_c5 = compute_c5(vec)
        if curr_c5 < best_c5:
            best_c5, best_h, best_n = curr_c5, vec.copy(), N

        # simulated‑annealing schedule
        T0, T_end = 0.02, 1e-4
        for it in range(max_iters):
            if time.time() - start_time > budget_s * 0.985:
                break
            i = rng.integers(N)
            j = rng.integers(N - 1)

                j += 1
            if vec[i] == vec[j]:
