        n = init.size
        target = n / 2.0
        h0 = _project_sum_box(init, target)
    except NameError:
        n = 80                           # a modest size, fast to evaluate
        target = n / 2.0
        h0 = rng.random(n)
        h0 = _project_sum_box(h0, target)

    best_h = h0.copy()
    best_c5 = _c5_of(best_h)

    cur_h = h0.copy()
    cur_c5 = best_c5

    start = time.time()
    limit = budget_s - 2.0          # keep a tiny safety margin

    # parameters for the annealing / gradient loop
    step0 = 0.6
    temp0 = 0.02
    it = 0
    max_it = 1_500_000

    while time.time() - start < limit and it < max_it:
        # ------------------------------------------------------------------
        # 1) find the lag with the smallest self‑overlap product
        corr = np.correlate(cur_h, cur_h, mode="full")
        prod = corr[n - 1 :]                 # prod[l] = Σ h[i]·h[i+l] for l≥0
        l_min = int(np.argmin(prod))

        # ------------------------------------------------------------------
        # 2) compute a gradient that increases this product
        grad = np.zeros_like(cur_h)
        l = l_min
        if l == 0:
            grad = 2 * cur_h
        else:
            grad[: n - l] += cur_h[l:]
            grad[l:] += cur_h[: n - l]

        # ------------------------------------------------------------------
        # 3) take a step, project back onto the feasible set
        progress = (time.time() - start) / limit
        step = step0 * (1 - progress) + 1e-6
        h_new = cur_h + step * grad
        h_new = _project_sum_box(h_new, target)

        new_c5 = _c5_of(h_new)

        # ------------------------------------------------------------------
        # 4) (simulated) acceptance
        if new_c5 < cur_c5:
            cur_h, cur_c5 = h_new, new_c5
        else:
            T = temp0 * (1 - progress) + 1e-6
            if random.random() < math.exp(-(new_c5 - cur_c5) / T):
                cur_h, cur_c5 = h_new, new_c5

        # keep the best ever seen
        if cur_c5 < best_c5:
            best_c5, best_h = cur_c5, cur_h.copy()
            if best_c5 <= 0.3808:          # reached target – stop early
                break

        # ------------------------------------------------------------------
        # 5) occasional random two‑point exchange to escape flat regions
        if it % 60 == 0:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                eps_low = -min(cur_h[i], 1.0 - cur_h[j])
                eps_high = min(1.0 - cur_h[i], cur_h[j])
                if eps_low < eps_high:
                    eps = rng.uniform(eps_low, eps_high)
                    trial = cur_h.copy()
                    trial[i] += eps
                    trial[j] -= eps
                    trial = _project_sum_box(trial, target)
                    c5_trial = _c5_of(trial)
                    if c5_trial < cur_c5:
                        cur_h, cur_c5 = trial, c5_trial
                        if c5_trial < best_c5:
                            best_c5, best_h = c5_trial, trial.copy()
                            if best_c5 <= 0.3808:
                                break

        it += 1

    # final safety projection and evaluation
    best_h = _project_sum_box(best_h, target)
    final_c5 = _c5_of(best_h)

    return best_h.tolist(), float(final_c5), int(n)
