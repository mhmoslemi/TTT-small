def project_to_box_sum(v, target, lo=0.0, hi=1.0, tol=1e-9):
    # ensures sum v = target after projection to [lo, hi] via Lagrange multipliers
    # binary search for lambda.
    lo_lambda = np.min(v - hi) - 1.0  # ensure f(lo_lambda) >= target
    hi_lambda = np.max(v - lo) + 1.0  # ensure f(hi_lambda) <= target
    for _ in range(100):
        lam = (lo_lambda + hi_lambda) / 2.0
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if np.abs(s - target) <= tol:
            return x
        if s > target:
            lo_lambda = lam
        else:
            hi_lambda = lam
    return np.clip(v - (lo_lambda+hi_lambda)/2, lo, hi)

def compute_gradient(h, p):
    n = len(h)
    dx = 2.0 / n
    # compute C vector
    C = np.correlate(h, 1 - h, mode='full') * dx
    # weight
    w = C ** (p - 1)  # shape 2n-1
    # Compute gradient
    grad = np.zeros_like(h)
    # For each shift index offset
    # correlation indices correspond to shift = k - (n-1)
    # For each k in range(len(C)): shift = k - (n-1)
    for k, wk in enumerate(w):
        shift = k - (n - 1)
        # add contribution wk * dx * (1 - h[i+shift]) for all i where i+shift in [0,n)
        # also subtract wk * dx * h[i-shift]
        # Equivalent: for each i, compute contributions from shift k.
        # Loop wise we can vectorize:
        # We'll compute for each shift, the term for all i in valid range.
        if wk == 0:
            continue
        if shift >= 0:
            # i from 0..n-1-shift: i+shift valid.
            grad[:n-shift] += wk * dx * (1 - h[shift:])
            # subtract term: i from shift..n-1: i - shift valid -> h[i-shift]
            grad[shift:] -= wk * dx * h[:n-shift]
        else:
            # shift negative: let s = -shift
            s = -shift
            # i from s..n-1: i+shift = i - s valid
            grad[s:] += wk * dx * (1 - h[:n-s])
            # subtract term: i from 0..n-1-s: i - shift = i + s
            grad[:n-s] -= wk * dx * h[s:]
    # This gradient is for f (smooth approx). Actually dimensionless.
    # We also consider -? maybe sign.
    return grad
```

def hill_climb(initial_h, budget_s, start_time):
    best = initial_h.copy()
    best_val = compute_C5(best)
    cur = best.copy()
    cur_val = best_val
    n = len(cur)
    # Parameters
    max_iters = 10**7  # large
    # schedule horizon: compute T_max = best_val maybe? Let's use acceptance probability based on temperature decreasing.
    # We'll compute a simple temperature schedule: T = start_temp * (1 - progress)
    start_temp = (best_val) * 0.1  # 10% relative maybe
    while time.time() - start_time < budget_s - 0.1:
        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # possible delta range
        max_dec = cur[i]
        max_inc = 1.0 - cur[j]
        max_delta = min(max_dec, max_inc)
        if max_delta <= 0:
            continue
        # Random delta up to max_delta
        delta = max_delta * np.random.rand()
        # Maybe also consider small delta for fine tuning: multiply by a factor
        if np.random.rand() < 0.5:
            delta *= 0.1  # small step
        new_h = cur.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # Evaluate
        new_val = compute_C5(new_h)
        # Accept if better
        if new_val < cur_val:
            cur = new_h
            cur_val = new_val
            if new_val < best_val:
                best = new_h.copy()
                best_val = new_val
        else:
            # Accept with probability exp(-(new_val - cur_val)/T)
            # Compute progress
            elapsed = time.time() - start_time
            progress = elapsed / budget_s
            T = start_temp * (1 - progress)
            if T > 0 and np.exp(-(new_val - cur_val) / T) > np.random.rand():
                cur = new_h
                cur_val = new_val
        # Occasionally adjust small random perturbations to escape
    return best, best_val

import numpy as np, time

def project_to_box_sum(v, target, lo=0.0, hi=1.0, tol=1e-9, max_iter=100):
    # Clip to bounds first
    v = np.clip(v, lo, hi)
    n = v.size
    lo_sum = lo * n
    hi_sum = hi * n
    if target <= lo_sum:
        return np.full_like(v, lo)
    if target >= hi_sum:
        return np.full_like(v, hi)
    # binary search for lambda such that sum(clip(v - lam, lo, hi)) = target
    lam_low = np.min(v - hi) - 1.0  # ensure sum > target
    lam_high = np.max(v - lo) + 1.0  # ensure sum < target
    for _ in range(max_iter):
        lam = (lam_low + lam_high) / 2.0
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    # after loop
    lam = (lam_low + lam_high) / 2.0
    x = np.clip(v - lam, lo, hi)
    # Adjust small residual via scaling of non-bound elements if needed
    # Compute residual
    res = target - x.sum()
    if abs(res) > 1e-6:
        # distribute residual to free variables (not at bounds) proportionally
        free = (x > lo + 1e-12) & (x < hi - 1e-12)
        if free.any():
            adjust = res / free.sum()
            x[free] += adjust
            x = np.clip(x, lo, hi)
    return x

def _project_to_box_sum(v, target, lo=0.0, hi=1.0, tol=1e-9, max_iter=100):
    """Euclidean projection onto the intersection of box [lo,hi] and hyperplane sum=v."""
    v = np.clip(v, lo, hi)
    n = v.size
    lo_sum = lo * n
    hi_sum = hi * n

    # binary search for λ in v - λ such that sum(clip(v-λ, lo, hi)) = target
    lam_low = np.min(v - hi) - 1.0   # guarantees sum > target
    lam_high = np.max(v - lo) + 1.0  # guarantees sum < target
    for _ in range(max_iter):
        lam = (lam_low + lam_high) * 0.5
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            lam_low = lam
        else:
            lam_high = lam
    lam = (lam_low + lam_high) * 0.5
    x = np.clip(v - lam, lo, hi)

    # small correction for residual numerical error
    residual = target - x.sum()
    if abs(residual) > 1e-8:
        free = (x > lo + 1e-12) & (x < hi - 1e-12)
        if free.any():
            adj = residual / free.sum()
            x[free] += adj
            x = np.clip(x, lo, hi)
    return x


def _compute_C5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx


def _perturb(h, target_sum, scale=0.05):
    """Add small uniform noise and re‑project to feasibility."""
    n = h.size
    noise = (np.random.rand(n) - 0.5) * scale
    new = h + noise
    new = np.clip(new, 0.0, 1.0)
    return _project_to_box_sum(new, target_sum)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # 1. Build an initial feasible step function
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        base_n = base.size
        factor = 4                     # up‑sample factor (keeps size modest)
        n = base_n * factor
        if n % 2 == 1:
            n += 1
        # linear interpolation on [0,2)
        x_old = np.linspace(0.0, 2.0, base_n, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, base)
        h = np.clip(h, 0.0, 1.0)
        target_sum = n / 2.0
        h = _project_to_box_sum(h, target_sum)
    else:
        n = 200
        target_sum = n / 2.0
        h = np.zeros(n, dtype=float)
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    # ensure exact feasibility
    h = _project_to_box_sum(h, target_sum)

    best_h = h.copy()
    best_val = _compute_C5(best_h)
    cur_h = best_h.copy()
    cur_val = best_val

    # ------------------------------------------------------------------
    # 2. Simple stochastic hill‑climbing with occasional perturbations
    # ------------------------------------------------------------------
    stagnation = 0
    stagnation_limit = 5000
    while time.time() - start_time < budget_s - 0.5:
        # propose a transfer of mass δ from i to j (maintains the total sum)
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        max_dec = cur_h[i]               # amount we may subtract from i
        max_inc = 1.0 - cur_h[j]         # amount we may add to j
        max_delta = min(max_dec, max_inc)
        if max_delta <= 0.0:
            continue

        # random step size – most steps are modest, occasional large jumps
        delta = max_delta * np.random.rand()
        if np.random.rand() < 0.3:       # 30 % small fine‑tuning steps
            delta *= 0.1

        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        new_val = _compute_C5(new_h)

        if new_val < cur_val:
            cur_h, cur_val = new_h, new_val
            stagnation = 0
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val
        else:
            stagnation += 1

        # if we have been stuck for a while, jump to a perturbed neighbourhood
        if stagnation >= stagnation_limit:
            cur_h = _perturb(best_h, target_sum, scale=0.02)
            cur_val = _compute_C5(cur_h)
            stagnation = 0

    # final re‑projection and recompute exact bound
    best_h = _project_to_box_sum(best_h, target_sum)
    best_val = _compute_C5(best_h)

    return best_h, float(best_val), int(n)
