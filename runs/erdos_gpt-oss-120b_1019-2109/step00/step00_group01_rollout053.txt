def max_overlap_and_shift(x):
    conv = np.correlate(x, 1 - x, mode='full')
    max_idx = int(conv.argmax())
    max_cnt = int(conv[max_idx])
    shift = max_idx - (len(x) - 1)
    return max_cnt, shift

    best_max, best_shift = max_overlap_and_shift(x)
    best_c5 = best_max * (2.0 / n)
    # parameters for annealing

    x = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, size=half, replace=False)
    x[ones] = 1

    best_x = x.copy()
    best_max, _ = _max_overlap_and_shift(x)
    best_c5 = best_max * (2.0 / n)

    # annealing parameters
    max_iter = 10_000_000  # safety upper bound
    T0 = 5.0
    iteration = 0
    last_improve = 0
    idx = np.arange(n)

    while True:
        # budget check – leave a small safety margin
        if time.time() - start_time > budget_s * 0.95:
            break

        cur_max, cur_shift = _max_overlap_and_shift(x)
        k = cur_shift

        # find positions that contribute to the worst shift
        if k >= 0:
            valid = idx + k < n
            i_mask = (x == 1) & valid & (x[idx + k] == 0)
            j_mask = (x == 0) & valid & (x[idx + k] == 1)
        else:
            kk = -k
            valid = idx - kk >= 0
            i_mask = (x == 1) & valid & (x[idx - kk] == 0)
            j_mask = (x == 0) & valid & (x[idx - kk] == 1)

        i_candidates = np.where(i_mask)[0]
        j_candidates = np.where(j_mask)[0]

        if i_candidates.size and j_candidates.size:
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)
        else:
            # fallback: any random 1↔0 exchange
            ones_idx = np.where(x == 1)[0]
            zeros_idx = np.where(x == 0)[0]
            i = rng.choice(ones_idx)
            j = rng.choice(zeros_idx)

        # propose the swap
        x[i], x[j] = 0, 1
        new_max, _ = _max_overlap_and_shift(x)

        # simulated‑annealing acceptance
        delta = new_max - cur_max
        T = T0 * (1.0 - iteration / max_iter)
        accept = False
        if delta <= 0:
            accept = True
        elif T > 0.0 and rng.random() < math.exp(-delta / T):
            accept = True

        if not accept:
            # revert the move
            x[i], x[j] = 1, 0
        else:
            # keep the move; update global best if improved
            if new_max < best_max:
                best_max = new_max
                best_x = x.copy()
                best_c5 = best_max * (2.0 / n)
                last_improve = iteration

        iteration += 1

        # occasional restart to escape deep local minima
        if iteration - last_improve > 50_000:
            x = np.zeros(n, dtype=np.int8)
            ones = rng.choice(n, size=half, replace=False)
            x[ones] = 1
            last_improve = iteration

    h_vals = best_x.astype(float)  # convert to float for the evaluator
    c5_bound = best_max * (2.0 / n)
    return h_vals, c5_bound, n
