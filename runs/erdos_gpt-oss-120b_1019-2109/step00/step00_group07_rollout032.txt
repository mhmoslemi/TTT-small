def proj_cap(v, s):
    # Clip to [0,1] after shifting by lambda.
    # Find lambda such that sum(clip(v - lambda, 0, 1)) = s.
    # Use binary search.
    lo = -1.0  # lambda lower bound
    hi = 1.0   # lambda upper bound
    # Actually v_i - lambda, so if lambda very negative, v - lambda large >1, then clip to 1 leads sum = N.
    # if lambda very positive, v - lambda small <0, clip 0 leads sum=0.
    # So we search lambda in range [min(v)-1, max(v)] maybe extended.
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0, 1)
        s_sum = proj.sum()
        if s_sum > s:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0, 1)

def gradient_step(h, eps, lr):
    corr = corr_func(h) # without dx
    # compute weights
    max_corr = np.max(corr) # for numerical stability can subtract max to avoid overflow
    # compute exp((corr - max_corr)/eps) to avoid overflow
    exp_vals = np.exp((corr - max_corr) / eps)
    weight = exp_vals
    weight_sum = weight.sum()
    probs = weight / weight_sum
    # compute gradient
    grad = np.zeros_like(h)
    # l from -(n-1) ... (n-1)
    # use vectorized loops
    offsets = np.arange(-(n-1), n)
    for idx, l in enumerate(offsets):
        w = probs[idx]
        if w == 0:
            continue
        if l >= 0:
            if l < n:  # always true
                grad[:n-l] += w * (1 - h[l:])
                grad[l:] -= w * h[:n-l]
        else:
            k = -l
            if k < n:
                grad[k:] += w * (1 - h[:n - k])
                grad[:n - k] -= w * h[k:]

    # gradient of smooth objective w.r.t h (not multiplied by dx)
    # update
    new_h = proj_cap(h - lr * grad, target_sum)
    return new_h

def proj_cap(v, s):
    lo = -1.0
    hi = 1.0
    # Expand bounds to ensure sum at lo > s and sum at hi < s
    # At lo = -1, v - lo = v+1 => >1 for many entries -> after clipping =1; sum = n
    # At hi = 1, v - hi = v-1 <=0 for many entries -> after clipping =0; sum =0
    # So s between 0 and n (it is n/2) reachable.
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0, 1)
        cur_sum = proj.sum()
        if cur_sum > s:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo+hi)/2.0, 0, 1)

def smooth_gradient_descent(h, dx, steps=2000, eps=0.001, lr=0.1):
    n = len(h)
    target_sum = n/2.0
    for step in range(steps):
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)  # for scaling
        # compute softmax weights
        exp_vals = np.exp((corr - max_corr) / eps)
        probs = exp_vals / exp_vals.sum()
        # compute gradient
        grad = np.zeros_like(h)
        for idx, l in enumerate(range(-n+1, n)):
            w = probs[idx]
            if w == 0:
                continue
            if l >= 0:
                grad[:n-l] += w * (1 - h[l:])
                grad[l:] -= w * h[:n-l]
            else:
                k = -l
                grad[k:] += w * (1 - h[:n - k])
                grad[:n - k] -= w * h[k:]
        # update
        h_new = proj_cap(h - lr * grad, target_sum)
        # evaluate
        c_old = np.max(corr) * dx
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c_new = np.max(corr_new) * dx
        # if improvement, accept
        if c_new <= c_old:
            h = h_new
            # optionally reduce lr gradually
            lr *= 0.99
        else:
            # no improvement, maybe reduce lr and try again
            lr *= 0.9
    return h

import numpy as np

def proj_cap(v, s):
    # binary search lambda
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        cur = proj.sum()
        if cur > s:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

import numpy as np
import random
import time
import math

def _proj_cap(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto the set {0<=x<=1, sum(x)=target_sum}."""
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c_value(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a given h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _simulated_annealing(start_h: np.ndarray, time_limit: float) -> tuple:
    """Simple SA that tries to lower the max correlation."""
    n = start_h.shape[0]
    target = n / 2.0
    dx = 2.0 / n

    h = start_h.copy()
    best_h = h.copy()
    best_c = _c_value(h)

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # current correlation and its maximal lag
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_c = float(corr.max() * dx)
        max_idx = int(corr.argmax())
        shift = max_idx - (n - 1)          # lag giving the max

        # annealing schedule
        T = 0.01 * math.exp(-it / 20000.0)
        step_scale = 0.5 * math.exp(-it / 50000.0)

        move_type = random.random()
        if move_type < 0.05:               # pure swap
            i, j = random.sample(range(n), 2)
            new_h = h.copy()
            new_h[i], new_h[j] = h[j], h[i]

        elif move_type < 0.55 and shift != 0:   # targeted transfer along the worst lag
            if shift > 0:
                i = random.randint(0, n - shift - 1)
                j = i + shift
            else:
                k = -shift
                i = random.randint(k, n - 1)
                j = i - k
            hi, hj = h[i], h[j]
            if random.random() < 0.5:
                max_transfer = min(hi, 1.0 - hj)
                if max_transfer <= 0.0:
                    it += 1
                    continue
                delta = random.random() * max_transfer * step_scale
                new_h = h.copy()
                new_h[i] = hi - delta
                new_h[j] = hj + delta
            else:
                max_transfer = min(hj, 1.0 - hi)
                if max_transfer <= 0.0:
                    it += 1
                    continue
                delta = random.random() * max_transfer * step_scale
                new_h = h.copy()
                new_h[i] = hi + delta
                new_h[j] = hj - delta

        else:                                   # generic transfer between two positions
            i, j = random.sample(range(n), 2)
            hi, hj = h[i], h[j]
            if random.random() < 0.5:
                max_transfer = min(hi, 1.0 - hj)
                if max_transfer <= 0.0:
                    it += 1
                    continue
                delta = random.random() * max_transfer * step_scale
                new_h = h.copy()
                new_h[i] = hi - delta
                new_h[j] = hj + delta
            else:
                max_transfer = min(hj, 1.0 - hi)
                if max_transfer <= 0.0:
                    it += 1
                    continue
                delta = random.random() * max_transfer * step_scale
                new_h = h.copy()
                new_h[i] = hi + delta
                new_h[j] = hj - delta

        new_c = float(np.correlate(new_h, 1.0 - new_h, mode="full").max() * dx)

        if new_c < cur_c or random.random() < math.exp(-(new_c - cur_c) / max(T, 1e-12)):
            h = new_h
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
                if best_c <= 0.3808:
                    return best_h, best_c
        it += 1
    return best_h, best_c


def _smooth_gradient_descent(h0: np.ndarray,
                             steps: int = 1500,
                             eps: float = 0.001,
                             lr: float = 0.08) -> np.ndarray:
    """Refine a candidate by minimizing a smooth approximation of the max."""
    n = h0.shape[0]
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    for _ in range(steps):
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        # soft‑max weights
        exp_vals = np.exp((corr - max_corr) / eps)
        probs = exp_vals / exp_vals.sum()
        grad = np.zeros_like(h)
        # gradient of each lag weighted by probs
        for idx, lag in enumerate(range(-n + 1, n)):
            w = probs[idx]
            if w == 0.0:
                continue
            if lag >= 0:
                grad[: n - lag] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[: n - lag]
            else:
                k = -lag
                grad[k:] += w * (1.0 - h[: n - k])
                grad[: n - k] -= w * h[k:]
        h_new = _proj_cap(h - lr * grad, target)
        old_c = float(corr.max() * dx)
        new_c = float(np.correlate(h_new, 1.0 - h_new, mode="full").max() * dx)
        if new_c < old_c:
            h = h_new
            lr *= 0.99
        else:
            lr *= 0.90
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Find a step function h on [0,2] with a small C5 bound."""
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------

    if 'initial_h_values' in globals():

    else:
        n = 200
        target = n / 2.0

    remaining = budget_s * 0.85

    if best_c > 0.3808:

    return best_h, float(final_c), n
