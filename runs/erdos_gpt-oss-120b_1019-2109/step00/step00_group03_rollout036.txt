def project_onto_box_sum(v, target, max_iter=100):
    v = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = v.sum()
        if abs(s - target) < 1e-10:
            break
        diff = target - s
        free = (v > 0) & (v < 1)
        free_count = free.sum()
        if free_count == 0:
            # all at bounds, can't adjust further; distribute to nearest bound
            # simply adjust by adding diff uniformly then clip again
            v = np.clip(v + diff / len(v), 0.0, 1.0)
        else:
            v[free] += diff / free_count
            # after adjustment, clip
            v = np.clip(v, 0.0, 1.0)
    # final clip
    v = np.clip(v, 0.0, 1.0)
    # adjust small residual by adjusting a random free element
    s = v.sum()
    if abs(s - target) > 1e-8:
        # distribute residual to first element not at bound
        idx = np.argmax((v > 0) & (v < 1))
        if v[idx] > 0 and v[idx] < 1:
            v[idx] = np.clip(v[idx] + (target - s), 0.0, 1.0)
        else:
            # fallback to scaling and clipping
            v = project_box_sum(v, target)  # recursion? careful.
    return v

import numpy as np
import time
import math

# -------------------------------------------------------------
# Helper functions (top‑level, no closures)
# -------------------------------------------------------------

def project_box_sum(v, target, tol=1e-12, max_iter=80):
    """
    Euclidean projection of vector v onto the set
        {x : 0 <= x_i <= 1 and sum(x) = target }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0          # guarantee all entries >1 after subtracting lo
    hi = np.max(v)                # guarantee all entries <0 after subtracting hi
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
        if hi - lo < tol:
            break
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    return x

def grad_overlap_for_lag(h, lag):
    """
    Gradient of the overlap integral for a single lag.
    Returns d/ dh of  ∫ h(x) (1 - h(x+lag)) dx   (discrete version).
    """
    N = h.shape[0]
    dx = 2.0 / N
    g = np.zeros(N, dtype=np.float64)
    if lag == 0:
        g -= 2.0 * h * dx
    elif lag > 0:
        g[: N - lag] -= h[lag:] * dx
        g[lag:]       -= h[: N - lag] * dx
    else:  # lag < 0
        L = -lag
        g[L:]       -= h[: N - L] * dx
        g[: N - L] -= h[L:] * dx
    return g

def max_overlap(h):
    """
    Compute C5 for a step function h (discrete version).
    """
    N = h.shape[0]
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def random_feasible(N, target, rng):
    """
    Produce a random vector in [0,1]^N with sum == target.
    """
    v = rng.random(N)
    v = v * (target / v.sum())
    v = np.clip(v, 0.0, 1.0)
    v = project_box_sum(v, target)
    return v

    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_box_sum(h0, target_sum)

    best_h = h0.copy()
    best_val = max_overlap(best_h)

    # -----------------------------------------------------------------
    # Optimization parameters
    # -----------------------------------------------------------------
    max_total_iters = 500000      # safety cap
    eta_initial = 1.0
    beta_smooth = 5.0             # smoothing for soft‑max
    restart_after = 8000          # iterations without improvement -> restart
    report_every = 2000           # evaluate and possibly record best
    # -----------------------------------------------------------------

    # We will carry out several independent trials (including the given start)
    trials = [h0] + [random_feasible(N, target_sum, rng) for _ in range(3)]

    for trial_h in trials:
        h = trial_h.copy()
        eta = eta_initial
        no_improve = 0

            corr = np.convolve(h, h[::-1], mode='full')

            idx_max = int(np.argmax(d))
            cur_val = float(d[idx_max])

            if cur_val + 1e-12 < best_val:
                best_val = cur_val
                best_h = h.copy()

            if it % 2 == 0:

                lag = idx_max - (N - 1)

            else:

                max_d = d[idx_max]

    best_h = project_box_sum(best_h, target_sum)
    final_val = max_overlap(best_h)

    return (best_h.tolist(), float(final_val), int(N))
