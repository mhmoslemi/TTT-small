def solve_binary_erdos(n, time_limit):
    import cvxpy as cp, numpy as np, time
    # define variables
    h = cp.Variable(n, boolean=True)
    P = cp.Variable((n, n), boolean=True)
    t = cp.Variable()
    constraints = []
    constraints.append(cp.sum(h) == n/2)
    # product constraints
    for i in range(n):
        for j in range(n):
            constraints.append(P[i, j] <= h[i])
            constraints.append(P[i, j] <= h[j])
            constraints.append(P[i, j] >= h[i] + h[j] - 1)
    # shift constraints
    for lag in range(-(n-1), n):
        start = max(0, -lag)
        end = min(n, n - lag)
        if start >= end:
            continue
        a_l = cp.sum(h[start:end])
        prod_sum = cp.sum([P[i, i + lag] for i in range(start, end)])
        constraints.append(a_l - prod_sum <= t)
    constraints.append(t >= 0)
    problem = cp.Problem(cp.Minimize(t), constraints)
    try:
        problem.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except Exception:
        try:
            problem.solve(solver=cp.GLPK_MI, glpk={'msg_lev':'GLP_MSG_OFF', 'tm_lim': int(time_limit*1000)}, verbose=False)
        except Exception:
            problem.solve(solver=cp.ECOS_BB, max_iters=200000, verbose=False)
    if h.value is None:
        return None
    h_sol = np.rint(h.value).astype(float)  # keep as float to not break evaluation.
    # ensure sum to n/2, maybe adjust small rounding issues.
    # evaluate C5
    dx = 2.0 / n
    c5 = np.max(np.correlate(h_sol, 1 - h_sol, mode='full')) * dx
    t_val = t.value
    return h_sol, float(c5), float(t_val)

import numpy as np, cvxpy as cp, time, math

def solve_binary_erdos(n, max_seconds):
    # Build MILP
    h = cp.Variable(n, boolean=True)
    P = cp.Variable((n,n), boolean=True)
    t = cp.Variable()
    cons = []
    cons.append(cp.sum(h) == n/2)
    # product constraints
    for i in range(n):
        for j in range(n):
            cons.append(P[i,j] <= h[i])
            cons.append(P[i,j] <= h[j])
            cons.append(P[i,j] >= h[i] + h[j] - 1)
    # shift constraints
    for lag in range(-(n-1), n):
        start = max(0, -lag)
        end = min(n, n - lag)
        if start >= end:
            continue
        a_l = cp.sum(h[start:end])
        # sum over product entries
        prod_terms = [P[i, i+lag] for i in range(start, end)]
        cons.append(a_l - cp.sum(prod_terms) <= t)
    cons.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), cons)
    # attempt solving
    try:
        prob.solve(solver=cp.CBC, verbose=False, max_seconds=max_seconds)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, verbose=False, glpk={'tm_lim': int(max_seconds*1000)})
        except Exception:
            prob.solve(solver=cp.ECOS_BB, verbose=False, max_iters=int(max_seconds*10000))
    if h.value is None:
        return None
    h_sol = np.rint(h.value).astype(float)
    dx = 2.0 / n
    c5 = float(np.max(np.correlate(h_sol, 1 - h_sol, mode='full')) * dx)
    # Return t for reference
    t_val = float(t.value) if t.value is not None else None
    return h_sol, c5, t_val

import time
import numpy as np
import cvxpy as cp

def _solve_binary_erdos(n: int, max_seconds: float):
    """Solve the discrete Erdős overlap problem for binary h of length n.
    Returns (h_array, C5_value, t_opt) or None on failure."""
    # variables
    h = cp.Variable(n, boolean=True)
    P = cp.Variable((n, n), boolean=True)  # product matrix h_i * h_j
    t = cp.Variable()
    cons = []

    # total mass = n/2 (n must be even)
    cons.append(cp.sum(h) == n / 2)

    # product constraints: P[i,j] = h[i] * h[j]
    for i in range(n):
        for j in range(n):
            cons.append(P[i, j] <= h[i])
            cons.append(P[i, j] <= h[j])
            cons.append(P[i, j] >= h[i] + h[j] - 1)

    # overlap constraints for every lag
    for lag in range(-(n - 1), n):
        start = max(0, -lag)
        end = min(n, n - lag)
        if start >= end:
            continue
        a_l = cp.sum(h[start:end])                     # sum of h over overlap
        prod_terms = [P[i, i + lag] for i in range(start, end)]
        cons.append(a_l - cp.sum(prod_terms) <= t)     # a_l - g_l <= t

    cons.append(t >= 0)

    prob = cp.Problem(cp.Minimize(t), cons)

    # try a few solvers, fall back to ECOS_BB if needed
    try:
        prob.solve(solver=cp.CBC, max_seconds=max_seconds, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI,
                       glpk={'msg_lev': 'GLP_MSG_OFF',
                             'tm_lim': int(max_seconds * 1000)},
                       verbose=False)
        except Exception:
            prob.solve(solver=cp.ECOS_BB,
                       max_iters=int(max_seconds * 10000),
                       verbose=False)

    if h.value is None:
        return None

    # round to exact binary values (solver may return tiny fractions)
    h_sol = np.rint(h.value).astype(float)

    dx = 2.0 / n
    c5_val = float(np.max(np.correlate(h_sol, 1 - h_sol, mode='full')) * dx)
    t_val = float(t.value) if t.value is not None else None
    return h_sol, c5_val, t_val


def _local_search_binary(h, max_iters, target):
    """Simple hill‑climbing: swap a 1 with a 0 if it improves C5."""
    n = len(h)
    dx = 2.0 / n
    best = h.copy()
    best_c5 = float(np.max(np.correlate(best, 1 - best, mode='full')) * dx)
    ones = np.where(best == 1)[0]
    zeros = np.where(best == 0)[0]

    for _ in range(max_iters):
        if best_c5 <= target:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cand = best.copy()
        cand[i], cand[j] = 0.0, 1.0
        c5 = float(np.max(np.correlate(cand, 1 - cand, mode='full')) * dx)
        if c5 < best_c5:
            best = cand
            best_c5 = c5
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
    return best, best_c5


def run(seed=42, budget_s=1100):
    np.random.seed(seed)
    start_time = time.time()

    # start from the provided construction if it exists

    # test a few even lengths (binary vectors)

    for n in candidates:

        # give each MILP a modest slice of the remaining budget

        max_sec = min(60.0, max(5.0, remaining - 10))

        h_sol, c5_val, _ = sol
        if c5_val < best_c5:

            if best_c5 <= 0.3808:          # already better than the record

    # If MILP did not improve the record, do a quick local search on the current best
    if best_c5 > 0.3808 and (time.time() - start_time) < budget_s - 5:
