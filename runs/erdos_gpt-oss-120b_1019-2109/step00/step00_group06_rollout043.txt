import numpy as np, time, random

def evaluate(h, dx):
   corr = np.correlate(h, 1 - h, mode='full')
   return np.max(corr) * dx

def build_h(N, a, c, v1, v2):
   b = a + c
   # ensure b <1 else no central plateau.
   if b >= 1:
        return None

   denom = 1 - b
   # compute v3 from integral condition: total area = N/2*d x =1
   # Actually using continuous: v1*a + v2*c + v3*(1 - b) = 0.5
   v3 = (0.5 - v1*a - v2*c) / denom
   if not (0 <= v3 <= 1):
        return None
   # optionally enforce monotonic
   if not (v1 <= v2 <= v3):
        return None

   # Build array using distances from endpoint
   dx = 2.0 / N
   xs = (np.arange(N) + 0.5) * dx
   t = np.minimum(xs, 2 - xs)
   # assign values
   h = np.where(t < a, v1, np.where(t < b, v2, v3))
   # adjust sum to target
   target_sum = N / 2
   cur_sum = h.sum()
   diff = target_sum - cur_sum
   if abs(diff) > 1e-12:
        # Find index where we can adjust without violating bounds

def eval_c5(h, dx):

    return np.max(corr) * dx

def build_h(N, a, c, v1, v2):
    b = a + c
    if b >= 1:
        return None
    v3 = (0.5 - v1*a - v2*c) / (1 - b)
    if not (0 <= v3 <= 1):
        return None

    if not (v1 <= v2 <= v3):
        return None
    xs = (np.arange(N) + 0.5) * (2.0 / N)
    t = np.minimum(xs, 2 - xs)
    h = np.where(t < a, v1, np.where(t < b, v2, v3))
    # adjust sum if needed (should be close)
    target = N/2

import numpy as np
import math
import time

def _c5_of_array(arr: np.ndarray) -> float:
    """return max_k ∫ h(x)(1-h(x+k))dx for the discretized step function."""
    n = arr.shape[0]
    dx = 2.0 / n
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * dx)


def _build_h(n: int, a: float, c: float, v1: float, v2: float) -> np.ndarray | None:
    """
    Construct a symmetric 5‑step shape on [0,2] with:
        - left/right edge length a (value v1)
        - inner side length c (value v2)
        - central plateau (value v3) filling the rest.
    Integral constraint ∫h = 1 is enforced exactly (up to floating error).
    """
    b = a + c
    if b >= 1.0:
        return None
    denom = 1.0 - b
    if denom <= 0.0:
        return None
    v3 = (0.5 - v1 * a - v2 * c) / denom
    if not (0.0 <= v3 <= 1.0):
        return None
    # optional monotonicity (helps convergence but not required)
    if not (v1 <= v2 <= v3):
        return None

    xs = (np.arange(n) + 0.5) * (2.0 / n)          # midpoint sampling
    t = np.minimum(xs, 2.0 - xs)                  # distance to nearest endpoint
    h = np.where(t < a, v1, np.where(t < b, v2, v3))

    # correct the tiny mass error that comes from discretisation
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 0.0) & (h < 1.0)
        if mask.any():
            idx = np.where(mask)[0][0]
            new_val = h[idx] + diff
            h[idx] = max(0.0, min(1.0, new_val))
        else:
            h += diff / n
            np.clip(h, 0.0, 1.0, out=h)
    return h


def _shape_a(n: int, a: float) -> np.ndarray | None:
    """Specialised 0‑½‑1 piecewise shape (edges 0, mid ½, centre 1)."""
    c = 1.0 - 2.0 * a          # length of the middle‑value region
    if c < 0.0:
        return None
    return _build_h(n, a, c, 0.0, 0.5)

        if mask.any():
            idx = np.where(mask)[0][0]
            best_h[idx] = max(0.0, min(1.0, best_h[idx] + diff))
        else:
            best_h += diff / n
            np.clip(best_h, 0.0, 1.0, out=best_h)

    # ------------------------------------------------------------------
    # 5. compute the final reported bound (use helper if available)
    # ------------------------------------------------------------------
    try:
        c5_bound = float(evaluate_erdos_solution(best_h))
    except Exception:
        c5_bound = float(_c5_of_array(best_h))

    return (best_h, c5_bound, n)
