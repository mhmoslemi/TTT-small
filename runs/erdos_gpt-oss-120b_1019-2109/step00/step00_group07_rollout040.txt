def project_to_constraints(h):
    # clip to [0,1]
    h = np.clip(h,0,1)
    # adjust to meet sum
    T = len(h)/2
    current = h.sum()
    if abs(current - T) < 1e-12:
        return h
    # compute difference
    diff = T - current
    # If diff > 0, need to add mass; if diff <0, need to subtract.
    # We'll adjust greedily on indices not at bound.
    if diff > 0:
        # available capacity
        cap = 1 - h
        total_cap = cap.sum()
        if total_cap > 0:
            # distribute proportionally
            h += diff * cap / total_cap
    else:
        # need to reduce mass
        avail = h # produce reduction up to h
        total_avail = avail.sum()
        if total_avail > 0:
            h -= (-diff) * avail / total_avail
    # Clip again to avoid rounding errors
    h = np.clip(h,0,1)
    # due to rounding may still be slightly off, perform final small adjustment:
    delta = T - h.sum()
    if abs(delta) > 1e-12:
        # add delta to a random index that can accommodate.
        idx = np.random.choice(np.where((h+delta >=0) & (h+delta <=1))[0]) if np.any((h+delta >=0) & (h+delta <=1)) else None
        if idx is not None:
            h[idx] += delta
    return h
```

import numpy as np, time, math

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.max(np.correlate(h, 1 - h, mode='full'))
    return c * dx

def project(h):
    n = len(h)
    T = n / 2
    # clip
    h = np.clip(h, 0.0, 1.0)
    # adjust sum via scaling of non-saturated elements
    total = h.sum()
    if abs(total - T) < 1e-10:
        return h
    # compute deficit/surplus
    diff = T - total
    # indices not at bound
    free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
    if free.size > 0:
        # distribute diff proportionally to remaining capacity
        # Use simple additive to each free variable.
        # We'll add diff / free_size to each, then reclip and loop.
        add = diff / free.size
        h[free] += add
        h = np.clip(h, 0.0, 1.0)
        # recursively adjust if still off
        return project(h)
    else:
        # If all at bounds, just adjust a random index that can be changed
        idx = np.random.randint(n)
        # adjust within bound
        max_move = min(diff, 1 - h[idx]) if diff > 0 else max(diff, -h[idx])
        h[idx] += max_move
        return h

def initial_h(n):
    # triangular shape
    x = np.linspace(0, 2, n, endpoint=False) + 1.0 / n  # shift? Not needed.
    # use midpoints: x_i = (i+0.5)*dx
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = np.abs(x - 1.0)
    # scale to meet sum constraint
    T = n / 2
    factor = T / h.sum()
    h = h * factor
    # ensure within bounds
    h = np.clip(h, 0.0, 1.0)
    # After clipping might change sum, apply project
    h = project(h)
    return h

    start = time.time()

    # Determine sample size
    init_vec = globals().get('initial_h_values')

        # Project onto constraints
        h = project(h)
    else:
        n = 200
        h = initial_h(n)

    best_h = h.copy()
    best_c = compute_c5(best_h)

    # Simulated annealing parameters
    T0 = 0.02  # initial temperature in C5 units scaling
    # We'll use temperature relative to c5 value differences
    temp = T0
    decay = 0.9995  # per iteration

    # step size for pair delta
    max_delta = 0.1  # will adapt

    # main loop
    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s - 0.5:
            break
        # Randomly pick pair for move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # magnitude
        delta = (rng.random() - 0.5) * 2 * max_delta
        hi = best_h[i] + delta
        hj = best_h[j] - delta
        # check bounds
        if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
            continue
        candidate = best_h.copy()
        candidate[i] = hi
        candidate[j] = hj
        # ensure sum unchanged automatically
        # evaluate
        cand_c = compute_c5(candidate)
        diff = cand_c - best_c
        if diff < 0 or rng.random() < math.exp(-diff / max(temp, 1e-12)):
            best_h = candidate
            best_c = cand_c
        # update temperature
        temp *= decay
        # adapt max_delta based on acceptance
        if it % 1000 == 0 and it > 0:
            # simple adaptation: if acceptance ratio low, reduce step

            pass

```
def project_onto_sum_and_box(x, T):
    x = np.clip(x, 0.0, 1.0)
    diff = T - x.sum()
    # waterfilling loop
    max_iter = 100
    for _ in range(max_iter):
        if abs(diff) < 1e-12:
            break
        free = (x > 1e-12) & (x < 1 - 1e-12)
        n_free = np.count_nonzero(free)
        if n_free == 0:
           # distribute diff to any variable that can be adjusted within bounds
            # pick a random index that can move
            idx = np.random.choice(np.arange(len(x)))
            # adjust within bound
            move = np.clip(diff, -x[idx], 1 - x[idx])
            x[idx] += move
            diff = T - x.sum()
            continue
        adjust = diff / n_free
        # apply adjust to free entries, but ensure not exceed bounds
        x_new = x.copy()
        # where adding adjust would exceed 1
        up = free & (x + adjust > 1)
        down = free & (x + adjust < 0)
        # For those exceed bounds, set to bound and remove from free next iteration
        x_new[up] = 1.0
        x_new[down] = 0.0
        # For remaining free entries, add adjust
        keep = free & ~up & ~down
        x_new[keep] += adjust
        # compute new diff
        diff = T - x_new.sum()
        x = x_new
    # final clip
    return np.clip(x, 0.0, 1.0)
```

def project(x, total):
    lo = -np.max(x)  # because x + λ >=0 => λ >= -x_i for each i (lowest requirement)
    hi = 1.0 - np.min(x)  # because x_i + λ <=1 => λ <= 1 - x_i (largest)
    for _ in range(50):
        mid = (lo + hi) / 2
        z = np.clip(x + mid, 0.0, 1.0)
        s = z.sum()
        if s > total:
            hi = mid
        else:
            lo = mid
    z = np.clip(x + (lo+hi)/2, 0.0, 1.0)
    # adjust for potential small sum error:
    diff = total - z.sum()
    if abs(diff) > 1e-9:
        # distribute among free variables not at bounds
        free = (z > 1e-12) & (z < 1 - 1e-12)
        if free.any():
            z[free] += diff / free.sum()
            z = np.clip(z, 0.0, 1.0)
    return z
```

def compute_gradient(h):
    n = len(h)
    # compute correlation raw for max lag
    corr_raw = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr_raw)
    k = idx - (n - 1)
    # compute shifted arrays
    # forward shifted: h_forward[i] = h[i + k] if i + k < n else 0
    # backward shifted: h_backward[i] = h[i - k] if i - k >=0 else 0
    if k >= 0:
        h_forward = np.concatenate([h[k:], np.zeros(k)])
        h_backward = np.concatenate([np.zeros(k), h[:n - k]])
    else:
        kk = -k
        h_forward = np.concatenate([np.zeros(kk), h[:n - kk]])
        h_backward = np.concatenate([h[kk:], np.zeros(kk)])
    gradient = 1.0 - h_forward - h_backward
    return gradient, idx, k, corr_raw[idx]
```

Note for k=0, both forward and backward are h, gradient = 1 - 2h.

import time, math
import numpy as np
from scipy.optimize import minimize_scalar

def _dx(n):
    return 2.0 / n

def _compute_c5(h):
    """C5 = max_k ∫ h(x)(1-h(x+k))dx approximated with dx."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * _dx(n)


def _project_sum_bounds(x, total):
    """
    Euclidean projection of x onto the set {z | sum(z)=total, 0<=z<=1}.
    Uses a simple water‑filling (binary search on the Lagrange multiplier).
    """
    lo = -np.max(x)                # ensure x+lam >=0
    hi = 1.0 - np.min(x)           # ensure x+lam <=1
    for _ in range(50):
        mid = (lo + hi) * 0.5
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > total:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for rounding errors
    diff = total - y.sum()
    if abs(diff) > 1e-12:
        free = (y > 1e-12) & (y < 1.0 - 1e-12)
        if free.any():
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y


def _worst_shift_and_gradient(h):
    """Return gradient of the worst (max) shift, its index, lag k and raw correlation."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    k = idx - (n - 1)                     # signed lag
    if k >= 0:
        forward = np.concatenate([h[k:], np.zeros(k)])
        backward = np.concatenate([np.zeros(k), h[: n - k]])
    else:
        kk = -k
        forward = np.concatenate([np.zeros(kk), h[: n - kk]])
        backward = np.concatenate([h[kk:], np.zeros(kk)])
    grad = 1.0 - forward - backward        # ∂/∂h_i of the worst f_k
    return grad, idx, k, corr[idx]


def _initial_triangular(n):
    """Triangular (|x-1|) init scaled to satisfy the integral exactly."""
    dx = _dx(n)
    x = (np.arange(n) + 0.5) * dx
    h = np.abs(x - 1.0)                   # values in [0,1]
    target = n / 2
    h = _project_sum_bounds(h, target)
    return h


def _search_from(start_h, budget, max_iter=20000):
    """Sub‑gradient descent with occasional jitter, returns best h and its C5."""
    n = start_h.shape[0]
    target = n / 2
    best_h = start_h.copy()
    best_c = _compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c = best_c
    start = time.time()
    step0 = 0.5

    for it in range(max_iter):
        # time guard
        if time.time() - start > budget - 0.5:
            break

        grad, idx, k, raw = _worst_shift_and_gradient(cur_h)
        cur_c = raw * _dx(n)

        # decaying step size
        eta = step0 / math.sqrt(it + 1)

        cand = cur_h - eta * grad
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_sum_bounds(cand, target)

        cand_raw = np.max(np.correlate(cand, 1.0 - cand, mode="full"))
        cand_c = cand_raw * _dx(n)

        if cand_c < cur_c:
            cur_h = cand
            cur_c = cand_c
            if cand_c < best_c:
                best_h = cand.copy()
                best_c = cand_c

        # occasional random jitter to escape plateaus
        if it % 1000 == 0:
            jitter = (np.random.rand(n) - 0.5) * eta * 0.3
            pert = cur_h + jitter
            pert = np.clip(pert, 0.0, 1.0)
            pert = _project_sum_bounds(pert, target)
            pert_raw = np.max(np.correlate(pert, 1.0 - pert, mode="full"))
            pert_c = pert_raw * _dx(n)
            if pert_c < best_c:
                best_h = pert.copy()
                best_c = pert_c
                cur_h = pert
                cur_c = pert_c

        # early stop if we already beat the target bound
        if best_c <= 0.3808:
            break

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points)
    where h_values satisfies the problem constraints and c5_bound is the achieved upper bound.
    """
    rng = np.random.default_rng(seed)
    # decide sample size – use the size of the provided construction if present
    init = globals().get("initial_h_values")
    if init is not None:
        n = int(len(init))
        h0 = np.array(init, dtype=float)
        h0 = _project_sum_bounds(h0, n / 2.0)
    else:
        n = 200                     # a modest size that respects the “<1k” guidance
        h0 = _initial_triangular(n)

    # run a first local optimisation from the initial vector
    remaining = budget_s
    start_all = time.time()
    best_h, best_c = _search_from(h0, remaining)

    # occasionally try a fresh random start to diversify
    for _ in range(3):
        if time.time() - start_all > budget_s - 0.5:
            break
        # random feasible start: Dirichlet and projection
        rand = rng.random(n)
        rand = _project_sum_bounds(rand, n / 2.0)
        h_rand, c_rand = _search_from(rand, budget_s - (time.time() - start_all))
        if c_rand < best_c:
            best_h, best_c = h_rand, c_rand

    # final projection (should already be satisfied)
    best_h = _project_sum_bounds(best_h, n / 2.0)
    final_c = _compute_c5(best_h)
    return best_h, final_c, n
