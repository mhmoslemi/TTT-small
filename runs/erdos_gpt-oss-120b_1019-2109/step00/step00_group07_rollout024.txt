import numpy as np, time

def project_onto_box_simplex(v, target):
    # Project v onto {h: sum h = target, 0<=h<=1}
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _evaluate(h, dx):
    # returns max overlap integral
    return (np.correlate(h, 1 - h, mode="full") * dx).max()

import numpy as np, time

def _project_box_simplex(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto {h: sum h = target, 0 <= h <= 1}."""
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    """Return max_{k} ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    return (np.correlate(h, 1.0 - h, mode="full") * dx).max()

    target = N / 2.0                # ∑h = target  (integral = 1)
    dx = 2.0 / N

    if init is not None:
        # linear up‑sampling of the parent construction
        old_x = np.linspace(0.0, 2.0, init.shape[0], endpoint=False)
        new_x = np.linspace(0.0, 2.0, N, endpoint=False)
        h0 = np.interp(new_x, old_x, init)
    else:
        h0 = rng.random(N)

    h0 = _project_box_simplex(h0, target)

    # ------------------------------------------------------------------ #
    # 2.   initialise the best solution (unscaled objective)
    # ------------------------------------------------------------------ #
    S = target                      # sum of h (fixed)
    Q = np.correlate(h0, h0, mode="full")
    best_h = h0.copy()
    best_unscaled = S - Q.min()     # = max overlap * (1/dx)
    best_scaled = best_unscaled * dx

    # ------------------------------------------------------------------ #
    # 3.   Differential Evolution (global exploration)
    # ------------------------------------------------------------------ #
    pop_size = max(15, N // 8)      # modest population
    F, CR = 0.8, 0.9
    max_de_evals = 20000
    evals = 0

    # population initialisation
    pop = []
    pop_fit = []                    # unscaled values
    for _ in range(pop_size - 1):
        v = rng.random(N)
        v = _project_box_simplex(v, target)
        pop.append(v)
        pop_fit.append(_eval_scaled(v, dx) / dx)
    pop.append(best_h.copy())
    pop_fit.append(best_unscaled)

    # DE main loop
    while evals < max_de_evals and (time.time() - start) < budget_s * 0.4:
        for i in range(pop_size):
            # mutation
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)
            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.clip(mutant, 0.0, 1.0)
            mutant = _project_box_simplex(mutant, target)

            # crossover
            jrand = rng.integers(N)
            mask = rng.random(N) < CR
            mask[jrand] = True
            trial = np.where(mask, mutant, pop[i])
            trial = np.clip(trial, 0.0, 1.0)
            trial = _project_box_simplex(trial, target)

            trial_scaled = _eval_scaled(trial, dx)
            trial_unscaled = trial_scaled / dx
            evals += 1

            if trial_unscaled < pop_fit[i]:
                pop[i] = trial
                pop_fit[i] = trial_unscaled
            if trial_unscaled < best_unscaled:
                best_unscaled = trial_unscaled
                best_h = trial.copy()
        # stop early if the time budget for DE is exhausted
        if (time.time() - start) >= budget_s * 0.4:
            break

    # ------------------------------------------------------------------ #
    # 4.   Local hill‑climbing with exact autocorrelation updates
    # ------------------------------------------------------------------ #
    # current state
    cur_h = best_h.copy()
    cur_Q = np.correlate(cur_h, cur_h, mode="full")
    cur_val = S - cur_Q.min()       # unscaled objective

    max_step, min_step = 0.2, 1e-5
    no_improve = 0

    while (time.time() - start) < budget_s * 0.95:
        # linearly decaying step size
        elapsed = time.time() - start
        frac = elapsed / budget_s
        step = max_step * (1.0 - frac) + min_step * frac

        i, j = rng.choice(N, size=2, replace=False)

        # decide direction of the transfer
        if rng.random() < 0.5:
            max_delta = min(step, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
        else:
            max_delta = min(step, cur_h[j], 1.0 - cur_h[i])
            if max_delta <= 0.0:
                continue
            delta = -rng.random() * max_delta

        # propose new vector
        new_h = cur_h.copy()
        new_h[i] -= delta
        new_h[j] += delta

        # incremental update of autocorrelation Q
        new_Q = cur_Q.copy()
        old_i, old_j = cur_h[i], cur_h[j]
        new_i, new_j = new_h[i], new_h[j]

        for p, old_val, new_val in ((i, old_i, new_i), (j, old_j, new_j)):
            s_min, s_max = -p, N - 1 - p
            for s in range(s_min, s_max + 1):
                idx = s + N - 1
                l = p + s
                # original second factor
                if l == i:
                    old_l = old_i
                elif l == j:
                    old_l = old_j
                else:
                    old_l = cur_h[l]
                # new second factor
                if l == i:
                    new_l = new_i
                elif l == j:
                    new_l = new_j
                else:
                    new_l = cur_h[l]
                new_Q[idx] += new_val * new_l - old_val * old_l

        new_val = S - new_Q.min()

        if new_val <= cur_val + 1e-14:          # accept non‑worsening move
            cur_h, cur_Q, cur_val = new_h, new_Q, new_val
            no_improve = 0
            if new_val < best_unscaled:
                best_unscaled = new_val
                best_h = new_h.copy()
        else:
            no_improve += 1

        # occasional random restart to escape plateaux
        if no_improve >= 10000:
            cur_h = best_h + 1e-2 * (rng.random(N) - 0.5)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_h = _project_box_simplex(cur_h, target)
            cur_Q = np.correlate(cur_h, cur_h, mode="full")
            cur_val = S - cur_Q.min()
            no_improve = 0

    # ------------------------------------------------------------------ #
    # 5.   final verification and return
    # ------------------------------------------------------------------ #
    best_h = _project_box_simplex(best_h, target)   # safety projection
    final_scaled = _eval_scaled(best_h, dx)
    return (best_h, float(final_scaled), N)
