import time
import numpy as np

def _c5_of(h):
    """Return the C5 value for a step function h."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)

def _triangle_h(n):
    """Construct the triangular step function with n samples."""
    dx = 2.0 / n
    idx = np.arange(n)
    x = (idx + 0.5) * dx                     # mid‑point sampling
    h = np.minimum(x, 2.0 - x)               # triangular shape
    # enforce the integral constraint exactly
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) > 1e-12:                    # tiny correction, keep bounds
        h[0] += diff
        if h[0] < 0.0:
            h[0] = 0.0
        elif h[0] > 1.0:
            h[0] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of samples."""
    rng = np.random.default_rng(seed)
    start = time.time()
    best_h = None

    best_n = None

    # Candidate sample sizes (even numbers give an exact integral for the triangle)
    candidate_ns = [200, 202, 220, 240, 260, 280, 300]

    for n in candidate_ns:
        if time.time() - start > budget_s * 0.95:
            break
        h = _triangle_h(n)
        c5 = _c5_of(h)
        if c5 < best_c5:
