def max_overlap_count(h):
    """Return the integer maximum overlap of h with (1‑h)."""
    # h is binary integer array
    return int(np.max(np.correlate(h, 1 - h, mode='full')))

    """Indices i where h[i]=1 and shifted partner is 0 for the given shift."""
    n = h.shape[0]
    if shift >= 0:
        i_range = np.arange(0, n - shift)
        mism = i_range[(h[i_range] == 1) & (h[i_range + shift] == 0)]
    else:
        s = -shift
        i_range = np.arange(s, n)
        mism = i_range[(h[i_range] == 1) & (h[i_range - s] == 0)]
    return mism

def improve_h_local(h, n_points):
    """
    Deterministic local improvement.
    Repeatedly tries swaps between a 1 that contributes to the current worst shift
    and any 0, accepting a swap that strictly reduces the maximal overlap.
    """
    while True:
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = int(corr.max())
        if max_val == 0:
            break
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n_points - 1)

        mism = compute_mismatches(h, shift)
        if mism.size == 0:
            break

        zeros = np.flatnonzero(h == 0)
        best_swap = None
        best_new_max = max_val

        # exhaustive search of all swaps (i from mismatching 1's, j from zeros)
        for i in mism:
            for j in zeros:
                if i == j:
                    continue
                new_h = h.copy()
                new_h[i] = 0
                new_h[j] = 1
                new_max = max_overlap_count(new_h)
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_swap = (i, j)
                    # a reduction by 1 is the best possible in one move
                    if best_new_max == max_val - 1:

        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            continue
        else:
