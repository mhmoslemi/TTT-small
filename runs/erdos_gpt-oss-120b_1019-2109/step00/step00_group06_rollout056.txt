    grad = np.zeros_like(h)
    # w array corresponds to shifts from -(n-1) to n-1 inclusive.
    shifts = np.arange(-n+1, n)
    for ws, s in zip(w, shifts):
        if ws == 0:
            continue
        if s >= 0:
            # first factor: indices i where i+s < n
            i_end = n - s
            if i_end > 0:
                grad[:i_end] += ws * (1 - h[s:])  # (1 - h[i+s]) for i = 0..i_end-1
            # second factor: i where i - s >=0 => i >= s
            if s < n:

            if k < n:
                grad[:n - k] -= ws * h[k:]
    return grad

import time, math, random
import numpy as np

def _proj_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] while forcing sum(v)=target."""
    lo, hi = -1e6, 1e6
    for _ in range(60):
        mid = (lo + hi) * 0.5
        z = np.clip(v - mid, 0.0, 1.0)
        s = z.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a given h."""
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * (2.0 / n)


def _grad_from_weights(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the weighted max‐correlation w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    off = n - 1
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - off          # shift
        if s >= 0:
            end = n - s
            if end > 0:
                grad[:end] += weight * (1 - h[s:])
            if s < n:
                grad[s:] -= weight * h[:n - s]
        else:
            k = -s
            if k < n:
                grad[k:] += weight * (1 - h[:n - k])
                grad[:n - k] -= weight * h[k:]
    return grad


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random feasible vector (values in [0,1], sum=n/2)."""
    v = np.random.rand(n)
    return _proj_sum_bounds(v, n / 2.0)

        h0 = np.full(n, 0.5, dtype=np.float64)

    best_h = h0.copy()
    best_c5 = _c5_of(best_h)

    start = time.time()
    limit = budget_s * 0.95
