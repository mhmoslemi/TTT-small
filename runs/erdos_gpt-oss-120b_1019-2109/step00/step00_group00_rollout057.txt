def gradient_for_shift(h, s, dx):
    n = len(h)
    grad = np.zeros_like(h)
    if s >= 0:
        # first term contributions
        grad[:n-s] += (1 - h[s:])   # shape
        # second term contributions
        grad[s:] -= h[:n-s]
    else:
        t = -s
        grad[t:] += (1 - h[:n-t])
        grad[:n-t] -= h[t:]
    grad *= dx
    return grad

def project_onto_constraints(v, target_sum):
    # clip
    v = np.clip(v, 0.0, 1.0)
    # adjust sum
    # Use water-filling iterative:
    max_iter = 100
    for _ in range(max_iter):
        total = v.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        # Determine free indices where not at bounds after clipping
        free = (v > 0) & (v < 1)
        n_free = free.sum()
        if n_free == 0:
            # cannot adjust exactly; break; maybe adjust all in small uniform
            # but then some will violate bounds.
            # We'll distribute diff equally to all entries and clip again
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / n_free
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final scaling maybe adjust small residual diff by distributing to free ones again without clipping
    # Ensure final sum close
    return v

def compute_gradient(h, shift):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[:n-shift] += 1.0 - h[shift:]
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        grad[s:] += 1.0 - h[:n-s]
        grad[:n-s] -= h[s:]
    grad *= dx
    return grad

def project_to_feasible(v, target_sum):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # water-filling method
    # We'll iterate a few times (max maybe 100) to adjust sum with bounds.
    for _ in range(100):
        total = v.sum()
        diff = target_sum - total
        if abs(diff) < 1e-12:
            break
        free = (v > 0) & (v < 1)
        n_free = free.sum()
        if n_free == 0:
            # distribute diff uniformly and clip again
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / n_free
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # final adjustment: ensure sum close
    # maybe adjust a few entries by small epsilon to exactly target.
    # We'll compute final diff and if still nonzero > tolerance, distribute across free ones ignoring bounds (they should be away from edges).
    diff = target_sum - v.sum()
    if abs(diff) > 1e-8:
        # add diff to some entries not near bound
        idx = np.where((v > 1e-12) & (v < 1 - 1e-12))[0]
        if len(idx) == 0:
            # adjust any
            idx = np.arange(len(v))
        # distribute diff equally
        v[idx] += diff / len(idx)
        v = np.clip(v, 0.0, 1.0)
    return v

def improve_h(h, n_iters, target_sum, time_deadline):
    best_h = h.copy()
    best_c5, _ = compute_c5(best_h)
    cur_h = h.copy()
    cur_c5 = best_c5
    n = len(h)
    for it in range(n_iters):
        now = time.time()
        if now >= time_deadline:
            break
        # compute gradient of leading shift
        _, shift = compute_c5(cur_h)
        grad = compute_gradient(cur_h, shift)
        # step size schedule: step = step0 / sqrt(it+1)
        step = 0.5 / np.sqrt(it + 1)
        # propose new
        new_h = project_to_feasible(cur_h - step * grad, target_sum)
        new_c5, _ = compute_c5(new_h)
        if new_c5 < cur_c5 - 1e-9:
            cur_h = new_h
            cur_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_h = new_h.copy()
                best_c5 = new_c5
        else:
            # maybe random perturbation
            # we can attempt a small random swap
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # move random delta
            max_transfer = min(cur_h[i], 1 - cur_h[j])
            if max_transfer <= 0:
                continue
            delta = np.random.rand() * max_transfer
            pert_h = cur_h.copy()
            pert_h[i] -= delta
            pert_h[j] += delta
            # Ensure sum unchanged
            pert_c5, _ = compute_c5(pert_h)
            if pert_c5 < cur_c5:
                cur_h = pert_h
                cur_c5 = pert_c5
                if pert_c5 < best_c5:
                    best_h = pert_h.copy()
                    best_c5 = pert_c5
    return best_h, best_c5

def shape_trapezoid(n, r):
    dx = 2.0 / n
    # positions define at midpoints x_i = (i+0.5)*dx
    xs = (np.arange(n) + 0.5) * dx
    m = 1.0 / (2.0 - r)
    # ramp region left: xs < r
    # plateau region: r <= xs <= 2 - r
    # right ramp: xs > 2 - r
    h = np.empty(n)
    left_mask = xs < r
    plateau_mask = (xs >= r) & (xs <= 2 - r)
    right_mask = xs > 2 - r
    # linear ramp from 0 at x=0 to m at x=r:
    h[left_mask] = (m / r) * xs[left_mask]   (if r>0)
    # plateau
    h[plateau_mask] = m
    # right ramp: decreasing linearly
    # For xs > 2 - r, distance from right edge: d = 2 - xs
    # h = (m / r) * d
    h[right_mask] = (m / r) * (2 - xs[right_mask])
    return h

import time
import numpy as np

def _eval_c5(h):
    """C5 bound for a discretised step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx, corr.argmax() - (n - 1)

def _gradient(h, shift):
    """Sub‑gradient of the maximal correlation term."""
    n = h.shape[0]
    dx = 2.0 / n
    grad = np.zeros_like(h)
    if shift >= 0:
        grad[: n - shift] += 1.0 - h[shift:]
        grad[shift:] -= h[: n - shift]
    else:
        s = -shift
        grad[s:] += 1.0 - h[: n - s]
        grad[: n - s] -= h[s:]
    grad *= dx
    return grad

def _project_feasible(v, target_sum):
    """Project onto {0≤h≤1, sum(h)=target_sum} (water‑filling)."""
    v = np.clip(v, 0.0, 1.0)
    for _ in range(100):
        cur = v.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (v > 0.0) & (v < 1.0)
        m = free.sum()
        if m == 0:
            v += diff / len(v)
            v = np.clip(v, 0.0, 1.0)
            continue
        delta = diff / m
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
    # tiny final adjustment (should be negligible)
    diff = target_sum - v.sum()
    if abs(diff) > 1e-10:
        idx = np.where((v > 1e-12) & (v < 1 - 1e-12))[0]
        if idx.size == 0:
            idx = np.arange(len(v))
        v[idx] += diff / len(idx)
        v = np.clip(v, 0.0, 1.0)
    return v

def _improve(h0, target_sum, deadline, max_iters=20000):
    """Local optimisation using a sub‑gradient step + tiny random swaps."""
    h = h0.copy()
    cur_val, _ = _eval_c5(h)
    best_h, best_val = h.copy(), cur_val
    n = h.shape[0]

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # sub‑gradient descent on the active shift
        _, shift = _eval_c5(h)
        g = _gradient(h, shift)
        step = 0.4 / np.sqrt(it + 1.0)
        cand = _project_feasible(h - step * g, target_sum)
        cand_val, _ = _eval_c5(cand)

        # accept if improvement
        if cand_val < cur_val - 1e-12:
            h, cur_val = cand, cand_val
        if cand_val < best_val - 1e-12:
            best_h, best_val = cand.copy(), cand_val

        # occasional random exchange to escape flats
        if np.random.rand() < 0.15:
            i, j = np.random.choice(n, 2, replace=False)
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer > 0:
                delta = np.random.rand() * max_transfer
                h2 = h.copy()
                h2[i] -= delta
                h2[j] += delta
                val2, _ = _eval_c5(h2)
                if val2 < cur_val - 1e-12:
                    h, cur_val = h2, val2
                if val2 < best_val - 1e-12:
                    best_h, best_val = h2.copy(), val2

    return best_h, best_val

def _shape_from_r(n, r):
    """Symmetric trapezoid: ramp width r, peak m=1/(2‑r)."""
    if r <= 0.0:
        return np.full(n, 0.5)
    xs = (np.arange(n) + 0.5) * (2.0 / n)
    m = 1.0 / (2.0 - r)
    h = np.empty(n)
    left = xs < r
    right = xs > (2.0 - r)
    mid = ~(left | right)
    h[left] = (m / r) * xs[left]
    h[mid] = m
    h[right] = (m / r) * (2.0 - xs[right])
    return h
