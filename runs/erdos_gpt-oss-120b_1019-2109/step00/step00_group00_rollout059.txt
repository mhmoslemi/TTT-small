import time, math
import numpy as np

# ------------------------------------------------------------
# Helper to evaluate the Erdős overlap constant for a vector
# ------------------------------------------------------------
def _raw_c5(h: np.ndarray) -> float:
    """Return the un‑scaled maximum of the correlation h⋆(1‑h)."""
    return np.max(np.correlate(h, 1 - h, mode="full"))


# ------------------------------------------------------------
# Simple hill‑climbing refinement (keeps the sum exactly N/2)
# ------------------------------------------------------------
def _hill_climb(start_h, max_iters, time_deadline, rng):
    N = start_h.shape[0]
    h = start_h.copy()
    best_h = h.copy()
    best_raw = _raw_c5(h)
    cur_raw = best_raw

    # initial step size; will decrease if we stall
    step = 0.3
    stall = 0
    it = 0
    while it < max_iters and time.time() < time_deadline:
        it += 1
        # pick two distinct indices
        i, j = rng.integers(N, size=2)
        if i == j:
            continue

        # we will move mass from one index to the other
        if h[i] <= 0.0 and h[j] <= 0.0:
            continue
        if h[i] >= 1.0 and h[j] >= 1.0:
            continue

        # direction: increase i, decrease j  (or opposite)
        if rng.random() < 0.5:
            up, down = i, j
        else:
            up, down = j, i

        max_inc = min(step, 1.0 - h[up])
        max_dec = min(step, h[down])
        delta = min(max_inc, max_dec)
        if delta <= 0.0:
            continue
        # small randomisation of delta
        delta *= rng.random()

        # apply move
        h[up] += delta
        h[down] -= delta

        new_raw = _raw_c5(h)

        # accept only improving moves (pure hill climb)
        if new_raw <= cur_raw + 1e-12:
            cur_raw = new_raw
            if new_raw < best_raw - 1e-12:
                best_raw = new_raw
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            # revert
            h[up] -= delta
            h[down] += delta
            stall += 1

        # adapt step size if we have stalled for a while
        if stall > 5000:
            step = max(0.01, step * 0.5)
            stall = 0

    return best_h, best_raw

    """

    start_time = time.time()

    # N = 44 gives dx = 2/44 and a target raw max ≤ 8 (since 8*dx≈0.3636).
    # --------------------------------------------------------
    N = 44
    dx = 2.0 / N
    target_c5 = 0.3808        # we aim to be below this
    target_raw = int(math.floor(target_c5 / dx + 1e-12))

    # --------------------------------------------------------
    # Attempt an exact integer programming solution.
    # Variables: binary h_i, binary w_{i,k} for each shift.
    # Minimise the worst‑case directional mismatch count z.
    # --------------------------------------------------------
    best_h = None
    best_raw = float("inf")
    attempt_ip = False
    # try a few solvers sequentially; any that works gives a solution
    try:
        import cvxpy as cp

        # define variables
        h = cp.Variable(N, boolean=True)
        z = cp.Variable(integer=True, nonneg=True)

        constraints = [cp.sum(h) == N // 2]          # exactly half the entries are 1
        # for each shift k = 1..N-1 enforce forward and backward mismatches ≤ z
        for k in range(1, N):
            L = N - k
            h1 = h[:L]
            h2 = h[k:]
            w = cp.Variable(L, boolean=True)       # w = h1 AND h2
            constraints += [
                w <= h1,
                w <= h2,
                w >= h1 + h2 - 1,
                cp.sum(h1 - w) <= z,                # forward mismatches
                cp.sum(h2 - w) <= z,                # backward mismatches
            ]

        prob = cp.Problem(cp.Minimize(z), constraints)

        attempt_ip = False

    if attempt_ip and z.value is not None:

        # sanity: enforce bounds and exact sum
        h_val = np.clip(h_val, 0, 1)

        raw = _raw_c5(h_val)

        # if the IP solution already beats the target, return immediately
        if raw * dx <= target_c5:

    # Build a reasonable initial vector: spread the ones as evenly as possible.

    remaining = budget_s - (time.time() - start_time) - 0.5
    if remaining < 5.0:

    # Run hill climbing with a generous iteration budget
    raw_limit = target_raw + 2  # safety margin
    hill_h, hill_raw = _hill_climb(

        max_iters=200_000,

    if hill_raw < best_raw:
        best_h, best_raw = hill_h, hill_raw

    # --------------------------------------------------------
    # Final check and return
    # --------------------------------------------------------
    final_c5 = best_raw * dx
    # Clip to [0,1] and enforce exact sum (tiny numerical drift possible)
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    # correct any drift by adjusting a random non‑boundary element
    if abs(cur_sum - N / 2) > 1e-9:
        diff = (N / 2) - cur_sum
        # pick an index that is not at its bound to absorb the diff
        candidates = np.where((best_h > 0) & (best_h < 1))[0]

            idx = 0
        else:
            idx = rng.choice(candidates)
        best_h[idx] = max(0.0, min(1.0, best_h[idx] + diff))

    return best_h, final_c5, N
