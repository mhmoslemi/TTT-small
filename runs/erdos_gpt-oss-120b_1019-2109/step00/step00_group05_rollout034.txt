```
def initial_shape(N):
    best_h = None
    best_c5 = np.inf
    for S in range(0, N//2 + 1):
        L = N - 2*S
        H = N / (2 * (N - S))
        # Build h
        if S > 0:
            up = H * np.arange(1, S+1) / (S+1)
            down = up[::-1]
        else:
            up = np.array([], dtype=float)
            down = np.array([], dtype=float)
        plateau = np.full(L, H)
        h = np.concatenate([up, plateau, down])
        # Ensure length N
        assert h.shape[0] == N
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    return best_h, best_c5
```

def compute_c5(h):

import time
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return the C5 value for a step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _repair(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust the sum to target_sum while staying feasible."""
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    diff = target_sum - s
    if abs(diff) < 1e-12:
        return h
    # indices that can still move without hitting a bound
    free = (h > 0.0) & (h < 1.0)
    if free.any():
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    else:
        # all entries are at a bound – just correct the first one
        i = 0
        h[i] = np.clip(h[i] + diff, 0.0, 1.0)
    return h
