def anneal(h, max_iters, start_temp, end_temp, start_time, time_limit):
    n = len(h)
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # Precompute indices sets always keep track
    # Actually we can recompute each iteration.
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        t = start_temp + (end_temp - start_temp) * (it / max_iters)  # linear cooling
        # choose swap
        ones = np.where(cur_h > 0.5)[0]
        zeros = np.where(cur_h <= 0.5)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_val = compute_c5(cur_h)
        delta = new_val - cur_val
        if delta < 0 or np.random.rand() < np.exp(-delta / (t + 1e-12)):
            # accept
            cur_val = new_val
            # update best
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # reject: revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        # continue
    return best_h, best_val
```

If we remove a 1 at position i (set to 0) and add a 1 at position j (set to 1), then for shift k0:

- For each position p such that either p = i or p = i - k0, the product h_p h_{p + k0} will be affected because h_i becomes 0. Initially contributions were:

- For new 1 at position j, contributions increase:

   - If j - k0 valid: before = h_{j - k0} * 0 = 0; after = h_{j - k0} * 1 = h_{j - k0} -> change = + h_{j - k0}
   - If j + k0 valid: before = 0 * h_{j + k0} = 0; after = 1 * h_{j + k0} = h_{j + k0} -> change = + h_{j + k0}

Thus total delta_S_k0 = + (h_{j - k0} if valid) + (h_{j + k0} if valid) - (h_{i - k0} if valid) - (h_{i + k0} if valid).

                    h[i] = 0

        best_n = 2
        best_c5 = compute_c5(best_h)
    # ensure sum constraint
    # return
    return best_h, best_c5, int(best_n)

def binary_annealing(h_init, max_iters, time_limit=None):
    import time, numpy as np, math, random
    n = len(h_init)
    h = h_init.copy()
    # Ensure binary
    h = (h > 0.5).astype(np.int8)
    # adjust sum to n/2 exactly
    current_sum = h.sum()
    target = n // 2
    if current_sum > target:
        # turn some ones to zeros
        ones = np.where(h == 1)[0]
        to_flip = current_sum - target
        idx = np.random.choice(ones, size=to_flip, replace=False)
        h[idx] = 0
    elif current_sum < target:
        zeros = np.where(h == 0)[0]
        to_flip = target - current_sum
        idx = np.random.choice(zeros, size=to_flip, replace=False)
        h[idx] = 1
    # compute correlation
    corr = np.correlate(h, h, mode='full')
    cur_min = corr.min()
    cur_c5 = 1 - (2 * cur_min) / n
    best_h = h.copy()
    best_c5 = cur_c5
    indices = np.arange(n)
    # precompute offsets matrix rows
    offsets = indices[None, :] - indices[:, None] + (n - 1)  # shape (n,n)
    if time_limit is not None:
        t0 = time.time()
    for it in range(max_iters):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        # temperature schedule
        T = 2.0 * (1 - it / max_iters) + 0.001
        # choose random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # compute delta
        delta = np.zeros(2 * n - 1, dtype=int)
        delta[offsets[i]] -= h
        delta[n - 1] += h[i]  # cancel self removal
        delta[offsets[j]] += h
        # h[j]==0 no self addition cancellation needed (or subtract 0)
        new_corr = corr + delta
        new_min = new_corr.min()
        # acceptance
        accept = False
        if new_min >= cur_min:
            accept = True
        else:
            prob = math.exp((new_min - cur_min) / T)
            if np.random.rand() < prob:
                accept = True
        if accept:
            # swap bits
            h[i] = 0
            h[j] = 1
            corr = new_corr
            cur_min = new_min
            cur_c5 = 1 - (2 * cur_min) / n
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        # else keep unchanged
    return best_h.astype(float), best_c5

import time
import math
import numpy as np

def compute_c5(h):
    """C5 value for a step function sampled in n points."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_to_feasible(x, target_sum, max_iter=60):
    """
    Project x onto the box [0,1] together with the equality
    sum(x) = target_sum using a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -1.0, 1.0
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(x - lam, 0.0, 1.0)


def gradient_descent_h(h0, max_iters=3000, start_step=0.6, end_step=0.02, time_limit=None):
    """
    Simple projected sub‑gradient descent on the original (continuous) problem.
    The objective is max_k  ∫ h(x)(1−h(x+k))dx, i.e. minimise the maximal mismatch.
    """
    n = h0.shape[0]
    target = n / 2.0
    h = h0.copy()
    start = time.time()
    for it in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break

        # autocorrelation and current mismatch profile
        corr = np.correlate(h, h, mode="full")
        w = h.sum()
        d = w - corr                     # d_k = w - S_k
        maxd = d.max()
        # active shifts (those attaining the maximum)
        mask = d >= maxd - 1e-12
        shifts = np.where(mask)[0] - (n - 1)
        cnt = shifts.size
        if cnt == 0:
            break

        # sub‑gradient of the max‑functional
        g = np.full(n, cnt, dtype=float)
        for k in shifts:
            if k > 0:
                g[:-k] -= h[k:]
                g[k:]  -= h[:-k]
            elif k < 0:
                kk = -k
                g[kk:] -= h[:-kk]
                g[:kk] -= h[kk:]
            else:                     # k == 0
                g -= 2.0 * h

        # step size schedule (linear)
        step = start_step + (end_step - start_step) * (it / max_iters)
        h_new = h - step * g
        h_new = _project_to_feasible(h_new, target)

        # accept only improvements (occasionally accept neutral moves)
        if compute_c5(h_new) <= compute_c5(h):
            h = h_new
    return h, compute_c5(h)


def binary_annealing(h0, max_iters=20000, time_limit=None):
    """
    Simulated‑annealing on binary vectors with exact half‑density.
    The move set is a swap of a 1 and a 0 which preserves the sum.
    """
    n = h0.shape[0]
    target_ones = n // 2
    # force binary
    h = (h0 > 0.5).astype(np.int8)
    cur_ones = h.sum()
    if cur_ones > target_ones:               # too many ones
        idx = np.where(h == 1)[0]
        drop = cur_ones - target_ones
        h[np.random.choice(idx, drop, replace=False)] = 0
    elif cur_ones < target_ones:             # too few ones
        idx = np.where(h == 0)[0]
        add = target_ones - cur_ones
        h[np.random.choice(idx, add, replace=False)] = 1

    # initial correlation and C5
    corr = np.correlate(h, h, mode="full")
    cur_min = corr.min()
    cur_c5 = 1.0 - (2.0 * cur_min) / n
    best_h, best_c5 = h.copy(), cur_c5

    # pre‑compute offset rows for rapid delta updates
    idx = np.arange(n)
    offsets = idx[None, :] - idx[:, None] + (n - 1)   # shape (n,n)

    start = time.time()
    for it in range(max_iters):
        if time_limit is not None and time.time() - start > time_limit:
            break

        # temperature schedule
        T = 2.0 * (1.0 - it / max_iters) + 1e-4

        # pick a random 1–0 pair to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)

        # compute correlation change caused by the swap
        delta = np.zeros(2 * n - 1, dtype=int)
        delta[offsets[i]] -= h
        delta[n - 1] += h[i]          # cancel the self‑term for i
        delta[offsets[j]] += h
        # (h[j] == 0, so no need to cancel a self‑addition)

        new_corr = corr + delta
        new_min = new_corr.min()

        # accept if not worsening, otherwise with Metropolis probability
        accept = False
        if new_min >= cur_min:
            accept = True
        else:
            prob = math.exp((new_min - cur_min) / T)
            if np.random.rand() < prob:
                accept = True

        if accept:
            # perform the swap permanently
            h[i] = 0
            h[j] = 1
            corr = new_corr
            cur_min = new_min
            cur_c5 = 1.0 - (2.0 * cur_min) / n
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h.astype(float), best_c5


def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that minimises
        C5 = max_k ∫ h(x)(1−h(x+k))dx .
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)

    start_global = time.time()
    best_h, best_val, best_n = None, float("inf"), None

    # --------------------------------------------------------------------
    # 1.  Use the supplied initial construction if present.
    # --------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_init = np.array(initial_h_values, dtype=float)
        n_init = h_init.shape[0]
        target_sum = n_init / 2.0
        h_init = _project_to_feasible(h_init, target_sum)
        # try a short gradient‑descent refinement
        time_left = budget_s - (time.time() - start_global)
        if time_left > 5:
            gd_h, gd_val = gradient_descent_h(
                h_init,
                max_iters=2500,
                start_step=0.5,
                end_step=0.01,
                time_limit=time_left * 0.4,
            )
            if gd_val < best_val:
                best_h, best_val, best_n = gd_h, gd_val, n_init

    # --------------------------------------------------------------------
    # 2.  Run binary simulated annealing on several even lengths.
    # --------------------------------------------------------------------
    even_lengths = [40, 44, 48, 56, 64, 80, 96, 112]
    for n in even_lengths:
        if time.time() - start_global > budget_s:
            break
        # starting vector: either the projected initial one (if same length) or random
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            h0 = np.array(initial_h_values, dtype=float)
            target_sum = n / 2.0
            h0 = _project_to_feasible(h0, target_sum)
        else:
            h0 = np.zeros(n, dtype=float)
            h0[: n // 2] = 1.0
            np.random.shuffle(h0)

        time_left = budget_s - (time.time() - start_global)
        if time_left < 5:
            break
        # allocate time proportionally (but not less than a few seconds)
        t_alloc = max(2.0, time_left * 0.15)

        # binary annealing
        ba_h, ba_val = binary_annealing(
            h0,
            max_iters=20000,
            time_limit=t_alloc,
        )
        if ba_val < best_val:
            best_h, best_val, best_n = ba_h, ba_val, n

    # --------------------------------------------------------------------
    # 3.  Fallback – a trivial construction if nothing succeeded.
    # --------------------------------------------------------------------
    if best_h is None:
        best_n = 2
        best_h = np.array([1.0, 0.0])
        best_val = compute_c5(best_h)

    # final sanity checks
    n_final = best_h.shape[0]
    dx = 2.0 / n_final
    assert best_h.min() >= -1e-12 and best_h.max() <= 1.0 + 1e-12
    assert abs(best_h.sum() * dx - 1.0) < 1e-9

    return best_h, best_val, int(n_final)
