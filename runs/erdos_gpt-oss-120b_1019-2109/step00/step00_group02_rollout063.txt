def compute_c5(h):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / len(h))

def compute_smooth_gradient(h, alpha):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # weights: exp(alpha*(corr - max_corr))
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    w = w / w_sum
    # compute shift_plus and shift_minus via convolution:
    w_rev = w[::-1]
    # conv with h:
    shift_minus = np.convolve(w, h, mode='full')[n-1:n-1+n]  # length n
    shift_plus = np.convolve(w_rev, h, mode='full')[n-1:n-1+n]
    grad = np.ones(n) - shift_minus - shift_plus
    return grad, np.max(corr)  # return raw max (without dx)

def compute_smooth_gradient(h, alpha):

    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    # We'll compute weights for each shift index
    w = np.exp(alpha * (corr - max_corr))
    w_sum = np.sum(w)
    w = w / w_sum

    # shift_minus = np.convolve(w, h, mode='full')[n-1:n-1+n]

    conv_full_rev = np.convolve(w[::-1], h, mode='full')

def project_to_box_simplex(y, target):
    n = y.shape[0]
    lo = -np.max(y) - 1.0
    hi = 1.0 - np.min(y) + 1.0

            hi = lam
        else:

    corr = np.correlate(h, 1 - h, mode='full')

import numpy as np
import time

# ---------- helper functions ----------

def _project_to_box_simplex(y: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector y onto the intersection of the box [0,1]^n and the affine
    hyperplane sum(x)=target.  Uses a simple bisection on the Lagrange multiplier.
    """
    n = y.shape[0]
    lo = -np.max(y) - 1.0          # large negative -> all entries become 0
    hi = 1.0 - np.min(y) + 1.0     # large positive -> all entries become 1
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    return np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)


def _c5_value(h: np.ndarray) -> float:
    """exact C5 value for a discrete step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))


def _max_shift_subgradient(h: np.ndarray):
    """
    Subgradient of the max‑overlap objective (the shift that attains the maximum).
    Returns the gradient vector and the current maximal correlation value.
    """
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)          # shift ∈ [-(n-1), n-1]

    grad = np.ones(n)

    if shift >= 0:
        # shift = +s
        s = shift
        if s < n:
            grad[: n - s] -= h[s:]
            grad[s:] -= h[: n - s]
    else:
        # shift = -t
        t = -shift
        if t < n:
            grad[: n - t] -= h[t:]
            grad[t:] -= h[: n - t]

    return grad, float(np.max(corr))


def _fejer_h(N: int, n: int) -> np.ndarray:
    """
    Build a Fejér‑kernel based step function with period 1 repeated twice
    on the interval [0,2].
    """
    # sample points (mid‑point rule)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    xm = np.mod(x, 1.0)                     # period‑1 coordinate
    sin_pi = np.sin(np.pi * xm)

    eps = 1e-12
    safe = np.where(np.abs(sin_pi) < eps, eps, sin_pi)

    # Fejér kernel:  (1/N) * (sin(Nπx)/sin(πx))**2
    F = (np.sin(np.pi * N * xm) / safe) ** 2 / N
    F = np.where(np.abs(sin_pi) < eps, float(N), F)

    S = (F - 1.0) / (N - 1.0)               # zero‑mean, range ~[-1/(N‑1),1]
    h = 0.5 + 0.5 * S
    h = np.clip(h, 0.0, 1.0)
    h = _project_to_box_simplex(h, n / 2)
    return h


# ---------- main optimisation routine ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # ---- choose discretisation size ----
    n = 300                     # even value < 1000
    if n % 2:
        n += 1
    target_sum = n / 2

    best_h = None
    best_val = np.inf

    # ---- try a family of Fejér constructions (quick cheap candidates) ----
    for N in range(2, 31, 2):               # N = 2,4,...,30
        h = _fejer_h(N, n)
        val = _c5_value(h)
        if val < best_val:
            best_val = val
            best_h = h.copy()

    # ---- incorporate the supplied initial construction if present ----
    try:
        init_h = np.asarray(initial_h_values, dtype=float)   # may raise NameError
        if init_h.shape[0] != n:
            # linear interpolation to the chosen resolution
            x_old = np.linspace(0.0, 2.0, init_h.shape[0], endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            from scipy.interpolate import interp1d
            f = interp1d(x_old, init_h, kind="linear", fill_value="extrapolate")
            init_h = np.clip(f(x_new), 0.0, 1.0)
        init_h = _project_to_box_simplex(init_h, target_sum)
        val = _c5_value(init_h)
        if val < best_val:
            best_val = val
            best_h = init_h.copy()
    except NameError:
        pass  # no initial_h_values supplied

    # ---- sub‑gradient descent from several starts ----
    max_time = budget_s * 0.95
    base_step = 0.7
    max_restarts = 8
    it_per_restart = 800

    for restart in range(max_restarts):
        if time.time() - start > max_time:
            break

        # initialise a new starting point
        if restart == 0 and best_h is not None:
            h = best_h.copy()
        else:
            # random point then project to feasible set
            h = _project_to_box_simplex(rng.random(n), target_sum)

        for it in range(it_per_restart):
            if time.time() - start > max_time:
                break

            grad, cur_corr = _max_shift_subgradient(h)
            step = base_step / (1.0 + it * 0.001)   # slowly diminishing step size
            h = _project_to_box_simplex(h - step * grad, target_sum)

            # periodic exact check (every 32 steps) – cheap enough
            if (it & 31) == 0:
                val = _c5_value(h)
                if val < best_val:
                    best_val = val
                    best_h = h.copy()

        # after each restart, inject a small random perturbation to aid exploration
        if restart < max_restarts - 1:
            perturb = rng.normal(scale=0.04, size=n)
            h = _project_to_box_simplex(best_h + perturb, target_sum)

    # final evaluation
    final_val = _c5_value(best_h)

    return best_h, float(final_val), n
