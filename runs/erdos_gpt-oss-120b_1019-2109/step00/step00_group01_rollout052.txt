def compute_c5(h):
    n = h.shape[0]
    N = 2 * n
    # allocate arrays for FFT
    hpad = np.empty(N, dtype=np.float64)
    gpad = np.empty(N, dtype=np.float64)
    hpad[:n] = h
    hpad[n:] = 0.0
    gpad[:n] = 1.0 - h
    gpad[n:] = 0.0
    H = np.fft.rfft(hpad)
    G = np.fft.rfft(gpad)
    corr = np.fft.irfft(H * np.conj(G), n=N)
    raw = np.max(corr[:2*n-1])
    return raw * (2.0 / n)

    b = np.empty(N, dtype=np.float64)
    a[:n] = arr
    a[n:] = 0.0
    b[:n] = 1.0 - arr
    b[n:] = 0.0
    # convolution via FFT (correlation)
    H = np.fft.rfft(a)
    G = np.fft.rfft(b)
    corr = np.fft.irfft(H * np.conj(G), n=N)
    raw = np.max(corr[: 2 * n - 1])
    return raw * (2.0 / n)  # multiply by dx = 2/n


def _project_sum(vec: np.ndarray) -> None:
    """Adjust `vec` in‑place so that sum(vec) = len(vec)/2  (keeps 0≤vec≤1)."""
    n = vec.shape[0]
    target = n * 0.5
    cur = vec.sum()
    if abs(cur - target) < 1e-12:
        return
    # Uniform scaling then clipping
    if cur > 0:
        vec *= target / cur
    np.clip(vec, 0.0, 1.0, out=vec)
    cur = vec.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return
    # Distribute the remaining tiny discrepancy
    if diff > 0:
        for i in range(n):
            inc = min(diff, 1.0 - vec[i])
            if inc > 0:
                vec[i] += inc
                diff -= inc
                if diff <= 1e-12:
                    break
    else:
        for i in range(n):
            dec = min(-diff, vec[i])
            if dec > 0:
                vec[i] -= dec
                diff += dec
                if diff >= -1e-12:
                    break
    # final safeguard
    vec[:] = np.clip(vec, 0.0, 1.0)


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C₅ problem."""
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # 1.  Build an initial feasible vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=np.float64)
        # ensure the length is even (simplifies the binary‑move logic)
        if base.shape[0] % 2 == 1:
            h = np.tile(base, 2)          # repeat twice → even length
        else:
            h = base.copy()
    else:
        # default: 200 points, exactly half of them equal to 1
        n = 200
        h = np.zeros(n, dtype=np.float64)
        idx = np.random.choice(n, n // 2, replace=False)
        h[idx] = 1.0

    n = h.shape[0]
    _project_sum(h)                     # guarantees Σh = n/2 and 0≤h≤1

    # ------------------------------------------------------------------
    # 2.  Simulated‑annealing style optimisation
    # ------------------------------------------------------------------
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5

    total_iters = 800_000               # generous upper bound; we stop by time
    move_scale = 0.7                    # maximal fraction of transferable mass
    T_start = 0.020                     # initial temperature
    T_end = 0.0005                      # final temperature

    for it in range(total_iters):
        if time.time() - start > budget_s * 0.95:
            break

        # temperature schedule (linear)
        prog = it / total_iters
        T = T_start * (1 - prog) + T_end * prog

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # random direction of mass transfer
        if np.random.rand() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        # how much can we move while staying in [0,1]?
        max_delta = min(h[src], 1.0 - h[dst])
        if max_delta <= 1e-12:
            continue

        delta = max_delta * np.random.rand() * move_scale

        # apply tentative move
        h[src] -= delta
        h[dst] += delta

        new_c5 = compute_c5(h)
        diff = new_c5 - cur_c5

        # Metropolis acceptance
        if diff < 0 or np.random.rand() < math.exp(-diff / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert
            h[src] += delta
            h[dst] -= delta

    # ------------------------------------------------------------------
    # 3.  Final clean‑up (tiny rounding errors)
    # ------------------------------------------------------------------
    _project_sum(best_h)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n
