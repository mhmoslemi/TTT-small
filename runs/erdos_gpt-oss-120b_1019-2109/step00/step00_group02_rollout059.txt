import time, math, random
import numpy as np

def _compute_c5(h):
    """C5 value for a given step function h (discrete version)."""
    n = len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _local_search(start_h, max_seconds):
    """Simple simulated‑annealing/greedy improvement for a fixed n."""
    n = len(start_h)
    h = start_h.copy()

    best_val = _compute_c5(best_h)

    start = time.time()
    # annealing schedule parameters
    T0 = 0.12          # initial temperature
    T_min = 1e-5
    step0 = 0.2        # initial maximal mass move
    while time.time() - start < max_seconds:
        elapsed = time.time() - start
        # linear cooling
        T = max(T_min, T0 * (1.0 - elapsed / max_seconds))
        # step size shrinks with time
        step = max(0.002, step0 * (1.0 - elapsed / max_seconds))

        # correlation to locate the worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        k_idx = int(np.argmax(corr))
        k = k_idx - (n - 1)                # shift value, can be negative

        if k == 0:                         # choose a random non‑zero shift
            k = random.randint(-n + 1, n - 1)
            if k == 0:
                continue

        # indices that actually overlap for this shift
        if k >= 0:
            inds = np.arange(n - k)
            hi = h[inds]
