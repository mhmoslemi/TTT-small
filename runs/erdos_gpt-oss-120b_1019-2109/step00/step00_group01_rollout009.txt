```python
def gradient_active_shifts(h, conv, n):
    dx = 2.0 / n
    # find max value
    max_val = conv.max()
    # locate shifts where conv equals max (within tolerance)
    # conv indices correspond to shift = idx - (n-1)
    idxs = np.where(conv >= max_val - 1e-12)[0]
    grads = np.zeros_like(h)
    for idx in idxs:
        k = idx - (n - 1)
        if k >= 0:
            l = n - k
            i = np.arange(l)
            j = i + k
            grads[i] += dx * (1.0 - h[j])
            grads[j] -= dx * h[i]
        else:
            kp = -k
            l = n - kp
            j = np.arange(kp, n)
            i = j - kp
            grads[j] += dx * (1.0 - h[i])
            grads[i] -= dx * h[j]
    # average across active shifts
    grads /= max(1, len(idxs))
    return grads
```

import numpy as np
import time

# ---------- helper utilities ----------

def _project_to_box_sum(x: np.ndarray, target: float) -> np.ndarray:
    """
    Projection of x onto the set {0 ≤ y ≤ 1, sum(y)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0               # wide enough interval for the multiplier
    for _ in range(60):               # enough for double precision
        mid = (lo + hi) * 0.5
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s < target:                # multiplier too large → sum too small
            hi = mid
        else:                         # multiplier too small → sum too large
            lo = mid
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    return y


def _c5_and_corr(h: np.ndarray):
    """
    Return (C5 value, full correlation array) for a step function h.
    """
    n = h.size
    conv = np.convolve(h, (1.0 - h)[::-1], mode='full')
    c5 = conv.max() * (2.0 / n)
    return c5, conv


def _gradient_for_active_shifts(h: np.ndarray, conv: np.ndarray):
    """
    Sub‑gradient of max_k  ∑ h_i (1‑h_{i+k}) for the active (worst) shifts.
    """
    n = h.size
    dx = 2.0 / n
    max_val = conv.max()
    # tolerance to catch all ties (floating point noise)
    act = np.where(conv >= max_val - 1e-12)[0]
    grad = np.zeros_like(h)
    for idx in act:
        k = idx - (n - 1)                     # shift corresponding to this index
        if k >= 0:
            l = n - k
            i = np.arange(l)                  # 0 .. n‑k‑1
            j = i + k
            grad[i] += dx * (1.0 - h[j])
            grad[j] -= dx * h[i]
        else:
            kp = -k
            l = n - kp
            j = np.arange(kp, n)              # kp .. n‑1
            i = j - kp
            grad[j] += dx * (1.0 - h[i])
            grad[i] -= dx * h[j]
    if act.size:
        grad /= float(act.size)
    return grad


# ---------- main optimisation routine ----------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where
        h_values : ndarray of shape (n_points,)
        c5_bound : float  – the evaluated bound
        n_points : int    – number of discretisation points (≤1000)
    """
    np.random.seed(seed)
    start = time.time()

    # ---- initialise problem size and starting vector -----------------
    init = globals().get('initial_h_values', None)
    if init is not None:
        init = np.asarray(init, dtype=np.float64)
    # target number of points – a compromise between flexibility and speed
    n = 300
    if init is not None:
        # up‑sample the provided construction to the target resolution
        m = init.shape[0]
        if m != n:
            x_old = np.linspace(0.0, 2.0, m, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(x_new, x_old, init)
    else:
        init = np.random.rand(n)

    # enforce feasibility (0 ≤ h ≤ 1, integral = 1 → sum = n/2)
    h = _project_to_box_sum(init, n / 2.0)

    best_h = h.copy()
    best_c, _ = _c5_and_corr(best_h)

    # ---- optimisation parameters ------------------------------------
    max_iter_per_restart = 5000
    step_initial = 0.45          # raw step size before projection
    step_decay = 0.9995          # exponential decay per iteration
    global_step = step_initial

    # ---- main optimisation loop (multiple restarts) -----------------
    while time.time() - start < budget_s * 0.95:
        # a fresh start – keep the original up‑sampled pattern with probability 0.2
        if np.random.rand() < 0.2:
            h = _project_to_box_sum(init, n / 2.0)
        else:
            h = np.random.rand(n)
            h = _project_to_box_sum(h, n / 2.0)

        step = step_initial

        for _ in range(max_iter_per_restart):
            # time guard
            if time.time() - start >= budget_s * 0.95:
                break

            c_val, conv = _c5_and_corr(h)
            if c_val < best_c:
                best_c, best_h = c_val, h.copy()

            # sub‑gradient step
            grad = _gradient_for_active_shifts(h, conv)
            h_new = h - step * grad
            h_new = _project_to_box_sum(h_new, n / 2.0)

            # occasional stochastic shake to avoid stagnation
            if np.random.rand() < 0.001:
                i, j = np.random.choice(n, 2, replace=False)
                lo = max(-h_new[i], h_new[j] - 1.0)
                hi = min(1.0 - h_new[i], h_new[j])
                if lo < hi:
                    delta = np.random.uniform(lo, hi)
                    h_new[i] += delta
                    h_new[j] -= delta
                    h_new = _project_to_box_sum(h_new, n / 2.0)

            h = h_new
            step *= step_decay
            # avoid vanishing step
            if step < 1e-6:
                break

    # final evaluation using the provided helper (ensures exact formatting)
    c5_bound = evaluate_erdos_solution(best_h)

    return best_h, c5_bound, n
