        else:
            # need less mass – pick a place that can be decreased
            candidates = np.where(v > 1e-12)[0]
            if candidates.size == 0:
                break
            i = np.random.choice(candidates)
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
    return v


def is_binary(v: np.ndarray) -> bool:
    """Check whether all entries are (almost) 0 or 1."""
    return np.all(np.isclose(v, np.round(v), atol=1e-9))


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step function h (as a list), its C5 value, and the number of points.
    The routine tries to improve on the supplied `initial_h_values` if present.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ----------------------------------------------------------------------
    # 0. Acquire initial vector, enforce constraints, decide binary/continuous.
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = len(h)
    else:
        n = 70
        # simple deterministic start: first half 1, rest 0
        h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(np.float64)

    target_sum = n / 2.0
    h = fix_sum(h, target_sum)

    binary_mode = is_binary(h)

    dx = 2.0 / n
    best_h = h.copy()
    best_cost = compute_cost(best_h, dx)
    cur_h = h.copy()
    cur_cost = best_cost

    # ----------------------------------------------------------------------
    # 1. Simulated‑annealing / stochastic local search.
    # ----------------------------------------------------------------------
    T = 0.5                # starting temperature
    T_min = 1e-6
    decay = 0.9995         # exponential cooling factor
    start_time = time.time()
    it = 0

    while time.time() - start_time < budget_s - 0.4:
        it += 1

        # ---- propose a neighbour -------------------------------------------------
        if binary_mode:
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h <= 0.5)[0]
            if ones.size == 0 or zeros.size == 0:
                continue
            i = int(np.random.choice(ones))
            j = int(np.random.choice(zeros))
            new_h = cur_h.copy()
            new_h[i] = 0.0
            new_h[j] = 1.0
        else:
            # continuous move: shift a small amount from one coordinate to another
            i, j = np.random.choice(n, size=2, replace=False)
            if random.random() < 0.5:
                max_inc = 1.0 - cur_h[i]
                max_dec = cur_h[j]
                max_delta = min(max_inc, max_dec)
                if max_delta <= 0:
                    continue
                delta = np.random.uniform(0, max_delta)
                new_h = cur_h.copy()
                new_h[i] += delta
                new_h[j] -= delta
            else:
                max_inc = 1.0 - cur_h[j]
                max_dec = cur_h[i]
                max_delta = min(max_inc, max_dec)
                if max_delta <= 0:
                    continue
                delta = np.random.uniform(0, max_delta)
                new_h = cur_h.copy()
                new_h[i] -= delta
                new_h[j] += delta

        # ---- evaluate -----------------------------------------------------------
        new_cost = compute_cost(new_h, dx)

        # ---- acceptance ---------------------------------------------------------
        if new_cost < cur_cost or random.random() < np.exp(-(new_cost - cur_cost) / max(T, 1e-12)):
            cur_h = new_h
            cur_cost = new_cost
            if new_cost < best_cost - 1e-12:
                best_h = new_h.copy()
                best_cost = new_cost

        # ---- temperature update -------------------------------------------------
        T = max(T * decay, T_min)

        # ---- occasional reset to best -------------------------------------------
        if it % 25000 == 0:
            cur_h = best_h.copy()
            cur_cost = best_cost

        # ---- optional targeted move every few hundred iterations ---------------
        if it % 2000 == 0 and not binary_mode:
            # identify the shift that actually attains the maximum
            corr = np.correlate(cur_h, 1.0 - cur_h, mode="full")
            max_idx = int(np.argmax(corr))
            shift = max_idx - (n - 1)            # signed shift
            if shift != 0:
                # compute per‑position contributions for this shift
                if shift > 0:
                    contrib = cur_h[: n - shift] * (1.0 - cur_h[shift:])
                else:
                    s = -shift
                    contrib = cur_h[s:] * (1.0 - cur_h[: n - s])
                # pick a high‑contribution index and a low‑contribution index
                high = np.where(contrib > np.median(contrib))[0]
                low = np.where(contrib < np.median(contrib))[0]
                if high.size and low.size:
                    if shift > 0:
                        i = int(np.random.choice(high))
                        j = int(np.random.choice(low) + shift)
                    else:
                        i = int(np.random.choice(high) + (-shift))
                        j = int(np.random.choice(low))
                    # move a modest amount from i to j
                    delta = min(0.1, cur_h[i], 1.0 - cur_h[j])
                    if delta > 1e-8:
                        cur_h[i] -= delta
                        cur_h[j] += delta
                        cur_cost = compute_cost(cur_h, dx)
                        if cur_cost < best_cost - 1e-12:
                            best_h = cur_h.copy()
                            best_cost = cur_cost

    # ----------------------------------------------------------------------
    # 2. Return the best found construction.
    # ----------------------------------------------------------------------
    return (best_h.tolist(), float(best_cost), n)
