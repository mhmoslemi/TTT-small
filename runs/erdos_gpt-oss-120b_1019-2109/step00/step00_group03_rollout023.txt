    if s >= 0:

        # indices i where i - s >= 0, i >= s

        # compute contributions symmetrically:
        # For s negative, h_i*(1 - h_{i-s}) where i-s corresponds to i + s

        # Actually g_i = (if i + (-s) < n) (1 - h[i - s]) - (if i - (-s) >=0) h[i + s]

def project_onto_capped_simplex(v, target_sum, lower=0.0, upper=1.0):
    # Ensures result within [lower, upper] and sum equals target_sum
    # Using bisection on lambda
    lo = v.min() - upper
    hi = v.max() - lower
    for _ in range(100):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if abs(s - target_sum) < 1e-10:
            return x
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, lower, upper)
```

import time
import numpy as np

def _project_capped_simplex(v, target, lo=0.0, hi=1.0, eps=1e-12, iters=100):
    """ Euclidean projection of v onto {x | lo <= x_i <= hi, sum x = target}. """
    #  solve sum_i clip(v_i - lam, lo, hi) = target for lam
    lower = v.min() - hi
    upper = v.max() - lo
    for _ in range(iters):
        lam = (lower + upper) * 0.5
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if abs(s - target) <= eps:
            return x
        if s > target:
            lower = lam
        else:
            upper = lam
    return np.clip(v - lam, lo, hi)


def _worst_shift(h, dx):
    """Return (raw_corr_array, index_of_max, shift) for the current h."""
    raw = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(raw))
    shift = idx - (len(h) - 1)          # shift in index units
    return raw, idx, shift


def _gradient_for_shift(h, shift):
    """Subgradient of the overlap term at the given (worst) shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        L = n - shift
        # indicator part (1 where i+shift inside)
        g[:L] += 1.0
        # -h_{i+shift}
        g[:L] -= h[shift:]
        # -h_{i-shift}
        g[shift:] -= h[:L]
    else:
        s = -shift
        L = n - s
        g[s:] += 1.0
        g[s:] -= h[:L]
        g[:L] -= h[s:]
    return g


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size and basic parameters
    # ------------------------------------------------------------------
    if "initial_h_values" in kwargs and kwargs["initial_h_values"] is not None:
        h0 = np.asarray(kwargs["initial_h_values"], dtype=np.float64)
        n = h0.shape[0]
    else:
        n = 70                     # default size used by the parent harness
        h0 = None

    total = n // 2                # sum of h must be exactly n/2
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # prepare initial feasible point
    # ------------------------------------------------------------------
    if h0 is not None:
        h = np.clip(h0, 0.0, 1.0)
        h = _project_capped_simplex(h, total, 0.0, 1.0)
    else:
        # start from a random feasible vector
        v = rng.random(n)
        h = _project_capped_simplex(v, total, 0.0, 1.0)

    best_h = h.copy()
    best_val = np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx

    start_time = time.time()
    step = 0.5                     # initial step size
    stagnation = 0                 # counter for iterations without improvement

    while time.time() - start_time < budget_s * 0.95:
        raw, idx, shift = _worst_shift(h, dx)
        cur_val = raw[idx] * dx

        # keep the best ever seen
        if cur_val < best_val - 1e-9:
            best_val = cur_val
            best_h = h.copy()
            stagnation = 0
            # stop early if we have already crossed the target record
            if best_val <= 0.3808:
                break
        else:
            stagnation += 1

        # compute (sub)gradient for the worst shift (average if several tie)
        max_raw = raw[idx]
        active = np.where(np.abs(raw - max_raw) < 1e-12)[0]
        grad = np.zeros_like(h)
        for a in active:
            s = a - (n - 1)
            grad += _gradient_for_shift(h, s)
        grad /= max(1, len(active))

        # gradient step and projection
        new_h = _project_capped_simplex(h - step * grad, total, 0.0, 1.0)

        new_raw = np.correlate(new_h, 1.0 - new_h, mode="full")
        new_val = new_raw.max() * dx

        # accept if improvement, otherwise shrink step
        if new_val < cur_val - 1e-12:
            h = new_h
            step = min(step * 1.05, 2.0)      # slight acceleration
        else:
            step *= 0.7
            if step < 1e-4:                    # too small → random perturbation
                pert = h + rng.normal(scale=0.1, size=n)
                h = _project_capped_simplex(pert, total, 0.0, 1.0)
                step = 0.5

        # occasional full restart if stuck for many iterations
        if stagnation >= 2000:
            v = rng.random(n)
            h = _project_capped_simplex(v, total, 0.0, 1.0)
            step = 0.5
            stagnation = 0

    # final safety check – if nothing improved, fall back to the initial vector
    if best_h is None or best_h.shape[0] == 0:
        best_h = h.copy()
        best_val = np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx

    return (best_h.astype(np.float64), float(best_val), n)
