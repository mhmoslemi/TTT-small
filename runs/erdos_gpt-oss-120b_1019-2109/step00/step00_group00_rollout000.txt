        it += 1

        # pick two distinct indices
        i, j = random.sample(range(n), 2)
        hi, hj = best_h[i], best_h[j]

        # adapt step size with temperature
        cur_step = max_step * (temp / temp_start)
        cur_step = max(cur_step, 1e-8)

        # feasible epsilon interval that keeps both entries in [0,1]
        low = max(-cur_step, -hi, hj - 1.0)
        high = min(cur_step, 1.0 - hi, hj)
        if low > high:
            temp *= cooling
            continue

        eps = random.random() * (high - low) + low

        # construct candidate
        cand = best_h.copy()
        cand[i] = hi + eps
        cand[j] = hj - eps

        cand_val = max_overlap(cand)
        delta = cand_val - best_val

        # Metropolis acceptance
        if delta < 0 or random.random() < math.exp(-delta / max(temp, 1e-12)):
            best_h, best_val = cand, cand_val

        temp *= cooling

    # ------------------------------------------------------------------
    # Final evaluation (using external helper if available)
    # ------------------------------------------------------------------
    try:
        # evaluate_erdos_solution may be provided by the harness
        c5_bound = float(evaluate_erdos_solution(best_h))
    except Exception:
        c5_bound = best_val

    # Convert to plain list for the output format
    h_out = best_h.tolist()
    return (h_out, c5_bound, n)
