def project_onto_box_sum(v, target):
    # Ensure box constraints [0,1] and sum target.
    # Use lambda binary search.
    low = -1.0
    high = 2.0
    for _ in range(50):
        mid = (low + high) / 2
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            low = mid
        else:
            high = mid
    y = np.clip(v + (low + high) / 2, 0.0, 1.0)
    # It may not exactly equal target due to rounding; adjust small diff.
    diff = target - y.sum()
    # If diff small (<1e-8) ignore; else tweak a random element not at bound.
    if abs(diff) > 1e-9:
        # Find indices not at bounds.
        free = np.where((y > 0.0) & (y < 1.0))[0]
        if free.size > 0:
            idx = np.random.choice(free)
            y[idx] = np.clip(y[idx] + diff, 0.0, 1.0)
    return y

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper: correlation based value of C5 for a discretised step function.
# ----------------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


# ----------------------------------------------------------------------
# Helper: Euclidean projection of a vector onto the box [0,1]^n
# intersected with the hyperplane sum(x)=target.
# ----------------------------------------------------------------------
def project_onto_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Find y = clamp(v + λ,0,1) with sum(y)=target.
    Uses a simple bisection on λ; works because the sum is monotone in λ.
    """
    low, high = -2.0, 2.0      # guarantees sum at low = 0, at high = n
    for _ in range(60):       # enough for double precision
        mid = (low + high) * 0.5
        y = np.clip(v + mid, 0.0, 1.0)
        s = y.sum()
        if s < target:
            low = mid
        else:
            high = mid
    y = np.clip(v + (low + high) * 0.5, 0.0, 1.0)

    # tiny correction for floating‑point drift
    diff = target - y.sum()
    if abs(diff) > 1e-9:
        free = np.where((y > 0.0) & (y < 1.0))[0]
        if free.size:
            i = free[np.random.randint(free.size)]
            y[i] = np.clip(y[i] + diff, 0.0, 1.0)
    return y

    greedy refinement.  If an initial construction `initial_h_values`
    is available it is used as a warm start; otherwise a random binary
    vector of length at least 200 is created.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Choose discretisation size.
    # ------------------------------------------------------------------
    try:

        n = init_vec.size
    except Exception:

        n = 200

    # ------------------------------------------------------------------
