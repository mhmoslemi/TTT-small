def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np
    import time

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Load or create initial step function h on [0,2]
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 100                     # fallback size
        h0 = rng.random(n)

    target_sum = n / 2.0            # ∫ h = 1  →  sum(h) = n/2  (dx = 2/n)

    # ------------------------------------------------------------------
    # 2. Projection onto the capped simplex  {0 ≤ x_i ≤ 1, sum x_i = target_sum}
    # ------------------------------------------------------------------
    def project(v):
        """Euclidean projection of v onto {0≤x≤1, sum x = target_sum}."""
        lo = np.min(v) - 1.0
        hi = np.max(v)
        for _ in range(60):        # binary search for Lagrange multiplier λ
            lam = (lo + hi) * 0.5
            x = np.clip(v - lam, 0.0, 1.0)
            s = x.sum()
            if s > target_sum:
                lo = lam
            else:
                hi = lam
        return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # enforce feasibility on the start vector
    cur_h = project(h0)

    # ------------------------------------------------------------------
    # 3. Helper to evaluate C5 and the lag attaining the maximum
    # ------------------------------------------------------------------
    def evaluate(h):
        """Return (C5, lag, full correlation array)."""
        conv = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(conv))
        lag = idx - (len(h) - 1)         # zero lag at index n-1
        c5 = conv[idx] * (2.0 / len(h))  # dx = 2/n
        return c5, lag, conv

    best_c5, best_lag, _ = evaluate(cur_h)
    best_h = cur_h.copy()

    # ------------------------------------------------------------------
    # 4. Main optimisation loop (simulated annealing + gradient steps)
    # ------------------------------------------------------------------
    start_time = time.time()
    T0, Tmin = 0.02, 1e-4               # temperature schedule
    step0, step_min = 0.12, 0.001       # step size for gradient moves

    iteration = 0
    cur_c5 = best_c5

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 0.2:
            break

        frac = elapsed / budget_s
        T = T0 * (1.0 - frac) + Tmin * frac
        # ------------------------------------------------------------------
        # 4a. Choose a move: gradient descent (30%) or random mass swap (70%)
        # ------------------------------------------------------------------
        if rng.random() < 0.30:                     # gradient step
            c5_cur, lag, _ = evaluate(cur_h)       # lag of the current worst shift
            # gradient of g_lag(h) = sum_i h_i (1-h_{i+lag})
            grad = np.ones(n, dtype=float)
            l = lag
            if l >= 0:
                if l < n:
                    grad[:n - l] -= cur_h[l:]
                    grad[l:]     -= cur_h[:n - l]
            else:
                la = -l
                if la < n:
                    grad[la:]    -= cur_h[:n - la]
                    grad[:n - la] -= cur_h[la:]
            step = step0 * (1.0 - frac) + step_min * frac
            cand = project(cur_h - step * grad)
        else:                                        # random swap of mass
            i, j = rng.choice(n, size=2, replace=False)
            # pick a direction: move mass from dec → inc
            if rng.random() < 0.5:
                inc, dec = i, j
            else:
                inc, dec = j, i
            max_inc = 1.0 - cur_h[inc]
            max_dec = cur_h[dec]
            max_transfer = min(max_inc, max_dec, 0.25)   # limit step size
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            cand = cur_h.copy()
            cand[inc] += delta
            cand[dec] -= delta

        cand_c5, _, _ = evaluate(cand)

        # ------------------------------------------------------------------
        # 4b. Acceptance rule (Metropolis)
        # ------------------------------------------------------------------
        if cand_c5 < cur_c5:
            accept = True
        else:
            prob = np.exp(-(cand_c5 - cur_c5) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_h = cand
            cur_c5 = cand_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()

        # ------------------------------------------------------------------
        # 4c. Periodic diversification
        # ------------------------------------------------------------------
        if iteration % 2000 == 0 and iteration > 0:
            if rng.random() < 0.5:
                cur_h = best_h.copy()
                cur_c5 = best_c5
            else:
                # completely new random feasible point
                cur_h = project(rng.random(n))
                cur_c5, _, _ = evaluate(cur_h)

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final projection & evaluation of the best found solution
    # ------------------------------------------------------------------
    best_h = project(best_h)
    final_c5, _, _ = evaluate(best_h)

    # Return as required: list of values, bound, and number of points
    return best_h.tolist(), float(final_c5), n
