def project_onto_feasible(x, target_sum):
    # Using bisection over lambda to satisfy sum(clipped(x - lam)) = target
    lo = -np.max(x) - 1.0 # ensure low enough
    hi = np.max(x) + 1.0  # high enough

        s = proj.sum()

            lo = lam
        else:

    return np.clip(x - (lo + hi) / 2.0, 0, 1)

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helpers
# --------------------------------------------------------------

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a step function represented by `h`."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) (full mode gives all integer shifts)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def worst_shift(h: np.ndarray) -> int:
    """Shift (as integer index) attaining the maximal overlap integral."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    return idx - (n - 1)


def gradient_shift(h: np.ndarray, s: int) -> np.ndarray:
    """
    Gradient of the objective f_s(h)=∑ h_i (1-h_{i+s}) with respect to `h`.
    The gradient of C5 (which is f_s * dx) is the same vector scaled by `dx`.
    """
    n = h.shape[0]
    g = np.ones_like(h)  # derivative of the linear part ∑ h_i
    if s >= 0:
        g[: n - s] -= h[s:]
        g[s:] -= h[: n - s]
    else:
        s = -s
        g[s:] -= h[: n - s]
        g[: n - s] -= h[s:]
    return g


def project_onto_feasible(z: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of vector `z` onto the set
        {x | 0 <= x_i <= 1, sum(x) = target_sum }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(z) - 1.0  # low enough so that all entries become 1 after shift
    hi = np.max(z) + 1.0   # high enough so that all entries become 0 after shift

    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(z - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)


def binarize_preserving_sum(v: np.ndarray) -> np.ndarray:
    """Return a binary (0/1) vector with exactly half the entries equal to 1."""
    n = v.shape[0]
    k = n // 2
    idx = np.argsort(v)[-k:]  # positions of the largest k entries
    h = np.zeros_like(v, dtype=float)
    h[idx] = 1.0
    return h


def random_binary(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random binary vector with exactly n/2 ones."""
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h


# --------------------------------------------------------------
# Main optimisation routine
# --------------------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function giving a low C5 value.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size and starting vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        n = init.shape[0]
        # make sure we have exactly half 1's (binary) – this is a good
        # starting point for the combinatorial part of the search
        h = binarize_preserving_sum(init)
    else:
        # no supplied construction – use a moderate size (keep evaluation cheap)
        n = 200
        if n % 2:
            n += 1
        h = random_binary(n, rng)

    target_sum = n / 2.0
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h)

    best_h = cur_h.copy()
    best_c5 = cur_c5

    # ------------------------------------------------------------------
    # Simulated‑annealing style hill climbing on binary vectors
    # ------------------------------------------------------------------
    max_iters = 5_000_000                # generous upper bound; will be cut by time
    init_temp = 0.02
    final_temp = 1e-6

    # Pre‑compute a mask of indices for quick swap selection
    for it in range(max_iters):
        # time check
        if time.time() - start_time > budget_s - 0.5:
            break

        # temperature schedule (exponential decay)
        t = init_temp * (final_temp / init_temp) ** (it / max_iters)

        # propose a swap of a 1 and a 0
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        if cur_h[i] == cur_h[j]:
            continue                      # not a real swap

        # execute swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_c5 = compute_c5(cur_h)
        delta = new_c5 - cur_c5

        if delta < -1e-12 or rng.random() < math.exp(-delta / t):
            # accept (improvement or probabilistic uphill)
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = cur_h.copy()
                # early stop if we beat the target record
                if best_c5 < 0.3808:
                    break
        else:
            # reject – revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        # Occasionally try a small random perturbation (adds diversity)
        if (it & 1023) == 0:   # every 1024 steps
            # perturb a few positions by adding tiny Gaussian noise and re‑project
            noise = rng.normal(scale=0.08, size=n)
            pert = cur_h + noise
            pert = np.clip(pert, 0.0, 1.0)
            pert = project_onto_feasible(pert, target_sum)
            pert_c5 = compute_c5(pert)
            if pert_c5 < cur_c5:
                cur_h = pert
                cur_c5 = pert_c5
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = cur_h.copy()
                    if best_c5 < 0.3808:
                        break

        # If we have been stagnant for a long time, restart from a fresh random binary vector
        if (it % 200_000) == 0 and it > 0:
            # keep the best seen so far, but give the optimizer a new chance
            cur_h = random_binary(n, rng)
            cur_c5 = compute_c5(cur_h)

    # ------------------------------------------------------------------
    # Return the best solution found
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_c5), n)
