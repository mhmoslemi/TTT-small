def compute_c_max(h, L):
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1 - h, n=L)

    return np.max(c[:2*n-1]) * (2.0 / len(h))

def project_feasible(y, target):
    # Clip
    x = np.clip(y, 0.0, 1.0)
    # Now adjust sum
    diff = target - x.sum()
    # tolerance
    if abs(diff) < 1e-10:
        return x
    # water-filling iterative:
    # Determine indices not at bound (free)
    free = np.arange(len(x))
    while abs(diff) > 1e-10 and len(free) > 0:
        # compute max possible adjustment in direction diff
        if diff > 0:
            capacities = 1.0 - x[free]
        else:
            capacities = -x[free]  # negative if we need decrease
        total_cap = capacities.sum()
        if total_cap == 0:
            break
        # compute uniform adjustment
        adj = diff / len(free)

        # If adj exceeds capacity for some, set those indices to bound and remove from free.
        exceed = (diff > 0) & (adj > capacities) | (diff < 0) & (adj < capacities)

import numpy as np, time, math

def project_to_sum_box(y, target):
    # projection onto [0,1] & sum target
    lo = np.min(y) - 1.0
    hi = np.max(y)
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo+hi)/2.0, 0.0, 1.0)

def max_overlap_c5(h, L, dx):
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    c = np.fft.irfft(H * np.conj(G), n=L)
    maxc = np.max(c[:2*len(h)-1])
    return maxc * dx

def autocorr(h, L):
    H = np.fft.rfft(h, n=L)
    c = np.fft.irfft(H * np.conj(H), n=L)
    return c[:2*len(h)-1]

import numpy as np
import time
import math

# ---- helper functions -------------------------------------------------

def _project_onto_sum_box(y, target):
    """
    Project vector y onto the set {0 <= x <= 1, sum(x)=target}
    using a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(y) - 1.0        # low enough that everything clips to 1
    hi = np.max(y)              # high enough that everything clips to 0
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(y - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_from_h(h, L, dx):
    """return C5 = max_k ∫ h(x)(1-h(x+k))dx  (discrete version)"""
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(1.0 - h, n=L)
    conv = np.fft.irfft(H * np.conj(G), n=L)[: 2 * len(h) - 1]
    return np.max(conv) * dx


def _autocorr(h, L):
    """full autocorrelation array of h (zero‑padded)"""
    H = np.fft.rfft(h, n=L)
    ac = np.fft.irfft(H * np.conj(H), n=L)[: 2 * len(h) - 1]
    return ac


# ---- main optimisation ------------------------------------------------

            ac = _autocorr(cur_h, L)
            ac[n - 1] = np.inf                      # ignore zero lag
