    max_s = np.max(s)  # Not needed; use -α s which will be negative;

def project_onto_bounds_and_sum(v, target_sum, lower=0.0, upper=1.0, tol=1e-9, max_iter=100):
    # Implements lambda projection: x_i = clip(v_i - lambda, lower, upper)
    # Find lambda using binary search.
    # v is np array.
    lo = -1.0
    hi = 1.0
    # expand bounds to ensure solution
    # Actually lambda may be between v-min upper and v-max lower; pick wide.
    # We'll use binary search for lambda.
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) / 2.0, lower, upper)
    # Adjust any residual diff due to clipping errors:
    diff = target_sum - x.sum()
    if abs(diff) > tol:
        # Distribute diff among non-bound indices
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if np.any(free):
            # add diff equally
            add = diff / np.count_nonzero(free)
            x[free] += add
        else:
            # adjust a few indices within bounds
            # find index with max potential to increase if diff>0 else decrease
            if diff > 0:
                # increase any variable not at upper
                idx = np.argmax(upper - x)
                delta = min(diff, upper - x[idx])
                x[idx] += delta
            else:
                idx = np.argmin(x - lower)
                delta = max(diff, lower - x[idx])  # diff negative
                x[idx] += delta
    # final clip
    x = np.clip(x, lower, upper)
    return x

def _project_onto_bounds_and_sum(v, target, lower=0.0, upper=1.0, max_iter=80):
    """Project v onto [lower,upper]^n with sum(v)=target."""
    lo, hi = -1e6, 1e6
    for _ in range(max_iter):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, lower, upper)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, lower, upper)
    # tiny residual correction
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > lower + 1e-12) & (x < upper - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
        else:
            if diff > 0:
                idx = np.argmax(upper - x)
                add = min(diff, upper - x[idx])
                x[idx] += add
            else:
                idx = np.argmin(x - lower)
                sub = min(-diff, x[idx] - lower)
                x[idx] -= sub
    return np.clip(x, lower, upper)


def _evaluate(h):
    """Return max overlap C5 for step function h."""
    n = h.shape[0]
    s = np.correlate(h, h, mode='full')[n - 1:]        # autocorrelation for shifts 0..n-1
    min_s = s.min()
    return 1.0 - (2.0 / n) * min_s, s


def _gradient_step(h, target, alpha, eta):
    """One projected gradient step aiming to increase the minimal autocorrelation."""
    n = h.shape[0]
    s_full = np.correlate(h, h, mode='full')
    s = s_full[n - 1:]                                # length n
    # soft‑min weights
    shift = s - s.min()                               # avoid overflow
    w = np.exp(-alpha * shift)
    w /= w.sum()
    grad = np.zeros_like(h)
    for ell in range(n):
        w_ell = w[ell]
        if w_ell == 0.0:
            continue
        if ell == 0:
            grad += 2.0 * w_ell * h
        else:
            grad[:n - ell] += w_ell * h[ell:]
            grad[ell:]   += w_ell * h[:n - ell]
    v = h + eta * grad
    return _project_onto_bounds_and_sum(v, target)

            else:

        else:

            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            hi, hj = cur_h[i], cur_h[j]
            max_inc = min(1.0 - hi, hj)     # increase i, decrease j

            delta = np.random.uniform(max_dec, max_inc)
