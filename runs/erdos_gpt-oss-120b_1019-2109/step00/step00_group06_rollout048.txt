    h = h_initial.copy()
    best = h.copy()
    best_obj = evaluate(h)
    start = time.time()
    while time.time() - start < time_limit:

        i, j = rng.integers(len(h), size=2)
        while i == j:
            j = rng.integers(len(h))
        # direction picked random: which to give?
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        # compute max transfer
        max_transfer = min(h[src], 1.0 - h[dst])
        if max_transfer <= 0:
            continue
        # choose delta randomly scaled (maybe up to max_transfer)
        delta = rng.random() * max_transfer
        if delta < 1e-12:
            continue
        # apply move
        h[src] -= delta
        h[dst] += delta
        obj = evaluate(h)
        if obj + 1e-12 < best_obj:  # improvement
            best_obj = obj
            best = h.copy()
        else:
            # revert
            h[src] += delta

        # optionally accept some non improvements with small probability to escape local minima
