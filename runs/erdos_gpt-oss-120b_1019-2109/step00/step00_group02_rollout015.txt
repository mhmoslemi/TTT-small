def project_feasible(h, target_sum):
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # If sum != target_sum, adjust using water-filling method.
    # We'll loop adjusting free variables.
    # Approach: while sum diff > tolerance:
    #   find indices not at bound.
    #   if none, break.
    #   compute adjustment = diff / count_free.
    #   add adjustment, clip again.
    diff = target_sum - h.sum()
    # We'll do iterative but for small diff this is fine.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        free = ( (h > 0 + 1e-12) & (h < 1 - 1e-12) )
        if not np.any(free):
            # modify all entries proportionally
            h = h + diff / len(h)
            # Clip again
            h = np.clip(h, 0, 1)
            diff = target_sum - h.sum()
            continue
        count = free.sum()
        adjust = diff / count
        h[free] += adjust
        # Clip again
        h = np.clip(h, 0, 1)
        diff = target_sum - h.sum()
    # Finally, if diff still not zero, adjust one arbitrary free index.
    if abs(diff) > 1e-9:
        # find index not at bound
        idx = np.argmax(h * (1 - h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')

    # Determine indices where contribution >0
    if shift >= 0:
        i_indices = np.arange(0, n - shift)
        j_indices = i_indices + shift
    else:
        i_indices = np.arange(-shift, n)

import time
import math
import numpy as np

def _compute_c5(h: np.ndarray) -> float:
    """Return the overlap constant C5 for a step function h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * (2.0 / n))


def _project_feasible(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    h = np.clip(h, 0.0, 1.0)
    # Simple scaling – may violate bounds, so iterate a few times.
    for _ in range(10):
        s = h.sum()
        if abs(s - target_sum) < 1e-12:
            break
        delta = target_sum - s
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if not free.any():
            # all at bounds – distribute proportionally
            h += delta / h.size
            h = np.clip(h, 0.0, 1.0)
            continue
        cnt = free.sum()
        adjust = delta / cnt
        h[free] += adjust
        h = np.clip(h, 0.0, 1.0)
    # Tiny residual fix
    residual = target_sum - h.sum()
    if abs(residual) > 1e-10:
        idx = np.argmax(h * (1.0 - h))  # interior point if any
        h[idx] = np.clip(h[idx] + residual, 0.0, 1.0)
    return h


def _random_move(h: np.ndarray, move_scale: float) -> np.ndarray | None:
    """Generate a feasible neighbour by moving mass between two indices."""
    n = h.shape[0]
    i = np.random.randint(n)
    j = np.random.randint(n - 1)
    if j >= i:
        j += 1
    # direction i -> j
    max_down_i = h[i]
    max_up_j = 1.0 - h[j]
    delta1 = min(max_down_i, max_up_j)
    # direction j -> i
    max_down_j = h[j]
    max_up_i = 1.0 - h[i]
    delta2 = min(max_down_j, max_up_i)

    if delta1 <= 0.0 and delta2 <= 0.0:
        return None
    if delta1 > delta2:
        delta = delta1 * move_scale * np.random.rand()
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        return new_h
    else:
        delta = delta2 * move_scale * np.random.rand()
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
        return new_h


def _directed_move(h: np.ndarray, move_scale: float) -> np.ndarray | None:
    """Move mass along the pair that currently contributes most to the max overlap."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)  # shift index (can be negative)

    if shift >= 0:
        i_idxs = np.arange(0, n - shift)
        j_idxs = i_idxs + shift
    else:
        i_idxs = np.arange(-shift, n)
        j_idxs = i_idxs + shift

    # contribution of each i/j pair
    contrib = h[i_idxs] * (1.0 - h[j_idxs])
    # we need a pair where we can transfer from i to j (i high, j low)
    feasible = (h[i_idxs] > 1e-12) & (h[j_idxs] < 1.0 - 1e-12)
    if not np.any(feasible):
        return None
    # choose the pair with the largest contribution among feasible
    idx = np.argmax(contrib * feasible)  # multiplies by bool to ignore infeasibles
    i = i_idxs[idx]
    j = j_idxs[idx]
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0.0:
        return None
    delta = max_delta * move_scale * np.random.rand()
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a low upper bound for C5.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()
    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():

    n = h.shape[0]
    target_sum = n / 2.0
    h = _project_feasible(h, target_sum)

    cur_h = h.copy()
    cur_val = _compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # ----- optimisation parameters ------------------------------------
    total_time = budget_s * 0.96  # leave a safety margin
    end_time = start_time + total_time
    T0 = 0.02          # initial temperature for SA
    T_end = 1e-6       # final temperature
    max_iters = 400000
    it = 0

    while time.time() < end_time and it < max_iters:
