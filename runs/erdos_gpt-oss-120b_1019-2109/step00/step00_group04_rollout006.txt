def optimize_h(n, time_budget):
    start = time.time()
    h = initial_h(n)
    dx = 2.0 / n
    best_h = h.copy()
    conv = np.correlate(h, 1 - h, mode='full') * dx
    best_f = np.max(conv)
    step = 0.5  # adjust
    no_improve_counter = 0
    while time.time() - start < time_budget:
        # compute correlation
        conv = np.correlate(h, 1 - h, mode='full') * dx
        f = np.max(conv)

        L = L_idx - (n - 1)

        for _ in range(10):

            h_new = project_h(h_new)

import time
import random
import numpy as np

def _project_to_box_sum(v):
    """Project a vector onto the intersection of [0,1]^n and sum(v)=n/2."""
    n = v.shape[0]
    target = n / 2.0
    # Bisection on the Lagrange multiplier λ for the box constraints.
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        lam = (lo + hi) / 2.0
        h = np.clip(v - lam, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # Tiny correction if rounding left a small residue.
    s = h.sum()
    if abs(s - target) > 1e-9:
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.any():
            delta = (target - s) / free.sum()
            h[free] += delta
            h = np.clip(h, 0.0, 1.0)
    return h

def _triangle_h(n):
    """Return a discretized triangular step function of length n."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx          # mid‑points of subintervals
    h0 = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    return _project_to_box_sum(h0)

def _c5_of(h):
    """Compute the max overlap integral for a discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(conv.max())

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C₅ bound and the number of points."""
    np.random.seed(seed)

    # Primary construction: a discretised triangle.
