def propose_move(h, max_transfer=0.2):
    i, j = np.random.randint(0, n, size=2)
    if i == j: return None  # no move
    # random direction
    direction = 1 if np.random.rand() < 0.5 else -1
    if direction == 1:
        # transfer from i to j
        max_delta = min(h[i], 1 - h[j])
    else:
        max_delta = min(h[j], 1 - h[i])
    if max_delta <= 1e-12:
        return None
    delta = np.random.rand() * max_delta
    # Apply
    new_h = np.copy(h)
    if direction == 1:
        new_h[i] -= delta
        new_h[j] += delta
    else:
        new_h[j] -= delta
        new_h[i] += delta
    return new_h

import time, math, random
import numpy as np

def _compute_c5(vec, dx):
    """Return max overlap integral for the given vector."""
    # correlation of h with (1-h), full mode
    corr = np.correlate(vec, 1.0 - vec, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimization for the Erdős minimum overlap constant C5.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray of shape (n_points,)
        c5_bound : float, the achieved upper bound
        n_points : int, number of samples used
    """
    start = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():

        h_cur = np.asarray(initial_h_values, dtype=np.float64).copy()

        # repair possible tiny violation of the integral constraint

        if abs(cur_sum - target_sum) > 1e-12:

            np.clip(h_cur, 0.0, 1.0, out=h_cur)
