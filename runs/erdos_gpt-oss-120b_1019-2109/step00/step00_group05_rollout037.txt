def project_to_box_sum(x, target, max_iter=100):

    for _ in range(max_iter):
        diff = target - y.sum()
        if abs(diff) < 1e-12:
            break
        # compute adjustable mask
        if diff > 0:
            mask = y < 1 - 1e-12
        else:
            mask = y > 1e-12
        if not np.any(mask):
            break
        # distribute diff equally among mask entries
        delta = diff / np.count_nonzero(mask)
        y[mask] += delta
        # clip again to ensure bounds
        y = np.clip(y, 0, 1)
    return y

            if abs(diff) < 1e-12:
                break
            if diff > 0:
                mask = y < 1.0 - 1e-12
            else:
                mask = y > 1e-12
            cnt = np.count_nonzero(mask)
            if cnt == 0:
                break
            delta = diff / cnt
            y[mask] += delta
            y = np.clip(y, 0.0, 1.0)
            diff = target_sum - y.sum()
        # final tiny adjustment if needed
        if abs(y.sum() - target_sum) > 1e-9:
            free = np.where((y > eps) & (y < 1 - eps))[0]
            if free.size:
                idx = np.random.choice(free)
                y[idx] += target_sum - y.sum()
                y = np.clip(y, 0.0, 1.0)
        return y

    def cost_vec(v):
        """C5 for a given h (using the same definition as the evaluator)."""
        conv = np.correlate(v, v, mode='full')
        min_corr = conv.min()
        # using the exact sum (should be target_sum)
        S = v.sum()
        return dx * (S - min_corr)

    def grad_weights(v, offset):
        """gradient weight vector for a fixed shift (offset)."""
        w = np.zeros_like(v)
        if offset >= 0:
            off = offset
            if off < n:
                w[:n - off] += v[off:]
                w[off:] += v[:n - off]
        else:
            off = -offset
            if off < n:
                w[off:] += v[:n - off]
                w[:n - off] += v[off:]
        return w

    # ensure feasibility
    h = project(h)
    best_h = h.copy()
    best_c = cost_vec(best_h)

    # ----------------------------------------------------------------------
    # optimisation parameters
    # ----------------------------------------------------------------------
    T = 0.01            # initial temperature for simulated annealing
    Tmin = 1e-6
    cooling = 0.9995
    it = 0

    # ----------------------------------------------------------------------
    # main optimisation loop
    # ----------------------------------------------------------------------
    while time.time() - start_time < budget_s:
        # current state
        conv = np.correlate(h, h, mode='full')
        min_idx = np.argmin(conv)
        offset = min_idx - (n - 1)           # shift that attains the worst overlap
        cur_c = dx * (target_sum - conv.min())

        # --------------------------------------------------------------
        # 1) gradient‑guided improvement for the worst shift
        # --------------------------------------------------------------
        w = grad_weights(h, offset)

        donor_mask = h > eps
        rec_mask = h < 1.0 - eps
        if donor_mask.any() and rec_mask.any():
            donor_candidates = np.where(donor_mask)[0]
            rec_candidates = np.where(rec_mask)[0]

            donor = donor_candidates[np.argmin(w[donor_mask])]
            rec   = rec_candidates[np.argmax(w[rec_mask])]
            if donor != rec:
                max_delta = min(h[donor], 1.0 - h[rec])
                if max_delta > eps:
                    improved = False
                    for factor in (1.0, 0.5, 0.25):
                        delta = max_delta * factor
                        if delta <= eps:
                            continue
                        h_tmp = h.copy()
                        h_tmp[donor] -= delta
                        h_tmp[rec]   += delta
                        new_c = cost_vec(h_tmp)
                        if new_c + 1e-12 < cur_c:
                            h = h_tmp
                            cur_c = new_c
                            improved = True
                            if new_c < best_c - 1e-12:
                                best_c = new_c
                                best_h = h.copy()
                            break
                    if improved:
                        it += 1
                        T = max(Tmin, T * cooling)
                        continue   # go to next iteration

        # --------------------------------------------------------------
        # 2) simulated‑annealing random move (mass exchange)
        # --------------------------------------------------------------
        donor_pool = np.where(h > eps)[0]
        rec_pool   = np.where(h < 1.0 - eps)[0]
        if donor_pool.size == 0 or rec_pool.size == 0:
            break
        donor = np.random.choice(donor_pool)
        rec   = np.random.choice(rec_pool)
        while rec == donor and rec_pool.size > 1:
            rec = np.random.choice(rec_pool)

        max_delta = min(h[donor], 1.0 - h[rec])
        if max_delta > eps:
            delta = np.random.rand() * max_delta
            if delta > eps:
                h_tmp = h.copy()
                h_tmp[donor] -= delta
                h_tmp[rec]   += delta
                new_c = cost_vec(h_tmp)
                # Metropolis acceptance
                if new_c < cur_c or np.random.rand() < np.exp(-(new_c - cur_c) / max(T, 1e-12)):
                    h = h_tmp
                    cur_c = new_c
                    if new_c < best_c - 1e-12:
                        best_c = new_c
                        best_h = h.copy()
        it += 1
        T = max(Tmin, T * cooling)

        # --------------------------------------------------------------
        # 3) occasional small perturbation of the best solution
        # --------------------------------------------------------------
        if it % 10000 == 0:
            pert = best_h.copy()
            for _ in range(5):
                i = np.random.randint(n)
                j = np.random.randint(n - 1)
                if j >= i:
                    j += 1
                # move a tiny amount from the richer to the poorer index
                if pert[i] > eps and pert[j] < 1 - eps:
                    md = min(pert[i], 1 - pert[j])
                    d = np.random.rand() * md * 0.1
                    pert[i] -= d
                    pert[j] += d
                elif pert[j] > eps and pert[i] < 1 - eps:
                    md = min(pert[j], 1 - pert[i])
                    d = np.random.rand() * md * 0.1
                    pert[j] -= d
                    pert[i] += d
            pert = project(pert)
            pc = cost_vec(pert)
            if pc < best_c - 1e-12:
                best_c = pc
                best_h = pert.copy()
                h = pert.copy()
                cur_c = pc

        # early exit if we already beat the target record
        if best_c <= 0.3808:
            break

    # final safety projection and cost recomputation
    best_h = project(best_h)
    final_c = cost_vec(best_h)

    return best_h, final_c, n
