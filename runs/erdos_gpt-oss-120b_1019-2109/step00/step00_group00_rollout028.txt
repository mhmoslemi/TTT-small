def compute_max_shift_and_gradient(h):
    n = len(h)
    dx = 2/n
    rev = 1 - h
    corr = np.correlate(h, rev, mode='full')  # length 2n-1
    max_idx = np.argmax(corr)  # index in [0, 2n-2]
    max_corr = corr[max_idx] * dx

    k = max_idx - (n-1)   # shift: negative to positive

    # Compute gradient (size n)
    g = np.zeros_like(h)

    if k >= 0:
        # term (1 - h[i+k]) for i=0..n-1-k
        i_range = np.arange(0, n - k)
        g[i_range] += 1 - h[i_range + k]
        # term -h[i-k] for i=k..n-1
        i_range2 = np.arange(k, n)
        g[i_range2] -= h[i_range2 - k]
    else:
        # k negative => shift left
        # Equivalent to shift = -abs_k
        kneg = -k
        # for k negative, c_{k} = sum_{i=kneg}^{n-1} h[i] * (1 - h[i - kneg])
        # i from kneg .. n-1; i - kneg >=0
        i_range = np.arange(kneg, n)
        g[i_range] += 1 - h[i_range - kneg]
        i_range2 = np.arange(0, n - kneg)
        g[i_range2] -= h[i_range2 + kneg]
    return max_corr, g

import time
import numpy as np
from scipy.optimize import minimize

# ----- global parameters used by callbacks (set inside run) -----
global_dx = 1.0
global_target = 0.0


def project_to_feasible(h: np.ndarray, target: float) -> np.ndarray:
    """Project h onto the box [0,1] and the hyperplane sum(h)=target."""
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
        if diff > 0.0:
            cap = 1.0 - h
        else:
            cap = h
        total = cap.sum()
        if total == 0.0:
            break
        delta = diff * cap / total
        h += delta
        h = np.clip(h, 0.0, 1.0)
    # tiny final correction if needed
    s = h.sum()
    diff = target - s
    if abs(diff) > 1e-12:
        if diff > 0.0:
            idx = np.where(h < 1.0 - 1e-12)[0][0]
            add = min(diff, 1.0 - h[idx])
            h[idx] += add
        else:
            idx = np.where(h > 1e-12)[0][0]
            sub = min(-diff, h[idx])
            h[idx] -= sub
    return np.clip(h, 0.0, 1.0)


def max_corr_and_grad(h: np.ndarray, dx: float):
    """Return max correlation value and its gradient (w.r.t. h)."""
    n = h.size
    rev = 1.0 - h
    corr = np.correlate(h, rev, mode="full")
    max_idx = int(np.argmax(corr))
    val = corr[max_idx] * dx
    k = max_idx - (n - 1)          # shift (positive = right)

    grad = np.zeros_like(h)
    if k >= 0:
        i1 = np.arange(0, n - k)
        grad[i1] += 1.0 - h[i1 + k]
        i2 = np.arange(k, n)
        grad[i2] -= h[i2 - k]
    else:
        kk = -k
        i1 = np.arange(kk, n)
        grad[i1] += 1.0 - h[i1 - kk]
        i2 = np.arange(0, n - kk)
        grad[i2] -= h[i2 + kk]
    grad *= dx
    return val, grad


def objective_fun(h: np.ndarray) -> float:
    """Objective for SLSQP (max correlation)."""
    val, _ = max_corr_and_grad(h, global_dx)
    return val


def gradient_fun(h: np.ndarray) -> np.ndarray:
    """Gradient of the objective."""
    _, g = max_corr_and_grad(h, global_dx)
    return g


def eq_constraint_fun(h: np.ndarray) -> float:
    """Equality: sum(h) = target."""
    return np.sum(h) - global_target


def eq_constraint_jac(h: np.ndarray) -> np.ndarray:
    """Jacobian of the equality constraint."""
    return np.ones_like(h)


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 value for a discrete step function."""
    dx = 2.0 / h.size
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with low overlap constant C5."""
    start_time = time.time()
    np.random.seed(seed)

    # number of sample points (adjust if you like)
    n = 200                     # even, < 1000 as required
    dx = 2.0 / n
    target_sum = n / 2.0        # because ∑h = n/2  ↔  ∫h = 1

    # make callbacks see the proper global parameters
    global global_dx, global_target
    global_dx = dx
    global_target = target_sum

    bounds = [(0.0, 1.0)] * n
    eq_cons = {
        "type": "eq",
        "fun": eq_constraint_fun,
        "jac": eq_constraint_jac,
    }

    best_h = None
    best_c5 = np.inf

    # --- helper to evaluate a candidate -------------------------------------------------
    def try_candidate(h0):
        nonlocal best_h, best_c5, start_time
        # SLSQP may stray from the hyperplane; we project before starting
        h0 = project_to_feasible(h0, target_sum)

        res = minimize(
            objective_fun,
            h0,
            method="SLSQP",
            jac=gradient_fun,
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        h_opt = project_to_feasible(res.x, target_sum)
        c5_val = compute_c5(h_opt)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt.copy()

    # -------------------------------------------------------------------------
    # 1) deterministic triangular start (often good)
    x = (np.arange(n) + 0.5) * dx
    tri = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    try_candidate(tri)

    # 2) a few random binary starts (half ones)
    for _ in range(6):
        if time.time() - start_time > budget_s - 5.0:
            break
        h_bin = np.zeros(n)
        h_bin[: n // 2] = 1.0
        np.random.shuffle(h_bin)
        # add a tiny random perturbation to break symmetry
        eps = np.random.uniform(-0.05, 0.05, size=n)
        h_bin = np.clip(h_bin + eps, 0.0, 1.0)
        try_candidate(h_bin)

    # 3) a few random continuous starts
    for _ in range(4):
        if time.time() - start_time > budget_s - 5.0:
            break
        h_rand = np.random.rand(n)
        # force the correct total mass by scaling and projecting
        h_rand = project_to_feasible(h_rand, target_sum)
        try_candidate(h_rand)

    # final safety check – ensure feasibility
    if best_h is not None:
        best_h = project_to_feasible(best_h, target_sum)
        best_c5 = compute_c5(best_h)

    # Return the best discovered solution
    return best_h, float(best_c5), n
