def local_optimize(h_initial, budget, start_time):
    best_h = h_initial.copy()
    best_val = compute_c5(best_h)
    h = h_initial.copy()
    val = best_val
    it = 0
    while it < max_iters and time.time() - start_time < budget:
        c, shift = compute_c5_and_shift(h)
        grad = grad_for_shift(h, shift) * dx
        # step size
        alpha = 0.5 / np.sqrt(it + 1)
        h_new = h - alpha * grad
        h_new = project(h_new, target_sum)
        new_val = compute_c5(h_new)
        # accept if improved, else maybe accept small deterioration
        if new_val < val or rng.random() < 0.001:
            h = h_new
            val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        it += 1
    return best_h, best_val
```

import time
import numpy as np

def _project_to_constraints(x, target_sum, max_iter=100):
    """Clip to [0,1] and adjust to meet the sum constraint."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        cur = x.sum()
        diff = target_sum - cur
        if abs(diff) < 1e-12:
            break
        free = (x > 1e-12) & (x < 1 - 1e-12)
        nfree = free.sum()
        if nfree == 0:
            # No free variables, distribute diff uniformly and clip again
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
            continue
        delta = diff / nfree
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    return x

def _grad_for_shift(h, shift):
    """Subgradient of  corr = h*(1-h_shift) for the given shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(n - shift, dtype=int)
        g[i] += (1.0 - h[i + shift])
        g[i + shift] += -h[i]
    else:
        s = -shift
        i = np.arange(n - s, dtype=int)
        g[i + s] += (1.0 - h[i])
        g[i] += -h[i + s]
    return g

def _max_corr_and_shift(h):
    """Return max correlation value (without dx) and the corresponding shift."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (len(h) - 1)
    return corr[idx], shift

def _upscale(h_small, n_target):
    """Linear interpolation of a smaller h to length n_target."""
    n_small = h_small.shape[0]
    if n_small == n_target:
        return h_small.astype(np.float64).copy()
    x_small = (np.arange(n_small) + 0.5) * (2.0 / n_small)
    x_target = (np.arange(n_target) + 0.5) * (2.0 / n_target)
    return np.interp(x_target, x_small, h_small.astype(np.float64))

def _local_optimize(h0, dx, target_sum, time_deadline, rng):
    """Simple projected sub‑gradient descent with occasional random jumps."""
    n = h0.shape[0]
    best_h = h0.copy()
    best_val = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    h = h0.copy()
    cur_val = best_val
    max_iters = 20000

    for it in range(max_iters):
        if time.time() > time_deadline:
            break

        # current worst shift and its subgradient
        corr_val, shift = _max_corr_and_shift(h)
        cur_val = corr_val * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        g = _grad_for_shift(h, shift) * dx
        # step size decreasing like 2/sqrt(it+1)
        step = 2.0 / np.sqrt(it + 1.0)
        h_new = h - step * g
        h_new = _project_to_constraints(h_new, target_sum)

        new_val = np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        # accept strictly improving moves
        if new_val < cur_val:
            h = h_new
            cur_val = new_val
        # every few hundred iterations try a random shake
        if (it + 1) % 400 == 0:
            shake = rng.normal(scale=0.03, size=n)
            shaken = np.clip(h + shake, 0.0, 1.0)
            shaken = _project_to_constraints(shaken, target_sum)
            shaken_val = np.max(np.correlate(shaken, 1 - shaken, mode='full')) * dx
            if shaken_val < cur_val:
                h = shaken
                cur_val = shaken_val
                if shaken_val < best_val:
                    best_val = shaken_val
                    best_h = shaken.copy()
        # early exit if we already beat the target
        if best_val <= 0.3808:
            break

    return best_h, best_val

        return h

    final_val = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h, float(final_val), n)
