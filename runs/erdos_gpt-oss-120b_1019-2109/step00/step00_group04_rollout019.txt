    # v: array, not necessarily feasible

import numpy as np
import time

def compute_cost(h):
    n = h.shape[0]
    dx = 2.0 / n
    # correlation: sum_i h[i] * (1 - h[i+s])
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_box_sum(v, target, tol=1e-9, max_iter=100):
    low = -1e6
    high = 1e6
    for _ in range(max_iter):
        lam = (low + high) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = lam
        else:
            high = lam
    # final projection
    return np.clip(v - lam, 0.0, 1.0)

def interpolate_h(orig, n):
    n0 = len(orig)
    dx0 = 2.0 / n0
    x0 = np.arange(n0) * dx0 + dx0 / 2.0
    dx = 2.0 / n
    x = np.arange(n) * dx + dx / 2.0
    h = np.interp(x, x0, orig, left=0.0, right=0.0)
    h = project_box_sum(h, n / 2.0)
    return h

def random_feasible_h(n):
    h = np.random.rand(n)
    return project_box_sum(h, n / 2.0)

import numpy as np
import time

def compute_cost(h):
    """C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx


def project_box_sum(v, target, tol=1e-9, max_iter=100):
    """Project v onto {0≤x_i≤1, sum x_i = target}."""
    low, high = -1e6, 1e6
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s > target:
            low = lam
        else:
            high = lam
    return np.clip(v - lam, 0.0, 1.0)


def interpolate_h(orig, n):
    """Upsample a parent vector to length n and project to the feasible set."""
    n0 = len(orig)
    dx0 = 2.0 / n0
    x0 = np.arange(n0) * dx0 + dx0 * 0.5
    dx = 2.0 / n
    x = np.arange(n) * dx + dx * 0.5
    h = np.interp(x, x0, orig, left=0.0, right=0.0)
    return project_box_sum(h, n / 2.0)


def random_feasible_h(n):
    """Generate a random feasible vector of length n."""
    h = np.random.rand(n)
    return project_box_sum(h, n / 2.0)

                best_h = h.copy()

    # fallback if the systematic search yields nothing (should not happen)
    if best_h is None:
        try:
            best_h = interpolate_h(initial_h_values, n)
        except NameError:
            best_h = random_feasible_h(n)

    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 1.0   # small safety margin
    if remaining > 0.0:
        best_h, best_val = optimize_h(best_h, n, remaining)

    # final projection onto the exact feasible set
    best_h = project_box_sum(best_h, target)
    best_val = compute_cost(best_h)

    return best_h, float(best_val), n
```
