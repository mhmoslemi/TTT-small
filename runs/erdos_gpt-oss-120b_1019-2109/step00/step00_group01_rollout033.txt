import numpy as np, time.

def compute_overlap_and_grad(h, beta=20.0, lam=100.0):
    N = len(h)
    dx = 2.0 / N
    # compute correlation: O_vec = dx * np.convolve(h, 1-h, mode='full')
    # Use np.convolve for speed:
    O_vec = np.convolve(h, 1 - h, mode='full') * dx
    # Softmax weights:
    maxO = O_vec.max()
    # subtract max for stability
    exps = np.exp(beta * (O_vec - maxO))
    Z = exps.sum()
    w = exps / Z
    # Compute gradient
    grad = np.zeros(N)
    # iterate shifts
    # shift range from -(N-1) to N-1
    shifts = np.arange(-(N-1), N)  # length 2N-1
    # iterate with vectorization? Could compute contributions via loops.
    for weight, s in zip(w, shifts):
        if weight == 0:
            continue
        if s >= 0:
            # term1 for i in [0, N-s-1]
            # grad[:N-s] += weight * dx * (1 - h[s:])
            grad[:N-s] += weight * dx * (1.0 - h[s:])
            # term2 for i in [s, N-1]
            grad[s:] -= weight * dx * h[:N-s]
        else:
            sn = -s
            # term1 for i in [sn, N-1]
            grad[sn:] += weight * dx * (1.0 - h[:N-sn])
            # term2 for i in [0, N-sn-1]
            grad[:N-sn] -= weight * dx * h[sn:]
    # penalty gradient
    sum_h = h.sum()
    grad += 2.0 * lam * (sum_h - N/2) * np.ones_like(h)
    # objective value: smooth max + penalty
    # smooth max = (1.0/beta) * (np.log(Z) + beta*maxO) (since we used shift)
    # Actually f = (1/beta) * np.log(np.exp(beta * O_vec).sum()) = (1/beta)*(np.log(Z) + beta*maxO) (since we subtracted maxO)
    smooth_max = (np.log(Z) + beta * maxO) / beta
    penalty = lam * (sum_h - N/2) ** 2
    return smooth_max + penalty, grad, O_vec

def project_to_box_and_sum(h, target_sum):
    # Clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # adjust sum
    N = len(h)
    current_sum = h.sum()
    # compute deficit
    deficit = target_sum - current_sum
    # tolerance
    if abs(deficit) < 1e-12:
        return h
    # water-filling:
    # We'll loop: while deficit not zero within tolerance
    # Determine free indices (can move up if deficit>0, down if deficit<0)
    # For up: idx where h < 1
    # For down: idx where h > 0
    max_iter=10
    for _ in range(max_iter):
        if abs(deficit) < 1e-12: break
        if deficit > 0:
            free = np.where(h < 1 - 1e-12)[0]
        else:
            free = np.where(h > 1e-12)[0]
        if free.size == 0:
            break
        # compute possible move per index without hitting bounds
        if deficit > 0:
            caps = 1 - h[free]
        else:
            caps = -h[free]  # negative amounts you can subtract
        total_cap = caps.sum()
        if abs(total_cap) < 1e-12:
            break
        # allocate proportionally to caps: allocate delta = deficit * (caps/total_cap)
        # This ensures we use full cap before hitting bound
        delta = deficit * (caps / total_cap)
        # Apply delta
        h[free] += delta
        # Clip to bounds to avoid numeric errors
        h = np.clip(h, 0.0, 1.0)
        deficit = target_sum - h.sum()
    # final clip
    h = np.clip(h,0,1)
    return h

    # early exit if sum close
    current_sum = h.sum()
    deficit = target_sum - current_sum
    if abs(deficit) <= eps:
        return h
    # water-filling
    # We'll loop until deficit <= eps or no free variables
    for _ in range(10):
        if abs(deficit) <= eps:
            break
        if deficit > 0:
            free = np.where(h < 1 - eps)[0]
            if free.size == 0:
                break
            caps = 1 - h[free]
        else:
            free = np.where(h > eps)[0]
            if free.size == 0:
                break
            caps = -h[free]  # negative capacities
        total_cap = caps.sum()
        if total_cap == 0:
            break
        # allocate proportionally to caps
        delta = deficit * (caps / total_cap)
        h[free] += delta
        # clamp to [0,1] to avoid small numeric overshoot
        h = np.clip(h, 0.0, 1.0)
        current_sum = h.sum()
        deficit = target_sum - current_sum
    # final clip
    h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

# Helper: project onto box [0,1] and enforce sum = target_sum
def _project_box_sum(v, target, eps=1e-12):
    v = np.clip(v, 0.0, 1.0)
    deficit = target - v.sum()
    if abs(deficit) <= eps:
        return v
    # water-filling style redistribution
    for _ in range(10):
        if abs(deficit) <= eps:
            break
        if deficit > 0:
            free = np.where(v < 1 - eps)[0]
            caps = 1.0 - v[free]
        else:
            free = np.where(v > eps)[0]
            caps = -v[free]               # negative capacities
        if free.size == 0:
            break
        total = caps.sum()
        if total == 0:
            break
        delta = deficit * (caps / total)
        v[free] += delta
        v = np.clip(v, 0.0, 1.0)
        deficit = target - v.sum()
    return v

# Compute smooth max objective, its gradient and the raw overlap vector
def _obj_grad_overlap(h, beta, lam):
    N = h.shape[0]
    dx = 2.0 / N
    # raw overlaps O(k) = ∑ h_i (1 – h_{i+k}) * dx
    O = np.correlate(h, 1.0 - h, mode='full') * dx
    m = O.max()
    exps = np.exp(beta * (O - m))
    Z = exps.sum()
    w = exps / Z                     # softmax weights, sum to 1
    # gradient of smooth max (weighted sum of sub‑gradients)
    grad = np.zeros(N)
    # mapping index -> shift
    offset = N - 1
    for idx, weight in enumerate(w):
        if weight == 0.0:
            continue
        s = idx - offset                # shift
        if s >= 0:
            end = N - s
            if end > 0:
                grad[:end] += weight * dx * (1.0 - h[s:])
                grad[s:]  -= weight * dx * h[:end]
        else:
            t = -s
            end = N - t
            if end > 0:
                grad[t:]    += weight * dx * (1.0 - h[:end])
                grad[:end]  -= weight * dx * h[t:]
    # penalty for the integral constraint
    cur_sum = h.sum()
    grad += 2.0 * lam * (cur_sum - N / 2.0) * np.ones_like(h)
    # smooth max value + quadratic penalty
    smooth = (np.log(Z) + beta * m) / beta
    penalty = lam * (cur_sum - N / 2.0) ** 2
    return smooth + penalty, grad, O

def _optimize(N, iters, lr0, beta_start, beta_end, lam, rng):
    target = N / 2.0
    h = rng.random(N)
    h = _project_box_sum(h, target)
    best_h = h.copy()
    best_val = np.correlate(h, 1 - h, mode='full').max() * (2.0 / N)

    lr = lr0
    for it in range(iters):
        beta = beta_start + (beta_end - beta_start) * it / max(iters - 1, 1)
        obj, grad, _ = _obj_grad_overlap(h, beta, lam)
        h = h - lr * grad
        h = _project_box_sum(h, target)
        # exact max overlap for monitoring
        cur = np.correlate(h, 1 - h, mode='full').max() * (2.0 / N)
        if cur < best_val:
            best_val = cur
            best_h = h.copy()
            if best_val < 0.3808:            # early success
                break
        lr *= 0.9995                          # gentle decay
    return best_h, best_val
