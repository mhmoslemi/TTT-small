def adjust_sum(h, target):
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # need to reduce total magnitude by diff
        cap = h.sum()  # total amount above 0
        if cap <= 0:
            return h
        # scaling factor
        factor = diff / cap
        h = h * (1 - factor)  # reduce each distance proportionally
        # Might cause tiny negative due numeric; clip
        h = np.clip(h, 0, 1)
    else:
        # need to increase sum
        inc_cap = (1 - h).sum()
        factor = -diff / inc_cap
        h = h + (1 - h) * factor
        h = np.clip(h, 0, 1)
    # after adjustment sum may be slightly off due to clipping, we can recalc recursively.
    return adjust_sum(h, target) if abs(h.sum() - target) > 1e-9 else h

    """Convert any vector to a binary vector with exactly sum(vec)=len(vec)/2."""
    n = len(vec)
    target = n // 2
    # sort by value descending, set top target to 1
    idx = np.argsort(-vec)
    bin_vec = np.zeros(n, dtype=float)
    bin_vec[idx[:target]] = 1.0
    return bin_vec

def _c5_raw(h):
    """Return raw maximal overlap deficiency (integer count) and the value."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max(), corr

def _c5_scaled(h, dx):
    raw, _ = _c5_raw(h)
    return raw * dx

def _swap(h, i, j):
    """Swap values at positions i (currently 1) and j (currently 0)."""
    h2 = h.copy()
    h2[i] = 0.0
    h2[j] = 1.0
    return h2

def _local_improve(h, n, max_trials=30):
    """Try a few targeted swaps that reduce the worst‐shift deficiency."""
    raw, corr = _c5_raw(h)
    p = int(np.argmax(corr) - (n - 1))            # shift giving the maximum
    start = max(0, -p)
    end   = min(n, n - p)
    i_range = np.arange(start, end, dtype=int)
    j_range = i_range + p

    # mismatches for this worst shift
    mis_one = i_range[(h[i_range] == 1) & (h[j_range] == 0)]
    mis_zero = i_range[(h[i_range] == 0) & (h[j_range] == 1)]

    if mis_one.size == 0 or mis_zero.size == 0:
        return h, raw

    for _ in range(max_trials):
        i = random.choice(mis_one)
        j = random.choice(mis_zero)
        h2 = _swap(h, i, j)
        raw2, _ = _c5_raw(h2)
        if raw2 < raw:
            return h2, raw2
    return h, raw
