def compute_c5(h):
    # h is a 1D numpy array of floats (0/1)
    # compute correlation
    c = np.correlate(h, 1-h, mode='full')
    dx = 2.0 / h.size

def random_binary_solution(n):

    return h

            best_h[idx] = 0.0

    return best_h, float(best_val), n
