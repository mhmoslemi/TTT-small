def c5_of_h(h, dx):
    a = h
    b = 1 - h
    # compute linear convolution of a with reversed b:
    n = a.size
    size = 2 * n - 1
    # choose power-of-two length for FFT
    fsize = 1 << ((size - 1).bit_length())
    # compute
    A = np.fft.rfft(a, fsize)
    B = np.fft.rfft(b[::-1], fsize)
    conv = np.fft.irfft(A * B, fsize)[:size] * dx
    # return max
    return conv.max()

def compute_c5(arr):
    a = arr
    n = a.size
    size = 2*n - 1
    fsize = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(a, fsize)
    B = np.fft.rfft(a[::-1], fsize)
    conv = np.fft.irfft(A * B, fsize)[:size] * (2.0 / n)
    min_corr = conv.min()
    return 1.0 - min_corr

import time, random, numpy as np

def _autocorr(arr):
    """Linear autocorrelation of arr with itself, scaled by dx=2/len(arr)."""
    n = arr.shape[0]
    size = 2 * n - 1
    fsize = 1 << ((size - 1).bit_length())
    A = np.fft.rfft(arr, fsize)
    B = np.fft.rfft(arr[::-1], fsize)
    conv = np.fft.irfft(A * B, fsize)[:size] * (2.0 / n)
    return conv

def _c5(arr):
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx = 1‑min autocorrelation."""
    r = _autocorr(arr)
    m = r.min()
    if m < 0.0:
        m = 0.0
    return 1.0 - m

def _w_for_lag(arr, lag):
    """Contribution weight w_i = arr[i+lag] + arr[i‑lag] (out‑of‑range = 0)."""
    n = arr.size
    if lag >= 0:
        forward = np.empty(n, dtype=arr.dtype)
        forward[:n - lag] = arr[lag:]
        forward[n - lag:] = 0.0
        backward = np.empty(n, dtype=arr.dtype)
        backward[lag:] = arr[:n - lag]
        backward[:lag] = 0.0
    else:
        sh = -lag
        forward = np.empty(n, dtype=arr.dtype)
        forward[sh:] = arr[:n - sh]
        forward[:sh] = 0.0
        backward = np.empty(n, dtype=arr.dtype)
        backward[:n - sh] = arr[sh:]
        backward[n - sh:] = 0.0
    return forward + backward
