def compute_objective(h, tau, alpha):
    n = h.shape[0]
    dx = 2.0 / n
    A = np.correlate(h, 1 - h, mode='full')

    logZ = scipy.special.logsumexp(A / tau)

def compute_gradient(h, tau, alpha, ind1, ind2):

    dx = 2.0 / n
    A = np.correlate(h, 1 - h, mode='full')
    # compute weights w for each lag

import time
import numpy as np
from scipy import special, optimize

# global placeholders (will be set by set_globals)
_CURRENT_TAU = None
_CURRENT_ALPHA = None
_TARGET_SUM = None
_DX = None
_IND1 = None
_IND2 = None
_N = None


def set_globals(n, tau, alpha):
    """Initialize global parameters for the current problem size and smoothing."""
    global _CURRENT_TAU, _CURRENT_ALPHA, _TARGET_SUM, _DX, _IND1, _IND2, _N
    _CURRENT_TAU = float(tau)
    _CURRENT_ALPHA = float(alpha)
    _N = int(n)
    _DX = 2.0 / _N
    _TARGET_SUM = _N / 2.0
    idx = np.arange(_N)
    diff = idx[:, None] - idx[None, :]          # i - j
    offset = _N - 1
    _IND1 = -diff + offset                      # indices for shift j-i
    _IND2 = diff + offset                       # indices for shift i-j


def objective_func(h):
    """Smooth max objective with quadratic penalty on the integral constraint."""
    # correlation A = h * (1 - h) (full mode)
    A = np.correlate(h, 1.0 - h, mode="full")
    # smooth max via log-sum-exp
    logZ = special.logsumexp(A / _CURRENT_TAU)
    smooth = _CURRENT_TAU * logZ
    # penalty for sum deviation
    s = np.sum(h) - _TARGET_SUM
    pen = _CURRENT_ALPHA * s * s
    return _DX * smooth + pen


def gradient_func(h):
    """Gradient of the smooth objective (including penalty)."""
    A = np.correlate(h, 1.0 - h, mode="full")
    B = A / _CURRENT_TAU
    logZ = special.logsumexp(B)
    w = np.exp(B - logZ)                # weights, sum to 1
    # build weighted Toeplitz matrices via index arrays
    W1 = w[_IND1]                       # shape (n, n)
    W2 = w[_IND2]                       # shape (n, n)
    term1 = W1.dot(1.0 - h)
    term2 = W2.dot(h)
    grad = _DX * (term1 - term2)
    # penalty gradient (same for each coordinate)
    s = np.sum(h) - _TARGET_SUM
    grad += 2.0 * _CURRENT_ALPHA * s
    return grad


def project_sum_box(h):
    """Project vector onto the hyperplane sum(h)=target while staying in [0,1]."""
    target = _TARGET_SUM
    # binary search for lambda such that sum(clip(h - lam,0,1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(60):

        proj = np.clip(h - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def compute_c5(h):
    """Exact C5 bound for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx
