import numpy as np, time, math

def _c5_val(h):
    n = len(h)
    # correlation h with (1-h)
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * (2.0 / n)

def _proj_to_sum(v, target, it=60):
    lo, hi = -2.0, 2.0
    for _ in range(it):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _random_feasible(n, rng):
    v = rng.random(n)
    return _proj_to_sum(v, n * 0.5)

            new_h = cur_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:                           # swap near‑binary values
            i = rng.integers(n)
            while cur_h[i] < 0.5:
                i = rng.integers(n)
            j = rng.integers(n)
            while cur_h[j] > 0.5 or j == i:
                j = rng.integers(n)
            new_h = cur_h.copy()
            new_h[i], new_h[j] = cur_h[j], cur_h[i]
        val = _c5_val(new_h)
        d = val - cur_val
        if d < 0.0 or rng.random() < math.exp(-d / max(T, 1e-12)):
            cur_h, cur_val = new_h, val
            if val < best_val:
                best_h, best_val = new_h.copy(), val
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_all = time.time()
    # initial vector (if provided)
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=float)
        n0 = len(base)
        target = n0 * 0.5
        if abs(base.sum() - target) > 1e-9:
            base = _proj_to_sum(base, target)
    else:
        n0 = 48               # even fallback size
        base = _random_feasible(n0, rng)

    best_h, best_val, best_n = base.copy(), _c5_val(base), n0
    max_n = 200
    sizes = [n0]
    while sizes[-1] * 2 <= max_n:
        sizes.append(sizes[-1] * 2)

    for idx, n in enumerate(sizes):
        if time.time() - start_all > budget_s - 5:
            break
        # build start vector for this resolution
        if n == n0:
            cur = base.copy()
        else:
            rep = n // n0
            extra = n % n0
            cur = np.concatenate([np.repeat(best_h, rep), best_h[:extra]]) if best_h.shape[0] == n0 else _random_feasible(n, rng)
        cur_val = _c5_val(cur)
        # allocate a slice of remaining time (roughly equal for each level)
        rem = budget_s - (time.time() - start_all)
        give = max(2.0, rem * 0.4 / (len(sizes) - idx))
        cur_opt, val_opt = _anneal(cur, cur_val, give, rng)
        if val_opt < best_val:
            best_h, best_val, best_n = cur_opt.copy(), val_opt, n
    return best_h, float(best_val), int(best_n)
