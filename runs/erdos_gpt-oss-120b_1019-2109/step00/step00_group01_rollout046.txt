    # algorithm: subtract tau from each component, then clamp to [0,1].

    # Use bisection to find tau.

        s = y.sum()
        if s > target:
            low = tau
        else:

def projected_subgradient_descent(h, iters, step0=0.5):
    for t in range(1, iters+1):
        # compute correlation and offset of max
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        offset = max_idx - (len(h) - 1)
        # compute gradient at this offset
        g = gradient_at_offset(h, offset)
        # step size
        step = step0 / np.sqrt(t)
        h_new = h - step * g
        # projection onto [0,1] and sum constraint
        h_new = project_onto_box_sum(h_new, target_sum = len(h)/2)
        h = h_new
    return h

def project_onto_box_sum(x, target):
    # clip to [0,1] then adjust with water-filling.
    # We'll use bisection for tau.
    # Find tau s.t. sum(clamp(x - tau, 0,1)) = target.
    low, high = -1.0, 1.0
    for _ in range(40):
        tau = (low + high)/2.0
        y = np.clip(x - tau, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(x - (low+high)/2.0, 0.0, 1.0)
    # Ensure sum exactly target by adjusting few entries
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # distribute diff among entries not at bounds (0 or 1)
        eps = diff / (np.sum((y > 0) & (y < 1)))
        # If there is no interior, adjust a random interior after slight relaxation
        # To avoid division zero, just adjust first element.
        if eps == 0 or np.isnan(eps):
            # find any index
            idx = np.argmax(y)
            y[idx] += diff
            y[idx] = np.clip(y[idx], 0.0, 1.0)
        else:
            # adjust interior
            mask = (y > 0) & (y < 1)
            y[mask] += eps
            # after rounding may violate bounds; clip again and final adjust small.
            y = np.clip(y, 0.0, 1.0)
            # recompute diff after clip; do simple fix.
            residual = target - y.sum()
            if abs(residual) > 1e-9:
                # adjust one element
                idx = np.argmax(y) if residual < 0 else np.argmin(y)
                y[idx] = np.clip(y[idx] + residual, 0.0, 1.0)
    return y

def gradient_at_offset(h, offset):
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if offset >= 0:
        a = offset
        # g[i] for i in 0..n-a-1 subtract h[i+a]
        g[:n - a] -= h[a:]
        # g[i] for i in a..n-1 subtract h[i-a]
        g[a:] -= h[:n - a]
    else:
        a = -offset
        g[a:] -= h[:n - a]
        g[:n - a] -= h[a:]
    return g

def project_box_sum(x, target):
    # we need to find tau such that y = clip(x - tau, 0, 1) => sum(y) = target.
    # Since function sum(y) is monotonic decreasing in tau.
    low = -1.0
    high = 1.0
    for _ in range(30):
        tau = (low + high) / 2.0
        y = np.clip(x - tau, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(x - (low + high) / 2.0, 0.0, 1.0)
    # final adjustment small if needed
    diff = target - y.sum()
    if abs(diff) > 1e-9:
        # adjust entries not at bounds slightly
        # find interior entries
        interior = (y > 0) & (y < 1)
        count = interior.sum()
        if count > 0:
            y[interior] += diff / count
            y = np.clip(y, 0.0, 1.0)
        else:
            # adjust one entry
            idx = np.argmax(y) if diff < 0 else np.argmin(y)
            y[idx] = np.clip(y[idx] + diff, 0.0, 1.0)
    return y

import time
import numpy as np

# ---------- helper functions ----------
def evaluate_c5(h):
    """Return C5 for a given step function h (array)."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def gradient_at_offset(h, off):
    """Subgradient of I(off) = ∫h(1−h(·+off)) at the current h."""
    n = h.shape[0]
    g = np.ones(n, dtype=np.float64)
    if off >= 0:
        k = off
        if k < n:
            g[: n - k] -= h[k:]
            g[k:] -= h[: n - k]
    else:
        k = -off
        if k < n:
            g[k:] -= h[: n - k]
            g[: n - k] -= h[k:]
    return g


def project_box_sum(x, target, iters=30):
    """
    L2 projection of x onto the set {y∈[0,1]^n : sum(y)=target}.
    Finds τ such that sum(clamp(x-τ,0,1)) = target.
    """
    low, high = -1.0, 1.0
    for _ in range(iters):
        tau = (low + high) * 0.5
        y = np.clip(x - tau, 0.0, 1.0)
        s = y.sum()
        if s > target:
            low = tau
        else:
            high = tau
    y = np.clip(x - (low + high) * 0.5, 0.0, 1.0)
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        interior = (y > 0) & (y < 1)
        cnt = interior.sum()
        if cnt > 0:
            y[interior] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
        else:
            # adjust a single entry if everything lies on a bound
            idx = np.argmax(y) if diff < 0 else np.argmin(y)
            y[idx] = np.clip(y[idx] + diff, 0.0, 1.0)
    return y


def upsample_initial(h0, n_target):
    """Linear interpolation of a coarse h0 to length n_target."""
    n0 = h0.shape[0]
    if n0 == n_target:
        return h0.copy()
    x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_target, endpoint=False)
    return np.interp(xn, x0, h0)
