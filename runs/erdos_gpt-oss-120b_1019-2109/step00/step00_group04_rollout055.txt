def project_to_sum_box(x, target):
    n = len(x)
    # initial: all free
    low = 0.0
    high = 1.0
    x_proj = x.copy()
    free = np.arange(n)
    while True:
        if len(free) == 0:
            break
        delta = (target - x_proj[free].sum()) / len(free)
        x_proj[free] += delta
        # check bounds
        below = np.where(x_proj[free] < low)[0]
        above = np.where(x_proj[free] > high)[0]
        if len(below)==0 and len(above)==0:
            break
        # fix those out-of-bounds
        if len(below)>0:
            idx = free[below]
            x_proj[idx] = low

def upsample_h(initial, N):
    old_N = len(initial)
    # use linear interpolation onto new grid
    xs_old = np.linspace(0, 2, old_N, endpoint=False)
    xs_new = np.linspace(0, 2, N, endpoint=False)
    new_h = np.interp(xs_new, xs_old, initial)
    # ensure within [0,1]
    new_h = np.clip(new_h, 0, 1)
    # adjust sum to target N/2 via scaling and projection
    target = N/2
    # simple scaling:
    sum_new = new_h.sum()
    if sum_new != 0:
        new_h *= target / sum_new
    # then project to bounds and sum
    new_h = project_to_sum_box(new_h, target)
    return new_h

Define compute_subgradient:

def compute_subgradient(h, max_idxs):
    N = len(h)
    grad = np.zeros_like(h)
    for idx in max_idxs:
        s = idx - (N - 1)  # shift s may be negative
        if s >= 0:
            # left part: i=0..N-1-s
            i1 = np.arange(N - s)
            grad[i1] += (1 - h[i1 + s])
            # right part: i=s..N-1
            i2 = np.arange(s, N)
            grad[i2] += -h[i2 - s]
        else:
            s = -s
            # shift negative
            # left part: i=s..N-1
            i1 = np.arange(s, N)
            grad[i1] += (1 - h[i1 - s])
            # right part: i=0..N-1-s
            i2 = np.arange(N - s)
            grad[i2] += -h[i2 + s]
    # average gradient
    grad /= len(max_idxs)
    return grad

def project_to_sum_box(x, target):
    # project onto [0,1] and sum target (hyperplane)
    # Use algorithm similar to water-filling
    # We'll solve by iterative clipping with lambda (dual variable)
    # Closed form exists: sort x, find lambda such that sum of clamp(x_i - lambda) = target
    # Equivalent to projecting onto simplex with lower bound 0 and upper bound 1.
    # We'll implement simple iterative method (bisection) using lambda.
    # Let x_i' = clip(x_i - lambda, 0, 1). We need sum x_i' = target.
    # lambda can be found via bisection in range [min(x_i)-1, max(x_i)].
    # Determine feasible lambda bounds: lower bound L = min(x) - 1 (makes all at most 1 after subtract) maybe negative large; Upper bound U = max(x) (makes all non-negative after subtract). We'll set L = -1, U = 1 (since x values in [0,1] after clipping? Actually x may be any after step).
    # We'll use binary search for lambda.
    x = np.asarray(x, dtype=float)
    # Clip to range [0,1] after adjusting? We'll not clip now; the formula uses clip(x - lambda, 0, 1)
    # Bisection tolerance
    lo = -1.0
    hi = 1.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    proj = np.clip(x - (lo+hi)/2.0, 0.0, 1.0)
    return proj

import time, math, numpy as np

# --------------------------------------------------------------
# helpers
# --------------------------------------------------------------
def compute_c5(h):
    """raw C5 value (max overlap * dx) for a vector h."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def project_to_sum_box(x, target):
    """
    Projection of x onto the polytope 0<=x<=1, sum(x)=target.
    Uses bisection on the dual variable lambda.
    """
    x = np.asarray(x, dtype=float)
    lo = x.min() - 1.0
    hi = x.max()
    for _ in range(50):
        lam = (lo + hi) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def subgradient(h, max_idxs):
    """subgradient of the max overlap w.r.t. h for the given indices."""
    N = len(h)
    g = np.zeros_like(h)
    for idx in max_idxs:
        s = idx - (N - 1)          # shift (negative possible)
        if s >= 0:
            # forward shift
            i1 = np.arange(N - s)
            g[i1] += (1.0 - h[i1 + s])
            i2 = np.arange(s, N)
            g[i2] += -h[i2 - s]
        else:
            s = -s
            i1 = np.arange(s, N)
            g[i1] += (1.0 - h[i1 - s])
            i2 = np.arange(N - s)
            g[i2] += -h[i2 + s]
    return g / len(max_idxs)


def smooth_optimize(h0, max_seconds):
    """gradient‑descent with a smooth max (log‑sum‑exp) approximation."""
    h = h0.astype(float).copy()
    N = len(h)
    target = N / 2.0
    best_h, best_val = h.copy(), compute_c5(h)
    start = time.time()
    it = 0
    eta0 = 0.7
    alpha = 1.0            # smoothing parameter
    while time.time() - start < max_seconds:
        # correlation and smooth max
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = corr.max()
        # indices attaining the current maximum (exact)
        idxs = np.where(np.abs(corr - max_val) < 1e-12)[0]
        # subgradient for those indices
        grad = subgradient(h, idxs)

        # step size schedule
        eta = eta0 / (1.0 + it * 0.001)
        # gradient step (we minimise the max, so move opposite to grad)
        h_new = h - eta * grad
        # enforce feasibility
        h_new = project_to_sum_box(h_new, target)

        new_val = compute_c5(h_new)
        if new_val < best_val:
            best_val, best_h = new_val, h_new.copy()

        # simple acceptance rule (deterministic descent)
        if new_val <= max_val:
            h = h_new
        else:
            # occasional uphill move (simulated annealing flavour)
            T = max(1e-4, 1.0 - (time.time() - start) / max_seconds)
            if np.random.rand() < math.exp(-(new_val - max_val) / (T * 0.05)):
                h = h_new
        it += 1
        # gradually tighten smoothing
        if it % 200 == 0 and alpha < 10.0:
            alpha *= 1.2
    return best_h, best_val


def binary_optimize(h0, max_seconds):
    """swap‑based simulated annealing on the binary feasible set."""
    h = h0.astype(np.int8).copy()
    N = len(h)
    target = N // 2
    best_h, best_val = h.copy(), compute_c5(h)
    ones = set(np.flatnonzero(h == 1))
    zeros = set(np.flatnonzero(h == 0))
    start = time.time()
    it = 0
    while time.time() - start < max_seconds:
        # pick a random 1 and a random 0 to swap
        i = np.random.choice(list(ones))
        j = np.random.choice(list(zeros))
        # perform swap
        h[i], h[j] = 0, 1
        # update sets
        ones.remove(i); zeros.add(i)
        zeros.remove(j); ones.add(j)

        new_val = compute_c5(h)
        if new_val < best_val:
            best_val, best_h = new_val, h.copy()
        # acceptance
        if new_val < best_val or np.random.rand() < 0.001:
            # keep the move (already in h)
            pass
        else:
            # revert
            h[i], h[j] = 1, 0
            ones.add(i); zeros.remove(i)
            zeros.add(j); ones.remove(j)
        it += 1
        # early stop if sufficiently good
        if best_val <= 0.3808:
            break
    return best_h.astype(float), best_val


def round_to_binary(h):
    """convert a continuous vector to the nearest feasible binary vector."""
    N = len(h)
    need = N // 2
    idx = np.argpartition(-h, need)[:need]
    bin_h = np.zeros(N, dtype=float)
    bin_h[idx] = 1.0
    return bin_h


def upscale_initial(init, N):
    """interpolate an existing pattern to a new resolution."""
    oldN = len(init)
    xs_old = np.linspace(0.0, 2.0, oldN, endpoint=False)
    xs_new = np.linspace(0.0, 2.0, N, endpoint=False)
    new = np.interp(xs_new, xs_old, init)
    new = np.clip(new, 0.0, 1.0)
    new = project_to_sum_box(new, N / 2.0)
    return new
