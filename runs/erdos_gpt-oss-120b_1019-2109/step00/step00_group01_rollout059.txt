def grad_h(h, beta):
    N = len(h)
    M = 2*N-1
    # compute f = correlation between h and 1-h (no scaling)
    f_all = np.correlate(h, 1 - h, mode='full')  # shape M
    # compute weights
    max_f = np.max(f_all)  # for numerical stability
    exp_vals = np.exp(beta*(f_all - max_f))  # subtract max to avoid overflow
    w = exp_vals / np.sum(exp_vals)
    # compute gradient
    g = np.zeros_like(h)
    # precompute 1-h
    one_minus = 1 - h
    # for l from -(N-1) to N-1:
    for idx, l in enumerate(range(-N+1, N)):
        w_l = w[idx]
        if w_l == 0:
            continue
        # term1: for each i where i + l in [0,N-1], add w_l * (1 - h_{i+l})
        # term2: for each i where i - l in range, subtract w_l * h_{i-l}
        # we can vectorize:
        if l >= 0:
            # i from 0 to N-l-1: i + l in bounds
            g[:N-l] += w_l * one_minus[l:]  # one_minus[i+l]
            # i from l to N-1: i - l >=0 => i-l in bounds, subtract h[i-l]
            g[l:] -= w_l * h[:N-l]
        else:
            # l negative: let k = -l >0

        hi = np.max(v) + 1.0

                x = np.clip(x, 0.0, 1.0)

def project(v):
    target = v.size/2 # N/2
    lower = 0.0
    upper = 1.0
    # initial λ bounds
    lo = np.min(v - upper) - 1.0
    hi = np.max(v - lower) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - hi, lower, upper)
    # adjust tiny residual
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # Distribute diff among interior entries
        interior = (x > lower + 1e-12) & (x < upper - 1e-12)
        if interior.any():
            incr = diff / interior.sum()
            x[interior] += incr
            # ensure still within bounds
            x = np.clip(x, lower, upper)
        else:
            # fallback: adjust all entries uniformly (while staying within bounds)
            x += diff / x.size
            x = np.clip(x, lower, upper)
            # after adjusting may not meet exactly but fine
    return x

import time
import numpy as np

def _project_to_capped_simplex(v, target):
    """Project v onto {x | sum x = target, 0 <= x <= 1} using water-filling."""
    lo = np.min(v - 1.0) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - hi, 0.0, 1.0)
    # tiny correction if needed
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        interior = (x > 0.0) & (x < 1.0)
        if interior.any():
            incr = diff / interior.sum()
            x[interior] += incr
            x = np.clip(x, 0.0, 1.0)
    return x

def _c5_from_h(h):
    """Return C5 for a discretised step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / h.size)

def _gradient(h, beta):
    """Gradient of the smooth max (log-sum-exp) of the overlap functional."""
    n = h.size
    f = np.correlate(h, 1.0 - h, mode='full')
    max_f = f.max()
    w = np.exp(beta * (f - max_f))
    w /= w.sum()
    one_minus = 1.0 - h
    g = np.zeros_like(h)
    # lags from -(n-1) .. (n-1)
    for idx, lag in enumerate(range(-n + 1, n)):
        wl = w[idx]
        if wl == 0.0:
            continue
        if lag >= 0:
            # (1 - h_{i+lag})
            g[: n - lag] += wl * one_minus[lag:]
            # -h_{i-lag}
            g[lag:] -= wl * h[: n - lag]
        else:
            k = -lag
            g[k:] += wl * one_minus[: n - k]
            g[: n - k] -= wl * h[k:]
    return g

def _optimize_fixed_n(n_points, init_h, rng, time_limit, start_time):
    """Projected gradient descent on a fixed grid size."""
    target_sum = n_points / 2.0
    if init_h is None:
        h = _project_to_capped_simplex(rng.random(n_points), target_sum)
    else:
        if len(init_h) != n_points:
            x_old = np.linspace(0.0, 2.0, len(init_h), endpoint=False)
            x_new = np.linspace(0.0, 2.0, n_points, endpoint=False)
            h = np.interp(x_new, x_old, init_h)
        else:
            h = init_h.copy()
        h = np.clip(h, 0.0, 1.0)
        h = _project_to_capped_simplex(h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    max_iters = 8000
    beta = 5.0
    beta_inc = (300.0 - beta) / max_iters
    lr = 0.12
    lr_decay = 0.9996

    for it in range(max_iters):
        if time.time() - start_time > time_limit - 5.0:
            break
        g = _gradient(h, beta)
        h_new = h - lr * g
        h_new = _project_to_capped_simplex(h_new, target_sum)
        c5_new = _c5_from_h(h_new)

        # accept if improvement, otherwise keep old (simple greedy)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= _c5_from_h(h):
            h = h_new
        else:
            # occasional random kick to escape plateaus
            if rng.random() < 0.02:
                h = _project_to_capped_simplex(h + 0.15 * (rng.random(n_points) - 0.5), target_sum)

        beta = min(300.0, beta + beta_inc)
        lr *= lr_decay

    # final fine-tuning with very large beta (essentially max)
    for _ in range(1500):
        if time.time() - start_time > time_limit - 5.0:
            break
        g = _gradient(h, 1e6)  # approximates sub‑gradient of the true max
        h_new = h - lr * g
        h_new = _project_to_capped_simplex(h_new, target_sum)
        c5_new = _c5_from_h(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new < _c5_from_h(h):
            h = h_new
        lr *= 0.9995
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) representing a
    step function on [0,2] with integral 1 that aims to minimise the
    Erdős C5 constant.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    time_limit = budget_s

    # Retrieve any provided parent construction.
    init_parent = globals().get('initial_h_values', None)
    if init_parent is not None:
        init_parent = np.asarray(init_parent, dtype=float)

    # List of grid sizes we will explore (coarse -> fine).
    grid_sizes = [120, 240, 480, 960]

    best_h = None
    best_c5 = np.inf
    best_n = None

    # First attempt using the parent if it exists.
    if init_parent is not None:
        n0 = init_parent.size
        h_opt, c5_opt = _optimize_fixed_n(n0, init_parent, rng, time_limit, start)
        if c5_opt < best_c5:
            best_c5, best_h, best_n = c5_opt, h_opt, n0

    # Progressive refinement.
    for n in grid_sizes:
        if time.time() - start > time_limit - 5.0:
            break
        if best_h is not None:
            # interpolate current best to new resolution
            x_old = np.linspace(0.0, 2.0, best_h.size, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            init = np.interp(x_new, x_old, best_h)
        else:
            init = None
        h_opt, c5_opt = _optimize_fixed_n(n, init, rng, time_limit, start)
        if c5_opt < best_c5:
            best_c5, best_h, best_n = c5_opt, h_opt, n
        # early exit if we beat the target record
        if best_c5 <= 0.3808:
            break

    # As a safety fallback create a simple linear shape if nothing succeeded.
    if best_h is None:
        n = 200
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        # h(x) = a - b*|x-1|  with a=0.7 (example)
        a, b = 0.7, 0.4
        best_h = a - b * np.abs(x - 1.0)
        best_h = np.clip(best_h, 0.0, 1.0)
        target = n / 2.0
        best_h = _project_to_capped_simplex(best_h, target)
        best_c5 = _c5_from_h(best_h)
        best_n = n

    # Final projection to guarantee constraints.
    best_h = _project_to_capped_simplex(best_h, best_h.size / 2.0)
    final_c5 = _c5_from_h(best_h)

    return best_h, float(final_c5), int(best_n)
