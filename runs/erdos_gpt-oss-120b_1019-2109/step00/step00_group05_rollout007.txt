def propose_directed_move(h):
    conv = np.correlate(h, 1 - h, mode='full')
    L_idx = np.argmax(conv) - (len(h)-1)  # shift value (range [-N+1,N-1])
    L = L_idx
    N = len(h)
    # Compute gradient
    first = np.ones(N)
    second = np.zeros(N)
    if L >= 0:
        first[:N-L] = 1 - h[L:]
        second[L:] = h[:N-L]
    else:
        # L negative
        Lneg = -L
        first[Lneg:] = 1 - h[:N-Lneg]
        second[:N-Lneg] = h[Lneg:]
    grad = first - second
    # indices for moving mass
    pos_idx = np.where(grad > 1e-9)[0]
    neg_idx = np.where(grad < -1e-9)[0]
    if pos_idx.size ==0 or neg_idx.size ==0:
        return None
    i = np.random.choice(pos_idx)
    j = np.random.choice(neg_idx)
    # Determine max transfer that respects bounds
    # Want to move from i (decrease) to j (increase)
    max_transfer = min(cur_h[i], 1 - cur_h[j])
    # Possibly also limited by max_delta
    max_transfer = min(max_transfer, max_delta)
    if max_transfer <= 1e-12:
        return None
    delta = np.random.rand() * max_transfer
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h

import time, math, numpy as np

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k)) dx using discrete correlation."""
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def fix_sum_and_bounds(h, target):
    """Clip to [0,1] and adjust to hit exact sum = target."""
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    # Greedy redistribution
    if diff > 0:
        # need to add mass
        idxs = np.where(h < 1.0)[0]
    else:
        # need to remove mass
        idxs = np.where(h > 0.0)[0]
    if idxs.size == 0:
        return h
    for idx in np.random.permutation(idxs):
        if diff > 0:
            cap = 1.0 - h[idx]
            inc = min(cap, diff)
            h[idx] += inc
            diff -= inc
        else:
            cap = h[idx]
            dec = min(cap, -diff)
            h[idx] -= dec
            diff += dec
        if abs(diff) < 1e-12:
            break
    np.clip(h, 0.0, 1.0, out=h)
    return h


def propose_random_move(h, max_delta):
    """Transfer a random amount of mass between two positions, preserving sum."""
    N = h.shape[0]
    i = np.random.randint(N)
    j = np.random.randint(N - 1)
    if j >= i:
        j += 1
    # direction
    if np.random.rand() < 0.5:
        # i -> j
        cap = min(max_delta, h[i], 1.0 - h[j])
        if cap <= 0:
            return None
        delta = np.random.rand() * cap
        nh = h.copy()
        nh[i] -= delta
        nh[j] += delta
    else:
        # j -> i
        cap = min(max_delta, h[j], 1.0 - h[i])
        if cap <= 0:
            return None
        delta = np.random.rand() * cap
        nh = h.copy()
        nh[i] += delta
        nh[j] -= delta
    return nh


def propose_directed_move(h, max_delta):
    """Use gradient of the worst lag to move mass from a high‑gradient index to a low‑gradient one."""
    N = h.shape[0]
    corr = np.correlate(h, 1 - h, mode="full")
    lag = np.argmax(corr) - (N - 1)   # integer lag in [-N+1, N-1]

    # compute gradient of the worst lag
    first = np.ones(N)
    second = np.zeros(N)
    if lag >= 0:
        first[: N - lag] = 1.0 - h[lag:]
        second[lag:] = h[: N - lag]
    else:
        lu = -lag
        first[lu:] = 1.0 - h[: N - lu]
        second[: N - lu] = h[lu:]
    grad = first - second

    pos = np.where(grad > 1e-12)[0]   # decreasing these lowers the lag value
    neg = np.where(grad < -1e-12)[0]  # increasing these lowers the lag value
    if pos.size == 0 or neg.size == 0:
        return None

    i = np.random.choice(pos)   # will be decreased
    j = np.random.choice(neg)   # will be increased
    cap = min(max_delta, h[i], 1.0 - h[j])
    if cap <= 0:
        return None
    delta = np.random.rand() * cap
    nh = h.copy()
    nh[i] -= delta
    nh[j] += delta
    return nh

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # --------------------------------------------------------
    # Initialise h
    # --------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.array(initial_h_values, dtype=np.float64)
        N0 = base.size
        # Double resolution if the base size is modest
        if N0 <= 100:
            h = np.repeat(base, 2)
        else:
            h = base.copy()
    else:
        # fallback: binary half‑filled vector
        N0 = 70
        h = np.concatenate([np.ones(N0 // 2), np.zeros(N0 - N0 // 2)])
        np.random.shuffle(h)

    N = h.size
    target_sum = N / 2.0
    h = fix_sum_and_bounds(h, target_sum)

    dx = 2.0 / N
    cur_h = h.copy()
    cur_c5 = compute_c5(cur_h, dx)
    best_h = cur_h.copy()
    best_c5 = cur_c5

    # --------------------------------------------------------
    # Simulated annealing parameters
    # --------------------------------------------------------
    temp = 0.05                # initial temperature
    min_temp = 1e-7
    decay = 0.9999             # slow cooling
    p_dir = 0.2                # probability to try directed move
    it = 0

    # --------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------
    while time.time() - start_time < budget_s - 0.3:
        # adapt step size to temperature
        max_delta = min(0.5, max(1e-6, temp * 10))

        if np.random.rand() < p_dir:
            cand = propose_directed_move(cur_h, max_delta)
            if cand is None:
                continue
        else:
            cand = propose_random_move(cur_h, max_delta)
            if cand is None:
                continue

        cand_c5 = compute_c5(cand, dx)

        if cand_c5 < cur_c5 - 1e-12:
            accept = True
        else:
            # Metropolis acceptance
            prob = math.exp(-(cand_c5 - cur_c5) / max(temp, 1e-12))
            accept = np.random.rand() < prob

        if accept:
            cur_h = cand
            cur_c5 = cand_c5
            if cand_c5 < best_c5 - 1e-12:
                best_c5 = cand_c5
                best_h = cand.copy()

        # periodic re‑projection to avoid drift
        if it % 1000 == 0 and it > 0:
            cur_h = fix_sum_and_bounds(cur_h, target_sum)
            best_h = fix_sum_and_bounds(best_h, target_sum)
            cur_c5 = compute_c5(cur_h, dx)
            best_c5 = compute_c5(best_h, dx)

        # cooling
        temp = max(temp * decay, min_temp)
        it += 1

    # --------------------------------------------------------
    # Final clean‑up
    # --------------------------------------------------------
    best_h = fix_sum_and_bounds(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)

    # Use the provided evaluator if available (more robust)
    try:
        final_c5 = evaluate_erdos_solution(best_h)
    except Exception:
        pass

    return best_h, float(final_c5), int(N)
