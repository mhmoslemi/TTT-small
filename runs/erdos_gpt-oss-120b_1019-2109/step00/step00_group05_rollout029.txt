import numpy as np
import time
import random

def _compute_c5(h: np.ndarray) -> float:
    """Return C5 for a step function h given by samples."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) (full mode)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_onto_box_and_sum(y: np.ndarray, total: float) -> np.ndarray:
    """
    Euclidean projection of y onto the intersection of
        0 <= x_i <= 1  and  sum x_i = total.
    Uses a simple bisection on the shift τ so that x_i = clip(y_i + τ,0,1)
    satisfies the sum constraint.
    """
    lo = -np.max(y) - 1.0          # guarantee y+lo <= 0
    hi = 1.0 - np.min(y) + 1.0    # guarantee y+hi >= 1
    for _ in range(60):
        tau = (lo + hi) * 0.5
        x = np.clip(y + tau, 0.0, 1.0)
        s = x.sum()
        if s < total:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) * 0.5
    return np.clip(y + tau, 0.0, 1.0)


def _gradient_of_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Sub‑gradient of the overlap functional for a given shift.
    shift is the integer offset corresponding to np.correlate's output
    (negative values are allowed).
    """
    n = h.shape[0]
    g = np.zeros_like(h)

    if shift >= 0:
        end = n - shift          # indices 0 .. end-1 appear in the forward sum
        if end > 0:
            g[:end] = 1.0                     # derivative of the linear term
            g[:end] -= h[shift:]              # -h_{i+shift} part
            g[shift:] -= h[:end]              # -h_{i-shift} part (i >= shift)
    else:
        t = -shift
        start = t                # indices t .. n-1 appear in the forward sum
        if start < n:
            g[start:] = 1.0
            g[start:] -= h[:n - t]           # -h_{i-shift}= -h_{i+t}
            g[:n - t] -= h[t:]                # -h_{i+shift}= -h_{i-t}
    return g


def _optimize(h0: np.ndarray, time_limit: float) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent on the worst shift.
    Returns the best h found and its C5 value.
    """
    n = h0.shape[0]
    total = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_val = _compute_c5(h)

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # current worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)          # map index to signed shift
        # sub‑gradient of that worst term
        g = _gradient_of_shift(h, shift)

        # diminishing step size
        step = 0.5 / np.sqrt(it + 1.0)

        # gradient descent step + projection onto feasible set
        y = h - step * g
        h = _project_onto_box_and_sum(y, total)

        cur_val = _compute_c5(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        it += 1
    return best_h, best_val


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) aiming at a low C5.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise a feasible h
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h_initial = np.array(initial_h_values, dtype=np.float64)
        n = h_initial.shape[0]
    else:
        n = 120                      # fall back size
        h_initial = np.random.rand(n)

    total = n / 2.0
    h_initial = np.clip(h_initial, 0.0, 1.0)
    h_initial = _project_onto_box_and_sum(h_initial, total)

    # ------------------------------------------------------------------
    # optimisation loop (multiple restarts)
    # ------------------------------------------------------------------
    best_h = h_initial.copy()
    best_c5 = _compute_c5(best_h)

    t_start = time.time()
    overall_budget = budget_s * 0.95          # keep a small safety margin
    while time.time() - t_start < overall_budget:
        remaining = overall_budget - (time.time() - t_start)
        if remaining < 0.5:
            break
        # allocate a chunk (max 30 s) for a single descent run
        chunk = min(30.0, remaining)

        # start from a slightly perturbed version of the current best
        pert = best_h + np.random.normal(scale=0.1, size=n)
        pert = np.clip(pert, 0.0, 1.0)
        pert = _project_onto_box_and_sum(pert, total)

        cand_h, cand_c5 = _optimize(pert, chunk)

        if cand_c5 < best_c5:
            best_c5 = cand_c5
            best_h = cand_h.copy()

    # final sanity check – enforce constraints exactly (tiny numerical drift)
    best_h = _project_onto_box_and_sum(best_h, total)

    return best_h, best_c5, n
