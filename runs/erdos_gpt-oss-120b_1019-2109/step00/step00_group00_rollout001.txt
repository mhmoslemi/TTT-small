import numpy as np

def _project_to_constraints(x, target):
    """Clip to [0,1] and adjust sum to target using a simple water‑filling."""
    x = np.clip(x, 0.0, 1.0)
    for _ in range(10):
        diff = x.sum() - target
        if abs(diff) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt == 0:
            break
        delta = -diff / cnt
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    # tiny residual fix
    residual = x.sum() - target
    if abs(residual) > 1e-9:
        idx = np.where((x > 0.0) & (x < 1.0))[0]
        if idx.size:
            i = idx[0]
            x[i] = np.clip(x[i] - residual, 0.0, 1.0)
        else:
            # fallback: redistribute uniformly
            x -= residual / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

def run(seed: int = 42, budget_s: float = 1100, **kwargs):

    n = 151                     # number of sample points (feel free to change)

    target_sum = n / 2.0        # ∑h = n/2  (integral constraint)

    # ----- initial guess -----

        if len(init) != n:

        else:

    except NameError:          # no parent construction available
        h0 = rng.uniform(0.0, 1.0, size=n)
        h0 = _project_to_constraints(h0, target_sum)

    best_h = h0.copy()
    best_val = _evaluate(best_h)

    # ----- optimisation parameters -----
    restarts = 6               # number of independent starts
    max_iters = 3000           # max inner iterations per restart
    step0 = 0.6                # initial step magnitude
    min_step = 1e-5

    idx_arr = np.arange(n)     # reused index array for speed

    for r in range(restarts):
        # initialise a starting point
        if r == 0:
            h = best_h.copy()
        else:
            h = rng.uniform(0.0, 1.0, size=n)
            h = _project_to_constraints(h, target_sum)

        it = 0
        while True:
            # time check
            if time.time() - start > budget_s - 0.5:
                break
            if it >= max_iters:
                break

            # correlation and worst shift
            corr = np.correlate(h, 1.0 - h, mode='full')
            C = corr * dx
            max_idx = int(np.argmax(C))
            cur_val = C[max_idx]
            lag = max_idx - (n - 1)   # shift corresponding to the maximum

            # gradient of the worst‑shift overlap (sub‑gradient)
            ip = idx_arr + lag
            im = idx_arr - lag

            plus = np.zeros(n)
            ok = (ip >= 0) & (ip < n)
            plus[ok] = 1.0 - h[ip[ok]]

            minus = np.zeros(n)
            ok = (im >= 0) & (im < n)
            minus[ok] = h[im[ok]]

            g = (plus - minus) * dx   # gradient (to be *subtracted*)

            # line‑search (back‑tracking) to guarantee improvement
            step = step0 / np.sqrt(it + 1.0)
            improved = False
            while step > min_step:
                cand = _project_to_constraints(h - step * g, target_sum)
                cand_val = _evaluate(cand)
                if cand_val < cur_val - 1e-12:
                    h = cand
                    cur_val = cand_val
                    improved = True
                    break
                step *= 0.5

            if not improved:
                # small random perturbation to escape possible plateaus
                pert = rng.normal(scale=0.02, size=n)
                h = _project_to_constraints(h + pert, target_sum)
                it += 1
                continue

            # record best solution seen so far
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = h.copy()

            it += 1

        if time.time() - start > budget_s - 0.5:
            break

    final_val = _evaluate(best_h)
    return (best_h.tolist(), float(final_val), n)
