import time, math
import numpy as np

def _c5_value(h, dx):
    """Return max_{k} ∫ h(x)*(1-h(x+k)) dx for a discrete step function."""
    # np.correlate implements the required linear correlation (zero padding)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def _balanced_initial(n, rng):
    """Balanced 0/1 sequence using a low‑discrepancy (golden‑ratio) rule."""
    phi = (np.sqrt(5) - 1) / 2          # ≈0.618
    i = np.arange(n, dtype=np.float64)
    frac = (i * phi) % 1.0
    order = np.argsort(frac)
    h = np.zeros(n, dtype=np.float64)
    h[order[: n // 2]] = 1.0            # exactly half ones
    return h

def _simulated_annealing(h, dx, budget, start):
    """Simple SA on binary vectors with swap moves."""
    n = h.shape[0]
    best_h = h.copy()
    best_val = _c5_value(h, dx)
    cur_h = h.copy()
    cur_val = best_val

    # temperature schedule – start roughly half the current value
    T0 = best_val * 0.5 + 1e-4
    Tmin = 1e-8

    while True:
        now = time.time()
        if now - start >= budget:
            break
        # linear cooling
        T = max(T0 * (1.0 - (now - start) / budget), Tmin)

        # propose a swap of opposite bits
        i = rng.integers(n)
        j = rng.integers(n)
        if cur_h[i] == cur_h[j]:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]          # execute swap
        new_val = _c5_value(cur_h, dx)
        delta = new_val - cur_val
        if delta <= 0.0 or rng.random() < math.exp(-delta / T):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert swap
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C₅ bound, and the number of sample points."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a problem size.  Even n gives an exact binary realisation of
    # the integral constraint.  n≈200 offers a good trade‑off between
    # expressiveness and speed.
    # ------------------------------------------------------------------
    n = 200
    dx = 2.0 / n

    # initial binary pattern – low‑discrepancy balanced word
    h0 = _balanced_initial(n, rng)

    start = time.time()
    # allocate ~90 % of the budget to the stochastic phase
    sa_budget = budget_s * 0.90
    best_h, best_val = _simulated_annealing(h0, dx, sa_budget, start)

    # ------------------------------------------------------------------
    # Final deterministic hill‑climbing: accept only improving swaps.
    # This fine‑tunes the solution found by SA while staying within the
    # remaining time.
    # ------------------------------------------------------------------
    cur_h = best_h.copy()
    cur_val = best_val
    while True:
        now = time.time()
        if now - start >= budget_s:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if cur_h[i] == cur_h[j]:
            continue
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
        new_val = _c5_value(cur_h, dx)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
            # keep the improving swap
        else:
            # revert
            cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

    # Return plain Python objects as required by the harness
    return (best_h.tolist(), float(best_val), n)
