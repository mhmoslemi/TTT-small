def hill_climb_continuous(h, n_steps, time_limit):
    best = h.copy()
    best_val = compute_c5(best)
    start = time.time()
    n = len(h)
    for step in range(n_steps):
        if time.time() - start > time_limit:
            break
        # select two indices i,j
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # max delta range
        max_down_i = best[i]
        max_up_i = 1 - best[i]
        max_down_j = best[j]
        max_up_j = 1 - best[j]
        # Proposed delta to transfer from i to j (i loses, j gains)
        delta_min = max(-max_down_i, -max_up_j)
        delta_max = min(max_up_i, max_down_j)
        if delta_max <= delta_min:
            continue
        delta = rng.uniform(delta_min, delta_max)
        # Apply
        best[i] -= delta
        best[j] += delta
        # Now sum unchanged
        val = compute_c5(best)
        if val < best_val:
            best_val = val
            # keep changes
        else:
            # revert
            best[i] += delta
            best[j] -= delta
    return best, best_val
```

def construct_h(L, Z, n):
    # L: m floats summing to 1
    # Z: m+1 floats summing to 1
    m = len(L)
    dx = 2.0 / n
    midpoints = (np.arange(n) + 0.5) * dx
    h = np.zeros(n, dtype=float)
    pos = 0.0
    # start with zero interval Z[0]
    segs = []
    for i in range(m):
        segs.append(('zero', Z[i]))
        segs.append(('one', L[i]))
    segs.append(('zero', Z[m]))

    for typ, seglen in segs:

        # due to floating errors, consider points where midpoint >= pos - eps and < next_pos + eps

        if zeros.size > 0:

        ones = np.where(h > 1 - 1e-12)[0]

def construct_h(L, Z, n):
    # L shape (m), Z shape (m+1)
    m = len(L)
    # Build lengths in order: zero[0], one[0], zero[1], one[1], ...
    lengths = np.empty(2*m+1)
    types = np.empty(2*m+1, dtype=int)
    lengths[0] = Z[0]
    types[0] = 0
    idx = 1
    for i in range(m):
        lengths[idx] = L[i]; types[idx] = 1; idx+=1
        lengths[idx] = Z[i+1]; types[idx] = 0; idx+=1
    # Compute cumulative edges
    edges = np.concatenate(([0.0], np.cumsum(lengths)))  # length 2*m+2
    # compute midpoints
    dx = 2.0 / n
    mids = (np.arange(n) + 0.5) * dx
    seg_idx = np.searchsorted(edges, mids, side='right') - 1
    # seg_idx in [0, 2*m]
    h = types[seg_idx].astype(float)
    # adjust sum to target
    # sum over h equals number of mids where types=1
    # The exact sum may not equal target due to rounding, adjust:
    target = n / 2
    cur_sum = h.sum()
    diff = target - cur_sum
    if diff > 1e-12:
        # find zeros
        zeros = np.where(h < 1e-12)[0]
        if zeros.size:
            h[zeros[0]] = diff
    elif diff < -1e-12:
        ones = np.where(h > 1-1e-12)[0]
        if ones.size:
            h[ones[0]] = 1.0 + diff
    return h

import time
import math
import numpy as np
import scipy.optimize as opt

# ---------- core evaluation ----------
def _c5_of_h(h: np.ndarray) -> float:
    """Exact discretized C5 for a step vector h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / h.size)


# ---------- construct binary step from interval lengths ----------
def _construct_h(L: np.ndarray, Z: np.ndarray, n: int) -> np.ndarray:
    """
    Build a 0/1 step function on [0,2] with total length of 1's = 1.
    L : lengths of the 'one' intervals (sum to 1)
    Z : lengths of the 'zero' intervals (len = len(L)+1, sum to 1)
    n : number of discretisation points (dx = 2/n)
    Returns an array of length n with values 0 or 1 (plus a single
    fractional correction to enforce the exact integral constraint).
    """
    m = L.size
    # interleave zero/one segments: zero0, one0, zero1, one1, ..., zerom
    seg_len = np.empty(2 * m + 1, dtype=float)
    seg_type = np.empty(2 * m + 1, dtype=int)   # 1 = one, 0 = zero
    seg_len[0] = Z[0]
    seg_type[0] = 0
    pos = 1
    for i in range(m):
        seg_len[pos] = L[i]
        seg_type[pos] = 1
        pos += 1
        seg_len[pos] = Z[i + 1]
        seg_type[pos] = 0
        pos += 1
    # cumulative edges of the pieces (start at 0, end at 2)
    edges = np.concatenate(([0.0], np.cumsum(seg_len)))
    # mid‑points of the discretisation cells
    dx = 2.0 / n
    mids = (np.arange(n) + 0.5) * dx
    # which segment each midpoint falls into?
    idx = np.searchsorted(edges, mids, side="right") - 1
    h = seg_type[idx].astype(float)

    # enforce exact sum = n/2 (i.e. integral = 1)
    target = n / 2.0
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        if diff > 0:
            zeros = np.where(h < 1e-12)[0]
            if zeros.size:
                h[zeros[0]] = diff
        else:
            ones = np.where(h > 1 - 1e-12)[0]
            if ones.size:
                h[ones[0]] = 1.0 + diff
    # safety clip
    np.clip(h, 0.0, 1.0, out=h)
    return h


# ---------- objective for DE ----------
def _bound_from_params(p: np.ndarray, m: int, n: int) -> float:
    """Map raw DE parameters to a C5 bound."""
    raw_one = p[:m]
    raw_zero = p[m:]
    # avoid zero‑division – the eps added is negligible for optimisation
    L = raw_one / (raw_one.sum() + 1e-12)
    Z = raw_zero / (raw_zero.sum() + 1e-12)
    h = _construct_h(L, Z, n)
    return _c5_of_h(h)


def _h_from_params(p: np.ndarray, m: int, n: int):
    """Return the step vector and its exact C5 bound for given DE parameters."""
    raw_one = p[:m]
    raw_zero = p[m:]
    L = raw_one / (raw_one.sum() + 1e-12)
    Z = raw_zero / (raw_zero.sum() + 1e-12)
    h = _construct_h(L, Z, n)
    return h, _c5_of_h(h)


# ---------- simple hill‑climbing for continuous vectors ----------
def _hill_climb(h0: np.ndarray, max_iters: int, time_limit: float, rng: np.random.Generator):
    """Transfer mass between two coordinates while keeping sum constant."""
    h = h0.copy()
    best_val = _c5_of_h(h)
    n = h.size
    start = time.time()
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # feasible delta range (add to i, subtract from j)
        max_up_i = 1.0 - h[i]
        max_down_i = h[i]
        max_up_j = 1.0 - h[j]
        max_down_j = h[j]
        d_min = max(-max_down_i, -max_up_j)
        d_max = min(max_up_i, max_down_j)
        if d_max <= d_min:
            continue
        delta = rng.uniform(d_min, d_max)
        h[i] += delta
        h[j] -= delta
        val = _c5_of_h(h)
        if val < best_val:
            best_val = val
        else:
            # revert
            h[i] -= delta
            h[j] += delta
    return h, best_val

    try:
        init = np.array(initial_h_values, dtype=float)  # noqa: F821
        n0 = init.size
        target0 = n0 / 2.0
        cur0 = init.sum()
        if abs(cur0 - target0) > 1e-9:
            init *= (target0 / cur0)
        init = np.clip(init, 0.0, 1.0)
        val0 = _c5_of_h(init)
        best_h, best_val = init, val0
    except Exception:
        pass  # no initial_h_values supplied

    # ---- 2. parameters of the discretisation ----
    n = 120                     # points on [0,2]; can be changed if time permits
    target = n / 2.0

    # ---- 3. optimise using interval‑parameter DE for several numbers of intervals ----
    for m in (3, 4, 5, 6):
        if time.time() - start_time > budget_s - 5:
            break
        dim = 2 * m + 1
        bounds = [(0.0, 1.0)] * dim

        # remaining time determines a safe maxiter count
        remaining = budget_s - (time.time() - start_time)
        # a modest budget per m (popsize 8, up to 150 generations)
        maxiter = min(150, max(10, int(remaining // 5)))
        popsize = 8

        # differential evolution
        result = opt.differential_evolution(
            lambda p, M=m: _bound_from_params(p, M, n),
            bounds,
            seed=int(rng.integers(0, 2**32 - 1)),
            maxiter=maxiter,
            popsize=popsize,
            polish=True,
            updating="deferred",
            atol=0,
            tol=0,
        )
        h_candidate, val_candidate = _h_from_params(result.x, m, n)
        if val_candidate < best_val:
            best_val = val_candidate
            best_h = h_candidate.copy()

        # ---- 4. local continuous refinement (hill‑climb) ----
        if best_h is not None:
            time_left = budget_s - (time.time() - start_time)
            if time_left > 1.0:
                # allocate a fraction of the remaining budget
                hc_iters = 200_000
                hc_time = min(0.6 * time_left, 5.0)
                best_h, best_val = _hill_climb(best_h, hc_iters, hc_time, rng)

    # ---- 5. final sanity adjustments ----
    if best_h is not None:
        # enforce exact integral again (possible tiny drift after hill‑climb)
        cur = best_h.sum()
        diff = target - cur
        if abs(diff) > 1e-12:
            if diff > 0:
                zeros = np.where(best_h < 1e-12)[0]
                if zeros.size:
                    best_h[zeros[0]] = diff
            else:
                ones = np.where(best_h > 1 - 1e-12)[0]
                if ones.size:
                    best_h[ones[0]] = 1.0 + diff
        np.clip(best_h, 0.0, 1.0, out=best_h)
        final_val = _c5_of_h(best_h)
    else:
        best_h = np.zeros(0, dtype=float)
        final_val = math.inf

    # return as required by the harness
    return best_h.tolist(), float(final_val), len(best_h)
