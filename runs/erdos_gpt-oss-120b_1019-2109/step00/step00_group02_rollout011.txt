    # Approach: First clip u to [0,1] resulting v.

import time

import math
import numpy as np

# projection onto the intersection of the box [0,1]^n and the affine hyperplane sum x = target
def _project_box_sum(v, target):
    # start by clipping to the box
    x = np.clip(v, 0.0, 1.0)
    # water‑filling on the free (non‑bound) coordinates
    for _ in range(100):
        s = x.sum()
        if abs(s - target) < 1e-12:
            break
        resid = target - s
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        nfree = int(free.sum())
        if nfree == 0:
            # all variables are at a bound – should not happen for feasible target
            break
        delta = resid / nfree
        x[free] += delta
        x = np.clip(x, 0.0, 1.0)
    return x
