def run(seed=42, budget_s=1100, **kwargs):

    start_time = time.time()

        if cur > target:                       # too many ones -> delete some
            idx = np.where(a == 1)[0]
            np.random.shuffle(idx)
            a[idx[:cur - target]] = 0
        elif cur < target:                     # too few ones -> add some
            idx = np.where(a == 0)[0]

    else:

    # ----------------------------------------------------------------------

        corr = np.correlate(arr, 1 - arr, mode='full')
        return corr.max() * dx, corr

        O = np.correlate(cur_arr, 1 - cur_arr, mode='full')
        max_O = O.max()
        # update global best if improved

        if cur_val < best_c5:

        max_lags = np.flatnonzero(O == max_O) - (n - 1)
