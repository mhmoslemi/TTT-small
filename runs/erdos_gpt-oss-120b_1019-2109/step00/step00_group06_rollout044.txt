def compute_gradient(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        # overlapping indices i = 0..n-k-1
        i = np.arange(0, n - k)
        grad[i] += 1 - h[i + k]
        grad[i + k] += -h[i]

import time
import random
import numpy as np

def compute_c5(h):
    """Return C5 = max_k ∫ h(x)(1-h(x+k)) dx for the discrete step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / n)


def compute_gradient_shift(h, k):
    """Gradient of C(k) = sum_i h_i(1-h_{i+k}) with respect to h, for a fixed shift k."""
    n = h.shape[0]
    grad = np.zeros(n, dtype=np.float64)
    if k >= 0:
        i_end = n - k
        # overlapping indices i = 0 .. i_end-1
        i = slice(0, i_end)
        grad[i] += 1.0 - h[i + k]          # derivative of linear part
        grad[i + k] += -h[i]                # derivative of quadratic part
    else:
        kp = -k
        i_start = kp
        # overlapping indices i = kp .. n-1
        i = slice(i_start, n)
        grad[i] += 1.0 - h[i + k]          # (i+k) is smaller index
        grad[i + k] += -h[i]
    return grad


def random_transfer(h, max_step):
    """Create a new vector by transferring a random amount from one index to another."""
    n = h.shape[0]
    i = np.random.randint(0, n)
    j = np.random.randint(0, n - 1)
    if j >= i:
        j += 1
    give = min(max_step, h[i])
    take = min(max_step, 1.0 - h[j])
    amt = min(give, take)
    if amt <= 0:
        return None, None
    delta = np.random.rand() * amt
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h, delta


def gradient_transfer(h, max_step, k):
    """Transfer mass from a positive‑gradient index to a negative‑gradient index for shift k."""
    grad = compute_gradient_shift(h, k)
    pos_mask = (grad > 1e-12) & (h > 1e-12)
    neg_mask = (grad < -1e-12) & (h < 1.0 - 1e-12)
    if not np.any(pos_mask) or not np.any(neg_mask):
        return None, None
    pos_idx = np.where(pos_mask)[0]
    neg_idx = np.where(neg_mask)[0]
    i = pos_idx[np.argmax(grad[pos_idx])]
    j = neg_idx[np.argmin(grad[neg_idx])]
    give = min(max_step, h[i])
    take = min(max_step, 1.0 - h[j])
    amt = min(give, take)
    if amt <= 0:
        return None, None
    delta = amt
    new_h = h.copy()
    new_h[i] -= delta
    new_h[j] += delta
    return new_h, delta
