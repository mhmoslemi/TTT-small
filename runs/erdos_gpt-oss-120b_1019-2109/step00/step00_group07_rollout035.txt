def project_to_constraints(v, target):
    # Clip to [0,1]
    v_clipped = np.clip(v, 0, 1)
    total = v_clipped.sum()
    if np.isclose(total, target):
        return v_clipped
    # Compute difference
    diff = target - total
    # We will adjust values that are not at bounds.
    # We'll distribute diff equally among free variables until they hit bounds.
    free = (v_clipped > 0) & (v_clipped < 1)
    # Use iterative water-filling:
    while abs(diff) > 1e-12:
        free_idxs = np.where((v_clipped > 0) & (v_clipped < 1))[0]
        if len(free_idxs) == 0:
            # add to all (should not happen)
            break
        # Amount to add per free variable
        add = diff / len(free_idxs)
        # Update
        v_clipped[free_idxs] += add
        # Clip again
        for idx in free_idxs:
            if v_clipped[idx] < 0:
                v_clipped[idx] = 0
            elif v_clipped[idx] > 1:
                v_clipped[idx] = 1
        total = v_clipped.sum()
        diff = target - total
        # break if maybe stuck
        if np.linalg.norm(add) < 1e-12:
            break
    return v_clipped

def build_h(n, L):
    # L in [0,1]
    if L <= 0:
        # constant 0.5
        return np.full(n, 0.5, dtype=float)
    H = 1.0 / (2.0 - L)
    a = 2.0 - 2.0 * L  # plateau width
    dx = 2.0 / n
    # compute x at midpoints: (np.arange(n) + 0.5) * dx
    xs = (np.arange(n) + 0.5) * dx
    h = np.empty(n, dtype=float)
    # left slope region where x < L
    mask_left = xs < L
    h[mask_left] = H * (xs[mask_left] / L)
    # plateau region where L <= x <= L + a
    mask_plateau = (xs >= L) & (xs <= L + a)
    h[mask_plateau] = H
    # right slope region > L + a
    mask_right = xs > L + a
    # For right region, distance to right end: 2 - x
    h[mask_right] = H * ((2.0 - xs[mask_right]) / L)
    return np.clip(h, 0.0, 1.0)

import time
import numpy as np

def _evaluate_c5(h):
    """Compute C5 using the same formula as the harness."""
    dx = 2.0 / h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_to_sum_box(v, target, iters=50):
    """
    Project vector v onto the set {0<=x_i<=1, sum(x)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0          # guarantees sum at lo >= target, at hi <= target
    for _ in range(iters):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def _build_h(n, L):
    """
    Symmetric piecewise‑linear function with slope length L (0≤L≤1).
    The function has height H = 1/(2-L) and a central plateau of width a = 2-2L.
    """
    if L <= 0.0:
        return np.full(n, 0.5, dtype=float)
    H = 1.0 / (2.0 - L)          # peak height
    a = 2.0 - 2.0 * L           # plateau width
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx   # midpoint sample points
    h = np.empty(n, dtype=float)

    left = xs < L
    h[left] = H * (xs[left] / L)

    plateau = (xs >= L) & (xs <= L + a)
    h[plateau] = H

    right = xs > L + a
    h[right] = H * ((2.0 - xs[right]) / L)

    # numerical safety
    return np.clip(h, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    """Find a step function h on [0,2] with small max overlap."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # 1. Determine problem size and a starting candidate
    # ------------------------------------------------------------------ #
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = int(h0.size)
        target = n / 2.0
        h0 = _project_to_sum_box(h0, target)
        best_h = h0.copy()
        best_val = _evaluate_c5(best_h)
    else:
        n = 200                     # a modest size, well below the 1k limit
        target = n / 2.0
        best_h = None
        best_val = np.inf

    # ------------------------------------------------------------------ #
    # 2. Simple 1‑parameter family (symmetric trapezoid) search
    # ------------------------------------------------------------------ #
    if best_h is None:            # no parent supplied – start from the family
        L_grid = np.linspace(0.0, 1.0, 101)
        for L in L_grid:
            h = _build_h(n, L)
            h = _project_to_sum_box(h, target)
            val = _evaluate_c5(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
                best_L = L

        # refine around the best L found
        for _ in range(5):
            step = 0.02 / (2 ** _)
            lo = max(0.0, best_L - step)
            hi = min(1.0, best_L + step)
            L_fine = np.arange(lo, hi + 1e-12, step / 10.0)
            for L in L_fine:
                h = _build_h(n, L)
                h = _project_to_sum_box(h, target)
                val = _evaluate_c5(h)
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
                    best_L = L

    # ------------------------------------------------------------------ #
    # 3. Short stochastic local search to polish the candidate
    # ------------------------------------------------------------------ #
    time_used = time.time() - start
    time_left = budget_s - time_used - 0.2   # tiny safety margin
    if time_left > 0.5:
        # simple hill‑climbing with Gaussian perturbations
        sigma_start, sigma_end = 0.2, 0.005
        iters = 0
        while time.time() - start < budget_s - 0.1:
            frac = (time.time() - start) / budget_s
            sigma = sigma_start * (1.0 - frac) + sigma_end * frac
            # perturb a random subset (≈10 % of entries) for efficiency
            idx = rng.choice(n, size=max(1, n // 10), replace=False)
            h_new = best_h.copy()
            delta = rng.normal(scale=sigma, size=idx.size)
            h_new[idx] += delta
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = _project_to_sum_box(h_new, target)
            val = _evaluate_c5(h_new)
            if val < best_val:
                best_val, best_h = val, h_new
            iters += 1
            if iters > 200000:      # safety abort
                break

    # final projection (ensures exact sum)
    best_h = _project_to_sum_box(best_h, n / 2.0)
    best_val = _evaluate_c5(best_h)

    return best_h.tolist(), float(best_val), int(n)
