def project_onto_box_simplex(v, target_sum):
    # Clip to [0,1] after subtracting lambda.
    # Use bisection to find lambda.
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0, 1)
        s = w.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - ((lo+hi)/2.0), 0, 1)
    return h

import time
import numpy as np

def _project_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Project v onto [0,1]^n with sum equal to target_sum."""
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    return h


def _c5_of(h: np.ndarray) -> float:
    """Maximum overlap value C5 for a given step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def _adjacent_sum(h: np.ndarray, offset: int) -> np.ndarray:
    """For a given shift offset, compute h[i+off] + h[i-off] (missing terms are 0)."""
    n = h.shape[0]
    out = np.zeros_like(h)
    if offset >= 0:
        out[: n - offset] += h[offset:]
        out[offset:] += h[: n - offset]
    else:
        d = -offset
        out[d:] += h[: n - d]
        out[: n - d] += h[d:]
    return out


def _random_feasible(n: int, target_sum: float, rng: np.random.RandomState) -> np.ndarray:
    """Generate a random vector in [0,1]^n with the prescribed sum."""
    v = rng.rand(n)
    return _project_box_sum(v, target_sum)


def _optimize_one_start(
    h0: np.ndarray,
    target_sum: float,
    max_iter: int,
    base_step: float,
    rng: np.random.RandomState,
    time_limit: float,
) -> tuple[np.ndarray, float]:

    n = h0.shape[0]
    dx = 2.0 / n
    cur = h0.copy()
    cur_val = _c5_of(cur)
    best_h, best_val = cur.copy(), cur_val
    start = time.time()

        if time.time() - start > time_limit:
            break

        corr = np.correlate(cur, 1.0 - cur, mode="full") * dx
        idx = int(np.argmax(corr))
        offset = idx - (n - 1)

        # sub‑gradient of the worst shift
        adj = _adjacent_sum(cur, offset)
        grad = adj - 1.0            # ∂C/∂h_i for the worst shift
        step = base_step * (1.0 - it / max_iter)

        # proposal and projection
        proposed = cur + step * grad
        new = _project_box_sum(proposed, target_sum)
        new_val = _c5_of(new)

        # accept monotone improvements
        if new_val <= cur_val:
            cur, cur_val = new, new_val
            if new_val < best_val:
                best_h, best_val = new.copy(), new_val

        # occasional tiny random jiggle to avoid stagnation
        if it % 200 == 0 and it > 0:
            i, j = rng.choice(n, 2, replace=False)
            if cur[i] < 1.0 and cur[j] > 0.0:
                eps = min(0.1, 1.0 - cur[i], cur[j]) * rng.rand()
                cur[i] += eps
                cur[j] -= eps
                cur = _project_box_sum(cur, target_sum)
                cur_val = _c5_of(cur)
                if cur_val < best_val:
                    best_h, best_val = cur.copy(), cur_val

    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.RandomState(seed)

    # ----------------------------------------------------
    # Determine problem size and a feasible starting vector
    # ----------------------------------------------------
    try:
        init_arr = np.array(initial_h_values, dtype=np.float64)  # provided by the harness
        n = init_arr.shape[0]
    except Exception:
        n = 180               # fall‑back size if no parent construction is present
        init_arr = None

    target = n / 2.0           # because ∫h = 1 and dx = 2/n → sum(h) = n/2

    # ensure feasibility of the parent vector (if present)
    if init_arr is not None:
        init_arr = _project_box_sum(init_arr, target)

    # ----------------------------------------------------
    # Optimization loop – several restarts within the time budget
    # ----------------------------------------------------
    best_h, best_c5 = None, np.inf
    start_time = time.time()
    max_iter = 5000
    base_step = 0.5

    # first try the parent solution (if any)
    candidates = []
    if init_arr is not None:
        candidates.append(init_arr)
    # always keep a random candidate ready
    candidates.append(_random_feasible(n, target, rng))

    while time.time() - start_time < budget_s - 4:
        if not candidates:
            candidates.append(_random_feasible(n, target, rng))
        h0 = candidates.pop(0)

        remaining = budget_s - (time.time() - start_time) - 2
        if remaining <= 0:
            break

        h_opt, c5_opt = _optimize_one_start(
            h0, target, max_iter, base_step, rng, remaining
        )
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt.copy(), c5_opt

        # push another random start for the next round
        candidates.append(_random_feasible(n, target, rng))

    # In the unlikely case nothing was found, fall back to a random feasible vector
    if best_h is None:
        best_h = _random_feasible(n, target, rng)
        best_c5 = _c5_of(best_h)

    # ----------------------------------------------------
    # Return result as required by the competition harness
    # ----------------------------------------------------
    return best_h.tolist(), float(best_c5), n
