def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx
```

def subgradient_step(h, shift, step):
    N = len(h)
    g = np.ones(N)
    if shift >= 0:
        g[:N-shift] -= h[shift:]
        g[shift:] -= h[:N-shift]
    else:
        s = -shift
        g[s:] -= h[:N-s]
        g[:N-s] -= h[s:]
    # update
    h_new = h - step * g
    # project
    h_new = project_feasible(h_new, target_sum=N/2)
    return h_new

                zeros = np.where(bc == 0)[0]
                choose = rng.choice(zeros, diff, replace=False)
                bc[choose] = 1.0
            elif diff < 0:
                ones = np.where(bc == 1)[0]
                choose = rng.choice(ones, -diff, replace=False)
                bc[choose] = 0.0
        # Run hill climbing
        final_h = hill_climb_binary(bc.copy())
        final_raw = compute_c5_raw(final_h)
        if final_raw < best_raw:
            best_h = final_h.copy()
            best_raw = final_raw

    # compute final bound
    c5 = best_raw * dx

    # Return solution as list (or numpy array)
    return (best_h.tolist(), float(c5), N)

import time, math
import numpy as np

def _project_feasible(x, target):
    """Project vector x onto [0,1]^n with sum == target via a simple bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(x - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)


def _max_corr_raw(h):
    """Raw maximum of the correlation used in the definition of C5."""
    return np.max(np.correlate(h, 1.0 - h, mode='full'))


def _hill_climb_binary(h, target_sum, rng):
    """
    Deterministic local improvement for a binary vector.
    Tries to eliminate mismatches at the worst shift.
    """
    n = h.shape[0]
    while True:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        cur_raw = float(corr[max_idx])
        shift = max_idx - (n - 1)

        # indices that contribute to the current worst value
        if shift >= 0:
            left = np.arange(n - shift)
            right = left + shift
        else:
            s = -shift
            left = np.arange(s, n)
            right = left - s

        bad = (h[left] > 0.5) & (h[right] < 0.5)
        bad_idx = left[bad]
        if bad_idx.size == 0:
            break  # nothing to improve

        # candidate zeros that would not increase the same shift if turned to 1
        zero_idxs = np.where(h < 0.5)[0]
        improved = False
        for i in bad_idx:
            # partner of i already caused a mismatch; turning it off removes one unit
            # look for a zero j whose partner is already a 1 (so the swap does not add a new mismatch)
            if shift >= 0:
                partner_ok = (j := zero_idxs + shift) < n
                good_j = zero_idxs[partner_ok & (h[j[partner_ok]] > 0.5)]
            else:
                s = -shift
                partner_ok = (j := zero_idxs - s) >= 0
                good_j = zero_idxs[partner_ok & (h[j[partner_ok]] > 0.5)]

            if good_j.size == 0:
                continue
            j = rng.choice(good_j)
            # perform the swap and see if it improves
            h[i], h[j] = 0.0, 1.0
            new_raw = _max_corr_raw(h)
            if new_raw < cur_raw - 1e-12:
                improved = True
                break
            # revert if no improvement
            h[i], h[j] = 1.0, 0.0

        if not improved:
            break
    # ensure exact sum (numerical issues)
    cur = h.sum()
    if cur != target_sum:
        diff = int(target_sum - cur)
        if diff > 0:
            zeros = np.where(h == 0)[0]
            h[rng.choice(zeros, diff, replace=False)] = 1.0
        elif diff < 0:
            ones = np.where(h == 1)[0]
            h[rng.choice(ones, -diff, replace=False)] = 0.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.time()

    # decide on the discretisation size
    if 'initial_h_values' in globals():
        init_arr = np.asarray(initial_h_values, dtype=np.float64)
        n_points = init_arr.size
    else:
        n_points = 70
        init_arr = None

    target_sum = n_points / 2.0
    dx = 2.0 / n_points

    # helper to keep the best solution seen so far
    best_h = None
    best_raw = np.inf

    def consider(candidate):
        nonlocal best_h, best_raw
        raw = _max_corr_raw(candidate)
        if raw < best_raw - 1e-12:
            best_raw = raw
            best_h = candidate.copy()

    # start from the provided construction if it exists and is feasible
    if init_arr is not None:
        h0 = np.clip(init_arr, 0.0, 1.0)
        if not np.isclose(h0.sum(), target_sum, atol=1e-9):
            h0 = _project_feasible(h0, target_sum)
        consider(h0)

    # utility to generate a random feasible binary vector
    def random_binary():
        h = np.zeros(n_points, dtype=np.float64)
        ones = rng.choice(n_points, n_points // 2, replace=False)
        h[ones] = 1.0
        return h

    # ---- simulated annealing on binary vectors ----
    cur_h = random_binary()
    cur_raw = _max_corr_raw(cur_h)
    consider(cur_h)

    T0 = 1.0
    max_no_improve = 200_000
    last_improve = 0
    it = 0
    while time.time() - start < budget_s * 0.95:
        it += 1
        # linear cooling based on elapsed portion
        elapsed_frac = (time.time() - start) / (budget_s * 0.95)
        T = T0 * max(0.001, 1.0 - elapsed_frac)

        # propose a swap of a 1 and a 0
        ones_idx = np.where(cur_h > 0.5)[0]
        zeros_idx = np.where(cur_h <= 0.5)[0]
        i = rng.choice(ones_idx)
        j = rng.choice(zeros_idx)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_raw = _max_corr_raw(cur_h)
        delta = new_raw - cur_raw
        accept = False
        if delta <= 0:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-delta / (T + 1e-12)):
                accept = True

        if accept:
            cur_raw = new_raw
            if new_raw < best_raw - 1e-12:
                best_raw = new_raw
                best_h = cur_h.copy()
                last_improve = it
        else:
            # revert the swap
            cur_h[i], cur_h[j] = 1.0, 0.0

        # occasional restart if stuck
        if it - last_improve > max_no_improve:
            cur_h = random_binary()
            cur_raw = _max_corr_raw(cur_h)
            last_improve = it

    # ---- deterministic hill‑climb on the best binary solution ----
    if best_h is not None:
        # make it strictly binary (0/1) before hill climbing
        binary_best = np.where(best_h > 0.5, 1.0, 0.0)
        # correct the total sum if rounding introduced a small error
        cur_sum = binary_best.sum()
        if cur_sum != target_sum:
            diff = int(target_sum - cur_sum)
            if diff > 0:
                zeros = np.where(binary_best == 0)[0]
                binary_best[rng.choice(zeros, diff, replace=False)] = 1.0
            elif diff < 0:
                ones = np.where(binary_best == 1)[0]
                binary_best[rng.choice(ones, -diff, replace=False)] = 0.0

        final_h = _hill_climb_binary(binary_best, target_sum, rng)
        final_raw = _max_corr_raw(final_h)
        if final_raw < best_raw - 1e-12:
            best_raw = final_raw
            best_h = final_h.copy()

    # final bound
    c5_bound = float(best_raw * dx)
    return (best_h.tolist(), c5_bound, n_points)
