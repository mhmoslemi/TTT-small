import numpy as np
import time
import random

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a discrete step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with its complement (1‑h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def thue_morse_seq(n: int) -> np.ndarray:
    """Binary Thue‑Morse sequence of length n (n may be any even number)."""
    return np.fromiter((bin(i).count("1") & 1 for i in range(n)),
                       dtype=np.int8, count=n)
