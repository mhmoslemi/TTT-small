def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    rev = np.flip(1 - h)
    L = 1 << ((2*n - 1).bit_length()) # next power of 2
    H = np.fft.fft(h, L)
    R = np.fft.fft(rev, L)
    conv = np.fft.ifft(H * R)[:2*n-1].real
    # Due to zero-padding, conv length = L but first 2n-1 entries correspond to linear convolution.
    max_val = conv.max()
    return max_val * dx

def propose(current, step):
    i, j = np.random.choice(n, 2, replace=False)
    max_delta = min(step, 1 - current[i], current[j]) # ensure i not exceed 1, j not go below 0
    min_delta = -min(step, current[i], 1 - current[j]) # delta negative allowed similarly.
    delta = np.random.uniform(min_delta, max_delta)
    new = current.copy()
    new[i] += delta
    new[j] -= delta
    return new

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # correlation of h with (1-h)
    rev = np.flip(1 - h)
    # convolution via FFT
    L = 1 << ((2*n - 1).bit_length())
    H = np.fft.rfft(h, n=L)
    R = np.fft.rfft(rev, n=L)
    conv = np.fft.irfft(H * R, n=L)[:2*n-1]
    max_val = conv.max()
    return max_val * dx

import numpy as np

def _project_onto_capped_simplex(v, target_sum):
    """Euclidean projection of v onto {0≤x≤1, sum(x)=target_sum}."""
    lo, hi = -1e5, 1e5          # wide enough interval for the Lagrange multiplier
    for _ in range(60):        # binary search – sufficient for double precision
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
