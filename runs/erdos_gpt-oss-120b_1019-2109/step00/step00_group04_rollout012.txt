import cvxpy as cp
import numpy as np
import time

def solve_for_n(n, time_limit=30):
   assert n%2==0
   x = cp.Variable(n, boolean=True)
   t = cp.Variable()
   constraints = [cp.sum(x) == n//2,
                  t >= 0]
   # define w_ℓ for ℓ=1..n-1
   for l in range(1, n):
        m = n - l
        w = cp.Variable(m, boolean=True)
        constraints += [w <= x[:m],
                        w <= x[l:],
                        w >= x[:m] + x[l:] - 1]
        # O_l
        O_l = cp.sum(x[:m]) - cp.sum(w)
        constraints.append(O_l <= t)
        # O_minus_l
        O_minus = cp.sum(x[l:]) - cp.sum(w)
        constraints.append(O_minus <= t)
   obj = cp.Minimize(t)
   prob = cp.Problem(obj, constraints)
   # Solve with CBC, with time limit:
   start = time.time()
   # Provide parameters to solver: if using 'CBC', we can specify maxSeconds:
   prob.solve(solver=cp.CBC, max_iters=1000000, warm_start=True)
   # Not sure about time limit; after solving, check elapsed.
   elapsed = time.time() - start
   # Return solution if feasible and status is optimal or optimal_inaccurate.
   if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None, elapsed
   h = np.array(x.value, dtype=float).round() (since binary)
   t_val = t.value
   return h, t_val

def solve_ilp(n, time_limit):
    import cvxpy as cp
    import numpy as np
    import time
    start = time.time()
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(x) == n//2, t >= 0]
    for shift in range(1, n):
        m = n - shift
        # product variables
        w = cp.Variable(m, boolean=True)
        # constraints linking w and x
        constraints.append(w <= x[:m])
        constraints.append(w <= x[shift:])
        constraints.append(w >= x[:m] + x[shift:] - 1)
        # O_ℓ positive shift
        O_plus = cp.sum(x[:m]) - cp.sum(w)
        constraints.append(O_plus <= t)
        # O_minus shift
        O_minus = cp.sum(x[shift:]) - cp.sum(w)
        constraints.append(O_minus <= t)
    obj = cp.Minimize(t)
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.GLPK_MI, glpk={'msg_lev': 'GLP_MSG_OFF'})
    except Exception as e:
        # fallback to CBC
        prob.solve(solver=cp.CBC, verbose=False)
    elapsed = time.time() - start
    if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        h = np.rint(x.value).astype(float)  # round to 0/1
        t_val = float(t.value)
        return h, t_val, elapsed
    else:
        return None, None, elapsed

def hill_climb(h, max_iters=10000, time_limit=60):
    '''
    h: numpy array with values 0 or 1 and sum = n/2
    Performs stochastic hill climbing using swap moves.
    Returns (h_best, bound_best)
    '''
    import numpy as np, random, time
    n = len(h)
    # indices of ones and zeros
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    best_h = h.copy()
    best_bound = compute_c5(best_h)
    start = time.time()
    # Simulated annealing parameters
    T0 = 1.0
    Tf = 0.001
    for iter in range(1, max_iters+1):
        if time.time() - start > time_limit:
            break
        # Temperature schedule
        T = T0 * ((Tf / T0) ** (iter / max_iters))
        # pick random swap
        i = random.choice(ones)
        j = random.choice(zeros)
        # propose swap
        h[i], h[j] = h[j], h[i]
        # compute bound
        cur_bound = compute_c5(h)
        # acceptance
        if cur_bound < best_bound - 1e-12 or random.random() < np.exp((best_bound - cur_bound) / T):
            # accept
            # update best if improved
            if cur_bound < best_bound - 1e-12:
                best_bound = cur_bound
                best_h = h.copy()
                # update indices
            # update one/zero lists after swap
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            # revert swap
            h[i], h[j] = h[j], h[i]
    return best_h, best_bound

import time, random, numpy as np, cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def solve_ilp(n):
    """MILP for a balanced binary step function of length n.
    Returns (h, t, elapsed) where h is a 0/1 array, t = max O_k, and elapsed time."""
    start = time.time()
    x = cp.Variable(n, boolean=True)
    t = cp.Variable()
    cons = [cp.sum(x) == n // 2, t >= 0]

    for shift in range(1, n):
        m = n - shift
        w = cp.Variable(m, boolean=True)          # product x[i]*x[i+shift]
        cons += [w <= x[:m],
                 w <= x[shift:],
                 w >= x[:m] + x[shift:] - 1]

        o_plus = cp.sum(x[:m]) - cp.sum(w)        # O_k for positive shift
        o_minus = cp.sum(x[shift:]) - cp.sum(w)   # O_k for negative shift
        cons += [o_plus <= t, o_minus <= t]

    prob = cp.Problem(cp.Minimize(t), cons)
    try:
        prob.solve(solver=cp.GLPK_MI, glpk={'msg_lev': 'GLP_MSG_OFF'})
    except Exception:
        prob.solve(solver=cp.CBC, verbose=False)

    elapsed = time.time() - start
    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None, elapsed
    h = np.rint(x.value).astype(float)
    return h, float(t.value), elapsed

        T = T0 * ((Tf / T0) ** (it / max_iters))
        i = random.choice(ones)
        j = random.choice(zeros)
        h[i], h[j] = h[j], h[i]                     # swap a 1 and a 0
        cur = compute_c5(h)
        improve = cur < best_score - 1e-12
        accept = improve or random.random() < np.exp((best_score - cur) / T)
        if accept:
            if improve:
                best_score = cur
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            h[i], h[j] = h[j], h[i]                 # undo
    return best_h, best_score

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    random.seed(seed)

    # ---- initial solution -------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        n0 = 50
        h0 = np.zeros(n0)
        h0[:n0 // 2] = 1
        np.random.shuffle(h0)
    best_h = h0
    best_n = len(best_h)
    best_bound = compute_c5(best_h)

    # ---- try a modest MILP (n = 56) ---------------------------------------
    if time.time() - start < budget_s * 0.90:
        cand_n = 56
        if cand_n != best_n:
            h_ilp, t_val, _ = solve_ilp(cand_n)
            if h_ilp is not None:
                b = compute_c5(h_ilp)
                if b < best_bound - 1e-6:
                    best_h, best_n, best_bound = h_ilp, cand_n, b

    # ---- local refinement if time remains ---------------------------------
    remaining = budget_s - (time.time() - start)
    if remaining > 5.0:
        # ensure exact half‑ones and 0/1 values
        h_bal = np.rint(best_h).astype(float)
        cur = int(np.sum(h_bal))
        target = best_n // 2
        if cur > target:
            idx = np.where(h_bal == 1)[0]
            flip = np.random.choice(idx, cur - target, replace=False)
            h_bal[flip] = 0
        elif cur < target:
            idx = np.where(h_bal == 0)[0]
            flip = np.random.choice(idx, target - cur, replace=False)
            h_bal[flip] = 1

        h_opt, b_opt = hill_climb(h_bal, max_iters=30000, time_limit=remaining)
        if b_opt < best_bound - 1e-9:
            best_h, best_bound = h_opt, b_opt
            best_n = len(best_h)

    # ---- final tidy‑up ----------------------------------------------------
    best_h = np.rint(best_h).astype(float)   # guarantee 0/1
    return best_h.tolist(), best_bound, best_n
