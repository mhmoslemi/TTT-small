            h[idx] += math.copysign(step, diff)
    target_sum = n / 2.0
    # enforce feasibility (box + sum)
    cur_sum = h.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        shift = (cur_sum - target_sum) / n
        h -= shift
        h = np.clip(h, 0.0, 1.0)
        err = target_sum - h.sum()
        mask = (h > 0) & (h < 1)
        if mask.any():
            h[mask] += err / mask.sum()
            h = np.clip(h, 0.0, 1.0)
        h *= target_sum / h.sum()
    cur_obj = _max_overlap(h)
    best_h = h.copy()
    best_obj = cur_obj
    start = time.time()
    max_iters = 2000000
    eps = 1e-12
    T0 = 0.002
    stale = 0
    for it in range(max_iters):
        if time.time() - start > budget_s * 0.95:
            break
        nloc = len(h)
        # current worst lag
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / nloc)
        idx = int(np.argmax(corr))
        lag = abs(idx - (nloc - 1))
        grad = _gradient(h, lag)
        pos = (grad > 0) & (h > eps)
        neg = (grad < 0) & (h < 1.0 - eps)
        if not pos.any() or not neg.any():
            # random tiny move
            i, j = random.sample(range(nloc), 2)
            if h[i] > eps and h[j] < 1.0 - eps:
                dm = min(h[i], 1.0 - h[j]) * random.random()
                h[i] -= dm
                h[j] += dm
                cur_obj = _max_overlap(h)
                if cur_obj < best_obj:
                    best_obj, best_h = cur_obj, h.copy()
            continue
        src = pos.nonzero()[0][int(np.argmax(grad[pos]))]
        tgt = neg.nonzero()[0][int(np.argmin(grad[neg]))]
        max_transfer = min(h[src] - eps, 1.0 - h[tgt] - eps)
        if max_transfer <= 0:
            continue
        delta = max_transfer * random.random()
        h[src] -= delta
        h[tgt] += delta
        new_obj = _max_overlap(h)
        if new_obj <= cur_obj:
            cur_obj = new_obj
            stale = 0
            if new_obj < best_obj:
                best_obj, best_h = new_obj, h.copy()
        else:
            T = T0 * (1.0 - it / max_iters)
            if T > 0 and random.random() < math.exp(-(new_obj - cur_obj) / T):
                cur_obj = new_obj
                stale = 0
                if new_obj < best_obj:
                    best_obj, best_h = new_obj, h.copy()
            else:
                h[src] += delta
                h[tgt] -= delta
                stale += 1
        if stale >= 2000:
            for _ in range(6):
                i, j = random.sample(range(nloc), 2)
                if h[i] > eps and h[j] < 1.0 - eps:
                    dm = min(h[i], 1.0 - h[j]) * random.random() * 0.1
                    h[i] -= dm
                    h[j] += dm
            cur_obj = _max_overlap(h)
            if cur_obj < best_obj:
                best_obj, best_h = cur_obj, h.copy()
            stale = 0
        if best_obj <= 0.3808:
            break
    best_h = np.clip(best_h, 0.0, 1.0)
    # final tiny correction of sum
    diff = best_h.sum() - target_sum
    if abs(diff) > 1e-9:
        mask = (best_h > 0) & (best_h < 1)
        if mask.any():
            best_h[mask] -= diff / mask.sum()
            best_h = np.clip(best_h, 0.0, 1.0)
    final_val = _max_overlap(best_h)
    return (best_h.tolist(), float(final_val), int(n))
