import numpy as np, time, random, math

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / n)

def propose_move(h, eps):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]
    max_inc_i = 1.0 - hi
    max_dec_i = hi
    max_inc_j = 1.0 - hj
    max_dec_j = hj
    low = -min(max_dec_i, max_inc_j)
    high = min(max_inc_i, max_dec_j)
    if low >= high:
        return None, None, None
    # caps by eps
    low = max(low, -eps)
    high = min(high, eps)
    if low >= high:
        return None, None, None
    delta = np.random.uniform(low, high)
    return i, j, delta

def random_feasible_h(n):

def simulated_annealing(h0, max_iter, time_limit, start_time):
    n = len(h0)
    h = h0.copy()
    cur_f = compute_c5(h)
    best_h = h.copy()
    best_f = cur_f
    T0 = 0.02
    eps0 = 0.5
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        elapsed = time.time() - start_time
        fraction = elapsed / time_limit
        T = T0 * (1.0 - fraction) + 1e-9
        eps = eps0 * (1.0 - fraction) + 1e-9
        i, j, delta = propose_move(h, eps)
        if i is None:
            continue
        # apply move
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # ensure bounds (should be by propose)
        new_f = compute_c5(h_new)
        dF = new_f - cur_f
        if dF <= 0 or random.random() < math.exp(-dF / T):
            h = h_new
            cur_f = new_f
            if new_f < best_f:
                best_f = new_f
                best_h = h.copy()
    return best_h, best_f

def _c5_value(h: np.ndarray) -> float:
    """Return the overlap constant for a step function h."""
    n = h.shape[0]
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)


def _propose_move(h: np.ndarray, eps: float):
    """Choose two indices and a delta that keeps h in [0,1] and preserves the sum."""
    n = h.shape[0]
    i, j = np.random.choice(n, 2, replace=False)
    hi, hj = h[i], h[j]

    max_inc_i = 1.0 - hi
    max_dec_i = hi
    max_inc_j = 1.0 - hj
    max_dec_j = hj

    low = -min(max_dec_i, max_inc_j)   # negative delta: i decreases, j increases
    high = min(max_inc_i, max_dec_j)   # positive delta: i increases, j decreases
    if low >= high:
        return None, None, None

    low = max(low, -eps)
    high = min(high, eps)

        return None, None, None

            np.clip(h0, 0.0, 1.0, out=h0)
            for _ in range(10):
                cur_sum = h0.sum()
                diff = target_sum - cur_sum
                if abs(diff) < 1e-12:
                    break
                mask = (h0 > 1e-12) & (h0 < 1.0 - 1e-12)
                free = mask.sum()
                if free == 0:
                    break
                step = diff / free
                h0[mask] += step
                np.clip(h0, 0.0, 1.0, out=h0)

        # Allocate a slice of the remaining budget to this run.
        deadline = start + remaining * 0.9  # keep a tiny safety margin
        h_opt, f_opt = _simulated_annealing(h0, deadline, start)

        if f_opt < best_f:
            best_f, best_h, best_n = f_opt, h_opt.copy(), n

        # If time is still ample, try a finer grid based on the current best.
        now = time.time()
        remaining = budget_s - (now - start)
        if remaining > 5.0 and best_n * 2 <= 200:
            # generate a doubled‑resolution version of the current best.
            candidates.append((best_n * 2, np.repeat(best_h, 2)))

    # ------------------------------------------------------------------
    # 5.  Final safety projection and return.
    # ------------------------------------------------------------------
    if best_h is None:                     # extremely unlikely fallback
        n = 60
        best_h = np.zeros(n)
        best_h[: n // 2] = 1.0
        best_n = n
        best_f = _c5_value(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    # final exact sum adjustment (tiny)
    diff = best_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        best_h += diff / best_n
        np.clip(best_h, 0.0, 1.0, out=best_h)

    return best_h.tolist(), float(best_f), int(best_n)
