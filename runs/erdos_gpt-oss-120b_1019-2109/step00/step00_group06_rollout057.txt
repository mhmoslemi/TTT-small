import time
import numpy as np
from scipy.optimize import minimize

_alpha = None   # smoothness for max‑approximation (global for callbacks)
_target = None  # required sum of h values (global for callbacks)


def evaluate_c5(vec: np.ndarray) -> float:
    """Exact C5 value for a step vector."""
    n = len(vec)
    dx = 2.0 / n
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return np.max(corr) * dx


def _corr_raw(vec: np.ndarray) -> np.ndarray:
    """Correlation without the dx factor (used inside the optimiser)."""
    return np.correlate(vec, 1.0 - vec, mode="full")


def softmax_objective(vec: np.ndarray, alpha: float) -> float:
    """Smooth approximation of the max overlap."""
    n = len(vec)
    dx = 2.0 / n
    c = _corr_raw(vec)
    scaled = alpha * c
    m = np.max(scaled)
    # stable softmax
    exp_scaled = np.exp(scaled - m)
    sum_exp = np.sum(exp_scaled)
    approx = (np.log(sum_exp) + m) / alpha
    return approx * dx


def softmax_gradient(vec: np.ndarray, alpha: float) -> np.ndarray:
    """Gradient of the soft‑max objective."""
    n = len(vec)
    dx = 2.0 / n
    c = _corr_raw(vec)
    scaled = alpha * c
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    sum_exp = np.sum(exp_scaled)
    w = exp_scaled / sum_exp                     # weights for every lag
    grad = np.zeros(n, dtype=np.float64)

    # each possible lag contributes with weight w[idx]
    for idx, wi in enumerate(w):
        if wi == 0.0:
            continue
        shift = idx - (n - 1)                    # integer lag (negative allowed)
        if shift >= 0:
            s = shift
            if s < n:
                grad[: n - s] += wi * (1.0 - vec[s:])
                grad[s:]      -= wi * vec[: n - s]
        else:
            p = -shift
            if p < n:
                grad[p:]      += wi * (1.0 - vec[: n - p])
                grad[: n - p] -= wi * vec[p:]
    return grad * dx


def objective_global(vec: np.ndarray) -> float:
    """Wrapper for scipy – uses the globally set _alpha."""
    return softmax_objective(vec, _alpha)


def gradient_global(vec: np.ndarray) -> np.ndarray:
    """Wrapper for scipy – uses the globally set _alpha."""
    return softmax_gradient(vec, _alpha)


def eq_constraint(vec: np.ndarray) -> float:
    """Equality constraint: sum(vec) must equal the target."""
    return np.sum(vec) - _target


def random_initial_vector(n: int, target: float) -> np.ndarray:
    """
    Produce a feasible start: a binary pattern with the required sum.
    For odd n the fractional remainder (always 0.5) is placed on a random
    zero position.
    """
    vec = np.zeros(n, dtype=np.float64)
    whole = int(np.floor(target))
    idx = np.random.choice(n, whole, replace=False)
    vec[idx] = 1.0
    rest = target - whole
    if rest > 1e-12:
        zero = np.where(vec == 0.0)[0]
        if zero.size:
            vec[np.random.choice(zero)] = rest
    return vec

    Optimise a step function h : [0,2] → [0,1] with ∫h = 1.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)

    # -------------------------------------------------

    if "initial_h_values" in globals():

    else:
        # fall back to a binary construction of length 120
        n0 = 120
        h0 = random_initial_vector(n0, n0 / 2.0)

    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n

    # store globals for the optimiser
    global _alpha, _target
    _alpha = 5.0               # smoothness parameter for the max‑approximation
    _target = target

    best_vec = h0.copy()
    best_val = evaluate_c5(best_vec)

        cand[i] += delta
        cand[j] -= delta
        cand_val = evaluate_c5(cand)

            if cand_val < best_val:
