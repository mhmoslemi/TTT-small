import time
import numpy as np
import math

# evaluate_erdos_solution is supplied in the execution environment
# initial_h_values is supplied when a parent solution exists


def _c5_value(h):
    """Return the C5 value for a candidate h using the provided evaluator."""
    # The evaluator may raise if constraints are violated; we assume h is feasible.
    return evaluate_erdos_solution(h)


def _random_pair_move(h, step):
    """Propose a new h by moving a random amount between two coordinates."""
    n = h.shape[0]
    i, j = np.random.randint(0, n, size=2)
    if i == j:
        return None, None, None
    delta = (np.random.rand() - 0.5) * step
    hi = h[i] + delta
    hj = h[j] - delta
    if hi < 0.0 or hi > 1.0 or hj < 0.0 or hj > 1.0:
        return None, None, None
    cand = h.copy()
    cand[i] = hi
    cand[j] = hj
    return cand, i, j


def _initial_candidate(n):
    """Construct a simple smooth feasible candidate (sin^2 shape)."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx          # mid‑point samples
    h = np.sin(np.pi * x / 2.0) ** 2        # sin^2(πx/2) integrates to 1 over [0,2]
    # Adjust for tiny discretisation error
    total = h.sum() * dx
    if abs(total - 1.0) > 1e-12:
        h *= (1.0 / total)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """
    Return a step‑function h (as a list), its C5 bound, and the number of points.
    Heuristic optimisation based on simulated‑annealing / pairwise swaps.
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----------------------------------------------------------------------
    # Choose discretisation size
    # ----------------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.shape[0]
    else:
        n = 200                     # a comfortable size when no parent is given
        h0 = _initial_candidate(n)

    # Ensure feasibility of the starting point (may be unnecessary but safe)
    dx = 2.0 / n
    cur = h0.copy()
    cur = np.clip(cur, 0.0, 1.0)
    # Project onto the hyperplane sum(h)=n/2 while staying inside bounds
    target_sum = n / 2.0
    cur_sum = cur.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # Simple correction: add the same offset to all interior points,
        # then clip again – the resulting sum is very close to the target.
        offset = (target_sum - cur_sum) / n
        cur += offset
        cur = np.clip(cur, 0.0, 1.0)

    cur_val = _c5_value(cur)

    # Keep the best ever seen
    best_h = cur.copy()
    best_val = cur_val

    # ----------------------------------------------------------------------
    # Simulated annealing parameters
    # ----------------------------------------------------------------------
    max_iter = 600000                     # generous upper bound

        if time.time() - start_time > budget_s * 0.95:

        # linear temperature schedule

        T = T0 * (1 - t) + T_min * t
        step = max_step * (T / T0)  # shrink step with temperature
