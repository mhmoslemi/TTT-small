def project_box_sum(v, target, tol=1e-8, max_iter=100):
    lo = -np.max(v) - 1.0  # low bound for lambda: enough negative to push all to 0.
    hi = 1.0 - np.min(v) + 1.0  # high bound for lambda: push all to 1.
    # Actually range should guarantee sum after clipping is monotonic: At lambda very low, all v_i+lambda <= 0 => x_i =0 => sum=0 < target. At lambda very high, x_i=1 => sum=n > target.
    # So we can use lo=-2, hi=2, but safer: -2,2.
    lo = -2.0
    hi = 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < tol:
            return x
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

import time, numpy as np

def project_box_sum(v, target, tol=1e-9, max_iter=100):
    lo = -2.0
    hi = 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) < tol:
            return x
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo+hi)/2, 0.0, 1.0)

def compute_obj_and_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    C0 = h.sum()
    conv = np.correlate(h, h, mode='full')
    O = dx * (C0 - conv)  # shape 2n-1, shift index from -(n-1) to (n-1)
    # Softmax for max approximation
    max_O = O.max()
    # For numerical stability
    exp_alpha = np.exp(alpha * (O - max_O))
    w = exp_alpha / exp_alpha.sum()
    # Compute gradient
    grad = np.zeros(n)
    # Precompute 1 - shifted h for each shift: we can compute via slices.
    # Loop over shifts
    for shift_idx, wj in enumerate(w):
        if wj == 0:
            continue
        # shift = shift_idx - (n-1)
        shift = shift_idx - (n - 1)
        if shift >= 0:
            i_end = n - shift
            if i_end <= 0:
                continue
            # slices
            i_slice = slice(0, i_end)
            j_slice = slice(shift, n)
            # Add contributions
            grad[i_slice] += dx * (1.0 - h[j_slice]) * wj
            grad[j_slice] -= dx * h[i_slice] * wj
        else:
            s = -shift
            i_start = s
            i_end = n
            # i from s..n-1, i+shift = i-s from 0..n-s-1
            i_slice = slice(i_start, i_end)
            j_slice = slice(0, n - s)
            grad[i_slice] += dx * (1.0 - h[j_slice]) * wj
            grad[j_slice] -= dx * h[i_slice] * wj
    # The gradient we computed corresponds to derivative of approximated objective F (softmax). Good.
    # Return max O as objective (actual max) also.
    obj = O.max()
    return obj, grad

def project_box_sum(v, target, tol=1e-8, max_iter=50):
    lo = -2.0
    hi = 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s < target:
            lo = mid
        else:
            hi = mid
    # Return after loop
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

import time
import numpy as np

def _project_box_sum(v: np.ndarray, target: float, tol: float = 1e-9, max_iter: int = 60) -> np.ndarray:
    """Project v onto {x | 0<=x_i<=1, sum(x)=target} using a simple bisection on the shift λ."""
    lo, hi = -5.0, 5.0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if abs(s - target) <= tol:
            return x
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


def _objective_and_grad(h: np.ndarray, alpha: float):
    """Return (max_overlap, gradient) for the smoothed problem with parameter alpha."""
    n = h.size
    dx = 2.0 / n
    # convolution h*h (full) gives inner products for all shifts
    conv = np.correlate(h, h, mode="full")
    C0 = h.sum()
    O = dx * (C0 - conv)                     # overlap for every shift (length 2n-1)

    # soft‑max weights approximating the max
    max_O = O.max()
    w_raw = np.exp(alpha * (O - max_O))
    w = w_raw / w_raw.sum()                  # sum(w)=1, length 2n-1

    # gradient: dx * (1 - Σ_j w_j h_{i+j} - Σ_j w_j h_{i-j})
    # first term a_i = Σ_j w_j h_{i+j}
    corr1 = np.correlate(w, h, mode="full")
    base = w.size - 1
    a = corr1[base:base + n]
    # second term b_i = Σ_j w_j h_{i-j}  = correlation with reversed w
    corr2 = np.correlate(w[::-1], h, mode="full")
    b = corr2[base:base + n]

    grad = dx * (1.0 - a - b)
    return O.max(), grad

    """Return a step function h, its C5 bound, and the number of samples."""
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s * 0.95  # safety margin

    # ------------------------------------------------------------------ #

    if "initial_h_values" in globals():

        h = rng.random(n0)
    n = h.size
    target_sum = n / 2.0
    h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1 - h, mode="full") * (2.0 / n))

    # simple schedule of smoothing parameters
    alphas = [2.0, 5.0, 10.0, 20.0, 40.0]
    base_lr = 0.5

    # ------------------------------------------------------------------ #
    # main optimisation loop
    while time.time() < deadline:
        # use a copy so we can revert if needed
        cur_h = h.copy()
        cur_n = cur_h.size
        cur_target = cur_n / 2.0
        dx = 2.0 / cur_n

        # inner iterations
        for alpha in alphas:
            for _ in range(30):
                if time.time() >= deadline:
                    break

                # evaluate objective and gradient
                cur_val, grad = _objective_and_grad(cur_h, alpha)

                # line search (back‑tracking)
                improved = False
                for factor in (1.0, 0.5, 0.25, 0.125):
                    step = base_lr * factor
                    cand = cur_h - step * grad
                    cand = _project_box_sum(cand, cur_target)
                    cand_val = np.max(np.correlate(cand, 1 - cand, mode="full") * dx)
                    if cand_val < cur_val - 1e-12:
                        cur_h = cand
                        cur_val = cand_val
                        improved = True
                        break

                if not improved:
