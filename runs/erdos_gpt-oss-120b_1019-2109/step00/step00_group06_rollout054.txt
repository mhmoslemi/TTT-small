During gradient calculation for shift k (positive maybe). For index i, we need to consider contributions from term where i is the "source" index (i in sum). Also contributions where i appears as the "target" index in the product for shift -k (i.e., j such that j+k = i). So gradient includes both.

Hence implement:

def gradient_of_O_k(h, k):
    N = len(h)
    dx = 2.0 / N
    g = np.zeros(N)
    if k >= 0:
        # source part: i from 0 to N-1-k
        g[:N-k] += dx * (1 - h[k:])   # (1 - h_{i+k})
        # target part: i from k to N-1   i is i such that (i - k) >=0
        g[k:] -= dx * h[:N-k]   # -h_{i-k}
    else:   # k negative
        kk = -k
        # for negative shift, rewrite as analogous
        # O(k) = Σ_i h_i * (1 - h_{i+k}) where i+k can be negative => out-of-range means term omitted
        # Overlap region i from kk .. N-1 (since i+k >=0)
        g[kk:] += dx * (1 - h[:N-kk])   # (1 - h_{i+k}) where i+k = i - kk => index i - kk
        # target part: i from 0 to N-1+ k = N-1-kk   i such that i - k = i+kk in range.
        g[:N-kk] -= dx * h[kk:]   # -h_{i - k}
    return g

def smooth_max_and_gradient(h, alpha, penalty):
    N = len(h)
    dx = 2.0 / N
    # O vector
    O = np.correlate(h, 1 - h, mode='full') * dx
    # find max for numerical stability
    max_O = O.max()
    # compute softmax weights
    exp_scaled = np.exp(alpha * (O - max_O))
    sum_exp = exp_scaled.sum()
    # f_smooth = max_O + log(sum_exp)/alpha
    f = max_O + np.log(sum_exp) / alpha

    # compute weighted gradient
    grad = np.zeros(N)
    # mapping indices to shift s
    # For idx in [0, 2N-2], shift s = idx - (N-1)
    N1 = N - 1
    # loop
    for idx in range(2*N - 1):
        w = exp_scaled[idx] / sum_exp
        if w == 0:
            continue
        s = idx - N1
        # compute gradient for shift s
        if s >= 0:
            # source part
            grad[:N-s] += w * dx * (1 - h[s:])
            # target part
            grad[s:] += -w * dx * h[:N-s]
        else:
            ss = -s
            grad[ss:] += w * dx * (1 - h[:N-ss])
            grad[:N-ss] += -w * dx * h[ss:]
    # penalty term for sum constraint
    K = N / 2
    diff = h.sum() - K
    f += penalty * diff * diff
    grad += 2 * penalty * diff
    return f, grad

def local_search(h, max_iter=500):
    N = len(h)
    best = h.copy()
    best_val = compute_c5(best)
    for _ in range(max_iter):
        # random pair i, j
        i, j = np.random.randint(N), np.random.randint(N)
        # propose small change epsilon
        eps = np.random.uniform(-0.1, 0.1)
        # modify h_i and h_j while maintaining sum
        # For simplicity, perform swap of values between i and j (exchange)
        # If values same - no effect.
        # else: we can adjust: new_h_i = h_i + eps; new_h_j = h_j - eps, clip and project.
        new_h = best.copy()
        # apply eps to i and -eps to j, clip to [0,1], and adjust accordingly
        delta = eps
        # enforce bounds: highest delta that keeps h_i+delta in [0,1] and h_j-delta in [0,1]
        max_delta = min(1 - new_h[i], new_h[j])
        min_delta = -min(new_h[i], 1 - new_h[j])
        # clamp delta
        delta = np.clip(delta, min_delta, max_delta)
        new_h[i] += delta
        new_h[j] -= delta
        # sum unchanged, bounds satisfied.
        val = compute_c5(new_h)
        if val < best_val - 1e-6:
            best = new_h
            best_val = val

import numpy as np
import time

# ------------------------------------------------------------
# Helper functions
# ------------------------------------------------------------
def compute_c5(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx for discrete h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx


def project_onto_bounds_sum(v: np.ndarray, target: float, tol: float = 1e-9,
                            max_iter: int = 60) -> np.ndarray:
    """
    Euclidean projection of v onto the set
        {x | 0 ≤ x_i ≤ 1, Σ x_i = target }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0      # low enough -> all entries clipped to 1
    hi = np.max(v) + 1.0      # high enough -> all entries clipped to 0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - lo, 0.0, 1.0)
    # tiny correction for round‑off
    diff = proj.sum() - target
    if abs(diff) > tol:
        free = (proj > 0.0) & (proj < 1.0)
        if free.any():
            proj[free] -= diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj


def subgradient_descent(h0: np.ndarray, max_iter: int,
                       step0: float = 5.0, decay: float = 0.0005) -> np.ndarray:
    """
    Projected sub‑gradient descent on
        f(h) = max_k ∫ h(1‑h shifted) .
    Only the worst shift (the sub‑gradient) is used.
    """
    n = h0.shape[0]
    dx = 2.0 / n
    target = n * 0.5
    h = h0.copy()
    for t in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)                     # shift value (can be negative)

        grad = np.zeros_like(h)
        if shift >= 0:
            grad[: n - shift] += dx * (1.0 - h[shift:])
            grad[shift:]      -= dx * h[: n - shift]
        else:
            ss = -shift
            grad[ss:]        += dx * (1.0 - h[: n - ss])
            grad[: n - ss]   -= dx * h[ss:]

        step = step0 / (1.0 + decay * t)
        h -= step * grad
        h = project_onto_bounds_sum(h, target)
    return h
