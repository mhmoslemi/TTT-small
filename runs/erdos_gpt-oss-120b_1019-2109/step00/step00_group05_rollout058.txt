import time, numpy as np

def compute_corr_raw(h):
    return np.correlate(h, 1-h, mode='full')

def compute_c5(h, dx):
    return np.max(compute_corr_raw(h)) * dx

def max_corr_shift(h):
    corr = compute_corr_raw(h)
    idx = np.argmax(corr)
    max_corr = corr[idx]
    shift = idx - (len(h) - 1)
    return max_corr, shift

def random_binary(n, target_sum):
    # generate vector of length n with exactly target sum ones.
    # choose target_sum positions randomly.
    h = np.zeros(n, dtype=np.int8)
    ones = np.random.choice(n, size=target_sum, replace=False)
    h[ones] = 1
    return h
```

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    try:
        init = np.array(initial_h_values, copy=False)
        n = init.shape[0]
    except NameError:
        n = 70
        init = None
    target_sum = n // 2  # because integral 1 -> sum = n/2

    if init is not None:
        # rounding to binary while preserving sum
        h = np.rint(init).astype(np.int8)
        # ensure sum correct
        diff = target_sum - h.sum()
        if diff > 0:
            zeros = np.where(h == 0)[0]
            chosen = rng.choice(zeros, size=diff, replace=False)
            h[chosen] = 1
        elif diff < 0:
            ones = np.where(h == 1)[0]
            chosen = rng.choice(ones, size=-diff, replace=False)
            h[chosen] = 0
    else:
        h = random_binary(n, target_sum)

    dx = 2.0 / n
    cur_raw_corr = compute_corr_raw(h)
    cur_max_raw = cur_raw_corr.max()
    best_h = h.copy()
    best_raw = cur_max_raw
    start = time.time()
    # main loop
    no_improve = 0
    # parameters
    max_no_improve_before_restart = max(5000, n*100)
    # loops
    while True:
        elapsed = time.time() - start
        if elapsed > budget_s * 0.95:
            break

        # compute worst shift
        max_idx = np.argmax(cur_raw_corr)
        shift = max_idx - (n-1)
        # find mismatched indices
        if shift >= 0:
            start_idx = 0
            end_idx = n - shift - 1
        else:
            start_idx = -shift
            end_idx = n - 1
        region = np.arange(start_idx, end_idx+1)
        # h region values and shifted region
        h_region = h[region]
        h_shifted = h[region + shift]
        mis_i = region[(h_region == 1) & (h_shifted == 0)]
        mis_j = region[(h_region == 0) & (h_shifted == 1)]

        # generate candidates
        candidates = []
        if mis_i.size and mis_j.size:
            # choose random pair from mis_i, mis_j
            i = rng.choice(mis_i)
            j = rng.choice(mis_j)
            candidates.append((i, j))
        if mis_i.size:
            # choose zero outside region
            outside_mask = np.ones(n, dtype=bool)
            outside_mask[region] = False
            zeros_out = np.where(outside_mask & (h == 0))[0]
            if zeros_out.size:
                i = rng.choice(mis_i)
                j = rng.choice(zeros_out)
                candidates.append((i, j))
        # random 1-0 swap as fallback
        if len(candidates) < 2:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if ones.size and zeros.size:
                i = rng.choice(ones)
                j = rng.choice(zeros)
                candidates.append((i, j))

        improved = False
        for i, j in candidates:
            # perform swap on copy
            h[i], h[j] = h[j], h[i]
            new_raw_corr = np.correlate(h, 1-h, mode='full')
            new_max_raw = new_raw_corr.max()
            if new_max_raw <= cur_max_raw:
                # accept
                cur_raw_corr = new_raw_corr
                cur_max_raw = new_max_raw
                improved = True
                if new_max_raw < best_raw:
                    best_raw = new_max_raw
                    best_h = h.copy()
                break
            else:
                # revert swap
                h[i], h[j] = h[j], h[i]

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

        # restart if stuck
        if no_improve >= max_no_improve_before_restart:
            # random restart
            h = random_binary(n, target_sum)
            cur_raw_corr = np.correlate(h, 1-h, mode='full')
            cur_max_raw = cur_raw_corr.max()
            no_improve = 0

    c5 = best_raw * dx
    return (best_h.astype(float), float(c5), n)

import time
import numpy as np

def _corr_raw(h):
    """Full correlation of h with (1‑h), returns raw (not scaled) values."""
    return np.correlate(h, 1.0 - h, mode='full')

def _max_corr_and_shift(h):
    """Maximum correlation value and the corresponding integer shift."""
    c = _corr_raw(h)
    idx = int(np.argmax(c))
    max_val = c[idx]
    shift = idx - (len(h) - 1)
    return max_val, shift, c

def _random_binary(rng, n, target):
    """Generate a binary vector of length n with exactly `target` ones."""
    h = np.zeros(n, dtype=np.int8)
    ones = rng.choice(n, size=target, replace=False)
    h[ones] = 1
    return h

    dx = 2.0 / n

    if init is not None:

        h = np.rint(init).astype(np.int8)
        diff = target_sum - int(h.sum())
        if diff > 0:
            zeros = np.where(h == 0)[0]
            pick = rng.choice(zeros, size=diff, replace=False)
            h[pick] = 1
        elif diff < 0:

            h[pick] = 0
    else:

    # ------------------------------------------------------------------

    cur_max, cur_shift, cur_corr = _max_corr_and_shift(h)

        shift = cur_shift
        if shift >= 0:

            end_idx = n - shift - 1
        else:
            start_idx = -shift
            end_idx = n - 1
        region = np.arange(start_idx, end_idx + 1)

        h_reg = h[region]
        h_shifted = h[region + shift]
        mis_one = region[(h_reg == 1) & (h_shifted == 0)]   # 1 that sees a 0

        if mis_one.size:
