def project_to_box_and_sum(v, target):
    # returns w ∈ [0,1]^n with sum target.
    lo = -1.0
    hi = 1.0
    for _ in range(30):
        mid = (lo + hi) / 2
        w = np.clip(v + mid, 0, 1)
        s = w.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    w = np.clip(v + lo, 0, 1)
    # after converging, due to numerical may be slightly off; adjust by distributing diff over non-bound entries.
    diff = target - w.sum()
    # If diff is small, ignore.
    if abs(diff) > 1e-9:
        # adjust free entries (those not at bounds) by adding diff/(free count)
        free = (w > 0) & (w < 1)
        cnt = free.sum()
        if cnt > 0:
            w[free] += diff / cnt
            w = np.clip(w, 0, 1)
    return w

def _project_box_sum(v, target):
    # Project onto [0,1]^n and sum = target using bisection on additive shift.
    lo, hi = -2.0, 2.0

        cnt = int(mask.sum())

            w[mask] += diff / cnt
            w = np.clip(w, 0.0, 1.0)
    return w
