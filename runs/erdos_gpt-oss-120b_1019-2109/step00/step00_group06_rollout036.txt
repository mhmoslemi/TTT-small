def project_to_sum_and_bounds(x, target):
    # Clip
    x = np.clip(x, 0.0, 1.0)
    # Compute diff
    diff = target - x.sum()
    # iterative redistribute among non-bound entries

            break
        adjust = diff / count
        # Apply adjust
        x[free] += adjust
        # Clip again
        x = np.clip(x, 0.0, 1.0)

        # If diff not changed significantly, break to avoid loop
        if abs(new_diff - diff) < 1e-12:
            break

    return x

def project_to_sum_and_bounds(v, target):
    # Clip
    v = np.clip(v, 0.0, 1.0)
    # compute excess
    total = v.sum()
    diff = target - total
    # iterative adjust by distributing diff equally among free entries
    # free = (v > 0 + 1e-12) & (v < 1 - 1e-12)
    # If diff >0 need to increase some free; if diff <0 need to decrease.
    # we do loop:
    for _ in range(10):
        free = (v > 1e-12) & (v < 1 - 1e-12)
        cnt = free.sum()
        if cnt == 0: break
        delta = diff / cnt
        v[free] += delta
        # Clip again
        v = np.clip(v, 0.0, 1.0)
        new_total = v.sum()
        diff = target - new_total
        if abs(diff) < 1e-12:
            break
    # If still diff not zero due to all at bounds, we may adjust distributed among all entries with possible rounding
    # Could simply adjust the first element within bounds.
    if abs(v.sum() - target) > 1e-8:
        # adjust any element not at bound

            if 0 < v[i] < 1:
                needed = target - v.sum()
                v[i] = np.clip(v[i] + needed, 0, 1)
                break
    return v

def project_box_sum(v, target, tol=1e-9, max_iter=100):
    lo = -2.0  # lower bound: because subtracting -2 adds 2, so clip becomes 1.
    hi = 2.0   # upper bound: subtracting 2 reduces to -2, clip 0.
    # Ensure lo and hi bracket root
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < tol:
            break
    return proj

import time
import math
import numpy as np

def _compute_c5(h):
    """return max_k ∫ h(x)(1-h(x+k))dx using the discrete definition."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_box_sum(v, target, tol=1e-12, max_iter=60):
    """
    L2‑projection of v onto the set {x | 0≤x_i≤1, Σx_i = target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo, hi = -2.0, 2.0          # enough to drive all entries to the bounds
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) <= tol:
            break
    return proj


def _gradient_softmax_full(h, beta):
    """
    Gradient of  (1/beta)·log Σ exp(beta·C_s)  w.r.t. h,
    where C_s = Σ_i h_i (1 - h_{i+s}) for shift s∈[-(n-1),…,n-1].
    """
    n = h.shape[0]
    c_full = np.correlate(h, 1.0 - h, mode="full")
    max_c = c_full.max()
    weights = np.exp(beta * (c_full - max_c))
    weights /= weights.sum()

    grad = np.zeros_like(h)
    mid = n - 1                     # index of shift 0
    for offset in range(-mid, mid + 1):
        w = weights[offset + mid]
        if w == 0.0:
            continue
        if offset >= 0:
            s = offset
            # ∂C/∂h_i = (1 - h_{i+s}) for i≤n-1-s  and  -h_{i-s} for i≥s
            grad[: n - s] += w * (1.0 - h[s:])
            grad[s:]      -= w * h[: n - s]
        else:
            k = -offset
            grad[k:]      += w * (1.0 - h[: n - k])
            grad[: n - k] -= w * h[k:]
    return grad


def run(seed=42, budget_s=1100, **kwargs):
    """Optimize a step‑function h to obtain a low Erdős C₅ bound."""
    rng = np.random.RandomState(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=float)   # provided by the harness
        n = h0.shape[0]
        target_sum = n / 2.0
        h = _project_box_sum(h0, target_sum)              # guarantee feasibility
    except Exception:                                    # no parent construction
        n = 200                                            # a modest size, fast to evaluate
        target_sum = n / 2.0
        h = np.zeros(n, dtype=float)
        idx = rng.choice(n, size=n // 2, replace=False)
        h[idx] = 1.0                                       # binary start with correct mass

    # ------------------------------------------------------------------
    # optimisation parameters
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_val = _compute_c5(best_h)

    beta_start = 5.0          # start with a smooth approximation
    beta_end   = 30.0         # finish with a sharper max‑approximation
    beta_steps = 50           # how many iterations before raising beta
    step0 = 0.5               # initial step length (will be decay‑scaled)

    start_time = time.time()
    it = 0
    beta = beta_start

    while time.time() - start_time < budget_s - 1.0:
        # ---- compute max–soft objective and its gradient ----------------
        grad = _gradient_softmax_full(h, beta)

        # descent direction (zero‑sum, so that the linear constraint stays feasible)
        d = -grad
        d -= d.mean()

        # step size schedule (simple 1/√t decay)
        step = step0 / math.sqrt(it + 1.0)

        h_new = h + step * d
        h_new = _project_box_sum(h_new, target_sum)

        new_val = _compute_c5(h_new)

        # keep the best ever seen
        if new_val < best_val:
            best_val = new_val
            best_h = h_new.copy()

        # accept greedy or with a tiny probability to escape plateaus
        cur_val = _compute_c5(h)
        if new_val <= cur_val or rng.rand() < 0.005:
            h = h_new

        # slowly raise β to focus on the true max
        if it % beta_steps == 0 and it > 0:
            beta = min(beta_end, beta * 1.2)

        it += 1

    # final bound (already scaled by dx inside _compute_c5)
    return (best_h.tolist(), best_val, n)
