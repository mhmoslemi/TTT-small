def project_feasible(h):
    n = h.size
    target = n / 2.0
    # clip
    h = np.clip(h, 0, 1)
    # compute diff
    diff = target - np.sum(h)
    # while diff not zero:
    # compute capacities: if diff > 0, capacity_i = 1 - h_i (space to add)
    # else capacity_i = h_i (space to subtract)
    capacities = (1 - h) if diff > 0 else h
    total_cap = np.sum(capacities)
    if total_cap == 0:
        # already at bounds can't adjust; return as is
        return h
    # allocate proportionally
    adjustment = diff * capacities / total_cap
    h += adjustment
    # after adjustment some may overshoot due to floating errors; clip again and iterate a few times
    h = np.clip(h, 0, 1)
    # we may need multiple passes
    # For simplicity, iterate up to 5 times
    for _ in range(5):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        capacities = (1 - h) if diff > 0 else h
        total_cap = np.sum(capacities)
        if total_cap == 0:
            break
        adjustment = diff * capacities / total_cap
        h += adjustment
        h = np.clip(h, 0, 1)
    return h

    cur_h = h_init.copy()
    cur_C = compute_C(cur_h)
    best_h = cur_h.copy()

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    # clip first
    x = np.clip(h, 0.0, 1.0)
    # adjust sum
    diff = target - np.sum(x)
    if abs(diff) < 1e-12:
        return x
    # iterative distribution
    # we will perform e.g., 10 passes.
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # need to add diff
            capacities = 1.0 - x
        else:
            capacities = x
        total_cap = np.sum(capacities)
        if total_cap <= 1e-12:
            break
        # allocate
        delta = diff * capacities / total_cap
        x += delta
        # clip any overflow
        x = np.clip(x, 0.0, 1.0)
        diff = target - np.sum(x)
    # After loop, adjust any remaining diff by adding to an arbitrary non-saturated entry
    if abs(diff) > 1e-8:
        # simple fix: add diff to first element (if possible)
        for i in range(n):
            if diff > 0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    # final clip
    x = np.clip(x, 0.0, 1.0)
    return x
```

def simulated_annealing(h0, time_limit):
    n = len(h0)
    dx = 2.0 / n
    compute_C = lambda hh: np.max(np.correlate(hh, 1.0 - hh, mode='full')) * dx
    cur = np.copy(h0)
    curC = compute_C(cur)
    best = np.copy(cur)
    bestC = curC
    start = time.time()
    # temperature schedule: start T0 based on initial C maybe 0.05
    T0 = 0.02  # small?
    Tf = 1e-6
    it = 0
    max_iters = 10**7  # large upper bound, while loop limited by time
    while time.time() - start < time_limit and it < max_iters:
        it += 1
        # fraction of time elapsed
        elapsed = time.time() - start
        frac = elapsed / time_limit
        T = T0 * (1 - frac) + Tf * frac
        # propose move
        i = np.random.randint(0, n)
        j = np.random.randint(0, n-1)
        if j >= i:
            j += 1  # ensure i!=j
        hi = cur[i]
        hj = cur[j]
        # epsilon range
        eps_low = max(-hi, hj - 1.0)
        eps_high = min(1.0 - hi, hj)
        if eps_low >= eps_high:
            continue
        # bound magnitude of epsilon relative to T
        max_step = 0.2 * (T / T0 + 0.01)  # up to 0.2 at start
        eps = np.random.uniform(eps_low, eps_high)
        if abs(eps) > max_step:
            eps = np.sign(eps) * max_step
        # apply move
        new = cur.copy()
        new[i] = hi + eps
        new[j] = hj - eps
        # Evaluate
        newC = compute_C(new)
        dC = newC - curC
        if dC < 0 or np.random.rand() < math.exp(-dC / (T + 1e-12)):
            cur = new
            curC = newC
            if newC < bestC:
                best = new.copy()
                bestC = newC
        # occasional random perturbation to avoid stagnation
        if it % 10000 == 0:
            # small random jitter on random subset
            idxs = np.random.choice(n, size=min(5,n), replace=False)
            for idx in idxs:
                change = np.random.uniform(-0.03, 0.03)
                cur[idx] = np.clip(cur[idx] + change, 0.0, 1.0)
            # reproject to maintain sum
            cur = project_feasible(cur)
            curC = compute_C(cur)
            if curC < bestC:
                best = cur.copy()
                bestC = curC
    return best, bestC

def slsqp_optimize(h0, time_limit):
    from scipy.optimize import minimize
    n = len(h0)
    target = n / 2.0
    dx = 2.0 / n
    def objective(h):
        # h is array
        return np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    # constraint
    cons = ({
        'type': 'eq',
        'fun': lambda h: np.sum(h) - target
    })
    bounds = [(0.0, 1.0)] * n
    # Set options with max time limit via iteration limit; choose maxiter based on time.
    maxiter = 1000
    # We'll adapt maxiter based on time left (e.g., 2000)
    # Use method 'SLSQP' which handles equality constraints.
    res = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=[cons],
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
    h_opt = res.x
    # reproject to ensure feasibility
    h_opt = project_feasible(h_opt)
    c_opt = objective(h_opt)
    return h_opt, c_opt

import numpy as np
import time
import math
import random
from scipy.optimize import minimize

def _compute_c5(h: np.ndarray) -> float:
    """return the Erdős C5 value for a step function h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _project_feasible(h: np.ndarray) -> np.ndarray:
    """Clip to [0,1] and adjust to make sum(h)=len(h)/2."""
    n = h.size
    target = n / 2.0
    x = np.clip(h, 0.0, 1.0)
    diff = target - x.sum()
    # iterative proportional redistribution
    for _ in range(10):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            caps = 1.0 - x          # space to increase
        else:
            caps = x                # space to decrease
        total = caps.sum()
        if total <= 0:
            break
        delta = diff * caps / total
        x += delta
        x = np.clip(x, 0.0, 1.0)
        diff = target - x.sum()
    # any remaining tiny discrepancy – fix greedily
    if abs(diff) > 1e-9:
        for i in range(n):
            if diff > 0 and x[i] < 1.0:
                add = min(diff, 1.0 - x[i])
                x[i] += add
                diff -= add
            elif diff < 0 and x[i] > 0.0:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def _upsample_h(h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a larger grid, then project onto feasible set."""
    old_n = h.size
    # centre of each cell as coordinate
    x_old = np.linspace(0, 2, old_n, endpoint=False) + (1.0 / old_n)
    x_new = np.linspace(0, 2, new_n, endpoint=False) + (1.0 / new_n)
    h_new = np.interp(x_new, x_old, h)
    return _project_feasible(h_new)

def _simulated_annealing(h0: np.ndarray, time_limit: float):
    """Simple SA that respects the sum constraint by moving mass between two indices."""
    n = h0.size
    dx = 2.0 / n
    target = n / 2.0

    def score(arr):
        return np.max(np.correlate(arr, 1.0 - arr, mode="full")) * dx

    cur = h0.copy()
    cur_score = score(cur)
    best = cur.copy()
    best_score = cur_score

    start = time.time()
    it = 0
    # schedule parameters
    T0 = 0.04
    Tf = 1e-6
    max_iters = 10_000_000

    while it < max_iters and time.time() - start < time_limit:
        it += 1
        t = (time.time() - start) / time_limit
        T = (1 - t) * T0 + t * Tf

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur[i], cur[j]

        # feasible epsilon range (move epsilon from j to i)
        eps_low = max(-hi, hj - 1.0)
        eps_high = min(1.0 - hi, hj)
        if eps_low >= eps_high:
            continue

        # limit step size with temperature
        max_step = 0.2 * (T / T0 + 0.01)
        eps = np.random.uniform(eps_low, eps_high)
        if abs(eps) > max_step:
            eps = math.copysign(max_step, eps)

        new = cur.copy()
        new[i] = hi + eps
        new[j] = hj - eps

        new_score = score(new)
        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-12)):
            cur = new
            cur_score = new_score
            if new_score < best_score:
                best = new.copy()
                best_score = new_score

        # occasional jitter to escape flat regions
        if it % 7000 == 0:
            idxs = np.random.choice(n, size=min(5, n), replace=False)
            for k in idxs:
                cur[k] = np.clip(cur[k] + np.random.uniform(-0.03, 0.03), 0.0, 1.0)
            cur = _project_feasible(cur)
            cur_score = score(cur)
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

    return best, best_score

def _slsqp_optimize(h0: np.ndarray, time_limit: float):
    """Refine a candidate using SLSQP with an equality sum constraint."""
    n = h0.size
    target = n / 2.0
    dx = 2.0 / n

    def obj(arr):
        return np.max(np.correlate(arr, 1.0 - arr, mode="full")) * dx

    cons = ({
        'type': 'eq',
        'fun': lambda arr: np.sum(arr) - target
    })
    bounds = [(0.0, 1.0)] * n

    # limit iterations proportionally to the allowed time
    maxiter = max(200, int(time_limit * 500))  # crude scaling
    res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                   constraints=[cons],
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
    h_opt = _project_feasible(res.x)
    return h_opt, obj(h_opt)
