def gradient_of_corr_at_shift(h, k):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros_like(h)
    if k == 0:
        g = dx * (1 - 2 * h)
        return g
    if k > 0:
        # first component for i in 0..n-1-k: add (1 - h[i+k])
        g[:n - k] += dx * (1 - h[k:])
        # second component for j in k..n-1: subtract h[j-k]
        g[k:] -= dx * h[:n - k]
    else:
        kp = -k
        # for i in kp..n-1: add (1 - h[i - kp])
        g[kp:] += dx * (1 - h[:n - kp])
        # for j in 0..n-1-kp: subtract h[j + kp]
        g[:n - kp] -= dx * h[kp:]
    return g

n = target_n

a = cp.Variable(n, boolean=True)

t = cp.Variable()

constraints = [cp.sum(a) == n/2]

def simulated_annealing(h0, time_deadline):
    n = len(h0)
    sum_target = n / 2.0
    cur_h = h0.copy()
    cur_c = compute_max_overlap(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c

    T = 0.001  # initial temperature
    step_factor = 1.0

    cooling = 0.9995
    min_T = 1e-9
    min_step = 1e-9

    while time.time() < time_deadline:
        # propose move
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            i, j = np.random.randint(0, n, size=2)
        # compute possible transfers
        max_inc = min(1.0 - cur_h[i], cur_h[j])
        max_dec = min(cur_h[i], 1.0 - cur_h[j])
        if max_inc > 0 and (max_dec == 0 or np.random.rand() < 0.5):
            delta = np.random.rand() * max_inc * step_factor
        elif max_dec > 0:
            delta = -np.random.rand() * max_dec * step_factor
        else:
            continue  # nothing to move
        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        new_c = compute_max_overlap(new_h)
        dE = new_c - cur_c
        if dE < 0 or np.exp(-dE / max(T, 1e-12)) > np.random.rand():
            cur_h, cur_c = new_h, new_c
            if cur_c < best_c:
                best_h, best_c = cur_h.copy(), cur_c
        # cooling
        T = max(min_T, T * cooling)
        step_factor = max(min_step, step_factor * cooling)
    return best_h, best_c

import time
import math
import random
import numpy as np

def _compute_max_overlap(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1‑h(x+k))dx for the discretized step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def _max_shift(h: np.ndarray):
    """Return the integer shift at which the overlap integral is maximal."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    shift = idx - (n - 1)
    return shift


def _gradient_at_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the max‑overlap term for a given shift k."""
    n = h.shape[0]
    dx = 2.0 / n
    g = np.zeros_like(h)
    if k == 0:
        g[:] = dx * (1.0 - 2.0 * h)
        return g
    if k > 0:
        # first part: i from 0 .. n-1-k  -> derivative w.r.t. h_i
        g[: n - k] += dx * (1.0 - h[k:])
        # second part: j from k .. n-1 -> derivative w.r.t. h_j (the shifted factor)
        g[k:] -= dx * h[: n - k]
    else:   # k < 0, treat symmetrically
        kp = -k
        g[kp:] += dx * (1.0 - h[: n - kp])
        g[: n - kp] -= dx * h[kp:]
    return g


def _project_capped_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto {x | 0 ≤ x_i ≤ 1, sum x_i = s}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0          # at lo all entries clip to 1 -> sum=n
    hi = np.max(v)                # at hi all entries clip to 0 -> sum=0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        tot = w.sum()
        if tot > s:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def _simulated_annealing(start_h: np.ndarray, deadline: float) -> tuple:
    """Simple SA that preserves the sum constraint by moving mass between two sites."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    cur_h = start_h.copy()
    cur_val = _compute_max_overlap(cur_h)

    best_h = cur_h.copy()
    best_val = cur_val

    T = 1e-3                     # initial temperature
    step = 1.0                   # initial step scale (≤1)
    cooling = 0.9995
    min_T = 1e-12
    min_step = 1e-12

    while time.time() < deadline:
        # propose a transfer between two distinct indices
        i, j = np.random.randint(0, n, size=2)
        while i == j:
            i, j = np.random.randint(0, n, size=2)

        # possible amount we can move from j to i (increase i)
        max_inc = min(1.0 - cur_h[i], cur_h[j])
        # possible amount we can move from i to j (decrease i)
        max_dec = min(cur_h[i], 1.0 - cur_h[j])

        if max_inc <= 0 and max_dec <= 0:
            continue

        if max_inc > 0 and (max_dec == 0 or random.random() < 0.5):
            delta = random.random() * max_inc * step
        else:
            delta = -random.random() * max_dec * step

        new_h = cur_h.copy()
        new_h[i] += delta
        new_h[j] -= delta

        new_val = _compute_max_overlap(new_h)
        dE = new_val - cur_val

        if dE < 0 or math.exp(-dE / max(T, 1e-12)) > random.random():
            cur_h, cur_val = new_h, new_val
            if cur_val < best_val:
                best_h, best_val = cur_h.copy(), cur_val

        # cool
        T = max(min_T, T * cooling)
        step = max(min_step, step * cooling)

    # final tiny projection to guarantee exact sum (helps with accumulated round‑off)
    best_h = _project_capped_simplex(best_h, target_sum)
    best_val = _compute_max_overlap(best_h)
    return best_h, best_val


def _gradient_descent(start_h: np.ndarray, steps: int = 200) -> tuple:
    """Few gradient steps with projection – fine‑tunes a good candidate."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    h = start_h.copy()
    cur_val = _compute_max_overlap(h)

    for it in range(steps):
        k = _max_shift(h)
        grad = _gradient_at_shift(h, k)
        step = 0.1 / (it + 1)          # diminishing step
        cand = _project_capped_simplex(h - step * grad, target_sum)
        cand_val = _compute_max_overlap(cand)

        if cand_val < cur_val:
            h, cur_val = cand, cand_val
        else:
            # try a smaller step just in case
            cand = _project_capped_simplex(h - step * 0.5 * grad, target_sum)
            cand_val = _compute_max_overlap(cand)
            if cand_val < cur_val:
                h, cur_val = cand, cand_val

    return h, cur_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős C5 problem.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # initialise
    init = globals().get('initial_h_values')
    if init is not None:
        h0 = np.array(init, dtype=float).copy()
    else:
        # fall‑back: 60 equally‑spaced samples, constant 0.5 (feasible)
        n = 60
        h0 = np.full(n, 0.5, dtype=float)

    n = h0.shape[0]
    target_sum = n / 2.0
    # enforce the integral exactly (may be off by rounding)
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-9:
        h0 = _project_capped_simplex(h0, target_sum)

    best_h = h0.copy()
    best_val = _compute_max_overlap(best_h)

    start_time = time.time()
    end_time = start_time + budget_s - 0.1   # tiny safety margin

    # ------------------------------------------------------------------
    # optimisation loop – a few restarts with SA + short GD fine‑tune
    restarts = 5
    for r in range(restarts):
        if time.time() >= end_time:
            break

        if r == 0 and init is not None:
            start_vec = h0.copy()
        else:
            # random binary start (half ones)
            start_vec = np.zeros(n, dtype=float)
            half = n // 2
            start_vec[:half] = 1.0
            np.random.shuffle(start_vec)

        # SA (will stop automatically when time runs out)
        vec_sa, val_sa = _simulated_annealing(start_vec, end_time)

        # short gradient descent for polishing
        vec_gd, val_gd = _gradient_descent(vec_sa, steps=200)

        if val_gd < best_val:
            best_h, best_val = vec_gd, val_gd

    # final projection to guarantee feasibility
    best_h = _project_capped_simplex(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)   # safety
    final_val = _compute_max_overlap(best_h)

    return (best_h, float(final_val), int(n))
