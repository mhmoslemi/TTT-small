            if new_val < best_val:
                best_val, best_h = new_val, new_h.copy()
        it += 1
        # stop early if we already beat the target bound
        if best_val < 0.3808:
            break

    # final renormalisation (tiny drift may appear)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h / (best_h.sum() * dx)
    final_val = _corr_max(best_h)

    return (best_h, final_val, n)
```
