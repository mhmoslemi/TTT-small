import math, time, numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_sum_and_bounds(h, target_sum):
    # projection onto [0,1] and sum target using bisection.
    n = len(h)
    lo = -1.0
    hi = 1.0
    # Expand bounds enough
    for _ in range(50):
        lam = (lo + hi) / 2.0
        proj = np.clip(h - lam, 0, 1)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return proj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    # correlation via convolution
    a = h
    b = 1 - h
    # reverse b for correlation
    b_rev = b[::-1]
    N = 1 << (2*n-1).bit_length()  # next power of two
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b_rev, n=N)
    conv = np.fft.irfft(fa * fb, n=N)
    corr = conv[:2*n-1]  # length
    return float(corr.max() * dx)

import numpy as np
import time

def _max_overlap(h):
    """Return C5 for a step function h using convolution."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation = convolution of h with reversed (1-h)
    a = h
    b = (1.0 - h)[::-1]
    N = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[: 2 * n - 1]
    return float(conv.max() * dx)


def _project_sum_bounds(x, target):
    """Project x onto the box [0,1] while fixing the sum to target."""
    lo, hi = -2.0, 2.0  # enough to drive all values to bounds
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(x - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return proj


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # starting function
    # ------------------------------------------------------------------
    try:
        h = np.array(initial_h_values, dtype=float)  # provided by the harness
        n = h.shape[0]
        target_sum = n / 2.0
        if not np.isclose(h.sum(), target_sum):
            h = _project_sum_bounds(h, target_sum)
    except NameError:
        # no parent construction – use a simple symmetric triangle
        n = 400                     # moderate resolution (<1000 points)
        dx = 2.0 / n
        x = (np.arange(n) + 0.5) * dx
        h = 1.0 - np.abs(x - 1.0)   # peak =1 at the centre, zero at the ends
        target_sum = n / 2.0

    # make sure constraints are satisfied
    h = np.clip(h, 0.0, 1.0)
    h = _project_sum_bounds(h, target_sum)

    best_h = h.copy()
    best_val = _max_overlap(best_h)

    # ------------------------------------------------------------------
    # Simple hill‑climbing / simulated‑annealing search
    # ------------------------------------------------------------------
    cur_h = h.copy()
    cur_val = best_val
    step = 0.2                     # initial magnitude of a transfer
    max_iters = 200000
    for it in range(max_iters):
        # respect time budget
        if time.time() - start_time > budget_s * 0.95:
            break

        # pick two distinct indices and propose a transfer
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = (np.random.rand() - 0.5) * step
        hi = cur_h[i] + delta
        hj = cur_h[j] - delta
        if not (0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0):
            continue

        new_h = cur_h.copy()
        new_h[i] = hi
        new_h[j] = hj
        new_val = _max_overlap(new_h)

        # accept only improving moves (monotone descent)
        if new_val < cur_val - 1e-12:
            cur_h, cur_val = new_h, new_val
            if new_val < best_val:
                best_h, best_val = new_h.copy(), new_val

        # slowly shrink the step size to fine‑tune
        step *= 0.99995
        if step < 1e-5:
            step = 1e-5

    # ------------------------------------------------------------------
    # Return result
    # ------------------------------------------------------------------
    return (best_h.tolist(), float(best_val), n)
