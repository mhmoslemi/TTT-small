import time
import math
import numpy as np

def max_overlap(h: np.ndarray) -> int:
    """Return the maximal overlap count ∑ h[i]*(1‑h[i+k]) for all integer shifts."""
    # use integer arithmetic for speed
    return int(np.correlate(h, 1 - h, mode='full').max())

def symmetric_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Construct a balanced 0/1 vector with many i,i+n/2 equal (helps lower shift N/2)."""
    half = n // 2
    # number of pairs with (1,1)
    double_pairs = half // 2            # 17 for n=70
    # choose indices for double 1s
    all_idx = np.arange(half)
    double_idx = rng.choice(all_idx, size=double_pairs, replace=False)
    remaining = np.setdiff1d(all_idx, double_idx, assume_unique=True)
    # choose one single‑1 pair
    single_idx = rng.choice(remaining, size=1)[0]
    # build the vector
    h = np.zeros(n, dtype=np.int8)
    # (1,1) pairs
    h[double_idx] = 1
    h[double_idx + half] = 1
    # (1,0) single pair (put the 1 in the first half)
    h[single_idx] = 1
    # already zero elsewhere, total ones = 2*double_pairs + 1 = n/2
    return h

def binary_from_initial(init_arr: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Derive a binary vector respecting the sum constraint from a possibly fractional init."""
    n = init_arr.size
    target = n // 2
    # start with rounding >0.5
    h = (init_arr > 0.5).astype(np.int8)
    cur = int(h.sum())
    if cur > target:
        # need to flip some 1→0, choose those with smallest init value (closest to 0.5)
        ones = np.where(h == 1)[0]
        order = ones[np.argsort(init_arr[ones])]  # ascending init
        flip = cur - target
        h[order[:flip]] = 0
    elif cur < target:
        zeros = np.where(h == 0)[0]
        order = zeros[np.argsort(-init_arr[zeros])]  # descending init
        flip = target - cur
        h[order[:flip]] = 1
    return h

    # -------------------------------------------------------------

    # a deterministic symmetric start (often good)

        cost_sym = max_overlap(h_sym)
