def random_intervals(N, k=5):
    target = N // 2
    zeros = N - target
    # Sample interval lengths using Dirichlet, produce real numbers
    probs = np.random.rand(k)
    lengths_f = probs / probs.sum()
    lengths = np.floor(lengths_f * target).astype(int)
    # Adjust to meet total:
    while lengths.sum() < target:
        idx = np.random.randint(k)
        lengths[idx] += 1
    while lengths.sum() > target:
        idx = np.random.randint(k)
        if lengths[idx] > 1:
            lengths[idx] -= 1
    # gaps: k+1
    probs_g = np.random.rand(k+1)
    gaps_f = probs_g / probs_g.sum()
    gaps = np.floor(gaps_f * zeros).astype(int)
    while gaps.sum() < zeros:
        idx = np.random.randint(k+1)
        gaps[idx] += 1
    while gaps.sum() > zeros:
        idx = np.random.randint(k+1)
        if gaps[idx] > 0:
            gaps[idx] -= 1
    # Build h
    h = np.zeros(N, dtype=float)
    pos = gaps[0]
    for i in range(k):
        if lengths[i] > 0:
            h[pos:pos+lengths[i]] = 1.0
        pos += lengths[i] + gaps[i+1]
    return h

def construct_h_from_lengths_gaps(lengths, gaps):
    N = sum(lengths) + sum(gaps)
    h = np.zeros(N, dtype=float)
    pos = gaps[0]
    for i in range(len(lengths)):
        L = lengths[i]
        if L > 0:
            h[pos:pos+L] = 1.0
        pos += L + gaps[i+1]
    return h

def random_intervals_fixed(N,k=5):
    target = N//2
    zeros = N - target
    # sample k lengths via Dirichlet then integer adjustments
    probs = np.random.rand(k)
    lengths_float = probs / probs.sum()
    lengths = np.floor(lengths_float * target).astype(int)
    # adjust to target
    while lengths.sum() < target:
        lengths[np.random.randint(k)] += 1
    while lengths.sum() > target:
        idx = np.random.randint(k)
        if lengths[idx] > 0:
            lengths[idx] -= 1
    # gaps
    probs_g = np.random.rand(k+1)
    gaps_float = probs_g / probs_g.sum()
    gaps = np.floor(gaps_float * zeros).astype(int)
    while gaps.sum() < zeros:
        gaps[np.random.randint(k+1)] += 1
    while gaps.sum() > zeros:
        idx = np.random.randint(k+1)
        if gaps[idx] > 0:
            gaps[idx] -= 1
    return lengths, gaps

import numpy as np
import time

# ---------- core evaluation ----------
def _c5_from_h(h: np.ndarray, size_fft: int) -> float:
    """Return C5 = max_k ∫ h(x)(1-h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    # correlation h * (1-h)  (full, non‑circular) via FFT
    f = h
    g = (1.0 - h)[::-1]               # reversed for correlation
    F = np.fft.rfft(f, n=size_fft)
    G = np.fft.rfft(g, n=size_fft)
    conv = np.fft.irfft(F * G, n=size_fft)[: 2 * n - 1]
    return conv.max() * (2.0 / n)    # dx = 2/n


# ---------- interval handling ----------
def _random_intervals(N: int, k: int = 5):
    """Generate integer interval lengths (k of them) and gaps (k+1) that sum to N."""
    target = N // 2                     # number of ones
    zeros = N - target                  # number of zeros

    # ----- interval lengths -----
    probs = np.random.rand(k)
    lengths = np.floor(probs / probs.sum() * target).astype(int)
    while lengths.sum() < target:
        lengths[np.random.randint(k)] += 1
    while lengths.sum() > target:
        i = np.random.randint(k)
        if lengths[i] > 0:
            lengths[i] -= 1

    # ----- gaps -----
    probs_g = np.random.rand(k + 1)
    gaps = np.floor(probs_g / probs_g.sum() * zeros).astype(int)
    while gaps.sum() < zeros:
        gaps[np.random.randint(k + 1)] += 1
    while gaps.sum() > zeros:
        i = np.random.randint(k + 1)
        if gaps[i] > 0:
            gaps[i] -= 1

    return lengths, gaps


def _build_h(lengths, gaps):
    """Construct binary vector from lengths/gaps (both integer arrays)."""
    N = int(lengths.sum() + gaps.sum())
    h = np.zeros(N, dtype=float)
    pos = gaps[0]
    for li, gi in zip(lengths, gaps[1:]):
        if li:
            h[pos : pos + li] = 1.0
        pos += li + gi
    return h
