import numpy as np, time

def evaluate(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def initial_shape(n):
    dx = 2.0 / n
    # midpoints
    x = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * x / 2.0)  # sine shape from 0 to 1
    # scale to integral 1
    total = h.sum() * dx
    if total != 0:
        h = h / total
    # ensure within [0,1]
    h = np.clip(h, 0.0, 1.0)
    return h

def hill_climb(h0, time_limit):
    n = len(h0)
    h = h0.copy()
    best = h.copy()
    best_score = evaluate(best)
    cur_score = best_score
    start = time.time()
    # temperature schedule
    T = best_score * 0.02 + 1e-4
    # Precompute max step: maybe 0.05 of average value 0.5? Use 0.2
    max_step = 0.2
    rng = np.random.RandomState(0)  # deterministic within run via seed
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # random index pair
        i, j = rng.randint(n), rng.randint(n)
        if i == j:
            continue
        # choose direction randomly
        if rng.rand() < 0.5:
            # move from i to j (decrease i, increase j)
            max_eps = min(h[i], 1.0 - h[j])
        else:
            # move from j to i
            max_eps = min(h[j], 1.0 - h[i])
            i, j = j, i  # swap to treat as moving from i to j
        if max_eps <= 0:
            continue
        eps = rng.rand() * max_step
        eps = min(eps, max_eps)
        if eps <= 0:
            continue
        # apply
        h[i] -= eps
        h[j] += eps
        # optional: keep sum exact; due to epsilon, sum identical within FP
        new_score = evaluate(h)
        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # simulated annealing
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.rand() < prob:
                accept = True
        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best = h.copy()
        else:
            # revert
            h[i] += eps
            h[j] -= eps
        # temperature cooling
        T *= 0.99995
        # periodically renormalize to guard numerical drift
        if it % 1000 == 0:
            s = h.sum()
            target = n / 2.0
            if abs(s - target) > 1e-8:
                h = h * (target / s)
                # clip again
                h = np.clip(h, 0.0, 1.0)
                # re-correct sum after clipping? Might change sum.
                s = h.sum()
                if abs(s - target) > 1e-8:
                    # distribute diff evenly
                    diff = target - s
                    h += diff / n
                    # ensure still within bounds
                    h = np.clip(h, 0.0, 1.0)
    return best, best_score

import time
import numpy as np

def evaluate_h(h: np.ndarray) -> float:
    """Return C5 for a given step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))


def build_initial_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a smooth start vector (scaled half‑sine) that satisfies the constraints."""
    dx = 2.0 / n
    # mid‑point sampling
    x = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * x / 2.0)                 # peaks at 1, zero at the ends
    total = h.sum() * dx
    if total != 0.0:
        h = h / total                           # scale to integral 1
    # after scaling the maximum is π/4 < 1, so clipping is unnecessary but keep it safe
    np.clip(h, 0.0, 1.0, out=h)
    return h


def hill_climb(initial_h: np.ndarray, time_budget: float, seed: int) -> tuple:
    """Simple simulated‑annealing optimizer that preserves the sum exactly."""
    rng = np.random.default_rng(seed)
    n = initial_h.shape[0]
    cur_h = initial_h.copy()
    cur_score = evaluate_h(cur_h)
    best_h = cur_h.copy()
    best_score = cur_score

    # annealing parameters
    T = max(0.01, cur_score * 0.02)
    decay = 0.99995
    max_step = 0.2                       # maximal mass moved per trial
    target_sum = n / 2.0

    start = time.time()
    iter_count = 0
    while time.time() - start < time_budget:
        iter_count += 1

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # decide direction of transfer
        if rng.random() < 0.5:
            src, dst = i, j   # move from src to dst
        else:
            src, dst = j, i

        max_eps = min(cur_h[src], 1.0 - cur_h[dst])
        if max_eps <= 0.0:
            continue

        eps = rng.random() * max_step
        eps = min(eps, max_eps)
        if eps <= 0.0:
            continue

        # propose change
        cur_h[src] -= eps
        cur_h[dst] += eps

        new_score = evaluate_h(cur_h)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[src] += eps
            cur_h[dst] -= eps

        # temperature decay
        T = max(T * decay, 1e-8)

        # occasional renormalisation to counter floating drift
        if iter_count % 1000 == 0:
            s = cur_h.sum()
            if abs(s - target_sum) > 1e-12:
                cur_h *= target_sum / s
                np.clip(cur_h, 0.0, 1.0, out=cur_h)

    return best_h, best_score


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    start_time = time.time()
    rng = np.random.default_rng(seed)

    # number of discretisation points (feel free to adjust via kwargs)
    n = int(kwargs.get("n_points", 100))
    if n < 2:
        n = 2

    # build a starting vector
    if "initial_h_values" in globals():
        init = globals()["initial_h_values"]
        init = np.asarray(init, dtype=float)
        if init.size != n:
            # resample the provided vector onto the new grid
            old_n = init.size
            dx_old = 2.0 / old_n
            x_old = (np.arange(old_n) + 0.5) * dx_old
            dx_new = 2.0 / n
            x_new = (np.arange(n) + 0.5) * dx_new
            init = np.interp(x_new, x_old, init)
        # enforce the integral constraint
        cur_sum = init.sum()
        target = n / 2.0
        if cur_sum != 0.0:
            init *= target / cur_sum
        np.clip(init, 0.0, 1.0, out=init)
        h0 = init
    else:
        h0 = build_initial_h(n, rng)

    # allocate most of the time budget to the optimizer
    remaining = max(0.0, budget_s - (time.time() - start_time))
    opt_time = remaining * 0.95

    best_h, best_score = hill_climb(h0, opt_time, seed + 12345)

    # final safety renormalisation
    target_sum = n / 2.0
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        best_h *= target_sum / cur_sum
        np.clip(best_h, 0.0, 1.0, out=best_h)

    final_score = evaluate_h(best_h)

    return best_h, final_score, n
