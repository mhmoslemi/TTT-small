def perturb(h, step=0.1):
    n = len(h)
    # pick 2 indices random
    i, j = np.random.choice(n, 2, replace=False)
    # delta randomly between -step and +step but respecting constraints.
    max_delta = min(step, 1 - h[i], h[j])  # increase i, decrease j
    min_delta = -min(step, h[i], 1 - h[j])  # decrease i, increase j
    delta = np.random.uniform(min_delta, max_delta)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def project_feasible(v, total):
    # clip to [0,1]
    v = np.clip(v, 0, 1)
    # compute current sum
    cur_sum = v.sum()
    if abs(cur_sum - total) < 1e-9:
        return v
    # need to shift interior values (those not at bounds) by lambda.
    # Use water-filling: find lambda s.t. sum_i clip(v_i - lambda, 0, 1) = total
    lo = -1  # lower bound for lambda (we can shift downward)

import time, math, numpy as np

def _project_to_sum_bounds(v, total):
    # projection onto [0,1]^n and sum == total using water-filling (binary search on lambda)
    v = np.clip(v, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def _random_feasible(n, rng):
    total = n / 2.0
    v = rng.rand(n)
    return _project_to_sum_bounds(v, total)

    # linear interpolation on [0,2)
    xs_orig = np.linspace(0.0, 2.0, len(orig), endpoint=False)
    xs_new = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.interp(xs_new, xs_orig, orig)
    h = np.clip(h, 0.0, 1.0)
    total = n / 2.0
    h = _project_to_sum_bounds(h, total)
    return h

    else:

            return h.copy()
        delta = max_delta * rng.rand()
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
    return h_new

def _greedy_step(h, rng, dx):

    max_idx = corr.argmax()

    k = max_idx - (n - 1)
    # choose a subset of promising indices
    if k >= 0:
        i_range = np.arange(0, n - k)
        j_offset = k
    else:
        kk = -k
        i_range = np.arange(kk, n)
        j_offset = -kk
    if i_range.size == 0:
        return h.copy(), best_val
    # sample a few positions

    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.RandomState(seed)
    target_n = 200

    # initialise h
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=float)
        if init.shape[0] != target_n:
            h = _upsample_initial(init, target_n)
        else:
            h = np.clip(init, 0.0, 1.0)
            total = target_n / 2.0
            if abs(h.sum() - total) > 1e-9:
                h = _project_to_sum_bounds(h, total)
    else:
        h = _random_feasible(target_n, rng)

    n = h.shape[0]
    dx = 2.0 / n

    best_h = h.copy()
    best_c5 = _c5_value(best_h, dx)
    current_h = h.copy()
    current_c5 = best_c5

    start = time.time()
    step_factor = 0.5

    while time.time() - start < budget_s * 0.95:
        # occasional random shake to avoid stagnation
        if rng.rand() < 0.02:
            cand = _random_pair_move(current_h, rng)
            cand_c5 = _c5_value(cand, dx)
            if cand_c5 < current_c5:
                current_h, current_c5 = cand, cand_c5
                if cand_c5 < best_c5:
                    best_h, best_c5 = cand.copy(), cand_c5
            continue

        cand_h, cand_c5 = _greedy_step(current_h, rng, dx, step_factor)

        if cand_c5 < current_c5 - 1e-12:
            current_h, current_c5 = cand_h, cand_c5
        else:
            # tiny probability of accepting a worsening move (simulated‑annealing style)
            elapsed = time.time() - start
            frac = elapsed / (budget_s * 0.95)
            temp = max(1e-6, (1.0 - frac) * 0.005)
            if rng.rand() < math.exp(-(cand_c5 - current_c5) / temp):
                current_h, current_c5 = cand_h, cand_c5

        if current_c5 < best_c5 - 1e-12:
            best_h, best_c5 = current_h.copy(), current_c5

        step_factor = max(0.01, step_factor * 0.9995)

    return best_h.tolist(), float(best_c5), int(n)
