import time, math, numpy as np

def _c5(h):
    """max_{k} ∫ h(x)(1‑h(x+k))dx approximated on a grid."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full").max() * dx

def _binary_hill_climb(n, rng, timelimit):
    """binary (0/1) hill‑climb with swaps, returns best vector and its C5."""
    cur = np.zeros(n, dtype=np.uint8)
    perm = rng.permutation(n)
    cur[perm[: n // 2]] = 1
    cur_c = _c5(cur)
    best = cur.copy()
    best_c = cur_c
    start = time.time()
    no_imp = 0
    while time.time() - start < timelimit:
        i, j = rng.integers(n, size=2)
        if cur[i] == cur[j]:
            continue
        cur[i], cur[j] = cur[j], cur[i]
        new_c = _c5(cur)
        if new_c <= cur_c:                     # accept improving swap
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best = cur.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:                                   # reject
            cur[i], cur[j] = cur[j], cur[i]
            no_imp += 1
        if no_imp >= 20000:                     # restart if stuck
            perm = rng.permutation(n)
            cur[:] = 0
            cur[perm[: n // 2]] = 1
            cur_c = _c5(cur)
            no_imp = 0
        if best_c <= 0.3808:
            break
    return best.astype(float), best_c

def _slsqp_refine(h0, rng, timelimit):
    """refine a feasible vector with SLSQP (if SciPy is available)."""
    try:
        from scipy.optimize import minimize
    except Exception:
        return h0, _c5(h0)

    n = h0.shape[0]
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = {"type": "eq", "fun": lambda x: x.sum() - n / 2.0}
    # objective: max correlation (scaled)
    def obj(x):
        return np.correlate(x, 1.0 - x, mode="full").max() * dx

    # limit iterations roughly by time budget
    maxit = int(200 * timelimit / 30.0) + 10   # very rough scaling
    opts = {"maxiter": maxit, "ftol": 1e-9, "disp": False}
    res = minimize(obj, h0, method="SLSQP", bounds=bounds,
                   constraints=cons, options=opts)
    if not res.success:
        # fallback to the starting point
        return h0, obj(h0)
    x = np.clip(res.x, 0.0, 1.0)
    # enforce exact sum (tiny correction)
    diff = (n / 2.0) - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += diff / free.sum()
            np.clip(x, 0.0, 1.0, out=x)
    return x, obj(x)
