def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    start = time.time()

    # Determine n and target sum
    if 'initial_h_values' in globals():

        n = init_h.size
    else:

    target_sum = n / 2.0
    dx = 2.0 / n

    # define evaluation functions
    def evaluate(h):
        c = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(c), int(np.argmax(c) - (n - 1))

    def project_box_sum(x):
        # search lambda s.t. sum clip(x - lam, 0,1) = target_sum
        lo = -np.max(x) - 2.0
        hi = np.max(x) + 2.0

            proj = np.clip(x - mid, 0.0, 1.0)
            s = proj.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

    def gradient(h, shift):
        # shift index: idx - (n-1)
        n = h.shape[0]
        grad = np.ones_like(h)
        if shift >= 0:
            grad[:n-shift] -= h[shift:]
            grad[shift:] -= h[:n-shift]
        else:
            s = -shift
            grad[s:] -= h[:n-s]
            grad[:n-s] -= h[s:]
        return grad

    # start with best from initial if exists else random
    if init_h is not None:
        h0 = project_box_sum(init_h.copy())
        best_h = h0.copy()
        best_val, _ = evaluate(best_h)
    else:
        # random start
        h0 = np.random.rand(n)
        h0 = project_box_sum(h0)
        best_h = h0.copy()
        best_val, _ = evaluate(best_h)

    # main optimization loop
    # Choose parameters
    # We'll decide number of iterations per "epoch" as maybe 2000, but may adjust.

    # Determine number of epochs based on time budget
    # We'll compute time per epoch adaptively
    epoch_iter = 2000
    # allocate approx 0.9 of budget for main loop

    while time.time() - start < budget_s * 0.95:
        # Choose random starting point occasionally, else start from best.
        if np.random.rand() < 0.3:
            start_h = np.random.rand(n)
        else:
            start_h = best_h.copy()
        start_h = project_box_sum(start_h)

    return (best_h.tolist(), float(best_val), n)

        corr = np.correlate(h, 1 - h, mode='full') * dx
        idx = int(np.argmax(corr))
        return float(corr[idx]), idx - (n - 1)

    def project(h):
        # project onto box [0,1] and sum target
        lo = -np.max(h) - 2.0
        hi = np.max(h) + 2.0
        for _ in range(50):
            mid = (lo + hi) * 0.5
            proj = np.clip(h - mid, 0.0, 1.0)
            s = proj.sum()
            if s > target:
                lo = mid
            else:
                hi = mid

                idx = np.flatnonzero(interior)[0]
                h_proj[idx] = np.clip(h_proj[idx] + diff, 0.0, 1.0)
            else:
                # fallback: adjust first element within bounds
                idx = 0
                h_proj[idx] = np.clip(h_proj[idx] + diff, 0.0, 1.0)
        return h_proj

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time
    np.random.seed(seed)
    start_time = time.time()

    # ----- problem size -------------------------------------------------
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values, dtype=np.float64)
        n = h_init.size
    else:
        # a reasonable default if no parent solution is supplied
        n = 80
        h_init = None
    target_sum = n / 2.0            # ∫h = 1  ⇒  sum(h) = n/2
    dx = 2.0 / n

    # ----- helpers -------------------------------------------------------
    def evaluate(h):
        """Return (max_overlap, shift_index) for a given h."""
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        idx = int(np.argmax(corr))
        # shift corresponds to index-(n-1); not needed for the value itself
        return float(corr[idx]), idx - (n - 1)

    def project_onto_box_and_sum(x):
        """Project x onto [0,1]^n while enforcing sum(x)=target_sum."""
        lo = -np.max(x) - 2.0   # ensures all entries become 1
        hi = np.max(x) + 2.0    # ensures all entries become 0
        for _ in range(50):
            mid = (lo + hi) * 0.5
            proj = np.clip(x - mid, 0.0, 1.0)
            s = proj.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

        # tiny correction to hit the sum exactly (numerical safety)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            interior = (h > 1e-9) & (h < 1.0 - 1e-9)
            if interior.any():
                i = np.flatnonzero(interior)[0]
                h[i] = np.clip(h[i] + diff, 0.0, 1.0)
            else:
                # fall back to the first coordinate (will stay in bounds)
                h[0] = np.clip(h[0] + diff, 0.0, 1.0)
        return h

    def gradient_worst_shift(h, shift):
        """Sub‑gradient of the worst overlap term with respect to h."""
        grad = np.ones_like(h)
        if shift >= 0:
            # terms involving h[i+shift] and h[i-shift]
            grad[: n - shift] -= h[shift:]
            grad[shift:] -= h[: n - shift]
        else:
            s = -shift
            grad[s:] -= h[: n - s]       # i+shift = i‑s
            grad[: n - s] -= h[s:]       # i‑shift = i+s
        return grad

    # ----- initialise best candidate ------------------------------------
    if h_init is not None:
        best_h = project_onto_box_and_sum(h_init.copy())
        best_val, _ = evaluate(best_h)
    else:
        rnd = np.random.rand(n)
        best_h = project_onto_box_and_sum(rnd)
        best_val, _ = evaluate(best_h)

    # ----- main stochastic search ----------------------------------------
    # timing constants
    TIME_LIMIT = budget_s * 0.94
    # parameters for inner optimisation
    INNER_ITERS = 3000
    while time.time() - start_time < TIME_LIMIT:
        # ----- choose a starting point ----------------------------------
        r = np.random.rand()
        if r < 0.35:
            # binary random pattern (exact half 1‑s)
            h0 = np.zeros(n, dtype=np.float64)
            ones = np.random.choice(n, size=n // 2, replace=False)
            h0[ones] = 1.0
        elif r < 0.7:
            # uniform random real vector
            h0 = np.random.rand(n)
        else:
            # perturb the current best solution
            h0 = best_h.copy()
            h0 += np.random.randn(n) * 0.02
        h0 = project_onto_box_and_sum(h0)

        # ----- projected gradient descent --------------------------------
        h = h0.copy()
        lr = 0.8                      # initial step size
        for it in range(INNER_ITERS):
            if time.time() - start_time >= TIME_LIMIT:
                break
            cur_val, cur_shift = evaluate(h)
            # store improvement
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            # gradient step on the worst shift
            g = gradient_worst_shift(h, cur_shift)
            h = h - lr * g
            h = project_onto_box_and_sum(h)
            lr *= 0.9995               # gradual decay

        # final check for this run
        cur_val, _ = evaluate(h)
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

    # ----- final polishing ------------------------------------------------
    best_h = project_onto_box_and_sum(best_h)
    best_val, _ = evaluate(best_h)

    # return in the required format
    return (best_h.tolist(), float(best_val), int(n))
