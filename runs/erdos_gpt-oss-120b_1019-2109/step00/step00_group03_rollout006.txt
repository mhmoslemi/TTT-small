def project_to_simplex(v, target):
    v = np.clip(v, 0.0, 1.0)
    current = v.sum()
    if abs(current - target) < 1e-12:
        return v
    diff = target - current
    # if diff >0: need to add mass
    if diff > 0:
        # find indices where v < 1
        can = np.where(v < 1)[0]
        if len(can) == 0:
            return v
        incr = diff / len(can)
        incr = min(incr, 1 - v[can])
        v[can] += incr
        # may still need more; repeat
    else:
        # diff < 0: need to subtract mass
        can = np.where(v > 0)[0]
        decr = (-diff) / len(can)
        decr = min(decr, v[can])
        v[can] -= decr
    # final clipping
    v = np.clip(v, 0, 1)
    # If still off due to rounding, adjust a random index.
    cur = v.sum()
    if cur < target:
        # add to first available
        for i in np.where(v < 1)[0]:
            need = target - cur
            add = min(need, 1 - v[i])
            v[i] += add
            cur = v.sum()
            if abs(cur - target) < 1e-12:
                break
    elif cur > target:
        for i in np.where(v > 0)[0]:
            need = cur - target
            sub = min(need, v[i])
            v[i] -= sub
            cur = v.sum()
            if abs(cur - target) < 1e-12:
                break
    return v

import numpy as np
import random
import time
import math

def _eval_c5(v, dx):
    # overlap = max_k ∫ h(x)*(1-h(x+k))dx = max(np.correlate(v,1-v))*dx
    corr = np.correlate(v, 1.0 - v, mode='full')
    return np.max(corr) * dx

def _project_to_sum(v, target):
    """Clip to [0,1] and adjust to have sum == target (exact)."""
    v = np.clip(v, 0.0, 1.0)
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    diff = target - cur
    if diff > 0:
        # need to add mass
        idx = np.where(v < 1.0)[0]
        for i in idx:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        diff = -diff
        idx = np.where(v > 0.0)[0]
        for i in idx:
            dec = min(diff, v[i])
            v[i] -= dec
            diff -= dec
            if diff <= 1e-12:
                break
    # final tiny correction if rounding left anything off
    cur = v.sum()
    if cur < target:
        for i in np.where(v < 1.0)[0]:
            add = min(target - cur, 1.0 - v[i])
            v[i] += add
            cur = v.sum()
            if abs(cur - target) < 1e-12:
                break
    elif cur > target:
        for i in np.where(v > 0.0)[0]:
            sub = min(cur - target, v[i])
            v[i] -= sub
            cur = v.sum()
            if abs(cur - target) < 1e-12:
                break
    return v

def run(seed=42, budget_s=1100, **kwargs):

    np.random.seed(seed + 12345)

    # --- initialise ---------------------------------------------------------

        n = h.shape[0]

        h = np.zeros(n, dtype=float)
        ones = np.random.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    target_sum = n / 2.0
    h = _project_to_sum(h, target_sum)

    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = _eval_c5(best_h, dx)
    cur_h = best_h.copy()
    cur_c5 = best_c5

    # --- annealing parameters -----------------------------------------------
    start = time.perf_counter()
    T0 = 0.05           # start temperature
    T_end = 1e-6        # final temperature
    stagnation = 0
    max_stag = 5000

    # --- main loop -----------------------------------------------------------
    while True:
        elapsed = time.perf_counter() - start
        if elapsed >= budget_s:
            break
        # temperature schedule (exponential)
        frac = elapsed / budget_s
        T = T0 * (T_end / T0) ** frac

        # propose a move

        if random.random() < 0.5:
            # swap values if they differ

        else:
