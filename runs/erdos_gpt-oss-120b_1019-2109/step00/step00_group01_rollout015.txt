    start = time.time()

    # Determine target N
    try:
        init = np.array(initial_h_values)
        base_len = init.shape[0]

    # Build initial binary vector h0

        rep = (N // base_len) + 1
        h_cont = np.tile(init, rep)[:N]

        idx_sorted = np.argsort(h_cont)
        h = np.zeros(N, dtype=np.float64)
        h[idx_sorted[:N//2]] = 1.0
    else:
        # random binary vector
        h = np.zeros(N, dtype=np.float64)
        ones_idx = rng.choice(N, N//2, replace=False)
        h[ones_idx] = 1.0

    # Precompute arrays for ones, zeros
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)

    # Evaluate current

        corr = np.correlate(vec, 1 - vec, mode='full')
        max_m = int(corr.max())
        return max_m, corr

    cur_max, cur_corr = evaluate_vec(h)
    best_h = h.copy()
    best_max = cur_max

    # Simulated annealing parameters
    total_time = budget_s
    T0 = 0.5
    # iteration loop
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > total_time * 0.95:
            break
        T = T0 * max(0.0, 1 - elapsed / total_time)  # linear cooling

        # get lag with max mismatch
        lag_index = int(np.argmax(cur_corr))
        lag = lag_index - (N - 1)

        # try targeted swap
        # determine bad ones (1->0) and good zeros (0->1)
        bad_ones = None
        good_zeros = None
        # compute arrays
        if lag >= 0:
            # slices
            a = h[:N - lag]
            b = h[lag:]
        else:

            good_zeros = np.where(good_mask)[0]
        else:

            good_zeros = np.where(good_mask)[0] + lag_abs

            j = rng.choice(good_zeros)
        else:

            j = rng.choice(zeros)

        h_new = h.copy()
        h_new[i] = 0
        h_new[j] = 1

        # Evaluate
        new_max, new_corr = evaluate_vec(h_new)

        delta = new_max - cur_max
        accept = False
        if delta <= 0:

        else:

            zeros = np.flatnonzero(h == 0)

                best_max = new_max
                best_h = h.copy()

    # final compute c5 bound
    dx = 2.0 / N
    c5 = best_max * dx
    return (best_h.tolist(), float(c5), N)

import time
import numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ------------------------------------------------------------------
    # Determine problem size (must be even for a binary solution)
    # ------------------------------------------------------------------
    try:
        init_vals = np.asarray(initial_h_values, dtype=float)
        init_len = init_vals.size
    except Exception:
        init_vals = None
        init_len = 0

    # Target number of points – a compromise between speed and granularity
    N = 600
    if N % 2:
        N += 1

    # ------------------------------------------------------------------
    # Helper: evaluate max mismatch (integer) and full correlation array
    # ------------------------------------------------------------------
    def eval_vec(vec):
        corr = np.correlate(vec, 1.0 - vec, mode='full')
        return int(corr.max()), corr

    # ------------------------------------------------------------------
    # Build an initial binary vector
    #   – start from a low-discrepancy construction (irrational rotation)
    #   – keep the best out of a few random seeds
    # ------------------------------------------------------------------
    best_vec = None
    best_max = N + 1        # large sentinel

    # use a few random slopes; the construction is: h[i]=1 iff {i*α}<0.5
    for _ in range(40):
        alpha = rng.random()
        frac = (np.arange(N, dtype=float) * alpha) % 1.0
        h = (frac < 0.5).astype(float)

        # enforce exact half mass (may be off by 1 due to rounding)
        cur = int(h.sum())
        target = N // 2
        if cur > target:
            # turn some ones (those closest to 0.5) into zeros
            ones = np.flatnonzero(h)
            d = frac[ones] - 0.5          # distance above 0.5 (positive)
            idx = ones[np.argsort(-d)[:cur - target]]
            h[idx] = 0.0
        elif cur < target:
            zeros = np.flatnonzero(h == 0)
            d = 0.5 - frac[zeros]          # distance below 0.5 (positive)
            idx = zeros[np.argsort(-d)[:target - cur]]
            h[idx] = 1.0

        m, _ = eval_vec(h)
        if m < best_max:
            best_max = m
            best_vec = h.copy()

    # If a parent construction exists, blend it in as an extra candidate
    if init_vals is not None and init_len > 0:
        # repeat / truncate to length N, then threshold to binary
        rep = N // init_len + 1
        cont = np.tile(init_vals, rep)[:N]
        h = np.zeros(N, dtype=float)
        idx = np.argsort(cont)[: N // 2]
        h[idx] = 1.0
        m, _ = eval_vec(h)
        if m < best_max:
            best_max = m
            best_vec = h.copy()

    # ------------------------------------------------------------------
    # Main optimization loop (targeted swaps with simulated annealing)
    # ------------------------------------------------------------------
    h = best_vec.copy()
    cur_max, cur_corr = eval_vec(h)

    # maintain index lists for fast random picks
    ones = np.flatnonzero(h == 1)
    zeros = np.flatnonzero(h == 0)

    best_h = h.copy()
    best_val = cur_max

    # annealing schedule
    T0 = 0.5
    no_improve = 0
    stagnation_limit = 5000

    while True:
        now = time.time()
        elapsed = now - start_time
        if elapsed > budget_s * 0.96:          # leave a small margin
            break
        # linear cooling
        T = T0 * max(0.0, 1.0 - elapsed / budget_s)

        # lag giving the current maximum mismatch
        lag_idx = int(np.argmax(cur_corr))
        lag = lag_idx - (N - 1)                # can be negative

        # compute mismatched 1→0 positions for this lag
        if lag >= 0:
            a = h[: N - lag]
            b = h[lag:]
            bad_mask = (a == 1) & (b == 0)     # 1 that becomes 0 after shift
            good_mask = (a == 0) & (b == 1)    # 0 that becomes 1 after shift
            bad_pos = np.where(bad_mask)[0]
            good_pos = np.where(good_mask)[0]
        else:
            L = -lag
            a = h[L:]
            b = h[: N - L]
            bad_mask = (a == 1) & (b == 0)
            good_mask = (a == 0) & (b == 1)
            bad_pos = np.where(bad_mask)[0] + L
            good_pos = np.where(good_mask)[0] + L

        if bad_pos.size and good_pos.size:
            i = rng.choice(bad_pos)
            j = rng.choice(good_pos)
        else:
            i = rng.choice(ones)
            j = rng.choice(zeros)

        # try the swap
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_max, new_corr = eval_vec(h_new)

        delta = new_max - cur_max
        accept = False
        if delta <= 0:
            accept = True
        elif T > 0:
            if rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            h = h_new
            cur_max = new_max
            cur_corr = new_corr
            # update index lists
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)

            if new_max < best_val:
                best_val = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # occasional random restart if stuck
        if no_improve > stagnation_limit:
            # re‑initialize from a fresh rotation sample
            alpha = rng.random()
            frac = (np.arange(N, dtype=float) * alpha) % 1.0
            h = (frac < 0.5).astype(float)
            cur = int(h.sum())
            target = N // 2
            if cur > target:
                ones_idx = np.flatnonzero(h)
                d = frac[ones_idx] - 0.5
                idx = ones_idx[np.argsort(-d)[:cur - target]]
                h[idx] = 0.0
            elif cur < target:
                zeros_idx = np.flatnonzero(h == 0)
                d = 0.5 - frac[zeros_idx]
                idx = zeros_idx[np.argsort(-d)[:target - cur]]
                h[idx] = 1.0
            cur_max, cur_corr = eval_vec(h)
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            no_improve = 0

    # ------------------------------------------------------------------
    # Final result
    # ------------------------------------------------------------------
    dx = 2.0 / N
    c5_bound = best_val * dx
    return (best_h.tolist(), float(c5_bound), int(N))
