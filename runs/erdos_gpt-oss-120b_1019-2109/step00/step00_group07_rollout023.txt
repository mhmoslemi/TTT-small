import numpy as np, cvxpy as cp, time

def solve_milp(n, time_limit):
    h = cp.Variable(n, boolean=True)
    t = cp.Variable()
    constraints = [cp.sum(h) == n/2]
    # t bounds
    constraints += [t >= 0, t <= 1]  # maybe
    # product vars
    y_vars = []
    # For each k, we will accumulate constraints linking y with h.
    for k in range(1, n):
        length = n - k
        y = cp.Variable(length, boolean=True)
        y_vars.append(y)
        # linking constraints
        constraints += [y <= h[:length]]
        constraints += [y <= h[k:]]
        constraints += [y >= h[:length] + h[k:] - 1]
        # sum overlapping ones
        constraints += [cp.sum(y) * (2.0/n) >= t]  # dx = 2/n
    # objective
    objective = cp.Maximize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.CBC, timeLimit=time_limit)
    return h.value, t.value, prob.status
```

def compute_c5(h):
    n = len(h)

    return np.max(np.correlate(h, 1 - h, mode='full') * dx)

def solve_via_milp(initial_h=None, time_budget=1100):
    start = time.time()
    # Determine n
    if initial_h is not None:
        n = len(initial_h)
    else:
        n = 80
    # Ensure n even
    if n % 2 != 0:
        n += 1
    # Use n points scheme; we can create extra dummy points? Optionally adjust.
    n_points = n
    dx = 2.0 / n_points

    h = cp.Variable(n_points, boolean=True)
    Cmax = cp.Variable()
    constraints = [cp.sum(h) == n_points / 2, Cmax >= 0]

        if length <= 0:
            continue
        h_prefix = h[:length]

        constraints += [y <= h_prefix,

        except Exception:
            # fallback to simple heuristic: use initial_h or random binary
            if initial_h is not None:
                h_sol = np.array(initial_h)

def _solve_milp_binary(n, init, time_limit):
    import cvxpy as cp, numpy as np, time
    dx = 2.0 / n
    h = cp.Variable(n, boolean=True)
    Cmax = cp.Variable()
    constraints = [cp.sum(h) == n/2,
                   Cmax >= 0]
    # for each lag
    for lag in range(1, n):
        length = n - lag
        if length == 0:
            continue
        h_prefix = h[:length]
        h_suffix = h[lag:]
        y = cp.Variable(length, boolean=True)
        constraints += [y <= h_prefix,
                        y <= h_suffix,
                        y >= h_prefix + h_suffix - 1]
        # prefix constraint
        constraints += [ (cp.sum(h_prefix) - cp.sum(y)) * dx <= Cmax ]
        # suffix constraint
        constraints += [ (cp.sum(h_suffix) - cp.sum(y)) * dx <= Cmax ]
    objective = cp.Minimize(Cmax)
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, maxSeconds=time_limit, verbose=False)
        except Exception:
            # fallback: return random binary solution
            h_vals = np.zeros(n)
            h_vals[np.random.choice(n, n//2, replace=False)] = 1.0
            c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full') * dx)
            return h_vals, c5, 'fallback'
    if h.value is None:
        # fallback random
        h_vals = np.zeros(n)
        h_vals[np.random.choice(n, n//2, replace=False)] = 1.0
        c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full') * dx)
        return h_vals, c5, 'fallback_none'
    h_vals = np.rint(h.value).astype(float)
    # ensure exact sum
    current_sum = int(h_vals.sum())
    target = n // 2
    if current_sum != target:
        # adjust by flipping
        diff = target - current_sum
        if diff > 0:
            zero_idxs = np.where(h_vals < 0.5)[0]
            if len(zero_idxs) >= diff:
                h_vals[zero_idxs[:diff]] = 1.0
        elif diff < 0:
            one_idxs = np.where(h_vals > 0.5)[0]
            if len(one_idxs) >= -diff:
                h_vals[one_idxs[: -diff]] = 0.0
    c5 = np.max(np.correlate(h_vals, 1 - h_vals, mode='full') * dx)
    return h_vals, float(c5), prob.status

def _hill_climb(h, current_c5, time_limit):
    import numpy as np, time
    start = time.time()
    n = len(h)
    dx = 2.0 / n
    # Precompute indices
    while time.time() - start < time_limit:
        # choose random indices i (1) and j (0)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h[i] = 0.0
        h[j] = 1.0
        # compute new C
        c = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if c < current_c5 - 1e-8:
            current_c5 = c
            # keep swap, continue
        else:
            # revert
            h[i] = 1.0
            h[j] = 0.0
    return h, float(current_c5)

    np.random.seed(seed)

    start = time.time()

    # determine n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    else:
        n = 80
        h = np.zeros(n)
        ones = np.random.choice(n, n//2, replace=False)
        h[ones] = 1.0

    # ensure sum constraint exactly
    # if sum != n/2, adjust:
    cur_sum = h.sum()
    target = n/2
    # if not integer? ensure n even
    diff = target - cur_sum
    if abs(diff) > 1e-8:

        # compute indices sorted by distance to 0.5 etc.
        # For simplicity, set some entries to 1 or 0
        idx_order = np.argsort(h) # zeros first
        if diff > 0: # need more ones
            idx = idx_order[:int(diff)]
            h[idx] = 1.0
        else:
            idx = idx_order[:int(-diff)]
            h[idx] = 0.0
    # compute initial bound
    dx = 2.0/n
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    best_h = h.copy()

            h[j] += delta

import time, random
import numpy as np

# helper: compute the C5 bound for a given step function h
def _c5_val(h):
    n = h.shape[0]
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))


# Solve a binary MILP: h ∈ {0,1}, sum h = n/2, minimize max_k ∫h(x)(1‑h(x+k))dx
def _solve_binary_milp(n, time_limit):
    try:
        import cvxpy as cp
    except Exception:  # pragma: no cover
        return None, None, "cvxpy_missing"

    dx = 2.0 / n
    h = cp.Variable(n, boolean=True)
    Cmax = cp.Variable()
    cons = [cp.sum(h) == n / 2, Cmax >= 0]

    for lag in range(1, n):
        length = n - lag
        if length == 0:
            continue
        hp = h[:length]          # prefix
        hs = h[lag:]             # shifted suffix
        y = cp.Variable(length, boolean=True)  # product proxy
        cons += [
            y <= hp,
            y <= hs,
            y >= hp + hs - 1,
            (cp.sum(hp) - cp.sum(y)) * dx <= Cmax,
            (cp.sum(hs) - cp.sum(y)) * dx <= Cmax,
        ]

    prob = cp.Problem(cp.Minimize(Cmax), cons)
    try:
        prob.solve(solver=cp.CBC, maxSeconds=time_limit, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLPK_MI, maxSeconds=time_limit, verbose=False)
        except Exception:
            return None, None, "solver_failed"

    if h.value is None:
        return None, None, "no_solution"

    # binary rounding and exact weight correction
    h_val = np.rint(h.value).astype(float)
    target = n // 2
    cur = int(h_val.sum())
    if cur != target:
        diff = target - cur
        if diff > 0:
            zeros = np.where(h_val < 0.5)[0]
            h_val[zeros[:diff]] = 1.0
        else:
            ones = np.where(h_val > 0.5)[0]
            h_val[ones[: -diff]] = 0.0
    return h_val, _c5_val(h_val), prob.status


# Simple greedy/local optimisation on a real‑valued vector.
def _local_improve(h, budget_sec, target=None):
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_val = _c5_val(best_h)
    cur_h = h.copy()
    cur_val = best_val
    start = time.time()
    while time.time() - start < budget_sec:
        # recompute all C_k values
        corr = np.correlate(cur_h, 1 - cur_h, mode="full") * dx
        lag = int(np.argmax(corr) - (n - 1))  # zero shift at n-1
        if target is not None and best_val <= target:
            break
        # overlapping index range for this lag
        lo = max(0, -lag)
        hi = min(n, n - lag)
        if hi - lo < 2:
            # cannot improve with this lag, random small perturbation
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            delta = min(cur_h[i], 1 - cur_h[j]) * 0.2
            if delta <= 0:
                continue
            cur_h[i] -= delta
            cur_h[j] += delta
        else:
            # pick a source i with h_i>0 and target j=i+lag with h_j<1
            i = np.random.randint(lo, hi)
            j = i + lag
            if cur_h[i] <= 0 or cur_h[j] >= 1:
                # try a few random trials
                for _ in range(5):
                    i = np.random.randint(lo, hi)
                    j = i + lag
                    if cur_h[i] > 0 and cur_h[j] < 1:
                        break
                else:
                    continue
            delta = min(cur_h[i], 1 - cur_h[j]) * 0.2
            if delta <= 0:
                continue
            cur_h[i] -= delta
            cur_h[j] += delta

        new_val = _c5_val(cur_h)
        # accept only if improvement (simple descent)
        if new_val + 1e-10 < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i] += delta
            cur_h[j] -= delta
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h, its C5 bound, and number of sample points."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()

    # ------------------------------------------------------------------
    # Determine problem size (use provided initial array if present)
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        h0 = np.array(init, dtype=float)
        n = h0.shape[0]
    else:
        n = 80
        h0 = np.zeros(n)
        ones = np.random.choice(n, n // 2, replace=False)
        h0[ones] = 1.0

    # Ensure exact weight (might be slightly off due to rounding of the parent)
    target = n // 2
    cur = int(h0.sum())
    if cur != target:
        diff = target - cur
        if diff > 0:
            zeros = np.where(h0 < 0.5)[0]
            h0[zeros[:diff]] = 1.0
        else:
            ones = np.where(h0 > 0.5)[0]
            h0[ones[: -diff]] = 0.0

    # ------------------------------------------------------------------
    # First attempt: binary MILP (quick, gives a good starting point)
    # ------------------------------------------------------------------
    time_left = max(0.0, budget_s - (time.time() - start_time))
    milp_time = min(300.0, time_left * 0.6)  # reserve time for refinement
    h_opt, c5_opt, status = _solve_binary_milp(n, milp_time)

    if h_opt is None:                     # fallback to the parent / random start
        h_opt = h0.copy()
        c5_opt = _c5_val(h_opt)

    # ------------------------------------------------------------------
    # Second phase: local greedy refinement on the (possibly) fractional vector
    # ------------------------------------------------------------------
    time_left = max(0.0, budget_s - (time.time() - start_time))
    if time_left > 2.0:
        # try to improve the bound further
        h_ref, c5_ref = _local_improve(h_opt, time_left * 0.9, target=0.3808)
        if c5_ref < c5_opt:
            h_opt, c5_opt = h_ref, c5_ref

    # final sanity check: respect constraints
    dx = 2.0 / n
    h_opt = np.clip(h_opt, 0.0, 1.0)
    cur_sum = h_opt.sum()
    if abs(cur_sum - target) > 1e-6:
        # simple proportional correction
        diff = target - cur_sum
        adjust = diff / n
        h_opt += adjust
        h_opt = np.clip(h_opt, 0.0, 1.0)

    final_c5 = _c5_val(h_opt)

    return h_opt, final_c5, n
