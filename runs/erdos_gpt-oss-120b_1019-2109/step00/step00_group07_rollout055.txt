def propose_transfer(h):
    i, j = random.sample(range(N), 2)
    # possible transfer from i to j (positive delta)
    max_pos = min(h[i], 1 - h[j])  # can't exceed these
    max_neg = min(h[j], 1 - h[i])  # transfer opposite direction
    if max_pos + max_neg == 0:
        return None  # no feasible move
    if random.random() < 0.5:
        # transfer from i to j
        delta = random.random() * max_pos
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
    else:
        delta = random.random() * max_neg
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
    return h_new
```

def project_onto_capped_simplex(v, S):
    # Find λ such that sum_i min(1, max(0, v_i + λ)) = S.

        if s > S:
            hi = mid
        else:
            lo = mid

def _project_capped_simplex(v, target_sum):
    """Project vector v to the set {0≤x≤1, sum(x)=target_sum}."""
    v = np.clip(v, 0.0, 1.0)
    lo = -np.max(v) - 10.0
    hi = 1.0 - np.min(v) + 10.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    return proj

def _c5_value(h, dx):
    """Return the C5 value for a step function h (discrete samples)."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)

def _worst_shift_index(corr, n):
    """Return the lag (index relative to zero) where the correlation is maximal."""
    idx = int(corr.argmax())
    return idx - (n - 1)

def _subgradient_step(h, k):
    """Compute a sub‑gradient direction for the worst shift k."""
    n = h.size
    grad = np.ones(n)
    if k >= 0:
        # i ≤ n‑1‑k  -> subtract h[i+k]
        if n - k > 0:
            grad[: n - k] -= h[k:]
        # i ≥ k      -> subtract h[i‑k]
        if k < n:
            grad[k:] -= h[: n - k]
    else:
        pk = -k
        if n - pk > 0:
            grad[: n - pk] -= h[pk:]
        if pk < n:
            grad[pk:] -= h[: n - pk]
    return grad

        n_pts = 80
        x = np.linspace(0.0, 2.0, n_pts, endpoint=False) + 1.0 / n_pts
        h0 = np.minimum(x, 2.0 - x)  # triangle rising to 1 then falling
    n = h0.size
    dx = 2.0 / n
    target_sum = n / 2.0                 # because ∫h = 1  ⇒  sum(h)*dx = 1
    h = _project_capped_simplex(h0, target_sum)

    best_h = h.copy()
    best_c5 = _c5_value(best_h, dx)

    start_time = time.time()
    it = 0
    # parameters for the hybrid search
    while time.time() - start_time < budget_s - 0.5:
        # evaluate current worst shift
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        cur_c5 = float(corr_full.max() * dx)
        k = _worst_shift_index(corr_full, n)

        # decide whether to do a deterministic sub‑gradient step or a random move
        if random.random() < 0.85:                # sub‑gradient most of the time
            grad = _subgradient_step(h, k)
            # diminishing step size
            eta = 0.5 / (1.0 + it * 0.001)
            h_new = _project_capped_simplex(h - eta * grad, target_sum)
        else:                                      # occasional random transfer
            i, j = random.sample(range(n), 2)
            # amount we can move from i to j (keeping bounds)
            max_pos = min(h[i], 1.0 - h[j])
            max_neg = min(h[j], 1.0 - h[i])
            if max_pos > 0 and random.random() < 0.5:
                delta = random.random() * max_pos
                h_new = h.copy()
                h_new[i] -= delta
                h_new[j] += delta
                h_new = _project_capped_simplex(h_new, target_sum)
            elif max_neg > 0:
                delta = random.random() * max_neg
                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta
                h_new = _project_capped_simplex(h_new, target_sum)
            else:
                it += 1
                continue

        new_c5 = _c5_value(h_new, dx)

        # accept if better, otherwise with a tiny probability (simulated‑annealing flavour)
        if new_c5 < cur_c5 or random.random() < 0.001:
            h = h_new
            cur_c5 = new_c5

        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # every few thousand iterations try a fresh random perturbation around the current best
        if it % 5000 == 0 and it > 0:
            pert = np.random.randn(n) * 0.05
            candidate = _project_capped_simplex(best_h + pert, target_sum)
            cand_c5 = _c5_value(candidate, dx)
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                h = candidate.copy()

        it += 1

    # final projection and evaluation
    best_h = _project_capped_simplex(best_h, target_sum)
    best_c5 = _c5_value(best_h, dx)

    return (best_h.tolist(), float(best_c5), int(n))
