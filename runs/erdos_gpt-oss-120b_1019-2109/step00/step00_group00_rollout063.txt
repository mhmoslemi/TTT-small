import time
import numpy as np
import cvxpy as cp


def _evaluate(h):
    """Return the C5 value for a step vector h."""
    N = h.size
    dx = 2.0 / N
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _round_binary(x, target_sum):
    """Round a (near‑binary) vector to exact 0/1 with given sum."""
    x = np.rint(x).astype(int)
    cur = int(x.sum())
    if cur == target_sum:
        return x.astype(float)
    # adjust based on fractional parts of the original values
    frac = x.astype(float) - np.floor(x)
    order = np.argsort(-frac)  # descending
    x[:] = 0
    x[order[:target_sum]] = 1
    return x.astype(float)


def _solve_milp(N, time_limit):
    """Solve the integer program for a given length N.
    Returns (h, c5) or (None, None) on failure / timeout."""
    n = N // 2
    # decision variables
    x = cp.Variable(N, boolean=True)                # 0/1 pattern
    t = cp.Variable(integer=True)                   # maximal overlap (integer)
    Y = cp.Variable((N, N), boolean=True)           # pairwise AND matrix

    constraints = [cp.sum(x) == n, t >= 0]

    # diagonal of Y must equal x (x_i * x_i = x_i)
    for i in range(N):
        constraints.append(Y[i, i] == x[i])

    # linearisation of AND for i<j
    for i in range(N):
        for j in range(i + 1, N):
            constraints.append(Y[i, j] <= x[i])
            constraints.append(Y[i, j] <= x[j])
            constraints.append(Y[i, j] >= x[i] + x[j] - 1)
            constraints.append(Y[j, i] == Y[i, j])

    # max‑overlap constraints for each positive shift
    for k in range(1, N):
        overlap = cp.sum([Y[i, i + k] for i in range(N - k)])
        constraints.append(t >= n - overlap)   # O_k = n - overlap <= t

    prob = cp.Problem(cp.Minimize(t), constraints)

    # try a few MILP solvers, stop as soon as one succeeds
    solvers = [cp.CBC, cp.GLPK_MI, cp.SCIP, cp.ECOS_BB]
    for s in solvers:
        try:
            prob.solve(solver=s, warm_start=True, verbose=False)
            if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
                break
        except Exception:
            continue

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None, None

    # extract a clean binary vector
    h = _round_binary(x.value, n)
    c5 = _evaluate(h)
    return h, c5

    """Return a step function (array), its C5 bound and length."""
    start_time = time.time()
    rng = np.random.default_rng(seed)

    target_c5 = 0.38085
    best_h = None
    best_c5 = float("inf")
    best_N = None

    # if a parent solution is supplied, use it as a baseline
    try:

        c5_init = _evaluate(init)
        if c5_init < best_c5:

    except Exception:
        pass

    # candidate lengths (even numbers, not too large)
    candidate_N = [48, 56, 64, 72, 80]

    for N in candidate_N:
        # leave some time for later steps
        if time.time() - start_time > budget_s * 0.9:
            break
        # simple random start for the fallback heuristic
        h_fallback = np.zeros(N)
        n = N // 2
        h_fallback[:n] = 1.0
        rng.shuffle(h_fallback)

        # try the exact MILP first
        remaining = budget_s - (time.time() - start_time)
        h, c5 = _solve_milp(N, remaining)

        # if MILP failed, use a quick local‑search heuristic
        if h is None:

            best_local = _evaluate(h)
            # hill‑climb: swap a 1 and a 0 if it improves the max overlap
            for _ in range(2000):

                new_c5 = _evaluate(h)

            c5 = best_local
