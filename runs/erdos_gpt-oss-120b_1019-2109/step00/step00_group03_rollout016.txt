import time
import numpy as np
import random

def _compute_c5(vec, dx):
    # vec : 0/1 integer array
    corr = np.correlate(vec.astype(float), 1.0 - vec.astype(float), mode="full")
    return float(np.max(corr) * dx)

def _binarize_initial(arr, target_sum):
    # turn any initial array into a binary vector with exactly target_sum ones
    arr = np.asarray(arr, dtype=float)
    # pick the largest entries to become 1
    idx = np.argsort(-arr)  # descending
    bin_vec = np.zeros_like(arr, dtype=np.int8)
    bin_vec[idx[:target_sum]] = 1
    return bin_vec

def _random_binary(n, target_sum):
    vec = np.zeros(n, dtype=np.int8)
    vec[:target_sum] = 1
    np.random.shuffle(vec)
    return vec

def _greedy_local_opt(vec, dx, time_deadline):
    """Iteratively apply the best improving swap until none exists."""
    best_vec = vec.copy()
    best_val = _compute_c5(best_vec, dx)
    # cheap bound on number of full scans
    while True:
        if time.time() >= time_deadline:
            break
        ones = np.flatnonzero(best_vec == 1)
        zeros = np.flatnonzero(best_vec == 0)
        improved = False
        best_swap = None
        best_swap_val = best_val
        # try all swaps (35*35 ≈ 1225) – cheap enough
        for i in ones:
            for j in zeros:
                cand = best_vec.copy()
                cand[i] = 0
                cand[j] = 1
                val = _compute_c5(cand, dx)
                if val < best_swap_val - 1e-12:
                    best_swap_val = val
                    best_swap = (i, j, cand)
        if best_swap is None:
            break
        # apply the improvement
        best_vec = best_swap[2]
        best_val = best_swap_val
        improved = True
        if not improved:
            break
    return best_vec, best_val

    if init_arr is not None:

    assert n % 2 == 0, "n_points must be even"
    target_sum = n // 2
    dx = 2.0 / n

    # ------------------------------------------------------------------
    # starting solution

    if init_arr is not None:
        cur_vec = _binarize_initial(init_arr, target_sum)
    else:

    cur_val = _compute_c5(cur_vec, dx)
    best_vec = cur_vec.copy()
    best_val = cur_val

    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    # annealing parameters
    T_start = 0.04
    T_end = 1e-6

    steps_without_improve = 0
