def project_onto_capped_simplex(v, target_sum):

    # returns w satisfying sum(w)=target_sum, 0≤w≤1, minimizing Euclidean distance.
    n = v.shape[0]

        s = w.sum()
        if s > target_sum:

        else:

def hill_climb_binary(h_start, target, time_limit, start_time):
    n = len(h_start)
    # ensure binary
    h = (h_start >= 0.5).astype(float)
    # adjust to have exactly n/2 ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    cur_ones = len(ones)
    target_ones = n // 2
    if cur_ones > target_ones:
        to_zero = np.random.choice(ones, cur_ones - target_ones, replace=False)
        h[to_zero] = 0.0
    elif cur_ones < target_ones:
        to_one = np.random.choice(zeros, target_ones - cur_ones, replace=False)
        h[to_one] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_h = h.copy()
    cur_c = best_c
    while time.time() - start_time < time_limit:
        # compute max shift
        cor = np.correlate(cur_h, 1 - cur_h, mode='full')
        max_idx = cor.argmax()
        shift = max_idx - (n - 1)
        # generate swap targeting mismatches
        i_start = max(0, -shift)
        i_end = min(n, n - shift)
        idx_range = np.arange(i_start, i_end)
        bad = (cur_h[idx_range] == 1) & (cur_h[idx_range + shift] == 0)
        good = (cur_h[idx_range] == 0) & (cur_h[idx_range + shift] == 1)
        bad_idx = idx_range[bad]
        good_idx = idx_range[good]
        if bad_idx.size > 0 and good_idx.size > 0:
            i = np.random.choice(bad_idx)
            j = np.random.choice(good_idx)
        else:
            # random swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
        # attempt Swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c = compute_c5(cur_h)
        if new_c <= cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = 1.0, 0.0
        # possible early break
        if best_c <= target:
            break
    return best_h, best_c

Now compute_gradient:

def compute_gradient(h, shift):
    n = len(h)
    grad = np.ones(n)
    if shift >= 0:
        if shift < n:
            grad[:n-shift] -= h[shift:]  # i+shift term
            grad[shift:] -= h[:n-shift]  # i-shift term
    else:
        k = -shift
        if k < n:
            grad[k:] -= h[:n-k]          # i+shift term (i - k)
            grad[:n-k] -= h[k:]          # i-shift term (i + k)
    return grad

import time
import numpy as np

# ---------- helper functions ----------
def _c5_of_h(h):
    """Compute C5 = max correlation * dx for a given h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _max_shift_info(h):
    """Return (shift, max_corr_value, full_corr_array)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = corr.argmax()
    # shift = idx - (n-1)
    return idx - (n - 1), corr[idx], corr


def _gradient_worst_shift(h, shift):
    """Gradient of the worst shift correlation w.r.t. h."""
    n = h.shape[0]
    grad = np.ones(n)           # derivative of Σ h_i part
    if shift >= 0:
        if shift < n:
            grad[: n - shift] -= h[shift:]          # -h_{i+shift}
            grad[shift:] -= h[: n - shift]          # -h_{i-shift}
    else:
        k = -shift
        if k < n:
            grad[k:] -= h[: n - k]                  # -h_{i+shift} (i+shift = i-k)
            grad[: n - k] -= h[k:]                  # -h_{i-shift} (i-shift = i+k)
    return grad


def _project_capped_simplex(v, target_sum):
    """
    Euclidean projection of v onto the set {w : 0<=w_i<=1, Σ w_i = target_sum}.
    Uses a simple bisection on the threshold λ for w_i = clip(v_i - λ,0,1).
    """
    lo = -np.max(v) - 1.0
    hi = np.max(v) + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - hi, 0.0, 1.0)


def _hill_climb_binary(start_h, time_limit, target):
    """
    Greedy hill‑climb on binary vectors (0/1) that keeps exactly half the mass.
    """
    n = start_h.shape[0]
    target_ones = n // 2
    h = (start_h >= 0.5).astype(float)

    # enforce exact half‑ones (or half‑ones + a 0.5 entry for odd n)
    cur_ones = int(h.sum())
    if cur_ones > target_ones:
        excess = cur_ones - target_ones
        idx = np.where(h == 1)[0]
        rm = np.random.choice(idx, excess, replace=False)
        h[rm] = 0.0
    elif cur_ones < target_ones:
        deficit = target_ones - cur_ones
        idx = np.where(h == 0)[0]
        add = np.random.choice(idx, deficit, replace=False)
        h[add] = 1.0

    best_h = h.copy()
    best_c = _c5_of_h(best_h)
    cur_h = h.copy()
    cur_c = best_c

    end_time = time.time() + time_limit
    while time.time() < end_time:
        shift, _, _ = _max_shift_info(cur_h)
        # overlapping region for this shift
        i_start = max(0, -shift)
        i_end = min(n, n - shift)
        idxs = np.arange(i_start, i_end)

        bad = (cur_h[idxs] == 1) & (cur_h[idxs + shift] == 0)
        good = (cur_h[idxs] == 0) & (cur_h[idxs + shift] == 1)

        bad_idx = idxs[bad]
        good_idx = idxs[good]

        if bad_idx.size and good_idx.size:
            i = np.random.choice(bad_idx)
            j = np.random.choice(good_idx)
        else:
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)

        # try the swap
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c = _c5_of_h(cur_h)

        if new_c <= cur_c:
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = 1.0, 0.0

        if best_c <= target:
            break
    return best_h, best_c


def _simulated_anneal(start_h, time_limit, target):
    """
    Very simple simulated‑annealing using swaps, works on binary vectors.
    """
    n = start_h.shape[0]
    target_ones = n // 2
    h = (start_h >= 0.5).astype(float)

    # enforce exact half‑ones
    cur_ones = int(h.sum())
    if cur_ones > target_ones:
        excess = cur_ones - target_ones
        idx = np.where(h == 1)[0]
        rm = np.random.choice(idx, excess, replace=False)
        h[rm] = 0.0
    elif cur_ones < target_ones:
        deficit = target_ones - cur_ones
        idx = np.where(h == 0)[0]
        add = np.random.choice(idx, deficit, replace=False)
        h[add] = 1.0

    best_h = h.copy()
    best_c = _c5_of_h(best_h)

    cur_h = h.copy()
    cur_c = best_c
    T = 0.02
    decay = 0.99995

    end_time = time.time() + time_limit
    while time.time() < end_time:
        # propose a random swap
        ones = np.where(cur_h == 1)[0]
        zeros = np.where(cur_h == 0)[0]
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        cur_h[i], cur_h[j] = 0.0, 1.0
        new_c = _c5_of_h(cur_h)
        delta = new_c - cur_c
        if delta <= 0 or np.random.rand() < np.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # revert
            cur_h[i], cur_h[j] = 1.0, 0.0

        T *= decay
        if T < 1e-5:
            T = 0.02

        if best_c <= target:
            break
    return best_h, best_c
