def evaluate_int(h):
    # h: binary np.array uint8
    # complementary
    g = 1 - h
    corr = np.correlate(h, g, mode='full')

def simulated_annealing(h0, time_limit):
    best = h0.copy()
    best_val = evaluate_int(best)
    current = best.copy()
    cur_val = best_val
    start = time.time()
    T = 1.0
    # schedule maybe T and exponent decay
    while time.time() - start < time_limit:
        # generate neighbor by swapping two positions of opposite bits
        # use random idx
        ones = np.where(current)[0]
        zeros = np.where(~current)[0] (or current==0)
        i = random.choice(ones)
        j = random.choice(zeros)
        neighbor = current.copy()
        neighbor[i] = 0
        neighbor[j] = 1
        # Evaluate neighbor
        val = evaluate_int(neighbor)
        delta = val - cur_val
        if delta < 0 or random.random() < np.exp(-delta / T):
            current = neighbor
            cur_val = val
            if val < best_val:
                best = neighbor.copy()
                best_val = val
                # optional print
        # update temperature
        # Maybe linear cooling:
        T = max(T * 0.9999, 1e-5)
    return best, best_val
```

import numpy as np, random, time, math

def evaluate_corr(h):
    corr = np.correlate(h, 1-h, mode='full')
    return int(corr.max())

def run(seed=42, budget_s=1100):
    random.seed(seed)
    np.random.seed(seed)
    # Determine n_points
    if 'initial_h_values' in globals():
        init = np.array(initial_h_values, dtype=np.float64)
        n = len(init)
    else:
        n = 70
    # Ensure integer n even; else adjust.
    # Define target number of ones
    num_ones = n // 2
    dx = 2.0 / n
    best_h = None
    best_val = None
    start_time = time.time()
    # Initialize from initial_h_values if possible: binarize to preserve sum.
    def to_binary(vec):
        # ensure sum equals num_ones by thresholding top values.
        inds = np.argsort(vec)[::-1]  # descending
        h = np.zeros_like(vec, dtype=np.uint8)
        h[inds[:num_ones]] = 1
        return h
    if 'initial_h_values' in globals():
        init_bin = to_binary(initial_h_values)
        cur_h = init_bin
        cur_val = evaluate_corr(cur_h)
        best_h = cur_h.copy()
        best_val = cur_val
    else:
        cur_h = None
        cur_val = None
    # Perform multiple random restarts with SA
    total_iters = 0
    # parameters
    T_initial = 2.0
    # dynamic schedule
    while time.time() - start_time < budget_s * 0.95:
        # generate random start if needed
        if cur_h is None or random.random() < 0.05:
            # random binary
            h = np.zeros(n, dtype=np.uint8)
            h[:num_ones] = 1
            np.random.shuffle(h)
            cur_h = h
            cur_val = evaluate_corr(cur_h)
        # Simulated annealing loop
        T = T_initial
        # inner loop limited by time and iteration count e.g., 5000 steps
        while time.time() - start_time < budget_s * 0.95:
            # pick random 1 position and random 0 position
            ones_idx = np.flatnonzero(cur_h)
            zeros_idx = np.flatnonzero(1 - cur_h)
            i = random.choice(ones_idx)
            j = random.choice(zeros_idx)
            # swap
            new_h = cur_h.copy()
            new_h[i] = 0
            new_h[j] = 1
            new_val = evaluate_corr(new_h)
            delta = new_val - cur_val
            if delta <= 0 or random.random() < math.exp(-delta / T):
                cur_h = new_h
                cur_val = new_val
                if best_val is None or new_val < best_val:
                    best_h = new_h.copy()
                    best_val = new_val
                    # early break if we achieve very low bound
                    # threshold - we desire <=13 maybe.
                    if best_val <= 13:
                        # return early
                        return (best_h.astype(float).tolist(), best_val*dx, n)
            # cooling
            T = max(T * 0.9995, 1e-4)
            # optional break if T low and no improvement for many steps
            total_iters += 1
            if total_iters % 10000 == 0:
                # optionally random restart if stuck
                if random.random() < 0.1:
                    break
    # final conversion to float list
    if best_h is None:
        # fallback to initial if exists else uniform
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values, dtype=np.float64)
            best_val = evaluate_corr(to_binary(initial_h_values))
        else:
            # trivial pattern half ones at start
            best_h = np.zeros(n, dtype=np.float64)
            best_h[:num_ones] = 1.0
            best_val = evaluate_corr(best_h.astype(np.uint8))
    return (best_h.astype(float).tolist(), best_val * dx, n)

def _evaluate_max_corr(h: np.ndarray) -> float:
    """Return the maximal overlap count (without multiplying by dx)."""
    # Correlation of h with (1-h) over the full range
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")))


def _project_onto_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to satisfy sum(v) == target_sum."""
    v = np.clip(v, 0.0, 1.0)
    # Simple water-filling style correction
    for _ in range(20):
        diff = v.sum() - target_sum
        if abs(diff) < 1e-12:
            break
        mask = (v > 0.0) & (v < 1.0)
        cnt = int(mask.sum())
        if cnt == 0:
            break
        v[mask] -= diff / cnt
        v = np.clip(v, 0.0, 1.0)
    return v


def _gradient_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the worst‑shift overlap w.r.t. h."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        end = n - shift
        # linear part: +1 for i where i+shift exists
        g[:end] += 1.0
        # -h_i * h_{i+shift} part
        g[:end] -= h[shift:]          # derivative w.r.t. h_i (first factor)
        g[shift:] -= h[:end]          # derivative w.r.t. h_{i+shift} (second factor)
    else:
        s = -shift

        g[start:] += 1.0
        g[start:] -= h[:length]       # derivative w.r.t. h_i (first factor)
        g[:length] -= h[s:]           # derivative w.r.t. h_{i+shift} (second factor)
    return g

            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    final_bound = best_val * dx
    return (best_h.astype(float).tolist(), final_bound, n)
