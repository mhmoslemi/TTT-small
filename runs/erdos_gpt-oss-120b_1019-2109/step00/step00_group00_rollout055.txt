def optimize(initial_h, time_limit):
    N = len(initial_h)
    target_sum = N/2.0
    current_h = initial_h.copy()
    current_score = evaluate(current_h)
    best_h = current_h.copy()
    best_score = current_score
    start = time.time()
    while time.time() - start < time_limit:
        # choose random i,j
        i, j = np.random.choice(N, 2, replace=False)
        # random delta
        max_inc_i = 1 - current_h[i]
        max_dec_i = current_h[i]
        max_inc_j = 1 - current_h[j]
        max_dec_j = current_h[j]
        # propose moving amount from i to j or vice versa
        # random direction
        if np.random.rand() < 0.5:
            # move from i to j
            max_delta = min(max_inc_j, max_dec_i)
            if max_delta <= 0:
                continue
            delta = (np.random.rand() * max_delta)
        else:
            # move from j to i
            max_delta = min(max_inc_i, max_dec_j)
            if max_delta <= 0:
                continue
            delta = -(np.random.rand() * max_delta)
        # apply delta
        new_h = current_h.copy()
        new_h[i] -= delta
        new_h[j] += delta
        # new_h sum unchanged

        new_score = evaluate(new_h)

        # Simple hill climbing: accept if better
        if new_score < current_score:
            current_h = new_h

import time, math, random
import numpy as np

def _proj_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x: 0<=x<=1, sum(x)=target}."""
    lo = -np.max(v)          # makes all v+lam <= 0
    hi = 1.0 - np.min(v)     # makes all v+lam >= 1
    for _ in range(60):
        lam = (lo + hi) * 0.5
        s = np.clip(v + lam, 0.0, 1.0).sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _make_h(mask: np.ndarray, half_idx: int) -> np.ndarray:
    """Construct the h-vector from a binary mask and the index of the half‑value."""
    h = mask.astype(float)
    h[half_idx] = 0.5
    return h

def _random_binary_half(N: int):
    """Random binary mask with (N‑1)//2 ones and a single 0.5 entry."""
    half_idx = random.randrange(N)
    idx = np.arange(N)
    idx = idx[idx != half_idx]
    ones = np.random.choice(idx, size=(N - 1) // 2, replace=False)
    mask = np.zeros(N, dtype=bool)
    mask[ones] = True
    return mask, half_idx

def _evaluate(mask: np.ndarray, half_idx: int) -> float:
    """Return the Erdős C₅ value for the vector defined by mask/half_idx."""
    h = _make_h(mask, half_idx)
    return float(evaluate_erdos_solution(h))

def _hill_climb(mask: np.ndarray, half_idx: int, time_allowed: float):
    """Simple hill‑climbing / simulated‑annealing search for a better mask."""
    N = len(mask)
    best_mask = mask.copy()
    best_half = half_idx
    best_score = _evaluate(mask, half_idx)

    cur_mask = mask.copy()
    cur_half = half_idx
    cur_score = best_score

    start = time.time()
    it_without_improve = 0

    while time.time() - start < time_allowed:
        # choose move type
        if random.random() < 0.5:
            # swap a 1 and a 0 (skip the half position)
            ones = np.where(cur_mask)[0]
            zeros = np.where(~cur_mask)[0]
            zeros = zeros[zeros != cur_half]
            if ones.size == 0 or zeros.size == 0:
                continue
            i = int(random.choice(ones))
            j = int(random.choice(zeros))
            new_mask = cur_mask.copy()
            new_mask[i] = False
            new_mask[j] = True
            new_half = cur_half
        else:
            # move the half entry to a new location
            j = random.randrange(N - 1)
            if j >= cur_half:
                j += 1
            new_mask = cur_mask.copy()
            if cur_mask[j]:
                # j currently a 1 -> become 0, old half becomes 1
                new_mask[cur_half] = True
                new_mask[j] = False
                new_half = j
            else:
                # j is a 0 -> old half becomes 0, new half is 0.5
                new_mask[cur_half] = False
                new_half = j
        new_score = _evaluate(new_mask, new_half)

        # acceptance: always accept improvement, otherwise with tiny probability
        if new_score < cur_score:
            cur_mask, cur_half, cur_score = new_mask, new_half, new_score
            it_without_improve = 0
            if new_score < best_score:
                best_mask, best_half, best_score = new_mask.copy(), new_half, new_score
        else:
            # simulated‑annealing style temperature (slow linear cooling)
            elapsed = time.time() - start
            prog = elapsed / time_allowed
            T = 0.002 * (1.0 - prog) + 1e-9
            if random.random() < math.exp(-(new_score - cur_score) / T):
                cur_mask, cur_half, cur_score = new_mask, new_half, new_score
            it_without_improve += 1

        # occasional random restart to escape deep plateaus
        if it_without_improve > 2000:
            cur_mask, cur_half = _random_binary_half(N)
            cur_score = _evaluate(cur_mask, cur_half)
            it_without_improve = 0

        # early exit if excellent bound reached
        if best_score < 0.3808:
            break

    return _make_h(best_mask, best_half), best_score

def run(seed=42, budget_s=1100, **kwargs):
    """
    Search for a step function h on [0,2] with integral 1 that
    minimizes max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size
    init = globals().get('initial_h_values')
    if init is not None:
        N = int(init.shape[0])
        init_h = np.clip(np.asarray(init, dtype=float), 0.0, 1.0)
        target_sum = N / 2.0
        # enforce exact sum (tiny numerical drift possible)
        if abs(init_h.sum() - target_sum) > 1e-12:
            init_h = _proj_sum_box(init_h, target_sum)
        best_h = init_h
        best_score = float(evaluate_erdos_solution(best_h))
    else:
        N = 45                     # default size when no parent is supplied
        target_sum = N / 2.0
        # start from a random binary+half configuration
        mask, half = _random_binary_half(N)
        best_h = _make_h(mask, half)
        best_score = float(evaluate_erdos_solution(best_h))

    start_time = time.time()
    # Main search loop – repeated short hill‑climbs until budget expires
    while time.time() - start_time < budget_s * 0.94:
        remaining = budget_s - (time.time() - start_time)
        # allocate a small slice of time to each climb
        slice_time = min(1.5, remaining)
        mask, half = _random_binary_half(N)
        cand_h, cand_score = _hill_climb(mask, half, slice_time)
        if cand_score < best_score:
            best_h, best_score = cand_h, cand_score
            # if we already beat the target, we can stop early
            if best_score < 0.3808:
                break

    # final projection to guarantee feasibility (tiny numeric drift may appear)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _proj_sum_box(best_h, target_sum)
    best_score = float(evaluate_erdos_solution(best_h))

    return (best_h.tolist(), best_score, int(N))
