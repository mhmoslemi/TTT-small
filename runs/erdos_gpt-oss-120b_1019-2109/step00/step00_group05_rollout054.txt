import time, math
import numpy as np

def _c5_score(arr, dx):
    """Return the C5 value for a step function given by arr."""
    # correlation of arr with (1-arr) over all integer shifts
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return float(np.max(corr) * dx)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise a binary candidate
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        N = h0.size
        target = N // 2
        # force binary while keeping the total number of ones fixed
        bin_h = (h0 >= 0.5).astype(np.float64)
        cur = int(bin_h.sum())
        diff = target - cur
        if diff > 0:
            zeros = np.where(bin_h == 0)[0]
            rng.shuffle(zeros)
            bin_h[zeros[:diff]] = 1.0
        elif diff < 0:
            ones = np.where(bin_h == 1)[0]
            rng.shuffle(ones)
            bin_h[ones[: -diff]] = 0.0
        h = bin_h
    else:
        # fall‑back: 140 points, exactly half ones
        N = 140
        target = N // 2
        h = np.zeros(N, dtype=np.float64)
        h[rng.choice(N, size=target, replace=False)] = 1.0

    dx = 2.0 / N

    # ------------------------------------------------------------------
    # evaluate the starting point
    # ------------------------------------------------------------------
    best_h = h.copy()
    best_score = _c5_score(best_h, dx)
    cur_h = h.copy()
    cur_score = best_score

    start = time.time()
    deadline = start + budget_s * 0.96  # a small safety margin

    # parameters for the stochastic local search
    no_improve = 0          # counts iterations without improvement
    TRIALS = 80             # how many random swaps are examined per outer step
    RESTART_AFTER = 400    # give up and restart after this many non‑improving rounds

    while time.time() < deadline:
        # positions of current ones and zeros
        ones = np.where(cur_h == 1.0)[0]
        zeros = np.where(cur_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # try a batch of random swaps and keep the best one
        best_local_h = None
        best_local_score = cur_score

        for _ in range(TRIALS):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cand = cur_h.copy()
            cand[i] = 0.0
            cand[j] = 1.0
            sc = _c5_score(cand, dx)
            if sc < best_local_score:

                best_local_h = cand

        if best_local_h is not None:
            cur_h = best_local_h
            cur_score = best_local_score
