def compute_f_and_grad(h):
    n = len(h)
    # compute all f_s for s from -(n-1) .. (n-1)
    # We'll compute using correlation:
    f = np.correlate(h, 1 - h, mode='full')
    # shape (2n-1).
    # Now compute gradients for each shift.
    grads = np.zeros((2*n-1, n))
    # For each offset idx and shift s = idx - (n-1), compute gradient
    for idx, s in enumerate(range(-(n-1), n)):
        # compute gradient for shift s
        grad = np.ones(n)
        if s >= 0:
            # subtract h[i+s] for i < n-s
            grad[:n-s] -= h[s:]
            # subtract h[i-s] for i >= s
            grad[s:] -= h[:n-s]
        else:
            s2 = -s
            # shift negative
            grad[s2:] -= h[:n-s2]
            grad[:n-s2] -= h[s2:]
        grads[idx] = grad
    return f, grads

def gradient_optimize(h, max_iters=2000):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    current_h = h.copy()
    for it in range(1, max_iters+1):
        # compute correlation
        f = np.correlate(current_h, 1 - current_h, mode='full')
        fmax = f.max()
        # compute gradient for worst shifts
        max_idxs = np.where(np.abs(f - fmax) < 1e-9)[0]
        grad = np.zeros_like(current_h)
        for idx in max_idxs:
            s = idx - (n - 1)
            # compute gradient for shift s
            g = np.ones(n)
            if s >= 0:
                g[:n-s] -= current_h[s:]
                g[s:] -= current_h[:n-s]
            else:
                s2 = -s
                g[s2:] -= current_h[:n-s2]
                g[:n-s2] -= current_h[s2:]
            grad += g
        grad /= len(max_idxs)
        # step size schedule
        lr = 0.1 / np.sqrt(it)
        new_h = current_h - lr * grad
        # project
        new_h = project_onto_constraints(new_h)
        # evaluate
        f_new = np.correlate(new_h, 1 - new_h, mode='full')
        val_new = f_new.max() * dx
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_h = new_h.copy()
        # accept new_h if improvement else maybe keep anyway (or accept if val_new <= fmax*dx?).
        # We'll accept only if val_new <= fmax*dx (non-increase) else keep old.
        if val_new <= fmax * dx + 1e-12:
            current_h = new_h
        else:
            # with small probability, accept for exploration
            if np.random.rand() < 0.01:
                current_h = new_h
    return best_h, best_val

    n = len(h_start)
    dx = 2.0/n
    cur_h = h_start.copy()
    cur_val = np.max(np.correlate(cur_h, 1 - cur_h, mode='full')) * dx
    best_h = cur_h.copy()

    start = time.time()
    init_T = 0.02
    while time.time() - start < time_budget:

def project_onto_constraints(v):
    n = v.shape[0]
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # adjust tiny discrepancy
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        # distribute diff to first element without breaking bounds
        proj[0] = np.clip(proj[0] + diff, 0.0, 1.0)
    return proj

            cur_h = cand_h
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = cur_h.copy()

            tot = cur_h[i] + cur_h[j]

    # final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h.tolist(), float(final_c5), int(n)

import time
import numpy as np


def _project_onto_constraints(v: np.ndarray) -> np.ndarray:
    """Project v onto the set 0≤h≤1, sum(h)=len(h)/2."""
    n = v.shape[0]
    target = n / 2.0
    lo, hi = -2.0, 2.0                     # enough to drive sum to n or 0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    # tiny correction for rounding errors
    diff = target - proj.sum()
    if abs(diff) > 1e-9:
        proj[0] = np.clip(proj[0] + diff, 0.0, 1.0)
    return proj


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] that yields a small
    C₅ = max_k ∫ h(x)(1‑h(x+k))dx.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # --- initial feasible point ------------------------------------------------
    if "initial_h_values" in globals():
        h_cur = np.array(initial_h_values, dtype=float)
    else:
        n = 80
        h_cur = rng.random(n)
    h_cur = _project_onto_constraints(h_cur)
    n = h_cur.size
    dx = 2.0 / n

    # current and best values (C₅ is the scaled max correlation)
    cur_val = np.max(np.correlate(h_cur, 1 - h_cur, mode="full")) * dx
    best_h = h_cur.copy()
    best_val = cur_val

    start = time.time()
    init_lr = 0.12          # base learning rate for gradient steps
    init_T = 0.025          # base temperature for simulated‑annealing acceptance

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.2:
            break

        # ---- correlation and indices attaining the max ------------------------
        corr = np.correlate(h_cur, 1 - h_cur, mode="full")
        max_corr = corr.max()
        max_idx = np.where(np.abs(corr - max_corr) < 1e-12)[0]

        # ---- subgradient of the worst shift(s) ------------------------------
        grad = np.zeros_like(h_cur)
        for idx in max_idx:
            shift = idx - (n - 1)          # shift ∈ [-(n‑1), n‑1]
            g = np.ones(n)
            if shift >= 0:
                g[: n - shift] -= h_cur[shift:]
                g[shift:] -= h_cur[: n - shift]
            else:
                s = -shift
                g[s:] -= h_cur[: n - s]
                g[: n - s] -= h_cur[s:]
            grad += g
        grad /= len(max_idx)

        # ---- gradient descent step (with projection) -------------------------
        lr = init_lr / np.sqrt(1.0 + elapsed)
        cand = h_cur - lr * grad
        cand = _project_onto_constraints(cand)
        cand_val = np.max(np.correlate(cand, 1 - cand, mode="full")) * dx

        # ---- acceptance (simulated annealing) -------------------------------
        delta = cand_val - cur_val
        T = init_T * max(0.0, 1.0 - elapsed / budget_s)
        if delta < 0.0 or np.exp(-delta / (T + 1e-12)) > rng.random():
            h_cur = cand
            cur_val = cand_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h_cur.copy()

        # ---- occasional random redistribution between a pair -----------------
        if rng.random() < 0.05:
            i, j = rng.choice(n, 2, replace=False)
            total = h_cur[i] + h_cur[j]
            lo = max(0.0, total - 1.0)
            hi = min(1.0, total)
            if hi - lo > 1e-12:

                new_j = total - new_i
                h2 = h_cur.copy()
                h2[i], h2[j] = new_i, new_j
                val2 = np.max(np.correlate(h2, 1 - h2, mode="full")) * dx
                d2 = val2 - cur_val
                if d2 < 0.0 or np.exp(-d2 / (T + 1e-12)) > rng.random():
                    h_cur = h2
                    cur_val = val2
                    if cur_val < best_val:
                        best_val = cur_val
                        best_h = h_cur.copy()

        # ---- occasional full restart -----------------------------------------
        if rng.random() < 0.01:
            h_cur = _project_onto_constraints(rng.random(n))
            cur_val = np.max(np.correlate(h_cur, 1 - h_cur, mode="full")) * dx

    # Final evaluation using the provided helper (ensures identical scaling)
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h.tolist(), float(final_c5), int(n)
