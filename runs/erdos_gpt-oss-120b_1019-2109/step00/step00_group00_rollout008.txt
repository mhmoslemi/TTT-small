def anneal(start_h, time_budget, rng):

    dx = 2.0 / n
    cur_h = start_h.copy()
    cur_val = evaluate_h(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val
    start = time.time()
    # parameters
    T_initial = 0.2
    T_final = 1e-5
    # Loop
    while True:
        elapsed = time.time() - start
        if elapsed >= time_budget:
            break
        # propose move
        i1 = rng.integers(n)
        i2 = rng.integers(n-1)
        if i2 >= i1:
            i2 += 1
        # compute allowed delta
        low = max(-cur_h[i1], cur_h[i2] - 1.0)
        high = min(1.0 - cur_h[i1], cur_h[i2])
        if high <= low:
            continue
        delta = rng.uniform(low, high)
        # apply move
        new_h = cur_h.copy()
        new_h[i1] += delta
        new_h[i2] -= delta
        new_val = evaluate_h(new_h)
        # temperature
        T = T_initial * (1.0 - elapsed / time_budget) + T_final * (elapsed / time_budget)
        # accept?
        if new_val < cur_val:
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
        else:
            # accept with prob
            if rng.random() < math.exp(-(new_val - cur_val) / max(T, 1e-12)):
                cur_h = new_h
                cur_val = new_val
        # optional random restart or small mutation? skip for now
    return best_h, best_val

import time, math
import numpy as np

def _evaluate(h: np.ndarray) -> float:
    """Compute C5 for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx


def _random_pattern(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random 0/1 (with possibly one fractional entry) pattern with sum=n/2."""
    target = n / 2.0
    integer = int(math.floor(target))
    frac = target - integer
    arr = np.zeros(n, dtype=float)
    if integer:
        arr[:integer] = 1.0
    if frac:
        arr[integer] = frac
    rng.shuffle(arr)
    return arr


def _anneal(start_h: np.ndarray, budget: float, rng: np.random.Generator) -> tuple:
    """Simple simulated annealing that respects the sum constraint."""
    n = start_h.size
    cur_h = start_h.copy()
    cur_val = _evaluate(cur_h)
    best_h, best_val = cur_h.copy(), cur_val

    start = time.time()
    T0, Tf = 0.5, 1e-5  # temperature schedule

    while True:
        elapsed = time.time() - start
        if elapsed >= budget:
            break

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # feasible delta so both stay in [0,1]
        low = max(-cur_h[i], cur_h[j] - 1.0)
        high = min(1.0 - cur_h[i], cur_h[j])
        if high <= low:
            continue

        delta = rng.uniform(low, high)

        # trial vector
        trial = cur_h.copy()
        trial[i] += delta
        trial[j] -= delta
        trial_val = _evaluate(trial)

        # temperature linear from T0 to Tf
        T = T0 * (1.0 - elapsed / budget) + Tf * (elapsed / budget)
        improve = trial_val < cur_val
        if improve or rng.random() < math.exp(-(trial_val - cur_val) / max(T, 1e-12)):
            cur_h, cur_val = trial, trial_val
            if trial_val < best_val:
                best_h, best_val = trial.copy(), trial_val

    return best_h, best_val
