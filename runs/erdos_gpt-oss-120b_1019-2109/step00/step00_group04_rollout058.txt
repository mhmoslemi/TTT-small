from scipy.optimize import minimize

def run(seed=42, budget_s=1100, **kwargs):
    start = time.time()
    np.random.seed(seed)
    # Determine number of points and initial h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 42
        # generate random binary with sum n/2
        h0 = np.zeros(n)
        idx = np.random.choice(n, n//2, replace=False)
        h0[idx] = 1.0

    dx = 2.0 / n
    target_val = 0.3808

    # objective: max correlation * dx
    def obj(h):
        return np.max(np.correlate(h, 1 - h, mode='full')) * dx

    # Equality constraint: sum h = n/2
    cons = ({'type': 'eq', 'fun': lambda h: np.sum(h) - n/2},)

    # bounds [0,1] for each variable
    bnds = [(0.0, 1.0) for _ in range(n)]

    best_h = h0.copy()
    best_val = obj(best_h)

    # Try a few SLSQP runs
    max_slsqp_runs = 6
    for run_idx in range(max_slsqp_runs):
        if time.time() - start > budget_s * 0.80:
            break
        # start point: different perturbations
        if run_idx == 0:
            x0 = best_h.copy()
        else:
            # random feasible point: sample uniform then project to simplex with box constraints
            y = np.random.rand(n)
            # Adjust to correct sum by shifting by constant and clipping iteratively
            # simple projection using bisection lambda:
            target = n/2
            lo, hi = -1.0, 1.0
            for _ in range(50):
                lam = (lo + hi) / 2.0
                z = np.clip(y - lam, 0.0, 1.0)
                s = np.sum(z)
                if s > target:
                    lo = lam
                else:
                    hi = lam
            x0 = np.clip(y - lam, 0.0, 1.0)

        res = minimize(obj, x0, method='SLSQP', bounds=bnds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-9, 'disp': False})
        if res.success:
            val = obj(res.x)
            if val < best_val - 1e-9:
                best_val = val
                best_h = res.x.copy()

    # Local greedy refinement moving mass from “bad’’ to “good’’ indices
    h = best_h.copy()
    step = 0.05  # maximal amount of mass moved each iteration
    # Helper sliding shift
    def shifted(arr, k):
        if k >= 0:
            out = np.empty_like(arr)
            out[:n - k] = arr[k:]
            out[n - k:] = 0.0
            return out
        else:
            s = -k
            out = np.empty_like(arr)
            out[s:] = arr[:n - s]
            out[:s] = 0.0
            return out

    while time.time() - start < budget_s * 0.95:
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = corr.max()
        if max_corr * dx < best_val - 1e-12:
            best_val = max_corr * dx
            best_h = h.copy()
        # stop if already below target
        if best_val < target_val:
            break
        max_idx = corr.argmax()
        shift = max_idx - (n - 1)
        neigh = shifted(h, shift)
        A = 1.0 - neigh
        # get candidates
        # high A & h>0
        idx_high = np.where(h > 1e-9)[0]
        if idx_high.size == 0:
            break
        A_high = A[idx_high]
        # choose i with maximal A among those
        i = idx_high[np.argmax(A_high)]
        # low A & h<1
        idx_low = np.where(h < 1 - 1e-9)[0]
        if idx_low.size == 0:
            break
        A_low = A[idx_low]

            if trans > 0:
                h[a] -= trans
                h[b] += trans

    # final projection onto feasible set (in case of round‑off)
    # simple bisection projection to sum = n/2 while respecting bounds
    total = np.sum(h)
    target = n/2
    if abs(total - target) > 1e-9:
        # find lambda to adjust
        lo, hi = -2.0, 2.0
        for _ in range(50):

            s = np.sum(z)
            if s > target:
                lo = lam
            else:
                hi = lam
        h = np.clip(h - lam, 0.0, 1.0)
    # ensure bounds
    h = np.clip(h, 0.0, 1.0)
    final_val = obj(h)
    # if our refined solution is not better keep previous best
    if final_val < best_val:
        best_h = h
        best_val = final_val

    return (best_h, float(best_val), n)

import time
import numpy as np
from scipy.optimize import minimize

def _project_to_feasible(v, target, lo=0.0, hi=1.0):
    """Project vector v onto the box [lo,hi] with sum == target."""
    # simple bisection on λ for projection onto the capped simplex
    l, r = -2.0, 2.0
    for _ in range(50):
        lam = (l + r) / 2.0
        z = np.clip(v - lam, lo, hi)
        s = z.sum()
        if s > target:
            l = lam
        else:
            r = lam
    return np.clip(v - lam, lo, hi)

def run(seed=42, budget_s=1100, **kwargs):
    """Return (h_values, c5_bound, n_points) for the Erdős C5 problem."""
    start = time.time()
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1.  Load or build an initial feasible vector
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float)
        n = h0.size
    else:
        n = 42                         # a modest size that works well
        h0 = np.zeros(n)
        idx = np.random.choice(n, n // 2, replace=False)
        h0[idx] = 1.0                 # binary start, sum == n/2

    dx = 2.0 / n
    target_c5 = 0.3808                 # we try to beat this value

    # ------------------------------------------------------------------
    # 2.  Objective: max_k ∫ h(x)(1‑h(x+k)) dx  (discretised)
    # ------------------------------------------------------------------
    def c5_obj(h):
        # value = max correlation * dx
        return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

    # equality constraint: ∑h = n/2  (integral == 1)
    constraints = ({
        "type": "eq",
        "fun": lambda h: np.sum(h) - n / 2.0,
    },)

    bounds = [(0.0, 1.0) for _ in range(n)]

    # ------------------------------------------------------------------
    # 3.  A few quick local SLSQP runs from random feasible starts
    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_val = c5_obj(best_h)

    for attempt in range(6):
        if time.time() - start > budget_s * 0.80:
            break

        if attempt == 0:
            x0 = best_h.copy()

            # random vector projected onto the feasible set
            rnd = np.random.rand(n)
            x0 = _project_to_feasible(rnd, n / 2.0)

            c5_obj,
            x0,
            method="SLSQP",
            bounds=bounds,
            constraints=constraints,
            options={"maxiter": 200, "ftol": 1e-9, "disp": False},
        )
        if res.success:
            val = c5_obj(res.x)
            if val < best_val - 1e-9:
                best_val = val
                best_h = res.x.copy()

    # ------------------------------------------------------------------
    # 4.  Greedy mass‑transport refinement (reducing the worst shift)
    # ------------------------------------------------------------------
    h = best_h.copy()
    step = 0.05                       # max mass moved per iteration

    def shift_array(arr, k):
        """Return arr shifted by k (zero‑padded)."""
        if k >= 0:
            out = np.empty_like(arr)
            out[: n - k] = arr[k:]
            out[n - k :] = 0.0
        else:
            s = -k
            out = np.empty_like(arr)
            out[s:] = arr[: n - s]
            out[:s] = 0.0
        return out

    while time.time() - start < budget_s * 0.95:
        corr = np.correlate(h, 1.0 - h, mode="full")
        max_corr = corr.max()
        cur_c5 = max_corr * dx
        if cur_c5 < best_val - 1e-12:
            best_val = cur_c5
            best_h = h.copy()
        if best_val < target_c5:
            break

        # shift with the currently largest correlation
        idx_max = int(corr.argmax())
        shift = idx_max - (n - 1)
        neigh = shift_array(h, shift)
        A = 1.0 - neigh                # contribution factor per index

        # candidates for decreasing / increasing mass
        pos = np.where(h > 1e-9)[0]
        if pos.size == 0:
            break
        i = pos[np.argmax(A[pos])]     # high A & positive mass

        neg = np.where(h < 1.0 - 1e-9)[0]
        if neg.size == 0:
            break
        j = neg[np.argmin(A[neg])]     # low A & room to receive mass

        delta = min(h[i], 1.0 - h[j], step)
        if delta <= 0:
            break
        h[i] -= delta
        h[j] += delta

        # occasional random jitter to avoid stagnation
        if np.random.rand() < 0.01:
            a, b = np.random.choice(n, 2, replace=False)
            trans = min(h[a], 1.0 - h[b], step * 0.5)
            if trans > 0:
                h[a] -= trans
                h[b] += trans

        # slowly shrink the step size if we have many iterations
        if np.random.rand() < 0.001:
            step = max(step * 0.7, 1e-4)

    # ------------------------------------------------------------------
    # 5.  Final projection onto the exact feasible set (numerical cleanup)
    # ------------------------------------------------------------------
    total = h.sum()
    if abs(total - n / 2.0) > 1e-9:
        h = _project_to_feasible(h, n / 2.0)

    h = np.clip(h, 0.0, 1.0)
    final_val = c5_obj(h)
    if final_val < best_val:
        best_val = final_val
        best_h = h.copy()

    return (best_h, float(best_val), n)
