import time, math
import numpy as np

def _evaluate(h, dx):
    """Compute C5 for a given step vector."""
    # correlation h with (1-h) over all linear shifts
    conv = np.correlate(h, 1.0 - h, mode='full')
    return np.max(conv) * dx

def _repair_sum(h, target):
    """Clip to [0,1] and adjust to meet the exact sum constraint."""
    np.clip(h, 0.0, 1.0, out=h)
    cur = h.sum()
    diff = target - cur
    # Distribute the remaining mass to non‑boundary entries
    for _ in range(100):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            inc = np.minimum(delta, 1.0 - h[idx])
        else:
            idx = np.where(h > 0.0)[0]
            if idx.size == 0:
                break
            delta = diff / idx.size
            inc = np.maximum(delta, -h[idx])
        h[idx] += inc
        cur = h.sum()
        diff = target - cur
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _random_swap(h, target, rng):
    """Move a random amount of mass from a 1‑ish entry to a 0‑ish entry."""
    ones = np.where(h > 1e-12)[0]
    zeros = np.where(h < 1.0 - 1e-12)[0]
    if ones.size == 0 or zeros.size == 0:
        return h.copy()
    i = rng.choice(ones)
    j = rng.choice(zeros)
    max_delta = min(h[i], 1.0 - h[j])
    if max_delta <= 0:
        return h.copy()
    delta = max_delta * rng.rand()
    h2 = h.copy()
    h2[i] -= delta
    h2[j] += delta
    return _repair_sum(h2, target)

def _gradient_step(h, target, dx):
    """Single gradient step aimed at the current worst shift."""
    n = h.size
    # correlation to locate worst shift
    conv = np.correlate(h, 1.0 - h, mode='full')
    max_idx = int(np.argmax(conv))
    s = max_idx - (n - 1)          # signed shift
    # gradient = 1 - (h_{i+s} + h_{i-s})  (missing terms are 0)
    g = np.ones_like(h)
    if s >= 0:
        g[:n - s] -= h[s:]
        g[s:]    -= h[:n - s]
    else:
        ss = -s
        g[ss:]   -= h[:n - ss]
        g[:n - ss] -= h[ss:]
    # remove the mean to keep the sum roughly constant
    g -= g.mean()
    # simple constant step size
    alpha = 0.02
    h_new = h - alpha * g
    return _repair_sum(h_new, target)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    start = time.time()

    # ----- initialise -------------------------------------------------
    try:
        h0 = np.asarray(initial_h_values, dtype=float)  # provided by harness
        n = h0.size
        target_sum = n / 2.0
        h0 = _repair_sum(h0, target_sum)
    except NameError:
        # no parent solution – create a smooth cosine‑squared start
        n = 120
        target_sum = n / 2.0
        x = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        h0 = np.cos(np.pi * (x - 1.0) / 2.0) ** 2
        h0 = _repair_sum(h0, target_sum)

    dx = 2.0 / n
    cur_h = h0.copy()
    cur_val = _evaluate(cur_h, dx)
    best_h = cur_h.copy()
    best_val = cur_val

    # ----- optimisation loop -------------------------------------------
    # parameters for the stochastic loop
    temp_start = 0.03
    temp_end   = 1e-4
    prob_grad  = 0.6      # probability of using a gradient step

    while True:
        now = time.time()
        if now - start > budget_s - 0.2:
            break

        # linear temperature cooling
        frac = (now - start) / (budget_s - 0.2)
        T = temp_start * (1.0 - frac) + temp_end * frac

        if rng.rand() < prob_grad:
            cand_h = _gradient_step(cur_h, target_sum, dx)
        else:
            cand_h = _random_swap(cur_h, target_sum, rng)

        cand_val = _evaluate(cand_h, dx)
        delta = cand_val - cur_val

        # accept if better, otherwise accept with a small probability
        if delta < 0 or rng.rand() < math.exp(-delta / max(T, 1e-12)):
            cur_h, cur_val = cand_h, cand_val
            if cand_val < best_val:
                best_h, best_val = cand_h.copy(), cand_val

        # occasional full restart to escape deep local minima
        if rng.rand() < 0.001:
            rnd = rng.rand(n)
            rnd = rnd / rnd.sum() * target_sum
            cur_h = _repair_sum(rnd, target_sum)
            cur_val = _evaluate(cur_h, dx)

    # final tidy‑up (ensure constraints exactly)
    best_h = _repair_sum(best_h, target_sum)
    best_val = _evaluate(best_h, dx)

    return (best_h.tolist(), float(best_val), int(n))
