def project_simplex_bounds(v, S):
    # ensure within 0..1
    v = np.clip(v, 0, 1)
    # compute current sum
    current = v.sum()
    # if close to S, done
    if abs(current - S) < 1e-9:
         return v
    # define direction: we need to add or subtract delta to adjust sum.
    # Use water-filling:
    d = current - S  # positive if we need to reduce sum
    # We'll iteratively adjust.
    # Use tolerance loop:
    while True:
        if abs(d) < 1e-9:
            break
        # Determine indices that can be moved in direction:
        if d > 0:
            # need to reduce sum, can reduce those that are >0
            mask = v > 0
        else:
            # need to increase sum, can increase those that are <1
            mask = v < 1
        if not np.any(mask):
            break  # cannot adjust further
        # distribute adjustment equally across all mask elements
        adjust = d / mask.sum()
        # Clip after adjustment
        v[mask] -= adjust
        # Clip to [0,1]
        v = np.clip(v, 0, 1)
        # recompute sum and diff
        new = v.sum()
        d = new - S
    return v

def project_onto_bounds_and_sum(x, S, tol=1e-9, max_iter=1000):
    x = np.clip(x, 0.0, 1.0)
    for _ in range(max_iter):
        sum_x = x.sum()
        diff = sum_x - S
        if abs(diff) < tol:
            break
        if diff > 0:
            # need to reduce mass
            # indices where x > 0 (allow reduce)
            mask = x > 0
            m = mask.sum()
            if m == 0:
                break
            delta = diff / m
            x[mask] -= delta
            # ensure no negative
            x = np.clip(x, 0.0, 1.0)
        else:
            # need to increase mass
            mask = x < 1
            m = mask.sum()
            if m == 0:
                break
            delta = (-diff) / m
            x[mask] += delta
            x = np.clip(x, 0.0, 1.0)
    return x

def worst_shift_and_gradient(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)  # index in full len 2*n-1
    k = idx - (n - 1)
    # compute shift arrays:
    if k >= 0:
        # shift right by k
        # h_r corresponds to h_i+k? Actually gradient uses h_{i+k} where i + k < n
        # For each i, term includes h[i+k] if i+k < n else 0.
        h_shift_plus = np.concatenate((np.zeros(k), h[:n - k]))
        # similarly h_{i - k}: shift left by k
        h_shift_minus = np.concatenate((h[k:], np.zeros(k)))
    else:
        kk = -k
        h_shift_plus = np.concatenate((h[kk:], np.zeros(kk)))  # k negative => i+k = i-kk
        h_shift_minus = np.concatenate((np.zeros(kk), h[:n - kk]))
    grad = 1.0 - (h_shift_plus + h_shift_minus)
    return k, grad, corr[idx] * (2.0 / n)

        # we can choose to move from j to i, which has amount = min(curr_h[j], 1 - curr_h[i])

            # move delta from i to j
            if max_transfer == 0:

            delta = rng.random() * max_transfer
            new_h = curr_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # move from j to i
            max_transfer2 = min(curr_h[j], 1 - curr_h[i])
            if max_transfer2 == 0:

            delta = rng.random() * max_transfer2

            curr_score = new_score

import time
import math
import numpy as np

# ------------------------------------------------------------
# Helper: correlation based C5 evaluation
def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)*(1-h(x+k)) dx for the discrete step function."""
    n = h.shape[0]
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    # dx = 2 / n   (integral over [0,2])
    return corr.max() * (2.0 / n)


# ------------------------------------------------------------
# Helper: projection onto the simplex with box [0,1] constraints
def project_onto_bounds_and_sum(v: np.ndarray, target_sum: float,
                                tol: float = 1e-9, max_iter: int = 500) -> np.ndarray:
    """Euclidean projection of v onto {x : 0≤x≤1, sum x = target_sum}."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(max_iter):
        s = x.sum()
        diff = s - target_sum
        if abs(diff) <= tol:
            break
        # choose variables that can move in the needed direction
        if diff > 0:          # need to decrease the sum
            mask = x > 0.0
        else:                 # need to increase the sum
            mask = x < 1.0
        m = mask.sum()
        if m == 0:
            break
        delta = diff / m
        # subtract delta (note: if diff<0, subtraction adds)
        x[mask] -= delta
        x = np.clip(x, 0.0, 1.0)
    return x


# ------------------------------------------------------------
# Helper: gradient of the worst (max) correlation term
def worst_shift_and_gradient(h: np.ndarray):
    """Return (shift, gradient vector, C5 value of that shift)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(corr.argmax())
    k = idx - (n - 1)          # signed shift
    # build shifted versions (zero outside the domain)
    if k >= 0:
        h_plus = np.concatenate((np.zeros(k), h[: n - k]))
        h_minus = np.concatenate((h[k:], np.zeros(k)))
    else:
        kk = -k
        h_plus = np.concatenate((h[kk:], np.zeros(kk)))
        h_minus = np.concatenate((np.zeros(kk), h[: n - kk]))
    grad = 1.0 - (h_plus + h_minus)
    return k, grad, corr[idx] * (2.0 / n)


# ------------------------------------------------------------
# Gradient descent on the worst shift (simple back‑tracking line search)
def gradient_descent(h: np.ndarray, target_sum: float,
                    max_iters: int = 200, init_step: float = 0.5) -> (np.ndarray, float):
    """Perform a few gradient steps to lower the maximal correlation."""
    n = h.shape[0]
    cur = h.copy()
    cur_val = compute_c5(cur)
    step = init_step
    for _ in range(max_iters):
        _, grad, _ = worst_shift_and_gradient(cur)
        # try a back‑tracking line search
        improved = False
        for _ in range(20):
            cand = cur - step * grad
            cand = project_onto_bounds_and_sum(cand, target_sum)
            cand_val = compute_c5(cand)
            if cand_val < cur_val - 1e-12:
                cur, cur_val = cand, cand_val
                step = min(step * 1.2, 1.0)
                improved = True
                break
            step *= 0.5
        if not improved:
            break
    return cur, cur_val

    best_val = compute_c5(best_h)
