def generate_two_step(n, e, a):
    if e * 2 > n:
        return None # invalid
    b = (n/2 - 2*e*a) / (n - 2*e) if n != 2*e else 0.0 # if n=2e, interior length 0; b unused.
    # Clamp b in [0,1]; if not, skip.
    if b < 0 or b > 1: return None
    h = np.full(n, b)
    if e > 0:
        h[:e] = a
        h[-e:] = a
    return h

import time
import numpy as np

def _c5_of_h(h):
    """Return C5 for a given step function array h."""
    # using the same definition as the evaluator in the harness
    n = h.shape[0]
    return np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / n)

def _two_step_h(n, e, a):
    """Construct a two‑level step function of length n.
    e   – number of points on each side having value a (edges)
    a   – edge value (0 ≤ a ≤ 1)
    The interior value b is chosen to satisfy the integral constraint.
    """
    target = n / 2.0                     # sum_i h[i] must equal target
    interior_len = n - 2 * e
    # solve 2*e*a + interior_len*b = target  => b = (target-2*e*a)/interior_len
    if interior_len == 0:
        # pure edge (no interior) – only possible when a = 0.5
        b = 0.0
    else:
        b = (target - 2.0 * e * a) / interior_len
    if b < 0.0 or b > 1.0:
        return None
    h = np.full(n, b, dtype=np.float64)
    if e > 0:
        h[:e] = a
        h[-e:] = a
    return h

def _search_two_step(n, rng):
    """Brute‑force search over two‑step shapes (edge length and value)."""
    best_c5 = np.inf
    best_h = None
    best_params = None
    target = n / 2.0
    # try every possible edge length (at least one point, leave interior)
    for e in range(1, n // 2):
        # feasible a satisfies b in [0,1]:
        # 0 ≤ (target - 2ea) / (n-2e) ≤ 1  →  max(0,1-n/(4e)) ≤ a ≤ min(1,n/(4e))
        a_low = max(0.0, 1.0 - n / (4.0 * e))
        a_high = min(1.0, n / (4.0 * e))
        if a_low > a_high:
            continue
        # coarse grid first
        for a in np.linspace(a_low, a_high, 41):
            h = _two_step_h(n, e, a)
            if h is None:
                continue
            c5 = _c5_of_h(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_params = (e, a, (target - 2.0 * e * a) / (n - 2.0 * e))
    # fine refinement around the best edge length
    if best_params is not None:
        e_best = best_params[0]
        a_best = best_params[1]
        # explore nearby edge lengths
        for de in range(-2, 3):
            e = e_best + de
            if e <= 0 or e >= n // 2:
                continue
            a_low = max(0.0, 1.0 - n / (4.0 * e))
            a_high = min(1.0, n / (4.0 * e))
            if a_low > a_high:
                continue
            # finer grid
            for a in np.linspace(max(a_low, a_best - 0.05),
                                 min(a_high, a_best + 0.05), 81):
                h = _two_step_h(n, e, a)
                if h is None:
                    continue
                c5 = _c5_of_h(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_params = (e, a,
                                   (target - 2.0 * e * a) / (n - 2.0 * e))
    return best_h, best_c5, best_params

def _hill_climb(initial_h, budget_s, start_time, rng):
    """Simple simulated‑annealing style hill climbing on the simplex."""
    n = initial_h.shape[0]
    h_cur = initial_h.copy()
    cur_c5 = _c5_of_h(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5

    # temperature schedule (linear cooling)
    T0 = 0.02
    while True:
        now = time.time()
        if now - start_time > budget_s * 0.95:
            break
        # pick two distinct indices
        i, j = rng.integers(0, n, size=2, replace=False)
        hi, hj = h_cur[i], h_cur[j]

        # maximum feasible delta keeping both values in [0,1]
        delta_min = max(-hi, hj - 1.0)
        delta_max = min(1.0 - hi, hj)
        if delta_max <= delta_min:
            continue
        # limit step size to keep moves local (helps convergence)
        max_step = 0.1
        delta_bound = min(delta_max, max_step)
        delta_low = max(delta_min, -max_step)
        delta = rng.uniform(delta_low, delta_bound)
        if delta == 0.0:
            continue

        # propose new vector
        h_new = h_cur.copy()
        h_new[i] = hi + delta
        h_new[j] = hj - delta

        new_c5 = _c5_of_h(h_new)

        # acceptance rule (simulated annealing)
        accept = False
        if new_c5 < cur_c5:
            accept = True
        else:
            # temperature declines linearly
            elapsed = now - start_time
            T = max(1e-6, T0 * (1.0 - elapsed / (budget_s * 0.9)))
            prob = np.exp((cur_c5 - new_c5) / T)
            if rng.random() < prob:
                accept = True
        if accept:
            h_cur = h_new
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where:
      - h_values : numpy array of step values satisfying the problem constraints
      - c5_bound : the corresponding C5 value (max overlap integral)
      - n_points : length of the discretisation (len(h_values))
    """
    rng = np.random.default_rng(seed)
    start = time.time()

    # Choose a moderate number of points (odd for symmetry)
    n_points = 401          # < 1000 as required
    # First, obtain a good two‑step candidate
    h, c5, _ = _search_two_step(n_points, rng)

    # If for any reason the search fails, fall back to a linear shape
    if h is None:
        # linear “V” shape : h_i = |i-(n-1)/2| / ((n-1)/2) scaled to satisfy the integral
        idx = np.arange(n_points, dtype=np.float64)
        centre = (n_points - 1) / 2.0
        h = np.abs(idx - centre) / centre
        scale = (n_points / 2.0) / h.sum()
        h = np.clip(h * scale, 0.0, 1.0)

    # Refine with hill climbing / simulated annealing
    h_opt, c5_opt = _hill_climb(h, budget_s, start, rng)

    # Final verification – use the supplied evaluator if available
    try:
        c5_bound = evaluate_erdos_solution(h_opt)
    except Exception:                     # fallback if helper not present
        c5_bound = _c5_of_h(h_opt)

    # Ensure the integral constraint is satisfied (within tiny tolerance)
    dx = 2.0 / n_points
    total = h_opt.sum() * dx
    if abs(total - 1.0) > 1e-9:
        # tiny renormalisation (won’t break bounds because deviation is minute)
        h_opt *= (1.0 / total)

    return h_opt, float(c5_bound), n_points
