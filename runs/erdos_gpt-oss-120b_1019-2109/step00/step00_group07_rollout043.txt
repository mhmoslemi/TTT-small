def project_simplex_bounds(v, total):

    w = np.clip(v, 0, 1)
    # Compute surplus = total - sum(w)
    surplus = total - w.sum()

            break

        free_indices = np.where(free_mask)[0]
        n_free = len(free_indices)
        if n_free == 0:
            # cannot adjust, break
            break
        delta = surplus / n_free
        w[free_indices] += delta
        # Clip again
        w = np.clip(w, 0, 1)

import time
import math
import random
import numpy as np

def _evaluate(h):
    """Return C5 for the discretized step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx


def _project_to_sum_and_bounds(v, total):
    """
    Euclidean projection of vector v onto the set
        { w : 0 <= w_i <= 1 , sum w_i = total }.
    Simple water‑filling iteration; fast for the modest sizes we use.
    """
    w = np.clip(v, 0.0, 1.0)
    # iterative adjustment of the free (non‑bounded) entries
    for _ in range(15):
        s = w.sum()
        diff = total - s
        if abs(diff) <= 1e-12:
            break
        free = (w > 0.0) & (w < 1.0)
        idx = np.where(free)[0]
        nfree = idx.size
        if nfree == 0:                     # all at a bound – distribute the tiny remaining error arbitrarily
            break
        delta = diff / nfree
        w[idx] += delta
        w = np.clip(w, 0.0, 1.0)
    # final tiny correction if ever needed
    s = w.sum()
    if abs(s - total) > 1e-9:
        # distribute the remaining error over a single free coordinate if possible
        for i in range(w.size):
            if 0.0 <= w[i] + (total - s) <= 1.0:
                w[i] += total - s
                break
    return w


def _grad_for_lag(h, lag):
    """Gradient of Q_l = sum_i h_i h_{i-lag} w.r.t. h (linear part, no bounds)."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if lag >= 0:
        # contribution from h_{i-lag}
        g[lag:] += h[: n - lag]
        # contribution from h_{i+lag}
        g[: n - lag] += h[lag:]
    else:
        l = -lag
        g[: n - l] += h[l:]
        g[l:] += h[: n - l]
    return g


def _local_optimize(start_h, total, max_iters, step0, time_limit, best_global):
    """
    Gradient ascent on the minimal self‑overlap (equivalently descent on max C5).
    Returns best (h, score) found from this start.
    """
    h = start_h.copy()
    n = h.shape[0]
    dx = 2.0 / n
    best_h = h.copy()
    best_score = _evaluate(h)

    # early exit if already good enough
    if best_score <= 0.3808:
        return best_h, best_score

    for it in range(max_iters):
        if time.time() >= time_limit:
            break

        # correlation for current h
        corr = np.correlate(h, 1 - h, mode="full")
        cur_score = np.max(corr) * dx
        # stop if we beat the global best already
        if cur_score < best_global:
            best_global = cur_score

        # worst lag (the one giving the maximal correlation)
        worst_idx = int(np.argmax(corr))
        lag = worst_idx - (n - 1)

        # gradient that would *increase* the overlap Q_l (i.e. reduce the correlation)
        g = _grad_for_lag(h, lag)

        # linearly decreasing step size
        step = step0 * (1.0 - it / max_iters)

        # take a step and project back onto feasible region
        h_new = h + step * g
        h_new = _project_to_sum_and_bounds(h_new, total)

        new_score = _evaluate(h_new)

        # accept if improvement, otherwise keep current h
        if new_score < cur_score:
            h = h_new
            if new_score < best_score:
                best_score = new_score
                best_h = h_new.copy()
        else:
            # occasional random jitter to escape tiny plateaus
            if random.random() < 0.02:
                jitter = (np.random.rand(n) - 0.5) * step
                h = _project_to_sum_and_bounds(h + jitter, total)

        # early exit if we have reached the target bound
        if best_score <= 0.3808:
            break

    return best_h, best_score

    """
    Optimize the step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ----------------------------------------------------------------------

    if 'initial_h_values' in globals():

    total = n / 2.0                # because integral of h over [0,2] must be 1
    h0 = _project_to_sum_and_bounds(h0, total)

    # ----------------------------------------------------------------------
    # Global bookkeeping

    start_time = time.time()
    time_limit = start_time + budget_s * 0.94   # keep a little slack

    best_h = h0.copy()
    best_score = _evaluate(best_h)

    # ----------------------------------------------------------------------
    # Perform several restarts with gradient ascent

    restarts = 6

    h_list = best_h.tolist()
    return (h_list, float(best_score), int(n))
