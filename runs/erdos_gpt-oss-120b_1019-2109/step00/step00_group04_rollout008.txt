import time
import math
import numpy as np

# ---------------------------------------------------------------
# Helper: evaluate the Erdős overlap bound for a given step vector
def _evaluate_h(h: np.ndarray) -> float:
    """Return C5 for step values h (dx = 2/n)."""
    n = h.shape[0]
    # full correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * (2.0 / n))


# ---------------------------------------------------------------
# Helper: Euclidean projection onto the box [0,1] with a fixed sum.
def _project_onto_box_sum(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Solve   min ||x-v||^2   s.t. 0<=x_i<=1 , sum x_i = target_sum.
    The solution is x_i = clip(v_i - λ,0,1) for a suitable λ.
    """
    # binary search for λ
    lo = -np.max(v) - 5.0   # large enough to force all entries to 1
    hi = np.max(v) + 5.0    # large enough to force all entries to 0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)
    # tiny correction for round‑off
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1.0 - 1e-12)
        if mask.any():
            cnt = mask.sum()
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            # all at a bound – adjust the first element
            x[0] = np.clip(x[0] + diff, 0.0, 1.0)
    return x


# ---------------------------------------------------------------
# Helper: one gradient descent step for the currently worst shift.
def _gradient_step(h: np.ndarray, lr: float, target_sum: float) -> np.ndarray:
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    d = idx - (n - 1)                     # shift giving the maximal overlap
    grad = np.ones_like(h)

    if d >= 0:
        grad[: n - d] -= h[d:]
        grad[d:]      -= h[: n - d]
    else:
        d = -d
        grad[d:]      -= h[: n - d]
        grad[: n - d] -= h[d:]

    v = h - lr * grad
    return _project_onto_box_sum(v, target_sum)


# ---------------------------------------------------------------
# Helper: produce a feasible vector by moving mass from i to j.
def _random_transfer(h: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    n = h.shape[0]
    i = rng.integers(0, n)
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1

    # direction: move from i to j or opposite, chosen uniformly
    if rng.random() < 0.5:
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:
            return None
        delta = rng.uniform(0.0, max_delta)
        new_h = h.copy()
        new_h[i] -= delta
        new_h[j] += delta
    else:
        max_delta = min(h[j], 1.0 - h[i])
        if max_delta <= 0.0:
            return None
        delta = rng.uniform(0.0, max_delta)
        new_h = h.copy()
        new_h[j] -= delta
        new_h[i] += delta
    return new_h

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] to lower the Erdős C5 bound.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # --- initialise ------------------------------------------------
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)
        # up‑sample once to give a finer discretisation without blowing up size
        factor = 2
        h0 = np.repeat(base, factor)
    else:
        # fallback: a uniform half‑filled vector
        n = 84
        h0 = np.full(n, 0.5, dtype=float)

    n = h0.shape[0]
    target_sum = n / 2.0
    # guarantee feasibility from the start
    h0 = _project_onto_box_sum(h0, target_sum)

    # best known solution so far
    best_h = h0.copy()
    best_val = _evaluate_h(best_h)

    # current point for the stochastic search
    cur_h = best_h.copy()
    cur_val = best_val

    # annealing schedule parameters
    T_start = 0.20          # relatively high to explore
    T_end = 5e-5
    # gradient step size
    lr = 0.5

    start_time = time.time()
    # we stop a little early to stay within the budget
    deadline = start_time + budget_s * 0.95

    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / (budget_s * 0.95)
        # linear temperature decay
        T = T_start * (1.0 - frac) + T_end * frac

        # Occasionally (10 % of iterations) try a deterministic gradient step
        if rng.random() < 0.10:
            cand = _gradient_step(cur_h, lr, target_sum)
            cand_val = _evaluate_h(cand)
            if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / max(T, 1e-12)):
                cur_h, cur_val = cand, cand_val
                if cand_val < best_val:

        cand = _random_transfer(cur_h, rng)

        cand = _project_onto_box_sum(cand, target_sum)
        cand_val = _evaluate_h(cand)

        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / max(T, 1e-12)):
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val:
                best_h, best_val = cand.copy(), cand_val

        # Occasionally jitter the learning rate of the gradient steps
        if rng.random() < 0.01:
            lr = max(0.01, min(1.0, lr * (0.9 + 0.2 * rng.random())))

    # final clean‑up: make sure the returned vector satisfies the constraints exactly
    best_h = _project_onto_box_sum(best_h, target_sum)
    final_val = _evaluate_h(best_h)

    return (best_h, float(final_val), int(n))
