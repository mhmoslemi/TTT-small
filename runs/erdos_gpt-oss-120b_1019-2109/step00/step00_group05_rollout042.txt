def project_to_feasible(h):
    # h may be out of [0,1] and sum may not equal target.
    # We need to produce h' in [0,1] and sum = target.
    # Approach: first clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Then adjust sum by distributing correction among elements not at bound.
    s = h.sum()
    target = target_sum
    diff = target - s
    if abs(diff) < 1e-12:
        return h
    # Use iterative water-filling: while diff not zero, adjust free entries.
    # Determine free indices where 0<h<1.
    free = (h > 0) & (h < 1)
    # If no free and diff !=0, we can't adjust. That would be degenerate.
    # We can adjust by moving some bound entries within epsilon.
    # But diff should be small.
    # We can use a more robust projection onto box-simplex using algorithm from 'Projection onto simplex with box constraints'.
    # Simpler: Use linear programming? Or use scipy.optimize.linprog? But that may be slower.
    # Let's implement iterative method: find free indices; if none, then skip.
    # else distribute diff equally among them; after distribution, some may hit bounds -> need to repeat.
    while True:
        free = (h > 1e-12) & (h < 1 - 1e-12)
        count = free.sum()
        if count == 0:
            # fallback: adjust by adding diff to the max element and clipping.
            # find index with largest capacity (if diff>0, choose an index <1)
            if diff > 0:
                # add diff to first element not at 1
                idx = np.argmax(1 - h)
                h[idx] = np.clip(h[idx] + diff, 0, 1)
                break
            else:
                idx = np.argmax(h)  # biggest h
                h[idx] = np.clip(h[idx] + diff, 0, 1)
                break
        # compute allowable adjustment per free to not exceed bounds
        # max increase per free: 1 - h[free]; max decrease: -h[free]
        # allocate diff equally:
        delta = diff / count
        # but limit
        delta = np.clip(delta, -h[free], 1 - h[free])
        # Apply
        h[free] += delta
        # update diff
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-12:
            break
    return h

def enforce_sum_and_bounds(h):
    # Clip
    h = np.clip(h, 0.0, 1.0)
    # Compute diff
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # adjust: find index with capacity to add/subtract
        # capacity = 1 - h[i] if diff >0 else h[i]
        if diff > 0:
            capacities = 1.0 - h
            # choose idx with max capacity
            idx = np.argmax(capacities)
            # add min(diff, capacity) (should be diff but capacity may be smaller)
            add = min(diff, capacities[idx])
            h[idx] += add
            diff -= add
            # If still diff>0 (unlikely), repeat loop adjusting next index.
            # For simplicity we loop while diff>1e-12
            while diff > 1e-12:
                capacities = 1.0 - h
                idx = np.argmax(capacities)
                add = min(diff, capacities[idx])
                if add < 1e-12: break
                h[idx] += add
                diff -= add
        else:
            # diff <0
            capacities = h
            idx = np.argmax(capacities)
            sub = min(-diff, capacities[idx])
            h[idx] -= sub
            diff += sub
            while diff < -1e-12:
                capacities = h
                idx = np.argmax(capacities)
                sub = min(-diff, capacities[idx])
                if sub < 1e-12: break
                h[idx] -= sub
                diff += sub
    # final clip again
    h = np.clip(h, 0.0, 1.0)
    # Now sum may be off small due to numeric, enforce final scaling with small fudge:
    # Optionally adjust by small proportion
    residual = target_sum - h.sum()
    if abs(residual) > 1e-9:
        # distribute residual uniformly across all entries not at bound
        free = (h > 0) & (h < 1)
        cnt = free.sum()
        if cnt>0:
            h[free] += residual / cnt
            h = np.clip(h,0.0,1.0)
    return h

```
def run(seed=42, budget_s=1100):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64)
        n = len(h)
        # Ensure feasibility
        h = project_to_feasible(h)
    else:
        n = 70  # default
        h = np.full(n, 0.5, dtype=np.float64)
        # Ensure sum = n/2
        h = project_to_feasible(h) # will adjust if not correct.
    target_sum = n/2
    dx = 2.0 / n

    # compute initial cost
    best_h = h.copy()
    best_cost = evaluate_erdos_solution(best_h)
    # local search
    start_time = time.time()
    iter = 0
    # parameters
    max_iters = 1000000  # adapt
    T0 = 0.1
    T = T0
    # schedule: T decays over time
    while time.time() - start_time < budget_s * 0.95:
        # propose move
        # select two indices i, j random distinct
        i, j = rng.integers(n, size=2)
        if i == j:
            continue
        # Choose epsilon as random up to min(0.1, available)
        # amount = delta to transfer from i to j
        # direction: if h[i] > 0 and h[j] < 1, we can move epsilon
        # else we could transfer in opposite direction.
        # Determine max possible transfer:
        max_transfer = min(h[i], 1 - h[j])
        if max_transfer <= 0:
            # try reverse direction
            max_transfer = min(h[j], 1 - h[i])
            if max_transfer <= 0:
                continue
            # swap roles
            i, j = j, i
        # epsilon random fraction of max_transfer
        eps = rng.random() * max_transfer
        # create new candidate
        h_new = best_h.copy()
        h_new[i] -= eps
        h_new[j] += eps
        # due to floating errors maybe small negative/over 1, clip
        # Evaluate cost
        cost_new = evaluate_erdos_solution(h_new)
        # Acceptance: if better or if random < exp((best_cost - cost_new) / T)
        if cost_new < best_cost or rng.random() < np.exp((best_cost - cost_new) / max(T, 1e-12)):
            # accept
            best_h = h_new
            best_cost = cost_new
            # optional store if improvement better than global best
        # Update temperature schedule
        iter += 1
        # cooling: T = T0 * (1 - elapsed/ budget) maybe linear or exponential
        elapsed = time.time() - start_time
        T = T0 * (1 - elapsed / (budget_s * 0.9))
        if T < 1e-4:
            T = 1e-4
    # final ensure feasibility -> project (maybe after moves sum unchanged)
    best_h = project_to_feasible(best_h)
    final_cost = evaluate_erdos_solution(best_h)
    return (best_h, final_cost, n)
```

def proj_to_bounds_and_sum(v, target):
    # clip
    v = np.clip(v, 0, 1)
    # compute difference
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    # Determine direction: if diff > 0 need to increase some components, else decrease.
    if diff > 0:
        # can increase components with <1.
        # Let free = indices where v < 1. We'll add to them.
        while diff > 1e-12:
            free = np.where(v < 1 - 1e-12)[0]
            if free.size == 0:
                # cannot increase further, break
                break
            add_per = diff / free.size
            # clamp addition to capacity
            capacities = 1 - v[free]
            add = np.minimum(add_per, capacities)
            v[free] += add
            diff = target - v.sum()
            # if any still under capacity continue
        # Finally clip
        v = np.clip(v, 0, 1)
        return v
    else:
        # diff <0: need to decrease some components with >0.
        while diff < -1e-12:
            free = np.where(v > 1e-12)[0]
            if free.size == 0:
                break
            sub_per = (-diff) / free.size
            capacities = v[free]
            sub = np.minimum(sub_per, capacities)
            v[free] -= sub
            diff = target - v.sum()
        v = np.clip(v, 0, 1)
        return v
```

            # choose idx with max capacity
            idx = np.argmax(caps)
            add = min(diff, caps[idx])
            v[idx] += add
```

def project_to_feasible(v, target):
    v = np.clip(v, 0.0, 1.0)
    # water-filling
    # We'll use simple loops: if sum < target, increase free entries equally until either sum reaches target or some hit bound.
    # If sum > target, decrease free entries equally.
    # 
    # Implementation:
    max_iter = 10 * len(v)
    for _ in range(max_iter):
        diff = target - v.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            # can increase
            free = v < 1 - 1e-12
        else:
            free = v > 1e-12
        if not np.any(free):
            break
        # amount to add per free entry
        inc = diff / free.sum()
        # limit
        if diff > 0:
            cap = 1 - v[free]
            add = np.minimum(inc, cap)
            v[free] += add
        else:
            cap = v[free]
            sub = np.maximum(inc, -cap)  # inc negative
            v[free] += sub
    # final clip due to floating errors
    v = np.clip(v, 0.0, 1.0)
    # ensure sum within tolerance: if not adjust a random index
    residual = target - v.sum()
    if abs(residual) > 1e-10:
        # adjust first index that can accommodate
        idx = (np.argmax(v) if residual < 0 else np.argmin(v))
        v[idx] = np.clip(v[idx] + residual, 0.0, 1.0)
    return v

def random_transfer(h, rng):
    n = h.shape[0]
    i, j = rng.integers(n, size=2)
    if i == j:
        return None
    # choose transfer direction
    # possible to move from i to j
    max_transfer = min(h[i], 1 - h[j])
    if max_transfer <= 0:
        # try opposite direction
        max_transfer = min(h[j], 1 - h[i])
        if max_transfer <= 0:
            return None
        i, j = j, i
    eps = rng.random() * max_transfer
    h_new = h.copy()
    h_new[i] -= eps
    h_new[j] += eps
    return h_new

import numpy as np
import time

def _compute_c5(h: np.ndarray) -> float:
    """Return the overlap bound C5 for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _project_feasible(v: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip to [0,1] and adjust to hit the required L1 sum."""
    v = np.clip(v, 0.0, 1.0)
    # water‑filling style adjustment
    for _ in range(100):
        diff = target_sum - v.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            free = v < 1.0 - 1e-12
            if not np.any(free):
                break
            inc = diff / free.sum()
            cap = 1.0 - v[free]
            add = np.minimum(inc, cap)
            v[free] += add
        else:
            free = v > 1e-12
            if not np.any(free):
                break
            dec = diff / free.sum()          # diff is negative
            cap = v[free]
            sub = np.maximum(dec, -cap)
            v[free] += sub
    v = np.clip(v, 0.0, 1.0)
    # tiny residual correction
    residual = target_sum - v.sum()
    if abs(residual) > 1e-10:
        idx = np.argmax(v) if residual < 0 else np.argmin(v)
        v[idx] = np.clip(v[idx] + residual, 0.0, 1.0)
    return v


def _active_gradient(h: np.ndarray):
    """Gradient of the currently maximal correlation term."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    lag = idx - (n - 1)                     # may be negative
    grad = np.zeros_like(h)

    if lag >= 0:
        # sum part (+1) on the prefix that participates
        grad[: n - lag] += 1.0
        # product part (-h[i+lag]) on the same prefix
        grad[: n - lag] -= h[lag:]
        # product part (-h[i-lag]) on the suffix that participates
        grad[lag:] -= h[: n - lag]
    else:
        L = -lag
        grad[L:] += 1.0
        grad[L:] -= h[: n - L]
        grad[: n - L] -= h[L:]

    return grad, lag


def _descent_direction(grad: np.ndarray) -> np.ndarray:
    """Direction that decreases the maximal correlation while preserving the sum."""
    d = -grad
    # remove the component along the all‑ones vector
    d -= np.mean(d)
    return d


def _random_transfer(h: np.ndarray, rng: np.random.Generator):
    """Move a random amount of mass from one coordinate to another."""
    n = h.shape[0]
    i, j = rng.integers(n, size=2)
    if i == j:
        return None
    # try i -> j
    max_transfer = min(h[i], 1.0 - h[j])
    if max_transfer <= 0.0:
        # try the opposite direction
        max_transfer = min(h[j], 1.0 - h[i])
        if max_transfer <= 0.0:
            return None
        i, j = j, i
    eps = rng.random() * max_transfer
    new_h = h.copy()
    new_h[i] -= eps
    new_h[j] += eps
    return new_h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.

    Returns
    -------
    (h_values, c5_bound, n_points)
        h_values : np.ndarray with shape (n_points,)
            The optimized step values in [0,1] satisfying the integral constraint.
        c5_bound : float
            The achieved upper bound C5.
        n_points : int
            Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise
    if "initial_h_values" in globals():
        h = np.asarray(initial_h_values, dtype=np.float64)
    else:
        # a modest default size – easy to work with and well within budget
        n_default = 70
        h = np.full(n_default, 0.5, dtype=np.float64)

    n = h.shape[0]
    target_sum = n / 2.0                     # because integral equals 1

    h = _project_feasible(h, target_sum)

    best_h = h.copy()
    best_cost = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # optimisation parameters
    lr = 0.5                                 # base learning rate
    lr_decay = 0.9995
    min_lr = 1e-4
    no_improve = 0
    start = time.time()
    time_limit = budget_s * 0.95

    while time.time() - start < time_limit:
        # 10 % of the time we try a random transfer move
        if rng.random() < 0.1:
            cand = _random_transfer(best_h, rng)
            if cand is None:
                continue
        else:
            grad, _ = _active_gradient(best_h)
            d = _descent_direction(grad)
            # randomised step length (helps escape flat regions)
            step = lr * (0.5 + 0.5 * rng.random())
            cand = best_h + step * d
            cand = _project_feasible(cand, target_sum)

        cand_cost = _compute_c5(cand)

        if cand_cost < best_cost - 1e-12:
            best_h = cand
            best_cost = cand_cost
            no_improve = 0
            lr = min(lr * 1.02, 5.0)         # a tiny boost after a win
        else:
            no_improve += 1
            lr = max(lr * lr_decay, min_lr)

        # occasional restart if we are stuck for a long while
        if no_improve > 5000:
            # fresh random feasible vector
            rand = rng.random(n)
            rand = rand / rand.sum() * target_sum
            rand = _project_feasible(rand, target_sum)
            cand_cost = _compute_c5(rand)
            if cand_cost < best_cost:
                best_h = rand
                best_cost = cand_cost
                no_improve = 0
                lr = 0.5
        # early exit if we already beat the target record
        if best_cost < 0.3808:
            break

    # final evaluation using the supplied helper (if present)
    if "evaluate_erdos_solution" in globals():
        final_c5 = float(evaluate_erdos_solution(best_h))
    else:
        final_c5 = float(_compute_c5(best_h))

    return best_h, final_c5, n
