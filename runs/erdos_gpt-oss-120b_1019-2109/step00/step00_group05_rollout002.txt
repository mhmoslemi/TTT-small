def _c5_value(arr):
    """Evaluate C5 using the provided helper if available, otherwise fallback."""
    if "evaluate_erdos_solution" in globals():
        return evaluate_erdos_solution(arr)
    # fallback: explicit correlation
    corr = np.correlate(arr, 1.0 - arr, mode="full")
    return corr.max() * (2.0 / arr.shape[0])


def _make_feasible(vec):
    """Project a vector onto the feasible set (0≤h≤1, sum = n/2)."""
    n = len(vec)
    target = n / 2.0
    v = np.clip(vec, 0.0, 1.0)
    s = v.sum()
    if np.isclose(s, target, atol=1e-12):
        return v
    # simple projection: preserve order, assign ones greedily then a possible fractional remainder
    idx = np.argsort(-v)
    h = np.zeros(n)
    whole = int(np.floor(target))
    h[idx[:whole]] = 1.0
    rem = target - whole
    if rem > 1e-12 and whole < n:
        h[idx[whole]] = rem
    return h
