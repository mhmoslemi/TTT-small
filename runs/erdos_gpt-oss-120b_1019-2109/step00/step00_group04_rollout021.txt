def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, cvxpy as cp
    start = time.time()
    # Determine n_points
    try:
        h0 = initial_h_values
        n = len(h0)
    except NameError:
        n = 42  # default
    # Ensure n even:
    if n % 2 != 0:
        n = n + 1
    # Solve MILP
    N = n
    h = cp.Variable(N, boolean=True)
    w = cp.Variable((N,N), boolean=True)
    t = cp.Variable()
    constraints = []
    constraints.append(cp.sum(h) == N/2)
    constraints.append(t >= 0)
    constraints.append(t <= 1)
    # product constraints
    for i in range(N):
        for j in range(N):
            constraints.append(w[i,j] <= h[i])
            constraints.append(w[i,j] <= h[j])
            constraints.append(w[i,j] >= h[i] + h[j] - 1)
    # overlap constraints for each shift
    two_over_N = 2.0 / N
    for s in range(-N+1, N):
        # compute sum of w[i,i+s] where indices valid
        if s >= 0:
            idx_i = range(0, N - s)
            sum_expr = cp.sum([w[i, i+s] for i in idx_i])
        else:
            idx_i = range(-s, N)
            sum_expr = cp.sum([w[i, i+s] for i in idx_i])
        constraints.append(t >= 1 - two_over_N * sum_expr)
    prob = cp.Problem(cp.Minimize(t), constraints)
    # Choose solver
    solved = False
    for solver in [cp.CBC, cp.GLPK_MI, cp.ECOS_BB]:
        try:
            prob.solve(solver=solver, verbose=False, max_iters=200000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                solved = True
                break
        except Exception:
            continue
    if not solved:
        # fallback random search
        np.random.seed(seed)
        N = n
        # binary random start
        h_vals = np.zeros(N)
        indices = np.random.choice(N, N//2, replace=False)
        h_vals[indices] = 1.0
        best_h = h_vals.copy()
        best_score = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0/N)
        # simple hill climbing for limited time
        end_time = start + budget_s - 0.5
        while time.time() < end_time:
            # propose swap
            one_idx = np.where(best_h == 1)[0]
            zero_idx = np.where(best_h == 0)[0]
            i = np.random.choice(one_idx)
            j = np.random.choice(zero_idx)
            new_h = best_h.copy()
            new_h[i] = 0
            new_h[j] = 1
            score = np.max(np.correlate(new_h, 1 - new_h, mode='full')) * (2.0/N)
            if score < best_score:
                best_h = new_h
                best_score = score
        h_vals = best_h
        c5 = best_score

        # ensure correct sum; if off adjust
        current_sum = h_vals.sum()
        target_sum = N/2
        if current_sum != target_sum:
            # adjust by flipping some bits
            diff = int(target_sum - current_sum)
            if diff > 0:
                # need to turn diff zeros to one
                zero_idx = np.where(h_vals == 0)[0]
                chosen = zero_idx[:diff]
                h_vals[chosen] = 1
            elif diff < 0:
                one_idx = np.where(h_vals == 1)[0]
                chosen = one_idx[: -diff]
                h_vals[chosen] = 0
        # Clip to bounds
        h_vals = np.clip(h_vals, 0, 1)
        # compute final bound
        c5 = float(np.max(np.correlate(h_vals, 1 - h_vals, mode='full')) * (2.0/N))

    assert abs(h_vals.sum() - N/2) < 1e-6
    return h_vals, c5, N

import time
import numpy as np
import cvxpy as cp

def _solve_milp(N):
    """Solve the binary MILP for given number of points N."""
    # binary variables for h
    h = cp.Variable(N, boolean=True)
    # binary matrix for products w[i,j] = h[i] * h[j]
    w = cp.Variable((N, N), boolean=True)
    # continuous variable for the maximal overlap t
    t = cp.Variable()
    constraints = []

    # integral constraint: sum h = N/2
    constraints.append(cp.sum(h) == N / 2)

    # bounds on t
    constraints.append(t >= 0)
    constraints.append(t <= 1)

    # product constraints w[i,j] = h[i] * h[j] (linearization)
    for i in range(N):
        for j in range(N):
            constraints.append(w[i, j] <= h[i])
            constraints.append(w[i, j] <= h[j])
            constraints.append(w[i, j] >= h[i] + h[j] - 1)

    # overlap constraints for every integer shift
    two_over_N = 2.0 / N
    for s in range(-N + 1, N):
        if s >= 0:
            idx = range(0, N - s)
            diag_sum = cp.sum([w[i, i + s] for i in idx])
        else:
            idx = range(-s, N)
            diag_sum = cp.sum([w[i, i + s] for i in idx])
        constraints.append(t >= 1 - two_over_N * diag_sum)

    # objective: minimize the worst-case overlap t
    prob = cp.Problem(cp.Minimize(t), constraints)

    # try a few MILP solvers
    for solver in (cp.CBC, cp.GLPK_MI, cp.ECOS_BB):
        try:
            prob.solve(solver=solver, verbose=False, max_iters=200000)
            if prob.status in ("optimal", "optimal_inaccurate"):
                return h, t
        except Exception:
            continue
    return None, None


def _hill_climb(h_vals, budget, start_time):
    """Simple hill‑climbing to lower the max overlap for binary vectors."""
    N = len(h_vals)
    dx = 2.0 / N
    # initial score
    best = h_vals.copy()
    best_score = float(np.max(np.correlate(best, 1 - best, mode='full')) * dx)
    # pre‑compute index sets
    ones = np.where(best == 1)[0]
    zeros = np.where(best == 0)[0]
    rng = np.random.default_rng()
    while time.time() - start_time < budget:
        # propose a swap
        i = rng.choice(ones)
        j = rng.choice(zeros)
        new = best.copy()
        new[i] = 0
        new[j] = 1
        score = float(np.max(np.correlate(new, 1 - new, mode='full')) * dx)
        if score < best_score - 1e-9:
            best = new
            best_score = score
            ones = np.where(best == 1)[0]
            zeros = np.where(best == 0)[0]
    return best, best_score
