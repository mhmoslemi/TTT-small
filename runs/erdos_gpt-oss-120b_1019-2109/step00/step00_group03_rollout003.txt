import time
import math
import numpy as np

def thue_morse(N: int) -> np.ndarray:
    """Return a binary Thue–Morse sequence of length N (N must be a power of two)."""
    return np.array([bin(i).count("1") & 1 for i in range(N)], dtype=np.float64)


def evaluate_h(vec: np.ndarray) -> float:
    """Compute the Erdős overlap bound for a discrete step function."""
    n = vec.size
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return np.max(corr) * (2.0 / n)


def simulated_annealing(start_vec: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Simple SA that swaps a 1 with a 0 while preserving the total mass."""
    cur = start_vec.copy()
    cur_cost = evaluate_h(cur)
    best = cur.copy()
    best_cost = cur_cost

    n = cur.size
    T = 0.5                     # initial temperature
    start = time.time()

    while time.time() - start < time_limit:
        # pick a random 1 and a random 0
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        i = rng.choice(ones)
        j = rng.choice(zeros)

        cand = cur.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_cost = evaluate_h(cand)

        # accept if better, otherwise with Metropolis probability
        if cand_cost < cur_cost or rng.random() < math.exp(-(cand_cost - cur_cost) / max(T, 1e-12)):
            cur = cand
            cur_cost = cand_cost
            if cur_cost < best_cost:
                best = cur.copy()
                best_cost = cur_cost

        # exponential cooling, with occasional reheating to avoid stagnation
        T *= 0.9995
        if T < 1e-6:
            T = 0.5

    return best, best_cost


def greedy_improve(start_vec: np.ndarray, time_limit: float, rng: np.random.Generator):
    """Hill‑climbing: accept only improving swaps."""
    cur = start_vec.copy()
    cur_cost = evaluate_h(cur)
    best = cur.copy()
    best_cost = cur_cost

    n = cur.size
    start = time.time()

    while time.time() - start < time_limit:
        ones = np.where(cur == 1)[0]
        zeros = np.where(cur == 0)[0]
        if ones.size == 0 or zeros.size == 0:
            break

        i = rng.choice(ones)
        j = rng.choice(zeros)

        cand = cur.copy()
        cand[i] = 0.0
        cand[j] = 1.0
        cand_cost = evaluate_h(cand)

        if cand_cost < cur_cost:
            cur = cand
            cur_cost = cand_cost
            if cand_cost < best_cost:
                best = cand.copy()
                best_cost = cand_cost

    return best, best_cost


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    total_start = time.time()

    # ------------------------------------------------------------------
    # Choose an initial vector.
    # Prefer the provided construction if it exists; otherwise fall back to
    # a Thue–Morse sequence of length 128 (a power of two).
    # ------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.size
        # safety: enforce the mass constraint exactly
        target = n // 2
        cur_sum = int(round(h0.sum()))
        if cur_sum != target:
            diff = target - cur_sum
            if diff > 0:
                zeros = np.where(h0 == 0)[0]
                flip = rng.choice(zeros, diff, replace=False)
                h0[flip] = 1.0
            elif diff < 0:
                ones = np.where(h0 == 1)[0]
                flip = rng.choice(ones, -diff, replace=False)
                h0[flip] = 0.0
    else:
        # default size – a power of two lets the Thue–Morse pattern have exact mass 1/2.
        n = 128
        h0 = thue_morse(n)

    # Verify constraints (they must hold for the optimizer)
    assert h0.shape == (n,)
    assert np.all((h0 >= 0) & (h0 <= 1))
    assert int(round(h0.sum())) == n // 2

    # ------------------------------------------------------------------
    # Phase 1 : Simulated annealing (≈80 % of the budget)
    # ------------------------------------------------------------------
    phase1_time = budget_s * 0.78
    best_vec, best_val = simulated_annealing(h0, phase1_time, rng)

    # ------------------------------------------------------------------
    # Phase 2 : Pure greedy hill‑climbing (remaining time)
    # ------------------------------------------------------------------
    remaining = budget_s - (time.time() - total_start)
    if remaining > 0.0:
        best_vec, best_val = greedy_improve(best_vec, remaining, rng)

    # Final evaluation (ensures the bound matches the returned vector)
    final_val = evaluate_h(best_vec)

    # ------------------------------------------------------------------
    # Return the result in the required format
    # ------------------------------------------------------------------
    return (best_vec, final_val, n)
