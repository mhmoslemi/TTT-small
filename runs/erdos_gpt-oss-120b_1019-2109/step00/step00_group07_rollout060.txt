import numpy as np
import random
import time
import math

def compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    # np.correlate implements linear cross‑correlation with zero padding
    corr = np.correlate(h, 1 - h, mode="full")
    return float(np.max(corr) * dx)


def _make_triangle(n: int) -> np.ndarray:
    """Triangular shape: 0→1→0 on [0,2]."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.where(xs <= 1.0, xs, 2.0 - xs)
    # enforce exact integral
    scale = (n / 2.0) / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    # re‑normalise if clipping changed the sum
    scale = (n / 2.0) / h.sum()
    h *= scale
    return h


def _make_sin2(n: int) -> np.ndarray:
    """h(x)=sin^2(pi x/2) on [0,2]."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    h = np.sin(np.pi * xs / 2.0) ** 2
    scale = (n / 2.0) / h.sum()
    h *= scale
    h = np.clip(h, 0.0, 1.0)
    scale = (n / 2.0) / h.sum()
    h *= scale
    return h


def _make_beta_pow(n: int, p: float) -> np.ndarray:
    """h(x) = (x(2-x))**p, scaled to satisfy the integral."""
    dx = 2.0 / n
    xs = (np.arange(n) + 0.5) * dx
    raw = (xs * (2.0 - xs)) ** p
    # avoid scaling >1 (would push some values >1)
    scale = (n / 2.0) / raw.sum()
    if scale > 1.0:          # infeasible with this p on this grid
        return None
    h = raw * scale
    h = np.clip(h, 0.0, 1.0)
    scale = (n / 2.0) / h.sum()
    h *= scale
    return h


def _gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of C(shift) = sum h_i (1-h_{i-shift}) w.r.t. h."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            grad[shift:] += 1.0 - h[: n - shift]
            grad[: n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            grad[: n - s] += 1.0 - h[s:]
            grad[s:] -= h[: n - s]
    return grad

        new_c5 = compute_c5(new_h)

        # temperature schedule – linearly decreasing
        elapsed = time.time() - start_time
        temperature = max(1e-3, (budget_s - elapsed) / budget_s)

        if new_c5 < current_c5 or random.random() < math.exp(-(new_c5 - current_c5) / temperature):
            current_h, current_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_h, best_c5 = new_h.copy(), new_c5
        # slowly shrink step size
        step = max(step * 0.9995, min_step)

        # early exit if we already beat the target comfortably
        if best_c5 < target * 0.998:
            break

    # ------------------------------------------------------------------
    # 4) Final sanity correction (tiny numerical drift)
    # ------------------------------------------------------------------
    # enforce exact integral
    cur_sum = best_h.sum()
    if abs(cur_sum - target_sum) > 1e-10:
        best_h *= target_sum / cur_sum
    # clip once more
    best_h = np.clip(best_h, 0.0, 1.0)

    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n)
