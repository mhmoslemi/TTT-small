import numpy as np
import random
import time
import math

def _compute_c5(h):
    """max_k ∫ h(x)(1‑h(x+k)) dx approximated on the grid."""
    n = h.size
    dx = 2.0 / n
    # np.correlate uses linear (zero‑padded) correlation
    return np.max(np.correlate(h, 1.0 - h, mode='full') * dx)

def _project_capped_simplex(v, target_sum, eps=1e-12):
    """
    Euclidean projection of v onto the set
        {x | 0 ≤ x_i ≤ 1,  sum x_i = target_sum }.
    The solution has the form x_i = clip(v_i - λ, 0, 1) for a suitable λ.
    """
    lo = np.min(v) - 1.0          # guarantees all entries become 1
    hi = np.max(v)                # guarantees all entries become 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        proj = np.clip(v - lam, 0.0, 1.0)
        s = proj.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

def _direction_for_shift(h, shift):
    """
    Gradient direction for a single shift (the worst shift).
    The direction corresponds to d_i = (h_{i+shift}+h_{i-shift})*dx.
    Out‑of‑range indices contribute 0.
    """
    n = h.size
    dx = 2.0 / n
    if shift >= 0:
        plus = np.empty_like(h)
        plus[:n - shift] = h[shift:]
        plus[n - shift:] = 0.0
        minus = np.empty_like(h)
        minus[shift:] = h[:n - shift]
        minus[:shift] = 0.0
    else:
        k = -shift
        plus = np.empty_like(h)
        plus[k:] = h[:n - k]
        plus[:k] = 0.0
        minus = np.empty_like(h)
        minus[:n - k] = h[k:]
        minus[n - k:] = 0.0
    return (plus + minus) * dx

def _optimize_continuous(start_h, time_limit):
    """Projected gradient descent on the smoothed max‑C₅ objective."""
    n = start_h.size
    target_sum = n / 2.0
    h = _project_capped_simplex(start_h, target_sum)
    best_h = h.copy()
    best_val = _compute_c5(h)

    start = time.time()
    it = 0
    # simple decreasing step schedule
    while time.time() - start < time_limit:
        # correlation to locate the worst shift
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        # gradient direction for that shift
        dir_vec = _direction_for_shift(h, shift)

        # step size (tuned empirically)
        step = 5.0 / (1.0 + it * 0.01)

        # tentative update
        h_new = h + step * dir_vec
        h_new = _project_capped_simplex(h_new, target_sum)

        val_new = _compute_c5(h_new)
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()

        # always move (the projection keeps us feasible)
        h = h_new
        it += 1

        # occasional tiny random perturbation to escape plateaus
        if it % 3000 == 0:
            noise = (np.random.rand(n) - 0.5) * 0.005
            h = _project_capped_simplex(h + noise, target_sum)

    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # resolution – 200 points give a good trade‑off between speed and accuracy
    n = 200
    target_sum = n / 2.0

    # ------------------------------------------------------------------
    # 1. Build a starting vector.
    # ------------------------------------------------------------------
    if 'initial_h_values' in globals():
        init = np.asarray(initial_h_values, dtype=float)
        m = init.size
        if m != n:
            # linear interpolation to the desired grid
            x_old = np.linspace(0.0, 2.0, m, endpoint=False)
            x_new = np.linspace(0.0, 2.0, n, endpoint=False)
            start_h = np.interp(x_new, x_old, init)
        else:
