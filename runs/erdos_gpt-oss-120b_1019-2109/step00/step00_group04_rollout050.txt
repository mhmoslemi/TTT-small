def gradient_descent_step(h, step=0.1):
    O = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(O)
    # compute derivative vector d
    d = np.empty_like(h)
    shift = max_idx - (len(h)-1)  # because correlate returns lags -n+1..n-1
    # For each i, compute i+shift:
    for i in range(n):
        idx = i + shift
        if 0 <= idx < n:
            d[i] = 1 - h[idx]
        else:
            d[i] = 1.0
    # pick i with h[i]>0 and max d
    candidates_i = np.where(h > 1e-12)[0]
    if candidates_i.size == 0:
        return False, h
    i = candidates_i[np.argmax(d[candidates_i])]
    # pick j with h[j] < 1 and min d
    candidates_j = np.where(h < 1-1e-12)[0]
    if candidates_j.size == 0:
        return False, h
    j = candidates_j[np.argmin(d[candidates_j])]
    # compute max delta
    max_delta = min(step, h[i], 1 - h[j])
    if max_delta <= 1e-12:
        return False, h

    delta_O = max_delta * (d[i] - d[j])

    h_new[i] -= max_delta
    h_new[j] += max_delta
    # verify O_new is lower
    new_O = np.max(np.correlate(h_new, 1 - h_new, mode='full'))

import time
import random
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdos C5 value for a step function h."""
    n = h.size
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode="full") * dx))

def random_binary_h(n: int) -> np.ndarray:
    """Binary vector of length n with exactly n/2 ones."""
    half = n // 2
    h = np.concatenate([np.ones(half), np.zeros(n - half)])
    np.random.shuffle(h)
    return h.astype(float)

def random_modify(h: np.ndarray):
    """Random small mass transfer between two positions, keeping sum constant."""
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    if np.random.rand() < 0.5:
        # increase i, decrease j
        maxdelta = min(1.0 - h[i], h[j])
        if maxdelta <= 0:
            return None
        delta = maxdelta * np.random.rand() * 0.5
        h2 = h.copy()
        h2[i] += delta
        h2[j] -= delta
    else:
        # increase j, decrease i
        maxdelta = min(1.0 - h[j], h[i])
        if maxdelta <= 0:
            return None
        delta = maxdelta * np.random.rand() * 0.5
        h2 = h.copy()
        h2[j] += delta
        h2[i] -= delta
    return h2

def gradient_descent_once(h: np.ndarray, step: float):
    """One deterministic move that reduces the current worst overlap if possible."""
    n = h.size
    # raw correlation (dx factor omitted – order unchanged)
    corr = np.correlate(h, 1 - h, mode="full")
    max_idx = int(np.argmax(corr))
    shift = max_idx - (n - 1)

    # derivative d_i = 1 - h[i+shift] (zero‑padded outside range)
    d = np.ones(n)
    if shift >= 0:
        d[: n - shift] = 1.0 - h[shift:]
    else:
        d[-shift:] = 1.0 - h[: n + shift]

    # choose donor i (h[i]>0) with largest d_i, receiver j (h[j]<1) with smallest d_j
    donors = np.where(h > 1e-12)[0]
    if donors.size == 0:
        return False, h
    i = donors[np.argmax(d[donors])]

    receivers = np.where(h < 1.0 - 1e-12)[0]
    if receivers.size == 0:
        return False, h
    j = receivers[np.argmin(d[receivers])]

    if d[i] <= d[j] + 1e-12:
        return False, h

    max_delta = min(step, h[i], 1.0 - h[j])
    if max_delta <= 1e-12:
        return False, h

    h_new = h.copy()
    h_new[i] -= max_delta
    h_new[j] += max_delta

    new_max = np.max(np.correlate(h_new, 1 - h_new, mode="full"))
    if new_max < np.max(corr) - 1e-12:
        return True, h_new
    return False, h

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function with a low C5 value."""
    np.random.seed(seed)
    random.seed(seed)
    start = time.time()
    best_h = None
    best_c5 = float("inf")
    best_n = None

    # decide which problem sizes to try
    n_candidates = []
    if "initial_h_values" in globals():
        n0 = len(initial_h_values)
        n_candidates.append(n0)
        if n0 * 2 <= 200:
            n_candidates.append(n0 * 2)
    else:
        n_candidates = [42, 60, 80, 100]

    for n in n_candidates:
        target_sum = n / 2.0

        # starting vector
        if "initial_h_values" in globals() and len(initial_h_values) == n:
            h = np.array(initial_h_values, dtype=float)
        else:
            h = random_binary_h(n)

        # exact integral correction (tiny)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-12:
            for idx in range(n):
                if 0.0 <= h[idx] + diff <= 1.0:
                    h[idx] += diff
                    break

        cur_c5 = compute_c5(h)
        if cur_c5 < best_c5:
            best_c5, best_h, best_n = cur_c5, h.copy(), n

        step = 0.5
        stagnant = 0
        it = 0

        # main optimisation loop
        while time.time() - start < budget_s * 0.98:
            improved = False
            # try deterministic gradient moves
            for _ in range(20):
                success, h_new = gradient_descent_once(h, step)
                if not success:
                    break
                h = h_new
                cur_c5 = compute_c5(h)
                improved = True
                if cur_c5 < best_c5:
                    best_c5, best_h, best_n = cur_c5, h.copy(), n

            if not improved:
                # random perturbation
                h_rand = random_modify(h)
                if h_rand is not None:
                    new_c5 = compute_c5(h_rand)
                    if new_c5 < cur_c5 or random.random() < 0.02:
                        h = h_rand
                        cur_c5 = new_c5
                        if new_c5 < best_c5:
                            best_c5, best_h, best_n = new_c5, h.copy(), n
                stagnant += 1
                if stagnant > 250:
                    # restart from fresh random binary vector
                    h = random_binary_h(n)
                    cur_c5 = compute_c5(h)
                    stagnant = 0
            else:
                stagnant = 0

            it += 1
            if it % 200 == 0:
                step = max(step * 0.7, 0.01)

        if time.time() - start >= budget_s * 0.99:
            break

    # final fine‑tuning on the best solution discovered
    if best_h is not None:
        h = best_h.copy()
        for step in (0.2, 0.1, 0.05, 0.02, 0.01):
            for _ in range(200):
                success, h_new = gradient_descent_once(h, step)
                if success:
                    h = h_new
        best_h = h
        best_c5 = compute_c5(best_h)
        best_n = best_h.size

    return best_h, best_c5, best_n
