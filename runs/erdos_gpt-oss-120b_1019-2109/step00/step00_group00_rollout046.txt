def local_search(h, time_limit, start_time):
    best = h.copy()
    best_score = compute_c5_raw(best) # not multiplied by dx; returns raw max correlation.
    n = len(h)
    # parameters
    max_iter = 1000000
    step = 0.2
    temp0 = 1.0
    for iter in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        # choose i, j distinct
        i, j = random.sample(range(n), 2)
        delta = (random.random() * 2 - 1) * step  # uniform [-step, step]
        if not (0 <= best[i] + delta <= 1 and 0 <= best[j] - delta <= 1):

        new_h[i] += delta
        new_h[j] -= delta

        if score < best_score - 1e-10:  # improvement
            best = new_h
            best_score = score
        else:

            # define temperature schedule: linear decay

            prob = np.exp((best_score - score) / t)  # if score higher, exponent negative
            if random.random() < prob:
                best = new_h
                best_score = score
        # maybe adapt step size every few iterations

def refine_with_slsqp(x0):
    n = len(x0)
    target_sum = n/2
    bounds = [(0.0,1.0)]*n
    cons = [{'type':'eq','fun': lambda x: np.sum(x) - target_sum}]
    res = minimize(lambda x: np.max(np.correlate(x, 1-x, mode='full')), x0,
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter':2000, 'ftol':1e-9, 'disp':False})
    if res.success:
        return res.x, np.max(np.correlate(res.x,1-res.x,mode='full'))
    else:
        return x0, np.max(np.correlate(x0,1-x0,mode='full'))
```

def random_feasible_vector(n, target_sum):
    # Start with zeros
    h = np.zeros(n)
    # For integer part
    int_part = int(math.floor(target_sum))
    # assign int_part ones randomly
    idx = np.random.choice(n, int_part, replace=False)
    h[idx] = 1.0
    remaining = target_sum - int_part
    if remaining > 1e-12:
        # assign fractional to a random index not already 1
        remaining_idx = np.random.choice([i for i in range(n) if h[i] < 1.0])
        h[remaining_idx] = remaining
    # else all fractional 0
    return h

def random_feasible_vector(n, target_sum):
    h = np.zeros(n)
    int_part = int(math.floor(target_sum))
    # choose random indices for 1's
    if int_part > 0:
        idx_ones = np.random.choice(n, int_part, replace=False)
        h[idx_ones] = 1.0
    remaining = target_sum - int_part
    if remaining > 1e-12:
        # choose index not already 1
        candidates = [i for i in range(n) if h[i] < 1.0]
        idx = np.random.choice(candidates)
        h[idx] = remaining
    return h

import time, math, random
import numpy as np
from scipy.optimize import minimize

def _raw_max(h):
    """Maximum of the discrete overlap (without dx scaling)."""
    return np.max(np.correlate(h, 1.0 - h, mode='full'))

def _enforce_sum(arr, target):
    """Adjust a single entry (if possible) to make the sum exactly target."""
    cur = arr.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return arr
    a = arr.copy()
    for i in range(len(a)):
        if 0.0 <= a[i] + diff <= 1.0:
            a[i] += diff
            return a
    # fallback: distribute diff evenly, then clip and fix residual
    a += diff / len(a)
    a = np.clip(a, 0.0, 1.0)
    cur = a.sum()
    diff = target - cur
    for i in range(len(a)):
        if 0.0 <= a[i] + diff <= 1.0:
            a[i] += diff
            break
    return a

def _random_feasible(n, target):
    """Binary (or one fractional) feasible vector with sum == target."""
    h = np.zeros(n, dtype=float)
    int_part = int(math.floor(target))
    if int_part > 0:
        idx = np.random.choice(n, int_part, replace=False)
        h[idx] = 1.0
    remain = target - int_part
    if remain > 1e-12:
        candidates = np.where(h < 1.0)[0]
        idx = np.random.choice(candidates)
        h[idx] = remain
    return h

def _local_search(h0, time_limit, start_time):
    """Simple simulated‑annealing hill‑climb keeping the total sum fixed."""
    n = len(h0)
    h = h0.copy()
    cur_raw = _raw_max(h)
    best_h, best_raw = h.copy(), cur_raw
    while time.time() - start_time < time_limit:
        elapsed = time.time() - start_time
        # temperature and step size decay linearly
        T = max(0.001, 1.0 - elapsed / time_limit)
        step = 0.5 * (1.0 - elapsed / time_limit) + 0.005
        i, j = random.sample(range(n), 2)
        delta = (random.random() * 2.0 - 1.0) * step
        new_i, new_j = h[i] + delta, h[j] - delta
        if not (0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0):
            continue
        h_new = h.copy()
        h_new[i], h_new[j] = new_i, new_j
        new_raw = _raw_max(h_new)
        # acceptance rule
        if new_raw < cur_raw or random.random() < math.exp((cur_raw - new_raw) / T):
            h, cur_raw = h_new, new_raw
            if new_raw < best_raw:
                best_h, best_raw = h_new.copy(), new_raw
    return best_h, best_raw

def _refine_slsqp(x0):
    """One SLSQP refinement from the given start point."""
    n = len(x0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    def obj(x):
        return np.max(np.correlate(x, 1.0 - x, mode='full'))
    res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': 1200, 'ftol': 1e-12, 'disp': False})
    if not res.success:

    # enforce exact sum (numerical drift) and bounds
    x = np.clip(res.x, 0.0, 1.0)
    x = _enforce_sum(x, target)
    return x, _raw_max(x)
