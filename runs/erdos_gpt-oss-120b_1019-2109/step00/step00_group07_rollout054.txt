import time, random, numpy as np

def _evaluate_step(h, dx):
    """Return C5 for a step vector h."""
    conv = np.correlate(h, 1.0 - h, mode='full')
    return conv.max() * dx

def _project_onto_capped_simplex(v, target):
    """Projection of v onto {x | 0<=x<=1, sum(x)=target}."""
    # binary search for lambda such that sum(clipped(v - lam)) == target
    lo, hi = -1.0, 1.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s < target:
            hi = lam
        else:
            lo = lam
    return np.clip(v - lo, 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
    else:
        # fallback: uniform 0.5 (area 1 on [0,2])
        n = 80
        h0 = np.full(n, 0.5, dtype=float)

    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0

    # make sure the start vector respects the constraints
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = _project_onto_capped_simplex(h0, target_sum)

    best_h = h0.copy()
    best_val = _evaluate_step(best_h, dx)

    start_time = time.time()
    time_limit = budget_s - 1.0                     # safety margin

    # ------------------------------------------------------------------
    # optional LP relaxation (cvxpy) – gives a quick good starting point
    try:
        import cvxpy as cp
        h = cp.Variable(n)
        Y = cp.Variable((n, n))
        w = cp.Variable()

        cons = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum,
            w >= 0
        ]

        # McCormick envelope for each product
        for i in range(n):
            for j in range(n):
                cons.append(Y[i, j] >= 0)
                cons.append(Y[i, j] <= h[i])
                cons.append(Y[i, j] <= h[j])
                cons.append(Y[i, j] >= h[i] + h[j] - 1)

        # constraints for every shift
        for lag in range(-n + 1, n):
            start = max(0, -lag)
            end = n - max(0, lag)
            if start >= end:
                continue
            sum_h = cp.sum(h[start:end])
            # sum of min(h_i,h_{i+lag}) will be attained at the upper McCormick bound
            y_terms = [Y[i, i + lag] for i in range(start, end)]
            sum_y = cp.sum(cp.hstack(y_terms))
            cons.append(w >= sum_h - sum_y)

        prob = cp.Problem(cp.Minimize(w), cons)
        prob.solve(solver=cp.ECOS, max_iters=2000,
                   abstol=1e-7, reltol=1e-7, feastol=1e-7)

        if prob.status in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
            h_lp = np.array(h.value).flatten()
            h_lp = _project_onto_capped_simplex(h_lp, target_sum)
            val_lp = _evaluate_step(h_lp, dx)
            if val_lp < best_val:
                best_val = val_lp
                best_h = h_lp.copy()
    except Exception:
        # if cvxpy is missing or the solve fails – ignore and continue
        pass

    # ------------------------------------------------------------------
    # local stochastic improvement (simulated annealing style)
    cur_h = best_h.copy()
    cur_val = best_val

    # parameters for the search
    base_step = 0.08            # maximum mass moved in a single operation
    temp0 = 0.04                # initial temperature for occasional uphill moves
    it = 0

    while time.time() - start_time < time_limit:
        it += 1
        # temperature linearly decays to zero
        t = temp0 * max(0.0, 1.0 - (time.time() - start_time) / time_limit)

        # decide move type
        if random.random() < 0.65:
            # gradient guided move – use the shift that currently dominates C5
            conv = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            lag = conv.argmax() - (n - 1)

            g = np.ones(n)
            if lag >= 0:
                nn = n - lag
                if nn > 0:
                    g[:nn] -= cur_h[lag:]
                    g[lag:] -= cur_h[:nn]
            else:
                l = -lag
                nn = n - l
                if nn > 0:
                    g[l:] -= cur_h[:nn]
                    g[:nn] -= cur_h[l:]

            pos = np.where(g > 1e-9)[0]
            neg = np.where(g < -1e-9)[0]
            if pos.size and neg.size:
                i = int(random.choice(pos))
                j = int(random.choice(neg))
                max_delta = min(base_step, cur_h[i], 1.0 - cur_h[j])
                if max_delta <= 1e-12:
                    continue
                delta = max_delta * random.random()
                cand = cur_h.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                # fall back to a plain random pair
                i, j = random.sample(range(n), 2)
                max_delta = min(base_step, cur_h[i], 1.0 - cur_h[j])
                if max_delta <= 1e-12:
                    continue
                delta = max_delta * random.random()
                cand = cur_h.copy()
                cand[i] -= delta
                cand[j] += delta
        else:
            # simple random transfer
            i, j = random.sample(range(n), 2)
            max_delta = min(base_step, cur_h[i], 1.0 - cur_h[j])
            if max_delta <= 1e-12:
                continue
            delta = max_delta * random.random()
            cand = cur_h.copy()
            cand[i] -= delta
            cand[j] += delta

        # evaluate candidate
        cand_val = _evaluate_step(cand, dx)

        # acceptance test
        if cand_val < cur_val - 1e-12 or random.random() < np.exp(-(cand_val - cur_val) / max(t, 1e-12)):
            cur_h, cur_val = cand, cand_val
            if cand_val < best_val - 1e-12:
                best_h, best_val = cand.copy(), cand_val

        # occasional refresh from a fresh random binary pattern
        if it % 15000 == 0:
            trial = np.zeros(n)
            idx = np.random.choice(n, n // 2, replace=False)
            trial[idx] = 1.0
            trial_val = _evaluate_step(trial, dx)
            if trial_val < best_val:
                cur_h, cur_val = trial.copy(), trial_val
                best_h, best_val = trial.copy(), trial_val

    # final projection to guarantee feasibility
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_onto_capped_simplex(best_h, target_sum)

    final_c5 = float(_evaluate_step(best_h, dx))
    return (best_h.tolist(), final_c5, n)
