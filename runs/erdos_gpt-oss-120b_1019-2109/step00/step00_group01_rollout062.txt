```python
def beatty_binary(n, alpha):
    xs = np.arange(n) * (2.0 / n)  # positions in [0,2)
    vals = (alpha * xs) % 2.0
    h = (vals < 1.0).astype(np.int8)
    # Ensure exactly n/2 ones: adjust by swapping if needed.
    current_ones = h.sum()
    target = n // 2
    if current_ones > target:
        # Turn some ones to zeros
        ones_idx = np.where(h == 1)[0]
        np.random.shuffle(ones_idx)
        h[ones_idx[:current_ones - target]] = 0
    elif current_ones < target:
        zeros_idx = np.where(h == 0)[0]
        np.random.shuffle(zeros_idx)
        h[zeros_idx[:target - current_ones]] = 1
    return h
```

```python
def hill_climb(h, n_iters, target_min_dot):
    n = len(h)
    best_h = h.copy()
    best_min = np.correlate(h, h, mode='full').min()
    best_c5 = 1 - (2 * best_min) / n

    ones = np.flatnonzero(best_h == 1)
    zeros = np.flatnonzero(best_h == 0)
    for it in range(n_iters):
        # random swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h[i], h[j] = 0, 1
        # compute min dot
        corr = np.correlate(h, h, mode='full')
        cur_min = corr.min()
        # if better (higher) min dot
        if cur_min > best_min:
            best_min = cur_min
            best_c5 = 1 - (2 * best_min) / n
            best_h = h.copy()
            # update ones zeros arrays
            ones = np.flatnonzero(h == 1)
            zeros = np.flatnonzero(h == 0)
            if best_min >= target_min_dot:
                break
        else:
            # revert if not accepted (or maybe accept with probability)
            # revert swap
            h[i], h[j] = 1, 0
        # Optionally also update ones zeros but they stay same.

    return best_h, best_c5, best_min
```

import time, math
import numpy as np

def _c5_of_h(h: np.ndarray) -> float:
    """Return the C5 value for a step function h on [0,2] discretized."""
    n = h.shape[0]
    # correlation of h with (1-h); zero padding is handled by np.correlate
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * (2.0 / n))


def _try_transfer(h: np.ndarray, src: int, dst: int, rng: np.random.Generator,
                  scale: float) -> float | None:
    """
    Move a small amount of mass from src to dst keeping 0<=h<=1.
    Returns the transferred amount or None if no transfer possible.
    """
    max_eps = min(h[src], 1.0 - h[dst])
    if max_eps <= 0.0:
        return None
    eps = rng.random() * max_eps * scale
    if eps <= 0.0:
        return None
    h[src] -= eps
    h[dst] += eps
    return eps


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step function h on [0,2] (discretised) that minimises
    the Erdős overlap constant C5.

    Returns
    -------
    h_values : np.ndarray
        The best candidate found (values in [0,1]).
    c5_bound : float
        The corresponding C5 value.
    n_points : int
        Number of discretisation points used.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.1  # safety margin

    # ------------------------------------------------------------------
    # 1.  Use the supplied initial construction if it exists.
    # ------------------------------------------------------------------
    best_h = None
    best_c5 = float("inf")
    if "initial_h_values" in globals():
        try:
            h0 = np.asarray(initial_h_values, dtype=float)
            c0 = _c5_of_h(h0)
            best_h = h0.copy()
            best_c5 = c0
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2.  Choose a working discretisation size (even for simplicity).
    # ------------------------------------------------------------------
    n = 200                     # even number gives exact half‑mass as integers
    target_sum = n / 2.0

    # initialise with a flat profile (h_i = 0.5) – respects the integral constraint
    h = np.full(n, 0.5, dtype=float)
    cur_c5 = _c5_of_h(h)
    if cur_c5 < best_c5:
        best_h = h.copy()
        best_c5 = cur_c5

    # ------------------------------------------------------------------
    # 3.  Simulated‑annealing style improvement.
    # ------------------------------------------------------------------
    T_start = 0.10               # initial "temperature"
    T_end = 1e-5                 # final temperature
    move_scale = 0.5             # maximal fraction of allowed transfer per step

    while time.time() < deadline:
        # linearly anneal temperature
        elapsed = time.time() - start
        progress = elapsed / (budget_s - 0.1)
        T = T_start * (1.0 - progress) + T_end * progress

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # decide direction of transfer
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i

        eps = _try_transfer(h, src, dst, rng, move_scale)
        if eps is None:
            continue

        new_c5 = _c5_of_h(h)
        delta = new_c5 - cur_c5
        if delta < 0.0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            # accept move
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                # early stop if we already beat the target record
                if best_c5 <= 0.38085:
                    break
        else:
            # reject – revert the transfer
            h[src] += eps
            h[dst] -= eps

    # ------------------------------------------------------------------
    # 4.  Final sanity adjustments (clip and enforce exact integral).
    # ------------------------------------------------------------------
    if best_h is None:
        best_h = h.copy()
        best_c5 = _c5_of_h(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)

    # enforce the exact integral condition sum(h) = n/2
    cur_sum = best_h.sum()
    diff = target_sum - cur_sum
    if abs(diff) > 1e-12:
        # distribute the tiny remainder uniformly and clip once more

        best_h = np.clip(best_h, 0.0, 1.0)
        # final renormalisation if necessary
        cur_sum = best_h.sum()
        if abs(cur_sum - target_sum) > 1e-8:
            best_h *= target_sum / cur_sum
            best_h = np.clip(best_h, 0.0, 1.0)

    # recompute final C5 for safety
    best_c5 = _c5_of_h(best_h)

    return best_h, best_c5, n
