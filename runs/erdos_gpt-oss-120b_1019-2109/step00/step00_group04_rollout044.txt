import time, math, random
import numpy as np

def _c5_of_h(h):
    """Compute the C5 value for a step function h on [0,2]."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _random_binary_feasible(n, rng):
    """Return a feasible binary vector (sum = n/2) for even n."""
    half = n // 2
    idx = np.arange(n)
    rng.shuffle(idx)
    h = np.zeros(n, dtype=float)
    h[idx[:half]] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function giving a good upper bound for C5."""
    random.seed(seed)
    rng = np.random.default_rng(seed)

    # ---- initialise ---------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        n = h0.shape[0]
    else:
        # fallback: even length, simple start
        n = 84
        h0 = np.full(n, 0.5, dtype=float)

    target_sum = n / 2.0
    # make sure the initial vector respects the constraints
    cur_sum = h0.sum()
    if abs(cur_sum - target_sum) > 1e-12:
        # tiny correction on first element, respecting bounds
        delta = target_sum - cur_sum
        h0[0] = np.clip(h0[0] + delta, 0.0, 1.0)

    # ensure exact sum after possible clipping
    diff = h0.sum() - target_sum
    if abs(diff) > 1e-12:
        # redistribute the tiny difference uniformly (will stay within [0,1] for the
        # tiny adjustments we are dealing with)
        h0 -= diff / n

    # ---- local search utilities --------------------------------------------
    def propose_swap(state):
        """Generate a feasible neighbour by swapping a delta between two entries."""
        i, j = random.sample(range(n), 2)
        hi, hj = state[i], state[j]

        # maximal allowed positive and negative delta preserving bounds
        max_pos = min(1.0 - hi, hj)      # increase i, decrease j
        max_neg = -min(hi, 1.0 - hj)     # decrease i, increase j
        if max_pos <= 0.0 and max_neg >= 0.0:
            return None, None, None, None

        # pick a delta inside the allowed interval, but limit step size
        raw = random.uniform(max_neg, max_pos)
        step_limit = 0.08  # maximal absolute change per move
        if raw > step_limit:
            raw = step_limit
        elif raw < -step_limit:
            raw = -step_limit
        # ensure after clipping we still respect bounds
        raw = max(raw, max_neg)
        raw = min(raw, max_pos)

        # apply
        state[i] += raw
        state[j] -= raw
        return (i, j, raw, (hi, hj))

    # ---- main optimisation loop --------------------------------------------
    best_h = h0.copy()
    best_val = _c5_of_h(best_h)

    # current state for the annealing run
    cur_h = h0.copy()
    cur_val = best_val

    start_time = time.time()
    last_improve = 0
    it = 0
    max_no_improve = 30000  # restart after this many iterations without improvement

    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.97:
            break

        # linear temperature schedule (very low, just to allow occasional uphill moves)
        frac = elapsed / budget_s
        temp = 1e-3 * (1.0 - frac) + 1e-6 * frac

        # propose a neighbour
        prop = propose_swap(cur_h)
        if prop[0] is None:
            continue
        i, j, delta, old_vals = prop

        new_val = _c5_of_h(cur_h)

        # acceptance test
        accept = False
        if new_val <= cur_val:
            accept = True
        else:
            # simulated annealing step
            prob = math.exp(-(new_val - cur_val) / temp) if temp > 0 else 0.0
            if random.random() < prob:
                accept = True

        if accept:
            cur_val = new_val
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_h = cur_h.copy()
                last_improve = it
        else:
            # revert move
            hi_old, hj_old = old_vals
            cur_h[i] = hi_old
            cur_h[j] = hj_old

        it += 1

        # occasional restart if stuck
        if it - last_improve > max_no_improve:
            # try a fresh random binary feasible vector
            cur_h = _random_binary_feasible(n, rng)
            cur_val = _c5_of_h(cur_h)
            last_improve = it

    # final evaluation (use the same routine that the evaluator expects)
    final_c5 = _c5_of_h(best_h)

    # Return list (or array) of values, the bound, and number of points.
    return (best_h.tolist(), float(final_c5), n)
