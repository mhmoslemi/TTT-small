import time
import numpy as np

def compute_c5(h: np.ndarray, dx: float) -> float:
    """Maximum overlap integral for a given step function."""
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def optimize_h(initial_h: np.ndarray, dx: float, budget_s: float,
               start_time: float, rng: np.random.Generator):
    """
    Heuristic local search for a step function with low C5.
    Returns the best vector found and its C5 value.
    """
    h = initial_h.astype(np.float64).copy()
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    eps = 1e-12
    stagnant = 0
    max_stagnant = 300_000

    # temperature for simulated‑annealing‑style uphill moves
    T = 0.02
    cooling = 0.9999
    it = 0
    idx_arr = np.arange(n)

    while True:
        # time limit
        if time.time() - start_time >= budget_s * 0.98:
            break

        # current overlap profile
        corr = np.correlate(h, 1.0 - h, mode="full")
        corr_vals = corr * dx
        max_idx = int(np.argmax(corr_vals))
        cur_c5 = float(corr_vals[max_idx])

        # keep the best ever seen
        if cur_c5 < best_c5 - eps:
            best_c5 = cur_c5
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

        # stop early if already very good
        if best_c5 <= 0.3808:
            break

        # shift that attains the maximum
        shift = max_idx - (n - 1)

        # neighbor values for that shift
        neighbor = np.zeros(n, dtype=np.float64)
        valid = (idx_arr + shift >= 0) & (idx_arr + shift < n)
        neighbor[valid] = h[idx_arr[valid] + shift]

        zero_mask = h < eps
        one_mask = h > 1.0 - eps

        # -------- greedy swap (always down‑hill) ----------
        if zero_mask.any() and one_mask.any():
            neigh_zero = np.where(zero_mask, neighbor, -np.inf)
            i_up = int(np.argmax(neigh_zero))
            neigh_one = np.where(one_mask, neighbor, np.inf)
            j_down = int(np.argmin(neigh_one))
            if neighbor[i_up] > neighbor[j_down] + eps:
                new_h = h.copy()

                new_h[j_down] = 0.0
                new_c5 = compute_c5(new_h, dx)
                if new_c5 < cur_c5 - eps:
                    h = new_h
                    it += 1

                    continue  # restart loop with improved state
