import time, numpy as np, math

def proj_box_simplex(v, s):
    n = len(v)
    # Binary search lambda
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0, 1)
        total = x.sum()
        if total < s:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo+hi)/2.0, 0, 1)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return float(np.max(corr) * dx)

def subgradient(h, d, dx):
    n = len(h)
    neigh = np.zeros_like(h)
    if d >= 0:

        if d_abs < n:
            neigh[d_abs:] += h[:n-d_abs]
            neigh[:n-d_abs] += h[d_abs:]

import time
import math
import numpy as np

def _proj_box_simplex(v, target_sum):
    """Project v onto {x: 0<=x<=1, sum(x)=target_sum}."""
    n = v.shape[0]
    # bounds for the Lagrange multiplier λ such that sum(clamp(v+λ,0,1))=target_sum
    lo = -np.max(v) - 1.0          # ensures all v+λ <= 0  -> sum = 0
    hi = 1.0 - np.min(v) + 1.0     # ensures all v+λ >= 1  -> sum = n
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        s = x.sum()
        if s < target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _c5_value(h, dx):
    """Return max_k ∫ h(x)(1-h(x+k))dx for the sampled h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * dx)

def _subgradient(h, shift, dx):
    """Subgradient of the max‑overlap objective at shift (integer offset)."""
    n = h.shape[0]
    neigh = np.zeros_like(h)
    if shift >= 0:
        if shift < n:
            neigh[:n - shift] += h[shift:]
            neigh[shift:] += h[:n - shift]
    else:
        d = -shift
        if d < n:
            neigh[d:] += h[:n - d]
            neigh[:n - d] += h[d:]
    return dx * (1.0 - neigh)

def _random_feasible(n, target_sum):
    """Uniform random point in the box-simplex."""
    v = np.random.rand(n)
    return _proj_box_simplex(v, target_sum)

def _refine(h, n, dx, time_limit):
    """Project‑gradient + occasional hill‑climb limited by time_limit seconds."""
    start = time.time()
    target_sum = n * 0.5
    best_h = h.copy()
    best_val = _c5_value(best_h, dx)
    cur_val = best_val
    it = 0
    while time.time() - start < time_limit:
        # correlation & shift giving the current maximum
        corr = np.correlate(h, 1.0 - h, mode='full')
        imax = int(np.argmax(corr))
        shift = imax - (n - 1)

        # subgradient for that shift
        g = _subgradient(h, shift, dx)

        # simple diminishing step size
        alpha = 0.2 / math.sqrt(it + 1.0)

        # projected gradient step
        h_new = _proj_box_simplex(h - alpha * g, target_sum)
        val_new = _c5_value(h_new, dx)

        # keep the best ever seen
        if val_new < best_val:
            best_val = val_new
            best_h = h_new.copy()

        # accept if we improved the current objective
        if val_new < cur_val - 1e-12:
            h = h_new
            cur_val = val_new

        # occasional small random exchange (preserves sum exactly)
        if it % 20 == 0:
            i, j = np.random.choice(n, 2, replace=False)
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max > 1e-12:
                delta = delta_max * np.random.rand()
                h_move = h.copy()
                h_move[i] -= delta
                h_move[j] += delta
                val_move = _c5_value(h_move, dx)
                if val_move < best_val:
                    best_val = val_move
                    best_h = h_move.copy()
                if val_move < cur_val:
                    h = h_move
                    cur_val = val_move

        it += 1
        # safety break on iteration count
        if it >= 5000:
            break
    return best_h, best_val
