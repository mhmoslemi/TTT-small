    # initial size

    else:

    # target size

        x_old = np.linspace(0,2, n0, endpoint=False)
        x_new = np.linspace(0,2, n, endpoint=False)
        h = np.interp(x_new, x_old, h0)

def project_onto_box_sum(y, target_sum):
    low = -np.max(y) - 10.0
    high = 1.0 - np.min(y) + 10.0
    for _ in range(60):
        lam = (low + high) / 2.0

        s = x.sum()

            high = lam
        else:
            low = lam

def upscale_h(h_old, n_new):
    n_old = h_old.size
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)
```

```
def gradient_worst_shift(h, dx):
    n = h.size
    conv = np.correlate(h, 1-h, mode='full')
    idx = np.argmax(conv)
    d = idx - (n-1)
    grad = np.zeros_like(h)
    low = max(0, -d)
    high = min(n-1, n-1-d)
    i = np.arange(low, high+1, dtype=int)
    j = i + d
    grad[i] += 1.0
    grad[i] -= h[j]
    grad[j] -= h[i]
    grad *= dx
    return grad, conv.max()*dx, d
```

def gd_optimize(h, max_iters, step0, decay, target_sum):

    dx = 2.0 / n

import numpy as np, time

def project_onto_box_sum(y, target):
    low = -np.max(y) - 1.0
    high = 1.0 - np.min(y) + 1.0
    for _ in range(60):
        lam = (low+high)*0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            high = lam
        else:
            low = lam
    return np.clip(y + (low+high)*0.5, 0.0, 1.0)

def upscale_h(old_h, new_n):
    old_n = old_h.size
    if old_n == new_n:
        return old_h.copy()
    x_old = np.linspace(0,2,old_n,endpoint=False) + 1.0/old_n  # shift to midpoints maybe not needed
    x_new = np.linspace(0,2,new_n,endpoint=False) + 1.0/new_n
    return np.interp(x_new, x_old, old_h)

def gradient_worst_shift(h, dx):
    n = h.shape[0]
    corr = np.correlate(h, 1 - h, mode='full')
    idx = np.argmax(corr)
    d = idx - (n-1)
    # compute gradient
    grad = np.zeros_like(h)
    low = max(0, -d)
    high = min(n-1, n-1-d)
    i = np.arange(low, high+1, dtype=int)
    j = i + d
    grad[i] += 1.0
    grad[i] -= h[j]
    grad[j] -= h[i]
    grad *= dx
    # return gradient, current value, shift
    val = corr[idx] * dx
    return grad, val, d

def gd_optimize(start_h, iters, step0, decay, target_sum):
    h = start_h.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1-h, mode='full')) * dx
    step = step0
    for _ in range(iters):
        grad, val, d = gradient_worst_shift(h, dx)
        # update best

            best_val = val
            best_h = h.copy()
        # gradient step
        y = h - step * grad

    # final check

        best_val = final_val
        best_h = h.copy()
    return best_h, best_val

import time
import numpy as np

def _project_onto_box_sum(y: np.ndarray, target_sum: float) -> np.ndarray:
    """Projection onto {0<=x_i<=1, sum(x)=target_sum} using a scalar shift."""
    low = -np.max(y) - 1.0   # makes all entries <=0 after shift → sum≈0
    high = 1.0 - np.min(y) + 1.0  # makes all entries >=1 after shift → sum≈n
    for _ in range(60):
        lam = (low + high) * 0.5
        x = np.clip(y + lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            high = lam
        else:
            low = lam
    return np.clip(y + (low + high) * 0.5, 0.0, 1.0)


def _upscale_h(old_h: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a step function sampled at old_n points to new_n points."""
    old_n = old_h.size
    if old_n == new_n:
        return old_h.copy()
    # sample at mid‑points for a more accurate integral approximation
    dx_old = 2.0 / old_n
    dx_new = 2.0 / new_n
    x_old = (np.arange(old_n) + 0.5) * dx_old
    x_new = (np.arange(new_n) + 0.5) * dx_new
    return np.interp(x_new, x_old, old_h)


def _gradient_worst_shift(h: np.ndarray, dx: float):
    """Gradient of the current worst (maximal) overlap term."""
    n = h.size
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    d = idx - (n - 1)               # integer shift index
    # overlapping index range
    lo = max(0, -d)
    hi = min(n - 1, n - 1 - d)
    i = np.arange(lo, hi + 1, dtype=int)
    j = i + d
    grad = np.zeros_like(h)
    grad[i] += 1.0
    grad[i] -= h[j]
    grad[j] -= h[i]
    grad *= dx
    val = corr[idx] * dx
    return grad, val, d


def _gd_optimize(start_h: np.ndarray, iters: int,
                 step0: float, decay: float,
                 target_sum: float):
    """Simple projected gradient descent on the max‑overlap objective."""
    h = start_h.copy()
    n = h.size
    dx = 2.0 / n
    best_h = h.copy()
    best_val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    step = step0
    for _ in range(iters):
        grad, val, _ = _gradient_worst_shift(h, dx)
        if val < best_val - 1e-12:
            best_val = val
            best_h = h.copy()
        y = h - step * grad
        h = _project_onto_box_sum(y, target_sum)
        step *= decay
    final_val = np.max(np.correlate(h, 1.0 - h, mode="full")) * dx
    if final_val < best_val:
        best_val = final_val
        best_h = h.copy()
    return best_h, best_val


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # --- resolution ---------------------------------------------------------
    n = 200                     # number of samples (fast enough, still accurate)
    target_sum = n / 2.0
    dx = 2.0 / n

    # --- initial construction -----------------------------------------------
    if "initial_h_values" in globals():

        h = _upscale_h(h0, n)
    else:
        # triangular prototype: h(x)=max(0,1‑|x‑1|)
        x = (np.arange(n) + 0.5) * dx
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))

    h = np.clip(h, 0.0, 1.0)
    h = _project_onto_box_sum(h, target_sum)

    best_h = h.copy()
    best_val = np.max(np.correlate(best_h, 1.0 - best_h, mode="full")) * dx

    # --- optimisation parameters --------------------------------------------
    step0 = 0.7
    decay = 0.995
    iters_per_restart = 3000

    # --- main optimisation loop ---------------------------------------------
    while time.time() - start_time < budget_s * 0.95:
        # perturb current best solution
        eps = 0.05 + 0.1 * rng.random()
        h_start = best_h + rng.normal(scale=eps, size=n)

            if val_opt2 < best_val:

                best_h = h_opt2.copy()

    # final evaluation (exact same formula used by the harness)
