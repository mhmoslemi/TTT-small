def project(v, S):
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo+hi)/2
        projected = np.clip(v+mid, 0, 1)
        total = projected.sum()
        if total > S:
            hi = mid
        else:
            lo = mid
    x = np.clip(v+ (lo+hi)/2, 0, 1)
    # after bisection ensure sum approx S; adjust small error by adding residual to a free index not at bound (if any)
    diff = S - x.sum()
    if abs(diff) > 1e-10:
        # find index not at bound if any
        free = np.where((x>0+1e-12) & (x<1-1e-12))[0]
        if free.size>0:
            i = free[0]
            x[i] = np.clip(x[i]+diff, 0, 1)
        else:
            # adjust first element
            i = np.argmax(x) if diff<0 else np.argmin(x)
            x[i] = np.clip(x[i]+diff, 0, 1)
    # final clip
    x = np.clip(x, 0, 1)
    return x

def compute_c5(h):
    # h array length n; values expected within [0,1]; sum = n/2.
    n = h.shape[0]
    dx = 2.0 / n
    # compute cross correlation of h and (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # max value
    max_corr = corr.max()
    return max_corr * dx

import time
import numpy as np
import scipy.optimize as _opt

_GLOBAL_N = None  # will be set inside run()


def _eq_sum(h):
    """Equality constraint: sum(h) == N/2."""
    if _GLOBAL_N is None:
        raise RuntimeError("Global N not set")
    return h.sum() - _GLOBAL_N / 2.0


def compute_c5_bound(h):
    """Return C₅ = max_k ∫ h(x)(1‑h(x+k)) dx for the discrete sample."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)


def _project_to_feasible(v, target_sum):
    """
    Project vector v onto the intersection of the hyperplane sum(x)=target_sum
    and the box 0 ≤ x ≤ 1.  Uses a simple bisection on the shift λ.
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction to satisfy the sum exactly (within floating error)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = np.where((x > 1e-9) & (x < 1.0 - 1e-9))[0]
        if free.size:
            i = free[0]
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
        else:
            # all coordinates are at a bound; adjust the one farthest from its bound
            i = np.argmin(x) if diff > 0 else np.argmax(x)
            x[i] = np.clip(x[i] + diff, 0.0, 1.0)
    return x

    start_time = time.time()

    # ----------------------------------------------------------------------
    # Choose discretisation size.
    # If the parent provides an initial vector, reuse its length (which is
    # usually small and already good).  Otherwise fall back to a modest size.
    # ----------------------------------------------------------------------
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
    else:
        # a default of 48 points (even) gives enough flexibility while staying fast
        h0 = np.full(48, 0.5, dtype=float)

    n = h0.shape[0]
    global _GLOBAL_N
    _GLOBAL_N = n
    target_sum = n / 2.0

    # Ensure the starting vector satisfies the constraints.
    h0 = _project_to_feasible(h0, target_sum)

    best_h = h0
    best_score = compute_c5_bound(best_h)

    # ----------------------------------------------------------------------
    # Helper objects for the optimizer.
    # ----------------------------------------------------------------------
    bounds = [(0.0, 1.0)] * n
    eq_cons = {"type": "eq", "fun": _eq_sum}

    # ----------------------------------------------------------------------
    # First run: try to improve the provided solution with a modest local
    # optimization (SLSQP).  This is cheap and may already give a better bound.
    # ----------------------------------------------------------------------
    try:
        res = _opt.minimize(
            compute_c5_bound,
            best_h,
            method="SLSQP",
            bounds=bounds,
            constraints=[eq_cons],
            options={"maxiter": 400, "ftol": 1e-12, "disp": False},
        )
        if res.success:
            cand = _project_to_feasible(res.x, target_sum)
            cand_score = compute_c5_bound(cand)
            if cand_score < best_score:
                best_h, best_score = cand, cand_score
    except Exception:
        # optimizer failure is non‑critical – we simply keep the current best.
        pass

    # ----------------------------------------------------------------------
    # Main randomized search loop.
    # We keep generating feasible starting points, run a short SLSQP
    # optimisation from each, and keep any improvement.
    # ----------------------------------------------------------------------
    iteration = 0
    while time.time() - start_time < budget_s * 0.90:
        iteration += 1

        # ---- random restart -------------------------------------------------
        rnd = np.random.rand(n)
        x0 = _project_to_feasible(rnd, target_sum)
        try:
            res = _opt.minimize(
                compute_c5_bound,
                x0,
                method="SLSQP",
                bounds=bounds,
                constraints=[eq_cons],
                options={"maxiter": 300, "ftol": 1e-12, "disp": False},
            )
            if res.success:
                cand = _project_to_feasible(res.x, target_sum)
                cand_score = compute_c5_bound(cand)
                if cand_score < best_score:
                    best_h, best_score = cand, cand_score
        except Exception:
            pass

        # ---- perturbation of the current best -------------------------------
        if iteration % 4 == 0:
            pert = best_h + 0.25 * np.random.randn(n)
            pert = _project_to_feasible(pert, target_sum)
            try:
                res = _opt.minimize(
                    compute_c5_bound,
                    pert,
                    method="SLSQP",
                    bounds=bounds,
                    constraints=[eq_cons],
                    options={"maxiter": 200, "ftol": 1e-12, "disp": False},
                )
                if res.success:
                    cand = _project_to_feasible(res.x, target_sum)
                    cand_score = compute_c5_bound(cand)
                    if cand_score < best_score:
                        best_h, best_score = cand, cand_score
            except Exception:
                pass

        # ------------------------------------------------------------------
        # occasional pure stochastic refinement (no optimizer) – cheap hill
        # climb that can escape tiny local minima.
        # ------------------------------------------------------------------
        if iteration % 7 == 0:
            cand = best_h + 0.10 * np.random.randn(n)
            cand = _project_to_feasible(cand, target_sum)
            cand_score = compute_c5_bound(cand)
            if cand_score < best_score:
                best_h, best_score = cand, cand_score

    # Final projection (safety) and bound computation.
    best_h = _project_to_feasible(best_h, target_sum)
    final_score = compute_c5_bound(best_h)

    return best_h, final_score, n
