def hill_climb(h, time_budget):
    start = time.time()
    n = len(h)
    best_h = h.copy()
    best_val = evaluate_c5(h)
    cur_val = best_val
    step = 0.1  # initial transfer amount
    eps = 1e-12
    while time.time() - start < time_budget * 0.95:   # keep some margin
        # compute worst shift
        corr = np.correlate(h, 1 - h, mode='full')
        idx = np.argmax(corr)
        shift = idx - (n - 1)
        d = abs(shift)
        # compute forward/backward
        forward = np.zeros(n)
        backward = np.zeros(n)
        if shift >= 0:
            if d < n:
                forward[:n-d] = h[d:]
                backward[d:] = h[:n-d]
        else:
            if d < n:
                forward[d:] = h[:n-d]
                backward[:n-d] = h[d:]
        g = 1 - forward - backward
        # find feasible i_high, i_low
        high_mask = (h > eps)
        low_mask = (h < 1 - eps)
        # Use only indices where gradient highest among high mask
        if not np.any(high_mask) or not np.any(low_mask):
            break
        # compute some candidate indices
        # maybe take top few to avoid ties
        high_candidates = np.where(high_mask)[0]
        low_candidates = np.where(low_mask)[0]
        # compute g for those masked
        i_high = high_candidates[np.argmax(g[high_candidates])]
        i_low = low_candidates[np.argmin(g[low_candidates])]
        # compute delta allowable
        delta = min(step, h[i_high], 1 - h[i_low])
        if delta <= 0:
            # reduce step perhaps
            step *= 0.5
            continue
        # apply move
        h[i_low] += delta
        h[i_high] -= delta
        new_val = evaluate_c5(h)
        # Accept move if improvement (lower)
        if new_val < cur_val:
            cur_val = new_val
            # maybe increase step slowly
            step = min(step * 1.05, 0.5)  # bound step
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i_low] -= delta
            h[i_high] += delta
            # reduce step
            step *= 0.7
        # add occasional random perturbation
        if np.random.rand() < 0.01:
            # random swap of small amount between two random indices
            i1, i2 = np.random.choice(n, 2, replace=False)

                        best_val = new_val2

def hill_climb(h, time_budget, dx):
    start = time.time()
    n = len(h)
    cur_h = h.copy()
    cur_val = evaluate(cur_h)
    best_h_local = cur_h.copy()
    best_val_local = cur_val
    step = 0.2
    eps = 1e-12
    while time.time() - start < time_budget:
        # compute correlation, max shift
        corr = np.correlate(cur_h, 1 - cur_h, mode='full')
        idx = corr.argmax()
        shift = idx - (n - 1)
        d = abs(shift)
        # build forward/backward
        forward = np.zeros(n)
        backward = np.zeros(n)
        if shift >= 0:
            if d < n:
                forward[:n-d] = cur_h[d:]
                backward[d:] = cur_h[:n-d]
        else:
            if d < n:
                forward[d:] = cur_h[:n-d]
                backward[:n-d] = cur_h[d:]
        g = 1 - forward - backward

        # find feasible indices
        # for decreasing: we need index with positive gradient (greater g) and h>eps (to reduce)
        # for increasing: lower gradient (smaller g) and h<1-eps

        high_mask = (cur_h > eps)
        low_mask = (cur_h < 1 - eps)

        if not np.any(high_mask) or not np.any(low_mask):
            break

        # compute argmax among high_mask
        candidates_high = np.where(high_mask)[0]
        i_high = candidates_high[np.argmax(g[candidates_high])]
        candidates_low = np.where(low_mask)[0]
        i_low = candidates_low[np.argmin(g[candidates_low])]

        # compute allowed transfer
        delta = min(step, cur_h[i_high], 1 - cur_h[i_low])
        if delta <= 0:
            step *= 0.5
            continue

        # apply
        cur_h[i_low] += delta
        cur_h[i_high] -= delta

        new_val = np.max(np.correlate(cur_h, 1 - cur_h, mode='full')) * dx

        if new_val < cur_val:
            cur_val = new_val
            # update best local if improved
            if new_val < best_val_local:
                best_val_local = new_val
                best_h_local = cur_h.copy()
            # increase step modestly
            step = min(step * 1.02, 0.5)
        else:
            # revert
            cur_h[i_low] -= delta
            cur_h[i_high] += delta
            step *= 0.7

        # occasional random random move
        if np.random.rand() < 0.01:
            i1, i2 = np.random.choice(n, 2, replace=False)
            # possible direction both ways? Choose random delta
            allowed = min(step, cur_h[i1], 1 - cur_h[i2])
            if allowed > 0:
                cur_h[i1] -= allowed
                cur_h[i2] += allowed
                new_val2 = np.max(np.correlate(cur_h, 1 - cur_h, mode='full')) * dx
                if new_val2 < cur_val:
                    cur_val = new_val2
                    if new_val2 < best_val_local:
                        best_val_local = new_val2
                        best_h_local = cur_h.copy()
                else:
                    # revert
                    cur_h[i1] += allowed
                    cur_h[i2] -= allowed

    return best_h_local, best_val_local

    dx = 2.0 / n

    def project_to_constraints(arr):
        # ensure bounds
        arr = np.clip(arr, 0.0, 1.0)
        # adjust sum to target
        cur_sum = arr.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # distribute diff uniformly to positions not at bounds
            free = np.where((arr > 0) & (arr < 1))[0]
            if free.size == 0:
                # fallback: adjust arbitrary
                free = np.arange(n)
            inc = diff / free.size
            arr[free] += inc
            # clip again
            arr = np.clip(arr, 0.0, 1.0)
        # final adjustment of residual
        cur_sum = arr.sum()
        diff = target_sum - cur_sum
        if abs(diff) > 1e-12:
            # adjust first element not at bound
            for i in range(n):
                if diff > 0 and arr[i] < 1 - 1e-12:
                    add = min(diff, 1 - arr[i])
                    arr[i] += add
                    diff -= add
                elif diff < 0 and arr[i] > 1e-12:
                    sub = min(-diff, arr[i])
                    arr[i] -= sub
                    diff += sub
                if abs(diff) < 1e-12:
                    break
        return arr

    if init_h is not None:
        # clip and adjust sum
        h0 = project_to_constraints(init_h.copy())

import numpy as np
import time
import random

def _shift_zero(arr: np.ndarray, shift: int) -> np.ndarray:
    """Shift arr by `shift` positions, filling the vacated places with zeros."""
    n = arr.shape[0]
    out = np.empty_like(arr)
    if shift >= 0:
        out[: n - shift] = arr[shift:]
        out[n - shift :] = 0
    else:
        s = -shift
        out[s:] = arr[: n - s]
        out[:s] = 0
    return out


def _c5_value(h: np.ndarray) -> float:
    """Evaluate the C5 bound for a step function h."""
    n = h.shape[0]
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def _improve_binary(start_h: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Heuristic improvement for a binary (0/1) array.
    Swaps a mismatching 1 with a mismatching 0 for the worst shift.
    """
    n = start_h.shape[0]
    dx = 2.0 / n
    best_h = start_h.copy()
    best_val = np.max(np.correlate(start_h, 1 - start_h, mode="full")) * dx
    cur_h = start_h.copy()
    cur_val = best_val
    start = time.time()

    while time.time() - start < time_budget:
        # locate the shift giving the maximal overlap integral
        corr = np.correlate(cur_h, 1 - cur_h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        shifted = _shift_zero(cur_h, shift)

        # positions where a 1 would become a mismatch and a 0 would turn into a match
        mis_one = np.where((cur_h == 1) & (shifted == 0))[0]
        mis_zero = np.where((cur_h == 0) & (shifted == 1))[0]

        if mis_one.size and mis_zero.size:
            i = np.random.choice(mis_one)
            j = np.random.choice(mis_zero)
        else:
            # fallback: any random 1‑0 pair
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if ones.size == 0 or zeros.size == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)

        # perform the swap
        cur_h[i], cur_h[j] = 0, 1
        new_val = np.max(np.correlate(cur_h, 1 - cur_h, mode="full")) * dx

        if new_val < cur_val:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = cur_h.copy()
        else:

            cur_h[i], cur_h[j] = 1, 0

        # occasional random perturbation to escape plateaus
        if np.random.rand() < 0.005:
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            if ones.size and zeros.size:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                cur_h[i], cur_h[j] = 0, 1

        # force binary representation while keeping the total mass
        h0 = (init >= 0.5).astype(float)
        # adjust to exact half‑density
        target = n // 2
        cur = int(h0.sum())
        while cur > target:
            idx = np.random.choice(np.where(h0 == 1)[0])
            h0[idx] = 0
            cur -= 1
        while cur < target:
            idx = np.random.choice(np.where(h0 == 0)[0])
            h0[idx] = 1
            cur += 1
        start_candidates = [h0]
    except Exception:
        # no parent supplied – fall back to a random binary start
        n = 80
        start_candidates = []

    # always include a fresh random binary candidate
    def random_binary():
        arr = np.zeros(n, dtype=float)
        arr[: n // 2] = 1.0
        np.random.shuffle(arr)
        return arr

    # -----------------------------------------------------------------
    # Main optimisation loop
    # -----------------------------------------------------------------
    best_h = None
    best_val = float("inf")
    start_time = time.time()

    # evaluate any supplied start
    for cand in start_candidates:
        val = _c5_value(cand)
        if val < best_val:
            best_val = val
            best_h = cand.copy()

    # repeated stochastic hill‑climbing runs
    while time.time() - start_time < budget_s * 0.92:
        # pick a fresh start (occasionally perturb the current best)
        if best_h is not None and np.random.rand() < 0.35:
            h0 = best_h.copy()
            # tiny random perturbation: swap a random 1‑0 pair
            ones = np.where(h0 == 1)[0]
            zeros = np.where(h0 == 0)[0]
            if ones.size and zeros.size:
                i = np.random.choice(ones)
                j = np.random.choice(zeros)
                h0[i], h0[j] = 0, 1
        else:
            h0 = random_binary()

        # allocate a slice of the remaining budget for this run
        remaining = budget_s - (time.time() - start_time)
        t_alloc = min(20.0, max(2.0, remaining * 0.25))

        h_opt, val_opt = _improve_binary(h0, t_alloc)

        if val_opt < best_val:
            best_val = val_opt
            best_h = h_opt.copy()
            if best_val <= 0.3808:  # early stop if we beat the target
                break

    # -----------------------------------------------------------------
    # Final sanity checks and return
    # -----------------------------------------------------------------
    if best_h is None:
        best_h = random_binary()
        best_val = _c5_value(best_h)

    # ensure exact integral constraint (sum == n/2)
    # binary swaps preserve it, but a final clipping guard does not hurt
    best_h = np.clip(best_h, 0.0, 1.0)
    cur_sum = best_h.sum()
    target_sum = n / 2.0
    if abs(cur_sum - target_sum) > 1e-12:
        # correct a tiny numerical drift
        diff = target_sum - cur_sum
        idxs = np.where((best_h > 0) & (best_h < 1))[0]
        if idxs.size:
            best_h[idxs[0]] += diff
        else:
            # fallback: adjust the first element safely
            best_h[0] = np.clip(best_h[0] + diff, 0.0, 1.0)

    final_val = _c5_value(best_h)
    return best_h, final_val, n
