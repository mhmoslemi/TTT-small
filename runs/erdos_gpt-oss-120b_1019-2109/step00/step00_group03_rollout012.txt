import time, math, numpy as np

def run(seed=42, budget_s=1100, **kwargs):
    # Setup RNGs
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # Determine problem size
    init = globals().get('initial_h_values', None)
    if init is not None:
        h0 = np.array(init, copy=False).astype(float)
        n = h0.shape[0]
    else:
        n = kwargs.get('n_points', 70)

    # Basic constants
    target = n // 2                     # sum of h must equal target
    dx = 2.0 / n
    # Helper to evaluate max correlation count (integer) and bound
    def max_corr_count(arr):
        corr = np.correlate(arr.astype(np.int32), (1 - arr).astype(np.int32), mode='full')
        return int(corr.max())
    # Random feasible binary vector
    def rand_vec():
        idx = rng.choice(n, target, replace=False)
        v = np.zeros(n, dtype=np.int8)
        v[idx] = 1
        return v

    # Repair a given vector to have correct sum (used for initial if needed)
    def repair(v):
        cur = int(v.sum())
        diff = cur - target
        if diff > 0:
            ones = np.flatnonzero(v)
            if len(ones):
                flip = rng.choice(ones, diff, replace=False)
                v[flip] = 0
        elif diff < 0:
            zeros = np.flatnonzero(1 - v)
            if len(zeros):
                flip = rng.choice(zeros, -diff, replace=False)
                v[flip] = 1
        return v

    # Initialise best solution
    if init is not None and h0.shape[0] == n:
        # Convert to binary (0/1) for search – we can also keep fractions, but binary suffices
        bh = (h0 > 0.5).astype(np.int8)
        bh = repair(bh)
        cur_best = bh.copy()
        best_corr = max_corr_count(cur_best)
    else:
        cur_best = rand_vec()
        best_corr = max_corr_count(cur_best)

    # Current state for SA
    cur = cur_best.copy()
    cur_corr = best_corr

    start = time.time()
    T = 0.5                     # initial temperature
    it = 0
    no_imp = 0                  # iterations without improvement

    # Main loop
    while time.time() - start < budget_s:
        # Occasionally recompute full correlation array
        corr_full = np.correlate(cur, 1 - cur, mode='full')
        max_idx = int(np.argmax(corr_full))
        shift = max_idx - (n - 1)          # signed shift
        m = abs(shift)

        improved = False

        # 1) targeted swap that directly reduces the current worst shift
        if m != 0:
            left = cur[: n - m]
            right = cur[m :]
            mism = np.flatnonzero((left == 1) & (right == 0))
            if mism.size:
                i = mism[rng.integers(mism.size)]
                j = i + m
                cur[i], cur[j] = cur[j], cur[i]          # perform swap
                new_corr = max_corr_count(cur)
                if new_corr <= cur_corr:
                    cur_corr = new_corr
                    improved = True
                else:
                    cur[i], cur[j] = cur[j], cur[i]      # revert

        # 2) fallback random swap with SA acceptance
        if not improved:
            ones = np.flatnonzero(cur)
            zeros = np.flatnonzero(1 - cur)
            i = ones[rng.integers(len(ones))]
            j = zeros[rng.integers(len(zeros))]
            cur[i], cur[j] = cur[j], cur[i]
            new_corr = max_corr_count(cur)
            delta = new_corr - cur_corr
            if delta <= 0 or (T > 1e-12 and rng.random() < math.exp(-delta / T)):
                cur_corr = new_corr
            else:
                cur[i], cur[j] = cur[j], cur[i]          # revert

        # Track best found
        if cur_corr < best_corr:
            best_corr = cur_corr
            cur_best = cur.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Cooling schedule
        it += 1
        if it % 2000 == 0:
            T *= 0.9
            if T < 1e-4:               # reheat and optionally restart
                T = 0.5
                cur = rand_vec()
                cur_corr = max_corr_count(cur)
                no_imp = 0

        # Random restart if stagnating
        if no_imp >= 5000:
            cur = rand_vec()
            cur_corr = max_corr_count(cur)
            no_imp = 0

        # Early exit if we already hit a very good bound
        if best_corr <= 13:            # corresponds to C5 <= 13/35 ≈ 0.3714
            break

    # Prepare return values
    h_values = cur_best.astype(float)          # float array in [0,1]
    c5_bound = best_corr * dx                  # final bound
    return (h_values, c5_bound, n)
