def greedy_step(h):
    # compute current counts and max
    conv = np.correlate(h, 1-h, mode='full')
    max_val = conv.max()
    max_idxs = np.where(conv == max_val)[0] # may be multiple shift indices
    # For simplicity try random worst shift
    shift_idx = random.choice(max_idxs)
    shift = shift_idx - (len(h)-1)
    # Determine i candidates
    # range for i: i such that i+shift in [0, n-1]
    if shift >= 0:
        i_start = 0
        i_end = len(h) - shift - 1
    else:
        i_start = -shift
        i_end = len(h) - 1
    i_range = np.arange(i_start, i_end+1)
    # i contributing where h[i]==1 and h[i+shift]==0
    i_cand = i_range[(h[i_range]==1) & (h[i_range + shift]==0)]

    # j candidates zero indices where after become one will not add to this shift
    zeros = np.where(h == 0)[0]
    # Determine for each zero if it would not become contributing after swap.
    # condition: either j+shift out of range or h[j+shift]==1

    # Compute boolean mask length of zeros
    if shift >= 0:
        cond = (zeros + shift >= len(h)) | (h[zeros + shift] == 1)
    else:
        cond = (zeros + shift < 0) | (h[zeros + shift] == 1)
    j_cand = zeros[cond]

    if i_cand.size > 0 and j_cand.size > 0:
        i = random.choice(i_cand)
        j = random.choice(j_cand)
        # Do swap
        h[i] = 0
        h[j] = 1
        # Return True
        return True
    else:
        return False

    # indices i where arr[i]=1 and arr[i+shift]=0 (contribute to worst)
    i_cands = []
    for i in range(n):
        j = i + shift
        if 0 <= j < n and arr[i] == 1 and arr[j] == 0:
            i_cands.append(i)

    if not i_cands:
        return None, None

    # zeros j where swapping to 1 would NOT add a contribution to the same shift
    j_cands = []
    for j in range(n):
        if arr[j] == 0:
            i2 = j + shift
            # either out of range or aligns with a 1 (no new conflict)
            if not (0 <= i2 < n and arr[i2] == 0):
                j_cands.append(j)

    if not j_cands:
        return None, None

    i = random.choice(i_cands)
    j = random.choice(j_cands)
    return i, j

def run(seed=42, budget_s=1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed + 12345)

    # ------------------------------------------------------------------
    # initialise h
    init = globals().get('initial_h_values')
    if init is not None:
        h = np.array(init, dtype=np.int8).copy()
        n = h.size
        h = np.where(h > 0.5, 1, 0).astype(np.int8)   # force binary
    else:
        n = 70
        h = np.zeros(n, dtype=np.int8)
        ones = np.random.choice(n, n // 2, replace=False)
        h[ones] = 1

    target_sum = n // 2
    h = repair_solution(h, target_sum)

    dx = 2.0 / n

    best_h = h.copy()
    best_max = max_overlap_count(h)

    start = time.time()
    elapsed = 0.0

    T0 = 5.0                      # initial temperature
    stagnation = 0
    stagn_limit = 20000          # restart after this many non‑improving steps

    while elapsed < budget_s:
        cur_max = max_overlap_count(h)

        # record best
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()
            stagnation = 0
            if best_max <= 13:           # meets target bound
                break

        # ------------------------------------------------------------------
        # propose a swap
        if random.random() < 0.5:
            i, j = targeted_swap(h)
            if i is None:
                ones_idx = np.where(h == 1)[0]
                zeros_idx = np.where(h == 0)[0]
                i = random.choice(ones_idx.tolist())
                j = random.choice(zeros_idx.tolist())
        else:
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            i = random.choice(ones_idx.tolist())
            j = random.choice(zeros_idx.tolist())

        # execute swap
        h[i] = 0
        h[j] = 1

        new_max = max_overlap_count(h)
        delta = new_max - cur_max
        # linear cooling schedule
        frac = elapsed / budget_s
        T = T0 * (1.0 - frac) + 0.1
        accept = False
        if delta <= 0:
            accept = True
        else:
            if random.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                stagnation = 0
                if best_max <= 13:
                    break
            else:
                stagnation += 1
        else:
            # revert
            h[i] = 1
            h[j] = 0
            stagnation += 1

        # restart if stuck
        if stagnation >= stagn_limit:
            h = np.zeros(n, dtype=np.int8)
            idx = np.random.choice(n, target_sum, replace=False)
            h[idx] = 1
            stagnation = 0

        elapsed = time.time() - start

    c5_bound = best_max * dx
    return (best_h.astype(float), float(c5_bound), n)
