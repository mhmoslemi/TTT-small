def _max_corr(arr):

import time, math
import numpy as np

def _max_corr(arr: np.ndarray) -> int:
    """Return the maximum overlap integral (as a count) for a binary array."""
    # np.correlate returns int64 for integer inputs
    return int(np.correlate(arr, 1 - arr, mode="full").max())

def _corr_array(arr: np.ndarray) -> np.ndarray:
    """Full correlation array (needed to locate the worst shift)."""
    return np.correlate(arr, 1 - arr, mode="full")

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h (as a vector), its C_5 bound, and the number of points.
    The function tries to minimise max_k ∫ h(x)(1‑h(x+k))dx using a
    simulated‑annealing / greedy search on a binary representation.
    """
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------ #
    # initialise h ------------------------------------------------------ #
    init_vec = globals().get("initial_h_values", None)
    if init_vec is not None:
        init = np.array(init_vec, dtype=float)
        n = init.size
        target_sum = n / 2.0
        # clip into [0,1] and force the exact sum by taking the largest entries
        init = np.clip(init, 0.0, 1.0)
        M = n // 2
        idx = np.argsort(-init)[:M]
        h = np.zeros(n, dtype=np.int8)
        h[idx] = 1
    else:
        n = 200                     # reasonable size when no parent is given
        M = n // 2
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, M, replace=False)] = 1

    dx = 2.0 / n
    cur_h = h.copy()
    cur_corr = _max_corr(cur_h)

    best_h = cur_h.copy()
    best_corr = cur_corr
    best_c5 = best_corr * dx

    # annealing schedule parameters
    T_start = 1.0
    T_min = 1e-4
    no_improve = 0
    restart_limit = 200_000   # iterations without improvement before a restart

    # ------------------------------------------------------------------ #
    # main optimisation loop ------------------------------------------- #
    while True:
        # time check
        if time.time() - start > budget_s * 0.95:
            break

        # temperature decreases linearly
        elapsed = time.time() - start
        T = max(T_min, T_start * (1.0 - elapsed / budget_s))

        # full correlation, locate worst shift
        corr_arr = _corr_array(cur_h)
        max_corr = int(corr_arr.max())
        if max_corr < best_corr:
            best_corr = max_corr
            best_h = cur_h.copy()
            best_c5 = best_corr * dx
            no_improve = 0
        else:
            no_improve += 1

        # stop early if perfect (cannot be <0)
        if best_corr == 0:
            break

        # shift that currently attains the maximum
        idx_max = int(np.argmax(corr_arr))
        k = idx_max - (n - 1)          # can be negative

        # partner values for this shift (0 where out of range)
        if k >= 0:
            partner = np.empty(n, dtype=np.int8)
            partner[: n - k] = cur_h[k:]
            partner[n - k :] = 0
        else:
            kp = -k
            partner = np.empty(n, dtype=np.int8)
            partner[kp:] = cur_h[: n - kp]
            partner[:kp] = 0

        # indices that cause a mismatch for the worst shift
        one_mis = np.where((cur_h == 1) & (partner == 0))[0]
        zero_ok = np.where((cur_h == 0) & (partner == 1))[0]

        if one_mis.size and zero_ok.size:
            i_one = rng.choice(one_mis)
            i_zero = rng.choice(zero_ok)
        else:
            # fall back to a completely random swap
            ones = np.where(cur_h == 1)[0]
            zeros = np.where(cur_h == 0)[0]
            i_one = rng.choice(ones)
            i_zero = rng.choice(zeros)

        # tentative swap
        cur_h[i_one] = 0
        cur_h[i_zero] = 1
        new_corr = _max_corr(cur_h)

        delta = new_corr - cur_corr
        accept = False
        if delta <= 0:
            accept = True
        else:
            # Metropolis acceptance
            exp_arg = -delta / T
            prob = 0.0 if exp_arg < -700 else math.exp(exp_arg)
            if rng.random() < prob:
                accept = True

        if accept:
            cur_corr = new_corr
            if new_corr < best_corr:
                best_corr = new_corr
                best_h = cur_h.copy()
                best_c5 = best_corr * dx
                no_improve = 0
        else:
            # revert swap
            cur_h[i_one] = 1
            cur_h[i_zero] = 0

        # occasional restart to escape deep local minima
        if no_improve > restart_limit:
            cur_h = np.zeros(n, dtype=np.int8)
            cur_h[rng.choice(n, M, replace=False)] = 1
            cur_corr = _max_corr(cur_h)
            no_improve = 0

    # final check (ensure feasibility)
    assert best_h.sum() == M
    assert best_h.dtype == np.int8
    return best_h.astype(float), best_c5, n
