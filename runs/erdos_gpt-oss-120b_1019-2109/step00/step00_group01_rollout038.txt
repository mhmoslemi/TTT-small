def project_sum_box(v, target):
    # Clip
    v = np.clip(v, 0.0, 1.0)
    # Binary search for lambda to adjust sum
    lo = -2.0
    hi = 2.0
    # Expand range if needed
    # Ensure that lower bound yields sum <= target
    while np.sum(np.clip(v + lo, 0.0, 1.0)) > target:
        lo *= 2
    while np.sum(np.clip(v + hi, 0.0, 1.0)) < target:
        hi *= 2
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:

        else:
            hi = mid

def subgradient_optimize(initial_h, max_iter, time_limit):
    h = initial_h.copy()
    n = h.size
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    eta0 = 0.5   # initial step size
    for it in range(1, max_iter+1):
        # time break
        if time.time() - start > time_limit:
            break
        # compute correlation
        conv = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(conv)
        k = max_idx - (n - 1)
        # gradient
        grad = 1.0 - shift(h, k) - shift(h, -k)

        eta = eta0 / np.sqrt(it)

def project_sum_box(v, target):
    # Clip to [0,1]
    v = np.clip(v, 0.0, 1.0)
    # if sum already target, done
    cur = v.sum()
    if abs(cur - target) < 1e-12:
        return v
    # Use binary search for lambda
    lo = -1.0
    hi = 1.0
    # Expand bounds
    while np.clip(v + lo, 0.0, 1.0).sum() > target:
        lo *= 2
        if lo < -1e6:
            break
    while np.clip(v + hi, 0.0, 1.0).sum() < target:
        hi *= 2
        if hi > 1e6:
            break
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def subgradient_optimize(h0, time_limit):
    import time
    n = h0.size
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5_full(h)
    start = time.time()
    iteration = 0
    while True:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        iteration += 1
        # compute correlation via fft for speed
        n = h.size
        m = 2 * n
        H = np.fft.fft(h, m)
        G = np.fft.fft(1 - h, m)
        conv = np.fft.ifft(H * np.conj(G)).real[:2 * n - 1]
        max_idx = np.argmax(conv)
        k = max_idx - (n - 1)
        # compute gradient
        grad = 1.0 - shift(h, k) - shift(h, -k)
        eta = 0.5 / np.sqrt(iteration)
        new_h = h - eta * grad
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_sum_box(new_h, target)
        c5 = compute_c5_full(new_h)  # use np.correlate for accurate
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
        h = new_h
    return best_h, best_c5

import time
import numpy as np

def _corr_max(h):
    """C5 value using the exact definition (np.correlate)."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * dx

def _shift(arr, k):
    """Return arr shifted by k (zero padding, not circular)."""
    if k == 0:
        return arr.copy()
    n = arr.shape[0]
    if k > 0:
        return np.concatenate([arr[k:], np.zeros(k, dtype=arr.dtype)])
    else:
        p = -k
        return np.concatenate([np.zeros(p, dtype=arr.dtype), arr[:n - p]])

def _project_sum_box(v, target):
    """Project v onto {x : sum(x)=target, 0<=x<=1} using a scalar lambda."""
    v = np.clip(v, 0.0, 1.0)
    # binary search for lambda such that sum(clip(v+lam)) = target
    lo, hi = -2.0, 2.0
    # enlarge interval if needed
    while np.clip(v + lo, 0.0, 1.0).sum() > target:
        lo *= 2.0
    while np.clip(v + hi, 0.0, 1.0).sum() < target:
        hi *= 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)

def _hill_climb(start_h, time_limit, rng):
    """Simple hill‑climbing with two‑index swaps."""
    n = start_h.shape[0]
    target = n * 0.5
    best_h = start_h.copy()
    best_val = _corr_max(best_h)
    cur_h = best_h.copy()
    cur_val = best_val
    step = 0.2                      # typical perturbation amplitude
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        i, j = rng.integers(0, n, size=2)
        while j == i:
            j = rng.integers(0, n)
        delta = (rng.random() - 0.5) * step
        cand = cur_h.copy()
        cand[i] += delta
        cand[j] -= delta
        cand = np.clip(cand, 0.0, 1.0)
        cand = _project_sum_box(cand, target)
        val = _corr_max(cand)
        if val < cur_val:
            cur_h, cur_val = cand, val
            if val < best_val:
                best_h, best_val = cand.copy(), val
        it += 1
        # occasional larger random reshuffle to avoid stagnation
        if it % 500 == 0:
            noise = rng.uniform(-step, step, size=n)
            cand = cur_h + noise
            cand = np.clip(cand, 0.0, 1.0)
