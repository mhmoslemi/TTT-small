def _evaluate(h):
    """Return the C5 value for a step vector h."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def _random_pair_move(h):
    """Propose a feasible move that transfers mass between two indices."""
    n = h.shape[0]
    i, j = random.sample(range(n), 2)
    lo = max(-h[i], h[j] - 1.0)      # keep h[i] ≥ 0, h[j] ≤ 1
    hi = min(1.0 - h[i], h[j])       # keep h[i] ≤ 1, h[j] ≥ 0
    if lo >= hi:
        return None
    delta = random.uniform(lo, hi)
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    return new_h

def _targeted_move(h, shift):
    """Propose a move that tries to increase the overlap for the worst shift."""
    n = h.shape[0]
    if shift == 0:
        # simple transfer from a high entry to a low entry
        i = int(np.argmin(h))
        mask = np.ones(n, dtype=bool)
        mask[i] = False
        j = int(np.argmax(np.where(mask, h, -1.0)))
        inc = 1.0 - h[i]
        dec = h[j]
        if inc <= 0 or dec <= 0:
            return None
        delta = random.random() * min(inc, dec)
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        return new_h

    # overlapping region for this shift
    if shift > 0:
        idx = np.arange(n - shift, dtype=int)
    else:
        idx = np.arange(-shift, n, dtype=int)

    # choose a pair with the smallest sum (both are low -> best to raise)
    pair_sum = h[idx] + h[idx + shift]
    i = int(idx[np.argmin(pair_sum)])
    partner = i + shift

    # choose a high index to take mass from
    mask = np.ones(n, dtype=bool)
    mask[i] = False
    mask[partner] = False
    if not mask.any():
        return None
    j = int(np.argmax(np.where(mask, h, -1.0)))

    # feasible amount we can move
    max_inc_i = 1.0 - h[i]
    max_inc_p = 1.0 - h[partner]
    max_dec_j = h[j]
    max_delta = min(max_inc_i, max_inc_p, max_dec_j / 2.0)
    if max_delta <= 0:
        return None
    delta = random.random() * max_delta
    new_h = h.copy()
    new_h[i] += delta
    new_h[partner] += delta
    new_h[j] -= 2.0 * delta
    return new_h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Find a step function h on [0,2] (discretised) that yields a small C5 value.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # initialise h
    if 'initial_h_values' in globals():
        h = np.asarray(initial_h_values, dtype=np.float64).copy()
    else:
        # fallback: half ones, half zeros
        n = 100
        h = np.concatenate([np.ones(n // 2), np.zeros(n - n // 2)]).astype(np.float64)

    n = h.shape[0]
    target_sum = n / 2.0
    # Correct any tiny deviation from the integral constraint
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # add the missing mass to the first entry that can absorb it
        for idx in range(n):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break

    best_h = h.copy()
    best_c = _evaluate(best_h)
    cur_h = best_h.copy()
    cur_c = best_c

    # algorithm parameters
    init_temp = 0.005
    prob_targeted = 0.7
    # ------------------------------------------------------------------
    # main optimisation loop
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > budget_s * 0.95:
            break

        # temperature linearly decreasing to near zero
        temp = init_temp * max(0.0, 1.0 - elapsed / budget_s) + 1e-12

        if random.random() < prob_targeted:
            # compute worst shift from current state
            corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            idx_max = int(np.argmax(corr))
            shift = idx_max - (n - 1)
            new_h = _targeted_move(cur_h, shift)
            if new_h is None:
                new_h = _random_pair_move(cur_h)
                if new_h is None:
                    continue
        else:
            new_h = _random_pair_move(cur_h)
            if new_h is None:
                continue

        new_c = _evaluate(new_h)
        delta_c = new_c - cur_c
        if delta_c <= 0 or random.random() < math.exp(-delta_c / temp):
            cur_h = new_h
            cur_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = new_h.copy()

    # ------------------------------------------------------------------
    # final projection to satisfy the exact integral constraint
    diff = target_sum - best_h.sum()
    if abs(diff) > 1e-9:
        for idx in range(n):
            if 0.0 <= best_h[idx] + diff <= 1.0:
                best_h[idx] += diff
                break

    final_c = _evaluate(best_h)
    return (best_h.tolist(), float(final_c), n)
