def _to_binary(h, half):
    """Convert an arbitrary vector to a binary vector with exactly `half` ones."""
    # pick indices of the largest entries
    idx = np.argsort(h)[::-1][:half]
    out = np.zeros_like(h, dtype=float)
    out[idx] = 1.0
    return out

def _best_shift_auto_corr(h):
    """Return the shift (relative index) that gives the smallest autocorrelation
    (i.e. the worst overlap)."""
    auto = np.correlate(h, h, mode='full')
    idx_min = int(np.argmin(auto))
    shift = idx_min - (len(h) - 1)
    return shift


def _improve_once(h, rng):
    """Perform a single improving swap aimed at the worst shift."""
    n = len(h)
    shift = _best_shift_auto_corr(h)

    # Define the index range where both i and i+shift are valid
    if shift >= 0:
        i_range = np.arange(0, n - shift)
        hi = h[i_range]
        hj = h[i_range + shift]
    else:
        s = -shift
        i_range = np.arange(s, n)
        hi = h[i_range]
        hj = h[i_range - s]

    # positions where we have (1,0) and (0,1) for this shift
    a_idx = i_range[(hi == 1) & (hj == 0)]
    b_idx = i_range[(hi == 0) & (hj == 1)]

    if a_idx.size and b_idx.size:
        i = int(rng.choice(a_idx))
        j = int(rng.choice(b_idx))
        new_h = h.copy()
        new_h[i], new_h[j] = new_h[j], new_h[i]   # swap 1↔0
        return new_h
    # fallback: random 1/0 swap
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    i = int(rng.choice(ones))
    j = int(rng.choice(zeros))
    new_h = h.copy()
    new_h[i] = 0.0
    new_h[j] = 1.0
    return new_h


def _c5_raw(h):
    """Raw (unscaled) maximum of the overlap integral: max_i sum h_i (1‑h_{i+k})"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr)
