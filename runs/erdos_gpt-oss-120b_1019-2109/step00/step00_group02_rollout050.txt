import time
import math
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return C5 for a step function represented by the array h."""
    n = h.shape[0]
    # correlation of h with 1‑h (full mode) gives the raw counts
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * (2.0 / n))

def _binary_from_continuous(v: np.ndarray) -> np.ndarray:
    """Turn a possibly fractional vector into a binary one with the same
    length and exactly half of its entries equal to 1 (rounded in a
    deterministic way)."""
    n = v.shape[0]
    half = n // 2
    # indices of the largest values become 1
    idx = np.argsort(-v, kind="stable")
    b = np.zeros(n, dtype=float)
    b[idx[:half]] = 1.0
    return b

def _hill_climb(start: np.ndarray,
                max_iter: int,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing hill‑climber for binary vectors."""
    cur = start.copy()
    cur_score = _c5_value(cur)
    best = cur.copy()
    best_score = cur_score
    n = cur.shape[0]
    # temperature schedule – a tiny temperature is enough because
    # evaluation noise is zero; we keep a tiny probability of uphill moves
    T = 0.015
    for it in range(max_iter):
        # choose move type ----------------------------------------------------
        # 70 %: targeted swap that tries to fix the worst lag;
        # 30 %: completely random swap.
        if rng.random() < 0.7:
            # identify the lag with maximal overlap
            corr = np.correlate(cur, 1.0 - cur, mode="full")
            lag = int(np.argmax(corr) - (n - 1))          # signed shift
            # find a 1 that mismatches and a 0 that mismatches for this lag
            if lag >= 0:
                # i+lag is inside the array
                mask1 = (cur == 1) & (np.arange(n) + lag < n) & (cur[np.arange(n) + lag] == 0)
                mask0 = (cur == 0) & (np.arange(n) + lag < n) & (cur[np.arange(n) + lag] == 1)
            else:
                # -lag is the forward shift
                shift = -lag
                mask1 = (cur == 1) & (np.arange(n) - shift >= 0) & (cur[np.arange(n) - shift] == 0)
                mask0 = (cur == 0) & (np.arange(n) - shift >= 0) & (cur[np.arange(n) - shift] == 1)

            mism1 = np.flatnonzero(mask1)
            mism0 = np.flatnonzero(mask0)
            if mism1.size and mism0.size:
                i = rng.choice(mism1)
                j = rng.choice(mism0)
            else:
                # fall back to a random swap
                i = rng.choice(np.flatnonzero(cur == 1))
                j = rng.choice(np.flatnonzero(cur == 0))
        else:
            i = rng.choice(np.flatnonzero(cur == 1))
            j = rng.choice(np.flatnonzero(cur == 0))

        # perform the swap
        nxt = cur.copy()
        nxt[i] = 0.0
        nxt[j] = 1.0
        nxt_score = _c5_value(nxt)

        delta = nxt_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / T):
            cur = nxt
            cur_score = nxt_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score

        # geometric cooling
        T *= 0.9997
        # very small temperature → pure hill climb
        if T < 1e-6:
            T = 1e-6
    return best, best_score


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Search for a step‑function h on [0,2] that yields a small C5 value.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)

    # --------------------------------------------------------------------
    # 1.  initialise with the supplied construction (if any)
    # --------------------------------------------------------------------
    best_h = None
    best_score = float("inf")
    best_n = None

    if "initial_h_values" in globals():
        init_arr = np.asarray(initial_h_values, dtype=float)
        if init_arr.ndim != 1:
            init_arr = init_arr.ravel()
        # make length even – the problem works for any length, but a
        # binary search is easier with an even number of samples.
        if init_arr.shape[0] % 2 == 1:
            init_arr = init_arr[:-1]
        # turn it into a binary vector (the known record is binary)
        binary = _binary_from_continuous(init_arr)
        score = _c5_value(binary)
        best_h, best_score, best_n = binary, score, binary.shape[0]

    # --------------------------------------------------------------------
    # 2.  define the search space (even lengths only)
    # --------------------------------------------------------------------
    # we try a variety of modest sizes; larger n gives finer resolution but
    # also makes each evaluation slightly more expensive.
    cand_ns = list(range(30, 122, 2))            # 30,32,…,120
    # ensure we keep the length from the initial construction in the list
    if best_n and best_n not in cand_ns:
        cand_ns.append(best_n)

    # --------------------------------------------------------------------
    # 3.  main optimisation loop – random restarts + hill climbing
    # --------------------------------------------------------------------
    # amount of time devoted overall to the meta‑search (90 % of the budget)
    total_search_time = budget_s * 0.90
    # each restart receives a tiny time slice; we keep launching new ones
    # until the budget is exhausted.
    while time.time() - start_time < total_search_time:
        # pick a length uniformly at random
        n = int(rng.choice(cand_ns))
        half = n // 2

        # initialise a random binary vector
        seq = np.zeros(n, dtype=float)
        seq[:half] = 1.0
        rng.shuffle(seq)

        # number of internal iterations – scaled with n (larger n → more work)
        max_iter = max(2000, n * 30)

        # run a short hill‑climb
        cand_h, cand_score = _hill_climb(seq, max_iter, rng)

        if cand_score < best_score:
            best_h, best_score, best_n = cand_h.copy(), cand_score, n

        # occasionally try to improve the current best with a few extra steps
        if best_h is not None and time.time() - start_time < total_search_time:
            # a tiny refinement on the global best (maybe escape a shallow local min)
            refined_h, refined_score = _hill_climb(best_h, max_iter // 2, rng)
            if refined_score < best_score:
                best_h, best_score, best_n = refined_h.copy(), refined_score, best_h.shape[0]

    # --------------------------------------------------------------------
    # 4.  final sanity check and return
    # --------------------------------------------------------------------
    # make absolutely sure the integral constraint holds (within floating error)
    if best_h is not None:
        n = best_h.shape[0]
        target_sum = n / 2.0
        cur_sum = best_h.sum()
        if abs(cur_sum - target_sum) > 1e-9:
            # project onto the feasible hyperplane while keeping box constraints
            # simple water‑filling style correction
            diff = cur_sum - target_sum
            if diff > 0:
                # need to decrease some entries
                idx = np.where(best_h > 0)[0]
                while diff > 1e-12 and idx.size:
                    i = rng.choice(idx)
                    delta = min(diff, best_h[i])
                    best_h[i] -= delta
                    diff -= delta
                    if best_h[i] < 1e-12:
                        idx = idx[idx != i]
            else:
                # need to increase some entries
                idx = np.where(best_h < 1)[0]
                while diff < -1e-12 and idx.size:
                    i = rng.choice(idx)
                    delta = min(-diff, 1.0 - best_h[i])
                    best_h[i] += delta
                    diff += delta
                    if best_h[i] > 1 - 1e-12:
                        idx = idx[idx != i]
        best_score = _c5_value(best_h)

    # return as a plain Python list for compatibility
    return (best_h.tolist() if best_h is not None else [], float(best_score), int(best_n))
