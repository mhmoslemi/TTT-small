        # first clip to [0,1]

        diff = target - h.sum()

        if abs(diff) < 1e-12:

        # iterate:
        while True:

import time
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős overlap constant for a given step function."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_hyperplane(x: np.ndarray,
                           target: float,
                           lower: float = 0.0,
                           upper: float = 1.0,
                           max_iter: int = 50) -> np.ndarray:
    """
    Project vector x onto the set {y | lower <= y_i <= upper, sum(y)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = lower - np.max(x) - 1.0   # certainly too low (all clipped to lower)
    hi = upper - np.min(x) + 1.0   # certainly too high (all clipped to upper)

    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, lower, upper)
        s = y.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) / 2.0
    return np.clip(x + lam, lower, upper)

    np.random.seed(seed)
