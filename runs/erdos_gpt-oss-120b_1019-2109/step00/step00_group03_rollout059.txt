import numpy as np
import time
import math

def compute_c5(arr: np.ndarray) -> float:
    """Return the max overlap integral for a binary (or [0,1]) vector."""
    n = arr.shape[0]
    conv = np.correlate(arr, 1 - arr, mode="full")
    return float(conv.max() * (2.0 / n))


def mismatched_indices(arr: np.ndarray, d: int) -> np.ndarray:
    """Indices i where arr[i]==1 and arr[i+d]==0 (contributing to the max shift)."""
    n = arr.shape[0]
    if d >= 0:
        length = n - d
        if length <= 0:
            return np.empty(0, dtype=int)
        mask = (arr[:length] == 1) & (arr[d:] == 0)
        return np.where(mask)[0]
    else:
        k = -d
        start = k
        length = n - k
        if length <= 0:
            return np.empty(0, dtype=int)
        mask = (arr[start:] == 1) & (arr[:length] == 0)
        return np.where(mask)[0] + start


def candidate_zero_indices(arr: np.ndarray, d: int) -> np.ndarray:
    """Zeros that, when turned into a one, do NOT create a new mismatch for shift d."""
    n = arr.shape[0]
    zeros = np.where(arr == 0)[0]
    if zeros.size == 0:
        return zeros
    if d >= 0:
        out_of = zeros + d >= n
        cond = out_of.copy()
        valid = ~out_of
        if np.any(valid):
            cond[valid] = arr[zeros[valid] + d] == 1
        return zeros[cond]
    else:
        k = -d
        out_of = zeros - k < 0
        cond = out_of.copy()
        valid = ~out_of
        if np.any(valid):
            cond[valid] = arr[zeros[valid] - k] == 1
        return zeros[cond]
