def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def local_search(h, time_limit, start_time):

    dx = 2.0 / n
    cur = h.copy()

    best = cur.copy()

        diff -= inc
        it += 1
    # If we still haven't hit the target (rare), fall back to a scaling step
    if abs(diff) > 1e-9:
        cur = h.sum()
        if cur > 0:
            factor = target / cur
            h[:] = np.clip(h * factor, 0.0, 1.0)
        # final pass – tiny residual handled greedily
        diff = target - h.sum()
        for i in range(n):
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                add = min(diff, 1.0 - h[i])
            else:
                add = max(diff, -h[i])
            h[i] += add
            diff -= add
    return h


def local_search(h_start: np.ndarray, max_seconds: float, start_time: float) -> (np.ndarray, float):
    """
    Simple projected‑pairwise‑exchange simulated‑annealing.
    Returns the best vector found and its C5.
    """
    h = h_start.copy()
    n = h.shape[0]
    best_h = h.copy()
    best_score = compute_c5(h)

    cur_score = best_score
    end_time = start_time + max_seconds

    # temperature schedule parameters
    T0 = 0.02   # initial temperature
    Tmin = 1e-5

    while time.time() < end_time:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        hi, hj = h[i], h[j]

        # allowable delta so that both stay in [0,1] after h[i]+=d, h[j]-=d
        d_min = max(-hi, hj - 1.0)
        d_max = min(1.0 - hi, hj)
        if d_min >= d_max:
            continue

        delta = np.random.uniform(d_min, d_max)
        h[i] = hi + delta
        h[j] = hj - delta

        new_score = compute_c5(h)

        if new_score < cur_score:
            accept = True
        else:
            # linear cooling
            elapsed = time.time() - start_time
            fraction = elapsed / max_seconds if max_seconds > 0 else 1.0
            T = max(T0 * (1.0 - fraction), Tmin)
            accept = np.random.rand() < np.exp(-(new_score - cur_score) / T)

        if accept:
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj

    return best_h, best_score

# Main entry point required by the harness
# ------------------------------------------------------------

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] to obtain a low C5 value.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # ----------------------------------------------------------------
    # 1. Initialise a feasible vector
    # ----------------------------------------------------------------
    try:
        init = initial_h_values  # supplied by the harness, if any
        n0 = len(init)
        h0 = np.array(init, dtype=float)
    except NameError:
        # No parent solution – fall back to a random binary vector with exact half mass
        n0 = 42
        h0 = np.zeros(n0, dtype=float)
        indices = np.random.choice(n0, n0 // 2, replace=False)
        h0[indices] = 1.0

    # enforce the integral constraint exactly (sum = n/2)
    target_sum = n0 / 2.0
    h0 = adjust_sum_to_target(h0, target_sum)

    # ----------------------------------------------------------------
    # 2. Local optimisation at the base resolution
    # ----------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    if remaining <= 0:
        return h0.tolist(), compute_c5(h0), n0

    # allocate ~30 % of the remaining budget to the first level
    base_time = min(remaining * 0.30, 300.0)
    best_h, best_score = local_search(h0, base_time, start_time)
    n_best = n0

    # ----------------------------------------------------------------
    # 3. Multi‑scale refinement (doubling the grid)
    # ----------------------------------------------------------------
    max_n = 200           # keep the problem size moderate for speed
    scale_factors = [2, 2]  # 42→84→168 (or whatever the start size is)

    h_current = best_h
    n_current = n0

    for factor in scale_factors:
        remaining = budget_s - (time.time() - start_time)
        if remaining <= 5.0:
            break
        n_new = n_current * factor
        if n_new > max_n:
            break

        # Upsample by linear interpolation on the interval [0,2)
        x_old = np.linspace(0.0, 2.0, n_current, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
        h_new = np.interp(x_new, x_old, h_current)

        # Normalise the new vector to the required integral
        target_sum_new = n_new / 2.0
        h_new = adjust_sum_to_target(h_new, target_sum_new)

        # Allocate a slice of the remaining budget to this level
        level_time = min(remaining * 0.40, 250.0)
        opt_h, opt_score = local_search(h_new, level_time, start_time)

        # Keep global best
        if opt_score < best_score:
            best_score = opt_score
            best_h = opt_h
            n_best = n_new

        # Prepare for possible further refinement
        h_current = opt_h
        n_current = n_new

    # ----------------------------------------------------------------
    # 4. Optional extra random restarts at the final resolution
    # ----------------------------------------------------------------
    remaining = budget_s - (time.time() - start_time)
    while remaining > 6.0:
        # short random‑restart search (5 s)
        t_this = min(5.0, remaining)
        # start from the current best with a small random perturbation
        h_restart = best_h.copy()
        # apply a few random pairwise swaps to diversify
        for _ in range(10):
            i, j = np.random.randint(0, n_best, size=2)
            hi, hj = h_restart[i], h_restart[j]
            d_min = max(-hi, hj - 1.0)
            d_max = min(1.0 - hi, hj)
            if d_min < d_max:
                d = np.random.uniform(d_min, d_max)
                h_restart[i] = hi + d
                h_restart[j] = hj - d
        # ensure sum is still exact (tiny drift possible)
        h_restart = adjust_sum_to_target(h_restart, n_best / 2.0)

        cand_h, cand_score = local_search(h_restart, t_this, start_time)
        if cand_score < best_score:
            best_score = cand_score
            best_h = cand_h
            n_best = n_best  # unchanged
        remaining = budget_s - (time.time() - start_time)

    # ----------------------------------------------------------------
    # 5. Final sanitisation and return
    # ----------------------------------------------------------------
    # Clip once more and re‑adjust the integral to be safe
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = adjust_sum_to_target(best_h, n_best / 2.0)

    final_c5 = compute_c5(best_h)
    return best_h.tolist(), float(final_c5), int(n_best)
