def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def project_sum_bounds(v, target):
    # binary search lam to enforce sum after clipping.
    lo, hi = -10.0, 10.0

        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()

            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo+hi)/2.0, 0.0, 1.0)

import time
import math
import numpy as np

def compute_c5(h: np.ndarray) -> float:
    """Return the Erdős C₅ value for a given step function sampled array."""
    n = h.shape[0]
    dx = 2.0 / n
    # cross‑correlation of h with (1‑h)
    corr = np.correlate(h, 1 - h, mode="full")
    return float(corr.max() * dx)


def _project_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project a vector onto the intersection of the box [0,1]^n and the affine
    hyperplane sum(v)=target using a simple bisection on a scalar shift.
    """
    lo, hi = -1e6, 1e6
    for _ in range(60):
        lam = (lo + hi) * 0.5
        w = np.clip(v + lam, 0.0, 1.0)
        s = w.sum()
        if s < target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v + lam, 0.0, 1.0)


def random_h(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random feasible h (bounds + integral)."""
    v = rng.random(n)
    v = v / v.sum() * (n / 2.0)
    return _project_sum_bounds(v, n / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = float(budget_s)

    # ----------------------------------------------------------------------
    # 1. Choose a working discretisation and initial point
    # ----------------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)  # type: ignore
        n0 = init.size
    except Exception:
        init = None
        n0 = None

    # Target size – if we have a parent solution we up‑sample it, otherwise use 120 points
    if init is not None:
        n = max(2 * n0, 120)
        # linear interpolation of the parent shape onto the new grid
        x_old = np.linspace(0.0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 2.0, n, endpoint=False)
        h = np.interp(x_new, x_old, init)
    else:
        n = 120
        # a smooth sine‑squared shape (average already 1/2)
        idx = np.arange(n, dtype=float) + 0.5
        h = np.sin(np.pi * idx / n) ** 2

    # enforce feasibility exactly
    h = _project_sum_bounds(h, n / 2.0)

    # ----------------------------------------------------------------------
    # 2. Simulated‑annealing style local search
    # ----------------------------------------------------------------------
    cur_h = h.copy()
    cur_val = compute_c5(cur_h)
    best_h = cur_h.copy()
    best_val = cur_val

    # initial temperature – a few hundredths works well for the scale of C₅
    T0 = 0.02

    # Main optimisation loop; stops when time budget is exhausted
    while time.time() - start_time < total_budget * 0.94:
        # pick two distinct positions and move a small amount of mass between them
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        hi, hj = cur_h[i], cur_h[j]

        # admissible amount that keeps both entries in [0,1]
        delta_low = -min(hi, 1.0 - hj)
        delta_high = min(1.0 - hi, hj)
        if delta_low >= delta_high:
            continue
        delta = rng.uniform(delta_low, delta_high)

        new_h = cur_h.copy()
        new_h[i] = hi + delta
        new_h[j] = hj - delta

        new_val = compute_c5(new_h)

        # linear cooling based on elapsed fraction of the budget
        progress = (time.time() - start_time) / total_budget
        T = T0 * (1.0 - progress)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur_h = new_h
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = new_h.copy()

        # occasional random restart when the temperature is very low
        if T < 1e-4 and rng.random() < 0.01:
            cur_h = random_h(n, rng)
            cur_val = compute_c5(cur_h)

    # ----------------------------------------------------------------------
    # 3. Final clean‑up and return
    # ----------------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_sum_bounds(best_h, n / 2.0)
    final_c5 = compute_c5(best_h)

    # Return as a plain Python list for compatibility
    return best_h.tolist(), float(final_c5), n
