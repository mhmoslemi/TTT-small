import time, random, math
import numpy as np

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------
    # problem dimensions (use the length of the provided parent if any)
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        n = len(initial_h_values)          # number of discretisation points
    else:
        n = 80                             # fallback size
    target_sum = n // 2                    # ∫h = 1  ⇒  sum(h) = n/2
    dx = 2.0 / n

    # target bound we would like to beat (0.3808)
    target_c5 = 0.3808
    target_int = math.floor(target_c5 / dx + 1e-12)   # allowed integer max value

    # ------------------------------------------------------------
    # helper utilities
    # ------------------------------------------------------------
    def max_corr_int(h):
        """maximum correlation value (integer) for array h."""
        c = np.correlate(h, 1 - h, mode='full')
        return int(c.max())

    def full_corr_array(h):
        """full correlation array (used to locate the worst shift)."""
        return np.correlate(h, 1 - h, mode='full')

    def gradient_of_shift(h, k):
        """sub‑gradient of the overlap for shift k (can be negative)."""
        grad = np.zeros_like(h, dtype=float)
        if k >= 0:
            if k < n:
                grad[: n - k] += 1.0
                grad[: n - k] -= h[k:]
                grad[k:]       -= h[: n - k]
        else:
            k = -k
            grad[k:]       += 1.0
            grad[k:]       -= h[: n - k]
            grad[: n - k]  -= h[k:]
        return grad

    def project_onto_feasible(z):
        """
        Project vector z onto the convex set {0≤h_i≤1, sum h_i = target_sum}.
        Uses a simple bisection on the Lagrange multiplier λ.
        """
        lo = -np.max(z) - 1.0   # makes all entries >1 after shift → clipped to 1
        hi =  np.max(z) + 1.0   # makes all entries <0 after shift → clipped to 0
        for _ in range(60):
            lam = (lo + hi) * 0.5
            y = np.clip(z - lam, 0.0, 1.0)
            s = y.sum()
            if s > target_sum:
                lo = lam
            else:
                hi = lam
        y = np.clip(z - (lo + hi) * 0.5, 0.0, 1.0)

        # tiny correction to hit the sum exactly (numerical safety)
        diff = target_sum - y.sum()
        if abs(diff) > 1e-10:
            if diff > 0:
                # increase some components that are not at the upper bound
                idx = np.argsort(y)
                for i in idx:
                    room = 1.0 - y[i]
                    if room <= 0.0:
                        continue
                    inc = min(room, diff)
                    y[i] += inc
                    diff -= inc
                    if diff <= 1e-12:
                        break
            else:
                # decrease some components that are not at the lower bound
                idx = np.argsort(-y)
                for i in idx:
                    room = y[i]
                    if room <= 0.0:
                        continue
                    dec = min(room, -diff)
                    y[i] -= dec
                    diff += dec
                    if diff >= -1e-12:
                        break
        return y

    def random_feasible():
        """draw a random point in the feasible polytope."""
        z = np.random.rand(n)
        return project_onto_feasible(z)

    # ------------------------------------------------------------
    # initial point
    # ------------------------------------------------------------
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_onto_feasible(h0)
    else:
        h0 = random_feasible()

    best_h = h0.copy()
    best_int = max_corr_int(best_h)

    cur_h = h0.copy()
    cur_int = best_int

    # ------------------------------------------------------------
    # optimisation loop (sub‑gradient on the worst shift)
    # ------------------------------------------------------------
    start = time.time()
    it = 0
    stagnation = 0
    init_step = 0.5
    min_step = 1e-4

    while True:
        elapsed = time.time() - start
        if elapsed >= budget_s - 0.5:
            break

        # compute current worst shift
        corr = full_corr_array(cur_h)
        max_val = int(corr.max())
        if max_val <= target_int:
            best_int = max_val
            best_h = cur_h.copy()
            break

        # indices achieving the maximum (could be several)
        worst_idx = np.flatnonzero(corr == max_val)
        k_idx = int(random.choice(worst_idx))
        k_shift = k_idx - (n - 1)        # lag value

        # gradient step
        grad = gradient_of_shift(cur_h, k_shift)
        frac = elapsed / (budget_s - 0.5)
        step = init_step * (1 - frac) + min_step * frac

        cand = cur_h - step * grad
        cand = project_onto_feasible(cand)
        cand_int = max_corr_int(cand)

        if cand_int <= cur_int or random.random() < 0.02:

            if cand_int < best_int:
                best_int, best_h = cand_int, cand.copy()
                stagnation = 0
            else:
                stagnation += 1
        else:
            stagnation += 1

        # occasional restart when stuck
        if stagnation > 5000:
            cur_h = random_feasible()
            cur_int = max_corr_int(cur_h)
            stagnation = 0

        # small periodic random perturbations for diversity
        if it % 2000 == 0 and it > 0:
            if random.random() < 0.1:
                cur_h = random_feasible()
                cur_int = max_corr_int(cur_h)

        it += 1

    # ------------------------------------------------------------
    # final polishing (ensure feasibility, evaluate bound)
    # ------------------------------------------------------------
    best_h = project_onto_feasible(best_h)
    final_int = max_corr_int(best_h)
    final_c5 = final_int * dx

        c5 = evaluate_erdos_solution(best_h)
    except Exception:
        c5 = final_c5

    return (best_h.astype(float), c5, n)
