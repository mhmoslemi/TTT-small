def shape_power_plateau(N, beta):

    # compute half-width a = (beta - 1)/(2*beta)

import numpy as np
import time

def _compute_c5(h):
    """Return the C5 value for a step function h (numpy array)."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def _scale_to_sum(raw, target):
    """
    Scale a non‑negative array `raw` to satisfy sum == target
    while respecting the box constraint 0 ≤ h ≤ 1.
    The scaling is performed by finding λ such that
        sum(min(1, λ·raw)) = target .
    """
    # quick path when scaling down (λ≤1) keeps everything ≤1
    s = raw.sum()
    if s >= target:
        lam = target / s
        h = raw * lam
        # clipping not needed because lam≤1 and raw≤1
        return h

    # otherwise we need saturation; binary search for λ
    lo, hi = 0.0, 1.0
    # increase hi until sum >= target
    while np.sum(np.minimum(1.0, raw * hi)) < target:
        hi *= 2.0
        if hi > 1e6:  # safety
            break
    for _ in range(60):
        mid = (lo + hi) * 0.5
        cur = np.sum(np.minimum(1.0, raw * mid))
        if cur < target:
            lo = mid
        else:
            hi = mid
    lam = hi
    h = np.minimum(1.0, raw * lam)

    # tiny adjustment to hit the target exactly (helps later checks)
    diff = np.sum(h) - target
    if abs(diff) > 1e-10:
        # adjust a free coordinate if possible
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if free.size:
            i = free[0]
            h[i] -= diff
            h[i] = max(0.0, min(1.0, h[i]))
        else:
            # fallback: modify the first element (still stays in bounds)
            h[0] = max(0.0, min(1.0, h[0] - diff))
    return h

def _shape_sin_pow(N, p):
    i = np.arange(N, dtype=float)
    raw = np.sin(np.pi * i / (N - 1)) ** p
    return _scale_to_sum(raw, N / 2.0)

def _shape_exp(N, c):
    x = np.linspace(0.0, 2.0, N, dtype=float)
    raw = np.exp(-c * np.abs(x - 1.0))
    return _scale_to_sum(raw, N / 2.0)

def _shape_beta(N, beta):
    # derived family with a central plateau and power‑law edges
    a = (beta - 1.0) / (2.0 * beta)          # half‑width of the plateau
    x = np.linspace(0.0, 2.0, N, dtype=float)
    t = np.abs(x - 1.0)
    mask = t <= a
    raw = np.empty_like(t)
    raw[mask] = 1.0
    raw[~mask] = ((1.0 - t[~mask]) / (1.0 - a)) ** beta
    return _scale_to_sum(raw, N / 2.0)

def _shape_triangle(N):
    x = np.linspace(0.0, 2.0, N, dtype=float)
    raw = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
    return _scale_to_sum(raw, N / 2.0)
