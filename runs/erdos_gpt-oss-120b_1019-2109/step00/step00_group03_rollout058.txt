        if s >= 0:

            i_range = np.arange(N - s)
            # mismatched ones where h[i]==1 and h[i+s]==0
            mismatched_ones = i_range[(current_h[i_range] == 1) & (current_h[i_range + s] == 0)]
            # mismatched zeros where h[j]==0 and h[j+s]==1
            mismatched_zeros = i_range[(current_h[i_range] == 0) & (current_h[i_range + s] == 1)]
        else:
            # s negative: shift left
            s_pos = -s
            # overlapping i: s_pos <= i < N
            i_range = np.arange(s_pos, N)
            mismatched_ones = i_range[(current_h[i_range] == 1) & (current_h[i_range + s] == 0)] # i+s is i - s_pos
            mismatched_zeros = i_range[(current_h[i_range] == 0) & (current_h[i_range + s] == 1)]

        if mismatched_ones.size == 0 or mismatched_zeros.size == 0:
            # cannot improve maybe random swap
            # do random swap
            i_candidates = np.flatnonzero(current_h == 1)
            j_candidates = np.flatnonzero(current_h == 0)
            i = rng.choice(i_candidates)
            j = rng.choice(j_candidates)

            j = rng.choice(mismatched_zeros)

            # compute acceptance probability

            elapsed = time.time() - start

import time
import math
import numpy as np


def _corr_max(h: np.ndarray):
    """Return max correlation value and full correlation array."""
    corr = np.correlate(h, 1 - h, mode="full")
    return int(corr.max()), corr


def _mismatched_indices(h: np.ndarray, shift: int):
    """
    For a given shift, return indices i where h[i]=1 & h[i+shift]=0
    and indices i where h[i]=0 & h[i+shift]=1 (both within bounds).
    """
    n = h.shape[0]
    if shift >= 0:
        start, end = 0, n - shift
    else:
        start, end = -shift, n
    i = np.arange(start, end, dtype=np.int32)
    t = i + shift
    ones_mask = (h[i] == 1) & (h[t] == 0)
    zeros_mask = (h[i] == 0) & (h[t] == 1)
    return i[ones_mask], i[zeros_mask]


def _local_search(start_h: np.ndarray, rng: np.random.Generator,
                  deadline: float, target_max: int):
    """
    Hill‑climbing / simulated‑annealing search.
    Returns best (h, max_corr) found before `deadline`.
    """
    n = start_h.shape[0]
    cur_h = start_h.copy()
    cur_max, cur_corr = _corr_max(cur_h)

    best_h = cur_h.copy()
    best_max = cur_max

    # parameters for restart / annealing
    no_improve = 0
    max_no_improve = 3000
    start_temp = 5.0  # initial temperature for uphill moves

    while time.time() < deadline:
        if cur_max <= target_max:
            best_h, best_max = cur_h.copy(), cur_max
            break

        # choose a shift with the current maximal value
        idxs = np.flatnonzero(cur_corr == cur_max)
        shift_idx = int(rng.choice(idxs))
        shift = shift_idx - (n - 1)

        mis_one, mis_zero = _mismatched_indices(cur_h, shift)

        # choose a swap candidate
        if mis_one.size and mis_zero.size:
            i = int(rng.choice(mis_one))
            j = int(rng.choice(mis_zero))
        else:
            # fallback random swap
            i = int(rng.choice(np.flatnonzero(cur_h == 1)))
            j = int(rng.choice(np.flatnonzero(cur_h == 0)))

        # perform swap
        cur_h[i], cur_h[j] = cur_h[j], cur_h[i]

        new_max, new_corr = _corr_max(cur_h)

        if new_max <= cur_max:
            # improvement or neutral – keep it
            cur_max, cur_corr = new_max, new_corr
            no_improve = 0
            if cur_max < best_max:
                best_max, best_h = cur_max, cur_h.copy()
        else:
            # possibly accept uphill move (simulated annealing)
            elapsed = time.time() - (deadline - (deadline - time.time()))
            total = max(deadline - time.time(), 1e-9)
            # linear cooling from start_temp to 0
            T = start_temp * max(0.0, 1.0 - (time.time() - (deadline - total)) / total)
            prob = math.exp(-(new_max - cur_max) / (T + 1e-12))
            if rng.random() < prob:
                cur_max, cur_corr = new_max, new_corr
                no_improve = 0
                if cur_max < best_max:
                    best_max, best_h = cur_max, cur_h.copy()
            else:
                # revert swap
                cur_h[i], cur_h[j] = cur_h[j], cur_h[i]
                no_improve += 1

        if no_improve >= max_no_improve:
            # restart from a fresh random configuration
            no_improve = 0
            cur_h[:] = 0
            cur_h[rng.choice(n, n // 2, replace=False)] = 1
            cur_max, cur_corr = _corr_max(cur_h)
            if cur_max < best_max:
                best_max, best_h = cur_max, cur_h.copy()

    return best_h, best_max


def run(seed=42, budget_s=1100, **kwargs):
    """
    Optimize a step function for the Erdős C₅ problem.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()

    # ------------------------------------------------------------------
    # initialise h (binary, integral = 1)
    # ------------------------------------------------------------------
    init = globals().get("initial_h_values")
    if init is not None:
        init_arr = np.asarray(init, dtype=np.float64)
        n = init_arr.shape[0]
        # binarise (threshold at 0.5) and force exact half‑ones
        h = (init_arr >= 0.5).astype(np.int8)
        need = n // 2 - int(h.sum())
        if need > 0:
            zero_idx = np.flatnonzero(h == 0)
            h[rng.choice(zero_idx, need, replace=False)] = 1
        elif need < 0:
            one_idx = np.flatnonzero(h == 1)
            h[rng.choice(one_idx, -need, replace=False)] = 0
    else:
        n = 70
        h = np.zeros(n, dtype=np.int8)
        h[rng.choice(n, n // 2, replace=False)] = 1

    dx = 2.0 / n
    target_bound = 0.3808
    target_max = int(math.floor(target_bound / dx))

    # ------------------------------------------------------------------
    # main search – run until budget exhausted
    # ------------------------------------------------------------------
    deadline = start_time + budget_s - 0.5  # small safety margin
    best_h, best_max = _local_search(h, rng, deadline, target_max)

    # final bound
    c5_bound = best_max * dx
    return best_h.astype(np.float64), c5_bound, n
