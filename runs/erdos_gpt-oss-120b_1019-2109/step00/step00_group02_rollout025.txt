def gradient_step(h, step):
    n = len(h)
    # compute autocorr full
    ac = np.correlate(h, h, mode='full')
    # remove center
    # Find min non-zero
    # Use np.argmin with mask

    mask[n-1] = False
    idx = np.argmin(ac, where=mask) # but np.argmin cannot accept where; so we can set mid value high.
    # easiest: set ac_center = np.inf
    ac2 = ac.copy()

    # compute gradient vector g
    g = np.zeros(n, dtype=np.float64)

        # For j:
        # each h_i contributes to g_i via partner h_i+shift, and g_{i+shift} via partner h_i.

        h_shift = h[shift:]  # length n-shift

        # For j in shift..n-1: add h_{j-shift}
        g[shift:] += h[:n-shift]

        g[k:] += h[:n-k]

    d = g - g.mean()

    h_new = h + step * d
    # project
    S = n / 2.0
    h_proj = project_to_box_simplex(h_new, S)

import time
import random
import math
import numpy as np

def _project_box_simplex(v, target_sum, max_iter=60):
    """
    Project vector `v` onto the set {x : sum(x)=target_sum, 0<=x<=1}.
    Uses bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0   # low enough so that v - lo > 1
    hi = np.max(v) + 1.0    # high enough so that v - hi < 0
    for _ in range(max_iter):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)


def _c5_from_h(h):
    """C5 = max_k ∫h(x)(1‑h(x+k))dx, discretised."""
    n = h.shape[0]
    # autocorrelation of h with itself
    ac = np.correlate(h, h, mode='full')
    center = n - 1
    # ignore zero shift (center)
    ac_no_center = np.concatenate([ac[:center], ac[center + 1:]])
    min_ac = np.min(ac_no_center)
    # max overlap = n/2 - min_ac
    return 1.0 - (2.0 * min_ac) / n


def _worst_shift_info(h):
    """Return (min_autocorr, shift) where shift ≠ 0 gives the smallest autocorrelation."""
    n = h.shape[0]
    ac = np.correlate(h, h, mode='full')
    center = n - 1
    ac_mod = ac.copy()
    ac_mod[center] = np.inf            # ignore zero shift
    idx = int(np.argmin(ac_mod))
    shift = idx - center
    return ac_mod[idx], shift, ac


def _gradient_step(h, step):
    """Perform one projected gradient step aimed at increasing the worst autocorrelation."""
    n = h.shape[0]
    target_sum = n * 0.5
    min_val, shift, _ = _worst_shift_info(h)

    # gradient of the worst autocorrelation w.r.t. h
    g = np.zeros_like(h)
    if shift != 0:
        s = abs(shift)
        # contributions from pairs (i,i+s)
        g[: n - s] += h[s:]          # h_i contributes h_{i+s}
        g[s:] += h[: n - s]          # h_{i+s} contributes h_i
    # project onto hyperplane sum=0 before the box‑simplex projection
    d = g - g.mean()
    h_new = h + step * d
    h_new = _project_box_simplex(h_new, target_sum)
    return h_new


def _local_optimize(start_h, budget):
    """Improve `start_h` within `budget` seconds using a simple sub‑gradient scheme."""
    n = start_h.shape[0]
    target_sum = n * 0.5
    h = _project_box_simplex(start_h, target_sum)

    best_h = h.copy()
    best_c5 = _c5_from_h(h)

    step = 0.5               # initial step size
    t0 = time.time()
    while time.time() - t0 < budget:
        # gradient ascent on the worst shift
        h_candidate = _gradient_step(h, step)
        cand_c5 = _c5_from_h(h_candidate)

        if cand_c5 < best_c5 - 1e-12:
            h, best_h, best_c5 = h_candidate, h_candidate.copy(), cand_c5
            step = min(step * 1.2, 1.0)          # increase step after improvement
            continue

        # no improvement – shrink step
        step *= 0.8
        if step < 1e-4:
            step = 0.5

        # occasional random exchange to escape flat regions
        if random.random() < 0.02:
            i_candidates = np.where(h > 1e-9)[0]
            j_candidates = np.where(h < 1 - 1e-9)[0]
            if i_candidates.size and j_candidates.size:
                i = random.choice(i_candidates)
                j = random.choice(j_candidates)
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta > 0:
                    delta = random.random() * max_delta * 0.5
                    h_rand = h.copy()
                    h_rand[i] -= delta
                    h_rand[j] += delta
                    h_rand = _project_box_simplex(h_rand, target_sum)
                    rand_c5 = _c5_from_h(h_rand)
                    if rand_c5 < best_c5 - 1e-12:
                        h, best_h, best_c5 = h_rand, h_rand.copy(), rand_c5
    return best_h, best_c5
