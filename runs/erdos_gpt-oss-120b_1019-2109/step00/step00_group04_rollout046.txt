def propose_move(h, step):
    n = h.shape[0]
    i, j = np.random.randint(n, size=2) # ensure i!=j?
    while i == j:
        j = np.random.randint(n)
    hi = h[i]; hj = h[j]
    # direction delta to add to i
    delta_max = min(step, 1 - hi, hj)
    delta_min = -min(step, hi, 1 - hj)
    if delta_max <= delta_min: # cannot move
        return None
    delta = np.random.uniform(delta_min, delta_max)
    h_new = h.copy()
    h_new[i] = hi + delta
    h_new[j] = hj - delta
    return h_new

import time
import math
import numpy as np

def _random_binary_feasible(n: int) -> np.ndarray:
    """Return a binary vector of length n with exactly n/2 ones (n must be even)."""
    half = n // 2
    a = np.zeros(n, dtype=float)
    a[:half] = 1.0
    np.random.shuffle(a)
    return a


def _compute_c5(arr: np.ndarray) -> float:
    """Overlap value C5 for a discrete step function."""
    conv = np.correlate(arr, 1.0 - arr, mode="full")
    return float(conv.max() * (2.0 / arr.shape[0]))


def run(seed: int = 42, budget_s: int = 1100, **kwargs):

    Returns a tuple (h_values, c5_bound, n_points).

    h_values – numpy array with values in [0,1] and sum = n_points/2.
