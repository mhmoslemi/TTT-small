def normalize_h(h, target):
    cur = h.sum()
    if cur == 0:
        return h
    scale = target / cur
    h = h * scale
    # Clip after scaling
    np.clip(h, 0.0, 1.0, out=h)
    # If sum deviates due to clipping, redistribute residual
    cur = h.sum()
    diff = target - cur
    if abs(diff) > 1e-12:
        # distribute diff uniformly over elements not at bounds
        mask = (h > 1e-12) & (h < 1 - 1e-12)
        # if no such elements, adjust any element within bounds
        if mask.any():
            adjustment = diff / mask.sum()
            h[mask] += adjustment
            np.clip(h, 0.0, 1.0, out=h)
        else:
            # fallback: adjust all
            h += diff / len(h)
            np.clip(h, 0.0, 1.0, out=h)
    return h

    start = time.time()
    try:
        import cvxpy as cp

        # binary variables for the step function
        x = cp.Variable(n, boolean=True)
        t = cp.Variable()                     # upper bound on the overlap
        cons = [cp.sum(x) == target_sum, t >= 0]

        # linearise the products x_i * x_{i+l}
        for lag in range(1, n):
            xi = x[: n - lag]
            xj = x[lag:]
            y = cp.Variable(n - lag, boolean=True)   # y = xi & xj
            cons += [
                y <= xi,
                y <= xj,
                y >= xi + xj - 1,
                target_sum - cp.sum(y) <= t,
            ]

        prob = cp.Problem(cp.Minimize(t), cons)
        # Use a MILP solver that is usually available (GLPK_MI); fallback to CBC.
        solver = "GLPK_MI"
        prob.solve(solver=solver, verbose=False, time_limit=int(budget_s * 0.8))
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE, cp.FEASIBLE]:
            h_bin = np.rint(x.value).astype(float)   # round to 0/1
            val = float(t.value) * dx
            best_h = h_bin
            best_val = val
    except Exception:
        # any problem (missing solver, timeout, etc.) just skip the MILP step
        pass

    # ------------------------------------------------------------------ #
    # 2) If the MILP did not give a solution or time permits, explore
    #    a small family of smooth shapes and then improve them with a
    #    simulated‑annealing style local search on real values.
    # ------------------------------------------------------------------ #
    elapsed = time.time() - start
    remaining = budget_s - elapsed
    if remaining > 5.0:   # need a tiny margin for the final return
        # ----- a) quick sweep over a one‑parameter family (p≥1) -----
        def shape(p):
            a = (p + 1.0) / (2.0 * p)               # half‑width of the flat top
            x = (np.arange(n) + 0.5) * dx
            h = np.empty(n, dtype=np.float64)
            left = x <= a
            right = x >= 2.0 - a
            middle = ~left & ~right
            h[left] = (x[left] / a) ** p
            h[middle] = 1.0
            h[right] = ((2.0 - x[right]) / a) ** p
            return normalise(h)

        # scan a modest range of p values
        p_vals = np.concatenate([np.linspace(1.0, 5.0, 41),
                                 np.linspace(5.5, 12.0, 14)])
        for p in p_vals:
            h = shape(p)
            val = max_overlap(h)
            if val < best_val:
                best_val, best_h = val, h.copy()
            if time.time() - start > budget_s * 0.6:
                break

        # ----- b) optionally binarise the best continuous shape -----
        if best_h is not None and best_h.ndim == 1:
            # produce a binary vector with exactly n/2 ones (round‑down if odd)
            k = int(round(target_sum))
            idx = np.argsort(-best_h)
            bin_candidate = np.zeros_like(best_h)
            bin_candidate[idx[:k]] = 1.0
            val = max_overlap(bin_candidate)
            if val < best_val:
                best_val, best_h = val, bin_candidate.copy()

        # ----- c) local search (simulated annealing) ----------------
        # work on the current best (may be binary or real)
        if best_h is not None:
            cur = best_h.copy()
            cur_val = best_val
            rng = np.random.default_rng(seed + 1)
            T0 = 0.02
            it = 0
            while time.time() - start < budget_s * 0.95:
                # pick two distinct positions
                i, j = rng.integers(n, size=2)
                if i == j:
                    continue
                # direction of mass transfer
                if rng.random() < 0.5:
                    src, dst = i, j
                else:
                    src, dst = j, i
                # maximal admissible delta
                max_delta = min(cur[src], 1.0 - cur[dst])
                if max_delta <= 1e-12:
                    continue
                delta = rng.random() * max_delta
                new = cur.copy()
                new[src] -= delta
                new[dst] += delta
                new_val = max_overlap(new)
                dcost = new_val - cur_val
                # temperature schedule
                T = T0 * math.exp(-it / 200000.0)
                accept = False
                if dcost <= 0:
                    accept = True
                elif T > 0 and rng.random() < math.exp(-dcost / T):
                    accept = True
                if accept:
                    cur, cur_val = new, new_val
                    if cur_val < best_val:
                        best_val, best_h = cur_val, cur.copy()
                it += 1
                # occasional re‑heat around the best found so far
                if it % 100000 == 0 and cur_val > best_val + 1e-5:
                    cur = best_h.copy()
                    cur_val = best_val

    # ------------------------------------------------------------------ #
    # 3) Final sanity checks and return
    # ------------------------------------------------------------------ #
    if best_h is None:
        # as a last resort fall back to a uniform constant 0.5 function
        best_h = np.full(n, 0.5, dtype=np.float64)
        best_h = normalise(best_h)
        best_val = max_overlap(best_h)

    # enforce constraints one more time (tiny numerical adjustments)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = normalise(best_h)

    final_val = max_overlap(best_h)

    # return the step‑function values, the bound, and the number of points
    return (best_h.tolist(), final_val, n)
