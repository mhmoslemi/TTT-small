def worst_shift_transfer(h, step_frac, rng):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # ignore zero shift
    zero_idx = n - 1
    corr[zero_idx] = -np.inf
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)
    if s == 0:
        # won't happen because set -inf
        return False
    if s >= 0:
        i_vals = np.arange(0, n - s)
        j_vals = i_vals + s
    else:
        i_vals = np.arange(-s, n)
        j_vals = i_vals + s
    # mask feasible pairs where i>0 and j<1
    feas_mask = (h[i_vals] > 1e-12) & (h[j_vals] < 1 - 1e-12)
    if not np.any(feas_mask):
        return False
    contrib = h[i_vals] * (1 - h[j_vals])
    # set to -inf where not feasible to ignore
    contrib[~feas_mask] = -np.inf
    idx_local = int(np.argmax(contrib))
    i = int(i_vals[idx_local])
    j = int(j_vals[idx_local])
    max_transfer = min(h[i], 1 - h[j])
    if max_transfer <= 0:
        return False
    t = max_transfer * step_frac
    h[i] -= t
    h[j] += t
    return True

import time
import numpy as np

def _c5_cost(vec):
    """Compute C5 for a given vector (uses the provided evaluator if present)."""
    # Use pre‑imported helper if it exists.
    if "evaluate_erdos_solution" in globals():
        try:
            return float(evaluate_erdos_solution(vec))
        except Exception:
            pass
    n = len(vec)
    corr = np.correlate(vec, 1.0 - vec, mode="full")
    return float(corr.max() * (2.0 / n))


def _fix_sum(h, target, rng):
    """Adjust h by pairwise transfers so that sum(h) == target (exactly)."""
    diff = target - h.sum()
    n = len(h)
    # simple pairwise fixing – each transfer preserves bounds.
    while abs(diff) > 1e-12:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        if diff > 0:
            inc = min(diff, 1.0 - h[i])
            dec = min(diff, h[j])
            t = min(inc, dec)
            if t <= 0:
                continue
            h[i] += t
            h[j] -= t
            diff -= t
        else:
            dec = min(-diff, h[i])
            inc = min(-diff, 1.0 - h[j])
            t = min(dec, inc)
            if t <= 0:
                continue
            h[i] -= t
            h[j] += t
            diff += t
    return h


def _best_feasible_pair(h):
    """Find the index pair (i,j) belonging to the worst (non‑zero) shift
    that can actually transfer mass (h[i]>0, h[j]<1).  Returns (i,j,transfer,shift)
    or (None, None, 0, None) if no feasible pair exists."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    zero_idx = n - 1
    corr[zero_idx] = -np.inf            # ignore zero shift
    max_idx = int(np.argmax(corr))
    s = max_idx - (n - 1)               # integer shift (may be negative)
    if s == 0:
        return None, None, 0.0, None    # shouldn't happen after masking
    # indices participating in this shift
    if s > 0:
        i_vals = np.arange(0, n - s, dtype=int)
        j_vals = i_vals + s
    else:
        i_vals = np.arange(-s, n, dtype=int)
        j_vals = i_vals + s
    # feasibility mask
    feasible = (h[i_vals] > 1e-12) & (h[j_vals] < 1.0 - 1e-12)
    if not np.any(feasible):
        return None, None, 0.0, None
    contrib = h[i_vals] * (1.0 - h[j_vals])
    contrib[~feasible] = -np.inf
    loc = int(np.argmax(contrib))
    i = int(i_vals[loc])
    j = int(j_vals[loc])
    max_transfer = min(h[i], 1.0 - h[j])
    return i, j, max_transfer, s


def run(seed=42, budget_s=1100, **kwargs):
    """Return a step‑function h (as a NumPy array), its C5 bound and length.
    The algorithm performs a focused local search that repeatedly moves mass
    from positions that contribute most to the worst overlap shift towards
    their partners, always preserving the integral and box constraints."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)   # provided by harness
    except Exception:

    target_sum = n / 2.0
    np.clip(h0, 0.0, 1.0, out=h0)
    h0 = _fix_sum(h0, target_sum, rng)

    # best solution seen so far
    best_h = h0.copy()
    best_cost = _c5_cost(best_h)

    # current solution (will be modified in place)
    cur_h = best_h.copy()
    cur_cost = best_cost

    start = time.time()
    total = budget_s
    # parameters for the annealing‑style schedule
    T0 = 0.002
    no_improve = 0
    max_no_improve = 2000

    while time.time() - start < total - 0.2:
        # ------------------------------------------------------------------
        # locate a feasible pair to move mass along the worst shift
        # ------------------------------------------------------------------
        i, j, max_tr, shift = _best_feasible_pair(cur_h)

        if i is None:
            # fallback: random transfer (always feasible because the sum constraint holds)
            i, j = rng.integers(n, size=2)
            if i == j:
                continue
            # choose direction that is feasible
            if rng.random() < 0.5:
                max_tr = min(cur_h[i], 1.0 - cur_h[j])
                if max_tr <= 0:
                    continue
                direction = -1   # i loses, j gains
            else:
                max_tr = min(cur_h[j], 1.0 - cur_h[i])
                if max_tr <= 0:
                    continue
                direction = 1    # i gains, j loses
                i, j = j, i      # swap to keep the same update rule later
        # step size shrinks as time passes
        elapsed = time.time() - start
        frac = max(0.001, 0.5 * (1.0 - elapsed / total))
        delta = max_tr * frac

        # apply the transfer (i loses delta, j gains delta)
        cur_h[i] -= delta
        cur_h[j] += delta

        # evaluate new cost
        new_cost = _c5_cost(cur_h)

        # annealing acceptance
        T = T0 * max(0.0, 1.0 - elapsed / total)
        accept = False
        if new_cost <= cur_cost:
            accept = True
        else:
            prob = np.exp((cur_cost - new_cost) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:

            if new_cost < best_cost - 1e-12:
                best_cost = new_cost
                best_h = cur_h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            # revert the move
            cur_h[i] += delta
            cur_h[j] -= delta
            no_improve += 1

        # ------------------------------------------------------------------
        # occasional perturbation to escape plateaus
        # ------------------------------------------------------------------

            cur_h += (rng.random(n) - 0.5) * eps

            cur_h = _fix_sum(cur_h, target_sum, rng)
            cur_cost = _c5_cost(cur_h)
            no_improve = 0

    # final sanity check – guarantee constraints exactly
    best_h = _fix_sum(best_h, target_sum, rng)
    final_cost = _c5_cost(best_h)

    return best_h, float(final_cost), len(best_h)
