def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(conv.max() * dx)

def gradient_worst_shift(h, k):
    n = h.shape[0]
    grad = np.ones(n)
    i = np.arange(n)
    i_plus = i + k
    i_minus = i - k
    mask_plus = (i_plus >= 0) & (i_plus < n)
    mask_minus = (i_minus >= 0) & (i_minus < n)
    grad[mask_plus] -= h[i_plus[mask_plus]]
    grad[mask_minus] -= h[i_minus[mask_minus]]
    return grad

def adjust_sum(v, target):
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    n = v.shape[0]
    # simple deterministic adjustment iterating through indices
    i = 0
    while abs(diff) > 1e-12 and i < n:
        if diff > 0:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        i += 1
    # If still not exact due to rounding, distribute uniformly
    if abs(diff) > 1e-12:
        v += diff / n
    return v

def random_initial(n, rng):
    # start from 0.5 and add zero-mean perturbation
    base = np.full(n, 0.5, dtype=np.float64)
    perturb = rng.normal(size=n)

    h = base + eps * perturb
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    h = adjust_sum(h, target)
    return h

    h_new[i] -= eps

import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Maximum overlap integral for a discretised step function."""
    n = h.shape[0]
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(conv.max() * dx)

def gradient_worst_shift(h: np.ndarray, k: int) -> np.ndarray:
    """Sub‑gradient of the worst‑case overlap term w.r.t. h."""
    n = h.shape[0]
    grad = np.ones(n)
    i = np.arange(n)
    ip = i + k
    im = i - k
    mask_p = (ip >= 0) & (ip < n)
    mask_m = (im >= 0) & (im < n)
    grad[mask_p] -= h[ip[mask_p]]
    grad[mask_m] -= h[im[mask_m]]
    return grad

def adjust_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Modify v in‑place so that sum(v) == target while respecting 0≤v≤1."""
    diff = target - v.sum()
    if abs(diff) < 1e-12:
        return v
    n = v.shape[0]
    i = 0
    while abs(diff) > 1e-12 and i < n:
        if diff > 0:
            inc = min(diff, 1.0 - v[i])
            v[i] += inc
            diff -= inc
        else:
            dec = min(-diff, v[i])
            v[i] -= dec
            diff += dec
        i += 1
    if abs(diff) > 1e-12:          # very rare fallback
        v += diff / n
    return v
