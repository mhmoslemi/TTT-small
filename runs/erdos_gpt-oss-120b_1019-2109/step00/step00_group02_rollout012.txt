def gradient_optimize(h0, max_iter, time_limit):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_cmax(h)
    t0 = time.time()
    # parameters
    step0 = 0.5
    alpha_max = 30.0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        # compute correlation vector c
        c = np.correlate(h, 1 - h, mode='full')
        # compute weights
        # Use scaling with dx for actual values? Should include dx? The values in c are unscaled by dx. Use original c (not scaled?) For gradient scaling we have dx factor; but weight can use same scaling; not a problem.
        # compute softmax weights
        # For smoothing we may use exponent based on c (unscaled) but need to scale effect.
        maxc = np.max(c)
        # choose alpha
        alpha = 1.0 + (alpha_max - 1.0) * (it / max_iter)
        exp_terms = np.exp(alpha * (c - maxc))
        w = exp_terms / exp_terms.sum()
        # compute conv_pos and conv_neg
        offsets = np.arange(-n+1, n)  # length 2n-1
        # Compute conv_pos via loops
        conv_pos = np.zeros(n)
        conv_neg = np.zeros(n)
        # we can vectorize with statements inside for loop but n perhaps <=200, loops okay
        for idx, offset in enumerate(offsets):
            w_off = w[idx]
            if w_off == 0.0:
                continue
            if offset >= 0:
                conv_pos[:n-offset] += w_off * h[offset:]
                conv_neg[offset:] += w_off * h[:n-offset]
            else:
                off = -offset
                conv_pos[off:] += w_off * h[:n-off]
                conv_neg[:n-off] += w_off * h[off:]
        grad = dx * (1.0 - conv_pos - conv_neg)
        # step size
        step = step0 / np.sqrt(it + 1)
        y = h - step * grad
        # projection onto capped simplex
        h = proj_capped_simplex(y, n / 2.0)
        # evaluate Cmax
        cur_c = compute_cmax(h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()

def hill_climb_binary(n, time_limit):

    h = beatty_pattern(n)
    cur_c = compute_cmax(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()

    while time.time() - start < time_limit:
        # Random swap
        ones = np.flatnonzero(h == 1)
        zeros = np.flatnonzero(h == 0)
        if len(ones) == 0 or len(zeros) == 0:

        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = 0, 1
        new_c = compute_cmax(h)
        # accept if improvement else revert
        if new_c <= cur_c:

            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # revert
            h[i], h[j] = 1, 0

    return best_h, best_c

import time
import math
import numpy as np

def compute_cmax(h):
    """Maximum overlap integral for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(corr.max())

def proj_capped_simplex(v, target):
    """
    Euclidean projection of v onto the set {x | 0<=x<=1, sum(x)=target}.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0   # makes v - lo > 1  -> all clipped to 1
    hi = np.max(v) + 1.0    # makes v - hi < 0  -> all clipped to 0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

def weighted_conv(h, w):
    """Convolution sum_i w_{i-j} * h_j for each i (zero‑padded)."""
    n = h.shape[0]
    conv = np.zeros(n)
    start = -n + 1
    for idx, wv in enumerate(w):
        if wv == 0.0:
            continue
        shift = start + idx
        if shift >= 0:
            conv[:n - shift] += wv * h[shift:]
        else:
            s = -shift
            conv[s:] += wv * h[:n - s]
    return conv

def gradient_optimize(h0, max_iter, time_limit):
    """Smooth max‑gradient descent with projection onto the capped simplex."""
    n = h0.shape[0]
    dx = 2.0 / n
    target = n / 2.0
    h = proj_capped_simplex(h0, target)
    best_h = h.copy()
    best_c = compute_cmax(h)
    t0 = time.time()
    alpha_max = 30.0
    step0 = 0.5
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        # correlation (without dx for weighting)
        c_raw = np.correlate(h, 1.0 - h, mode='full')
        # smooth weights
        alpha = 1.0 + (alpha_max - 1.0) * it / max_iter
        mx = c_raw.max()
        expw = np.exp(alpha * (c_raw - mx))
        w = expw / expw.sum()
        # gradients via two weighted convolutions
        conv_pos = weighted_conv(h, w)
        conv_neg = weighted_conv(h, w[::-1])
        grad = dx * (1.0 - conv_pos - conv_neg)
        step = step0 / math.sqrt(it + 1.0)
        y = h - step * grad
        h = proj_capped_simplex(y, target)
        cur_c = compute_cmax(h)
        if cur_c < best_c:
            best_c = cur_c
            best_h = h.copy()
    return best_h, best_c

def beatty_pattern(n):
    """Balanced 0/1 pattern using a Beatty sequence (golden ratio)."""
    phi = (1.0 + np.sqrt(5.0)) * 0.5
    frac = (np.arange(n, dtype=np.float64) * phi) % 1.0
    idx = np.argsort(frac)
    h = np.zeros(n, dtype=np.float64)
    h[idx[:n // 2]] = 1.0
    if n % 2:
        # one extra slot: give it the average value 0.5 to keep the integral exact
        h[idx[n // 2]] = 0.5
    return h

def hill_climb_binary(n, time_limit):
    """Simple hill‑climbing on binary vectors (swap a 0 and a 1)."""
    h = beatty_pattern(n)  # good deterministic start
    cur_c = compute_cmax(h)
    best_h = h.copy()
    best_c = cur_c
    start = time.time()
    ones = np.flatnonzero(h == 1.0)
    zeros = np.flatnonzero(h == 0.0)
    # temperature schedule for occasional uphill moves
    T0 = 0.05
    while time.time() - start < time_limit:
        if ones.size == 0 or zeros.size == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h[i], h[j] = 0.0, 1.0
        new_c = compute_cmax(h)
        if new_c < cur_c:
            cur_c = new_c
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            # simulated‑annealing acceptance
            t = T0 * (1.0 - (time.time() - start) / time_limit)
            if np.random.rand() < math.exp(-(new_c - cur_c) / max(t, 1e-8)):
                cur_c = new_c
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[i], h[j] = 1.0, 0.0   # revert
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_total = time.time()
    best_h = None
    best_c = float('inf')
    best_n = None

    # ---- candidates -------------------------------------------------
    candidates = []

    # use the provided initial construction if present
    try:
        init = np.asarray(initial_h_values, dtype=np.float64)
        candidates.append(('initial', init))
    except NameError:
        pass

    # deterministic Beatty patterns for several even sizes
    for n in (44, 48, 56, 64, 80, 96, 112, 128):
        if n % 2 == 0:
            candidates.append((f'beatty_{n}', beatty_pattern(n)))

    # a few random binary starts
    for n in (64, 80, 100):
        if n % 2:
            continue
        h = np.zeros(n, dtype=np.float64)
        h[np.random.choice(n, n // 2, replace=False)] = 1.0
        candidates.append((f'random_{n}', h))

    # allocate time roughly equally among candidates, leaving a margin for final refinement
    remaining_budget = budget_s - 0.2 * budget_s
    per_candidate = max(0.05, remaining_budget / max(len(candidates), 1))

    for name, h0 in candidates:
        if time.time() - start_total >= budget_s:
            break
        n = h0.shape[0]
        # decide which optimiser to use
        if np.all(np.logical_or(h0 == 0.0, h0 == 1.0)):
            # binary optimisation
            h_opt, c_opt = hill_climb_binary(n, per_candidate)
        else:
            # continuous optimisation (projection keeps the mean exact)
            h_opt, c_opt = gradient_optimize(h0, max_iter=8000, time_limit=per_candidate)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = n

    # ---- final refinement on the current best -----------------------
    if best_h is not None and time.time() - start_total < budget_s - 0.1:
        spare = budget_s - (time.time() - start_total) - 0.05
        # a final longer gradient descent run
        h_ref, c_ref = gradient_optimize(best_h, max_iter=20000, time_limit=spare)
        if c_ref < best_c:
            best_c = c_ref
            best_h = h_ref.copy()
            best_n = best_h.shape[0]

    # final safety projection and recomputation
    if best_h is not None:
        best_h = proj_capped_simplex(best_h, best_h.shape[0] / 2.0)
        best_c = compute_cmax(best_h)
        best_n = int(best_h.shape[0])

    # return as Python list for the harness
    return (best_h.tolist() if best_h is not None else [], float(best_c), int(best_n))
