import numpy as np, time, math

def compute_max_overlap(h, dx):
    O = np.correlate(h, 1 - h, mode='full') * dx
    return O.max(), O

def project_to_capped_simplex(v, target_sum):
    # v: np.ndarray
    # find lambda such that sum of clip(v - lambda, 0, 1) = target_sum
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)
```

import time, math
import numpy as np

def _compute_overlap(h, dx):
    """Return (max_overlap, full_overlap_array)."""
    O = np.correlate(h, 1.0 - h, mode="full") * dx
    return O.max(), O

def _proj_capped_simplex(v, target):
    """
    Project v onto the set {x | 0<=x_i<=1, sum(x)=target}.
    Uses the fact that the solution is x_i = min(1, max(0, v_i - lam))
    for a suitable lam found by bisection.
    """
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)

def _random_feasible(rng, n, target):
    """Generate a random vector in [0,1]^n with sum = target."""
    v = rng.random(n)
    return _proj_capped_simplex(v, target)

def run(seed=42, budget_s=1100, **kwargs):
