def run(...):
    n = 200
    dx = 2.0 / n

    # Initialize random vector with half ones
    rng = np.random.default_rng(seed)
    h = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=n//2, replace=False)
    h[idx] = 1.0

    # compute cost
    def compute_cost(arr):
        return np.max(np.correlate(arr, 1 - arr, mode='full'))

    best_h = h.copy()
    best_cost = compute_cost(h)
    current_h = h.copy()
    current_cost = best_cost

    start = time.time()
    time_limit = budget_s - 0.05

    # SA parameters
    T = 2.0 * best_cost  # maybe start high
    while time.time() < time_limit:
        # propose move: swap random 1 and 0
        ones = np.where(current_h > 0.5)[0]
        zeros = np.where(current_h < 0.5)[0]
        i = rng.choice(ones)
        j = rng.choice(zeros)
        # swap
        current_h[i], current_h[j] = 0, 1
        new_cost = compute_cost(current_h)
        delta = new_cost - current_cost
        accept = False
        if delta <= 0:
            accept = True
        else:
            prob = np.exp(-delta / T)
            if rng.random() < prob:
                accept = True
        if accept:
            current_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = current_h.copy()
        else:
            # revert swap
            current_h[i], current_h[j] = 1, 0

        # cooling
        T *= 0.99995   # slow cooling

    # return best solution
    c5_bound = best_cost * dx
    return (best_h, float(c5_bound), n)

def propose_move(h, rng):
    # choose two distinct indices
    i, j = rng.choice(n, size=2, replace=False)
    # ensure we increase hi and decrease hj if possible else reverse
    hi = h[i]
    hj = h[j]
    # compute deltas
    if hi < hj:
        # we desire to move mass from j to i
        max_delta = min(1 - hi, hj)
        if max_delta <= 1e-12:
            return None
        delta = rng.random() * max_delta
        # apply
        h[i] += delta
        h[j] -= delta
        return (i, j, delta, True) # True indicating we increased i
    elif hi > hj:
        # move from i to j
        max_delta = min(hi, 1 - hj)
        if max_delta <= 1e-12:
            return None
        delta = rng.random() * max_delta
        h[i] -= delta
        h[j] += delta
        return (i, j, delta, False)
    else:
        # equal, skip
        return None

import time
import numpy as np

def _compute_raw_cost(h):
    """raw (unnormalized) max overlap = max_k sum_i h[i]*(1-h[i+k])"""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr)

def run(seed: int = 42, budget_s: float = 1100.0, **kwargs):
    """
    Simulated‑annealing search for a step function h on [0,2] (discretised)
    that minimises   C5 = max_k ∫ h(x)(1-h(x+k))dx .
    The function returns (h_values, C5_bound, n_points).
    """
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------------
    # Choose a working discretisation.
    # 200 points gives a good trade‑off between quality and speed.
    # -------------------------------------------------------------
    n = 200
    dx = 2.0 / n

    # start from the constant 0.5 function (feasible and easy)
    h = np.full(n, 0.5, dtype=np.float64)

    # -------------------------------------------------------------
    # Helper to keep the sum constraint exact (up to floating noise)
    # -------------------------------------------------------------
    target_sum = n / 2.0                       # integral must be 1 → sum = n/2
    eps = 1e-12

    # -------------------------------------------------------------
    # Simulated‑annealing parameters
    # -------------------------------------------------------------
    best_h = h.copy()
    best_raw = _compute_raw_cost(best_h)

    current_h = h.copy()
    current_raw = best_raw

    # temperature works in the *raw* cost domain (integer‑like values)
    T = max(1.0, best_raw * 0.5)
    decay = 0.9995                     # slow exponential cooling

    start_time = time.time()
    end_time = start_time + budget_s - 0.05   # give a tiny safety margin

    target_bound = 0.38085                   # we want to be below this value
    target_raw = target_bound / dx           # corresponding raw count

    # -------------------------------------------------------------
    # Main optimisation loop
    # -------------------------------------------------------------
    while time.time() < end_time:
        # recompute correlations only when needed
        corr = np.correlate(current_h, 1.0 - current_h, mode="full")
        current_raw = np.max(corr)

        # early exit if we already beat the target
        if current_raw <= target_raw:
            best_h = current_h.copy()
            best_raw = current_raw
            break

        # identify the lag that currently attains the maximum
        lag_index = np.argmax(corr)                 # index in full convolution
        lag = lag_index - (n - 1)                    # convert to shift value (positive = right)

        # compute contribution of each pair for this lag
        if lag >= 0:
            # terms involve h[i] * (1 - h[i+lag]), i = 0 .. n-lag-1
            first = current_h[: n - lag]
            second = current_h[lag:]
            contrib = first * (1.0 - second)
            # pick the pair with the largest contribution (worst offender)
            i_local = int(np.argmax(contrib))
            i = i_local
            j = i + lag
        else:
            # lag < 0  →  use shift = -k
            k = -lag
            # terms involve h[i] * (1 - h[i-k]), i = k .. n-1
            first = current_h[k:]
            second = current_h[: n - k]
            contrib = first * (1.0 - second)
            i_local = int(np.argmax(contrib))
            i = i_local + k
            j = i - k

        hi, hj = current_h[i], current_h[j]

        # we would like to move mass from the larger entry to the smaller one
        if hi > hj:
            max_delta = min(hi, 1.0 - hj)
            if max_delta < eps:
                # fallback: random swap of a 1‑pixel and a 0‑pixel
                ones = np.where(current_h > 0.5)[0]
                zeros = np.where(current_h < 0.5)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    continue
                i, j = rng.choice(ones), rng.choice(zeros)
                max_delta = min(1.0 - current_h[i], current_h[j])
                if max_delta < eps:
                    continue
                delta = max_delta * rng.random()
                current_h[i] += delta
                current_h[j] -= delta
            else:
                delta = max_delta * rng.random()
                current_h[i] -= delta
                current_h[j] += delta
        elif hj > hi:
            # reverse situation – move mass the other way
            max_delta = min(hj, 1.0 - hi)
            if max_delta < eps:
                continue
            delta = max_delta * rng.random()
            current_h[i] += delta
            current_h[j] -= delta
        else:
            # perfectly equal – perform a tiny random perturbation elsewhere
            idx1, idx2 = rng.choice(n, size=2, replace=False)
            hi, hj = current_h[idx1], current_h[idx2]
            if hi == hj:
                continue
            if hi > hj:
                max_delta = min(hi, 1.0 - hj)
                delta = max_delta * rng.random()
                current_h[idx1] -= delta
                current_h[idx2] += delta
            else:
                max_delta = min(hj, 1.0 - hi)
                delta = max_delta * rng.random()
                current_h[idx1] += delta
                current_h[idx2] -= delta

        # re‑enforce bounds (numerical safety)
        np.clip(current_h, 0.0, 1.0, out=current_h)

        # recompute raw cost after the move
        new_raw = _compute_raw_cost(current_h)
        delta_raw = new_raw - current_raw

        # Metropolis acceptance
        if delta_raw <= 0 or rng.random() < np.exp(-delta_raw / max(T, eps)):
            current_raw = new_raw
            if new_raw < best_raw:
                best_raw = new_raw
                best_h = current_h.copy()
        else:
            # revert the move
            # (the reverse operation is simply to undo the last delta)
            # we stored the indices and delta implicitly via the update above;
            # recompute the opposite change
            if hi > hj:
                # we decreased hi and increased hj
                current_h[i] += delta
                current_h[j] -= delta
            elif hj > hi:
                current_h[i] -= delta
                current_h[j] += delta
            else:
                # fallback random pair
                current_h[idx1] = hi
                current_h[idx2] = hj

        # temperature cooling
        T *= decay

    # final cost (scaled)
    final_raw = _compute_raw_cost(best_h)
    c5_bound = final_raw * dx

    # final correction of the sum (tiny floating‑point drift)
    current_sum = np.sum(best_h) * dx
    if abs(current_sum - 1.0) > 1e-12:
        # distribute the tiny error uniformly (still respects bounds)
        correction = (1.0 - current_sum) / n
        best_h += correction
        np.clip(best_h, 0.0, 1.0, out=best_h)

    return (best_h, float(c5_bound), n)
