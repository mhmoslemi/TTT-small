def make_trapezoid(n, r):
    m = n//2 - r
    a = (n//2 - r)//2
    # create array of zeros length n
    h = np.zeros(n, dtype=np.float64)
    # left ramp indices: from a to a+r-1
    left_start = a
    left_end = a + r
    # values for left ramp
    if r>0:
        ramp_vals = (np.arange(1, r+1) / (r+1))
        h[left_start:left_end] = ramp_vals
        # plateau
        plateau_start = left_end
        plateau_end = plateau_start + m
        h[plateau_start:plateau_end] = 1.0
        # right ramp values decreasing
        h[plateau_end:plateau_end + r] = ramp_vals[::-1]
    else:
        # just zeros left, plateau full of ones
        plateau_start = a
        plateau_end = a + m
        h[plateau_start:plateau_end] = 1.0
    # remaining zeros automatically
    return h

    np.random.seed(seed)

    # Determine n

        n = len(initial_h_values)

    dx = 2.0 / n

    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    best_h = None
    best_c5 = float('inf')

    if init_h is not None:
        best_h = init_h.copy()
        best_c5 = compute_c5(best_h)

    # search over trapezoid shapes
    max_r = n // 2
    for r in range(0, max_r + 1, 2):   # step 2 for parity
        m = n // 2 - r
        a = (n // 2 - r) // 2
        h = np.zeros(n, dtype=np.float64)
        if r > 0:
            ramp = (np.arange(1, r + 1) / (r + 1))
            h[a:a + r] = ramp
            h[a + r:a + r + m] = 1.0
            h[a + r + m:a + r + m + r] = ramp[::-1]
        else:
            h[a:a + m] = 1.0
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # local hill climbing from best_h
    start = time.time()
    eps = 0.1
    # number of iterations determined by time
    while time.time() - start < budget_s * 0.9:
        # generate perturbation
        i, j = random.sample(range(n), 2)
        # compute allowed delta range
        low = max(-best_h[i], best_h[j] - 1.0, -eps)
        high = min(1.0 - best_h[i], best_h[j], eps)
        if low >= high:
            continue
        delta = random.uniform(low, high)
        h_new = best_h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        # evaluate
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new
            # maybe reduce eps a bit when improvement
            eps = max(eps * 0.99, 1e-4)
        else:
            # occasionally reduce eps
            eps = max(eps * 0.9999, 1e-4)

    # final verification: ensure constraints
    # clip within [0,1] due to numerical error
    best_h = np.clip(best_h, 0.0, 1.0)
    # adjust sum to exact n/2 by distributing error to central region
    current_sum = best_h.sum()
    target_sum = n / 2.0
    diff = target_sum - current_sum
    if abs(diff) > 1e-9:

        indices = np.where((best_h > 0) & (best_h < 1))[0]

            idx = indices[len(indices)//2]
            best_h[idx] = np.clip(best_h[idx] + diff, 0.0, 1.0)

    final_c5 = compute_c5(best_h)

    return best_h, final_c5, n

def _c5_value(h):
    """Compute the Erdős overlap constant for a given vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # Linear (non‑circular) correlation as described in the statement
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _make_trapezoid(n, r):
    """
    Build a symmetric trapezoidal step function of length n.
    r : length of each ramp (must have the same parity as n//2).
    The construction guarantees 0 ≤ h[i] ≤ 1 and sum(h)=n/2.
    """
    half = n // 2
    m = half - r                     # length of the flat top (value 1)
    a = (half - r) // 2              # length of each zero tail

    h = np.zeros(n, dtype=np.float64)

    if r > 0:
        # linear ramp values: 1/(r+1), 2/(r+1), ..., r/(r+1)
        ramp = np.arange(1, r + 1, dtype=np.float64) / (r + 1.0)
        # left ramp
        h[a:a + r] = ramp
        # flat top
        h[a + r:a + r + m] = 1.0
        # right ramp (mirrored)
        h[a + r + m:a + r + m + r] = ramp[::-1]
    else:
        # pure block (no ramps)
        h[a:a + m] = 1.0

    return h
