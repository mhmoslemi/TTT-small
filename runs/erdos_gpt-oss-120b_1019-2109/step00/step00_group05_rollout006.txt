def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project vector v onto the hyperplane sum(v)=target with 0<=v<=1."""
    lo, hi = -1.0, 1.0
    # binary search on lambda such that sum(clamp(v - λ,0,1)) = target
    for _ in range(60):
        mid = (lo + hi) / 2.0
        w = np.clip(v - mid, 0.0, 1.0)
        s = w.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)


def _max_corr_and_offset(h: np.ndarray):
    """Return max correlation value, the offset that attains it, and the full array."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    offset = idx - (len(h) - 1)
    return float(corr[idx]), offset, corr


def _compute_c5(h: np.ndarray) -> float:
    """C5 value for step function h."""
    max_corr, _, _ = _max_corr_and_offset(h)
    return max_corr * (2.0 / h.size)

        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            last_improve = now

        # adaptive parameters
        elapsed_frac = (now - start_time) / budget_s
        step_size = min_step + (max_step_initial - min_step) * (1.0 - elapsed_frac)
        temperature = max_temp_initial * (1.0 - elapsed_frac)

        # sub‑gradient based move (most of the time)
        if rng.random() < 0.85:
            g = _subgradient_worst(h, offset)
            dec_candidates = np.where((g > 1e-12) & (h > 1e-12))[0]
            inc_candidates = np.where((g < -1e-12) & (h < 1.0 - 1e-12))[0]

            if dec_candidates.size and inc_candidates.size:
                i = rng.choice(dec_candidates)
                j = rng.choice(inc_candidates)
                delta_max = min(h[i], 1.0 - h[j], step_size)
                if delta_max > 0:
                    delta = delta_max * rng.random()
                    h[i] -= delta
                    h[j] += delta
                else:
                    continue
            else:
                # fallback to a random pair if gradient gives no direction
                i, j = rng.choice(n, size=2, replace=False)
                delta_max = min(h[i], 1.0 - h[j], step_size)
                if delta_max <= 0:
                    continue
                delta = delta_max * rng.random()
                h[i] -= delta
                h[j] += delta
        else:
            # pure random perturbation
            i, j = rng.choice(n, size=2, replace=False)
            delta_max = min(h[i], 1.0 - h[j], step_size)
            if delta_max <= 0:
                continue
            delta = delta_max * rng.random()
            h[i] -= delta
            h[j] += delta

        # evaluate new candidate
        new_max_corr = np.max(np.correlate(h, 1.0 - h, mode='full'))
        new_c5 = new_max_corr * (2.0 / n)

        # decide acceptance
        if new_c5 <= c5 + 1e-12:
            # improvement (or ties) – keep it
            continue
        else:
            # possible uphill move with Metropolis rule
            delta_val = new_c5 - c5
            accept_prob = math.exp(-delta_val / max(temperature, 1e-12))
            if rng.random() < accept_prob:
                # accept uphill
                continue
            # reject: roll back
            h[i] += delta
            h[j] -= delta

        # occasional restart if stuck for a while
        if now - last_improve > improve_interval:
            # perturb the current best a little to escape flat regions
            h = best_h.copy()
            # add small random noise then re‑project to feasibility
            noise = (rng.random(n) - 0.5) * 0.02  # tiny jitter
            h = np.clip(h + noise, 0.0, 1.0)
            h = _project_box_sum(h, target_sum)
            last_improve = now

    # final evaluation (make sure best_h still feasible)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_box_sum(best_h, target_sum)
    final_c5 = _compute_c5(best_h)

    return best_h, final_c5, n
