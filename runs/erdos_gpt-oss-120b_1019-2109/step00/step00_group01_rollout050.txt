def c5_of_h(h):
    n = h.shape[0]
    # pad length to next power of 2 for speed: m = 1 << (2*n-1).bits_length()
    m = 1 << ((2*n-1).bit_length())  # next power of two
    fh = np.fft.rfft(h, n=m)
    fgh = np.fft.rfft(1 - h, n=m)
    conv = np.fft.irfft(fh * np.conj(fgh), n=m)[:2*n-1]  # real part
    max_val = np.max(conv) * (2.0 / n)
    return max_val

def project_sum_bounds(v, target):
    # project onto [0,1] and sum = target using bisection on offset
    lo = -2.0
    hi = 2.0

        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:

        else:
            hi = lam

        if free.any():

            x = np.clip(x, 0.0, 1.0)

import time
import numpy as np

def _c5_value(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k)) dx for a step vector."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx


def _project_sum_bounds(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project ``v`` onto the box [0,1] with the linear equality ``sum(v)=target``.
    Uses a simple bisection on the offset λ for the formula x_i = clip(v_i‑λ,0,1).
    """
    lo, hi = -2.0, 2.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction in case of numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function h on [0,2] for the Erdős minimum overlap constant C5.
    Returns (h_values, c5_bound, n_points).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # ----------------------------------------------------------------------
    # 1. Choose resolution and build an initial feasible vector
    # ----------------------------------------------------------------------
    N_target = 200                     # moderate size – fast enough, good flexibility
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        if base.size != N_target:
            # linear interpolation to the target size
            old_n = base.size
            old_dx = 2.0 / old_n
            old_x = (np.arange(old_n) + 0.5) * old_dx
            new_dx = 2.0 / N_target
            new_x = (np.arange(N_target) + 0.5) * new_dx
            h = np.interp(new_x, old_x, base, left=base[0], right=base[-1])
        else:
            h = base.copy()
    else:
        # a simple triangular start (area exactly 1 after scaling)
        dx = 2.0 / N_target
        x = (np.arange(N_target) + 0.5) * dx
        h = np.maximum(0.0, 1.0 - np.abs(x - 1.0))
        target_sum = N_target / 2.0
        cur_sum = h.sum()
        if cur_sum > 0:
            h = np.clip(h * (target_sum / cur_sum), 0.0, 1.0)

    # enforce hard constraints (sum = N/2, 0<=h<=1)
    N = h.shape[0]
    target_sum = N / 2.0
    if (abs(h.sum() - target_sum) > 1e-12) or (h.min() < 0) or (h.max() > 1):
        h = _project_sum_bounds(h, target_sum)

    # ----------------------------------------------------------------------
    # 2. Simulated‑annealing style local search
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_c5 = _c5_value(best_h)

    cur_h = h.copy()
    cur_c5 = best_c5

    # schedule parameters
    eps_start, eps_end = 0.20, 0.001       # max transfer amount
    temp_start, temp_end = 0.020, 1e-5    # temperature for Metropolis acceptance
    max_iters = 2_000_000                 # generous upper bound; real stop by time

    it = 0
    while True:

        if elapsed >= budget_s - 0.15:       # a tiny safety margin

        if (new_c5 < cur_c5) or (rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-12))):
            cur_h, cur_c5 = new_h, new_c5
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h.copy()

        it += 1

    # 3. Final clean‑up (ensure feasibility)
    # ----------------------------------------------------------------------
    best_h = _project_sum_bounds(best_h, target_sum)
    best_c5 = _c5_value(best_h)
