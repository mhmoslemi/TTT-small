The strategy will optimize a step function defined by its symmetry (h(x) = h(2-x)) to minimize the maximum overlap integral. We use a 23-variable representation and SLSQP optimization with integral enforcement. The objective is the maximum overlap correlation multiplied by dx.

import numpy as np
from scipy.optimize import minimize

n_points = 45
dx = 2.0 / n_points

def run(seed=42, budget_s=1100, **kwargs):
    # Extract supplied functions if available, fallback to initialized defaults
    evaluate_erdos_solution = kwargs.get('evaluate_erdos_solution', lambda h: None)
    initial_h_values = kwargs.get('initial_h_values')

    if initial_h_values is not None:
        parent_available = True
    else:
        parent_available = False

    # Input validation for parent
    if parent_available:
        # Must be a 45-point array assuming symmetry
        assert len(initial_h_values) == n_points, "Parent solution must have 45 points"

    def build_symmetric_array(x):
        # x has 23 elements
        h_arr = np.zeros(n_points)
        h_arr[:23] = x
        # Symmetric part: indices23 to 44 are the mirror of indices0 to 21
        h_arr[23:] = x[:22][::-1]
        return h_arr

    def compute_C5(h):
        corr = np.correlate(h, 1-h, mode='full')
        return np.max(corr) * dx

    def objective(x):
        h_arr = build_symmetric_array(x)
        return compute_C5(h_arr)

    def constraint_sum(x):
        # Sum the entire function via symmetry and vectorization
        # The first 22 variables are doubled, last is single
        total = 2 * np.sum(x[:22]) + x[-1]
        # Normalize by dx to ensure integral=1
        return total - (n_points/2)

    # Bounds and initial guess
    bounds = [(0,1)] * 23
    initial_g = initial_h_values[:23] if parent_available else np.ones(23)*0.5

    success, x_opt, C5_opt, message = False, None, None, None
    
    try:
        # Use local optimization starting from parent or uniform initialization
        res = minimize(
            objective,
            initial_g,
            method='SLSQP',
            bounds=bounds,
            constraints={'type':'eq', 'fun':constraint_sum},
            options={'disp': True, 'maxiter': 5000}
        )
        success = res.success
        x_opt = res.x
        message = res.message
        C5_opt = compute_C5(build_symmetric_array(x_opt))
    except Exception as e:
        message = f"Optimization failed: {str(e)}"
        if parent_available:
            # Fallback: use parent
            h_fallback = initial_h_values
            C5_fallback = compute_C5(h_fallback)
            success, C5_opt, x_opt = True, C5_fallback, initial_h_values[:23]

    # Final validation and fallback preparation
    reconstructed = build_symmetric_array(x_opt)
    integral = np.sum(reconstructed) * dx
    if abs(integral - 1) > 1e-5 and parent_available:
        # If exactly violated and parent is available, return parent
        reconstructed = initial_h_values

    # Ensure valid bounds (optional but may not be necessary)
    if np.any(reconstructed < 0) or np.any(reconstructed > 1):
        if parent_available:
            reconstructed = initial_h_values

    # Evaluation handover to the Erdős specific function if provided
    if evaluate_erdos_solution:
        evaluate_erdos_solution(reconstructed)

    return reconstructed, C5_opt, n_points
