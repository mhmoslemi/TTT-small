import time, random, math
import numpy as np

def _project_sum_box(v: np.ndarray, total: float) -> np.ndarray:
    """Euclidean projection of v onto the set {0≤x≤1, sum x = total}."""
    x = np.clip(v, 0.0, 1.0)
    for _ in range(20):                     # water‑filling, converges fast
        excess = np.sum(x) - total
        if abs(excess) < 1e-12:
            break
        free = (x > 0.0) & (x < 1.0)
        cnt = np.count_nonzero(free)
        if cnt == 0:                         # all at a bound – distribute uniformly
            delta = excess / x.size
            x = np.clip(x - delta, 0.0, 1.0)
            continue
        delta = excess / cnt
        x[free] -= delta
        x = np.clip(x, 0.0, 1.0)
    # numerical clean‑up
    x = np.clip(x, 0.0, 1.0)
    return x

def _max_corr_raw(h: np.ndarray) -> tuple:
    """Return (max_raw, lag) of np.correlate(h,1-h,'full')."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    max_raw = float(corr[idx])
    lag = idx - (h.size - 1)
    return max_raw, lag

def _grad_for_lag(h: np.ndarray, lag: int) -> np.ndarray:
    """Sub‑gradient of the raw correlation corresponding to `lag`."""
    n = h.size
    start = max(0, -lag)
    end = min(n - 1, n - 1 - lag)
    i = np.arange(start, end + 1, dtype=int)
    j = i + lag
    grad = np.zeros(n, dtype=float)
    grad[i] += 1.0
    grad[i] -= h[j]
    grad[j] -= h[i]
    return grad

def _c5_from_raw(max_raw: float, n: int) -> float:
    """Scale raw correlation to the integral value."""
    return max_raw * (2.0 / n)
