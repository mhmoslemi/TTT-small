def gradient_corr_shift(h, shift):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if shift >= 0:
        i = np.arange(n - shift)
        j = i + shift
    else:
        # shift negative => shift = -d
        d = -shift
        i = np.arange(n - d)
        j = i + shift  # j < i
    # contributions
    g[i] += 1 - h[j]
    g[j] -= h[i]
    return g

import time
import numpy as np

def evaluate_c5(h, dx):
    """Maximum overlap integral for step function h."""
    # correlation of h with (1-h), linear (non‑circular) convolution
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * dx

def gradient_corr_shift(h, shift):
    """Gradient of the correlation term for a given integer shift."""
    n = h.shape[0]
    g = np.zeros_like(h)
    if shift >= 0:
        i = np.arange(n - shift)
        j = i + shift
    else:
        d = -shift
        i = np.arange(n - d)
        j = i + shift          # j = i - d
    # contribution from the first factor h[i] * (1 - h[j])
    g[i] += 1.0 - h[j]
    # contribution from the second factor -h[i] * h[j]
    g[j] -= h[i]
    return g

def project_onto_box_sum(v, target, lo=0.0, hi=1.0, eps=1e-12, max_iter=60):
    """Projection of v onto {x: lo<=x<=hi, sum(x)=target}."""
    # binary search on lambda for x = clip(v - λ, lo, hi)
    low, high = -10.0, 10.0
    for _ in range(max_iter):
        lam = (low + high) * 0.5
        x = np.clip(v - lam, lo, hi)
        s = x.sum()
        if s > target:
            low = lam
        else:
            high = lam
    lam = (low + high) * 0.5
    x = np.clip(v - lam, lo, hi)
    diff = target - x.sum()
    if abs(diff) > eps:
        free = (x > lo + eps) & (x < hi - eps)
        if free.any():
            add = diff / free.sum()
            x[free] += add
            x = np.clip(x, lo, hi)
    return x

def gradient_descent(start_h, target_sum, dx, rng, max_iters, time_deadline):
    """Local descent on the worst‑shift objective."""
    h = start_h.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h, dx)
    step = 0.02
    n = h.shape[0]

    for it in range(max_iters):
        if time.time() > time_deadline:
            break
        # worst shift
        corr = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)

        g = gradient_corr_shift(h, shift)

        # trial step
        trial = h - step * g
        trial = project_onto_box_sum(trial, target_sum)
        val = evaluate_c5(trial, dx)

        if val < best_val - 1e-12:
            h = trial
            best_h = trial
            best_val = val
            step = min(step * 1.05, 0.5)
        else:
            step *= 0.5
            if step < 1e-7:
                # random perturbation to escape stagnation
                pert = rng.normal(scale=0.03, size=n)
                trial = h + pert
                trial = project_onto_box_sum(trial, target_sum)
                val = evaluate_c5(trial, dx)
                if val < best_val:
                    h = trial
                    best_h = trial
                    best_val = val
                    step = 0.02

        # occasional small random tweak
        if it % 250 == 0:
            cand = h + rng.normal(scale=0.015, size=n)
            cand = project_onto_box_sum(cand, target_sum)
            cand_val = evaluate_c5(cand, dx)
            if cand_val < best_val:
                h = cand
                best_h = cand
                best_val = cand_val
                step = 0.02

    return best_h, best_val
