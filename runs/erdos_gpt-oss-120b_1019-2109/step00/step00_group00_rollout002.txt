import numpy as np, time, math

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def neighbor_sum(h, d):
    N = h.shape[0]
    s = np.zeros(N, dtype=h.dtype)
    if d >= 0:
        limit = N - d
        if limit > 0:
            s[:limit] += h[d:]
            s[d:] += h[:limit]
    else:
        d = -d
        limit = N - d
        if limit > 0:
            s[d:] += h[:limit]
            s[:limit] += h[d:]
    return s

def random_feasible(N, rng):
    # start from uniform half
    h = np.full(N, 0.5, dtype=float)
    # perform many random pair adjustments
    for _ in range(2000):
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        # random signed delta up to +-0.2
        delta = (rng.random() - 0.5) * 0.4
        # bound delta to keep within [0,1]
        max_up = 1.0 - h[i]
        max_down = h[j]
        # delta positive adds to i, subtract from j
        # if delta negative, we swap roles
        if delta > 0:
            delta = min(delta, max_up, max_down)
            if delta <= 0:
                continue
            h[i] += delta
            h[j] -= delta
        else:
            delta = -delta
            max_up2 = 1.0 - h[j]
            max_down2 = h[i]
            delta = min(delta, max_up2, max_down2)
            if delta <= 0:
                continue
            h[j] += delta
            h[i] -= delta
    return h

import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1‑h(x+k)) dx for the discrete step function."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def neighbor_sum(h: np.ndarray, d: int) -> np.ndarray:
    """For a given shift d, return for each position i the sum h[i+d]+h[i‑d]
    with out‑of‑range values taken as 0."""
    N = h.shape[0]
    s = np.zeros(N, dtype=h.dtype)
    if d >= 0:
        lim = N - d
        if lim > 0:
            s[:lim] += h[d:]
            s[d:] += h[:lim]
    else:
        d = -d
        lim = N - d
        if lim > 0:
            s[d:] += h[:lim]
            s[:lim] += h[d:]
    return s


def random_feasible(N: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random vector in [0,1]^N with sum N/2 using pairwise swaps."""
    h = np.full(N, 0.5, dtype=float)
    for _ in range(2000):
        i, j = rng.integers(N, size=2)
        if i == j:
            continue
        delta = (rng.random() - 0.5) * 0.4          # signed proposal in [-0.2,0.2]
        if delta > 0:
            max_up = 1.0 - h[i]
            max_down = h[j]
            delta = min(delta, max_up, max_down)
            if delta <= 0:
                continue
            h[i] += delta
            h[j] -= delta
        else:
            delta = -delta
            max_up = 1.0 - h[j]
            max_down = h[i]
            delta = min(delta, max_up, max_down)
            if delta <= 0:
                continue
            h[j] += delta
            h[i] -= delta
    return h


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a small C5 value."""
    start = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise
    if "initial_h_values" in globals():
        h0 = np.asarray(initial_h_values, dtype=float).copy()
        N = h0.shape[0]
        target_sum = N / 2.0
        diff = target_sum - h0.sum()
        if abs(diff) > 1e-12:                     # tiny correction if needed
            h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)
    else:
        N = 45
        h0 = np.full(N, 0.5, dtype=float)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    target = 0.3808                                     # stop when we beat this

    # ------------------------------------------------------------------
    # main optimisation loop
    while True:
        if time.time() - start > budget_s - 0.5:
            break

        # ------------------------------------------------------------------
        # pick a start point for this trial
        if rng.random() < 0.2:
            h = random_feasible(N, rng)
        else:
            h = best_h.copy()
            # a few tiny random swaps from the current best
            for _ in range(5):
                i, j = rng.integers(N, size=2)
                if i == j:
                    continue
                max_up = 1.0 - h[i]
                max_down = h[j]
                if max_up <= 0 or max_down <= 0:
                    continue
                delta = min(0.05, max_up, max_down) * rng.random()
                h[i] += delta
                h[j] -= delta

        # ------------------------------------------------------------------
        # local improvement on the chosen start
        step = 0.1
        max_inner = 4000
        for it in range(max_inner):

                best_c5 = c5
                best_h = h.copy()
                if best_c5 < target:

            diff -= inc
