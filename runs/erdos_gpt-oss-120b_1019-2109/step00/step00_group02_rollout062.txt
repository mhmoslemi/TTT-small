import time
import random
import math
import numpy as np

def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Heuristic search for a step function h on [0,2] (discretized) that minimizes
        C5 = max_k ∫ h(x) (1 - h(x+k)) dx .
    The evaluation is performed by the pre‑imported function `evaluate_erdos_solution`.
    The algorithm combines random binary constructions (exactly half the entries are 1)
    with a simple simulated‑annealing / hill‑climbing loop and a short fractional
    refinement stage.
    """
    # --------------------------------------------------------------
    # basic setup
    random.seed(seed)
    np.random.seed(seed)

    # helpers -------------------------------------------------------
    def eval_h(arr):
        """evaluate a candidate using the provided evaluator."""
        return float(evaluate_erdos_solution(arr))

    def random_binary_vec(n):
        """binary vector of length n with exactly n/2 ones (n must be even)."""
        m = n // 2
        idx = np.arange(n)
        np.random.shuffle(idx)
        h = np.zeros(n, dtype=float)
        h[idx[:m]] = 1.0
        return h

    # initial solution (if supplied) --------------------------------
    best_h = None
    best_c = float('inf')
    best_n = None

    init = globals().get('initial_h_values')
    if init is not None:
        h0 = np.asarray(init, dtype=float)
        c0 = eval_h(h0)
        best_h, best_c, best_n = h0, c0, len(h0)

    # candidate sizes ------------------------------------------------
    # keep sizes modest (fast evaluation) but allow a few different resolutions
    n_candidates = [44, 48, 52, 60, 80, 100, 120]
    n_candidates = [n for n in n_candidates if n % 2 == 0]

    start_time = time.time()
    time_limit = budget_s - 0.5           # safety margin

    # --------------------------------------------------------------
    # outer search loop (simulated annealing on binary vectors)
    while time.time() - start_time < time_limit:
        n = random.choice(n_candidates)

        # initialise current state
        if best_h is not None and len(best_h) == n:
            cur = best_h.copy()
        else:
            cur = random_binary_vec(n)

        cur_c = eval_h(cur)

        # inner SA parameters
        max_inner = 3000
        T_start, T_end = 0.3, 0.001

        for it in range(max_inner):
            # abort if out of time
            if time.time() - start_time >= time_limit:
                break

            # temperature schedule (linear)
            prog = it / max_inner
            T = T_start * (1 - prog) + T_end * prog

            # propose a swap between a 1 and a 0
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)

            cand = cur.copy()
            cand[i] = 0.0
            cand[j] = 1.0
            cand_c = eval_h(cand)

            delta = cand_c - cur_c
            if delta < 0 or random.random() < math.exp(-delta / max(T, 1e-12)):
                cur, cur_c = cand, cand_c
                if cand_c < best_c:
                    best_h, best_c, best_n = cand.copy(), cand_c, n

            # occasional random restart to diversify
            if (it % 600 == 0) and (random.random() < 0.05):
                cur = random_binary_vec(n)
                cur_c = eval_h(cur)

        # end of inner loop, check overall time
        if time.time() - start_time >= time_limit:
            break

    # --------------------------------------------------------------
    # fractional refinement (small mass transfers) -----------------
    if best_h is not None:
        h_cur = best_h.copy()
        cur_c = best_c
        n = best_n
        max_frac = 5000
        for it in range(max_frac):
            if time.time() - start_time >= time_limit:
                break
            i, j = np.random.choice(n, 2, replace=False)

            # how much can be moved from i to j while staying in [0,1]?
            max_minus = h_cur[i]                 # amount we can subtract
            max_plus = 1.0 - h_cur[j]            # amount we can add
            if max_minus == 0 and max_plus == 0:
                continue

            eps = (random.random() - 0.5) * 0.2   # try a change up to ±0.1
            eps = max(-max_minus, min(eps, max_plus))
            if abs(eps) < 1e-9:
                continue

            cand = h_cur.copy()
            cand[i] -= eps
            cand[j] += eps
            cand_c = eval_h(cand)

            if cand_c < cur_c:
                h_cur, cur_c = cand, cand_c
                if cand_c < best_c:
                    best_h, best_c, best_n = cand.copy(), cand_c, n

    # --------------------------------------------------------------
    # final fallback (should never happen)
    if best_h is None:
        n = 100
        best_h = np.full(n, 0.5, dtype=float)
        best_c = eval_h(best_h)
        best_n = n

    return best_h, float(best_c), int(best_n)
