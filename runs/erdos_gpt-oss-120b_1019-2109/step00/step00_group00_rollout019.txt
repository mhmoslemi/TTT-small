import time
import numpy as np

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] while enforcing sum(v)=target.
    Finds λ such that sum(clip(v+λ,0,1)) = target."""
    lo = -np.max(v)                # makes all entries ≤0
    hi = 1.0 - np.min(v)           # makes all entries ≥1
    # binary search for λ
    for _ in range(60):
        mid = (lo + hi) * 0.5
        shifted = np.clip(v + mid, 0.0, 1.0)
        s = shifted.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    λ = (lo + hi) * 0.5
    return np.clip(v + λ, 0.0, 1.0)


def _c5_value(h: np.ndarray):
    """Return (C5, full correlation array) for a step vector h."""
    N = h.shape[0]
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr


def _sgd_step(h: np.ndarray, shift: int, eta: float, target_sum: float) -> np.ndarray:
    """One projected‑gradient step aiming to reduce the worst‑shift C5."""
    N = h.shape[0]
    dx = 2.0 / N
    step = eta * dx                     # factor multiplying (neighbor‑sum‑1)
    neighbor = np.zeros_like(h)

    if shift > 0:
        neighbor[: N - shift] += h[shift:]
        neighbor[shift:] += h[: N - shift]
    elif shift < 0:
        k = -shift
        neighbor[k:] += h[: N - k]
        neighbor[: N - k] += h[k:]
    else:                                # shift==0 never the worst shift in practice
        neighbor = 2.0 * h

    h_new = h + step * (neighbor - 1.0)   # gradient ascent on overlap
    return _project_to_sum(h_new, target_sum)

        start_vectors.append(_project_to_sum(rnd, N / 2.0))

    # deterministic “ends‑filled” pattern
    quarter = N // 4
    ends = np.concatenate([np.ones(quarter),
                           np.zeros(N - 2 * quarter),
                           np.ones(quarter)])
    start_vectors.append(ends)

    target_sum = N / 2.0
    best_h = None
    best_val = float("inf")
    eta_initial = 0.8                     # initial learning rate factor

    # ----- main optimisation loop (multiple restarts) -----
    for h_start in start_vectors:
        h = h_start.copy()
        eta = eta_initial
        # a few thousand SGD steps per start; stop early if out of time
        for _ in range(2500):
            if time.time() - start_time > budget_s - 2.0:
                break

            val, corr = _c5_value(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()

            # worst shift (largest correlation)
            idx_max = int(np.argmax(corr))
            shift = idx_max - (N - 1)      # shift in index units, can be negative

            if shift == 0:                # should not happen, but keep moving a bit
                h = _project_to_sum(
                    np.clip(h + (np.random.rand(N) - 0.5) * 0.01, 0.0, 1.0),
                    target_sum)
                continue

            # one SGD step directed at the current worst shift
            h = _sgd_step(h, shift, eta, target_sum)

            # slowly decay the learning rate
            eta = max(eta * 0.9995, 0.05)

        if time.time() - start_time > budget_s - 2.0:
            break

    # final evaluation (just in case the best was never updated)
    if best_h is None:
        # fallback – use the first start vector
        best_h = start_vectors[0].copy()
        best_val, _ = _c5_value(best_h)

    # guarantee feasibility one last time
    best_h = _project_to_sum(best_h, target_sum)

    return best_h, float(best_val), N
