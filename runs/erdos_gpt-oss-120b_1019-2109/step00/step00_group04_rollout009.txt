def project_to_feasible(x, S):
    lo = min(x) - 1.0 # ensure sum <= S, but we need to guarantee range.
    hi = max(x) + 1.0 
    for _ in range(60):
        lam = (lo+hi)/2.0
        h = np.clip(x - lam, 0.0, 1.0)
        cur = h.sum()
        if cur > S:
            lo = lam
        else:
            hi = lam
    h = np.clip(x - (lo+hi)/2.0, 0.0, 1.0)
    # final adjust to correct sum due to numeric rounding: compute diff = S - sum.
    # Add diff to entries not at bounds.
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        # distribute diff equally among non-bound entries
        free = np.where((h>0+1e-12) & (h<1-1e-12))[0]
        if free.size > 0:
            h[free] += diff / free.size
            # clip again to be safe
            h = np.clip(h, 0.0, 1.0)
    return h

def project_to_feasible(x, S):
    # x: np.ndarray shape (n,)
    # Return h in [0,1] with sum S
    # Use water-filling binary search
    lo = -np.max(x) - 1.0  # ensure sum > S
    hi = np.max(x) + 1.0   # ensure sum < S
    for _ in range(60):
        lam = (lo + hi) / 2.0
        h = np.clip(x - lam, 0.0, 1.0)
        cur = h.sum()
        if cur > S:
            lo = lam
        else:
            hi = lam
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Correct small residual
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        # adjust among non-bound indices
        free = np.where((h > 1e-9) & (h < 1 - 1e-9))[0]
        if free.size > 0:
            # distribute equally
            h[free] += diff / free.size
            # clip again
            h = np.clip(h, 0.0, 1.0)
    return h

def project_to_simplex_box(x, S):
    # x: np.ndarray
    # Return projection onto [0,1] and sum = S
    lo = -np.max(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        h = np.clip(x - lam, 0.0, 1.0)
        cur = h.sum()
        if cur > S:
            lo = lam
        else:
            hi = lam
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

        if free.size > 0:

            h = np.clip(h, 0.0, 1.0)
    return h

import time
import numpy as np

# ---------- helper functions (top level) ----------
def project_to_simplex_box(x: np.ndarray, S: float) -> np.ndarray:
    """
    Orthogonal projection of x onto the box-constrained simplex:
        0 <= h_i <= 1,   sum_i h_i = S
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = -np.max(x) - 1.0   # sum will be > S
    hi = np.max(x) + 1.0    # sum will be < S
    for _ in range(60):
        lam = (lo + hi) * 0.5
        h = np.clip(x - lam, 0.0, 1.0)
        cur = h.sum()
        if cur > S:
            lo = lam
        else:
            hi = lam
    h = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction to hit the target sum exactly
    diff = S - h.sum()
    if abs(diff) > 1e-12:
        free = np.where((h > 1e-9) & (h < 1 - 1e-9))[0]
        if free.size:
            h[free] += diff / free.size
            h = np.clip(h, 0.0, 1.0)
    return h


def generate_cos_shape(n: int) -> np.ndarray:
    """h(x) = (1 + cos(pi * x)) / 2 sampled at midpoints."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    return (1.0 + np.cos(np.pi * x)) * 0.5


def generate_triangular_shape(n: int) -> np.ndarray:
    """h(x) = max(0, 1 - |x-1|) sampled at midpoints."""
    dx = 2.0 / n
    x = (np.arange(n) + 0.5) * dx
    h = 1.0 - np.abs(x - 1.0)
    return np.clip(h, 0.0, 1.0)


def compute_c5(h: np.ndarray) -> float:
    """Exact evaluation of the overlap constant for a discrete step function."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)
