def project(y, target_sum):
    # y is numpy array.
    lo = -1.0  # lambda lower bound (shift)
    hi = 1.0   # maybe bigger
    # Initial lo/hi such that sum of clipped(y - λ) > target (or < target). We need monotonic decreasing function of λ.
    # compute sum when λ = -inf => all clipped to 1 => sum = n. When λ = +inf => all clipped to 0 => sum=0.
    # So we can start with lo = -1e6, hi = 1e6.
    lo = -1e6
    hi = 1e6
    for _ in range(60):
        mid = (lo+hi)/2
        p = np.clip(y - mid, 0.0, 1.0)
        s = p.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo+hi)/2, 0.0, 1.0)

import time
import numpy as np
from scipy.interpolate import interp1d

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()

    best_h = None
    best_c5 = np.inf

    # Helper: projection onto [0,1]^n with sum = target_sum
    def make_projector(target_sum):
        def project(y):
            lo = y.min() - 1.0
            hi = y.max()
            for _ in range(50):
                mid = (lo + hi) * 0.5
                p = np.clip(y - mid, 0.0, 1.0)
                s = p.sum()
                if s > target_sum:
                    lo = mid
                else:
                    hi = mid
            return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
        return project

    # Helper: autocorrelation based C5 evaluator
    def make_c5_computer(dx):
        def compute(h):
            conv = np.correlate(h, h, mode='full')
            vals = (h.sum() - conv) * dx
            return vals.max()
        return compute

    # Helper: gradient of the overlap term for a given lag
    def grad_for_lag(h, lag):
        n = h.shape[0]
        g = np.zeros_like(h)
        if lag >= 0:
            g[: n - lag] += h[lag:]
            g[lag:] += h[: n - lag]
        else:
            k = -lag
            g[k:] += h[: n - k]
            g[: n - k] += h[k:]
        return g

    # Candidate sample sizes (prefer modest <1000)
    cand_sizes = []
    if 'initial_h_values' in globals():
        cand_sizes.append(len(initial_h_values))
    cand_sizes.extend([144, 200, 250, 300])
    cand_sizes = sorted(set(cand_sizes))

    for n in cand_sizes:
        # stop if out of time
        if time.time() - start_time > budget_s - 5:
            break

        dx = 2.0 / n
        target_sum = n / 2.0
        proj = make_projector(target_sum)
        compute_c5 = make_c5_computer(dx)

        # Prepare starting vectors
        starts = []

        # use parent if compatible
        if 'initial_h_values' in globals() and len(initial_h_values) == n:
            starts.append(proj(np.clip(initial_h_values.astype(float), 0.0, 1.0)))
        elif 'initial_h_values' in globals():
            # linear interpolation to required length
            src_x = np.linspace(0.0, 2.0, len(initial_h_values), endpoint=False)
            tgt_x = np.linspace(0.0, 2.0, n, endpoint=False)
            f = interp1d(src_x, initial_h_values, kind='linear', fill_value='extrapolate')
            interp_h = np.clip(f(tgt_x), 0.0, 1.0)
            starts.append(proj(interp_h))

        # a few random starts
        for _ in range(5):

            starts.append(proj(rnd))

        # gradient descent parameters
        step0 = 0.5

            h = h0.copy()
            cur_c5 = compute_c5(h)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()

            for it in range(max_iters):
                if time.time() - start_time > budget_s - 2:
                    break

                # correlation and worst lag
                conv = np.correlate(h, h, mode='full')
                vals = (h.sum() - conv) * dx
                idx = int(np.argmax(vals))
                cur_c5 = vals[idx]
                lag = idx - (n - 1)
