def run(seed=42, budget_s=1100, **kwargs):
    import time, math, numpy as np, random
    try:
        import cvxpy as cp
    except Exception:
        cp = None

    # Determine n
    try:
        init_arr = np.array(initial_h_values, dtype=float)
        n = init_arr.shape[0]
    except Exception:
        n = 80
        init_arr = None

    target_sum = n / 2
    dx = 2.0 / n

    # function to compute C5
    def compute_c5(vec):
        return np.max(np.correlate(vec, 1 - vec, mode='full') * dx)

    # Shortcut: if MIP possible, try solve.
    best_h = None
    best_score = None
    start = time.time()
    remaining = budget_s - (time.time() - start)
    if cp is not None and remaining > 10:
        # attempt MIP for small time budget
        try:
            # Setup variables
            y = cp.Variable(n, boolean=True)
            r = cp.Variable(integer=True)
            # Z matrix for products
            Z = cp.Variable((n, n), boolean=True)
            constraints = [cp.sum(y) == target_sum,
                           r >= 0,
                           r <= target_sum]  # r <= m
            # diagonal
            for i in range(n):
                constraints.append(Z[i, i] == y[i])
            # product constraints for i<j
            for i in range(n):
                for j in range(i+1, n):
                    constraints.append(Z[i, j] <= y[i])
                    constraints.append(Z[i, j] <= y[j])
                    constraints.append(Z[i, j] >= y[i] + y[j] - 1)
            # overlap constraints
            for k in range(1, n):
                # sum of Z[i,i+k] for i valid
                constraints.append(cp.sum(Z[:n-k, k:]) >= target_sum - r)
            prob = cp.Problem(cp.Minimize(r), constraints)
            # time limit in milliseconds for GLPK
            glp_seconds = max(1, int((budget_s - (time.time() - start)) * 0.9))
            # Choose GLPK_MI if available
            prob.solve(solver=cp.GLPK_MI, glpk={'msg_lev': 'GLP_MSG_OFF', 'tm_lim': glp_seconds * 1000})
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                y_val = np.rint(y.value).astype(float)  # round, should be 0/1
                # Safety: adjust sum if off by 1
                cur_sum = y_val.sum()
                if cur_sum != target_sum:
                    diff = int(target_sum - cur_sum)
                    # adjust by flipping some bits
                    indices = np.where(y_val == 0)[0] if diff > 0 else np.where(y_val == 1)[0]
                    if len(indices) >= abs(diff):
                        pick = rng.choice(indices, size=abs(diff), replace=False)
                        y_val[pick] = 1 if diff > 0 else 0
                best_h = y_val
                best_score = compute_c5(best_h)
            # else fallback
        except Exception as e:
            # fallback to heuristic
            pass

    # If MIP didn't yield result
    if best_h is None:
        # use heuristic (simulated annealing)
        rng = np.random.default_rng(seed)
        # initialize best
        if init_arr is not None:
            # ensure feasibility
            h0 = init_arr.copy()
            # clip and adjust sum
            h0 = np.clip(h0, 0.0, 1.0)
            s0 = np.sum(h0)
            if not np.isclose(s0, target_sum):
                # simple scaling
                if s0 > 0:
                    h0 = h0 * (target_sum / s0)
                else:
                    h0 = np.full(n, target_sum / n)
                h0 = np.clip(h0, 0.0, 1.0)
                # final fix
                diff = target_sum - np.sum(h0)
                if abs(diff) > 1e-8:
                    # adjust some non-bounded element
                    idx = np.where((h0 > 0) & (h0 < 1))[0]
                    if idx.size > 0:
                        h0[idx[0]] += diff
                        h0 = np.clip(h0, 0.0, 1.0)
            cur_h = h0.copy()
        else:
            # start with binary random
            cur_h = np.zeros(n)
            cur_h[:int(target_sum)] = 1.0
            rng.shuffle(cur_h)
        cur_score = compute_c5(cur_h)
        best_h = cur_h.copy()
        best_score = cur_score

        # Simulated annealing
        it = 0
        no_improve = 0
        while True:
            elapsed = time.time() - start
            if elapsed > budget_s * 0.95:
                break
            it += 1
            # adaptive temperature
            t_frac = elapsed / budget_s
            T = 0.5 * (1 - t_frac) + 1e-6
            step = 0.2 * (1 - t_frac) + 1e-4
            # propose swap
            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)
            # compute max delta allowed
            max_inc_i = 1.0 - cur_h[i]
            max_dec_i = cur_h[i]
            max_inc_j = 1.0 - cur_h[j]
            max_dec_j = cur_h[j]
            # delta from i to j
            delta = (rng.random() * 2 - 1) * step
            if delta > 0:
                delta = min(delta, max_inc_i, max_dec_j)
            else:
                delta = -min(-delta, max_inc_j, max_dec_i)
            if abs(delta) < 1e-12:
                continue
            new_h = cur_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
            new_score = compute_c5(new_h)
            # acceptance
            if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / T):
                cur_h = new_h
                cur_score = new_score
                no_improve = 0
                if new_score < best_score:
                    best_h = new_h.copy()
                    best_score = new_score
            else:
                no_improve += 1
            # occasional random restart
            if no_improve > 200000:
                # random restart from best with jitter
                cur_h = best_h.copy()
                # add small jitter
                jitter = rng.normal(scale=0.05, size=n)
                cur_h = np.clip(cur_h + jitter, 0.0, 1.0)
                # renorm to target sum
                s = np.sum(cur_h)
                if s != 0:
                    cur_h = cur_h * (target_sum / s)
                cur_h = np.clip(cur_h, 0.0, 1.0)
                cur_score = compute_c5(cur_h)
                no_improve = 0

        # finalize best_h
        # ensure constraints exactly
        h = best_h.copy()
        h = np.clip(h, 0.0, 1.0)
        s = np.sum(h)
        if not np.isclose(s, target_sum, atol=1e-8):
            # adjust
            diff = target_sum - s
            # add diff to a free index
            idxs = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
            if idxs.size == 0:
                idxs = np.arange(n)
            h[idxs[0]] += diff
            h = np.clip(h, 0.0, 1.0)
        best_h = h
        best_score = compute_c5(best_h)

    # compute final score using provided evaluator if exists

        c5_val = float(evaluate_erdos_solution(best_h))
    except Exception:
        c5_val = float(best_score)

    return (best_h.tolist(), c5_val, n)

                        y_val[idx[:abs(diff)]] = 1 if diff > 0 else 0
                best_h = y_val
                best_score = compute_c5(best_h)
        except Exception:
            # any problem – simply fall back to heuristic below
            best_h = None

    # ------------------------------------------------------------------
    # Heuristic (simulated annealing / swap moves) – used if MIP failed
    # ------------------------------------------------------------------
    if best_h is None:
        # start point
        if init_arr is not None:
            cur = init_arr.copy()
            cur = np.clip(cur, 0.0, 1.0)
            s = cur.sum()
            if not np.isclose(s, target_sum):
                if s > 0:
                    cur *= target_sum / s
                else:
                    cur[:] = target_sum / n
                cur = np.clip(cur, 0.0, 1.0)
        else:
            cur = np.zeros(n)
            cur[:target_sum] = 1.0
            rng.shuffle(cur)

        cur_score = compute_c5(cur)
        best_h = cur.copy()
        best_score = cur_score

        it = 0
        no_imp = 0
        while True:
            elapsed = time.time() - start
            if elapsed > budget_s * 0.95:
                break
            it += 1
            # temperature and step size schedules
            frac = elapsed / budget_s
            T = 0.5 * (1.0 - frac) + 1e-6
            step = 0.2 * (1.0 - frac) + 1e-4

            i, j = rng.integers(0, n, size=2)
            while i == j:
                j = rng.integers(0, n)

            max_inc_i = 1.0 - cur[i]
            max_dec_i = cur[i]
            max_inc_j = 1.0 - cur[j]
            max_dec_j = cur[j]

            delta = (rng.random() * 2.0 - 1.0) * step
            if delta > 0:
                delta = min(delta, max_inc_i, max_dec_j)
            else:
                delta = -min(-delta, max_inc_j, max_dec_i)
            if abs(delta) < 1e-12:
                continue

            new = cur.copy()
            new[i] += delta
            new[j] -= delta
            new_score = compute_c5(new)

            if new_score < cur_score or rng.random() < math.exp((cur_score - new_score) / T):
                cur = new
                cur_score = new_score
                no_imp = 0
                if new_score < best_score:
                    best_h = new.copy()
                    best_score = new_score
            else:
                no_imp += 1

            # occasional random restart with a jitter around current best
            if no_imp > 200_000:
                cur = best_h.copy()
                jitter = rng.normal(scale=0.05, size=n)
                cur = np.clip(cur + jitter, 0.0, 1.0)
                cur = cur * (target_sum / cur.sum())
                cur = np.clip(cur, 0.0, 1.0)
                cur_score = compute_c5(cur)
                no_imp = 0

        # final feasibility tweak
        h = np.clip(best_h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) > 1e-8:
            free = np.where((h > 1e-12) & (h < 1 - 1e-12))[0]
            if free.size == 0:
                free = np.arange(n)
            h[free[0]] += diff
            h = np.clip(h, 0.0, 1.0)
        best_h = h
        best_score = compute_c5(best_h)

    # ------------------------------------------------------------------
    # final evaluation (use the supplied evaluator if present)
    # ------------------------------------------------------------------
    try:
        c5_val = float(evaluate_erdos_solution(best_h))
    except Exception:
        c5_val = float(best_score)

    return (best_h.tolist(), c5_val, n)
