def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
```

import numpy as np
import time
import math
import random

def compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k))dx using FFT‑based correlation."""
    n = h.shape[0]
    dx = 2.0 / n
    g = (1.0 - h)[::-1]                     # reversed complement for correlation
    m = 2 * n - 1
    N = 1 << (m - 1).bit_length()          # next power of two ≥ m
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(g, N)
    conv = np.fft.irfft(H * G, N)[:m]      # linear correlation (full mode)
    return float(np.max(conv) * dx)

def adjust_sum(h: np.ndarray, target: float) -> np.ndarray:
    """Small projection of h onto the hyperplane sum(h)=target, staying in [0,1]."""
    cur = h.sum()
    diff = target - cur
    if abs(diff) < 1e-12:
        return h
    n = h.shape[0]
    if diff > 0:
        cand = np.where(h < 1.0 - 1e-12)[0]
        np.random.shuffle(cand)
        for idx in cand:
            room = 1.0 - h[idx]
            inc = min(diff, room)
            h[idx] += inc
            diff -= inc
            if diff <= 1e-12:
                break
    else:
        cand = np.where(h > 1e-12)[0]
        np.random.shuffle(cand)
        for idx in cand:
            room = h[idx]
            dec = min(-diff, room)
            h[idx] -= dec
            diff += dec
            if diff >= -1e-12:
                break
    # tiny numerical drift is acceptable
    return h

def perturb(h: np.ndarray, target: float) -> np.ndarray:
    """Add a tiny random noise to h and re‑project to the feasible set."""
    n = h.shape[0]
    noise = (np.random.rand(n) - 0.5) * 0.02  # max ±0.01
    h2 = h + noise
    np.clip(h2, 0.0, 1.0, out=h2)
    h2 = adjust_sum(h2, target)
    return h2

def anneal(start_h: np.ndarray, max_seconds: float) -> tuple:
    """
    Simple simulated‑annealing on the vector h.
    Returns (best_h, best_cost) found within the allotted time.
    """
    n = start_h.shape[0]
    target_sum = n / 2.0
    h = adjust_sum(start_h.copy(), target_sum)
    cur_cost = compute_c5(h)
    best_h = h.copy()
    best_cost = cur_cost

    start = time.time()
    T0 = 0.5
    T = T0
    decay = 0.9995

    while time.time() - start < max_seconds:
        # choose a source (decrease) and a destination (increase)
        i = np.random.randint(n)
        j = np.random.randint(n)
        while i == j:
            j = np.random.randint(n)

        max_sub = h[i]                # we can subtract at most this
        max_add = 1.0 - h[j]          # we can add at most this
        max_delta = min(max_sub, max_add)
        if max_delta <= 0.0:
            continue

        # delta magnitude shrinks with temperature
        delta = max_delta * np.random.rand() * max(0.1, T / T0)

        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta

        new_cost = compute_c5(h_new)

        if new_cost < cur_cost or np.random.rand() < math.exp(-(new_cost - cur_cost) / (T + 1e-12)):
            h = h_new
            cur_cost = new_cost
            if cur_cost < best_cost:
                best_h = h.copy()
                best_cost = cur_cost

        T = max(T * decay, 1e-6)

    # final safety projection (tiny numerical drift)
    best_h = adjust_sum(best_h, target_sum)
    best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_cost
