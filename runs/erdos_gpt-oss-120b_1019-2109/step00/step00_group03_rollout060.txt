import numpy as np, time

def project_to_sum_and_bounds(z, target):
    # binary search for lambda
    lo, hi = -1.0, 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        x = np.clip(z + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(z + lo, 0.0, 1.0)
    # adjust for small numerical errors
    diff = target - x.sum()
    if abs(diff) > 1e-9:
        # distribute diff to non-bound entries proportionally
        free = (x > 0) & (x < 1)
        if np.any(free):
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(corr))

def min_corr_sum(h):
    corr = np.correlate(h, h, mode='full')
    return np.min(corr)

def compute_c5(h):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(corr))

def gradient_cross_corr(h, shift):
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        ov = n - shift
        if ov <= 0:
            return g
        g[:ov] += 1.0 - h[shift:]
        g[shift:] -= h[:ov]
    else:
        d = -shift
        ov = n - d
        if ov <= 0:
            return g
        g[d:] += 1.0 - h[:ov]
        g[:ov] -= h[d:]
    return g

import time
import numpy as np

def _project_box_sum(z: np.ndarray, target: float) -> np.ndarray:
    """Project z onto the box [0,1]^n and the hyperplane sum(z)=target."""
    # binary search for lambda such that sum(clip(z+lam,0,1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        x = np.clip(z + lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = lam
        else:
            lo = lam
    x = np.clip(z + lo, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            x[free] += diff / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x


def _c5_value(h: np.ndarray) -> float:
    """Return the Erdős C5 value for a step vector h."""
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(corr.max())


def _grad_worst_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient (w.r.t. h) of the cross‑correlation term that attains the maximum."""
    n = h.size
    g = np.zeros_like(h)
    if shift >= 0:
        ov = n - shift
        if ov <= 0:
            return g
        # derivative of Σ h_i term (+1) and of -h_i h_{i+shift} term (−h_{i+shift})
        g[:ov] += 1.0 - h[shift:]
        # counterpart contribution for the second factor in the product
        g[shift:] -= h[:ov]
    else:
        d = -shift
        ov = n - d
        if ov <= 0:
            return g
        # now the sum runs over indices i=d..n-1
        g[d:] += 1.0 - h[:ov]
        g[:ov] -= h[d:]
    return g


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Produce a step function h on [0,2] with ∫h = 1 that (approximately) minimizes
    C5 = max_k ∫ h(x)·(1−h(x+k)) dx.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start = time.time()

    # ------------------------------------------------------------------
    # Choose discretisation size
    if "initial_h_values" in globals():
        h0 = np.array(initial_h_values, dtype=float)
        n = h0.size
        target = n / 2.0
        h = np.clip(h0, 0.0, 1.0)
        h = _project_box_sum(h, target)
    else:
        n = 200                     # default resolution
        target = n / 2.0
        h = np.full(n, 0.5, dtype=float)
        h = _project_box_sum(h, target)

    best_h = h.copy()
    best_c5 = _c5_value(h)

    # early exit if the supplied construction already beats the record
    if best_c5 <= 0.38085:
        return best_h, best_c5, n

    # ------------------------------------------------------------------
    # Sub‑gradient descent on the worst shift
    step = 0.12                     # initial step length
    decay = 0.9995                  # step decay factor
    it = 0
    zero_idx = n - 1

    while time.time() - start < budget_s * 0.94:
        # evaluate all shifts
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst_idx = int(np.argmax(corr))
        shift = worst_idx - zero_idx

        # gradient of the worst term (we move opposite to the gradient)
        g = _grad_worst_shift(h, shift)
        y = h - step * g
        h = _project_box_sum(y, target)

        cur_c5 = _c5_value(h)
        if cur_c5 < best_c5:

            step = 0.12               # reset step when improvement occurs

        step *= decay

        # occasional random jitter to escape stagnation

            cand = np.clip(cand, 0.0, 1.0)

        if best_c5 <= 0.3808:      # target reached – stop early
            break

    return best_h, best_c5, n
