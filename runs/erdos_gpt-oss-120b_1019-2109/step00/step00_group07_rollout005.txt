def enforce_sum(h, target_sum):
    total = h.sum()
    diff = target_sum - total
    if abs(diff) < 1e-12:
        return h
    # adjust a single entry to fix diff, respecting bounds
    if diff > 0:
        # need to add mass
        # find index with capacity > 0
        caps = 1.0 - h
        idx = np.argmax(caps)
        add = min(diff, caps[idx])
        h[idx] += add
        diff -= add
        # if still diff >0 (shouldn't), distribute among others
        # fallback: add to all with capacity proportionally
        if diff > 1e-12:
            # distribute evenly among all with capacity > 0, loop until diff satisfied
            while diff > 1e-12:
                caps = 1.0 - h
                avail = caps > 1e-12
                if not np.any(avail):
                    break
                per = diff / np.sum(avail)
                inc = np.minimum(per, caps[avail])
                h[avail] += inc

import numpy as np
import time
import random

def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize the Erdős minimum overlap problem for C5.
    Returns a tuple (h_values, c5_bound, n_points).
    """
    # -------------------------------------------------------------
    # Helper utilities
    # -------------------------------------------------------------
    def init_h():
        """Initial step function."""
        try:
            # use the provided construction when available
            h0 = np.array(initial_h_values, dtype=np.float64)
        except NameError:
            # fallback: uniform 0.5 values (guaranteed to satisfy the sum)
            N = 80
            h0 = np.full(N, 0.5, dtype=np.float64)
        N = h0.shape[0]
        target_sum = N / 2.0
        cur_sum = h0.sum()
        if abs(cur_sum - target_sum) > 1e-12:
            # simple scaling, then a tiny correction to keep within [0,1]
            h0 = h0 * (target_sum / cur_sum)
            h0 = np.clip(h0, 0.0, 1.0)
            diff = target_sum - h0.sum()
            # adjust the first component to enforce the exact sum
            h0[0] = np.clip(h0[0] + diff, 0.0, 1.0)
        return h0

    def compute_c5(h):
        """C5 = max_k ∫ h(x) (1‑h(x+k)) dx  using the discretised correlation."""
        N = h.shape[0]
        dx = 2.0 / N
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return float(corr.max())

    def worst_lag_index(h):
        """Return the shifted‑index (lag) that attains the maximum overlap."""
        N = h.shape[0]
        full = np.correlate(h, 1.0 - h, mode='full')
        idx = int(full.argmax())                # index in 0 … 2N‑2
        return idx - (N - 1)                    # convert to signed lag

    def compute_w(h, lag):
        """Contribution w_i = 1 - h_{i+lag} for the lag under consideration."""
        N = h.shape[0]
        w = np.zeros(N, dtype=np.float64)
        if lag >= 0:
            valid = N - lag
            if valid > 0:
                w[:valid] = 1.0 - h[lag:]
        else:
            L = -lag
            valid = N - L
            if valid > 0:
                w[L:] = 1.0 - h[:valid]
        return w

    def project_bounds_and_sum(h, target_sum):
        """Project onto [0,1] box and enforce the exact sum constraint."""
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            return h
        N = h.shape[0]
        # simple greedy redistribution
        if diff > 0.0:
            for i in range(N):
                cap = 1.0 - h[i]
                if cap <= 0.0:
                    continue
                add = min(cap, diff)
                h[i] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            diff = -diff
            for i in range(N):
                if h[i] <= 0.0:
                    continue
                sub = min(h[i], diff)
                h[i] -= sub
                diff -= sub
                if diff <= 1e-12:
                    break
        return h

    def gradient_f(h, alpha):
        """
        Gradient of the smooth approximation
            F(h)=logsumexp(alpha * g(h))/alpha,
        where g_k(h)=∑_i h_i (1‑h_{i+lag_k})*dx.
        """
        N = h.shape[0]
        dx = 2.0 / N
        # raw (non‑scaled) correlation vector
        g_full = np.correlate(h, 1.0 - h, mode='full')
        g = g_full * dx

        # soft‑max weights (numerically stable)
        g_max = g.max()
        exp_vals = np.exp(alpha * (g - g_max))
        w = exp_vals / exp_vals.sum()               # shape (2N‑1,)

        grad = np.zeros(N, dtype=np.float64)

        for idx, wk in enumerate(w):
            if wk == 0.0:
                continue
            lag = idx - (N - 1)
            if lag >= 0:
                L = lag
                valid = N - L
                if valid == 0:
                    continue
                # ∂/∂h_i part (where i is the left index of the product)
                grad[:valid] += wk * (1.0 - h[L:]) * dx
                # ∂/∂h_{i+lag} part (right index)
                grad[L:] += wk * (-h[:valid]) * dx
            else:
                L = -lag
                valid = N - L
                if valid == 0:
                    continue
                # symmetric case for negative lag
                grad[L:] += wk * (1.0 - h[:valid]) * dx
                grad[:valid] += wk * (-h[L:]) * dx
        return grad

    # -------------------------------------------------------------
    # Main optimisation loop
    # -------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    h = init_h()
    N = h.shape[0]
    target_sum = N / 2.0

    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    cur_c5 = best_c5

    start = time.time()
    deadline = start + budget_s - 0.05          # small safety margin
    # hyper‑parameters
    alpha = 10.0                                # smooth‑max sharpness
    lr = 0.05                                   # initial learning rate
    min_lr = 5e-4
    max_lr = 0.08

    # progress tracking
    no_improve = 0
    IMPROVE_LIMIT = 2000

    while time.time() < deadline:
        # schedule learning‑rate (linear decay)
        elapsed = time.time() - start
        frac = elapsed / (budget_s - 0.05)
        lr = max_lr * (1.0 - frac) + min_lr * frac

        # gradient step
        grad = gradient_f(h, alpha)
        step = -lr * grad
        h_new = h + step
        h_new = project_bounds_and_sum(h_new, target_sum)

        new_c5 = compute_c5(h_new)

        if new_c5 < cur_c5 - 1e-10:
            # genuine improvement
            h = h_new
            cur_c5 = new_c5
            no_improve = 0
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h_new.copy()
                # early exit if we beat the target bound
                if best_c5 <= 0.3808:
                    break
            continue

        # ---------------------------------------------------------
        # Simple fallback move when gradient step fails to improve
        # ---------------------------------------------------------
        # move a small amount of mass from a “bad” index to a “good” one
        lag = worst_lag_index(h)
        w_vec = compute_w(h, lag)

        src_idx = None
        tgt_idx = None
        src_cand = np.where((h > 1e-8) & (w_vec > 1e-8))[0]
        if src_cand.size:
            src_idx = src_cand[np.argmax(w_vec[src_cand])]

        tgt_cand = np.where((h < 1.0 - 1e-8))[0]
        if tgt_cand.size:
            tgt_idx = tgt_cand[np.argmin(w_vec[tgt_cand])]

        made_move = False
        if src_idx is not None and tgt_idx is not None and src_idx != tgt_idx:
            max_delta = min(0.2, h[src_idx], 1.0 - h[tgt_idx])
            if max_delta > 0:
                delta = max_delta * random.random()
                h_tmp = h.copy()
                h_tmp[src_idx] -= delta
                h_tmp[tgt_idx] += delta
                h_tmp = project_bounds_and_sum(h_tmp, target_sum)
                cand_c5 = compute_c5(h_tmp)
                if cand_c5 < cur_c5 - 1e-10:
                    h = h_tmp
                    cur_c5 = cand_c5
                    if cand_c5 < best_c5 - 1e-12:
                        best_c5 = cand_c5
                        best_h = h_tmp.copy()
                        if best_c5 <= 0.3808:
                            break
                    made_move = True

        if not made_move:
            # random tiny perturbation as a last resort
            i, j = np.random.choice(N, 2, replace=False)
            if h[i] > 1e-8 and h[j] < 1.0 - 1e-8:
                max_delta = min(0.1, h[i], 1.0 - h[j])
                if max_delta > 0:
                    delta = max_delta * random.random()
                    h_tmp = h.copy()
                    h_tmp[i] -= delta
                    h_tmp[j] += delta
                    h_tmp = project_bounds_and_sum(h_tmp, target_sum)
                    cand_c5 = compute_c5(h_tmp)
                    if cand_c5 < cur_c5 - 1e-10:
                        h = h_tmp
                        cur_c5 = cand_c5
                        if cand_c5 < best_c5 - 1e-12:
                            best_c5 = cand_c5
                            best_h = h_tmp.copy()
                            if best_c5 <= 0.3808:
                                break
        # stagnation handling
        no_improve += 1
        if no_improve > IMPROVE_LIMIT:
            # re‑seed from the best known solution with a small random jitter
            jitter = (np.random.rand(N) - 0.5) * 0.02
            h = best_h + jitter
            h = np.clip(h, 0.0, 1.0)
            h = project_bounds_and_sum(h, target_sum)
            cur_c5 = compute_c5(h)
            no_improve = 0

    # final clean‑up : ensure exact sum constraint
    best_h = project_bounds_and_sum(best_h, target_sum)
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), N)
