    n = len(b)

def continuous_refine(h, max_iters, rng):
    n = len(h)
    cur_val = max_overlap(h)
    best_val = cur_val
    best_h = h.copy()
    step = 0.5
    for it in range(max_iters):
        # pick two indices i, j distinct
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = h[i], h[j]
        # Compute feasible delta so that h[i]+delta in [0,1] and h[j]-delta in [0,1]
        delta_min = max(-hi, hj - 1.0)  # because hi + delta >=0 => delta >= -hi; hj - delta >=0 => delta <= hj
        delta_max = min(1.0 - hi, hj)   # hi+delta <=1 => delta <= 1-hi; hj - delta <=1 => delta >= hj-1 (already covered)

        # Conditions:
        # hi + delta ∈ [0,1] => delta ∈ [-hi, 1-hi]
        # hj - delta ∈ [0,1] => delta ∈ [hj-1, hj]
        # Intersection is [max(-hi, hj-1), min(1-hi, hj)]
        delta_low = max(-hi, hj - 1.0)
        delta_high = min(1.0 - hi, hj)
        if delta_low >= delta_high:
            continue
        # propose delta as random uniform in range scaled by step factor

            # revert
            h[i] = hi
            h[j] = hj

    return best_h, best_val

import time
import numpy as np

def _max_overlap(h: np.ndarray) -> float:
    """Return max_{k} ∫ h(x)(1-h(x+k)) dx for discrete samples."""
    # full linear cross‑correlation (zero‑padded)
    conv = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(conv) * (2.0 / h.shape[0]))


def _binary_hill_climb(b: np.ndarray, max_iters: int, rng: np.random.Generator, deadline: float):
    """Hill‑climb a binary (0/1) vector by swapping a 1 with a 0."""
    n = b.shape[0]
    cur_val = _max_overlap(b)
    best_val = cur_val
    best_b = b.copy()

    ones = list(np.flatnonzero(b))
    zeros = list(np.flatnonzero(1.0 - b))

    no_improve = 0
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        # pick a random 1 and a random 0
        idx_one = rng.integers(len(ones))
        idx_zero = rng.integers(len(zeros))
        i_one = ones[idx_one]
        i_zero = zeros[idx_zero]

        # perform swap
        b[i_one] = 0.0
        b[i_zero] = 1.0

        new_val = _max_overlap(b)
        if new_val <= cur_val:
            # accept
            cur_val = new_val
            ones[idx_one] = i_zero
            zeros[idx_zero] = i_one
            no_improve = 0
            if new_val < best_val:
                best_val = new_val
                best_b = b.copy()
        else:
            # revert
            b[i_one] = 1.0
            b[i_zero] = 0.0
            no_improve += 1

        if no_improve >= 5000:
            break
    return best_b, best_val


def _continuous_refine(h: np.ndarray, max_iters: int, rng: np.random.Generator, deadline: float):
    """Greedy pair‑wise adjustment preserving the total sum."""
    n = h.shape[0]
    cur_val = _max_overlap(h)
    best_val = cur_val
    best_h = h.copy()
    step = 0.5

    for _ in range(max_iters):
        if time.time() > deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        hi, hj = h[i], h[j]

        # feasible delta for h[i] += delta, h[j] -= delta
        delta_low = max(-hi, hj - 1.0)          # lower bound
        delta_high = min(1.0 - hi, hj)          # upper bound
        if delta_low >= delta_high:
            continue

        # propose a step‑scaled delta
        delta = step * rng.uniform(-1.0, 1.0)
        delta = max(delta_low, min(delta_high, delta))
        if abs(delta) < 1e-12:
            continue

        # apply
        h[i] = hi + delta
        h[j] = hj - delta

        new_val = _max_overlap(h)
        if new_val < cur_val - 1e-12:
            cur_val = new_val
            step = max(step * 0.99, 1e-6)
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] = hi
            h[j] = hj
    return best_h, best_val
