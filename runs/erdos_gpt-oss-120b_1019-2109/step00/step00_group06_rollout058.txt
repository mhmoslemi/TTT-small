def enforce_sum(h, T):
    # iterative water-filling
    for _ in range(10):
        diff = T - h.sum()
        if abs(diff) < 1e-10:
            break
        free = (h > 0) & (h < 1)
        if not np.any(free):
            # All at bounds; adjust by clipping some values within tolerance (should not happen often)
            # apply uniform shift and clip again
            h += diff / len(h)
            h = np.clip(h, 0, 1)
            continue
        adjust = diff / free.sum()
        # add adjust to free indices, but ensure we stay within bounds
        h_new = h.copy()
        h_new[free] += adjust
        # clip again
        h_new = np.clip(h_new, 0, 1)
        # if clipping changed something, need to recompute diff
        h = h_new
    return h

def subgradient_step(h, η):
    n = len(h)
    # compute correlation full
    corr = np.correlate(h, 1-h, mode='full')
    # find index where max occurs
    k = np.argmax(corr) - (n-1)  # shift offset (k negative if index < n-1)
    # compute gradient contributions
    # For index j: g_j = dx * ((1 - h[j+k]) - h[j - k]) with appropriate bounds.
    # Need to handle j+k and j-k indices within [0,n-1]; else term is 0 for out-of-range.
    # We'll compute vectorized.
    dx = 2.0 / n
    # shift forward
    idx_fwd = np.arange(n) + k
    term_fwd = np.zeros_like(h)
    mask_fwd = (idx_fwd >= 0) & (idx_fwd < n)
    term_fwd[mask_fwd] = 1.0 - h[idx_fwd[mask_fwd]]
    # shift backward
    idx_bwd = np.arange(n) - k
    term_bwd = np.zeros_like(h)
    mask_bwd = (idx_bwd >= 0) & (idx_bwd < n)
    term_bwd[mask_bwd] = h[idx_bwd[mask_bwd]]
    g = dx * (term_fwd - term_bwd)
    h_new = h - η * g
    # enforce constraints
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = enforce_sum(h_new, n/2)
    return h_new

def project_onto_sum_box(v, T):
    # v: numpy array
    # solve min_{x} ||x - v||_2^2 subject to sum(x)=T, 0≤x≤1
    # Using water-filling binary search for lambda.
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s < T:
            hi = mid
        else:
            lo = mid
    x = np.clip(v - lo, 0.0, 1.0)
    # Ensure sum close to T
    diff = T - x.sum()
    if abs(diff) > 1e-10:
        # distribute diff uniformly over interior (non-bound positions)
        interior = (x > 0) & (x < 1)
        if interior.any():
            x[interior] += diff / interior.sum()
            # Clip again
            x = np.clip(x, 0.0, 1.0)
        else:
            # adjust all (could be at bounds), just add small uniform and renorm
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

def subgradient_step(h, eta, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    # find idx of max (multiple possible)
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)  # shift
    # compute gradient
    # forwarded indices j+k
    idx_fwd = np.arange(n) + k
    term_fwd = np.zeros_like(h)
    mask_fwd = (idx_fwd >= 0) & (idx_fwd < n)
    term_fwd[mask_fwd] = 1.0 - h[idx_fwd[mask_fwd]]
    idx_bwd = np.arange(n) - k
    term_bwd = np.zeros_like(h)
    mask_bwd = (idx_bwd >= 0) & (idx_bwd < n)
    term_bwd[mask_bwd] = h[idx_bwd[mask_bwd]]
    g = dx * (term_fwd - term_bwd)
    h_new = h - eta * g
    # project onto constraints
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_onto_sum_box(h_new, T)
    return h_new

def perturb_move(h, delta_max):
    n = len(h)
    i, j = np.random.choice(n, 2, replace=False)
    # Determine allowed deltas that keep both in [0,1]
    # new h[i] = h[i] + d, h[j] = h[j] - d
    # Need 0 <= h[i] + d <= 1, 0 <= h[j] - d <= 1
    d_min = max(-h[i], h[j] - 1)   # lower bound on d (negative)
    d_max = min(1 - h[i], h[j])   # upper bound on d (positive)
    # Limit to delta_max magnitude
    lo = max(d_min, -delta_max)
    hi = min(d_max, delta_max)
    if lo >= hi:
        return h, False
    d = np.random.uniform(lo, hi)
    h_new = h.copy()
    h_new[i] += d
    h_new[j] -= d
    # sum unchanged, constraints satisfied
    return h_new, True

import numpy as np, time

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def project_sum_box(v, T):
    # solves min||x - v||^2 s.t sum x = T, 0 <= x <= 1 using water-filling
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s < T:
            hi = mid
        else:
            lo = mid
    x = np.clip(v - lo, 0.0, 1.0)
    # small correction
    diff = T - x.sum()
    if abs(diff) > 1e-12:
        interior = (x > 0) & (x < 1)
        if interior.any():
            x[interior] += diff / interior.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            # any adjustment needed will be tiny - just distribute uniformly
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

def subgradient_step(h, eta, T):
    n = h.size
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    k = max_idx - (n - 1)
    # compute forward and backward contributions
    idx = np.arange(n)
    fwd = idx + k
    bwd = idx - k
    term_fwd = np.zeros_like(h)
    mask_fwd = (fwd >= 0) & (fwd < n)
    term_fwd[mask_fwd] = 1.0 - h[fwd[mask_fwd]]
    term_bwd = np.zeros_like(h)
    mask_bwd = (bwd >= 0) & (bwd < n)
    term_bwd[mask_bwd] = h[bwd[mask_bwd]]
    g = dx * (term_fwd - term_bwd)
    h_new = h - eta * g
    # enforce box and sum
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, T)
    return h_new

def perturb_move(h, delta_max):
    n = h.size
    i, j = np.random.choice(n, 2, replace=False)
    # compute allowed delta
    d_min = max(-h[i], h[j] - 1.0)  # lower bound for delta (negative)
    d_max = min(1.0 - h[i], h[j])   # upper bound (positive)
    # restrict magnitude
    lo = max(d_min, -delta_max)
    hi = min(d_max, delta_max)
    if lo >= hi:
        return h, False
    d = np.random.uniform(lo, hi)
    h_new = h.copy()
    h_new[i] += d
    h_new[j] -= d
    # ensure still in [0,1]
    # due to bounds this is guaranteed
    return h_new, True

import numpy as np

def _c5_of(h: np.ndarray) -> float:
    """Compute the Erdős C₅ value for a discrete step function."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _project_sum_box(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the feasible set
        { x | 0 ≤ x_i ≤ 1,  sum(x)=target }.
    Uses a water‑filling binary search for the Lagrange multiplier.
    """
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        if x.sum() < target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v - lo, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        interior = (x > 0.0) & (x < 1.0)
        if interior.any():
            x[interior] += diff / interior.sum()
            x = np.clip(x, 0.0, 1.0)
        else:
            x += diff / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x
