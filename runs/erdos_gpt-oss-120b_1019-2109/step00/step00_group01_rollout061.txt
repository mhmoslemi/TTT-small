def project_sum(h, target_sum):
    # simple iterative scaling and clipping
    n = len(h)
    # clip to bounds
    h = np.clip(h, 0, 1)
    total = h.sum()
    # if already close, return
    if abs(total - target_sum) < 1e-10:

    # compute correction delta to distribute uniformly
    # using while loop for a few iterations
    for _ in range(10):

        # distribute diff uniformly
        delta = diff / n
        h += delta
        np.clip(h, 0, 1, out=h)
        total = h.sum()
        if abs(total - target_sum) < 1e-10:

    return h

def upsample_h(h_old, new_n):
    old_n = len(h_old)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)

    return project_sum(h_new, new_n / 2.0)

import numpy as np
import time
import math

def compute_c5(h):
    """Maximum overlap integral for a discretised step function."""
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / h.size)

def project_sum(h, target):
    """Project vector onto the box [0,1] while fixing the sum = target."""
    h = np.clip(h, 0.0, 1.0)
    total = h.sum()
    if abs(total - target) < 1e-12:
        return h
    n = h.size
    # simple water‑filling: distribute the deficit uniformly, repeat a few times
    for _ in range(10):
        diff = target - total
        if abs(diff) < 1e-12:
            break
        delta = diff / n
        h += delta
        np.clip(h, 0.0, 1.0, out=h)
        total = h.sum()
    return h

def upsample_h(old_h, new_n):
    """Linear interpolation of an existing vector to a new resolution."""
    old_n = old_h.size
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, new_n, endpoint=False)
    new_h = np.interp(x_new, x_old, old_h)
    new_h = np.clip(new_h, 0.0, 1.0)
    return project_sum(new_h, new_n / 2.0)

def generate_shape(p, N):
    """Family h(x)=s*|x-1|^p, s chosen so that ∫h=1 (h∈[0,1])."""
    x = np.linspace(0.0, 2.0, N, endpoint=False) + (1.0 / N)  # centre of bins
    t = np.abs(x - 1.0)
    s = (p + 1.0) / 2.0  # scaling for unit integral
    h = s * (t ** p)
    h = np.clip(h, 0.0, 1.0)
    return project_sum(h, N / 2.0)

        max_idx = int(np.argmax(conv))
