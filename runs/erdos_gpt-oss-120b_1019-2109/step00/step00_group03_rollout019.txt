def compute_c5_fft(h):
    n = len(h)
    N = 1 << ((2*n-1).bit_length())
    h_fft = np.fft.rfft(h, n=N)
    rev_g = (1 - h)[::-1]
    rev_g_fft = np.fft.rfft(rev_g, n=N)
    conv = np.fft.irfft(h_fft * rev_g_fft, n=N)[:2*n-1]
    return conv.max() * (2.0 / n)

def compute_c5(h):
    n = h.shape[0]
    N = 1 << ((2*n-1).bit_length())
    h_fft = np.fft.rfft(h, n=N)
    rev_g = (1 - h)[::-1]
    rev_g_fft = np.fft.rfft(rev_g, n=N)
    conv = np.fft.irfft(h_fft * rev_g_fft, n=N)[:2*n-1]
    return conv.max() * (2.0 / n)

        cur_h[i_one] = 0.0

        else:

            if T > 0:

                best_c5 = new_c5
                best_h = cur_h.copy()
                last_improve_iter = it

            cur_h[i_one] = 1.0

    return (best_h, best_c5, best_h.shape[0])

def compute_c5_cor(h):
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode='full')
    return conv.max() * (2.0 / n)

def max_missing_and_shift(h):
    n = h.shape[0]
    conv = np.correlate(h, 1 - h, mode='full')
    max_val = conv.max()
    idx = conv.argmax()
    shift = idx - (n - 1)
    return max_val, shift, conv

def _c5_from_h(h: np.ndarray) -> float:

    return conv.max() * (2.0 / n)


def _max_missing_and_shift(h: np.ndarray):
    """Largest missing overlap and the corresponding integer shift."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(conv.argmax())
    return float(conv[idx]), idx - (n - 1), conv

            new_miss, _, _ = _max_missing_and_shift(cur_h)
            new_c5 = new_miss * (2.0 / n)

            if new_c5 + 1e-12 < cur_c5:
                cur_c5 = new_c5
                improved = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = cur_h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                cur_h[i_one] = 1.0
                cur_h[i_zero] = 0.0
                no_improve += 1

        it += 1

        # restart if stuck
        if no_improve >= max_no_improve:
            cur_h[:] = 0.0
            idx = rng.choice(n, size=n // 2, replace=False)
            cur_h[idx] = 1.0
            no_improve = 0

    # final sanity
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = _c5_from_h(best_h)

    return best_h, best_c5, best_h.shape[0]
