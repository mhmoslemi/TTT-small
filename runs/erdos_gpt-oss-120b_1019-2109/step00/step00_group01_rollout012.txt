import time
import random
import numpy as np

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] while forcing the sum to equal target."""
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        w = np.clip(v + mid, 0.0, 1.0)
        if w.sum() < target:
            lo = mid
        else:
            hi = mid
    w = np.clip(v + hi, 0.0, 1.0)
    # tiny correction to hit the target sum more accurately
    diff = target - w.sum()
    if abs(diff) > 1e-10:
        free = (w > 1e-12) & (w < 1 - 1e-12)
        if free.any():
            w[free] += diff / free.sum()
            w = np.clip(w, 0.0, 1.0)
    return w


def _compute_c5(h: np.ndarray) -> float:
    """Return the C5 value for a given step function."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)


def _grad_for_max_conv(h: np.ndarray):
    """Gradient of the worst (max) overlap term."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))            # index of the worst shift
    start = max(0, idx - (n - 1))
    end = min(n - 1, idx)                 # inclusive
    i = np.arange(start, end + 1)
    grad = np.zeros_like(h)
    grad[i] = 1.0 - 2.0 * h[idx - i]
    return grad, conv[idx]                # raw (unnormalised) max value


def _initial_h(n_target: int) -> np.ndarray:
    """Build a feasible initial vector of length n_target."""
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        n0 = base.shape[0]
        if n_target == n0:
            h = base.copy()
        else:
            # linear interpolation from the provided samples
            x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
            xt = np.linspace(0.0, 2.0, n_target, endpoint=False)
            h = np.interp(xt, x0, base)
    else:
        h = np.random.rand(n_target)
    # enforce constraints
    return _project_box_sum(h, n_target / 2.0)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return a step function h, its C5 bound, and the number of points."""
    random.seed(seed)
    np.random.seed(seed)

    start_time = time.time()
    # choose a reasonable resolution (keep it < 500 for speed)
    if "initial_h_values" in globals():
        base_n = len(initial_h_values)
    else:
        base_n = 47
    n = max(200, base_n * 4)
    n = min(500, n)

    # prepare a few diverse starting points
    starts = [_initial_h(n)]

    # cosine-shaped start (smooth candidate)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h_cos = 0.5 * (1.0 + np.cos(np.pi * x))
    h_cos = _project_box_sum(h_cos, n / 2.0)
    starts.append(h_cos)

    # a few random starts
    for _ in range(3):
        rnd = np.random.rand(n)
        starts.append(_project_box_sum(rnd, n / 2.0))

    best_h = None
    best_val = np.inf

    # main optimisation loop (time‑controlled)
    for h0 in starts:
        h = h0.copy()
        cur_val = _compute_c5(h)
        if cur_val < best_val:
            best_val, best_h = cur_val, h.copy()

        step = 0.2
        it = 0
        while True:
            if time.time() - start_time > budget_s * 0.96:
                break

            grad, _ = _grad_for_max_conv(h)

            # try a few step sizes (simple backtracking line search)
            improved = False
            trial_step = step
            for _ in range(5):
                h_new = _project_box_sum(h - trial_step * grad, n / 2.0)
                new_val = _compute_c5(h_new)
                if new_val < cur_val - 1e-12:
                    h, cur_val = h_new, new_val
                    improved = True
                    step = trial_step * 1.1   # a little optimism
                    break
                trial_step *= 0.5

            if not improved:
                # no progress – shrink step and try a small random perturbation
                step *= 0.7
                if random.random() < 0.03:
                    noise = (np.random.rand(n) - 0.5) * 0.02
                    noise -= noise.mean()
                    h_noise = _project_box_sum(h + noise, n / 2.0)
                    val_noise = _compute_c5(h_noise)
                    if val_noise < cur_val or random.random() < 0.1:
                        h, cur_val = h_noise, val_noise

            # record global best
            if cur_val < best_val - 1e-12:
                best_val, best_h = cur_val, h.copy()

            it += 1
            if it % 500 == 0 and step < 1e-4:
                step = 0.2

        if time.time() - start_time > budget_s * 0.96:
            break

    # final projection to guarantee feasibility
    best_h = _project_box_sum(best_h, best_h.size / 2.0)
    final_c5 = _compute_c5(best_h)

    return best_h, final_c5, best_h.size
