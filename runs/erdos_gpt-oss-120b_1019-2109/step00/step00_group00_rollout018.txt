    # binary search for lambda
    lo = np.min(z) - 1
    hi = np.max(z) + 1
    for _ in range(50):
        lam = (lo+hi)/2

            lo = lam
        else:

    return np.clip(z - ((lo+hi)/2), 0, 1)

    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(max_iter):

        s = x.sum()

import numpy as np
import time

def project_capped_simplex(z, S, tol=1e-12, max_iter=60):
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(z - lam, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    # fix any tiny diff
    diff = S - x.sum()
    if abs(diff) > tol:
        mask = (x > 0) & (x < 1)
        if np.any(mask):
            cnt = mask.sum()
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            # all at bounds, distribute diff arbitrarily
            idx = np.argmin(np.abs(z - (lo+hi)/2.0))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

def compute_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()

import numpy as np

def project_capped_simplex(z, S, tol=1e-12, max_iter=60):
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(z - lam, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > tol:
        mask = (x > 0) & (x < 1)
        if mask.any():
            cnt = mask.sum()
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.argmin(np.abs(z - (lo+hi)/2.0))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x

import numpy as np
import time

def project_capped_simplex(z, S, tol=1e-12, max_iter=60):
    """
    Euclidean projection of z onto the set {x | 0<=x_i<=1, sum(x)=S}.
    """
    lo = np.min(z) - 1.0   # guarantees all entries clipped to 1
    hi = np.max(z) + 1.0   # guarantees all entries clipped to 0
    for _ in range(max_iter):
        lam = (lo + hi) / 2.0
        x = np.clip(z - lam, 0.0, 1.0)
        s = x.sum()
        if s > S:
            lo = lam
        else:
            hi = lam
    x = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    diff = S - x.sum()
    if abs(diff) > tol:
        mask = (x > 0) & (x < 1)
        if mask.any():
            cnt = mask.sum()
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.argmin(np.abs(z - (lo + hi) / 2.0))
            x[idx] = np.clip(x[idx] + diff, 0.0, 1.0)
    return x


def compute_c5(h):
    """
    Exact C5 for a discretized step function h.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return corr.max()


def random_feasible(rng, n, S):
    """
    Generate a random point in the feasible set.
    """
    z = rng.random(n)
    return project_capped_simplex(z, S)


def run(seed=42, budget_s=1100, **kwargs):

    n = 200                     # number of discretisation points
    S = n / 2.0                 # required sum of h_i
    # ------------------------------------------------------------------
    # 1. starting point

        init_vals = initial_h_values
        has_init = True
    except NameError:
        has_init = False

    if has_init:
        x0 = np.linspace(0.0, 2.0, len(init_vals), endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        h0 = np.interp(x, x0, init_vals)
        h0 = np.clip(h0, 0.0, 1.0)
        h0 = project_capped_simplex(h0, S)
    else:
        h0 = random_feasible(rng, n, S)

    best_h = h0.copy()
    best_val = compute_c5(best_h)

    start_time = time.time()
    time_limit = budget_s * 0.95    # safety margin
    restart = 0

    # ------------------------------------------------------------------
    # 2. main optimisation loop (projected gradient descent)
    while time.time() - start_time < time_limit:
        if restart == 0 and has_init:
            h = h0.copy()
        else:
            h = random_feasible(rng, n, S)

        # hyper‑parameters for this run (a bit of randomisation)
        lr = rng.uniform(0.2, 0.6)          # initial step size
        alpha = rng.uniform(20.0, 80.0)     # smooth‑max strength
        max_iters = 2000

            if time.time() - start_time > time_limit:
                break

            # ------- evaluate current objective ----------
            conv_hh = np.correlate(h, h, mode='full')
            f = 1.0 - (2.0 / n) * conv_hh          # vector of overlap costs
            C = f.max()
            if C < best_val:
                best_val = C
                best_h = h.copy()

            # ------- smooth max weights -------------------
            max_f = f.max()
            exp_term = np.exp(alpha * (f - max_f))
            w = exp_term / exp_term.sum()
