def project_onto_box_simplex(v, total):
    # total = n/2
    # returns x = clip(v - lam, 0, 1) such that sum(x) = total
    # lam determined via bisection
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi)/2
        x = np.clip(v - lam, 0, 1)
        s = np.sum(x)
        if s > total:
            lo = lam
        else:

import numpy as np, time, math

def project_box_sum(v,total):
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi)/2.0
        x = np.clip(v - lam, 0, 1)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0, 1)

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')
def evaluate(h):
    return np.max(compute_corr(h)) * 2.0 / len(h)

def gradient(h,weights):
    n = len(h)
    grad = np.zeros_like(h)
    # iterate over shifts
    for shift_idx, w in enumerate(weights):
        shift = shift_idx - (n - 1)
        if w == 0:
            continue
        if shift >= 0:
            k = shift
            if n - k > 0:
                grad[:n - k] += w * (1 - h[k:])
                grad[k:] -= w * h[:n - k]
        else:
            k = -shift
            if n - k > 0:
                grad[k:] += w * (1 - h[:n - k])
                grad[:n - k] -= w * h[k:]
    return grad

def optimize(n, budget, seed):
    np.random.seed(seed)
    # start with random 0-1 pattern maybe half ones, plus random small noise to be fractional
    h = np.random.rand(n)
    # project to feasible set: adjust to have sum = n/2
    h = project_box_sum(h, n/2.0)
    best_h = h.copy()
    best_c = evaluate(h)
    start = time.time()
    it = 0
    # parameters for schedule
    max_iter = 1000000
    while time.time() - start < budget * 0.95 and it < max_iter:
        # compute correlation array C
        C = compute_corr(h)
        maxC = C.max()
        # if better
        if maxC < best_c:
            best_c = maxC
            best_h = h.copy()
        # smoothing param
        alpha = 10.0  # can adapt maybe increase over time
        maxC_val = maxC
        # compute smooth weights
        # Use log-sum-exp weights: w_i = exp(alpha * C_i) / sum
        exp_vals = np.exp(alpha * (C - maxC_val))
        weights = exp_vals / exp_vals.sum()
        # compute gradient of smoothed max
        g = gradient(h, weights)
        # step size adaptive
        eta = 0.5 / (np.sqrt(it+1))
        h_new = h - eta * g
        # project onto constraints
        h_new = project_box_sum(h_new, n/2.0)
        h = h_new
        it += 1
    return best_h, evaluate(best_h), n

import time
import numpy as np

def _project_onto_box_sum(v: np.ndarray, total: float) -> np.ndarray:
    """Project v onto the set {0<=x<=1, sum(x)=total}."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(50):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    return np.clip(v - lam, 0.0, 1.0)


def _correlate_full(h: np.ndarray) -> np.ndarray:
    """Linear correlation of h with (1‑h)."""
    return np.correlate(h, 1.0 - h, mode="full")


def _c5_from_h(h: np.ndarray) -> float:
    """Return C5 = max_k ∫ h(x)(1‑h(x+k))dx approximated on the grid."""
    n = h.size
    return float(_correlate_full(h).max() * (2.0 / n))


def _grad_one_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """Gradient of the single correlation term for a given shift."""
    n = h.size
    g = np.zeros_like(h)
    L = n - abs(shift)
    if L <= 0:
        return g
    if shift >= 0:
        k = shift
        # indices 0 .. L‑1 correspond to i, i+k are k .. k+L‑1
        g[:L] += 1.0 - h[k:k + L]
        g[k:k + L] -= h[:L]
    else:
        k = -shift
        # shift < 0 : i ranges k .. k+L‑1, i‑k ranges 0 .. L‑1
        g[k:k + L] += 1.0 - h[:L]
        g[:L] -= h[k:k + L]
    return g


def _optimize_once(n: int, seed: int, max_iters: int, time_limit: float):
    """Run a single gradient‑descent run, return best h and its raw max correlation."""
    rng = np.random.default_rng(seed)
    # start from a random point and project onto the feasible set
    h = rng.random(n, dtype=np.float64)
    h = _project_onto_box_sum(h, n * 0.5)

    best_h = h.copy()
    best_raw = _correlate_full(h).max()
    start = time.time()

    for it in range(1, max_iters + 1):
        if time.time() - start > time_limit:
            break

        corr = _correlate_full(h)
        cur_raw = corr.max()

        if cur_raw < best_raw:
            best_raw = cur_raw
            best_h = h.copy()
            # stop early if already below target
            if best_raw * (2.0 / n) <= 0.3808:
                break

        # pick one of the (possibly several) maximal shifts
        idxs = np.where(corr >= cur_raw - 1e-12)[0]
        idx = rng.choice(idxs)
        shift = idx - (n - 1)

        # gradient of the chosen worst shift
        g = _grad_one_shift(h, shift)

        # simple diminishing step size
        eta = 0.5 / np.sqrt(it)

        h_new = h - eta * g
        h = _project_onto_box_sum(h_new, n * 0.5)

        # occasional tiny random jitter to escape plateaus
        if it % 3000 == 0:
            jitter = (rng.random(n) - 0.5) * 0.02
            h = np.clip(h + jitter, 0.0, 1.0)
            h = _project_onto_box_sum(h, n * 0.5)

    return best_h, best_raw


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimize a step function for the Erdős minimum overlap problem.
    Returns (h_values, c5_bound, n_points).
    """
    start_total = time.time()
    # target bound (stop early if reached)
    target = 0.3808

    # choose a modest size – large enough for a decent bound, small enough to stay fast
    n = 200
    if n % 2:
        n += 1

    best_h = None
    best_bound = float("inf")
    restart = 0

    # keep restarting until time is exhausted or target achieved
    while time.time() - start_total < budget_s * 0.90:
        remaining = budget_s * 0.90 - (time.time() - start_total)
        # give each run a few seconds – 30 000 iterations is plenty for n=200
        max_iters = 30000
        h_candidate, raw_max = _optimize_once(n, seed + restart, max_iters, remaining)
        bound = raw_max * (2.0 / n)

        if bound < best_bound:
            best_bound = bound
            best_h = h_candidate.copy()

        if best_bound <= target:
            break
        restart += 1

    # final sanity projection (eliminate tiny drift)
    best_h = _project_onto_box_sum(best_h, n * 0.5)
    final_bound = _c5_from_h(best_h)

    return best_h, float(final_bound), n
