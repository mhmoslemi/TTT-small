def solve_lp(N):
    h = cp.Variable(N)
    lamb = cp.Variable()
    constraints = [h >= 0, h <= 1, cp.sum(h) == N/2]
    # For k=0:
    # w0 = h
    # sum w0 = N/2 >= lamb (but N/2 is larger > unknown, but we include anyway)

    # For each k>=1
    for k in range(1, N):
        length = N - k
        # define w_k of size length
        w = cp.Variable(length)
        # constraints w <= h[:length], w <= h[k:]
        constraints.append(w <= h[:length])
        constraints.append(w <= h[k:])
        # (optional) w >= h[:length] + h[k:] - 1
        constraints.append(w >= h[:length] + h[k:] - 1)
        constraints.append(w >= 0)
        constraints.append(cp.sum(w) >= lamb)
        # store w maybe not needed later

    # maximize lambda
    prob = cp.Problem(cp.Maximize(lamb), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    return h.value, lamb.value, prob.status

import time
import numpy as np
import cvxpy as cp

def _c5_of(h):
    """Return the C5 value for a step vector h."""
    N = h.shape[0]
    dx = 2.0 / N
    # correlation in “full” mode gives all integer lags
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def _milp_binary(N, time_budget):
    """
    Solve the integer program for a binary vector of length N.
    Returns (h, t) where h is a binary array and t is the maximal conflict.
    If the solver does not finish in time_budget seconds, returns (None, None).
    """
    # time guard – we rely on the outer caller to respect the overall budget
    start = time.time()
    # binary variables
    h = cp.Variable(N, boolean=True)
    t = cp.Variable(integer=True)

    constraints = [
        cp.sum(h) == N // 2,          # exactly half the entries are 1
        t >= 0,
        t <= N,                       # trivial bound
    ]

    # for each positive lag L we create the product variables w_{i,L}
    for L in range(1, N):
        length = N - L
        w = cp.Variable(length, boolean=True)          # w[i] = h[i] * h[i+L]

        # McCormick (exact for binaries)
        constraints += [
            w <= h[:length],
            w <= h[L:],
            w >= h[:length] + h[L:] - 1,
        ]

        # forward mismatches  (i, i+L)
        constraints.append(cp.sum(h[:length]) - cp.sum(w) <= t)
        # backward mismatches (i+L, i)
        constraints.append(cp.sum(h[L:]) - cp.sum(w) <= t)

    # minimise the worst‑case conflict
    prob = cp.Problem(cp.Minimize(t), constraints)

    # try a few MILP solvers; GLPK_MI is the most widely available
    solved = False
    for solver in (cp.GLPK_MI, cp.CBC, cp.SCIP):
        try:
            prob.solve(solver=solver, warm_start=True, msg=False)
        except Exception:
            continue
        if prob.status in ("optimal", "optimal_inaccurate"):
            solved = True
            break
        # some solvers may return infeasible quickly – fall back to next
    if not solved or time.time() - start > time_budget:
        return None, None

    h_val = np.rint(np.array(h.value).flatten())
    t_val = int(round(t.value))
    return h_val, t_val


def _lp_relax(N):
    """
    Solve the linear‑programming relaxation that maximises the minimal
    overlap Σ min(h_i,h_{i+L}).  The returned h is a feasible real vector.
    """
    h = cp.Variable(N)
    lam = cp.Variable()               # lower bound on the sum of mins

    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == N / 2,
        lam >= 0,
        cp.sum(h) >= lam,              # shift 0 gives sum(h)=N/2
    ]

    # for each positive lag L
    for L in range(1, N):
        length = N - L
        w = cp.Variable(length)

        constraints += [
            w <= h[:length],
            w <= h[L:],
            w >= h[:length] + h[L:] - 1,
            cp.sum(w) >= lam,
        ]

    prob = cp.Problem(cp.Maximize(lam), constraints)
    # ECOS handles large LPs efficiently
    prob.solve(solver=cp.ECOS, max_iters=5000, verbose=False)
    if h.value is None:
        # fall back to a slower interior‑point solver if needed
        prob.solve(solver=cp.SCS, verbose=False)
    h_val = np.clip(np.array(h.value).flatten(), 0.0, 1.0)
    # enforce the exact integral constraint (tiny rounding error fix)
    target = N / 2.0
    diff = target - h_val.sum()
    if abs(diff) > 1e-9:
        h_val += diff / N
        h_val = np.clip(h_val, 0.0, 1.0)
    return h_val


def _local_improve(h, budget, seed):
    """
    Simple hill‑climbing / simulated‑annealing style local search.
    Works for both binary and fractional vectors.
    """
    rng = np.random.default_rng(seed)
    N = h.shape[0]
    best_h = h.copy()
    best_val = _c5_of(best_h)
    cur_h = h.copy()
    cur_val = best_val
    t0 = time.time()
    # step size for fractional moves – will be reduced over time
    step = 0.2

    while time.time() - t0 < budget:
        # binary case: swap a 1 and a 0
        if np.all(np.isclose(cur_h, np.round(cur_h))):
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            cur_h[i] = 0.0
            cur_h[j] = 1.0
            new_val = _c5_of(cur_h)
            if new_val <= cur_val:
                cur_val = new_val
                if new_val < best_val:
                    best_val, best_h = new_val, cur_h.copy()
            else:
                # revert
                cur_h[i] = 1.0
                cur_h[j] = 0.0
        else:
            # fractional move – transfer a small amount from one coordinate to another
            i, j = rng.choice(N, size=2, replace=False)
            # amount that can be moved from i to j while staying in [0,1]
            max_down = min(step, cur_h[i])               # what we can take from i
            max_up = min(step, 1.0 - cur_h[j])           # what we can give to j
            delta = rng.uniform(0.0, min(max_down, max_up))
            if delta < 1e-8:
                continue
            cur_h[i] -= delta
            cur_h[j] += delta
            new_val = _c5_of(cur_h)
            if new_val <= cur_val:
                cur_val = new_val
                if new_val < best_val:
                    best_val, best_h = new_val, cur_h.copy()
            else:
                # revert
                cur_h[i] += delta
                cur_h[j] -= delta
        # gradually shrink step size
        step *= 0.9999
    return best_h, best_val
