import numpy as np
import time
import math

def _evaluate_h(h: np.ndarray) -> float:
    """
    Compute C5 for a given discretised step function h.
    C5 = max_k ∫ h(x) (1 - h(x+k)) dx  ->  max convolution * dx
    """
    n = h.shape[0]
    # Cross‑correlation h with (1‑h) (full mode)
    c = np.correlate(h, 1.0 - h, mode="full")
    return np.max(c) * (2.0 / n)   # dx = 2/n


def _project_onto_constraints(v: np.ndarray, total: float) -> np.ndarray:
    """
    Project vector v onto the box 0≤x≤1 and the affine hyperplane sum(x)=total.
    Uses a simple bisection on the Lagrange multiplier for the equality.
    """
    v = np.clip(v, 0.0, 1.0)

    lo, hi = -1.0, 1.0           # bounds for lambda
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction for numerical drift
    diff = total - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            add = diff / free.sum()
            x[free] = np.clip(x[free] + add, 0.0, 1.0)
    return np.clip(x, 0.0, 1.0)


def run(seed: int = 42, budget_s: int = 1100):
    """
    Stochastic optimisation that attempts to lower the Erdős C5 constant.
    Returns (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ----- initialise -------------------------------------------------
    if 'initial_h_values' in globals():
        base = np.asarray(initial_h_values, dtype=float)
        n = base.shape[0]
        h = base.copy()
    else:
        n = 200                     # choose a modest size (< 1000)
        h = np.zeros(n, dtype=float)
        ones = np.random.choice(n, n // 2, replace=False)
        h[ones] = 1.0

    # enforce constraints exactly
    h = _project_onto_constraints(h, n / 2.0)

    cur_val = _evaluate_h(h)
    best_h = h.copy()
    best_val = cur_val

    # ----- simulated‑annealing parameters -----------------------------
    T0 = 0.05                      # initial temperature
    T = T0
    it = 0
    target = 0.3808                # we stop early if we hit this

    # ----- main optimisation loop ------------------------------------
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.5:          # safety margin
            break
        if best_val <= target:
            break

        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # decide move type
        if np.random.rand() < 0.5:
            # ---- full swap (binary‑style) ----
            if h[i] == h[j]:
                continue
            h_new = h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # ---- fractional transfer ----
            # feasible delta so that both stay in [0,1]
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi <= lo:

            delta = lo + np.random.rand() * (hi - lo)
            h_new = h.copy()
            h_new[i] += delta
            h_new[j] -= delta

        new_val = _evaluate_h(h_new)
