import numpy as np
import time

def compute_c5(h: np.ndarray) -> float:
    """Return max overlap integral for a discretised step function."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_onto_capped_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the set
        { x : 0 <= x_i <= 1 , sum_i x_i = target_sum }.
    Uses a simple bisection on the Lagrange multiplier.
    """
    lo = np.min(v - 1.0)          # makes all entries clamp to 1
    hi = np.max(v)                # makes all entries clamp to 0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # small correction for numerical drift
    diff = target_sum - x.sum()
    if abs(diff) > 1e-9:
        free = (x > 0.0) & (x < 1.0)
        if free.any():
            adjust = diff / free.sum()
            x[free] += adjust
            x = np.clip(x, 0.0, 1.0)
    return x


def _gradient_of_max_overlap(h: np.ndarray):
    """
    Gradient of the currently maximal overlap term.
    Returns (grad, lag, corr) where corr is the full correlation array.
    """
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx_max = int(np.argmax(corr))
    lag = idx_max - (n - 1)          # integer shift corresponding to the max

    g = np.zeros_like(h)

    if lag >= 0:
        # first part: (1 - h[i+lag]) * dx for i where i+lag is valid
        i1 = np.arange(0, n - lag)
        g[i1] = (1.0 - h[i1 + lag]) * dx
        # second part: -h[i-lag] * dx for i where i-lag is valid
        i2 = np.arange(lag, n)
        g[i2] -= h[i2 - lag] * dx
    else:
        lag_abs = -lag
        # first part (lag negative)
        i1 = np.arange(lag_abs, n)
        g[i1] = (1.0 - h[i1 - lag_abs]) * dx
        # second part
        i2 = np.arange(0, n - lag_abs)
        g[i2] -= h[i2 + lag_abs] * dx
    return g, lag, corr


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function h on [0,2] to minimise the Erdős overlap constant C5.

    Returns a tuple (h_values, c5_bound, n_points).
    """
    np.random.seed(seed)
    start_time = time.time()

    # ---- initialise -------------------------------------------------
    init_vec = globals().get("initial_h_values")
    if init_vec is not None:
        h = np.array(init_vec, dtype=float).copy()
        n = h.shape[0]
        target = n / 2.0
        h = _project_onto_capped_simplex(h, target)
    else:
        # a reasonable default size (small enough for fast optimisation)
        n = 150
        target = n / 2.0
        xs = np.linspace(0.0, 2.0, n, endpoint=False) + (1.0 / n)
        # quadratic hump, normalised to integral 1
        h0 = 0.75 * (1.0 - (xs - 1.0) ** 2)
        h0 = np.clip(h0, 0.0, 1.0)
        h = _project_onto_capped_simplex(h0, target)

    best_h = h.copy()
    best_val = compute_c5(best_h)

    # ---- optimisation parameters ------------------------------------
    lr = 5.0                     # current step size
    min_lr = 1e-4
    it = 0

    while time.time() - start_time < budget_s - 0.5:
        grad, lag, _ = _gradient_of_max_overlap(h)

        # simple back‑tracking line search
        trial_lr = lr
        improved = False
        for _ in range(10):
            cand = h - trial_lr * grad
            cand = _project_onto_capped_simplex(cand, n / 2.0)
            cand_val = compute_c5(cand)
            if cand_val < best_val:
                improved = True
                break
            trial_lr *= 0.5

        if improved:
            best_val = cand_val
            best_h = cand.copy()
            h = cand
            lr = min(trial_lr * 1.2, 10.0)   # slightly enlarge step after success
        else:
            lr = max(lr * 0.7, min_lr)       # shrink step if no progress

        # occasional random jitter to escape plateaus
        if it % 30 == 0:
            remaining = budget_s - (time.time() - start_time)
            amp = 0.25 * (remaining / budget_s)   # amplitude shrinks to zero
            noise = (np.random.rand(n) - 0.5) * amp
            jitter = np.clip(h + noise, 0.0, 1.0)
            jitter = _project_onto_capped_simplex(jitter, n / 2.0)
            jitter_val = compute_c5(jitter)
            if jitter_val < best_val:
                best_val = jitter_val
                best_h = jitter.copy()
                h = jitter
                lr = 5.0

        it += 1

    return (best_h, best_val, n)
