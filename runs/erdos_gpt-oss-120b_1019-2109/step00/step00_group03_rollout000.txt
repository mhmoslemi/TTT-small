def compute_cost(h):
    n = len(h)
    g = 1 - h
    # reverse g for correlation
    g_rev = g[::-1]
    # choose size = next power of two >= 2n-1
    L = 1 << ((2*n - 1).bit_length())
    # zero pad
    H = np.fft.rfft(h, n=L)
    G = np.fft.rfft(g_rev, n=L)
    corr = np.fft.irfft(H * G, n=L)[:2*n-1]   # real values
    max_val = np.max(corr)
    return max_val * (2.0 / n)

def run(seed=42, budget_s=1100, **kwargs):
    import time
    import numpy as np

    start_time = time.time()
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size and initial vector
    # ------------------------------------------------------------------
    try:
        init = initial_h_values  # provided by the harness
        n = len(init)
        init_arr = np.asarray(init, dtype=float)
        half = n // 2
        # take the half largest values as ones (ensures sum = n/2)
        idx = np.argpartition(init_arr, -half)[-half:]
        h = np.zeros(n, dtype=int)
        h[idx] = 1
    except Exception:
        # fallback: use 70 points, half ones at random
        n = 70
        half = n // 2
        h = np.zeros(n, dtype=int)
        h[rng.choice(n, size=half, replace=False)] = 1

    dx = 2.0 / n

    # ------------------------------------------------------------------
    # Helper: compute the max overlap (cost)
    # ------------------------------------------------------------------
    def compute_cost(vec):
        # correlation between vec and its complement
        corr = np.correlate(vec, 1 - vec, mode='full')
        return np.max(corr) * dx

    current_cost = compute_cost(h)
    best_h = h.copy()
    best_cost = current_cost

    # ------------------------------------------------------------------
    # Simulated annealing parameters
    # ------------------------------------------------------------------
    T_initial = 0.05          # starting temperature
    T = T_initial
    decay = 0.99997           # exponential cooling factor
    restart_limit = 250000    # steps without improvement before a restart
    steps_since_best = 0

    # pre‑compute lags for possible use (not needed with full recompute)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s * 0.96:
        # choose a random 1 and a random 0 to swap
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(h == 0)
        if ones.size == 0 or zeros.size == 0:
            break
        i = rng.choice(ones)
        j = rng.choice(zeros)

        # propose new vector by swapping i (1→0) and j (0→1)
        new_h = h.copy()
        new_h[i] = 0
        new_h[j] = 1

        new_cost = compute_cost(new_h)

        # acceptance test
        if new_cost < current_cost or rng.random() < np.exp(-(new_cost - current_cost) / T):
            h = new_h
            current_cost = new_cost
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = h.copy()
                steps_since_best = 0
            else:
                steps_since_best += 1
        else:
            steps_since_best += 1

        # cooling
        T *= decay
        if T < 1e-7:
            T = T_initial

        # occasional restart to escape deep local minima
        if steps_since_best > restart_limit:
            # random new start (preserving half ones)
            h = np.zeros(n, dtype=int)
            h[rng.choice(n, size=half, replace=False)] = 1
            current_cost = compute_cost(h)
            T = T_initial
            steps_since_best = 0

    # ------------------------------------------------------------------
    # Return the best found solution
    # ------------------------------------------------------------------
    best_h_float = best_h.astype(float)
    # Re‑evaluate to avoid any rounding drift
    best_cost = compute_cost(best_h_float)
    return best_h_float, float(best_cost), n
