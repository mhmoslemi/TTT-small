def project_box_sum(v, target):
    # clip into [0,1] after shifting by lambda
    # We need lambda such that sum_i clip(v_i + lam, 0,1) = target.
    lo = -1.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def project_box_sum(v, target):
    # v is any real vector
    # Find lambda such that sum(clip(v + lambda, 0, 1)) = target
    lo = -1.0
    hi = 1.0
    # Because adding -1 to any entry reduces to 0, adding 1 increases at most to 1.
    # Dit ensures we have feasible region.
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)
    # Ensure exact sum within tolerance: small adjust by distributing leftover.
    # Could optionally do a small fix.
    return proj

def gradient_smoothmax(h, alpha):
    n = h.size
    f = np.correlate(h, 1 - h, mode='full')
    # stability
    max_f = np.max(f)
    exp_f = np.exp(alpha * (f - max_f))
    w = exp_f / np.sum(exp_f)
    grad = np.zeros_like(h)
    # Loop over shifts
    for shift_idx in range(2 * n - 1):
        shift = shift_idx - (n - 1)
        weight = w[shift_idx]
        if weight == 0:
            continue
        if shift >= 0:
            # slice length = n - shift
            l = n - shift
            # grad[0:l] += weight * (1 - h[shift:])
            grad[:l] += weight * (1 - h[shift:])
            grad[shift:] += weight * (-h[:l])
        else:
            d = -shift
            l = n - d
            grad[d:] += weight * (1 - h[:l])
            grad[:l] += weight * (-h[d:])
    return grad

import time
import numpy as np

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project vector v onto the intersection of the box [0,1]^n and the hyperplane sum(v)=target.
    Uses a simple water‑filling (binary search on a Lagrange multiplier).
    """
    lo, hi = -1.0, 1.0          # enough to drive all entries to 0 or 1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        proj = np.clip(v + mid, 0.0, 1.0)
        s = proj.sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)


def _evaluate(h: np.ndarray) -> float:
    """Return C5 for the current step function h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * (2.0 / h.size)


def _grad_smooth_max(h: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of the smooth‑max approximation of C5.
    The smooth max is (1/α)·log Σ exp(α·f_k) where f_k are the raw correlation values.
    """
    n = h.size
    f = np.correlate(h, 1.0 - h, mode='full')
    # stable soft‑max weights
    max_f = np.max(f)
    exp_f = np.exp(alpha * (f - max_f))
    w = exp_f / exp_f.sum()

    grad = np.zeros_like(h)
    for idx in range(2 * n - 1):
        shift = idx - (n - 1)
        weight = w[idx]
        if weight == 0.0:
            continue
        if shift >= 0:
            l = n - shift
            # i = 0..l-1 , j = i+shift = shift..n-1
            grad[:l]     += weight * (1.0 - h[shift:])
            grad[shift:] += weight * (-h[:l])
        else:
            d = -shift
            l = n - d
            # i = d..n-1 , j = i-shift = i-d = 0..l-1
            grad[d:]     += weight * (1.0 - h[:l])
            grad[:l]     += weight * (-h[d:])
    return grad


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Optimise a step function `h` on [0,2] with ∫h = 1, 0≤h≤1,
    minimising C₅ = max_k ∫ h(x)(1−h(x+k))dx.

    Returns
    -------
    h_values : np.ndarray
        Optimised discretised function (length n).
    c5_bound : float
        Corresponding C₅ value.
    n_points : int
        Number of discretisation points.
    """
    np.random.seed(seed)

    # --- problem size -------------------------------------------------
    try:
        n = len(initial_h_values)          # provided by the harness
    except NameError:
        n = 70                               # fallback
    target_sum = n / 2.0                     # because ∫h = 1 and dx = 2/n
    dx = 2.0 / n

    # --- initialise h -------------------------------------------------
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=np.float64).copy()
        h = np.clip(h, 0.0, 1.0)
        h = _project_box_sum(h, target_sum)
    else:
        h = np.random.rand(n)
        h = _project_box_sum(h, target_sum)

    best_h = h.copy()
    best_val = _evaluate(h)
    cur_val = best_val

    # --- optimisation parameters ---------------------------------------
    step = 0.5               # initial learning rate
    alpha = 5.0              # soft‑max sharpness
    it = 0
    start = time.time()
    no_improve = 0
    max_no_improve = 25000
    temp_start = 0.02        # annealing temperature for occasional uphill moves

    while time.time() - start < budget_s * 0.95:
        # gradient of smooth max
        grad = _grad_smooth_max(h, alpha)

        # propose a step and project back to feasible set
        h_new = _project_box_sum(h - step * grad, target_sum)
        new_val = _evaluate(h_new)

        # simple simulated‑annealing acceptance
        delta = new_val - cur_val
        temperature = temp_start * (0.9995 ** it)
        accept = (delta <= 0.0) or (np.random.rand() < np.exp(-delta / max(temperature, 1e-12)))

        if accept:
            h = h_new
            cur_val = new_val
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # schedule adjustments
        if it % 200 == 0 and alpha < 200.0:
            alpha *= 1.04
        if it % 500 == 0 and step > 0.02:
            step *= 0.93

        # restart if stuck
        if no_improve > max_no_improve:
            h = np.random.rand(n)
            h = _project_box_sum(h, target_sum)
            cur_val = _evaluate(h)
            step = 0.5
            alpha = 5.0
            no_improve = 0

        it += 1

    # final projection (ensures feasibility)
    best_h = _project_box_sum(best_h, target_sum)
    final_c5 = _evaluate(best_h)

    return best_h, final_c5, n
