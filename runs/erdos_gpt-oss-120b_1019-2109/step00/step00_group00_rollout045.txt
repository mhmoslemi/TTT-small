    for _ in range(60):
        mid = (low + high) / 2

            low = mid
        else:

import numpy as np, time
from scipy.special import logsumexp

def compute_c5(h):
    N = h.shape[0]
    dx = 2.0 / N
    return np.max(np.correlate(h, 1 - h, mode='full')) * dx

def project_capped_simplex(y, target_sum):
    low = np.min(y) - 1.0
    high = np.max(y)
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(y - (low + high) / 2.0, 0.0, 1.0)

def symmetrize(h, target_sum):
    h_sym = (h + h[::-1]) / 2.0
    return project_capped_simplex(h_sym, target_sum)

def compute_weights(conv, alpha):
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    w = exp_scaled / sum_exp
    return w, max_scaled, sum_exp

def compute_gradient(h, w):
    N = h.shape[0]
    g = np.zeros_like(h)
    shifts = np.arange(-N+1, N)
    idx = np.arange(N)
    for ind, k in enumerate(shifts):
        wk = w[ind]
        if wk == 0:
            continue
        i_k = idx + k
        mask1 = (i_k >= 0) & (i_k < N)
        if mask1.any():
            g[idx[mask1]] += wk * (1 - h[i_k[mask1]])
        i_mk = idx - k
        mask2 = (i_mk >= 0) & (i_mk < N)
        if mask2.any():
            g[idx[mask2]] -= wk * h[i_mk[mask2]]
    return g

def gradient_descent(h0, target_sum, alpha=200.0, max_iter=800, init_step=0.5):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)

import numpy as np
import time
from scipy.special import logsumexp


def compute_c5(h: np.ndarray) -> float:
    """Exact C5 for a discretized step function."""
    N = h.shape[0]
    dx = 2.0 / N
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx


def project_capped_simplex(y: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project onto the set {x | 0 <= x_i <= 1, sum x_i = target_sum}.
    Uses a bisection on the Lagrange multiplier.
    """
    low = np.min(y) - 1.0  # makes all entries 1 after clipping
    high = np.max(y)       # makes all entries 0 after clipping
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        s = x.sum()
        if s > target_sum:
            low = mid
        else:
            high = mid
    return np.clip(y - (low + high) / 2.0, 0.0, 1.0)


def symmetrize(h: np.ndarray, target_sum: float) -> np.ndarray:
    """Enforce h(x)=h(2-x) by averaging with its reverse and re‑project."""
    h_sym = (h + h[::-1]) / 2.0
    return project_capped_simplex(h_sym, target_sum)


def compute_weights(conv: np.ndarray, alpha: float):
    """Softmax (with stabilization) of alpha*conv."""
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    sum_exp = exp_scaled.sum()
    w = exp_scaled / sum_exp
    return w, max_scaled, sum_exp


def compute_gradient(h: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max objective.
    g_i = sum_k w_k * (1 - h_{i+k}) - sum_k w_k * h_{i-k}.
    Out‑of‑range indices are ignored (treated as 0).
    """
    N = h.shape[0]
    g = np.zeros_like(h)
    shifts = np.arange(-N + 1, N)          # corresponds to convolution indices
    idx = np.arange(N)

    for ind, k in enumerate(shifts):
        wk = w[ind]
        if wk == 0.0:
            continue

        i_k = idx + k
        mask1 = (i_k >= 0) & (i_k < N)
        if mask1.any():
            g[idx[mask1]] += wk * (1 - h[i_k[mask1]])

        i_mk = idx - k
        mask2 = (i_mk >= 0) & (i_mk < N)
        if mask2.any():
            g[idx[mask2]] -= wk * h[i_mk[mask2]]

    return g


def gradient_descent(
    h0: np.ndarray,
    target_sum: float,
    alpha: float = 200.0,
    max_iter: int = 800,
    init_step: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the smooth approximation of max C5."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = init_step

    for _ in range(max_iter):
        conv = np.correlate(h, 1 - h, mode="full")
        w, max_scaled, sum_exp = compute_weights(conv, alpha)
        val_curr = (np.log(sum_exp) + max_scaled) / alpha

        g = compute_gradient(h, w)

        # simple backtracking line‑search
        step_tmp = step
        improved = False
        for _ in range(10):
            h_new = project_capped_simplex(h - step_tmp * g, target_sum)
            conv_new = np.correlate(h_new, 1 - h_new, mode="full")
            _, max_scaled_new, sum_exp_new = compute_weights(conv_new, alpha)
            val_new = (np.log(sum_exp_new) + max_scaled_new) / alpha
            if val_new < val_curr - 1e-12:
                improved = True
                break
            step_tmp *= 0.5

        if not improved:
            step *= 0.5
            if step < 1e-8:
                break
            continue

        h = h_new
        step = step_tmp
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()

    return best_h, best_c5


def hill_climb(
    h0: np.ndarray,
    target_sum: float,
    max_moves: int,
    dx: float,
) -> tuple[np.ndarray, float]:
    """
    Random pairwise mass transfers (preserving sum) with a tiny annealing schedule.
    """
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    N = h.shape[0]

    for move in range(max_moves):
        i, j = np.random.choice(N, size=2, replace=False)
        max_delta = min(h[i], 1.0 - h[j])
        if max_delta <= 0.0:

            if temp > 0 and np.random.rand() < np.exp(-(c5_new - best_c5) / temp):

    return best_h, best_c5
