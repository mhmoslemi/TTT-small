import time
import numpy as np

def compute_c5(h):
    """Return C5 for a step function h on [0,2]."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * (2.0 / n)


def _project_sum(h, target):
    """Adjust h so that sum(h) == target while keeping 0<=h<=1."""
    # simple affine shift then clipping, repeated once if needed
    diff = target - h.sum()
    h += diff / h.size
    np.clip(h, 0.0, 1.0, out=h)
    # if clipping disturbed the sum, fix the remaining error on the first element
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        h[0] = np.clip(h[0] + diff2, 0.0, 1.0)
    return h


def _optimize_one(start_h, time_budget):
    """Simulated‑annealing style refinement of a single candidate."""
    n = start_h.shape[0]
    target_sum = n / 2.0
    h = start_h.copy()
    # enforce the sum constraint precisely at the start
    h = _project_sum(h, target_sum)

    cur_val = compute_c5(h)
    best_h = h.copy()
    best_val = cur_val

    start = time.time()
    # annealing schedule parameters
    T_start = 0.25
    T_end = 1e-4
    max_iters = 800000
    it = 0
    rng = np.random.default_rng()

    while True:
        if time.time() - start > time_budget:
            break
        # temperature decreasing linearly
        T = T_start * (1 - it / max_iters) + T_end * (it / max_iters)

        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # direction of transfer
        if rng.random() < 0.5:
            delta_max = min(h[i], 1.0 - h[j])
            if delta_max <= 0.0:
                it += 1
                continue
            delta = delta_max * rng.random()
            h_new = h.copy()
            h_new[i] -= delta
            h_new[j] += delta
        else:
            delta_max = min(h[j], 1.0 - h[i])
            if delta_max <= 0.0:
                it += 1
                continue
            delta = delta_max * rng.random()
            h_new = h.copy()
            h_new[j] -= delta
            h_new[i] += delta

        new_val = compute_c5(h_new)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            h = h_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h_new.copy()

        it += 1

    # final projection to guarantee exact sum
    best_h = _project_sum(best_h, target_sum)
    best_val = compute_c5(best_h)
    return best_h, best_val
