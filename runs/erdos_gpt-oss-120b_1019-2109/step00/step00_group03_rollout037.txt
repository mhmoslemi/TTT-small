def project_onto_bounds_and_sum(v, target, lower=0.0, upper=1.0, max_iter=60):
    lo = np.min(v) - upper  # guess lower bound
    hi = np.max(v) - lower   # guess upper bound
    # Expand bounds if needed
    for _ in range(max_iter):
        # compute sum at midpoints?
        # Actually binary search
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo+hi)/2.0, lower, upper)
    # Due to floating errors, may need to adjust small leftover
    # Optionally adjust a few entries to match target exactly:
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff uniformly among interior (non-bound elements)
        interior = (x > lower + 1e-12) & (x < upper - 1e-12)
        if interior.any():
            x[interior] += diff / interior.sum()
            x = np.clip(x, lower, upper)
    return x
```

def compute_grad_for_lag(h, L):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if L >= 0:
        # overlapping indices i = 0 .. n-1-L
        # term1 = 1 for i in that range
        # for j in overlap range:
        # grad[j] += 1
        # grad[j] -= h[j+L]
        # also for j >= L: grad[j] -= h[j-L]
        # implement vectorized
        overlap_len = n - L
        # term1
        grad[:overlap_len] += 1.0
        # subtract h[j+L]
        grad[:overlap_len] -= h[L:]  # h[j+L] for j in 0..overlap_len-1
        # subtract h[j-L] for j >= L
        grad[L:] -= h[:n - L]   # h[j-L] for j = L .. n-1
    else:
        p = -L
        overlap_start = p
        overlap_len = n - p
        # term1 = 1 for j >= p (i.e., indices overlap_start..n-1)
        grad[overlap_start:] += 1.0
        # subtract h[j+L] = h[j-p] for j >= p
        grad[overlap_start:] -= h[:n - p]   # h[j-p] corresponds to indices 0..n-p-1
        # subtract h[j-L] = h[j+p] for j <= n-1-p
        grad[:n-p] -= h[p:]   # h[j+p] for j=0..n-p-1
    return grad
```

import numpy as np
import time

def project_onto_bounds_and_sum(v, target, lower=0.0, upper=1.0, max_iter=60):
    lo = np.min(v) - upper
    hi = np.max(v) - lower
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, lower, upper)
    # small correction
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        interior = (x > lower + 1e-12) & (x < upper - 1e-12)
        if interior.any():
            x[interior] += diff / interior.sum()
            x = np.clip(x, lower, upper)
    return x

def compute_grad_for_lag(h, L):
    n = h.shape[0]
    grad = np.zeros_like(h)
    if L >= 0:
        overlap_len = n - L
        if overlap_len > 0:
            grad[:overlap_len] += 1.0
            grad[:overlap_len] -= h[L:]
            grad[L:] -= h[:overlap_len]
    else:
        p = -L
        overlap_len = n - p
        if overlap_len > 0:
            start = p
            grad[start:] += 1.0
            grad[start:] -= h[:n-p]  # h[j-p]
            grad[:n-p] -= h[p:]  # h[j+p]
    return grad

def max_corr_and_lags(h):
    # returns full correlation array and the max value and list of lag indices achieving max
    conv = np.correlate(h, 1 - h, mode='full')
    mx = conv.max()
    # compute lag indices where value is near max
    eps = 1e-12
    idxs = np.where(conv >= mx - eps)[0]
    return conv, mx, idxs

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    start = time.time()
    np.random.seed(seed)
    # pick resolution
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = None
    best_val = math.inf
    # generate initial vectors
    init_vectors = []
    # triangle shape (linear)
    x_center = (np.arange(n) + 0.5) * dx
    triangle = np.clip(1.0 - np.abs(x_center - 1.0), 0.0, 1.0)
    init_vectors.append(triangle)
    # uniform
    init_vectors.append(np.full(n, 0.5))
    # if initial_h_values exists, upsample it
    if 'initial_h_values' in globals():
        try:
            base = np.array(initial_h_values, dtype=float)
            m = base.shape[0]
            # original points at centers
            x0 = (np.arange(m) + 0.5) * (2.0 / m)
            init_up = np.interp(x_center, x0, base, left=0.0, right=0.0)
            init_vectors.append(init_up)
        except Exception:
            pass
    # random starts
    for _ in range(5):
        rand = np.random.rand(n)
        proj = project_onto_bounds_and_sum(rand, target_sum)
        init_vectors.append(proj)
    # main optimization loop
    # set some parameters
    max_total_iters = 200000  # safety upper bound
    it = 0
    # iterate over each start
    for init in init_vectors:
        h = project_onto_bounds_and_sum(init, target_sum)
        # evaluate initial
        conv, cur_max, _ = max_corr_and_lags(h)
        cur_val = cur_max * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()
        # inner loop
        # schedule: simple diminishing step size
        while True:
            if time.time() - start > budget_s * 0.95:
                break
            it += 1
            # compute correlation and gradients
            conv, cur_max, max_idxs = max_corr_and_lags(h)
            cur_val = cur_max * dx
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
            # compute aggregate gradient over all max lags
            grad = np.zeros_like(h)
            for idx in max_idxs:
                L = idx - (n - 1)
                grad += compute_grad_for_lag(h, L)
            grad /= len(max_idxs)
            # step size schedule: a/t^{0.5}
            alpha = 0.5 / math.sqrt(it + 1.0)
            # gradient descent step
            h_new = h - alpha * grad
            # project onto feasible set
            h = project_onto_bounds_and_sum(h_new, target_sum)
            # break condition after iteration limit per start
            if it >= max_total_iters:
                break
        if time.time() - start > budget_s * 0.95:
            break
    # final best computation (just in case)
    final_c5 = best_val
    # Ensure returned h meets constraints (clip)
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        # final projection to guarantee sum exactly target
        best_h = project_onto_bounds_and_sum(best_h, target_sum)
        final_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx
    else:
        # fallback: uniform
        best_h = np.full(n, 0.5)
        final_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx
    return (best_h, float(final_c5), n)

import time
import math
import numpy as np

def project_onto_bounds_and_sum(v, target, lower=0.0, upper=1.0, max_iter=60):
    """Project v onto the box [lower,upper] while enforcing sum(v)=target."""
    lo = np.min(v) - upper
    hi = np.max(v) - lower
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        x = np.clip(v - mid, lower, upper)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(v - (lo + hi) / 2.0, lower, upper)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        interior = (x > lower + 1e-12) & (x < upper - 1e-12)
        if interior.any():
            x[interior] += diff / interior.sum()
            x = np.clip(x, lower, upper)
    return x

def compute_grad_for_lag(h, L):
    """Subgradient of the overlap value for a given lag L."""
    n = h.shape[0]
    grad = np.zeros_like(h)
    if L >= 0:
        ov = n - L
        if ov > 0:
            grad[:ov] += 1.0                # +1 for the linear term
            grad[:ov] -= h[L:]              # -h[i+L]
            grad[L:]  -= h[:ov]              # -h[i-L]
    else:
        p = -L
        ov = n - p
        if ov > 0:
            start = p
            grad[start:] += 1.0             # +1 term
            grad[start:] -= h[:ov]          # -h[i+L] (i+L = i-p)
            grad[:ov]   -= h[p:]            # -h[i-L] (i-L = i+p)
    return grad

def max_corr_and_lags(h):
    """Return full correlation array, its maximum, and all lag indices attaining it."""
    conv = np.correlate(h, 1 - h, mode='full')
    mx = conv.max()
    eps = 1e-12
    idxs = np.where(conv >= mx - eps)[0]
    return conv, mx, idxs

def run(seed=42, budget_s=1100, **kwargs):
    """Search for a step function giving a low C₅ bound."""
    start_time = time.time()
    np.random.seed(seed)

    # resolution – 200 points gives a good balance between quality and speed
    n = 200
    dx = 2.0 / n
    target_sum = n / 2.0

    best_h = None
    best_val = math.inf

    # --- build a list of initial vectors ---------------------------------
    init_vectors = []

    # 1. triangular shape (linear rise to the centre)
    x_center = (np.arange(n) + 0.5) * dx
    triangle = np.clip(1.0 - np.abs(x_center - 1.0), 0.0, 1.0)
    init_vectors.append(triangle)

    # 2. sinusoidal shape (scaled sine, integral = 1)
    sine = (np.pi / 4.0) * np.sin(np.pi * x_center / 2.0)
    init_vectors.append(sine)

    # 3. constant 0.5 (feasible but not optimal, gives a baseline)
    init_vectors.append(np.full(n, 0.5))

    # 4. upsample provided initial_h_values if it exists
    if 'initial_h_values' in globals():
        try:
            base = np.asarray(initial_h_values, dtype=float)
            m = base.shape[0]
            x0 = (np.arange(m) + 0.5) * (2.0 / m)
            up = np.interp(x_center, x0, base, left=0.0, right=0.0)
            init_vectors.append(up)
        except Exception:
            pass

    # 5. a few random starts
    for _ in range(5):
        rand = np.random.rand(n)
        proj = project_onto_bounds_and_sum(rand, target_sum)
        init_vectors.append(proj)

    # --- main optimisation loop -----------------------------------------
    it = 0
    # safety cap on total iterations (will also stop on time limit)
    max_total_iters = 500000

    for init in init_vectors:
        h = project_onto_bounds_and_sum(init, target_sum)
        conv, cur_max, _ = max_corr_and_lags(h)
        cur_val = cur_max * dx
        if cur_val < best_val:
            best_val = cur_val
            best_h = h.copy()

        while True:
            # time guard
            if time.time() - start_time > budget_s * 0.95:
                break
            it += 1
            if it > max_total_iters:
                break

            conv, cur_max, max_idxs = max_corr_and_lags(h)
            cur_val = cur_max * dx
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()

            # aggregate subgradient over all maximal lags
            grad = np.zeros_like(h)
            for idx in max_idxs:
                L = idx - (n - 1)
                grad += compute_grad_for_lag(h, L)
            grad /= len(max_idxs)

            # step size schedule (simple diminishing step)
            alpha = 0.8 / math.sqrt(it + 1.0)
            h_new = h - alpha * grad
            h = project_onto_bounds_and_sum(h_new, target_sum)

        if time.time() - start_time > budget_s * 0.95:
            break

    # final projection to guarantee feasibility
    if best_h is not None:
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_onto_bounds_and_sum(best_h, target_sum)
        final_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx
    else:
        # fallback – uniform distribution
        best_h = np.full(n, 0.5)
        final_c5 = np.correlate(best_h, 1 - best_h, mode='full').max() * dx

    return (best_h, float(final_c5), n)
