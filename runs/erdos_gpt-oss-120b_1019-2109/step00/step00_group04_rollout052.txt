def simulated_annealing(h, max_time, start_time, rng):
    n = len(h)
    best_h = h.copy()
    best_val = compute_c(best_h)
    cur_val = best_val
    # Pre compute...
    while True:
        if time.time() - start_time > max_time:
            break
        # compute fraction
        elapsed = time.time() - start_time
        frac = elapsed / max_time
        step = max_step * (1 - frac) + min_step * frac
        T = T0 * (1 - frac) + T_min * frac
        # compute current shift of max correlation
        corr = np.correlate(h, 1 - h, mode='full')
        idx_max = np.argmax(corr)
        s = idx_max - (n - 1)
        # if random
        if rng.random() < 0.7:
            # targeted
            if s >= 0:
                i = rng.integers(0, n - s)
                j = i + s
            else:
                i = rng.integers(-s, n)
                j = i + s
            hi = h[i]; hj = h[j]
            if hi < hj:
                max_delta = min(step, 1 - hi, hj)
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                new_hi = hi + delta
                new_hj = hj - delta
            else:
                max_delta = min(step, hi, 1 - hj)
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                new_hi = hi - delta
                new_hj = hj + delta
            # store
            old_hi, old_hj = hi, hj
            h[i] = new_hi
            h[j] = new_hj
        else:
            # random pair move
            i = rng.integers(0, n)
            j = rng.integers(0, n-1)
            while j >= i:
                j = rng.integers(0, n)
            # ensure i!=j
            hi = h[i]; hj = h[j]
            max_pos = min(step, hi, 1.0 - hj) # move i->j
            max_neg = min(step, hj, 1.0 - hi) # move j->i
            total = max_pos + max_neg
            if total <= 0:
                continue
            delta = rng.random() * total - max_neg
            # apply
            # if delta>0: move from i to j
            # if delta<0: move from j to i (i.e., i gains -delta, j loses -delta)
            old_hi, old_hj = hi, hj
            if delta >= 0:
                h[i] = hi - delta
                h[j] = hj + delta
            else:
                h[i] = hi - delta  # subtract negative -> add
                h[j] = hj + delta
        # compute new value
        new_val = compute_c(h)
        d = new_val - cur_val
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):

            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # revert
            h[i] = old_hi
            h[j] = old_hj

import time
import numpy as np

def _compute_c(h):
    """Return the Erdős overlap constant for a step vector h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) gives the numerator of the integral
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _simulated_annealing(h_init, time_budget, start_time, rng):
    """Simple SA that respects 0≤h≤1 and Σh = n/2."""
    n = h_init.shape[0]
    target_sum = n / 2.0
    # parameters (tuned for speed vs. quality)
    max_step = 0.30
    min_step = 0.01
    T0 = 0.02
    T_min = 1e-6

    h = h_init.copy()
    best_h = h.copy()
    best_val = _compute_c(best_h)
    cur_val = best_val

    # pre‑compute a mask of indices for fast random pair generation
    while True:
        if time.time() - start_time > time_budget:
            break

        elapsed = time.time() - start_time
        frac = elapsed / time_budget
        step = max_step * (1.0 - frac) + min_step * frac
        T = T0 * (1.0 - frac) + T_min * frac

        # locate shift giving the current worst (largest) overlap integral
        corr_full = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(np.argmax(corr_full))
        shift = idx_max - (n - 1)   # integer shift in index units

        # ----- propose a move -------------------------------------------------
        if rng.random() < 0.7:   # targeted move aimed at the worst shift
            if shift >= 0:
                i = rng.integers(0, n - shift)
                j = i + shift
            else:
                i = rng.integers(-shift, n)
                j = i + shift

            hi, hj = h[i], h[j]
            # move mass from the larger entry to the smaller one (increase product)
            if hi < hj:
                max_delta = min(step, 1.0 - hi, hj)
                if max_delta <= 0.0:
                    continue
                delta = rng.random() * max_delta
                new_hi = hi + delta
                new_hj = hj - delta
            else:
                max_delta = min(step, hi, 1.0 - hj)
                if max_delta <= 0.0:
                    continue
                delta = rng.random() * max_delta
                new_hi = hi - delta
                new_hj = hj + delta

            old_hi, old_hj = hi, hj
            h[i], h[j] = new_hi, new_hj

        else:                     # generic random redistribution between two sites
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1                       # guarantee i != j
            hi, hj = h[i], h[j]

            max_pos = min(step, hi, 1.0 - hj)   # move from i to j
            max_neg = min(step, hj, 1.0 - hi)   # move from j to i
            total = max_pos + max_neg
            if total <= 0.0:
                continue
            delta = rng.random() * total - max_neg
            old_hi, old_hj = hi, hj
            if delta >= 0.0:
                h[i] = hi - delta
                h[j] = hj + delta
            else:
                h[i] = hi - delta      # delta<0 adds to hi
                h[j] = hj + delta

        # ----- evaluate -------------------------------------------------------
        new_val = _compute_c(h)
        d = new_val - cur_val
        if d < 0.0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            # accept
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        else:
            # reject – revert
            h[i] = old_hi
            h[j] = old_hj

    # final sanity: enforce exact sum (tiny numerical drift)
    diff = best_h.sum() - target_sum
    if abs(diff) > 1e-12:
        best_h -= diff / n
        best_h = np.clip(best_h, 0.0, 1.0)
    return best_h, best_val
