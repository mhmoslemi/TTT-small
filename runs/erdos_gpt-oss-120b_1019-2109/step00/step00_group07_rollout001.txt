import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper functions (top level, no closures)
# -------------------------------------------------------------
def max_overlap(h: np.ndarray) -> float:
    """C5 value: max_k ∫ h(x)(1−h(x+k))dx approximated on the grid."""
    n = h.shape[0]
    dx = 2.0 / n
    # linear correlation (zero padding outside the interval)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def worst_shift_and_value(h: np.ndarray):
    """Return shift (as integer offset) that attains the maximal overlap
    and the corresponding value."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(corr))
    val = float(corr[idx] * (2.0 / n))
    shift = idx - (n - 1)          # convert index to signed shift
    return shift, val


def gradient_for_shift(h: np.ndarray, shift: int) -> np.ndarray:
    """
    Gradient of the overlap for a fixed shift.
    For each i:
        d/ dh_i  [∑ h_i - ∑ h_i h_{i+shift}] = 1 - h_{i+shift} - h_{i-shift}
    (terms omitted when the index falls outside the interval).
    """
    n = h.shape[0]
    g = np.ones_like(h)
    if shift >= 0:
        if shift < n:
            # part where i+shift is inside
            g[: n - shift] -= h[shift:]
            # part where i-shift is inside
            g[shift:] -= h[: n - shift]
    else:  # shift < 0
        s = -shift
        if s < n:
            g[s:] -= h[: n - s]
            g[: n - s] -= h[s:]
    return g


def project_to_sum_bounds(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Project vector v onto the set {x | 0 ≤ x_i ≤ 1, sum x_i = target_sum}.
    This is a simple water‑filling binary search for the Lagrange multiplier λ:
        x_i = clip(v_i - λ, 0, 1)
    """
    lo, hi = -2.0, 2.0            # wide enough interval for λ
    for _ in range(60):           # enough for double precision
        lam = (lo + hi) * 0.5
        w = np.clip(v - lam, 0.0, 1.0)
        s = w.sum()
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)

    """Generate a random feasible vector (values in [0,1], sum = n/2)."""
    v = np.random.rand(n).astype(np.float64)
    return project_to_sum_bounds(v, n / 2.0)

            # occasional uphill moves (simulated‑annealing style)
            temp = max(0.001, 0.02 * (1.0 - t_frac))
            if np.random.rand() < math.exp(-(cand_score - cur_score) / temp):
                cur_h, cur_score = cand, cand_score
                no_improve = 0
            else:
                no_improve += 1

        # record the best overall solution
        if cur_score < best_score:
            best_h, best_score = cur_h.copy(), cur_score
            no_improve = 0
            # stop early if we have already beaten the target bound
            if best_score <= 0.3808:
                break

        # random restart if stagnation persists
        if no_improve > 5000:
            cur_h = random_feasible(n)
            cur_score = max_overlap(cur_h)
            no_improve = 0

        it += 1

    # -----------------------------------------------------------------
    # Final sanitisation of the best vector (tiny numerical drift)
    # -----------------------------------------------------------------
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_to_sum_bounds(best_h, target_sum)
    final_score = max_overlap(best_h)

    return best_h.tolist(), float(final_score), n
