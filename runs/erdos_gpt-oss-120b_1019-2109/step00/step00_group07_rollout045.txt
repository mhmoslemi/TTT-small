    import time, numpy as np, random, math
    start = time.time()

    # Determine n_points & init vector

        n = 80
        init = np.zeros(n)
    dx = 2.0 / n

    def calc_c5(vec):

        return np.max(np.correlate(vec, 1-vec, mode='full')) * dx
    best = b.copy()
    best_c5 = calc_c5(best.astype(float))
    curr = best.copy()
    curr_c5 = best_c5
    # Simulated annealing parameters
    max_iter = 1000000  # will break on time
    # temperature schedule: start 0.5 * best_c5 maybe; end near 1e-6

    # start time loops

    while True:

        T = T0 * (1 - fraction) + Tmin * fraction
        # propose a swap: pick random one index and zero index

        cand[i] = 0
        cand[j] = 1
        cand_c5 = calc_c5(cand.astype(float))
        # Accept?
        delta = cand_c5 - curr_c5
        if delta < 0 or random.random() < math.exp(-delta / (T+1e-12)):

def run(seed=42, budget_s=1100, **kwargs):
    import time, math, random
    import numpy as np

    start = time.time()
    random.seed(seed)
    np.random.seed(seed)

    # ----- problem size -------------------------------------------------
    # Use the provided initial vector if it exists, otherwise choose a moderate size.
    init_vec = globals().get('initial_h_values')
    if init_vec is not None:
        n = len(init_vec)
        # ensure even length (required for the exact half‑mass constraint)
        if n % 2:
            n += 1
    else:
        n = 120                     # a convenient even size when no parent is given
    if n % 2:
        n += 1
    dx = 2.0 / n                    # step length, used for scaling the integral

    # ----- initial binary construction -----------------------------------
    if init_vec is not None and len(init_vec) == n:
        # set the n/2 largest entries to 1, the rest to 0
        order = np.argsort(-np.array(init_vec, dtype=float))
        h = np.zeros(n, dtype=int)
        h[order[: n // 2]] = 1
    else:
        # random half‑filled binary vector
        h = np.zeros(n, dtype=int)
        h[: n // 2] = 1
        np.random.shuffle(h)

    # keep explicit index lists for fast random access
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))

    # ----- evaluation ----------------------------------------------------
    def compute_c5(arr):
        """Return (C5, full correlation array) for a 0/1 array."""
        corr = np.correlate(arr, 1 - arr, mode='full')
        return corr.max() * dx, corr

    best_c5, curr_corr = compute_c5(h)
    best_h = h.copy()
    curr_c5 = best_c5

    # target bound – stop early if we achieve it
    TARGET = 0.3808

    # ----- simulated annealing parameters --------------------------------
    p_targeted = 0.3                # probability of using a targeted swap
    T0 = 0.04                       # initial temperature (scale of C5 differences)
    Tmin = 1e-6                     # final temperature

    # ----- main optimisation loop ----------------------------------------
    while True:
        now = time.time()
        elapsed = now - start
        if elapsed > budget_s - 0.5:               # leave a little margin
            break

        # linear cooling schedule
        frac = elapsed / budget_s
        T = T0 * (1 - frac) + Tmin * frac

        # shift giving the current worst overlap
        idx_max = int(np.argmax(curr_corr))
        shift = idx_max - (n - 1)                  # shift = lag – (n‑1)

        # ----- choose indices to swap ------------------------------------
        if random.random() < p_targeted:
            # try a swap that directly reduces the worst shift
            if shift >= 0:
                i_range = np.arange(n - shift)
                hi = h[i_range]
                hj = h[i_range + shift]
            else:
                i_range = np.arange(-shift, n)
                hi = h[i_range]
                hj = h[i_range + shift]
            mism1 = i_range[(hi == 1) & (hj == 0)]
            mism0 = i_range[(hi == 0) & (hj == 1)]
            if mism1.size and mism0.size:
                i = int(random.choice(mism1))
                j = int(random.choice(mism0))
            else:
                i = random.choice(ones)
                j = random.choice(zeros)
        else:
            i = random.choice(ones)
            j = random.choice(zeros)

        # ----- try the swap (temporary) ----------------------------------
        h[i], h[j] = h[j], h[i]                    # perform swap
        new_c5, new_corr = compute_c5(h)
        delta = new_c5 - curr_c5

        # ----- Metropolis acceptance --------------------------------------
        if delta < 0 or random.random() < math.exp(-delta / (T + 1e-15)):
            # keep the new state
            curr_c5 = new_c5
            curr_corr = new_corr
            # update the index lists
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            # record best solution
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                if best_c5 <= TARGET:
                    break
        else:
            # reject – revert the swap
            h[i], h[j] = h[j], h[i]

    # ----- optional greedy hill‑climbing ---------------------------------
    # try a few deterministic improvements on the best vector found
    h = best_h.copy()
    ones = list(np.flatnonzero(h))
    zeros = list(np.flatnonzero(1 - h))
    best_c5, _ = compute_c5(h)

    improving = True
    while improving and (time.time() - start) < budget_s - 0.5:
        improving = False
        for _ in range(200):                     # limited attempts per pass
            i = random.choice(ones)
            j = random.choice(zeros)
            h[i], h[j] = h[j], h[i]
            new_c5, _ = compute_c5(h)
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_h = h.copy()
                # update lists accordingly
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                improving = True
                break
            else:
                h[i], h[j] = h[j], h[i]

    # ----- final result --------------------------------------------------
    h_float = best_h.astype(float)   # step function values in [0,1]
    return (h_float, float(best_c5), n)
