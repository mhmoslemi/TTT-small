import numpy as np, time

def project_to_sum_box(v, target_sum):
    # binary search lambda
    low = -np.max(v) # ensures all v+lambda <= 0 -> sum=0
    high = 1 - np.min(v) # ensures all v+lambda >=1 -> sum=n
    for _ in range(40):
        mid = (low + high) / 2
        clipped = np.clip(v + mid, 0, 1)
        s = clipped.sum()
        if s < target_sum:
            low = mid
        else:
            high = mid
    return np.clip(v + (low+high)/2, 0, 1)

def compute_max_corr(h):
    return np.max(np.correlate(h, 1 - h, mode='full'))

def gradient_step(h, shift):
    n = len(h)
    grad = np.zeros_like(h)
    if shift >= 0:
        # i from 0 to n-1-shift: term (1 - h[i+shift])
        grad[:n-shift] += 1 - h[shift:]
        # i from shift to n-1: term -h[i-shift]
        grad[shift:] -= h[:n-shift]
    else:
        s = -shift
        # shift negative: effectively shift left
        # we can handle by symmetry: gradient for -s = -gradient for s with reversed indices?
        # Derive: C_{-s} = sum_i h_i * (1 - h_{i-s}) where i-s >=0.
        # So same as for positive shift but with indices reversed.
        # For shift = -s:
        # for i from s to n-1: term (1 - h[i-s])
        # for i from 0 to n-1-s: term -h[i+s]
        grad[s:] += 1 - h[:n-s]
        grad[:n-s] -= h[s:]
    return grad
```

def project_to_sum_box(v, target):
    # Use bisection to find lambda such that sum_i clip(v_i + lam, 0, 1) == target
    lo = -np.max(v) - 1e-12
    hi = 1 - np.min(v) + 1e-12
    for _ in range(60):
        lam = (lo + hi)/2
        clipped = np.clip(v + lam, 0, 1)
        total = clipped.sum()
        if total < target:
            lo = lam
        else:
            hi = lam
    return np.clip(v + (lo+hi)/2, 0, 1)

        for _ in range(60):                     # bisection on the Lagrange multiplier
            mid = (lo + hi) * 0.5
            clipped = np.clip(v + mid, 0.0, 1.0)
            s = clipped.sum()
            if s < target_sum:
                lo = mid
            else:
                hi = mid
        out = np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)
        # tiny correction to enforce the sum exactly (numerical safety)
        diff = target_sum - out.sum()
        if abs(diff) > 1e-12:
            free = (out > 0.0) & (out < 1.0)
            if free.any():
                idx = np.flatnonzero(free)[0]
                out[idx] = np.clip(out[idx] + diff, 0.0, 1.0)
            else:
                out[0] = np.clip(out[0] + diff, 0.0, 1.0)
        return out

    h = project(h)

    # ----------------------------------------------------------------------
    # Utility functions
    # ----------------------------------------------------------------------
    def max_corr(vec: np.ndarray) -> float:
        """Maximum of the overlap integral (without dx factor)."""
        return float(np.max(np.correlate(vec, 1.0 - vec, mode='full')))

    # ----------------------------------------------------------------------
    # Main optimisation loop – projected gradient with occasional random moves
    # ----------------------------------------------------------------------
    best_h = h.copy()
    best_corr = max_corr(best_h)
    start_time = time.time()
    iter_cnt = 0
    no_improve = 0
    idx_all = np.arange(n)

    while time.time() - start_time < budget_s - 0.5:
        # current correlation profile
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = float(corr.max())

        # collect all shifts that are essentially at the max (equioscillation)
        tol = max(1e-12, 0.001 * cur_max)
        near_idx = np.where(corr >= cur_max - tol)[0]

        # accumulated gradient over the near‑max shifts
        grad = np.zeros_like(h)
        for ix in near_idx:
            shift = int(ix - (n - 1))
            i = idx_all
            i_shift = i + shift
            mask = (i_shift >= 0) & (i_shift < n)
            grad[mask] += 1.0 - h[i_shift[mask]]
            i_minus = i - shift
            mask2 = (i_minus >= 0) & (i_minus < n)
            grad[mask2] -= h[i_minus[mask2]]
        if near_idx.size:
            grad *= (1.0 / near_idx.size)

        # step size slowly decays
        alpha = 0.5 / (1.0 + iter_cnt * 0.001)
        v = h - alpha * grad
        h_new = project(v)

        new_max = max_corr(h_new)

        # accept improvements (or a tiny chance of uphill move)
        if new_max < cur_max or rng.random() < 0.002:
            h = h_new
            if new_max < best_corr:
                best_corr = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            # occasional stochastic perturbation to escape plateaus
            if rng.random() < 0.001:
                h = best_h + rng.normal(scale=0.05, size=n)
                h = np.clip(h, 0.0, 1.0)
                h = project(h)

        # hard restart if stuck for a while
        if no_improve > 5000:
            h = best_h + rng.normal(scale=0.2, size=n)
            h = np.clip(h, 0.0, 1.0)
            h = project(h)
            no_improve = 0

        iter_cnt += 1

    # ----------------------------------------------------------------------
    # Final local refinement – pairwise swaps that preserve sum exactly
    # ----------------------------------------------------------------------
    h_refine = best_h.copy()
    cur_best = best_corr
    for _ in range(20000):
        i, j = rng.choice(n, size=2, replace=False)
        # allowable delta so that both stay in [0,1]
        max_delta = min(1.0 - h_refine[i], h_refine[j])
        min_delta = -min(h_refine[i], 1.0 - h_refine[j])
        if max_delta <= min_delta:
            continue
        delta = rng.uniform(min_delta, max_delta)
        cand = h_refine.copy()
        cand[i] += delta
        cand[j] -= delta
        cand_corr = max_corr(cand)
        if cand_corr < cur_best:
            h_refine = cand
            cur_best = cand_corr
            best_h = cand.copy()
            best_corr = cand_corr

    # ----------------------------------------------------------------------
    # Return result
    # ----------------------------------------------------------------------
    final_c5 = best_corr * (2.0 / n)   # incorporate dx = 2/n
    return best_h.tolist(), final_c5, n
```
