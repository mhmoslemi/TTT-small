def project_onto_box_simplex(v, s):
    # v is array of any real numbers
    n = len(v)
    # Clip lower bound: v = np.maximum(v,0)
    # We'll treat algorithm: solve for lambda such that sum_i min(1, max(0, v_i + lambda)) = s.
    # We can find lambda via bisection.
    lo = -np.max(v) - 1  # low enough
    hi = 1 - np.min(v) + 1
    for _ in range(50):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0, 1)
        total = x.sum()
        if total > s:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo+hi)/2, 0, 1)

def project_to_box_simplex(v, s):
    """
    Project vector v onto the set {x: 0 <= x <= 1, sum x = s}
    using bisection on Lagrange multiplier.
    """
    lo = -np.max(v) - 1.0  # ensure all v+lam <= 0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        lam = (lo + hi) / 2.0
        x = np.clip(v + lam, 0.0, 1.0)
        total = x.sum()
        if total > s:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

import time
import math
import random
import numpy as np

# ------------------------------------------------------------
# Helper: project a vector onto the box [0,1]^n intersect {sum = s}
# ------------------------------------------------------------
def _project_to_box_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Find x = clip(v + lam, 0, 1) such that sum(x) = s.
    Uses a simple bisection on the Lagrange multiplier lam.
    """
    lo = -1.0e6
    hi = 1.0e6
    # 60 iterations are enough for double precision
    for _ in range(60):
        lam = (lo + hi) * 0.5
        x = np.clip(v + lam, 0.0, 1.0)
        total = x.sum()
        if total > s:
            hi = lam
        else:
            lo = lam
    return np.clip(v + (lo + hi) * 0.5, 0.0, 1.0)


# ------------------------------------------------------------
# Helper: compute C5 for a given step function h
# ------------------------------------------------------------
def _c5_value(h: np.ndarray) -> float:
    n = h.shape[0]
    # correlation of h with (1-h) over the linear (zero‑padded) domain
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


# ------------------------------------------------------------
# Core optimisation routine for a single length n
# ------------------------------------------------------------
def _optimise_length(
    n: int,
    init_h: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Return (best_h, best_score) for a given n within time_limit seconds."""
    target_sum = n / 2.0
    h = init_h.copy()
    best_h = h.copy()
    best_score = _c5_value(h)

    # Basic parameters
    max_no_improve = 40000  # restart after this many useless steps
    steps_since_improve = 0
    start = time.time()

    while time.time() - start < time_limit:
        # Periodically restart if we are stuck
        if steps_since_improve > max_no_improve:
            h = _project_to_box_simplex(rng.random(n), target_sum)
            cur_score = _c5_value(h)
            steps_since_improve = 0
            continue

        # ----------------------------------------------------
        # Determine worst lag (the one giving the current max)
        # ----------------------------------------------------
        corr = np.correlate(h, 1.0 - h, mode="full")
        cur_score = corr.max() * (2.0 / n)
        lag = int(corr.argmax()) - (n - 1)   # lag ∈ [-(n-1), n-1]

        # ----------------------------------------------------
        # Propose a move that directly attacks the worst lag.
        # ----------------------------------------------------
        if lag >= 0:
            # contributions = h[i] * (1 - h[i+lag])
            length = n - lag
            hi = h[:length]
            hj = h[lag:]
            contrib = hi * (1.0 - hj)
            i = int(contrib.argmax())
            j = i + lag
        else:
            # lag < 0 : shift to the left
            L = -lag
            length = n - L
            hi = h[L:]                # later index
            hj = h[:length]           # earlier index
            contrib = hi * (1.0 - hj)
            i = int(contrib.argmax()) + L
            j = i - L

        # Move mass from i (the factor h_i) to j (the factor reducing 1‑h_j)
        delta_max = min(h[i], 1.0 - h[j])
        if delta_max <= 0.0:
            # fallback to a random cheap move
            i, j = rng.choice(n, size=2, replace=False)
            # feasible interval for delta so that both stay in [0,1]
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if hi <= lo:
                steps_since_improve += 1
                continue
            delta = rng.uniform(lo, hi)
        else:
            # Take a random fraction of the allowed transfer (avoid too tiny steps)
            delta = delta_max * rng.random() * 0.5 + delta_max * 0.1 * rng.random()
        # Apply move
        h[i] += delta
        h[j] -= delta

        new_score = _c5_value(h)

        if new_score < cur_score - 1e-12:
            # Accept improvement
            cur_score = new_score
            steps_since_improve = 0
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_h = h.copy()
        else:
            # reject – revert
            h[i] -= delta
            h[j] += delta
            steps_since_improve += 1

    # Ensure the best solution satisfies constraints exactly
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = _project_to_box_simplex(best_h, target_sum)
    best_score = _c5_value(best_h)
    return best_h, float(best_score)


# ------------------------------------------------------------
# Main entry point required by the harness
# ------------------------------------------------------------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points).

    h_values : numpy array of shape (n_points,) with entries in [0,1] and sum = n_points/2.
    c5_bound : the achieved value of C5 (the maximum overlap integral).
    n_points : length of the discretisation.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    total_budget = float(budget_s)

    # ----------------------------------------------------------------
    # Assemble a collection of candidate discretisation sizes.
    # ----------------------------------------------------------------
    candidates = set()
    if "initial_h_values" in globals():
        candidates.add(len(initial_h_values))
    # add a few even lengths (binary friendly) and a couple odd ones
    for n in (40, 44, 48, 56, 64, 80, 96, 112):
        candidates.add(n)
    candidates = sorted(candidates)

    # Allocate most of the budget to the search; keep a tiny safety margin.
    search_budget = total_budget * 0.93
    per_n_budget = search_budget / max(1, len(candidates))

    best_overall_h = None
    best_overall_score = float("inf")
    best_overall_n = None

    # ----------------------------------------------------------------
    # Loop over each candidate length and run a dedicated local search.
    # ----------------------------------------------------------------
    for n in candidates:
        # Time slice for this length
        remaining = start_time + total_budget - time.time()
        if remaining <= 0.0:
            break
        this_budget = min(per_n_budget, remaining * 0.9)

        # ----- Build an initial vector for this length -----
        if "initial_h_values" in globals() and n == len(initial_h_values):
            init = np.asarray(initial_h_values, dtype=np.float64).copy()
            # re‑project to satisfy the exact sum constraint
            target = n / 2.0
            init = _project_to_box_simplex(init, target)
        else:
            # Start from a random point on the bounded simplex
            target = n / 2.0
            init = _project_to_box_simplex(rng.random(n), target)

        # ----- Optimise for this length -----
        h_opt, score_opt = _optimise_length(n, init, this_budget, rng)

        # ----- Keep the best across all lengths -----
        if score_opt < best_overall_score:
            best_overall_score = score_opt
            best_overall_h = h_opt.copy()
            best_overall_n = n

    # ----------------------------------------------------------------
    # Final sanity check – ensure constraints are satisfied exactly.
    # ----------------------------------------------------------------
    if best_overall_h is None:
        # Fallback: a trivial feasible vector (n=2)
        best_overall_n = 2
        best_overall_h = np.array([1.0, 0.0], dtype=np.float64)
        best_overall_score = _c5_value(best_overall_h)

    best_overall_h = np.clip(best_overall_h, 0.0, 1.0)
    best_overall_h = _project_to_box_simplex(best_overall_h, best_overall_n / 2.0)
    final_score = _c5_value(best_overall_h)

    return (best_overall_h, float(final_score), int(best_overall_n))
