def _project_onto_box_sum(v, target):
    # Solve min ||x - v||^2 s.t. 0 <= x <= 1 and sum x = target.
    # Use lambda water-filling via bisection
    low = -1.0
    high = 1.0
    # Expand bounds adaptively
    # Actually we need to cover lambda range. Since x_i = clip(v_i - λ, 0,1).
    # So λ can be as low as min(v_i) - 1 (if we need to clip at upper bound 1).
    # And as high as max(v_i) (if we need lower bound 0).
    low = min(v) - 1.0
    high = max(v)
    # Ensure any lambda beyond give sum below/above target
    # We'll do binary search for 60 iter
    for _ in range(60):
        mid = (low + high) / 2.0
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(v - (low + high) / 2.0, 0.0, 1.0)
    # Because of numeric errors, adjust diff
    diff = target - x.sum()
    if abs(diff) > 1e-8:
        # distribute diff among non-bound indices
        free = (x > 1e-12) & (x < 1 - 1e-12)
        if free.any():
            # add equal share
            add = diff / free.sum()
            x[free] += add
            # clip again (just in case)
            x = np.clip(x, 0.0, 1.0)
    # final clip again to ensure bounds
    return x

    h = h0.copy()
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c = _evaluate_c5(h)
    # parameters
    step = 0.5 # may adjust
    alpha = 8.0
    # iteration counters
    it = 0
    patience = 0
    while True:
        if time.time() - start_time >= max_seconds:
            break
        # compute correlation vector
        c = np.correlate(h, 1 - h, mode='full')
        max_c = np.max(c)
        # compute weights for log-sum-exp
        # To keep numerical stable:
        exp_vals = np.exp(alpha * (c - max_c))
        w = exp_vals / np.sum(exp_vals)
        grad = np.zeros_like(h)
        # compute gradient contributions
        # vectorization using loops per shift
        # we will access w via index offset.
        for offset in range(-n+1, n):
            w_s = w[offset + n - 1]
            L = offset
            if L >= 0:
                if L < n:  # ensures ranges exist
                    grad[: n - L] += w_s * (1 - h[L: n])
                if L > 0:
                    grad[L:] -= w_s * h[: n - L]
            else:
                Lp = -L
                grad[Lp:] += w_s * (1 - h[: n - Lp])
                grad[: n - Lp] -= w_s * h[Lp:]
        # project gradient to zero-sum subspace
        grad -= np.mean(grad)

        # gradient step
        y = h - step * grad
        # projection to feasible set
        h_new = _project_onto_box_sum(y, target_sum)

        new_c = _evaluate_c5(h_new)

        if new_c < best_c - 1e-10:
            best_c = new_c
            best_h = h_new.copy()
            h = h_new

            step = min(step * 1.1, 2.0)
            patience = 0

            patience += 1

            # Occasionally random perturb

        it += 1

        if step < 1e-6:
            # maybe break if step too small
            if patience > 500:
                break
    # final best
    return best_h, best_c

def _evaluate_c5(h):
    n = h.shape[0]
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)


def _project_onto_box_sum(v, target):
    """ Euclidean projection of v onto {0<=x<=1, sum(x)=target}. """
    low = np.min(v) - 1.0
    high = np.max(v)
    for _ in range(60):
        mid = (low + high) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            low = mid
        else:
            high = mid
    x = np.clip(v - (low + high) * 0.5, 0.0, 1.0)
    # tiny correction for numerical drift
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            add = diff / free.sum()
            x[free] += add
            x = np.clip(x, 0.0, 1.0)
    return x


def _smooth_weights(corr, alpha):
    """log‑sum‑exp weights for a smooth max."""
    m = np.max(corr)
    w = np.exp(alpha * (corr - m))
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _gradient(h, alpha):
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    w = _smooth_weights(corr, alpha)

    grad = np.zeros_like(h)
    # contribution of each lag
    for off in range(-n + 1, n):
        ws = w[off + n - 1]
        if off >= 0:
            L = off
            if L < n:
                grad[: n - L] += ws * (1.0 - h[L:])
            if L > 0:
                grad[L:] -= ws * h[: n - L]
        else:
            L = -off
            grad[L:] += ws * (1.0 - h[: n - L])
            grad[: n - L] -= ws * h[L:]
    # enforce zero‑sum direction (preserve total mass)
    grad -= grad.mean()
    return grad


def _optimize_h(h0, max_seconds, start_time):
    """Projected gradient descent with occasional random perturbations."""
    h = h0.copy()
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c = _evaluate_c5(h)

    step = 0.5
    alpha = 8.0
    patience = 0
    itr = 0

    while time.time() - start_time < max_seconds:
        grad = _gradient(h, alpha)
        y = h - step * grad
        h_new = _project_onto_box_sum(y, target)
        c_new = _evaluate_c5(h_new)

        if c_new < best_c - 1e-12:
            best_c, best_h = c_new, h_new.copy()
            h = h_new
            step = min(step * 1.1, 2.0)
            patience = 0
        else:
            patience += 1
            step *= 0.9
            if patience > 200:
                # random rescue around current best
                cand = best_h + 0.1 * np.random.randn(n)
                cand = np.clip(cand, 0.0, 1.0)
                h = _project_onto_box_sum(cand, target)
                patience = 0
                step = 0.5

        # slowly tighten the smoothness
        if itr % 100 == 0 and alpha < 30.0:
            alpha = min(alpha * 1.1, 30.0)
        itr += 1
        if step < 1e-6 and patience > 500:
            break
    return best_h, best_c


def _refine_local(h, steps=2000):
    """simple pair‑wise mass‑transfer hill climbing."""
    n = h.shape[0]
    target = n / 2.0
    best_h = h.copy()
    best_c = _evaluate_c5(best_h)
    for _ in range(steps):
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move a small amount from i to j (keeping sum)
        delta = min(best_h[i], 1.0 - best_h[j])
        if delta <= 0:

        delta *= np.random.rand() * 0.5
        cand = best_h.copy()
        cand[i] -= delta
        cand[j] += delta
        c = _evaluate_c5(cand)
        if c < best_c - 1e-12:
            best_h, best_c = cand, c
    return best_h, best_c
