import numpy as np, time, math, random

def _compute_c5(arr):
    """Return max_k ∫ h(x)(1-h(x+k)) dx for a uniform grid."""
    n = arr.size
    dx = 2.0 / n
    # correlation of h with (1-h)
    corr = np.correlate(arr, 1.0 - arr, mode="full") * dx
    return corr.max()

def _init_h(n, seed):
    """Generate a feasible starting vector (sum = n/2, 0≤h≤1)."""
    rng = np.random.RandomState(seed)
    # start from a binary vector with exactly n/2 ones
    h = np.zeros(n, dtype=float)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    # introduce a small amount of fractional jitter
    eps = 0.05
    for _ in range(3 * n):
        i, j = rng.choice(n, 2, replace=False)
        low = max(-h[i], h[j] - 1.0)
        high = min(1.0 - h[i], h[j])
        if high <= low:
            continue
        delta = eps * (high - low) * rng.rand() + low
        h[i] += delta
        h[j] -= delta
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """Return a step function h that (approximately) minimizes C5."""
    rng = np.random.RandomState(seed)
    # ------------------------------------------------------------------
    # initialise ---------------------------------------------------------
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=float)

    else:
        n = 120                         # a modest size when no parent is present
        h = _init_h(n, seed)

    target_sum = n / 2.0                # ∫h = 1  ⇒  sum(h)=n/2
    # enforce the sum exactly (tiny numerical drift may appear)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        # shift a single non‑boundary element to fix the sum
        for idx in range(n):
            if 0.0 <= h[idx] + diff <= 1.0:
                h[idx] += diff
                break

    best_h = h.copy()
    best_val = _compute_c5(best_h)

    # ------------------------------------------------------------------
    # optimisation loop -------------------------------------------------
    start = time.time()
    max_time = budget_s * 0.95          # keep a safety margin
    step = 0.4                          # maximal amount transferred per move
    it = 0
    no_improve = 0

    while time.time() - start < max_time:
        # current objective and the worst shift
        cur_val = _compute_c5(h)
        if cur_val < best_val - 1e-12:
            best_val, best_h = cur_val, h.copy()
            no_improve = 0
        else:
            no_improve += 1

        # gradient of the worst shift (pairwise transfer direction)
        corr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        k_idx = int(np.argmax(corr))
        k = k_idx - (n - 1)                       # signed shift

        # neighbour sum for this shift
        neigh = np.zeros_like(h)
        if k >= 0:
            if k < n:
                neigh[: n - k] += h[k:]
                neigh[k:]      += h[: n - k]
        else:
            kp = -k
            if kp < n:
                neigh[kp:]     += h[: n - kp]
                neigh[: n - kp] += h[kp:]

        # gradient of C(k) = dx*(1 - neigh)
        dx = 2.0 / n
        grad = dx * (1.0 - neigh)

        # pick indices with opposite sign gradients
        i_pos = int(np.argmax(grad))   # most positive → should decrease
