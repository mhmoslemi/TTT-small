import numpy as np, time, random

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    return max_val * dx

def simulated_annealing_binary(h, time_budget, start_time):
    n = len(h)
    ones = np.flatnonzero(h > 0.5) # indices where h == 1
    zeros = np.flatnonzero(h <= 0.5)
    curr_c5 = compute_c5(h)
    best_c5 = curr_c5
    best_h = h.copy()
    # temperature schedule: start high reduce linearly
    T0 = 0.1
    # param: iterations per temperature step maybe adapt.
    it = 0
    while True:
        t_now = time.time()
        if t_now - start_time >= time_budget:
            break
        # propose swap
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # swap
        h[i], h[j] = h[j], h[i]
        # update ones, zeros sets
        # after swap: i becomes 0, j becomes 1
        # We'll update arrays:
        # but removal/insertion for many updates may be expensive; we can just recompute on each iteration new ones and zeros via condition using array; but that will be O(n) each iteration. Maybe okay for n=200.
        # Simpler: after swap, recompute arrays:
        # but O(n) each iteration (200) negligible.
        curr_c5_new = compute_c5(h)
        delta = curr_c5_new - curr_c5
        # compute temperature based on progress
        elapsed = t_now - start_time
        T = T0 * (1 - elapsed / time_budget)
        if delta <= 0 or np.random.rand() < np.exp(-delta / (T + 1e-9)):
            # accept
            curr_c5 = curr_c5_new
            if curr_c5 < best_c5:

            h[i], h[j] = h[j], h[i]
        # update ones zeros using condition
        # next iteration compute new indices
        it += 1
        # we could also occasionally shuffle to avoid stagnation
    return best_h, best_c5

def simulated_annealing_fractional(h, time_budget, start_time):
    # h is float vector with sum fixed

        max_add = 1 - h[j]  # can't exceed 1

        else:
            # move from j to i
            max_sub2 = h[j]
            max_add2 = 1 - h[i]
            max_delta2 = min(max_sub2, max_add2)

        h[i_src] -= delta
        h[i_dst] += delta
        curr_c5_new = compute_c5(h)
        delta_c5 = curr_c5_new - curr_c5
        # temperature schedule maybe separate
        elapsed = t_now - start_time

                best_h = h.copy()

            # revert
            h[i_src] += delta
            h[i_dst] -= delta
        it += 1
    return best_h, best_c5

import numpy as np, time, math

def compute_c5(h):
    """Discrete C5 for step function defined by samples h."""
    n = h.shape[0]
    dx = 2.0 / n
    # correlation of h with (1-h) without wrap‑around
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def random_initial_h(n, rng):
    """Binary (or ½) vector with sum = n/2."""
    h = np.zeros(n, dtype=float)
    half = n // 2
    idx = rng.choice(n, size=half, replace=False)
    h[idx] = 1.0
    if n % 2:
        # one extra half‑unit to meet the exact integral
        remaining = np.setdiff1d(np.arange(n), idx)
        h[rng.choice(remaining)] = 0.5
    return h

def propose_fractional_move(h, rng, max_step):
    """Pick two distinct indices i,j and a feasible delta to move mass from i to j."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # direction i -> j
    sub = h[i]
    add = 1.0 - h[j]
    move = min(sub, add)
    if move <= 0.0:
        # try opposite direction
        sub = h[j]
        add = 1.0 - h[i]
        move = min(sub, add)
        if move <= 0.0:
            return None
        i, j = j, i
    # delta is randomly sized but not larger than max_step
    delta = rng.random() * move
    if delta > max_step:
        delta = max_step
    return i, j, delta

def anneal_fractional(start_h, budget, start_time):
    """Simple simulated‑annealing on the continuous domain."""
    rng = np.random.default_rng()
    h = start_h.copy()
    n = h.shape[0]
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    # temperature schedule parameters
    T0 = 0.02                # initial temperature
    step0 = 0.2              # initial maximal delta
    while True:
        now = time.time()
        if now - start_time >= budget:
            break
        # linear cooling
        elapsed = now - start_time
        frac = 1.0 - elapsed / budget
        T = max(T0 * frac, 1e-9)
        max_step = max(step0 * frac, 1e-4)
        prop = propose_fractional_move(h, rng, max_step)
        if prop is None:
            continue
        i, j, delta = prop
        h[i] -= delta
        h[j] += delta
        new_c5 = compute_c5(h)
        delta_c5 = new_c5 - cur_c5
        accept = False
        if delta_c5 <= 0.0:
            accept = True
        else:
            if rng.random() < math.exp(-delta_c5 / T):
                accept = True
        if accept:
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # revert move
            h[i] += delta
            h[j] -= delta
    return best_h, best_c5

def enforce_constraints(h):
    """Clip to [0,1] and adjust sum to exactly n/2."""
    n = h.shape[0]
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) > 1e-12:
        # scale uniformly then re‑clip small residual
        if s > 0:
            factor = target / s
            h *= factor
        else:
            # degenerate case: all zeros – put mass on first entry
            h[0] = target
        h = np.clip(h, 0.0, 1.0)
        # final tiny correction
        diff = target - h.sum()
        if abs(diff) > 1e-12:
            # add/subtract diff from a non‑boundary element
            if diff > 0:
                cand = np.where(h < 1.0 - 1e-12)[0]
                if cand.size:
                    h[rng.choice(cand)] += diff
            else:
                cand = np.where(h > 1e-12)[0]
                if cand.size:
                    h[rng.choice(cand)] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns (h_values, c5_bound, n_points) where h_values is a numpy array
    representing the step function on [0,2], c5_bound is its overlap bound,
    and n_points is the number of discretisation points.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # decide length
    if 'initial_h_values' in globals() and isinstance(initial_h_values, np.ndarray):
        h0 = np.array(initial_h_values, copy=True, dtype=float)
        n = h0.shape[0]
    else:
        n = 200  # even, gives reasonable resolution
        h0 = random_initial_h(n, rng)
    # guarantee constraints for the starter
    h0 = enforce_constraints(h0)
    # allocate the whole budget (minus a tiny safety margin)
    total_budget = budget_s - 0.5
    best_h, best_c5 = anneal_fractional(h0, total_budget, start)
    # final clean‑up
    best_h = enforce_constraints(best_h)
    # evaluate with the official helper (if available) for a consistent value
    try:
        final_c5 = float(evaluate_erdos_solution(best_h))
    except Exception:
        final_c5 = float(compute_c5(best_h))
    return best_h, final_c5, n
