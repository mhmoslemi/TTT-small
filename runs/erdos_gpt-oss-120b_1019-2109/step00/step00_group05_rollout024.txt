def project_to_bounds_sum(v, target):
    # Perform water-filling to satisfy sum constraint with bounds [0,1]
    # Using binary search on lambda.
    lo = -1.0
    hi = 1.0
    # Expand boundaries to ensure feasible.
    # Actually lambda may need to be outside [-1,1], but we can expand until sum at lo>target and sum at hi<target.
    # Compute sum at lo = np.clip(v - lo, 0, 1). With lo very negative, v - lo >> 1 => clip to 1 → sum = n (max). So at lo = -inf, sum = n > target.
    # With hi large positive, v - hi <<0 => clip to 0 → sum = 0 < target.
    # So we can set lo=-10, hi=10 scanning.
    lo = -10.0
    hi = 10.0
    # binary search
    for _ in range(60):
        lam = (lo + hi) / 2.0
        q = np.clip(v - lam, 0, 1)
        s = q.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    q = np.clip(v - (lo+hi)/2.0, 0, 1)
    # Due to numerical error sum may not be exactly target. Adjust by distributing tiny delta over non-bound entries.
    diff = target - q.sum()
    if abs(diff) > 1e-8:
        # find indices not at bounds
        free = (q > 1e-12) & (q < 1-1e-12)
        if free.any():
            # distribute proportionally
            q[free] += diff / free.sum()
            # ensure still within bounds
            q = np.clip(q, 0, 1)
    return q

import time
import numpy as np

def _project_to_bounds_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1]^n while enforcing sum(v)=target.
    Uses a simple water‑filling (binary search on λ) which solves
        q_i = clip(v_i - λ, 0, 1)
    for the unique λ with Σ q_i = target.
    """
    lo, hi = -10.0, 10.0
    for _ in range(60):
        lam = (lo + hi) * 0.5
        q = np.clip(v - lam, 0.0, 1.0)
        s = q.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    q = np.clip(v - (lo + hi) * 0.5, 0.0, 1.0)

    # tiny correction in case of numerical error
    diff = target - q.sum()
    if abs(diff) > 1e-12:
        free = (q > 1e-12) & (q < 1.0 - 1e-12)
        if free.any():
            q[free] += diff / free.sum()
            q = np.clip(q, 0.0, 1.0)
    return q


def _random_feasible(n: int, rng: np.random.Generator) -> np.ndarray:
    """Generate a random vector in [0,1]^n with sum = n/2."""
    target = n / 2.0
    v = rng.random(n)                # uniform in (0,1)
    return _project_to_bounds_sum(v, target)


def _cost_and_autocorr(h: np.ndarray):
    """Return C5 value and the full autocorrelation array of h."""
    n = h.shape[0]
    ac = np.correlate(h, h, mode='full')
    min_val = ac.min()
    c5 = 1.0 - min_val * (2.0 / n)   # because ∫h = 1
    return c5, ac


def _pair_move(h: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Random pair move that preserves the sum and box constraints."""
    n = h.shape[0]
    i, j = rng.choice(n, size=2, replace=False)
    # delta will be added to i and subtracted from j
    max_inc = min(1.0 - h[i], h[j])          # increase i, decrease j
    max_dec = min(h[i], 1.0 - h[j])          # decrease i, increase j
    if max_inc + max_dec == 0.0:
        return None
    # sample uniformly in the admissible interval
    delta = rng.random() * (max_inc + max_dec) - max_dec
    # optionally shrink step size a bit (helps stability)
    delta *= 0.5
    new_h = h.copy()
    new_h[i] += delta
    new_h[j] -= delta
    # Clip tiny numerical excursions
    new_h[i] = min(1.0, max(0.0, new_h[i]))
    new_h[j] = min(1.0, max(0.0, new_h[j]))
    return new_h


def _targeted_move(h: np.ndarray, ac: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move mass into the region that is responsible for the minimal autocorrelation."""
    n = h.shape[0]
    # index of the worst shift (minimum autocorrelation)
    min_idx = ac.argmin()
    k = min_idx - (n - 1)                     # shift value, can be negative
    # indices i that participate in this shift (i and i+k are both inside [0,n))
    if k >= 0:
        i_start, i_end = 0, n - k
    else:
        i_start, i_end = -k, n
    support = np.arange(i_start, i_end, dtype=int)

    # candidates that can receive additional mass (still <1)
    inc_mask = h[support] < 1.0 - 1e-12
    inc_idx = support[inc_mask]
    if inc_idx.size == 0:
        return None

    # weighting: prefer positions whose partner (i+k) already holds large value
    partner = inc_idx + k
    partner_vals = h[partner]
    weights = partner_vals + 1e-12               # avoid zero weight
    w_sum = weights.sum()
    if w_sum == 0.0:
        i_sel = rng.choice(inc_idx)
    else:
        # sample proportional to partner value
        r = rng.random() * w_sum
        cum = 0.0
        for ii, ww in zip(inc_idx, weights):
            cum += ww
            if r <= cum:
                i_sel = ii
                break

    # choose a donor index outside the support (so we do not hurt the worst shift)
    outside = np.ones(n, dtype=bool)
    outside[support] = False
    donor_mask = (h > 1e-12) & outside
    donor_idx = np.where(donor_mask)[0]
    if donor_idx.size == 0:
        # fallback to any positive entry
        donor_idx = np.where(h > 1e-12)[0]
        if donor_idx.size == 0:
            return None
    j_sel = rng.choice(donor_idx)

    max_inc = 1.0 - h[i_sel]
    max_dec = h[j_sel]
    delta_max = min(max_inc, max_dec)
    if delta_max <= 0.0:
        return None
    delta = rng.random() * delta_max
    new_h = h.copy()
    new_h[i_sel] += delta
    new_h[j_sel] -= delta
    return new_h
