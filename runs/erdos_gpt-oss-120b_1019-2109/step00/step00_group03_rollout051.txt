def parametric_search(N):
    total = N // 2  # assuming N even.
    best_h = None
    best_score = np.inf
    for n0 in range(N//2 + 1):
        max_ns = (N - 2*n0) // 2
        for ns in range(max_ns + 1):
            nc = N - 2*n0 - 2*ns
            if ns == 0:
                # Only central plateau and zeros
                if nc == total:
                    # central plateau length total, zeros elsewhere
                    h = np.zeros(N)
                    start = n0
                    h[start:start+nc] = 1
                else:
                    continue
            else:
                a = (total - nc) / (2*ns)
                # if a out of [0,1], skip
                if a < 0 or a > 1:
                    continue
                if nc < 0:
                    continue
                # shape
                h = np.zeros(N)
                # side left
                start_side = n0
                h[start_side:start_side+ns] = a
                # central
                start_center = n0 + ns
                h[start_center:start_center+nc] = 1
                # side right
                start_side_r = start_center + nc
                h[start_side_r:start_side_r+ns] = a
                # zeros elsewhere automatically
            # compute score
            sc = compute_score(h)
            if sc < best_score:
                best_score = sc
                best_h = h.copy()
    return best_h, best_score

def project_onto_box_and_sum(v, total):
    # v array
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) / 2.0
        x = np.clip(v - lam, 0.0, 1.0)

import time
import numpy as np

# ---------- helper functions ----------
def shift(arr: np.ndarray, k: int) -> np.ndarray:
    """Shift arr by k positions with zero padding.
    Positive k shifts to the right (arr[i‑k]), negative k to the left."""
    n = arr.shape[0]
    if k > 0:
        return np.concatenate((np.zeros(k, dtype=arr.dtype), arr[: n - k]))
    if k < 0:
        k = -k
        return np.concatenate((arr[k:], np.zeros(k, dtype=arr.dtype)))
    return arr.copy()


def project_onto_box_and_sum(v: np.ndarray, total: float) -> np.ndarray:
    """Project v onto the set {x: 0≤x≤1, sum(x)=total}."""
    lo, hi = -1.0, 1.0
    for _ in range(30):
        lam = (lo + hi) * 0.5
        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > total:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def compute_score(h: np.ndarray, dx: float) -> float:
    """C5 value for a step function h (discretized)."""
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)


def parametric_search(N: int, total: float, dx: float) -> tuple:
    """Three‑level shape: 0 – a – 1 – a – 0.
    Returns the best found h and its score."""
    best_h = None
    best_sc = np.inf
    for n0 in range(N // 2 + 1):
        max_ns = (N - 2 * n0) // 2
        for ns in range(max_ns + 1):
            nc = N - 2 * n0 - 2 * ns
            if ns == 0:
                # only a central block of ones
                if nc == total:
                    h = np.zeros(N)
                    h[n0 : n0 + nc] = 1.0
                else:
                    continue
            else:
                a = (total - nc) / (2.0 * ns)
                if a < 0.0 or a > 1.0:
                    continue
                if nc < 0:
                    continue
                h = np.zeros(N)
                left_start = n0
                left_end = left_start + ns
                h[left_start:left_end] = a
                cen_start = left_end
                cen_end = cen_start + nc
                h[cen_start:cen_end] = 1.0
                right_start = cen_end
                right_end = right_start + ns
                h[right_start:right_end] = a
            sc = compute_score(h, dx)
            if sc < best_sc:
                best_sc = sc
                best_h = h.copy()
    return best_h, best_sc


def gradient_descent(h0: np.ndarray, total: float, dx: float,
                    budget_s: float, start_time: float) -> tuple:
    """Projected sub‑gradient descent on C5."""
    h = h0.copy()
    N = h.shape[0]
    best_h = h.copy()
    best_sc = compute_score(h, dx)
    lr = 0.5
    iter_cnt = 0
    stagnant = 0
    while time.time() - start_time < budget_s * 0.93:
        conv = np.correlate(h, 1.0 - h, mode="full")
        idx = int(np.argmax(conv))
        lag = idx - (N - 1)
        # gradient of the active lag
        hv_minus = shift(h, lag)
        hv_plus = shift(h, -lag)
        grad = 1.0 - (hv_plus + hv_minus)
        # step and project
        v = h - lr * grad
        v = np.clip(v, 0.0, 1.0)
        h = project_onto_box_and_sum(v, total)

        sc = compute_score(h, dx)
        if sc + 1e-12 < best_sc:
            best_sc = sc
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1

        iter_cnt += 1
        if iter_cnt % 500 == 0:
            lr = max(lr * 0.9, 0.01)

        # occasional random perturbation if stuck
        if stagnant > 5000:
            noise = (np.random.rand(N) - 0.5) * 0.1
            h = project_onto_box_and_sum(np.clip(best_h + noise, 0.0, 1.0), total)
            lr = 0.5
            stagnant = 0
    return best_h, best_sc


def hill_climb(h0: np.ndarray, total: float, dx: float,
               budget_s: float, start_time: float) -> tuple:
    """Simple pairwise mass‑exchange hill climbing."""
    h = h0.copy()
    N = h.shape[0]
    best_sc = compute_score(h, dx)
    best_h = h.copy()
    attempts = 0
    while time.time() - start_time < budget_s * 0.95:
        i, j = np.random.randint(N), np.random.randint(N)
        if i == j:
            continue
        # propose moving ε from i to j
        epsilon = np.random.rand() * 0.1
        epsilon = min(epsilon, h[i], 1.0 - h[j])
        if epsilon <= 1e-12:
            continue
        h_new = h.copy()
        h_new[i] -= epsilon
        h_new[j] += epsilon
        sc = compute_score(h_new, dx)
        attempts += 1
        if sc + 1e-12 < best_sc:
            h = h_new
            best_sc = sc
            best_h = h_new.copy()
            if best_sc < 0.3808:
                break
        # occasionally re‑project to keep exact sum
        if attempts % 1000 == 0:
            h = project_onto_box_and_sum(h, total)
    return best_h, best_sc


# ---------- main entry point ----------
def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Return (h_array, C5_bound, n_points) within the time budget."""
    np.random.seed(seed)
    start = time.time()

    # ----- initialise -----
    if "initial_h_values" in globals():
        init = np.asarray(initial_h_values, dtype=float)
        # up‑sample the given construction (factor 2) for finer control
        factor = 2
        N = init.shape[0] * factor
        x_old = np.linspace(0.0, 2.0, init.shape[0], endpoint=False)
        x_new = np.linspace(0.0, 2.0, N, endpoint=False)
        h0 = np.interp(x_new, x_old, init)
    else:
        N = 120  # a modest even size
        h0 = np.full(N, 0.5)

    total = N / 2.0
    dx = 2.0 / N
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_onto_box_and_sum(h0, total)

    # ----- coarse parametric search -----
    best_h, best_sc = parametric_search(N, total, dx)

    # also consider the (possibly up‑sampled) initial construction
    sc_init = compute_score(h0, dx)
    if sc_init < best_sc:
        best_h, best_sc = h0.copy(), sc_init

    # ----- refine with gradient descent -----
    h_gd, sc_gd = gradient_descent(best_h, total, dx, budget_s, start)
    if sc_gd < best_sc:
        best_h, best_sc = h_gd, sc_gd

    # ----- final local search -----
    h_hc, sc_hc = hill_climb(best_h, total, dx, budget_s, start)
    if sc_hc < best_sc:
        best_h, best_sc = h_hc, sc_hc

    # final projection to guarantee feasibility
    best_h = project_onto_box_and_sum(best_h, total)
    best_sc = compute_score(best_h, dx)

    return best_h, float(best_sc), N
