def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    # try to use provided initial_h_values if exists
    try:
        h0 = np.array(initial_h_values, dtype=np.float64)
        n = h0.shape[0]
    except NameError:
        n = 200  # default size
        h0 = np.full(n, 0.5)  # start uniform
    dx = 2.0 / n
    target_sum = n / 2.0
    # ensure sum constraint via projection
    h0 = project_box_sum(h0, target_sum)  # define project function earlier
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr.max() * dx
    best_h = h0.copy()
    best_val = compute_c5(best_h)
    current_h = best_h.copy()
    current_val = best_val
    # SA parameters
    T0 = best_val * 0.5 + 0.01  # starting temperature maybe a fraction.
    T = T0
    start_time = time.time()
    iter_count = 0
    # main loop
    while True:
        now = time.time()
        if now - start_time > budget_s * 0.98:  # leave margin
            break
        # propose move
        i, j = rng.choice(n, size=2, replace=False)
        # direction: move mass from i to j or j to i randomly
        if rng.random() < 0.5:
            # move from i to j
            max_transfer = min(current_h[i], 1 - current_h[j])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = current_h.copy()
            new_h[i] -= delta
            new_h[j] += delta
        else:
            # move from j to i
            max_transfer = min(current_h[j], 1 - current_h[i])
            if max_transfer <= 0:
                continue
            delta = rng.random() * max_transfer
            new_h = current_h.copy()
            new_h[i] += delta
            new_h[j] -= delta
        # new_h already respects bounds and sum
        new_val = compute_c5(new_h)
        # accept
        if new_val < current_val:
            accept = True
        else:
            # Metropolis
            prob = math.exp(-(new_val - current_val) / T) if T > 1e-12 else False
            accept = rng.random() < prob
        if accept:
            current_h = new_h
            current_val = new_val
            if new_val < best_val:
                best_h = new_h.copy()
                best_val = new_val
        # temperature schedule
        # we can reduce T slowly
        iter_count += 1
        if iter_count % 1000 == 0:
            # exponential cooling
            T = T * 0.99
            # ensure T does not go negative
    # after loop, ensure best_h satisfies constraints due to floating errors
    # Clip and project for safety
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_box_sum(best_h, target_sum)
    final_val = compute_c5(best_h)
    return (best_h.tolist(), float(final_val), n)

import numpy as np

def _project_box_sum(v: np.ndarray, target: float) -> np.ndarray:
    """L2 projection of v onto the box [0,1]^n with sum == target."""
    lo = np.min(v) - 1.0
    hi = np.max(v)
    for _ in range(60):
        lam = (lo + hi) * 0.5
        p = np.clip(v - lam, 0.0, 1.0)
        s = p.sum()
        if s > target:
            lo = lam
        else:
            hi = lam
    lam = (lo + hi) * 0.5
    return np.clip(v - lam, 0.0, 1.0)


def _c5_value(h: np.ndarray, dx: float) -> float:
    """Return the C5 value for a given step vector."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(corr.max() * dx)
