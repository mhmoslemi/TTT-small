def gradient_step(h, alpha):

    conv = np.correlate(h, 1 - h, mode='full')
    idx = conv.argmax()
    L = idx - (len(h) - 1)
    n = len(h)

def gradient_step(h, alpha, target):
    n = len(h)
    idx = np.argmax(np.correlate(h, 1 - h, mode='full'))
    L = idx - (n - 1)
    grad = np.ones(n)
    if L >= 0:
        if L < n:
            grad[:n - L] -= h[L:]  # i+L valid
            grad[L:] -= h[:n - L]
    else:
        Lpos = -L
        if Lpos < n:
            grad[Lpos:] -= h[:n - Lpos]
            grad[:n - Lpos] -= h[Lpos:]
    h_new = h - alpha * grad
    # Project
    h_proj = project_to_sum(h_new, target)
    return h_proj, idx

import time, random, numpy as np

def _project_to_sum(v: np.ndarray, target: float) -> np.ndarray:
    """Project v onto the box [0,1] and the affine hyperplane sum(v)=target."""
    lo = v.min() - 1.0
    hi = v.max()
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v - mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - lo, 0.0, 1.0)


def _c5_of(h: np.ndarray) -> float:
    """Calculate C5 for a step vector h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """Search for a step function with a small Erdős C5 bound."""
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # initialise h
    # ------------------------------------------------------------------
    try:
        h = np.array(initial_h_values, dtype=np.float64).copy()
        n = h.shape[0]
    except NameError:
        n = 150                     # reasonable size, well under the 1000 limit
        target = n / 2.0
        h = _project_to_sum(np.random.rand(n), target)

    target = n / 2.0               # ∑h = n/2  (∫h = 1)

    # ensure feasibility of the starting point
    h = _project_to_sum(h, target)

    best_h = h.copy()
    best_c5 = _c5_of(best_h)

    start_time = time.time()
    alpha = 0.5                    # initial step size for gradient moves
    it = 0
    last_improve = 0

    # ------------------------------------------------------------------
    # main optimisation loop (gradient + random pair swaps)
    # ------------------------------------------------------------------
    while time.time() - start_time < budget_s - 0.2:
        # current C5 and worst shift
        conv_full = np.correlate(h, 1.0 - h, mode="full")
        idx_max = int(conv_full.argmax())
        L = idx_max - (n - 1)               # integer shift causing the maximum

        # gradient of the worst shift (see analysis)
        grad = np.ones(n)
        if L >= 0:
            if L < n:
                grad[: n - L] -= h[L:]
                grad[L:]       -= h[: n - L]
        else:
            Lp = -L
            if Lp < n:
                grad[Lp:]      -= h[: n - Lp]
                grad[: n - Lp] -= h[Lp:]

        # gradient descent step with projection
        h_new = h - alpha * grad
        h_new = _project_to_sum(h_new, target)
        c5_new = _c5_of(h_new)
        c5_cur = _c5_of(h)

        # deterministic acceptance of improvement, stochastic otherwise
        if c5_new < c5_cur:
            h = h_new

                h = h_new

        # ------------------------------------------------------------------
        # occasional random pair swap (keeps sum exactly)
        # ------------------------------------------------------------------
        if it % 20 == 0:

                eps = min(h[i], 1.0 - h[j])

                    cand = h.copy()
                    cand[i] -= eps
                    cand[j] += eps
                else:
                    cand = None

                alpha = 1e-4

    # ------------------------------------------------------------------
