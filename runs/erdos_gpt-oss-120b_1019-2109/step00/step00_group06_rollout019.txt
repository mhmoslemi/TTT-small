def hill_climb(h, current_c5, time_limit, start_time):
    n = len(h)
    ones_idx = np.where(h==1)[0]
    zeros_idx = np.where(h==0)[0]
    while time.time() - start_time < time_limit:
        i = np.random.choice(ones_idx)
        j = np.random.choice(zeros_idx)
        # swap
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5_vec(h)
        if new_c5 < current_c5:
            current_c5 = new_c5
            best_h = h.copy()
            # update indices lists
            # update ones_idx and zeros_idx accordingly:
            # could recompute each iteration; cheap for n small.
            ones_idx = np.where(h==1)[0]
            zeros_idx = np.where(h==0)[0]
        else:
            # revert swap
            h[i] = 1
            h[j] = 0
    return best_h, current_c5

def resample_initial(initial_h, n):
    n0 = len(initial_h)
    x0 = np.linspace(0,2, n0, endpoint=False)
    x = np.linspace(0,2, n, endpoint=False)
    interp = np.interp(x, x0, initial_h)
    # thresh: set top m entries to 1
    m = n//2
    order = np.argsort(-interp)
    out = np.zeros(n)
    out[order[:m]] = 1.0
    return out
```

def run(seed=42, budget_s=1100, **kwargs):
    import numpy as np, random, time, math, sys

    # ----------------------------------------------------------------------
    # Helper: compute the C5 bound for a binary (or [0,1]) vector h
    # ----------------------------------------------------------------------
    def compute_c5(h):
        n = h.shape[0]
        dx = 2.0 / n
        # correlation of h with (1-h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        return np.max(corr) * dx

    # ----------------------------------------------------------------------
    # Candidate generators
    # ----------------------------------------------------------------------
    def random_binary(n):
        m = n // 2
        arr = np.zeros(n, dtype=np.float64)
        arr[:m] = 1.0
        np.random.shuffle(arr)
        return arr

    def thue_morse(n):
        # produce first n bits of Thue–Morse; works best when n is a power of 2
        seq = np.fromiter(((bin(i).count("1") & 1) for i in range(n)),
                          dtype=np.int8, count=n)
        return seq.astype(np.float64)

    def sturmian(n):
        # golden ratio based Sturmian word (cutting sequence)
        phi = (math.sqrt(5) - 1) / 2.0  # ≈0.618
        t = (np.arange(n, dtype=np.float64) * phi) % 1.0
        arr = (t < 0.5).astype(np.float64)
        # enforce exact half weight
        m = n // 2
        s = int(arr.sum())
        if s != m:
            diff = m - s
            # flip the diff entries with smallest distance to threshold
            if diff > 0:
                # need more ones -> turn some zeros (t>=0.5) into ones
                idx = np.argsort(t)  # smallest t are already ones
                zeros = np.where(arr == 0)[0]
                choose = zeros[np.argsort(t[zeros])[:diff]]
                arr[choose] = 1.0
            else:
                # need fewer ones -> turn some ones into zeros
                idx = np.argsort(-t)  # largest t already zeros
                ones = np.where(arr == 1)[0]
                choose = ones[np.argsort(-t[ones])[: -diff]]
                arr[choose] = 0.0
        return arr

    def arith_progression(n, step):
        """binary set using a single step (coprime to n)"""
        m = n // 2
        arr = np.zeros(n, dtype=np.float64)
        pos = 0
        for _ in range(m):
            arr[pos] = 1.0
            pos = (pos + step) % n
        return arr

    def resample_initial(init_h, n):
        # linear interpolation then keep top n/2 entries
        init_h = np.asarray(init_h, dtype=np.float64)
        n0 = init_h.shape[0]
        x0 = np.linspace(0.0, 2.0, n0, endpoint=False)
        x = np.linspace(0.0, 2.0, n, endpoint=False)
        interp = np.interp(x, x0, init_h)
        m = n // 2
        order = np.argsort(-interp)  # descending
        out = np.zeros(n, dtype=np.float64)
        out[order[:m]] = 1.0
        return out

    # ----------------------------------------------------------------------
    # Choose problem size
    # ----------------------------------------------------------------------
    # Use an even number of points; 200 gives a good trade‑off.
    n = 200
    if n % 2:
        n += 1
    half = n // 2

    # ----------------------------------------------------------------------
    # Build a pool of initial candidates
    # ----------------------------------------------------------------------
    candidates = []

    # 1. Random binary
    candidates.append(random_binary(n))

    # 2. Thue–Morse (if length appropriate)
    candidates.append(thue_morse(n))

    # 3. Sturmian (golden‑ratio cutting)
    candidates.append(sturmian(n))

    # 4. Arithmetic progressions with coprime steps
    #    Try a handful of steps; avoid those giving the alternating pattern.
    from math import gcd
    steps = [p for p in range(3, n, 2) if gcd(p, n) == 1]
    # limit to a reasonable subset to keep init time low
    random.seed(seed)
    random.shuffle(steps)
    for p in steps[:30]:
        candidates.append(arith_progression(n, p))

    # 5. Resampled initial_h_values, if supplied
    if 'initial_h_values' in globals():
        try:
            candidates.append(resample_initial(initial_h_values, n))
        except Exception:
            pass

    # ----------------------------------------------------------------------
    # Evaluate all candidates, keep the best
    # ----------------------------------------------------------------------
    best_h = None
    best_c5 = float('inf')
    for cand in candidates:
        c5 = compute_c5(cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = cand.copy()

    # ----------------------------------------------------------------------
    # Local optimisation (hill‑climbing with occasional random restarts)
    # ----------------------------------------------------------------------
    start_time = time.time()
    time_limit = float(budget_s) * 0.95   # keep a small safety margin

    cur_h = best_h.copy()
    cur_c5 = best_c5

    stagnation = 0          # how many iterations without improvement
    max_stagnation = 20000  # trigger a restart
    iteration = 0

    # Parameters for the simple simulated‑annealing acceptance
    T = 0.02                # initial temperature (in C5 units)
    T_min = 1e-5
    decay = 0.99996

    # Pre‑compute index arrays for speed
    def ones_zeros(arr):
        ones = np.flatnonzero(arr)
        zeros = np.flatnonzero(1.0 - arr)
        return ones, zeros

    while True:
        # time check
        if time.time() - start_time > time_limit:
            break

        # pick a random 1 and a random 0 to swap
        ones, zeros = ones_zeros(cur_h)
        if ones.size == 0 or zeros.size == 0:
            break
        i = random.choice(ones)
        j = random.choice(zeros)

        # perform swap
        cur_h[i] = 0.0
        cur_h[j] = 1.0
        new_c5 = compute_c5(cur_h)

        delta = new_c5 - cur_c5
        accept = False
        if delta <= 0:
            accept = True
        else:
            # probabilistic uphill move
            if T > T_min and random.random() < math.exp(-delta / T):
                accept = True

        if accept:
            cur_c5 = new_c5
            stagnation = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
                # if we already beat the target, we can stop early
                if best_c5 <= 0.3808:
                    break
        else:
            # revert swap
            cur_h[i] = 1.0
            cur_h[j] = 0.0
            stagnation += 1

        # temperature cooling
        T = max(T_min, T * decay)

        # occasional restart to avoid being stuck
        if stagnation > max_stagnation:
            # random restart (biased towards good arithmetic steps)
            if random.random() < 0.5 and steps:
                p = random.choice(steps)
                cur_h = arith_progression(n, p)
            else:
                cur_h = random_binary(n)
            cur_c5 = compute_c5(cur_h)
            stagnation = 0
            T = 0.02  # reset temperature

        iteration += 1

    # ----------------------------------------------------------------------
    # Return the best construction found
    # ----------------------------------------------------------------------
    # Ensure exact integral constraint (sum should be n/2)
    # Slight numeric drift can appear only if we accidentally used non‑binary values.
    # Force binary by thresholding the best solution (won’t affect C5 noticeably).
    best_h = (best_h > 0.5).astype(np.float64)
    # Re‑compute final bound after possible rounding
    final_c5 = compute_c5(best_h)

    return (best_h, float(final_c5), n)
