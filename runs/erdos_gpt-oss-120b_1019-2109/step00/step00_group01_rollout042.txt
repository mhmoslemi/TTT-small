def improve_h(h, budget, target):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    current_h = h.copy()
    current_c5 = best_c5
    start = time.time()
    # temperature schedule
    T0 = 0.1
    it = 0
    while time.time() - start < budget:
        # propose swap
        # find indices of ones and zeros
        ones = np.flatnonzero(current_h > 0.5)
        zeros = np.flatnonzero(current_h < 0.5)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        # perform swap
        candidate = current_h.copy()
        candidate[i] = 0.0
        candidate[j] = 1.0
        cand_c5 = compute_c5(candidate)
        delta = cand_c5 - current_c5
        # acceptance
        T = T0 * (0.9995 ** it)
        if delta < 0 or np.random.rand() < math.exp(-delta / (T + 1e-12)):
            current_h = candidate
            current_c5 = cand_c5
            if cand_c5 < best_c5:
                best_c5 = cand_c5
                best_h = candidate.copy()
                if best_c5 <= target:
                    break
        it += 1
    return best_h, best_c5

def local_optimize(h, time_limit, target):
    import time, numpy as np
    n = h.shape[0]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    # For speed, precompute constant factor dx? not needed.
    while time.time() - start < time_limit:
        # compute conv and find worst lag
        conv = np.correlate(best_h, 1 - best_h, mode='full')
        max_idx = int(np.argmax(conv))
        k = max_idx - (n - 1)
        if k < 0:
            k = -k
        # ignore k == 0 (if max is at zero shift, pick next)
        if k == 0:
            # set conv[max_idx] = -inf temporarily
            # We'll just set to -1 to ignore and recompute
            conv[max_idx] = -1
            max_idx = int(np.argmax(conv))
            k = abs(max_idx - (n - 1))
            if k == 0:
                # no improvement possible
                break
        L = n - k
        left = best_h[:L]
        right = best_h[k:]
        B = np.where((left == 1) & (right == 0))[0]
        R = np.where((left == 0) & (right == 1))[0]
        improved = False
        if len(B) > 0 and len(R) > 0:
            i = np.random.choice(B)
            j = np.random.choice(R)
            # swap values in a copy
            h_new = best_h.copy()
            h_new[i], h_new[j] = 0, 1
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_h = h_new
                best_c5 = c5_new
                improved = True
        if not improved:
            # random swap fallback
            ones = np.where(best_h == 1)[0]
            zeros = np.where(best_h == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            h_new = best_h.copy()
            h_new[i], h_new[j] = 0, 1

import time, math, random
import numpy as np

def _c5_of_h(h):
    """return C5 for a binary (or [0,1]) vector h."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    return conv.max() * (2.0 / n)


def _improve_one_step(h, target):
    """Try to improve h by a single targeted swap (works for binary h)."""
    n = h.shape[0]
    conv = np.correlate(h, 1.0 - h, mode="full")
    max_idx = int(np.argmax(conv))
    k = max_idx - (n - 1)
    if k < 0:
        k = -k
    if k == 0:                     # ignore the trivial lag
        conv[max_idx] = -1.0
        max_idx = int(np.argmax(conv))
        k = abs(max_idx - (n - 1))
        if k == 0:
            return h, False
    L = n - k
    left = h[:L]
    right = h[k:]
    bad_left = np.where((left == 1) & (right == 0))[0]
    bad_right = np.where((left == 0) & (right == 1))[0]
    if bad_left.size and bad_right.size:
        i = np.random.choice(bad_left)
        j = np.random.choice(bad_right)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_c5 = _c5_of_h(h_new)
        if new_c5 + 1e-12 < _c5_of_h(h):
            return h_new, True
    # fallback: random 1‑0 swap
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if ones.size and zeros.size:
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_new = h.copy()
        h_new[i] = 0.0
        h_new[j] = 1.0
        new_c5 = _c5_of_h(h_new)
        if new_c5 + 1e-12 < _c5_of_h(h):
            return h_new, True
    return h, False


def _local_search(start_h, time_limit, target):
    """hill‑climbing / simulated‑annealing style search."""
    best_h = start_h.copy()
    best_c5 = _c5_of_h(best_h)
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        # temperature schedule (very simple)
        T = 0.02 * (0.9995 ** it)
        h_candidate, improved = _improve_one_step(best_h, target)
        cand_c5 = _c5_of_h(h_candidate)
        accept = False
        if cand_c5 < best_c5 - 1e-12:
            accept = True
        elif not improved:
            # sometimes accept a worsening move (simulated annealing)
            delta = cand_c5 - best_c5
            if random.random() < math.exp(-delta / (T + 1e-12)):
                accept = True
        if accept:
            best_h = h_candidate
            best_c5 = cand_c5
        it += 1
        if best_c5 <= target:
            break
    return best_h, best_c5


def run(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Find a binary step function that gives a good upper bound for C5.
    Returns (h_values, c5_bound, n_points).
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Choose the discretisation length.
    # ------------------------------------------------------------------
    # If the harness provides an initial construction, use it as a seed.
    if "initial_h_values" in globals():
        base = np.asarray(initial_h_values, dtype=np.float64)
        # make sure it is binary (the original construction is binary)
        base = (base > 0.5).astype(np.float64)
        n0 = base.size
        # duplicate once so we have an even number of points (helps the
        # half‑mass constraint) and enough resolution.
        h0 = np.tile(base, 2)
    else:
        # default size – 200 points is a good trade‑off between speed and
        # solution quality.
        n = 200
        h0 = np.zeros(n, dtype=np.float64)
        h0[: n // 2] = 1.0
        np.random.shuffle(h0)

    n = h0.size
    # guarantee the integral condition exactly (sum = n/2)
    target_sum = n // 2
    cur_sum = int(h0.sum())
    if cur_sum != target_sum:
        diff = target_sum - cur_sum
        if diff > 0:
            zeros = np.where(h0 == 0)[0]
            idx = np.random.choice(zeros, diff, replace=False)
            h0[idx] = 1.0
        else:
            ones = np.where(h0 == 1)[0]
            idx = np.random.choice(ones, -diff, replace=False)
            h0[idx] = 0.0

    # ------------------------------------------------------------------
    # 2. Optimisation loop – many short local‑search runs.
    # ------------------------------------------------------------------
    best_h = h0.copy()
    best_c5 = _c5_of_h(best_h)

    deadline = time.time() + budget_s - 0.5          # keep a small safety margin
    target = 0.3808

    # perform a sequence of local searches, restarting if progress stalls
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # each local search gets at most 5 seconds (or the whole remaining time)
        chunk = min(5.0, remaining)
        h_candidate, c5_candidate = _local_search(best_h.copy(), chunk, target)
        if c5_candidate < best_c5 - 1e-12:
            best_h, best_c5 = h_candidate, c5_candidate
        if best_c5 <= target:
            break
        # occasional random restart to escape possible plateaus
        if random.random() < 0.15:
            h_rand = np.zeros(n, dtype=np.float64)
            h_rand[: n // 2] = 1.0
            np.random.shuffle(h_rand)
            best_h, best_c5 = h_rand, _c5_of_h(h_rand)

    final_c5 = _c5_of_h(best_h)

    # The harness expects a float array in [0,1]; return it as float.
    return best_h.astype(float), float(final_c5), n
