def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1 - h, mode='full')
    return corr.max() * dx

def project_to_sum_and_bounds(v, target):
    # find lambda via binary search
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + lo, 0.0, 1.0)  # or (lo+hi)/2
    # due to numeric tolerance we might adjust final sum by scaling small
    if abs(x.sum() - target) > 1e-8:
        # adjust by distributing deficit proportionally among interior vals not at bounds
        diff = target - x.sum()
        # adjust values not at bounds
        free = (x > 0) & (x < 1)
        count = free.sum()
        if count > 0:
            x[free] += diff / count
            # clip again
            x = np.clip(x, 0, 1)
    return x

def worst_shift_gradient(h):
    N = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    max_val = corr.max()
    idxs = np.flatnonzero(corr == max_val)
    grad = np.zeros(N)
    for idx in idxs:
        L = idx - (N - 1)
        if L >= 0:
            # first term
            grad[:N - L] += (1 - h[L:])
            # second term
            grad[L:] -= h[:N - L]
        else:
            k = -L
            grad[k:] += (1 - h[:N - k])
            grad[:N - k] -= h[k:]
    grad /= len(idxs)
    return grad, max_val

import time
import numpy as np

def _c5_of_h(h):
    """Compute C5 for a given step function h using the prescribed metric."""
    n = h.shape[0]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * dx

def _project_to_box_sum(v, target):
    """Project v onto the box [0,1]^n while enforcing sum(v)=target."""
    # binary search for a shift lambda such that sum(clip(v+lambda,0,1)) = target
    lo = -np.max(v) - 1.0
    hi = 1.0 - np.min(v) + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        x = np.clip(v + mid, 0.0, 1.0)
        s = x.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + lo, 0.0, 1.0)
    # tiny correction for numerical drift
    residual = target - x.sum()
    if abs(residual) > 1e-12:
        free = (x > 0.0) & (x < 1.0)
        cnt = free.sum()
        if cnt:
            x[free] += residual / cnt
            x = np.clip(x, 0.0, 1.0)
        else:
            # all at bounds – distribute uniformly (should be rare)
            x += residual / len(x)
            x = np.clip(x, 0.0, 1.0)
    return x

def _worst_shift_gradient(h):
    """Sub‑gradient of max_lag  Σ h_i (1−h_{i+ℓ}) w.r.t. h, using the worst lag(s)."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    maxv = corr.max()
    idxs = np.flatnonzero(corr == maxv)
    grad = np.zeros_like(h)
    for idx in idxs:
        ell = idx - (n - 1)          # actual shift (may be negative)
        if ell >= 0:
            grad[: n - ell] += (1.0 - h[ell:])
            grad[ell:] -= h[: n - ell]
        else:
            k = -ell
            grad[k:] += (1.0 - h[: n - k])
            grad[: n - k] -= h[k:]
    grad /= len(idxs)
    return grad, maxv
