def run(seed=42, budget_s=1100, **kwargs):
    """
    Constructs a simple linear step‐function on [0,2] that satisfies the
    constraints and gives a very good upper bound for C₅.
    The function h(x) = x/2 (discretised) yields

    # Choose a moderate even number of samples (smaller than 1000 for speed)
    n = int(kwargs.get("n_points", 200))
    if n % 2 == 1:

    # Linear function h(x)=x/2 on [0,2], discretised as (i+0.5)/n

    # The construction guarantees this exactly in exact arithmetic.
    dx = 2.0 / n
    # Compute the C₅ bound using the same formula as the evaluator
    c5 = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
