def build_lp(h0):
    N = len(h0)
    dx = 2.0/N
    num_shifts = 2*N-1
    A_ub = np.empty((num_shifts, N+1), dtype=np.float64)
    b_ub = np.empty(num_shifts, dtype=np.float64)
    for idx, s in enumerate(range(-N+1,N)):
        abs_s = abs(s)
        O_s0 = 1.0 - dx * np.dot(h0[:N-abs_s], h0[abs_s:])
        # grad
        grad = np.zeros(N)
        if s >= 0:
            grad[:N-s] += -dx * h0[s:]
            grad[s:] += -dx * h0[:N-s]
        else:
            k = -s
            grad[k:] += -dx * h0[:N-k]
            grad[:N-k] += -dx * h0[k:]
        rhs = O_s0 - grad @ h0
        # row: grad coefficients for h, -1 for t
        A_ub[idx,:N] = grad
        A_ub[idx,N] = -1.0
        b_ub[idx] = -rhs
    # equality constraint: sum h = N/2
    A_eq = np.zeros((1,N+1))
    A_eq[0,:N] = 1.0

    c = np.zeros(N+1); c[N]=1.0 # minimize t
    bounds = [(0.0,1.0)]*N + [(0.0, None)]
    return c, A_ub, b_ub, A_eq, b_eq, bounds

def run(seed=42,budget_s=1100,**kwargs):
    import numpy as np, time
    from scipy.optimize import linprog

    np.random.seed(seed)
    start = time.time()
    # Determine initial h0 and N
    if 'initial_h_values' in globals():
        base = np.array(initial_h_values, dtype=np.float64)
        base_len = base.shape[0]
        # Choose factor to get N moderate at most 500
        factor = 5
        N = base_len * factor
        h0 = np.repeat(base, factor)
    else:
        N = 200
        h0 = np.zeros(N, dtype=np.float64)
        idx = np.random.choice(N, N//2, replace=False)
        h0[idx] = 1.0

    dx = 2.0 / N

    def compute_c5(vec):
        O = np.correlate(vec, 1-vec, mode='full') * dx
        return O.max()

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # iteration parameters
    max_iter = 50
    iter_num = 0

    while time.time() - start < budget_s * 0.95 and iter_num < max_iter:
        iter_num += 1
        # Build linear program data for current h0
        N = h0.shape[0]
        dx = 2.0 / N
        num_cons = 2 * N - 1
        # allocate arrays
        A_ub = np.empty((num_cons, N + 1), dtype=np.float64)
        b_ub = np.empty(num_cons, dtype=np.float64)

        # constraints for each shift s
        const_idx = 0
        for s in range(-N + 1, N):
            # overlapping index range
            if s >= 0:
                i_start = 0
                i_end = N - s
                x = h0[i_start:i_end]
                y = h0[i_start + s:i_end + s]  # = h0[s:]
                # O_s0
                O0 = dx * np.dot(x, 1.0 - y)
                # gradient
                grad = np.zeros(N, dtype=np.float64)
                # linear term indicator for prefix
                grad[i_start:i_end] += dx
                # forward shift term
                grad[i_start:i_end] -= dx * y
                # backward shift term on suffix part
                grad[s:] -= dx * x
            else:
                k = -s
                i_start = k
                i_end = N
                x = h0[i_start:i_end]               # suffix
                y = h0[0:i_end - i_start]           # same length = h0[:N-k]
                O0 = dx * np.dot(x, 1.0 - y)
                grad = np.zeros(N, dtype=np.float64)
                # linear term for suffix indices
                grad[i_start:i_end] += dx
                # forward shift term (h_{i+s}=h_{i-k})
                grad[i_start:i_end] -= dx * y
                # backward shift term (h_{i-s}=h_{i+k})
                grad[0:i_end - i_start] -= dx * x

            rhs = O0 - grad.dot(h0)
            # build row for A_ub: grad for h, -1 for t
            A_ub[const_idx, :N] = grad
            A_ub[const_idx, N] = -1.0
            b_ub[const_idx] = -rhs
            const_idx += 1

        # equality constraint sum h = N/2
        A_eq = np.zeros((1, N + 1), dtype=np.float64)
        A_eq[0, :N] = 1.0
        b_eq = np.array([N / 2.0], dtype=np.float64)

        c = np.zeros(N + 1, dtype=np.float64)
        c[N] = 1.0  # minimize t

        bounds = [(0.0, 1.0)] * N + [(0.0, None)]

        # warm start initial guess if we have best_h approx?
        # linprog does not support warm start directly.

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            # if LP fails, break out
            break

        x = res.x
        h_new = x[:N]
        # Clip to [0,1] to correct small numerical violations
        h_new = np.clip(h_new, 0.0, 1.0)
        # Ensure sum close to N/2
        cur_sum = h_new.sum()
        if abs(cur_sum - N/2.0) > 1e-8:
            # renormalize proportionally while staying in bounds
            diff = N/2.0 - cur_sum
            # Adjust uniformly then clip
            h_new += diff / N
            h_new = np.clip(h_new, 0.0, 1.0)

        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-7:
            best_c5 = c5_new
            best_h = h_new.copy()
        # update h0 for next iteration
        h0 = h_new.copy()

        # early exit if we already beat target
        if best_c5 <= 0.3808:
            break

    n_points = best_h.shape[0]
    return (best_h.tolist(), float(best_c5), n_points)

                # forward shift (h_{i+s}=h_{i-k})
                grad[i_start:i_end] -= dx * y
                # backward shift (h_{i-s}=h_{i+k})
                grad[0:i_end - i_start] -= dx * x

            rhs = O0 - grad.dot(h0)          # constant of the linearised inequality
            # inequality: t - grad·h >= rhs  ->  grad·h - t <= -rhs
            A_ub[row, :N] = grad
            A_ub[row, N] = -1.0
            b_ub[row] = -rhs
            row += 1

        # equality: sum(h) = N/2
        A_eq = np.zeros((1, N + 1), dtype=np.float64)
        A_eq[0, :N] = 1.0
        b_eq = np.array([N / 2.0], dtype=np.float64)

        c = np.zeros(N + 1, dtype=np.float64)
        c[N] = 1.0                     # minimise t

        bounds = [(0.0, 1.0)] * N + [(0.0, None)]

        # solve the LP
        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break

        x = res.x
        h_new = np.clip(x[:N], 0.0, 1.0)

        # correct tiny drift in the integral constraint
        cur_sum = h_new.sum()
        if abs(cur_sum - N / 2.0) > 1e-8:
            diff = N / 2.0 - cur_sum
            h_new += diff / N
            h_new = np.clip(h_new, 0.0, 1.0)

        val = c5_of(h_new)

        if val < best_val - 1e-7:
            best_val = val
            best_h = h_new.copy()

        h0 = h_new.copy()

        if best_val <= 0.3808:
            break

    n_pts = best_h.shape[0]
    return (best_h.tolist(), float(best_val), n_pts)
