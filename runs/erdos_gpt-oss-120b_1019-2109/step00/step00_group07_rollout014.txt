def propose(h):
    i, j = np.random.choice(N, 2, replace=False)
    max_delta = min(0.1, 1 - h[i], h[j])
    min_delta = -min(0.1, h[i], 1 - h[j]) # moving from j to i
    if max_delta <= min_delta:
        return None # can't move
    delta = np.random.uniform(min_delta, max_delta)
    h_new = h.copy()
    h_new[i] += delta
    h_new[j] -= delta
    return h_new

def repair(h):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    total = h.sum()
    if total == target:
        return h
    # compute scaling factor, but if scaling pushes out-of-bounds, iterative
    for _ in range(10):
        scale = target / total
        h = np.clip(h * scale, 0.0, 1.0)
        total = h.sum()
        if abs(total - target) < 1e-12:
            break
    # final linear adjustment
    diff = target - total
    if abs(diff) > 1e-12:
        # distribute diff across entries not at bounds
        free = (h > 0) & (h < 1)
        if np.any(free):
            # distribute equally
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            # fallback naive: adjust first entry
            h[0] = np.clip(h[0] + diff, 0.0, 1.0)
    # final clipping
    h = np.clip(h, 0.0, 1.0)
    return h

            if total > 0:

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: enforce the box constraints and the exact integral constraint
# ----------------------------------------------------------------------
def _repair(vec, target):
    """Project vec onto the box [0,1] with sum(vec)=target."""
    v = np.clip(vec, 0.0, 1.0)
    s = v.sum()
    if abs(s - target) > 1e-12:
        # simple scaling loop – it converges quickly for our small vectors
        for _ in range(10):
            if s == 0:
                break
            scale = target / s
            v *= scale
            v = np.clip(v, 0.0, 1.0)
            s = v.sum()
            if abs(s - target) < 1e-12:
                break
        # distribute any tiny remainder uniformly over non‑saturated entries
        diff = target - s
        if abs(diff) > 1e-12:
            free = (v > 0.0) & (v < 1.0)
            if free.any():
                v[free] += diff / free.sum()
                v = np.clip(v, 0.0, 1.0)
            else:
                # fallback – adjust the first entry
                v[0] = np.clip(v[0] + diff, 0.0, 1.0)
    return v


# ----------------------------------------------------------------------
# Helper: evaluate the objective (uses the provided evaluator if present)
# ----------------------------------------------------------------------
def _c5_value(vec):
    """Return the C5 value for a candidate vector."""
    N = vec.size
    # Prefer the official evaluator when it exists
    try:
        # evaluate_erdos_solution is injected in the environment
        return evaluate_erdos_solution(vec)
    except NameError:
        # fallback – the specification’s reference implementation
        corr = np.correlate(vec, 1.0 - vec, mode="full")
        return np.max(corr) * (2.0 / N)


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def run(seed=42, budget_s=1100, **kwargs):
    """
    Returns a tuple (h_values, c5_bound, n_points) where
        h_values : 1‑D NumPy array of the step function values,
        c5_bound : the obtained C5 value,
        n_points : length of the discretisation.
    """

    # --------------------------------------------------------------
    # initialise problem size and starting vector
    # --------------------------------------------------------------
    try:
        init = np.array(initial_h_values, dtype=float)
        N = init.size
    except NameError:
        N = 80                     # a modest default size
        init = np.full(N, 0.5)    # uniform start (satisfies the integral)

    target_sum = N / 2.0
    h = _repair(init.copy(), target_sum)

    # initial evaluation
    cur_val = _c5_value(h)
    best_h = h.copy()
    best_val = cur_val

    # if the given construction already beats the target we can exit early
    TARGET = 0.38085
    if best_val <= TARGET:
        return best_h, best_val, N

    # --------------------------------------------------------------
    # simulated‑annealing loop
    # --------------------------------------------------------------
    start_time = time.time()
    max_iter = 2_000_000                     # upper bound – real stop is time‑based
    TEMP0 = 0.12
    STEP0 = 0.35

    for it in range(max_iter):
        # time‑budget guard
        if time.time() - start_time > budget_s * 0.95:
            break

        # linear cooling schedules
        progress = it / max_iter
        temp = TEMP0 * (1.0 - progress) + 1e-6
        step = STEP0 * (1.0 - progress) + 0.005

        # ------------------------------------------------------------------
        # propose a move that preserves the total mass exactly
        # ------------------------------------------------------------------
        i, j = rng.integers(0, N, size=2)
        if i == j:

        # move mass from i to j (or the opposite, chosen at random)
        if rng.random() < 0.5:

        else:

        max_transfer = min(step, h[src], 1.0 - h[dst])

        h_new = h.copy()
        h_new[src] -= delta
