        def project_to_feasible(x):
            # project onto { sum = S, 0<=x_i<=1 }
            S = n/2
            x = np.clip(x, 0.0, 1.0)
            # iterative algorithm: water-filling
            # Start with x; adjust to meet sum.
            # While sum != S, adjust remaining free variables.
            # We'll implement using algorithm for projection onto capped simplex.
            # Use sorted array
            # Source: Duchi et al.
            # We need projection onto hyperplane with box constraints.
            # Equivalent to solve min ||y - x||^2 subject to sum(y) = S, 0 <= y <=1.
            # This can be done using lambda shift and clipping.
            # Let y_i = clip(x_i + lambda, 0,1). Find lambda such that sum y = S.
            # This is monotonic in lambda; solve via binary search.
            low = -1.0  # enough negative
            high = 1.0  # enough positive

                if c < best_c:

                    best_h = h.copy()
            # occasionally do a random restart if stuck?

def project_to_feasible(x):
    S = len(x) / 2.0
    # lambda binary search
    lo = -1.0
    hi = 1.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = np.sum(y)
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

import time, math, random
import numpy as np

def _project_to_feasible(x: np.ndarray) -> np.ndarray:
    """Project x onto {0<=y_i<=1, sum(y)=len(x)/2}."""
    n = x.shape[0]
    S = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(40):
        lam = (lo + hi) / 2.0
        y = np.clip(x + lam, 0.0, 1.0)
        s = y.sum()
        if s > S:
            hi = lam
        else:
            lo = lam
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)


def _compute_c5(h: np.ndarray) -> float:
    """Return max_k ∫ h(x)(1-h(x+k)) dx approximated on the discrete grid."""
    n = h.shape[0]
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr.max() * (2.0 / n)


def _random_feasible(n: int) -> np.ndarray:
    """Generate a random point satisfying the constraints."""
    x = np.random.rand(n)
    return _project_to_feasible(x)


def run(seed: int = 42, budget_s: int = 1100, **kwargs):
    """
    Return a step function h (as a NumPy array), its C5 bound, and point count.
    """
    random.seed(seed)
    np.random.seed(seed)

    # Determine problem size -------------------------------------------------
    init_vals = globals().get("initial_h_values")
    if init_vals is not None:
        h0 = np.array(init_vals, dtype=np.float64)
        n = h0.shape[0]
    else:
        n = 100
        h0 = None

    # Prepare initial candidates ---------------------------------------------
    candidates = []
    if h0 is not None:
        h0 = _project_to_feasible(h0)
        candidates.append(h0.copy())
    for _ in range(3):
        candidates.append(_random_feasible(n))

    best_h = None
    best_c = float("inf")

    # Evaluate all initial candidates
    for h in candidates:
        c = _compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()

    start_time = time.time()
    time_limit = budget_s * 0.95  # margin

    # Simulated‑annealing loop -----------------------------------------------
    def _anneal(start_h: np.ndarray):
        nonlocal best_h, best_c, start_time
        h = start_h.copy()
        c = _compute_c5(h)

        max_iters = 800_000
        init_T, final_T = 0.12, 1e-6

        for it in range(max_iters):
            # stop if out of time budget
            if time.time() - start_time > time_limit:
                break
            # linear cooling
            T = init_T + (final_T - init_T) * (it / max_iters)

            # ----------------------------------------------------------------
            # Targeted move (probability 0.6)
            # ----------------------------------------------------------------
            if np.random.rand() < 0.6:
                corr = np.correlate(h, 1.0 - h, mode="full")
                idx = int(corr.argmax())
                shift = idx - (n - 1)

                # indices where the pair (i, i+shift) exists
                if shift >= 0:
                    i_vals = np.arange(0, n - shift)
                else:
                    i_vals = np.arange(-shift, n)

                if i_vals.size:
                    diff = h[i_vals] - h[i_vals + shift]
                    i_sel = i_vals[np.argmax(diff)]
                    j_sel = i_sel + shift
                    if h[i_sel] > h[j_sel]:
                        max_move = min(h[i_sel], 1.0 - h[j_sel])
                        if max_move > 0:
                            eps = np.random.rand() * max_move
                            h_new = h.copy()
                            h_new[i_sel] -= eps
                            h_new[j_sel] += eps
                    else:
                        max_move = min(h[j_sel], 1.0 - h[i_sel])
                        if max_move > 0:
                            eps = np.random.rand() * max_move
                            h_new = h.copy()
                            h_new[j_sel] -= eps
                            h_new[i_sel] += eps
                    else_skip = False
                else:
                    else_skip = True

                if not else_skip:

                    delta = c_new - c
                    if delta < 0 or math.exp(-delta / (T + 1e-12)) > np.random.rand():
                        h, c = h_new, c_new
                        if c < best_c:
                            best_c, best_h = c, h.copy()
                    continue  # go to next iteration

            # Random pairwise swap move
            # ----------------------------------------------------------------
            i, j = np.random.randint(0, n, size=2)
            while i == j:
                j = np.random.randint(0, n)

            if np.random.rand() < 0.5:

                if max_move <= 0:
                    continue

                h_new = h.copy()
                h_new[i] += eps
                h_new[j] -= eps
            else:
                max_move = min(h[i], 1.0 - h[j])

                eps = np.random.rand() * max_move

        best_c = _compute_c5(best_h)

    return best_h, best_c, n
