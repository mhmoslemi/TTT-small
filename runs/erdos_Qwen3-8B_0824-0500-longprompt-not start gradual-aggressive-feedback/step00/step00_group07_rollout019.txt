```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial guess
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h = (n_points / 2) / np.sum(h) * h
    
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h_pad = np.concatenate([h, np.zeros_like(h)])
        corr = signal.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_n = n_points
    
    # Try reducing n_points to speed up computation
    for n in range(100, n_points, -1):
        if n < 50:
            break
        # Generate random samples
        h = np.random.rand(n)
        h = (n / 2) / np.sum(h) * h
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
    
    # Now do a more refined search around the best_h
    for _ in range(5):
        # Use scipy optimization
        def objective(x):
            h = x.copy()
            dx = 2.0 / len(h)
            h_pad = np.concatenate([h, np.zeros_like(h)])
            corr = signal.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        # Bounds and initial guess
        bounds = [(0.0, 1.0) for _ in range(len(best_h))]
        result = opt.minimize(objective, best_h, bounds=bounds, method='L-BFGS-B', tol=1e-6)
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h)
        else:
            break
    
    # Ensure sum of h is correct
    sum_h = np.sum(best_h)
    if np.abs(sum_h - best_n / 2) > 1e-6:
        best_h = (best_n / 2) / sum_h * best_h
    
    # Final check
    best_c5 = compute_c5(best_h)
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return best_h, best_c5, best_n
    
    return best_h, best_c5, best_n
```
