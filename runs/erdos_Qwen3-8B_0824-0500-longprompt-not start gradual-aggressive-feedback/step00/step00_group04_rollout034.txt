```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.optimize import minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Initial number of points
    n_points = 100  # Start with 100 points for speed
    max_points = 200  # Limit to prevent excessive computation
    best_c5 = float('inf')
    best_h = None
    best_n = n_points
    
    def evaluate(h):
        # Ensure h is within [0,1]
        h_clamped = np.clip(h, 0, 1)
        # Compute the overlap integral using convolution
        conv = np.correlate(h_clamped, 1 - h_clamped, mode="full")
        dx = 2.0 / len(h_clamped)
        c5 = np.max(conv) * dx
        return c5

    def objective_func(h):
        # Objective function: minimize the maximum overlap
        return evaluate(h)

    def initial_h_values_func(n_points):
        # Start with a uniform distribution
        h = np.ones(n_points) * (n_points / 2) / 2
        return h

    # Start with the initial h
    initial_h = initial_h_values_func(n_points)
    initial_c5 = evaluate(initial_h)
    best_c5 = initial_c5
    best_h = initial_h.copy()
    best_n = n_points

    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Try increasing the number of points
        if n_points < max_points:
            n_points += 1
            initial_h = initial_h_values_func(n_points)
            initial_c5 = evaluate(initial_h)
            best_c5 = initial_c5
            best_h = initial_h.copy()
            best_n = n_points
        
        # Use a simple gradient descent with line search
        result = minimize(objective_func, best_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
                best_n = n_points
        
        # Try a few random perturbations
        for _ in range(5):
            perturbation = np.random.normal(0, 0.05, n_points)
            h_perturbed = best_h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            c5_perturbed = evaluate(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
                best_n = n_points
    
    return best_h, best_c5, best_n
```
