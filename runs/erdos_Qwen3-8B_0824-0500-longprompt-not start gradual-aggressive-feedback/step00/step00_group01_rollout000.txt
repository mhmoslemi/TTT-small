import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random

# Pre-imported helper functions and initial values
# evaluate_erdos_solution() is used to compute C_5 for a given h
# initial_h_values is a sample initial configuration

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points_initial = len(initial_h_values)
    h_initial = initial_h_values
    
    # Try improving the initial configuration
    n_points = n_points_initial
    h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Try different numbers of points
    for n_points in [n_points_initial, n_points_initial // 2, n_points_initial * 2]:
        if time.time() - start_time > budget_s - 3:
            break
        # Ensure n_points is at least 20
        if n_points < 20:
            n_points = 20
        
        # Create initial guess
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        total_area = n_points / 2
        
        # Random initial guess
        h = np.random.rand(n_points)
        h = h * (total_area / np.sum(h))
        
        # Optimization using scipy.optimize
        def objective(x):
            h = x.copy()
            # Ensure h stays within [0, 1]
            h = np.clip(h, 0, 1)
            # Normalize to have area 1
            h = h * (total_area / np.sum(h))
            # Compute the maximum overlap
            corr = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Constraints: sum(h) == total_area
        cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - total_area})
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Run optimization
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * (total_area / np.sum(h))
        
        # Evaluate the result
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Return the best solution found
    return (best_h, best_c5, len(best_h))
