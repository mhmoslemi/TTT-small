```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions and variables
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assume this is a preloaded numpy array

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different numbers of points
    n_points_list = [200, 300, 400]
    best_c5 = np.inf
    best_h = None
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess based on initial_h_values or random
        if n_points < len(initial_h_values):
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        else:
            h = np.random.rand(n_points)
        
        # Normalize h to meet the constraint sum(h) == n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Define decision variables
        h_var = cp.Variable(n_points, nonneg=True, boolean=False)
        dx = 2.0 / n_points
        total_area = cp.sum(h_var) * dx
        constraint = total_area == 1
        
        # Define the overlap integral for each shift k
        # We'll evaluate shifts k in {-n_points//2, ..., n_points//2}
        k_values = np.arange(-n_points//2, n_points//2 + 1)
        overlap_integrals = []
        
        for k in k_values:
            if k < 0:
                h_shifted = np.zeros(n_points)
                for i in range(n_points):
                    j = (i + k) % n_points
                    h_shifted[i] = h_var[j]
            elif k > 0:
                h_shifted = np.zeros(n_points)
                for i in range(n_points):
                    j = (i - k) % n_points
                    h_shifted[i] = h_var[j]
            else:
                h_shifted = h_var
            
            overlap_integral = cp.sum(h_var * (1 - h_shifted)) * dx
            overlap_integrals.append(overlap_integral)
        
        # Objective is to minimize the maximum overlap integral
        objective = cp.Minimize(cp.Max(overlap_integrals))
        prob = cp.Problem(objective, [constraint])
        
        try:
            prob.solve(solver=cp.SCS, max_iter=100, verbose=False)
            if prob.status == 'optimal':
                h_opt = h_var.value
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
        except:
            pass
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s:
            break
    
    # Return the best solution found
    return (best_h, best_c5, len(best_h))
```
