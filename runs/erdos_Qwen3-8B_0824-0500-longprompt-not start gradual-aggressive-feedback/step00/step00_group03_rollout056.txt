```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import sys

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Example initial configuration

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points for efficiency
    max_iter = 100
    best_c5 = float('inf')
    best_h = None
    
    # Start with the initial configuration
    h = np.copy(initial_h_values[:n_points])
    h = h / (np.sum(h) * 2.0 / n_points) * (n_points / 2.0)
    
    # Evaluate initial solution
    current_c5 = evaluate_erdos_solution(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Optimization loop
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 0.5:
            break
        
        # Define optimization variables
        h_var = cp.Variable(n_points, bounds=(0, 1))
        constraints = [
            cp.sum(h_var) == n_points / 2,  # Enforce integral constraint
        ]
        
        # Objective: minimize the maximum overlap
        max_overlap = cp.Maximize(cp.Max(cp.sum(h_var * (1 - h_var[k:] - h_var[:-k])) for k in range(1, n_points)))
        prob = cp.Problem(max_overlap, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iter=100)
            if prob.status == "optimal":
                h_new = np.round(h_var.value, 5)
                current_c5 = evaluate_erdos_solution(h_new)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_new.copy()
        except:
            pass
        
        # Introduce small perturbations to escape local minima
        h = best_h + 0.05 * np.random.randn(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h)
    return (best_h, best_c5, n_points)
```
