import numpy as np

# Pre-imported functions

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial parameters
    n_points = 45  # Starting with 45 samples
    dx = 2.0 / n_points
    
    # Initial h values
    h = np.array(initial_h_values).copy()
    h = h[:n_points]  # Ensure it's of correct length
    
    # Ensure the integral condition is satisfied
    h_sum = np.sum(h) * dx
    if abs(h_sum - 1) > 1e-8:
        # Normalize if needed
        h = h * (1 / h_sum) if h_sum != 0 else np.zeros_like(h)
    
    # Function to compute the maximum overlap
    def compute_c5(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        h = np.concatenate([h, np.zeros_like(h)])  # Extend to avoid boundary issues
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Function to optimize the function h
    def optimize_h(h_initial, n_points, budget_s):
        h = h_initial.copy()
        h = h[:n_points]  # Ensure the length is correct
        h = h.reshape(-1)
        h = np.concatenate([h, np.zeros_like(h)])  # Extend to avoid boundary issues
        
        # Objective: minimize the maximum overlap
        n = len(h)
        dx = 2.0 / n_points
        h = h[:n_points]
        
        # Optimization variables
        x = cp.Variable(n_points, lb=0, ub=1)
        
        # Objective: minimize the maximum overlap
        # We'll use a convex reformulation with binary variables and constraints
        
        # Create copies of h for all shifts
        h_shifted = [np.roll(x, k) for k in range(-n_points + 1, n_points)]
        
        # Objective function: max over all k of integral(h(x) * (1 - h(x + k)) dx)
        # This is approximated with the correlation
        corr = np.array([np.dot(x, 1 - np.roll(x, k)) for k in range(-n_points + 1, n_points)])
        obj = cp.Minimize(cp.Max(corr * dx))
        
        # Constraints: integral(h(x)) = 1
        constraint = [cp.sum(x) * dx == 1]
        
        # Solve the problem
        prob = cp.Problem(obj, constraint)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4, max_iters=1000, time_limit=budget_s - (time.time() - start_time))

                h_opt = x.value
                return h_opt
            else:
                print("Optimization failed with status:", prob.status)
                return h
        except:
            print("Optimization failed")
            return h
    
    # Run the optimization
    h_opt = optimize_h(h, n_points, budget_s)
    
    # Evaluate the result
    c5_bound = compute_c5(h_opt)
    
    # Output
    return h_opt, c5_bound, n_points
