    def compute_c5(h_values):
        h = np.array(h_values)
        # Ensure h is length n_points
        h = h[:n_points]
        # Ensure sum(h) is correct
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Compute the max overlap integral
        # The overlap integral is the max of the correlation scaled by dx
        c5 = np.max(corr) * dx
        return c5
    
    # Function to minimize the overlap integral
    def minimize_overlap(h_values):
        h = np.array(h_values)

        h = np.clip(h, 0, 1)

        # Initial guess
