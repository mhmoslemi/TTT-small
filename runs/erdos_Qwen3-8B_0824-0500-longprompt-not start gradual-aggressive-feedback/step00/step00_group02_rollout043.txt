```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions and variables
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points and initial solution
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    
    # Initial C5 value
    c5_initial = evaluate_erdos_solution(h_initial)
    
    best_h = h_initial
    best_c5 = c5_initial
    best_n_points = n_points_initial
    
    # Try a few different numbers of points (e.g., 100, 200, 300) to avoid local optima
    for n_points in [100, 200, 300]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Check if we have enough time to try this n_points
        if n_points >= n_points_initial:
            # Optimize directly from the initial solution
            h = h_initial.copy()
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), h)
            h = np.clip(h, 0, 1)
            
            # Optimize using coordinate descent
            for _ in range(50):  # Limit number of iterations to save time
                # Compute the overlap integral for all shifts
                overlap = np.zeros(n_points)
                for k in range(n_points):
                    shifted = np.roll(h, k)
                    overlap[k] = np.dot(h, 1 - shifted) * (2.0 / n_points)
                
                # Find the maximum overlap
                max_overlap = np.max(overlap)
                
                # If max_overlap is already low, we can stop early
                if max_overlap < best_c5:
                    best_c5 = max_overlap
                    best_h = h.copy()
                    best_n_points = n_points
                    break
                
                # Update h to reduce the largest overlap
                max_k = np.argmax(overlap)
                shifted = np.roll(h, max_k)
                h = h * (1.0 - (overlap[max_k] / np.dot(h, 1 - shifted)))
                h = np.clip(h, 0, 1)
        
        # Evaluate the current solution
        c5_current = evaluate_erdos_solution(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            best_n_points = n_points
    
    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h)
    return best_h, best_c5, best_n_points
```
