```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 points
    max_iter = 100  # Max iterations for optimization
    eps = 1e-4  # Tolerance for convergence
    
    # Initial guess
    h = np.array(initial_h_values)  # Use provided initial values
    h = h[:n_points]  # Ensure it's within the n_points limit
    h = np.roll(h, int(np.random.rand() * len(h)))  # Randomly shift to avoid local optima
    
    # Ensure the integral condition is satisfied
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h if sum_h != 0 else h
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Define variables
        h_var = cp.Variable(n_points, lower=0, upper=1)
        
        # Objective: minimize the max overlap
        # We will use a convex relaxation by minimizing the maximum over all shifts
        # For simplicity, we'll optimize for a subset of shifts
        shifts = np.arange(1, n_points // 2 + 1)
        obj = cp.Minimize(cp.Max([cp.sum(h_var * (1 - cp.roll(h_var, shift))) for shift in shifts]))
        
        # Constraints: integral condition
        constraint = [cp.sum(h_var) == n_points / 2]
        
        # Solve the problem using ECOS (which is suitable for this type of problem)
        prob = cp.Problem(obj, constraint)
        try:
            prob.solve(solver=cp.ECOS, eps=eps)
        except:
            # If optimization fails, try with a smaller epsilon
            prob.solve(solver=cp.ECOS, eps=eps / 10)
        
        if prob.status != 'OPTIMAL':
            break
        
        # Update the solution
        h = h_var.value
        current_c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Adjust n_points if needed (optional)
        # n_points = min(n_points + 10, 500)  # Increase sample points gradually
    
    # Ensure the final h is within bounds and integral condition is met
    sum_h = np.sum(best_h)
    best_h = best_h * (n_points / 2) / sum_h if sum_h != 0 else best_h
    
    return (best_h, best_c5, n_points)
```
