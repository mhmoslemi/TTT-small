```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 70
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values[:n_points])
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        shifted_h = np.zeros_like(h)
        for k in range(1, n_points):
            shifted_h[k:] = h[:-k]
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Optimization function
    def optimize_h(initial_h, max_iter=100, tolerance=1e-5):
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h)
        best_h_values = best_h.copy()
        
        for _ in range(max_iter):
            # Create a new h with small perturbations
            h = best_h + 0.01 * np.random.randn(len(best_h))
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_h_values = h.copy()
        
        return best_h_values, best_c5
    
    # Start optimization
    start_time = time.time()
    best_h, best_c5 = optimize_h(h_initial)
    
    # Try improving with more points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        n_points = 100
        dx = 2.0 / n_points
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        best_h, best_c5 = optimize_h(h_initial, max_iter=50)
    
    # Final check: use CVXPY for more accurate optimization if time allows
    if remaining_time > 20:
        n_points = 100
        dx = 2.0 / n_points
        x = cp.Variable(n_points)
        constraints = [cp.minimize(0), cp.sum(x) == n_points / 2, cp.minimize(0)]
        obj = cp.Maximize(cp.max(cp.sum(x * (1 - x[cp.diag(np.eye(n_points))])) * dx))
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.SCS, max_iter=100)
        best_h = x.value
        best_c5 = result
    
    return best_h, best_c5, n_points
```
