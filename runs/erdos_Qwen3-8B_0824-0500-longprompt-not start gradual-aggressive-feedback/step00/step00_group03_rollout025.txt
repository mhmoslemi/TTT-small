```python
import numpy as np
import scipy.optimize as opt
import time
import copy
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run_seedless(budget_s=60, n_points=100, seed=42):
    # Use the initial_h_values as a starting point
    h_start = initial_h_values.copy()
    # Normalize to ensure the integral is exactly 1
    h_start /= np.sum(h_start) * 2.0 / n_points
    h_start = np.clip(h_start, 0, 1)
    
    # Initialize best solution
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    # Optimization parameters
    max_iterations = 50
    learning_rate = 0.1
    min_learning_rate = 1e-5
    decay_rate = 0.995
    epsilon = 1e-8
    
    # Main optimization loop
    start_time = time.time()
    
    for i in range(max_iterations):
        # Compute current C5
        current_c5 = evaluate_erdos_solution(best_h, n_points)
        
        # If we've reached the budget, stop
        if time.time() - start_time > budget_s:
            break
        
        # Compute gradient of C5 with respect to h
        # Use finite differences for gradient estimation
        grad = np.zeros_like(best_h)
        for idx in range(n_points):
            # Small perturbation
            h_perturbed = best_h.copy()
            h_perturbed[idx] += epsilon
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
            # Compute C5 for perturbed h
            c5_perturbed = evaluate_erdos_solution(h_perturbed, n_points)
            
            # Compute gradient
            grad[idx] = (c5_perturbed - current_c5) / epsilon
        
        # Update h using gradient descent
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-6:
            h_update = -learning_rate * grad / grad_norm
            best_h += h_update
            best_h = np.clip(best_h, 0, 1)
        
        # Adjust learning rate
        learning_rate *= decay_rate
        if learning_rate < min_learning_rate:
            learning_rate = min_learning_rate
        
        # Ensure the integral constraint is satisfied
        integral = np.sum(best_h) * 2.0 / n_points
        if abs(integral - 1) > 1e-6:
            # Normalize h to maintain integral constraint
            best_h = best_h / integral * 1
            best_h = np.clip(best_h, 0, 1)
        
        # Update best solution
        current_c5 = evaluate_erdos_solution(best_h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return best_h, final_c5, n_points

def run(seed=42, budget_s=60, **kwargs):
    # Set default n_points to 100, which is faster to optimize
    n_points = 100
    h_values, c5_bound, n_points = run_seedless(budget_s, n_points, seed)
    return h_values, c5_bound, n_points
```
