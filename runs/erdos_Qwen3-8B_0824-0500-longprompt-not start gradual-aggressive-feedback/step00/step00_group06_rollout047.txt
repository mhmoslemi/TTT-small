```python
import numpy as np
import scipy.optimize as opt
import sys
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    h_initial = initial_h_values.copy()
    n_points_initial = len(h_initial)
    
    # Try different n_points starting from the initial one
    for n_points in [n_points_initial, n_points_initial // 2, n_points_initial // 4, n_points_initial // 8, n_points_initial // 16]:
        if n_points < 4:
            continue
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Initialize h with random values in [0, 1]
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        
        # Try improving from initial solution
        if n_points == n_points_initial:
            h = copy.copy(h_initial)
        
        # Optimization with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Define objective function
        def objective(x):
            h = x
            integral = 0.0
            for k in range(1, n_points):
                if k >= n_points:
                    break
                shifted_h = np.roll(h, k)
                integral += np.dot(h, 1 - shifted_h)
            integral /= (n_points - 1)
            return integral
        
        # Use L-BFGS-B as the optimization method
        result = opt.minimize(
            objective,
            h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        # Get the best solution
        h_opt = result.x
        c5_bound = evaluate_erdos_solution(h_opt, n_points, dx)
        
        # Check if we are within the budget
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        
        # Print the result
        print(f"n_points={n_points}, c5_bound={c5_bound}")
        
        # Check if we have a better solution
        if c5_bound < 0.380876:
            print(f"Improved bound found: {c5_bound}")
            return (h_opt, c5_bound, n_points)
    
    # If no better solution found, return the best one so far
    return (h_initial, evaluate_erdos_solution(h_initial, n_points_initial, 2.0 / n_points_initial), n_points_initial)
```
