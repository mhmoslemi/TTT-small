```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from functools import lru_cache

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Chosen for a balance between accuracy and performance
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    initial_h = np.resize(initial_h, n_points) * (n_points / 2) / np.sum(initial_h)
    h = initial_h.copy()

    def objective(h):
        h = np.resize(h, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        # Compute the overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def simulate_and_optimize(h_start):
        h = h_start.copy()
        for _ in range(100):
            # Use L-BFGS-B for optimization
            res = opt.minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 10, 'ftol': 1e-7, 'gtol': 1e-7}
            )
            h = res.x
            if res.success:
                c5 = objective(h)
                if c5 < 0.3808:
                    print(f"Found better solution: {c5}")
                    return h
        return h

    start_time = time.time()
    best_h = simulate_and_optimize(h)
    best_c5 = objective(best_h)
    
    # Final refinement using convex optimization
    h_refined = cp.Variable(n_points)
    constraints = [
        cp.sum(h_refined) == n_points / 2,
        0 <= h_refined,
        h_refined <= 1
    ]
    
    problem = cp.Problem(cp.Minimize(cp.Max(cp.multiply(h_refined, 1 - h_refined.shift(1, axis=0)))), constraints)
    problem.solve(solver='SCIPY', max_iter=100)
    best_h_refined = h_refined.value
    best_c5_refined = objective(best_h_refined)
    
    # Ensure it's within budget
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Do one more refinement step
        h_refined = cp.Variable(n_points)
        constraints = [
            cp.sum(h_refined) == n_points / 2,
            0 <= h_refined,
            h_refined <= 1
        ]
        
        problem = cp.Problem(cp.Minimize(cp.Max(cp.multiply(h_refined, 1 - h_refined.shift(1, axis=0)))), constraints)
        problem.solve(solver='SCIPY', max_iter=100)
        best_h_refined = h_refined.value
        best_c5_refined = objective(best_h_refined)
    
    return (best_h_refined, best_c5_refined, n_points)
```
