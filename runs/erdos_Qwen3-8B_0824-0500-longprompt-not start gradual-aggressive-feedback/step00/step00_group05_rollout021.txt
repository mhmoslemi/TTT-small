```python
import numpy as np
import scipy.optimize as opt
import scipy.interpolate as interp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))
initial_h_values = np.random.rand(70) * 2 - 1  # placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points for a balance between speed and accuracy
    h_values = np.random.rand(n_points) * 2 - 1  # Initial random guess

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        return evaluate_erdos_solution(h)

    # Function to evaluate the objective (we minimize the max overlap)
    def objective(h):
        return evaluate_overlap(h)

    # Function to compute the constraint (integral of h(x) over [0, 2] should be 1)
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Constraints for the optimization
    constraints = ({'type': 'eq', 'fun': constraint})
    bounds = [(0, 1) for _ in range(n_points)]

    # Run optimization
    result = opt.minimize(
        objective,
        h_values,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        tol=1e-6,
        options={'maxiter': 100}
    )

    # Check if we exceeded time budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:
        return result.x, evaluate_overlap(result.x), n_points

    # Try increasing n_points if time allows
    remaining_time = budget_s - elapsed
    if remaining_time > 10:
        n_points = 200
        h_values = np.random.rand(n_points) * 2 - 1

        # Re-run optimization with more points
        result = opt.minimize(
            objective,
            h_values,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            tol=1e-6,
            options={'maxiter': 50}
        )

        # Check if we still have time
        elapsed = time.time() - start_time
        if elapsed < budget_s:
            return result.x, evaluate_overlap(result.x), n_points
        else:
            return result.x, evaluate_overlap(result.x), n_points
    else:
        return result.x, evaluate_overlap(result.x), n_points
```
