import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 samples, can increase if needed
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()  # Use provided initial guess
    initial_h = np.array(initial_h[:n_points])  # Truncate to n_points
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2)  # Ensure integral is 1
    h_values = initial_h.copy()
    
    best_c5 = np.inf
    best_h = h_values.copy()
    best_n_points = n_points
    
    start_time = time.time()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)  # Ensure integral is 1
        h = np.concatenate([h, np.zeros(n_points)])  # Pad for convolution
        result = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(result)
    
    # Optimization function
    def optimize_h(h_values, budget_s):
        nonlocal best_c5, best_h
        t0 = time.time()
        remaining = budget_s - (time.time() - t0)
        
        # Use a local optimization method
        result = optimize.minimize(
            lambda x: compute_c5(x),
            x0=h_values,
            bounds=[(0, 1) for _ in range(len(h_values))],
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6},
            jac=False
        )
        
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
                best_n_points = len(current_h)
        
        # Try increasing n_points if budget allows
        if remaining > 10:
            new_n_points = min(2 * n_points, 500)
            new_dx = 2.0 / new_n_points
            new_h = np.interp(np.linspace(0, 2, new_n_points), np.linspace(0, 2, len(h_values)), h_values)
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (new_n_points / 2)
            optimize_h(new_h, remaining - 10)
        
        return best_h, best_c5, best_n_points
    
    # Run the optimization
    final_h, final_c5, final_n_points = optimize_h(h_values, budget_s)
    
    # Ensure the final_h has the correct integral
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (final_n_points / 2)
    
    return final_h, final_c5, final_n_points
