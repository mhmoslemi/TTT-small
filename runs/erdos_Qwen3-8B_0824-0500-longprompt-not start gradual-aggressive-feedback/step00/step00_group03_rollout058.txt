```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with an initial solution
    n_points = 70
    h = initial_h_values.copy()
    
    # Optimize with fewer points for speed
    for n_points in [70, 60, 50, 40, 30, 20]:
        if time.time() - start_time > budget_s:
            break
        
        # Rescale to [0, 2] with n_points samples
        dx = 2.0 / n_points
        h = np.linspace(0, 2, n_points)
        h_values = np.zeros(n_points)
        h_values[:] = np.random.rand(n_points)
        h_values = h_values * 2.0 / (np.sum(h_values) / dx)
        h_values = h_values / (np.sum(h_values) / dx)
        
        # Evaluate the current solution
        current_c5 = evaluate_erdos_solution(h_values, n_points)
        print(f"n_points={n_points}, C₅={current_c5}")
        
        # Try to improve the solution
        # Define the optimization problem
        h_var = cp.Variable(n_points, bounds=(0.0, 1.0))
        objective = cp.Minimize(cp.Max([cp.sum(h_var * (1 - h_var[k:])) * dx for k in range(n_points)]))
        prob = cp.Problem(objective, [cp.sum(h_var) * dx == 1])
        
        # Solve the optimization problem
        try:
            result = prob.solve(solver=cp.SCS, max_iter=100, time_limit=5)
            if result is not None:
                h_values = h_var.value
                current_c5 = evaluate_erdos_solution(h_values, n_points)
                print(f"Optimized n_points={n_points}, C₅={current_c5}")
        except:
            pass
        
        # Update the best solution
        if current_c5 < 0.380876:
            best_h = h_values
            best_c5 = current_c5
            best_n = n_points
    
    # Return the best solution
    return (best_h, best_c5, best_n)
```
