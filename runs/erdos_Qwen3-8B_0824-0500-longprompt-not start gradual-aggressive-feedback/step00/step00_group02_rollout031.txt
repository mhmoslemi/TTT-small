def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute the cross-correlation with (1 - h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Find the maximum value of the correlation (overlap integral)
    c5 = np.max(corr) * dx
    return c5
