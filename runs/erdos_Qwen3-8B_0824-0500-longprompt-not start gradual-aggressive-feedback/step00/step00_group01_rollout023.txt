```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 256  # Start with 256 samples for efficiency
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Start from initial_h_values, rescaled to fit n_points
    if len(initial_h_values) > 0:
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    # Ensure sum(h) * dx = 1
    h = h / (np.sum(h) * dx) * (n_points / 2)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()

    # Optimization strategy: Use a grid search with simplex for faster convergence
    for _ in range(10):  # Iterative refinement
        # Define variables
        h_var = cp.Variable(n_points, name='h')
        # Objective: minimize the maximum overlap integral
        # For a fixed h, compute the correlation
        def objective_func(h):
            corr = signal.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx

        # Use a numerical optimization to find a better h
        res = opt.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'gtol': 1e-5, 'maxiter': 100}
        )
        h = res.x
        # Ensure sum(h) * dx = 1
        h = h / (np.sum(h) * dx) * (n_points / 2)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        if time.time() - start_time > budget_s - 1:
            break

    return best_h, best_c5, n_points
```
