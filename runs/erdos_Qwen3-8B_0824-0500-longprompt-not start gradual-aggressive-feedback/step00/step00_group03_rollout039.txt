```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import copy
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points, can increase if needed
    max_iterations = 100  # Limit the number of iterations for time constraints
    dx = 2.0 / n_points

    # Initial guess from provided initial_h_values (scaled to n_points=100)
    initial_h = initial_h_values
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = initial_h * (n_points / 2)  # Ensure sum is n_points / 2
    initial_h = initial_h / (np.sum(initial_h) / (n_points / 2))  # Normalize to ensure correct sum

    best_h = copy.deepcopy(initial_h)
    best_c5 = evaluate_erdos_solution(best_h, n_points, dx)

    for it in range(max_iterations):
        # Create a copy for optimization
        h = copy.deepcopy(best_h)
        h = h.astype(float)

        # Define the objective function with constraints
        def objective(x):
            return evaluate_erdos_solution(x, n_points, dx)

        # Define constraints: h[i] must be in [0,1], sum(h) = n_points / 2
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        ]

        # Use L-BFGS-B optimizer
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )

        # Check if the optimization was successful
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                print(f"Improved C5: {best_c5:.6f}")
        else:
            print("Optimization failed, using previous best")

        # Check time budget
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s - 5:  # Leave 5 seconds for final evaluation
            break

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    print(f"Final C5: {final_c5:.6f}")
    return (best_h, final_c5, n_points)
```
