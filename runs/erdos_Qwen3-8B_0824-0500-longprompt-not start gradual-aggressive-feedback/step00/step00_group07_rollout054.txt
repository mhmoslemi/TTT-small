import numpy as np
import scipy.optimize as optimize
import scipy.interpolate as interp
import time
from scipy.optimize import minimize
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Define parameters
    n_points = 80  # Start with the initial 80 samples
    dx = 2.0 / n_points
    max_iterations = 100
    tolerance = 1e-4  # Tolerance for convergence
    
    # Initial guess from pre-imported values
    h_initial = np.array(initial_h_values)  # Ensure it's a numpy array
    h_initial = h_initial[:n_points]  # Trim to n_points if necessary
    
    # Scale h_initial to satisfy the integral constraint
    h_initial = h_initial / (np.sum(h_initial) * dx)  # Normalize so that integral is 1

    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)  # Ensure values are in [0, 1]
        # Compute the overlap integral for all k
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        # Take the maximum overlap
        c5 = np.max(overlap)
        return c5

    def objective_with_gradient(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)  # Ensure values are in [0, 1]
        # Compute the overlap integral for all k
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        # Take the maximum overlap
        c5 = np.max(overlap)
        # Compute gradient numerically
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            c5_plus = objective(h_plus)
            grad[i] = (c5_plus - c5) / 1e-6
        return c5, grad.flatten()

    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = minimize(
        fun=objective,
        x0=h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        jac=None,
        options={
            'maxiter': max_iterations,
            'ftol': tolerance,
            'gtol': tolerance,
            'disp': False
        }
    )

    # Extract the best solution
    h_best = result.x
    h_best = np.clip(h_best, 0, 1)
    c5_best = objective(h_best)

    # Check if we can refine further or increase n_points
    # If time is left, try increasing n_points
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        # Try increasing n_points to 128
        n_points_new = 128
        dx_new = 2.0 / n_points_new
        h_best_new = np.zeros(n_points_new)
        
        # Interpolate the current best solution to new n_points
        h_best_interp = interp.interp1d(np.linspace(0, 2, n_points), h_best, kind='linear', fill_value="extrapolate")
        h_best_new = h_best_interp(np.linspace(0, 2, n_points_new))
        h_best_new = np.clip(h_best_new, 0, 1)
        
        # Normalize to ensure the integral is 1
        h_best_new = h_best_new / (np.sum(h_best_new) * dx_new) * dx_new * n_points_new / 2
        
        # Optimize the new solution
        result_new = minimize(
            fun=objective,
            x0=h_best_new,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points_new)],
            jac=None,
            options={
                'maxiter': 50,
                'ftol': tolerance,
                'gtol': tolerance,
                'disp': False
            }
        )
        h_best = result_new.x
        h_best = np.clip(h_best, 0, 1)
        c5_best = objective(h_best)
        n_points = n_points_new

    # Final result
    return (h_best, c5_best, n_points)
