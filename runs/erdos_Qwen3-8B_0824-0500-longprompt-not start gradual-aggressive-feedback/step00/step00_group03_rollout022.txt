```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import cvxpy as cp

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70  # Starting with the initial n_points
    h_values = initial_h_values.copy()
    
    # Function to compute the overlap integral C5
    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Function to compute the cost for optimization
    def objective(h_flat):
        h = h_flat.reshape(-1)
        return compute_c5(h)
    
    # Function to enforce the integral constraint
    def constraint_func(h_flat):
        h = h_flat.reshape(-1)
        return np.sum(h) - n_points / 2  # Enforces sum(h) == n_points / 2
    
    # Initial optimization: try to find a better h_values
    # We will use a local optimization approach with bounds and constraints
    
    # Bounds for h: [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    # Initial guess
    initial_guess = h_values.flatten()
    
    # Set up the constraint
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Optimization
    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=cons)
    
    # Extract the best value found
    best_h = result.x.reshape(n_points)
    best_c5 = compute_c5(best_h)
    
    # Check if we have time to try a higher resolution
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 2:
        # Try increasing n_points to 100
        n_points = 100
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy the integral constraint
        
        # Refine using local optimization
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        initial_guess = h.copy()
        
        # Optimization
        result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=cons)
        best_h = result.x.reshape(n_points)
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
```
