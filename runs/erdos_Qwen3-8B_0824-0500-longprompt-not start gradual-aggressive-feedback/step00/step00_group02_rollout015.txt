import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values
    n_points = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    dx = 2.0 / n_points
    
    # Objective function to minimize maximum overlap
    def objective(h):
        h = np.array(h)
        # Ensure h is within [0,1] and sum(h) == n_points / 2
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            return float('inf')  # Penalize for not meeting the constraint
        
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # The overlap integral is the correlation multiplied by dx
        # We take the maximum of these values
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    # Optimization: use differential evolution to find a good h
    result = optimize.differential_evolution(
        objective,
        bounds=[(0, 1) for _ in range(n_points)],
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-4,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        disp=False,
        x0=h_initial
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Ensure the solution satisfies the integral constraint
    sum_h = np.sum(best_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        # If not satisfied, adjust to meet the constraint
        # Normalize h to ensure sum_h == n_points / 2
        best_h = best_h * (n_points / 2) / sum_h
    
    # Evaluate the final overlap
    final_c5 = objective(best_h)
    
    # Return the best solution, its C5 value, and number of points
    return (best_h.tolist(), final_c5, n_points)
