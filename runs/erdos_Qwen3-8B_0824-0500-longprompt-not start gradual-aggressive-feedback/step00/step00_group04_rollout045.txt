```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # start with 100 points
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)[:n_points]  # take first n_points samples
    
    # Ensure the initial h satisfies the constraint sum(h) == n_points / 2
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-5:
        h += (n_points / 2 - total) / n_points

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Try increasing n_points to 200 if time allows
    if budget_s > 10:
        n_points = 200
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-5:
            h += (n_points / 2 - total) / n_points

        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    # Try more advanced optimization if time allows
    if budget_s > 30:
        # Convert to CVXPY variables
        h_var = cp.Variable(n_points)
        constraints = [0 <= h_var, h_var <= 1]
        constraints.append(cp.sum(h_var) == n_points / 2)

        # Objective: minimize the maximum overlap
        # We minimize the maximum of the convolution
        # Since it's challenging to directly minimize max over a convolution, we use a heuristic
        # We minimize the mean of the convolutions
        convolutions = []
        for k in range(1, n_points):
            conv = cp.multiply(h_var, 1 - h_var[k:])
            convolutions.append(conv)
        objective = cp.Minimize(cp.mean(convolutions))

        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=100)
            if prob.status == 'optimal':
                h = h_var.value
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
        except:
            pass

        # If not optimal, try another heuristic
        if best_c5 > 0.38:
            # Try a more aggressive random perturbation
            h = best_h + 0.1 * np.random.rand(n_points) - 0.05
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            if abs(total - n_points / 2) > 1e-5:
                h += (n_points / 2 - total) / n_points
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    # Final check
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # If time remains, try another random perturbation of the best solution
        h = best_h + 0.05 * np.random.rand(n_points) - 0.025
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if abs(total - n_points / 2) > 1e-5:
            h += (n_points / 2 - total) / n_points
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    # Final result
    return (best_h, best_c5, n_points)
```
