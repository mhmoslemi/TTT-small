evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 100  # A reasonable starting point
    h = np.zeros(n_points)
    h[:n_points // 2] = 1.0  # Start with a simple step function
    
    # Refine h to match the area constraint
    total_area = np.sum(h) * (2.0 / n_points)
    required_area = 1.0
    scale_factor = required_area / total_area
    h *= scale_factor
    
    # Function to compute the maximum overlap
    def compute_c5(h):
        if len(h) < 2:
            return float('inf')
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / len(h))
    
    # Function to refine h with CVXOPT
    def optimize_h(h_initial):
        n_points = len(h_initial)
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        obj = cp.Minimize(cp.max(cp.multiply(h, 1 - h)))
        prob = cp.Problem(obj, constraints)
        result = prob.solve()
        return h.value
    
    # Try multiple refinements
    for _ in range(3):
        h = optimize_h(h)
        c5 = compute_c5(h)
        print(f"Current C5: {c5}")
        if c5 < 0.380876:
            break
    
    # Additional refinement with gradient descent
    def objective(h):
        return compute_c5(h)
    
    h = np.copy(h)
    for _ in range(10):
        grad = np.gradient(h)
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h)
        print(f"Refined C5: {c5}")
        if c5 < 0.380876:
            break
    
    return h, compute_c5(h), len(h)
