import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.correlate(h, 1 - h, mode="full") * dx
initial_h_values = np.load('initial_h_values.npy')  # Load initial values

def run(seed=42, budget_s=60, **kwargs):
    n_points = 100  # Choose a reasonable number of samples
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Initialize with a simple piecewise constant function
    h[:n_points // 2] = 0.5
    h[n_points // 2:] = 0.5
    # Ensure the integral constraint is satisfied
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization settings
    max_iter = 100
    t0 = time.time()
    
    for _ in range(max_iter):
        if time.time() - t0 > budget_s:
            break
        
        # Objective: minimize the maximum overlap
        # We use a convex relaxation to approximate the max function
        # We consider the overlap for all shifts k in [0, 2]
        # The overlap integral is computed using cross-correlation
        
        # Evaluate current C5
        current_overlap = evaluate_erdos_solution(h)
        c5_bound = np.max(current_overlap)
        
        # Optimize h to reduce the maximum overlap
        # Use local search with simulated annealing
        h_new = h + np.random.normal(0, 0.05, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new C5
        new_overlap = evaluate_erdos_solution(h_new)
        new_c5 = np.max(new_overlap)
        
        # Accept the new solution if it's better (lower c5)
        if new_c5 < c5_bound:
            h = h_new
            c5_bound = new_c5
    
    # Final evaluation
    final_overlap = evaluate_erdos_solution(h)
    final_c5 = np.max(final_overlap)
    
    # Return the best solution found
    return (h, final_c5, n_points)
