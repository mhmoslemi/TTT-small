```python
import numpy as np
import scipy.optimize as optimize
import time
import warnings

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 95  # Starting with the initial sample count
    h_initial = initial_h_values
    dx = 2.0 / n_points
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Evaluate the initial solution
    start_time = time.time()
    c5_initial = evaluate_erdos_solution(h_initial, n_points, dx)
    elapsed_time = time.time() - start_time
    
    # Early exit if we have already exceeded the budget
    if elapsed_time >= budget_s:
        return h_initial, c5_initial, n_points
    
    # Try a finer grid if possible
    n_points = min(500, int(n_points * 2))
    dx = 2.0 / n_points
    
    # Initialize h with the initial solution scaled to the new grid
    h = np.zeros(n_points)
    h[:len(h_initial)] = h_initial * (n_points / 2) / np.sum(h_initial)
    h = h / (np.sum(h) / (n_points / 2))
    
    def objective(h_flat):
        h = h_flat.reshape((n_points,))
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Ensure the integral condition is met
        h = h / (np.sum(h) / (n_points / 2))
        # Compute the overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use L-BFGS-B for faster convergence
    result = optimize.minimize(
        objective,
        h,
        bounds=bounds,
        method="L-BFGS-B",
        options={"maxiter": 100, "ftol": 1e-6, "gtol": 1e-6}
    )
    
    # Final evaluation
    h_final = result.x
    c5_bound = evaluate_erdos_solution(h_final, n_points, dx)
    elapsed_time = time.time() - start_time
    
    # Early exit if time is up
    if elapsed_time >= budget_s:
        return h_final, c5_bound, n_points
    
    # Perform a local search with a small perturbation
    for _ in range(5):
        perturbation = np.random.normal(0, 0.01, size=n_points)
        perturbed_h = h_final + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h / (np.sum(perturbed_h) / (n_points / 2))
        c5_perturbed = evaluate_erdos_solution(perturbed_h, n_points, dx)
        if c5_perturbed < c5_bound:
            h_final = perturbed_h
            c5_bound = c5_perturbed
    
    return h_final, c5_bound, n_points
```
