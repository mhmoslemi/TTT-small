def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    shifted_h = np.zeros_like(h)
    for k in range(1, n_points):
        shifted_h = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial solution
    h_values = initial_h_values()
    n_points = len(h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h_values)
    target_sum = n_points / 2
    
    # Evaluate initial solution
    max_overlap = evaluate_erdos_solution(h_values)
    best_c5 = max_overlap
    best_h = h_values.copy()
    best_n_points = n_points
    
    # Try reducing the number of points to speed up evaluation
    for n_points in [100, 200, 300, 400, 500]:
        # Generate a random starting point
        h_values = np.random.rand(n_points)
        h_values = (h_values * (target_sum / np.sum(h_values))).astype(np.float64)
        
        # Evaluate initial solution
        start_time = time.time()
        max_overlap = evaluate_erdos_solution(h_values)
        end_time = time.time()
        elapsed = end_time - start_time
        
        # Check if we can run more iterations within the budget
        if elapsed < budget_s:
            # Run a gradient-based optimization
            def objective(h):
                h = np.clip(h, 0, 1)
                overlap = evaluate_erdos_solution(h)
                return overlap
            
            bounds = [(0, 1) for _ in range(n_points)]
            result = optimize.minimize(objective, h_values, method='L-BFGS-B', bounds=bounds)
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = (h_opt * (target_sum / np.sum(h_opt))).astype(np.float64)
            
            # Evaluate optimized solution
            max_overlap_opt = evaluate_erdos_solution(h_opt)
            if max_overlap_opt < best_c5:
                best_c5 = max_overlap_opt
                best_h = h_opt.copy()
                best_n_points = n_points
    
    # Final output
    return best_h, best_c5, best_n_points
