def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define parameters
    n_points = 100  # Start with 100 samples, can be increased if needed
    dx = 2.0 / n_points
    max_iterations = 50  # Limit the number of optimization steps to stay within time budget
    
    # Initialize h_values
    h = np.random.rand(n_points)
    h = h * (n_points / 2)  # Normalize to meet the integral constraint
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        # Evaluate current C5
        current_c5 = evaluate_erdos_solution(h)
        
        # Check if we're improving
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # If we're already at the target, break early
        if best_c5 <= 0.380876:
            break
        
        # Optimization step: use gradient descent-like approach
        # We consider the overlap integral as a function of h, and minimize it
        # We penalize deviation from the integral constraint
        
        # Define the objective function
        def objective(x):
            h_new = x.reshape(n_points)
            # Compute the overlap integral
            overlap = np.max(np.correlate(h_new, 1 - h_new, mode="full")) * dx
            # Penalize deviation from the integral constraint
            penalty = (np.sum(h_new) - n_points / 2) ** 2
            return overlap + 1e-4 * penalty
        
        # Gradient descent step
        h_new = h.copy()
        learning_rate = 0.1
        for _ in range(10):
            # Compute gradient numerically
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_temp = h_new.copy()
                h_temp[i] += 1e-5
                grad[i] = (objective(h_temp) - objective(h_new)) / 1e-5
            
            # Update h_new
            h_new -= learning_rate * grad
        
        # Ensure h_new is within [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Check if the integral constraint is satisfied
        if np.abs(np.sum(h_new) - n_points / 2) > 1e-2:
            # If not, move towards the correct value
            diff = (n_points / 2 - np.sum(h_new)) / n_points
            h_new += diff
        
        # Update h
        h = h_new
        
        # Check if we're out of time
        if time.time() - start_time > budget_s - 1:
            break
    
    return best_h, best_c5, n_points
