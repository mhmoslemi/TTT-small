```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assuming initial_h.npy contains the initial 47-sample vector

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(initial_h_values)
    max_attempts = 10  # Limit number of attempts for efficiency
    min_points = 100  # Minimum number of points to ensure sufficient resolution

    # Try a few different starting points based on initial_h_values
    for scale in [1.0, 0.8, 0.6, 0.5, 0.4, 0.3, 0.2]:
        h = initial_h_values * scale
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.round(h).astype(float)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

        # Try different point counts if needed
        n_points_candidate = np.random.randint(min_points, 1000)
        dx = 2.0 / n_points_candidate
        x = np.linspace(0, 2, n_points_candidate)
        h_candidate = np.zeros(n_points_candidate)
        h_candidate[:n_points] = h[:n_points]
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points_candidate / 2) / np.sum(h_candidate)
        h_candidate = np.round(h_candidate).astype(float)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points_candidate / 2) / np.sum(h_candidate)

        c5 = evaluate_erdos_solution(h_candidate)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate
            best_n_points = n_points_candidate

        if time.time() - start_time > budget_s - 1:
            break

    # Optional fine-tuning with convex optimization
    try:
        n_points_candidate = len(best_h)
        dx = 2.0 / n_points_candidate
        x = np.linspace(0, 2, n_points_candidate)
        h_var = cp.Variable(n_points_candidate, bounds=(0, 1))
        constraints = [cp.sum(h_var) == n_points_candidate / 2]
        objective = cp.Maximize(cp.max(cp.correlate(h_var, 1 - h_var, mode="full")) * dx)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == 'optimal':
            h_candidate = h_var.value
            c5 = evaluate_erdos_solution(h_candidate)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate
                best_n_points = n_points_candidate
    except:
        pass

    return best_h, best_c5, best_n_points
```
