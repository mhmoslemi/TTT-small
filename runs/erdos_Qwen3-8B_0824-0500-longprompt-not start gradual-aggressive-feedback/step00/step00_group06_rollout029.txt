import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    n_points = 200  # Start with 200 samples for faster computation
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Take initial values and adjust to n_points
    h = (h * n_points / np.sum(h)).astype(float)  # Rescale to satisfy integral constraint
    h = np.clip(h, 0, 1)  # Ensure values are in [0, 1]

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    # Optimization loop
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break

        # Use a simple gradient descent approach to minimize C5
        # Use finite differences to approximate the gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = np.clip(h_plus, 0, 1)
            c5_plus = evaluate_erdos_solution(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            h_minus = np.clip(h_minus, 0, 1)
            c5_minus = evaluate_erdos_solution(h_minus)

            grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)

        # Update h using gradient descent
        h -= 0.1 * grad
        h = np.clip(h, 0, 1)
        h = (h * n_points / np.sum(h)).astype(float)  # Re-normalize to satisfy integral constraint

        # Evaluate and update best solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    return best_h, best_c5, n_points
