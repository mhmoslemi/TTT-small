import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
# evaluate_erdos_solution() is used to compute the overlap integral
# initial_h_values is the starting point for the optimization

def run(seed=42, budget_s=60, **kwargs):
    # Define parameters
    n_points = 100  # Reduced from initial 95 to speed up optimization
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)  # Starting point
    
    # Resample initial_h to n_points
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    
    # Define the function to evaluate C5
    def evaluate_c5(h):
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
        return c5
    
    # Define the function to optimize
    def objective(h):
        return evaluate_c5(h)
    
    # Define the constraints
    def constraint1(h):
        return np.sum(h) - n_points / 2
    
    # Set up the optimization problem using CVXPY
    h = cp.Variable(n_points)
    obj = cp.Minimize(objective(h))
    constraints = [constraint1(h) == 0, 0 <= h, h <= 1]
    
    # Solve the optimization problem
    start_time = time.time()
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-5, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
    except:
        pass
    
    # Extract the solution
    best_h = h.value
    best_c5 = evaluate_c5(best_h)
    
    # If optimization failed, use the initial solution
    if best_h is None or np.isnan(best_c5):
        best_h = initial_h
        best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, n_points
