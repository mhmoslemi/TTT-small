import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 samples for faster optimization
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Use initial values as starting point
    h = h / np.sum(h) * (n_points / 2)  # Ensure the integral is 1

    start_time = time.time()

    # Function to evaluate the Erdős constant C₅
    def evaluate_c5(h_vals):
        h_vals = np.array(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)  # Ensure the integral is 1
        h = h_vals
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Function to optimize h_vals with constraints
    def optimize_h_vals(h_vals):
        n_points = len(h_vals)
        dx = 2.0 / n_points
        x = cp.Variable(n_points)
        # Constraint: sum(h_vals) * dx = 1
        constraints = [cp.sum(x) == n_points / 2]
        # Objective: minimize the maximum overlap
        overlap = np.correlate(x.value, 1 - x.value, mode="full")
        c5 = np.max(overlap) * dx if x.value is not None else float('inf')
        # Simple gradient descent approach for demonstration
        for _ in range(100):  # Limited iterations for speed
            # Evaluate C5
            overlap = np.correlate(x.value, 1 - x.value, mode="full")
            c5 = np.max(overlap) * dx
            # Gradient of C5 with respect to h_vals
            # Simplified approach: perturb h_vals slightly and compute gradient
            perturbation = 0.01
            c5_plus = evaluate_c5(h_vals + perturbation)
            c5_minus = evaluate_c5(h_vals - perturbation)
            gradient = (c5_plus - c5_minus) / (2 * perturbation)
            # Update h_vals in the direction of minimizing C5
            h_vals -= 0.01 * gradient
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals / np.sum(h_vals) * (n_points / 2)

        return h_vals

    # Initial optimization
    h_vals = optimize_h_vals(h)
    best_c5 = evaluate_c5(h_vals)
    best_h = h_vals.copy()

    # Try a higher resolution (if time allows)
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        dx = 2.0 / n_points
        h_vals = np.random.rand(n_points) * 2 - 1  # Random starting point
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        h_vals = optimize_h_vals(h_vals)
        current_c5 = evaluate_c5(h_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals

    # Final check
    if time.time() - start_time < budget_s - 2:
        h_vals = np.random.rand(n_points) * 2 - 1  # Random starting point
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
        h_vals = optimize_h_vals(h_vals)
        current_c5 = evaluate_c5(h_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals

    # Return the best solution found
    return best_h, best_c5, n_points
