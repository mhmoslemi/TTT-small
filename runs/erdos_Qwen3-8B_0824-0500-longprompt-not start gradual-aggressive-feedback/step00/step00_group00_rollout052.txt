        return np.max(correlation * dx)

    def objective(x):
        x = np.clip(x, 0, 1)
        return evaluate_c5(x)

    def constraint_sum(x):
        return np.sum(x) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Use bounded optimization with constraints
    result = optimize.minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-6)
    best_h = result.x
    best_c5 = evaluate_c5(best_h)
    
    # Refine using local search
    for _ in range(3):
        perturbations = np.random.uniform(-0.1, 0.1, size=n_points)
        candidate = best_h + perturbations
        candidate = np.clip(candidate, 0, 1)
        candidate_sum = np.sum(candidate) * dx
        if np.abs(candidate_sum - 1) < 1e-6:
            c5_candidate = evaluate_c5(candidate)
            if c5_candidate < best_c5:
                best_h = candidate
                best_c5 = c5_candidate
