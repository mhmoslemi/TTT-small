```python
import numpy as np
import scipy.optimize as optimize
import numpy.random as npr
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial number of points
    n_points = 100
    max_points = 1000
    step = 2.0 / n_points
    
    # Initial guess based on provided starting point
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]
    
    # Objective function for optimization
    def objective(h):
        h = np.array(h, dtype=np.float64)
        dx = 2.0 / len(h)
        h = np.concatenate([h, h[:2]])  # Extend for convolution
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(corr) * dx
        return overlap
    
    # Constraint: sum(h) must be n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Initial evaluation
    best_h = h_initial
    best_c5 = objective(best_h)
    
    # Try optimizing with different starting points
    for _ in range(5):
        # Perturb initial guess slightly
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = np.clip(h_initial + perturbation, 0.0, 1.0)
        
        # Optimize with scipy
        result = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        # Update best solution
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Try increasing number of points if time permits
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        n_points = min(n_points + 100, max_points)
        step = 2.0 / n_points
        h_initial = np.random.rand(n_points)
        h_initial = np.clip(h_initial, 0.0, 1.0)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # Final evaluation
    best_c5 = objective(best_h)
    return (best_h, best_c5, n_points)
```
