```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 80  # Start with 80 samples, can increase if needed
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)  # Start with uniform distribution

    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate((h, h))  # To handle convolution properly
        shifted = np.array([1 - h[i] for i in range(len(h))])
        conv = np.correlate(h, shifted, mode="full")
        conv = conv[n_points - 1 : 2 * n_points - 1]
        max_overlap = np.max(conv * dx)
        return max_overlap

    def objective_function(h_values):
        return evaluate_c5(h_values)

    # Initial guess
    h_initial = initial_h_values.copy()

    # Use a gradient-free optimizer for robustness
    result = optimize.minimize(
        objective_function,
        h_initial,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        options={'ftol': 1e-6, 'maxiter': 100}
    )

    best_h = result.x
    best_c5 = evaluate_c5(best_h)

    # Try a few more iterations with a different starting point
    for _ in range(3):
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(best_h + perturbation, 0.0, 1.0)
        result = optimize.minimize(
            objective_function,
            h_perturbed,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'ftol': 1e-6, 'maxiter': 50}
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

    # Optionally, try a larger n_points if time allows
    if budget_s > 10:
        n_points = 160
        dx = 2.0 / n_points
        h_initial = np.zeros(n_points)
        h_initial[:n_points//2] = 1.0
        h_initial[n_points//2:] = 0.0
        h_initial = np.array([0.5] * n_points)  # Start with uniform distribution

        def evaluate_c5_large(h):
            h = np.array(h)
            h = np.concatenate((h, h))
            shifted = np.array([1 - h[i] for i in range(len(h))])
            conv = np.correlate(h, shifted, mode="full")
            conv = conv[n_points - 1 : 2 * n_points - 1]
            max_overlap = np.max(conv * dx)
            return max_overlap

        def objective_function_large(h_values):
            return evaluate_c5_large(h_values)

        result = optimize.minimize(
            objective_function_large,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'ftol': 1e-6, 'maxiter': 50}
        )

        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

    return (best_h, best_c5, n_points)
```
