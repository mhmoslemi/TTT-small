import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import sys

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 1000
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
        initial_h_values = np.roll(initial_h_values, int(np.random.rand() * n_points))
        initial_h_values = np.clip(initial_h_values, 0, 1)
        initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    def objective_function(h_values):
        h = h_values.reshape(-1)
        h = np.clip(h, 0, 1)
        one_minus_h = 1 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = optimize.minimize(objective_function, initial_h_values, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 0.5:
        return (result.x, result.fun, n_points)
    
    # Refine with another round of optimization
    result = optimize.minimize(objective_function, result.x, method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-6)
    return (result.x, result.fun, n_points)

start_time = time.time()
run()
