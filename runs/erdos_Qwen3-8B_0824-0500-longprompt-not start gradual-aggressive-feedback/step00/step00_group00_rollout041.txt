def run(seed=42, budget_s=60, **kwargs):

    n_points = len(initial_h_values)

    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:1]])  # Extend for full convolution
        result = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(result)

    def compute_cost(h):
        return compute_c5(h)

        if c5 < best_c5:

            best_c5 = c5

    def objective_function(h):
        return compute_c5(h)

        h_values -= learning_rate * grad
