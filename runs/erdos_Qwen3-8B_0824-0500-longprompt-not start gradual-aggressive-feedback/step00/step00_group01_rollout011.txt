```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Start with the initial guess
    n_points_initial = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    
    # Evaluate the initial solution
    c5_initial = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = c5_initial
    best_n_points = n_points_initial
    
    # Try smaller n_points
    n_points_candidates = [n_points_initial // 2, n_points_initial // 4, n_points_initial // 8]
    for n_points in n_points_candidates:
        if n_points < 4:
            continue
        
        # Generate random samples over [0, 2] with uniform spacing
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initialize h with random values in [0, 1]
        h = np.random.uniform(0, 1, n_points)
        # Normalize to ensure ∫h(x)dx = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate the solution
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n_points = n_points
    
    # Try to optimize the best solution with fewer points
    n_points = best_n_points
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)
    h = best_h.copy()
    
    # Define the objective function to minimize C₅
    def objective(params):
        h = params.reshape(-1)
        c5 = evaluate_erdos_solution(h)
        return c5
    
    # Define constraints
    def constraint_sum(params):
        return np.sum(params) - (n_points / 2)
    
    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'ftol': 1e-8, 'maxiter': 200}
    )
    
    h_opt = result.x
    c5_opt = result.fun
    
    if c5_opt < best_c5:
        best_h = h_opt
        best_c5 = c5_opt
    
    # Final check to ensure the solution is within budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, best_n_points)
    
    return (best_h, best_c5, best_n_points)
```
