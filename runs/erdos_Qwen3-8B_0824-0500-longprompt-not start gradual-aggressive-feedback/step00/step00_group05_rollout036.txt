```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 samples
    dx = 2.0 / n_points
    
    # Define initial h_values
    h_values = initial_h_values.copy()[:n_points]
    h_values = h_values / np.sum(h_values) * (n_points / 2)  # Normalize to ensure sum(h) == n_points / 2
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        k_values = np.arange(-n + 1, n)
        overlaps = []
        for k in k_values:
            if k < 0:
                h_shift = np.roll(h, -k)
            else:
                h_shift = np.roll(h, k)
            overlap = np.sum(h * (1 - h_shift)) * dx
            overlaps.append(overlap)
        return np.max(overlaps)
    
    # Optimization function
    def objective_function(h):
        return compute_c5(h)
    
    # Optimization settings
    max_iter = 100
    tol = 1e-4
    best_h = h_values
    best_c5 = compute_c5(best_h)
    best_n_points = n_points
    time_start = time.time()
    
    # Try with different n_points
    for n_points in [100, 200, 300, 400, 500]:
        dx = 2.0 / n_points
        h_values = np.random.rand(n_points)
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        
        # Optimization
        h = cp.Variable(n_points)
        objective = cp.Minimize(objective_function(h))
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        prob = cp.Problem(objective, constraints)
        
        # Solve using ECOS
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-4, max_iter=100)
            if result is not None and result <= best_c5:
                best_h = h.value
                best_c5 = result
                best_n_points = n_points
        except:
            pass
        
        # Local optimization with gradient descent
        h_values = np.random.rand(n_points)
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        for _ in range(max_iter):
            h_values = np.clip(h_values, 0, 1)
            c5 = compute_c5(h_values)
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_values[i] += 1e-4
                c5_plus = compute_c5(h_values)
                h_values[i] -= 1e-4
                c5_minus = compute_c5(h_values)
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-4)
            h_values -= 0.1 * grad
            h_values = np.clip(h_values, 0, 1)
            h_values = h_values / np.sum(h_values) * (n_points / 2)
            c5 = compute_c5(h_values)
            if c5 < best_c5:
                best_h = h_values.copy()
                best_c5 = c5
                best_n_points = n_points
        
        # Check time budget
        elapsed = time.time() - time_start
        if elapsed >= budget_s:
            break
    
    # Evaluate best solution
    result = evaluate_erdos_solution(best_h)
    
    return best_h, result, best_n_points
```
