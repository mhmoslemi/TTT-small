```python
import numpy as np
import scipy.optimize as optimize
import time
import sys

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Example initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with a smaller number of points for faster evaluation

    # Try a few different numbers of points to find a good balance between speed and quality
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 2:
            break

        # Generate initial guess based on the provided initial solution
        h_initial = np.zeros(n_points)
        h_initial[:len(initial_h_values)] = initial_h_values[:n_points]
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)

        # Optimization: minimize the maximum overlap integral
        def objective(x):
            h = x.reshape(-1)
            return evaluate_erdos_solution(h)

        # Constraints: sum(h) = n_points / 2
        def constraint(x):
            return np.sum(x) - n_points / 2

        # Bounds: 0 <= h[i] <= 1
        bounds = [(0, 1) for _ in range(n_points)]

        # Use SLSQP to find the minimum
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-4, 'xatol': 1e-4}
        )

        if result.success:
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

        # Early stopping if time is up
        if time.time() - start_time > budget_s - 2:
            break

    # If no solution found, fallback to the initial solution
    if best_h is None:
        best_h = initial_h_values[:n_points]

    return best_h, best_c5, n_points
```
