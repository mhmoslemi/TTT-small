import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 100  # Start with 100 samples, can increase if needed
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])  # Start with initial values
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h = np.pad(h, (0, n_points), 'constant')  # Pad for full correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Function to update h using gradient descent-like approach
    def update_h(h, learning_rate=0.01):
        h = np.array(h)
        # Compute the overlap integral for all shifts
        corr = signal.correlate(h, 1 - h, mode='full')
        c5_values = corr * dx
        max_c5_index = np.argmax(c5_values)
        
        # Determine the shift that contributes the most to the max C5
        shift = max_c5_index - len(h) + 1
        if shift < 0:
            shift = 0
        elif shift > len(h) - 1:
            shift = len(h) - 1
        
        # Compute derivative of h with respect to the overlap integral
        # This is an approximation using finite differences
        h_plus = np.copy(h)
        h_plus[shift] += 0.01
        c5_plus = compute_c5(h_plus)
        
        h_minus = np.copy(h)
        h_minus[shift] -= 0.01
        c5_minus = compute_c5(h_minus)
        
        # Update h to reduce the overlap integral
        derivative = (c5_plus - c5_minus) / 0.02
        h[shift] -= learning_rate * derivative
        h = np.clip(h, 0, 1)
        
        # Normalize to maintain the integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        return h.tolist()
    
    # Optimization loop
    best_h = h
    best_c5 = compute_c5(h)
    for _ in range(1000):  # Max 1000 iterations
        if time.time() - start_time > budget_s:
            break
        h = update_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    # Try a few random perturbations to escape local optima
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(np.array(best_h) + perturbation, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    n_points = len(best_h)
    return best_h, best_c5, n_points
