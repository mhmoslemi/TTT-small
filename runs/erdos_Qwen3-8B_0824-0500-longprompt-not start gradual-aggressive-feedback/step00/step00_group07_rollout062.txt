```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Start with initial solution
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    
    # Define the overlap integral function
    def compute_c5(h_values):
        h = h_values
        n = len(h)
        dx = 2.0 / n
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Compute the max overlap integral
        c5 = np.max(corr) * dx
        return c5
    
    # Define the optimization problem
    def optimize_h(h_initial=None):
        if h_initial is None:
            h_initial = np.random.rand(n_points)
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
            h_initial = np.clip(h_initial, 0, 1)
        else:
            h_initial = np.array(h_initial)
        
        # Objective: minimize the maximum overlap integral
        # Constraints: sum(h) = n_points / 2 and 0 <= h <= 1
        
        # Convert to CVXPY variables
        h_cvx = cp.Variable(n_points, bounds=(0, 1))
        objective = cp.Minimize(cp.Max(cp.correlate(h_cvx, 1 - h_cvx, mode="full") * (2.0 / n_points)))
        constraints = [cp.sum(h_cvx) == n_points / 2]
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.SCS)
        
        if result is None:
            return h_initial
        else:
            return h_cvx.value
    
    # Initial optimization
    best_h = optimize_h(h)
    best_c5 = compute_c5(best_h)
    
    # Iterative refinement with local search
    for _ in range(10):
        # Perturb the solution slightly
        perturbation = np.random.randn(n_points) * 0.05
        h_perturbed = np.clip(best_h + perturbation, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Optimize the perturbed solution
        h_optimized = optimize_h(h_perturbed)
        current_c5 = compute_c5(h_optimized)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h_optimized
            best_c5 = current_c5
    
    # Final optimization with more aggressive perturbations
    for _ in range(5):
        perturbation = np.random.randn(n_points) * 0.1
        h_perturbed = np.clip(best_h + perturbation, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Optimize the perturbed solution
        h_optimized = optimize_h(h_perturbed)
        current_c5 = compute_c5(h_optimized)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h_optimized
            best_c5 = current_c5
    
    # Ensure the solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_h = np.clip(best_h, 0, 1)
    
    # Check elapsed time and return
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Time budget exceeded. Returning best solution so far.")
    
    return best_h.tolist(), best_c5, n_points
```
