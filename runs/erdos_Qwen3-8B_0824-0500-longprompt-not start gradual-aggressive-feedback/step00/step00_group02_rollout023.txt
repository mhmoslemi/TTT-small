```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Try different n_points to find the best balance between resolution and speed
    n_points_list = [50, 75, 100, 125, 150, 175, 200]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Initial guess - start from the current construction, scaled to n_points
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
        
        # Define variables for optimization
        h = cp.Variable(n_points, bounds=(0, 1))
        constraints = [cp.sum(h) == n_points / 2]
        
        # Function to minimize max overlap
        def objective_func(h_values):
            h_array = np.array(h_values)
            corr = np.correlate(h_array, 1 - h_array, mode='full')
            return np.max(corr) * dx
        
        # We minimize the maximum overlap using a convex optimization approach
        # Note: This is a simplified approach and might not capture the full problem structure
        # But it gives a direction to find a better upper bound
        
        # Use a numerical optimization method
        def evaluate_objective(x):
            return objective_func(x)
        
        # Use L-BFGS-B to optimize
        result = opt.minimize(evaluate_objective, initial_h, method='L-BFGS-B', tol=1e-6)
        
        h_values = result.x
        c5_bound = objective_func(h_values)
        
        # Check if this is the best solution
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_values
            best_n_points = n_points
        
        print(f"n_points={n_points}, c5_bound={c5_bound}")
    
    # Return the best solution found
    return best_h, best_c5, best_n_points
```
