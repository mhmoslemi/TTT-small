        objective = cp.Minimize(cp.Max([cp.sum(cp.multiply(h, 1 - h[i:i+n])) * dx for i in range(n-1)]))
        problem = cp.Problem(objective, constraints)
        start_time = time.time()
        try:
            result = problem.solve(solver=cp.SCS, verbose=False, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
