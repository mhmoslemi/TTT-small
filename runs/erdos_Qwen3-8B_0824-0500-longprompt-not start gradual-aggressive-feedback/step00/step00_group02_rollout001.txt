import numpy as np
import scipy.optimize as optimize
import warnings
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 45  # Starting with the initial construction
    dx = 2.0 / n_points
    
    # Function to compute overlap integral
    def compute_c5(h):
        h = np.array(h)
        h_extended = np.zeros((len(h), len(h)))
        h_extended = np.vstack((h, np.zeros_like(h)))
        cross_corr = np.correlate(h, 1 - h, mode="full")
        cross_corr = cross_corr * dx
        return np.max(cross_corr)
    
    # Function to evaluate the solution
    def evaluate_solution(h):
        nonlocal n_points, dx
        h = np.array(h)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        return compute_c5(h)
    
    # Initial solution
    initial_h = initial_h_values
    initial_c5 = evaluate_solution(initial_h)
    
    # Function to optimize the solution using simulated annealing
    def optimize_solution(initial_h, budget_s):
        nonlocal n_points, dx
        best_h = initial_h.copy()
        best_c5 = initial_c5
        time_start = time.time()
        
        # Simulated Annealing parameters
        temp = 1.0
        cooling_rate = 0.995
        iterations = 1000
        max_time = budget_s
        
        for _ in range(iterations):
            elapsed = time.time() - time_start
            if elapsed > max_time:
                break
            temp *= cooling_rate
            
            # Perturb the current solution
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturbed = np.clip(best_h + perturbation, 0, 1)
            
            # Compute the new c5 value
            current_c5 = evaluate_solution(h_perturbed)
            
            # Acceptance probability
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5) / temp):
                best_h = h_perturbed
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Run optimization
    best_h, best_c5 = optimize_solution(initial_h, budget_s)
    
    # Return the best solution found
    return best_h, best_c5, n_points
