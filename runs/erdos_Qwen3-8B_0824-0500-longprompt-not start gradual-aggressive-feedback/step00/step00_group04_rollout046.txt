import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Helper function to compute the overlap integral
def compute_overlap_integral(h, dx):
    h_shifted = np.roll(h, 1)
    overlap = np.correlate(h, 1 - h_shifted, mode="full")
    return np.max(overlap) * dx

# Function to evaluate the solution
def evaluate_solution(h_values, dx):
    return compute_overlap_integral(h_values, dx)

# Function to run the optimization
def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial number of points
    n_points = 100
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=float)
    
    # Normalize h to ensure the integral is 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Define optimization variables
    x = np.random.rand(n_points)
    x = x * (n_points / 2) / np.sum(x)
    
    # Define the objective function to minimize
    def objective(x):
        h = x.copy()
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_solution(h, dx)
    
    # Define constraints: sum(h) == n_points / 2
    def constraint_func(x):
        return np.sum(x) - n_points / 2
    
    constraints = ({'type': 'eq', 'fun': constraint_func})
    
    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run the optimization
    result = optimize.minimize(
        objective,
        x0=x,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    # Extract the best solution
    best_h = result.x
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Compute the best C5 value
    best_c5 = evaluate_solution(best_h, dx)
    
    # Check if we have time to refine the solution with more points
    if time.time() - start_time < budget_s - 10:
        # Try with more points
        n_points = 200
        dx = 2.0 / n_points
        best_h = np.random.rand(n_points)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        def objective_refined(x):
            h = x.copy()
            h = h * (n_points / 2) / np.sum(h)
            return evaluate_solution(h, dx)
        
        constraints_refined = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds_refined = [(0, 1) for _ in range(n_points)]
        
        result_refined = optimize.minimize(
            objective_refined,
            x0=best_h,
            method='SLSQP',
            bounds=bounds_refined,
            constraints=constraints_refined,
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        
        best_h = result_refined.x
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_solution(best_h, dx)
    
    # Ensure the best_h has values in [0, 1]
    best_h = np.clip(best_h, 0, 1)
    
    # Check if we have time to refine once more
    if time.time() - start_time < budget_s - 20:
        n_points = 300
        dx = 2.0 / n_points
        best_h = np.random.rand(n_points)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        def objective_final(x):
            h = x.copy()
            h = h * (n_points / 2) / np.sum(h)
            return evaluate_solution(h, dx)
        
        constraints_final = ({'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2})
        bounds_final = [(0, 1) for _ in range(n_points)]
        
        result_final = optimize.minimize(
            objective_final,
            x0=best_h,
            method='SLSQP',
            bounds=bounds_final,
            constraints=constraints_final,
            options={'ftol': 1e-8, 'maxiter': 300}
        )
        
        best_h = result_final.x
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_solution(best_h, dx)
    
    best_h = np.clip(best_h, 0, 1)
    return best_h.tolist(), best_c5, n_points
