import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial solution from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    # Check initial C5 value
    def compute_c5(h):
        h = np.array(h)
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Take the max value
        max_corr = np.max(corr)
        # Multiply by dx to get the integral
        return max_corr * dx

    # Start with initial solution
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    best_n_points = n_points

    # Try with smaller n_points
    for n in range(100, 500, 50):
        # Generate a random initial guess with n samples
        h = np.random.uniform(0, 1, n)
        # Normalize to satisfy the integral constraint
        h = h * (n / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        # Compute initial C5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n_points = n
        # Early stopping if we find a better solution
        if best_c5 < 0.380876:
            break
    
    # Refine the best solution with more points
    for n in range(best_n_points, best_n_points + 100, 10):
        # Generate a refined initial guess with n samples
        h = np.random.uniform(0, 1, n)
        # Normalize to satisfy the integral constraint
        h = h * (n / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        # Compute initial C5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n_points = n
    
    # Now perform optimization using CVXPY
    time_start = time.time()
    while time.time() - time_start < budget_s:
        # Convert to a more structured optimization problem
        n = len(best_h)
        h_var = cp.Variable(n)
        constraints = [cp.sum(h_var) == n / 2]
        obj = cp.Minimize(cp.max(cp.correlate(h_var.value, 1 - h_var.value, mode="full")) * dx)
        prob = cp.Problem(obj, constraints)
        
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-6)
            if result is not None and result < best_c5:
                best_h = h_var.value
                best_c5 = result
        except:
            pass
        
        # Early stopping if we find a better solution
        if best_c5 < 0.380876:
            break

    return best_h, best_c5, best_n_points
