import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.roll(h, shift=1, axis=0)
    overlap = np.correlate(h, 1 - shifted_h, mode="full")
    return np.max(overlap) * dx

def initial_h_values():
    # Placeholder for initial construction (n=70)
    return np.random.rand(70) * 2 - 1  # Just a dummy initial guess

# Strategy: Use a combination of random search with local optimization to find a good h
# Start from initial guess, then perturb and refine using numerical methods
# Track the best solution within time budget

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    n_points = 100  # Start with 100 points for efficiency

    # Initial guess from provided function
    h = initial_h_values()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Ensure integral equals 1

    # Evaluate initial solution
    c5 = evaluate_erdos_solution(h)
    best_c5 = c5
    best_h = h.copy()

    # Try multiple perturbations
    for _ in range(50):  # 50 iterations within budget
        if time.time() - start_time > budget_s - 1:
            break

        # Perturb the solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h + perturbation, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)

        # Local optimization using gradient descent
        learning_rate = 0.01
        for _ in range(100):  # 100 steps of gradient descent
            if time.time() - start_time > budget_s - 1:
                break
            # Evaluate current solution
            current_c5 = evaluate_erdos_solution(h_perturbed)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturbed.copy()

            # Gradient descent step
            # Compute loss function
            def loss(h_vals):
                h = np.clip(h_vals, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                return evaluate_erdos_solution(h)

            # Numerical gradient
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h_perturbed.copy()
                h_plus[i] += 1e-5
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                loss_plus = loss(h_plus)
                h_minus = h_perturbed.copy()
                h_minus[i] -= 1e-5
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                loss_minus = loss(h_minus)
                grad[i] = (loss_plus - loss_minus) / (2 * 1e-5)

            # Update h_perturbed
            h_perturbed -= learning_rate * grad

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
