import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        h = h / sum_h * (n_points / 2)
    
    # Function to compute C5
    def compute_c5(h):
        dx = 2.0 / len(h)
        h_pad = np.zeros(len(h) * 2 - 1)
        h_pad[:len(h)] = h
        corr = np.correlate(h_pad, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Initial C5 value
    c5_bound = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_bound
    best_n_points = n_points
    
    # Try reducing the number of points if possible
    for n_points in range(n_points, 10, -1):
        if n_points < 10:
            break
        # Downsample the current solution
        h_downsampled = np.zeros(n_points)
        for i in range(n_points):
            h_downsampled[i] = np.mean(h[int(i * len(h) / n_points):int((i + 1) * len(h) / n_points)] + 1e-10)
        h_downsampled = h_downsampled / np.sum(h_downsampled) * (n_points / 2)
        c5 = compute_c5(h_downsampled)
        if c5 < best_c5:
            best_h = h_downsampled.copy()
            best_c5 = c5
            best_n_points = n_points
        if time.time() - start_time > budget_s - 1:
            break
    
    # Try optimizing with more points if time allows
    if time.time() - start_time < budget_s - 10:
        # Use more points for optimization
        n_points_opt = min(200, best_n_points * 2)
        h_opt = np.random.uniform(0, 1, n_points_opt)
        h_opt = h_opt / np.sum(h_opt) * (n_points_opt / 2)
        
        # Define the optimization problem
        h_var = cp.Variable(n_points_opt)
        constraints = [
            cp.sum(h_var) == n_points_opt / 2,
            0 <= h_var,
            h_var <= 1
        ]
        
        # Objective: minimize the maximum overlap
        # Note: This is a difficult problem to optimize directly, so we use a bilevel approach
        # Here, we approximate by minimizing one of the overlaps
        
        def compute_overlap(h):
            dx = 2.0 / len(h)
            h_pad = np.zeros(len(h) * 2 - 1)
            h_pad[:len(h)] = h
            corr = np.correlate(h_pad, 1 - h, mode="full")
            return corr
        
        def objective(h):
            corr = compute_overlap(h)
            return np.max(corr) * dx
        
        # Use a numerical optimizer to minimize the overlap
        result = optimize.minimize(
            objective,
            h_opt,
            method="L-BFGS-B",
            bounds=[(0, 1) for _ in range(n_points_opt)],
            options={"ftol": 1e-5, "gtol": 1e-5, "maxiter": 100}
        )
        h_opt = result.x
        h_opt = h_opt / np.sum(h_opt) * (n_points_opt / 2)
        c5_opt = compute_c5(h_opt)
        
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            best_n_points = n_points_opt
    
    return best_h, best_c5, best_n_points
