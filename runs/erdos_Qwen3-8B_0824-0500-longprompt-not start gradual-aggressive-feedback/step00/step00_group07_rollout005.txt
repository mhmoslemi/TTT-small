```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    n_points = 200  # Use 200 points for better precision within time budget
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h = copy.deepcopy(initial_h_values)
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    
    # Function to compute C5 given h
    def compute_c5(h_values):
        h = np.array(h_values)
        h = np.concatenate([h, h[:n_points - 1]])  # For full convolution
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    # Objective: minimize C5
    def objective(h_values):
        return compute_c5(h_values)
    
    # Constraints:
    # 1. h[i] ∈ [0, 1]
    # 2. ∑h[i] * dx = 1 → ∑h[i] = n_points / 2
    
    # Using constrained optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
    ]
    
    # Initial solution
    initial_guess = h
    
    # Run optimization
    result = optimize.minimize(
        objective, 
        initial_guess, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    # Check if we are under time limit
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 5:
        return (result['x'], compute_c5(result['x']), n_points)
    
    # If optimization not successful, try a local search
    if result.success:
        h_values = result['x']
    else:
        # Fallback to initial guess
        h_values = initial_guess
    
    # Evaluate the final C5
    c5_bound = compute_c5(h_values)
    
    # Check if we can refine further with a local optimization
    result_refine = optimize.minimize(
        objective, 
        h_values, 
        method='L-BFGS-B', 
        bounds=bounds, 
        constraints=constraints,
        tol=1e-8,
        options={'maxiter': 50, 'disp': False}
    )
    
    if result_refine.success:
        h_values = result_refine['x']
    else:
        h_values = h_values
    
    c5_bound = compute_c5(h_values)
    
    # Final check
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 5:
        return (h_values, c5_bound, n_points)
    
    return (h_values, c5_bound, n_points)
```
