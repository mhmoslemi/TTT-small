import numpy as np
import scipy.optimize as optimize
import scipy.interpolate as interpolate
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points (faster to optimize)

    # Function to evaluate the overlap integral
    def evaluate_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        # Use convolution to compute max overlap
        corr = np.correlate(h, 1 - h, mode="full")
        # Only consider the relevant part of the convolution
        corr = corr[len(h)-1:2*len(h)-1]
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to optimize h with constraints
    def optimize_h(h_initial):
        n = len(h_initial)
        dx = 2.0 / n
        # Set up optimization
        x = np.arange(n) * dx
        h = np.array(h_initial, dtype=float)

        # Objective function to minimize
        def objective(h_vec):
            h = h_vec
            corr = np.correlate(h, 1 - h, mode="full")
            corr = corr[len(h)-1:2*len(h)-1]
            return np.max(corr) * dx

        # Constraints: sum(h) == n / 2
        def constraint(h_vec):
            return np.sum(h_vec) - n / 2

        # Bounds: 0 <= h[i] <= 1
        bounds = [(0, 1) for _ in range(n)]

        # Optimization using SLSQP
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        return result.x

    # Initial guess from provided initial_h_values
    initial_h = initial_h_values
    if len(initial_h) != n_points:
        # Resample to n_points
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    result = optimize_h(initial_h)
    best_h = result
    best_c5 = evaluate_c5(result)
    best_n_points = n_points

    # Try increasing the number of points (up to 500) to improve the bound
    for n_points in [200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 5:
            break
        # Resample from best_h (interpolate to new n_points)
        x_new = np.linspace(0, 2, n_points)
        x_old = np.linspace(0, 2, len(best_h))
        h_new = np.interp(x_new, x_old, best_h)
        h_new = np.clip(h_new, 0, 1)
        # Optimize the new h
        result = optimize_h(h_new)
        current_c5 = evaluate_c5(result)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result
            best_n_points = n_points

    # Final evaluation
    return (best_h, best_c5, best_n_points)
