```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Load initial solution

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    n_points = len(best_h)
    dx = 2.0 / n_points
    sum_h = np.sum(best_h)
    
    # Convert to optimization variables
    h = cp.Variable(n_points)
    
    # Objective: minimize the maximum overlap
    # We will use a convex optimization approach to minimize the maximum overlap
    # We will fix the maximum overlap and try to find a feasible solution
    
    # For each shift k, we compute the overlap integral
    # But due to computational cost, we use a smaller n_points
    # Start with the same n_points as the initial solution
    
    # We will run a few rounds of convex optimization to refine the solution
    for _ in range(5):
        # Set up the constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # Define objective: minimize the maximum overlap
        # We'll use a convex relaxation by introducing a variable c5
        c5 = cp.Variable()
        constraints += [c5 >= np.correlate(h.value, 1 - h.value, mode="full") * dx]
        
        # Objective is to minimize c5
        objective = cp.Minimize(c5)
        problem = cp.Problem(objective, constraints)
        
        # Solve the problem
        try:
            result = problem.solve(solver=cp.ECOS)
            if result is not None and c5.value < best_c5:
                best_c5 = c5.value
                best_h = h.value
                print(f"Improved c5: {best_c5}")
        except:
            pass
    
    # Evaluate the best solution
    c5_bound = evaluate_erdos_solution(best_h)
    
    # Ensure we don't exceed the time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        print("Time budget exceeded, returning best solution so far")
    
    return best_h, c5_bound, n_points
```
