```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 70  # Starting with the initial construction
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    
    # Objective: minimize the maximum overlap integral
    # We use a greedy approach to refine the initial construction
    # Try increasing number of points if time permits
    
    # Initial evaluation
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Trial with more points if time allows
    for n_points in [70, 100, 150, 200]:
        start_time = time.time()
        dx = 2.0 / n_points
        h = np.random.rand(n_points) * 2 - 1  # Random initial guess (scaled to [0, 2])
        h = np.clip(h, 0, 2)  # Ensure in [0, 2]
        h /= 2  # Scale to [0, 1]
        h = h * (n_points / 2)  # Ensure integral is 1
        h = np.clip(h, 0, n_points / 2)  # Clamp to valid range
        
        # Evaluate initial guess
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Optimization loop
        for _ in range(50):  # Number of iterations
            if time.time() - start_time > budget_s:
                break
            
            # Construct the cost function for optimization
            # We use a convex optimization approach to minimize the maximum overlap
            x = cp.Variable(n_points)
            constraints = [
                cp.sum(x) == n_points / 2,  # Integral is 1
                0 <= x,
                x <= n_points / 2
            ]
            
            # Compute the overlap for all k
            overlap = np.zeros(n_points)
            for k in range(1, n_points):
                shifted = np.roll(x, k)
                overlap[k] = cp.max(cp.multiply(x, 1 - shifted))
            
            # Objective: minimize the maximum overlap
            obj = cp.Minimize(cp.max(overlap))
            prob = cp.Problem(obj, constraints)
            prob.solve(solver='ECOS', eps=1e-6, verbose=False)
            
            if prob.status != 'OPTIMAL':
                break
            
            # Update the best solution
            h = x.value
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        
        # Check if we have improved the best solution
        if best_c5 < evaluate_erdos_solution(best_h):
            best_h = h.copy()
            best_c5 = c5
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, len(best_h)
```
