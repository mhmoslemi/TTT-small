import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different numbers of points
    n_points_list = [100, 200, 300, 400]
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Define the grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initial guess
        h_initial = initial_h_values.copy()
        h_initial = np.interp(np.linspace(0, 2, n_points), x, h_initial)
        
        # Bounds for h
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the objective function
        def objective(h_values):
            h = h_values.reshape(-1)
            h = np.clip(h, 0, 1)
            overlap = np.correlate(h, 1 - h, mode="full") * dx
            c5 = np.max(overlap)
            return c5
        
        # Define the constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
        def constraint(h_values):
            return np.sum(h_values) - n_points / 2
        
        # Optimization setup
        h = cp.Variable(n_points)
        obj = cp.Minimize(objective(h.value))
        cons = [cp.sum(h) == n_points / 2]
        prob = cp.Problem(obj, cons)
        
        try:
            result = prob.solve(solver=cp.SCS, eps=1e-4, verbose=False)
            if result is not None:
                h_values = h.value
                c5 = objective(h_values)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
                    best_n = n_points
        except:
            pass
        
        # Try a local optimization from initial guess
        result = optimize.minimize(objective, h_initial, bounds=bounds, method='L-BFGS-B', tol=1e-4, options={'maxiter': 100})
        if result.success:
            h_values = result.x
            c5 = objective(h_values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values
                best_n = n_points
    
    # Evaluate the best solution
    best_h = np.clip(best_h, 0, 1)
    c5_bound = evaluate_erdos_solution(best_h, n_points=best_n)
    
    return best_h, c5_bound, best_n
