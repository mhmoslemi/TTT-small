def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros(2 * n_points - 1)
    shifted[:n_points] = h
    shifted[n_points:] = h[:-1]
    one_minus_h = 1 - h
    correlation = np.correlate(h, one_minus_h, mode="full")
    c5 = np.max(correlation) * dx
    return c5
