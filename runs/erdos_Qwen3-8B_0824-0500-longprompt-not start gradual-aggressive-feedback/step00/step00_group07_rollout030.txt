import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points and initial guess
    n_points = 200  # Starting with 200 points, can adjust if needed
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()
    
    # Ensure initial_h is of correct size and normalized
    initial_h = initial_h[:n_points]
    initial_h = initial_h / (np.sum(initial_h) * dx) * (n_points / 2)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h = np.pad(h, (0, 1), mode='constant')  # Pad to handle full convolution
        shifted = 1 - np.roll(h, 1)
        conv = np.correlate(h, shifted, mode='full')
        c5 = np.max(conv) * dx
        return c5
    
    # Function to optimize h with constraints
    def optimize_h(h_prev):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        objective = cp.Minimize(compute_c5(h))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False, time_limit=budget_s)
            if prob.status == 'OPTIMAL':
                return h.value
        except:
            pass
        return h_prev
    
    # Optimization loop with time tracking
    start_time = time.time()
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    for _ in range(5):  # Perform a few rounds of optimization
        current_h = optimize_h(best_h)
        if current_h is None:
            break
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
