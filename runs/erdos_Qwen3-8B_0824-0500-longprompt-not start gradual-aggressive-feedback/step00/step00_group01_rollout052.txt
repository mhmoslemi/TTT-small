import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
# evaluate_erdos_solution() and initial_h_values are available

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Start from the initial construction
    n_points = len(initial_h_values)
    h = initial_h_values.copy()

    # Initial evaluation
    def compute_c5(h):
        h = np.array(h)
        dh = 2.0 / len(h)
        conv = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv * dh)
        return c5

    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5

    # Function to optimize
    def optimize_h(h):
        n = len(h)
        h = np.array(h)
        # Objective: minimize the maximum overlap
        # We convert the problem to minimize the maximum value of the convolution
        # This is a non-convex optimization problem, so we use a convex relaxation
        # We minimize the maximum value of the convolution using a convex optimization approach

        # Define variables
        x = cp.Variable(n)
        y = cp.Variable(n)
        obj = cp.Minimize(cp.Max(cp.sum(x * (1 - y)) / (2.0 / n)))

        # Constraints
        constraints = [
            cp.sum(x) == n / 2,
            0 <= x,
            x <= 1,
            0 <= y,
            y <= 1
        ]

        # Solve
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.ECOS, verbose=False)
        if result is not None and result >= 0:
            return x.value
        else:
            return h

    # Optimization loop
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        h = optimize_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Final evaluation
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
