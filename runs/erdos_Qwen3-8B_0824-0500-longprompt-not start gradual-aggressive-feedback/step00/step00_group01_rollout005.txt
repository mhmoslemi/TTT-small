import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Use the initial H values as a starting point
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        shifted = np.concatenate([np.zeros_like(h), h[:-1]])
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to evaluate the objective and constraints
    def objective(h):
        return compute_c5(h)
    
    # Optimization: Use a gradient-free optimization
    # Try different starting points to avoid local optima
    for _ in range(3):  # Try 3 different starting points
        # Random perturbation
        h_values = np.array(initial_h_values)
        h_values += np.random.normal(0, 0.05, size=h_values.shape)
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        # Optimization settings
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use differential evolution to find a better solution
        result = optimize.differential_evolution(objective, bounds, 
                                                 strategy='best1bin', 
                                                 maxiter=100, 
                                                 popsize=15, 
                                                 tol=1e-5, 
                                                 recombination=0.9, 
                                                 mutation=0.2,
                                                 seed=seed + _)
        
        # Check if we're under time
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s - 1:
            break
        
        # Update best solution
        best_h = result.x
        best_c5 = result.fun
        h_values = best_h
        print(f"Best C5 found: {best_c5:.6f}")
    
    # Evaluate the final solution
    final_c5 = compute_c5(h_values)
    print(f"Final C5: {final_c5:.6f}")
    
    return h_values, final_c5, n_points
