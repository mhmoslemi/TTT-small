```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    h = initial_h_values[:n_points].copy()  # Start from initial construction
    h = h / np.sum(h) * (n_points / 2)  # Normalize to satisfy integral constraint
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points, dx)

    # Try increasing n_points if time allows
    elapsed = 0
    for n_points in [100, 150, 200, 250, 300, 350, 400, 450, 500]:
        if elapsed >= budget_s:
            break
        
        start_time = time.time()
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        # Use a simple heuristic: divide [0, 2] into equal segments and assign values
        # to minimize overlap
        for i in range(n_points):
            h[i] = 1 if (i * dx) % 1 < 0.5 else 0
        
        # Enforce integral constraint
        total = np.sum(h) * dx
        scale = (n_points / 2) / total
        h = h * scale

        # Evaluate
        c5 = evaluate_erdos_solution(h, n_points, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        elapsed = time.time() - start_time

    # Try local optimization around the best solution
    for _ in range(5):
        if elapsed >= budget_s:
            break
        
        start_time = time.time()
        h = best_h.copy()
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        scale = (n_points / 2) / total
        h = h * scale

        # Evaluate
        c5 = evaluate_erdos_solution(h, n_points, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        elapsed = time.time() - start_time

    # Try convex optimization approach with fewer points
    for n_points in [100, 150, 200]:
        if elapsed >= budget_s:
            break
        
        start_time = time.time()
        dx = 2.0 / n_points
        h = cp.Variable(n_points)
        constraints = [h >= 0, h <= 1, cp.sum(h) * dx == n_points / 2]
        objective = cp.Maximize(cp.max([cp.sum(h[i:i + n_points] * (1 - h[i + k:i + n_points + k])) for k in range(1, n_points // 2 + 1)]))
        problem = cp.Problem(objective, constraints)
        try:
            problem.solve(solver=cp.MOSEK)
            if problem.status == 'OPTIMAL':
                h_val = h.value
                c5 = evaluate_erdos_solution(h_val, n_points, dx)
                if c5 < best_c5:
                    best_h = h_val.copy()
                    best_c5 = c5
        except:
            pass
        elapsed = time.time() - start_time

    return best_h, best_c5, n_points
```
