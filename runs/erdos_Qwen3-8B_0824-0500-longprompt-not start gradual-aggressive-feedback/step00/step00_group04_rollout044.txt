import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different numbers of points
    n_points_list = [100, 200, 300, 400]
    
    best_c5 = float('inf')
    best_h = None
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)
        
        # Initial guess: start from the given initial_h_values or a uniform distribution
        if n_points == initial_h_values.shape[0]:
            h_initial = initial_h_values.copy()
        else:
            h_initial = np.random.uniform(0, 1, n_points)
            # Ensure the integral condition is satisfied
            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        # Define variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [h <= 1, h >= 0, cp.sum(h) == n_points / 2]
        
        # Objective function: minimize the maximum overlap integral
        # For efficiency, we'll use a convex relaxation
        # Since the max is non-convex, we'll use a heuristic search
        
        # Use a simulated annealing-like optimization
        def objective_func(h_values):
            h_array = h_values.copy()
            # Compute the correlation
            corr = np.correlate(h_array, 1 - h_array, mode="full")
            # Convert to integral by multiplying by dx
            c5 = np.max(corr * dx)
            return c5
        
        # Use optimization to minimize the maximum overlap
        def optimize_h(n_points, dx, x, initial_h, max_iterations=1000):
            h = initial_h.copy()
            best_h = h.copy()
            best_c5 = objective_func(h)
            
            for _ in range(max_iterations):
                # Perturb the solution
                perturbation = np.random.normal(0, 0.05, n_points)
                h_new = np.clip(h + perturbation, 0, 1)
                h_new = h_new / np.sum(h_new) * (n_points / 2)
                
                # Evaluate new solution
                new_c5 = objective_func(h_new)
                
                # Accept better solution with some probability
                if new_c5 < best_c5:
                    h = h_new
                    best_c5 = new_c5
                    best_h = h.copy()
                else:
                    # With some probability, accept worse solution
                    if np.random.rand() < np.exp((best_c5 - new_c5) / 0.1):
                        h = h_new
                        best_c5 = new_c5
                        best_h = h.copy()
            return best_h, best_c5
        
        # Optimize
        h_opt, c5_opt = optimize_h(n_points, dx, x, h_initial)
        
        # Update best solution
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Format the output
    n_points = best_h.shape[0]
    h_values = best_h
    c5_bound = best_c5
    return h_values, c5_bound, n_points
