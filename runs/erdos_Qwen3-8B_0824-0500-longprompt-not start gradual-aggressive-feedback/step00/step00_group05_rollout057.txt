import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp
import scipy.sparse.linalg as spla
import time
from scipy.optimize.slsqp import minimize_slsqp
from scipy.optimize import Bounds
import warnings
warnings.filterwarnings("ignore")

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction (from initial_h_values)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    assert np.isclose(sum_h, n_points / 2), "Initial h does not satisfy the integral constraint"
    
    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate((h, h[:1]))  # pad to make full convolution possible
        conv = np.correlate(h, 1 - h, mode='full')
        conv = conv[1:-1]  # remove padding
        c5 = np.max(conv * dx)
        return c5
    
    # Function to compute the gradient of C5 (for optimization)
    def compute_gradient(h):
        h = np.array(h)
        # Create a grid of x values
        x = np.linspace(0, 2, n_points)
        # Create the function h(x) and 1 - h(x + k)
        # For efficiency, use vectorized operations
        h_func = np.interp(x, x, h)
        h_func = np.concatenate((h_func, h_func[:1]))  # pad for full convolution
        conv = np.correlate(h_func, 1 - h_func, mode='full')
        conv = conv[1:-1]
        max_index = np.argmax(conv)
        # Compute the gradient for the max convolution
        h_func_padded = np.concatenate((h_func, h_func[:1]))
        h_func_padded = h_func_padded[1:-1]
        conv_derivative = np.zeros_like(h)
        for i in range(n_points):
            dx = x[i] - x
            h_func_padded = np.zeros_like(h_func_padded)
            h_func_padded[i] = 1
            conv_padded = np.correlate(h_func_padded, 1 - h_func_padded, mode='full')
            conv_padded = conv_padded[1:-1]
            conv_derivative[i] = conv_padded[max_index] * dx
        return conv_derivative
    
    # Function to evaluate the constraint (integral constraint)
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Function for the optimization
    def optimization_objective(h):
        c5 = evaluate_c5(h)
        return c5
    
    # Bounds on h
    bounds = Bounds(0, 1)
    
    # Start with the initial guess
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Try a few starting points with slight perturbations
    for perturbation in [0.0, 0.01, 0.05, 0.1]:
        perturbed_h = initial_h + perturbation * np.random.randn(n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        c5 = evaluate_c5(perturbed_h)
        if c5 < best_c5:
            best_h = perturbed_h
            best_c5 = c5
    
    # Optimization loop
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Optimization with SLSQP
        result = minimize_slsqp(
            optimization_objective, 
            best_h,
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            jac=compute_gradient,
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        if result.success:
            best_h = result.x
            best_c5 = evaluate_c5(best_h)
        else:
            # If optimization fails, try another perturbation
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            perturbed_h = best_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0, 1)
            c5 = evaluate_c5(perturbed_h)
            if c5 < best_c5:
                best_h = perturbed_h
                best_c5 = c5
        
        # Early stopping if we reached the target
        if best_c5 < 0.380876:
            break
    
    return (best_h, best_c5, n_points)
