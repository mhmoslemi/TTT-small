import numpy as np

# Pre-imported functions

def run(seed=42, budget_s=TIME_BUDGET, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial guess
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values)
    best_c5 = evaluate_erdos_solution(h_values)
    
    # Set up time tracking
    start_time = time.time()
    
    # Use a local optimization strategy with gradient descent
    def objective(h):
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute overlap integral
        h = h.reshape(-1, 1)
        h_shifted = np.roll(h, shift=1, axis=0)
        overlap = np.sum(h * (1 - h_shifted)) * (2.0 / n_points)
        return overlap
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Gradient descent with momentum
    h = h_values.copy()
    lr = 0.01
    momentum = 0.9
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for step in range(1000):
        # Compute objective and gradient (approximate)
        h_tiled = h.reshape(-1, 1)
        h_shifted = np.roll(h_tiled, shift=1, axis=0)
        overlap = np.sum(h_tiled * (1 - h_shifted)) * (2.0 / n_points)
        
        # Compute gradient (approximate)
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus_tiled = h_plus.reshape(-1, 1)
            h_plus_shifted = np.roll(h_plus_tiled, shift=1, axis=0)
            overlap_plus = np.sum(h_plus_tiled * (1 - h_plus_shifted)) * (2.0 / n_points)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus_tiled = h_minus.reshape(-1, 1)
            h_minus_shifted = np.roll(h_minus_tiled, shift=1, axis=0)
            overlap_minus = np.sum(h_minus_tiled * (1 - h_minus_shifted)) * (2.0 / n_points)
            
            grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
        
        # Update
        grad_proj = grad - (np.sum(grad) / n_points)  # Project gradient to be mean-zero
        grad_proj = np.clip(grad_proj, -1, 1)
        h = h - lr * grad_proj
        
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        h = np.roll(h, shift=1, axis=0)  # Shift to avoid bias
        
        # Check constraint
        if np.abs(np.sum(h) - n_points / 2) > 1e-6:
            h = h_values.copy()
        
        # Compute current C5
        current_c5 = evaluate_erdos_solution(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Check time budget
        if time.time() - start_time > budget_s:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
