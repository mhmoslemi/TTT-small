import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the initial number of points
    n_points = 70
    h_initial = initial_h_values.copy()
    
    # Function to evaluate the C5 value
    def evaluate_c5(h_values):
        h = np.array(h_values)
        h_shifted = np.roll(h, shift=1, axis=0)
        corr = signal.correlate(h, 1 - h, mode='full')
        dx = 2.0 / len(h)
        c5 = np.max(corr * dx)
        return c5
    
    # Function to compute the overlap integral
    def compute_overlap(h_values):
        h = np.array(h_values)
        corr = signal.correlate(h, 1 - h, mode='full')
        dx = 2.0 / len(h)
        return corr * dx
    
    # Function to optimize h_values
    def optimize_h(n_points, initial_h, budget_s):
        h = initial_h.copy()
        best_h = h.copy()
        best_c5 = evaluate_c5(h)
        best_overlap = compute_overlap(h)
        
        start_time = time.time()
        
        # Use L-BFGS-B for optimization
        def objective_function(x):
            h = x.reshape((-1, 1))
            h = np.clip(h, 0, 1)
            dx = 2.0 / len(h)
            corr = signal.correlate(h, 1 - h, mode='full')
            c5_values = corr * dx
            return np.max(c5_values)
        
        def constraint_sum(x):
            return np.sum(x) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = optimize.minimize(
            objective_function,
            x0=h,
            bounds=bounds,
            constraints=constraints,
            method='L-BFGS-B',
            options={'ftol': 1e-8, 'maxiter': 1000}
        )
        
        if result.success:
            h_optimal = result.x
            c5_optimal = evaluate_c5(h_optimal)
            if c5_optimal < best_c5:
                best_h = h_optimal
                best_c5 = c5_optimal
            best_overlap = compute_overlap(best_h)
        
        # Try perturbing the solution if not improved
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturbed = np.clip(best_h + perturbation, 0, 1)
            c5_perturbed = evaluate_c5(h_perturbed)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
        
        elapsed_time = time.time() - start_time
        if elapsed_time < budget_s:
            # Try a few more iterations if time allows
            for _ in range(3):
                result = optimize.minimize(
                    objective_function,
                    x0=best_h,
                    bounds=bounds,
                    constraints=constraints,
                    method='L-BFGS-B',
                    options={'ftol': 1e-8, 'maxiter': 100}
                )
                if result.success:
                    h_optimal = result.x
                    c5_optimal = evaluate_c5(h_optimal)
                    if c5_optimal < best_c5:
                        best_h = h_optimal
                        best_c5 = c5_optimal
                elapsed_time = time.time() - start_time
                if elapsed_time >= budget_s:
                    break
        
        return best_h, best_c5
    
    # Run the optimization
    best_h, best_c5 = optimize_h(n_points, h_initial, budget_s)
    
    # Evaluate the solution
    c5_final = evaluate_c5(best_h)
    
    return best_h, c5_final, n_points
