def run(seed=42, budget_s=60, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 2 - 1  # Random initial guess
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the integral constraint
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    elapsed_time = 0
    
    # Define the objective function to minimize the maximum overlap
    def objective_func(h_values):
        c5 = evaluate_erdos_solution(h_values)
        return c5
    
    # Optimization loop
    for _ in range(100):  # Run for a fixed number of iterations
        start_time = time.time()
        
        # Local search with gradient descent
        h_new = h + np.random.normal(0, 0.05, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = objective_func(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
        
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
