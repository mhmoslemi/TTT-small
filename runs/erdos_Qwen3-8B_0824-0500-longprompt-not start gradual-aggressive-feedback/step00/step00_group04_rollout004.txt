```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=np.float64) * (n_points / 2) / np.sum(initial_h_values[:n_points])
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)

    start_time = time.time()
    elapsed = 0
    max_iter = 100  # Limit number of iterations to stay within budget

    for _ in range(max_iter):
        if elapsed > budget_s - 1:
            break
        # Convert h to a variable for optimization
        h_var = cp.Variable(n_points, bounds=(0, 1))
        objective = cp.Minimize(cp.Max([cp.sum(h_var * (1 - h_var[k:])) * dx for k in range(n_points)]))
        prob = cp.Problem(objective)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4, max_iters=1000)
            if prob.status == 'optimal':
                h = h_var.value
                h = np.clip(h, 0, 1)
                current_c5 = evaluate_erdos_solution(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                elapsed = time.time() - start_time
            else:
                break
        except:
            break

    # If we didn't find a better solution, try random perturbations
    if best_c5 >= best_c5:
        for _ in range(10):
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            h = best_h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate_erdos_solution(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    return (best_h, best_c5, n_points)
```
