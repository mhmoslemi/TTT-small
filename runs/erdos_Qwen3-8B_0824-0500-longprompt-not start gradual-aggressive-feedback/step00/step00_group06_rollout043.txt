```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points for speed
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=float)
    h = h / (dx * n_points) * n_points / 2  # Ensure sum(h) * dx = 1
    
    # Function to compute overlap integral
    def compute_c5(h):
        h = np.array(h, dtype=float)
        h = h / (dx * n_points) * n_points / 2
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Initial evaluation
    c5_initial = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_initial
    best_n_points = n_points
    
    # Try increasing n_points if time permits
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        dx = 2.0 / n_points
        h = np.random.rand(n_points) * 0.5  # Random initial guess
        h = h / (dx * n_points) * n_points / 2  # Ensure sum(h) * dx = 1
        
        # Optimization using CVXPY
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h_var) == n_points / 2]
        objective = cp.Maximize(cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, 1))))
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.ECOS, eps=1e-5, verbose=False)
            if result is not None:
                h_opt = h_var.value
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5
                    best_n_points = n_points
        except:
            pass
    
    # Try further optimization on best solution
    n_points = best_n_points
    dx = 2.0 / n_points
    h = best_h.copy()
    h = h / (dx * n_points) * n_points / 2
    
    # Use scipy.optimize to minimize the max overlap
    def objective(x):
        x = np.array(x, dtype=float)
        x = x / (dx * n_points) * n_points / 2
        corr = np.correlate(x, 1 - x, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=bounds, tol=1e-5, options={'maxiter': 100})
    h_opt = result.x
    c5_opt = compute_c5(h_opt)
    
    if c5_opt < best_c5:
        best_h = h_opt.copy()
        best_c5 = c5_opt
    
    return best_h, best_c5, best_n_points
```
