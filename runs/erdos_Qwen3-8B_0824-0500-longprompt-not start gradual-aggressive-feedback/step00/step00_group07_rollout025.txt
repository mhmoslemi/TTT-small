def run(seed=42, budget_s=60, **kwargs):

        h_shifted = h_shifted[:n_points]
        overlap = np.correlate(h, 1 - h_shifted, mode="full")

        h = np.array(h)

        h_shifted = np.roll(h, 1)
        h_shifted = h_shifted[:n_points]
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap) * dx

    def compute_c5_with_derivative(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points - 1]])
        h_shifted = np.roll(h, 1)
        h_shifted = h_shifted[:n_points]
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap) * dx

    def compute_derivative(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points - 1]])
        h_shifted = np.roll(h, 1)
        h_shifted = h_shifted[:n_points]
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap) * dx

        max_index = np.argmax(overlap)
        # Compute derivative
        derivative = np.zeros(n_points)
        for i in range(n_points):
            # For each position i, compute derivative
            # We differentiate the overlap with respect to h[i]
            # For the overlapping integral at position max_index, derivative is:
            # sum over k of (h[i] * (1 - h[i + k + 1]) - h[i] * (1 - h[i + k]))
            # where the indices are properly wrapped
            # This is not easy to compute analytically, so we approximate numerically
            # We use a small perturbation to approximate the derivative
            perturbation = 1e-6
            h_perturbed = h.copy()
            h_perturbed[i] += perturbation
            c5_perturbed = compute_c5_with_derivative(h_perturbed)[0]
            derivative[i] = (c5_perturbed - c5) / perturbation
        return derivative

    def objective_and_derivative(h):
        c5 = compute_c5(h)
        derivative = compute_derivative(h)

        return sum(h) - n_points / 2

    def optimize_h(h):

        learning_rate = 0.1

            h = h - learning_rate * derivative

            h = np.clip(h, 0, 1)
            # Enforce the constraint sum(h) = n_points / 2
            # Compute the constraint violation
            constraint_violation = constraint_sum(h)
            # Adjust h to satisfy the constraint
            if constraint_violation > 0:
                # Need to decrease sum(h)
                h -= np.abs(constraint_violation) / n_points
            elif constraint_violation < 0:
                # Need to increase sum(h)
                h += np.abs(constraint_violation) / n_points
            # Clamp h again
            h = np.clip(h, 0, 1)
            # Enforce the constraint
            h = h - constraint_violation / n_points
            h = np.clip(h, 0, 1)
        return h
    
    # Optimize h
    h = optimize_h(h)
    
    # Evaluate the final result
    final_c5 = compute_c5(h)
    # Return the result
    return (h.tolist(), final_c5, n_points)
