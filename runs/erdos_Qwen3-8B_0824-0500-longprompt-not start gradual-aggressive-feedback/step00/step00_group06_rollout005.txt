import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for actual initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for speed
    dx = 2.0 / n_points

    # Initial guess from provided initial values
    h = initial_h_values.copy()
    h = h[:n_points]  # Trim to n_points if necessary
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)  # Ensure it's over [0, 2]

    # Normalize to satisfy the integral constraint
    sum_h = np.sum(h) * dx
    h = h * (1.0 / sum_h) * (n_points / 2.0)

    def objective(h_values):
        # Ensure values are in [0, 1]
        h_values = np.clip(h_values, 0, 1)
        # Compute the overlap integral
        h = h_values.reshape(-1)
        h_expanded = np.zeros_like(h)
        h_expanded[:len(h)] = h
        h_expanded[len(h):] = h
        overlap = signal.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

    def constraints(h_values):
        # Ensure the integral constraint is met
        sum_h = np.sum(h_values) * dx
        return [sum_h - (n_points / 2.0)]

    # Optimization setup
    bounds = Bounds(0, 1)
    constraints = LinearConstraint(np.ones(n_points), [n_points / 2.0], [n_points / 2.0])

    # Start optimization
    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-8, 'maxiter': 100})
    elapsed = time.time() - start_time

    # If the optimization is not completed within the time, return the initial guess
    if elapsed >= budget_s:
        return initial_h_values[:n_points], 0.38092, n_points

    # Evaluate the result
    h_values = result.x
    c5_bound = objective(h_values)
    return h_values, c5_bound, n_points
