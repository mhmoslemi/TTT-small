import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming initial values are loaded here

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial values
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h)
    
    # Define a function to evaluate the overlap integral
    def evaluate_overlap(h_vals):
        return evaluate_erdos_solution(h_vals)
    
    # Define a function to optimize the function h_vals
    def optimize_h_vals(h_vals, n_points, dx, budget_s):
        # Convert to variable
        h_var = cp.Variable(n_points, bounds=(0, 1))
        # Constraint: integral of h is 1
        constraint = cp.sum(h_var) == n_points / 2
        
        # Objective: minimize max overlap
        # To handle max, we use a convex optimization trick
        # We introduce a variable t and minimize t such that all overlaps are <= t
        t = cp.Variable()
        constraints = [cp.sum(h_var) == n_points / 2]
        for k in range(1, n_points):
            # For each k, compute the overlap
            # Overlap for shift k is sum(h * (1 - h[k:])) * dx
            # We use the fact that np.correlate(h, 1 - h, mode='full') gives the overlaps
            # We can approximate this using the correlation
            # Since we are minimizing the maximum overlap, we can use the t constraint
            
            # For each k, compute the correlation
            # We use the current h_vals to approximate this
            # Note: we could implement this with a more precise formulation, but for the sake of time and code simplicity,
            # we use a simple numerical approximation
            
            # Compute the overlap for shift k
            corr_val = np.correlate(h_vals, 1 - h_vals, mode='full')[k]
            # Add constraint: corr_val * dx <= t
            constraints.append(t >= corr_val * dx)
        
        # Solve the problem
        prob = cp.Problem(cp.Minimize(t), constraints)
        result = prob.solve(solver=cp.CVXOPT, verbose=False)
        if result is None:
            return h_vals, np.inf
        return h_var.value, t.value
    
    # Optimize
    optimized_h, new_c5 = optimize_h_vals(h, n_points, dx, budget_s)
    
    # Check if we have time to further refine
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        # Use local optimization to further reduce C5
        def objective_func(h_vals):
            return evaluate_overlap(h_vals)
        result = opt.minimize(objective_func, optimized_h, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
        optimized_h = result.x
        new_c5 = evaluate_overlap(optimized_h)
    
    # Final result
    final_c5 = new_c5
    return optimized_h, final_c5, n_points
