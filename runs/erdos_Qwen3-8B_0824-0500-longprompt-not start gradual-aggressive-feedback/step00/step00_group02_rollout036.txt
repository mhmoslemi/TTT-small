import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Define parameters
    n_points = 100
    
    # Initial guess
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial[0::2] = 0.5
    h_initial[1::2] = 0.5
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Objective function and constraints
    def objective(h):
        c5 = evaluate_erdos_solution(h)
        return c5
    
    # Optimization with L-BFGS-B
    result = opt.minimize(
        fun=objective,
        x0=h_initial,
        bounds=[(0, 1) for _ in range(n_points)],
        method='L-BFGS-B',
        options={'maxiter': 100, 'disp': False}
    )
    
    # Check if we have time to refine
    elapsed = time.time() - start_time
    if elapsed < budget_s - 5:
        # Use a fine-tuning approach with simulated annealing
        result = opt.minimize(
            fun=objective,
            x0=result.x,
            bounds=[(0, 1) for _ in range(n_points)],
            method='annealing',
            options={'maxiter': 100, 'disp': False}
        )
    
    # Final result
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, n_points
