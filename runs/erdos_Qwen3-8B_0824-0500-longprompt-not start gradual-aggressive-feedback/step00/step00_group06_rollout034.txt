```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
from scipy.optimize import Bounds, minimize, LinearConstraint
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder, assuming initial construction is loaded

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 500  # Choose a moderate number of points for good balance between speed and accuracy
    
    # Initialize h with some structure or random starting point
    h = np.random.rand(n_points) * 2 - 1  # Random samples in [-1, 1]
    h = np.clip((h + 1) / 2, 0, 1)  # Normalize to [0, 1]
    h = initial_h_values[:n_points]  # Use the initial construction as a starting point
    
    # Ensure the integral constraint is satisfied
    integral = np.sum(h) * (2.0 / n_points)
    h = h * (1 / integral) * (n_points / 2)  # Scale to meet the constraint
    
    # Optimize h to minimize the maximum overlap
    def objective(h_vals):
        h = h_vals
        corr = signal.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        return np.max(corr)
    
    # Use bounds and constraints
    bounds = Bounds(0, 1)
    
    # Use a constrained optimization approach
    def optimize_h(h0):
        result = minimize(
            fun=objective,
            x0=h0,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        return result.x
    
    # Run the optimization
    h_opt = optimize_h(h)
    
    # Ensure the integral constraint is still satisfied after optimization
    integral = np.sum(h_opt) * (2.0 / n_points)
    h_opt = h_opt * (1 / integral) * (n_points / 2)
    
    # Evaluate the result
    c5_bound = evaluate_erdos_solution(h_opt)
    
    # Check if we're within the time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return h_opt, c5_bound, n_points
    
    # Further refinement with more advanced optimization
    # Here we use a convex optimization approach for better performance
    x = cp.Variable(n_points)
    constraints = [cp.sum(x) == n_points / 2]
    obj = cp.Maximize(cp.max(cp.multiply(x, 1 - x)))
    prob = cp.Problem(obj, constraints)
    result = prob.solve(solver=cp.CVXOPT, abstol=1e-8, reltol=1e-8)
    h_cvx = x.value
    
    # Ensure the integral constraint is satisfied after optimization
    integral = np.sum(h_cvx) * (2.0 / n_points)
    h_cvx = h_cvx * (1 / integral) * (n_points / 2)
    
    # Evaluate the result
    c5_bound = evaluate_erdos_solution(h_cvx)
    
    # Return the best solution found
    return h_cvx, c5_bound, n_points
```
