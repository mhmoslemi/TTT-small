```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set a random seed for reproducibility
    np.random.seed(seed)

    # Define the number of points; we'll use 200 points for a good balance
    n_points = 200
    dx = 2.0 / n_points

    # Initial guess from the provided construction
    h_initial = np.array(initial_h_values[:n_points], dtype=np.float64)
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)

    # Function to evaluate C5 for a given h
    def evaluate_c5(h):
        # Make sure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Rescale to [0, 2] by taking the middle part
        corr = corr[n_points - 1: 2 * n_points - 1]
        # Multiply by dx to get the integral
        c5 = np.max(corr * dx)
        return c5

    # Define the objective function for optimization
    def objective(h):
        return evaluate_c5(h)

    # Define the constraint: sum(h) = n_points / 2
    def constraint(h):
        return np.sum(h) - (n_points / 2)

    # Use L-BFGS-B optimizer
    result = opt.minimize(
        fun=objective,
        x0=h_initial,
        bounds=[(0, 1) for _ in range(n_points)],
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    # Check if the optimization was successful
    if result.success:
        best_h = result.x
    else:
        # If optimization failed, use the initial guess
        best_h = h_initial

    # Evaluate the best solution
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, n_points
```
