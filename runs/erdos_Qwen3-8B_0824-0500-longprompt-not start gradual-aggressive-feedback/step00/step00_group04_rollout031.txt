import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Initialize parameters
    n_points = 42  # Start with the initial sample size
    dx = 2.0 / n_points
    max_iterations = 100  # Limit the number of iterations for time constraints
    tolerance = 1e-4  # Tolerance for convergence
    
    # Initial solution from the provided initial_h_values
    h_values = np.array(initial_h_values)
    h_values = h_values[:n_points]  # Ensure it's the correct length
    
    # Function to compute C_5
    def compute_c5(h):
        h = h.reshape(-1)
        h_ext = np.concatenate([h, np.zeros_like(h)])
        corr = np.correlate(h, 1 - h_ext, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute the overlap integral for all possible shifts
    def compute_all_overlaps(h):
        h = h.reshape(-1)
        h_ext = np.concatenate([h, np.zeros_like(h)])
        corr = np.correlate(h, 1 - h_ext, mode="full")
        overlaps = corr * dx
        return overlaps
    
    # Optimization function
    def optimize_h(h_prev, max_iter=100, tol=1e-4):
        h = h_prev.copy()
        best_h = h.copy()
        best_c5 = compute_c5(h)
        for i in range(max_iter):
            # Compute all overlaps
            overlaps = compute_all_overlaps(h)
            # Find the shift with the maximum overlap
            max_overlap_index = np.argmax(overlaps)
            shift = max_overlap_index - len(h) + 1
            if shift < 0:
                shift += len(h)
            # Construct the new objective: minimize overlap for this shift
            # Objective is to minimize the overlap at this particular shift
            # Formulate as constrained optimization
            x = cp.Variable(len(h))
            # Objective: minimize the overlap at this shift
            overlap = cp.multiply(h, 1 - h[shift:])
            obj = cp.Minimize(cp.sum(overlap))
            # Constraints: sum(h) == n_points / 2, 0 <= h <= 1
            constraints = [
                cp.sum(x) == n_points / 2,
                0 <= x,
                x <= 1
            ]
            # Solve the optimization problem
            prob = cp.Problem(obj, constraints)
            try:
                prob.solve(solver=cp.SCS, eps=1e-5)
                if prob.status != 'optimal':
                    break
                h = x.value
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            except:
                break
            # Early stopping if the change is small
            if np.linalg.norm(h - best_h) < tol:
                break
        return best_h, best_c5
    
    # Run optimization
    best_h, best_c5 = optimize_h(h_values, max_iter=max_iterations, tol=tolerance)
    
    # Check if we have time to increase the sample size
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 5:
        # Try with more points
        n_points *= 2
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        # Initialize with a simple pattern
        for i in range(n_points):
            if i % 4 == 0 or i % 4 == 1:
                h_values[i] = 0.5
            else:
                h_values[i] = 0.75
        h_values = h_values / np.sum(h_values) * n_points / 2  # Adjust to satisfy the integral constraint
        best_h, best_c5 = optimize_h(h_values, max_iter=50, tol=1e-3)
    
    # Ensure the solution is valid
    h_values = best_h
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values / np.sum(h_values) * n_points / 2
    
    # Final evaluation
    final_c5 = compute_c5(h_values)
    
    # Return the results
    return (h_values, final_c5, n_points)
