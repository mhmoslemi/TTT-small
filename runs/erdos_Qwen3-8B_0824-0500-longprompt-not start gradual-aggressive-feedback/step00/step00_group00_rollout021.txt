import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
# initial_h_values is a numpy array with n_points samples
# evaluate_erdos_solution takes h and returns the C5 value

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different numbers of points, starting from a small number
    n_points_list = [45, 80, 120, 160, 200]
    
    best_c5 = float('inf')
    best_h = np.zeros(1)
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initial guess from provided initial_h_values (resized if needed)
        if hasattr(initial_h_values, "__array__"):
            if initial_h_values.shape[0] > n_points:
                initial_h_values = initial_h_values[:n_points]
            elif initial_h_values.shape[0] < n_points:
                pad = np.zeros(n_points - initial_h_values.shape[0])
                initial_h_values = np.concatenate((initial_h_values, pad))
        
        # Ensure the sum is correct
        h_initial = initial_h_values.copy()
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        # Optimization variables
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # Objective: minimize the maximum correlation
        # We reformulate the problem as minimizing the maximum value among all shifts
        # To do this, we maximize a value for each shift and minimize the maximum over all shifts
        
        # Precompute the shifted values
        max_shift = 1  # Maximum shift to consider (since the interval is [0, 2])
        correlations = []
        for shift in np.linspace(0, max_shift, 100):  # use discrete shifts for approximation
            shifted_h = np.roll(h.value, int(shift / dx)) if h.value is not None else np.zeros(n_points)
            shifted_1_minus_h = 1 - shifted_h
            # compute the convolution
            # to approximate the integral, we use the discrete correlation
            corr = np.correlate(h.value, shifted_1_minus_h, mode="full")
            corr = corr * dx
            correlations.append(np.max(corr))
        
        # We'll use a heuristic approach to find a good h that reduces the max correlation
        # We'll use a local optimization routine with the current h_initial
        
        # Heuristic: optimize for a few key shifts
        # For simplicity, we'll consider the correlations for shifts in [0, 0.5, 1.0]
        # These are the shifts that can cause significant overlaps
        
        # Objective: minimize the maximum of the three correlations
        obj = cp.Minimize(cp.Max([cp.Max(np.correlate(h, 1 - np.roll(h, int(shift / dx)), mode="full") * dx) for shift in [0, 0.5, 1.0]]))
        
        # Solve the problem
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.SCS, eps=1e-4)
        except:
            # If the solver fails, try a different approach
            # Use a local optimization method to refine the initial guess
            def objective_func(h_values):
                h_values = h_values.reshape(n_points)
                # Normalize to ensure sum is n_points / 2
                h_values = h_values / np.sum(h_values) * (n_points / 2)
                # Compute the maximum correlation
                max_corr = 0
                for shift in np.linspace(0, 1, 10):
                    shifted_h = np.roll(h_values, int(shift / dx))
                    corr = np.correlate(h_values, 1 - shifted_h, mode="full") * dx
                    max_corr = max(max_corr, np.max(corr))
                return max_corr
            
            # Use scipy's optimization to find a minimum
            result = opt.minimize(objective_func, h_initial, method='L-BFGS-B', bounds=[(0, 1)]*n_points, tol=1e-4)
            h_opt = result.x
            c5 = result.fun
            
            # Update the best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        else:
            # Update the best solution
            if prob.status == 'optimal':
                h_opt = h.value
                c5 = np.max(np.correlate(h_opt, 1 - np.roll(h_opt, int(0.5 / dx)), mode="full") * dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
    
    # Return the best solution found
    return (best_h, best_c5, len(best_h))
