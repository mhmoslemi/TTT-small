import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points])  # Start with the provided initial values
    
    # Function to compute the overlap integral
    def compute_c5(h_vals):
        h = h_vals.copy()
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Compute the maximum overlap integral
        c5 = np.max(corr * dx)
        return c5
    
    # Function to optimize h with constraints
    def optimize_h(h_vals, n_points, max_iter=100):
        h = cp.Variable(n_points)
        # Objective: minimize the maximum overlap integral
        # Note: This is a convex optimization problem (minimax)
        # We use a convex relaxation by minimizing the maximum using a single constraint
        # We use a threshold value and iterate
        max_c5 = cp.Maximize(cp.minimize(cp.multiply(cp.diag(h), 1 - h)))
        constraints = [
            cp.sum(h) == n_points / 2,  # Integral constraint
            0 <= h,
            h <= 1
        ]
        # Optimization
        prob = cp.Problem(cp.Minimize(cp.Max(cp.multiply(h, 1 - h))), constraints)
        result = prob.solve()
        return h.value
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Try different resolutions within the budget
    for resolution in range(100, 500, 50):
        if time.time() - start_time > budget_s - 1:
            break
        n_points = resolution
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2)  # Scale to meet integral constraint
        h = optimize_h(h, n_points)
        h = np.clip(h, 0, 1)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, len(best_h)
