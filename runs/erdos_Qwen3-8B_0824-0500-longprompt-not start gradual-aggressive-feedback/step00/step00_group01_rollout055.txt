```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial point
    n_points = 200
    h = np.array(initial_h_values[:n_points])
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    def objective(h):
        # Convert to function
        h_func = np.interp(np.linspace(0, 2, n_points), np.arange(n_points), h)
        # Compute overlap integral
        overlap = np.correlate(h_func, 1 - h_func, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    def optimize_h():
        # Initial attempt with simulated annealing
        result = optimize.minimize(
            fun=objective,
            x0=h,
            method="L-BFGS-B",
            bounds=[(0, 1) for _ in range(n_points)],
            jac=None,
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        best_h = result.x
        best_c5 = objective(best_h)
        return best_h, best_c5
    
    best_h, best_c5 = optimize_h()
    
    # Refinement with gradient descent
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        
        # Gradient descent step
        h_func = np.interp(np.linspace(0, 2, n_points), np.arange(n_points), best_h)
        overlap = np.correlate(h_func, 1 - h_func, mode="full") * (2.0 / n_points)
        max_overlap = np.max(overlap)
        idx = np.argmax(overlap)
        
        # Perturb the function to reduce max overlap
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        # Evaluate
        c5_perturbed = objective(perturbed_h)
        if c5_perturbed < best_c5:
            best_h = perturbed_h
            best_c5 = c5_perturbed
    
    # Final evaluation
    final_h = best_h
    final_c5 = objective(final_h)
    return final_h, final_c5, n_points
```
