import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with n=100 for balance between performance and accuracy
    
    # Initial guess based on the provided initial_h_values
    h_initial = np.array(initial_h_values[:n_points])
    h_initial = np.interp(h_initial, (h_initial.min(), h_initial.max()), (0, 1))  # Normalize
    h_initial = h_initial * (n_points / 2)  # Scale to satisfy the integral constraint
    
    def objective(h):
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        # Compute the overlap integral
        h_fft = np.fft.fft(h)
        one_minus_h_fft = np.fft.fft(1 - h)
        overlap_fft = h_fft * np.conj(one_minus_h_fft)
        overlap = np.real(np.fft.ifft(overlap_fft))
        c5 = np.max(overlap * dx)
        return c5
    
    def constraint(h):
        return np.sum(h) - n_points / 2  # Enforce ∫h(x)dx = 1
    def constraint_func(h):
        return constraint(h)
    
    # Optimization with bounds
    h = cp.Variable(n_points, lower=0, upper=1)
    objective_func = objective(h)
    constraints = [constraint(h) == 0]
    problem = cp.Problem(cp.Minimize(objective_func), constraints)
    
    # Solve the problem
    try:
        result = problem.solve(solver=cp.SCS, max_iter=100, time_limit=budget_s - (time.time() - start_time))
        if result is not None:
            h_values = np.round(h.value, 4)
            c5_bound = objective(h_values)
            return (h_values, c5_bound, n_points)
    except:
        pass
    
    # Fallback: Use gradient descent with constraints
    h = np.clip(h_initial, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to ensure integral is 1
    
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Compute objective
        dx = 2.0 / n_points
        h_fft = np.fft.fft(h)
        one_minus_h_fft = np.fft.fft(1 - h)
        overlap_fft = h_fft * np.conj(one_minus_h_fft)
        overlap = np.real(np.fft.ifft(overlap_fft))
        c5 = np.max(overlap * dx)
        
        # Gradient descent step
        grad = np.gradient(overlap * dx)
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    c5_bound = objective(h)
    return (h, c5_bound, n_points)
