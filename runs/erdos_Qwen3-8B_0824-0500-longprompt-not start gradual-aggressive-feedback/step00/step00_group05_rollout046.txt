import numpy as np

# Pre-imported functions for evaluation and initial h
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Set up parameters
    max_points = 100  # We'll try up to 100 points for speed
    best_c5 = np.inf
    best_h = None
    best_n_points = 0
    
    # Try progressively larger n_points up to max_points
    for n_points in range(50, max_points + 1, 10):
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)
        
        # Start from the initial h if available, or use random
        if n_points <= len(initial_h_values):
            h = np.array(initial_h_values[:n_points], dtype=float)
        else:
            h = np.random.uniform(0, 1, n_points)
        
        # Ensure the integral is exactly 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Optimization: minimize the max overlap
        # We'll use a heuristic to find a good h
        
        # Define the overlap function
        def compute_c5(h):
            # Compute 1 - h shifted versions
            h_shifted = np.zeros((n_points, n_points))
            for i in range(n_points):
                h_shifted[i, :] = np.roll(h, i)
            # Compute the overlap
            overlap = np.zeros(n_points)
            for i in range(n_points):
                overlap[i] = np.sum(h * (1 - h_shifted[i, :])) * dx
            return np.max(overlap)
        
        # Optimization: use gradient descent to minimize the max overlap
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            return compute_c5(h)
        
        # Initial guess
        h_flat = h.copy()
        
        # Run gradient descent for a limited number of steps
        for _ in range(100):
            # Evaluate the objective
            current_c5 = objective_func(h_flat)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_flat.copy()
                best_n_points = n_points
            # Compute gradient numerically
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h_flat.copy()
                h_plus[i] += 1e-5
                grad[i] = (objective_func(h_plus) - objective_func(h_flat)) / 1e-5
            # Update the h
            h_flat -= 0.1 * grad
        
        # Check if we have improved the best solution
        if compute_c5(h_flat) < best_c5:
            best_c5 = compute_c5(h_flat)
            best_h = h_flat.copy()
            best_n_points = n_points
    
    # Evaluate the best solution
    c5_bound = evaluate_erdos_solution(best_h)
    
    # Return the best solution found
    return (best_h, c5_bound, best_n_points)
