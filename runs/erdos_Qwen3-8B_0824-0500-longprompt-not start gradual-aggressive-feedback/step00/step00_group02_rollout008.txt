import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Starting with 100 points, can be adjusted
    dx = 2.0 / n_points
    total_integral = n_points / 2.0  # Ensures ∫h(x)dx = 1

    # Initial guess based on initial_h_values
    h_initial = np.array(initial_h_values[:n_points], dtype=float)
    h_initial = np.clip(h_initial, 0, 1)  # Ensure within bounds
    h_initial = h_initial * (total_integral / np.sum(h_initial))  # Normalize to meet the integral constraint

    def objective(h):
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * dx
        if abs(integral - 1) > 1e-6:
            return np.inf  # Penalize for not meeting the constraint
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5

    def constraint(h):
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * dx
        return integral - 1

    # Bounds for each sample
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess
    x0 = h_initial.copy()

    # Optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        tol=1e-6,
        options={'maxiter': 100, 'disp': False}
    )

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h_initial, evaluate_erdos_solution(h_initial), n_points)

    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)

    # Try refining with a smaller step size
    n_points_refine = 200
    dx_refine = 2.0 / n_points_refine
    total_integral_refine = n_points_refine / 2.0

    # Refine the solution
    h_refine = np.zeros(n_points_refine)
    h_refine[:n_points] = best_h
    h_refine = np.clip(h_refine, 0, 1)
    h_refine = h_refine * (total_integral_refine / np.sum(h_refine))

    def objective_refine(h):
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * dx_refine
        if abs(integral - 1) > 1e-6:
            return np.inf
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx_refine
        return c5

    def constraint_refine(h):
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * dx_refine
        return integral - 1

    bounds_refine = [(0, 1) for _ in range(n_points_refine)]
    x0_refine = h_refine.copy()

    result_refine = optimize.minimize(
        objective_refine,
        x0_refine,
        method='L-BFGS-B',
        bounds=bounds_refine,
        constraints=[{'type': 'eq', 'fun': constraint_refine}],
        tol=1e-6,
        options={'maxiter': 100, 'disp': False}
    )

    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (h_refine, evaluate_erdos_solution(h_refine), n_points_refine)

    best_h_refine = result_refine.x
    best_c5_refine = evaluate_erdos_solution(best_h_refine)

    # Final selection between initial and refined
    if best_c5_refine < best_c5:
        final_h = best_h_refine
        final_c5 = best_c5_refine
        n_points_final = n_points_refine
    else:
        final_h = best_h
        final_c5 = best_c5
        n_points_final = n_points

    return (final_h, final_c5, n_points_final)
