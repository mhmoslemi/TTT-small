import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 points for faster initial evaluation
    max_iterations = 100  # Limit iterations to stay within budget
    learning_rate = 0.01
    tolerance = 1e-4
    
    # Start with initial h values
    h_values = initial_h_values.copy()
    h_values = h_values[:n_points]  # Ensure it fits the number of points
    
    # Ensure the integral condition is met
    total_integral = np.sum(h_values) * (2.0 / n_points)
    if abs(total_integral - 1.0) > 1e-6:
        # Normalize to meet the constraint
        h_values = h_values * (1.0 / total_integral) * (2.0 / n_points)
    
    best_c5 = evaluate_erdos_solution(h_values, n_points)
    best_h = h_values.copy()
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute the current C5 value
        current_c5 = best_c5
        
        # Try a small random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        
        # Clip to [0, 1]
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        
        # Ensure the integral condition is met
        total_integral = np.sum(perturbed_h) * (2.0 / n_points)
        if abs(total_integral - 1.0) > 1e-6:
            # Normalize to meet the constraint
            perturbed_h = perturbed_h * (1.0 / total_integral) * (2.0 / n_points)
        
        # Evaluate the new C5 value
        new_c5 = evaluate_erdos_solution(perturbed_h, n_points)
        
        # Accept the perturbation if it improves the result
        if new_c5 < current_c5 + tolerance:
            best_c5 = new_c5
            best_h = perturbed_h.copy()
        
        # Early stopping if we're close enough
        if best_c5 < 0.380876 + 1e-5:
            break
    
    # Return the best values found
    return (best_h, best_c5, n_points)
