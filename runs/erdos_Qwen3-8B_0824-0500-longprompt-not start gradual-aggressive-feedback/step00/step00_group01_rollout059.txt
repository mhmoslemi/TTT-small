```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    
    # Normalize h to satisfy the integral constraint
    h /= (np.sum(h) / (n_points / 2))
    
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Try small perturbations to improve the solution
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate a random index to perturb
        idx = np.random.randint(n_points)
        
        # Try flipping the value at idx
        h_flip = h.copy()
        h_flip[idx] = 1 - h_flip[idx]
        
        # Check if the integral constraint is satisfied
        if np.abs(np.sum(h_flip) - n_points / 2) < 1e-6:
            c5 = evaluate_erdos_solution(h_flip)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_flip.copy()
    
    # Try a few more iterations with gradient descent-like approach
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate small perturbations
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturb = h + perturb
        
        # Clip to [0, 1] and ensure the integral constraint
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
        
        c5 = evaluate_erdos_solution(h_perturb)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturb.copy()
    
    # Try a few more iterations with local search
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate a random index to perturb
        idx = np.random.randint(n_points)
        
        # Try flipping the value at idx
        h_flip = h.copy()
        h_flip[idx] = 1 - h_flip[idx]
        
        # Check if the integral constraint is satisfied
        if np.abs(np.sum(h_flip) - n_points / 2) < 1e-6:
            c5 = evaluate_erdos_solution(h_flip)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_flip.copy()
    
    # Try to find a better solution with more points
    for n_points in [50, 100, 150, 200]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create a new h with n_points samples
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h.tolist(), best_c5, len(best_h)
```
