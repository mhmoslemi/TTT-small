```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial solution
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=float)
    
    # Function to compute C5
    def compute_c5(h):
        dx = 2.0 / len(h)
        h = np.concatenate([h, h[:2]])  # Extend for correlation
        result = np.correlate(h, 1 - h[:len(h)], mode="full")
        overlap = result * dx
        return np.max(overlap)
    
    # Objective function to minimize
    def objective(x):
        nonlocal h
        h = x.copy()
        h = np.clip(h, 0, 1)
        return compute_c5(h)
    
    # Constraint: sum(h) == n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint})
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    initial_guess = h.copy()
    
    # Run optimization
    result = minimize(objective, initial_guess, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    
    # Check if we have time left to refine
    elapsed = time.time() - start_time
    if elapsed < budget_s - 5:
        # Try a local search with perturbations
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = result.x + perturb
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = (perturbed_h * n_points / 2).astype(float) / (n_points / 2)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        def perturb_objective(x):
            nonlocal h
            h = x.copy()
            h = np.clip(h, 0, 1)
            return compute_c5(h)
        
        perturb_result = minimize(perturb_objective, perturbed_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        result = perturb_result
    
    # Final check
    final_h = np.clip(result.x, 0, 1)
    final_h = (final_h * n_points / 2).astype(float) / (n_points / 2)
    final_h = np.clip(final_h, 0, 1)
    c5_bound = compute_c5(final_h)
    
    return final_h, c5_bound, n_points
```
