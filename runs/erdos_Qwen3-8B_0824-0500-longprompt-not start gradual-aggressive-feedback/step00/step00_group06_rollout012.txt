import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 100  # Starting with 100 points for faster computation
    dx = 2.0 / n_points

    # Initial guess from provided initial_h_values
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]
    initial_h = np.clip(initial_h, 0.0, 1.0)
    sum_h = np.sum(initial_h) * dx
    if abs(sum_h - 1.0) > 1e-6:
        initial_h = initial_h * (1.0 / sum_h) * dx
    initial_h = np.clip(initial_h, 0.0, 1.0)
    initial_h = initial_h * (n_points / 2.0)  # Ensure sum is n_points / 2

    # Optimization function
    def objective(h):
        nonlocal best_c5
        h = np.clip(h, 0.0, 1.0)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        h = h.reshape(-1)
        conv = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Perform optimization
    result = optimize.minimize(
        objective,
        initial_h,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        options={'ftol': 1e-6, 'maxiter': 100}
    )

    # Check if we can improve with more points
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        dx = 2.0 / n_points
        initial_h = best_h.copy()
        initial_h = initial_h[:n_points]
        initial_h = np.clip(initial_h, 0.0, 1.0)
        sum_h = np.sum(initial_h) * dx
        if abs(sum_h - 1.0) > 1e-6:
            initial_h = initial_h * (1.0 / sum_h) * dx
        initial_h = np.clip(initial_h, 0.0, 1.0)
        initial_h = initial_h * (n_points / 2.0)

        def objective_high_res(h):
            nonlocal best_c5
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            if abs(sum_h - n_points / 2) > 1e-6:
                return np.inf
            h = h.reshape(-1)
            conv = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(conv) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            return c5

        result_high_res = optimize.minimize(
            objective_high_res,
            initial_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'ftol': 1e-6, 'maxiter': 100}
        )

    # Use the best solution found
    final_h = best_h
    final_c5 = best_c5

    # If no improvement, fallback to initial solution
    if final_h is None:
        final_h = initial_h
        final_c5 = evaluate_erdos_solution(final_h)

    # Ensure final_h has n_points
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = final_h * (n_points / 2.0)
    final_h = final_h.reshape(-1)

    # Compute final C5
    final_h = np.clip(final_h, 0.0, 1.0)
    sum_h = np.sum(final_h) * dx
    if abs(sum_h - 1.0) > 1e-6:
        final_h = final_h * (1.0 / sum_h) * dx
    final_h = np.clip(final_h, 0.0, 1.0)
    final_h = final_h * (n_points / 2.0)
    conv = signal.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(conv) * dx

    # Final check
    if final_c5 > 0.380876:
        final_h = initial_h
        final_c5 = evaluate_erdos_solution(final_h)

    elapsed_time = time.time() - start_time
    return (final_h, final_c5, n_points)
