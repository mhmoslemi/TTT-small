import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import sys
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.concatenate([h[-1:], h[:-1]], axis=0)
    corr = signal.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    # A base initial construction with n=80 samples
    n = 80
    h = np.zeros(n)
    for i in range(n):
        if i < 40:
            h[i] = 1.0
        else:
            h[i] = 0.0
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = np.array([])
    n_points = 100  # Start with n=100 samples as a reasonable size for optimization
    dx = 2.0 / n_points
    
    # Initial guess
    h_initial = initial_h_values()
    h_initial = np.resize(h_initial, n_points)
    
    # Optimization bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Objective function
    def objective(x):
        h = x
        shifted = np.concatenate([h[-1:], h[:-1]], axis=0)
        corr = signal.correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * dx
        return c5_val
    
    # Optimization settings
    max_iter = 100
    tol = 1e-5
    
    # Try a few different starting points to avoid local minima
    for i in range(3):
        h_start = h_initial + np.random.uniform(-0.1, 0.1, n_points)
        h_start = np.clip(h_start, 0.0, 1.0)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Perform gradient descent optimization
        result = optimize.minimize(objective, h_start, method='L-BFGS-B', bounds=bounds, tol=tol, options={'maxiter': max_iter})
        
        if result.success:
            h_candidate = result.x
            c5_val = result.fun
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_candidate
    
    # Refine with a smaller n_points if time allows
    if time.time() - start_time < budget_s - 5:
        n_points = 150
        dx = 2.0 / n_points
        h_initial = initial_h_values()
        h_initial = np.resize(h_initial, n_points)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Try a few different starting points again
        for i in range(3):
            h_start = h_initial + np.random.uniform(-0.1, 0.1, n_points)
            h_start = np.clip(h_start, 0.0, 1.0)
            h_start = h_start * (n_points / 2) / np.sum(h_start)
            
            result = optimize.minimize(objective, h_start, method='L-BFGS-B', bounds=bounds, tol=tol, options={'maxiter': max_iter})
            
            if result.success:
                h_candidate = result.x
                c5_val = result.fun
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_candidate
    
    # Final check
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
