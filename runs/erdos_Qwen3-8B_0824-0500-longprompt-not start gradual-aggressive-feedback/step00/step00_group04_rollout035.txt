import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions and initial construction
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction as starting point
    n_points = 100  # Start with 100 points for faster evaluation
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Define function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr * dx)
    
    # Optimization loop
    for _ in range(10):  # Try multiple optimization steps
        # Convert h to a variable for optimization
        h_var = cp.Variable(len(h), bounds=(0, 1))
        # Integrate constraint: sum(h) = n_points / 2
        constraint = cp.sum(h_var) == len(h) / 2
        # Objective: minimize the maximum overlap integral
        # We use a convex optimization approach with a convex relaxation
        # We use a convex relaxation by minimizing the maximum of a convex function
        # This is a convex problem, so we can solve it with CVXPY
        # Use a small epsilon to avoid numerical issues
        epsilon = 1e-5
        # We minimize the maximum overlap integral with a small perturbation
        obj = cp.Minimize(cp.Max(cp.multiply(1 - h_var, h_var), epsilon))
        # Solve the problem
        prob = cp.Problem(obj, [constraint])
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status != 'optimal':
                break
            h_new = h_var.value
            # Check if the solution is feasible
            if np.all(h_new >= 0) and np.all(h_new <= 1) and np.isclose(np.sum(h_new), len(h) / 2):
                # Compute the C5 value
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                h = h_new
        except:
            break
        
        # Check if time budget is exhausted
        if time.time() - start_time > budget_s - 1:
            break
    
    # Try a few more points to see if we can improve the bound
    for n_points in [200, 300, 400]:
        h_new = np.zeros(n_points)
        # Spread out the h values to minimize overlap
        for i in range(n_points):
            h_new[i] = 1.0 / (n_points / 2)
        # Use the best_h as a starting point
        h_new[:len(best_h)] = best_h
        # Normalize to ensure the integral is 1
        h_new = h_new / (np.sum(h_new) / (n_points / 2))
        
        # Check feasibility
        if np.all(h_new >= 0) and np.all(h_new <= 1) and np.isclose(np.sum(h_new), n_points / 2):
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
    
    # Return the best solution found
    return (best_h, best_c5, len(best_h))
