def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    
    # Optimize with increasing n_points up to a limit
    max_points = 100  # Can be increased if time allows
    
    best_c5 = float('inf')
    best_h = h_initial
    
    for n in range(n_points, max_points + 1):
        if time.time() - start_time > budget_s - 2:
            break
        
        dx = 2.0 / n
        h = np.random.rand(n) * 2 - 1  # Initial guess
        h = np.clip(h, 0, 1)
        h = h * (n / 2)  # Scale to meet the integral constraint
        
        # Optimization loop
        for _ in range(100):  # Max iterations
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute overlaps
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Update h using gradient descent
            # Compute the derivative of C5 with respect to h[i]
            # Approximate derivative using finite differences
            h_perturb = h.copy()
            h_perturb += 1e-5
            overlap_perturb = np.correlate(h_perturb, 1 - h_perturb, mode="full")
            c5_perturb = np.max(overlap_perturb) * dx
            grad = (c5_perturb - c5) / 1e-5
            
            # Gradient descent step
            h -= 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n / 2)
    
    # Final check
    if time.time() - start_time > budget_s - 2:
        pass
    
    return (best_h, best_c5, len(best_h))
