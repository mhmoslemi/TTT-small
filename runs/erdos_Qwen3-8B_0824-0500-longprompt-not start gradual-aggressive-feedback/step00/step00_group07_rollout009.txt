import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial parameters
    n_points = 80  # Initial number of points
    dx = 2.0 / n_points
    sum_h = n_points / 2  # Ensure the integral is 1
    
    # Start with the initial construction
    h = np.array(initial_h_values)
    h = h[:n_points]
    
    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.array(h)
        h = np.pad(h, (0, 1), 'constant')  # Pad to allow full convolution
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Function to optimize h
    def optimize_h(h, n_points):
        # Convert to optimization variables
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [
            cp.sum(h_var) == sum_h,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Objective is to minimize the maximum overlap integral
        # We use a convex relaxation where we minimize the maximum overlap
        # We use a convex relaxation by minimizing the maximum of the overlap integrals
        
        # Since maximum is non-convex, we use a convex relaxation by minimizing the maximum via a single constraint
        # We can approximate the maximum by minimizing a variable that is an upper bound of the overlap integrals
        
        # Evaluate all possible shifts (up to n_points)
        max_overlap = 0.0
        for k in range(1, n_points):
            overlap = np.correlate(h_var.value, 1 - h_var.value, mode='full')[k] * dx
            max_overlap = max(max_overlap, overlap)
        
        # Use an iterative approach with a convex optimization
        # We can use a binary search to find the minimal C5
        
        # Set up the problem
        obj = cp.Minimize(cp.Max(cp.multiply(cp.ones(n_points - 1), np.correlate(h_var, 1 - h_var, mode='full')[1:] * dx)))
        
        # Solve the problem
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        # Return the optimized h
        return h_var.value
    
    # Function to refine the solution
    def refine_h(h, n_points):
        # Start with a refined number of points
        refined_n_points = 100
        refined_dx = 2.0 / refined_n_points
        refined_sum_h = refined_n_points / 2
        
        # Interpolate the current h to the refined grid
        refined_h = np.zeros(refined_n_points)
        for i in range(refined_n_points):
            x = i * refined_dx
            # Find the closest points in the original h
            idx = np.argmin(np.abs(np.linspace(0, 2, n_points) - x))
            refined_h[i] = h[idx]
        
        # Optimize the refined h
        refined_h_opt = optimize_h(refined_h, refined_n_points)
        return refined_h_opt
    
    # Function to evaluate the current solution
    def evaluate_solution(h):
        # Evaluate the current C5
        c5 = evaluate_c5(h)
        return c5
    
    # Main optimization loop
    best_h = h[:]
    best_c5 = evaluate_c5(best_h)
    
    # Try optimizing with the initial number of points
    optimized_h = optimize_h(h, n_points)
    current_c5 = evaluate_c5(optimized_h)
    if current_c5 < best_c5:
        best_h = optimized_h[:]
        best_c5 = current_c5
    
    # Refine the solution
    refined_h = refine_h(best_h, n_points)
    current_c5 = evaluate_c5(refined_h)
    if current_c5 < best_c5:
        best_h = refined_h[:]
        best_c5 = current_c5
    
    # Check if we have enough time to optimize further
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Try optimizing with a slightly higher number of points
        refined_n_points = 120
        refined_dx = 2.0 / refined_n_points
        refined_sum_h = refined_n_points / 2
        
        # Interpolate the best solution to the refined grid
        refined_h = np.zeros(refined_n_points)
        for i in range(refined_n_points):
            x = i * refined_dx
            # Find the closest points in the best_h
            idx = np.argmin(np.abs(np.linspace(0, 2, len(best_h)) - x))
            refined_h[i] = best_h[idx]
        
        # Optimize the refined h
        refined_h_opt = optimize_h(refined_h, refined_n_points)
        current_c5 = evaluate_c5(refined_h_opt)
        if current_c5 < best_c5:
            best_h = refined_h_opt[:]
            best_c5 = current_c5
    
    # Return the best solution found
    return (best_h, best_c5, len(best_h))
