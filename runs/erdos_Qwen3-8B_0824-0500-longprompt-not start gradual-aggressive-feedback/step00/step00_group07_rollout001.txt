import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values, dx: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * dx
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Choose a starting point
    n_points = 100  # Start with 100 samples
    h = np.zeros(n_points)
    h[::2] = 1  # Initial guess: alternating 1 and 0
    h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy ∫h(x)dx = 1
    
    # Evaluate initial C5
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Optimization loop
    for _ in range(50):  # 50 iterations
        if time.time() - start_time > budget_s:
            break
        
        # Try random perturbation
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h + perturb, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Evaluate C5
        c5 = evaluate_erdos_solution(h_perturbed, dx)
        if c5 < best_c5:
            best_h = h_perturbed.copy()
            best_c5 = c5
        
        # Try gradient descent-like update
        # Compute overlap integral for all shifts
        overlap = np.correlate(h, 1 - h, mode="full")
        overlap = overlap * dx
        max_overlap = np.max(overlap)
        # Try to reduce the maximum overlap by adjusting h
        # Use a simple gradient descent approach
        grad = np.zeros(n_points)
        for shift in range(len(overlap)):
            if shift < n_points:
                # For each position, adjust h to reduce its contribution to max_overlap
                # This is heuristic, for demonstration purposes
                grad += (h * (1 - h) * (overlap[shift] - max_overlap)) * (1 - 2 * h)
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate and update
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Try a different starting point
        if np.random.rand() < 0.2:
            n_points = np.random.randint(100, 200)
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            dx = 2.0 / n_points
            c5 = evaluate_erdos_solution(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
    
    # Final evaluation
    dx = 2.0 / n_points
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return best_h, final_c5, n_points
