import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from current construction
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values)
    
    # Optimization parameters
    max_iter = 100
    tol = 1e-5
    
    start_time = time.time()
    
    def objective(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        total = np.sum(h) * (2.0 / n_points)
        if abs(total - 1.0) > 1e-6:
            return np.inf
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Set up bounds for optimization
    bounds = Bounds(0, 1)
    
    # Use Sequential Least Squares Quadratic Programming (SLSQP)
    result = optimize.minimize(
        objective,
        h_values,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': max_iter, 'ftol': tol, 'disp': False}
    )
    
    h_values = result.x
    c5_bound = objective(h_values)
    
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Further refinement if time allows
        for _ in range(20):
            result = optimize.minimize(
                objective,
                h_values,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 10, 'ftol': tol, 'disp': False}
            )
            h_values = result.x
            c5_bound = objective(h_values)
            if elapsed_time + (time.time() - start_time) > budget_s:
                break
    
    return (h_values, c5_bound, n_points)
