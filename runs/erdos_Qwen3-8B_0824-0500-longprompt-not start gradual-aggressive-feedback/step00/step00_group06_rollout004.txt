def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess from the pre-defined starting point
    h_initial = initial_h_values.copy()
    n_points = len(h_initial)
    
    # Function to compute the overlap integral (C5)
    def compute_c5(h):
        dx = 2.0 / n_points
        shifted_h = np.zeros_like(h)
        for k in range(1, n_points):
            shifted_h[k:] = h[:n_points - k]
        overlap = np.correlate(h, 1 - shifted_h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Function to evaluate the objective (C5) and constraints for optimization
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        return c5
    
    # Optimization using local search with simulated annealing
    def optimize_h(h_initial, budget_s):
        best_h = h_initial.copy()
        best_c5 = compute_c5(best_h)
        time_used = 0
        
        # Simulated Annealing Parameters
        temp = 100.0
        cooling_rate = 0.995
        steps = 1000
        while time_used < budget_s and temp > 1e-3:
            for _ in range(steps):
                if time_used >= budget_s:
                    break
                # Perturb the current solution
                perturbation = np.random.normal(0, 0.05, size=n_points)
                h_perturbed = best_h + perturbation
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
                
                # Compute the new cost
                new_c5 = compute_c5(h_perturbed)
                # Acceptance probability
                if np.random.rand() < np.exp((best_c5 - new_c5) / temp):
                    best_h = h_perturbed
                    best_c5 = new_c5
                time_used = time.time() - start_time
            temp *= cooling_rate
        
        return best_h, best_c5
    
    # Additional refinement using gradient descent
    def gradient_descent(h_initial, budget_s):
        h = h_initial.copy()

        h = h * (n_points / 2) / np.sum(h)
        for _ in range(1000):

                break

            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_c5(h_plus)
                
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                c5_minus = compute_c5(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            
            # Update step
            step_size = 0.01
            h = h - step_size * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        return h, compute_c5(h)
    
    # Optimize using simulated annealing
    best_h, best_c5 = optimize_h(h_initial, budget_s)
    best_h, best_c5 = gradient_descent(best_h, budget_s)
    
    # Final refinement
    for _ in range(10):

            break
        perturbation = np.random.normal(0, 0.01, size=n_points)
        h_candidate = best_h + perturbation
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
    
    # Ensure the solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return best_h, best_c5, n_points
