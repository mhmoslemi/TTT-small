import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    # Initial solution from the current construction
    n_points = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    
    # Define the function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        h = np.concatenate([h, h[0:1]])  # Extend to handle wrap-around
        overlap = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * (2.0 / len(h))  # dx = 2 / n_points
    
    # Define the optimization problem
    def optimize_h(n_points, initial_h=None):
        h = cp.Variable(n_points, nonneg=True)
        # Constraint: sum(h) = n_points / 2
        constraints = [cp.sum(h) == n_points / 2]
        # Objective: minimize the maximum overlap
        # Since we can't directly minimize the max, we use a minimax approach with a helper variable
        t = cp.Variable()
        constraints += [t >= evaluate_overlap(h)]
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        start_time = time.time()
        result = prob.solve(solver=cp.SCS, max_iter=1000, time_limit=budget_s - (time.time() - start_time))
        if result is None:
            return initial_h, float('inf')
        return h.value, t.value
    
    # Start with the initial solution
    h_values = h_initial.copy()
    best_c5 = evaluate_overlap(h_values)
    best_h = h_values
    
    # Try several points around the initial solution
    for n in [n_points - 10, n_points, n_points + 10]:
        if n < 50:
            continue
        try:
            h_opt, c5 = optimize_h(n, initial_h=initial_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        except:
            continue
    
    # Return the best solution found
    return (best_h, best_c5, len(best_h))
