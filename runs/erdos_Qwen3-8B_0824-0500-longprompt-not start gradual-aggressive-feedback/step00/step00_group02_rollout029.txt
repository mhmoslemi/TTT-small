import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 200  # Start with 200 samples
    max_iterations = 50  # Max number of iterations
    time_budget = budget_s  # Time budget in seconds
    
    start_time = time.time()
    
    # Create initial guess from initial_h_values scaled to n_points
    initial_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    best_h = initial_guess.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    for iteration in range(max_iterations):
        # Evaluate current best
        current_c5 = evaluate_erdos_solution(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        
        # Check if time budget is exceeded
        elapsed_time = time.time() - start_time
        if elapsed_time >= time_budget:
            break
        
        # Perturb the solution slightly
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        
        # Rescale to maintain integral constraint
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        
        # Evaluate the perturbed solution
        perturbed_c5 = evaluate_erdos_solution(perturbed_h)
        
        # Update best solution
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
