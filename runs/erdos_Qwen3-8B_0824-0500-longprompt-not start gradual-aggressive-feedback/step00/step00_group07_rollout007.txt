import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros_like(h)
    for k in range(1, n_points):
        shifted_h = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap

def initial_h_values():
    # A simple initial construction
    n_points = 80
    h = np.zeros(n_points)
    h[:20] = 1.0
    h[40:] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values()
    n_points = len(best_h)
    
    # Set up initial parameters
    n_points = 80
    dx = 2.0 / n_points
    h_initial = np.zeros(n_points)
    h_initial[:20] = 1.0
    h_initial[40:] = 1.0
    h_initial = h_initial.tolist()
    
    # Define the objective function
    def objective(h_values):
        nonlocal best_c5
        h = np.array(h_values)
        max_overlap = 0.0
        for k in range(1, n_points):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        c5 = max_overlap
        if c5 < best_c5:
            best_c5 = c5
        return c5
    
    # Define bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Set up constraints
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint}]
    
    # Set up optimization parameters
    method = 'L-BFGS-B'
    options = {'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    
    # Run optimization
    result = minimize(
        objective,
        h_initial,
        method=method,
        bounds=bounds,
        constraints=cons,
        options=options
    )
    
    # Extract the best solution
    best_h = result.x.tolist()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Check if we have time left to refine
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.1:
        # Perform a local refinement
        refined_h = best_h.copy()
        for i in range(n_points):
            refined_h[i] = np.clip(refined_h[i] + np.random.normal(0, 0.05), 0, 1)
        refined_c5 = evaluate_erdos_solution(refined_h)
        if refined_c5 < best_c5:
            best_h = refined_h
            best_c5 = refined_c5
    
    return best_h, best_c5, n_points
