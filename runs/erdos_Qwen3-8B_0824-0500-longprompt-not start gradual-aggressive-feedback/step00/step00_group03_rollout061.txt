```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Initial point
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    h = h / np.sum(h) * (n_points / 2)
    
    # Parameters
    max_iterations = 100
    tolerance = 1e-6
    max_time_per_iter = 0.5  # seconds per iteration
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    for i in range(max_iterations):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
        
        # Create a small perturbation
        perturbation = np.random.normal(0, 0.05, n_points)
        h_candidate = h + perturbation
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate / np.sum(h_candidate) * (n_points / 2)
        
        # Evaluate the candidate
        c5_candidate = evaluate_erdos_solution(h_candidate)
        
        # Update best solution
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
        
        # Update h
        h = h_candidate.copy()
        
        # Check convergence
        if abs(best_c5 - evaluate_erdos_solution(h)) < tolerance:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h.tolist(), final_c5, len(best_h)
```
