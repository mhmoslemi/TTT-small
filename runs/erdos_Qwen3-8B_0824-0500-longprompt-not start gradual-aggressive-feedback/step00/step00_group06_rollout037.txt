import numpy as np

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    n_points = 95
    h = np.array(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / sum_h
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = h.astype(float)
        n = len(h)
        dx = 2.0 / n
        h_extended = np.zeros(2 * n - 1)
        h_extended[:n] = h
        h_shifted = np.zeros(2 * n - 1)
        h_shifted[n-1:] = h
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Function to compute the overlap integral for a given h
    def objective(h_vec):
        h = h_vec.astype(float)
        n = len(h)
        dx = 2.0 / n
        h_extended = np.zeros(2 * n - 1)
        h_extended[:n] = h
        h_shifted = np.zeros(2 * n - 1)
        h_shifted[n-1:] = h
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Function to compute the constraints for optimization
    def constraints(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Optimization function to minimize the maximum overlap
    def optimize_h(h_initial):
        h = h_initial.copy()
        h = h.astype(float)
        n = len(h)
        dx = 2.0 / n
        h_extended = np.zeros(2 * n - 1)
        h_extended[:n] = h
        h_shifted = np.zeros(2 * n - 1)
        h_shifted[n-1:] = h
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    def local_optimize(h_initial, max_iter=100, step_size=0.01):
        h = h_initial.copy()
        h = h.astype(float)
        n = len(h)
        dx = 2.0 / n
        current_c5 = compute_c5(h)
        for _ in range(max_iter):
            h_extended = np.zeros(2 * n - 1)
            h_extended[:n] = h
            h_shifted = np.zeros(2 * n - 1)
            h_shifted[n-1:] = h
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap * dx)
            
            # Compute gradient (approximate)
            grad = np.zeros(n)
            for i in range(n):
                h_plus = h.copy()
                h_plus[i] += step_size
                h_plus = np.clip(h_plus, 0, 1)
                c5_plus = compute_c5(h_plus)
                grad[i] = (c5_plus - c5) / step_size
                
                h_minus = h.copy()
                h_minus[i] -= step_size
                h_minus = np.clip(h_minus, 0, 1)
                c5_minus = compute_c5(h_minus)
                grad[i] = (c5 - c5_minus) / step_size
            
            # Update h
            h -= grad * 0.1
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            if np.abs(compute_c5(h) - current_c5) < 1e-6:
                break

    if c5_small < c5:

    # Check time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points_small)
    
    # Try to further optimize with convex optimization
    n_points_cvx = 100
    h_cvx = np.random.rand(n_points_cvx)

    # Solve the optimization problem
    try:
        problem = cp.Problem(obj, constraints)
        problem.solve(solver=cp.CVXOPT)
        h_cvx_opt = h_var.value
        c5_cvx = compute_c5(h_cvx_opt)
        if c5_cvx < best_c5:
            best_h = h_cvx_opt
            best_c5 = c5_cvx
    except:
        pass
    
    # Return the best solution found
    return (best_h, best_c5, n_points_cvx)
