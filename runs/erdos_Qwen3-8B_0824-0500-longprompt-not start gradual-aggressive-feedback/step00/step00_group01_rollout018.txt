import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    sum_h = np.sum(h) * dx
    if abs(sum_h - 1.0) > 1e-10:
        h = h * (1.0 / sum_h) * 2.0  # Normalize to ensure integral is 1

    def compute_c5(h_values):
        h = np.array(h_values, dtype=np.float64)
        dx = 2.0 / len(h)
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Extract overlapping parts
        offset = len(h) - 1
        corr = corr[offset:-offset]
        # Multiply by dx
        corr *= dx
        # Return maximum value
        return np.max(corr)

    # Function to minimize C5
    def obj_func(h_values):
        c5 = compute_c5(h_values)
        return c5

    # Function to optimize with constraints
    def optimize_h(initial_h, n_points, budget_s):
        h = np.copy(initial_h)
        dx = 2.0 / n_points
        sum_h = np.sum(h) * dx
        if abs(sum_h - 1.0) > 1e-10:
            h = h * (1.0 / sum_h) * 2.0  # Normalize to ensure integral is 1

        # Use a simple gradient descent with a penalty for constraint violation
        max_iter = int(budget_s / 0.1)  # Adjust based on time budget
        learning_rate = 0.01
        for _ in range(max_iter):
            # Compute current C5
            current_c5 = compute_c5(h)
            # Compute gradient (approximate)
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                c5_plus = compute_c5(h_plus)
                grad[i] = (c5_plus - current_c5) / 1e-5
            # Adjust h to minimize C5
            h -= learning_rate * grad
            # Enforce constraints
            h = np.clip(h, 0, 1)
            sum_h_current = np.sum(h) * dx
            if abs(sum_h_current - 1.0) > 1e-10:
                h = h * (1.0 / sum_h_current) * 2.0

        return h

    # Run optimization
    h_optimized = optimize_h(h, n_points, budget_s)

    # Compute final C5
    final_c5 = compute_c5(h_optimized)

    # Return best solution
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        print(f"Time budget exceeded, returning initial solution")
        return (initial_h_values, compute_c5(initial_h_values), len(initial_h_values))

    return (h_optimized.tolist(), final_c5, len(h_optimized))
