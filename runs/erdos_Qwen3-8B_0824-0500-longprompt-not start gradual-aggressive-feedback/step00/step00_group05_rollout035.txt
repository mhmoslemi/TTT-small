```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial guess from the provided initial construction
    n_points_initial = len(initial_h_values)
    h_initial = initial_h_values.copy()
    
    # Function to compute the overlap integral for a given h
    def compute_c5(h):
        dx = 2.0 / len(h)
        # Compute the correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        # The maximum overlap integral corresponds to the maximum value in the correlation
        c5 = np.max(corr) * dx
        return c5
    
    # Function to evaluate the objective for optimization
    def objective(h):
        return compute_c5(h)
    
    # Function to enforce the constraint sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - len(h) / 2
    
    # Optimization settings
    n_points = n_points_initial  # Start with the same number of points as the initial guess
    max_iterations = 100  # Limit to prevent infinite loops
    time_limit = budget_s  # Time budget in seconds
    
    start_time = time.time()
    
    best_c5 = np.inf
    best_h = h_initial.copy()
    
    # Try different numbers of points, starting from lower values to speed up
    for n_points in [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]:
        if time.time() - start_time > time_limit:
            break
        
        # Generate initial guess with uniform spacing
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = h_initial[i * n_points_initial // n_points]
        
        # Use scipy's optimization with bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint_sum}]
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-6, 'maxiter': max_iterations}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
    
    # Final evaluation of the best solution
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
```
