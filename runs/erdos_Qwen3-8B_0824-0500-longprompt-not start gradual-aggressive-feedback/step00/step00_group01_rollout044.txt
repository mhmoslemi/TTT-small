def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute 1 - h
    one_minus_h = 1.0 - h
    # Compute the cross-correlation (full mode)
    corr = correlate(h, one_minus_h, mode='full')
    # The max overlap integral is the max value in the cross-correlation multiplied by dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def initial_h_values(n_points):
    # Initial guess based on a simple construction
    # You can replace this with any initial guess you want
    h = np.zeros(n_points)
    # For example, a piecewise constant function that is 1 on [0, 0.5] and 0 on [0.5, 2]
    # This is just an example to get started
    for i in range(n_points):
        x = i * 2.0 / n_points
        if x <= 1.0:
            h[i] = 1.0
        else:
            h[i] = 0.0
    # Adjust to satisfy the integral constraint
    total = np.sum(h) * 2.0 / n_points
    if total != 1.0:
        # Scale h to satisfy the integral constraint
        h = h * 1.0 / total
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    n_points = 100  # Using 100 points for a reasonable balance between speed and accuracy
    dx = 2.0 / n_points
    # Initial guess with a simple construction
    h_initial = initial_h_values(n_points)
    # Convert to numpy array for easier manipulation
    h = np.array(h_initial, dtype=np.float64)

    # Define the objective function (we want to minimize the maximum overlap)
    def objective(h_flat):
        h = h_flat.reshape((n_points,))
        # Compute the overlap integral
        corr = correlate(h, 1.0 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define bounds for h values

    result = minimize(
