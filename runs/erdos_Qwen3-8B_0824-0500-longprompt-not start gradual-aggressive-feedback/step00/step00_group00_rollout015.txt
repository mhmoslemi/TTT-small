        # We use a convex relaxation by minimizing an upper bound
        # Convert the max to a convex form using a new variable
        corr = cp.Variable((2 * n_points - 1, 1))
        constraints = []
        for i in range(2 * n_points - 1):
            # Build the correlation expression
            corr_expr = 0
            for j in range(n_points):
                corr_expr += h_opt[j] * (1 - h_opt[(i - j) % n_points])
            constraints.append(corr[i] >= corr_expr)
        # Objective: minimize the maximum correlation
        obj = cp.Minimize(cp.Max(corr))
        problem = cp.Problem(obj, constraints)

            problem.solve(solver=cp.ECOS, eps=1e-5, verbose=False)
