```python
import numpy as np
import scipy.optimize as optimize
import time
import copy
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Placeholder for initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for efficiency
    
    # Initialize with the provided initial values
    initial_h = initial_h_values[:n_points]
    initial_h = np.clip(initial_h, 0, 1)
    sum_h = np.sum(initial_h) * (2.0 / n_points)
    if abs(sum_h - 1.0) > 1e-6:
        initial_h = initial_h * (1.0 / sum_h) * (2.0 / n_points)  # Normalize
    
    # Function to compute C5
    def compute_c5(h_val):
        h_val = np.clip(h_val, 0, 1)
        if len(h_val) != n_points:
            h_val = np.pad(h_val, (0, n_points - len(h_val)), 'constant')
        h_val = h_val[:n_points]
        c5 = evaluate_erdos_solution(h_val)
        return c5
    
    # Function to optimize the h_values
    def optimize_h(h_initial):
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        objective = cp.Minimize(cp.max(cp.correlate(h.value, 1 - h.value, mode="full") * (2.0 / n_points)))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=1000, verbose=False)
            if prob.status == 'OPTIMAL':
                return h.value
        except:
            pass
        return h_initial
    
    # Initial optimization
    h_opt = optimize_h(initial_h)
    c5 = compute_c5(h_opt)
    best_c5 = c5
    best_h = h_opt
    
    # Try increasing n_points to 200 if time allows
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        # Initialize with a better starting point
        h_initial = np.zeros(n_points)
        h_initial[:n_points // 2] = 1.0
        h_initial[n_points // 2:] = 0.0
        h_initial = np.random.permutation(h_initial)
        h_initial = h_initial[:n_points]
        sum_h = np.sum(h_initial) * (2.0 / n_points)
        if abs(sum_h - 1.0) > 1e-6:
            h_initial = h_initial * (1.0 / sum_h) * (2.0 / n_points)
        
        # Optimize again
        h_opt = optimize_h(h_initial)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
    
    # Final result
    if time.time() - start_time > budget_s:
        return (best_h, best_c5, len(best_h))
    
    return (best_h, best_c5, len(best_h))
```
