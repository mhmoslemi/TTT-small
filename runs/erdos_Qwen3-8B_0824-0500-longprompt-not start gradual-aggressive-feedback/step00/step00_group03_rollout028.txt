import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.zeros(2 * n_points)
    h_shifted[:n_points] = h
    conv = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(conv) * dx
    return c5

def initial_h_values():
    # Placeholder for initial solution (n=70 samples)
    return np.array([0.0 if i % 3 == 0 else 1.0 if i % 3 == 1 else 0.5 for i in range(70)])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial sample count
    n_points = 100  # Start with 100 samples for faster convergence
    
    # Initial guess
    h_initial = initial_h_values()
    h_initial = np.resize(h_initial, n_points)
    
    # Optimization
    def objective(h_values):
        # Convert to numpy array
        h = np.array(h_values)
        # Ensure sum is correct
        sum_h = np.sum(h)
        # Compute the overlap integral
        h_shifted = np.zeros(2 * n_points)
        h_shifted[:n_points] = h
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * (2.0 / n_points)
        return c5
    
    # Optimization with constraints
    def optimize_h(n_points, h_initial):
        h = cp.Variable(n_points)
        # Constraint: sum(h) == n_points / 2
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        # Objective: minimize the maximum overlap
        # For simplicity, we convert to a single scalar optimization with constraints
        # This is an approximate approach since the max is non-convex
        
        # Discretize the overlap integral and minimize its maximum value
        # Here we use a numerical optimization approach
        def f(x):
            h_vals = x
            h_shifted = np.zeros(2 * n_points)
            h_shifted[:n_points] = h_vals
            conv = np.correlate(h_vals, 1 - h_vals, mode="full")
            return np.max(conv) * (2.0 / n_points)
        
        # Use scipy optimization to minimize the C5 value
        result = optimize.minimize(
            f,

            bounds=[(0, 1) for _ in range(n_points)],
            options={"maxiter": 100, "ftol": 1e-5}
        )

    best_c5 = float('inf')

        h_opt = optimize_h(n_points, h_initial)
        c5 = evaluate_erdos_solution(h_opt)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
