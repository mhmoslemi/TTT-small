import scipy.optimize as optimize
import time


def run(seed=42, budget_s=60, **kwargs):

    n_points = len(initial_h_values)
    dx = 2.0 / n_points

    def evaluate_c5(h_vals):
        h_vals = np.clip(h_vals, 0.0, 1.0)
        h_vals = h_vals.reshape(-1, 1)
        one_minus_h = 1.0 - h_vals
        # Compute full convolution
        conv = np.convolve(h_vals.flatten(), one_minus_h.flatten(), mode="full")
        # Multiply by dx and find max
        c5 = np.max(conv * dx)
        return c5

    def objective(h_vals):
        return evaluate_c5(h_vals)

    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2

    # Use Scipy's optimization with bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)}]

    # Initial guess
    initial_guess = h.copy()

    start_time = time.time()
    result = optimize.minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        tol=1e-6,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    elapsed = time.time() - start_time

    if elapsed >= budget_s:
        # If time is up, return the best found solution
        return (h, evaluate_c5(h), n_points)
    else:

        best_h = result.x
        best_c5 = result.fun
