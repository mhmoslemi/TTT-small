    best_h = h.copy()

    # Try refining with a smaller number of points starting from the initial one
    for n_points in [80, 100, 120, 140, 160, 180, 200]:
        if n_points == 0:
            continue
        
        # Generate a random sample of n_points from the initial h values
        h = np.random.choice(initial_h_values, size=n_points, replace=True)
        h = h / np.sum(h) * (n_points / 2)  # Normalize to satisfy the integral constraint
        
        # Convert to a continuous function for optimization
        h = h.astype(float)
        
        # Evaluate initial C5
        c5 = evaluate_erdos_solution(h)
        
        # Local search
        for _ in range(100):  # Perform local search for 100 iterations
            # Randomly perturb one point
            idx = np.random.randint(n_points)
            h[idx] = np.random.uniform(0, 1)

            if c5 < best_c5:

        if best_c5 < current_c5:
            current_c5 = best_c5
            best_h = best_h.copy()
    
    # Evaluate the best solution found
    final_c5 = evaluate_erdos_solution(best_h)
