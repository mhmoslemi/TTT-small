```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 samples
    dx = 2.0 / n_points
    max_iterations = 100
    cooling_factor = 0.995
    n_iterations = 0
    
    # Initial solution
    h_values = initial_h_values.copy()
    h_values = h_values[:n_points]
    h_values = np.array(h_values) * (n_points / 2) / np.sum(h_values)
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values / np.max(h_values)
    
    best_c5 = evaluate_erdos_solution(h_values, dx)
    best_h = copy.copy(h_values)
    
    # Optimization loop
    while n_iterations < max_iterations and time.time() - start_time < budget_s:
        n_iterations += 1
        
        # Perturb the solution
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_values = h_values + perturbation
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values / np.max(h_values)
        
        # Normalize to satisfy the integral constraint
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        # Compute the C5 value
        current_c5 = evaluate_erdos_solution(h_values, dx)
        
        # Accept the new solution if it's better or with some probability
        if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / (n_iterations * 0.1)):
            best_c5 = current_c5
            best_h = copy.copy(h_values)
        
        # Adaptively adjust the number of points if needed
        if n_iterations % 10 == 0:
            if best_c5 < 0.380876:
                n_points = min(200, n_points + 10)
            else:
                n_points = max(50, n_points - 10)
            dx = 2.0 / n_points
            h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
            h_values = h_values * (n_points / 2) / np.sum(h_values)
            h_values = np.clip(h_values, 0, 1)
            h_values = h_values / np.max(h_values)
    
    return best_h, best_c5, n_points
```
