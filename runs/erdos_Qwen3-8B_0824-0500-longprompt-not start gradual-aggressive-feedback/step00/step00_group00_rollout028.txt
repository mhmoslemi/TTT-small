    for n_points in [n_points_initial, 100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 2:

        if n_points % 2 != 0:
            n_points += 1
        
        # Create the grid
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=True)
        
        # Start with initial h_values, but scale to fit the new n_points
        if n_points_initial != n_points:
            h_values = np.interp(x, np.linspace(0, 2, n_points_initial), initial_h_values)
        
        # Ensure the integral of h is 1
        total = np.sum(h_values * dx)
        h_values = h_values * (1 / total) * (n_points / 2)
        
        # Define the objective function
        def objective(h):
            h = np.clip(h, 0, 1)
            # Compute the overlap integral
            overlap = np.correlate(h, 1 - h, mode="full") * dx
            c5 = np.max(overlap)
            return c5
        
        # Define bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Perform optimization
        result = opt.minimize(
            objective,
            h_values,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-5}
        )
        
        current_h = result.x
        c5 = objective(current_h)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
            best_n_points = n_points
        
        # If we have a very good solution, break early
        if best_c5 < 0.3808:
            break
