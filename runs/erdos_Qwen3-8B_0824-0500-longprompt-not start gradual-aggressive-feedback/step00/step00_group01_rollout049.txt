```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h) * dx
    if abs(sum_h - 1.0) > 1e-6:
        h = h / sum_h * 1.0
    h = np.clip(h, 0.0, 1.0)
    
    best_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    
    # Try different n_points for faster optimization
    for n_points in [100, 200, 300, 400, 500, 600, 800, 1000]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Initialize h with a random construction
        h = np.random.uniform(0.0, 1.0, n_points)
        sum_h = np.sum(h)
        h = h / sum_h * (n_points / 2.0)
        h = np.clip(h, 0.0, 1.0)
        
        # Optimization
        def objective(x):
            h = x.reshape(n_points)
            dx = 2.0 / n_points
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        def constraint(x):
            h = x.reshape(n_points)
            return np.sum(h) - (n_points / 2.0)
        
        x = cp.Variable(n_points)
        obj = cp.Minimize(objective(x))
        cons = [constraint(x)]
        prob = cp.Problem(obj, cons)
        try:
            result = prob.solve(solver=cp.SCS, eps=1e-4, max_iters=1000)
            if result is not None:
                h_opt = x.value
                h_opt = np.clip(h_opt, 0.0, 1.0)
                c5 = objective(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
        except:
            pass
        
        # Local search
        for _ in range(10):
            if time.time() - start_time > budget_s - 1:
                break
            h = best_h.copy()
            perturb = np.random.uniform(-0.1, 0.1, n_points)
            h += perturb
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            h = h / sum_h * (n_points / 2.0)
            c5 = evaluate_erdos_solution(h, 2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Return best solution
    dx = 2.0 / len(best_h)
    return best_h, best_c5, len(best_h)
```
