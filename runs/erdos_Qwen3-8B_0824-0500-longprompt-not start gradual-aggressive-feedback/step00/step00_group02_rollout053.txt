import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal

def run(seed=42, budget_s=60, **kwargs):
    # Set the random seed
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()

    n_points = 45
    h = np.array(initial_h_values, dtype=np.float64)

    # Function to compute C5
    def compute_c5(h):
        h = np.array(h, dtype=np.float64)
        dx = 2.0 / len(h)
        # Compute the correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        # Compute max integral
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Function to optimize h
    def optimize_h(h_initial, n_points, max_iter=100, learning_rate=0.01, tol=1e-5):
        h = np.copy(h_initial)
        best_c5 = compute_c5(h)
        best_h = h.copy()
        best_n_points = n_points
        
        for _ in range(max_iter):
            # Compute current C5
            current_c5 = compute_c5(h)
            
            # Compute gradient of C5 w.r.t. h
            # Since this is a numerical optimization, we use finite differences
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                c5_plus = compute_c5(h_plus)
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                c5_minus = compute_c5(h_minus)
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            
            # Update h using gradient descent
            h -= learning_rate * grad
            
            # Ensure h is within [0, 1]
            h = np.clip(h, 0, 1)
            
            # Ensure integral is 1
            h = h * (n_points / 2) / np.sum(h)
            
            # Check for convergence
            if np.abs(current_c5 - compute_c5(h)) < tol:
                break
            
            # Update best solution
            if compute_c5(h) < best_c5:
                best_c5 = compute_c5(h)
                best_h = h.copy()
                best_n_points = n_points
        
        return best_h, best_c5, best_n_points
    
    # Try different numbers of points to find the best solution
    best_c5 = float('inf')
    best_h = np.array([])
    best_n_points = 0

    h_opt, c5_opt, n_points_opt = optimize_h(h, n_points)

        best_c5 = c5_opt
        best_h = h_opt
        best_n_points = n_points_opt
    
    # Try with more points
    for n_points in [60, 75, 90, 105, 120]:
        # Create initial guess based on the best solution so far
        h_initial = np.linspace(0, 2, n_points)

            best_c5 = c5_opt
            best_h = h_opt
            best_n_points = n_points

    elapsed_time = time.time() - start_time

            best_h = h_opt
