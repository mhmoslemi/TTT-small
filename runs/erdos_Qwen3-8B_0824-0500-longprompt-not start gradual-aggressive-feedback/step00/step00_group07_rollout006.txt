import numpy as np
import scipy.optimize as optimize
import time
import random
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 80  # Starting with 80 points
    dx = 2.0 / n_points
    h = initial_h_values.copy() / 2  # Normalize sum to n_points / 2
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h_extended = np.concatenate([h, np.zeros_like(h)])
        result = np.correlate(h, 1 - h, mode="full")
        return np.max(result) * dx
    
    # Function to optimize h with constraints
    def optimize_h(h):
        n = len(h)
        dx = 2.0 / n
        h = np.array(h)
        h = np.clip(h, 0, 1)
        
        # Objective: minimize maximum overlap
        def objective(h):
            h = np.array(h)
            h = np.clip(h, 0, 1)
            h_extended = np.concatenate([h, np.zeros_like(h)])
            result = np.correlate(h, 1 - h, mode="full")
            return np.max(result) * dx
        
        # We use a local optimization with constraints
        # Since we cannot use global optimization in 60 seconds, we use a gradient-based approach
        # with constraints on the sum of h
        h = h.astype(float)
        bounds = [(0, 1) for _ in range(n)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2}
        
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints=cons,
            method='L-BFGS-B',
            options={'maxiter': 1000, 'ftol': 1e-8}
        )
        return result.x
    
    # Initial evaluation
    h = optimize_h(h)
    best_h = h
    best_c5 = evaluate_overlap(h)
    
    # Try increasing n_points to potentially improve the bound
    for n in [100, 120, 140]:
        if time.time() - start_time > budget_s - 5:
            break
        dx = 2.0 / n
        h = np.random.rand(n)
        h = h / np.sum(h) * (n / 2)
        h = np.clip(h, 0, 1)
        h = optimize_h(h)
        c5 = evaluate_overlap(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # Final check with the initial solution
    if time.time() - start_time < budget_s - 5:
        h_initial = initial_h_values.copy() / 2
        h_initial = np.clip(h_initial, 0, 1)
        c5_initial = evaluate_overlap(h_initial)
        if c5_initial < best_c5:
            best_h = h_initial
            best_c5 = c5_initial
    
    # Return the result
    return (best_h, best_c5, len(best_h))
