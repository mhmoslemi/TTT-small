def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 100  # Choose a reasonable number of points
    h = np.zeros(n_points)
    
    # Start with the initial solution
    h = np.copy(initial_h_values[:n_points])
    h = np.clip(h, 0.0, 1.0)
    
    # Ensure the integral constraint
    total_integral = np.sum(h) * (2.0 / n_points)
    scale_factor = 1.0 / total_integral
    h = h * scale_factor
    
    # Optimization: Use a gradient-based approach
    def objective(h):
        # Compute the max overlap
        h_correlate = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(h_correlate) * (2.0 / n_points)
        return c5
    
    # Define bounds for h
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use scipy optimization
    result = optimize.minimize(
        objective,
        h,
        method="L-BFGS-B",
        bounds=bounds,
        options={'disp': False, 'ftol': 1e-9, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Refine further with a few iterations of local search
    for _ in range(5):
        # Add random perturbations to explore the space
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        perturb = np.clip(perturb, -0.5, 0.5)
        candidate_h = best_h + perturb
        candidate_h = np.clip(candidate_h, 0.0, 1.0)
        # Ensure integral constraint
        total_integral = np.sum(candidate_h) * (2.0 / n_points)
        scale_factor = 1.0 / total_integral

        c5_candidate = objective(candidate_h)
        if c5_candidate < best_c5:
            best_h = candidate_h
            best_c5 = c5_candidate
    
    # Final evaluation to ensure constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    total_integral = np.sum(best_h) * (2.0 / n_points)
    scale_factor = 1.0 / total_integral
    best_h = best_h * scale_factor
    
    return best_h, best_c5, n_points
