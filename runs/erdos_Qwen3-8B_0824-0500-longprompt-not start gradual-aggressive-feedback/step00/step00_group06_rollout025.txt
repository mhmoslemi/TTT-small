```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Start with the initial construction
    initial_h = initial_h_values
    n_points_initial = len(initial_h)
    if n_points_initial < 1000:
        h = initial_h
        n_points = n_points_initial
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n_points = n_points

    # Try improving with a smaller number of points
    for n_points in [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]:
        if time.time() - start_time > budget_s - 1:
            break

        # Create a grid of x values
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        h = np.random.rand(n_points)
        h = h * (n_points / 2)  # Ensure sum(h) == n_points / 2
        h = np.clip(h, 0, 1)

        # Define objective function for optimization
        def objective(h):
            h = np.clip(h, 0, 1)
            correlation = signal.correlate(h, 1 - h, mode='full')
            c5_val = np.max(correlation) * dx
            return c5_val

        # Optimization: minimize max correlation
        def optimize_h(h_initial):
            h = cp.Variable(n_points)
            obj = cp.Minimize(cp.Max(signal.correlate(h.value, 1 - h.value, mode='full') * dx))
            constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
            problem = cp.Problem(obj, constraints)
            result = problem.solve(solver='SCIPY')
            if result is not None:
                return np.array(h.value)
            return h_initial

        # Run the optimization
        h_opt = optimize_h(h)
        c5_val = objective(h_opt)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_opt
            best_n_points = n_points

    # Final check with the best solution found
    if best_h is not None:
        return (best_h, best_c5, best_n_points)
    else:
        return (initial_h, best_c5, best_n_points)
```
