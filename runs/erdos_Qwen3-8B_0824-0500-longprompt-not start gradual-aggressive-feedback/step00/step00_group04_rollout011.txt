import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can adjust if needed
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    # Start with a uniform distribution as initial guess
    initial_h_values[:] = 1.0 / n_points * n_points / 2
    h = initial_h_values.copy()
    
    # Function to compute the overlap integral and max over k
    def compute_c5(h):
        h = np.array(h)
        # Normalize h to ensure sum(h) == n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        # Compute the convolution
        conv = np.correlate(h, 1 - h, mode="full")
        # Compute the maximum overlap integral
        c5 = np.max(conv) * dx
        return c5
    
    # Function to optimize h using gradient descent
    def optimize_h(h, max_iter=500, learning_rate=0.01, tol=1e-5):
        h = np.array(h)
        best_h = h.copy()
        best_c5 = compute_c5(h)
        start_time = time.time()
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            # Compute the current c5
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            # Compute the gradient (approximate by finite differences)
            eps = 1e-6
            grad = np.zeros_like(h)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_c5(h_plus)
                h_minus = h.copy()
                h_minus[j] -= eps
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                c5_minus = compute_c5(h_minus)
                grad[j] = (c5_plus - c5_minus) / (2 * eps)
            # Update h using gradient descent
            h -= learning_rate * grad
            # Ensure h stays within [0, 1] and sum(h) == n_points / 2
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        return best_h, best_c5
    
    # Run the optimization
    best_h, best_c5 = optimize_h(h)
    return best_h, best_c5, n_points
