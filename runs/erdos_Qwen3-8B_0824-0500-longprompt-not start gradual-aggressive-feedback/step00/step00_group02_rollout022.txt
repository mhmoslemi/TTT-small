```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can be increased if needed
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points]) * (n_points / 2) / np.sum(initial_h_values[:n_points])
    h = np.clip(h, 0, 1)

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        # Compute convolution
        corr = np.correlate(h, 1 - h, mode="full")
        # Extract relevant part of the convolution
        corr = corr[n_points - 1:]
        # Multiply by dx
        c5 = np.max(corr * dx)
        return c5

    # Function to optimize h
    def optimize_h(h_vals):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        objective = cp.Minimize(cp.Max(cp.multiply(cp.sum(cp.multiply(h, 1 - h)) / n_points, 1)))
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.CVXOPT, verbose=False)
        if result is None:
            return h_vals
        return h.value

    # Try to improve the solution
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_n_points = n_points

    # Try increasing n_points if time allows
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        dx = 2.0 / n_points
        h = np.array(initial_h_values[:n_points]) * (n_points / 2) / np.sum(initial_h_values[:n_points])
        h = np.clip(h, 0, 1)

        # Use optimization to refine h
        h_opt = optimize_h(h)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5
            best_n_points = n_points

        # Print progress
        print(f"n_points={n_points}, c5={best_c5:.6f}")

    # Return the best result
    return (best_h, best_c5, best_n_points)
```
