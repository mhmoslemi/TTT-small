import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
# evaluate_erdos_solution() is used to evaluate a solution
# initial_h_values is the starting point (n=95 samples)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with n_points = 100 for faster optimization

    # Initial guess from provided initial_h_values
    initial_h = initial_h_values.copy()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    # Rescale to satisfy the integral constraint
    scale = (n_points / 2) / np.sum(initial_h)
    initial_h = initial_h * scale

    # Evaluate the initial guess
    h_values = initial_h
    c5 = evaluate_erdos_solution(h_values)
    best_c5 = c5
    best_h = h_values.copy()

    # Try optimization for different n_points
    for n_points in [100, 150, 200, 250, 300, 350, 400]:
        if time.time() - start_time > budget_s - 1:
            break

        # Define the optimization problem
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        objective = cp.Minimize(
            cp.Max(
                cp.sum(h * (1 - h[:(n_points - k)])) * (2.0 / n_points) for k in range(1, n_points)
            )
        )
        problem = cp.Problem(objective, constraints)

        try:
            result = problem.solve(solver=cp.SCS, max_iters=100, verbose=False)
            if result is not None and np.isfinite(result):
                # Extract the optimized h values
                h_values = h.value
                # Evaluate the C5
                c5 = evaluate_erdos_solution(h_values)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values.copy()
        except:
            # If optimization fails, use the initial guess
            h_values = initial_h.copy()
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values.copy()

    # Return the best solution found
    return best_h, best_c5, n_points
