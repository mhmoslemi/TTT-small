import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 200  # Start with a manageable number of points
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values[:n_points])  # Use initial construction
    h_initial = np.clip(h_initial, 0, 1)  # Ensure values are within [0, 1]
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Normalize to meet the integral constraint
    
    # Define the function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure the integral equals 1
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5
    
    # Define the objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Define bounds for h
    bounds = Bounds(0, 1)
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-5
    
    # Use L-BFGS-B for optimization
    result = opt.minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, tol=tolerance, options={'maxiter': max_iter})
    
    # Check if we have time left to refine the solution
    elapsed = time.time() - start_time
    if elapsed < budget_s - 5:
        # Try a more refined approach with higher n_points
        n_points = 400
        dx = 2.0 / n_points
        h_initial = np.zeros(n_points)
        # Start with a uniform distribution
        h_initial[:n_points//2] = 1
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Re-define compute_c5 with new n_points
        def compute_c5_refined(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            conv = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(conv) * dx
            return c5
        
        # Re-define objective function
        def objective_refined(h):
            return compute_c5_refined(h)
        
        # Re-define bounds
        bounds_refined = Bounds(0, 1)
        
        # Optimize again
        result_refined = opt.minimize(objective_refined, h_initial, method='L-BFGS-B', bounds=bounds_refined, tol=tolerance, options={'maxiter': max_iter})
        best_h = result_refined.x
        best_c5 = compute_c5_refined(best_h)
    else:
        best_h = result.x
        best_c5 = compute_c5(best_h)
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
