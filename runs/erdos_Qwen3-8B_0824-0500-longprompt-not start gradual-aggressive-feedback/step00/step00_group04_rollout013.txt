import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    
    # Initial guess from provided initial_h_values
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = (h * 2.0 - 1.0) * 0.5 + 0.5  # Random initialization in [0, 1]
    
    # Normalize to meet the integral constraint
    h /= (np.sum(h) * dx)
    h = np.clip(h, 0, 1)  # Ensure values are within [0, 1]
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop
    for step in range(100):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Compute current C5
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Try a small perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h /= (np.sum(perturbed_h) * dx)
        
        # Evaluate perturbed solution
        perturbed_c5 = evaluate_erdos_solution(perturbed_h)
        if perturbed_c5 < best_c5:
            best_c5 = perturbed_c5
            best_h = perturbed_h.copy()
        
        # Try gradient descent-like update
        # Compute overlap integrals for all shifts
        overlap = np.correlate(h, 1 - h, mode="full")
        overlap *= dx
        max_overlap = np.max(overlap)
        
        # Create a cost function to minimize
        cost = np.zeros(n_points)
        for i in range(n_points):
            cost[i] = np.max(overlap[i:i + n_points])
        
        # Gradient descent update
        grad = np.gradient(cost)
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        h /= (np.sum(h) * dx)
        
        # Update best solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
