```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
import random

# Pre-imported functions
import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different numbers of points
    n_points_list = [200, 300, 400, 500]
    best_c5 = float('inf')
    best_h = None
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Initialize h with a random perturbation of the initial solution
        h = np.array(initial_h_values) * (n_points / len(initial_h_values))
        h = np.round(h).astype(float)
        h = h[:n_points]
        
        # Perturb h to avoid getting stuck in local minima
        for _ in range(3):
            idxs = np.random.choice(n_points, size=5, replace=False)
            h[idxs] = np.random.uniform(0, 1, size=5)
        
        # Ensure the integral constraint
        total = np.sum(h) * dx
        scaling_factor = 1.0 / total
        h = h * scaling_factor
        
        # Function to compute C5
        def compute_c5(h):
            h = np.clip(h, 0, 1)
            h_extended = np.concatenate([h, np.zeros_like(h)])
            corr = np.correlate(h_extended, 1 - h, mode="full")
            overlaps = corr[1:-1] * dx
            return np.max(overlaps)
        
        # Use a simple gradient descent approach
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            
            # Compute current C5
            current_c5 = compute_c5(h)
            
            # Compute gradients for all shifts (k = 1 to n_points - 1)
            # For each k, compute the derivative of the overlap with respect to h[i]
            gradients = np.zeros_like(h)
            for k in range(1, n_points):
                overlap = np.correlate(h, 1 - h, mode="full")[k]
                for i in range(n_points):
                    # Derivative of overlap with respect to h[i]
                    # overlap = sum_{j} h[j] * (1 - h[j + k])
                    # derivative = (1 - h[j + k]) - h[j] * (1 - 2h[j + k])
                    if i + k < n_points:
                        derivative = (1 - h[i + k]) - h[i] * (1 - 2 * h[i + k])
                    else:
                        derivative = 0
                    gradients[i] += derivative * dx
            
            # Update h using gradient descent
            step_size = 0.01
            h = h - step_size * gradients
            h = np.clip(h, 0, 1)
            
            # Ensure the integral constraint
            total = np.sum(h) * dx
            scaling_factor = 1.0 / total
            h = h * scaling_factor
        
        # Evaluate the best solution found
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    # Final evaluation
    if best_h is not None:
        result = evaluate_erdos_solution.evaluate_erdos_solution(best_h)
        print(f"Final C5 bound: {best_c5:.6f}")
        print(f"Result from evaluate_erdos_solution: {result}")
        return (best_h.tolist(), best_c5, len(best_h))
    else:
        return ([], float('inf'), 0)
```
