import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    import numpy as np
    import scipy.optimize as optimize
    import time
    
    # Set random seed
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial number of points
    n_points = 80
    
    # Initial h_values
    h_values = initial_h_values.copy()
    
    # Check initial values meet constraints
    assert np.all(0 <= h_values) and np.all(h_values <= 1), "Initial h_values out of bounds"
    assert np.isclose(np.sum(h_values) * (2.0 / n_points), 1.0), "Initial h_values sum not equal to 1"
    
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        h_extended = np.pad(h, (0, n), mode='constant', constant_values=0)
        corr = np.correlate(h_extended, 1 - h, mode='full')
        corr = corr[n - 1: 2 * n - 1]
        c5 = np.max(corr * dx)
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    h_guess = h_values.copy()
    
    # Optimization settings
    max_iter = 100
    tol = 1e-5
    
    # Optimization
    result = optimize.minimize(
        objective,
        h_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': max_iter, 'ftol': tol, 'disp': False}
    )
    
    # If optimization fails, use initial values
    if result.success:
        h_values = result.x
    else:
        h_values = h_guess
    
    # Ensure h_values are within [0,1]
    h_values = np.clip(h_values, 0.0, 1.0)
    
    # Ensure the sum constraint is satisfied
    sum_h = np.sum(h_values) * (2.0 / n_points)
    if not np.isclose(sum_h, 1.0):
        # Adjust slightly to satisfy the constraint
        h_values = h_values / sum_h * 1.0
    
    # Compute final C5
    c5_bound = compute_c5(h_values)
    
    # Check time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (h_values, c5_bound, n_points)
    
    # Try increasing the number of points to improve the bound
    n_points_new = min(200, n_points * 2)
    h_values_new = np.zeros(n_points_new)
    dx_new = 2.0 / n_points_new
    
    # Initialize new h_values with the previous solution
    for i in range(n_points):
        h_values_new[i] = h_values[i] * dx_new * n_points / 2.0
    
    # Optimization with new number of points
    bounds_new = [(0.0, 1.0) for _ in range(n_points_new)]
    
    result_new = optimize.minimize(
        lambda h: compute_c5(h),
        h_values_new,
        method='L-BFGS-B',
        bounds=bounds_new,
        constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points_new / 2}],
        options={'maxiter': 50, 'ftol': 1e-5, 'disp': False}
    )
    
    if result_new.success:
        h_values = result_new.x
    else:
        h_values = h_values_new
    
    h_values = np.clip(h_values, 0.0, 1.0)
    sum_h = np.sum(h_values) * dx_new
    if not np.isclose(sum_h, 1.0):
        h_values = h_values / sum_h * 1.0
    
    c5_bound = compute_c5(h_values)
    n_points = n_points_new
    
    return (h_values, c5_bound, n_points)
