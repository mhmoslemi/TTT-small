```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can scale up if needed
    
    # Function to compute C5 for a given h
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Extract the relevant part for overlap
        c5 = np.max(corr) * dx
        return c5
    
    # Function to optimize h using CVXPY
    def optimize_h(h_init):
        n = len(h_init)
        h = cp.Variable(n)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n / 2
        ]
        # Objective: minimize the max correlation
        # We use a trick to minimize the maximum value
        t = cp.Variable()
        constraints.append(t >= 0)
        for k in range(-n+1, n):
            # Shift h by k
            shifted_h = np.roll(h, k)
            # Compute the integral of h(x)(1 - h(x + k))
            # This is approximated by sum(h * (1 - shifted_h)) * dx
            integral = cp.sum(h * (1 - shifted_h)) * (2.0 / n)
            constraints.append(t >= integral)
        # Minimize t
        prob = cp.Problem(cp.Minimize(t), constraints)
        prob.solve(solver='SCIPY')  # Or 'CVXOPT' if SCIPY fails
        if prob.status not in ['optimal', 'success']:
            return h_init
        return h.value
    
    # Start with initial h
    h = initial_h_values.copy()
    h = np.array(h)
    best_h = h
    best_c5 = compute_c5(h)
    
    # Refine h using optimization
    h = optimize_h(h)
    best_c5 = compute_c5(h)
    best_h = h
    
    # Try refining with more points if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h = np.zeros(n_points)
        # Distribute the integral evenly
        h[:int(n_points / 2)] = 1.0
        # Randomly perturb to avoid local minima
        perturbation = np.random.rand(n_points) * 0.2 - 0.1
        h += perturbation
        h = np.clip(h, 0, 1)
        h = optimize_h(h)
        best_c5 = compute_c5(h)
        best_h = h
    
    # Final check
    if time.time() - start_time < budget_s:
        # Try one more refinement
        h = optimize_h(best_h)
        best_c5 = compute_c5(h)
        best_h = h
    
    return (best_h.tolist(), best_c5, n_points)
```
