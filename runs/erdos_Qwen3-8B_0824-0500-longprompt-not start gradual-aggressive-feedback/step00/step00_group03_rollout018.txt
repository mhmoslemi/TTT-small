```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster evaluation

    # Check if the initial solution is valid
    initial_h = initial_h_values[:n_points]
    if len(initial_h) != n_points:
        initial_h = np.random.rand(n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    initial_c5 = evaluate_erdos_solution(initial_h)
    best_c5 = initial_c5
    best_h = initial_h.copy()

    # Try a few different point counts
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        # Initialize with a random valid h
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = h.astype(np.float64)

        # Define the optimization problem
        x = cp.Variable(n_points)
        objective = cp.Maximize(cp.sum(cp.multiply(x, 1 - cp.roll(x, 1, axis=0))))
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4)
            if prob.status == 'optimal':
                h_opt = np.round(x.value, 4)
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
        except:
            pass

        # Try local search from current best
        if best_h is not None:
            h_local = best_h.copy()
            for _ in range(10):
                if time.time() - start_time > budget_s - 1:
                    break
                # Small perturbation
                idx = np.random.randint(n_points)
                h_local[idx] += np.random.uniform(-0.1, 0.1)
                h_local = np.clip(h_local, 0, 1)
                h_local = h_local * (n_points / 2) / np.sum(h_local)
                c5 = evaluate_erdos_solution(h_local)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_local.copy()

    return (best_h, best_c5, n_points)
```
