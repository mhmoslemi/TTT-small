import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)

    # Start time
    start_time = time.time()

    # Initial configuration
    n_points = 45
    h_initial = initial_h_values
    h = copy.deepcopy(h_initial)
    h_values = h

    # Objective function
    def objective(h_values):
        # Ensure h_values is a numpy array
        h = np.array(h_values, dtype=np.float64)
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Compute the maximum value of the correlation
        max_corr = np.max(corr)
        return max_corr * (2.0 / n_points)

    # Constraint function
    def constraint_sum(h_values):
        h = np.array(h_values, dtype=np.float64)
        return np.sum(h) - n_points / 2

    # Optimization options
    method = 'SLSQP'
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda x: np.ones_like(x)}]

    # Run optimization
    result = optimize.minimize(
        objective,
        h,
        method=method,
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    # Get best solution
    best_h = result.x
    best_c5 = result.fun
    best_n_points = n_points

    # Check if we have time to try with more points
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 10:
        # Try with double the number of points
        n_points_new = 90
        h_new = np.random.rand(n_points_new) * 0.5  # Initial guess

        # Rescale to satisfy the integral constraint
        h_new = h_new * (n_points_new / 2) / np.sum(h_new)

        # Optimization with more points
        def objective_new(h_values):
            h = np.array(h_values, dtype=np.float64)
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr)
            return max_corr * (2.0 / n_points_new)

        def constraint_sum_new(h_values):
            h = np.array(h_values, dtype=np.float64)
            return np.sum(h) - n_points_new / 2

        constraints_new = [{'type': 'eq', 'fun': constraint_sum_new, 'jac': lambda x: np.ones_like(x)}]
        bounds_new = [(0.0, 1.0) for _ in range(n_points_new)]

        result_new = optimize.minimize(
            objective_new,
            h_new,
            method=method,
            bounds=bounds_new,
            constraints=constraints_new,
            options={'ftol': 1e-8, 'maxiter': 100}
        )

        best_h = result_new.x
        best_c5 = result_new.fun
        best_n_points = n_points_new

    # Return the best solution
    return best_h, best_c5, best_n_points
