    for n_points in [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Interpolate initial h to current n_points
        x_initial = np.linspace(0, 2, len(h_initial))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_initial, h_initial)
        
        # Rescale to meet the integral constraint
        dx = 2.0 / n_points
        total = np.sum(h) * dx
        h = h * (1 / total)
        
        # Define the overlap integral function
        def compute_c5(h):
            h_shifted = np.concatenate([h[-1:], h[:-1]])
            corr = signal.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        # Define variables for optimization
        x = cp.Variable(n_points, boolean=True)

        constraint1 = cp.sum(x) == n_points / 2  # Integral constraint
        constraint2 = cp.Minimize(cp.max(cp.multiply(x, 1 - x) * dx))
        
        # Solve optimization problem
        problem = cp.Problem(constraint2, [constraint1])
        try:
            problem.solve(solver=cp.MOSEK, eps=1e-4, verbose=False)
        except:
            problem.solve(solver=cp.SCS, eps=1e-4, verbose=False)
        
        # Extract optimal h
        if problem.status == 'optimal':
            h_opt = x.value
            c5_opt = compute_c5(h_opt)
            h_values = h_opt

            h = h.copy()

            h[1::2] = 0.0
            h = h * (n_points / 2) / np.sum(h)
            c5 = compute_c5(h)
            h_values = h
            c5_bound = c5
