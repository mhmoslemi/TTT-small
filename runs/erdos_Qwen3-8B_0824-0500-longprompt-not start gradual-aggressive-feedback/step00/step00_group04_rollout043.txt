import numpy as np
import scipy.optimize as optimize
import time
import copy
from scipy.optimize import Bounds
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = 42  # Starting with the initial construction
    
    # Initial values from the given initial_h_values
    h_values = initial_h_values.copy()
    
    # Function to evaluate the C5 bound
    def evaluate_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        # Compute the cross-correlation
        correlation = np.correlate(h, 1 - h, mode="full")
        # Multiply by dx to get the integral approximation
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    # Function to check if the solution meets the constraints
    def is_valid(h):
        return np.isclose(np.sum(h) * 2.0 / len(h), 1.0)
    
    # Function to perform optimization with bounds and constraints
    def optimize_h(n_points, initial_h=None):
        if initial_h is None:
            initial_h = np.random.uniform(0, 1, n_points)
        initial_h = np.clip(initial_h, 0, 1)
        # Normalize to satisfy the integral constraint
        initial_h = initial_h / (np.sum(initial_h) * 2.0 / n_points)
        
        # Bounds for h[i] (0 to 1)
        bounds = Bounds(0, 1)
        
        # Objective function: minimize the maximum overlap
        def objective(x):
            return evaluate_c5(x)
        
        # Constraints
        def constraint(x):
            return np.sum(x) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint}
        
        # Perform optimization
        result = optimize.minimize(
            objective,
            initial_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'disp': False}
        )
        
        # Return the optimized h values
        return result.x
    
    # Function to refine the solution using a quadratic programming approach
    def refine_h(h):
        n = len(h)
        dx = 2.0 / n
        # Objective: minimize the maximum overlap
        # We can approximate this by minimizing the maximum value of the correlation
        # We'll use a quadratic programming approach
        # Since we cannot directly minimize a max function, we minimize the maximum
        # by introducing a variable for the maximum value
        # This is a convex optimization problem
        
        # Objective: minimize max_overlap
        # We'll minimize a single variable max_overlap and ensure that
        # the correlation at each point is less than or equal to max_overlap
        # and that the integral constraint is satisfied
        
        # Create variables
        h = cp.Variable(n)
        max_overlap = cp.Variable()
        
        # Objective: minimize max_overlap
        objective = cp.Minimize(max_overlap)
        
        # Constraints: correlation at each point <= max_overlap
        for k in range(n):
            # Compute the correlation at shift k
            # This is equivalent to the integral of h(x) * (1 - h(x + k))
            # We'll use a numerical approach here
            # For each x, h(x) * (1 - h((x + k) % 2))
            # We can approximate this by evaluating the integral numerically
            # But to make this efficient, we'll use a sampled approach
            # For now, we'll use a simple correlation approximation
            corr = 0
            for i in range(n):
                x = i * dx
                x_shifted = (x + k) % 2
                shift_index = int(x_shifted / dx)
                if shift_index < 0 or shift_index >= n:
                    continue
                corr += h[i] * (1 - h[shift_index])
            corr /= n
            constraint = corr <= max_overlap
        
        # Constraint: integral of h is 1
        constraint = cp.sum(h) * dx == 1
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver='SCIPY')
        
        # Return the optimized h values
        return h.value
    
    # Main optimization loop
    best_h = h_values
    best_c5 = evaluate_c5(h_values)
    best_n_points = n_points
    
    # Try increasing the number of points up to 1000
    for n_points in range(42, 1001):
        if time.time() - start_time > budget_s:
            break
        
        # Optimize the solution
        h = optimize_h(n_points)
        
        # Refine the solution
        h = refine_h(h)
        
        # Check if the solution is valid
        if not is_valid(h):
            continue
        
        # Evaluate the C5 value
        c5 = evaluate_c5(h)
        
        # Update the best solution
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n_points = n_points
    
    # Return the best solution found
    return (best_h, best_c5, best_n_points)
