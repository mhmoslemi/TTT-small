```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial construction
    n_points = 45
    h_initial = initial_h_values
    h_initial = h_initial[:n_points]
    
    # Try with smaller n_points for faster computation
    # Try with n_points = 50, 60, 70, etc., up to maximum that fits within budget
    # Start with n_points = 50
    n_points = 50
    dx = 2.0 / n_points
    sum_h = n_points / 2  # Must satisfy ∫h(x)dx = 1 => sum(h) * dx = 1
    h = np.full(n_points, 0.5)
    h[:int(n_points / 2)] = 1.0
    h[int(n_points / 2):] = 0.0
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    best_n = n_points
    
    # Try different n_points sizes up to a maximum that still allows for fast evaluation
    for n in [50, 60, 70, 80, 90, 100]:
        dx = 2.0 / n
        sum_h = n / 2
        
        # Initialize h with a simple pattern
        h = np.full(n, 0.5)
        h[:int(n / 2)] = 1.0
        h[int(n / 2):] = 0.0
        
        # Try a few random perturbations to escape local optima
        for _ in range(3):
            # Perturb the h values with small random changes
            h_perturb = h.copy()
            perturbations = np.random.uniform(-0.1, 0.1, size=n)
            h_perturb += perturbations
            h_perturb = np.clip(h_perturb, 0.0, 1.0)
            
            # Check if the sum condition is met
            if np.abs(np.sum(h_perturb) - sum_h) > 1e-6:
                continue
            
            # Evaluate the overlap
            c5 = evaluate_erdos_solution(h_perturb, n, dx)
            
            # Update best solution if this is better
            if c5 < best_c5:
                best_h = h_perturb.copy()
                best_c5 = c5
                best_n = n
        
        # Stop early if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Return the best solution found
    return best_h, best_c5, best_n
```
