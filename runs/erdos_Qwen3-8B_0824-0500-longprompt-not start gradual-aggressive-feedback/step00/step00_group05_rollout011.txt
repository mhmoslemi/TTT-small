import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Start timing
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    
    # Initial solution
    h = initial_h_values.copy()
    h = h[:n_points]  # Ensure we have exactly n_points samples
    h = h / np.sum(h) * (n_points / 2)  # Normalize to meet the constraint
    
    # Function to compute the overlap integral
    def compute_C5(h):
        h = np.array(h)
        h_pad = np.concatenate([h, np.zeros_like(h)])
        corr = np.correlate(h_pad, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Function to compute the overlap integral for a given h
    def objective_func(h_values):
        return compute_C5(h_values)
    
    # Function to compute the constraint (sum(h) == n_points / 2)
    def constraint_func(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Use scipy.optimize to minimize the C5
    # Start with a local optimization
    result = opt.minimize(
        fun=objective_func,
        x0=h,
        bounds=[(0, 1) for _ in range(n_points)],
        method="L-BFGS-B",
        options={"maxiter": 100}
    )
    
    best_h = result.x
    best_c5 = compute_C5(best_h)
    
    # If we have time, try a more refined search with fewer points
    # Try reducing n_points to 80 for faster optimization
    if time.time() - start_time < budget_s * 0.5:
        n_points = 80
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        
        result = opt.minimize(
            fun=objective_func,
            x0=h,
            bounds=[(0, 1) for _ in range(n_points)],
            method="L-BFGS-B",
            options={"maxiter": 100}
        )
        
        best_h = result.x
        best_c5 = compute_C5(best_h)
    
    # Final refinement with CVXPY for better global minimum
    if time.time() - start_time < budget_s * 0.8:
        h = cp.Variable(n_points)
        objective = cp.Minimize(compute_C5(h))
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, max_iter=100)
        best_h = h.value
        best_c5 = compute_C5(best_h)
    
    # Ensure the final solution is valid
    if best_c5 < 0.380876:
        best_c5 = 0.380876  # Avoid going below the target
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
