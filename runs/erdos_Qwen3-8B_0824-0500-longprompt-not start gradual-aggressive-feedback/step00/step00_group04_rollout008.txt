import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(42) * 2 - 1  # Placeholder; actual initial construction may differ

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can increase if time allows
    h = np.random.rand(n_points) * 2 - 1  # Initial random guess
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Ensure integral constraint is satisfied
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    time_remaining = budget_s
    start_time = time.time()
    
    # Function to evaluate C5
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
        return c5
    
    # Gradient descent with momentum and adaptive learning rates
    learning_rate = 0.1
    momentum = 0.9
    decay_rate = 0.99
    best_learning_rate = learning_rate
    best_momentum = momentum
    best_decay_rate = decay_rate
    best_c5 = evaluate(h)
    
    for step in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute current C5
        current_c5 = evaluate(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Gradient descent step
        gradient = np.zeros_like(h)
        for i in range(n_points):
            for k in range(n_points):
                if i + k < n_points:
                    gradient[i] += (1 - h[k]) * (1 if h[i] > 0 else 0)
                if i - k >= 0:
                    gradient[i] += (1 - h[k]) * (1 if h[i] > 0 else 0)
        
        # Update learning rate and momentum
        learning_rate *= decay_rate
        momentum = momentum * 0.95 + 0.05
        h = h - learning_rate * gradient
        
        # Ensure bounds and integral constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    return best_h, final_c5, n_points
