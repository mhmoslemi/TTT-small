def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points, can be adjusted if needed
    
    # Initialize with the provided initial solution
    h = initial_h_values.copy()
    n_points_initial = len(h)
    if n_points_initial != n_points:
        h = np.zeros(n_points)
        h[:n_points_initial] = initial_h_values
        h = np.repeat(h, n_points // n_points_initial)
        h = h[:n_points]
    
    dx = 2.0 / n_points
    # Ensure integral constraint is satisfied
    total = np.sum(h) * dx
    scale = 1.0 / total
    h *= scale
    
    # Function to compute C₅
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        one_minus_h = 1.0 - h
        corr = signal.correlate(h, one_minus_h, mode='full')
        return np.max(corr) * dx
    
    # Function to minimize C₅ using optimization
    def optimize_h(h):
        result = optimize.minimize(
            compute_c5,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        return result
    
    # Initial optimization
    result = optimize_h(h)
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Additional optimization runs to improve the result
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        # Generate a small perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h /= np.sum(perturbed_h) * dx
        result = optimize_h(perturbed_h)
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    # Final check of the best solution
    best_h = np.clip(best_h, 0, 1)
    best_h /= np.sum(best_h) * dx
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, n_points
