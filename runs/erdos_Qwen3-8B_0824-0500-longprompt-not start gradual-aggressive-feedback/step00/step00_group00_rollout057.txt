import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Initial number of points
    n_points = 45
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        h = np.array(h_values)
        shifted_h = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted_h[k:] = h[:len(h) - k]
        integrals = np.zeros(len(h) - 1)
        for k in range(len(h) - 1):
            integrals[k] = np.sum(h * (1 - shifted_h[k:])) * dx
        return np.max(integrals)
    
    # Function to compute the overlap integral with given h
    def objective_func(h_values):
        return compute_c5(h_values)
    
    # Function to compute the integral constraints
    def constraint_func(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Initialize with the given initial solution
    current_best = h.copy()
    best_c5 = compute_c5(current_best)
    
    # Try improving with a few iterations of optimization
    for _ in range(3):
        # Use scipy.optimize to minimize the C5 value
        def objective(x):
            return objective_func(x)
        
        def constraint(x):
            return constraint_func(x)
        
        # Bounds for each h[i] (0 <= h[i] <= 1)
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Solve the optimization problem
        result = opt.minimize(objective, current_best, bounds=bounds, method='L-BFGS-B')
        
        if result.success:
            current_best = result.x
            current_c5 = compute_c5(current_best)
            
            # Check if this is an improvement
            if current_c5 < best_c5:
                best_c5 = current_c5
        else:
            break
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Return the best solution found
    return (current_best, best_c5, n_points)
