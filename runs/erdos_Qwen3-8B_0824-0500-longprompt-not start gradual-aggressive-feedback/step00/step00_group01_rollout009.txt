```python
import numpy as np
import scipy.optimize
import scipy.signal
import time
import cvxpy as cp
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(47) * 2 - 1  # Example initial values (not actual initial construction)

def generate_initial_guess(n_points):
    """Generate a random initial guess that satisfies the constraints."""
    h = np.random.rand(n_points)
    h -= np.min(h)
    h /= np.max(h)
    h *= 0.5
    h += 0.5
    return h

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    elapsed_time = 0

    # Start with the initial guess
    best_c5 = evaluate_erdos_solution(best_h)
    best_h = best_h.copy()
    print(f"Initial C5: {best_c5:.6f} (n_points={n_points})")

    # Try increasing n_points up to 1000
    for n_points in range(50, 1000, 50):
        # Generate a new initial guess
        h = generate_initial_guess(n_points)
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        h *= (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)

        # Use CVXPY to optimize the function
        x = cp.Variable(n_points)
        constraints = [0 <= x, x <= 1, cp.sum(x) == n_points / 2]
        # Objective function is the maximum of the overlap integrals
        # Approximate max using a large epsilon for each shift
        eps = 1e-3
        overlaps = []
        for k in range(-n_points + 1, n_points):
            if k == 0:
                continue
            if abs(k) > n_points:
                continue
            shifted = np.roll(x, k)
            overlap = cp.sum(x * (1 - shifted))
            overlaps.append(overlap)
        
        # Approximate max using a convex relaxation
        if len(overlaps) > 0:
            obj = cp.Minimize(cp.max(overlaps))
        else:
            obj = cp.Minimize(0)

        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - elapsed_time)
            if prob.status == 'optimal':
                h_opt = x.value
                h_opt = np.clip(h_opt, 0, 1)
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    print(f"Found better C5: {best_c5:.6f} (n_points={n_points})")
        except:
            pass
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break

    return best_h, best_c5, len(best_h)
```
