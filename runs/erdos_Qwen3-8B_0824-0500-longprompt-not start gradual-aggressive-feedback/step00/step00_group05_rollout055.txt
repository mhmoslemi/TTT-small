        def objective(x):
            h = x.copy()
            dx = 2.0 / len(h)
            overlap = np.correlate(h, 1 - h, mode="full")
            return np.max(overlap) * dx
        
        # Define bounds
        bounds = Bounds(0, 1)
        
        # Optimization
        start_time = time.time()
        result = optimize.minimize(
            objective, 
            h,

            bounds=bounds,
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        elapsed = time.time() - start_time
        
        if elapsed < budget_s:
            current_c5 = result.fun
            if current_c5 < best_c5:

                best_h = result.x.copy()
