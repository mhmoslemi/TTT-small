import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions and initial values
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the correlation
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Normalize and find the maximum
    c5 = np.max(corr * dx)
    return c5

initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for faster optimization
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Start with initial values (if provided)
    if hasattr(initial_h_values, '__len__') and len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        # Random initialization
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2  # Ensure sum is n_points / 2
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    # Bounds on h
    bounds = Bounds(0, 1)
    
    # Constraint: sum(h) = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    
    # Initial guess
    h_flat = h.reshape(-1)
    
    # Perform optimization
    start_time = time.time()
    result = opt.minimize(
        objective,
        h_flat,
        method='SLSQP',
        bounds=bounds,
        constraints=constraint,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Get best solution
    best_h = result.x
    best_c5 = result.fun
    elapsed_time = time.time() - start_time
    
    # Check if we need to increase the number of points
    if elapsed_time < budget_s - 10:
        # Try with more points (e.g., 200) for better accuracy
        n_points = 200
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        h = best_h.reshape(-1)  # Use previous best as initial guess
        h = h[:n_points]  # Truncate if necessary
        h = h / np.sum(h) * n_points / 2  # Ensure sum is n_points / 2
        
        def objective_large(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5
        
        bounds_large = Bounds(0, 1)
        constraint_large = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        
        result_large = opt.minimize(
            objective_large,
            h.reshape(-1),
            method='SLSQP',
            bounds=bounds_large,
            constraints=constraint_large,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        best_h = result_large.x
        best_c5 = result_large.fun
    
    return best_h, best_c5, n_points
