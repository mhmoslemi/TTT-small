import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions and variables
initial_h_values = np.load('initial_h_values.npy')  # 80 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess
    n_points = 80  # Start with 80 points
    h = initial_h_values.copy()
    h = h[:n_points]
    
    # Define the function to compute C5
    def compute_c5(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5
    
    # Define the optimization objective (minimize C5)
    def objective(h_flat):
        h = h_flat.reshape(-1)
        return compute_c5(h)
    
    # Define constraints
    def constraints(h_flat):
        h = h_flat.reshape(-1)
        return np.sum(h) - n_points / 2  # Enforce ∫h(x)dx = 1
    
    # Initial guess
    h_initial = h.copy()
    h_initial = h_initial.reshape(-1)
    
    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraints}
    
    # Run optimization
    result = opt.minimize(
        objective,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Check if we can try with more points (if time allows)
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        n_points = 120
        h_new = np.zeros(n_points)
        h_new[:int(n_points / 2)] = 1.0  # Initial guess for more points
        h_new = h_new.reshape(-1)
        
        # Re-define objectives and constraints
        def compute_c5_new(h_flat):
            h = h_flat.reshape(-1)
            dx = 2.0 / len(h)
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap * dx)
            return c5
        
        def objective_new(h_flat):
            h = h_flat.reshape(-1)
            return compute_c5_new(h)
        
        def constraints_new(h_flat):
            h = h_flat.reshape(-1)
            return np.sum(h) - n_points / 2  # Enforce ∫h(x)dx = 1
        
        bounds_new = [(0.0, 1.0) for _ in range(n_points)]
        cons_new = {'type': 'eq', 'fun': constraints_new}
        
        result_new = opt.minimize(
            objective_new,
            h_new,
            method='SLSQP',
            bounds=bounds_new,
            constraints=cons_new,
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        
        best_h_new = result_new.x
        best_c5_new = compute_c5_new(best_h_new)
        
        if best_c5_new < best_c5:
            best_h = best_h_new
            best_c5 = best_c5_new
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
