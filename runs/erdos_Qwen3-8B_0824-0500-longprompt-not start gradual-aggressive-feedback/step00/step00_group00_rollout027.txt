```python
import numpy as np
import scipy.signal
import scipy.optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    max_points = 1000  # Max number of points to consider
    max_iterations = 100  # Max number of iterations for optimization
    n_points = 100  # Start with 100 points
    
    # Initial guess with 100 points
    h = np.array(initial_h_values)
    if len(h) > n_points:
        h = h[::int(len(h)/n_points)]
    elif len(h) < n_points:
        h = np.concatenate([h, np.zeros(n_points - len(h))])
    h = np.clip(h, 0, 1)
    h = h[:n_points]
    h = h * (n_points / 2) / np.sum(h)  # Normalize to ensure integral is 1
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Try increasing the number of points gradually
    for n_points in range(100, max_points + 1, 50):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Resample to n_points
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Normalize
        
        # Function to evaluate C5
        def evaluate_c5(h):
            dx = 2.0 / n_points
            h_shifted = np.concatenate([h[-2:], h[:-2]])
            corr = scipy.signal.correlate(h, 1 - h_shifted, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        # Optimization: minimize C5
        def objective(x):
            h = x.reshape(n_points)
            dx = 2.0 / n_points
            h_shifted = np.concatenate([h[-2:], h[:-2]])
            corr = scipy.signal.correlate(h, 1 - h_shifted, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        # Constraints: sum(h) == n_points / 2
        x = cp.Variable(n_points)
        constraints = [cp.sum(x) == n_points / 2, 0 <= x, x <= 1]
        
        # Optimization problem
        prob = cp.Problem(cp.Minimize(objective(x)), constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=100, verbose=False)
        except:
            pass
        
        if prob.status == "optimal":
            h_opt = x.value
            c5_opt = evaluate_c5(h_opt)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, len(best_h)
```
