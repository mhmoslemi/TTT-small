import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(signal.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.array([0.0]*95)  # Placeholder for initial values (to be filled)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Define the number of points (adjustable for performance)
    n_points = 256  # We'll use a moderate number to balance speed and performance
    
    # Initialize h_values with a basic construction
    h_values = np.zeros(n_points)
    h_values[::2] = 1.0  # A simple checkerboard pattern
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(h_values) * (2.0 / n_points)
    if abs(total_integral - 1.0) > 1e-6:
        scale = 1.0 / total_integral
        h_values *= scale
    
    # Function to evaluate C5
    def objective(h_values):
        h = h_values.copy()
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Gradient-based optimization
    def optimize_solution(h_vals):
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Objective function to minimize the maximum overlap
        corr = signal.correlate(h.value, 1 - h.value, mode="full") if h.value is not None else np.zeros_like(h.value)
        max_overlap = np.max(corr) * (2.0 / n_points)
        
        # We need to use a different approach, as CVXPY can't directly minimize a max of correlations
        # We instead use a numerical optimization approach with constraints
        
        # Numerical optimization with constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [(0 <= h_vals[i], h_vals[i] <= 1) for i in range(n_points)]
        constraints.append((np.sum(h_vals) == n_points / 2))
        
        result = optimize.minimize(
            objective,
            h_vals,
            bounds=bounds,
            constraints=constraints,
            method='L-BFGS-B',
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        return result.x
    
    # Run optimization
    best_h = optimize_solution(h_values)
    best_c5 = objective(best_h)
    
    # Print result
    elapsed = time.time() - start_time
    print(f"Optimized h_values with C5: {best_c5:.6f} (Time: {elapsed:.2f}s)")
    
    return best_h, best_c5, n_points
