import scipy.optimize as opt
import time

def run(seed=42, budget_s=60, **kwargs):

    dx = 2.0 / n_points
    
    def objective(h_values):

        # Compute the overlap integrals for all shifts
        overlap = np.correlate(h, 1 - h, mode="full")

        return np.max(overlap)
