import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    n_points = 42
    initial_h_values = np.array(initial_h_values)
    
    # Try starting from the initial construction
    def objective(h_values):
        h = h_values.reshape(-1)
        # Ensure h has the right shape
        h = h.reshape(n_points)
        # Ensure the integral constraint is satisfied
        integral = np.sum(h) * (2.0 / n_points)
        # Adjust h to meet the integral constraint
        if abs(integral - 1.0) > 1e-6:
            # Normalize h
            h = h / integral * 1.0
        # Compute the overlap integral
        shifted_h = np.roll(h, 1)
        overlap = correlate(h, 1 - shifted_h, mode='full')
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    def objective_with_constraints(h_values):
        h = h_values.reshape(n_points)
        integral = np.sum(h) * (2.0 / n_points)
        if abs(integral - 1.0) > 1e-6:
            # Adjust h to meet the integral constraint
            h = h / integral * 1.0
        # Compute the overlap integral
        shifted_h = np.roll(h, 1)
        overlap = correlate(h, 1 - shifted_h, mode='full')
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    # Try a few different n_points values
    n_points_list = [42, 60, 80, 100, 120]
    best_c5 = float('inf')
    best_h = np.array([])
    
    for n_points in n_points_list:
        if time.time() > time.time() + budget_s:
            break
        print(f"Trying n_points = {n_points}")
        # Generate initial guess
        initial_h = np.random.rand(n_points)
        # Normalize to meet the integral constraint
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2.0)
        # Bounds for h[i] between 0 and 1
        bounds = Bounds(0.0, 1.0)
        # Optimization
        result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, tol=1e-5)
        # Check if we found a better solution
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        # Try starting from the initial construction
        initial_h = np.random.rand(n_points)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2.0)
        result = minimize(objective, initial_h, method='L-BFGS-B', bounds=bounds, tol=1e-5)
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
    
    # Return the best solution
    return (best_h, best_c5, len(best_h))
