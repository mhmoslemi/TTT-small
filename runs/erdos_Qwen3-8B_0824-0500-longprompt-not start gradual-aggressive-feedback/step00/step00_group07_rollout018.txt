import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions and initial construction
initial_h_values = np.load('initial_h_values.npy')  # Assume initial construction is provided
n_points_initial = len(initial_h_values)

def evaluate_erdos_solution(h):
    # Evaluate the overlap integral for the given h
    dx = 2.0 / len(h)
    h_extended = np.concatenate([h, np.zeros(len(h))])
    shift = np.arange(-len(h)+1, len(h))
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap * dx)
    return c5

def run(seed=42, budget_s=60, **kwargs):
    # Parameters for the optimization
    n_points = 100  # Start with a moderate number of points
    max_iter = 100  # Max number of iterations for optimization
    tolerance = 1e-4  # Tolerance for stopping criterion
    dx = 2.0 / n_points

    # Initialize h with the provided initial construction, or a uniform distribution
    h = np.zeros(n_points)
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * 2 - 1  # Ensure sum is around n_points / 2
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / len(h)
        h_extended = np.concatenate([h, np.zeros(len(h))])
        shift = np.arange(-len(h)+1, len(h))
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    start_time = time.time()

    for _ in range(max_iter):
        # Convert h to a CVXPY variable
        h_var = cp.Variable(n_points, nonneg=True)
        objective = cp.Minimize(cp.Max(cp.multiply(1 - h_var, cp.reshape(cp.fft.fft(h_var), (n_points, 1)))))

        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            0 <= h_var,
            h_var <= 1
        ]

        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.SCS, max_iter=100, eps=1e-4, verbose=False)
            if result is not None:
                h = h_var.value
                h = np.clip(h, 0, 1)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            # If solver fails, use gradient descent
            pass

        # Gradient descent as fallback
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        best_c5 = c5

        # Check for early stopping
        if time.time() - start_time > budget_s - 1:
            break

    return best_h, best_c5, n_points
