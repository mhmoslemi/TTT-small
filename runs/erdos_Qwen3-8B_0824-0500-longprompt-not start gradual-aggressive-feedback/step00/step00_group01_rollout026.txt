    n_points = 100  # Start with 100 points, can be adjusted if needed

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:n_points-1]], axis=0)  # Extend for full convolution
        shifted = np.roll(h, shift=1, axis=0)
        overlap = np.abs(np.correlate(h, 1 - shifted, mode="full")) * dx
        return np.max(overlap)
    
    # Function to compute the cost for optimization
    def cost_function(h_values):
        h_values = np.array(h_values)
        h_values = np.concatenate([h_values, h_values[:n_points-1]], axis=0)
        shifted = np.roll(h_values, shift=1, axis=0)
        overlap = np.abs(np.correlate(h_values, 1 - shifted, mode="full")) * dx
        return np.max(overlap)
    
    # Function to compute the constraint sum(h) == n_points / 2
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Define the optimization problem
    h = cp.Variable(n_points)
    objective = cp.Minimize(cost_function(h))
    constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
    
    # Solve using a solver
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=1000, eps=1e-6, verbose=False)
        if prob.status == 'optimal':
            h_opt = h.value
            c5_bound = compute_c5(h_opt)
            return (h_opt, c5_bound, n_points)
        else:
            # Fallback to gradient descent if not optimal
            h_grad_desc = np.copy(h)
            learning_rate = 0.01
            for _ in range(1000):
                h_values = np.copy(h_grad_desc)
                h_values = np.concatenate([h_values, h_values[:n_points-1]], axis=0)
                shifted = np.roll(h_values, shift=1, axis=0)
                overlap = np.abs(np.correlate(h_values, 1 - shifted, mode="full")) * dx
                c5 = np.max(overlap)
                # Compute gradient using finite differences
                grad = np.zeros(n_points)
                for i in range(n_points):
                    h_tmp = np.copy(h_values)
                    h_tmp[i] += 1e-4
                    h_tmp = np.concatenate([h_tmp, h_tmp[:n_points-1]], axis=0)
                    shifted_tmp = np.roll(h_tmp, shift=1, axis=0)
                    overlap_tmp = np.abs(np.correlate(h_tmp, 1 - shifted_tmp, mode="full")) * dx
                    c5_tmp = np.max(overlap_tmp)
                    grad[i] = (c5_tmp - c5) / 1e-4
                grad = -grad
                # Project gradient onto feasible set
                grad[grad > 0] = 0
                grad[grad < 0] = 0
                h_grad_desc += learning_rate * grad
                h_grad_desc = np.clip(h_grad_desc, 0, 1)
            c5_bound = compute_c5(h_grad_desc)
            return (h_grad_desc, c5_bound, n_points)
    except:
        # Fallback to random search
        best_c5 = float('inf')
        best_h = np.copy(h)
        for _ in range(100):
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = np.clip(h_rand, 0, 1)
            h_rand = np.concatenate([h_rand, h_rand[:n_points-1]], axis=0)
            shifted = np.roll(h_rand, shift=1, axis=0)
            overlap = np.abs(np.correlate(h_rand, 1 - shifted, mode="full")) * dx
            c5 = np.max(overlap)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_rand
        return (best_h[:n_points], best_c5, n_points)
    
    # Check if time is exceeded
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return (h, compute_c5(h), n_points)
