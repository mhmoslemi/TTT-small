def run(seed=42, budget_s=60, **kwargs):

    start_time = time.time()
    n_points = 100  # Use 100 points for a balance between speed and accuracy
    dx = 2.0 / n_points

    # Initial guess from the provided initial_h_values, if it has the same length
    if len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5  # Random initialization in [0, 1]

    # Ensure the integral constraint is met
    total_integral = np.sum(h) * dx
    h = h * (1 / total_integral) * (n_points / 2)  # Adjust to meet the constraint

    # Function to compute C5
    def compute_c5(h):
        h_pad = np.concatenate([h, np.zeros(n_points)])
        h_shifted = np.zeros(2 * n_points)
        for k in range(n_points):
            h_shifted[k:n_points + k] = h
        h_1_minus = 1 - h_shifted
        convolution = np.correlate(h_pad, h_1_minus, mode="full")
        overlap_integral = convolution * dx
        return np.max(overlap_integral)

    # Optimization objective: minimize the maximum overlap integral
    # We use a non-convex optimization approach with gradient-based methods
    def objective_func(x):
        h = x.copy()
        total_integral = np.sum(h) * dx
        h = h * (1 / total_integral) * (n_points / 2)  # Ensure constraint is met
        return compute_c5(h)

    # Gradient descent with momentum
    learning_rate = 0.01
    momentum = 0.9
    max_iterations = 1000
    tolerance = 1e-5
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros_like(h)

    for _ in range(max_iterations):
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Compute gradient numerically
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = h_plus * (1 / (np.sum(h_plus) * dx)) * (n_points / 2)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / 1e-6

        # Update velocity and parameters
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity

        # Ensure h remains within [0, 1]
        h = np.clip(h, 0, 1)

        # Check for early stopping
        if time.time() - start_time > budget_s - 1:
            break

    # Final evaluation
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
