```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
from time import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Pre-loaded initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with the initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    best_n_points = n_points

    # Try reducing the number of points to speed up the search
    for n_points in [100, 200, 300, 400, 500, 600, 700, 800, 900]:
        if time() - start_time > budget_s - 1:
            break
        
        # Random perturbation of the initial solution
        h = np.random.rand(n_points) * 2 - 1  # Random initial guess
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        h = h * (1 / sum_h) * (n_points / 2)  # Normalize to meet the constraint
        
        # Optimization with CVXPY
        x = cp.Variable(n_points)
        constraints = [
            0 <= x,
            x <= 1,
            cp.sum(x) == n_points / 2
        ]
        objective = cp.Maximize(cp.norm(cp.correlate(x, 1 - x, mode="full") * dx, 1))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, time_limit=budget_s - (time() - start_time))
            if prob.status == 'OPTIMAL':
                h_opt = x.value
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5
                    best_n_points = n_points
        except:
            pass
    
    return best_h, best_c5, best_n_points
```
