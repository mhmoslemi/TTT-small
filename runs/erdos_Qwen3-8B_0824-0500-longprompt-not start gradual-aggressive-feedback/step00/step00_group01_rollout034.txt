```python
import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Set up initial parameters
    n_points_initial = len(initial_h_values)
    dx = 2.0 / n_points_initial
    h_initial = np.array(initial_h_values)
    
    # Evaluate initial solution
    c5_initial = evaluate_erdos_solution(h_initial)
    
    # Try reducing the number of points
    for n_points in range(n_points_initial, 16, -1):
        if n_points < 16:
            break
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = h_initial[i * n_points_initial // n_points]
        h = h / (np.sum(h) * dx) * (n_points / 2)
        h = np.clip(h, 0, 1)
        
        # Compute initial C5
        c5 = evaluate_erdos_solution(h)
        if c5 < c5_initial:
            c5_initial = c5
            h_initial = h.copy()
    
    # Start with the best found so far
    h = h_initial.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    c5 = c5_initial
    
    # Define the objective function to minimize (we minimize max correlation)
    def objective(x):
        nonlocal c5
        h_new = x.copy()
        h_new = np.clip(h_new, 0, 1)
        # Compute the correlation
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        c5_candidate = np.max(corr * dx)
        if c5_candidate < c5:
            c5 = c5_candidate
        return c5_candidate
    
    # Set up constraints
    def constraint_sum(x):
        return np.sum(x) - n_points / 2
    
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Perform optimization
    start_time = time.time()
    result = opt.minimize(objective, h, method='L-BFGS-B', constraints=cons, tol=1e-6, options={'maxiter': 100})
    
    if result.success:
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5_opt = np.max(corr * dx)
    else:
        h_opt = h.copy()
        c5_opt = c5
    
    # Final evaluation
    corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_opt = np.max(corr * dx)
    
    return (h_opt, c5_opt, n_points)
```
