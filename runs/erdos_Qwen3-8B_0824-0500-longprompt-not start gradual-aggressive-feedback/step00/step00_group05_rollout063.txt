import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions and variables
# initial_h_values is a list of n_points samples over [0, 2]
# evaluate_erdos_solution(h_values) computes C5 for the given h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Initial number of points
    initial_n_points = len(initial_h_values)
    n_points = initial_n_points
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h_values):
        h = np.array(h_values)
        dx = 2.0 / n_points
        h_shifted = np.concatenate([np.zeros(n_points), h[:-1]])
        overlap = np.correlate(h, 1 - h_shifted, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to compute the cost (C5) for a given h_values
    def cost_function(h_values):
        return evaluate_overlap(h_values)
    
    # Function to compute the constraint (sum(h) == n_points / 2)
    def constraint_function(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Function to optimize h_values using gradient descent
    def optimize_h_values(h_initial):
        h = np.copy(h_initial)
        learning_rate = 0.01
        max_iterations = 1000
        for _ in range(max_iterations):
            # Compute current C5
            current_c5 = evaluate_overlap(h)
            # If we're below the target, we're done
            if current_c5 <= 0.380876:
                return h, current_c5
            
            # Compute the gradient of the overlap integral with respect to h
            # We approximate the gradient numerically
            grad = np.zeros_like(h)
            for i in range(n_points):
                # Small perturbation
                h_perturbed = np.copy(h)
                h_perturbed[i] += 1e-5
                c5_perturbed = evaluate_overlap(h_perturbed)
                grad[i] = (c5_perturbed - current_c5) / 1e-5
            
            # Update h with gradient descent
            h -= learning_rate * grad
            
            # Ensure h stays within [0, 1]
            h = np.clip(h, 0, 1)
            
            # Ensure sum(h) == n_points / 2
            sum_h = np.sum(h)
            if sum_h != 0:
                h *= (n_points / 2) / sum_h
        
        return h, current_c5
    
    # Function to run the optimization
    def run_optimization():
        # Start with initial values
        h_initial = np.array(initial_h_values)
        best_h = h_initial
        best_c5 = evaluate_overlap(h_initial)
        
        # Try increasing the number of points to see if we can improve the result
        for n_points in [initial_n_points, initial_n_points * 2, initial_n_points * 3, initial_n_points * 4]:
            if time.time() - start_time > budget_s - 5:
                break
            # Initialize h with uniform distribution
            h = np.random.rand(n_points) * 2
            # Ensure sum(h) == n_points / 2
            h = h * (n_points / 2) / np.sum(h)
            # Optimize
            h_opt, c5_opt = optimize_h_values(h)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        
        return best_h, best_c5, n_points
    
    # Run the optimization
    h_best, c5_best, n_points_best = run_optimization()

    if time.time() - start_time < budget_s - 10:
        # Use CVXOPT for constrained optimization
        h = cp.Variable(n_points)
        # Objective: minimize C5
        # We use the current h_best as the initial guess
        h_initial = np.array(h_best)
        h = cp.Variable(n_points)
        obj = cp.Minimize(evaluate_overlap(h))
        constr = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        prob = cp.Problem(obj, constr)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                h_best = h.value
                c5_best = evaluate_overlap(h_best)
        except:
            pass
    
    # Return the best solution found
    return h_best, c5_best, n_points_best
