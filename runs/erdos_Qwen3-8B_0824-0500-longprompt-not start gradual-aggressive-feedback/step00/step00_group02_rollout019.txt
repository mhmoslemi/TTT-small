import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 45
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = (n_points / 2) / sum_h * h
    
    # Function to evaluate C5
    def evaluate_c5(h_vals):
        h_vals = np.array(h_vals)
        dx = 2.0 / len(h_vals)
        h = h_vals.reshape(-1, 1)
        one_minus_h = 1 - h
        correlate = signal.correlate(h, one_minus_h, mode="full")
        c5 = np.max(correlate) * dx
        return c5
    
    # Local search with simulated annealing
    def local_search(h_vals, max_iter=100, temp=1.0, cooling_rate=0.995, tol=1e-5):
        h_vals = h_vals.copy()
        best_h = h_vals.copy()
        best_c5 = evaluate_c5(best_h)
        for _ in range(max_iter):
            # Random perturbation
            i = np.random.randint(len(h_vals))
            delta = np.random.uniform(-0.1, 0.1)
            h_vals[i] = np.clip(h_vals[i] + delta, 0, 1)
            # Check if the sum constraint is maintained
            if np.abs(np.sum(h_vals) - len(h_vals) / 2) > 1e-5:
                h_vals = best_h.copy()
                continue
            current_c5 = evaluate_c5(h_vals)
            if current_c5 < best_c5:
                best_h = h_vals.copy()
                best_c5 = current_c5
            else:
                # Simulated annealing acceptance
                if np.random.random() < np.exp((current_c5 - best_c5) / temp):
                    best_h = h_vals.copy()
                    best_c5 = current_c5
            temp *= cooling_rate
            if abs(best_c5 - current_c5) < tol:
                break
        return best_h, best_c5
    
    # Perform local search
    h, c5 = local_search(h)
    best_h = h.copy()
    best_c5 = c5
    
    # Try multiple starting points
    for _ in range(5):
        # Random perturbation to the initial point
        h = initial_h_values.copy()
        dx = 2.0 / len(h)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, len(h) / 2):
            h = (len(h) / 2) / sum_h * h
        h += np.random.uniform(-0.1, 0.1, len(h))
        h = np.clip(h, 0, 1)
        h, c5 = local_search(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    # Check time remaining
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        # Try a global optimization approach using CVXPY for a few iterations
        n_points = 200
        dx = 2.0 / n_points
        x = cp.Variable(n_points)
        objective = cp.Minimize(cp.max(signal.correlate(x, 1 - x, mode="full") * dx))
        constraints = [cp.sum(x) == n_points / 2, 0 <= x, x <= 1]
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, eps=1e-4, max_iter=100)
            if prob.status == 'optimal':
                best_h = x.value
                best_c5 = prob.objective.value
        except:
            pass
    
    # Final check
    if best_c5 < 0.380876:
        return best_h, best_c5, len(best_h)
    else:
        return best_h, best_c5, len(best_h)
