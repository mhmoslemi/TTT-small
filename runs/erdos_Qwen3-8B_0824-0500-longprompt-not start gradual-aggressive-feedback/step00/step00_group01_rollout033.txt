import numpy as np

def run(seed=42, budget_s=60, **kwargs):
    # Set a random seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Set up the problem
    n_points = 100  # Start with 100 points (can be increased if time allows)
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Initialize with a uniform distribution to satisfy the integral constraint
    h[:] = 0.5
    # Start with the initial solution
    if hasattr(initial_h_values, "__len__") and len(initial_h_values) == n_points:
        h = np.array(initial_h_values, dtype=np.float64)
    
    # Ensure the integral constraint is met
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = np.interp(h, np.linspace(0, 2, n_points), np.linspace(0, 2, n_points))
        h = np.array(initial_h_values, dtype=np.float64)
        sum_h = np.sum(h)
    
    # Define the function to compute the maximum overlap
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_full = np.zeros(2 * n_points - 1)
        h_full[:n_points] = h
        h_full[n_points:] = 1 - h
        corr = np.correlate(h, h_full, mode='full')
        dx = 2.0 / n_points
        c5 = np.max(corr * dx)
        return c5
    
    # Define the objective function for optimization
    def objective_and_gradient(h_vec):
        h = h_vec.reshape(n_points)
        c5 = compute_c5(h)
        gradient = np.zeros(n_points)
        # Compute the gradient numerically
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            c5_plus = compute_c5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            c5_minus = compute_c5(h_minus)
            gradient[i] = (c5_plus - c5_minus) / (2 * 1e-6)
        return c5, gradient.flatten()
    
    # Optimizer settings
    max_iter = 100
    tol = 1e-5
    
    # Perform optimization
    h_opt = h.copy()
    best_c5 = compute_c5(h_opt)
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Use gradient descent
        c5, grad = objective_and_gradient(h_opt)
        if np.max(np.abs(grad)) < tol:
            break
        
        # Take a step in the direction opposite to the gradient
        step_size = 0.1 / np.max(np.abs(grad))
        h_opt -= step_size * grad
        
        # Clip values to [0, 1]
        h_opt = np.clip(h_opt, 0, 1)
        
        # Ensure the integral constraint is satisfied
        sum_h = np.sum(h_opt)
        if not np.isclose(sum_h, n_points / 2):
            h_opt = np.interp(h_opt, np.linspace(0, 2, n_points), np.linspace(0, 2, n_points))
        
        # Update best value
        current_c5 = compute_c5(h_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            h_opt = h_opt.copy()

    if time.time() - start_time < budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        if hasattr(initial_h_values, "__len__") and len(initial_h_values) == n_points:
            h = np.array(initial_h_values, dtype=np.float64)
        else:
            h = np.array([0.5 for _ in range(n_points)])
        
        # Ensure the integral constraint is met
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            h = np.interp(h, np.linspace(0, 2, n_points), np.linspace(0, 2, n_points))
        
        # Optimization with higher points
        h_opt = h.copy()
        best_c5 = compute_c5(h_opt)
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            
            c5, grad = objective_and_gradient(h_opt)
            if np.max(np.abs(grad)) < tol:
                break
            
            step_size = 0.1 / np.max(np.abs(grad))
            h_opt -= step_size * grad
            h_opt = np.clip(h_opt, 0, 1)
            sum_h = np.sum(h_opt)
            if not np.isclose(sum_h, n_points / 2):
                h_opt = np.interp(h_opt, np.linspace(0, 2, n_points), np.linspace(0, 2, n_points))
            
            current_c5 = compute_c5(h_opt)
            if current_c5 < best_c5:
                best_c5 = current_c5
                h_opt = h_opt.copy()
    
    # Final evaluation
    final_c5 = compute_c5(h_opt)
    return h_opt.tolist(), final_c5, n_points
