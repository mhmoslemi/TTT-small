import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))
initial_h_values = np.random.rand(95) * 2 - 1  # Placeholder for actual initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial guess
    n_points = 100
    h_initial = np.zeros(n_points)
    h_initial[0:n_points//2] = 1.0
    h_initial = (h_initial + 1.0) / 2.0  # Normalize to [0,1]
    
    # Convert to numpy array
    h = np.array(h_initial, dtype=np.float64)
    
    # Function to evaluate C5
    def evaluate_C5(h):
        h = np.clip(h, 0.0, 1.0)
        overlap = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * (2.0 / len(h))
        return c5
    
    # Function to evaluate and return C5
    def objective_func(x):
        h = x.copy()
        h = np.clip(h, 0.0, 1.0)
        # Enforce the integral constraint
        integral = np.sum(h) * (2.0 / len(h))
        # Scale to meet the constraint
        scale = 1.0 / integral if integral != 0 else 1.0
        h_scaled = h * scale
        # Evaluate C5
        overlap = signal.correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(overlap) * (2.0 / len(h_scaled))
        return c5
    
    # Optimization settings
    max_iter = 100
    tol = 1e-5
    
    # Simulated annealing-like optimization
    for _ in range(max_iter):
        # Perturb the current solution
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturb
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        # Enforce the integral constraint
        integral = np.sum(h_perturbed) * (2.0 / n_points)
        scale = 1.0 / integral if integral != 0 else 1.0
        h_scaled = h_perturbed * scale
        # Evaluate C5
        c5_perturbed = evaluate_C5(h_scaled)
        # Compare to current C5
        c5_current = evaluate_C5(h)
        # Accept the perturbation if it improves the solution
        if c5_perturbed < c5_current:
            h = h_scaled
    
    # If time allows, run a local optimization
    if time.time() - start_time < budget_s - 10:
        # Use CVXPY to minimize the maximum overlap
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        # Objective is to minimize the maximum overlap
        # Since we can't directly minimize the max, we reformulate it
        # We create a variable to represent the maximum overlap
        max_overlap = cp.Variable()
        constraints += [
            max_overlap >= signal.correlate(x, 1 - x, mode='full')[i] * (2.0 / n_points)
            for i in range(len(signal.correlate(x, 1 - x, mode='full')))
        ]
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.SCS, verbose=False)
            if result is not None:
                h = np.clip(x.value, 0.0, 1.0)
        except:
            pass
    
    # Final evaluation
    c5_bound = evaluate_C5(h)
    return h, c5_bound, n_points
