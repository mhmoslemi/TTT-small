import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=60, **kwargs):
    n_points = 100
    dx = 2.0 / n_points
    h_initial = np.array(initial_h_values) if hasattr(initial_h_values, '__iter__') else np.random.rand(n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)  # Ensure total integral is 1

    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)

    end_time = time.time() + budget_s

    while time.time() < end_time:
        # Define objective and constraints
        h = cp.Variable(n_points)
        constraints = [cp.sum(h) == n_points / 2]
        for k in range(n_points):
            overlap = cp.sum(cp.multiply(h, 1 - cp.roll(h, k)))
            constraints.append(overlap <= best_c5)

        # Objective: minimize the maximum overlap
        objective = cp.Minimize(best_c5)

        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve()
            if result is not None and result < best_c5:
                best_c5 = result
                best_h = h.value
        except:
            pass

    return best_h, best_c5, n_points
