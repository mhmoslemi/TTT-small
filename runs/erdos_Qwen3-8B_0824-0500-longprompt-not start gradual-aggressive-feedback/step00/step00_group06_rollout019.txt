```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Example initial h_values for n=95

def run(seed=42, budget_s=60, **kwargs):
    # Define parameters
    n_points = 100  # Start with 100 points, can be adjusted if needed
    dx = 2.0 / n_points
    max_iterations = 100  # Limit iterations to stay within budget

    # Initial guess
    h = np.zeros(n_points)
    if initial_h_values.size == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.sort(h)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)

    # Ensure the integral condition is satisfied
    total_area = np.sum(h) * dx
    h = h * (1 / total_area) * (n_points / 2)

    # Optimization loop
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    start_time = time.time()

    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Define the objective
        def objective(x):
            h_current = x.copy()
            h_current = np.clip(h_current, 0, 1)
            c5 = evaluate_erdos_solution(h_current)
            return c5

        # Define constraints
        def constraints(x):
            return np.sum(x) - (n_points / 2)

        # Use scipy's optimization to find better h
        result = optimize.minimize(
            objective,
            h,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP"
        )

        # Update best solution
        if result.success:
            h = result.x
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
```
