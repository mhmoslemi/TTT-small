import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from the provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h) * dx
    if abs(sum_h - 1.0) > 1e-6:
        initial_h = (initial_h * (1.0 / sum_h)) * 1.0
    
    # Evaluate initial C5
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h_shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            h_shifted[k:] = h[:-k]
        overlap = np.correlate(h, 1.0 - h_shifted, mode="full")
        c5 = np.max(overlap * dx)
        return c5

    best_h = initial_h.copy()
    best_c5 = compute_c5(initial_h)
    best_n = n_points
    
    # Try lower n_points
    for n in range(20, 60):
        if time.time() - start_time > budget_s - 1:
            break
        h = np.random.rand(n) * 1.0
        total = h.sum()
        h = h * (1.0 / total) * 1.0  # Normalize to ensure integral is 1
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_n = n
    
    # Refine with optimization
    n_refine = 100
    h_refine = np.random.rand(n_refine) * 1.0
    total = h_refine.sum()
    h_refine = h_refine * (1.0 / total) * 1.0  # Normalize to ensure integral is 1
    c5_refine = compute_c5(h_refine)
    if c5_refine < best_c5:
        best_h = h_refine.copy()
        best_c5 = c5_refine
        best_n = n_refine
    
    # Final optimization with CVXOPT
    start_time = time.time()
    n_opt = 100
    h_opt = np.random.rand(n_opt) * 1.0
    total = h_opt.sum()
    h_opt = h_opt * (1.0 / total) * 1.0
    
    # Define the optimization problem
    h = cp.Variable(n_opt)
    constraints = [h >= 0, h <= 1, cp.sum(h) == n_opt / 2]
    
    # Objective: minimize the maximum overlap
    def objective(h_values):
        h_values = np.array(h_values)
        dx = 2.0 / len(h_values)
        h_shifted = np.zeros_like(h_values)
        for k in range(1, len(h_values) - 1):
            h_shifted[k:] = h_values[:-k]
        overlap = np.correlate(h_values, 1.0 - h_shifted, mode="full")
        return np.max(overlap * dx)
    
    # Use a gradient-based optimizer
    result = optimize.minimize(objective, h_opt, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_opt)])
    h_opt = result.x
    c5_opt = compute_c5(h_opt)
    if c5_opt < best_c5:
        best_h = h_opt.copy()
        best_c5 = c5_opt
        best_n = n_opt
    
    return (best_h, best_c5, best_n)
