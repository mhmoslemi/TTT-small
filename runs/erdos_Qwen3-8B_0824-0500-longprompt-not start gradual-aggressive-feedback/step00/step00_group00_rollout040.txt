# Strategy:
# 1. Start with an initial guess (e.g., uniform or random).
# 2. Use a gradient-based optimizer (e.g., BFGS) to minimize the maximum overlap integral.
# 3. Use a discretization of the function h with n_points samples.
# 4. Evaluate the overlap for all possible shifts and take the maximum.
# 5. Use box constraints [0, 1] for h[i].
# 6. Limit the number of samples to under 1000 for speed.
# 7. Use the current best solution as a starting point for optimization.

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = 100  # Reduce number of samples for speed
    dx = 2.0 / n_points
    
    # Initial guess: uniform distribution
    h = np.random.rand(n_points)
    h = h * (n_points / 2) / np.sum(h)  # Ensure the integral is 1
    
    # Objective function: maximize the overlap integral
    def objective(x):
        h_vals = x.reshape(-1)
        max_overlap = 0.0
        for k in range(1, n_points):
            shifted_h = np.zeros_like(h_vals)
            shifted_h[k:] = h_vals[:-k]
            overlap = np.dot(h_vals, 1 - shifted_h)
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap * dx
    
    # Constraints: sum(h) == n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 1000}
    )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun
    
    # Evaluate to ensure bounds are respected
    if np.any(best_h < 0) or np.any(best_h > 1):
        best_h = np.clip(best_h, 0, 1)
        best_c5 = evaluate_erdos_solution(best_h)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Time budget exceeded")
    
    return (best_h, best_c5, n_points)
