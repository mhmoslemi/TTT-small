```python
import numpy as np
import scipy.optimize as optimize
import time
import functools

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided starting point
    n_points = len(initial_h_values)
    h_values = initial_h_values.copy()
    
    # Evaluate initial solution
    def objective(h):
        h = np.array(h)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Optimization function to minimize C5
    def optimize_h(h_values):
        result = optimize.minimize(
            objective,
            h_values,
            method="L-BFGS-B",
            bounds=[(0, 1) for _ in range(n_points)],
            options={"maxiter": 100, "ftol": 1e-6}
        )
        return result.x
    
    # Track best solution
    best_h = h_values
    best_c5 = objective(h_values)
    start_time = time.time()
    
    # Try optimizing with current initial values
    optimized_h = optimize_h(h_values)
    optimized_c5 = objective(optimized_h)
    if optimized_c5 < best_c5:
        best_h = optimized_h
        best_c5 = optimized_c5
    
    # Try small perturbations to escape local minima
    for _ in range(5):
        perturbed_h = best_h + np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_c5 = objective(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5
    
    # Evaluate the best solution
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 0.5:
        # Try another optimization run with current best
        optimized_h = optimize_h(best_h)
        optimized_c5 = objective(optimized_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    total_integral = np.sum(best_h) * (2.0 / n_points)
    if not np.isclose(total_integral, 1.0, atol=1e-5):
        # Force normalization if needed
        scale = 1.0 / (total_integral + 1e-10)
        best_h *= scale
    
    # Ensure the best_h is properly formatted for output
    return (best_h, best_c5, n_points)
```
