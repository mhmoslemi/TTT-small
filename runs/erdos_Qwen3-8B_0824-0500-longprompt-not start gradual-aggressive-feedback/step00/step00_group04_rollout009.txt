import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
import sys

# Pre-imported functions
# These are assumed to be available as global variables
# evaluate_erdos_solution = ...  # Function to evaluate the C5 value
# initial_h_values = ...        # Initial guess for h_values

def run(seed=42, budget_s=60, **kwargs):
    # Define the parameters
    n_points = 100  # Initial number of points
    dx = 2.0 / n_points
    max_iter = 100  # Maximum number of iterations
    tolerance = 1e-4  # Tolerance for convergence

    # Set a seed for reproducibility (optional)
    np.random.seed(seed)

    # Initial guess based on the provided initial_h_values or a random one
    if hasattr(initial_h_values, "__len__") and len(initial_h_values) == n_points:
        h_values = np.array(initial_h_values, dtype=np.float64)
    else:
        h_values = np.random.rand(n_points)
        h_values = h_values * (n_points / 2)  # Ensure sum(h_values) * dx ≈ 1

    # Enforce the constraint sum(h_values) = n_points / 2
    h_values = h_values / np.sum(h_values) * (n_points / 2)

    best_c5 = np.inf
    best_h = h_values.copy()

    start_time = time.time()

    # Define the objective function to minimize the maximum overlap
    def objective(h):
        h = h.copy()
        # Ensure values are within [0, 1]
        h = np.clip(h, 0, 1)
        # Enforce the area constraint
        area = np.sum(h) * dx
        if abs(area - 1) > 1e-6:
            return np.inf  # Penalize for not meeting the area constraint
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)
        return c5

    # Define the bounds for the optimization variables
    bounds = Bounds(0, 1)

    # Optimization loop
    for _ in range(max_iter):
        # Check if we have run out of time
        if time.time() - start_time > budget_s - 1:
            break

        # Perform optimization step
        result = optimize.minimize(
            objective,
            h_values,
            method='L-BFGS-B',
            bounds=bounds,
            jac=None,
            options={'maxiter': 10, 'ftol': 1e-6}
        )

        # Update the best solution if improved
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x.copy()

        # Update the current guess
        h_values = result.x.copy()

        # Early stopping if convergence is achieved
        if abs(best_c5 - np.inf) < tolerance:
            break

    # Return the best solution found
    return (best_h, best_c5, n_points)
