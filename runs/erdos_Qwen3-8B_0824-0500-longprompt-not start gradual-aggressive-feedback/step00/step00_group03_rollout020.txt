```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assuming this file exists with the initial construction


def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with a reasonable number of points
    
    # Try different numbers of points
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s:
            break
        
        # Generate an initial guess based on the provided initial_h_values
        if n_points < len(initial_h_values):
            h = np.zeros(n_points)
            h[:n_points] = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        else:
            # If we need more points, interpolate or generate a new initial guess
            h = np.random.uniform(0, 1, n_points)
            # Enforce the integral constraint
            h = h * (n_points / 2) / np.sum(h)
        
        # Optimization with CVXPY
        try:
            # Define variables and constraints
            h_var = cp.Variable(n_points, name='h')
            constraints = [
                0 <= h_var,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            
            # Objective is to minimize the maximum overlap integral
            # We use a convex relaxation by minimizing the maximum of the correlations
            # This is an approximation due to the non-convexity of the max function
            # Evaluate all shifted versions of h and find the maximum
            max_overlap = 0
            for k in range(1, n_points):
                if k >= n_points:
                    break
                shift = np.roll(h_var, k)
                correlation = cp.multiply(h_var, 1 - shift)
                overlap = cp.sum(correlation)
                max_overlap = cp.Max(max_overlap, overlap)
            
            # Define the objective
            objective = cp.Minimize(max_overlap)
            
            # Solve the optimization problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, eps=1e-6, verbose=False)
            
            if prob.status == 'optimal':
                h_opt = h_var.value
                c5 = evaluate_erdos_solution(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
                    print(f"Improved C5: {best_c5:.6f} with n={n_points}")
        except:
            pass
    
    # After trying all configurations, return the best result
    return best_h, best_c5, len(best_h)
```
