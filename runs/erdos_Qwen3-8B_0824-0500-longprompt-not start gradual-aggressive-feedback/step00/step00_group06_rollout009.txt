import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from the provided initial_h_values
    initial_h_values = evaluate_erdos_solution()  # Assuming this is a function that returns the initial h_values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    
    # Ensure the initial sum is correct
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h_values = np.array(initial_h_values, dtype=float)
        initial_h_values = initial_h_values / (sum_h / (n_points / 2))
    
    def objective_function(h_values):
        h = h_values.reshape(-1)
        # Compute the overlap integral for all k
        h_ext = np.concatenate([h, np.zeros_like(h)])
        corr = np.correlate(h_ext, 1 - h, mode="full")
        # The overlap integral for each k is corr[k] * dx, but we are only interested in k from 1 to n_points-1
        c5_values = corr[1:n_points] * dx
        return np.max(c5_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Define bounds for h_values (0 to 1)
    bounds = Bounds(0, 1)
    
    # Set up optimization
    h_initial = initial_h_values.copy()
    result = optimize.minimize(
        objective_function,
        h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        tol=1e-8,
        options={'maxiter': 100, 'disp': False}
    )
    
    # Extract the best solution found
    best_h_values = result.x
    best_c5 = objective_function(best_h_values)
    
    # Ensure the best_h_values are within bounds and the sum constraint is satisfied
    best_h_values = np.clip(best_h_values, 0, 1)
    if abs(np.sum(best_h_values) - n_points / 2) > 1e-6:
        best_h_values = best_h_values / (np.sum(best_h_values) / (n_points / 2))
    
    # Evaluate the final c5 value
    final_c5 = objective_function(best_h_values)
