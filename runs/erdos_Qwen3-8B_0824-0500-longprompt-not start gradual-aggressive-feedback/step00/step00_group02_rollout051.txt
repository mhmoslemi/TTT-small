```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial parameters
    n_points = 45  # Start with the initial value
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:1]])  # Pad for full convolution
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5
    
    # Function to compute the objective and constraints for optimization
    def objective_and_constraints(h_values):
        h = np.array(h_values)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            return np.inf  # Penalize for not meeting the integral constraint
        
        c5 = compute_c5(h)
        return c5
    
    # Function to optimize h using local search
    def local_search(h_start, max_iter=100, max_time=30):
        best_h = h_start.copy()
        best_c5 = compute_c5(best_h)
        best_h_vals = best_h.copy()
        
        for _ in range(max_iter):
            # Perturb the current solution
            perturbation = np.random.uniform(-0.1, 0.1, size=len(best_h))
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_h = np.round(perturbed_h, 5)  # Reduce numerical error
            
            # Check constraint
            sum_perturbed = np.sum(perturbed_h)
            if not np.isclose(sum_perturbed, n_points / 2):
                continue
            
            c5 = compute_c5(perturbed_h)
            if c5 < best_c5:
                best_h = perturbed_h.copy()
                best_c5 = c5
                best_h_vals = best_h.copy()
        
        return best_h_vals, best_c5
    
    # Function to perform hill climbing
    def hill_climbing(h_start, max_iter=100, max_time=30):
        best_h = h_start.copy()
        best_c5 = compute_c5(best_h)
        best_h_vals = best_h.copy()
        
        for _ in range(max_iter):
            # Perturb the current solution
            perturbation = np.random.uniform(-0.1, 0.1, size=len(best_h))
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_h = np.round(perturbed_h, 5)
            
            # Check constraint
            sum_perturbed = np.sum(perturbed_h)
            if not np.isclose(sum_perturbed, n_points / 2):
                continue
            
            c5 = compute_c5(perturbed_h)
            if c5 < best_c5:
                best_h = perturbed_h.copy()
                best_c5 = c5
                best_h_vals = best_h.copy()
        
        return best_h_vals, best_c5
    
    # Function to improve h using convex optimization
    def convex_optimizer(h_start):
        h = cp.Variable(len(h_start))
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        objective = cp.Minimize(cp.Max(cp.correlate(h, 1 - h, mode="full") * dx))
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=1000)
            if prob.status == 'optimal':
                return h.value.tolist()
            else:
                return h_start.copy()
        except:
            return h_start.copy()
    
    # Function to perform adaptive search
    def adaptive_search(h_start, max_iter=100, max_time=30):
        best_h = h_start.copy()
        best_c5 = compute_c5(best_h)
        best_h_vals = best_h.copy()
        
        for _ in range(max_iter):
            # Perturb the current solution
            perturbation = np.random.uniform(-0.1, 0.1, size=len(best_h))
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_h = np.round(perturbed_h, 5)
            
            # Check constraint
            sum_perturbed = np.sum(perturbed_h)
            if not np.isclose(sum_perturbed, n_points / 2):
                continue
            
            c5 = compute_c5(perturbed_h)
            if c5 < best_c5:
                best_h = perturbed_h.copy()
                best_c5 = c5
                best_h_vals = best_h.copy()
        
        return best_h_vals, best_c5
    
    # Function to perform simulated annealing
    def simulated_annealing(h_start, max_iter=100, max_time=30):
        T = 1.0  # Initial temperature
        cooling_rate = 0.995
        best_h = h_start.copy()
        best_c5 = compute_c5(best_h)
        best_h_vals = best_h.copy()
        
        for _ in range(max_iter):
            # Perturb the current solution
            perturbation = np.random.uniform(-0.1, 0.1, size=len(best_h))
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_h = np.round(perturbed_h, 5)
            
            # Check constraint
            sum_perturbed = np.sum(perturbed_h)
            if not np.isclose(sum_perturbed, n_points / 2):
                continue
            
            c5 = compute_c5(perturbed_h)
            if c5 < best_c5 or np.random.rand() < np.exp((c5 - best_c5) / T):
                best_h = perturbed_h.copy()
                best_c5 = c5
                best_h_vals = best_h.copy()
            
            T *= cooling_rate
        
        return best_h_vals, best_c5
    
    # Main optimization loop
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h)
    best_h_vals = best_h.copy()
    remaining_time = budget_s - (time.time() - start_time)
    
    # Run hill climbing
    if remaining_time > 5:
        best_h_vals, best_c5 = hill_climbing(best_h, max_iter=50, max_time=remaining_time - 5)
        remaining_time -= 5
    
    # Run simulated annealing
    if remaining_time > 5:
        best_h_vals, best_c5 = simulated_annealing(best_h_vals, max_iter=50, max_time=remaining_time - 5)
        remaining_time -= 5
    
    # Run convex optimization
    if remaining_time > 5:
        best_h_vals = convex_optimizer(best_h_vals)
        best_c5 = compute_c5(best_h_vals)
        remaining_time -= 5
    
    # Run local search
    if remaining_time > 5:
        best_h_vals, best_c5 = local_search(best_h_vals, max_iter=50, max_time=remaining_time - 5)
        remaining_time -= 5
    
    # Return the best solution found
    return (best_h_vals, best_c5, n_points)
```
