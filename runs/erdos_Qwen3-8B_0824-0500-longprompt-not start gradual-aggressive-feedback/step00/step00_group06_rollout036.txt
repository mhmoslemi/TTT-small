import numpy as np

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    h = np.copy(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Define the objective function for optimization
    def objective(h_values):
        nonlocal best_c5, best_h
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values.copy()
        return c5
    
    # Optimization method: Use coordinate descent with random restarts
    for _ in range(5):  # 5 random restarts
        if time.time() - start_time > budget_s - 1:
            break
        
        # Random perturbation of the initial configuration
        h = np.copy(best_h)
        for i in range(len(h)):
            if np.random.rand() < 0.1:
                h[i] = np.random.uniform(0, 1)
        
        # Ensure sum(h) is approximately n_points / 2
        total = np.sum(h)
        if total > 0:
            h = h * (n_points / 2) / total
        
        # Optimization using coordinate descent
        for _ in range(100):  # 100 iterations
            if time.time() - start_time > budget_s - 1:
                break
            
            # Evaluate current solution
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # For each position, try adjusting the value
            for i in range(len(h)):
                if np.random.rand() < 0.1:  # Randomly choose a position to update
                    # Try increasing h[i]
                    h_new = h.copy()
                    h_new[i] = min(1.0, h_new[i] + 0.05)
                    total = np.sum(h_new)
                    if total > 0:
                        h_new = h_new * (n_points / 2) / total
                    c5_new = evaluate_erdos_solution(h_new)
                    if c5_new < current_c5:
                        h = h_new
                        current_c5 = c5_new
                    # Try decreasing h[i]
                    h_new = h.copy()
                    h_new[i] = max(0.0, h_new[i] - 0.05)
                    total = np.sum(h_new)
                    if total > 0:
                        h_new = h_new * (n_points / 2) / total
                    c5_new = evaluate_erdos_solution(h_new)
                    if c5_new < current_c5:
                        h = h_new
                        current_c5 = c5_new
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, len(best_h)
