```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions and initial construction
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load("initial_h_values.npy")  # Assuming this is loaded with n_points = 95

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different number of points
    n_points_options = [200, 300, 400]
    best_c5 = float('inf')
    best_h_values = None
    
    for n_points in n_points_options:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess based on initial_h_values or random
        if n_points < len(initial_h_values):
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        else:
            h = np.random.rand(n_points)
        
        # Ensure sum(h) == n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Optimization using coordinate descent
        for _ in range(200):
            if time.time() - start_time > budget_s:
                break
            # Evaluate C5
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h_values = h.copy()
            
            # Compute shifts and update h
            shifts = np.arange(-1, 2, 0.1)  # Adjust based on n_points
            for shift in shifts:
                if shift == 0:
                    continue
                h_shifted = np.roll(h, int(shift * n_points / 2))
                overlap = np.sum(h * (1 - h_shifted))
                if overlap > 0:
                    h = h * (1 - overlap / np.sum(h)) + h_shifted * (overlap / np.sum(h_shifted))
            
            # Normalize to ensure sum(h) == n_points / 2
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h_values)
    return best_h_values, final_c5, len(best_h_values)
```
