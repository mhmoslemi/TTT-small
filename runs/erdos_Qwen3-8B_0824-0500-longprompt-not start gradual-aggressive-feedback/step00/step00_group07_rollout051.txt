import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    
    # Initial guess from the provided initial_h_values
    # Ensure it's scaled to the correct number of points
    initial_h = initial_h_values.copy()
    if len(initial_h) != n_points:
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    
    # Ensure the integral constraint is satisfied
    sum_h = np.sum(initial_h) * dx
    if abs(sum_h - 1) > 1e-6:
        # Scale to satisfy the constraint
        initial_h = initial_h * (1 / sum_h)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = np.repeat(h, 2)  # Extend for convolution
        one_minus_h = 1 - h
        correlation = signal.correlate(h, one_minus_h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Objective function for optimization
    def objective(h):
        return compute_c5(h)
    
    # Optimization using gradient-based method
    # We'll use a modified version of the initial guess
    # We'll optimize using a simplex method for robustness
    # Initial guess
    h_guess = initial_h.copy()
    
    # Optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perform the optimization
    result = opt.minimize(
        objective,
        h_guess,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # If optimization failed, fall back to the initial guess
    if result.success:
        best_h = result.x
    else:
        best_h = initial_h
    
    # Evaluate the best solution
    best_c5 = compute_c5(best_h)
    
    # Check if we can improve with more points, if time permits
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        # Reconstruct initial guess with new points
        initial_h = initial_h_values.copy()
        if len(initial_h) != n_points:
            initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        sum_h = np.sum(initial_h) * dx
        if abs(sum_h - 1) > 1e-6:
            initial_h = initial_h * (1 / sum_h)
        
        # Re-define objective and bounds
        def compute_c5_new(h):
            h = np.clip(h, 0, 1)
            h = np.repeat(h, 2)
            one_minus_h = 1 - h
            correlation = signal.correlate(h, one_minus_h, mode="full")
            c5 = np.max(correlation) * dx
            return c5
        
        def objective_new(h):
            return compute_c5_new(h)
        
        bounds_new = [(0, 1) for _ in range(n_points)]
        
        # Optimize again with more points
        result_new = opt.minimize(
            objective_new,
            initial_h,
            method='L-BFGS-B',
            bounds=bounds_new,
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        # Update best solution
        if result_new.success:
            best_h = result_new.x
            best_c5 = compute_c5_new(best_h)
    
    return (best_h, best_c5, n_points)
