import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points (as per initial_h_values)
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = np.array(initial_h_values, dtype=float)
    
    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:1]])  # pad to allow full convolution
        shifted = np.zeros_like(h)
        for k in range(1, len(h) - 1):
            shifted[k] = 1 - h[k]
        conv = np.correlate(h, shifted, mode="full")
        conv = conv[1:-1]  # remove padding
        c5 = np.max(conv) * dx
        return c5
    
    def objective(h_vals):
        return evaluate_c5(h_vals)
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    # Initial guess
    initial_guess = h.copy()
    
    # Setup optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Optimization settings
    options = {
        'disp': False,
        'maxiter': 100,
        'ftol': 1e-8,
        'gtol': 1e-8,
        'xtol': 1e-8,
        'callback': lambda x: None
    }
    
    # Perform optimization
    start_time = time.time()
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options=options
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        # If time is up, return the best solution found so far
        best_h = initial_guess
        best_c5 = evaluate_c5(best_h)
    else:
        # Use the result if it improved
        best_h = result.x
        best_c5 = evaluate_c5(best_h)
    
    # Ensure the solution satisfies the sum constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    sum_h = np.sum(best_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        # If not satisfied, use initial guess
        best_h = initial_guess
        best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, n_points)
