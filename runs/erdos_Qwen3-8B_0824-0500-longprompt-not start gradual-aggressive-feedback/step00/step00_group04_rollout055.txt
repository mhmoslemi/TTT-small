    # Define the number of points
    n_points = 100  # A value less than 1k for efficiency
    dx = 2.0 / n_points
    initial_h = initial_h_values.copy()  # Start from the initial construction
    
    # Ensure the initial_h is of the correct length
    if len(initial_h) != n_points:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
    
    # Objective function to minimize
    def objective(h):
        # Ensure h is of the correct type
        h = np.array(h, dtype=float)
        # Constrain h to [0, 1]
        h = np.clip(h, 0.0, 1.0)
        # Ensure the integral constraint is satisfied
        integral = np.sum(h) * dx
        if abs(integral - 1.0) > 1e-6:
            return np.inf  # Penalize for not meeting the constraint
        
        # Compute h(x)(1 - h(x + k)) for all k
        # Use numpy correlate with mode 'full' to compute the overlap
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        
        # Find the maximum overlap
        c5 = np.max(overlap)
        return c5
    
    # Constraints: h(x) ∈ [0, 1], ∑h(x) * dx = 1
    # Since the integral is a constraint, we include it in the objective function with a penalty
    
    # Optimization settings
    # Use the initial guess
    h_initial = initial_h.copy()
    
    # Use L-BFGS-B for optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_c5 = objective(best_h)
    elapsed_time = time.time() - start_time
    
    # Final check: ensure the constraints are satisfied
    best_h = np.clip(best_h, 0.0, 1.0)
    integral = np.sum(best_h) * dx
    if abs(integral - 1.0) > 1e-6:
        best_c5 = np.inf  # Penalize for not meeting the constraint
    
    return (best_h, best_c5, n_points)
