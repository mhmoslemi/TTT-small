import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    # Computes C5 as the max of the overlap integrals
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute the cross-correlation with (1 - h_values)
    corr = correlate(h_values, 1 - h_values, mode="full")
    # The relevant part of the correlation corresponds to the overlap integrals
    # We are only interested in the overlap integrals for k = 0 to n_points-1
    # The correlation result for k = 0 to n_points-1 is from index n_points-1 to 2*n_points-2
    relevant_corr = corr[n_points-1:2*n_points-1]
    # Compute the max of the integrals (multiply by dx)
    c5 = np.max(relevant_corr * dx)
    return c5

def initial_h_values():
    # Returns the initial construction (n=47 samples)
    # This is the current best known construction for C₅
    # (values correspond to the initial solution with n=47)
    return np.array([
        0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000,
        0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000, 0.0000,
        0.0000, 0.0484, 0.0905, 0.1370, 0.1845, 0.2320, 0.2785, 0.3250, 0.3695,
        0.4120, 0.4545, 0.4970, 0.5395, 0.5810, 0.6225, 0.6625, 0.7025, 0.7425,
        0.7825, 0.8225, 0.8650, 0.9000, 0.9350, 0.9700, 1.0000, 1.0000, 1.0000,
        1.0000, 1.0000, 1.0000, 1.0000, 1.0000, 1.0000, 1.0000, 1.0000, 1.0000,
        1.0000, 1.0000, 1.0000, 1.0000, 1.0000, 1.0000, 1.0000, 1.0000, 1.0000
    ])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different numbers of points
    n_points_list = [64, 96, 128]  # Try different resolutions
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initial guess: start from the initial construction, scaled to n_points
        initial_h = initial_h_values()
        initial_h = np.resize(initial_h, n_points)
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2)
        initial_h = np.round(initial_h).astype(float)
        initial_h = initial_h / (n_points / 2)
        initial_h = np.clip(initial_h, 0, 1)

        # Define the variables
        h = cp.Variable(n_points, nonneg=True, boolean=True)

        # Objective: minimize the maximum overlap integral
        # This is a minimax optimization problem
        # We can use a convex relaxation approach
        # We define a variable to represent the maximum overlap
        max_overlap = cp.Variable()

        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]

        # Define the overlap integral for each k
        # We use correlation to find all the overlaps
        # For each k, compute the overlap integral as integral(h(x) * (1 - h(x + k)) dx)
        # We are to minimize the maximum value of these integrals

        # Define the objective: minimize max_overlap, subject to max_overlap >= overlap_integral for all k
        for k in range(n_points):
            # Compute the overlap integral for shift k
            # Since we are dealing with a grid, the overlap for k is equivalent to the sum over i of h[i] * (1 - h[(i + k) % n_points])
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            # Add constraint max_overlap >= overlap * dx
            constraints.append(max_overlap >= overlap * (2.0 / n_points))

        # Define the problem
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)

        # Solve the problem
        try:
            prob.solve(solver=cp.SCS, eps=1e-6, max_iters=1000)
        except:
            prob.solve(solver=cp.ECOS, eps=1e-6, max_iters=1000)

        if prob.status == 'optimal':
            h_values = h.value
            c5 = max_overlap.value
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values
                best_n = n_points

        # Optional: perform local improvement
        # Perform a few steps of local search
        for _ in range(5):
            if time.time() - start_time > budget_s:
                break
            # Perturb the current solution
            h_perturbed = h_values.copy()
            idx = np.random.randint(0, n_points)
            h_perturbed[idx] = np.random.uniform(0, 1)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2)
            h_perturbed = np.round(h_perturbed).astype(float)
            h_perturbed = h_perturbed / (n_points / 2)
            h_perturbed = np.clip(h_perturbed, 0, 1)

            # Compute the overlap for the perturbed solution
            c5_perturbed = evaluate_erdos_solution(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed
                best_n = n_points

    # If no solution found, return the initial solution
    if best_h is None:
        best_h = initial_h_values()
        best_h = np.resize(best_h, n_points)
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2)
        best_h = np.round(best_h).astype(float)
        best_h = best_h / (n_points / 2)
        best_h = np.clip(best_h, 0, 1)
        best_c5 = evaluate_erdos_solution(best_h)

    # Convert the best solution to a numpy array and return
    return (best_h, best_c5, best_n)
