import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    max_points = 1000  # Maximum number of points to use
    n_points = 100  # Start with a smaller number of points
    dx = 2.0 / n_points
    
    # Initial guess based on provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]  # Ensure it fits the n_points
    h_initial /= np.sum(h_initial) * dx  # Normalize to satisfy integral constraint
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        h = np.hstack([h, h[:1]])  # Extend to handle circular shifts
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(corr * dx)
        return overlap
    
    # Define a function to minimize overlap
    def objective_func(h_values):
        return evaluate_overlap(h_values)
    
    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run optimization
    result = optimize.minimize(
        objective_func,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        tol=1e-6,
        options={'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    
    # Optionally, try increasing the number of points if we have time
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5:
        n_points = 200
        dx = 2.0 / n_points
        h_initial = np.random.rand(n_points)
        h_initial /= np.sum(h_initial) * dx  # Normalize to satisfy integral constraint
        
        result = optimize.minimize(
            objective_func,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            tol=1e-6,
            options={'maxiter': 50}
        )
        
        best_h = result.x
        best_c5 = evaluate_overlap(best_h)
    
    # Final check: ensure integral constraint is satisfied
    integral = np.sum(best_h) * dx
    if np.abs(integral - 1) > 1e-6:
        print("Integral constraint not satisfied")
    
    return best_h, best_c5, n_points
