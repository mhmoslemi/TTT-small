```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is pre-loaded with n=95 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess based on provided initial_h_values
    n_points = len(initial_h_values)
    h = np.copy(initial_h_values)
    
    # Function to evaluate the C5 value
    def objective(h_values):
        h_values = h_values.astype(float)
        # Ensure values are in [0,1] and normalize
        h_values = np.clip(h_values, 0, 1)
        total = np.sum(h_values)
        if total == 0:
            return np.inf
        h_values = h_values * (n_points / 2) / total
        # Compute the maximum overlap
        overlap = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(overlap * (2.0 / n_points))
        return c5
    
    # Function to perform a gradient-based optimization
    def optimize_h():
        nonlocal h
        h = np.copy(h)
        # Initial optimization using BFGS
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=1e-5,
            options={'maxiter': 100, 'disp': False}
        )
        best_h = result.x
        best_c5 = objective(best_h)
        
        # Perform a few iterations of coordinate descent for refinement
        for _ in range(5):
            best_h = np.copy(best_h)
            for i in range(n_points):
                # For each point, try flipping it to 0 or 1
                for val in [0, 1]:
                    temp_h = np.copy(best_h)
                    temp_h[i] = val
                    # Ensure normalization
                    total = np.sum(temp_h)
                    if total == 0:
                        continue
                    temp_h = temp_h * (n_points / 2) / total
                    c5 = objective(temp_h)
                    if c5 < best_c5:
                        best_h = temp_h
                        best_c5 = c5
            if best_c5 >= objective(best_h):
                break
        
        h = best_h
        return best_c5, h
    
    best_c5, best_h = optimize_h()
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Perform additional refinement if time allows
        for _ in range(2):
            new_c5, new_h = optimize_h()
            if new_c5 < best_c5:
                best_c5, best_h = new_c5, new_h
            elapsed_time = time.time() - start_time
            if elapsed_time >= budget_s:
                break
    
    return best_h, best_c5, n_points
```
