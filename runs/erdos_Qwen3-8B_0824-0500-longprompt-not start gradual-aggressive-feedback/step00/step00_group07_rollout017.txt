from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess

    h_initial = initial_h_values.copy()
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    
    def evaluate_h(h):
        # Ensure h is valid
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            return float('inf')
        
        # Compute overlap integral
        h_pad = np.pad(h, (n_points, 0), 'constant')
        conv = correlate(h_pad, 1 - h, mode='full')
        max_overlap = np.max(conv[n_points:-n_points] * (2.0 / n_points))
        return max_overlap
    
    # Initial evaluation
    current_c5 = evaluate_h(h_initial)
    best_h = h_initial.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    
    # Optimization loop
    for n_points in [100, 200, 300, 400]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate a random perturbation
        h = h_initial.copy()
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h += perturbation
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            h = h_initial.copy()
        
        # Evaluate
        c5 = evaluate_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Fine-tune with gradient descent
    h = best_h.copy()
    learning_rate = 0.01
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute derivatives using finite differences
        h_plus = np.clip(h + 1e-5, 0, 1)
        h_minus = np.clip(h - 1e-5, 0, 1)
        c5_plus = evaluate_h(h_plus)
        c5_minus = evaluate_h(h_minus)
        grad = (c5_plus - c5_minus) / (2 * 1e-5)
        
        # Update h
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h != n_points / 2:
            h = best_h.copy()
        
        c5 = evaluate_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5, n_points
