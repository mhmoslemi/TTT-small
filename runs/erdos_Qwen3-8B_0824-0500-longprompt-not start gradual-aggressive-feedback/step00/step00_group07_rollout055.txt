```python
import numpy as np
import scipy.optimize as opt
import time
from functools import partial

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    n_points = 80
    h = initial_h_values.copy()
    h = h[:n_points]
    
    # Try with lower n_points to save time
    for n in [40, 32, 24, 16, 12, 8]:
        # Ensure the function is valid
        if np.sum(h) * 2 / n_points != 1:
            h = np.random.rand(n) * 2 - 1
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
        
        # Optimization
        def objective(x):
            x = x.reshape(n)
            h = x * (n / 2) / np.sum(x)
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * 2 / n
            return c5
        
        def constraint(x):
            x = x.reshape(n)
            return np.sum(x) - n / 2
        
        x0 = np.random.rand(n) * 2 - 1
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * (n / 2) / np.sum(x0)
        cons = {'type': 'eq', 'fun': constraint}
        bounds = [(0, 1) for _ in range(n)]
        
        try:
            result = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                h_candidate = result.x * (n / 2) / np.sum(result.x)
                c5_candidate = np.max(np.correlate(h_candidate, 1 - h_candidate, mode="full")) * 2 / n
                if c5_candidate < 0.380876:
                    h = h_candidate
                    n_points = n
                    break
        except:
            pass
        
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    c5_final = evaluate_erdos_solution(h)
    return h, c5_final, n_points
```
