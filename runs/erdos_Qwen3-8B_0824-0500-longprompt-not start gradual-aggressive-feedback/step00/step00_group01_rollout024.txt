import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assuming initial_h.npy is loaded with the initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Start with the initial construction
    n_points_initial = len(initial_h_values)
    h = initial_h_values.copy()
    
    # Function to compute the C5 value
    def compute_c5(h):
        return evaluate_erdos_solution(h)
    
    # Function to compute the integral of h
    def integral(h):
        return np.sum(h) * (2.0 / len(h))
    
    # Optimization parameters
    num_iterations = 100
    max_tries = 5
    best_c5 = float('inf')
    best_h = h.copy()
    n_points = n_points_initial
    best_n_points = n_points_initial
    
    # Try different numbers of points
    for n in [n_points_initial, n_points_initial // 2 + 1, n_points_initial * 2, n_points_initial * 3]:
        if n < 10:
            continue  # Skip too small n
        if n > 1000:
            continue  # Skip too large n to save time
        
        # Try multiple initializations
        for _ in range(max_tries):
            # Generate a random initial guess with the correct integral
            h_rand = np.random.rand(n)
            h_rand = h_rand / np.sum(h_rand) * (n / 2)
            h_rand = np.clip(h_rand, 0, 1)
            
            # Perform gradient descent to minimize C5
            h_current = h_rand.copy()
            learning_rate = 0.01
            for _ in range(num_iterations):
                # Compute the current C5 value
                current_c5 = compute_c5(h_current)
                
                # If current_c5 is better than best, update
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_current.copy()
                    best_n_points = n
                
                # Compute the derivative of C5 with respect to h
                # Using finite differences
                perturbation = 1e-6
                grad = np.zeros_like(h_current)
                for i in range(len(h_current)):
                    h_perturbed = h_current.copy()
                    h_perturbed[i] += perturbation
                    grad[i] = (compute_c5(h_perturbed) - current_c5) / perturbation
                
                # Update h_current using gradient descent
                h_current -= learning_rate * grad
                
                # Ensure h_current stays within [0, 1] and integral constraint is maintained
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current / np.sum(h_current) * (n / 2)
        
        # Check if this n gives a better result
        if compute_c5(best_h) < best_c5 and best_n_points == n:
            best_c5 = compute_c5(best_h)
    
    # Ensure we don't exceed the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return best_h, best_c5, best_n_points
    
    return best_h, best_c5, best_n_points
