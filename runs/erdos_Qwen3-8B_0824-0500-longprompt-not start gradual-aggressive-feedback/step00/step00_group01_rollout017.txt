        c5 = np.max(corr) * dx
        return c5

    # Optimization
    result = optimize.minimize(
        fun=objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],

    )

    best_h = result.x
    best_c5 = objective(best_h)
