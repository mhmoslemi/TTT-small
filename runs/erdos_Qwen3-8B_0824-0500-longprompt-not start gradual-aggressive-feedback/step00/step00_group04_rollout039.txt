import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions and variables
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.2443, 0.2443, 0.2443, 0.2443, 0.2443, 0.2443, 0.2443, 0.2443, 0.2443, 0.2443,

                             0.2443, 0.2443, 0.2443, 0.2443, 0.2443, 0.2443, 0.2443, 0.2443, 0.2443, 0.2443])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for faster computation
    dx = 2.0 / n_points
    h = np.random.rand(n_points) * 0.5  # Initial random guess
    h = h / np.sum(h) * n_points / 2  # Normalize to satisfy the integral constraint

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    # Try increasing n_points if time allows
    for n_points in [100, 200, 300, 400, 500, 600, 700, 800]:
        if time.time() - start_time > budget_s - 5:
            break
        dx = 2.0 / n_points
        h = np.random.rand(n_points) * 0.5
        h = h / np.sum(h) * n_points / 2

        # Use optimization to refine the solution
        x = cp.Variable(n_points)
        constraints = [
            0 <= x,
            x <= 1,
            cp.sum(x) == n_points / 2
        ]
        obj = cp.Minimize(cp.Max(cp.correlate(x.value, 1 - x.value, mode="full") * dx))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == 'optimal':
                h = x.value
                c5 = evaluate_erdos_solution(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            pass

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
