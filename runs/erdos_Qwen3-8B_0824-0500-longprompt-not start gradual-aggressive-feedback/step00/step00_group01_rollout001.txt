```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
import cvxpy as cp
import warnings

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

# Define the objective function to minimize C5
def objective_function(h_values, dx):
    # Compute the correlation
    correlation = np.correlate(h_values, 1 - h_values, mode="full")
    # Compute the max overlap integral
    c5 = np.max(correlation) * dx
    return c5

# Define the constraints function for optimization
def constraints_function(h_values, dx, n_points):
    # Constraint 1: The integral of h(x) over [0, 2] is 1
    integral = np.sum(h_values) * dx
    # Constraint 2: h(x) ∈ [0, 1]
    bounds = Bounds(0, 1)
    return integral, bounds

# Main function to run the optimization
def run(seed=42, budget_s=60, **kwargs):
    # Set the time limit
    start_time = time.time()
    # Initial number of points (start with 100)
    n_points = 100
    dx = 2.0 / n_points
    # Initial guess from the provided initial_h_values
    h_initial = initial_h_values
    # Resample to n_points
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    # Optimize
    result = minimize(
        fun=lambda h: objective_function(h, dx),
        x0=h_initial,
        bounds=Bounds(0, 1),
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-9, 'disp': False},
        tol=1e-8
    )
    # Extract the best solution
    h_best = result.x
    # Evaluate the C5 value
    c5_best = objective_function(h_best, dx)
    
    # If we have time, try with more points
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        dx = 2.0 / n_points
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        result = minimize(
            fun=lambda h: objective_function(h, dx),
            x0=h_initial,
            bounds=Bounds(0, 1),
            method='SLSQP',
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False},
            tol=1e-8
        )
        h_best = result.x
        c5_best = objective_function(h_best, dx)
    
    # Return the best solution
    return (h_best, c5_best, n_points)
```
