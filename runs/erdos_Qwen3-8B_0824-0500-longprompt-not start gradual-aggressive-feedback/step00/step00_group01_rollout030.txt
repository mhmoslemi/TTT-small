```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
# evaluate_erdos_solution = ...  # Not used in this implementation
# initial_h_values = ...  # Starting point for optimization

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different numbers of points
    n_points_list = [50, 65, 80, 100, 120]
    
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initialize with a random function satisfying the constraint
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        
        # Define variables for optimization
        h_var = cp.Variable(n_points, low=0, up=1)
        constraint = cp.sum(h_var) == n_points / 2
        
        # Evaluate the overlap integral for all shifts
        def objective(h_vals):
            h = h_vals
            max_overlap = 0.0
            for k in range(1, n_points):
                if k >= n_points:
                    continue
                overlap = np.sum(h * (1 - h[k:])) * dx
                if overlap > max_overlap:
                    max_overlap = overlap
            return max_overlap
        
        # Use a gradient-based optimization method
        result = optimize.minimize(
            fun=objective,
            x0=h,
            method='L-BFGS-B',
            options={'maxiter': 100}
        )
        
        h_candidate = result.x
        c5 = objective(h_candidate)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate
    
    # Convert best_h to a list for output
    best_h_list = best_h.tolist()
    
    return (best_h_list, best_c5, len(best_h_list))
```
