import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    h = initial_h_values.copy()  # Start with initial_h_values (n=45 samples)
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    h = np.clip(h, 0, 1)
    
    # Optimization function for C5
    def objective(x):
        h_values = x
        h_values = np.clip(h_values, 0, 1)
        # Compute the integral of h(x)(1 - h(x+k)) for all k
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        # The overlap integral is the correlation scaled by dx
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization settings
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 100})
    
    # Check if we're under time budget and consider increasing n_points
    elapsed_time = time.time() - time.time()
    if elapsed_time < budget_s:
        # Try with more points if time allows
        n_points = 200
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        # Ensure the integral constraint is satisfied
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 100})
    
    # Evaluate the final solution
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_c5 = objective(best_h)
    
    # Ensure the integral constraint is exactly satisfied
    total_integral = np.sum(best_h) * dx
    if abs(total_integral - 1) > 1e-6:
        # Adjust to satisfy the constraint
        scale = 1 / total_integral
        best_h = best_h * scale
    
    return best_h, best_c5, n_points
