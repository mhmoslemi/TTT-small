import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points values
    n_points_list = [256, 512, 1024, 2048]
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Compute dx
        dx = 2.0 / n_points
        
        # Initialize h with a good starting point
        h = np.array(initial_h_values, dtype=float)
        h = np.resize(h, (n_points,))
        
        # Ensure the sum is correct
        h_sum = np.sum(h)
        required_sum = n_points / 2
        if np.abs(h_sum - required_sum) > 1e-6:
            # Scale to meet the constraint
            h = h * required_sum / h_sum
        
        # Function to compute C₅
        def compute_c5(h_vals):
            h_vals = np.resize(h_vals, (n_points,))
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals) if np.sum(h_vals) != 0 else h_vals
            h_vals = np.clip(h_vals, 0, 1)
            
            # Compute the overlap integral
            h = h_vals.reshape((n_points, 1))
            one_minus_h = 1 - h_vals
            correlation = np.correlate(h, one_minus_h, mode='full')
            c5 = np.max(correlation) * dx
            
            return c5
        
        # Optimization: minimize C₅
        def objective(x):
            return compute_c5(x)
        
        # Bounds for h values
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Optimization with L-BFGS-B
        result = opt.minimize(
            objective, 
            h, 
            method='L-BFGS-B', 
            bounds=bounds,
            jac=None,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        # Evaluate the result
        best_h_current = result.x
        c5_current = compute_c5(best_h_current)
        
        # Update best solution
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = best_h_current.copy()
            best_n_points = n_points
        
        print(f"n_points={n_points}, C5={c5_current:.6f}")
    
    # Return the best solution found
    return (best_h, best_c5, best_n_points)
