import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 70
    h = np.array(initial_h_values)
    dx = 2.0 / n_points
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-6:
        h = h / h_sum * (n_points / 2)
    
    # Function to compute C5
    def compute_c5(h_values):
        h = h_values / (n_points / 2)
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, np.zeros(n_points)])
        result = np.correlate(h, 1 - h, mode="full")
        result = result[:n_points + 1]
        c5 = np.max(result * dx)
        return c5
    
    # Optimization function
    def optimize_h():
        nonlocal h
        h = h / (n_points / 2)
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, np.zeros(n_points)])
        result = np.correlate(h, 1 - h, mode="full")
        result = result[:n_points + 1]
        c5 = np.max(result * dx)
        return c5
    
    def objective(h_values):
        nonlocal h
        h = h_values
        return optimize_h()
    
    # Gradient descent with momentum
    def gradient_descent(h, max_iter=1000, learning_rate=0.1, momentum=0.9, tolerance=1e-4):
        h = h.copy()
        v = np.zeros_like(h)
        for _ in range(max_iter):
            grad = np.zeros_like(h)
            # Compute gradient numerically
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                c5_plus = optimize_h()
                h_minus = h.copy()
                h_minus[i] -= 1e-6
                c5_minus = optimize_h()
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-6)
            # Update velocity and position
            v = momentum * v - learning_rate * grad
            h += v
            h = np.clip(h, 0, 1)
            # Check for convergence
            if np.max(np.abs(grad)) < tolerance:
                break
        return h
    
    # Run optimization
    h_opt = gradient_descent(h, max_iter=1000, learning_rate=0.1, momentum=0.9, tolerance=1e-4)
    h = h_opt * (n_points / 2)
    
    # Refinement with local search
    for _ in range(50):
        idx = np.random.randint(0, n_points)
        h[idx] = np.random.uniform(0, 1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)
    
    # Final evaluation
    c5_value = compute_c5(h)
    
    # Ensure the sum constraint is satisfied
    total = np.sum(h) * dx
    if abs(total - 1) > 1e-6:
        h = h / total * 1
        h = h * (n_points / 2)
    
    # Final check
    c5_value = compute_c5(h)
    
    # Ensure we don't exceed the time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (h, c5_value, n_points)
    
    return (h, c5_value, n_points)
