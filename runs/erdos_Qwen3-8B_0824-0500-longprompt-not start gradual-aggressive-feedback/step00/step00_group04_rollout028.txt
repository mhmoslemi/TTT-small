import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Set initial number of points
    n_points = 42  # Start with the provided initial construction
    
    # Initial guess of h_values
    h_values = initial_h_values.copy()
    
    # Check constraint: sum(h_values) must be n_points / 2
    if not np.isclose(np.sum(h_values), n_points / 2):
        raise ValueError("Initial h_values does not satisfy the constraint")
    
    # Evaluate initial C_5
    c5_initial = evaluate_erdos_solution(h_values)
    best_c5 = c5_initial
    best_h_values = h_values.copy()
    
    # Define the objective function for optimization
    def objective_function(h_values, n_points):
        # Make a copy to avoid modifying the original
        h_values = h_values.copy()
        h_values = np.clip(h_values, 0, 1)
        
        # Ensure the sum constraint is satisfied
        if not np.isclose(np.sum(h_values), n_points / 2):
            # Penalize deviation from the constraint
            penalty = 1000 * np.abs(np.sum(h_values) - n_points / 2)
            return np.inf + penalty  # Return a large value to avoid this solution
        
        # Compute the overlap integral
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        h[:] = h_values
        shifted_h = np.zeros((n_points, n_points))
        for i in range(n_points):
            shifted_h[i] = np.roll(h, -i)
        
        # Compute 1 - h(x + k)
        ones_minus_h = 1 - h
        overlap = np.zeros(n_points)
        for k in range(n_points):
            overlap[k] = np.sum(h * ones_minus_h[k:]) * dx
        
        return np.max(overlap)
    
    # Define the optimization function
    def optimize_h_values(h_values, n_points, max_iterations=100):
        nonlocal best_c5, best_h_values
        for _ in range(max_iterations):
            start_time = time.time()
            
            # Define the optimization problem
            h = cp.Variable(n_points, name="h")
            constraint = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
            
            # Objective: minimize the maximum overlap integral
            dx = 2.0 / n_points
            h_val = cp.Parameter(n_points, value=h_values)
            ones_minus_h = 1 - h_val
            overlap = cp.Variable(n_points)
            
            for k in range(n_points):
                overlap[k] = cp.sum(h * ones_minus_h[k:])
            
            max_overlap = cp.Max(overlap)
            objective = cp.Minimize(max_overlap)
            
            # Solve the optimization problem
            prob = cp.Problem(objective, constraint)
            try:
                prob.solve(solver=cp.SCS, max_iter=1000, verbose=False)
            except:
                # If solver fails, try with another solver
                try:
                    prob.solve(solver=cp.CVXOPT, verbose=False)
                except:
                    prob.solve(solver=cp.GUROBI, verbose=False)
            
            if prob.status != 'OPTIMAL':
                break
            
            # Update best solution
            current_h = h.value
            current_c5 = objective.value * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h_values = current_h.copy()
            
            # Stop early if we've reached the budget
            if time.time() - start_time > budget_s:
                break
        
        return best_h_values, best_c5
    
    # Optimize the solution
    h_values, c5_bound = optimize_h_values(h_values, n_points)
    
    # Return the best solution found
    return (h_values, c5_bound, n_points)
