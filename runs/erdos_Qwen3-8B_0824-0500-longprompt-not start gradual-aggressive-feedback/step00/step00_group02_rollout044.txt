import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
import cvxpy as cp

# Pre-imported functions
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5], dtype=np.float64)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points, can increase if time permits
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Start with initial_h_values, scaled appropriately
    if n_points > len(initial_h_values):
        # Interpolate or extend if needed
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        h = initial_h_values[:n_points]
    
    # Normalize to satisfy integral constraint
    total = np.sum(h) * dx
    h = h * (1.0 / total) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    # Function to compute C5
    def compute_c5(h_values):
        h = h_values
        dx = 2.0 / len(h)
        # Create 1 - h
        one_minus_h = 1.0 - h
        # Perform full convolution
        conv = np.correlate(h, one_minus_h, mode='full')
        # Multiply by dx and find max
        c5 = np.max(conv) * dx
        return c5
    
    # Function to optimize h with constraints
    def optimize_h(h_initial, max_iterations=100, max_time=30):
        n = len(h_initial)
        dx = 2.0 / n
        
        # Objective: minimize the maximum overlap
        # We can't directly minimize a max function, so we use a minimax approach
        # Introduce a variable to bound the max
        h = cp.Variable(n)
        obj = cp.Minimize(cp.Max(cp.sum(cp.multiply(h, 1 - h[i]) for i in range(n))))
        
        # Constraints
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) * dx == 1
        ]
        
        # Solve with CVXOPT
        prob = cp.Problem(obj, constraints)
        try:
            result = prob.solve(solver=cp.CVXOPT, verbose=False)
            if result is not None:
                return h.value
            else:
                return h_initial
        except:
            return h_initial
    
    # Perform optimization with multiple restarts
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try several restarts
    for _ in range(3):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Perturb initial guess
        h = h.copy()
        perturb = np.random.uniform(-0.05, 0.05, size=n_points)
        h += perturb
        h = np.clip(h, 0, 1)
        
        # Normalize to satisfy integral constraint
        total = np.sum(h) * dx
        h = h * (1.0 / total) * (n_points / 2)
        h = np.clip(h, 0, 1)
        
        # Run optimization
        h_opt = optimize_h(h, max_iterations=50, max_time=20)
        h_opt = np.clip(h_opt, 0, 1)
        
        # Normalize again
        total = np.sum(h_opt) * dx
        h_opt = h_opt * (1.0 / total) * (n_points / 2)
        h_opt = np.clip(h_opt, 0, 1)
        
        # Compute C5
        current_c5 = compute_c5(h_opt)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_opt.copy()
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, n_points)
