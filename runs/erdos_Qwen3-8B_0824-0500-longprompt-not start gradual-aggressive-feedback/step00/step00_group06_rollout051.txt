import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions and initial solution
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Choose n_points based on the budget
    # For 60 seconds, we choose a relatively large n_points that allows optimization within the time
    n_points = 500  # Adjusted based on computational budget
    
    # Initialize h_values using the provided initial solution
    # If initial_h_values is not available, we use a simple uniform distribution
    if initial_h_values is not None:
        h_values = initial_h_values.copy()
    else:
        h_values = np.random.rand(n_points)
        h_values = (h_values * 2 - 1) * 0.5 + 0.5  # Normalize to [0,1]
    
    # Rescale h_values to satisfy the integral constraint
    dx = 2.0 / n_points
    total_integral = np.sum(h_values) * dx
    scaling_factor = 1.0 / total_integral
    h_values *= scaling_factor
    
    # Function to evaluate the overlap integral (C₅)
    def evaluate_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        # Compute 1 - h shifted by k for k = 1 to len(h)-1
        # Using numpy.correlate in 'full' mode to get all shifts
        corr = np.correlate(h, 1 - h, mode="full")
        # The relevant part corresponds to k = -len(h) + 1 to len(h) - 1
        # For our case, the relevant shifts are from 1 to len(h) - 1
        # But we'll take the maximum from all shifts
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Function to compute the objective for optimization
    def objective(h):
        return evaluate_c5(h)
    
    # Function to compute the constraint (integral constraint)
    def constraint(h):
        dx = 2.0 / len(h)
        return np.sum(h) * dx - 1.0
    
    # Optimization: Use genetic algorithm for global search
    # This is a heuristic approach given the complexity of the problem
    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    elite_size = 5
    
    # Genetic algorithm
    def genetic_algorithm(initial_population):
        population = initial_population.copy()
        best_solution = population[0]
        best_fitness = objective(best_solution)
        
        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective(ind) for ind in population]
            fitness = np.array(fitness)
            
            # Select elite solutions
            elite_indices = np.argsort(fitness)[:elite_size]
            elite = population[elite_indices]
            
            # Generate new population
            new_population = []
            for _ in range(population_size - elite_size):
                # Select two parents using tournament selection
                tournament = np.random.choice(len(population), size=2)
                parent1 = population[tournament[0]]
                parent2 = population[tournament[1]]
                if fitness[tournament[0]] < fitness[tournament[1]]:
                    parent1, parent2 = parent2, parent1
                
                # Crossover
                crossover_point = np.random.randint(1, len(parent1) - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                
                # Mutation
                for i in range(len(child)):
                    if np.random.rand() < mutation_rate:
                        child[i] = np.random.uniform(0, 1)
                
                # Ensure the constraint is satisfied (integral is 1)
                child_sum = np.sum(child)
                child = child * (1.0 / (child_sum * 2.0 / n_points)) if child_sum != 0 else child
                
                new_population.append(child)
            
            # Combine elite and new population
            population = np.vstack([elite, new_population])
            population = population[0:population_size]
            
            # Update best solution
            current_best = population[np.argmin(fitness)]
            if objective(current_best) < best_fitness:
                best_solution = current_best
                best_fitness = objective(current_best)
            
            # Check for time budget
            if time.time() - start_time > budget_s:
                break
        
        return best_solution, best_fitness
    
    # Initial population for genetic algorithm
    initial_population = []
    for _ in range(population_size):
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5  # Normalize to [0,1]
        h_sum = np.sum(h)
        h = h * (1.0 / (h_sum * 2.0 / n_points)) if h_sum != 0 else h
        initial_population.append(h)
    
    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(np.array(initial_population))
    
    # Evaluate the best solution
    best_c5 = evaluate_c5(best_h)
    
    # Return the result
    return (best_h, best_c5, n_points)
