```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define a function to compute C5 for given h_values
    def compute_C5(h_values, dx):
        h = np.array(h_values)
        shifted_h = np.zeros_like(h)
        for i in range(len(h)):
            shifted_h = np.roll(h, i)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > current_max:
                current_max = overlap
        return current_max
    
    # Initial setup
    n_points = 100  # Number of samples
    dx = 2.0 / n_points
    h_init = initial_h_values[:n_points]
    h_init = np.array(h_init) * (n_points / 2)  # Ensure integral is 1
    
    # Optimization
    start_time = time.time()
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Define a function to compute C5 for a given h
    def objective(h_vec):
        h = h_vec.reshape(-1)
        dx = 2.0 / len(h)
        # Compute all shifted versions of h and compute the max overlap
        max_overlap = 0.0
        for shift in range(len(h)):
            shifted_h = np.roll(h, shift)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [cp.sum(h) == n_points / 2]
    
    # Use scipy.optimize to minimize the C5 value
    result = opt.minimize(
        objective,
        x0=best_h,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-9, 'maxiter': 100}
    )
    
    # Update best solution
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Early exit if time is up
    if time.time() - start_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # Try a few more points with different initializations
    for _ in range(3):
        # Random perturbation to avoid local optima
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        h_perturb = h_perturb * (n_points / 2)
        
        # Compute C5 for the perturbed solution
        c5_perturb = evaluate_erdos_solution(h_perturb, dx)
        
        if c5_perturb < best_c5:
            best_h = h_perturb
            best_c5 = c5_perturb
    
    return (best_h.tolist(), best_c5, n_points)
```
