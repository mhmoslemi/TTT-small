```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported helpers
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    dx_initial = 2.0 / n_points_initial
    sum_h_initial = np.sum(h_initial) * dx_initial
    assert np.isclose(sum_h_initial, 1.0), "Initial h must integrate to 1"

    # Optimization parameters
    n_points = n_points_initial  # Start with the initial construction size
    max_iterations = 100
    convergence_threshold = 1e-5
    iteration = 0

    # Evaluate the initial solution
    def evaluate(h):
        dx = 2.0 / len(h)
        kernel = 1 - h
        corr = np.correlate(h, kernel, mode="full")
        c5 = np.max(corr) * dx
        return c5

    best_h = h_initial.copy()
    best_c5 = evaluate(best_h)
    best_n_points = n_points_initial

    # Optimization loop
    while iteration < max_iterations and time.time() - start_time < budget_s:
        # Try reducing n_points if possible
        if n_points > 20:
            n_points -= 10
            h = np.zeros(n_points)
            dx = 2.0 / n_points
            # Rescale initial h to fit new n_points
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), h_initial)
            # Ensure h is in [0,1]
            h = np.clip(h, 0, 1)
            # Ensure integral is 1
            sum_h = np.sum(h) * dx
            scale = 1.0 / sum_h
            h *= scale
        else:
            # If n_points is small, do not reduce further
            pass

        # Define optimization variables
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]

        # Objective: minimize the maximum overlap
        def objective(h):
            dx = 2.0 / len(h)
            kernel = 1 - h
            corr = np.correlate(h.value, kernel.value, mode="full")
            return np.max(corr) * dx

        # Use a local optimization method with a starting point
        if n_points < 100:
            method = 'SLSQP'
        else:
            method = 'L-BFGS-B'

        result = opt.minimize(
            fun=objective,
            x0=h_initial[:n_points] if n_points < n_points_initial else np.random.rand(n_points),
            method=method,
            bounds=[(0, 1) for _ in range(n_points)],
            options={'ftol': 1e-7, 'maxiter': 100}
        )

        if result.success:
            h_opt = result.x
            c5 = evaluate(h_opt)
            if c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5
                best_n_points = n_points

        iteration += 1

    # Final evaluation
    final_c5 = evaluate(best_h)
    return best_h.tolist(), final_c5, best_n_points
```
