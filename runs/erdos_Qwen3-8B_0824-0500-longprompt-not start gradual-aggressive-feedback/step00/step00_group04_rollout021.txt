def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h_extended = np.concatenate((h, h[:n_points//2]))
    correlations = np.correlate(h_extended, 1 - h, mode="full")
    c5_values = correlations[n_points-1:] * dx
    return np.max(c5_values)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial solution
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Initial evaluation
    c5_initial = evaluate_erdos_solution(h)
    best_c5 = c5_initial
    best_h = h.copy()
    
    start_time = time.time()
    
    # Define optimization variables
    x = cp.Variable(n_points)
    constraints = [
        cp.sum(x) == n_points / 2,
        0 <= x,
        x <= 1
    ]
    
    for _ in range(10):  # Number of iterations
        if time.time() - start_time > budget_s - 5:
            break
        
        # Define objective
        def objective_func(x_val):
            h_val = x_val.copy()
            n_points = len(h_val)
            dx = 2.0 / n_points
            h_extended = np.concatenate((h_val, h_val[:n_points//2]))
            correlations = np.correlate(h_extended, 1 - h_val, mode="full")
            c5_values = correlations[n_points-1:] * dx
            return np.max(c5_values)
        
        # Define optimization problem
        obj = cp.Minimize(objective_func(x))
        prob = cp.Problem(obj, constraints)
        
        # Solve
        try:
            result = prob.solve(solver=cp.SCS, max_iters=100, abstol=1e-8, reltol=1e-8, time_limit=5)
            if result == 0:
                # If solve fails, use current solution
                pass
            else:
                h = x.value
                c5_current = evaluate_erdos_solution(h)
                if c5_current < best_c5:
                    best_c5 = c5_current
                    best_h = h.copy()
        except:
            # If any error occurs during optimization, use current solution
            pass
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, len(best_h)
