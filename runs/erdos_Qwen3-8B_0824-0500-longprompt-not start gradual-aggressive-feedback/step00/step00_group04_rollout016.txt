import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Start with 200 points, can be increased if needed

    # Initial guess: use the provided initial_h_values or sample uniformly
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) == n_points:
        h_values = initial_h_values.copy()
    else:
        # Random initialization with uniform distribution, ensuring sum(h) = n_points / 2
        h_values = np.random.rand(n_points)
        h_values /= np.sum(h_values)
        h_values *= n_points / 2

    dx = 2.0 / n_points

    def evaluate_h(h):
        h = h.reshape(-1)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Ensure the integral equals 1
        total = np.sum(h) * dx
        h = h * (1 / total) * n_points / 2 if total != 0 else h
        # Compute the overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def objective(h):
        return evaluate_h(h)

    # Bounds: h must be in [0, 1]
    bounds = Bounds(0, 1)

    # Gradient-based optimization using L-BFGS-B
    result = opt.minimize(
        objective,
        h_values,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    # Refine with a local search if time allows
    elapsed_time = time.time() - time.time()
    if elapsed_time < budget_s - 10:
        # Use a local search with a random perturbation
        h_refined = result.x.copy()
        perturbation = np.random.rand(n_points) * 0.1 - 0.05
        h_refined += perturbation
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        h_refined = np.clip(h_refined, 0, 1)
        c5_refined = evaluate_h(h_refined)
        if c5_refined < result.fun:
            result = opt.minimize(
                objective,
                h_refined,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 50, 'ftol': 1e-8}
            )

    # Final evaluation
    h_final = result.x.copy()
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    h_final = np.clip(h_final, 0, 1)
    c5_bound = evaluate_h(h_final)

    return h_final, c5_bound, n_points
