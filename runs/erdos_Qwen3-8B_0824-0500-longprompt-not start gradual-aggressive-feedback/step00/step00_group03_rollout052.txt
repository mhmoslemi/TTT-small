import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
# evaluate_erdos_solution() is used for evaluating the solution
# initial_h_values is the starting point for the optimization

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = 70  # Starting with 70 points as provided
    dx = 2.0 / n_points
    
    # Initial solution from initial_h_values
    h = initial_h_values.copy()
    h = h[:n_points]
    
    # Normalize the initial solution to satisfy the integral constraint
    sum_h = np.sum(h)
    h = h * (n_points / 2) / sum_h
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        n = len(h_values)
        dx = 2.0 / n
        overlap = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to evaluate the solution
    def evaluate(solution):
        return compute_c5(solution)
    
    # Optimization settings
    max_iterations = 100  # Limit the number of iterations to stay within the budget
    tolerance = 1e-5  # Stopping criterion
    
    # Define the objective function for optimization
    def objective(h_values):
        return evaluate(h_values)
    
    # Define the constraints
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Set up bounds for h_values (each between 0 and 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use the SLSQP method for constrained optimization
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': max_iterations, 'ftol': tolerance, 'disp': False}
    )
    
    # If the optimization was successful, use the result
    if result.success:
        h_optimized = result.x
    else:
        # If optimization fails, use the initial guess
        h_optimized = h
    
    # Evaluate the best solution found
    best_c5 = compute_c5(h_optimized)
    
    # Check if we can increase the number of points for better accuracy
    # Since we are under time budget, we can try a few more points if feasible
    n_points = min(n_points + 10, 200)  # Increase by 10 points, up to 200
    dx = 2.0 / n_points
    
    # Initialize with the optimized solution and scale it
    h = h_optimized.copy()
    h = h * (n_points / 2) / np.sum(h)
    
    # Run a quick optimization with the new number of points
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        options={'maxiter': 50, 'ftol': tolerance, 'disp': False}
    )
    
    if result.success:
        h_optimized = result.x
    else:
        h_optimized = h
    
    # Evaluate the final solution
    final_c5 = compute_c5(h_optimized)
    
    # Return the best solution and the C5 value
    return (h_optimized, final_c5, n_points)
