import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = None
initial_h_values = None

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess: use the provided initial_h_values (n=95)
    n_points_initial = len(initial_h_values)
    if n_points_initial == 0:
        n_points_initial = 1000  # fallback
    
    # Start with the initial guess
    n_points = n_points_initial
    h = np.copy(initial_h_values) if initial_h_values is not None else np.random.rand(n_points)
    h = h * 2.0 / n_points  # scale to make sum(h) = n_points / 2 (since sum(h) * dx = 1)
    h = np.clip(h, 0, 1)
    
    # Function to compute C5
    def compute_c5(h):
        dx = 2.0 / n_points
        h_shifted = np.zeros((n_points, n_points))
        for i in range(n_points):
            h_shifted[i, :] = h[(i + 1) % n_points:]
        overlap = np.sum(h * (1 - h_shifted), axis=1)
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to compute objective and constraints for optimization
    def optimize_h(h):
        h = np.copy(h)
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        h_shifted = np.zeros((n_points, n_points))
        for i in range(n_points):
            h_shifted[i, :] = h[(i + 1) % n_points:]
        overlap = np.sum(h * (1 - h_shifted), axis=1)
        c5 = np.max(overlap) * dx
        return c5
    
    # Optimization function
    def optimize_function(x):
        h = x.reshape(n_points)
        h = np.clip(h, 0, 1)
        h = h * 2.0 / n_points
        return optimize_h(h)
    
    # Optimization with L-BFGS-B
    start_time = time.time()
    result = optimize.minimize(
        optimize_function,
        x0=h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={"maxiter": 100, "ftol": 1e-6, "gtol": 1e-6, "disp":False}
    )
    
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        print("Time budget exceeded, returning initial solution")
        return (h, compute_c5(h), n_points)
    
    # Refine with another round
    h_refined = result.x
    h_refined = np.clip(h_refined, 0, 1)
    h_refined = h_refined * 2.0 / n_points
    c5_refined = optimize_h(h_refined)
    
    # Check if we can reduce n_points
    for n in range(n_points - 10, 10, -1):
        if n < 10:
            break
        h_new = np.linspace(0, 2, n)
        h_new = np.random.rand(n) * 2.0 / n
        h_new = np.clip(h_new, 0, 1)
        c5_new = optimize_h(h_new)
        if c5_new < c5_refined:
            n_points = n
            h_refined = h_new
            c5_refined = c5_new
    
    return (h_refined, c5_refined, n_points)
