    for _ in range(200):  # Limit iterations to stay within budget
        start_time = time.time()
        
        # Create a copy to perturb
        h_perturb = h.copy()
        # Random perturbation: flip some values between 0 and 1
        indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
        h_perturb[indices] = 1 - h_perturb[indices]
        
        # Ensure the integral constraint is satisfied
        sum_h = np.sum(h_perturb)
        scale_factor = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h_perturb *= scale_factor
        
        # Evaluate the new C5
        c5 = evaluate_erdos_solution(h_perturb)
        
        # Update best solution if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturb.copy()
        
        # Check if time is running out
        if time.time() - start_time > budget_s - 1:
            break
    
    # Try a few local optimization steps using convex optimization
    # Simplified version to avoid complexity
    for _ in range(10):
        # Define variables
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        
        # Objective: minimize the maximum overlap
        # This is a complex objective; we approximate by optimizing a few shifts
        obj = cp.Minimize(0)
        for k in [1, 2, 3, 4, 5]:
            # Compute the overlap for shift k
            overlap = cp.sum(x * (1 - x[k:])) / (n_points / 2)
            obj = cp.Minimize(cp.Max(obj, overlap))
        
        # Solve the problem
        prob = cp.Problem(obj, constraints)

            prob.solve(solver=cp.SCS, verbose=False)
