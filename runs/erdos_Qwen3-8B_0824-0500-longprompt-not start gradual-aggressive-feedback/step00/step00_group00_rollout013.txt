import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with a reasonable number of points
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)
    
    # Function to evaluate the C5 bound
    def evaluate_c5(h):
        h = np.array(h)
        h = np.pad(h, (0, n_points), 'constant')  # Pad to make full convolution possible
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute the overlap integral for a given h
    def objective(h):
        return evaluate_c5(h)
    
    # Function to compute the constraint: integral of h over [0, 2] = 1
    def constraint(h):
        return np.sum(h) * dx - 1
    
    # Function to compute bounds on h: 0 <= h[i] <= 1
    def bounds(h):
        return np.all((h >= 0) & (h <= 1))
    
    # Initial guess: flat 0.5
    h_initial = np.array([0.5] * n_points)
    
    # Use a local optimizer to minimize the C5 bound
    result = opt.minimize(
        fun=objective,
        x0=h_initial,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        method='L-BFGS-B',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = evaluate_c5(best_h)
    
    # Try increasing n_points if time allows
    elapsed = time.time() - time.time()
    remaining = budget_s - elapsed
    if remaining > 0.5:
        n_points = min(200, int(remaining * 10))
        dx = 2.0 / n_points
        h_initial = np.array([0.5] * n_points)
        
        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            method='L-BFGS-B',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        
        best_h = result.x
        best_c5 = evaluate_c5(best_h)
    
    # Final check for bounds and integral constraint
    if not bounds(best_h):
        best_c5 = np.inf
    
    return best_h, best_c5, n_points
