import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(70) * 2 - 1  # Placeholder initial construction (should be actual initial h_values)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 points
    dx = 2.0 / n_points
    max_iter = 100  # Max number of iterations
    improvement_threshold = 1e-4  # Stop if no improvement in C5 for some iterations
    time_limit = budget_s  # Time limit in seconds
    start_time = time.time()
    
    # Initial guess
    h = np.random.rand(n_points)
    h = np.clip((h - 0.5) * 2, 0, 1)  # Initial guess with values in [0, 1]
    # Ensure integral is 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    iteration = 0
    
    while time.time() - start_time < time_limit and iteration < max_iter:
        # Objective: minimize the maximum overlap integral
        # We can use a convex optimization approach to minimize the maximum over k
        
        # Define variables
        h_var = cp.Variable(n_points, nonneg=True)
        # Ensure h is in [0, 1]
        constraints = [0 <= h_var, h_var <= 1]
        # Ensure integral is 1
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Compute all shifts and their overlaps
        overlap_vals = []
        for k in range(1, n_points):
            if k >= n_points:
                break
            # Shift h by k
            shifted_h = np.roll(h_var.value, k)
            overlap = cp.multiply(h_var, 1 - shifted_h)
            overlap_vals.append(overlap)
        
        # Maximize the overlap vals across all k
        max_overlap = cp.Max(overlap_vals)
        
        # Objective: minimize the max overlap
        problem = cp.Problem(cp.Minimize(max_overlap), constraints)
        try:
            problem.solve(solver=cp.ECOS, eps=1e-6)
            if problem.status != 'OPTIMAL':
                break
            h = h_var.value
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5 - improvement_threshold:
                best_h = h.copy()
                best_c5 = current_c5
            iteration += 1
        except:
            # If solver fails, try a simpler optimization
            # Try gradient descent on the current h
            h = h_var.value
            learning_rate = 0.01
            for _ in range(10):
                # Compute the overlap integrals
                overlaps = []
                for k in range(1, n_points):
                    if k >= n_points:
                        break
                    shifted_h = np.roll(h, k)
                    overlap = np.sum(h * (1 - shifted_h)) * dx
                    overlaps.append(overlap)
                max_overlap = np.max(overlaps)
                # Gradient descent step
                grads = np.zeros_like(h)
                for k in range(1, n_points):
                    if k >= n_points:
                        break
                    shifted_h = np.roll(h, k)
                    grad = np.zeros_like(h)
                    for i in range(n_points):
                        # Contribution to gradient from h[i]
                        grad[i] = (1 - shifted_h[i]) * dx
                    grads += grad
                # Update h
                h -= learning_rate * grads
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5 - improvement_threshold:
                best_h = h.copy()
                best_c5 = current_c5
            iteration += 1
    
    return best_h, best_c5, n_points
