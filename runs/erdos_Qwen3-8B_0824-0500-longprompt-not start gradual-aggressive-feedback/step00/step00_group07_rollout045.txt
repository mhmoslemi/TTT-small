import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the number of points and the grid
    n_points = 80  # Start with a small number of points
    dx = 2.0 / n_points
    x_grid = np.linspace(0, 2, n_points)
    
    # Initial guess based on the provided initial_h_values
    h_values = np.array(initial_h_values[:n_points])
    h_values = h_values / np.sum(h_values) * (n_points / 2)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) / (n_points / 2))
        shift = np.arange(-n_points + 1, n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        corr = corr * dx
        return np.max(corr)
    
    # Function to optimize with CVXPY
    def optimize_solution(h_values):
        h = cp.Variable(n_points)
        # Constraint: sum(h) == n_points / 2
        constraints = [cp.sum(h) == n_points / 2]
        # Objective: minimize the maximum overlap integral
        # We can use a minimax formulation with auxiliary variables
        t = cp.Variable()
        constraints += [t >= cp.multiply(h, 1 - cp.roll(h, shift)) for shift in range(1, n_points)]
        constraints += [t >= cp.multiply(h, 1 - cp.roll(h, shift)) for shift in range(n_points, 2 * n_points - 1)]
        objective = cp.Minimize(t)
        problem = cp.Problem(objective, constraints)
        result = problem.solve()
        if result is None:
            return h_values
        return np.array([float(val) for val in h])
    
    # Optimize the solution
    best_h = h_values
    best_c5 = compute_c5(best_h)
    
    # Run the optimization
    try:
        h_opt = optimize_solution(best_h)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
        c5 = compute_c5(h_opt)
        best_h = h_opt
        best_c5 = c5
    except:
        pass
    
    # Further refine with scipy optimization
    def objective_func(h):
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        return compute_c5(h)
    
    result = optimize.minimize(
        objective_func,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 100, 'ftol': 1e-6}
    )
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * (n_points / 2)
    best_c5 = compute_c5(best_h)
    
    # Track elapsed time and return the best solution
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return best_h, best_c5, n_points
    
    # If time allows, increase the number of points and re-optimize
    if elapsed_time < budget_s * 0.8:
        n_points *= 2
        dx = 2.0 / n_points
        x_grid = np.linspace(0, 2, n_points)
        h_values = np.zeros(n_points)
        h_values[::2] = 1
        h_values = h_values / np.sum(h_values) * (n_points / 2)
        
        def compute_c5_new(h):

            shift = np.arange(-n_points + 1, n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            corr = corr * dx
            return np.max(corr)
        
        def optimize_solution_new(h_values):
            h = cp.Variable(n_points)

            constraints = [cp.sum(h) == n_points / 2]
            # Objective: minimize the maximum overlap integral
            # We can use a minimax formulation with auxiliary variables
            t = cp.Variable()
            constraints += [t >= cp.multiply(h, 1 - cp.roll(h, shift)) for shift in range(1, n_points)]
            constraints += [t >= cp.multiply(h, 1 - cp.roll(h, shift)) for shift in range(n_points, 2 * n_points - 1)]
            objective = cp.Minimize(t)
            problem = cp.Problem(objective, constraints)
            result = problem.solve()
            if result is None:
                return h_values
            return np.array([float(val) for val in h])
        
        best_h_new = h_values
        best_c5_new = compute_c5_new(best_h_new)
        
        # Optimize the solution
        try:
            h_opt_new = optimize_solution_new(best_h_new)
            h_opt_new = np.clip(h_opt_new, 0, 1)
            h_opt_new = h_opt_new / np.sum(h_opt_new) * (n_points / 2)
            c5_new = compute_c5_new(h_opt_new)
            best_h_new = h_opt_new
            best_c5_new = c5_new
        except:
            pass
        
        # Further refine with scipy optimization
        def objective_func_new(h):
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            return compute_c5_new(h)
        
        result_new = optimize.minimize(
            objective_func_new,

            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        best_h_new = result_new.x
        best_h_new = np.clip(best_h_new, 0, 1)
        best_h_new = best_h_new / np.sum(best_h_new) * (n_points / 2)
        best_c5_new = compute_c5_new(best_h_new)
        
        # Compare the best solutions
        if best_c5_new < best_c5:
            best_h = best_h_new
            best_c5 = best_c5_new
    
    return best_h, best_c5, n_points
