import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for speed

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h_extended = np.concatenate((h, np.zeros_like(h)))
        shifted = np.zeros_like(h_extended)
        for k in range(1, len(h)):
            shifted[k:] = h_extended[:-k]
        overlap = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlap)

    # Initial guess from provided initial_h_values
    initial_h = initial_h_values.copy()
    n_points_initial = len(initial_h)
    if n_points_initial <= 1000:
        n_points = n_points_initial
        h = initial_h.copy()
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Try a few perturbations
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            if np.abs(np.sum(h_perturbed) - n_points / 2) < 1e-6:
                c5_perturbed = compute_c5(h_perturbed)
                if c5_perturbed < best_c5:
                    best_c5 = c5_perturbed
                    best_h = h_perturbed.copy()
    else:
        # Use a smaller n_points
        n_points = 100
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h)
        best_c5 = c5
        best_h = h.copy()

    # Optimization loop with decreasing n_points
    for n_points in [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]:
        if time.time() - start_time > budget_s - 5:
            break
        # Use a smaller n_points
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Perform gradient descent with constraints
        def fitness(x):
            h = x.reshape(-1)
            dx = 2.0 / len(h)
            h_extended = np.concatenate((h, np.zeros_like(h)))
            shifted = np.zeros_like(h_extended)
            for k in range(1, len(h)):
                shifted[k:] = h_extended[:-k]
            overlap = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(overlap)

        def constraint(x):
            return np.sum(x) - n_points / 2

        x0 = h.copy()
        result = opt.minimize(fitness, x0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        if result.success:
            h_opt = result.x
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        else:
            h_opt = h.copy()

    # Final refinement with CVXPY
    if time.time() - start_time < budget_s - 5:
        n_points_refine = min(1000, len(best_h))
        h_refine = best_h[:n_points_refine]
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points_refine / 2) / np.sum(h_refine)
        x = cp.Variable(n_points_refine)
        obj = cp.Minimize(cp.Max(cp.correlate(x, 1 - x, mode='full') * (2.0 / n_points_refine)))
        cons = [cp.sum(x) == n_points_refine / 2]
        prob = cp.Problem(obj, cons)
        result = prob.solve(solver='SCIP')
        if result is not None:
            h_opt = x.value
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Ensure best_h has the correct length
    best_h = best_h[:n_points]
    return best_h, best_c5, n_points
