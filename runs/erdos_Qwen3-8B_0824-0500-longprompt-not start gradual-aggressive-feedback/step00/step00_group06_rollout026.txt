import numpy as np
import scipy.optimize as optimize

import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h) * dx
    # Normalize to ensure the integral is 1
    h = h * (1.0 / sum_h) * 1.0
    
    def evaluate_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        # Compute the overlap integral
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        c5 = evaluate_c5(h)
        return c5
    
    def constraint_sum(x):
        return np.sum(x) - (n_points / 2)
    
    # Optimization using gradient descent with a learning rate
    learning_rate = 0.01
    max_iterations = 1000
    for _ in range(max_iterations):
        # Compute the objective and constraints
        current_c5 = evaluate_c5(h)
        # Compute the gradient (approximate using finite differences)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            grad[i] = (evaluate_c5(h_plus) - evaluate_c5(h_minus)) / (2 * 1e-5)
        
        # Update the weights
        h = h - learning_rate * grad
        # Clip to [0, 1] and normalize the sum
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Check if we are exceeding the budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Evaluate the final solution
    final_c5 = evaluate_c5(h)
    return (h.tolist(), final_c5, len(h))
