import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set the random seed for reproducibility
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Choose a number of points (smaller is better for speed)
    n_points = 100  # Small enough for fast evaluation
    
    # Generate initial guess
    h_initial = np.array(initial_h_values[:n_points])  # Use the initial construction
    h_initial = np.clip(h_initial, 0, 1)
    
    # Normalize the initial guess to meet the integral constraint
    total_integral = np.sum(h_initial) * (2.0 / n_points)
    h_initial = h_initial * (1.0 / total_integral) * (2.0 / n_points)
    
    # Objective function to minimize max overlap
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h.reshape(-1, 1)
        # Compute the overlap integral for all lags
        overlaps = np.zeros(h.shape[0] - 1)
        for k in range(1, h.shape[0]):
            overlap = np.sum(h[:-k] * (1 - h[k:])) * (2.0 / n_points)
            overlaps[k - 1] = overlap
        return np.max(overlaps)
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    max_iterations = 200
    tolerance = 1e-5
    
    # Use a local optimizer with a good initial guess
    result = optimize.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        jac=None,
        options={'maxiter': max_iterations, 'ftol': tolerance}
    )
    
    # Extract the best solution found
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (1.0 / np.sum(best_h) * (2.0 / n_points))  # Ensure integral constraint
    
    # Evaluate the best solution
    c5_bound = evaluate_erdos_solution(best_h, n_points)
    
    # Ensure we stay within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:  # Leave some margin for safety
        return (best_h, c5_bound, n_points)
    
    # Optionally perform additional local refinements if time permits
    # For brevity, we'll return the best solution found so far
    
    return (best_h, c5_bound, n_points)
