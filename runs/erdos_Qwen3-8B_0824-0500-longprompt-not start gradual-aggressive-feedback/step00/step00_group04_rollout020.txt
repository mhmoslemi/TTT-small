import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
from functools import partial

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Initial construction from initial_h_values
    initial_h = initial_h_values
    n_points_initial = len(initial_h)
    dx_initial = 2.0 / n_points_initial
    # Convert to float for optimization
    h_initial = np.array(initial_h, dtype=float)
    
    # Start with the initial construction
    best_h = h_initial.copy()
    best_c5 = compute_c5(best_h, dx=2.0 / len(best_h))
    
    # Attempt to improve with different n_points
    for n_points in [64, 128, 256]:
        if time.time() > time.time() + budget_s:
            break
        
        # Define the dx for this n_points
        dx = 2.0 / n_points
        
        # Define the variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,  # Integral over [0, 2] is 1
            h >= 0,
            h <= 1
        ]
        
        # Objective is to minimize the maximum overlap integral
        # Compute cross-correlation and find max
        def objective(h_val):
            h_val = np.array(h_val)
            corr = signal.correlate(h_val, 1 - h_val, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        # Use a gradient-based approach for faster optimization
        # Convert to a vectorized function
        def objective_vectorized(h_val):
            h_val = np.array(h_val)
            corr = signal.correlate(h_val, 1 - h_val, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        # Set up the optimization problem with a local solver
        # We use scipy to optimize the function directly
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Initial guess from initial_h scaled to n_points
        initial_guess = np.array(initial_h) * (n_points / n_points_initial)
        initial_guess = np.clip(initial_guess, 0, 1)
        
        # Use L-BFGS-B for optimization
        result = optimize.minimize(
            objective_vectorized,
            initial_guess,
            bounds=bounds,
            method='L-BFGS-B',
            options={'gtol': 1e-5, 'disp': False}
        )
        
        # Update the best solution if improved
        if result.success:
            h_candidate = result.x
            c5_candidate = compute_c5(h_candidate, dx=dx)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        
    # Return the best solution found
    return best_h, best_c5, len(best_h)

def compute_c5(h, dx=1.0):
    corr = signal.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap
