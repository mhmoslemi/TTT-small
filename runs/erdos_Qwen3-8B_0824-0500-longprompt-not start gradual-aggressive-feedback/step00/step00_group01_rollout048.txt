def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points (can increase if needed)
    dx = 2.0 / n_points
    h_values = np.zeros(n_points)
    
    # Use the initial_h_values if available
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) == n_points:
        h_values = np.array(initial_h_values)
    else:
        # Initialize h_values with a simple uniform distribution
        h_values = np.ones(n_points) * (n_points / 2) / (2.0 * dx)
    
    # Optimization function: minimize the maximum overlap integral
    def objective(h):
        h = np.clip(h, 0.0, 1.0)
        h = np.concatenate((h, np.zeros(n_points)))  # Extend for convolution
        h_shifted = np.roll(h, shift=1, axis=0)
        overlap = np.correlate(h, 1 - h_shifted, mode='full') * dx
        return np.max(overlap)
    
    # Use a gradient-based optimization method
    def optimize_h(n_points, max_iter=100, tol=1e-5):
        h = np.clip(np.random.uniform(0.0, 1.0, n_points), 0.0, 1.0)
        h = h * (n_points / 2) / (2.0 * dx)  # Ensure the integral is 1
        
        for i in range(max_iter):
            # Compute the overlap integral
            h = np.clip(h, 0.0, 1.0)
            h_extended = np.concatenate((h, np.zeros(n_points)))
            h_shifted = np.roll(h_extended, shift=1, axis=0)
            overlap = np.correlate(h, 1 - h_shifted, mode='full') * dx
            current_c5 = np.max(overlap)
            
            if current_c5 < tol:
                break
            
            # Compute gradient via finite differences
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0.0, 1.0)
                h_plus_extended = np.concatenate((h_plus, np.zeros(n_points)))
                h_plus_shifted = np.roll(h_plus_extended, shift=1, axis=0)
                overlap_plus = np.correlate(h_plus, 1 - h_plus_shifted, mode='full') * dx
                grad[j] = np.max(overlap_plus) - current_c5
            
            # Update using gradient descent
            h -= 0.1 * grad
            h = np.clip(h, 0.0, 1.0)
            h = h * (n_points / 2) / (2.0 * dx)
        
        return h, current_c5
    
    # Initial optimization
    h, c5_initial = optimize_h(n_points)
    
    # Try increasing n_points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0:
        n_points = min(200, int(n_points * 1.5))
        dx = 2.0 / n_points
        h = np.random.uniform(0.0, 1.0, n_points)
        h = h * (n_points / 2) / (2.0 * dx)
        h, c5_final = optimize_h(n_points)
    else:
        h = h_values
        c5_final = c5_initial
    
    # Final evaluation
    h = np.clip(h, 0.0, 1.0)
    h_extended = np.concatenate((h, np.zeros(n_points)))
    h_shifted = np.roll(h_extended, shift=1, axis=0)
    overlap = np.correlate(h, 1 - h_shifted, mode='full') * dx
    c5_bound = np.max(overlap)
    
    # Ensure the integral is 1
    integral = np.sum(h) * dx
    if np.abs(integral - 1.0) > 1e-6:
        raise ValueError(f"Integral not equal to 1.0: {integral}")
    
    # Return the best solution
    return h, c5_bound, n_points
