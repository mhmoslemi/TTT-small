import numpy as np
import scipy.optimize as optimize

from scipy.optimize.optimize import OptimizeResult

import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    # Placeholder for actual evaluation function
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shift = np.arange(-n_points + 1, n_points)
    shifted_h = np.zeros((n_points, n_points))
    for i in range(n_points):
        shifted_h[i, :] = np.roll(h, -i)
    shifted_1_minus_h = 1 - shifted_h
    cross_corr = np.sum(h * shifted_1_minus_h, axis=1)
    c5 = np.max(cross_corr * dx)
    return c5

def initial_h_values():
    # Placeholder for initial construction
    return np.array([0.5]*45)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Initial guess
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    required_sum = n_points / 2
    # Adjust initial guess to meet the sum constraint
    if abs(sum_h - required_sum) > 1e-6:
        # Normalize
        initial_h = initial_h * required_sum / sum_h
    
    # Define variables
    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1]
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Objective function: minimize max_k ∫ h(x)(1 - h(x + k)) dx
    # To approximate max_k, we can minimize the maximum over a few k's
    # We'll evaluate for a few shifts k to get a proxy for the max
    # For simplicity, evaluate for k in [-n_points//2, n_points//2]
    # For a lower bound, we can consider evaluating for k in [-5, 5]
    
    # Evaluate for k in range(-5, 6)
    max_c5 = cp.Maximize(0)
    for k in range(-5, 6):
        if k == 0:
            continue
        # Shift h by k positions
        shifted_h = cp.reshape(cp.roll(h, -k), (n_points, 1))
        # Compute 1 - h(x + k)
        one_minus_shifted_h = 1 - shifted_h
        # Compute the integral: ∫ h(x)(1 - h(x + k)) dx
        # Approximate with dot product * dx
        integral = cp.sum(h * one_minus_shifted_h) * dx
        # Add to the max constraint
        max_c5 = cp.Maximize(cp.min(cp.stack([max_c5.expr, integral])))
    
    # Optimization problem
    prob = cp.Problem(max_c5, constraints)
    
    # Solve with a solver
    try:
        result = prob.solve(solver=cp.MOSEK, verbose=False)
    except:
        result = prob.solve(solver=cp.CVXOPT, verbose=False)
    
    if result is None:
        # Fallback to a simple optimization
        h_values = initial_h.copy()
        for _ in range(100):
            # Simple gradient descent
            # Compute the max_c5 for the current h
            current_c5 = evaluate_erdos_solution(h_values)
            # Simple update
            h_values = np.clip(h_values + np.random.normal(0, 0.01, n_points), 0, 1)
            h_values = h_values * (n_points / 2) / np.sum(h_values)
            # Early stopping if time is up
            if time.time() - start_time > budget_s - 1:
                break
        return h_values, current_c5, n_points
    
    # Get the optimized h values
    h_values = h.value
    c5_bound = evaluate_erdos_solution(h_values)
    
    # Early stopping if time is up
    if time.time() - start_time > budget_s - 1:
        return h_values, c5_bound, n_points
    
    # Refine with a local optimization
    for _ in range(10):
        # Define a new optimization problem to minimize the max
        # For simplicity, we'll minimize the integral for a few shifts
        # We'll use a gradient-based optimization
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            # Compute the maximum overlap integral
            max_int = 0
            for k in range(-5, 6):
                if k == 0:
                    continue
                # Shift h by k positions
                shifted_h = np.roll(h, -k)
                # Compute integral
                integral = np.sum(h * (1 - shifted_h)) * dx
                if integral > max_int:
                    max_int = integral
            return max_int
        
        # Gradient of the objective
        def gradient(h_flat):
            h = h_flat.reshape(n_points)
            grad = np.zeros(n_points)
            for k in range(-5, 6):
                if k == 0:
                    continue
                # Shift h by k positions
                shifted_h = np.roll(h, -k)
                # Compute derivative
                derivative = np.zeros(n_points)
                for i in range(n_points):
                    # For each x, the derivative is (1 - shifted_h[i]) - h[i] * (derivative of shifted_h)
                    # The derivative of shifted_h[i] with respect to h[j] is 1 if j = (i + k) mod n_points
                    # So the derivative of the integral with respect to h[j] is (1 - shifted_h[i]) when i = (j - k) mod n_points
                    # This is complicated; we'll approximate using finite differences
                    # Use a small epsilon
                    eps = 1e-6
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    integral_plus = np.sum(h_plus * (1 - np.roll(h_plus, -k))) * dx
                    derivative[j] = (integral_plus - objective(h_flat)) / eps
            # Average the derivatives
            return np.mean(derivative, axis=0)
        
        # Perform gradient descent
        h_values = np.clip(h_values + np.random.normal(0, 0.01, n_points), 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        for _ in range(50):
            grad = gradient(h_values)
            h_values = np.clip(h_values - 0.01 * grad, 0, 1)
            h_values = h_values * (n_points / 2) / np.sum(h_values)
            # Early stopping if time is up
            if time.time() - start_time > budget_s - 1:
                break
    
    # Final evaluation
    c5_bound = evaluate_erdos_solution(h_values)
    
    return h_values, c5_bound, n_points
