import numpy as np

import time

from functools import partial

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):

    n_points_initial = len(initial_h_values)

    target_sum = n_points_initial / 2

    h_initial *= scale
    h_initial = np.clip(h_initial, 0, 1)

    def compute_c5(h):
        n_points = len(h)
        dx = 2.0 / n_points
        h = np.clip(h, 0, 1)
        h_pad = np.concatenate([h, np.zeros_like(h)])
        corr = np.correlate(h_pad, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx
    
    initial_c5 = compute_c5(h_initial)
    
    # Try with different n_points
    best_h = h_initial
    best_c5 = initial_c5
    n_points_best = n_points_initial
    
    # Function to evaluate a solution
    def evaluate_solution(h):
        nonlocal best_h, best_c5, n_points_best
        start_time = time.time()
        h = np.clip(h, 0, 1)
        n_points = len(h)
        dx = 2.0 / n_points
        h_pad = np.concatenate([h, np.zeros_like(h)])
        corr = np.correlate(h_pad, 1 - h, mode="full")
        max_corr = np.max(corr)
        c5 = max_corr * dx
        elapsed_time = time.time() - start_time
        return c5, elapsed_time
    
    # Try with a few different n_points
    for n_points in [n_points_initial, n_points_initial + 10, n_points_initial + 20, n_points_initial + 30]:
        if n_points < 2:
            continue
        
        # Create an initial guess with n_points
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        h[0] = 1.0
        h[-1] = 1.0
        h[1:-1] = 0.5
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h *= scale
        h = np.clip(h, 0, 1)
        
        # Define the optimization problem
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.SCS, max_iter=100, eps=1e-4, verbose=False)
            if result is not None:
                h_opt = x.value
                c5, _ = evaluate_solution(h_opt)
                if c5 < best_c5:
                    best_h = h_opt
                    best_c5 = c5
                    n_points_best = n_points
        except:
            pass
    
    # Try with a refinement step using gradient descent
    def objective_function(h):
        return compute_c5(h)
    
    def gradient_descent(h, learning_rate=0.01, n_iter=1000, tolerance=1e-4):
        for _ in range(n_iter):
            grad = np.zeros_like(h)
            n_points = len(h)
            dx = 2.0 / n_points
            h_pad = np.concatenate([h, np.zeros_like(h)])
            corr = np.correlate(h_pad, 1 - h, mode="full")
            max_corr_index = np.argmax(corr)
            max_corr = corr[max_corr_index]
            
            # Compute the gradient
            for i in range(n_points):
                if max_corr_index >= i:
                    grad[i] = -max_corr * (1 - h[i]) * dx
                else:
                    grad[i] = max_corr * h[i] * dx
            
            # Update the solution
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h
            h *= scale
            h = np.clip(h, 0, 1)
            
            # Check for convergence
            if np.linalg.norm(grad) < tolerance:
                break
        return h
    
    h_refined = gradient_descent(best_h.copy(), learning_rate=0.01, n_iter=1000, tolerance=1e-4)
    c5_refined, _ = evaluate_solution(h_refined)
    
    if c5_refined < best_c5:
        best_h = h_refined
        best_c5 = c5_refined
    
    # Try with a few random initializations
    for _ in range(5):
        h_rand = np.random.rand(n_points_best)
        sum_h = np.sum(h_rand)
        scale = (n_points_best / 2) / sum_h
        h_rand *= scale
        h_rand = np.clip(h_rand, 0, 1)
        
        # Use gradient descent on this initial guess
        h = gradient_descent(h_rand.copy(), learning_rate=0.01, n_iter=1000, tolerance=1e-4)
        c5, _ = evaluate_solution(h)
        
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    # Return the best solution found
    best_h = np.clip(best_h, 0, 1)
    return best_h.tolist(), best_c5, n_points_best
