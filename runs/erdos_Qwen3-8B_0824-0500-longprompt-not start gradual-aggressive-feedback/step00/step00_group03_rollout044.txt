def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initialize with the given starting point
    initial_h_values = np.array(initial_h_values)
    n_points = len(initial_h_values)
    
    # Define the overlap integral function for a given h
    def compute_c5(h_values):
        h = h_values
        conv = signal.correlate(h, 1 - h, mode="full")
        # Only consider the relevant part of the correlation (overlap)
        # The overlap starts at index n_points - 1 and ends at 2 * n_points - 2
        start_idx = n_points - 1
        end_idx = 2 * n_points - 2
        overlap = conv[start_idx:end_idx + 1]
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    # Define the objective function for optimization
    def objective(h_values):
        return compute_c5(h_values)
    
    # Define constraint: sum(h_values) == n_points / 2
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Define bounds: h_values in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using Sequential Least Squares Programming (SLSQP)
    result = opt.minimize(
        fun=objective,
        x0=initial_h_values,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"maxiter": 100, "ftol": 1e-8, "eps": 1e-8}
    )
    
    # Extract the best solution found
    best_h_values = result.x
    best_c5 = compute_c5(best_h_values)
    
    # If the optimization was successful, return the result
    if result.success:
        return (best_h_values, best_c5, n_points)
    else:
        # If the optimization failed, return the initial solution
        return (initial_h_values, compute_c5(initial_h_values), n_points)
