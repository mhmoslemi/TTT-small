import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with initial solution if available
    if hasattr(globals(), 'initial_h_values'):
        h = np.array(initial_h_values, dtype=np.float64)
        n_points = len(h)
    else:
        # Default starting point: uniform distribution
        h = np.full(45, 1/45, dtype=np.float64)
        n_points = 45
    
    dx = 2.0 / n_points
    sum_h = h.sum()
    if not np.isclose(sum_h, n_points / 2):
        h = h * (n_points / 2) / sum_h
    
    def evaluate_c5(h_values):
        h = np.array(h_values, dtype=np.float64)
        dx = 2.0 / len(h)
        n = len(h)
        shifted_h = np.zeros(2 * n - 1)
        for i in range(n):
            shifted_h[i:i + n] = h
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5

    def optimize_c5(h_values, max_iter=100, tol=1e-4):
        best_h = h_values.copy()
        best_c5 = evaluate_c5(best_h)
        time_used = 0
        
        for _ in range(max_iter):
            start_time = time.time()
            
            # Use a simple gradient descent with line search
            h = best_h.copy()
            h = np.clip(h, 0, 1)
            dx = 2.0 / len(h)
            n = len(h)
            
            # Compute the cross correlation
            shifted_h = np.zeros(2 * n - 1)
            for i in range(n):
                shifted_h[i:i + n] = h
            cross_corr = np.correlate(h, 1 - h, mode="full")
            
            # Compute the maximum value
            c5 = np.max(cross_corr) * dx
            
            # Compute the derivative
            grad = np.zeros_like(h)
            for i in range(n):
                # For each point, compute the impact on the maximum correlation
                # Simplified for speed
                for k in range(1, n):
                    if i + k < n:
                        grad[i] += (cross_corr[i + k] - cross_corr[i + k - 1]) * dx
                    if i - k >= 0:
                        grad[i] -= (cross_corr[i - k] - cross_corr[i - k + 1]) * dx
            grad = -grad * dx
            
            # Line search
            alpha = 1.0
            while True:
                h_new = h + alpha * grad
                h_new = np.clip(h_new, 0, 1)
                new_c5 = evaluate_c5(h_new)
                if new_c5 < c5:
                    h = h_new
                    c5 = new_c5
                    break
                else:
                    alpha *= 0.5
                    if alpha < 1e-6:
                        break
            
            # Update best solution
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
            time_used += time.time() - start_time
            if time_used >= budget_s:
                break
        
        return best_h, best_c5

    # Run optimization
    best_h, best_c5 = optimize_c5(h)
    
    # Fine-tune with a few more iterations
    for _ in range(5):
        start_time = time.time()
        best_h, best_c5 = optimize_c5(best_h)
        if time.time() - start_time >= budget_s:
            break
    
    n_points = len(best_h)
    return best_h, best_c5, n_points
