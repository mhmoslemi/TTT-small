import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
import cvxpy as cp
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
initial_h_values = np.array([0.0] * 70)  # Placeholder; actual initial value is loaded from somewhere

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100  # Start with 100 points; can increase if time allows
    dx = 2.0 / n_points

    # Initial guess based on the provided initial_h_values
    h_initial = np.zeros(n_points)
    if len(initial_h_values) == n_points:
        h_initial = initial_h_values.copy()
    else:
        # Fallback: uniform distribution
        h_initial = np.ones(n_points) * (n_points / 2) / n_points

    # Function to evaluate the C5 value
    def objective(h):
        h = h.reshape(-1)
        # Ensure h is within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Function to compute the overlap integral for given h
    def compute_overlap(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx

    # Optimization loop
    best_h = h_initial.copy()
    best_c5 = compute_overlap(best_h)

    # Try various local search strategies
    for _ in range(100):  # Limited number of iterations to stay within budget
        # Random perturbation
        h_perturbed = best_h.copy()
        idx = np.random.randint(0, n_points)
        h_perturbed[idx] += np.random.uniform(-0.1, 0.1)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Ensure the integral constraint
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        c5_perturbed = compute_overlap(h_perturbed)
        if c5_perturbed < best_c5:
            best_h = h_perturbed
            best_c5 = c5_perturbed

        # Local search with gradient descent
        h_local = best_h.copy()
        # Gradient approximation
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_local.copy()
            h_plus[i] += 1e-4
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_overlap(h_plus)
            h_minus = h_local.copy()
            h_minus[i] -= 1e-4
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = compute_overlap(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * 1e-4)
        # Update in the direction of descent
        h_local -= 0.01 * grad
        h_local = np.clip(h_local, 0, 1)
        h_local = h_local * (n_points / 2) / np.sum(h_local)
        c5_local = compute_overlap(h_local)
        if c5_local < best_c5:
            best_h = h_local
            best_c5 = c5_local

        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break

    # Try convex optimization
    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
    obj = cp.Minimize(cp.Max(cp.correlate(h, 1 - h, mode="full") * dx))
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=100)
        if prob.status == 'optimal':
            best_h = h.value
            best_c5 = compute_overlap(best_h)
    except:
        pass

    # Final check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = compute_overlap(best_h)
    return (best_h, best_c5, n_points)
