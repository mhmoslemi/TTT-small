import numpy as np

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:-k]
        overlap = np.correlate(h, 1 - shifted, mode="full")
        c5 = np.max(overlap) * dx
        if c5 < 0.380876:
            return c5
    return np.max(np.correlate(h, 1 - h, mode="full")) * dx

def initial_h_values():
    # A basic initial guess with a simple pattern
    n_points = 95
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            h[i] = 1.0
    h = h / np.sum(h) * (n_points / 2)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial guess
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    
    # Define the function to minimize C5
    def objective_function(h_values):
        h = np.array(h_values)
        # Compute overlap integral for all k
        overlaps = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlaps) * dx
        return c5
    
    # Optimizer settings
    max_iterations = 100
    tolerance = 1e-5
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        # Perform gradient descent step
        h = np.array(best_h)
        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlaps) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            print(f"New best C5: {best_c5}")
        
        # Check if time budget is exceeded
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb the solution slightly
        perturbation = np.random.normal(0, 0.05, size=n_points)
        h_candidate = np.clip(h + perturbation, 0, 1)
        # Normalize to satisfy constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate candidate
        overlaps = np.correlate(h_candidate, 1 - h_candidate, mode="full")
        c5_candidate = np.max(overlaps) * dx
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
    
    return best_h, best_c5, n_points
