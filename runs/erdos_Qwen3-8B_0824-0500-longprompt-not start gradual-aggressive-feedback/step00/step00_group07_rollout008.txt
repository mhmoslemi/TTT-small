import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial guess
    n_points = 80
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    
    # Ensure sum is exactly n_points / 2
    if abs(sum_h - n_points / 2) > 1e-6:
        h += (n_points / 2 - sum_h) / n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h = np.pad(h, (0, 1), mode='constant', constant_values=0)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization function: minimize C5 with constraints
    def optimize_h(h_values):
        h = np.array(h_values)
        dx = 2.0 / len(h)
        sum_h = np.sum(h)
        
        # Constraint: sum(h) = n_points / 2
        if abs(sum_h - n_points / 2) > 1e-6:
            return np.inf
        
        # Compute C5
        h = np.pad(h, (0, 1), mode='constant', constant_values=0)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        
        return c5
    
    # Simulated annealing with a small number of iterations
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_n_points = n_points
    iterations = 0
    
    while time.time() - start_time < budget_s - 1:
        # Perturb the current solution
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = np.clip(h + perturbation, 0, 1)
        perturbed_h = np.pad(perturbed_h, (0, 1), mode='constant', constant_values=0)
        sum_perturbed = np.sum(perturbed_h)
        
        # Adjust to meet the constraint
        if abs(sum_perturbed - n_points / 2) > 1e-6:
            adjustment = (n_points / 2 - sum_perturbed) / n_points
            perturbed_h += adjustment
        
        # Compute C5 for the perturbed solution
        current_c5 = compute_c5(perturbed_h)
        
        # Accept the new solution with a probability based on temperature
        temperature = max(0.99 ** iterations, 0.01)
        if current_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - current_c5) / temperature):
            best_h = perturbed_h
            best_c5 = current_c5
            best_n_points = n_points
        
        iterations += 1
    
    return best_h[:n_points], best_c5, best_n_points
