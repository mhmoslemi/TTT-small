import numpy as np
import scipy.optimize as optimize
import time
import copy
from scipy.optimize import Bounds
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Define the number of points and the step size
    n_points = 200  # Choose a reasonable number of points
    dx = 2.0 / n_points
    n_half = n_points // 2
    
    # Initial guess from provided initial_h_values (scaled to n_points)
    initial_h = np.array(initial_h_values).copy()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    
    # Function to compute C5 value given a function h
    def compute_c5(h):
        h = h.reshape(-1)
        h = np.concatenate((h, h[:n_half]))  # Pad for correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr[n_half:-n_half]) * dx
        return c5
    
    # Define the objective function for optimization: minimize C5
    def objective(h):
        return compute_c5(h)
    
    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i] (0 <= h[i] <= 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h_initial = initial_h.copy()
    
    # Start timer
    start_time = time.time()
    
    # Use L-BFGS-B optimization with bounds
    result = optimize.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # If optimization didn't improve, try a random perturbation
    if best_c5 > 0.38092:  # Current record
        # Perturb the initial guess slightly and optimize again
        np.random.seed(seed + 1)
        h_initial = np.random.uniform(0, 1, n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        result = optimize.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            jac=None,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        best_h = result.x
        best_c5 = compute_c5(best_h)
    
    # Check time budget and return the best solution
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Further refine using a local search
        np.random.seed(seed + 2)
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            h_perturbed = best_h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            current_c5 = compute_c5(h_perturbed)
            if current_c5 < best_c5:
                best_h = h_perturbed
                best_c5 = current_c5
    
    return best_h, best_c5, n_points
