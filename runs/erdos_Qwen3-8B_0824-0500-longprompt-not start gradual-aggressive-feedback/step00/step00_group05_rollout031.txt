```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 70
    h = initial_h_values.copy()
    
    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / n_points
        h_vec = np.array(h)
        # Shift h by k = 1 to 2 (since we're working on [0,2] and h is 1D)
        # For simplicity, we only check k=1 and k=2 (shifts that overlap)
        # This is not exhaustive but gives a rough estimate
        c5_values = []
        for k in range(1, 3):
            shifted = np.roll(h_vec, k)
            overlap = np.sum(h_vec * (1 - shifted)) * dx
            c5_values.append(overlap)
        return max(c5_values)
    
    # Optimization objective: minimize C5
    def objective(x):
        h = x.copy()
        return compute_c5(h)
    
    # Optimization bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Start with initial guess
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_n_points = n_points
    
    # Try increasing number of points to improve accuracy
    for n_points in [70, 100, 150, 200, 250, 300, 350, 400, 450, 500]:
        # Create new h with n_points samples
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (n_points / 2)  # Ensure the integral is 1
        h = np.clip(h, 0, 1)
        
        # Use a gradient-based optimization to minimize C5
        start_time = time.time()
        result = opt.minimize(objective, h, bounds=bounds, method='L-BFGS-B', options={'ftol': 1e-6, 'maxiter': 100})
        elapsed = time.time() - start_time
        
        if elapsed > budget_s:
            break
        
        current_h = result.x
        current_c5 = compute_c5(current_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            best_n_points = n_points
    
    # Evaluate the best solution and return
    return best_h, best_c5, best_n_points
```
