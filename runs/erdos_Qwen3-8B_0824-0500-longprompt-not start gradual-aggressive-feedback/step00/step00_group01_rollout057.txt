```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(47) * 2 - 1  # Example initial values (for demonstration)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for better resolution and speed
    
    # Initial guess from initial_h_values
    initial_h = np.interp(np.linspace(0, 2, n_points), np.sort(initial_h_values), np.linspace(0, 1, len(initial_h_values)))
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)  # Ensure integral is 1
    
    # Optimization loop
    for _ in range(5):  # Limited iterations to stay within time budget
        # Define variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        
        # Maximize the overlap integral over all k
        max_overlap = 0
        for k in range(1, n_points):
            if k >= n_points:
                break
            # Compute the overlap for this shift
            overlap = cp.sum(h * (1 - h[k:])) / (n_points / 2)
            max_overlap = cp.Max(max_overlap, overlap)
        
        # Objective is to minimize the maximum overlap
        objective = cp.Minimize(max_overlap)
        
        # Solve the convex problem
        prob = cp.Problem(objective, constraints)
        try:
            result = prob.solve(solver=cp.SCS, eps=1e-4, max_iter=100)
            if result is not None:
                current_h = np.array(h.value)
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * (n_points / 2) / np.sum(current_h)  # Ensure integral is 1
                
                # Evaluate the C5 bound
                c5 = evaluate_erdos_solution(current_h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = current_h
        except:
            pass
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break
    
    # Final evaluation
    if best_h is None:
        best_h = initial_h
    c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, c5, n_points)
```
