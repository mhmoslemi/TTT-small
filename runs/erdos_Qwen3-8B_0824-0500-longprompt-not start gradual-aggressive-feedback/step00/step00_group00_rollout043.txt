import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp

# Pre-imported helper functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial solution
    h_values = initial_h_values
    n_points = len(h_values)
    dx = 2.0 / n_points
    integral = np.sum(h_values) * dx
    if abs(integral - 1.0) > 1e-8:
        print(f"Warning: Initial solution integral is {integral}, should be 1.0")
    
    # Compute initial C5
    c5 = evaluate_erdos_solution(h_values, dx)
    
    # Try improving
    for _ in range(5):  # Try a few iterations of improvement
        # Try a few different strategies for improvement
        # Strategy 1: Try random perturbations
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h_values + perturbation, 0.0, 1.0)
        integral_perturbed = np.sum(h_perturbed) * dx
        if abs(integral_perturbed - 1.0) < 1e-8:
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < c5:
                h_values = h_perturbed
                c5 = c5_perturbed
        else:
            # Adjust to maintain integral constraint
            delta = (1.0 - integral_perturbed) / (n_points * dx)
            h_perturbed = np.clip(h_values + perturbation + delta, 0.0, 1.0)
            integral_perturbed = np.sum(h_perturbed) * dx
            c5_perturbed = evaluate_erdos_solution(h_perturbed, dx)
            if c5_perturbed < c5:
                h_values = h_perturbed
                c5 = c5_perturbed
        
        # Strategy 2: Try a more structured approach (e.g., periodic patterns)
        # Try a periodic pattern with small adjustments
        period = n_points // 3
        if period > 0:
            h_periodic = np.zeros(n_points)
            for i in range(period):
                h_periodic[i] = 0.5
            h_periodic = np.tile(h_periodic, n_points // period + 1)[:n_points]
            h_periodic = np.clip(h_periodic + np.random.uniform(-0.1, 0.1, size=n_points), 0.0, 1.0)
            integral_periodic = np.sum(h_periodic) * dx
            if abs(integral_periodic - 1.0) < 1e-8:
                c5_periodic = evaluate_erdos_solution(h_periodic, dx)
                if c5_periodic < c5:
                    h_values = h_periodic
                    c5 = c5_periodic
            else:
                # Adjust to maintain integral constraint
                delta = (1.0 - integral_periodic) / (n_points * dx)
                h_periodic = np.clip(h_periodic + delta, 0.0, 1.0)
                integral_periodic = np.sum(h_periodic) * dx
                c5_periodic = evaluate_erdos_solution(h_periodic, dx)
                if c5_periodic < c5:
                    h_values = h_periodic
                    c5 = c5_periodic
        
        # Stop if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Optionally, try a larger set of points for more precision
    if n_points < 500:
        n_points_new = min(500, int(n_points * 2))
        dx_new = 2.0 / n_points_new
        # Interpolate the current h_values to the new number of points
        h_new = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, n_points), h_values)
        h_new = np.clip(h_new, 0.0, 1.0)
        integral_new = np.sum(h_new) * dx_new
        if abs(integral_new - 1.0) > 1e-8:
            # Adjust to maintain integral constraint
            delta = (1.0 - integral_new) / (n_points_new * dx_new)
            h_new = np.clip(h_new + delta, 0.0, 1.0)
        c5_new = evaluate_erdos_solution(h_new, dx_new)
        if c5_new < c5:
            h_values = h_new
            c5 = c5_new
            n_points = n_points_new
            dx = dx_new
    
    # Return the best solution found
    return (h_values, c5, n_points)
