import numpy as np
import scipy.optimize as optimize
import scipy.sparse as sparse
import scipy.sparse.linalg as splinalg
import time
from scipy.optimize.minimize import Bounds
from scipy.optimize import LinearConstraint
import cvxpy as cp
import random
import sys

# Pre-imported functions and variables
evaluate_erdos_solution = None
initial_h_values = None

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    best_c5 = evaluate_erdos_solution(h)
    
    # Try a few different constructions to avoid local optima
    for _ in range(3):
        # Generate a random starting point with fixed integral
        n_points = 100  # Try with 100 points for better resolution
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)  # Ensure h stays within [0, 1]
        
        # Evaluate initial C₅
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Run gradient descent for a few iterations
        max_iter = 50
        learning_rate = 0.01
        for i in range(max_iter):
            # Compute overlaps
            overlaps = np.correlate(h, 1 - h, mode="full")
            corr = overlaps * (2.0 / n_points)
            c5 = np.max(corr)
            
            # Compute gradient for each point
            grad = np.zeros(n_points)
            for k in range(1, n_points - 1):
                # For each k, compute derivative of integral wrt h[i]
                # The derivative of the overlap integral with respect to h[i] is
                # (1 - h[i + k]) - h[i] * (1 - h[i + k])' + ... (depends on overlap)
                # For simplicity, we use a numerical gradient here
                h_plus = h.copy()
                h_plus[i] += 1e-6
                c5_plus = evaluate_erdos_solution(h_plus)
                grad[i] = (c5_plus - c5) / 1e-6
            
            # Update h using gradient descent
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Return the best solution found
    return (best_h.tolist(), best_c5, n_points)
