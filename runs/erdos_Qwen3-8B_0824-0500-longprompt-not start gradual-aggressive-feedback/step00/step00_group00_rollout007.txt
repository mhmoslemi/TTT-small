import numpy as np
import scipy.optimize as opt
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initial parameters
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Adjust initial_h to satisfy the constraint sum(h) = n_points / 2
    # Calculate the difference
    target_sum = n_points / 2
    delta = target_sum - sum_h
    # Add delta to the first element (randomly selected)
    initial_h[0] += delta
    
    def compute_c5(h):
        h = np.array(h)
        # Ensure h is properly scaled
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        shifted_h = np.roll(h, 1)
        overlap = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * dx
        return c5
    
    def objective_func(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization settings
    max_iter = 100
    time_limit = budget_s
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    # Use a local optimizer for faster convergence
    # Use a local search with random restarts
    for _ in range(10):
        # Perturb initial_h to avoid local optima
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = initial_h + perturbation
        h = np.clip(h, 0, 1)
        h[0] = (n_points / 2 - np.sum(h[1:])) / 1  # Ensure constraint
        
        # Use a local optimization method
        res = opt.minimize(objective_func, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
    
    # Use a global optimization method for better convergence
    # Use a genetic algorithm for global search
    # Initial population
    population_size = 20
    population = np.random.uniform(0, 1, (population_size, n_points))
    for i in range(population_size):
        population[i] = np.clip(population[i], 0, 1)
        population[i][0] = (n_points / 2 - np.sum(population[i][1:])) / 1  # Ensure constraint
    
    for _ in range(10):
        # Evaluate fitness
        fitness = np.array([compute_c5(ind) for ind in population])
        
        # Select the best individuals
        best_indices = np.argsort(fitness)[:population_size // 2]
        best_population = population[best_indices]
        
        # Apply crossover
        for i in range(population_size // 2):
            parent1 = best_population[i]
            parent2 = best_population[np.random.choice(len(best_population))]

            for j in range(n_points):

                    child[j] = parent1[j]

            child[0] = (n_points / 2 - np.sum(child[1:])) / 1

        # Apply mutation
        for i in range(population_size):
            for j in range(n_points):
                if np.random.rand() < 0.1:
                    population[i][j] += np.random.uniform(-0.1, 0.1)

        if np.min(fitness) < best_c5:
            best_c5 = np.min(fitness)
            best_h = population[np.argmin(fitness)]
    
    # Final optimization using CVXPY
    h = cp.Variable(n_points)
    objective = cp.Minimize(compute_c5(h))
    constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
    prob = cp.Problem(objective, constraints)
    start_time = time.time()
    prob.solve(solver=cp.SCS, max_iter=100)
    elapsed_time = time.time() - start_time
    
    if prob.status == 'optimal' and elapsed_time < budget_s:
        best_c5 = prob.value
        best_h = h.value
    
    # Ensure best_h satisfies the constraint
    best_h = np.clip(best_h, 0, 1)
    best_h[0] = (n_points / 2 - np.sum(best_h[1:])) / 1
    
    return (best_h, best_c5, n_points)
