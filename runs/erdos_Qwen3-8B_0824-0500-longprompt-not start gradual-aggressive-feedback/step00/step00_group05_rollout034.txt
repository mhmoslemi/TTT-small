import scipy.optimize as optimize
import time

from scipy.signal import correlate
from scipy.optimize import minimize

# Pre-imported functions
def evaluate_erdos_solution(h):
    dx = 2.0 / len(h)
    # Compute the full cross-correlation
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum value of the overlap and normalize by dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def initial_h_values(n_points):
    # Sample initial construction
    h = np.zeros(n_points)
    # Sample initial h values with a simple construction
    # For example, evenly distribute mass in multiple intervals
    for i in range(n_points):
        if i < n_points // 5:
            h[i] = 1.0
        elif i < 2 * n_points // 5:
            h[i] = 0.5
        elif i < 3 * n_points // 5:
            h[i] = 0.0
        elif i < 4 * n_points // 5:

        else:
            h[i] = 1.0
    return h

def run(seed=42, budget_s=60, **kwargs):

    start_time = time.time()

    n_points = 300  # A reasonable number of samples for quick evaluation

    h_initial = initial_h_values(n_points)

    bounds = [(0, 1) for _ in range(n_points)]
