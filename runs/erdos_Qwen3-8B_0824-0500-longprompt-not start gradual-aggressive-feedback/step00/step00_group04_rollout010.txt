def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    
    # Initial settings
    n_points = 100  # Start with 100 points
    max_iter = 100  # Max number of iterations for optimization
    h = np.copy(initial_h_values[:n_points])  # Start from initial construction
    dx = 2.0 / n_points
    h = h * (n_points / 2)  # Scale to make sum(h) == n_points / 2
    
    # Optimization function to minimize C5
    def objective(h_values):
        h_values = np.clip(h_values, 0, 1)
        integral = np.correlate(h_values, 1 - h_values, mode="full") * dx
        return np.max(integral)
    
    # Bounds for the optimization
    bounds = Bounds(0, 1)
    
    # Optimization using L-BFGS-B
    result = minimize(objective, h, bounds=bounds, method='L-BFGS-B', jac=lambda x: None, options={'maxiter': max_iter})
    
    # Update h with optimized values
    h_opt = result.x
    
    # Compute C5 value for the optimized solution
    c5_bound = evaluate_erdos_solution(h_opt)
    
    # If we have time, try increasing n_points for better accuracy
    if time.time() - start_time < budget_s - 5:
        n_points = 150
        h = np.clip(h_opt, 0, 1)
        h = h * (n_points / 2) / (len(h_opt) / 2)  # Adjust scaling
        h = np.pad(h, (0, n_points - len(h)), 'constant')  # Pad to match new n_points

    # Ensure h is within [0,1]
    h_opt = np.clip(h_opt, 0, 1)
    # Return the final solution
    return h_opt, c5_bound, len(h_opt)
