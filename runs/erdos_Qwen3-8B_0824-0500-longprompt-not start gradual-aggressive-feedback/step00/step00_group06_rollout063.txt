```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = signal.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    # Initial construction with 95 points
    return np.load('initial_h.npy')  # Placeholder for actual initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess with 95 points
    n_points = 95
    initial_h = initial_h_values()
    h = np.copy(initial_h)
    
    # Function to evaluate C5
    def evaluate(h_vals):
        h = np.array(h_vals)
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / len(h))
        return c5

    # Optimization function to minimize C5
    def objective(h_vals):
        return evaluate(h_vals)

    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [cp.sum(h) == n_points / 2]

    # Optimization using CVXPY with a solver
    h_var = cp.Variable(n_points, bounds=bounds)
    obj = cp.Minimize(evaluate(h_var))
    prob = cp.Problem(obj, constraints)

    # Run the optimization
    try:
        result = prob.solve(solver='SCIPY', maxiter=1000, tol=1e-6)
        if result is None:
            # If optimization fails, use the initial guess
            h_opt = h
        else:
            h_opt = h_var.value
    except:
        h_opt = h
    
    # Evaluate the result
    c5_bound = evaluate(h_opt)
    elapsed_time = time.time() - start_time
    
    # If we have time, try with a smaller number of points
    if elapsed_time < budget_s - 10:
        n_points = 60
        h = np.zeros(n_points)
        dx = 2.0 / n_points
        h[:int(0.5 * n_points)] = 1.0  # Start with a simple step function
        h = h + np.random.rand(n_points) * 0.1 - 0.05  # Perturb slightly
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        c5_bound = evaluate(h)
        h_opt = h
    
    return h_opt.tolist(), c5_bound, n_points
```
