import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions and initial construction
initial_h_values = np.load('initial_h_values.npy')  # n=70 samples

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Function to compute overlap integral for a given h
    def compute_overlap(h):
        n_points = len(h)
        dx = 2.0 / n_points
        # Compute (1 - h) shifted by k for k = 0, 1, ..., n_points - 1
        # Use convolution to compute overlaps efficiently
        # Note: We need to evaluate for k = -n_points + 1 to n_points - 1
        # But since h is periodic with period 2, we need to consider shifts within [0, 2]
        # We'll use cross-correlation with mode="full" to get all possible shifts
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        # We only care about shifts within [0, 2], which corresponds to k in [0, n_points - 1]
        # So take the middle part of the result
        return np.max(overlap)
    
    # Initial n_points
    n_points_initial = len(initial_h_values)
    h_initial = initial_h_values.copy()
    
    # Try with a smaller number of points first
    n_points = n_points_initial
    
    # Try decreasing n_points to reduce computation
    # Try a few different starting points
    starting_points = [n_points_initial, n_points_initial - 10, n_points_initial - 20]
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points_initial
    
    for n in starting_points:
        if n < 2:
            continue
        
        # Generate initial guess by resampling the initial h
        h = np.zeros(n)
        indices = np.linspace(0, 2, n_points_initial, endpoint=False)
        idx = np.searchsorted(indices, np.linspace(0, 2, n, endpoint=False))
        h = np.zeros(n)
        for i in range(n):
            h[i] = initial_h_values[idx[i] % n_points_initial]
        h = h / (n / 2)  # Ensure area is 1
        
        # Define optimization variables
        h_var = cp.Variable(n)
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n / 2
        ]
        
        # Define overlap integral
        def compute_overlap_with_cvxpy(h_var):
            overlap = np.correlate(h_var.value, 1 - h_var.value, mode="full") * (2.0 / n)
            return np.max(overlap)
        
        # Optimization loop
        for _ in range(10):  # Run a few iterations
            # Objective: minimize max overlap
            # Use a convex relaxation or try to find local minima
            # Here, we try to minimize the maximum overlap using a convex problem
            
            # Create a convex problem
            obj = cp.Minimize(cp.Max(cp.multiply(h_var, 1 - h_var)))
            prob = cp.Problem(obj, constraints)
            try:
                prob.solve(solver=cp.SCS, max_iter=100, eps=1e-5)
                if prob.status == 'optimal':
                    h = h_var.value
                    current_c5 = compute_overlap(h)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h
                        best_n_points = n
            except:
                pass
        
        # Also try local optimization
        def objective(h_values):
            return compute_overlap(h_values)
        
        # Local optimization
        result = opt.minimize(objective, h, bounds=[(0, 1) for _ in range(n)])
        if result.success:
            h = result.x
            current_c5 = compute_overlap(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
                best_n_points = n
    
    # Return the best solution found
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        # If we exceed the time budget, return the best solution found so far
        return (best_h, best_c5, best_n_points)
    else:
        # If we have time left, do one more try with n_points = 50
        n = 50
        h = np.random.rand(n)
        h = h / (n / 2)
        
        # Optimizer
        result = opt.minimize(objective, h, bounds=[(0, 1) for _ in range(n)])
        if result.success:
            h = result.x
            current_c5 = compute_overlap(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
                best_n_points = n
        
        return (best_h, best_c5, best_n_points)
