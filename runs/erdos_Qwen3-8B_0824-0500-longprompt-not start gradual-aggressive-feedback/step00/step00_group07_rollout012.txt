import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import copy
from scipy.optimize import Bounds

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Start with 100 points (adjust if needed)
    dx = 2.0 / n_points
    max_iterations = 100  # Limit the number of iterations to stay within time budget
    
    # Initial guess
    h = np.copy(initial_h_values[:n_points])
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet the constraint
    
    # Define the function to minimize
    def objective_function(h_values):
        h = h_values.copy()
        # Ensure h values are within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        corr = signal.correlate(h, 1 - h, mode='full')
        # Compute C5 as the maximum of the correlations multiplied by dx
        c5 = np.max(corr) * dx
        return c5
    
    # Define the constraint: sum(h) == n_points / 2
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Setup optimization problem
    bounds = Bounds(0, 1)
    cons = {'type': 'eq', 'fun': constraint}
    
    # Optimization
    best_h = h
    best_c5 = objective_function(h)
    
    start_time = time.time()
    
    for _ in range(max_iterations):
        # Perform a local optimization step
        res = optimize.minimize(objective_function, h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        if res.success:
            h = res.x
            current_c5 = objective_function(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
            # Early stopping if we're close to the target
            if best_c5 <= 0.380876:
                break
        else:
            # If optimization fails, try a random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, size=h.shape)
            h = np.clip(h + perturbation, 0, 1)
    
    # Final evaluation
    final_c5 = objective_function(best_h)
    
    # Return the result
    return (best_h, final_c5, n_points)
