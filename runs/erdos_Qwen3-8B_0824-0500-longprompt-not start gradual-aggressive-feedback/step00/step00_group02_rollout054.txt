import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Starting with 45 points as per initial_h_values
    n_points = 45
    h = copy.deepcopy(initial_h_values)
    
    # Function to compute the overlap integral C_5
    def compute_c5(h):
        dx = 2.0 / n_points
        h = np.array(h)
        h_shifted = np.zeros((n_points, n_points))
        for i in range(n_points):
            h_shifted[i, :] = h[i:] if i <= n_points else h[i - n_points:]
        overlap = np.zeros(n_points)
        for k in range(n_points):
            overlap[k] = np.dot(h, 1 - h_shifted[k, :]) * dx
        return np.max(overlap)
    
    # Function to update h with a small perturbation
    def perturb_h(h, magnitude=0.05):
        h = np.array(h)
        perturbation = np.random.uniform(-magnitude, magnitude, size=h.shape)
        h = np.clip(h + perturbation, 0, 1)
        return h.tolist()
    
    # Optimization loop
    best_h = h
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    for _ in range(1000):  # Limit iterations to stay within time budget
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the current h values
        h = perturb_h(h)
        
        # Ensure the integral constraint is satisfied
        dx = 2.0 / n_points
        total_integral = np.sum(h) * dx
        if abs(total_integral - 1) > 1e-6:
            # Adjust h to meet the integral constraint
            delta = (1 - total_integral) / n_points
            h = np.clip(h + delta, 0, 1).tolist()
        
        # Compute C_5 for the new h
        current_c5 = compute_c5(h)
        
        # Update best solution if current is better
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
