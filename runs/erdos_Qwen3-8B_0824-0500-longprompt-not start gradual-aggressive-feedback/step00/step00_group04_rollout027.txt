import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp


def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values (n=42 samples)
    n_points = len(initial_h_values)
    h = np.array(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(h) * dx
    if abs(sum_h - 1.0) > 1e-6:
        h = h * (1.0 / sum_h) * 2.0  # Normalize to ensure ∫ h(x) dx = 1
    
    def evaluate_c5(h_values):
        h = np.array(h_values)
        dx = 2.0 / len(h)
        # Compute (1 - h(x + k)) for k = 0, ..., len(h)-1
        # We use numpy.correlate with mode="full" for convolution
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode="full")
        # The relevant part is from index len(h)-1 to 2*len(h)-2
        relevant_corr = corr[len(h)-1:2*len(h)-1]
        # The overlap integral is corr * dx
        c5 = np.max(relevant_corr) * dx
        return c5
    
    def objective_function(h_values):
        c5 = evaluate_c5(h_values)
        return c5
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2  # Ensure ∫ h(x) dx = 1
    
    # Optimization setup
    h_vars = cp.Variable(n_points, lb=0.0, ub=1.0)
    objective = cp.Minimize(objective_function(h_vars))
    constraints = [constraint_sum(h_vars) == 0]
    
    # Solve with CVXOPT (or any other solver if needed)
    prob = cp.Problem(objective, constraints)
    
    start_time = time.time()
    try:
        result = prob.solve(solver=cp.CVXOPT, verbose=False)
        if result is not None:
            h_opt = h_vars.value
            c5_bound = evaluate_c5(h_opt)
            return (h_opt.tolist(), c5_bound, n_points)
    except:
        pass
    
    # If CVXOPT fails, try a local optimization method such as SLSQP
    h_opt = h.copy()
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    
    def objective_func(x):
        return evaluate_c5(x)
    
    result = optimize.minimize(objective_func, h_opt, bounds=bounds, constraints=constraints, method='SLSQP', tol=1e-6)
    if result.success:
        h_opt = result.x
        c5_bound = evaluate_c5(h_opt)
        return (h_opt.tolist(), c5_bound, n_points)
    
    # If optimization fails, return the initial values
    return (h.tolist(), evaluate_c5(h), n_points)
