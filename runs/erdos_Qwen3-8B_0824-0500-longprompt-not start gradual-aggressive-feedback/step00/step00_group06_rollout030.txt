def run(seed=42, budget_s=60, **kwargs):

    dx = 2.0 / n_points
    
    def objective(h):
        # Ensure h is within [0,1]
        h = np.clip(h, 0, 1)
        # Ensure integral is 1
        total = np.sum(h) * dx
        h = h / (total / dx) if total != 0 else h
        # Compute the overlap integral
        h_shifted = np.roll(h, 1)
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    def constraint_total(h):
        return np.sum(h) - n_points / 2
    
    # Objective function to minimize (we minimize the max overlap)
    def objective_wrapper(h):
        return objective(h)
    
    # Constraint to ensure the integral is 1
    cons = [{'type': 'eq', 'fun': constraint_total}]
    
    # Bounds: each h_i must be in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Starting point
    start_time = time.time()
    result = opt.minimize(objective_wrapper, h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-4, options={'maxiter': 100})
    
    # Best solution found
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Check if we have time left to try another start
    if time.time() - start_time < budget_s - 5:
        # Try a different random start
        np.random.seed(seed + 1)
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand) if np.sum(h_rand) != 0 else h_rand
        result_rand = opt.minimize(objective_wrapper, h_rand, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-4, options={'maxiter': 100})
        best_h = result_rand.x
        best_c5 = objective(best_h)
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h) * dx
    best_h = best_h * (n_points / 2) / total if total != 0 else best_h
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
