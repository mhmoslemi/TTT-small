```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import sys

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Parameters
    n_points_list = [50, 100, 150, 200]  # Try different resolutions
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)
        
        # Initial guess - use initial_h_values if available
        if hasattr(initial_h_values, '__iter__') and len(initial_h_values) == n_points:
            h_initial = np.array(initial_h_values, dtype=np.float64)
        else:
            h_initial = np.random.uniform(0, 1, n_points)
        
        # Normalize to satisfy the integral constraint
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define function to evaluate C5 for given h
        def objective(h):
            h = np.clip(h, 0, 1)
            integral = np.correlate(h, 1 - h, mode="full") * dx
            c5 = np.max(integral)
            return c5
        
        # Define bounds for h
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use differential evolution for global optimization
        result = optimize.differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            seed=seed,
            disp=False
        )
        
        # Extract best solution
        h_opt = result.x
        c5 = result.fun
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n_points = n_points
        
        print(f"n_points={n_points}, c5={c5:.6f}, best_c5={best_c5:.6f}")
    
    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_erdos_solution(best_h)
    print(f"Final c5 bound: {final_c5:.6f}")
    
    # Return best solution and bound
    return (best_h, final_c5, best_n_points)
```
