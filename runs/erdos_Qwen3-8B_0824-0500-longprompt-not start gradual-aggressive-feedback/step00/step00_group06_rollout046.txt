import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Placeholder for actual initial values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try with a small number of points (n_points = 100)
    n_points = 100
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    # Start with a uniform distribution
    h = np.random.rand(n_points)
    h = h / np.sum(h) * (n_points / 2)
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        corr = signal.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Function to compute the overlap integral
    def objective(h):
        return evaluate_overlap(h)
    
    # Function to apply gradient descent
    def gradient_descent(initial_h, max_iter=1000, learning_rate=0.01, tolerance=1e-6):
        h = initial_h.copy()
        for i in range(max_iter):
            current_value = evaluate_overlap(h)
            # Compute gradient numerically
            grad = np.zeros_like(h)
            for j in range(len(h)):
                h_plus = h.copy()
                h_plus[j] += 1e-6
                h_minus = h.copy()
                h_minus[j] -= 1e-6
                grad[j] = (evaluate_overlap(h_plus) - evaluate_overlap(h_minus)) / (2 * 1e-6)
            # Update h
            h -= learning_rate * grad
            # Clip h to [0, 1]
            h = np.clip(h, 0, 1)
            # Enforce integral constraint
            h = h * (n_points / 2) / np.sum(h)
            # Check for convergence
            if i % 100 == 0:
                print(f"Iteration {i}: C5 = {current_value}")
            if np.abs(grad).max() < tolerance:
                break
        return h
    
    # Run gradient descent
    h_opt = gradient_descent(h, max_iter=1000, learning_rate=0.01)
    c5 = evaluate_overlap(h_opt)
    print(f"Final C5 with n_points={n_points}: {c5}")
    
    # If time allows, try with a higher number of points
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        h_opt = gradient_descent(h, max_iter=1000, learning_rate=0.01)
        c5 = evaluate_overlap(h_opt)
        print(f"Final C5 with n_points={n_points}: {c5}")
    
    # Return the best solution found
    best_h = h_opt
    best_c5 = c5
    return best_h, best_c5, n_points
