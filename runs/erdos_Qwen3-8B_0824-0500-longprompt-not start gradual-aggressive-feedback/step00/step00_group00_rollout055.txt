import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1-h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75, 0.75])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    best_n_points = len(best_h)
    n_points = len(best_h)
    
    def evaluate(h):
        nonlocal best_c5, best_n_points
        if len(h) != n_points:
            h = np.resize(h, n_points)
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n_points
        return c5
    
    # Initial evaluation
    c5 = evaluate(best_h)
    
    # Local search with perturbations
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        perturb = np.random.choice([0, 1], size=n_points, p=[0.95, 0.05])
        perturb *= np.random.uniform(-0.2, 0.2, size=n_points)
        h = best_h + perturb
        h = np.clip(h, 0, 1)
        if np.abs(np.sum(h) - n_points / 2) < 1e-6:
            c5 = evaluate(h)
    
    # Try increasing n_points
    for n in range(len(best_h) + 1, min(len(best_h) + 10, 1000)):
        if time.time() - start_time > budget_s - 1:
            break
        # Interpolate initial_h_values to n points
        h = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        h = np.clip(h, 0, 1)
        # Perturb
        perturb = np.random.choice([0, 1], size=n, p=[0.95, 0.05])
        perturb *= np.random.uniform(-0.2, 0.2, size=n)
        h = h + perturb
        h = np.clip(h, 0, 1)
        if np.abs(np.sum(h) - n / 2) < 1e-6:
            c5 = evaluate(h)
    
    # Try using a more structured approach (e.g., two different segments)
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        # Split into two segments
        split = np.random.randint(1, n_points - 1)
        h1 = np.random.uniform(0, 1, split)
        h2 = np.random.uniform(0, 1, n_points - split)
        h = np.concatenate([h1, h2])
        h = np.clip(h, 0, 1)
        if np.abs(np.sum(h) - n_points / 2) < 1e-6:
            c5 = evaluate(h)
    
    # Return best solution
    return (best_h, best_c5, best_n_points)
