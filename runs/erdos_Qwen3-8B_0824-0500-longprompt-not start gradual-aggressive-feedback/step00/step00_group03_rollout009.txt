# Define the objective function to minimize the maximum overlap integral
def compute_c5(h, dx):
    h = np.array(h)
    # Compute 1 - h
    one_minus_h = 1.0 - h
    # Compute the correlation
    corr = np.correlate(h, one_minus_h, mode="full")
    # Normalize by dx
    c5 = np.max(corr * dx)
    return c5

def constraint_sum(h, dx):

def constraint_bounds(h):
    return np.all((h >= 0) & (h <= 1))

# Define the run function
def run(seed=42, budget_s=60, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)

    # Start timing
    start_time = time.time()
    remaining_time = budget_s

    # Initial guess: use the provided initial_h_values
    initial_samples = initial_h_values
    n_points = len(initial_samples)
    dx = 2.0 / n_points

    # Evaluate the initial solution
    c5_initial = compute_c5(initial_samples, dx)
    best_c5 = c5_initial
    best_h = initial_samples.copy()

    # Optimization parameters
    max_iterations = 50
    for iteration in range(max_iterations):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time <= 0:
            break

        # Create bounds for the optimization
        bounds = [(0.0, 1.0) for _ in range(n_points)]

        # Set up the optimization problem
        # Objective function: minimize the maximum overlap integral
        # We use a scalar optimization with a penalty approach

        # Define the optimization problem
        def objective_function(h_values):
            h = np.array(h_values)
            # Check bounds
            if not constraint_bounds(h):
                return float('inf')
            # Compute c5
            c5 = compute_c5(h, dx)
            return c5

        # Run the optimization
        result = opt.minimize(
            objective_function,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 10, 'ftol': 1e-6, 'disp': False}
        )

        # Check if the optimization was successful
        if result.success:
            current_h = result.x
            current_c5 = compute_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()

        # Print progress
        if iteration % 10 == 0:
            print(f"Iteration {iteration}, Best C5: {best_c5:.6f}")

    # Final evaluation
    final_c5 = compute_c5(best_h, dx)

    # Return the result
    return (best_h, final_c5, n_points)
