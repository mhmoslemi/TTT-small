```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.random.rand(80) * 2 - 1  # Example initial guess

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try different numbers of points
    n_points_list = [100, 150, 200, 250, 300]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Initial guess: spread out the 1's
        initial_h = np.zeros(n_points)
        initial_h[::n_points//5] = 1.0
        
        # Optimization with CVXPY
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Objective: minimize the maximum overlap
        overlap = np.zeros(n_points)
        for k in range(n_points):
            overlap[k] = cp.sum(h * (1 - h[:(n_points - k)]))
        
        objective = cp.Minimize(cp.Max(overlap))
        prob = cp.Problem(objective, constraints)
        
        try:
            result = prob.solve(solver='ECOS', eps=1e-5)
            if result is not None:
                h_values = h.value
                c5 = evaluate_erdos_solution(h_values)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
                    best_n_points = n_points
        except:
            pass
        
        # Fallback to random initialization if optimization fails
        if best_c5 == float('inf'):
            h_values = np.random.rand(n_points)
            h_values = (h_values * 2 - 1) * (n_points / 2) / np.sum(h_values)
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values
                best_n_points = n_points
    
    # Return best solution found
    return (best_h, best_c5, best_n_points)
```
