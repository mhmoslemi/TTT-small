    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)

    def compute_c5(h_values):
        h = h_values.reshape((n_points, 1))
        shifted = np.roll(h, shift=1, axis=0)
        overlap = np.sum(h * (1 - shifted))
        return overlap * (2.0 / n_points)
    
    # Define the optimization problem
    # Objective: minimize the maximum overlap
    # Constraints: sum(h) = n_points / 2, 0 <= h[i] <= 1
    
    # We will use a heuristic optimization approach due to the complexity of the problem
    # We will perform a gradient-based search with some random perturbations
    
    def objective_function(h_values):
        # Compute the maximum overlap for all shifts
        overlaps = []
        for shift in range(1, n_points // 2 + 1):
            shifted = np.roll(h_values, shift=shift)
            overlap = np.sum(h_values * (1 - shifted)) * (2.0 / n_points)
            overlaps.append(overlap)
        return np.max(overlaps)
    
    # Define the optimization bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization using L-BFGS-B
    result = optimize.minimize(
        objective_function,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # If time allows, perform a local search around the result
    if time.time() - start_time < budget_s - 10:
        # Define a local search
        def local_search(h_values):
            perturb = np.random.uniform(-0.05, 0.05, size=h_values.shape)
            perturbed = np.clip(h_values + perturb, 0.0, 1.0)
            return perturbed
    
        # Perform a few local searches
        for _ in range(5):
            h_perturbed = local_search(result.x)
            result = optimize.minimize(
                objective_function,
                h_perturbed,
                method='L-BFGS-B',
                bounds=bounds,

            )

    final_c5 = objective_function(result.x)
