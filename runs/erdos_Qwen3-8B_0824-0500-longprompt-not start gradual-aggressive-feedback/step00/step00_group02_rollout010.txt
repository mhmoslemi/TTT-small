```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import sys
import math

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define helper functions
    def compute_overlap(h):
        h = np.array(h)
        dx = 2.0 / len(h)
        h_pad = np.concatenate([h, np.zeros_like(h)])
        h_shifted = np.zeros_like(h_pad)
        for k in range(len(h)):
            h_shifted[k:] = h[:len(h_pad) - k]
        integral = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(integral)
    
    def objective_function(x):
        return compute_overlap(x)
    
    def constraint_sum(x):
        return np.sum(x) - len(x) / 2
    
    def bounds():
        return Bounds(0, 1)
    
    # Initial guess
    n_points = 100
    h_initial = initial_h_values[:n_points]
    h_initial = np.array(h_initial)
    h_initial = h_initial[:n_points]
    
    # Refine n_points if needed
    n_points = min(500, n_points)  # Limit to 500 for performance
    
    # Set up optimization problem
    x0 = h_initial
    bounds = Bounds(0, 1)
    cons = LinearConstraint(np.ones(n_points), lb=0, ub=n_points / 2)
    
    # Optimization settings
    max_iter = 100
    tolerance = 1e-4
    
    start_time = time.time()
    result = minimize(objective_function, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=tolerance, options={'maxiter': max_iter})
    
    # Check if we have time left for further refinements
    elapsed = time.time() - start_time
    if elapsed < budget_s - 10:
        # Try with more points if time permits
        n_points = min(600, n_points + 100)
        x0 = np.linspace(0, 1, n_points)
        x0 = np.random.rand(n_points)
        x0 = x0 * np.sum(h_initial) / np.sum(x0)
        x0 = np.array(x0)
        x0 = x0 * (len(x0) / 2) / np.sum(x0)
        bounds = Bounds(0, 1)
        cons = LinearConstraint(np.ones(n_points), lb=0, ub=n_points / 2)
        result = minimize(objective_function, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=tolerance, options={'maxiter': max_iter})
    
    h_values = result.x
    c5_bound = compute_overlap(h_values)
    n_points = len(h_values)
    
    return h_values, c5_bound, n_points
```
