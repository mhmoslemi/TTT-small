import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
# evaluate_erdos_solution() is used for evaluation
# initial_h_values is the starting point for the optimization

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 256  # Start with 256 points, can be adjusted if needed

    # Initial guess
    h_initial = initial_h_values.copy()
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        nonlocal best_c5
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = np.concatenate([h, np.zeros(n_points)])  # Extend for full convolution
        corr = signal.correlate(h, 1 - h, mode="full")
        overlap = corr * (2.0 / n_points)
        c5 = np.max(overlap)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h[:n_points].copy()
        return c5

    # Function to convert h to a list for optimization
    def objective(h_list):
        return evaluate_overlap(h_list)

    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    result = optimize.minimize(
        objective,
        x0=h_initial,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-7, 'maxiter': 50}
    )
    
    # If optimization is not sufficient, try a more intensive method
    if time.time() - start_time < budget_s - 5:
        result = optimize.minimize(
            objective,
            x0=result.x,
            bounds=bounds,
            method='SLSQP',
            options={'ftol': 1e-7, 'maxiter': 30}
        )
    
    # Final evaluation
    final_c5 = evaluate_overlap(result.x)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = result.x.copy()
    
    # Trim to the best solution found
    return (best_h, best_c5, n_points)
