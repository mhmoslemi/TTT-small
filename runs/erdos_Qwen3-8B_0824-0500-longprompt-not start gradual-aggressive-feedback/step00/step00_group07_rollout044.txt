```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80  # Start with 80 points to balance speed and accuracy
    dx = 2.0 / n_points
    target_C5 = 0.380876
    
    # Initialize with the current construction
    h = np.array(initial_h_values, dtype=np.float64)
    h = h[:n_points]  # Ensure it's the correct length
    h = h / (np.sum(h) * dx)  # Normalize to satisfy the integral constraint
    
    # Function to compute the overlap integral
    def compute_C5(h):
        h = np.pad(h, (0, 1), mode='constant')  # Pad for full convolution
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Optimization objective: minimize the maximum overlap integral
    def objective(h):
        return compute_C5(h)
    
    # Optimization with bounds and constraints
    def optimize_h(h):
        n = len(h)
        x = cp.Variable(n)
        constraints = [
            0 <= x,
            x <= 1,
            cp.sum(x) == n / 2  # Ensure the integral constraint is satisfied
        ]
        prob = cp.Problem(cp.Minimize(objective(x)), constraints)
        prob.solve(solver=cp.SCS, eps=1e-4)
        if prob.status not in ['optimal', 'success']:
            return h
        return x.value
    
    # Iterative optimization
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Try a few iterations of optimization
    for _ in range(3):
        h = optimize_h(h)
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
        if time.time() - start_time > budget_s:
            break
    
    # Try increasing n_points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5 * budget_s:
        n_points = 120
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h / (np.sum(h) * dx)
        for _ in range(2):
            h = optimize_h(h)
            current_C5 = compute_C5(h)
            if current_C5 < best_C5:
                best_h = h.copy()
                best_C5 = current_C5
            if time.time() - start_time > budget_s:
                break
    
    # Final evaluation
    result = evaluate_erdos_solution(best_h)
    return (best_h, result, n_points)
```
