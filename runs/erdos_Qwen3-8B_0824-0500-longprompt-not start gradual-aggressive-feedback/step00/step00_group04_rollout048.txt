```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy import sparse
from scipy.sparse import lil_matrix
from scipy.sparse.linalg import lsqr
import cvxpy as cp

# Pre-imported helper functions
initial_h_values = np.load('initial_h_values.npy')  # Assume this is the initial construction

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 400  # Start with 400 points for faster initial evaluation
    
    # Initial guess
    h_initial = np.copy(initial_h_values[:n_points])
    # Normalize to meet the integral constraint
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.pad(h, (0, 1), mode='constant')  # Pad for full convolution
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    # Function to optimize h
    def optimize_h(h_guess):
        nonlocal best_c5, best_h
        h = np.copy(h_guess)
        # Normalize to meet the integral constraint
        h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = np.copy(h)
        return c5
    
    # Simple gradient descent with momentum
    def gradient_descent(h_initial, iterations=1000, learning_rate=0.01, momentum=0.9):
        h = np.copy(h_initial)
        velocity = np.zeros_like(h)
        for i in range(iterations):
            # Compute the gradient of the cost function
            # For simplicity, use finite differences
            grad = np.zeros_like(h)
            for idx in range(len(h)):
                h_plus = np.copy(h)
                h_plus[idx] += 1e-5
                h_minus = np.copy(h)
                h_minus[idx] -= 1e-5
                c_plus = compute_c5(h_plus)
                c_minus = compute_c5(h_minus)
                grad[idx] = (c_plus - c_minus) / (2 * 1e-5)
            # Update velocity and h
            velocity = momentum * velocity - learning_rate * grad
            h += velocity
            # Clip h to [0, 1]
            h = np.clip(h, 0, 1)
            # Normalize to meet the integral constraint
            h = h * (n_points / 2) / np.sum(h)
            # Early stopping if time is up
            if time.time() - start_time > budget_s - 1:
                break
        return h
    
    # Run the optimization
    best_h = gradient_descent(h_initial, iterations=500)
    best_c5 = compute_c5(best_h)
    
    # Check if we can improve by increasing the number of points
    if time.time() - start_time < budget_s - 10:
        n_points = 800
        h_initial = np.zeros(n_points)
        h_initial[:len(initial_h_values)] = initial_h_values[:len(initial_h_values)]
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        best_h = gradient_descent(h_initial, iterations=300)
        best_c5 = compute_c5(best_h)
    
    # Final check
    if time.time() - start_time < budget_s - 5:
        # Refine with another round
        best_h = gradient_descent(best_h, iterations=200)
        best_c5 = compute_c5(best_h)
    
    # Ensure best_h is within [0, 1] and normalized
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return best_h, best_c5, n_points
```
