```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial solution
    initial_h_values = np.load("initial_h.npy")
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    
    # Initial check if it satisfies the constraint
    if np.isclose(sum_h, n_points / 2):
        h = initial_h_values.copy()
    else:
        # Adjust initial solution to satisfy the constraint
        h = initial_h_values.copy()
        excess = sum_h - n_points / 2
        # Distribute the excess evenly
        h += excess / n_points
        h = np.clip(h, 0, 1)
    
    # Define the function to evaluate C5
    def evaluate_c5(h_values):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        n_points = len(h)
        dx = 2.0 / n_points
        # Compute the cross-correlation
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Function to compute the cost for optimization
    def objective(h_values):
        return evaluate_c5(h_values)
    
    # Function to compute the constraint
    def constraint_func(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Optimization parameters
    max_iter = 50
    n_points_list = [80, 100, 120, 160, 200]
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Create the optimization variable
        h = cp.Variable(n_points)
        # Objective: minimize the max overlap
        obj = cp.Minimize(objective(h))
        # Constraint: sum of h is n_points / 2
        cons = [constraint_func(h) == 0]
        
        # Solve the optimization problem
        try:
            prob = cp.Problem(obj, cons)
            prob.solve(solver=cp.SCS, max_iter=1000)
            if prob.status == 'optimal':
                h_values = h.value
                current_c5 = evaluate_c5(h_values)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_values.copy()
        except:
            pass
    
    return best_h, best_c5, len(best_h)
```
