import random
import time

    generations = 100

    mutation_strength = 0.05
    elite_fraction = 0.2
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def evaluate(h):
        h = np.array(h)
        h = normalize(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(overlap)
    
    # Initialize population
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    population = [normalize(p) for p in population]
    
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    for generation in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate population
        fitness = [evaluate(individual) for individual in population]
        
        # Find the best individual
        best_index = np.argmin(fitness)
        best_c5_current = fitness[best_index]
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = population[best_index].copy()
        
        # Select parents
        parents = np.argsort(fitness)[:int(elite_fraction * population_size)]
        parents = [population[i] for i in parents]
        
        # Create new population
        new_population = []
        for _ in range(population_size - len(parents)):
            # Select two parents
            parent1 = random.choice(parents)
            parent2 = random.choice(parents)

                child += mutation

            # Normalize
            child = normalize(child)
            new_population.append(child)
        
        # Replace population
        population = parents + new_population
    
    return (best_h, best_c5, n_points)
