import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    random.seed(seed)
    
    # Initial guess: uniform distribution
    n_points = 70
    initial_h = np.random.uniform(0, 1, n_points)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Function to evaluate the overlap integral with penalty for constraint violation
    def objective_with_penalty(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        # Penalty for constraint violation
        constraint_violation = abs(np.sum(h) * (2.0 / n_points) - 1.0)
        return c5 + 1000 * constraint_violation
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0

    def local_optimization(h_initial):

            objective_with_penalty,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},

        )
