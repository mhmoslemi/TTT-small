            return h
        return h * (n_points / 2) / total

    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        # Compute the correlation
        overlap = np.correlate(h, 1 - h, mode="full")
        # Scale the result
        overlap_scaled = overlap * (2.0 / n_points)
        # Find the maximum overlap
        c5 = np.max(overlap_scaled)
        return c5

        h = np.array(h)

        constraint_violation = abs(np.sum(h) * (2.0 / n_points) - 1.0)
        return c5 + 100 * constraint_violation

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint(h):
        h = np.array(h)

        return total - 1.0

    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
