import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):

    initial_h = initial_h_values
    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Local optimization with SLSQP
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)
    
    # Function to create adaptive perturbations
    def adaptive_perturb(h, magnitude=0.01, seed=None):
        if seed is not None:
            np.random.seed(seed)
        perturbation = np.random.uniform(-magnitude, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        np.random.seed()  # Reset seed
        return h_perturbed
    
    # Function to generate random samples around the current best
    def random_sample_around(h, magnitude=0.01):
        perturbation = np.random.uniform(-magnitude, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization loop with adaptive perturbations and global sampling
    for _ in range(100):

        perturbed_h = random_sample_around(best_h, magnitude=0.01)

            best_c5 = c5_lo
