import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Generate multiple initial candidates
    n_points = 100  # Use 100 points for a good balance between resolution and speed
    initial_candidates = []
    for _ in range(5):  # Generate 5 initial candidates
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        initial_candidates.append(h)
    
    best_c5 = np.inf
    best_h = initial_candidates[0].copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Function to apply adaptive perturbation
    def adaptive_perturb(h, magnitude=0.01, decay=0.999):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization loop
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Select a random initial candidate
        h = initial_candidates[np.random.randint(len(initial_candidates))]
        
        # Perturb the solution
        h_perturbed = adaptive_perturb(h, magnitude=0.01, decay=0.999)
        
        # Refine the perturbed solution with local optimization
        result = optimize.minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,

        )
        
        h_lo = result.x
