import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Local optimization with SLSQP
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)
    
    # Function to create adaptive perturbations
    def adaptive_perturb(h, magnitude=0.01, seed=None):
        if seed is not None:
            np.random.seed(seed)
        perturbation = np.random.uniform(-magnitude, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        np.random.seed()  # Reset seed
        return h_perturbed
    
    # Function to perform block coordinate descent
    def block_coordinate_descent(h, num_blocks=10, block_size=10):
        h = h.copy()
        for _ in range(num_blocks):
            for i in range(0, len(h), block_size):
                block = h[i:i+block_size]
                # Local optimization on the block
                h_block = optimize.minimize(
                    lambda x: objective(np.concatenate((h[:i], x, h[i+block_size:]))),
                    block,
                    method='L-BFGS-B',
                    bounds=[(0.0, 1.0) for _ in range(len(block))],
                    options={'maxiter': 50, 'ftol': 1e-9, 'disp': False}
                ).x
                h = np.concatenate((h[:i], h_block, h[i+block_size:]))
        return h
    
    # Main optimization loop with adaptive perturbations and block coordinate descent
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        # Perturb the current best solution with adaptive directions
        perturbed_h = adaptive_perturb(best_h, magnitude=0.005)
        # Refine the perturbed solution with block coordinate descent
        h_bc = block_coordinate_descent(perturbed_h)
        # Refine further with local optimization
        h_lo, c5_lo = local_optimization(h_bc)
        # Update the best solution if improvement is found
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
