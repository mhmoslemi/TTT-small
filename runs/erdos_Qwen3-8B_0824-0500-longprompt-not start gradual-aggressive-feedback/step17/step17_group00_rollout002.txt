import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    random.seed(seed)
    
    # Define a different initial construction
    n_points = 1000
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            h_initial[i] = 1.0
        elif i % 3 == 1:
            h_initial[i] = 0.5
        else:
            h_initial[i] = 0.0
    
    best_c5 = np.inf
    best_h = h_initial.copy()
    best_n = n_points
    
    start_time = time.time()

    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0

    def perturb(h, magnitude=0.01):

        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization loop with differential evolution
    for _ in range(10):

        result = optimize.minimize(
