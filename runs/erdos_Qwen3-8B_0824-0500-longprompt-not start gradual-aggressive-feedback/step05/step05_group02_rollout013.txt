    n_points = 100
    dx = 2.0 / n_points
    
    def objective(h):
        one_minus_h = 1 - h
        overlap = scipy.signal.correlate(h, one_minus_h, mode='full') * dx

        return objective(h)

                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        return h

                h[i] = h2[i]
        return h
