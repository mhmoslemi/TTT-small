$$
C_5 = \max_k \int h(x)(1 - h(x+k)) dx
$$

under the constraints:
- $ \int_0^2 h(x) dx = 1 $
- $ h(x) \in [0, 1] $

import random

    def objective(h_vec):
        h = np.array(h_vec)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5

    def constraint(h_vec):
        h = np.array(h_vec)
        total = np.sum(h) * (2.0 / n_points)

        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed

    for _ in range(300):
