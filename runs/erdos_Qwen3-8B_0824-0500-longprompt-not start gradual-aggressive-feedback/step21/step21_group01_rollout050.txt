import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define initial construction
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is preloaded
    best_c5 = np.inf
    best_h = None
    best_n = 0
    start_time = time.time()
    
    # Try multiple grid sizes
    for n_points in [70, 100, 150]:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with initial_h_values or random if not available
        if n_points == 70:
            h_initial = initial_h_values.copy()
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        # Define the objective function
        def objective(h):
            h = np.array(h)
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            c5 = np.max(overlap)
            return c5
        
        # Define the constraint function
        def constraint(h):
            h = np.array(h)
            total = np.sum(h) * (2.0 / n_points)
            return total - 1.0
        
        # Function to perturb the solution
        def perturb(h, magnitude=0.01):
            perturbation = np.random.normal(0, magnitude, len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            return h_perturbed
        
        # Refine the initial solution with local optimization
        result = optimize.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        
        h_lo = result.x
        c5_lo = result.fun
        
        # Update the best solution if improvement is found
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
