import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Function to apply adaptive perturbation with learning rate
    def adaptive_perturb(h, magnitude=0.002, decay=0.999):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Function to normalize the solution to maintain the integral constraint
    def normalize_solution(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        scale = 1.0 / total if total != 0 else 1.0
        return np.clip(h * scale, 0, 1)
    
    # Main optimization loop with parallel exploration and learning rate adaptation
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Generate multiple perturbations
        perturbed_hs = [adaptive_perturb(best_h, magnitude=0.002, decay=0.999) for _ in range(5)]
        
        # Refine each perturbed solution with local optimization
        for perturbed_h in perturbed_hs:
            # Normalize the perturbed solution
            perturbed_h = normalize_solution(perturbed_h)
            
            result = optimize.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
            )
            
            h_lo = result.x
            c5_lo = result.fun
            
            # Update the best solution if improvement is found
            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
                best_n = n_points
    
    return (best_h, best_c5, best_n)
