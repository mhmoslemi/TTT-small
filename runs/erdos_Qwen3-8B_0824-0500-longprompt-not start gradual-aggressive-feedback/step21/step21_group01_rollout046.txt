import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define the initial construction
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is a pre-loaded array
    
    best_c5 = np.inf
    best_h = None
    best_n = 0
    start_time = time.time()
    
    # Try multiple grid sizes
    for n_points in [70, 100, 150]:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with the initial construction or random
        if n_points == 70:
            h_initial = initial_h_values.copy()
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        # Function to evaluate the overlap integral
        def objective(h_vec):
            h = np.array(h_vec)
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(overlap)
        
        # Function to maintain the integral constraint
        def constraint(h_vec):
            h = np.array(h_vec)
            total = np.sum(h) * (2.0 / n_points)
            return total - 1.0
        
        # Function to perturb the solution
        def perturb(h_vec, magnitude):
            perturbation = np.random.normal(0, magnitude, len(h_vec))
            h_perturbed = np.clip(h_vec + perturbation, 0, 1)
            return h_perturbed
        
        # Main optimization loop
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
            
            remaining_time = budget_s - (time.time() - start_time)
            perturbation_magnitude = 0.01 * (1 - (remaining_time / budget_s))
            perturbed_h = perturb(h_initial, magnitude=perturbation_magnitude)
            
            # Refine the perturbed solution with local optimization
            result = optimize.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
            )
            
            h_lo = result.x
            c5_lo = result.fun
            
            # Update the best solution if improvement is found
            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
                best_n = n_points
                h_initial = h_lo  # Update the initial for next iteration
        
    return (best_h, best_c5, best_n)
