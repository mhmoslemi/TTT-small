import time
import random

    def objective(h_vec):
        h = np.array(h_vec)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5

    def constraint(h_vec):

        return total - 1.0

        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
