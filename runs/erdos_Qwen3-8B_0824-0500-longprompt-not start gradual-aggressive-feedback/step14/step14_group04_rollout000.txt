import random

    def normalize(h, n_points):

    def objective(h):

        c5 = np.max(overlap)
        return c5

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def local_optimization(h_initial):
        result = optimize.minimize(

            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
