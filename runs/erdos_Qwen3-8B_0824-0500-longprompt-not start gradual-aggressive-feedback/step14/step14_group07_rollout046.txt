import scipy.optimize as optimize
import time

    initial_h = initial_h_values

    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h_norm = normalize(h, n_points)
        overlap = np.correlate(h_norm, 1 - h_norm, mode="full") * (2.0 / n_points)
        return np.max(overlap)
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h_array = np.array(h)
        total = np.sum(h_array) * (2.0 / n_points)
        return total - 1.0

        result = optimize.minimize(

            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},

        )
        return result.x, evaluate_overlap(result.x)
