import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    # Use the initial H values provided
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.array(h)
        h = normalize(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5

        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)
    
    h_lo, c5_lo = local_optimization(best_h)

        best_h = h_lo.copy()
