$$
C_5 = \max_k \int h(x)(1 - h(x+k)) dx
$$

With the constraints:
1. $ h(x) \in [0, 1] $ for all $ x $
2. $ \int_0^2 h(x) dx = 1 $

1. Starting from an initial guess (using `initial_h_values`).
2. Normalizing the function to meet the integral constraint.
3. Using local optimization techniques like SLSQP to refine the solution.
4. Applying small random perturbations to escape local minima.
5. Iteratively improving the solution within the given time budget.

import time
import random

    n_points = len(initial_h)
    
    best_c5 = np.inf

    best_n = n_points

    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Local optimization with SLSQP
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)

        return h_perturbed
    
    # Main optimization loop
    for _ in range(150):

            best_c5 = c5_lo
            best_h = h_lo.copy()
