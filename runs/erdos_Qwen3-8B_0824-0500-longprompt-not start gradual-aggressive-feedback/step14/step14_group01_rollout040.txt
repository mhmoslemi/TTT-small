import scipy.optimize as optimize
import time
import random

        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.zeros(2 * n_points - 1)
        for k in range(n_points - 1):
            overlap[k] = np.dot(h[:n_points - k], (1 - h[k:]))
        c5 = np.max(overlap)
        return c5

        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0

            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},

        )
