import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Use the initial H values provided
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)
    
    def simulated_annealing(h_initial, temp=1.0, cooling_rate=0.995, max_iter=1000):
        current_h = h_initial.copy()
        current_c5 = objective(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s:
                break
            perturbation = np.random.rand(n_points) * 0.03
            candidate_h = np.clip(current_h + perturbation, 0, 1)
            candidate_h = normalize(candidate_h, n_points)
            candidate_c5 = objective(candidate_h)
            
            delta = candidate_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                current_h = candidate_h
                current_c5 = candidate_c5
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
            temp *= cooling_rate
        return best_h, best_c5
    
    h_lo, c5_lo = local_optimization(best_h)
    if c5_lo < best_c5:
        best_c5 = c5_lo
        best_h = h_lo.copy()
        best_n = n_points
    
    for _ in range(20):
        if time.time() - start_time > budget_s:
            break
        deltas = np.linspace(-0.01, 0.01, 11)
        for delta in deltas:

            h_lo, c5_lo = local_optimization(h_perturbed)
            if c5_lo < best_c5:

                best_h = h_lo.copy()
                best_n = n_points

            best_h = h_lo.copy()

    # Simulated Annealing
    h_lo, c5_lo = simulated_annealing(best_h)
    if c5_lo < best_c5:
        best_c5 = c5_lo
        best_h = h_lo.copy()
        best_n = n_points
    
    return (best_h, best_c5, best_n)
