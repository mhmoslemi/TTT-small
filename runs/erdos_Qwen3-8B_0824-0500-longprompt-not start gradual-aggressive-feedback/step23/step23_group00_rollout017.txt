import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    initial_h = initial_h_values

    mutation_magnitude = 0.01

    max_generations = 200
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    start_time = time.time()

    def objective(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Function to apply mutation
    def mutate(h, mutation_magnitude):
        h_mut = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h_mut[i] += np.random.uniform(-mutation_magnitude, mutation_magnitude)
                h_mut[i] = np.clip(h_mut[i], 0.0, 1.0)
        return h_mut
    
    # Function to perform crossover
    def crossover(parent1, parent2):
        child = parent1.copy()
        for i in range(len(parent1)):
            if np.random.rand() < crossover_rate:
                child[i] = parent2[i]
        return child
    
    # Adaptive mutation schedule
    def adaptive_mutation_magnitude(iteration, max_iterations, initial_magnitude=0.01):
        return initial_magnitude * (1 - iteration / max_iterations)

    for generation in range(max_generations):
        if time.time() - start_time > budget_s:

        # Initialize population
        population = [initial_h.copy() for _ in range(population_size)]
        for i in range(population_size):
            population[i] = mutate(population[i], mutation_magnitude)
        
        # Evaluate population
        population_fitness = []
        for h in population:
            result = optimize.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
            )
            population_fitness.append(result.fun)
        
        # Select best individuals
        indices = np.argsort(population_fitness)

            )

                best_c5 = result.fun
                best_h = result.x.copy()
                best_n = n_points
