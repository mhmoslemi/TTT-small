import random
import time

import scipy.optimize as optimize

def run(seed=42, budget_s=60, **kwargs):

    best_n = n_points

        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(overlap)
    
    def normalize(h):
        total = np.sum(h) * (2.0 / n_points)
        factor = 1.0 / total if total != 0 else 1.0
        return np.clip(h * factor, 0, 1)
    
    def mutate(h, mutation_rate=0.05):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
        return h
    
    def crossover(parent1, parent2):
        h = parent1.copy()
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = parent2[i]
        return h
    
    def local_refinement(h):
        n = len(h)
        # Convert h to decision variable
        x = np.array(h, dtype=np.float64)
        x = x.copy()
        
        # Define the objective function
        def objective(x):
            return np.max(np.correlate(x, 1 - x, mode="full")) * (2.0 / n)
        
        # Define the constraint: integral must be equal to 1
        def constraint(x):
            return np.sum(x) * (2.0 / n) - 1.0
        
        # Set up optimization problem
        res = optimize.minimize(objective, x, method='SLSQP', constraints={'type': 'eq', 'fun': constraint})
        if res.success:
            return res.x
        else:
            return x
    
    for generation in range(100):
        if time.time() - start_time > budget_s:

        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        
        # Find the best individual
        best_index = np.argmin(fitness)
        best_fitness = fitness[best_index]
        
        # Update best solution
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = population[best_index].copy()
            best_n = n_points
        
        # Apply elitism
        if best_fitness < np.min(fitness):
            population[best_index] = best_h.copy()

            parent2 = random.choice(population)
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutate
            child = mutate(child)
            
            # Normalize
            child = normalize(child)
            
            # Local refinement
            child = local_refinement(child)
            
            new_population.append(child)
        
        # Replace old population
        population = new_population
    
    return (best_h, best_c5, best_n)
