    def objective(h):

        dx = 2.0 / n_points
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)
        return c5

    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / len(h))
        return total - 1.0

    def perturb(h, magnitude=0.01):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Function to improve the solution with local search
    def improve_solution(h, n_points, budget_s):
        h = np.array(h)
        best_h = h.copy()
        best_c5 = objective(h)
        
        for _ in range(30):
            if time.time() - start_time > budget_s:
                break
            # Perturb the solution
            perturb_magnitude = 0.01 * np.random.rand()
            h_perturbed = perturb(h, magnitude=perturb_magnitude)
            # Refine with local optimization
            result = optimize.minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
            )
            h_lo = result.x
            c5_lo = result.fun
            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
            h = best_h.copy()
        return best_h, best_c5

    for n_points in [70, 100, 150]:

        h_improved, c5_improved = improve_solution(h_initial, n_points, budget_s)
