import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Use the provided initial construction
    h_initial = initial_h_values.copy()
    
    best_c5 = np.inf
    best_h = None
    best_n = len(h_initial)
    best_time = 0
    
    start_time = time.time()
    
    # Define the evaluation function
    def evaluate(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        c5 = np.max(overlap)
        return c5
    
    # Define the constraint function
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / len(h))
        return total - 1.0
    
    # Define the perturbation function
    def perturb(h, magnitude=0.01):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        # Normalize to maintain integral constraint
        total = np.sum(h_perturbed) * (2.0 / len(h_perturbed))
        if total != 0:
            h_perturbed = h_perturbed * (1.0 / total)
        return h_perturbed
    
    # Main optimization loop
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        
        # Adaptive perturbation magnitude
        remaining_time = budget_s - (time.time() - start_time)
        perturbation_magnitude = 0.01 * (1 - (remaining_time / budget_s))
        
        # Generate multiple perturbations
        candidates = [perturb(h_initial, magnitude=perturbation_magnitude) for _ in range(5)]
        
        # Local optimization with SLSQP
        for perturbed_h in candidates:
            result = optimize.minimize(
                evaluate,
                perturbed_h,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(len(h_initial))],
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
            )
            
            h_lo = result.x
            c5_lo = result.fun
            
            # Update the best solution if improvement is found
            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
                best_n = len(h_initial)
                best_time = time.time() - start_time
    
    return (best_h, best_c5, best_n)
