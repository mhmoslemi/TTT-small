import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    random.seed(seed)

    def objective(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        c5 = np.max(overlap)
        return c5

    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / len(h))
        return total - 1.0

    def perturb(h, magnitude=0.01):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed

            solutions.append(perturbed)
        return solutions
