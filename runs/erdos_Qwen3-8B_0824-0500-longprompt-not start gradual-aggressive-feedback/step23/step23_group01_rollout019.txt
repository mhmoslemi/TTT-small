import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Use the provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Function to apply random perturbations
    def random_perturb(h, magnitude=0.005):
        # Perturb in a random direction
        direction = np.random.rand(len(h)) * 2 - 1
        direction /= np.linalg.norm(direction)
        perturbation = direction * magnitude
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Function to apply gradient-based refinement
    def gradient_refinement(h, step_size=0.01, max_iter=100):
        for _ in range(max_iter):
            h = np.array(h)
            # Compute objective and gradient
            obj = objective(h)
            grad = np.zeros_like(h)
            # Finite difference approximation of the gradient
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-6
                grad[i] = (objective(h_plus) - obj) / 1e-6
            # Update step
            h -= step_size * grad
            # Clip to [0, 1]
            h = np.clip(h, 0, 1)
        return h
    
    # Main optimization loop
    for iteration in range(200):
        if time.time() - start_time > budget_s:
            break
        
        # Random perturbation
        perturbed_h = random_perturb(best_h, magnitude=0.005)
        
        # Gradient refinement
        refined_h = gradient_refinement(perturbed_h, step_size=0.01, max_iter=50)
        
        # Refine the refined solution with local optimization
        result = optimize.minimize(
            objective,
            refined_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        
        h_lo = result.x
        c5_lo = result.fun
        
        # Update the best solution if improvement is found
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
