import random
import time
import scipy.optimize as opt

def run(seed=42, budget_s=60, **kwargs):

    random.seed(seed)

    initial_h = initial_h_values
    n_points = len(initial_h)
    
    # Initialize population with the initial solution and some perturbations
    population_size = 50
    population = [initial_h.copy() for _ in range(population_size)]
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    def evaluate(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(overlap)
    
    def normalize(h):
        total = np.sum(h) * (2.0 / n_points)
        if total == 0:
            return np.zeros_like(h)
        factor = 1.0 / total
        return np.clip(h * factor, 0, 1)
    
    def objective_function(x, h, k_values):
        result = 0
        for k in k_values:
            result = max(result, np.dot(x, 1 - h[k:]) * (2.0 / n_points))
        return result
    
    def mutation(h, mutation_rate=0.05):
        h = h.copy()
        for i in range(len(h)):
            if random.random() < mutation_rate:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
        return h
    
    def crossover(parent1, parent2):
        h = parent1.copy()
        for i in range(len(h)):
            if random.random() < 0.5:
                h[i] = parent2[i]
        return h
    
    # Define k values for the overlap calculation
    k_values = np.arange(1, n_points)
    
    for generation in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [evaluate(ind) for ind in population]
        best_index = np.argmin(fitness)
        best_fitness = fitness[best_index]
        
        # Update best solution
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = population[best_index].copy()
            best_n = n_points
        
        # Apply elitism
        if best_fitness < np.min(fitness):
            population[best_index] = best_h.copy()
        
        # Generate new population
        new_population = []
        for _ in range(population_size // 2):
            # Select parents
            parent1 = random.choice(population)
            parent2 = random.choice(population)
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutate
            child = mutation(child)
            
            # Normalize
            child = normalize(child)
            
            new_population.append(child)
        
        # Replace old population
        population = new_population
    
    # Local optimization on the best solution

            return objective_function(x, h, k_values)
        
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        return result.x
    
    best_h = local_optimize(best_h)
    best_h = normalize(best_h)
    
    return (best_h, best_c5, best_n)
