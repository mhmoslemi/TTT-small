import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate
from functools import partial
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    # Initial sampling from provided initial_h_values
    n_points = 100
    initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values is stored in a file
    h_values = initial_h_values.copy()
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    def objective(x):
        h = x.astype(float)
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint})
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Genetic algorithm setup
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    elite_size = 10
    
    # Initial population
    population = np.random.rand(population_size, n_points)
    population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)
    population = np.clip(population, 0, 1)
    
    def fitness(individual):
        return objective(individual)
    
    best_h = h_values.copy()
    best_c5 = objective(h_values)
    
    for generation in range(generations):
        # Evaluate fitness
        fitness_scores = np.array([fitness(ind) for ind in population])
        
        # Select elite
        elite_indices = np.argsort(fitness_scores)[:elite_size]
        elite = population[elite_indices]
        
        # Select parents
        selected_indices = np.random.choice(population_size, size=population_size, p=fitness_scores / fitness_scores.sum())
        parents = population[selected_indices]
        
        # Crossover
        offspring = []
        for i in range(population_size // 2):
            parent1 = parents[2 * i]
            parent2 = parents[2 * i + 1]
            child1 = parent1 * 0.5 + parent2 * 0.5
            child2 = parent2 * 0.5 + parent1 * 0.5
            offspring.append(child1)
            offspring.append(child2)
        
        # Mutation
        for i in range(len(offspring)):
            if np.random.rand() < mutation_rate:
                offspring[i] += np.random.normal(0, 0.05, n_points)
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = offspring[i] * (n_points / 2) / np.sum(offspring[i])
        
        # Combine elite and offspring
        population = np.vstack((elite, offspring))
        population = np.random.permutation(population)
        
        # Update best solution
        current_best = population[np.argmin(fitness_scores)]
        current_c5 = fitness(current_best)
        if current_c5 < best_c5:
            best_h = current_best
            best_c5 = current_c5
        
        # Check time budget
        elapsed_time = generation / generations * budget_s
        if elapsed_time >= budget_s:
            break
    
    # Refine with gradient-based optimization
    remaining_time = budget_s - (generation / generations * budget_s)
    if remaining_time > 0:
        result = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-7)
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, len(best_h)
