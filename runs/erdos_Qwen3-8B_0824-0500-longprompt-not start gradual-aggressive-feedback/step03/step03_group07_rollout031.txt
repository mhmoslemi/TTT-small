        overlap = correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    # Start with initial guess
    h_guess = h_initial.copy()
    h_guess = normalize_h(h_guess)
    
    # Optimization with SLSQP
    start_time = time.time()
    result = opt.minimize(
        objective,
        h_guess,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)},
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8},
        tol=1e-8
    )
    
    h_opt = normalize_h(result.x)
