import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 500  # Use 500 points for balance between accuracy and speed
    dx = 2.0 / n_points
    initial_h_values = np.random.rand(n_points) * 2 - 1  # Random initial guess
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = initial_h_values * (n_points / 2) / np.sum(initial_h_values)
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        overlap = correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def normalize_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    # Setup genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.rand)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    toolbox.register("evaluate", lambda ind: compute_c5(ind))
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initialize population
    pop = toolbox.population(n=50)
    h_guess = initial_h_values.copy()
    h_guess = normalize_h(h_guess)
    
    # Start with initial guess
    h_initial = creator.Individual(h_guess)
    h_initial.fitness.values = (compute_c5(h_initial),)
    pop = [h_initial] + pop
    
    # Genetic algorithm optimization
    start_time = time.time()
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
        fits = [toolbox.evaluate(ind) for ind in offspring]
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = toolbox.select(pop + offspring, len(pop))
    
    # Refine with SLSQP
    h_opt = np.array([ind for ind in pop][0])
    h_opt = normalize_h(h_opt)
    
    def objective_slsqp(h):
        return compute_c5(h)
    
    def constraint_sum_slsqp(h):
        return np.sum(h) - n_points / 2
    
    result = opt.minimize(
        objective_slsqp,
        h_opt,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint_sum_slsqp, 'jac': lambda h: np.ones_like(h)},
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8},
        tol=1e-8
    )
    
    h_opt = normalize_h(result.x)
    c5_bound = compute_c5(h_opt)
    
    # Ensure final constraint is satisfied
    h_opt = normalize_h(h_opt)
    c5_bound = compute_c5(h_opt)
    
    return (h_opt, c5_bound, n_points)
