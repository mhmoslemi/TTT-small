import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    initial_h_values = np.load("initial_h_values.npy")  # Assume initial_h_values is pre-loaded
    n_points_candidates = [100, 80, 70, 60, 50, 40, 30, 20]  # Try varying n_points values

    best_c5 = np.inf
    best_h = np.zeros(1)
    best_n = 1
    start_time = time.time()

    for n_points in n_points_candidates:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial values for this n_points
        if n_points == 70:
            h_values = initial_h_values.copy()
        else:
            h_values = np.random.rand(n_points)
            h_values = h_values * (n_points / 2) / np.sum(h_values)
        h_values = np.clip(h_values, 0, 1)
        
        # Normalize function
        def normalize(h):
            total = np.sum(h)
            if total == 0:
                return h
            return h * (n_points / 2) / total
        
        # Objective function
        def objective(h):
            h = np.array(h)
            h = normalize(h)
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            c5 = np.max(overlap)
            return c5
        
        # Constraint function
        def constraint(h):
            h = np.array(h)
            total = np.sum(h) * (2.0 / n_points)
            return total - 1.0
        
        # Bounds
        bounds = Bounds(0, 1)
        
        # Simulated Annealing
        def simulated_annealing(h_initial, T=1.0, T_min=1e-5, cooling_rate=0.995, max_iter=500):
            h = h_initial.copy()
            current_cost = objective(h)
            best_h = h.copy()
            best_cost = current_cost
            for i in range(max_iter):
                T = T * cooling_rate
                if T < T_min:
                    break
                perturbation = np.random.normal(0, 0.1, n_points)
                h_perturbed = np.clip(h + perturbation, 0, 1)
                cost_perturbed = objective(h_perturbed)
                if np.random.rand() < np.exp((current_cost - cost_perturbed) / T):
                    h = h_perturbed
                    current_cost = cost_perturbed
                    if current_cost < best_cost:
                        best_h = h.copy()
                        best_cost = current_cost
            return best_h, best_cost
        
        # Gradient-based optimization
        def gradient_optimization(h_initial):

                h_initial,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
            )

            best_h = h_sa.copy()
            best_n = n_points
        
        # Run gradient-based optimization
        h_gr, c5_gr = gradient_optimization(h_sa)
        if c5_gr < best_c5:
            best_c5 = c5_gr
            best_h = h_gr.copy()
            best_n = n_points

    return (best_h, best_c5, best_n)
