import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:-k]
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def initial_h_values():
    # Initial construction with 80 samples
    n = 80
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 1.0
        elif i < 40:
            h[i] = 0.5
        elif i < 60:
            h[i] = 0.0
        else:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 80  # Use 80 samples for a balance between speed and quality
    
    # Initialize the best solution with the initial guess
    initial_guess = initial_h_values()
    best_h = initial_guess
    best_c5 = evaluate_erdos_solution(best_h)
    
    # PSO parameters
    n_particles = 30
    inertia_weight = 0.8
    cognitive_coeff = 1.5
    social_coeff = 1.5
    max_iterations = 100
    max_time = budget_s - 1  # Allow some margin for computation
    
    # PSO loop
    particles = np.array([initial_guess.copy() for _ in range(n_particles)])
    velocities = np.zeros((n_particles, n_points))
    
    for _ in range(max_iterations):
        if time.time() - start_time > max_time:
            break
        
        # Evaluate fitness for all particles
        fitness = np.array([evaluate_erdos_solution(p) for p in particles])
        
        # Find the best particle
        best_idx = np.argmin(fitness)
        best_fitness = fitness[best_idx]
        
        # Update velocities and positions
        for i in range(n_particles):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            velocities[i] = inertia_weight * velocities[i] + \
                            cognitive_coeff * r1 * (particles[i] - np.random.rand(n_points)) + \
                            social_coeff * r2 * (particles[best_idx] - particles[i])
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure positions are within [0, 1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Enforce constraint: sum(particles[i]) = n_points / 2
            total = np.sum(particles[i])
            if total > 0:
                particles[i] = particles[i] * (n_points / 2) / total
        
        # Update best solution
        current_fitness = evaluate_erdos_solution(particles[best_idx])
        if current_fitness < best_c5:
            best_c5 = current_fitness
            best_h = particles[best_idx].tolist()
    
    return best_h, best_c5, n_points
