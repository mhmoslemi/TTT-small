import scipy.optimize as optimize
import time
import random

    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:

        return h * (n_points / 2) / total

    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint(h):

    def adaptive_perturb(h, magnitude=0.005, decay=0.999):
