To improve the upper bound for the Erdős minimum overlap problem constant $ C_5 $, we aim to find a step function $ h $ that minimizes the maximum overlap integral:

$$
C_5 = \max_k \int h(x)(1 - h(x+k)) dx
$$

subject to the constraint:

$$
\int_0^2 h(x) dx = 1
$$

3. **Local Optimization**: Use a local optimization method such as SLSQP to refine perturbed solutions while maintaining the integral constraint.

4. **Early Termination**: Since we have a time budget, we'll limit the number of iterations and refine solutions efficiently.

5. **Adaptive Perturbation**: This helps in exploring the neighborhood of the current solution more effectively while ensuring convergence.

import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start from the initial solution
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Function to apply adaptive perturbation
    def adaptive_perturb(h, magnitude=0.005, decay=0.999):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization loop
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Perturb the current best solution with adaptive magnitude
        perturbed_h = adaptive_perturb(best_h, magnitude=0.005, decay=0.999)
        # Refine the perturbed solution with local optimization
        result = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-10, 'disp': False}
        )
        h_lo = result.x
        c5_lo = result.fun
        # Update the best solution if improvement is found
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
