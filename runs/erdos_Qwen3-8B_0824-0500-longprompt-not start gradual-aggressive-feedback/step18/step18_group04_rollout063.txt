import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    random.seed(seed)

    initial_h_values = [0.0] * 70  # Placeholder for the actual initial_h_values

    def objective(h):
        h = np.array(h)

        c5 = np.max(overlap)
        return c5

    def constraint(h):

        total = np.sum(h) * (2.0 / n_points)

        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
