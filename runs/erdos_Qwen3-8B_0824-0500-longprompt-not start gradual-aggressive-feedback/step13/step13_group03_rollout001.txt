import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with the initial guess
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Local optimization with SLSQP
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)
    
    # Global search with random restarts
    def global_search(iterations=10):
        nonlocal best_c5, best_h
        for _ in range(iterations):
            if time.time() - start_time > budget_s:
                break
            # Random restart
            h_restart = np.random.uniform(0, 1, n_points)
            h_normalized = normalize(h_restart, n_points)
            # Local optimization
            h_lo, c5_lo = local_optimization(h_normalized)
            # Update best solution if improvement is found
            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
                best_n = n_points
    
    # Perform global search
    global_search(iterations=20)
    
    return (best_h, best_c5, best_n)
