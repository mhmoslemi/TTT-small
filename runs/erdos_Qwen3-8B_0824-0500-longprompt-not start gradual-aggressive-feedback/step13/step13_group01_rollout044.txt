import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    initial_h_values = np.load('initial_h_values.npy')
    n_points = len(initial_h_values)
    
    best_c5 = np.inf
    best_h = initial_h_values.copy()
    best_n = n_points
    
    start_time = time.time()
    
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='Nelder-Mead',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)
    
    h_lo, c5_lo = local_optimization(best_h)
    if c5_lo < best_c5:
        best_c5 = c5_lo
        best_h = h_lo.copy()
        best_n = n_points
    
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        perturbation = np.random.uniform(-0.005, 0.005, n_points)
        h_perturbed = np.clip(best_h + perturbation, 0, 1)
        h_lo, c5_lo = local_optimization(h_perturbed)
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h_initial = np.random.rand(n_points)
        h_initial = normalize(h_initial, n_points)
        h_lo, c5_lo = local_optimization(h_initial)
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
