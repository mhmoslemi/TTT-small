import scipy.optimize as optimize
import time
import random
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=60, **kwargs):

    random.seed(seed)

    initial_h = initial_h_values
    n_points = len(initial_h)

    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Function to evaluate the overlap integral with penalty for constraint violation
    def objective_with_penalty(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        # Penalty for constraint violation
        constraint_violation = abs(np.sum(h) * (2.0 / n_points) - 1.0)
        return c5 + 100 * constraint_violation

    bounds = [(0.0, 1.0) for _ in range(n_points)]

        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0

        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
        )
        return result.x, objective(result.x)

    creator.create("Individual", list, fitness=creator.FitnessMin)

    toolbox.register("attr_float", np.random.rand)

            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
