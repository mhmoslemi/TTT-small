import numpy as np
import scipy.optimize as optimize
import time
import random
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    toolbox.register("evaluate", objective)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0.0, sigma=0.05, indpb=1.0/n_points)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Local optimization with SLSQP
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) * (2.0 / n_points) - 1.0},
            options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
        )
        return result.x, objective(result.x)
    
    # Run genetic algorithm
    population = toolbox.population(n=50)
    fitnesses = list(map(toolbox.evaluate, population))
    for ind, fit in zip(population, fitnesses):
        ind.fitness.values = fit
    
    for gen in range(100):
        if time.time() - start_time > budget_s:
            break
        offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.1)
        fits = map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        population = toolbox.select(population + offspring, len(population))
    
    # Refine the best solution from genetic algorithm
    best_genetic = min(population, key=lambda ind: ind.fitness.values[0])
    h_genetic = np.array(best_genetic)
    h_genetic_normalized = normalize(h_genetic, n_points)
    c5_genetic = objective(h_genetic_normalized)
    
    if c5_genetic < best_c5:
        best_c5 = c5_genetic
        best_h = h_genetic_normalized.copy()
        best_n = n_points
    
    # Local optimization on the best solution
    h_lo, c5_lo = local_optimization(best_h)
    if c5_lo < best_c5:
        best_c5 = c5_lo
        best_h = h_lo.copy()
        best_n = n_points
    
    return (best_h, best_c5, best_n)
