import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):

    n_points = 100  # Use 100 points for faster optimization

    def normalize(h, n_points):

    def evaluate_overlap(h):

    def simulated_annealing(initial_guess):

            bounds=bounds,
