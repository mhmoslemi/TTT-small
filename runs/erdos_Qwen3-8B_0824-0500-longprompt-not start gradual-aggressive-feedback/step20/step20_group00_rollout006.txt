import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    random.seed(seed)

    initial_h_values = np.random.rand(70)

    best_n = 70

        def objective(h):
            h = np.array(h)
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            c5 = np.max(overlap)
            return c5
        
        # Function to maintain the integral constraint
        def constraint(h):
            h = np.array(h)
            total = np.sum(h) * (2.0 / n_points)
            return total - 1.0
