import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess with n = 100 samples
    n_points = 100
    initial_h = np.zeros(n_points)
    initial_h[:50] = 0.5  # First half as 0.5
    initial_h[50:] = 0.0  # Second half as 0.0
    h = initial_h.copy()
    
    best_c5 = np.inf
    best_h = h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Function to apply adaptive perturbation
    def adaptive_perturb(h, magnitude=0.002, decay=0.999):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization loop
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the current best solution with adaptive magnitude
        perturbed_h = adaptive_perturb(best_h, magnitude=0.0015, decay=0.9995)
        
        # Refine the perturbed solution with local optimization
        result = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )

        c5_lo = result.fun

            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
