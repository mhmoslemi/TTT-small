import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial construction provided
    initial_h_values = np.load('initial_h_values.npy')
    best_c5 = np.inf
    best_h = None
    best_n = 0
    start_time = time.time()
    
    # Try different sample sizes and perturbations
    for n_points in [70, 100]:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with initial construction or random samples
        if n_points == 70:
            h_initial = initial_h_values.copy()
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        # Function to evaluate the overlap integral
        def objective(h):
            h = np.array(h)
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(overlap)
        
        # Function to maintain the integral constraint
        def constraint(h):
            h = np.array(h)
            total = np.sum(h) * (2.0 / n_points)
            return total - 1.0
        
        # Function to perturb the solution
        def perturb(h, magnitude=0.01):
            perturbation = np.random.normal(0, magnitude, len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            return h_perturbed
        
        # Main optimization loop with adaptive perturbations
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break

            remaining_time = budget_s - (time.time() - start_time)

            result = optimize.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
            )
            h_lo = result.x

                best_c5 = c5_lo

                best_n = n_points
    
    return (best_h, best_c5, best_n)
