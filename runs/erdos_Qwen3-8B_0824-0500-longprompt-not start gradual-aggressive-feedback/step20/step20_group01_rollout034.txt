import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess with 70 points
    initial_h_values = np.array([0.0] * 70)
    for i in range(70):
        initial_h_values[i] = 1.0 if i < 35 else 0.0  # Simple symmetric pattern
    initial_h = initial_h_values.copy()

    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = 70
    start_time = time.time()

    # Optimization parameters
    n_points_options = [70, 100, 150]
    perturbation_magnitude = 0.01
    num_iterations_per_n = 100

    for n_points in n_points_options:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with initial_h for n=70, else random
        if n_points == 70:
            h_initial = initial_h.copy()
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        
        # Objective function
        def objective(h):
            h = np.array(h)
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(overlap)
        
        # Constraint: sum(h) * dx = 1 => sum(h) == n_points / 2
        def constraint(h):
            h = np.array(h)
            return np.sum(h) - (n_points / 2)
        
        # Perturbation function
        def perturb(h, magnitude=0.01):
            perturbation = np.random.normal(0, magnitude, len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            return h_perturbed
        
        # Local optimization
        for _ in range(num_iterations_per_n):
            if time.time() - start_time > budget_s:
                break
            h_initial = perturb(h_initial, magnitude=perturbation_magnitude)
            result = optimize.minimize(
                objective,
                h_initial,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
            )
            h_lo = result.x
            c5_lo = result.fun
            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
                best_n = n_points

    return (best_h, best_c5, best_n)
