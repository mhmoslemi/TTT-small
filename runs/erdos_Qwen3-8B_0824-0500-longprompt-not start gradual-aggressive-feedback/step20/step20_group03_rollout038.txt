import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    try:
        initial_h = initial_h_values
    except NameError:
        initial_h = np.ones(100) / 100 * (100 / 2)
    
    n_points = len(initial_h)
    h = initial_h.copy()
    h = h / np.sum(h) * (n_points / 2)
    
    best_c5 = np.inf
    best_h = h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    def objective(h_vec):
        h = np.array(h_vec)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    def constraint(h_vec):
        h = np.array(h_vec)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    def perturb(h, magnitude=0.01):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    for _ in range(300):
        if time.time() - start_time > budget_s:
            break
        perturbation_magnitude = 0.01 * (1 - (time.time() - start_time) / budget_s)
        perturbed_h = perturb(best_h, magnitude=perturbation_magnitude)
        result = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        h_lo = result.x
        c5_lo = result.fun
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
