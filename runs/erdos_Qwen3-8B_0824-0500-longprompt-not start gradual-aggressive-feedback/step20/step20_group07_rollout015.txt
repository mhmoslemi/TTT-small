import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

        np.random.rand(len(initial_h_values)),  # Random

        np.ones(len(initial_h_values))  # All ones (with constraint)
    ]
    
    best_c5 = np.inf
    best_h = None
    best_n = 0
    start_time = time.time()
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / len(h))
        return total - 1.0

        perturbation = np.random.normal(0, mag, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Refinement with local optimization
    def refine_solution(h, n_points):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, result.fun

    for initial_h in initial_guesses:

        if n_points < 10:

                best_n = n_points
            current_h = refined_h
