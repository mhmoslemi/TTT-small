        return total - 1.0

    def perturb(h, magnitude=0.01):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization loop with genetic algorithm and local optimization
    population_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.7
    elite_size = 5
    
    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate the population
        fitness = [objective(ind) for ind in population]
        
        # Select the elites
        elite_indices = np.argsort(fitness)[:elite_size]
        elites = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elites.copy()
        
        # Generate offspring
        while len(new_population) < population_size:
