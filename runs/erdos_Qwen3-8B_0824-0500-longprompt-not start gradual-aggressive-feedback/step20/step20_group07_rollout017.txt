import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guesses (multiple starting points for better exploration)
    initial_h_values = [
        [0.0] * 70 + [1.0] * 70,  # Example initial guess
        [0.5] * 70,              # Uniform distribution
        [1.0 if i % 3 == 0 else 0.0 for i in range(70)],  # Periodic pattern
        [1.0 if i % 5 == 0 else 0.0 for i in range(70)],  # Another periodic pattern
        [0.0 if i % 7 == 0 else 1.0 for i in range(70)],  # More sparse pattern
        # Additional starting points can be added as needed
    ]
    
    best_c5 = np.inf
    best_h = None
    best_n = 0
    best_n_points = 70  # Default number of points

    start_time = time.time()

    def objective(h, n_points):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5

    def constraint(h, n_points):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0

    def adaptive_perturb(h, magnitude, decay, iteration, max_iterations):
        perturbation = np.random.normal(0, magnitude * (1 - iteration / max_iterations), len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed

    for initial_h in initial_h_values:
        n_points = len(initial_h)
        if n_points != best_n_points:
            # Only use the same number of points for consistency
            continue
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        best_h_current = initial_h.copy()
        best_c5_current = objective(best_h_current, n_points)

        for iteration in range(200):
            if time.time() - start_time > budget_s:
                break
            current_magnitude = 0.005 * (1 - iteration / 200)
            perturbed_h = adaptive_perturb(best_h_current, current_magnitude, 0.999, iteration, 200)
            result = optimize.minimize(
                lambda x: objective(x, n_points),
                perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': lambda x: constraint(x, n_points)},
                options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
            )
            h_lo = result.x
            c5_lo = result.fun
            if c5_lo < best_c5_current:
                best_c5_current = c5_lo
                best_h_current = h_lo.copy()

        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
            best_n = n_points

    return (best_h, best_c5, best_n)
