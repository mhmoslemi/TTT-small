import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction provided
    initial_h_values = np.array([0.5 + 0.5 * np.cos(np.pi * i / 100) for i in range(100)])
    initial_h_values = initial_h_values * (100 / np.sum(initial_h_values))
    
    # Try different n_points to balance between accuracy and performance
    n_points_list = [100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    start_time = time.time()
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with the initial construction scaled to n_points
        h_initial = np.zeros(n_points)
        for i in range(n_points):
            h_initial[i] = 0.5 + 0.5 * np.cos(np.pi * i / n_points)
        h_initial = h_initial * (n_points / np.sum(h_initial))
        
        h_values = h_initial.copy()
        
        # Optimization parameters
        max_iter = 50
        tol = 1e-5
        
        def objective(h):
            h = np.array(h)
            h = np.clip(h, 0, 1)
            # Ensure the integral constraint is satisfied
            total = np.sum(h) * (2.0 / n_points)
            if abs(total - 1.0) > 1e-6:
                return np.inf
            # Compute the overlap integral
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            c5 = np.max(overlap)
            return c5
        
        def constraint(h):
            h = np.array(h)
            total = np.sum(h) * (2.0 / n_points)
            return total - 1.0
        
        # Set up bounds for optimization
        bounds = Bounds(0, 1)
        
        # Use SLSQP for initial optimization
        result = optimize.minimize(
            objective,
            h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': max_iter, 'ftol': tol, 'disp': False}
        )
        
        h_values = result.x
        c5_bound = objective(h_values)
        
        # Early stopping if budget is exhausted
        if time.time() - start_time > budget_s:
            break
        
        # Refine with L-BFGS-B
        for _ in range(20):
            result = optimize.minimize(
                objective,
                h_values,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 10, 'ftol': tol, 'disp': False}
            )
            h_values = result.x
            c5_bound = objective(h_values)
            if time.time() - start_time > budget_s:
                break
        
        # Randomized initialization to avoid local optima
        for _ in range(5):
            h_rand = np.random.uniform(0, 1, n_points)
            h_rand = h_rand * (1.0 / (np.sum(h_rand) * (2.0 / n_points)))
            result = optimize.minimize(
                objective,
                h_rand,
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 50, 'ftol': tol, 'disp': False}
            )
            h_rand = result.x
            c5_rand = objective(h_rand)
            if c5_rand < c5_bound:
                c5_bound = c5_rand
                h_values = h_rand
        
        # Track the best solution
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_values
            best_n = n_points
    
    return (best_h, best_c5, best_n)
