```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:-k]
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def initial_h_values():
    # Initial construction with 80 samples
    n = 80
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 1.0
        elif i < 40:
            h[i] = 0.5
        elif i < 60:
            h[i] = 0.0
        else:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points_list = [80, 100, 120]  # Try different sample sizes

    for n_points in n_points_list:
        dx = 2.0 / n_points

        # Generate a better initial guess based on the initial construction
        h_initial = np.zeros(n_points)
        for i in range(n_points):
            if i < 20:
                h_initial[i] = 1.0
            elif i < 40:
                h_initial[i] = 0.5
            elif i < 60:
                h_initial[i] = 0.0
            else:
                h_initial[i] = 1.0
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

        # Gradient-based optimization with constraint
        def objective(x):
            h = np.array(x)
            overlap = correlate(h, 1 - h, mode="full") * dx
            return np.max(overlap)

        def constraint(x):
            return np.sum(x) - n_points / 2

        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint}
        )

        current_h = result.x
        current_c5 = evaluate_erdos_solution(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

        # Early stopping if time is up
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:
            break

    return best_h.tolist(), best_c5, len(best_h)
```
