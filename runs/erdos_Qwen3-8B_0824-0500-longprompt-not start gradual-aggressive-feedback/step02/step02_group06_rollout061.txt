import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):

    # Initial construction based on known heuristic
    def initial_balanced_h(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.5 * np.cos(np.pi * (i) / n_points)

        max_iter_slsqp = 50
        max_iter_lbfgs = 20
        max_random_init = 5

            h = np.array(h)

            total = np.sum(h) * (2.0 / n_points)
            return total - 1.0
