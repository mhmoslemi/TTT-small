import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    def initial_balanced_h(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.5 * np.sin(np.pi * (i) / n_points)
        integral = np.sum(h) * (2.0 / n_points)
        h = h * (1.0 / integral)
        return h

    n_points_list = [100, 150, 200]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    start_time = time.time()

    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break

        h_initial = initial_balanced_h(n_points)
        h_values = h_initial.copy()

        def objective(h):
            h = np.array(h)
            h = np.clip(h, 0, 1)
            total = np.sum(h) * (2.0 / n_points)
            if abs(total - 1.0) > 1e-6:
                return np.inf
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            return np.max(overlap)

        def constraint(h):
            h = np.array(h)
            total = np.sum(h) * (2.0 / n_points)
            return total - 1.0

        bounds = Bounds(0, 1)

        result = optimize.minimize(
            objective,
            h_values,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 50, 'ftol': 1e-6, 'disp': False}
        )

        h_values = result.x
        c5_bound = objective(h_values)

        if time.time() - start_time > budget_s:
            break

        for _ in range(20):
            result = optimize.minimize(
                objective,
                h_values,
                method='L-BFGS-B',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 10, 'ftol': 1e-6, 'disp': False}
            )
            h_values = result.x
            c5_bound = objective(h_values)
            if time.time() - start_time > budget_s:
                break

        for _ in range(5):
            h_rand = np.random.uniform(0, 1, n_points)

            result = optimize.minimize(
                objective,
                h_rand,

                h_values = h_rand

            best_c5 = c5_bound
            best_h = h_values
            best_n = n_points

    return (best_h, best_c5, best_n)
