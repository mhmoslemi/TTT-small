import scipy.optimize as optimize
import time

from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def initial_h_values():
    # Initial construction with 80 samples
    n = 80
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 1.0
        elif i < 40:
            h[i] = 0.5
        elif i < 60:
            h[i] = 0.0
        else:
            h[i] = 1.0
    return h.tolist()

        # Define the objective function
        def objective(h_values):

            return np.max(overlap)
