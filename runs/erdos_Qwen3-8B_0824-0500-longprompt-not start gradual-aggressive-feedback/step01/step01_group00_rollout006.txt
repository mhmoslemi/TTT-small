```python
import numpy as np
import scipy.optimize as optimize
import numpy.random as npr
from scipy.optimize import minimize
from scipy.signal import correlate
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=60, **kwargs):
    # Start with a higher resolution to better approximate the optimal solution
    n_points = 200
    # Initialize with a simple alternating pattern to spread out the values
    h_values = np.zeros(n_points)
    h_values[::2] = 0.5
    h_values[1::2] = 0.5
    h_values = np.random.permutation(h_values)
    # Normalize to meet the integral constraint
    h_values = h_values * (n_points / 2) / np.sum(h_values)
    h_values = np.clip(h_values, 0, 1)
    
    dx = 2.0 / n_points
    
    # Objective function to minimize the maximum overlap
    def objective(x):
        h = x.reshape(n_points)
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: sum(h) == n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    cons = ({'type': 'eq', 'fun': constraint})
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    x0 = h_values.copy()
    
    # Optimize using L-BFGS-B for faster convergence
    result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
    best_h = result.x
    best_c5 = result.fun
    
    # Try refining with a smaller step size and tighter tolerance
    result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-8)
    best_h = result.x
    best_c5 = result.fun
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, len(best_h)
```
