import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:-k]
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def initial_h_values():
    # Initial construction with 80 samples
    n = 80
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 1.0
        elif i < 40:
            h[i] = 0.5
        elif i < 60:
            h[i] = 0.0
        else:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 80  # Start with 80 points for faster exploration
    
    # Start from initial construction
    h = np.array(initial_h_values())
    h = np.resize(h, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    # Use a gradient-based optimization method
    def objective(h_values):
        h = np.array(h_values)
        overlapped = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(overlapped)
    
    # Define bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Use L-BFGS-B optimization
    def optimize_h(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            jac=None,
            options={'disp': False, 'maxiter': 50}
        )
        return result.x
    
    # Perform multiple rounds of optimization
    for _ in range(5):
        h = optimize_h(h)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Try a slightly larger number of points for better resolution
    n_points = 100
    dx = 2.0 / n_points
    h = np.array(best_h)
    h = np.resize(h, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    # Refine with L-BFGS-B again
    for _ in range(5):
        h = optimize_h(h)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final evaluation
    current_c5 = evaluate_erdos_solution(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return best_h.tolist(), best_c5, n_points
    
    return best_h.tolist(), best_c5, n_points
