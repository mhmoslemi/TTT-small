import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.optimize import minimize

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Define the objective function for optimization
    def objective_function(h_values):
        # Ensure the integral constraint is met
        h_values = h_values * (len(h_values) / 2) / np.sum(h_values)
        # Evaluate the overlap integral
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * (2.0 / len(h_values))
    
    # Define the optimization problem
    def optimization_step(current_h):
        # Define the objective function with current_h as initial guess
        def objective(x):
            return objective_function(x)
        
        # Define bounds for the variables
        bounds = [(0.0, 1.0) for _ in range(len(current_h))]
        
        # Use L-BFGS-B for optimization
        result = minimize(objective, current_h, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        
        # Ensure the solution meets the integral constraint
        result.x = result.x * (len(result.x) / 2) / np.sum(result.x)
        
        return result.x
    
    # Initial guess: start with the initial_h_values scaled to n_points
    n_points = 100  # Start with 100 samples for faster evaluation
    initial_guess = initial_h_values.copy()
    initial_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_guess)
    
    best_h = initial_guess.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    # Optimization loop
    for iteration in range(100):
        # Evaluate current best
        current_c5 = evaluate_erdos_solution(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        
        # Check time budget
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
        
        # Optimize the current solution
        best_h = optimization_step(best_h)
        
        # Early stopping if no improvement
        if np.isclose(evaluate_erdos_solution(best_h), best_c5, atol=1e-6):
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
