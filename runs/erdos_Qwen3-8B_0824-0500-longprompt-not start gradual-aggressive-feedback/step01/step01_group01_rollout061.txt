def run(seed=42, budget_s=60, **kwargs):

    max_iter_refine = 100
    tol = 1e-5

            return np.inf

        overlap = []

            if k >= n_points:
                break
            overlap_k = np.sum(h[:n_points - k] * (1 - h[k:])) * (2.0 / n_points)
            overlap.append(overlap_k)
        c5 = np.max(overlap)
        return c5

        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0

            self.n_particles = n_particles
            self.n_iterations = n_iterations
            self.w = w
            self.c1 = c1
            self.c2 = c2
            
        def optimize(self, objective, bounds, initial_guess=None):

            particles = np.random.uniform(bounds[0], bounds[1], (self.n_particles, n))
            velocities = np.zeros((self.n_particles, n))

            gbest = pbest[np.argmin(pbest_obj)]
