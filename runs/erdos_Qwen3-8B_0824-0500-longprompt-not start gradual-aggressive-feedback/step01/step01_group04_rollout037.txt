import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:-k]
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def initial_h_values():
    # Initial construction with 80 samples
    n = 80
    h = np.zeros(n)

        if i < 20:
            h[i] = 1.0
        elif i < 40:
            h[i] = 0.5
        elif i < 60:
            h[i] = 0.0
        else:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):

    def compute_c5(h_values):
        h = np.array(h_values)
        shifted = np.zeros_like(h)

            shifted[k:] = h[:-k]

        return np.max(overlap)
    
    # Define the optimization objective function
    def objective_func(h_values):
        return compute_c5(h_values)

            h -= 0.1 * grad

    for _ in range(5):
