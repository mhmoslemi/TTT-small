import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200  # Start with 200 samples
    time_budget = budget_s  # Time budget in seconds
    
    start_time = time.time()
    
    # Initial guess from initial_h_values scaled to n_points
    initial_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    # Function to evaluate the overlap integral
    def objective_function(h):
        return evaluate_erdos_solution(h)
    
    # Function to enforce the integral constraint
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Optimization
    result = optimize.minimize(
        fun=objective_function,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint_func},
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    # Check if we reached the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s:
        # Return the best solution found so far
        best_h = initial_guess
        best_c5 = evaluate_erdos_solution(best_h)
        return best_h, best_c5, n_points
    
    # Check if optimization was successful
    if result.success:
        best_h = result.x
    else:
        # Fallback to initial guess
        best_h = initial_guess
    
    best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, n_points
