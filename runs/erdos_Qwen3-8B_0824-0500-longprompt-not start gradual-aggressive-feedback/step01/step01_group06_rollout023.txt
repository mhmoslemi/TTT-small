def run(seed=42, budget_s=60, **kwargs):

        h_current = np.clip(x, 0, 1)
        one_minus_h = 1 - h_current
        overlap = scipy.signal.correlate(h_current, one_minus_h, mode='full') * dx
        c5 = np.max(overlap)
        return c5

    current_h = best_h.copy()
