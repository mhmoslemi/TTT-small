import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:-k]
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def initial_h_values():
    # Initial construction with 80 samples
    n = 80
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 1.0
        elif i < 40:
            h[i] = 0.5
        elif i < 60:
            h[i] = 0.0
        else:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 80  # Start with 80 points for faster exploration
    
    # Start from initial construction
    h = np.array(initial_h_values())
    h = np.resize(h, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h_values):
        h = np.array(h_values)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(overlap)
    
    # Define the constraint function to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Use SLSQP optimization with bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Update best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Evaluate and refine with local gradient descent
    for _ in range(20):
        h = best_h.copy()
        overlapped = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        max_overlap = np.max(overlapped)
        
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            overlapped_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * (2.0 / n_points)
            grad[i] = (np.max(overlapped_plus) - max_overlap) / 1e-5
        
        # Update h using gradient descent
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping
        if np.abs(np.sum(h) - n_points / 2) < 1e-5:
            break
        
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    # Try a slightly higher resolution
    n_points = 100
    dx = 2.0 / n_points
    best_h = np.resize(best_h, n_points)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Further refine with smaller step size
    for _ in range(20):
        overlapped = np.correlate(best_h, 1 - best_h, mode="full") * dx
        max_overlap = np.max(overlapped)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += 1e-5
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            overlapped_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * dx
            grad[i] = (np.max(overlapped_plus) - max_overlap) / 1e-5
        
        # Update h
        best_h -= 0.01 * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        # Early stopping
        if np.abs(np.sum(best_h) - n_points / 2) < 1e-5:
            break
        
        current_c5 = evaluate_erdos_solution(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return best_h.tolist(), best_c5, n_points
    
    return best_h.tolist(), best_c5, n_points
