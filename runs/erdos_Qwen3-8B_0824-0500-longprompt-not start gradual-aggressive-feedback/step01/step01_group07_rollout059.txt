import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 200  # Start with 200 samples
    max_iterations = 50  # Max number of iterations
    time_budget = budget_s  # Time budget in seconds
    
    start_time = time.time()
    
    # Create initial guess based on initial_h_values
    initial_guess = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    best_h = initial_guess.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Evaluate best_h
    current_c5 = evaluate_erdos_solution(best_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
    
    # Define objective function
    def obj_func(h):
        # Ensure h is valid
        h = np.clip(h, 0.0, 1.0)
        # Rescale to maintain integral constraint
        h = h * (n_points / 2) / np.sum(h)
        # Evaluate the overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / len(h))
        return c5
    
    # Define constraints
    def constraints(h):
        # Ensure integral constraint
        integral = np.sum(h) * (2.0 / n_points)
        return [integral - 1.0]
    
    # Optimization
    for iteration in range(max_iterations):
        elapsed_time = time.time() - start_time
        if elapsed_time >= time_budget:
            break
        
        # Use L-BFGS-B optimizer
        h = best_h.copy()
        result = optimize.minimize(obj_func, h, method='L-BFGS-B', bounds=[(0.0, 1.0) for _ in range(n_points)], constraints=constraints)
        
        # Check if optimization was successful
        if result.success:
            best_h = result.x
            best_c5 = obj_func(best_h)
        
        # Evaluate current best
        current_c5 = obj_func(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
