import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

    n_points = 128

        c5 = np.max(overlap)
        return c5

        return total - 1.0

    bounds = Bounds(0, 1)
    
    # Initial optimization with L-BFGS-B
