import numpy as np
import scipy.optimize as optimize
import time

from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:-k]
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def initial_h_values():
    # Initial construction with 80 samples
    n = 80
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 1.0
        elif i < 40:
            h[i] = 0.5
        elif i < 60:
            h[i] = 0.0
        else:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for better resolution and faster convergence
    
    # Start from initial construction
    h = np.array(initial_h_values())
    h = np.resize(h, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    # Define a function to compute the overlap integral
    def compute_overlap(h_values):
        h = np.array(h_values)
        dx = 2.0 / len(h)
        shifted = np.zeros_like(h)
        for k in range(1, len(h)):
            shifted[k:] = h[:-k]
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Define a function to compute the gradient of the overlap integral
    def compute_gradient(h_values):
        h = np.array(h_values)
        dx = 2.0 / len(h)
        grad = np.zeros(len(h))
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = h_plus * (len(h) / 2) / np.sum(h_plus)
            overlapped_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * dx
            grad[i] = (np.max(overlapped_plus) - compute_overlap(h_values)) / 1e-5
        return grad
    
    # Use a local optimization algorithm to refine the solution
    def optimize_h(h_initial):
        h = np.array(h_initial)
        h = h * (n_points / 2) / np.sum(h)
        for _ in range(100):
            # Compute the overlap for all shifts
            max_overlap = compute_overlap(h)
            
            # Compute the gradient
            grad = compute_gradient(h)
            
            # Update h using gradient descent
            h -= 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Early stopping
            if np.abs(np.sum(h) - n_points / 2) < 1e-5:
                break
        
        return h.tolist()
    
    # Refine the solution
    for _ in range(5):
        h = optimize_h(h)
        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    # Try a smaller number of samples for faster exploration

    h = np.array(best_h)

    # Further refine with smaller step size

        grad = compute_gradient(h)
        
        # Update h
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)

        if np.abs(np.sum(h) - n_points / 2) < 1e-5:

    current_c5 = compute_overlap(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h
    
    elapsed_time = time.time() - start_time

    return best_h, best_c5, n_points
