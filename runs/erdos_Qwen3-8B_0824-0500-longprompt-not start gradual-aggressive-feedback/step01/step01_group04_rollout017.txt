```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:-k]
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def initial_h_values():
    # Initial construction with 80 samples
    n = 80
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 1.0
        elif i < 40:
            h[i] = 0.5
        elif i < 60:
            h[i] = 0.0
        else:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for better resolution and faster convergence
    dx = 2.0 / n_points
    
    # Start from initial construction
    h = np.array(initial_h_values())
    h = np.resize(h, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    # Use gradient-based optimization with L-BFGS-B
    def objective(h_values):
        h = np.array(h_values)
        overlapped = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlapped)
    
    # Define bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial optimization
    result = optimize.minimize(
        fun=objective,
        x0=h,
        bounds=bounds,
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    h_optimized = result.x
    h_optimized = h_optimized * (n_points / 2) / np.sum(h_optimized)
    current_c5 = evaluate_erdos_solution(h_optimized)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_optimized
    
    # Refine with local gradient descent
    for _ in range(50):
        # Compute overlap
        overlapped = np.correlate(h_optimized, 1 - h_optimized, mode="full") * dx
        max_overlap = np.max(overlapped)
        
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_optimized.copy()
            h_plus[i] += 1e-5
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            overlapped_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * dx
            grad[i] = (np.max(overlapped_plus) - max_overlap) / 1e-5
        
        # Update h
        h_optimized -= 0.01 * grad
        h_optimized = np.clip(h_optimized, 0, 1)
        h_optimized = h_optimized * (n_points / 2) / np.sum(h_optimized)
        
        # Early stopping if constraint is satisfied
        if np.abs(np.sum(h_optimized) - n_points / 2) < 1e-5:
            break
    
    current_c5 = evaluate_erdos_solution(h_optimized)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_optimized
    
    # Try a smaller grid for faster exploration and better local minima
    n_points = 80
    dx = 2.0 / n_points
    h = np.array(best_h)
    h = np.resize(h, n_points)
    h = h * (n_points / 2) / np.sum(h)
    
    # Further optimization with gradient descent
    for _ in range(50):
        overlapped = np.correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlapped)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            overlapped_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * dx
            grad[i] = (np.max(overlapped_plus) - max_overlap) / 1e-5
        
        # Update h
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping
        if np.abs(np.sum(h) - n_points / 2) < 1e-5:
            break
    
    current_c5 = evaluate_erdos_solution(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
```
