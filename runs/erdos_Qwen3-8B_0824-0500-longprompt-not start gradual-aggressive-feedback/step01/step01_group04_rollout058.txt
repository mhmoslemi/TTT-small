import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate
from scipy.optimize import minimize

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:-k]
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def initial_h_values():
    # Initial construction with 80 samples
    n = 80
    h = np.zeros(n)
    for i in range(n):
        if i < 20:
            h[i] = 1.0
        elif i < 40:
            h[i] = 0.5
        elif i < 60:
            h[i] = 0.0
        else:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 80
    n_points = 80
    dx = 2.0 / n_points

    # Start from initial construction
    h = np.array(initial_h_values())
    h = np.resize(h, n_points)
    h = h * (n_points / 2) / np.sum(h)
    current_c5 = evaluate_erdos_solution(h)
    best_c5 = current_c5
    best_h = h

    # Define a function to evaluate the objective and constraints
    def objective_function(h_values):
        h = np.array(h_values)
        # Ensure integral constraint
        integral = np.sum(h) * dx
        if abs(integral - 1.0) > 1e-6:
            return float('inf')  # Penalize deviation from integral constraint
        # Compute overlap integral
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Define the constraints for the optimization
    def constraints(h_values):
        h = np.array(h_values)
        return [np.sum(h) * dx - 1.0]  # Integral constraint
    
    # Gradient-based optimization with bounds
    def optimize_h(h_initial):
        # Convert to numpy array and ensure bounds
        h = np.array(h_initial, dtype=float)
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)
        result = minimize(
            fun=objective_function,
            x0=h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0},
            options={'ftol': 1e-8, 'maxiter': 100}
        )
        return result.x.tolist()
    
    # Run multiple rounds of optimization with different starting points
    for _ in range(3):
        h = optimize_h(best_h)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    # Try with a smaller number of points for faster exploration
    n_points = 60
    dx = 2.0 / n_points
    h = np.array(best_h)
    h = np.resize(h, n_points)
    h = h * (n_points / 2) / np.sum(h)
    current_c5 = evaluate_erdos_solution(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h
    
    # Further refine using gradient descent
    for _ in range(20):
        h = np.array(best_h)
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)
        overlapped = np.correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlapped)
        
        # Compute gradient with finite differences
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            overlapped_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * dx
            grad[i] = (np.max(overlapped_plus) - max_overlap) / 1e-5
        
        # Update h using gradient descent
        h -= 0.01 * grad
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
