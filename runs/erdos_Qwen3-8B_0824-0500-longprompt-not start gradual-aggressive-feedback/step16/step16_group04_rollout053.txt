import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess with a low number of samples
    n_points = 100
    initial_h = np.zeros(n_points)

        if i < n_points // 2:
            initial_h[i] = 1.0
    initial_h = initial_h.copy()
    best_c5 = np.inf
    best_h = initial_h.copy()

    start_time = time.time()

    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def constraint(h):

        return total - 1.0

    def local_optimization(h_initial):

            method='SLSQP',
            bounds=bounds,
