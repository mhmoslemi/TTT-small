import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess with structured pattern to minimize overlap
    n_points = 100  # Use a manageable number of points for faster optimization
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        # Create a periodic pattern with alternating regions of 0 and 1
        if (i % 5) < 3:
            initial_h[i] = 1.0
        else:
            initial_h[i] = 0.0
    # Normalize to satisfy the integral constraint
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Local optimization with SLSQP
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)
    
    # Function to create structured perturbations based on local gradients
    def gradient_based_perturb(h, magnitude=0.01):
        # Compute the gradient of the objective function at the current point
        # This is a simplified version - actual gradient computation would require more effort
        perturbation = np.zeros_like(h)
        perturbation[::2] = np.random.uniform(-magnitude, magnitude, len(perturbation[::2]))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization loop with gradient-based perturbations
    for _ in range(150):
        if time.time() - start_time > budget_s:
            break
        # Perturb the current best solution with gradient-based directions
        perturbed_h = gradient_based_perturb(best_h, magnitude=0.005)
        # Refine the perturbed solution with local optimization
        h_lo, c5_lo = local_optimization(perturbed_h)
        # Update the best solution if improvement is found
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
