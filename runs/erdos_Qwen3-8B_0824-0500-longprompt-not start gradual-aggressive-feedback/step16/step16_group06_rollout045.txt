import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess: a simpler structured solution
    n_points = 100
    initial_h = np.zeros(n_points)
    initial_h[0:50] = 1.0
    initial_h = (initial_h * (n_points / 2)) / np.sum(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.array(h)
        h = normalize(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-12, 'disp': False}
        )
        return result.x, objective(result.x)
    
    # Initial optimization
    h_lo, c5_lo = local_optimization(best_h)
    if c5_lo < best_c5:
        best_c5 = c5_lo
        best_h = h_lo.copy()

    for _ in range(30):

        perturbation = np.random.uniform(-0.02, 0.02, size=n_points)
        h_perturbed += perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_lo, c5_lo = local_optimization(h_perturbed)
        if c5_lo < best_c5:
            best_c5 = c5_lo
