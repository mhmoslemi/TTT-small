import scipy.optimize as optimize
import time
import random

initial_h_values = np.random.rand(70)  # Example placeholder, actual initial values would be defined elsewhere

def run(seed=42, budget_s=60, **kwargs):

    random.seed(seed)

    initial_h = initial_h_values.copy()
    n_points = len(initial_h)

    start_time = time.time()

    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:

        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5

    def local_optimization(h_initial):
