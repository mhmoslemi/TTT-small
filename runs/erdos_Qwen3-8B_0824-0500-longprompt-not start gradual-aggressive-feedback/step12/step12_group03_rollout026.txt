import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Local optimization with SLSQP
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)
    
    # Function to create a random perturbation
    def perturb(h, magnitude=0.01):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Function to create a new initial guess based on previous best
    def generate_new_initial_guess(h_best):
        # Create a new guess by splitting and recombining segments
        n = len(h_best)
        split_point = n // 2
        h1 = h_best[:split_point]
        h2 = h_best[split_point:]
        h_new = np.concatenate([h2, h1])
        return h_new
    
    # Main optimization loop
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        # Perturb the current best solution
        perturbed_h = perturb(best_h)
        # Refine the perturbed solution with local optimization
        h_lo, c5_lo = local_optimization(perturbed_h)
        # Update the best solution if improvement is found
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
        # Generate a new initial guess based on the best solution
        new_initial_guess = generate_new_initial_guess(best_h)
        # Refine the new initial guess with local optimization
        h_lo_new, c5_lo_new = local_optimization(new_initial_guess)
        # Update the best solution if improvement is found
        if c5_lo_new < best_c5:
            best_c5 = c5_lo_new
            best_h = h_lo_new.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
