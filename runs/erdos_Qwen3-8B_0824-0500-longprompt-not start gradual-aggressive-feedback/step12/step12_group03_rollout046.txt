import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess
    n_points = 200  # Increased sample size for better resolution
    initial_h = np.zeros(n_points)
    initial_h[::2] = 1.0  # Simple initial guess: alternating 0 and 1
    initial_h = initial_h.copy()
    
    # Define the normalization function to maintain the integral constraint
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def compute_c5(h):
        h = normalize(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(overlap)
    
    # Genetic algorithm parameters
    population_size = 50
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_fraction = 0.2
    
    # Initialize population
    population = []
    for _ in range(population_size):
        h = np.random.rand(n_points)
        h = normalize(h)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = [compute_c5(ind) for ind in population]
        
        # Select elite individuals
        elite_size = int(elite_fraction * population_size)
        elite_indices = np.argsort(fitness)[:elite_size]
        elite_population = [population[i] for i in elite_indices]

        new_population = elite_population.copy()
