import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Function to evaluate the overlap integral with penalty for constraint violation
    def objective_with_penalty(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        # Penalty for constraint violation
        constraint_violation = abs(np.sum(h) * (2.0 / n_points) - 1.0)
        return c5 + 1000 * constraint_violation
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Local optimization with SLSQP
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)
    
    # Adaptive perturbation and optimization
    def adaptive_perturbation(h):
        h = np.array(h)
        for _ in range(10):
            if time.time() - start_time > budget_s:
                break
            # Generate a random perturbation
            perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
            candidate_h = np.clip(h + perturbation, 0, 1)
            candidate_h = normalize(candidate_h, n_points)
            candidate_c5 = objective(candidate_h)
            
            # Accept if better
            if candidate_c5 < objective(h):
                h = candidate_h
        return h
    
    # Run global optimization with adaptive perturbation
    for _ in range(10):
        if time.time() - start_time > budget_s:
            break
        best_h = adaptive_perturbation(best_h)
        h_lo, c5_lo = local_optimization(best_h)
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
