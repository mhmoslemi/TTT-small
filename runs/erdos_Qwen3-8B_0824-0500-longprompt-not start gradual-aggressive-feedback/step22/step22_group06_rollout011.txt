import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    best_n = 0

    def objective(h):

        c5 = np.max(overlap)

    def perturb(h, magnitude=0.01):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization loop
    for _ in range(200):
        if time.time() - start_time > budget_s:

        perturbed_h = perturb(h_initial, magnitude=perturbation_magnitude)

        h_initial = h_lo
