import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Predefined initial construction
    n_initial = len(initial_h_values)
    h_initial = initial_h_values.copy()
    
    best_c5 = np.inf
    best_h = None
    best_n = n_initial
    
    start_time = time.time()
    
    # Adaptive perturbation parameters
    max_perturbation = 0.15
    perturbation_decay = 0.95
    
    # Local optimization function with normalization
    def optimize_with_normalization(h, n_points, remaining_time):
        def objective(h_vec):
            h = np.array(h_vec)
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            c5 = np.max(overlap)
            return c5
        
        def constraint(h_vec):
            h = np.array(h_vec)
            total = np.sum(h) * (2.0 / n_points)
            return total - 1.0
        
        # Perform constrained optimization
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        
        return result.x, result.fun
    
    # Main optimization loop
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Perturb the current solution
        perturbation = np.random.normal(0, max_perturbation, len(h_initial))
        h_perturbed = np.clip(h_initial + perturbation, 0, 1)
        
        # Normalize the perturbed solution to maintain integral constraint
        h_normalized = h_perturbed * (n_initial / np.sum(h_perturbed)) if np.sum(h_perturbed) > 0 else h_perturbed
        
        # Refine the normalized solution
        h_refined, c5 = optimize_with_normalization(h_normalized, len(h_normalized), budget_s - (time.time() - start_time))
        
        # Update the best solution if improvement is found
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_refined.copy()
            best_n = len(h_refined)
        
        # Decrease perturbation magnitude to refine the search
        max_perturbation *= perturbation_decay
    
    return (best_h, best_c5, best_n)
