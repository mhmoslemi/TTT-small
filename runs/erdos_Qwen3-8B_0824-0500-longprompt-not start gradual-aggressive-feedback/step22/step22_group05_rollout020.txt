import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    best_c5 = np.inf

        if n_points == len(initial_h):
            h_initial = initial_h.copy()

            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)

        def objective(h):
            h = np.array(h)
            overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
            c5 = np.max(overlap)
            return c5
        
        # Function to maintain the integral constraint
        def constraint(h):
            h = np.array(h)
            total = np.sum(h) * (2.0 / n_points)
            return total - 1.0
        
        # Main optimization loop with adaptive perturbations
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
            
            # Perturb the current solution with adaptive magnitude
            remaining_time = budget_s - (time.time() - start_time)
            perturbation_magnitude = 0.01 * (1 - (remaining_time / budget_s))
            perturbed_h = np.clip(h_initial + np.random.normal(0, perturbation_magnitude, n_points), 0, 1)
            
            # Refine the perturbed solution with local optimization
            result = optimize.minimize(
                objective,
                perturbed_h,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
            )
            
            h_lo = result.x
            c5_lo = result.fun
            
            # Update the best solution if improvement is found
            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
                best_n = n_points
