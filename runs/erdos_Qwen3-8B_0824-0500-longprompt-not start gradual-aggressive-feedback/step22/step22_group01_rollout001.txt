import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a different initial construction
    h_initial = np.random.rand(100)

    h_initial = h_initial / np.sum(h_initial) * (100 / 2)

    best_n = 100

    # Define the evaluation function
    def evaluate(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        c5 = np.max(overlap)
        return c5

    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / len(h))
        return total - 1.0
    
    # Define the perturbation function
    def perturb(h, magnitude=0.01):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed

    for _ in range(150):
