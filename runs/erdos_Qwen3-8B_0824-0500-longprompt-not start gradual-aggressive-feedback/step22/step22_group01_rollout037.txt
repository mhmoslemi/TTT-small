import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):

    best_n = n_points

    def evaluate(h):

        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
        c5 = np.max(overlap)

        h = np.array(h)
