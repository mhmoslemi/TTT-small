    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    def objective(h):
        one_minus_h = 1 - h
        overlap = scipy.signal.correlate(h, one_minus_h, mode='full') * dx
        return np.max(overlap)
    
    def fitness(h):
        return -objective(h)  # We want to minimize the overlap, so maximize the fitness
