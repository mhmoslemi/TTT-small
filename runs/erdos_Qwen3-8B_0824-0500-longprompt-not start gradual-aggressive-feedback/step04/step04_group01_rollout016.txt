def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 50
    individual_size = n_points
    
    # Define the fitness function
    def evaluate(individual):
        h_values = np.array(individual, dtype=float)
        h_values = np.clip(h_values, 0, 1)
        # Enforce integral constraint
        integral = np.sum(h_values) * dx
        if abs(integral - 1.0) > 1e-6:
            return (1e6,)
        one_minus_h = 1 - h_values
        overlap = scipy.signal.correlate(h_values, one_minus_h, mode='full') * dx
        c5 = np.max(overlap)
        return (c5,)
    
    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", lambda: np.random.uniform(0, 1))
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=individual_size)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    toolbox.register("evaluate", evaluate)
    
    # Start with the initial solution
    initial_individual = initial_h_values.copy()
    initial_individual = np.clip(initial_individual, 0, 1)
    initial_individual = initial_individual / np.sum(initial_individual) * (n_points / 2)
    initial_individual = np.round(initial_individual).astype(float)

    population = [initial_individual]
    
    # Evaluate the initial population
    fitnesses = list(map(toolbox.evaluate, population))
    for ind, fit in zip(population, fitnesses):
        ind.fitness = fit
    
    # Genetic algorithm parameters
    generations = 50
    population = toolbox.population(n=population_size)
    start_time = time.time()
    
    # Run the genetic algorithm
    for gen in range(generations):
        if time.time() - start_time > budget_s:

        population = toolbox.select(population + offspring, k=population_size)
    
    # Find the best individual
    best_individual = min(population, key=lambda ind: ind.fitness.values[0])
    best_c5 = best_individual.fitness.values[0]
