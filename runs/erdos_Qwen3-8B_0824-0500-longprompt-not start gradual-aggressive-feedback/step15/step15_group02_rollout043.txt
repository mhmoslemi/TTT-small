import random

        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-10, 'disp': False}
        )
        return result.x, objective(result.x)
    
    def differential_evolution(population_size=20, generations=15):
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [normalize(ind, n_points) for ind in population]
        
        for _ in range(generations):
            if time.time() - start_time > budget_s:
                break
