import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with a smaller number of samples for faster evaluation and refinement
    n_points = 100
    initial_h_values = np.random.rand(n_points)
    initial_h_values = (initial_h_values * n_points / 2).astype(float)
    
    best_c5 = np.inf
    best_h = initial_h_values.copy()
    best_n = n_points
    
    start_time = time.time()
    
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)
    
    # Initial optimization with the initial guess
    h_lo, c5_lo = local_optimization(best_h)
    if c5_lo < best_c5:
        best_c5 = c5_lo
        best_h = h_lo.copy()

            break

        h_lo, c5_lo = local_optimization(h_initial)
        if c5_lo < best_c5:
            best_c5 = c5_lo
