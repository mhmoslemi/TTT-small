import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100  # Use 100 samples for a balance between accuracy and speed
    dx = 2.0 / n_points
    initial_h = np.random.rand(n_points)  # Initial guess
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    
    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h):
        total = np.sum(h) * dx
        if total == 0:
            return h
        return h * (1.0 / total)
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = normalize(h)
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)
        return c5
    
    # Define bounds

    # Genetic algorithm parameters
    population_size = 50
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initial population
    population = np.random.rand(population_size, n_points)
    
    for _ in range(generations):
        if time.time() - start_time > budget_s:
            break
        
        # Evaluate fitness
        fitness = np.array([objective(ind) for ind in population])
        
        # Select the best individuals
        best_indices = np.argsort(fitness)[:int(population_size / 2)]
        best_population = population[best_indices]
        
        # Generate new population
        new_population = np.zeros((population_size, n_points))
        
        for i in range(population_size):

                parent1 = best_population[np.random.randint(len(best_population))]

                child = np.clip(child, 0, 1)

        population = new_population
    
    # Evaluate the best solution
    best_h = population[np.argmin(fitness)]
    best_c5 = objective(best_h)
    
    return (best_h, best_c5, n_points)
