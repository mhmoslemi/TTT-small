import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initialize with a structured pattern to maintain integral constraint
    n_points = 100
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        initial_h_values[i] = 1.0 if i % 5 == 0 else 0.0
    initial_h_values = initial_h_values / (np.sum(initial_h_values) / (n_points / 2))
    
    best_c5 = np.inf
    best_h = initial_h_values.copy()
    best_n = n_points
    
    start_time = time.time()
    
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-10, 'disp': False}
        )
        return result.x, objective(result.x)
    
    def evolutionary_search(population_size=20, generations=10):
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [normalize(ind, n_points) for ind in population]
        
        for _ in range(generations):
            if time.time() - start_time > budget_s:
                break
            for i in range(population_size):
                a, b, c = np.random.choice(population_size, 3, replace=False)
                mutant = population[a] + np.random.rand(n_points) * (population[b] - population[c])
                mutant = np.clip(normalize(mutant, n_points), 0, 1)
                if objective(mutant) < objective(population[i]):
                    population[i] = mutant
        
        best = min(population, key=objective)
        return best, objective(best)
    
    # Initial optimization
    h_lo, c5_lo = local_optimization(best_h)
    if c5_lo < best_c5:

        best_h = h_lo.copy()
        best_n = n_points
    
    # Evolutionary search for global optimization
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        h_de, c5_de = evolutionary_search()
        if c5_de < best_c5:
            best_c5 = c5_de
            best_h = h_de.copy()
            best_n = n_points

            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
                best_n = n_points

        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
