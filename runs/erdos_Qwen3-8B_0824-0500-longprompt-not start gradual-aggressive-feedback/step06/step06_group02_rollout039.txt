import scipy.optimize as optimize
import time
import random
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=60, **kwargs):

    random.seed(seed)

    initial_h = initial_h_values
    n_points = len(initial_h)

    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total

    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    # Initialization
    def init个体():
        return [np.random.uniform(0, 1) for _ in range(n_points)]
    
    toolbox = base.Toolbox()
    toolbox.register("individual", tools.initRepeat, creator.Individual, init个体, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.05, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    toolbox.register("evaluate", objective)
    
    # Genetic algorithm parameters
    pop_size = 50
    n_gen = 100
    cxpb = 0.5
    mutpb = 0.1
    
    # Run genetic algorithm
    pop = toolbox.population(n=pop_size)
    for gen in range(n_gen):

        for fit, ind in zip(fits_offspring, offspring):

        pop = toolbox.select(offspring + pop, k=len(pop))
    
    # Extract best individual from genetic algorithm
    best_h_genetic = min(pop, key=lambda ind: ind.fitness.values[0])
    best_c5_genetic = best_h_genetic.fitness.values[0]

    # Run local optimization on the best individual from genetic algorithm
    h_lo, c5_lo = local_optimization(best_h)
    if c5_lo < best_c5:
        best_c5 = c5_lo
        best_h = h_lo.copy()
        best_n = n_points
    
    return (best_h, best_c5, best_n)
