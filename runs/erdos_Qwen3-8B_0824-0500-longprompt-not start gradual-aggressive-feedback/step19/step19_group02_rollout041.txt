import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess with a uniform distribution
    n_points = 100
    initial_h = np.full(n_points, 1.0 / n_points)
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Function to apply adaptive perturbation
    def adaptive_perturb(h, magnitude=0.01, decay=0.999):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization loop with genetic algorithm
    for _ in range(100):
        if time.time() - start_time > budget_s:
            break
        
        # Generate a new population with random mutations
        new_population = []
        for _ in range(20):
            perturbed_h = adaptive_perturb(best_h, magnitude=0.01, decay=0.999)
            new_population.append(perturbed_h)
        
        # Evaluate the new population
        for h in new_population:
            result = optimize.minimize(
                objective,
                h,
                method='SLSQP',
                bounds=[(0.0, 1.0) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': constraint},
                options={'maxiter': 10, 'ftol': 1e-9, 'disp': False}
            )
            h_lo = result.x
            c5_lo = result.fun
            
            # Update the best solution if improvement is found
            if c5_lo < best_c5:
                best_c5 = c5_lo
                best_h = h_lo.copy()
                best_n = n_points
        
    return (best_h, best_c5, best_n)
