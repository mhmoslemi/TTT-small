import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100  # Number of points for the step function
    dx = 2.0 / n_points
    
    # Initial guess: uniform distribution
    initial_h = np.full(n_points, 1.0 / n_points)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        c5 = np.max(overlap)
        return c5
    
    # Define bounds
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * dx
        return total - 1.0
    
    # Function to apply adaptive perturbation
    def adaptive_perturb(h, magnitude=0.005, decay=0.999):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization loop
    for _ in range(200):
        if time.time() - start_time > budget_s:
            break
        # Perturb the current best solution with adaptive magnitude
        perturbed_h = adaptive_perturb(best_h, magnitude=0.005, decay=0.999)
        # Refine the perturbed solution with local optimization
        result = optimize.minimize(
            objective,
            perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-9, 'disp': False}
        )
        h_lo = result.x
        c5_lo = result.fun

            best_c5 = c5_lo
            best_h = h_lo.copy()
            best_n = n_points
    
    return (best_h, best_c5, best_n)
