import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define an alternative initial construction with more even distribution
    n_initial = 100
    h_initial = np.random.rand(n_initial)
    h_initial = h_initial / np.sum(h_initial) * (n_initial / 2)
    
    best_c5 = np.inf
    best_h = None
    best_n = n_initial
    
    start_time = time.time()
    
    # Define the objective function
    def objective(h):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / len(h))
        return c5
    
    # Define the constraint function maintaining the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / len(h))
        return total - 1.0
    
    # Define a perturbation function
    def perturb(h, magnitude=0.01):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization with Bayesian optimization
    for _ in range(50):  # Max number of trials
        if time.time() - start_time > budget_s:
            break
        
        # Create a new initial guess with small perturbation
        h_initial = perturb(h_initial, magnitude=0.01)
        
        # Define bounds and constraints for optimization
        bounds = Bounds(0, 1)
        cons = [{'type': 'eq', 'fun': constraint}]
        
        # Use Bayesian optimization with local refinement
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 200, 'ftol': 1e-9, 'disp': False}
        )
        
        h_lo = result.x
        c5_lo = result.fun
        
        # Update the best solution if improvement is found

            best_c5 = c5_lo
            best_h = h_lo.copy()

    return (best_h, best_c5, best_n)
