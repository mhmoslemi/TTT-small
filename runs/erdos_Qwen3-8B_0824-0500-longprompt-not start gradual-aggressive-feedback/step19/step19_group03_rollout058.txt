            h_vec,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint},

        )
        return result.x, result.fun

    population_size = 20
    generations = 100
    mutation_rate = 0.1
    crossover_rate = 0.8

    population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]

        fitness = [objective(ind) for ind in population]
