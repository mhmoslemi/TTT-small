import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define initial construction
    initial_h_values = np.load("initial_h_values.npy")  # Assume initial_h_values is preloaded
    
    best_c5 = np.inf
    best_h = None
    best_n = 0
    start_time = time.time()
    
    # Define a function to create a structured initial guess based on a base pattern
    def structured_initial_guess(n_points, base_pattern):
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = base_pattern[i % len(base_pattern)]
        h = h / np.sum(h) * (n_points / 2)
        return h
    
    # Define multiple base patterns for structured initialization
    base_patterns = [
        np.random.rand(7),  # Random pattern
        np.sin(np.linspace(0, np.pi, 7)),  # Sinusoidal pattern
        np.cos(np.linspace(0, 2 * np.pi, 7)),  # Cosine pattern
        np.concatenate([np.ones(3), np.zeros(4)])  # Alternating pattern
    ]
    
    # Evaluate a given h vector
    def evaluate_overlap(h, n_points):
        h = np.array(h)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(overlap)
    
    # Function to maintain the integral constraint
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Function to perturb the solution
    def perturb(h, magnitude, n_points):
        perturbation = np.random.normal(0, magnitude, len(h))
        h_perturbed = np.clip(h + perturbation, 0, 1)
        return h_perturbed
    
    # Main optimization loop
    for n_points in [70, 100, 150]:
        if time.time() - start_time > budget_s:
            break
        
        # Use a structured or random initial guess
        if n_points == 70:
            # Start with the initial construction
            h_initial = np.array(initial_h_values)
            h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        else:
            # Use a structured initial guess based on a random base pattern
            base_pattern = np.random.choice(base_patterns)
            h_initial = structured_initial_guess(n_points, base_pattern)

        def objective(h):
            return evaluate_overlap(h, n_points)

                best_h = h_lo.copy()
                best_n = n_points
    
    return (best_h, best_c5, best_n)
