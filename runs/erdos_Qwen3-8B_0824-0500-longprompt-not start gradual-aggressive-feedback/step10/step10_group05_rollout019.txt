import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial guess
    n_points = 70
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        initial_h[i] = 1 if i < n_points // 2 else 0
    initial_h = np.random.permutation(initial_h)
    best_h = initial_h.copy()
    best_c5 = np.inf
    best_n = n_points
    
    start_time = time.time()
    
    # Normalize function to maintain the integral constraint
    def normalize(h):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Evaluate overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    # Constraint function
    def constraint(h):
        h = np.array(h)
        return np.sum(h) - n_points / 2
    
    # Bounds for each parameter
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Define a function to perform a local optimization
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 300, 'ftol': 1e-9, 'disp': False}
        )
        return result.x, objective(result.x)
    
    # Main optimization loop

        if c5_lo < best_c5:

            best_h = h_lo.copy()
    
    return (best_h, best_c5, best_n)
