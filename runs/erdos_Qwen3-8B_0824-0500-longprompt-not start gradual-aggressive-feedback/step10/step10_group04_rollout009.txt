import numpy as np
import scipy.optimize as optimize
import time
import random

    n_points = len(initial_h)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    start_time = time.time()
    
    # Function to normalize the solution to meet the integral constraint
    def normalize(h, n_points):
        total = np.sum(h)
        if total == 0:
            return h
        return h * (n_points / 2) / total
    
    # Function to evaluate the overlap integral
    def objective(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        return c5
    
    # Function to evaluate the overlap integral with penalty for constraint violation
    def objective_with_penalty(h):
        h = np.array(h)
        h = normalize(h, n_points)
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        c5 = np.max(overlap)
        # Penalty for constraint violation
        constraint_violation = abs(np.sum(h) * (2.0 / n_points) - 1.0)
        return c5 + 100 * constraint_violation
    
    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint function for the integral
    def constraint(h):
        h = np.array(h)
        total = np.sum(h) * (2.0 / n_points)
        return total - 1.0
    
    # Local optimization with SLSQP
    def local_optimization(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
        )
        return result.x, objective(result.x)
    
    # Perturbation-based optimization
    def perturbation_optimization(h_initial, perturbation_size=0.05, max_perturbations=50):
        h = h_initial.copy()
        for _ in range(max_perturbations):
            # Randomly perturb a few points
            indices = np.random.choice(n_points, size=5, replace=False)
            h[indices] += np.random.uniform(-perturbation_size, perturbation_size, size=5)
            h = np.clip(h, 0.0, 1.0)
            h = normalize(h, n_points)
            # Local optimization
            h_lo, c5 = local_optimization(h)
            if c5 < objective(h_lo):
                h = h_lo.copy()
        return h, objective(h)
    
    # Run perturbation-based optimization
    best_h, best_c5 = perturbation_optimization(best_h)
