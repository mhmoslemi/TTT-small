import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA

def run(seed=42, budget_s=60):
    np.random.seed(seed)

    initial_h = initial_h_values
    n_points = len(initial_h)
    n_points = max(200, n_points)

    initial_h_scaled = initial_h * scale
    initial_h_scaled = np.clip(initial_h_scaled, 0, 1)
    
    # Evaluate the initial solution
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Define bounds for all h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a combination of Nelder-Mead and L-BFGS-B for optimization
    def minimize_with_restart(method='Nelder-Mead', max_iter=100):
        result = minimize(
            objective,
            x0=initial_h_scaled,
            method=method,
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-8}
        )
        return result
    
    # Try different methods and restarts
    best_c5 = float('inf')
    best_h = np.random.rand(n_points)
    
    start_time = time.time()
    while time.time() - start_time < budget_s - 1:
        # Run Nelder-Mead
        result_nm = minimize_with_restart(method='Nelder-Mead', max_iter=50)
        h_nm = result_nm.x
        h_nm = np.clip(h_nm, 0, 1)
        h_nm = h_nm * (1 / (np.sum(h_nm) * dx))
        corr_nm = correlate(h_nm, 1 - h_nm, mode='full')
        c5_nm = np.max(corr_nm) * dx
        if c5_nm < best_c5:
            best_c5 = c5_nm
            best_h = h_nm
        
        # Run L-BFGS-B
        result_lbfgs = minimize_with_restart(method='L-BFGS-B', max_iter=50)
        h_lbfgs = result_lbfgs.x
        h_lbfgs = np.clip(h_lbfgs, 0, 1)

        c5_lbfgs = np.max(corr_lbfgs) * dx

            best_c5 = c5_lbfgs

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
