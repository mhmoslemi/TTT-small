import numpy as np
import time
from scipy.signal import correlate

    dx = 2.0 / n_points
    
    def normalize(h):
        total = np.sum(h)
        return h * (n_points / 2) / total
    
    def evaluate(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = normalize(h)
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    def fitness(h):
        return evaluate(h)
    
    class Chromosome:
        def __init__(self, genes):
            self.genes = genes
    
    def create_chromosome(genes):
        return Chromosome(genes)
    
    def crossover(parent1, parent2):
        n = len(parent1.genes)
        split = np.random.randint(1, n - 1)
        child_genes = np.concatenate((parent1.genes[:split], parent2.genes[split:]))
        return create_chromosome(child_genes)
    
    def mutate(chromosome, mutation_rate=0.1):
        n = len(chromosome.genes)
        for i in range(n):
            if np.random.rand() < mutation_rate:
                chromosome.genes[i] = np.random.uniform(0, 1)
        return chromosome
    
    def genetic_search(initial_population_size=20, generations=100, mutation_rate=0.1):
        population = [create_chromosome(np.random.uniform(0, 1, n_points)) for _ in range(initial_population_size)]
        best = min(population, key=fitness)
        best_fitness = fitness(best)
        
        for gen in range(generations):
            if time.time() - start_time > budget_s - 1:
                break
            if best_fitness <= 0.38:
                break
            new_population = []
            for _ in range(initial_population_size):
                parent1 = np.random.choice(population)
                parent2 = np.random.choice(population)
                child = crossover(parent1, parent2)
                child = mutate(child, mutation_rate)
                new_population.append(child)
            population = new_population
            current_best = min(population, key=fitness)
            if fitness(current_best) < best_fitness:
                best_fitness = fitness(current_best)
                best = current_best
        return best.genes, best_fitness, n_points
    
    start_time = time.time()
