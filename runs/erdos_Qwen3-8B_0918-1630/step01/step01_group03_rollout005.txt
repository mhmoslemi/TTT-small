    def run_optimization(initial_guess, ftol=1e-8, gtol=1e-8, maxiter=1000, **kwargs):

    # Main optimization loop with time tracking
    start_time = time.time()
    elapsed = time.time() - start_time
    
    # Run several optimization passes with increasing parameters
    for _ in range(3):
