import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    initial_sum = sum(initial_h)

    # Ensure initial sum is exactly n_points / 2
    initial_h = np.array(initial_h)
    initial_h = initial_h / (initial_sum / (n_points / 2))

    def evaluate(h):
        h = np.array(h)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            return np.inf
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap

    def objective(x):
        x = x.reshape(-1)
        return evaluate(x)

    bounds = Bounds(0.0, 1.0)

    def optimize_h(initial_guess, max_iter=1000):
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': max_iter, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if result.success:
            return result.x
        return initial_guess

    # Initial evaluation
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)

    start_time = time.time()

    # First optimization
    optimized_h = optimize_h(best_h)
    best_c5 = evaluate(optimized_h)
    best_h = optimized_h

    # Perturbation and refinement
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_c5 = evaluate(perturbed_h)
        if perturbed_c5 < best_c5:
            best_c5 = perturbed_c5
            best_h = perturbed_h

    # Try a few more iterations
    optimized_h = optimize_h(best_h)
    best_c5 = evaluate(optimized_h)
    best_h = optimized_h

    return best_h, best_c5, n_points
