import numpy as np
import time
import random
from scipy.optimize import minimize
from scipy.stats import norm

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 250
    dx = 2.0 / n_points
    h_initial = initial_h_values(n_points)
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    h0 = h_initial
    cons = ({'type': 'eq', 'fun': constraint})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    def simulated_annealing():
        T = 100.0
        T_min = 1e-5
        cooling_rate = 0.9995
        h_current = h0.copy()
        best_h = h_current.copy()
        best_c5 = evaluate_erdos_solution(h_current, n_points)
        for step in range(1000):
            T = T * cooling_rate
            if T < T_min:
                break
            for i in range(n_points):
                if random.random() < 0.1:
                    perturbation = np.random.choice([-1, 1]) * np.random.uniform(0.05, 0.2)
                    h_new = h_current.copy()
                    h_new[i] += perturbation
                    h_new = np.clip(h_new, 0.0, 1.0)
                    c5_new = evaluate_erdos_solution(h_new, n_points)
                    delta = c5_new - best_c5
                    if delta < 0 or np.random.random() < np.exp(-delta / T):
                        h_current = h_new
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_current.copy()
        return best_h, best_c5
    
    h_best, c5_best = simulated_annealing()
    
    return h_best, c5_best, n_points
