import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

    dx = 2.0 / n_points

        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), 0.0, n_points / 2)
    
    h0 = h_initial
    
    def grad_objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        max_overlap = np.max(overlap)
        derivative = np.zeros(n_points)
        for i in range(n_points):
            delta = np.zeros(n_points)
            delta[i] = 1.0
            delta_h = h + delta
            delta_h_1_minus = 1.0 - delta_h
            correlate_delta = np.correlate(delta_h, delta_h_1_minus, mode="full")
            overlap_delta = correlate_delta * dx
            max_overlap_delta = np.max(overlap_delta)
            derivative[i] = (max_overlap_delta - max_overlap) / dx
        return derivative
    
    result = minimize(objective_func, h0, method='L-BFGS-B', bounds=bounds, constraints=constraint, jac=grad_objective, options={'ftol': 1e-8, 'maxiter': 1000})
    
    if time.time() - time.time() > budget_s:
        return np.array(h_initial), evaluate_erdos_solution(h_initial, n_points), n_points
    
    h_values = result.x
    c5_bound = evaluate_erdos_solution(h_values, n_points)
