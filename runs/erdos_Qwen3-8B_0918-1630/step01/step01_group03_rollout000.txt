import numpy as np
import time
import random
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    def evaluate(h):
        h = np.array(h)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            return np.inf
        
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    best_c5 = evaluate(initial_h)
    best_h = initial_h.copy()
    
    def objective(x):
        return evaluate(x)
    
    bounds = Bounds(0.0, 1.0)
    
    def optimize_h(initial_guess):
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if result.success:
            return result.x
        return initial_guess
    
    def perturb(h, step=0.1):
        h = np.array(h)
        perturbation = np.random.uniform(-step, step, h.shape)
        perturbed = np.clip(h + perturbation, 0, 1)
        return perturbed
    
    def optimize_with_perturbations(initial_guess, max_perturbations=3):
        h = optimize_h(initial_guess)
        best_h = h
        best_c5 = evaluate(h)
        
        for _ in range(max_perturbations):
            h_perturbed = perturb(h, step=0.05)
            c5_perturbed = evaluate(h_perturbed)
            if c5_perturbed < best_c5:
                best_h = h_perturbed
                best_c5 = c5_perturbed
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h_optimized, best_c5_optimized = optimize_with_perturbations(initial_h)
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        # Try slightly larger n_points if possible
        for n in [n_points + 50, n_points + 100]:
            if n > 200:
                break
            dx_new = 2.0 / n
            h_new = np.zeros(n)
            for i in range(n):
                h_new[i] = best_h_optimized[int(i * n_points / n)]
            h_new = np.clip(h_new, 0, 1)
            sum_h_new = np.sum(h_new)
            if not np.isclose(sum_h_new, n / 2):
                continue
            corr = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr * dx_new)
            if c5_new < best_c5_optimized:
                best_c5_optimized = c5_new
                best_h_optimized = h_new
    
    return best_h_optimized, best_c5_optimized, n_points
