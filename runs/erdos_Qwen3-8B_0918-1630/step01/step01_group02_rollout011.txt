```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA
from itertools import product
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points_list = [128, 256, 512]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 256

    for n_points in n_points_list:
        if n_points > 1000:
            continue
        dx = 2.0 / n_points
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                initial_h[i] = 0.5
            else:
                initial_h[i] = 0.0
        initial_h = np.clip(initial_h, 0, 1)
        sum_h = np.sum(initial_h)
        initial_h += (n_points / 2 - sum_h) / n_points

        # Generate diverse initial points
        initial_points = []
        for i in range(5):
            initial_points.append(np.zeros(n_points))
            for j in range(n_points // 2):
                initial_points[-1][j] = 1.0
            initial_points[-1] = np.clip(initial_points[-1], 0, 1)
            sum_h = np.sum(initial_points[-1])
            initial_points[-1] += (n_points / 2 - sum_h) / n_points

        for i in range(5):
            initial_points.append(np.random.rand(n_points))
            initial_points[-1] = np.clip(initial_points[-1], 0, 1)
            sum_h = np.sum(initial_points[-1])
            initial_points[-1] += (n_points / 2 - sum_h) / n_points

        initial_points = np.array(initial_points)
        pca = PCA(n_components=2)
        initial_points = pca.fit_transform(initial_points)
        initial_points = np.clip(initial_points, 0, 1)
        initial_points = pca.inverse_transform(initial_points)

        # Generate more diverse points using KMeans
        kmeans = KMeans(n_clusters=5, n_jobs=-1)
        kmeans.fit(initial_points)
        centroids = kmeans.cluster_centers_
        initial_points = np.vstack([centroids, initial_points])
        initial_points = np.array([np.clip(p, 0, 1) for p in initial_points])

        # Run optimization with multiple initial points
        start_time = time.time()
        for initial in initial_points:
            initial = np.array(initial)
            initial = np.clip(initial, 0, 1)
            sum_h = np.sum(initial)
            initial += (n_points / 2 - sum_h) / n_points

            def evaluate_overlap(h):
                h = np.clip(h, 0, 1)
                conv = np.correlate(h, 1 - h, mode="full")
                c5 = np.max(conv) * dx
                return c5

            def objective(x):
                return evaluate_overlap(x)

            def constraint_total_area(x):
                return np.sum(x) - n_points / 2

            bounds = Bounds(0, 1)
            cons = (LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2),)

            try:
                result = minimize(
                    objective,
                    initial,
                    method='L-BFGS-B',
                    bounds=bounds,
                    constraints=cons,
                    tol=1e-5,
                    options={'disp': False, 'maxiter': 100}
                )
                h_candidate = result.x
                c5_value = evaluate_overlap(h_candidate)
                if c5_value < best_c5:
                    best_c5 = c5_value
                    best_h = h_candidate
                    best_n_points = n_points
            except:
                pass

            if time.time() - start_time > budget_s:
                break

    return (best_h, best_c5, best_n_points)
```
