```python
import numpy as np
import time
import random
from scipy.optimize import minimize
from functools import partial

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def perturb_h(h, scale=0.1, seed=None):
    if seed is not None:
        np.random.seed(seed)
    h = np.array(h)
    indices = np.random.permutation(len(h))
    for i in indices[:len(h) // 4]:
        h[i] = min(max(h[i] + np.random.uniform(-scale, scale), 0.0), 1.0)
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 256
    initial_h = initial_h_values(n_points)
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    elapsed_time = 0
    
    def objective(h):
        return evaluate_erdos_solution(h, n_points)
    
    for _ in range(50):
        if elapsed_time > budget_s:
            break
        current_h = perturb_h(best_h, scale=0.2, seed=seed + _)
        c5 = evaluate_erdos_solution(current_h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h
        elapsed_time = time.time() - time.time()
        time.sleep(0.1)
    
    return best_h, best_c5, n_points
```
