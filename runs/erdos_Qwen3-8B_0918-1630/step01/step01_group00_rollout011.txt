import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.stats import truncnorm

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    n_points = 500
    dx = 2.0 / n_points
    x_vals = np.linspace(0, 2, n_points, endpoint=False)
    
    # Define initial diverse constructions
    def construct_uniform():
        return np.array([1.0 / n_points] * n_points)
    
    def construct_sparse():
        idx = np.random.choice(n_points, size=100, replace=False)
        h = np.zeros(n_points)
        h[idx] = 1.0
        return h / np.sum(h)
    
    def construct_periodic():
        h = np.sin(2 * np.pi * (x_vals - 1.0) / 2.0)
        h = (h + 1.0) / 2.0
        h = h.clip(0, 1)
        return h
    
    def construct_random():
        h = np.random.rand(n_points)
        h = (h - 0.5) * 2.0
        h = np.clip(h, 0, 1)
        return h / np.sum(h)
    
    candidates = [construct_uniform(), construct_sparse(), construct_periodic(), construct_random()]
    candidates = [np.array(c) for c in candidates]
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    def objective(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return float('inf')
        h_scaled = h / total * (1 / dx)
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Bounds for all h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    for i, init in enumerate(candidates):
        result = minimize(
            objective,
            x0=init,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 500, 'ftol': 1e-8}
        )
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            if total == 0:
                continue
            h_scaled = h / total * (1 / dx)
            corr = correlate(h_scaled, 1 - h_scaled, mode='full')
            c5 = np.max(corr) * dx
