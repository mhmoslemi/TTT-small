import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from skopt import BayesSearchSpace, BayesianOptimization

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points_list = [256, 512, 1024]  # Try different n_points sizes
    best_c5 = float('inf')
    best_h = np.array([])
    best_n_points = 0
    
    for n_points in n_points_list:
        if n_points > 1024:
            continue
        
        dx = 2.0 / n_points
        initial_h_values = np.random.uniform(0, 1, n_points)
        initial_h_values = np.sort(initial_h_values)
        initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        
        def objective(x):
            x = np.clip(x, 0, 1)
            conv = np.correlate(x, 1 - x, mode="full")
            c5 = np.max(conv) * dx
            return c5
        
        def constraint_total_area(x):
            return np.sum(x) - n_points / 2
        
        bounds = Bounds(0, 1)
        cons = (LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2),)
        
        # Define a Bayesian optimization search space
        search_space = BayesSearchSpace([
            ('h', BayesSearchSpace('real', low=0.0, high=1.0, dimension=n_points))
        ])
        
        # Set up the Bayesian optimizer
        opt = BayesianOptimization(objective, search_space, n_initial_points=10, n_iter=20, random_state=seed)
        
        start_time = time.time()
        for _ in range(20):
            if time.time() - start_time > budget_s - 3:
                break
            result = opt.ask_and_tell()
        
        best = opt.get_best()
        best_h_current = best['x']
        current_c5 = objective(best_h_current)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h_current
            best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
