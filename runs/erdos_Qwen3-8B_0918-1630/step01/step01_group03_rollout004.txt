import time
from scipy.optimize import minimize

    def evaluate(h):
        h = np.array(h)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            return np.inf
        
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    # Initial evaluation
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    # Function to generate a new initial guess with perturbation
    def generate_perturbed_guess(h, perturbation_strength=0.1, seed=None):
        if seed is not None:
            np.random.seed(seed)
        perturbation = np.random.uniform(-perturbation_strength, perturbation_strength, len(h))
        perturbed_h = np.clip(h + perturbation, 0, 1)
        return perturbed_h
    
    # Optimization function with bounds and constraints
    def objective(x):
        x = x.reshape(-1)
        return evaluate(x)
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization strategy: try several initial guesses
    def optimize_h(initial_guess, num_restarts=5, max_iter=200):
        best_result = np.inf
        best_solution = initial_guess
        for seed in range(num_restarts):
            perturbed_guess = generate_perturbed_guess(initial_guess, seed=seed)
            result = minimize(
                objective,
                perturbed_guess,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': max_iter, 'ftol': 1e-10, 'gtol': 1e-10}
            )
            if result.success and result.fun < best_result:
                best_result = result.fun
                best_solution = result.x
        return best_solution, best_result
    
    # Main optimization loop
    start_time = time.time()
    optimized_h, best_c5 = optimize_h(best_h, num_restarts=5, max_iter=200)
    best_h = optimized_h
    
    # If there is time left, try a more aggressive perturbation strategy
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        for _ in range(3):
            perturbation = np.random.uniform(-0.2, 0.2, n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_c5 = evaluate(perturbed_h)
            if perturbed_c5 < best_c5:
                best_c5 = perturbed_c5
                best_h = perturbed_h
    
    return best_h, best_c5, n_points
