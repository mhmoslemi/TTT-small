import numpy as np
import time
import random
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    random.seed(seed)
    np.random.seed(seed)

    # Try with different n_points
    n_points_list = [100, 150, 200]
    best_c5 = float('inf')
    best_h = None

    for n_points in n_points_list:
        dx = 2.0 / n_points
        # Initial guess: Random structured pattern
        h_initial = np.zeros(n_points)
        for i in range(n_points):
            if i % 2 == 0:
                h_initial[i] = 1.0
        h_initial = h_initial.tolist()

        def objective(h_vec):
            h = np.array(h_vec, dtype=np.float64)
            h_1_minus = 1.0 - h
            correlate = np.correlate(h, h_1_minus, mode="full")
            overlap = correlate * dx
            return np.max(overlap)

        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)

        # Run optimization
        start_time = time.time()
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-9, 'maxiter': 200})
        elapsed = time.time() - start_time

        if elapsed > budget_s:
            continue

        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x

    return best_h, best_c5, n_points_list[-1]
