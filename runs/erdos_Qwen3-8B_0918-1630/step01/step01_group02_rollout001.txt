import numpy as np
import time
from scipy.optimize import Bounds, LinearConstraint, minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 512  # 512 samples for better resolution under 1k
    dx = 2.0 / n_points

    # Initial guess: Use a smooth transition from 0 to 1 to avoid sharp jumps
    initial_h_values = np.sin(np.linspace(0, np.pi, n_points) ** 2)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    initial_h_values = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))
    initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    sum_h = np.sum(initial_h_values)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        initial_h_values += (n_points / 2 - sum_h) / n_points

    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5

    def objective(x):
        return evaluate_overlap(x)

    def constraint_total_area(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = (LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2),)

    result = minimize(objective, initial_h_values, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'disp': False, 'ftol': 1e-5, 'maxiter': 100})

    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    return (best_h, best_c5, n_points)
