import numpy as np
import time
from skopt import BayesSearchSpace, BayesianOptimization

from skopt.optimizer import Optimizer
from skopt.sampler import qeseries
from skopt.constraints import Interval, Constraint

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = sum(initial_h)
    
    def evaluate(h):
        h = np.array(h)
        if not np.isclose(np.sum(h), n_points / 2):
            return np.inf
        corr = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    def objective(params):
        h = np.zeros(n_points)
        start = params['start']
        width = params['width']
        height = params['height']
        for i in range(len(start)):
            pos = int(start[i] * n_points)
            if pos < 0:
                pos = 0
            if pos >= n_points:
                pos = n_points - 1
            for j in range(pos, min(pos + int(width[i] * n_points), n_points)):
                h[j] = height[i]
        return evaluate(h)
    
    def report(y):
        print(f"Found C5 value: {y}")
        return 0
    
    search_space = BayesSearchSpace(
        {
            'start': Interval(0.0, 2.0, 100),
            'width': Interval(0.0, 2.0, 100),
            'height': Interval(0.0, 1.0, 100),
        }
    )
    
    opt = Optimizer(
        space=search_space,
        objective=objective,
        n_jobs=1,
        acq_func='EI',
        maxiter=100,
        n_points=5,
        random_state=seed
    )
    
    best_result = np.inf
    best_params = None
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        result = opt.ask()
        if result is None:
            break
        params = result
        val = objective(params)
        opt.tell(val, report)
        if val < best_result:
            best_result = val
            best_params = params
    
    h = np.zeros(n_points)
    for i in range(len(best_params['start'])):
        pos = int(best_params['start'][i] * n_points)
        if pos < 0:
            pos = 0
        if pos >= n_points:
            pos = n_points - 1
        for j in range(pos, min(pos + int(best_params['width'][i] * n_points), n_points)):
            h[j] = best_params['height'][i]
    
    return h, best_result, n_points
