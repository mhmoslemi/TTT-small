import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    total_area = sum(initial_h) * dx
    initial_h = np.array(initial_h)
    initial_h = initial_h / total_area * 1  # Normalize to ensure integral is 1

    def evaluate(h):
        h = np.array(h)
        sum_h = np.sum(h) * dx
        if not np.isclose(sum_h, 1.0):
            return np.inf
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap

    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)

    def objective(x):
        return evaluate(x)

    bounds = [(0.0, 1.0) for _ in range(n_points)]

    def optimize_h(initial_guess):
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if result.success:
            return result.x
        return initial_guess

    start_time = time.time()
    optimized_h = optimize_h(best_h)
    best_c5 = evaluate(optimized_h)
    best_h = optimized_h
    elapsed = time.time() - start_time

    if elapsed < budget_s - 5:
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_c5 = evaluate(perturbed_h)
            if perturbed_c5 < best_c5:
                best_c5 = perturbed_c5
                best_h = perturbed_h

    return best_h.tolist(), best_c5, n_points
