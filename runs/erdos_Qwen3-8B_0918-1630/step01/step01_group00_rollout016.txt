import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from random import random, seed

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points / 2:
            initial_h_values[i] = 1.0
        else:
            initial_h_values[i] = 0.0
    initial_h_values = np.random.permutation(initial_h_values)
    initial_h_values = np.random.permutation(initial_h_values)

    def objective(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        scale = 1.0 / total
        h_scaled = h * scale
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def update_h(h, temp):
        h = h.copy()
        for i in range(n_points):
            if random() < 0.1:
                h[i] = 1.0 if h[i] == 0.0 else 0.0
            elif random() < 0.2:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.1), 0, 1)
        total = np.sum(h) * dx
        scale = 1.0 / total
        h_scaled = h * scale
        return h_scaled

    def sa_step(h, temp):
        h_new = update_h(h, temp)
        obj_new = objective(h_new)
        obj_old = objective(h)
        if obj_new < obj_old:
            return h_new, obj_new
        else:
            return h, obj_old

    best_h = initial_h_values
    best_c5 = objective(best_h)
    current_h = initial_h_values
    current_c5 = best_c5
    temp = 1.0
    decay_rate = 0.999

    start_time = time.time()
    while time.time() - start_time < budget_s:
        for _ in range(100):
            new_h, new_c5 = sa_step(current_h, temp)
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
        temp *= decay_rate

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    corr = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    return best_h, c5_bound, n_points
