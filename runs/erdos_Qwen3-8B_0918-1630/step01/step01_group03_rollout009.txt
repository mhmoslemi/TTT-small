import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import random

def run(seed=42, budget_s=60, **kwargs):
    # Predefined initial sample from initial_h_values, with n_points=47
    initial_h_values = [0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                       0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                       0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                       0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                       0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5]
    n_points = len(initial_h_values)
    initial_h = np.array(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Define the evaluation function
    def evaluate(h):
        h = h.copy()
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2):
            return np.inf  # Penalize deviation from integral constraint
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    # Initialize best solution
    best_h = initial_h.copy()
    best_c5 = evaluate(best_h)
    
    # Define the objective for minimization
    def objective(x):
        x = x.reshape(-1)
        return evaluate(x)
    
    # Define bounds for optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization function with some heuristics and parallel trials
    def optimize_h(initial_guess):
        result = minimize(
            objective,
            initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 300, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if result.success:
            return result.x
        return initial_guess
    
    # Start time tracking
    start_time = time.time()
    
    # Run multi-start optimization with perturbations
    multi_start = 3
    for _ in range(multi_start):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        optimized_h = optimize_h(perturbed_h)
        current_c5 = evaluate(optimized_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Try extending the search with different n_points (adaptive grid size)
    adaptive_n_points = [50, 100, 150, 200]
    for n in adaptive_n_points:
        if time.time() - start_time > budget_s - 1:
            break
        # Resample to new n_points
        new_dx = 2.0 / n
        new_h = np.linspace(0, 2, n)
        new_h_values = np.interp(new_h, np.linspace(0, 2, n_points), best_h)
        new_h_values = np.clip(new_h_values, 0, 1)
        new_h_values[0] = 0.5  # Ensure boundary condition
        new_h_values[-1] = 0.5
        optimized_h = optimize_h(new_h_values)
        current_c5 = evaluate(optimized_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
    
    # Handle time budget and return
    elapsed = time.time() - start_time
    if elapsed < budget_s:
        # Final perturbation sweep
        for _ in range(5):
            perturbation = np.random.uniform(-0.05, 0.05, n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            current_c5 = evaluate(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed_h
    
    best_h = np.clip(best_h, 0, 1)
    
    return best_h.tolist(), best_c5, n_points
