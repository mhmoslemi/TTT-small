import numpy as np
import time
import random
from scipy.optimize import minimize
from scipy.optimize import Bounds
import multiprocessing
from functools import partial

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def generate_multi_perturbations(h, n_perturb, scale=0.1):
    perturbations = []
    for _ in range(n_perturb):
        perturbation = np.random.normal(0, scale, len(h))
        perturbation = np.clip(perturbation, -0.5, 0.5)
        perturbed_h = h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbations.append(perturbed_h)
    return perturbations

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points_list = [200, 300, 400]
    
    max_iterations = 50
    best_c5 = float('inf')
    best_h = np.zeros(200)
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        initial_h = initial_h_values(n_points)
        perturbations = generate_multi_perturbations(initial_h, 20, scale=0.2)
        perturbations.append(initial_h)
        
        def objective(h):
            h = np.array(h)
            h_1_minus = 1.0 - h
            correlate = np.correlate(h, h_1_minus, mode="full")
            overlap = correlate * dx
            return np.max(overlap)
        
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = Bounds(0.0, 1.0)
        for pert in perturbations:
            for _ in range(max_iterations):
                try:
                    h0 = pert.copy()
                    cons = [{'type': 'eq', 'fun': constraint}]
                    result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-9, 'maxiter': 100})
                    if not result.success:
                        continue
                    c5 = result.fun
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = result.x
                    break
                except:
                    continue
    
    return best_h, best_c5, len(best_h)
