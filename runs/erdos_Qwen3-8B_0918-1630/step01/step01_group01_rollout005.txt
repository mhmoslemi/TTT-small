import numpy as np
import time
import random
from scipy.optimize import minimize, Bounds
from scipy.optimize import Bounds, LinearConstraint
from scipy.optimize import Bounds, minimize

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    h_initial = initial_h_values(n_points)
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    h0 = h_initial
    cons = ({'type': 'eq', 'fun': constraint})
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    start_time = time.time()
    result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 200})
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        return np.array(h_initial), evaluate_erdos_solution(h_initial, n_points), n_points
    
    h_values = result.x
    c5_bound = evaluate_erdos_solution(h_values, n_points)
    return h_values, c5_bound, n_points
