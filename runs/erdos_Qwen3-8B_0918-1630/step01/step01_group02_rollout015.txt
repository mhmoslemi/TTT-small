```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import differential_evolution
from functools import partial

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    n_points = 512  # Reduced for better performance
    dx = 2.0 / n_points
    
    # Base initial guess: low-frequency wave-like pattern
    initial_h_base = np.sin(np.linspace(0, 2 * np.pi, n_points))
    initial_h_base = (initial_h_base + 1) / 2
    initial_h_base = np.clip(initial_h_base, 0, 1)
    total_area = np.sum(initial_h_base) * dx
    initial_h_base = initial_h_base * (n_points / 2) / total_area
    
    # Alternative initial guess: alternating bins
    initial_h_alt = np.zeros(n_points)
    initial_h_alt[:n_points // 2] = 1
    initial_h_alt[n_points // 2:] = 0
    total_area_alt = np.sum(initial_h_alt) * dx
    initial_h_alt = initial_h_alt * (n_points / 2) / total_area_alt
    
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5

    def objective(x):
        return evaluate_overlap(x)

    def constraint_total_area(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = (LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2),)

    def optimize_with_restart(initial_guess, max_iter=100, tol=1e-5):
        result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=cons, tol=tol, options={'disp': False, 'ftol': tol, 'maxiter': max_iter})
        if result.success:
            return result.x, result.fun
        else:
            return initial_guess, np.inf

    def differential_evolutionsolver():
        def fitness(x):
            return evaluate_overlap(x)

        def constraint_func(x):
            return np.sum(x) - n_points / 2

        def feasibility(x):
            return constraint_func(x) / (n_points / 2)

        def constraint(x):
            return feasibility(x)

        bounds = [(0, 1) for _ in range(n_points)]
        result = differential_evolution(fitness, bounds, constraints=constraint, strategy='best1bin', popsize=20, maxiter=50, tol=1e-5, mutation=(0.5, 1), recombination=0.7, disp=False)
        if result.success:
            return result.x, result.fun
        else:
            return np.random.uniform(0, 1, n_points), np.inf

    candidates = [initial_h_base, initial_h_alt]
    best_h = None
    best_c5 = np.inf
    elapsed = time.time() - start_time

    # Run SLSQP with initial guesses
    for guess in candidates:
        if elapsed > budget_s - 1:
            break
        h, c5 = optimize_with_restart(guess)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        elapsed = time.time() - start_time

    if elapsed < budget_s - 1:
        # Run differential evolution as fallback
        h, c5 = differential_evolutionsolver()
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    if best_c5 == np.inf:
        best_h = initial_h_base
        best_c5 = evaluate_overlap(best_h)

    return (best_h, best_c5, n_points)
```
