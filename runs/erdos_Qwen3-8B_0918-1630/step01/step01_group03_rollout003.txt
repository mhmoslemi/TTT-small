    def genetic_algorithm(pop_size=100, generations=50, mutation_rate=0.1, tournament_size=5):
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        best_individual = population[0]
        best_score = evaluate(best_individual)

            scores = [evaluate(ind) for ind in population]

            parents = []
            for _ in range(pop_size // 2):
                tournament = np.random.choice(population, tournament_size)
                winner = tournament[np.argmin([evaluate(ind) for ind in tournament])]
                parents.append(winner)
            
            # Create new population
            new_population = []
            for _ in range(pop_size // 2):
                parent1 = np.random.choice(parents)
                parent2 = np.random.choice(parents)
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                    if np.random.rand() < mutation_rate:
                        child[i] = np.clip(child[i] + np.random.uniform(-0.1, 0.1), 0, 1)
                new_population.append(child)
            
            population = new_population
            best_score = min([s for s in scores + [best_score]])
