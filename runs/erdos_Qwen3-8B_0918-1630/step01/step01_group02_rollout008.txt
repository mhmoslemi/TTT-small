import numpy as np
import time
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 512  # Use 512 samples for higher precision
    dx = 2.0 / n_points

    # Initial guess: load from initial_h_values if provided, or use a simple pattern
    initial_h_values = np.array([0.5 if i % 2 == 0 else 1.0 for i in range(n_points)])
    initial_h = np interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)

    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5

    def constraint_total_area(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = (LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2),)

    # Use a local optimization with initial guess
    result = minimize(evaluate_overlap, initial_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-7, options={'maxiter': 100})

    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    return (best_h, best_c5, n_points)
