import numpy as np
import time
import random
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import differential_evolution

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with smaller point count for faster exploration
    n_points = 300
    dx = 2.0 / n_points
    
    # Initial guess: periodic pattern
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 10 == 0:
            h_initial[i] = 1.0
    
    # Function to evaluate the objective
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    # Constraint: total integral of h is 1
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Local optimization with SLSQP
    def local_optimize(h_initial):
        cons = [{'type': 'eq', 'fun': constraint}]
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 200})
        return result.x
    
    # Simulated annealing with fixed step size
    def simulated_annealing(h_initial):
        h = h_initial.copy()
        temp = 1.0
        cooling_rate = 0.995
        best_h = h.copy()
        best_value = objective(h)
        
        for _ in range(1000):
            step = 0.1
            i = np.random.randint(0, n_points)
            perturbation = np.random.uniform(-step, step)
            h_new = h.copy()
            h_new[i] += perturbation
            h_new[i] = np.clip(h_new[i], 0.0, 1.0)
            
            if np.sum(h_new) * dx == 1.0:
                value_new = objective(h_new)
                delta = value_new - best_value
                if delta < 0 or np.random.random() < np.exp(-delta / temp):
                    h = h_new
                    if value_new < best_value:
                        best_value = value_new
                        best_h = h.copy()
            temp *= cooling_rate
        return best_h
    
    # Differential evolution
    def differential_evolution_run():
        def objective_adev(h):
            h = np.array(h)
            h_1_minus = 1.0 - h
            correlate = np.correlate(h, h_1_minus, mode="full")
            overlap = correlate * dx
            return np.max(overlap)
        
        def constraint_adev(h):
            return np.sum(h) - n_points / 2
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        result = differential_evolution(objective_adev, bounds, constraints=constraint_adev, strategy='best1bin', 
                                      popsize=20, maxiter=100, tol=1e-8, mutation=(0.5, 1), recombination=0.8, 
                                      workers=1, polish=True, callback=None)
        return result.x
    
    # Optimization steps
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(h_initial, n_points)
    
    # Try local optimization
    h_local = local_optimize(best_h)
    c5_local = evaluate_erdos_solution(h_local, n_points)
    if c5_local < best_c5:
        best_h = h_local
        best_c5 = c5_local
    
    # Try simulated annealing
    h_sa = simulated_annealing(best_h)
    c5_sa = evaluate_erdos_solution(h_sa, n_points)
    if c5_sa < best_c5:
        best_h = h_sa
        best_c5 = c5_sa
    
    # Try differential evolution
    h_de = differential_evolution_run()
    c5_de = evaluate_erdos_solution(h_de, n_points)
    if c5_de < best_c5:
        best_h = h_de
        best_c5 = c5_de
    
    # Final evaluation
    elapsed = time.time() - time.time()
    return best_h, best_c5, n_points
