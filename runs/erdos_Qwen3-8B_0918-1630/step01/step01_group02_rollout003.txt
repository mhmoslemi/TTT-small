def run(seed=42, budget_s=60, **kwargs):

    mutation_rate = 0.1
    crossover_rate = 0.7

        # A smooth, periodic-like function starting from 0 to 1
        x = np.linspace(0, 2, n)

    initial_population = []

        sum_h = np.sum(h)

            h += delta

def evaluate_erdos_solution(h, dx):
    h = np.clip(h, 0, 1)
    conv = np.correlate(h, 1 - h, mode="full")
    return np.max(conv) * dx
