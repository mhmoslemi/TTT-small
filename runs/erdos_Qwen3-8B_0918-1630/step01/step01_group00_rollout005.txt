import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from functools import partial

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial guess: use the provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    scale = 1.0 / (sum_h * dx)
    initial_h_scaled = initial_h * scale
    initial_h_scaled = np.clip(initial_h_scaled, 0, 1)
    
    # Evaluation function
    def evaluate(h, dx):
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to optimize
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Run L-BFGS-B with multiple restarts
    best_c5 = float('inf')
    best_h = initial_h_scaled
    
    start_time = time.time()
    remaining_time = budget_s - (time.time() - start_time)
    
    # Try optimizing with different initial guesses
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        # Generate new initial guess (random)
        new_h = np.random.uniform(0, 1, n_points)
        new_h = new_h * (1 / (np.sum(new_h) * dx))
        new_h = np.clip(new_h, 0, 1)
        
        result = minimize(
            objective,
            x0=new_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (1 / (np.sum(current_h) * dx))

        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
