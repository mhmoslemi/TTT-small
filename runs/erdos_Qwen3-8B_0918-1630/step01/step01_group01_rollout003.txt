```python
import numpy as np
import time
import random
from scipy.optimize import minimize
from scipy.optimize import Bounds

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Try with different n_points to find the best solution
    n_points_list = [100, 200, 300]
    best_c5 = float('inf')
    best_h = initial_h_values(100)
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        h_initial = initial_h_values(n_points)
        
        def objective(h):
            h = np.array(h)
            h_1_minus = 1.0 - h
            correlate = np.correlate(h, h_1_minus, mode="full")
            overlap = correlate * dx
            return np.max(overlap)
        
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        bounds = Bounds(0.0, 1.0)
        cons = ({'type': 'eq', 'fun': constraint})
        
        # Try different optimization methods to avoid local optima
        methods = ['L-BFGS-B', 'SLSQP', 'COBYLA']
        for method in methods:
            h0 = h_initial
            start_time = time.time()
            result = minimize(objective, h0, method=method, bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 200})
            elapsed = time.time() - start_time
            
            if elapsed > budget_s:
                continue
            
            h_values = result.x
            c5_bound = evaluate_erdos_solution(h_values, n_points)
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_values
    
    return best_h, best_c5, len(best_h)
```
