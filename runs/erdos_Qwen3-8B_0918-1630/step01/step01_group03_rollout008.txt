import time
from scipy.optimize import differential_evolution

    dx = 2.0 / n_points
    sum_h = sum(initial_h)
    
    def evaluate(h):

        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr * dx)
        return max_overlap
    
    def objective(x):
        return evaluate(x)
    
    def generate_initial_population(size):
        population = []
        for _ in range(size):
            h = np.random.uniform(0, 1, n_points)
            if np.isclose(np.sum(h), n_points / 2):
                population.append(h)
            else:
                adjustment = (n_points / 2 - np.sum(h)) / n_points
                h = np.clip(h + adjustment, 0, 1)
                population.append(h)
        return population
    
    def fitness(individual):
        return objective(individual)
    
    start_time = time.time()
    
    population_size = 20
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=population_size,
        maxiter=100,
        mutation=(0.5, 1),
        recombination=0.9,
        tol=1e-10,
        mutation_strength=0.5,
        mutation_probability=0.8,
        polish=True,
        disp=False,
        vectorized=True
    )
    
    best_h = result.x
    best_c5 = evaluate(best_h)
    
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5:
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_c5 = evaluate(perturbed_h)
            if perturbed_c5 < best_c5:
                best_c5 = perturbed_c5
                best_h = perturbed_h
