```python
import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel
from joblib import Parallel, delayed

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    def generate_initial_guess(n_points):
        # Generate a starting guess with structured patterns
        candidates = []
        for _ in range(5):
            # Symmetric, periodic, and sparse random patterns
            if np.random.rand() > 0.5:
                # Symmetric pattern with 3 large peaks and 1 small
                peaks = np.random.choice(range(n_points), 4, replace=False)
                peaks = np.sort(peaks)
                h = np.zeros(n_points)
                for peak in peaks:
                    h[peak] = 1
                h[peaks[0]] = 0.5
                h[peaks[1]] = 0.75
                h[peaks[2]] = 0.8
                h[peaks[3]] = 0.6
                h = h / (np.sum(h) * 2.0 / n_points)
                candidates.append(h)
            else:
                # Random sparse pattern with 2-4 sharp peaks
                num_peaks = np.random.randint(2, 5)
                peaks = np.random.choice(range(n_points), num_peaks, replace=False)
                h = np.zeros(n_points)
                for peak in peaks:
                    val = np.random.uniform(0.3, 0.8)
                    h[peak] = val
                h = h / (np.sum(h) * 2.0 / n_points)
                candidates.append(h)
        return candidates
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * 2.0 / n_points))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    def evaluate_candidate(h_values, n_points):
        h = np.array(h_values)
        h = h * (1 / (np.sum(h) * 2.0 / n_points))
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * (2.0 / n_points)
    
    n_points = 100
    initial_candidates = generate_initial_guess(n_points)
    
    def optimize_with_candidate(candidate):
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            objective,
            x0=candidate,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (1 / (np.sum(best_h) * 2.0 / n_points))
        return best_h, evaluate_candidate(best_h, n_points)
    
    def parallel_optimize(candidate):
        start = time.time()
        result = optimize_with_candidate(candidate)
        elapsed = time.time() - start
        return result, elapsed
    
    results = []
    start_time = time.time()
    
    # Run parallel optimization on initial candidates
    parallel_results = Parallel(n_jobs=-1, backend="threading", verbose=0)(
        delayed(parallel_optimize)(candidate) for candidate in initial_candidates
    )
    
    # Select the best result so far
    best_result = min(parallel_results, key=lambda x: x[0][1])
    best_h = best_result[0][0]
    best_c5 = best_result[0][1]
    best_time = time.time() - start_time
    
    # If we still have budget, perform Bayesian optimization
    if best_time < budget_s - 1:
        # Build a GP model to guide the next search
        X = np.array([np.random.rand(n_points) for _ in range(20)])
        X = np.vstack(X)
        Y = np.array([evaluate_candidate(x, n_points) for x in X])
        kernel = ConstantKernel(1.0) * RBF(length_scale=1.0)
        gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=10)
        gp.fit(X, Y)
        
        def acquisition(X_test, gp, bounds, n_points):
            mu, std = gp.predict(X_test, return_std=True)
            acq = mu - 1.96 * std
            return acq
        
        def propose_new_point(gp, bounds, n_points):
            X_test = np.random.rand(100, n_points)
            acq = acquisition(X_test, gp, bounds, n_points)
            idx = np.argmax(acq)
            return X_test[idx]
        
        for _ in range(5):
            if time.time() - start_time >= budget_s - 1:
                break
            new_point = propose_new_point(gp, bounds, n_points)
            new_c5 = evaluate_candidate(new_point, n_points)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = new_point
        
        # Perform a final L-BFGS-B optimization on the best point
        result = minimize(
            objective,
            x0=best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (1 / (np.sum(best_h) * 2.0 / n_points))
        best_c5 = evaluate_candidate(best_h, n_points)
    
    return best_h, best_c5, n_points
```
