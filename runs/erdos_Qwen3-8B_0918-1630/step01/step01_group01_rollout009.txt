```python
import numpy as np
import time
import random
from scipy.optimize import minimize
from scipy.optimize import Bounds

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def cosine_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        h[i] = 0.5 * (1 + np.cos(np.pi * x))
    return h.tolist()

def wave_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i * 2.0 / n_points
        h[i] = 0.5 * (1 + np.sin(np.pi * x))
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    candidates = [initial_h_values, cosine_h_values, wave_h_values]
    best_score = float('inf')
    best_h = None
    best_n = 100
    
    for n_points in [100, 200, 300]:
        if time.time() - time.time() > budget_s:
            break
        
        for init_func in candidates:
            try:
                h_initial = init_func(n_points)
                dx = 2.0 / n_points
                h = np.array(h_initial)
                
                def objective(h_vals):
                    h = np.array(h_vals)
                    h_1_minus = 1.0 - h
                    correlate = np.correlate(h, h_1_minus, mode="full")
                    overlap = correlate * dx
                    return np.max(overlap)
                
                def constraint(h_vals):
                    return np.sum(h_vals) - n_points / 2
                
                bounds = [(0.0, 1.0) for _ in range(n_points)]
                cons = ({'type': 'eq', 'fun': constraint})
                
                result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'ftol': 1e-8, 'maxiter': 200})
                
                if result.success:
                    current_score = evaluate_erdos_solution(result.x, n_points)
                    if current_score < best_score:
                        best_score = current_score
                        best_h = result.x
                        best_n = n_points
            except:
                pass
    
    if best_h is None:
        best_h = initial_h_values(100)
        best_score = evaluate_erdos_solution(best_h, 100)
    
    return best_h, best_score, best_n
```
