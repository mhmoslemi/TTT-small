import numpy as np

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 256  # Reasonable size under 1k
    dx = 2.0 / n_points
    # Initial guess: uniform distribution
    initial_h = np.full(n_points, 0.5)
    # Ensure total area is 1
    initial_h = np.clip(initial_h, 0, 1)
    sum_h = np.sum(initial_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        initial_h += (n_points / 2 - sum_h) / n_points

    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5

    def objective(x):
        return evaluate_overlap(x)

    def constraint_total_area(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = (LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2),)

    # Define parameters for optimization
    max_iter = 100
    tol = 1e-5
    # Define a genetic algorithm approach
    def genetic_search(pop_size=32, generations=20, mutation_rate=0.1, crossover_rate=0.7):
        def fitness(individual):
            return evaluate_overlap(individual)

        def mutate(individual):
            mutation = np.random.normal(0, mutation_rate, size=n_points)
            return np.clip(individual + mutation, 0, 1)

        def crossover(parent1, parent2):
            mask = np.random.rand(n_points) < crossover_rate
            return np.where(mask, parent1, parent2)

        # Initialize population
        population = [np.random.uniform(0, 1, n_points) for _ in range(pop_size)]
        best_individual = population[0]
        best_fitness = fitness(best_individual)
        for _ in range(generations):
            fitness_scores = [fitness(ind) for ind in population]
            parents = np.array(population)[np.argsort(fitness_scores)[:int(pop_size / 2)]]
            next_gen = []
            for i in range(int(pop_size / 2)):
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = crossover(p1, p2)
                child = mutate(child)
                next_gen.append(child)
            population = next_gen
            current_best = np.argmin(fitness_scores)
            if fitness_scores[current_best] < best_fitness:
                best_fitness = fitness_scores[current_best]
                best_individual = population[current_best]
        return best_individual, best_fitness

    # Run genetic search
    best_solution, best_c5 = genetic_search(pop_size=32, generations=20, mutation_rate=0.1, crossover_rate=0.7)
    return (best_solution, best_c5, n_points)
