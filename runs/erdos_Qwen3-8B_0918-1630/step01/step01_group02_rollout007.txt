import numpy as np

from scipy.optimize import minimize
from scipy.optimize import Bounds

from functools import partial

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 512  # Stay under 1k samples
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5 for _ in range(n_points)])
    
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5

    def objective(x):
        return evaluate_overlap(x)

    def constraint_total_area(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = (LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2),)

    def generate_candidate():
        h = np.random.uniform(0, 1, n_points)
        total = np.sum(h)
        h = h / total * (n_points / 2)
        return h

    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    start_time = time.time()

    for _ in range(200):
        h = generate_candidate()
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'ftol': 1e-7, 'maxiter': 20})
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

        if time.time() - start_time > budget_s - 1:
            break

    return (best_h, best_c5, n_points)
