import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.optimize.optimize import OptimizeResult
from scipy.stats import qmc

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 128  # Reasonable size under 1k
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h)
    initial_h = np.clip(initial_h, 0, 1)
    sum_h = np.sum(initial_h)
    if np.abs(sum_h - n_points / 2) > 1e-6:
        initial_h += (n_points / 2 - sum_h) / n_points

    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5

    def objective(x):
        return evaluate_overlap(x)

    def constraint_total_area(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = (LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2),)

    def optimize_with_perturbation(h_start, max_iter=150, tol=1e-5):
        best_h = h_start.copy()
        best_c5 = evaluate_overlap(best_h)
        for _ in range(max_iter):
            perturbation = np.random.normal(0, 0.05, n_points)
            perturbed_h = np.clip(h_start + perturbation, 0, 1)
            result = minimize(objective, perturbed_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=tol, options={'disp': False, 'maxiter': 100})
            current_h = result.x
            current_c5 = evaluate_overlap(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
            print(f"Current best C5: {best_c5:.6f}")
        return best_h, best_c5

    # Hybrid approach with multiple starting points
    start_time = time.time()
    current_best_h = initial_h
    current_best_c5 = evaluate_overlap(current_best_h)
    for _ in range(5):
        print(f"Starting optimization round {_ + 1}")
        result = minimize(objective, current_best_h, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-5, options={'disp': False, 'ftol': 1e-5, 'maxiter': 150})
        current_best_h = result.x
        current_best_c5 = evaluate_overlap(current_best_h)
        if time.time() - start_time > budget_s - 10:
            break
    # Add Sobol sequence exploration
    sobol = qmc.Sobol(d=n_points)
    samples = sobol.random(n=5)
    for sample in samples:
        perturbed_h = np.interp(sample, np.linspace(0, 1, n_points), np.linspace(0, 1, n_points))
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) + (1 - perturbed_h) * (n_points / 2)
        result = minimize(objective, perturbed_h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-5, options={'disp': False, 'maxiter': 150})
        current_best_h = result.x
        current_best_c5 = evaluate_overlap(current_best_h)
    return (current_best_h, current_best_c5, n_points)
