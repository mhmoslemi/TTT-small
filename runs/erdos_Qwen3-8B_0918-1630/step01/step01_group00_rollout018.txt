import numpy as np
import time
from scipy.optimize import Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial guess: use the provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    scale = 1.0 / (sum_h * dx)
    initial_h_scaled = initial_h * scale
    initial_h_scaled = np.clip(initial_h_scaled, 0, 1)
    
    def compute_C5(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Ensure integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def fitness(h):
        return compute_C5(h)
    
    class Particle:
        def __init__(self, n_points):
            self.position = np.random.uniform(0, 1, n_points)
            self.velocity = np.random.uniform(-0.1, 0.1, n_points)
            self.pbest_pos = self.position.copy()
            self.pbest_fit = fitness(self.position)
            self.best_pos = self.position.copy()
            self.best_fit = self.pbest_fit
    
    class PSO:
        def __init__(self, n_points, n_particles, max_iter, seed):
            np.random.seed(seed)
            self.n_points = n_points
            self.n_particles = n_particles
            self.max_iter = max_iter
            self.particles = [Particle(n_points) for _ in range(n_particles)]
            self.gbest_pos = self.particles[0].position.copy()
            self.gbest_fit = self.particles[0].pbest_fit
            
            self.inertia = 0.9
            self.cognitive = 2.5
            self.social = 2.5
            self.mutation_rate = 0.1
            self.mutation_factor = 0.5
            self.time_budget = budget_s
            
        def update_velocity(self, particle, global_best):
            r1 = np.random.rand(self.n_points)
            r2 = np.random.rand(self.n_points)
            cognitive = self.cognitive * r1 * (particle.pbest_pos - particle.position)
            social = self.social * r2 * (global_best - particle.position)
            particle.velocity = self.inertia * particle.velocity + cognitive + social
        
        def update_position(self, particle):
            particle.position += particle.velocity
            particle.position = np.clip(particle.position, 0, 1)
        
        def evaluate(self):
            for particle in self.particles:
                fit = fitness(particle.position)
                if fit < particle.pbest_fit:
                    particle.pbest_pos = particle.position.copy()
                    particle.pbest_fit = fit
                if fit < self.gbest_fit:
                    self.gbest_pos = particle.position.copy()
                    self.gbest_fit = fit
        
        def mutate(self, particle):
            if np.random.rand() < self.mutation_rate:
                mutation = np.random.uniform(-self.mutation_factor, self.mutation_factor, self.n_points)
                particle.position += mutation
                particle.position = np.clip(particle.position, 0, 1)
        
        def run(self):
            start_time = time.time()
            for iter in range(self.max_iter):
                if time.time() - start_time > self.time_budget:
                    break
                for particle in self.particles:
                    self.update_velocity(particle, self.gbest_pos)
                    self.update_position(particle)
                    self.mutate(particle)
                self.evaluate()
            
            best_h = self.gbest_pos
            best_h = best_h * (1 / (np.sum(best_h) * dx))
            best_h = np.clip(best_h, 0, 1)
            return best_h, self.gbest_fit, self.n_points
    
    n_points = 100
    pso = PSO(n_points, n_particles=50, max_iter=200, seed=seed)
    best_h, best_c5, n = pso.run()
    
    return best_h, best_c5, n
