import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from deap import base, creator, algorithms, tools

def run(seed=42, budget_s=60):
    np.random.seed(seed)

    # Initial guess
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    scale = 1.0 / (sum_h * dx)
    initial_h_scaled = initial_h * scale
    initial_h_scaled = np.clip(initial_h_scaled, 0, 1)
    
    # Define population parameters
    POP_SIZE = 50
    MUT_PB = 0.1
    CXPB = 0.8
    MAX_GEN = 100
    TOURN_SIZE = 3

    # Create the fitness function
    def evaluate(individual):
        h = np.clip(individual, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Enforce integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5,

    # Create the DEAP evolutionary algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, 
                    toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initIterate, list, 
                    tools.initRepeat, creator.Individual, 
                    toolbox.attr_float, n=n_points)
    toolbox.register("evaluate", evaluate)
    toolbox.register("select", tools.selTournament, tournsize=TOURN_SIZE)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, 
                    indpb=0.1)
    toolbox.register("best", tools.selBest, k=1)

    # Initialize population with initial solution
    pop = [creator.Individual(initial_h_scaled)]

    # Run the algorithm
    start_time = time.time()

    for gen in range(MAX_GEN):
        # Evaluate the current population
        offspring = []
        for ind in pop:
            if np.isnan(ind.fitness.values[0]):
                ind.fitness.values = (np.inf,)
            else:
                ind.fitness.values = (ind.fitness.values[0],)

        # Generate offspring
        offspring = [toolbox.clone(ind) for ind in pop]
        for child1, child2 in zip(offspring[::2], offspring[1::2]):
            if np.random.random() < CXPB:
                toolbox.mate(child1, child2)
                for child in (child1, child2):
                    if np.random.random() < MUT_PB:
                        toolbox.mutate(child)
            del child1.fitness, child2.fitness

        # Evaluate offspring
        for ind in offspring:
            if np.isnan(ind.fitness.values[0]):
                ind.fitness.values = (np.inf,)
            else:
                ind.fitness.values = (ind.fitness.values[0],)

        # Select the next population
        pop = tools.selBest(pop + offspring, k=POP_SIZE)

        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break

    # Extract the best solution
    best = tools.selBest(pop, k=1)[0]
    best_h = np.clip(best, 0, 1)
    best_h = best_h * (1 / (np.sum(best_h) * dx))  # Enforce integral constraint

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx

    return best_h, c5_bound, len(best_h)
