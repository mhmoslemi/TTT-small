```python
import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    max_n_points = 256
    n_points = max_n_points
    dx = 2.0 / n_points
    n_iter = 20
    best_c5 = 1.0
    best_h = np.zeros(n_points)

    # Predefined initial patterns for guided search
    initial_patterns = [
        np.zeros(n_points),  # all zero
        np.ones(n_points),   # all ones
        np.linspace(0, 1, n_points),  # linear
        np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 5), [0, 1, 0, 1, 0]),  # 5-point discrete
        np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 3), [0, 1, 0]),  # 3-point discrete
        np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 10), np.concatenate([np.zeros(5), np.ones(5)])),  # 10-point discrete
    ]

    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        conv = correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        return c5

    def constraint_total_area(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    cons = (LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2),)

    for pattern in initial_patterns:
        # Resample pattern to match n_points
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(pattern)), pattern)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if np.abs(sum_h - n_points / 2) > 1e-6:
            h += (n_points / 2 - sum_h) / n_points
        h = np.clip(h, 0, 1)

        # Local optimization
        result = minimize(
            lambda x: evaluate_overlap(x),
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': 50, 'ftol': 1e-6}
        )

        if result.success:
            current_h = result.x
            current_c5 = evaluate_overlap(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

        # Early stopping if time is up
        elapsed = time.time() - time.time()
        if elapsed >= budget_s - 1:
            break

    return (best_h, best_c5, n_points)
```
