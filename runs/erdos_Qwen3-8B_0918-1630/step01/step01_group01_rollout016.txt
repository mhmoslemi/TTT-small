```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points_options = [100, 150, 200]
    best_c5 = float('inf')
    best_h = initial_h_values(100)
    
    for n_points in n_points_options:
        dx = 2.0 / n_points
        h_initial = initial_h_values(n_points)
        
        def objective(h):
            h = np.array(h)
            h_1_minus = 1.0 - h
            correlate = np.correlate(h, h_1_minus, mode="full")
            overlap = correlate * dx
            return np.max(overlap)
        
        def constraint(h):
            return np.sum(h) - n_points / 2
        
        h0 = h_initial
        cons = ({'type': 'eq', 'fun': constraint})
        bounds = Bounds(0.0, 1.0)
        
        start_time = time.time()
        result = minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, options={'ftol': 1e-10, 'maxiter': 300})
        elapsed = time.time() - start_time
        
        if elapsed > budget_s:
            continue
        
        current_c5 = evaluate_erdos_solution(result.x, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = result.x
    
    return best_h, best_c5, len(best_h)
```
