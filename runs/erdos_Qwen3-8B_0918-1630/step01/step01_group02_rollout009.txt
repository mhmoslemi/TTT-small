import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import random
import warnings

#--------------------------------------------------
# Objective function and constraints
#--------------------------------------------------

def calculate_overlap(h, dx):
    conv = np.correlate(h, 1 - h, mode="full")
    return np.max(conv) * dx

def constraint_sum(h):
    return np.sum(h) - len(h) / 2

def objective_overlap(h, dx):
    return calculate_overlap(h, dx)

#--------------------------------------------------
# Genetic Algorithm Setup
#--------------------------------------------------

def genetic_algorithm(initial_pop, pop_size=100, generations=30, dp=0.1, cm=0.8, mut_rate=0.1, fitness_func, dx=1.0):
    population = [individual for individual in initial_pop]
    best = min(population, key=fitness_func)
    for _ in range(generations):
        fitness = np.array([fitness_func(ind) for ind in population])
        parents = np.random.choice(population, size=pop_size, p=fitness / fitness.sum())
        children = []
        for _ in range(pop_size // 2):
            p1 = random.choice(parents)
            p2 = random.choice(parents)
            child = np.copy(p1)
            for i in range(len(child)):
                if random.random() < cp:
                    child[i] = p1[i] * (1 - dp) + p2[i] * dp
                if random.random() < mut_rate:
                    child[i] = np.clip(child[i] + np.random.normal(0, 0.05), 0, 1)
            children.append(child)
        population = children
        best = min(population, key=fitness_func)
    return best
