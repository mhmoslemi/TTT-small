```python
import numpy as np
import time
import random
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import minimize_scalar
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel as C

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h_initial[i] = 1.0
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def perturb_solution(h, perturbation_strength=0.1):
        h_perturbed = h.copy()
        for i in range(n_points):
            if np.random.rand() < perturbation_strength:
                h_perturbed[i] = np.clip(h_perturbed[i] + np.random.normal(0, 0.1), 0, 1)
        return h_perturbed
    
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Simulated annealing-like approach with multiple starting points
        for _ in range(5):
            h_start = perturb_solution(best_h)
            h_start = np.clip(h_start, 0, 1)
            
            def objective_with_bounds(h):
                h = np.array(h)
                h_1_minus = 1.0 - h
                correlate = np.correlate(h, h_1_minus, mode="full")
                overlap = correlate * dx
                return np.max(overlap)
            
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            
            result = minimize(objective_with_bounds, h_start, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint}, options={'ftol': 1e-8, 'maxiter': 100})
            
            if result.success:
                h_candidate = result.x
                c5_candidate = evaluate_erdos_solution(h_candidate, n_points)
                if c5_candidate < best_c5:
                    best_h = h_candidate
                    best_c5 = c5_candidate
                print(f"New best C5: {best_c5}")
        
        # Try increasing n_points if budget allows
        if time.time() - start_time < budget_s - 5:
            n_points = 150
            dx = 2.0 / n_points
            best_h = np.clip(best_h, 0, 1)
            best_h = np.resize(best_h, n_points)
            best_h = np.pad(best_h, (0, n_points - len(best_h)))
            best_h = np.repeat(best_h, 2)
            best_h = best_h[:n_points]
            best_c5 = evaluate_erdos_solution(best_h, n_points)
            print(f"Switched to n_points = {n_points}")
    
    return best_h, best_c5, n_points
```
