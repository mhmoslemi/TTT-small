def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    h_initial = initial_h_values(n_points)
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def optimize_h(h_initial, n_points):
        def objective_func(h):
            return evaluate_erdos_solution(h, n_points)
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = differential_evolution(objective_func, bounds, strategy='best1bin', maxiter=100, popsize=30, tol=1e-8, mutation=(0.5, 1), recombination=0.9, polish=True, disp=False, workers=1)
        return result.x
    
    def evaluate_with_rounding(h_values, n_points):
        h = np.array(h_values, dtype=np.float64)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * (2.0 / n_points)
        return np.max(overlap)
    
    best_solution = h_initial
    best_c5 = evaluate_erdos_solution(h_initial, n_points)
    best_n_points = n_points
    
    for trial in range(5):
        start_time = time.time()
        h = optimize_h(h_initial, n_points)
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            return best_solution, best_c5, best_n_points
        
        current_c5 = evaluate_with_rounding(h, n_points)
        if current_c5 < best_c5:
            best_solution = h
            best_c5 = current_c5
            best_n_points = n_points
    
    return best_solution, best_c5, best_n_points
