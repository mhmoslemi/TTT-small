    def compute_c5(h):
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * dx
        h = h / integral
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def objective(h):
        return compute_c5(h)

    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bayesian Optimization
    def bayesian_optimization():
        samples = 100
        grid = np.linspace(0, 1, samples)
        X = np.random.rand(samples, n_points)
        X = np.clip(X, 0, 1)
        X = X * (n_points / 2)
        y = np.array([compute_c5(X[i]) for i in range(samples)])
        kernel = ConstantKernel(1.0) * RBF(length_scale=1.0, length_scale_bounds=(0.01, 10.0))
        gp = GaussianProcessRegressor(kernel=kernel, n_restarts_optimizer=10)
        gp.fit(X, y)
        best_idx = np.argmin(y)
        best_h = X[best_idx]
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2)
        return best_h

    def make_initial_guesses():
        guesses = []
        h_base = initial_h.copy()
        guesses.append(h_base)
        h_shifted = np.roll(h_base, 1)
        guesses.append(h_shifted)
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points / 2)
        guesses.append(h_random)
        return guesses

    # Try multiple initial guesses
    best_h = None
    best_c5 = float('inf')
    guesses = make_initial_guesses()
    for i, guess in enumerate(guesses):
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, x0=guess, method='L-BFGS-B', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}])
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < best_c5:

                best_h = h_candidate

    # Try Bayesian optimization
    best_bayesian = bayesian_optimization()
    best_bayesian = np.clip(best_bayesian, 0, 1)
    best_bayesian = best_bayesian * (n_points / 2)
    c5_bayesian = compute_c5(best_bayesian)
    if c5_bayesian < best_c5:
        best_c5 = c5_bayesian
        best_h = best_bayesian
