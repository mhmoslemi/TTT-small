import numpy as np
import time
from skopt import BayesSearchSpace, BayesianOptimization
from skopt.utils import use_vectorized_obj
from scipy.signal import correlate
from scipy.optimize import Bounds

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    def objective(x):
        h = np.clip(x, 0, 1)
        sum_h = np.sum(h)
        norm = sum_h / 2.0 if sum_h != 0 else 1.0
        h = h / norm
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / len(h))
        return c5
    
    def evaluate_erdos_solution(x):
        return objective(x)
    
    n_points = 1000
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    initial_h[:int(n_points / 2)] = 1.0
    initial_h = initial_h + np.random.normal(0, 0.1, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (2.0 / n_points) / np.sum(initial_h)
    
    def surrogate(x):
        return evaluate_erdos_solution(x)
    
    space = BayesSearchSpace()
    space.continuous({'h': (0, 1)})
    
    bo = BayesianOptimization(
        surrogate,
        space,
        n_points=n_points,
        n_init=10,
        n_iter=200,
        maxiter=200,
        acq_func='ei',
        n_jobs=-1
    )
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        bo.fit()
        best_model = bo.get_model()
        best_params = best_model.get_params()
        best_point = best_model.get_best()
        best_result = evaluate_erdos_solution(best_point)
        print(f"Current best C5: {best_result}")
    
    best_model = bo.get_model()
    best_point = best_model.get_best()
    best_result = evaluate_erdos_solution(best_point)
    h_candidate = best_point
    c5_candidate = best_result
    return h_candidate, c5_candidate, len(h_candidate)
