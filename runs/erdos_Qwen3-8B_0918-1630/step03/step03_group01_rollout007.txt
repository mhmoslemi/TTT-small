```python
import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize.lbfgsb import fmin_l_bfgs_b
from itertools import product
from multiprocessing import Pool

def evaluate_overlap(h, n_points):
    dx = 2.0 / n_points
    h_scaled = h / (np.sum(h) * dx)
    corr = correlate(h_scaled, 1 - h_scaled, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = None
    best_n = 200

    def optimize(n_points):
        dx = 2.0 / n_points
        bounds = [(0, 1) for _ in range(n_points)]
        h0 = np.random.rand(n_points)
        h0 = h0 * (1 / (np.sum(h0) * dx))
        h0 = np.clip(h0, 0, 1)

        def objective(h):
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            scale = 1.0 / (sum_h * dx) if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = correlate(h_scaled, 1 - h_scaled, mode='full')
            return np.max(corr) * dx

        result = minimize(objective, x0=h0, method='L-BFGS-B', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}])
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
            c5_candidate = evaluate_overlap(h_candidate, n_points)
            return h_candidate, c5_candidate
        else:
            return h0, float('inf')

    def worker(n_points):
        nonlocal best_c5, best_h, best_n
        t0 = time.time()
        h, c5 = optimize(n_points)
        elapsed = time.time() - t0
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n_points
        return elapsed

    n_values = [200, 250, 300, 350, 400]
    for n in n_values:
        if time.time() - time.time() > budget_s:
            break
        worker(n)

    return best_h, best_c5, best_n
```
