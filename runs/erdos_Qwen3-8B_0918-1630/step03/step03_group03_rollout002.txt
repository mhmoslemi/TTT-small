import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Start with a basic initial guess
    initial_h_values = np.random.uniform(0, 1, 200)
    initial_h_values[initial_h_values > 0.5] = 1.0
    initial_h_values[initial_h_values <= 0.5] = 0.0
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_initial = np.sum(initial_h_values)
    scale = 1.0 / (sum_initial * dx)
    h_initial = initial_h_values * scale
    h_initial = np.clip(h_initial, 0, 1)
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        h = h * (1 / (sum_h * dx))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use Adam optimizer for better convergence
    from scipy.optimize import minimize
    result = minimize(objective, x0=h_initial, method='L-BFGS-B', bounds=bounds)
    
    if result.success:
        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        sum_h = np.sum(h_candidate)
        if sum_h == 0:
            c5_candidate = float('inf')
        else:
            h_candidate = h_candidate * (1 / (sum_h * dx))
            corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
    
    # Perform some random perturbations
    for _ in range(10):
        perturb = np.random.uniform(-0.2, 0.2, n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        sum_h = np.sum(h_perturb)
        if sum_h == 0:
            c5_perturb = float('inf')
        else:
            h_perturb = h_perturb * (1 / (sum_h * dx))
            corr = correlate(h_perturb, 1 - h_perturb, mode='full')
            c5_perturb = np.max(corr) * dx
        if c5_perturb < best_c5:
            best_c5 = c5_perturb
            best_h = h_perturb
    
    return best_h, best_c5, len(best_h)
