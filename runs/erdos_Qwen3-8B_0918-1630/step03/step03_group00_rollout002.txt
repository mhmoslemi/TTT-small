import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.optimize import Bounds

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    n_points = 400
    dx = 2.0 / n_points
    max_evals = 200
    
    # Initial samples
    initial_patterns = [
        np.concatenate([np.linspace(0, 1, n_points // 2), np.linspace(1, 0, n_points // 2)]),
        np.sin(np.linspace(0, np.pi, n_points)) ** 2,
        np.random.rand(n_points) * 2 - 1,
        np.concatenate([np.zeros(n_points // 2), np.ones(n_points // 2)]),
        np.linspace(0, 1, n_points),
    ]
    
    # Evaluate initial patterns
    best_c5 = float('inf')
    best_h = None
    best_n = n_points
    
    for interval in initial_patterns:
        interval = np.clip(interval, 0, 1)
        sum_h = np.sum(interval)
        scale = 1.0 / (sum_h * dx)
        h = interval * scale
        h = np.clip(h, 0, 1)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n = n_points
    
    # Define a function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Define objective function to minimize
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        return compute_c5(h)
    
    # Define bounds for all h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint: sum(h) * dx = 1
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Optimization with SLSQP
    result = minimize(objective, x0=best_h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': constraint_func}])
    
    if result.success:
        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
    
    return best_h, best_c5, best_n
