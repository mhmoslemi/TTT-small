```python
import numpy as np
import time
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate
from scipy.optimize import COBYLA, SLSQP

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial grid using Chebyshev nodes
    def chebyshev_nodes(n):
        return np.cos(np.pi * (2 * np.arange(n) + 1) / (2 * n))
    
    initial_h = chebyshev_nodes(200)
    initial_h = np.clip(initial_h, 0, 1)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to evaluate C5
    def evaluate_C5(h):
        h = np.clip(h, 0, 1)
        scaled = h * (1 / (np.sum(h) * dx))
        corr = correlate(scaled, 1 - scaled, mode='full')
        return np.max(corr) * dx
    
    # Function to compute constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Meta-search: evaluate multiple initial points
    def meta_search(initial_guesses, max_iter=100):
        best_c5 = float('inf')
        best_h = None
        for guess in initial_guesses:
            h = np.array(guess)
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))
            result = minimize(evaluate_C5, x0=h, method=COBYLA, bounds=Bounds(0, 1),
                              constraints=[{'type': 'eq', 'fun': constraint}])
            if result.success:
                c5 = result.fun
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = result.x
            else:
                print("Optimization failed for this initial guess.")
        return best_h, best_c5
    
    # Generate multiple initial guesses
    initial_guesses = [initial_h]
    for _ in range(3):
        perturb = np.random.rand(n_points) * 0.2 - 0.1
        initial_guesses.append(np.clip(initial_h + perturb, 0, 1))
    
    best_h, best_c5 = meta_search(initial_guesses)
    
    # Final refinement with SLSQP
    result = minimize(evaluate_C5, x0=best_h, method=SLSQP, bounds=Bounds(0, 1),
                      constraints=[{'type': 'eq', 'fun': constraint}])
    
    if result.success:
        best_c5 = result.fun
        best_h = result.x
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    
    return best_h, best_c5, len(best_h)
```
