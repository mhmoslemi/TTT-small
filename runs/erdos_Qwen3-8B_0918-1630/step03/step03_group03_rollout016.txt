import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import LinearConstraint
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Starting from a different construction to avoid local optima
    n_points = 400
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Initial guess: piecewise constant function with non-uniform distribution
    initial_h = np.zeros(n_points)
    initial_h[0:100] = 1.0
    initial_h[100:200] = 0.5
    initial_h[200:300] = 0.2
    initial_h[300:] = 0.1
    initial_h = np.clip(initial_h, 0, 1)
    
    # Scale to meet the integral constraint
    initial_sum = np.sum(initial_h) * dx
    scale = 1.0 / initial_sum
    h_initial = initial_h * scale
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    def objective(h):
        h = np.clip(h, 0, 1)
        total_integral = np.sum(h) * dx
        h_scaled = h / total_integral
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = LinearConstraint(np.ones(n_points), -np.inf, n_points / 2)
    
    # Use differential evolution for global optimization
    result = differential_evolution(
        objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=100,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False
    )
    
    if result.success:
        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        total_integral = np.sum(h_candidate) * dx
        h_candidate_scaled = h_candidate / total_integral
        corr = correlate(h_candidate_scaled, 1 - h_candidate_scaled, mode='full')
        c5_candidate = np.max(corr) * dx
        best_c5 = c5_candidate
        best_h = h_candidate_scaled
    
    # Try a few random starting points for fine-tuning
    for _ in range(5):
        perturb = np.random.uniform(-0.2, 0.2, n_points)
        h_perturb = np.clip(best_h + perturb, 0, 1)
        total_integral = np.sum(h_perturb) * dx
        h_perturb_scaled = h_perturb / total_integral
        corr = correlate(h_perturb_scaled, 1 - h_perturb_scaled, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturb_scaled
    
    return best_h, best_c5, n_points
