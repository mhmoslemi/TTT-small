import numpy as np
import time

from scipy.optimize import minimize

from scipy.optimize import minimize

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    n_points = 500
    dx = 2.0 / n_points
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        h = h / total
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def fitness(h):
        return compute_c5(h)
    
    def initial_h_values(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 10 == 0:
                h[i] = 1
        return h
    
    initial_h = initial_h_values(n_points)
    initial_h = np.array(initial_h)
    
    def create_population(size):
        return np.random.uniform(0, 1, (size, n_points))
    
    def crossover(parent1, parent2):
        child = np.zeros_like(parent1)
        for i in range(n_points):
            if np.random.rand() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return child
    
    def mutate(individual, mutation_rate=0.05):
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                individual[i] = np.random.uniform(0, 1)
        return individual
    
    def evaluate(individual):
        return -fitness(individual)
    
    def run_genetic_algorithm():
        population_size = 20
        mutation_rate = 0.15
        generations = 10
        population = create_population(population_size)
        for gen in range(generations):
            fitnesses = np.array([evaluate(ind) for ind in population])
            best_index = np.argmax(fitnesses)
            best = population[best_index]
            population = [mutate(ind) for ind in population]
            population = [crossover(best, ind) for ind in population]
        
        return best
    
    best_h = run_genetic_algorithm()
    
    def refine_solution(h):
        def objective(x):
            return compute_c5(x)
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, x0=h, method='L-BFGS-B', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}])
        if result.success:
            return result.x
        return h
    
    best_h = refine_solution(best_h)
    
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    corr = correlate(best_h, 1 - best_h, mode='full')
    c5 = np.max(corr) * dx
    
    return best_h, c5, n_points
