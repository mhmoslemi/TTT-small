import numpy as np
import time
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate
from scipy.optimize import COBYLA, SLSQP

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial configuration with a different construction for diversity
    n_points = 200
    initial_h = np.zeros(n_points)
    initial_h[::2] = 1.0
    initial_h[1::2] = 0.5
    initial_h = initial_h[:n_points]
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * (2.0 / n_points)))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Define bounds for all h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective function to minimize the maximum overlap
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * (2.0 / n_points)))
        return compute_c5(h)
    
    # Constraint: integral constraint
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization with COBYLA for better handling of constraints
    result = minimize(objective, x0=initial_h, method='COBYLA', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': constraint}])
    
    if result.success:
        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (1 / (np.sum(h_candidate) * (2.0 / n_points)))
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr) * (2.0 / n_points)
        
        return h_candidate, c5_candidate, len(h_candidate)
    
    # If optimization fails, try with a smaller n_points and another initial guess
    n_points = 100
    initial_h = np.zeros(n_points)
    initial_h[::3] = 1.0
    initial_h[1::3] = 0.75
    initial_h[2::3] = 0.5
    initial_h = initial_h[:n_points]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(objective, x0=initial_h, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': constraint}])
    
    if result.success:
        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (1 / (np.sum(h_candidate) * (2.0 / n_points)))
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr) * (2.0 / n_points)
        
        return h_candidate, c5_candidate, len(h_candidate)
    
    # Fallback to initial_h if all else fails
    h_candidate = initial_h
    h_candidate = np.clip(h_candidate, 0, 1)
    h_candidate = h_candidate * (1 / (np.sum(h_candidate) * (2.0 / n_points)))
    corr = correlate(h_candidate, 1 - h_candidate, mode='full')
    c5_candidate = np.max(corr) * (2.0 / n_points)
    
    return h_candidate, c5_candidate, len(h_candidate)
