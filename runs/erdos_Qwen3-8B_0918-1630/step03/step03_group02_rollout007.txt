import numpy as np
import time
from scipy.optimize import Bounds
from scipy.signal import correlate
from scipy.optimize import minimize
from deap import base, creator, algorithms, tools

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial configuration
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Create a GA toolbox
    creator.create("FitnessMin", float, weight=-1.0)
    creator.create("Individual", list, fitness=creator.FitnessMin)

   toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    # Fitness evaluation
    def eval_func(individual):
        h = np.array(individual)
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return (c5,)
    
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxBlend, alpha=0.1)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.05)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Genetic algorithm parameters
    n_gen = 50
    pop_size = 50
    cxpb = 0.8
    mutpb = 0.2
    n_offsprings = int(np.floor(pop_size * cxpb))
    
    # Start with initial population
    pop = toolbox.population(n=pop_size)
    best_ind = None
    best_score = float('inf')
    
    start_time = time.time()
    
    # Run genetic algorithm
    for gen in range(n_gen):
        offspring = algorithms.varAnd(pop, toolbox, cxpb=cxpb, mutpb=mutpb)
        offspring = list(offspring)
        fits = list(map(toolbox.evaluate, offspring))
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = toolbox.select(pop + offspring, pop_size)
        
        # Update best individual
        for ind in pop:
            if not ind.fitness.worse(best_score):
                best_score = ind.fitness.values[0]
                best_ind = ind
        
        # Check time budget
        if time.time() - start_time > budget_s - 5:
            break
    
    # Refine best individual using L-BFGS-B
    if best_ind is not None:
        h_best = np.array(best_ind)
        h_best = np.clip(h_best, 0, 1)
        h_best = h_best * (1 / (np.sum(h_best) * dx))
        
        # Define objective function for refinement
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Constraints
        def constraint(h):
            h = np.clip(h, 0, 1)
            return np.sum(h) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, x0=h_best, method='L-BFGS-B', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': constraint}])
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
            corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            return h_candidate, c5_candidate, len(h_candidate)
        else:
            h_candidate = h_best
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
            corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            return h_candidate, c5_candidate, len(h_candidate)
    else:
        h_candidate = initial_h
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr) * dx
        return h_candidate, c5_candidate, len(h_candidate)
