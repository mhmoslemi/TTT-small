import numpy as np
import time
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate
from scipy.optimize import minimize, Bounds
from scipy.optimize.cobyla import minimize_cobyla
from scipy.sparse import diags
from cvxpy import Variable, Problem, minimize as cvx_minimize, constraints, Max, Min, norm

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Predefined initial constructions to explore
    initial_h_values = np.repeat([1, 0], 100)
    def h_uniform(x):
        return np.floor(x / (2.0 / 400)) % 2 == 0
    def h_symmetric(x):
        return (np.floor(x * 2.5) % 2) == 0
    def h_random(x):
        return np.random.rand(len(x)) > 0.5
    
    # Generate initial candidates
    candidates = []
    for i in range(3):
        if i == 0:
            h = initial_h_values
        elif i == 1:
            h = (np.floor(np.linspace(0, 2, 400) * 2.5) % 2).astype(int)
        elif i == 2:
            h = np.random.rand(400) > 0.5
        candidates.append(h)
    
    best_h = candidates[0]
    best_c5 = float('inf')
    best_n = 400
    
    for h_initial in candidates:
        h_initial = np.clip(h_initial, 0, 1)
        n_points = len(h_initial)
        dx = 2.0 / n_points
        
        # Ensure the integral constraint is satisfied
        sum_h = np.sum(h_initial)
        scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h_initial = np.clip(h_initial * scale, 0, 1)
        
        # Define objective function
        def objective(params):
            h = np.clip(params, 0, 1)
            sum_h = np.sum(h)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h_scaled = h * scale
            corr = correlate(h_scaled, 1 - h_scaled, mode='full')
            return np.max(corr) * dx
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use COBYLA for constrained optimization
        result = minimize(objective, x0=h_initial, bounds=bounds, constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}])
        
        if result.success:
            h_candidate = np.clip(result.x, 0, 1)
            sum_h_candidate = np.sum(h_candidate)
            scale = (n_points / 2) / sum_h_candidate if sum_h_candidate != 0 else 1.0
            h_scaled = h_candidate * scale
            corr = correlate(h_scaled, 1 - h_scaled, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate
                best_n = n_points
        
        # Local search with random perturbations
        for _ in range(2):
            perturb = np.random.rand(len(h_candidate)) * 0.2 - 0.1
            h_candidate = np.clip(h_candidate + perturb, 0, 1)
            sum_h_candidate = np.sum(h_candidate)
            scale = (n_points / 2) / sum_h_candidate if sum_h_candidate != 0 else 1.0
            h_scaled = h_candidate * scale
            corr = correlate(h_scaled, 1 - h_scaled, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n = n_points
    
    return best_h, best_c5, best_n
