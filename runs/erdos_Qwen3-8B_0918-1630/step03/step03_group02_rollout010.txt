```python
import numpy as np
import time
from scipy.optimize import Bounds
from scipy.signal import correlate
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    n_points = 600
    dx = 2.0 / n_points
    
    initial_guesses = []
    # 1. Uniform distribution
    initial_guesses.append(np.full(n_points, 1.0 / n_points))
    # 2. Alternating pattern
    initial_guesses.append(np.array([1.0 if i % 2 == 0 else 0.0 for i in range(n_points)]))
    # 3. Sparse clusters
    initial_guesses.append(np.zeros(n_points))
    for i in range(5):
        idx = np.random.choice(n_points, 20, replace=False)
        initial_guesses[-1][idx] = 1.0
    # 4. Random step function
    initial_guesses.append(np.random.rand(n_points))
    
    best_h = None
    best_c5 = float('inf')
    
    elapsed_time = 0.0
    start_time = time.time()
    
    for guess in initial_guesses:
        # Define function to compute C5
        def compute_c5(h):
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Objective function with integral constraint
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))
            return compute_c5(h)
        
        # Bounds for all h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the constraint: sum(h) * dx = 1
        constraint = LinearConstraint(np.ones(n_points), 1.0, 1.0)
        
        # Optimize
        result = minimize(objective, x0=guess, method='L-BFGS-B', bounds=bounds,
                          constraints=[constraint], tol=1e-8, options={'maxiter': 1000, 'ftol': 1e-8})
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
            corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                elapsed_time = time.time() - start_time
                if elapsed_time > budget_s - 1:
                    break
        
    # If no better solution found, fallback to best guess
    if best_h is None:
        best_h = initial_guesses[0]
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (1 / (np.sum(best_h) * dx))
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return best_h, best_c5, n_points
```
