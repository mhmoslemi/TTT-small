import numpy as np
import time
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate
from scipy.optimize import Bounds, minimize
from cma import CMAES

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial configurations for exploration
    initial_h_values = np.load("initial_h_values.npy")
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    scale = 1.0 / (sum_h * dx)
    h_initial = initial_h_values * scale
    h_initial = np.clip(h_initial, 0, 1)
    
    best_c5 = float('inf')
    best_h = h_initial
    best_n = n_points
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Define the objective function for optimization
    def objective(x, dx):
        h = np.clip(x, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Define the constraint function for the integral
    def constraint_func(x, dx):
        return np.sum(x) - n_points / 2
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def optimize_h(h_initial, dx, n_points):
        result = minimize(lambda x: objective(x, dx), x0=h_initial, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: constraint_func(x, dx)}])
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
            c5_candidate = compute_c5(h_candidate)
            return h_candidate, c5_candidate
        return h_initial, float('inf')
    
    # Swarm-based search with CMA-ES
    def cma_es_search(n_points, dx):
        def fitness_func(x):
            h = np.clip(x, 0, 1)
            h = h * (1 / (np.sum(h) * dx))
            corr = correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx
        
        # Ensure bounds are in [0, 1]
        bounds = [0, 1]
        # Use CMA-ES with appropriate parameters
        opt = CMAES(obj_func=fitness_func, bounds=bounds, sigma=0.1, popsize=20, maxiter=100)
        best = opt.run()
        h_candidate = best[0]
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
        c5_candidate = compute_c5(h_candidate)
        return h_candidate, c5_candidate
    
    # Try initial_h_values and optimize
    h_initial = initial_h_values
    dx_initial = 2.0 / len(h_initial)
    sum_h = np.sum(h_initial)
    scale = 1.0 / (sum_h * dx_initial)
    h_initial_scaled = h_initial * scale
    h_initial_scaled = np.clip(h_initial_scaled, 0, 1)
    h_candidate, c5_candidate = optimize_h(h_initial_scaled, dx_initial, len(h_initial))
    if c5_candidate < best_c5:
        best_c5 = c5_candidate
        best_h = h_candidate
        best_n = len(h_initial)
    
    # Try structured patterns
    patterns = [
        # Checkerboard pattern
        np.concatenate([np.zeros(100), np.ones(100)]),
        # Sinusoidal pattern
        np.sin(np.linspace(0, np.pi, 200)) ** 2,
        # Linear rise and fall
        np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)]),
        # Random pattern
        np.random.rand(200)
    ]
    for pattern in patterns:
        n_points = len(pattern)
        if n_points > 200:
            continue
        dx = 2.0 / n_points
        sum_h = np.sum(pattern)
        scale = 1.0 / (sum_h * dx)
        h_initial = pattern * scale
        h_initial = np.clip(h_initial, 0, 1)
        h_candidate, c5_candidate = optimize_h(h_initial, dx, n_points)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
            best_n = n_points
    
    # Try CMA-ES search
    try:
        h_candidate, c5_candidate = cma_es_search(200, 2.0 / 200)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
            best_n = 200
    except:
        pass
    
    # Early stopping if time is up
    elapsed = time.time() - time.time()
    if elapsed > budget_s:
        return best_h, best_c5, best_n
    
    return best_h, best_c5, best_n
