from scipy.optimize import LinearConstraint

        h = np.clip(h, 0, 1)
        h = h * (1 / np.sum(h))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    bounds = [(0, 1) for _ in range(n_points)]

    # Use L-BFGS-B for faster convergence
    result = minimize(objective, x0=h_initial, method='L-BFGS-B', bounds=bounds, constraints=constraint)

        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (1 / np.sum(h_candidate))
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr) * dx

        best_h = h_candidate
    
    # Try random perturbations of the best solution to avoid local optima

        h_perturb = h_perturb * (1 / np.sum(h_perturb))
        corr = correlate(h_perturb, 1 - h_perturb, mode='full')
