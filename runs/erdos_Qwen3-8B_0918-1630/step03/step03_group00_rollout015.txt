import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial construction from provided variable
    initial_h_values = np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    scale = 1.0 / (sum_h * dx)
    h_initial = initial_h_values * scale
    h_initial = np.clip(h_initial, 0, 1)
    
    best_c5 = float('inf')
    best_h = h_initial
    best_n = n_points
    
    # Precomputed patterns for exploration
    patterns = [
        np.zeros(n_points),  # Constant 0
        np.ones(n_points),   # Constant 1
        np.linspace(0, 1, n_points),  # Linear rise
        np.linspace(1, 0, n_points),  # Linear fall
        np.sin(np.linspace(0, np.pi, n_points)) ** 2,  # Sinusoidal pattern
        np.random.rand(n_points) * 2 - 1,  # Random pattern
        np.concatenate([np.zeros(50), np.ones(50)]),  # Step pattern
        np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)])  # Initial pattern
    ]
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        return compute_c5(h)
    
    # Optimization function with constraints
    def optimize_from_initial(x0, n_points, dx):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        result = minimize(objective, x0=x0, bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 100})
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
            return h_candidate, compute_c5(h_candidate)
        return None, float('inf')
    
    start_time = time.time()
    for pattern in patterns:
        n = len(pattern)
        if n > 1000:
            continue
        
        dx = 2.0 / n
        sum_h = np.sum(pattern)
        scale = 1.0 / (sum_h * dx)
        h_initial = pattern * scale
        h_initial = np.clip(h_initial, 0, 1)
        
        h_candidate, c5_candidate = optimize_from_initial(h_initial, n, dx)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
            best_n = n
    
    # Final refinement with a few more patterns
    additional_patterns = [
        np.concatenate([np.linspace(0, 1, 50), np.linspace(1, 0, 50)]),
        np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)]),
        np.array([0.5 if i % 2 == 0 else 0.5 for i in range(n_points)]),
        np.array([0.5 if i % 3 == 0 else 0.5 for i in range(n_points)]),
    ]
    
    for pattern in additional_patterns:
        n = len(pattern)
        if n > 1000:
            continue
        
        dx = 2.0 / n
        sum_h = np.sum(pattern)
        scale = 1.0 / (sum_h * dx)
        h_initial = pattern * scale
        h_initial = np.clip(h_initial, 0, 1)
        
        h_candidate, c5_candidate = optimize_from_initial(h_initial, n, dx)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
            best_n = n
    
    # Ensure the best_h is properly scaled and normalized
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (1 / (np.sum(best_h) * (2.0 / best_n)))
    
    # Check remaining time and force early return if necessary
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return best_h, best_c5, best_n
    
    return best_h, best_c5, best_n
