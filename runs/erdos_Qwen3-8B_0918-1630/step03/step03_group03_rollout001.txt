import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import LinearConstraint
from functools import partial

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Define an initial random step function with uniform distribution
    n_points = 100
    initial_h = np.random.uniform(0, 1, n_points)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h)
    initial_h = initial_h * (1 / initial_sum) * dx
    initial_h = np.clip(initial_h, 0, 1)
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    best_h_values = initial_h.copy()
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = LinearConstraint(np.ones(n_points), -np.inf, n_points / 2)
    
    # Initial optimization with SLSQP
    result = minimize(objective, x0=initial_h, method='SLSQP', bounds=bounds, constraints=constraint)
    
    if result.success:
        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr) * dx
        best_c5 = c5_candidate
        best_h_values = h_candidate
    
    # Try different perturbations and refinements
    for _ in range(10):
        perturb = np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(best_h_values + perturb, 0, 1)
        h_perturb = h_perturb * (1 / (np.sum(h_perturb) * dx))
        corr = correlate(h_perturb, 1 - h_perturb, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h_values = h_perturb
    
    # Try a second optimization pass with fine-tuning
    result = minimize(objective, x0=best_h_values, method='SLSQP', bounds=bounds, constraints=constraint)
    
    if result.success:
        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr) * dx
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h_values = h_candidate
    
    return best_h_values, best_c5, n_points
