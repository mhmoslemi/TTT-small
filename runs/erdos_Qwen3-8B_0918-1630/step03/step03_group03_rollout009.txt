        (np.abs(np.sin(np.linspace(0, 4*np.pi, 600)) - 1) + 1) / 2,
        np.kaiser(600, 5)
    ]
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for initial_h in initial_constructions:
        n_points = len(initial_h)
        if n_points > 600:
            continue
        dx = 2.0 / n_points
        sum_initial = np.sum(initial_h)
        scale = 1.0 / (sum_initial * dx)
        h_initial = initial_h * scale
        h_initial = np.clip(h_initial, 0, 1)
        
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = LinearConstraint(np.ones(n_points), -np.inf, n_points / 2)
        
        # Use Adam-like optimization with exponential decay for step size
        def adam_optimize(f, x0, bounds, constraint, max_iter=200, step_size=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
            m = np.zeros_like(x0)
            v = np.zeros_like(x0)
            t = 0
            x = x0.copy()
            for _ in range(max_iter):
                t += 1
                grad = np.gradient(f(x))
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * grad**2
                m_hat = m / (1 - beta1**t)
                v_hat = v / (1 - beta2**t)
                x = x - step_size * m_hat / (np.sqrt(v_hat) + eps)
                x = np.clip(x, bounds[0], bounds[1])
                x = x * (1 / (np.sum(x) * dx))
                x = np.clip(x, 0, 1)
                if np.isnan(f(x)).any():
                    break
            return x
        
        result = adam_optimize(objective, x0=h_initial, bounds=bounds, constraint=constraint)
        
        if result is not None:
            h_candidate = result
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
            corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n_points = n_points
    
    # Final check with random perturbations

                best_c5 = c5
                best_h = h_perturb
                best_n_points = len(best_h)
    
    return best_h, best_c5, best_n_points
