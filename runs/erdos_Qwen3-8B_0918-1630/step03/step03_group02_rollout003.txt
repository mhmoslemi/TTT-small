import numpy as np
import time
from scipy.optimize import Bounds
from scipy.signal import correlate
from scipy.optimize import minimize
from functools import partial
from deap import base, creator, algorithms
import random

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initial configuration
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    
    # Function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Function to evaluate a candidate solution
    def evaluate(individual):
        h = np.clip(individual, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return (c5,)
    
    # Genetic algorithm setup
    def genetic_run(n_points):
        # Define the initial population
        initial_pop = np.random.uniform(0, 1, (10, n_points))
        initial_pop = np.array([np.clip(p, 0, 1) for p in initial_pop])
        initial_pop = np.array([p * (1 / (np.sum(p) * dx)) for p in initial_pop])
        
        # Set up the genetic algorithm
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)
        
        toolbox = base.Toolbox()
        toolbox.register("attr_float", np.random.uniform, 0, 1)
        toolbox.register("individual", base.Toolbox.individual, creator.Individual, 
                        lambda: np.array([toolbox.attr_float() for _ in range(n_points)]))
        toolbox.register("population", base.Toolbox.population, creator.Individual, n=10)
        toolbox.register("mate", base.Toolbox.mate, crossover=base.cxBlend, alpha=0.5)
        toolbox.register("mutate", base.Toolbox.mutate, mutation=base.mutGaussian, 
                        mu=0.1, sigma=0.1, indpb=0.1)
        toolbox.register("select", base.Toolbox.select, tournament_size=3)
        toolbox.register("evaluate", evaluate)
        
        # Run the genetic algorithm
        pop = toolbox.population()
        stats = base.Statistics(key=lambda ind: ind.fitness.values)
        logbook = base.Logbook()
        
        for gen in range(100):
            offspring = list(map(toolbox.clone, pop))
            for child in offspring:
                if random.random() < 0.2:
                    toolbox.mutate(child)
            offspring = toolbox.select(pop + offspring, len(pop))
            pop = offspring
            fits = list(map(toolbox.evaluate, pop))
            for ind, fit in zip(pop, fits):
                ind.fitness.values = fit
            logbook.record(gen=gen, **stats.compile(pop))
            if logbook.get_last('min') < 0.380876:
                break
        
        best_ind = min(pop, key=lambda x: x.fitness.values[0])
        return best_ind
    
    # Try different n_points values
    best_c5 = float('inf')
    best_h = initial_h
    best_n = n_points
    
    for n in [100, 200, 400, 800]:
        start_time = time.time()
        if n < n_points:
            h = np.random.uniform(0, 1, n)
            h = h * (1 / (np.sum(h) * (2.0 / n)))
        elif n > n_points:
            h = np.repeat(initial_h, 2)
            if len(h) > n:
                h = h[:n]
        else:
            h = initial_h
        
        # Use genetic algorithm
        if n < 500:
            best_h = genetic_run(n)
            best_c5 = compute_c5(best_h)
        else:
            best_h = initial_h
            best_c5 = compute_c5(best_h)
        
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
