import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import LinearConstraint
from itertools import combinations
from collections import deque

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Define initial constructions
    initial_h_values = np.random.uniform(0, 1, 200)
    initial_h_values = initial_h_values[np.argsort(initial_h_values)]
    initial_h_values[initial_h_values < 0.5] = 0
    initial_h_values[initial_h_values > 0.5] = 1
    initial_h_values = (initial_h_values - 0.5) * 2 + 0.5
    initial_h_values = np.sort(initial_h_values)  # For periodic structure
    initial_h_values = initial_h_values[np.argsort(np.random.rand(len(initial_h_values)))]
    
    def generate_initial_h(n_points):
        h = np.random.uniform(0, 1, n_points)
        h = np.sort(h)
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * 2.0 / n_points))
        return h
    
    # Create a list of initial constructions
    initial_h_list = [
        generate_initial_h(100),
        generate_initial_h(150),
        generate_initial_h(200)
    ]
    
    # Combine with the provided initial_h_values
    initial_h_list.append(initial_h_values)
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    def evaluate_h(h):
        nonlocal best_c5, best_h, best_n_points
        h = np.clip(h, 0, 1)
        total_mass = np.sum(h) * (2.0 / len(h))
        scale = 1.0 / total_mass
        h = h * scale
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / len(h))
        return c5
    
    def objective(x, n_points):
        return evaluate_h(x)
    
    # Set time budget
    start_time = time.time()
    elapsed = 0
    
    # Population-based optimization
    population_size = 20
    generations = 50
    mutation_rate = 0.1
    crossover_rate = 0.9
    
    for init_h in initial_h_list:
        n_points = len(init_h)
        h_initial = init_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = LinearConstraint(np.ones(n_points), -np.inf, n_points / 2)
        
        # Genetic algorithm
        population = [np.random.uniform(0, 1, n_points) for _ in range(population_size)]
        population = [h * (1 / (np.sum(h) * 2.0 / n_points)) for h in population]
        
        for gen in range(generations):
            if time.time() - start_time > budget_s:
                break
            # Evaluate fitness
            fitness = [evaluate_h(h) for h in population]
            # Select top performers
            indices = np.argsort(fitness)
            selected = [population[i] for i in indices[:int(len(population) * 0.5)]]
            # Crossover
            new_population = []
            while len(new_population) < population_size:
                a, b = np.random.choice(selected, 2, replace=False)
                if np.random.rand() < crossover_rate:
                    point = np.random.randint(1, n_points - 1)
                    child = np.concatenate((a[:point], b[point:]))
                else:
                    child = a
                new_population.append(child)
            population = new_population
            
            # Mutation
            for i in range(len(population)):
                if np.random.rand() < mutation_rate:
                    idx = np.random.choice(n_points)
                    population[i][idx] = np.random.uniform(0, 1)
                    population[i] = population[i] * (1 / (np.sum(population[i]) * 2.0 / n_points))
            
            # Update best solution
            for h in population:
                c5 = evaluate_h(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    best_n_points = n_points
    
    # Final refinement with local optimizer
    if best_h is not None:
        bounds = [(0, 1) for _ in range(best_n_points)]
        constraint = LinearConstraint(np.ones(best_n_points), -np.inf, best_n_points / 2)
        
        def refine_objective(x):
            x = np.clip(x, 0, 1)
            x = x * (1 / (np.sum(x) * 2.0 / best_n_points))
            corr = correlate(x, 1 - x, mode='full')
            c5 = np.max(corr) * (2.0 / best_n_points)
            return c5
        
        result = minimize(
            refine_objective,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraint
        )
        
        if result.success:
            best_h = result.x
            best_c5 = evaluate_h(best_h)
    
    # Check the best solution within time budget
    if time.time() - start_time > budget_s:
        return best_h, best_c5, best_n_points if best_h is not None else np.array([0.5]), best_c5, best_n_points
    
    return best_h, best_c5, best_n_points
