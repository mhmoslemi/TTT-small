import numpy as np
import time
from scipy.optimize import Bounds
from numpy.random import rand, choice
from sklearn.neighbors import NearestNeighbors

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial patterns to seed the search
    initial_patterns = [
        np.concatenate([np.zeros(100), np.ones(100)]),  # Checkerboard
        np.sin(np.linspace(0, np.pi, 200)) ** 2,       # Sinusoidal
        np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)]),  # Linear rise and fall
        np.random.rand(200),                            # Random initial guess
    ]
    
    def compute_overlap_integral(h, dx):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint(h, dx):
        return np.sum(h) - (len(h) / 2)
    
    def particle_swarm_optimize(pattern, n_particles=50, max_iter=200, adaptive_learning=True):
        n_points = len(pattern)
        dx = 2 / n_points
        
        # Initialize particles
        particles = np.random.rand(n_particles, n_points)
        particles = np.clip(particles, 0, 1)
        velocities = np.random.randn(n_particles, n_points)
        
        # Initialize bests
        best_positions = particles.copy()
        best_scores = np.array([compute_overlap_integral(p, dx) for p in best_positions])
        best_global = best_positions[np.argmin(best_scores)]
        best_global_score = np.min(best_scores)
        
        # PSO parameters
        inertia = 0.9
        cognitive = 1.5
        social = 1.5
        decay_rate = 0.995
        temp = 1.0
        cool_rate = 0.99
        
        for iter in range(max_iter):
            if adaptive_learning:
                inertia *= decay_rate
                cognitive *= decay_rate
                social *= decay_rate
            
            for i in range(n_particles):
                r1, r2 = np.random.rand(2)
                velocity = inertia * velocities[i] + cognitive * r1 * (best_positions[i] - particles[i]) + social * r2 * (best_global - particles[i])
                velocities[i] = velocity
                
                particles[i] += velocity
                particles[i] = np.clip(particles[i], 0, 1)
                
                # Apply constraint: normalize to maintain integral constraint
                particles[i] = particles[i] * (len(particles[i]) / 2) / np.sum(particles[i])
                
                # Compute score
                score = compute_overlap_integral(particles[i], dx)
                
                if score < best_scores[i]:
                    best_scores[i] = score
                    best_positions[i] = particles[i].copy()
            
            # Update global best
            best_global = best_positions[np.argmin(best_scores)]
            best_global_score = np.min(best_scores)
            
            # Cool down
            if adaptive_learning:
                temp *= cool_rate
        
        return best_global, best_global_score
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    best_time = 0
    
    for pattern in initial_patterns:
        n_points = len(pattern)
        dx = 2.0 / n_points
        if n_points < 100:
            continue
        
        # Scale initial pattern to satisfy integral constraint
        sum_p = np.sum(pattern)
        scaled_pattern = pattern * (n_points / 2) / sum_p
        scaled_pattern = np.clip(scaled_pattern, 0, 1)
        
        # Run PSO
        start_time = time.time()
        h_candidate, c5_candidate = particle_swarm_optimize(scaled_pattern, max_iter=100)
        time_taken = time.time() - start_time
        
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
            best_n = n_points
            best_time = time_taken
    
    return best_h, best_c5, best_n
