import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize.lbfgsb import fmin_l_bfgs_b
from functools import partial

# Pre-imported initial_h_values
initial_h_values = np.array([0.0] * 200)

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    n_points_start = 200
    n_points_final = 500
    max_iterations = 50
    perturbation_factor = 0.2
    temp_start = 1.0
    temp_end = 0.1
    cooling_rate = 0.995

    def objective(h, dx):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = 1.0 / (sum_h * dx) if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx

    def simulate_anneal(initial_h, n_points, dx, temp_start, temp_end, cooling_rate, max_iterations):
        h = initial_h.copy()
        best_h = h.copy()
        best_c5 = objective(h, dx)
        temp = temp_start
        for iter in range(max_iterations):
            for _ in range(10):
                perturb = np.random.rand(n_points) * 2 * perturbation_factor - perturbation_factor
                h_candidate = np.clip(h + perturb, 0, 1)
                h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
                c5_candidate = objective(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
            temp *= cooling_rate
            if temp < temp_end:
                break
        return best_h, best_c5

    def evaluate_h(h, n_points):
        dx = 2.0 / n_points
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = 1.0 / (sum_h * dx) if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5, h_scaled

    initial_h = initial_h_values.copy()
    n_points = n_points_start
    dx = 2.0 / n_points
    initial_c5, _ = evaluate_h(initial_h, n_points)
    best_h = initial_h
    best_c5 = initial_c5
    best_n_points = n_points

    for _ in range(max_iterations):
        n_points = n_points_start + int(np.random.uniform(0, 300))
        if n_points < 10:
            continue
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = h * (1 / (np.sum(h) * dx))
        h = np.clip(h, 0, 1)
        c5, _ = evaluate_h(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n_points

    final_h = best_h.copy()
    final_n_points = best_n_points
    final_dx = 2.0 / final_n_points
    final_c5, _ = evaluate_h(final_h, final_n_points)

    return final_h, final_c5, final_n_points
