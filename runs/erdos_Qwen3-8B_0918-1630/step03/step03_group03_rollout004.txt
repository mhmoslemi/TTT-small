import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import LinearConstraint
from skopt import BayesSearchSpace, BayesOptimizer

    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_initial = np.sum(initial_h)
    scale = 1.0 / (sum_initial * dx)

    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = LinearConstraint(np.ones(n_points), -np.inf, n_points / 2)
    
    # Define Bayesian Optimization Space
    search_space = BayesSearchSpace()
    for i in range(n_points):
        search_space.add_variable(('x' + str(i)), (0, 1))
    
    # Bayesian Optimization with early stopping
    optimizer = BayesOptimizer(
        objective=objective,
        search_space=search_space,
        n_jobs=1,
        random_state=seed,
        n_iter=30,
        acq_func='ei',
        n_points=10,
        verbose=False
    )
    
    # Initial evaluation of the initial guess
    h_candidate = h_initial.copy()
    h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
    corr = correlate(h_candidate, 1 - h_candidate, mode='full')
    c5_candidate = np.max(corr) * dx
    best_c5 = c5_candidate
    best_h = h_candidate
    
    # Run Bayesian Optimization
    start_time = time.time()
    while time.time() - start_time < budget_s:
        try:
            result = optimizer.ask_and_tell(
                x0=h_candidate,
                constraints=constraint,
                bounds=bounds
            )
            if result is None:
                break

            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))

            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_candidate
        except Exception as e:
            print(f"Bayesian Optimization failed: {e}")
            break
    
    # Final refinement with SLSQP
    result = minimize(

        x0=best_h,

        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
