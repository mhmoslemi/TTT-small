import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=60):

    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_initial = np.sum(initial_h)
    scale = 1.0 / (sum_initial * dx)
    h_initial = initial_h * scale
    h_initial = np.clip(h_initial, 0, 1)
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    best_n = n_points
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def objective(h):
        return compute_c5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = LinearConstraint(np.ones(n_points), -np.inf, n_points / 2)
    
    def perturb_h(h, scale_factor=0.1):
        h = np.copy(h)
        h += np.random.uniform(-scale_factor, scale_factor, h.shape)
        h = np.clip(h, 0, 1)
        return h
    
    def evolve_h(h):
        h = perturb_h(h)
        h = h * (1 / (np.sum(h) * dx))
        return h
    
    def local_search(h):
        result = minimize(objective, x0=h, method='SLSQP', bounds=bounds, constraints=constraint)
        if result.success:
            return result.x
        return h
    
    def genetic_search(h, n_pop=20, n_gen=10, mutation_rate=0.2, crossover_rate=0.8):
        pop = [h.copy() for _ in range(n_pop)]
        for _ in range(n_gen):
            fitness = [compute_c5(ind) for ind in pop]
            best_idx = np.argmin(fitness)
            best = pop[best_idx]
            new_pop = [best.copy() for _ in range(n_pop)]
            for i in range(n_pop):
                if np.random.rand() < crossover_rate:
                    j = np.random.randint(n_pop)
                    child = (pop[i] + pop[j]) / 2
                    if np.random.rand() < mutation_rate:
                        child = perturb_h(child)
                    new_pop[i] = child
            pop = new_pop
        return np.min(fitness)
    
    def run_evolution(h):
        h = evolve_h(h)
        for _ in range(10):
            h = evolve_h(h)
            h = h * (1 / (np.sum(h) * dx))
        return h
    
    def eval_and_store(h, n, best_c5, best_h, best_n):
        h = h * (1 / (np.sum(h) * dx))
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n
        return best_c5, best_h, best_n
    
    start_time = time.time()

    ga_result = genetic_search(h_initial)
    best_c5, best_h, best_n = eval_and_store(ga_result, n_points, best_c5, best_h, best_n)
    
    # Local search on GA result
    local_result = local_search(best_h)
    best_c5, best_h, best_n = eval_and_store(local_result, n_points, best_c5, best_h, best_n)
    
    # Evolutionary refinement
    for _ in range(5):
        h = run_evolution(best_h)
        best_c5, best_h, best_n = eval_and_store(h, n_points, best_c5, best_h, best_n)
    
    # Perturb and refine
    for _ in range(10):
        perturbed = perturb_h(best_h, 0.1)
        perturbed = perturbed * (1 / (np.sum(perturbed) * dx))
        best_c5, best_h, best_n = eval_and_store(perturbed, n_points, best_c5, best_h, best_n)
    
    # Final local search
    local_result = local_search(best_h)
    best_c5, best_h, best_n = eval_and_store(local_result, n_points, best_c5, best_h, best_n)
    
    # Return the best solution
    return best_h, best_c5, best_n
