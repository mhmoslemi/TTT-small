import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import LinearConstraint
from deap import base, creator, tools
import random

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define initial configurations with different n and structures
    initial_configs = {
        100: np.linspace(0, 2, 100),
        150: np.linspace(0, 2, 150),
        200: np.linspace(0, 2, 200),
        250: np.linspace(0, 2, 250),
        300: np.linspace(0, 2, 300),
        350: np.linspace(0, 2, 350),
        400: np.linspace(0, 2, 400)
    }
    
    best_c5 = float('inf')
    best_h = np.array([])
    best_n = 0
    
    for n_points, x_values in initial_configs.items():
        if n_points > 400:
            continue
        
        # Randomly perturb x_values to avoid regular sampling
        perturb = np.random.uniform(-0.1, 0.1, n_points)
        x_values = np.clip(x_values + perturb, 0, 2)
        
        # Create initial h as a function of x_values with random values
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
        dx = 2.0 / n_points
        
        # Define objective
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = LinearConstraint(np.ones(n_points), -np.inf, n_points / 2)
        
        # Use SLSQP optimization
        result = minimize(objective, x0=h_initial, method='SLSQP', bounds=bounds, constraints=constraint)
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
            corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n = n_points
        
        # Evaluate with genetic algorithm
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", list, fitness=creator.FitnessMin)
        
        toolbox = base.Toolbox()
        toolbox.register("attr_float", lambda: np.random.uniform(0, 1))
        toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
        toolbox.register("population", tools.initRepeat, list, toolbox.individual)
        
        def eval_func(individual):
            h = np.array(individual)
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return (c5,)
        
        toolbox.register("evaluate", eval_func)
        toolbox.register("select", tools.selTournament, tournsize=3)
        toolbox.register("mate", tools.cxTwoPoint)
        toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
        
        pop = toolbox.population(n=20)
        for gen in range(20):
            offspring = list()
            for ind in pop:
                offspring.append(ind)
            offspring = [toolbox.clone(ind) for ind in offspring]
            for child1, child2 in zip(offspring[::2], offspring[1::2]):
                if random.random() < 0.5:
                    toolbox.mate(child1, child2)
                    del child1.fitness.values
                    del child2.fitness.values
            for idx, ind in enumerate(offspring):
                if random.random() < 0.1:
                    toolbox.mutate(ind)
                    del ind.fitness.values
            invalid_ind = [ind for ind in offspring if not ind.fitness.valid]
            for ind in invalid_ind:
                ind.fitness = toolbox.evaluate(ind)
            pop = tools.selBest(offspring, k=len(pop))
        
        for ind in pop:
            if not ind.fitness.worse:
                h_candidate = np.array(ind)
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
                corr = correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr) * dx
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate
                    best_n = n_points
        
    return best_h, best_c5, len(best_h)
