import numpy as np
import time
from scipy.optimize import Bounds
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Define a base initial guess based on a low-frequency sinusoid
    base_h = np.sin(np.linspace(0, np.pi, 200)) * 0.5 + 0.5
    base_h = np.clip(base_h, 0, 1)
    base_h = base_h * (200 / np.sum(base_h))  # Normalize to satisfy integral constraint
    
    # Function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization with multiple restarts and adaptive bounds
    best_c5 = float('inf')
    best_h = base_h.copy()
    n_points = len(base_h)
    dx = 2.0 / n_points
    
    # Bounds for all h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define an objective function that includes the integral constraint
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        return compute_c5(h)
    
    # Optimization with integral constraint
    def optimize_from_guess(guess):
        result = minimize(objective, x0=guess, method='L-BFGS-B', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}],
                          options={'maxiter': 100, 'disp': False})
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
            corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            return h_candidate, c5_candidate
        return None, float('inf')
    
    # Run multiple restarts
    restarts = 5
    start_time = time.time()
    for _ in range(restarts):
        if time.time() - start_time > budget_s - 1:
            break
        guess = np.random.uniform(0, 1, n_points)
        guess = guess * (n_points / 2)  # Initial guess satisfying integral constraint
        h_candidate, c5_candidate = optimize_from_guess(guess)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
