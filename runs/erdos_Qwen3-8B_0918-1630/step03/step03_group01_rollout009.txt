import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.optimize import Bounds
import random

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    initial_h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            initial_h[i] = 0.5
        else:
            initial_h[i] = 1.0
    initial_h = initial_h * (1.0 / (np.sum(initial_h) * dx)) * (1.0 / 0.5)
    initial_h = np.clip(initial_h, 0, 1)
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = 1.0 / (sum_h * dx) if sum_h != 0 else 1.0
        h_scaled = h * scale
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        return np.max(corr) * dx
    
    def objective(h):
        return compute_c5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def optimize_h(h_initial):
        result = minimize(objective, x0=h_initial, bounds=bounds, constraints=[{'type': 'eq', 'fun': constraint}])
        if result.success:
            return result.x
        else:
            return h_initial
    
    best_h = initial_h
    best_c5 = compute_c5(best_h)
    
    def perturb(h, scale=0.1):
        h = h.copy()
        for i in range(n_points):
            h[i] += np.random.normal(0, scale)
            h[i] = np.clip(h[i], 0, 1)
        return h
    
    for _ in range(10):
        h_candidate = perturb(best_h)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        h_candidate = np.clip(h_candidate, 0, 1)
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
    
    return best_h, best_c5, n_points
