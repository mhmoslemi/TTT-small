import numpy as np
import time
from scipy.optimize import Bounds, minimize, differential_evolution
from scipy.signal import correlate
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Candidate initial patterns
    patterns = [
        np.concatenate([np.zeros(100), np.ones(100)]),  # Checkerboard pattern
        np.sin(np.linspace(0, np.pi, 200)) ** 2,       # Sinusoidal pattern
        np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)]),  # Linear rise and fall
        np.random.rand(200) * 2 - 1,                   # Random initial guess
        np.zeros(200),                                  # Constant zero
        np.ones(200),                                   # Constant one
    ]
    
    best_c5 = float('inf')
    best_h = np.zeros(200)
    best_n = 200
    best_time = 0
    
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        return compute_c5(h)
    
    for pattern in patterns:
        n_points = len(pattern)
        if n_points > 600:
            continue
        dx = 2.0 / n_points
        sum_h = np.sum(pattern)
        scale = 1.0 / (sum_h * dx)
        h_initial = pattern * scale
        h_initial = np.clip(h_initial, 0, 1)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_func(h):
            return np.sum(h) - n_points / 2
        
        try:
            result = minimize(objective, x0=h_initial, method='SLSQP', bounds=bounds,
                              constraints=[{'type': 'eq', 'fun': constraint_func}])
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
                c5_candidate = compute_c5(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate
                    best_n = n_points
                    best_time = time.time()
        except:
            pass
        
        if time.time() - best_time > budget_s - 1:
            break
    
    def evolution_objective(x):
        x = np.clip(x, 0, 1)
        x = x * (1 / (np.sum(x) * dx))
        return compute_c5(x)
    
    try:
        result = differential_evolution(evolution_objective, bounds=[(0, 1)] * best_n,
                                       strategy='best1bin', popsize=20, maxiter=200,
                                       tol=1e-6, mutation=(0.5, 1), recombination=0.8,
                                       polish=True)
        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate
            best_n = best_n
    except:
        pass
    
    return best_h, best_c5, best_n
