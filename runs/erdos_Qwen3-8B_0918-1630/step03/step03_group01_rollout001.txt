import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Predefined initial step function constructions
    initial_constructions = [
        np.array([0.5 for _ in range(400)]),  # Uniform
        np.array([0.5 if i % 2 == 0 else 0 for i in range(400)]),  # Alternating
        np.array([1.0 if i < 200 else 0.0 for i in range(400)]),  # Half-1
        np.array([1.0 if i < 100 else 0.0 for i in range(400)]),  # Half-2
        np.array([0.5 if i % 3 == 0 else 0 for i in range(400)]),  # Mod-3
    ]
    
    best_h = None
    best_c5 = float('inf')
    best_n = 0
    
    for initial_h in initial_constructions:
        n_points = len(initial_h)
        dx = 2.0 / n_points
        
        # Ensure integral constraint by scaling
        scale = 1.0 / (np.sum(initial_h) * dx)
        h = initial_h * scale
        
        # Define the objective function
        def objective(x):
            h = np.clip(x, 0, 1)
            sum_h = np.sum(h)
            dx = 2.0 / len(h)
            h_scaled = h * (1.0 / (sum_h * dx)) if sum_h != 0 else h
            corr = correlate(h_scaled, 1 - h_scaled, mode='full')
            return np.max(corr) * dx
        
        # Bounds for all h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use L-BFGS-B for optimization
        result = minimize(objective, x0=h, method='L-BFGS-B', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}])
        
        if result.success:
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
            # Rescale to maintain integral constraint
            sum_h = np.sum(current_h)
            dx = 2.0 / len(current_h)
            current_h = current_h * (1.0 / (sum_h * dx)) if sum_h != 0 else current_h
            corr = correlate(current_h, 1 - current_h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                best_n = n_points
        
        # Test a few random perturbations
        for _ in range(3):
            h_candidate = np.clip(current_h + np.random.rand(len(current_h)) * 0.2 - 0.1, 0, 1)
            sum_h = np.sum(h_candidate)
            dx = 2.0 / len(h_candidate)
            h_candidate = h_candidate * (1.0 / (sum_h * dx)) if sum_h != 0 else h_candidate
            corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
                best_n = len(h_candidate)
    
    return best_h, best_c5, best_n
