```python
import numpy as np
import time
from scipy.optimize import differential_evolution
from scipy.signal import correlate
from functools import partial

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    n_points = 1000
    dx = 2.0 / n_points
    
    # Initial guess: periodic-based with adjusted samples
    initial_h = np.sin(np.linspace(0, np.pi, n_points) * 2) / 2 + 0.5
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (2.0 / n_points) / np.mean(initial_h)
    
    # Objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx)) if np.sum(h) != 0 else h
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Bounds for all h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Objective with fixed bounds
    obj_fixed = partial(objective, bounds=bounds)
    
    # Use differential evolution
    result = differential_evolution(obj_fixed, bounds, strategy='best1bin', popsize=30, maxiter=100, tol=1e-6, mutation=(0.5, 1), recombination=0.9, polish=False, seed=seed)
    
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (1 / (np.sum(best_h) * dx))
    corr = correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Try different initial configurations
    initial_configs = [
        np.random.rand(n_points) * 0.5 + 0.5,
        np.sin(np.linspace(0, 2 * np.pi, n_points)) / 2 + 0.5,
        np.cos(np.linspace(0, 2 * np.pi, n_points)) / 2 + 0.5,
        np.heaviside(np.linspace(0, 2, n_points) - 1, 0.5),
        np.random.permutation(initial_h)
    ]
    
    for config in initial_configs:
        config = np.clip(config, 0, 1)
        config = config * (1 / (np.sum(config) * dx))
        corr = correlate(config, 1 - config, mode='full')
        c5_candidate = np.max(corr) * dx
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = config
    
    return best_h, best_c5, n_points
```
