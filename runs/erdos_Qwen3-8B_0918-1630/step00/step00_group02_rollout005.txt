import numpy as np
import scipy.optimize as optimize
import time

initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial guess

def evaluate_erdos_solution(h_values, n_points, dx):
    h = np.array(h_values)
    shifted = np.roll(h, shift=1)
    corr = np.correlate(h, 1 - h, mode='full')
    overlap = corr * dx
    return np.max(overlap)

def objective_function(h, n_points, dx):
    h_array = np.array(h)
    shifted = np.roll(h_array, shift=1)
    corr = np.correlate(h_array, 1 - h_array, mode='full')
    overlap = corr * dx
    return np.max(overlap)

def constraint_sum(h, n_points):
    return np.sum(h) - n_points / 2

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 256
    dx = 2.0 / n_points
    initial_h = initial_h_values[:n_points]
    best_c5 = float('inf')
    best_h = initial_h
    best_n = n_points

    def perturb_h(h):
        perturbation = np.random.normal(0, 0.01, h.shape)
        return np.clip(h + perturbation, 0, 1)

    def optimize_h(start_h, max_iter=100):
        h = start_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            lambda x: objective_function(x, n_points, dx),
            h,
            bounds=bounds,
            method='L-BFGS-B',
            options={'maxiter': max_iter}
        )
        return result.x

    start_time = time.time()

    for _ in range(5):
        start_h = perturb_h(initial_h)
        optimized_h = optimize_h(start_h)
        current_c5 = objective_function(optimized_h, n_points, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = optimized_h
            best_n = n_points

    if time.time() - start_time < budget_s:
        for _ in range(3):
            start_h = perturb_h(best_h)
            optimized_h = optimize_h(start_h)
            current_c5 = objective_function(optimized_h, n_points, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = optimized_h
                best_n = n_points

    return best_h, best_c5, best_n
