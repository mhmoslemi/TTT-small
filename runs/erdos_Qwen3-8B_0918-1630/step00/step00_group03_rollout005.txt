import numpy as np
import random
import time
from scipy.optimize import minimize

# Pre-imported functions
initial_h_values = np.load("initial_h_values.npy")

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100
    h = np.array(initial_h_values[:n_points])
    h = h / np.sum(h) * n_points / 2
    best_overlap = evaluate_erdos_solution(h)
    
    # Define the objective function
    def objective(h):
        h = np.array(h)
        h = h / np.sum(h) * n_points / 2
        return evaluate_erdos_solution(h)
    
    # Define the constraints
    def constraint(h):
        h = np.array(h)
        h = h / np.sum(h) * n_points / 2
        return np.sum(h) - n_points / 2
    
    # Define the bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use the Simulated Annealing method
    result = minimize(
        objective,
        x0=h,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='SLSQP',
        options={'ftol': 1e-8, 'max_iter': 100, 'disp': False}
    )
    
    h = result.x
    h = h / np.sum(h) * n_points / 2
    best_overlap = evaluate_erdos_solution(h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        print(f"Time budget exceeded, returning best-so-far solution")
        return (h, best_overlap, n_points)
    
    return (h, best_overlap, n_points)
