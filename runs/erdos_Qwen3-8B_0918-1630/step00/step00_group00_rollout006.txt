def run(seed=42, budget_s=60):

        trial_h = trial_h * (n_points / 2) / np.sum(trial_h)
        starting_points.append(trial_h)

    def cost_func(h):
        dx = 2.0 / len(h)
        shifted_h = np.roll(h, shift=1)
        corr = np.correlate(h, 1 - shifted_h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define the constraints
    def constraints_func(h):
        return np.sum(h) - n_points / 2

    # Define the bounds
    bounds = [(0, 1) for _ in range(n_points)]

    for start in starting_points:

        try:
            result = optimize.minimize(
                cost_func,
                start,
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraints_func}],
                method='L-BFGS-B',
