        initial_h = np.array([0.5 for _ in range(n_points)])

    total = np.sum(h)

        return h

    total = np.sum(h)

    scale = (n_points / 2) / total

    return h

def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Create 1 - h for correlation
    h_ones_complement = 1.0 - h
    # Compute the cross-correlation
    corr = correlate(h, h_ones_complement, mode='full')
    # The overlap integral is the maximum of corr * dx
    max_overlap = np.max(corr) * dx
    return max_overlap
