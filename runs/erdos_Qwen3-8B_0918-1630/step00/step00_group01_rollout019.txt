import time
from scipy.optimize import minimize, Bounds
from scipy.stats import norm

def run(seed=42, budget_s=60, **kwargs):

    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Objective function to minimize C_5
    def objective(h):
        h = np.array(h)
        # Compute all overlaps using convolution
        overlaps = np.correlate(h, 1 - h, mode='full')
        # Normalize the overlaps by dx
        overlaps /= dx
        # C_5 is the maximum overlap
        return np.max(overlaps)

    # Constraint: sum of h must be n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2

    # Bounds: each h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess
    x0 = initial_h.copy()

    # Optimization settings
    max_iter = 100
    tolerance = 1e-6

    start_time = time.time()
    best_solution = x0
    best_c5 = objective(x0)

    # Use Nelder-Mead with bounds
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': tolerance, 'maxiter': max_iter}
    )

    # Check if result is better
    if result.success and result.fun < best_c5:
        best_solution = result.x
        best_c5 = result.fun

    # If not improved, use the initial solution
    if best_c5 > objective(x0):
        best_solution = x0
        best_c5 = objective(x0)

    # Check if we are under time limit
    if time.time() - start_time < budget_s:
        # Perform a few random perturbations to explore the space
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
