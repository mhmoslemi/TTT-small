import numpy as np

from scipy.optimize import minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    max_k = n_points // 2  # Limit to reasonable shifts

    def objective(h):
        h = np.array(h)
        max_val = 0
        for k in range(1, max_k + 1):
            corr = np.sum(h * (1 - np.roll(h, k)) * dx)
            max_val = max(max_val, corr)
        return max_val

    def constraint(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, x0=initial_h, bounds=bounds, constraints={"type": "eq", "fun": constraint})
    best_h = result.x
    best_c5 = objective(best_h)
    return best_h, best_c5, n_points
