    h = np.array(initial_h[:n_points], dtype=np.float64)
    h = np.resize(h, (n_points,))
    
    def objective(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        x = x * dx
        max_overlap = 0.0
        for k in range(1, n_points):
            shifted = np.roll(x, k)
            overlap = np.sum(x * (1 - shifted))
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    def compute_overlap(h_values):
        n = len(h_values)
        dx = 2.0 / n
        max_overlap = 0.0
        for k in range(1, n):
            shifted = np.roll(h_values, k)
            overlap = np.sum(h_values * (1 - shifted)) * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    def objective_with_constraints(x):
        x = x.reshape(n_points)
        x = np.clip(x, 0, 1)
        if np.abs(np.sum(x) * dx - 1) > 1e-6:
            return float('inf')
        return compute_overlap(x)
    
    best_overlap = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    # Optimization with constraints
    result = opt.minimize(
        objective_with_constraints,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 1000}
    )
    best_h = result.x
