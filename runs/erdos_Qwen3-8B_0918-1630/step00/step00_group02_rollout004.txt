import numpy as np
import scipy.optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Precompute initial_h_values
    initial_h_values = np.load("initial_h_values.npy")
    n_points_initial = len(initial_h_values)
    n_points = n_points_initial  # Start with the initial configuration
    
    def evaluate_h(h):
        n = len(h)
        dx = 2.0 / n
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * n / 2  # Enforce integral constraint
        
        # Compute all correlations
        overlap = []
        for k in range(1, n):
            # Compute the overlap integral: sum(h[i] * (1 - h[(i + k) % n])) * dx
            corr = 0.0
            for i in range(n):
                j = (i + k) % n
                corr += h[i] * (1 - h[j])
            overlap.append(corr * dx)
        return np.max(overlap)
    
    def objective(h):
        return evaluate_h(h)
    
    def gradient_descent(h0, max_iter=1000, learning_rate=0.01, tol=1e-5):
        h = h0.copy()
        for _ in range(max_iter):
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = np.copy(h)
                h_plus[i] += 1e-6
                h_minus = np.copy(h)
                h_minus[i] -= 1e-6
                dh = (evaluate_h(h_plus) - evaluate_h(h_minus)) / (2e-6)
                grad[i] = dh
            grad = grad / np.linalg.norm(grad)
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * len(h) / 2
            if np.linalg.norm(h - h0) < tol:
                break
        return h
    
    def simulated_annealing(start_h, temp=1.0, cooling_rate=0.995, max_iter=1000):
        h = start_h.copy()
        best_h = h.copy()
        best_value = evaluate_h(h)
        for _ in range(max_iter):
            perturbation = np.random.normal(0, 0.05, len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            h_perturbed = h_perturbed / np.sum(h_perturbed) * len(h_perturbed) / 2
            value = evaluate_h(h_perturbed)
            if np.random.rand() < np.exp((best_value - value) / temp):
                h = h_perturbed
                if value < best_value:
                    best_value = value
                    best_h = h.copy()
            temp *= cooling_rate
        return best_h, best_value
    
    def optimize_with_scheduled_budget(n_points, budget_s):
        start_time = time.time()
        h = initial_h_values[:n_points].copy()
        h = h / np.sum(h) * n_points / 2
        history = []
        
        # First perform simulated annealing
        print(f"Starting with {n_points} points...")
        sa_h, sa_value = simulated_annealing(h, max_iter=500)
        history.append(sa_value)
        
        # Refine with gradient descent
        gd_h = gradient_descent(sa_h, max_iter=500)
        gd_value = evaluate_h(gd_h)
        history.append(gd_value)
        
        # Final evaluation
        final_value = evaluate_h(gd_h)
        print(f"Final C5 value: {final_value}")
        return gd_h, final_value, n_points
    
    # Start with the initial n_points and try to refine
    best_h = initial_h_values.copy()
    best_value = evaluate_h(best_h)
    best_n = len(best_h)
    
    # Try small subsets
    for n_points in [100, 200, 300]:
        if time.time() - start_time > budget_s - 1:
            break
        h = initial_h_values[:n_points].copy()
        h = h / np.sum(h) * n_points / 2
        h_opt = gradient_descent(h, max_iter=200)
        value = evaluate_h(h_opt)
        if value < best_value:
            best_value = value
            best_h = h_opt
            best_n = n_points
    
    print(f"Final best C5 value: {best_value}, with {best_n} points")
    return best_h, best_value, best_n
