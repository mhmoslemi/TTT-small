```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode='full')) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # placeholder for initial guess

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    max_n_points = 400  # balance between resolution and speed
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Start from initial_h_values if provided
    if hasattr(initial_h_values, '__array__'):
        initial_h = initial_h_values.copy()
        n_points = len(initial_h)
        if n_points > max_n_points:
            initial_h = initial_h[:max_n_points]
            n_points = len(initial_h)
        if np.sum(initial_h) * 2.0 / n_points != 1:
            raise ValueError("Initial h does not satisfy the integral constraint")
    else:
        initial_h = np.random.rand(max_n_points)
        initial_h = (initial_h * 2.0 / np.sum(initial_h)).clip(0, 1)
    
    def objective(h):
        nonlocal best_c5
        h = h.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / len(h))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    def constraint(h):
        return np.sum(h) - len(h) / 2  # ensure ∫h dx = 1
    
    # Perform local optimization
    result = minimize(objective, initial_h, method='L-BFGS-B', bounds=[(0, 1)] * len(initial_h),
                      constraints=[{'type': 'eq', 'fun': constraint}])
    
    # Final evaluation
    final_h = result.x
    final_c5 = evaluate_erdos_solution(final_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (best_h, best_c5, len(best_h))
    
    # Additional refinement with hill climbing
    for _ in range(10):
        if elapsed > budget_s - 1:
            break
        h = best_h.copy()
        for i in range(len(h)):
            for delta in [0.05, -0.05]:
                h[i] += delta
                h[i] = np.clip(h[i], 0, 1)
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                h[i] -= delta
        elapsed = time time.time() - start_time
    
    return (best_h, best_c5, len(best_h))
```
