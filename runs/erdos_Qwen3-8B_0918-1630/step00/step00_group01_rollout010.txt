import numpy as np
import cma
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values is provided
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    
    def objective(x):
        h = np.clip(x, 0, 1)
        # Normalize h to ensure sum is n_points / 2
        total = np.sum(h)
        if total == 0:
            return np.inf
        h = h * (n_points / 2) / total
        # Compute the convolution
        conv = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    # Initial guess
    initial_guess = initial_h_values
    # Adjust initial guess to be within [0, 1]
    initial_guess = np.clip(initial_guess, 0, 1)
    total = np.sum(initial_guess)
    if total != 0:
        initial_guess = initial_guess * (n_points / 2) / total
    else:
        initial_guess = np.ones(n_points) * (n_points / 2) / n_points
    
    # Run CMA-ES
    result = cma.CMAEvolutionStrategy(
        initial_guess,
