import numpy as np
from scipy.optimize import minimize
import time

# Pre-imported functions (simulated here)
initial_h_values = np.random.rand(45) * 2 - 1  # Simulate initial values
def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    h = np.clip(h, 0, 1)
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 64  # Smaller n for faster evaluation
    dx = 2.0 / n_points
    initial_guess = np.random.rand(n_points)
    initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)

    def objective(h_vec):
        h = np.clip(h_vec, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    def constraint(h_vec):
        return np.sum(h_vec) - n_points / 2

    # Initial guess
    initial_guess = np.random.rand(n_points) * 2 - 1
    initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)

    # Optimization
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'ftol': 1e-10, 'maxiter': 1000}
    )

    if result.success:
        best_h = result.x
    else:
        best_h = initial_guess

    # Final check and evaluation
    best_h = np.clip(best_h, 0, 1)
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, n_points
