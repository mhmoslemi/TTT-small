from functools import lru_cache

initial_h_values = np.load('initial_h_values.npy')  # Assuming this is preloaded

def run(seed=42, budget_s=60, **kwargs):

    np.random.seed(seed)

    h_initial = initial_h_values

        if np.sum(h) != n_points / 2:

        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr * dx)
        return c5

        return np.sum(h) - n_points / 2
