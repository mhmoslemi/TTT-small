import numpy as np
import scipy.optimize as optimize
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from provided initial_h_values
    initial_h_values = np.load('initial_h_values.npy')  # Assume this is a pre-loaded initial guess
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    if n_points < 100:
        n_points = 200
        dx = 2.0 / n_points
        # Resample initial_h_values to match new n_points
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    def objective(h_values):
        # Shift the function and compute overlap
        h = h_values.reshape(-1)
        max_overlap = 0.0
        for k in range(n_points):
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted)) * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization using L-BFGS-B
    result = minimize(
        objective,
        initial_h_values,
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        method='L-BFGS-B',
        options={'maxiter': 500, 'ftol': 1e-8}
    )
    
    best_h_values = result.x
    best_c5 = objective(best_h_values)
    
    # Return the best solution found
    return (best_h_values, best_c5, n_points)
