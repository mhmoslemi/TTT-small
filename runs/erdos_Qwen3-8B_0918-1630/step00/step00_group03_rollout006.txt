def overlap_integral(h, dx):
    n = len(h)
    max_c5 = 0.0
    for k in range(n):
        shift = k * dx
        c = 0.0

            x = i * dx

            if x_shifted >= 2.0:

        max_c5 = max(max_c5, c)
    return max_c5
