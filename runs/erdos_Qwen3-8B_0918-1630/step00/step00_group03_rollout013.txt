import numpy as np
import time
from scipy.optimize import Bounds, minimize
from itertools import combinations

# Constants
MAX_BUDGET_S = 60
MAX_EVALS = 500
MIN_N_POINTS = 50
MAX_N_POINTS = 100
TARGET_C5 = 0.380876

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all shifts
    result = []
    for k in range(1, int(2 * dx) + 1):
        shifted = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted)) * dx
        result.append(overlap)
    return np.max(result)

def build_candidate(n_points):
    # Random candidate with sum(h) = n_points / 2
    h = np.random.uniform(0, 1, n_points)
    scaling = (n_points / 2) / np.sum(h)
    h *= scaling
    return h

def generate_initial(n_points):
    # Start with initial_h_values if available, else random
    global initial_h_values
    if initial_h_values is not None:
        h = np.array(initial_h_values[:n_points])
        scaling = (n_points / 2) / np.sum(h)
        h *= scaling
        return h
    return build_candidate(n_points)

def optimize_erdos(n_points, seed=42):
    # Set random seed for reproducibility
    np.random.seed(seed)
    # Generate initial candidate
    h_values = generate_initial(n_points)
    best_h = h_values.copy()
    best_c5 = evaluate_erdos_solution(h_values, n_points)
    best_c5 = best_c5

    # Optimization loop
    for _ in range(MAX_EVALS):
        # Perturb the current solution
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.2, 0.2)
        h = best_h.copy()
        h[idx] += delta
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Enforce integral constraint
        sum_h = np.sum(h)
        scaling = (n_points / 2) / sum_h if sum_h != 0 else 1.0
        h *= scaling
        # Evaluate
        current_c5 = evaluate_erdos_solution(h, n_points)
        # Accept better solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            print(f"New best C5: {best_c5:.6f}")

    return best_h, best_c5, n_points

def run(seed=42, budget_s=60, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n = 0

    # Try different n_points
    for n_points in range(MIN_N_POINTS, MAX_N_POINTS + 1):
        if time.time() - start_time > budget_s:
            break
        print(f"Trying n={n_points}")
        try:
            h, c5, _ = optimize_erdos(n_points, seed)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_n = n_points
        except:
            pass

    return best_h, best_c5, best_n
