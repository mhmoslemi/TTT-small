import time
import scipy.optimize as optimize

    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = n_points / 2.0  # so that sum(h) * dx = 1

    # Objective function to minimize the maximum overlap
    def objective(h):
        h = np.array(h, dtype=np.float64)
        if np.abs(sum(h) - sum_h) > 1e-8:
            return float('inf')
        # Compute all shifted overlaps
        overlaps = np.zeros(n_points - 1)
        for k in range(1, n_points - 1):
            # Pad h to handle the full convolution
            h_padded = np.zeros(n_points + k)
            h_padded[:n_points] = h
            h_shifted = np.zeros(n_points + k)
            h_shifted[k:] = h[:n_points - k]
            overlap = np.sum(h_padded * h_shifted) * dx
            overlaps[k - 1] = overlap
        return np.max(overlaps)  # Max overlap across all k

    # Bounds for function values
    bounds = Bounds(0, 1)

    # Use L-BFGS-B with restarts or try a smarter approach
    result = optimize.minimize(
        objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'gtol': 1e-7, 'maxiter': 1000, 'disp': False}
    )

    # Early termination if time runs out
    if time.time() - start_time > budget_s:
        return (initial_h, objective(initial_h), n_points)

    # Return the best configuration found
    return (result.x, objective(result.x), n_points)
