```python
import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate
import time

# Pre-imported helper functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Start from initial_h_values
    h = np.array(initial_h_values[:n_points])
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h)) if np.sum(h) != 0 else h

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        a = h
        b = 1 - h
        corr = correlate(a, b, mode="full")
        c5 = np.max(corr * dx)
        return c5

    def objective(h):
        return compute_c5(h)

    def constraint(h):
        return np.sum(h) - target_sum

    def bounds(h):
        return np.array([0.0, 1.0])

    result = optimize.minimize(
        objective,
        h,
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints=[{'type': 'eq', 'fun': constraint}],
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    h_best = result.x
    c5_best = compute_c5(h_best)

    # Ensure h_best is within [0, 1] and sum is correct
    h_best = np.clip(h_best, 0, 1)
    h_best = h_best * (target_sum / np.sum(h_best)) if np.sum(h_best) != 0 else h_best

    return h_best, c5_best, n_points
```
