    # Objective function: max over k of the overlap integral
    def objective(h):
        nonlocal n_points
        dx = 2 / n_points
        k_values = np.arange(0, n_points * dx, dx)
        max_overlap = 0.0
        for k in k_values:
            if k >= 2:
                continue
            overlap = 0.0
            for i in range(n_points):
                x_i = i * dx
                x_i_plus_k = x_i + k
                if x_i_plus_k >= 2:
                    continue
                idx = np.searchsorted(x_new, x_i_plus_k)
                h_i_plus_k = h[idx] if idx < n_points else 0
                overlap += h[i] * (1 - h_i_plus_k) * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap
    
    # Constraints: sum(h) = n / 2
    def constraint(h):
        return np.sum(h) - n / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = Bounds(0, 1)
    
    # Optimization
    start_time = time.time()
