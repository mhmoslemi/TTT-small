import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h_values = np.load('initial_h_values.npy')  # Assume this file exists with the initial guess
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    n = n_points
    
    def overlap(h):
        h = np.array(h)
        signal = np.concatenate([h, np.zeros(n)])
        kernel = 1 - h
        result = np.correlate(signal, kernel, mode='full')
        result = result[n:-n] / dx
        return np.max(result)
    
    def objective(x):
        return overlap(x)
    
    def constraint_func(x):
        return np.sum(x) - (n / 2)
    
    cons = {'type': 'eq', 'fun': constraint_func}
    
    result = optimize.minimize(
        objective,
        initial_h_values,
        bounds=[(0, 1) for _ in range(n)],
        constraints=cons,
        method='L-BFGS-B'
    )
    
    h_values = result.x
    c5_bound = overlap(h_values)
    # Check if we are under time budget and return
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (h_values, c5_bound, n)
    
    # If not, try a larger n_points
    n = 512
    dx = 2.0 / n
    initial_h_values = np.random.rand(n)
    initial_h_values = (initial_h_values / np.sum(initial_h_values)) * (n / 2)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def overlap_large(h):
        h = np.array(h)
        signal = np.concatenate([h, np.zeros(n)])
        kernel = 1 - h
        result = np.correlate(signal, kernel, mode='full')
        result = result[n:-n] / dx
        return np.max(result)
    
    result_large = optimize.minimize(
        overlap_large,
        initial_h_values,
        bounds=[(0, 1) for _ in range(n)],
        constraints=cons,
        method='L-BFGS-B'
    )
    
    h_values = result_large.x
    c5_bound = overlap_large(h_values)
    return (h_values, c5_bound, n)

start_time = time.time()
