def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    
    # Start from initial_h_values or random if not available
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.minimum(1.0, np.maximum(0.0, h))
        h = np.sort(h)
        h = np.interp(np.linspace(0, 1, n_points), np.sort(h), np.arange(n_points))
        h = np.sort(h)
    
    # Ensure sum is n_points / 2
    h = np.sort(h)
    h = np.interp(np.linspace(0, 1, n_points), np.sort(h), np.arange(n_points))
    h = np.sort(h)

    h = np.sort(h)
    h = np.interp(np.linspace(0, 1, n_points), np.sort(h), np.arange(n_points))

    h = np.interp(np.linspace(0, 1, n_points), np.sort(h), np.arange(n_points))

    h = np.sort(h)
    
    # Function to compute C5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Optimization function
    def objective(h):
        return compute_c5(h)
    
    # Constraint: sum(h) * dx = 1 (which is sum(h) = n_points / 2)
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    
    # Initial guess
    h0 = h.copy()
    
    # Optimization settings
    max_iter = 100
    result = minimize(objective, h0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], constraints=cons, options={'maxiter': max_iter})
    
    h_values = result.x
    best_c5 = compute_c5(h_values)
    
    # Early stopping if budget is exceeded (simulate)
    elapsed_time = time.time() - time.time()  # Dummy, in real code this would be tracked
    if elapsed_time > budget_s:
        return (h_values, best_c5, n_points)
    
    return (h_values, best_c5, n_points)
