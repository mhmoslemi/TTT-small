import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate
import time

initial_h_values = np.load('initial_h_values.npy')

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h = np.array(initial_h_values[:n_points], dtype=np.float64)
    h = h / np.sum(h) * n_points / 2
    h = np.clip(h, 0, 1)

    def compute_C5(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        h_func = np.interp(np.linspace(0, 2, 1000), np.arange(n_points), h)
        max_overlap = 0
        for k in range(1, 100):
            shifted_h = np.roll(h_func, k)
            overlap = np.sum(h_func * (1 - shifted_h)) * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap

    def objective(x):
        global h
        h = np.clip(x, 0, 1)
        h = h * n_points / 2
        return compute_C5(h)

    def constraint(x):
        return np.sum(x) - n_points / 2

    cons = {'type': 'eq', 'fun': constraint}
    bounds = [(0, 1) for _ in range(n_points)]

    start_time = time.time()

    try:
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, jac=False)
        h_best = result.x
        c5_best = result.fun
    except Exception as e:
        print(f"Optimization failed: {e}")
        h_best = h
        c5_best = compute_C5(h)

    elapsed = time.time() - start_time

    if elapsed > budget_s:
        print("Time exceeded")
        return (h, c5_best, n_points)

    return (h_best, c5_best, n_points)
