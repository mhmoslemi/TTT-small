import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Reasonable number for speed and accuracy
    dx = 2.0 / n_points
    initial_h_values = np.array([0.5] * n_points)  # Placeholder for initial values

    # Start from the given initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h_values = np.array(initial_h_values)
    else:
        initial_h_values = np.array([0.5] * n_points)

    def objective(h):
        h = h.reshape(-1)
        h = np.clip(h, 0, 1)
        offset = np.concatenate([h, np.zeros_like(h)])
        h_1_minus = 1 - h
        result = np.correlate(h_1_minus, h, mode='full')
        c5 = np.max(result * dx)
        return c5

    def constraint(h):
        h = h.reshape(-1)
        return np.sum(h) - n_points / 2  # Enforce integral constraint

    # Initial guess based on given or placeholder
    initial_guess = initial_h_values.reshape(-1).copy()
    initial_guess = np.clip(initial_guess, 0, 1)
    initial_guess = initial_guess + np.random.normal(0, 0.1, initial_guess.shape)
    initial_guess = np.clip(initial_guess, 0, 1)
    initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)

    bounds = Bounds(0, 1)

    result = minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 1000},
        tol=1e-6
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = objective(best_h)

    # Final check to ensure integral constraint
    final_sum = np.sum(best_h) * dx
    print(f"Final sum: {final_sum}")
    best_h = np.clip(best_h, 0, 1)

    return best_h, best_c5, n_points
