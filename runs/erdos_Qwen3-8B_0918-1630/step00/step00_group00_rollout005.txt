import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize.optimize import OptimizeResult
import time
import random

# Preloaded initial_h_values (n=45)
initial_h_values = np.array([...]) # Placeholder for actual initial values

def evaluate_erdos_solution(h, n_points):
    dx = 2 / n_points
    h = np.array(h)
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Multiply by dx to get integral values
    correlation_values = corr * dx
    # The overlap integral is max over all shifts
    return np.max(correlation_values)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Try with different n_points
    n_points_list = [200, 400, 600]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for n in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Initialize with a perturbed version of the initial solution
        if hasattr(initial_h_values, '__len__') and len(initial_h_values) == n:
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n)
            h = np.clip(h, 0, 1)
        
        # Normalize h to satisfy the integral constraint
        h = h / np.sum(h) * (n / 2)
        
        # Define bounds for each h[i]
        bounds = [(0, 1) for _ in range(n)]
        
        # Define the objective function
        def objective(x):
            nonlocal n, best_c5
            x = np.clip(x, 0, 1)
            corr = np.correlate(x, 1 - x, mode='full')
            c5 = np.max(corr * (2 / n))
            return c5
        
        # Optimization
        res = minimize(objective, h, method='Nelder-Mead', bounds=bounds, options={'maxiter': 100, 'xatol': 1e-6})
        
        # Update best result
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
                best_n = n
        
    # Return best solution found
    return (best_h, best_c5, best_n)
