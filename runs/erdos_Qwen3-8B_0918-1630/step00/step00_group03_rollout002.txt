import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
import time
import copy

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    def objective(h):
        n = len(h)
        dx = 2.0 / n
        h_corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(h_corr) * dx
        return c5

    initial_h = copy.copy(initial_h_values)
    n_points = len(initial_h)
    dx = 2.0 / n_points
    best_c5 = objective(initial_h)
    best_h = initial_h
    best_h_values = initial_h_values

    def evaluate(h_values):
        nonlocal best_c5, best_h_values
        h = np.array(h_values, dtype=np.float64)
        if np.any(h < 0) or np.any(h > 1):
            return float('inf')
        c5_val = objective(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h_values = h.copy()
        return c5_val

    start_time = time.time()

    # Try with initial_h_values
    result = evaluate(initial_h_values)
    if result < best_c5:
        best_c5 = result
        best_h_values = initial_h_values

    # Try with smaller n_points
    for n in [100, 200, 300, 400, 500]:
        if n < n_points:
            continue
        new_h = np.zeros(n)
        for i in range(n_points):
            new_h[i * n // n_points] = initial_h_values[i]
        new_h = np.array([new_h])
        if np.sum(new_h) * 2.0 / n > 1.0:
            new_h = np.array([new_h[0] * (n / n_points)])
        new_h = np.array([new_h], dtype=np.float64)[0]
        if np.any(new_h < 0) or np.any(new_h > 1):
            continue
        c5 = objective(new_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h_values = new_h
        print(f"Evaluated n={n}, C5={c5}")

    # Try with small perturbations
    for p in [0.1, 0.2, 0.3]:
        perturbed = initial_h_values * (1 + np.random.normal(0, p, n_points))
        perturbed = np.clip(perturbed, 0, 1)
        c5 = objective(perturbed)
        if c5 < best_c5:
            best_c5 = c5
            best_h_values = perturbed

    # Try with binary step function
    binary_h = np.zeros(n_points)
    binary_h[:n_points//2] = 1
    c5 = objective(binary_h)
    if c5 < best_c5:
        best_c5 = c5
        best_h_values = binary_h

    # Try with optimization
    for _ in range(10):
        if time.time() - start_time > budget_s - 5:
            break
        bounds = Bounds(0, 1)
        result = opt.minimize(objective, initial_h_values, method='L-BFGS-B', bounds=bounds)
        if result.success:
            best_c5 = result.fun
            best_h_values = result.x

    if time.time() - start_time > budget_s - 5:
        print(f"Time budget exceeded, returning best solution found so far")
    else:
        print(f"Optimization complete with C5 = {best_c5}")

    return best_h_values, best_c5, len(best_h_values)
