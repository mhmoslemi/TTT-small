    # Objective function with penalty for constraint violation
    def objective(h):

        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess
    h0 = initial_h.copy()

    # Optimization settings
    method = 'L-BFGS-B'  # Efficient for box constraints
    maxiter = 1000
    time_remaining = budget_s

    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    time_start = time.time()

    for _ in range(10):
        if time.time() - time_start > time_remaining:
            break
        
        result = minimize(
            objective,
            h0,
            bounds=bounds,
            method=method,
            options={'maxiter': maxiter, 'disp': False}
        )
        
        if result.success:
            h_candidate = result.x
            c5_candidate = evaluate_erdos_solution(h_candidate)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
            h0 = h_candidate
        else:
            break

    # Try small perturbations to avoid local optima
    for _ in range(5):
        if time.time() - time_start > time_remaining:
            break
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h_candidate = np.clip(h0 + perturbation, 0, 1)
        c5_candidate = evaluate_erdos_solution(h_candidate)
        if c5_candidate < best_c5:
            best_h = h_candidate
            best_c5 = c5_candidate
