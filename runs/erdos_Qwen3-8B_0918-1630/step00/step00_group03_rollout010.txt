import numpy as np
import scipy.optimize as optimize
from scipy.signal import correlate
import time

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Assuming this is provided with correct data

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.array(h)
    result = correlate(h, 1 - h, mode='full') * dx
    return np.max(result)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced for speed
    dx = 2.0 / n_points
    initial_h = initial_h_values[:n_points] if len(initial_h_values) > n_points else initial_h_values
    initial_h = np.clip(initial_h[:n_points], 0, 1)
    sum_h = np.sum(initial_h) * dx
    adjustment = (1 - sum_h) / dx
    initial_h[0] += adjustment if adjustment > 0 else -adjustment
    initial_h = np.clip(initial_h, 0, 1)
    initial_h[0] = np.clip(initial_h[0], 0, 1)

    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        adjustment = (1 - sum_h) / dx
        if adjustment > 1e-8:
            h[0] += adjustment
        elif adjustment < -1e-8:
            h[0] += adjustment
        h = np.clip(h, 0, 1)
        max_overlap = evaluate_erdos_solution(h)
        return max_overlap

    def constraint(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * dx
        adjustment = (1 - sum_h) / dx
        if adjustment > 1e-8:
            h[0] += adjustment
        elif adjustment < -1e-8:
            h[0] += adjustment
        h = np.clip(h, 0, 1)
        return np.sum(h) * dx - 1

    # Initial guess
    h0 = initial_h.copy()
    h0 = np.clip(h0, 0, 1)
    sum_h = np.sum(h0) * dx
    adjustment = (1 - sum_h) / dx
    if adjustment > 1e-8:
        h0[0] += adjustment
    elif adjustment < -1e-8:
        h0[0] += adjustment
    h0 = np.clip(h0, 0, 1)

    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        h0,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxfun': 1000, 'maxiter': 1000},
        tol=1e-8
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h) * dx
    adjustment = (1 - sum_h) / dx
    if adjustment > 1e-8:
        best_h[0] += adjustment
    elif adjustment < -1e-8:
        best_h[0] += adjustment
    best_h = np.clip(best_h, 0, 1)

    best_c5 = evaluate_erdos_solution(best_h)

    # Ensure we stay within time budget
    t0 = time.time()
    while time.time() - t0 < budget_s:
        # Try different initializations for diversity
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed = best_h + perturbation
        perturbed = np.clip(perturbed, 0, 1)
        sum_h = np.sum(perturbed) * dx
        adjustment = (1 - sum_h) / dx
        if adjustment > 1e-8:
            perturbed[0] += adjustment
        elif adjustment < -1e-8:
            perturbed[0] += adjustment
        perturbed = np.clip(perturbed, 0, 1)

        result = optimize.minimize(
            objective,
            perturbed,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxfun': 500, 'maxiter': 500},
            tol=1e-8
        )

        candidate_h = result.x
        candidate_h = np.clip(candidate_h, 0, 1)
        sum_h = np.sum(candidate_h) * dx
        adjustment = (1 - sum_h) / dx
        if adjustment > 1e-8:
            candidate_h[0] += adjustment
        elif adjustment < -1e-8:
            candidate_h[0] += adjustment
        candidate_h = np.clip(candidate_h, 0, 1)
        candidate_c5 = evaluate_erdos_solution(candidate_h)

        if candidate_c5 < best_c5:
            best_c5 = candidate_c5
            best_h = candidate_h

    return best_h.tolist(), best_c5, n_points
