```python
import numpy as np
import time
from scipy.optimize import minimize, Bounds
from functools import partial

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess with fewer samples
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    
    # Define the cost function
    def cost(h_vals):
        h = np.array(h_vals, dtype=np.float64)
        n = len(h)
        max_int = 0.0
        for k in range(1, n):
            idx = np.arange(n - k)
            val = h[:n - k] * (1 - h[k:])
            max_int = max(max_int, np.sum(val) * dx)
        return max_int
    
    # Bounds: each h[i] must be in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization
    result = minimize(
        cost,
        initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    # Check if we have a better solution than the initial one
    best_h = result.x
    best_c5 = cost(best_h)
    
    # Generate a higher-resolution version to get a better estimate of C5
    n_high = 400
    h_high = np.zeros(n_high)
    for i in range(n_high):
        x = i * 2.0 / n_high
        h_high[i] = best_h[np.argmin(np.abs(initial_h - x))]
    
    # Final evaluation with the high-resolution h
    final_c5 = cost(h_high)
    
    # Return the best solution found
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print(f"Warning: Exceeded time budget of {budget_s} seconds.")
    
    return best_h, final_c5, n_points
```
