                j = (i + k) % n_points

        return compute_c5(h)
    
    def constraint(h):

    cons = ({'type': 'eq', 'fun': constraint})
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    while time.time() - start_time < budget_s:
        res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x.copy()
        h = np.random.uniform(0, 1, n_points)
