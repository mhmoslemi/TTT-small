```python
import numpy as np
import matplotlib.pyplot as plt
import time
from scipy.optimize import minimize
from scipy.stats import truncnorm

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100
    max_iter = 1000
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    def objective(h):
        nonlocal best_c5
        t0 = time.time()
        t = t0
        overlaps = np.correlate(h, 1 - h, mode="full")
        c5 = max(overlaps) * (2.0 / len(h))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    def generate_initial_h(n):
        h = np.zeros(n)
        total = n / 2
        h[:int(total)] = 1.0
        return h
    
    def perturb_h(h, perturb_strength=0.1):
        idxs = np.random.choice(len(h), size=10, replace=False)
        for idx in idxs:
            delta = perturb_strength * (np.random.rand() * 2 - 1)
            h[idx] = np.clip(h[idx] + delta, 0, 1)
        return h
    
    h = generate_initial_h(n_points)
    for _ in range(max_iter):
        t0 = time.time()
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            print(f"New best C5: {best_c5:.6f}")
        # Perturb the solution
        h = perturb_h(h)
        if time.time() - t0 > budget_s:
            break
    
    return best_h, best_c5, len(best_h)
```
