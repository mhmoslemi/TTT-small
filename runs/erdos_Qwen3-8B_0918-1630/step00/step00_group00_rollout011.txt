import numpy as np
import time
import scipy.optimize as optimize
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h_values = evaluate_erdos_solution()
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()
    
    def objective(h_values):
        h_full = np.concatenate((h_values, h_values[:5]))  # extend to avoid wrap-around
        corr = np.correlate(h_full, 1 - h_full, mode="full")
        c5 = np.max(corr * dx)
        return c5
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint},
                      options={'disp': False, 'maxiter': 1000})
    
    best_h = result['x']
    best_c5 = objective(best_h)
    
    if best_c5 < 0.380876:
        print(f"Improved bound found: {best_c5}")
    
    return best_h, best_c5, len(best_h)
