import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess
    initial_h = initial_h_values
    n_points = len(initial_h)
    
    # Rescale initial_h to have integral 1
    dx = 2.0 / n_points
    initial_h = np.array(initial_h, dtype=float)
    initial_h = initial_h / (np.sum(initial_h) * dx)
    
    # Objective function
    def obj_func(h_flat):
        h = h_flat.reshape(n_points)
        # Compute (1 - h) for overlap
        one_minus_h = 1.0 - h
        # Compute all shifted correlations
        corr = np.correlate(h, one_minus_h, mode='full')
        # Get the maximum value
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Constraints
    def cons_func(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    cons = LinearConstraint(np.ones(n_points), lb=0, ub=1, name='integral')
    
    # Bounds for each h[i]
    bounds = Bounds(0, 1)
    
    # Initial guess
    x0 = initial_h.flatten()
    
    # Optimization settings
    start_time = time.time()
    
    result = minimize(
        fun=obj_func,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    end_time = time.time()
    elapsed_time = end_time - start_time
    
    if elapsed_time > budget_s:
        print(f"Time exceeded budget. Stopping early.")
        return (initial_h_values, obj_func(x0), n_points)
    
    best_h = result.x
    h_values = best_h.reshape(n_points)
    c5_bound = result.fun
    n_points = len(h_values)
    
    return (h_values, c5_bound, n_points)
