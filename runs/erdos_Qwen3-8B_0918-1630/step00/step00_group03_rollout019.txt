import numpy as np
import scipy.signal as signal
import time
import random

initial_h_values = np.load('initial_h_values.npy')

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    conv = signal.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=60, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 256
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None

    initial_h_values = np.load('initial_h_values.npy')
    for _ in range(10):
        h = np.copy(initial_h_values)
        for i in range(n_points):
            h[i] = np.random.uniform(0, 1)
        total = np.sum(h) * dx
        scale = (1.0 / total) if total != 0 else 1.0
        h *= scale
        
        conv = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Try perturbing the initial guess
    for _ in range(5):
        h = np.copy(initial_h_values)
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h[i] = np.random.uniform(0, 1)
        total = np.sum(h) * dx
        scale = (1.0 / total) if total != 0 else 1.0
        h *= scale
        
        conv = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(conv) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5, n_points
