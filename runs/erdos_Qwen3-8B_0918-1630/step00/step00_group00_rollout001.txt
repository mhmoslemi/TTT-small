import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.sparse import diags
from scipy.linalg import lu_factor, lu_solve

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 99  # 99 samples for faster evaluation and optimization
    dx = 2.0 / n_points
    initial_h = np.array([0.5] * n_points)  # Start with uniform distribution
    if 'initial_h_values' in kwargs:
        initial_h = kwargs['initial_h_values']
    initial_h = np.clip(initial_h, 0.0, 1.0)
    sum_h = np.sum(initial_h)
    if sum_h != n_points / 2:
        initial_h = np.array([0.5] * n_points)

    def compute_c5(h):
        h = np.clip(h, 0.0, 1.0)
        h = np.concatenate([h, h])
        result = np.zeros(n_points + 1)
        for shift in range(1, n_points + 1):
            if shift == n_points:
                continue
            result[shift] = correlate(h, 1 - h[shift:n_points + shift], mode="full")[n_points - shift]
        max_overlap = np.max(result) * dx
        return max_overlap

    def objective(h):
        return compute_c5(h)

    def gradient_step(h, step_size=0.01, cap=0.01):
        h = h.copy()
        for i in range(n_points):
            if h[i] < 1e-6:
                continue
            overlap = np.zeros(n_points)
            for k in range(1, n_points):
                if i + k < n_points:
                    overlap[k] = h[i] * (1 - h[i + k])
                if i - k >= 0:
                    overlap[k] += h[i] * (1 - h[i - k])
            max_overlap = np.max(overlap)
            if max_overlap > 1e-6:
                if i + 1 < n_points and h[i + 1] < 1:
                    h[i + 1] = np.clip(h[i + 1] + step_size * h[i], 0.0, 1.0)
                if i - 1 >= 0 and h[i - 1] > 0:
                    h[i - 1] = np.clip(h[i - 1] - step_size * h[i], 0.0, 1.0)
                h[i] = np.clip(h[i] - step_size * max_overlap, 0.0, 1.0)
        return h

    best_h = initial_h.copy()
    best_c5 = compute_c5(best_h)
    time_start = time.time()
    for _ in range(1000):
        current_time = time.time()
        remaining_time = budget_s - (current_time - time_start)
        if remaining_time < 1:
            break
        h = gradient_step(best_h, step_size=0.05)
        h = np.clip(h, 0.0, 1.0)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
        if np.abs(best_c5 - 0.379) < 1e-6:
            break
    return best_h, best_c5, n_points
