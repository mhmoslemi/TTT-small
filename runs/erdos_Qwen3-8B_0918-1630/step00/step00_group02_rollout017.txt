```python
import numpy as np
import scipy.optimize as optimize
import time

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # This is a pre-loaded initial guess

def evaluate_erdos_solution(h_values, dx):
    n_points = len(h_values)
    x = np.linspace(0, 2, n_points)
    c5 = 0.0
    max_k = 2
    for k in range(-int(max_k / dx), int(max_k / dx) + 1):
        h_shifted = np.roll(h_values, k)
        overlap = np.sum(h_values * (1 - h_shifted)) * dx
        c5 = max(c5, overlap)
    return c5

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Define initial settings
    max_n_points = 100  # Maximum allowed n_points for the budget
    dx = 2.0 / max_n_points
    x = np.linspace(0, 2, max_n_points)
    
    # Start with initial configuration
    h = initial_h_values.copy()
    h = np.interp(h, [0, len(h) - 1], [0, 2])  # Apply to [0, 2] domain
    h = np.interp(h, [0, 2], [0, 1])  # Rescale to [0, 1]
    h = np.interp(h, [0, 2], [0, 1])  # Ensure the h values are in [0, 1]
    
    # Adjust to exactly fit the required total integral
    integral = np.sum(h) * dx
    scaling = 1.0 / integral if integral != 0 else 1.0
    h = np.clip(h * scaling, 0.0, 1.0)
    
    # Define the function to minimize
    def objective(h_vals):
        return evaluate_erdos_solution(h_vals, dx)
    
    # Define the constraints
    def constraint(h_vals):
        return np.sum(h_vals) * dx - 1.0
    
    # Set up optimization problem
    bounds = [(0.0, 1.0) for _ in range(max_n_points)]
    
    try:
        result = optimize.minimize(
            objective,
            x0=h,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
    except:
        # Fallback to random perturbation if optimization fails
        result = optimize.minimize(
            objective,
            x0=np.random.uniform(0, 1, max_n_points),
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 100}
        )
    
    h_opt = result.x
    c5_opt = evaluate_erdos_solution(h_opt, dx)
    
    return h_opt, c5_opt, max_n_points
```
