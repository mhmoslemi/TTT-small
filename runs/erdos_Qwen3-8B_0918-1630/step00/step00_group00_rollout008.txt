import numpy as np
import numpy.random as npr
import scipy.optimize as opt
import time

initial_h_values = np.load("initial_h_values.npy")  # Assume this is a pre-loaded array

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Use a smaller grid for faster evaluation
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with the given initial_h_values or random values
    h_initial = initial_h_values[:n_points]
    if len(h_initial) != n_points:
        h_initial = np.random.uniform(0, 1, n_points)
    
    # Ensure the initial h satisfies the integral constraint
    required_sum = n_points / 2
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = (h_initial * required_sum) / np.sum(h_initial)
    
    # Define the objective function
    def objective(x):
        h = x.reshape(n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Define constraints
    def constraint(x):
        return np.sum(x) - required_sum
    
    cons = ({'type': 'eq', 'fun': constraint})
    
    # Setup optimization problem
    x0 = h_initial.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Run optimization
    start_time = time.time()
    result = opt.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8)
    elapsed = time.time() - start_time
    h_values = result.x
    
    # Final evaluation to ensure it meets the constraints
    h_values = np.clip(h_values, 0, 1)
    h_values = (h_values * required_sum) / np.sum(h_values)
    c5_bound = evaluate_erdos_solution(h_values)
    
    return h_values, c5_bound, n_points
