import numpy as np
import scipy.optimize as optimize
import numpy.random as npr
import time

# Pre-imported global variables
initial_h_values = np.array([...])  # Replace with actual initial values

def evaluate_erdos_solution(h_values, n_points, dx, max_k):
    h = np.array(h_values)
    overlap = np.correlate(h, 1 - h, mode="full")
    overlap = overlap * dx
    max_overlap = np.max(overlap[:max_k+1])
    return max_overlap

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 500
    dx = 2.0 / n_points

    # Helper: objective function for optimization
    def objective(h_vec):
        nonlocal best_c5
        h = h_vec
        if np.sum(h) * dx != 1:
            penalty = np.abs(np.sum(h) * dx - 1)
            return 10000 * penalty + np.max(np.correlate(h, 1 - h, mode="full") * dx)
        else:
            return np.max(np.correlate(h, 1 - h, mode="full") * dx)

    # Initial guess - use given initial_h_values or a random one
    initial_guess = initial_h_values if len(initial_h_values) == n_points else np.random.rand(n_points)

    # Run optimization
    result = optimize.minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'ftol': 1e-6, 'maxiter': 100}
    )

    best_h = result.x
    best_c5 = result.fun
    elapsed = time.time() - start_time

    # If time allows, perform a few more steps with simulated annealing
    if elapsed < budget_s - 10:
        from scipy.optimize import dual_annealing

        def anneal_objective(h_vec):
            return objective(h_vec)

        result_anneal = dual_annealing(
            anneal_objective,
            bounds=[(0, 1) for _ in range(n_points)],
            maxiter=100,
            disp=False
        )

        if result_anneal.fun < best_c5:
            best_c5 = result_anneal.fun
            best_h = result_anneal.x

    return (best_h, best_c5, n_points)
