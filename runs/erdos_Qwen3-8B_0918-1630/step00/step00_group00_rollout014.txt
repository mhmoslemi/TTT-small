import numpy as np
import time
import random
from scipy.optimize import minimize

# Pre-imported functions
# evaluate_erdos_solution()
# initial_h_values (n=45 samples)

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_start = 100
    n_points = 100
    dx = 2.0 / n_points
    
    def objective(h_values):
        h = np.array(h_values).astype(float)
        total = np.sum(h)
        if total != n_points / 2:
            return float('inf')
        
        max_overlap = 0.0
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            max_overlap = max(max_overlap, overlap * dx)
        return max_overlap
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    def generate_initial_solution():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        # Normalize to meet the area constraint
        factor = (n_points / 2) / np.sum(h)
        h = np.clip(h * factor, 0, 1)
        return h
    
    # Initial solution
    h_initial = generate_initial_solution()
    
    # Try multiple starting points
    best_c5 = float('inf')
    best_h = h_initial
    
    start_time = time.time()
    remaining_time = budget_s
    
    for _ in range(5):
        if time.time() - start_time > remaining_time:
            break
        h = generate_initial_solution()
        result = minimize(objective, h, constraints=[{'type': 'eq', 'fun': constraint_sum}])
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return (best_h, best_c5, len(best_h))
