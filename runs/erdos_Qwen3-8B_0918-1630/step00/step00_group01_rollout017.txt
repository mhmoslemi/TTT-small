import numpy as np
import scipy.optimize as opt
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Assuming initial_h_values is provided

def objective(h_values, n_points):
    h = h_values.copy()
    if len(h) != n_points:
        raise ValueError(f"Input vector length ({len(h)}) does not match n_points ({n_points})")
    dx = 2.0 / n_points
    # Compute all shifted overlaps
    shifts = np.arange(n_points)
    max_overlap = 0.0
    for k in shifts:
        overlap = 0.0
        for i in range(n_points):
            x = i * dx
            shifted_index = (i + k) % n_points
            overlap += h[i] * (1 - h[shifted_index])
        max_overlap = max(max_overlap, overlap * dx)
    return max_overlap

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Starting with a moderate grid size
    max_iterations = 100
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    # Try a few starting points
    for start in range(3):
        if start == 0:
            h0 = np.load("initial_h_values.npy")  # Use the initial construction
        elif start == 1:
            h0 = np.random.rand(n_points)
            h0 = (h0 - h0.min()) / (h0.max() - h0.min())  # Normalize to [0,1]
        else:
            h0 = np.zeros(n_points)
            h0[:n_points // 2] = 1.0  # Simple step function
        # Scale to meet the area constraint
        h0 = h0 * (n_points / 2) / np.sum(h0)
        h0 = np.clip(h0, 0, 1)

        result = opt.minimize(
            lambda x: objective(x, n_points),
            x0=h0,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'maxiter': max_iterations, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    # Attempt with smaller n_points if time allows
    if budget_s > 10:
        n_points = 50
        h0 = np.load("initial_h_values.npy")
        h0 = h0[:n_points]
        h0 = h0 * (n_points / 2) / np.sum(h0)
        h0 = np.clip(h0, 0, 1)

        result = opt.minimize(
            lambda x: objective(x, n_points),
            x0=h0,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'maxiter': 50, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    # Final check with smaller n_points
    if budget_s > 20:
        n_points = 30
        h0 = np.random.rand(n_points)
        h0 = (h0 - h0.min()) / (h0.max() - h0.min())
        h0 = h0 * (n_points / 2) / np.sum(h0)
        h0 = np.clip(h0, 0, 1)

        result = opt.minimize(
            lambda x: objective(x, n_points),
            x0=h0,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            options={'maxiter': 50, 'disp': False}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, len(best_h)
