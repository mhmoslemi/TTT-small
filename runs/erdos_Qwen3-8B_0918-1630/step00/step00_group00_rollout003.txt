```python
import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds, differential_evolution
from functools import lru_cache
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h, dx: np.max(np.correlate(h, 1 - h, mode="full")) * dx
initial_h_values = np.load("initial_h_values.npy")  # Assume this is a pre-loaded array

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Initial guess
    h = initial_h_values
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Function to compute overlap integral
    def objective(x):
        h_new = x.copy()
        # Ensure values are in [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Compute overlap
        corr = np.correlate(h_new, 1 - h_new, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Bounds for each parameter: [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    initial_guess = h.copy()
    
    # Optimization with L-BFGS-B
    result = minimize(objective, initial_guess, bounds=bounds, method='L-BFGS-B', tol=1e-6, options={'disp': False})
    
    best_h = result.x
    best_c5 = result.fun
    
    # Local search with Nelder-Mead
    result_nm = minimize(objective, best_h, bounds=bounds, method='Nelder-Mead', tol=1e-5, options={'disp': False})
    best_h = result_nm.x
    best_c5 = result_nm.fun
    
    # Final check
    final_c5 = result_nm.fun
    final_h = best_h
    
    # Ensure correct sample size
    n_points = len(final_h)
    dx = 2.0 / n_points
    
    # Return the best solution found
    return (final_h, final_c5, n_points)

if __name__ == "__main__":
    run()
```
