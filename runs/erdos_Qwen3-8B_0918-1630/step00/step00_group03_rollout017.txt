import numpy as np
import time
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = n_points / 2
    
    # Function to evaluate C5
    def evaluate(h):
        h = np.array(h, dtype=np.float64)
        max_overlap = 0.0
        for k in range(1, n_points):
            corr = np.correlate(h, 1 - h, mode="full")
            overlap = corr[k] * dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap
    
    # Simulated annealing
    def objective(h):
        return evaluate(h)
    
    # Initial solution
    h = initial_h.copy()
    current_c5 = evaluate(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Temperature schedule
    temp = 1.0
    cooling_rate = 0.995
    iterations = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Mutate
        i = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h[i] = np.clip(h[i] + delta, 0.0, 1.0)
        # Ensure sum constraint
        excess = np.sum(h) - sum_h
        if excess > 1e-6:
            h[i] -= excess
        elif excess < -1e-6:
            h[i] += -excess
        h[i] = np.clip(h[i], 0.0, 1.0)
        
        # Evaluate
        new_c5 = evaluate(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Cool down
        temp *= cooling_rate
        iterations += 1
    
    return best_h, best_c5, n_points
