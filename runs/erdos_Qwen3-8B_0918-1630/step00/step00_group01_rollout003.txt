import numpy as np
import scipy.optimize as opt
import time
import random

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    initial_h = initial_h_values if 'initial_h_values' in globals() else np.random.uniform(0, 1, n_points)
    initial_h = np.clip(initial_h, 0, 1)
    # Normalize to satisfy the integral constraint
    initial_h = (initial_h * n_points) / 2.0
    initial_h = np.clip(initial_h, 0, 1)
    h_values = initial_h.copy()

    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = (h * n_points) / 2.0
        h = np.clip(h, 0, 1)
        ones_minus_h = 1.0 - h
        corr = np.correlate(h, ones_minus_h, mode='full')
        return np.max(corr * dx)

    def objective(x):
        nonlocal h_values
        h_values = x
        return compute_c5(h_values)

    def gradient(x):
        # Placeholder for gradient calculation; uses finite differences
        eps = 1e-6
        grad = np.zeros_like(x)
        for i in range(len(x)):
            x_plus = x.copy()
            x_plus[i] += eps
            x_minus = x.copy()
            x_minus[i] -= eps
            grad[i] = (objective(x_plus) - objective(x_minus)) / (2 * eps)
        return grad

    # Use L-BFGS-B with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(objective, h_values, bounds=bounds, method='L-BFGS-B', jac=gradient, options={'maxiter': 200})
    best_h = result.x
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, n_points)
