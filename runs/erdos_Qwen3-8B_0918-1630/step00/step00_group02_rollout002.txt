def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 500
    dx = 2.0 / n_points
    initial_h = np.copy(initial_h_values)
    initial_h = np.resize(initial_h, n_points)
    initial_h = np.pad(initial_h, (0, n_points - len(initial_h)), 'constant')
    initial_h = initial_h[:n_points]
    initial_h = np.clip(initial_h, 0.0, 1.0)
    sum_h = np.sum(initial_h)
    initial_h *= (n_points / 2.0) / sum_h if sum_h != 0 else 1.0

    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0.0, 1.0)
        x = np.arange(n_points)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(np.abs(corr) * dx)
        return c5

    def evaluate(h):
        return objective(h)

    # Initial solution
    best_h = np.copy(initial_h)
    best_c5 = evaluate(best_h)

    for _ in range(10):
        time_start = time.time()
        if time.time() - time_start > budget_s:
            break

        # Local search with simulated annealing
        h = np.copy(best_h)
        for _ in range(100):
            if time.time() - time_start > budget_s:
                break
            idx = np.random.randint(n_points)
            h[idx] = np.random.uniform(0.0, 1.0)
            h = np.clip(h, 0.0, 1.0)
            sum_h = np.sum(h)
            if sum_h == 0:
                continue
            h = h * (n_points / 2.0) / sum_h
            c5 = evaluate(h)
            if c5 < best_c5:
                best_h = np.copy(h)
                best_c5 = c5

    return best_h, best_c5, n_points
