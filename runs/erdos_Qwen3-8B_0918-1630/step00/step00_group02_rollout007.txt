    def compute_c5(h):
        n = len(h)
        max_overlap = 0.0
        for k in range(n):
            overlap = 0.0
            for i in range(n):
                x = i * dx
                x_k = x + k * dx
                if x_k > 2.0:
                    continue
                idx = int(x_k / dx)
                if idx >= n:
                    continue
                if idx < 0:
                    continue
                overlap += h[i] * (1.0 - h[idx])
            overlap *= dx
            if overlap > max_overlap:
                max_overlap = overlap
        return max_overlap

    # Objective function to minimize maximum overlap
    def objective(h_flat, n_points, dx):
        h = h_flat.reshape((n_points,))
        max_overlap = compute_c5(h)
        return max_overlap

    # Constraints
    def constraint_sum(h_flat):
        return np.sum(h_flat) - n_points / 2

    # Bounds
    bounds = Bounds(0.0, 1.0)

    # Initial guess
    x0 = np.full(n_points, 0.5)

    # Use optimization to refine the initial guess
    result = opt.minimize(
        lambda x: objective(x, n_points, dx),
        x0,
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 100}
    )

    best_h = result.x
    best_c5 = compute_c5(best_h)
