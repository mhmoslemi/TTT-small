```python
import numpy as np
import time
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.optimize import DifferentialEvolution
from functools import partial

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 4:
            h[i] = 1.0
        elif i > 3 * n_points // 4:
            h[i] = 1.0
    return h.tolist()

def symmetric_initial(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def random_binary(n_points):
    h = np.random.binomial(1, 0.5, n_points)
    h = h / np.sum(h) * n_points / 2
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 800
    dx = 2.0 / n_points

    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)

    initial_guesses = [
        initial_h_values(n_points),
        symmetric_initial(n_points),
        random_binary(n_points)
    ]

    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    best_n = n_points

    for guess in initial_guesses:
        start_time = time.time()
        result = minimize(objective, guess, method='L-BFGS-B', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'maxiter': 100})
        elapsed = time.time() - start_time

        if elapsed > budget_s - 5:
            continue

        h_values = result.x
        c5 = evaluate_erdos_solution(h_values, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values
            best_n = n_points

        if best_c5 < 0.380876:
            break

    if best_c5 < 0.380876:
        return best_h, best_c5, best_n
    else:
        return np.zeros(n_points), best_c5, best_n
```
