import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from deap import base, creator, algorithms
import random

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    h[:n_points//4] = 1.0
    h[n_points//2 - n_points//8 : n_points//2 + n_points//8] = 0.5
    h[-n_points//4:] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 500
    dx = 2.0 / n_points
    
    # Create a population of initial guesses with different structures
    def create_initial_pop(pop_size):
        pop = []
        for _ in range(pop_size):
            h = np.zeros(n_points)
            segment1 = np.random.randint(0, n_points // 2)
            segment2 = np.random.randint(segment1 + 1, n_points)
            segment3 = np.random.randint(segment2 + 1, n_points)
            h[segment1:segment2] = 0.5
            h[segment3:] = 1.0
            h[:segment1] = 1.0
            pop.append(h.tolist())
        return pop
    
    # Genetic algorithm parameters
    POP_SIZE = 20
    MAX_GENERATIONS = 10
    TOURNAMENT_SIZE = 3
    CROSSOVER_RATE = 0.9
    MUTATION_RATE = 0.1
    MUTATION_STRENGTH = 0.1
    
    # Custom fitness function
    def fitness(individual):
        h = np.array(individual)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap),
    
    # Create the DEAP fitness and individual
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    # Genetic algorithm setup
    toolbox = base.Toolbox()
    toolbox.register("select", base.tools.selTournament, fit_attr="fitness", k=TOURNAMENT_SIZE)
    toolbox.register("mate", base.tools.cxTwoPoint)
    toolbox.register("mutate", base.tools.mutUniformInt, low=0, up=1, mutatator=base.tools.mutUniform, indpb=MUTATION_RATE)
    toolbox.register("evaluate", fitness)
    toolbox.register("individual", base.tools.initIterate, base.tools.initRepeat, creator.Individual, lambda: initial_h_values(n_points))
    toolbox.register("population", base.tools.initRepeat, list, toolbox.individual)
    
    # Initialize population
    population = toolbox.population(n=POP_SIZE)
    
    # Local refinement using SLSQP
    def local_refine(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        c5 = np.max(overlap)
        bounds = Bounds(0.0, 1.0)
        constraint = LinearConstraint(np.ones(n_points), lb=n_points / 2, ub=n_points / 2)
        def objective(x):
            return np.max(np.correlate(x, 1.0 - x, mode="full") * dx)
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'maxiter': 50})
        return result.x
    
    # Run genetic algorithm
    start_time = time.time()
    for gen in range(MAX_GENERATIONS):
        offspring = list()
        for _ in range(POP_SIZE):
            ind = toolbox.select(population)
            ind = toolbox.mate(*ind)
            ind[0] = toolbox.mutate(ind[0])
            ind[0].fitness.values = fitness(ind[0])
            offspring.append(ind[0])
        population = offspring
    
    # Evaluate best candidate and refine it locally
    best = min(population, key=lambda x: x.fitness)
    best = local_refine(best)
    
    # Final evaluation
    c5_bound = evaluate_erdos_solution(best, n_points)
    
    # Ensure we do not exceed the budget
    elapsed = time.time() - start_time
    if elapsed > budget_s:
        return np.array(initial_h_values(n_points)), evaluate_erdos_solution(initial_h_values(n_points), n_points), n_points
    
    return best, c5_bound, n_points
