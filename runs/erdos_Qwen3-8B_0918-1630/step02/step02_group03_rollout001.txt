import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import Bounds

def run(seed=42, budget_s=60):
    np.random.seed(seed)

    # Initial guess: use the provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points

    # Define bounds for all h[i]
    bounds = [(0, 1) for _ in range(n_points)]

    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = 1.0 / (sum_h * dx)
        h_scaled = h * scale
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5

    # Optimization with multiple restarts
    best_c5 = float('inf')
    best_h = np.zeros(n_points)

    for restart in range(5):
        # Perturb the initial guess for diversity
        perturb = np.random.normal(0, 0.1, size=n_points)
        initial_perturbed = np.clip(initial_h + perturb, 0, 1)
        initial_scaled = initial_perturbed * (1 / (np.sum(initial_perturbed) * dx))
        initial_scaled = np.clip(initial_scaled, 0, 1)

        result = minimize(
            objective,
            x0=initial_scaled,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8}
        )

        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (1 / (np.sum(current_h) * dx))
        current_h = np.clip(current_h, 0, 1)

        corr = correlate(current_h, 1 - current_h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h

        if time.time() - time.time() > budget_s - 1:
            break

    return best_h, best_c5, len(best_h)
