import numpy as np
import time
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import minimize

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    n_points = 600
    dx = 2.0 / n_points
    
    def build_h_partition(n_segments, n_points):
        segment_length = n_points // n_segments
        h = np.zeros(n_points)
        for i in range(n_segments):
            start = i * segment_length
            end = min((i + 1) * segment_length, n_points)
            if i % 3 == 0:
                h[start:end] = 1.0
            elif i % 3 == 1:
                h[start:end] = 0.5
            else:
                h[start:end] = 0.0
        return h
    
    best_h = np.zeros(n_points)
    best_c5 = 1.0
    
    for n_segments in [5, 6, 7, 8, 9, 10]:
        h = build_h_partition(n_segments, n_points)
        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Add random perturbation to escape local optima
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h = best_h + perturbation
        h = np.clip(h, 0.0, 1.0)
        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Further fine-tuning
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    
    start_time = time.time()
    
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'maxiter': 100})
    
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        return best_h, best_c5, n_points
    
    h_values = result.x
    c5_bound = evaluate_erdos_solution(h_values, n_points)
    
    return h_values, c5_bound, n_points
