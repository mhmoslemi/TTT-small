import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from functools import lru_cache

# Predefined initial pattern for starting points
initial_h_values = np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)])

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    time_start = time.time()
    
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        dx = 2.0 / len(h)
        h = h * (1 / (sum_h * dx))  # Scale to maintain integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Use the initial_h_values as a starting point
    n_initial = len(initial_h_values)
    dx_initial = 2.0 / n_initial
    sum_initial = np.sum(initial_h_values)
    h_initial = initial_h_values * (1 / (sum_initial * dx_initial))
    h_initial = np.clip(h_initial, 0, 1)
    
    # Use SLSQP for constrained optimization
    result_initial = minimize(objective, x0=h_initial, method='SLSQP', bounds=[(0, 1) for _ in range(n_initial)],
                              constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_initial / 2}])
    
    if result_initial.success:
        h_candidate = result_initial.x
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (1 / (np.sum(h_candidate) * 2.0 / len(h_candidate)))
        corr = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr) * (2.0 / len(h_candidate))
        best_c5 = c5_candidate
        best_h = h_candidate
        best_n = len(h_candidate)
    
    # Try different starting patterns in the same time budget
    for pattern in [np.sin(np.linspace(0, np.pi, 100)) ** 2 * 0.5 + 0.5,
                   np.concatenate([np.linspace(0, 1, 50), np.linspace(1, 0, 50)]),
                   np.random.rand(100) * 2 - 1]:
        n_points = len(pattern)
        dx = 2.0 / n_points
        sum_h = np.sum(pattern)
        h_initial = pattern * (1 / (sum_h * dx))
        h_initial = np.clip(h_initial, 0, 1)
        
        result = minimize(objective, x0=h_initial, method='SLSQP', bounds=[(0, 1) for _ in range(n_points)],
                          constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}])
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * 2.0 / len(h_candidate)))
            corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * (2.0 / len(h_candidate))
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n = len(h_candidate)
    
    # Use simulated annealing on the best solution so far
    if best_h is not None:
        def sa_objective(h):
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            dx = 2.0 / len(h)
            h = h * (1 / (sum_h * dx))
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Define bounds for all h[i]
        bounds = [(0, 1) for _ in range(len(best_h))]
        
        # Perform simulated annealing on the best_h
        best_h_sa = best_h.copy()
        current_h = best_h.copy()
        temperature = 1.0
        cooling_rate = 0.9995
        iterations = 1000
        max_time = budget_s - (time.time() - time_start)
        
        for _ in range(iterations):
            if time.time() - time_start > budget_s:
                break
            # Generate a neighbor solution
            idx = np.random.randint(0, len(current_h))
            delta = np.random.uniform(-0.1, 0.1)
            new_h = current_h.copy()
            new_h[idx] += delta
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (1 / (np.sum(new_h) * 2.0 / len(new_h)))
            
            # Compute objective
            obj_current = sa_objective(current_h)
            obj_new = sa_objective(new_h)
            
            # Acceptance probability
            if obj_new < obj_current:
                current_h = new_h
            else:
                prob = np.exp((obj_current - obj_new) / temperature)
                if np.random.rand() < prob:
                    current_h = new_h
            
            # Update temperature
            temperature *= cooling_rate
        
        if sa_objective(current_h) < best_c5:
            best_h = current_h
            best_c5 = sa_objective(current_h)
    
    return best_h, best_c5, len(best_h)
