import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import Bounds
from scipy.optimize import minimize

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial construction provided (n=200)
    initial_h_values = np.load('initial_h_values.npy')
    n_points_initial = len(initial_h_values)
    dx_initial = 2.0 / n_points_initial
    sum_h_initial = np.sum(initial_h_values)
    scale_initial = 1.0 / (sum_h_initial * dx_initial)
    h_initial = initial_h_values * scale_initial
    h_initial = np.clip(h_initial, 0, 1)
    
    def objective(h, dx, n_points):
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        # Scale to maintain integral
        if total == 0:
            return np.inf
        scale = 1.0 / total
        h_scaled = h * scale
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def optimize_h(h0, dx, n_points):
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
        result = minimize(
            lambda h: objective(h, dx, n_points),
            x0=h0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-9, 'maxiter': 500}
        )
        if result.success:
            return result.x
        else:
            return h0
    
    def evaluate(h, dx, n_points):
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        if total == 0:
            return np.inf
        scale = 1.0 / total
        h_scaled = h * scale
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Optimize initial construction
    optimized_initial = optimize_h(h_initial, dx_initial, n_points_initial)
    c5_initial = evaluate(optimized_initial, dx_initial, n_points_initial)
    best_c5 = c5_initial
    best_h = optimized_initial
    best_n = n_points_initial
    
    # Generate more patterns
    patterns = [
        np.concatenate([np.zeros(100), np.ones(100)]),  # Checkerboard
        np.sin(np.linspace(0, np.pi, 200)) ** 2,       # Sinusoidal
        np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)]),  # Linear
        np.random.rand(200) * 2 - 1,                   # Random
    ]
    
    pattern_results = []
    for pattern in patterns:
        n_points = len(pattern)
        dx = 2.0 / n_points
        sum_h = np.sum(pattern)
        scale = 1.0 / (sum_h * dx)
        h_initial = pattern * scale
        h_initial = np.clip(h_initial, 0, 1)
        
        # Optimize
        h_opt = optimize_h(h_initial, dx, n_points)
        c5 = evaluate(h_opt, dx, n_points)
        pattern_results.append((c5, h_opt, n_points))
    
    # Evaluate all pattern results
    for c5, h_opt, n_points in pattern_results:
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n = n_points
    
    # Final refinement
    if best_n > 0:
        dx_final = 2.0 / best_n
        best_h = np.clip(best_h, 0, 1)
        total = np.sum(best_h) * dx_final
        scale = 1.0 / total
        best_h = best_h * scale
        best_c5 = evaluate(best_h, dx_final, best_n)
    
    return best_h, best_c5, len(best_h)
