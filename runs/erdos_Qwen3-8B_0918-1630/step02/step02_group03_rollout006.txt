            if T < T_min:

                # Random perturbation
                indices = np.random.choice(n_points, size=10, replace=False)
                for i in indices:
                    delta = np.random.uniform(-0.1, 0.1)
                    h[i] = np.clip(h[i] + delta, 0, 1)

                total_area = np.sum(h) * dx

                    h = initial_h_scaled.copy()

                new_c5 = objective(h_scaled)
