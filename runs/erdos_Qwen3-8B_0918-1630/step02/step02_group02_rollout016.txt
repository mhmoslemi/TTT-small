import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from itertools import product

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 256
    dx = 2.0 / n_points
    
    # Generate a few diverse initial patterns
    def generate_initial_patterns(n_points):
        patterns = []
        # Constant half (base case)
        h_const = np.zeros(n_points)
        h_const[:n_points//2] = 1.0
        patterns.append(h_const)
        
        # Sinusoidal-like pattern
        h_sinus = np.zeros(n_points)
        for i in range(n_points):
            h_sinus[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
        h_sinus = np.clip(h_sinus, 0, 1)
        patterns.append(h_sinus)
        
        # Two plateaus with a gap
        h_two_plateaus = np.zeros(n_points)
        h_two_plateaus[:(n_points//3)] = 1.0
        h_two_plateaus[(n_points//2 - n_points//6):(n_points//2 + n_points//6)] = 0.5
        h_two_plateaus[-(n_points//3):] = 1.0
        patterns.append(h_two_plateaus)
        
        # Periodic-like pattern with multiple plateaus
        h_periodic = np.zeros(n_points)
        for i in range(n_points):
            h_periodic[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / (n_points//2))
        h_periodic = np.clip(h_periodic, 0, 1)
        patterns.append(h_periodic)
        
        # Random perturbation of the base case
        h_perturb = np.zeros(n_points)
        h_perturb[:n_points//2] = 1.0
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h_perturb[i] = np.random.uniform(0, 1)
        patterns.append(h_perturb)
        
        # Binary-like pattern with random intervals
        h_binary = np.random.binomial(1, 0.5, n_points)
        h_binary = np.where(h_binary == 1, 1.0, 0.0)
        h_binary = h_binary * (1.0 / np.sum(h_binary)) * n_points / 2
        patterns.append(h_binary)
        
        return patterns
    
    initial_patterns = generate_initial_patterns(n_points)
    initial_candidates = [p.tolist() for p in initial_patterns]
    best_c5 = np.inf
    best_h = initial_candidates[0]
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    
    for seed in range(5):
        np.random.seed(seed)
        h_initial = initial_candidates[np.random.choice(len(initial_candidates))]
        start_time = time.time()
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=[constraint], options={'ftol': 1e-12, 'maxiter': 300})
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            continue
        
        h_values = result.x
        c5 = evaluate_erdos_solution(h_values, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values
    
    return best_h, best_c5, n_points
