```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from skopt import BayesSearchSpace, BayesianOptimization
from skopt.utils import use_positional_args

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 128
    dx = 2.0 / n_points
    
    # Generate an initial function with more complex structure
    h_initial = np.zeros(n_points)
    h_initial[:n_points//4] = 1.0
    h_initial[n_points//2 - n_points//8 : n_points//2 + n_points//8] = 0.5
    h_initial[-n_points//4:] = 1.0
    h_initial = h_initial.tolist()
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    
    @use_positional_args
    def wrapped_objective(params):
        h = params
        return objective(h)
    
    search_space = BayesSearchSpace()
    for i in range(n_points):
        search_space.add_dimension((0.0, 1.0))
    
    bo = BayesianOptimization(
        wrapped_objective,
        search_space,
        n_points=20,
        n_iter=50,
        acq_func='ei',
        random_state=seed
    )
    
    els = []
    start_time = time.time()
    
    for _ in range(50):
        if time.time() - start_time > budget_s:
            break
        res = bo.suggest_and_notify()
        els.append(res)
    
    best_params = els[0]
    for param in els:
        if wrapped_objective(param) < wrapped_objective(best_params):
            best_params = param
    
    h_values = best_params
    c5_bound = evaluate_erdos_solution(h_values, n_points)
    
    return h_values, c5_bound, n_points
```
