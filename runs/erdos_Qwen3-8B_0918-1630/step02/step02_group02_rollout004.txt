import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from pyswarm import pso  # Particle Swarm Optimization

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    h[:n_points//3] = 1.0
    h[n_points//3:n_points//3*2] = 0.5
    h[-n_points//3:] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 500
    dx = 2.0 / n_points
    
    # Initial guess based on known constructions
    h_initial = np.zeros(n_points)
    h_initial[:n_points//3] = 1.0
    h_initial[n_points//3:n_points//3*2] = 0.5
    h_initial[-n_points//3:] = 1.0
    h_initial = h_initial.tolist()
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    
    def run_pso():
        start_time = time.time()
        result = pso(objective, bounds=(0,1), f_ieqcons=None, seed=seed, maxiter=100, np=20, swarmsize=20, debug=False)
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            return np.array(h_initial), evaluate_erdos_solution(h_initial, n_points), n_points
        return result.x, result.fval, n_points
    
    try:
        h_values, c5_bound, n_points = run_pso()
        return h_values, c5_bound, n_points
    except:
        return np.array(h_initial), evaluate_erdos_solution(h_initial, n_points), n_points
