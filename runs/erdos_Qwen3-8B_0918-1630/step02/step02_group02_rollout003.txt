import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from sklearn.gaussian_process import GaussianProcessRegressor
from sklearn.gaussian_process.kernels import RBF, ConstantKernel

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    h[:n_points//4] = 1.0
    h[n_points//2 - n_points//8 : n_points//2 + n_points//8] = 0.5
    h[-n_points//4:] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try with different point numbers for better performance
    n_points = 600
    dx = 2.0 / n_points
    
    # Initial guess with alternating patterns and density constraint
    h_initial = np.zeros(n_points)
    h_initial[:n_points//4] = 1.0
    h_initial[n_points//2 - n_points//8 : n_points//2 + n_points//8] = 0.5
    h_initial[-n_points//4:] = 1.0
    h_initial = h_initial.tolist()
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    
    # Start with SLSQP
    start_time = time.time()
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'maxiter': 200})
    elapsed = time.time() - start_time
    
    # If time is left, do a local search with random perturbations
    if elapsed < budget_s:
        h = result.x
        h_perturbed = h.copy()
        for _ in range(5):
            idx = np.random.choice(n_points, 10, replace=False)
            h_perturbed[idx] = np.clip(h_perturbed[idx] + np.random.uniform(-0.1, 0.1), 0.0, 1.0)
            constraint_val = np.sum(h_perturbed)
            if constraint_val < n_points / 2:
                h_perturbed[idx] = np.clip(h_perturbed[idx] + (n_points / 2 - constraint_val) / len(idx), 0.0, 1.0)
            else:
                h_perturbed[idx] = np.clip(h_perturbed[idx] + (n_points / 2 - constraint_val) / len(idx), 0.0, 1.0)
            result = minimize(objective, h_perturbed, method='L-BFGS-B', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'maxiter': 50})
            h = result.x
    
    h_values = h
    c5_bound = evaluate_erdos_solution(h_values, n_points)
    
    return h_values, c5_bound, n_points
