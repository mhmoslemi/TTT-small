```python
import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Define initial patterns
    def checkerboard(n):
        return np.concatenate([np.zeros(n//2), np.ones(n//2)])
    
    def sine_pattern(n):
        x = np.linspace(0, np.pi, n)
        return np.sin(x)**2
    
    def linear_ramp(n):
        return np.concatenate([np.linspace(0, 1, n//2), np.linspace(1, 0, n//2)])
    
    def random_noise(n):
        return np.random.rand(n) * 2 - 1
    
    initial_patterns = [
        checkerboard, sine_pattern, linear_ramp, random_noise
    ]
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    
    for pattern_func in initial_patterns:
        for n_points in [200, 300, 400, 500, 600]:
            try:
                if n_points < 10:
                    continue
                pattern = pattern_func(n_points)
                dx = 2.0 / n_points
                sum_h = np.sum(pattern)
                
                # Scale pattern to maintain integral constraint
                scale = 1.0 / (sum_h * dx)
                h_initial = pattern * scale
                h_initial = np.clip(h_initial, 0, 1)
                
                # Define objective function
                def objective(h):
                    h = np.clip(h, 0, 1)
                    sum_h_val = np.sum(h)
                    scale = 1.0 / (sum_h_val * dx)
                    h_scaled = h * scale
                    corr = correlate(h_scaled, 1 - h_scaled, mode='full')
                    c5 = np.max(corr) * dx
                    return c5
                
                # Define bounds for all h[i]
                bounds = [(0, 1) for _ in range(n_points)]
                
                # Use L-BFGS-B for unconstrained optimization with bounds
                result = minimize(objective, x0=h_initial, method='L-BFGS-B', bounds=bounds,
                                  constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}])
                
                if result.success:
                    h_candidate = result.x
                    h_candidate = np.clip(h_candidate, 0, 1)
                    sum_h_val = np.sum(h_candidate)
                    scale = 1.0 / (sum_h_val * dx)
                    h_scaled = h_candidate * scale
                    corr = correlate(h_scaled, 1 - h_scaled, mode='full')
                    c5_candidate = np.max(corr) * dx
                    
                    if c5_candidate < best_c5:
                        best_c5 = c5_candidate
                        best_h = h_scaled
                        best_n = n_points
            except:
                continue
            
            if time.time() - time.time() > budget_s:
                break
    
    return best_h, best_c5, len(best_h)
```
