    def objective(h):
        h = np.clip(h, 0, 1)

        c5 = np.max(corr) * dx

    def evaluate_solution(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def generate_initial_guesses(n_guesses=5):
        guesses = []

            h = np.random.rand(n_points)
            h = h * (1 / (np.sum(h) * dx))
            guesses.append(h)
        return guesses

    # Try different initial guesses and keep the best
    best_solution = initial_h_scaled
    best_c5 = evaluate_solution(best_solution)

    guesses = [initial_h_scaled]
    guesses += generate_initial_guesses(4)

    for guess in guesses:

            objective,
            x0=guess,
            method='L-BFGS-B',
