import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import differential_evolution

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            h[i] = 1.0
        elif i < 2 * n_points // 3:
            h[i] = 0.5
        else:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 500
    dx = 2.0 / n_points
    
    # Initial guess with more complex structure
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 3:
            h_initial[i] = 1.0
        elif i < 2 * n_points // 3:
            h_initial[i] = 0.5
        else:
            h_initial[i] = 1.0
    h_initial = h_initial.tolist()
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    
    start_time = time.time()
    
    # Use differential evolution for global optimization
    def objective_for_de(h):
        return objective(h)
    
    result = differential_evolution(objective_for_de, bounds, strategy='best1bin', 
                                    maxiter=100, popsize=30, tol=1e-10, mutation=(0.5, 1), recombination=0.7, 
                                    polish=True, seed=seed)
    
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        return np.array(h_initial), evaluate_erdos_solution(h_initial, n_points), n_points
    
    h_values = result.x
    c5_bound = evaluate_erdos_solution(h_values, n_points)
    
    return h_values, c5_bound, n_points
