    ]

    best_c5 = float('inf')

        # Scale to satisfy integral constraint
        sum_config = np.sum(config) * dx

            h = np.clip(h, 0, 1)

            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        def constraint(h):
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
            sum_h = np.sum(h)
            return sum_h - 1
        
        result = minimize(
            objective,
            x0=config_scaled,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 150, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success:
            current_h = result.x
            current_h = np.clip(current_h, 0, 1)
