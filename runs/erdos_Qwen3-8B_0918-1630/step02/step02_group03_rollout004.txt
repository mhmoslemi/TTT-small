import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from functools import lru_cache
from cma import CMAEs

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Define initial block-based construction with known patterns
    def block_based_h(n_points):
        positions = np.sort(np.random.choice(np.arange(0, n_points), size=5, replace=False))
        block_sizes = np.random.choice([1, 2, 3], size=5)
        h = np.zeros(n_points)
        for i in range(5):
            start = int(positions[i])
            end = start + block_sizes[i]
            if end > n_points:
                end = n_points
            h[start:end] = 1.0
        # Rescale to satisfy integral constraint
        integral = np.sum(h) * (2.0 / n_points)
        h = h * (1.0 / integral)
        return h
    
    # Evaluate the overlap integral
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (1.0 / (np.sum(h) * (2.0 / len(h))))  # Rescale to satisfy integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / len(h))
        return c5
    
    # Define bounds for all h[i]
    def bounds(n_points):
        return [(0, 1) for _ in range(n_points)]
    
    # CMA-ES with adaptive parameters
    def cma_es_optimize(n_points, budget_s):
        start_time = time.time()
        t0 = 0.5
        sigma = 0.5
        cost = np.inf
        best_h = np.random.rand(n_points)
        best_h = best_h * (1.0 / np.sum(best_h) * (2.0 / n_points))
        
        while time.time() - start_time < budget_s:
            cma = CMAEs(
                fitness_function=lambda x: evaluate(x),
                sigma=sigma,
                population_size=15,
                bounds=bounds(n_points),
                maxiter=1,
                seed=seed,
                verb_flag=False
            )
            cma.optimize()
            cma_results = cma.get_result()
            best_h = cma_results[0]
            current_cost = evaluate(best_h)
            if current_cost < cost:
                cost = current_cost
            sigma = max(sigma * 0.99, 0.1)
        
        return best_h, cost
    
    # L-BFGS-B refinement
    def lbfgs_refine(h, n_points, budget_s):
        start_time = time.time()
        bounds = [(0, 1) for _ in range(n_points)]
        h_val = h.copy()
        h_val = h_val * (1.0 / np.sum(h_val) * (2.0 / n_points))
        
        def objective(x):
            x = np.clip(x, 0, 1)
            x = x * (1.0 / np.sum(x) * (2.0 / n_points))
            corr = correlate(x, 1 - x, mode='full')
            c5 = np.max(corr) * (2.0 / len(x))
            return c5
        
        result = minimize(
            objective,
            x0=h_val,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if time.time() - start_time < budget_s:
            return result.x, evaluate(result.x)
        else:
            return h_val, evaluate(h_val)
    
    # Initial block-based construction with 200 samples
    n_points = 200
    initial_h = block_based_h(n_points)
    initial_cost = evaluate(initial_h)
    
    # CMA-ES optimization with reduced steps
    cma_result, cma_cost = cma_es_optimize(n_points, budget_s * 0.7)
    
    # Local L-BFGS refinement
    final_h, final_cost = lbfgs_refine(cma_result, n_points, budget_s * 0.3)
    
    # Final result
    return final_h, final_cost, len(final_h)
