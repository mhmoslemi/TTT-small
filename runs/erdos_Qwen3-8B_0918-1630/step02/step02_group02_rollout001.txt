import numpy as np
import time
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from itertools import product
from sklearn.decomposition import PCA

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 400
    dx = 2.0 / n_points
    
    # Generate an initial function with more complex structure
    h_initial = np.zeros(n_points)
    n_blocks = 10
    block_size = n_points // n_blocks
    for i in range(n_blocks):
        start = i * block_size
        end = start + block_size
        if i % 2 == 0:
            h_initial[start:end] = 1.0
        else:
            h_initial[start:end] = 0.5
    
    # Further refine the initial guess with alternating patterns
    for i in range(n_points):
        if i % 2 == 0:
            h_initial[i] = 1.0
        else:
            h_initial[i] = 0.5
    
    h_initial = h_initial.tolist()
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    
    start_time = time.time()
    
    # Use SLSQP with a more sophisticated initial guess and more iterations
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'maxiter': 300})
    
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        return np.array(h_initial), evaluate_erdos_solution(h_initial, n_points), n_points
    
    h_values = result.x
    c5_bound = evaluate_erdos_solution(h_values, n_points)
    
    # Adaptive refinement to further decrease overlap
    def refine_overlap(h, n_refine=5):
        for _ in range(n_refine):
            h = h.copy()
            dx = 2.0 / len(h)
            h_1_minus = 1.0 - h
            correlate = np.correlate(h, h_1_minus, mode='full')
            max_pos = np.argmax(correlate)
            k = max_pos - len(h) + 1
            if k < 0:
                k = 0
            if k > len(h) - 1:
                k = len(h) - 1
            
            # This is a simple improvement heuristics: shift the function to reduce overlap
            h_new = np.roll(h, k)
            h = (h * np.abs(1.0 - h_new) + h_new * np.abs(1.0 - h)) / (np.abs(1.0 - h_new) + np.abs(1.0 - h))
            h = np.clip(h, 0.0, 1.0)
        
        return h
    
    for _ in range(3):
        h_values = refine_overlap(h_values)
    
    c5_bound = evaluate_erdos_solution(h_values, n_points)
    
    return h_values, c5_bound, n_points
