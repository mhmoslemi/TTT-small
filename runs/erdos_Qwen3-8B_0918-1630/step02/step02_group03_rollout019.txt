import numpy as np

# Pre-imports for evaluation
evaluate_erdos_solution = lambda h: np.max(correlate(h, 1 - h, mode='full')) * (2. / len(h))
initial_h_values = np.random.rand(70) * 1

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Define the number of points and initial configurations
    n_points = 500
    sample_size = n_points
    dx = 2.0 / n_points

    # Generate initial configurations
    initial_candidates = []
    initial_candidates.append(initial_h_values)
    for _ in range(10):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * dx
        initial_candidates.append(h)
    
    # Function to evaluate the C5 value for a given h
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Function to optimize a candidate h
    def optimize_h(h_initial):
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
            corr = correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            objective,
            x0=h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-8}
        )
        return result.x

    # Bayesian optimization loop
    def bayesian_optimization():
        def objective(x):
            x = x.reshape(-1, n_points)
            return np.array([evaluate(h) for h in x])

        def fitness(x):
            return objective(x)

        X = np.random.rand(10, n_points)
        y = objective(X)

        model = GaussianProcessRegressor(
            kernel=ConstantKernel(1.0) * RBF(length_scale=1.0, length_scale_bounds=(0.1, 10.0)),
            n_restarts_optimizer=5
        )
        model.fit(X, y)

        best_idx = np.argmin(y)
        best_h = X[best_idx]
        best_value = y[best_idx]

        for _ in range(100):
            # Predict on a grid
            X_new = np.random.rand(10, n_points)
            y_pred, _ = model.predict(X_new, return_std=True)
            acquisition = np.min(y_pred)
            idx = np.argmin(y_pred)
            best_h = X_new[idx]
            best_value = y_pred[idx]
            model.fit(np.vstack((X, X_new)), np.hstack((y, y_pred)))
            if best_value < 0.380876:
                break

        return best_h, best_value

    # Try with initial candidates
    best_h = None
    best_value = float('inf')

    for candidate in initial_candidates:
        h = candidate.copy()
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        value = evaluate(h)
        if value < best_value:
            best_value = value
            best_h = h

    # Perform Bayesian optimization
    best_h, best_value = bayesian_optimization()

    # Final optimization
    final_h = optimize_h(best_h)
    final_value = evaluate(final_h)

    return final_h, final_value, len(final_h)
