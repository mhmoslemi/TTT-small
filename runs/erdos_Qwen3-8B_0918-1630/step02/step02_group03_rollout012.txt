        elapsed = time.time() - start_time
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        print(f"Initial {i+1}/{len(initial_points)}: c5={c5:.8f}, time={elapsed:.2f}s")

    cluster_points = np.array([np.random.rand(n_points) for _ in range(50)])

                h_opt, c5 = optimize_with_bounds(h)
