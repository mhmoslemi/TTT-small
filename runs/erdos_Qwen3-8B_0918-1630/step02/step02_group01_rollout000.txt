import numpy as np

from scipy import stats

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Predefined initial solutions for diversification
    initial_candidates = [
        initial_h_values,
        np.random.uniform(0, 1, 150),
        np.random.uniform(0, 1, 200),
        np.random.uniform(0, 1, 250)
    ]
    
    best_h = None
    best_c5 = float('inf')
    n_points = 200
    
    time_start = time.time()
    
    for candidate in initial_candidates:
        if len(candidate) != n_points:
            candidate = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(candidate)), candidate)
        
        # Scale to satisfy the integral constraint
        sum_h = np.sum(candidate)
        scale = 1.0 / (sum_h * 2.0 / n_points)
        scaled_h = candidate * scale
        scaled_h = np.clip(scaled_h, 0, 1)
        
        def objective(h):
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            dx = 2.0 / n_points
            scaled_h = h * (1 / (sum_h * dx))
            corr = correlate(scaled_h, 1 - scaled_h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Define bounds for all h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(
            objective,
            x0=scaled_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        
        # Rescale to maintain integral constraint
        sum_h = np.sum(current_h)
        dx = 2.0 / n_points
        scaled_h = current_h * (1 / (sum_h * dx))
        scaled_h = np.clip(scaled_h, 0, 1)
        
        corr = correlate(scaled_h, 1 - scaled_h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = scaled_h
    
    if best_h is None:
        # Fallback to a simple uniform distribution
        n_points = 200
        dx = 2.0 / n_points
        best_h = np.ones(n_points) * (1 / n_points) * 2.0
        best_h = np.clip(best_h, 0, 1)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return best_h, best_c5, len(best_h)
