```python
import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from random import uniform, choice
import copy

# Preload initial solution
initial_h_values = np.array([0.0] * 200)

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    start_time = time.time()
    
    # Base parameters
    max_n_points = 400
    population_size = 30
    generations = 200
    mutation_rate = 0.1
    crossover_rate = 0.7
    learning_rate = 0.05
    tolerance = 1e-5
    best_c5 = float('inf')
    best_h = np.zeros(200)
    
    # Initialize population with initial solution and random variations
    population = []
    for _ in range(population_size):
        h = copy.deepcopy(initial_h_values)
        h = h * (np.random.rand(len(h)) * 0.5 + 0.5)
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * 2.0 / len(h)))
        population.append(h)
    
    # Optimization loop
    for gen in range(generations):
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / len(h))
            fitness.append(c5)
        
        # Select top performers
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size // 2]
        top_population = [population[i] for i in top_indices]
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Crossover
            if np.random.rand() < crossover_rate:
                p1 = np.random.choice(top_indices)
                p2 = np.random.choice(top_indices)
                child = (population[p1] + population[p2]) / 2
            else:
                child = np.random.rand(len(population[0])) * 0.5 + 0.5
            
            # Mutation
            for i in range(len(child)):
                if np.random.rand() < mutation_rate:
                    child[i] = max(0, min(1, child[i] + np.random.normal(0, 0.05)))
            
            # Enforce integral constraint
            total = np.sum(child) * 2.0 / len(child)
            child = child * (1 / (total))
            
            new_population.append(child)
        
        # Add diversity by adding random samples
        for _ in range(population_size // 2):
            h = np.random.rand(len(initial_h_values)) * 0.5 + 0.5
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * 2.0 / len(h)))
            new_population.append(h)
        
        population = new_population
        
        # Update best solution
        for i, h in enumerate(population):
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * (2.0 / len(h))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
        # Check time
        if time.time() - start_time > budget_s - 5:
            break
    
    # Refine with gradient descent
    h_refined = best_h
    def objective(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h) * 2.0 / len(h)
        h = h * (1 / (total))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * (2.0 / len(h))
        return c5
    
    bounds = [(0, 1) for _ in range(len(best_h))]
    result = minimize(objective, x0=h_refined, method='L-BFGS-B', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - len(h) / 2}])
    
    if result.success:
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (1 / (np.sum(best_h) * 2.0 / len(best_h)))
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * (2.0 / len(best_h))
    
    return best_h, best_c5, len(best_h)
```
