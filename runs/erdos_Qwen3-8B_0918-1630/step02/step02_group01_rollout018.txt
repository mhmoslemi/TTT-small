import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy import stats
import random

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial guess: use the provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    scale = 1.0 / (sum_h * dx)
    initial_h_scaled = initial_h * scale
    initial_h_scaled = np.clip(initial_h_scaled, 0, 1)
    
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        h = h / (sum_h * dx)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return np.inf
        h = h / (sum_h * dx)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        return sum_h - 1
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def perturb_solution(h, perturbation_factor=0.05):
        h_perturb = h.copy()
        for i in range(len(h_perturb)):
            h_perturb[i] += np.random.uniform(-perturbation_factor, perturbation_factor)
            h_perturb[i] = np.clip(h_perturb[i], 0, 1)
        sum_h = np.sum(h_perturb)
        if sum_h == 0:
            return h
        scale = 1.0 / (sum_h * dx)
        h_perturb = h_perturb * scale
        return h_perturb
    
    def simulated_annealing():
        temp = 1.0
        cooling_rate = 0.995
        max_iter = 1000
        best_h = initial_h_scaled.copy()
        best_c5 = evaluate_overlap(best_h)
        
        for _ in range(max_iter):
            h = perturb_solution(best_h)
            current_c5 = evaluate_overlap(h)
            if current_c5 < best_c5:
                best_h = h
                best_c5 = current_c5
            temp *= cooling_rate
        
        return best_h, best_c5
    
    def local_optimize(h):
        result = minimize(
            objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)
        if sum_h == 0:
            return h
        best_h = best_h / (sum_h * dx)
        return best_h
    
    def optimize_with_restart():
        best_h = initial_h_scaled.copy()
        best_c5 = evaluate_overlap(best_h)
        best_h = local_optimize(best_h)
        best_c5 = evaluate_overlap(best_h)
        
        for _ in range(5):
            perturbed = perturb_solution(best_h)
            perturbed = local_optimize(perturbed)
            c5 = evaluate_overlap(perturbed)
            if c5 < best_c5:
                best_h = perturbed
                best_c5 = c5
        
        return best_h, best_c5
    
    start_time = time.time()
    best_h, best_c5 = optimize_with_restart()
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return best_h, best_c5, len(best_h)
    
    for _ in range(3):
        if elapsed_time > budget_s - 1:
            break
        h_perturb = perturb_solution(best_h)
        h_perturb = local_optimize(h_perturb)
        c5 = evaluate_overlap(h_perturb)
        if c5 < best_c5:
            best_h = h_perturb
            best_c5 = c5
        time.sleep(0.1)
    
    return best_h, best_c5, len(best_h)
