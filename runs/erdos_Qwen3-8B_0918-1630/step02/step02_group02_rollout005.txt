import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from functools import partial

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points and initial guesses
    n_points_list = [100, 200, 300, 400, 500, 600, 700, 800]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    elapsed_time = 0

    for n_points in n_points_list:
        if elapsed_time > budget_s:
            break
        dx = 2.0 / n_points
        
        # Use a variety of initial guesses
        initial_guesses = [
            np.zeros(n_points),
            np.full(n_points, 0.5),
            np.array([1.0 if (i % 4 == 0) else 0.0 for i in range(n_points)]),
            np.array([1.0 if (i % 5 == 0) else 0.0 for i in range(n_points)]),
            np.array([1.0 if (i % 3 == 0) else 0.0 for i in range(n_points)]),
            np.array([1.0 if (i % 10 == 0) else 0.0 for i in range(n_points)]),
            np.array([1.0 if (i % 7 == 0) else 0.0 for i in range(n_points)]),
            np.array([1.0 if (i % 15 == 0) else 0.0 for i in range(n_points)]),
            np.array([1.0 if (i % 25 == 0) else 0.0 for i in range(n_points)]),
            np.array([1.0 if (i % 30 == 0) else 0.0 for i in range(n_points)]),
        ]
        
        for guess in initial_guesses:
            if np.sum(guess) * dx < 1 or np.sum(guess) * dx > 1:
                continue
            
            h_initial = guess.copy()
            h_initial = h_initial.tolist()
            
            def objective(h):
                h = np.array(h)
                h_1_minus = 1.0 - h
                correlate = np.correlate(h, h_1_minus, mode="full")
                overlap = correlate * dx
                return np.max(overlap)
            
            def constraint_sum(h):
                return np.sum(h) - n_points / 2
            
            bounds = Bounds(0.0, 1.0)
            constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
            
            start_time = time.time()
            
            result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'maxiter': 200})
            elapsed = time.time() - start_time
            elapsed_time += elapsed
            
            if elapsed > budget_s:
                break
            
            h_values = result.x
            current_c5 = evaluate_erdos_solution(h_values, n_points)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_values
                best_n = n_points
                
    return best_h, best_c5, best_n
