        np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)]),  # Linear rise and fall
        np.random.rand(400) * 2 - 1,                   # Random initial guess
        np.abs(np.sin(np.linspace(0, 2 * np.pi, 400)))  # Periodic pattern
    ]

    best_h = None
    best_n = 0
    
    # Function to compute the overlap integral
    def compute_c5(h):
        dx = 2.0 / len(h)
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Function to minimize
    def objective(h_values):
        return compute_c5(h_values)
    
    # Function to enforce the integral constraint
    def constraint_func(h_values):
        return np.sum(h_values) - len(h_values) / 2
    
    # Experiment with different starting points
    for pattern in candidate_patterns:

        if n_points > 1000:
            continue

        scale = 1.0 / (np.sum(pattern) * (2.0 / n_points))

        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization
        try:
            result = minimize(
                objective,
                x0=h_initial,
                method='SLSQP',
                bounds=bounds,
                constraints=[{'type': 'eq', 'fun': constraint_func}]
            )
