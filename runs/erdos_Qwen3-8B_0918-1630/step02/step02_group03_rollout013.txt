import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from functools import lru_cache
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=60):
    np.random.seed(seed)

    # Initial grid points
    n_points = 60
    dx = 2.0 / n_points

    # Heuristic starting points: try symmetric, periodic, and random based
    def generate_starting_point():
        # Symmetric base case
        h = np.zeros(n_points)
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 0.5
            else:
                h[i] = 0.5
        # Perturb to avoid flatness
        perturb = np.random.normal(0, 0.1, n_points)
        h += perturb
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        return h

    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def optimize_h(starting_h):
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            lambda x: evaluate(x),
            x0=starting_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (1 / (np.sum(best_h) * dx))
        return best_h, evaluate(best_h)

    # Try multiple starting points
    starting_points = [generate_starting_point() for _ in range(5)]

    best_h = None
    best_c5 = float('inf')

    with Pool(processes=cpu_count()) as pool:
        results = pool.map(optimize_h, starting_points)
        for h, c5 in results:
            if c5 < best_c5:
                best_h = h
                best_c5 = c5

    return best_h, best_c5, n_points
