import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from sklearn.gaussian_process.kernels import RBF, ConstantKernel

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial patterns
    initial_patterns = [
        np.concatenate([np.zeros(100), np.ones(100)]),
        np.sin(np.linspace(0, np.pi, 200)) ** 2,
        np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)]),
        np.random.rand(200) * 2 - 1,
        np.concatenate([np.linspace(0, 0.5, 100), np.linspace(0.5, 1, 100)]),
        np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)]),
    ]
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    start_time = time.time()
    
    for pattern in initial_patterns:
        n_points = len(pattern)
        dx = 2.0 / n_points
        sum_h = np.sum(pattern)
        scale = 1.0 / (sum_h * dx)
        h_initial = pattern * scale
        h_initial = np.clip(h_initial, 0, 1)
        
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try various optimization methods
        for method in ['L-BFGS-B', 'SLSQP', 'Nelder-Mead']:
            result = minimize(objective, x0=h_initial, method=method, bounds=bounds,
                              constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}])
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
                corr = correlate(h_candidate, 1 - h_candidate, mode='full')
                c5_candidate = np.max(corr) * dx
                
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate
                    best_n = n_points
            if time.time() - start_time > budget_s - 1:
                break
    
    # Use adaptive simulated annealing if no full optimizers worked well
    if best_c5 == float('inf'):
        n_points = 200
        dx = 2.0 / n_points
        h_initial = np.random.rand(n_points) * 2 - 1
        h_initial = np.clip(h_initial, -1, 1)
        h_initial = h_initial * (1 / (np.sum(h_initial) * dx))
        h_initial = np.clip(h_initial, 0, 1)
        
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(objective, x0=h_initial, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}])
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
            corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n = n_points
    
    return best_h, best_c5, len(best_h)
