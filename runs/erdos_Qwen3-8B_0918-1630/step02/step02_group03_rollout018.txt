import numpy as np

def run(seed=42, budget_s=60):
    np.random.seed(seed)

    # Initial guess: use the provided initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h)
    scale = 1.0 / (sum_h * dx)
    initial_h_scaled = initial_h * scale
    initial_h_scaled = np.clip(initial_h_scaled, 0, 1)

    class PSO:
        def __init__(self, objective, bounds, n_particles=50, max_iter=100, inertia=0.7, cognitive=1.5, social=1.5):
            self.objective = objective
            self.bounds = bounds
            self.n_particles = n_particles
            self.max_iter = max_iter
            self.inertia = inertia
            self.cognitive = cognitive
            self.social = social
            self.particles = np.random.rand(n_particles, len(bounds)) * (np.array(bounds)[:, 1] - np.array(bounds)[:, 0]) + np.array(bounds)[:, 0]
            self.velocities = np.random.rand(n_particles, len(bounds)) * 0.01
            self.pbest = self.particles.copy()
            self.pbest_values = np.array([self.objective(p) for p in self.particles])
            self.gbest = self.pbest[np.argmin(self.pbest_values)]
            self.gbest_value = np.min(self.pbest_values)

        def update_velocity(self):
            r1 = np.random.rand(self.n_particles, len(self.bounds))
            r2 = np.random.rand(self.n_particles, len(self.bounds))
            self.velocities = self.inertia * self.velocities + self.cognitive * r1 * (self.pbest - self.particles) + self.social * r2 * (self.gbest - self.particles)

        def update_position(self):
            self.particles = np.clip(self.particles + self.velocities, np.array(self.bounds)[:, 0], np.array(self.bounds)[:, 1])

        def optimize(self):
            for _ in range(self.max_iter):
                self.update_velocity()
                self.update_position()
                current_values = np.array([self.objective(p) for p in self.particles])
                improved = current_values < self.pbest_values
                self.pbest[improved] = self.particles[improved]
                self.pbest_values[improved] = current_values[improved]
                self.gbest = self.pbest[np.argmin(self.pbest_values)]
                self.gbest_value = np.min(self.pbest_values)
            return self.gbest, self.gbest_value

    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    bounds = [(0, 1) for _ in range(n_points)]

    # Create PSO optimizer
    pso = PSO(objective, bounds, n_particles=50, max_iter=50, inertia=0.7, cognitive=1.5, social=1.5)
    best_h, best_c5 = pso.optimize()

    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (1 / (np.sum(best_h) * dx))  # Scale to maintain integral constraint

    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx

    return best_h, c5_bound, len(best_h)
