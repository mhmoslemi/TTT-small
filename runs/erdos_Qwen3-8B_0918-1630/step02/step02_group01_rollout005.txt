    def evaluate_overlap(h):

        h_scaled = h / total_area

        if total_area == 0:
            return float('inf')
        h_scaled = h / total_area
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5

    opt = BayesianOptimization(

    )

    for _ in range(50):

        next_params = opt.ask()
        next_value = evaluate(next_params)
        if next_value < best_value:
            best_value = next_value
            best_params = next_params

    final_h = np.array(best_params)
