```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from itertools import combinations

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try multiple starting points with different structures
    initial_points = [
        initial_h_values(800),
        [0.5 if i < 400 else 1.0 for i in range(800)],
        [1.0 if i % 3 == 0 else 0.0 for i in range(800)],
        [1.0 if i % 2 == 0 else 0.5 for i in range(800)]
    ]
    
    best_c5 = float('inf')
    best_h = initial_h_values(800)
    
    for h_initial in initial_points:
        n_points = len(h_initial)
        dx = 2.0 / n_points
        
        def objective(h):
            h = np.array(h)
            h_1_minus = 1.0 - h
            correlate = np.correlate(h, h_1_minus, mode="full")
            overlap = correlate * dx
            return np.max(overlap)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        bounds = Bounds(0.0, 1.0)
        constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
        
        start_time = time.time()
        
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'maxiter': 300})
        
        elapsed = time.time() - start_time
        
        if elapsed > budget_s:
            continue
        
        h_values = result.x
        c5_bound = evaluate_erdos_solution(h_values, n_points)
        
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_values
    
    return best_h, best_c5, len(best_h)
```
