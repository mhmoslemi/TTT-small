import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import Bounds
from scipy.optimize import minimize

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 500
    dx = 2.0 / n_points

    # Spaced intervals with fixed width and spacing
    interval_width = 0.25
    interval_spacing = 0.25
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        if any((x >= (start - 0.01) and x <= (start + 0.01)) for start in np.arange(0, 2, interval_spacing)):
            if any((x >= (start - 0.01) and x <= (start + 0.01)) for start in np.arange(0, 2, interval_spacing)):
                if any((x >= start - interval_width and x <= start + interval_width) for start in np.arange(0, 2, interval_spacing)):
                    h_initial[i] = 1.0
    h_initial = h_initial.tolist()

    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)

    start_time = time.time()

    result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'gtol': 1e-8, 'maxiter': 1000})

    elapsed = time.time() - start_time

    if elapsed > budget_s:
        return np.array(h_initial), evaluate_erdos_solution(h_initial, n_points), n_points

    h_values = result.x
    c5_bound = evaluate_erdos_solution(h_values, n_points)

    return h_values, c5_bound, n_points
