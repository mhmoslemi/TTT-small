    def objective(h):

        c5 = np.max(corr) * dx
        return c5

            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            perturbed_h = initial_h_scaled + perturbation
            perturbed_h = np.clip(perturbed_h, 0, 1)

            result = minimize(
