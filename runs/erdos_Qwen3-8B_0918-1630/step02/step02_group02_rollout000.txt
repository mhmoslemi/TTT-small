import numpy as np
import time
from scipy.optimize import minimize

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 300
    dx = 2.0 / n_points
    n_evals = 50  # Number of function evaluations per iteration
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = np.array([[0.0, 1.0] for _ in range(n_points)])
    
    def perturb(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h[i] = np.clip(h[i] + np.random.uniform(-0.2, 0.2), 0.0, 1.0)
        return h
    
    def generate_random_candidate():
        h = np.random.uniform(0.0, 1.0, n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h.tolist()
    
    best_h = initial_h_values(n_points)
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    for _ in range(10):
        candidates = [best_h]
        for _ in range(n_evals - 1):
            candidates.append(perturb(candidates[-1]))
        
        for h in candidates:
            if np.sum(h) != n_points / 2:
                continue
            c5 = evaluate_erdos_solution(h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    return best_h, best_c5, n_points
