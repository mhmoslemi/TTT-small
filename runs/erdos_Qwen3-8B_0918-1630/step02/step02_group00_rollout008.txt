    scale = 1.0 / (sum_h * dx)
    h_initial = initial_h * scale
    h_initial = np.clip(h_initial, 0, 1)

    def objective(h, k_values):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))
        corr = []
        for k in k_values:
            shifted = np.roll(h, int(k * n_points / 2))
            corr.append(np.sum(h * (1 - shifted)) * dx)
        return np.max(corr)
    
    # Define bounds for all h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define kernel values to test (small range for faster evaluation)
    k_values = np.linspace(-1, 1, 100)
    
    # Optimization with SLSQP to find a better construction
    def optimized_objective(h):
        return objective(h, k_values)
    
    result = minimize(optimized_objective, x0=h_initial, method='SLSQP', bounds=bounds,
                      constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}])
    
    if result.success:
        h_candidate = result.x
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
        c5_candidate = objective(h_candidate, k_values)
        
        # Save best result
        best_h = h_candidate
        best_c5 = c5_candidate
        
        # Test with a few additional initializations for diversity
        additional_patterns = [
            np.sin(np.linspace(0, np.pi, n_points)) ** 2,

        ]

            scale = 1.0 / (np.sum(p) * dx)
            p_scaled = p * scale
