import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import Bounds
import random

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 4:
            h[i] = 1.0
        elif n_points // 4 <= i < n_points // 2:
            h[i] = 0.5
        elif n_points // 2 <= i < 3 * n_points // 4:
            h[i] = 0.0
        else:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 600
    dx = 2.0 / n_points
    
    # Generate multiple initial guesses with structured patterns
    def generate_pattern(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 10 == 0:
                h[i] = 1.0
            elif i % 10 == 5:
                h[i] = 0.5
            elif i % 5 == 0:
                h[i] = 0.0
            elif i % 10 == 2 or i % 10 == 3:
                h[i] = 1.0
            elif i % 10 == 7 or i % 10 == 8:
                h[i] = 0.5
            else:
                h[i] = 0.0
        return h.tolist()
    
    # Generate 5 initial patterns
    initial_guesses = [generate_pattern(n_points) for _ in range(5)]
    initial_guesses.append(initial_h_values(n_points))
    
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    start_time = time.time()
    
    for guess in initial_guesses:
        def objective(h):
            h = np.array(h)
            h_1_minus = 1.0 - h
            correlate = np.correlate(h, h_1_minus, mode="full")
            overlap = correlate * dx
            return np.max(overlap)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        bounds = Bounds(0.0, 1.0)
        constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
        
        result = minimize(objective, guess, method='SLSQP', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'maxiter': 300})
        
        if time.time() - start_time > budget_s:
            break
        
        if result.success:
            current_c5 = evaluate_erdos_solution(result.x, n_points)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
