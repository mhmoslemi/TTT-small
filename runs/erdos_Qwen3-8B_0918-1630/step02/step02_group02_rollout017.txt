import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Start with a moderate number of points to balance speed and resolution
    n_points = 1000
    dx = 2.0 / n_points
    
    # Generate a more complex initial guess with alternating segments
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 4:
            h_initial[i] = 1.0
        elif n_points // 4 <= i < 3 * n_points // 4:
            h_initial[i] = 0.5
        else:
            h_initial[i] = 1.0
    
    # Add some small perturbations to break symmetry
    perturbation = np.random.normal(0, 0.05, n_points)
    h_initial += perturbation
    h_initial = np.clip(h_initial, 0, 1)
    
    def objective(h):
        h = np.array(h)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    
    start_time = time.time()
    
    # Use SLSQP with a refined initial guess and more iterations
    result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'maxiter': 500})
    
    elapsed = time.time() - start_time
    
    if elapsed > budget_s:
        return np.array(h_initial), evaluate_erdos_solution(h_initial, n_points), n_points
    
    h_values = result.x
    c5_bound = evaluate_erdos_solution(h_values, n_points)
    
    return h_values, c5_bound, n_points
