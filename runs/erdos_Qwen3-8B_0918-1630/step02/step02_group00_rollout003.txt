import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import DifferentialEvolution

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Predefined patterns for starting points
    patterns = [
        np.concatenate([np.zeros(100), np.ones(100)]),  # Checkerboard pattern
        np.sin(np.linspace(0, np.pi, 400)) ** 2,       # Sinusoidal pattern
        np.concatenate([np.linspace(0, 1, 100), np.linspace(1, 0, 100)]),  # Linear rise and fall
        np.random.rand(400) * 2 - 1,                   # Random initial guess
        np.concatenate([np.zeros(100), np.ones(100), np.zeros(100)]),  # Triple checkerboard
        np.tile([0, 1, 0, 1, 0, 1], 30),                 # Periodic pattern
        np.concatenate([np.linspace(0, 1, 50), np.linspace(1, 0, 50)]),  # Zig-zag pattern
    ]
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    best_time = 0
    
    for pattern in patterns:
        n_points = len(pattern)
        if n_points > 2000:
            continue
        
        dx = 2.0 / n_points
        sum_h = np.sum(pattern)
        
        # Scale pattern to maintain integral constraint
        scale = 1.0 / (sum_h * dx)
        h_initial = pattern * scale
        h_initial = np.clip(h_initial, 0, 1)
        
        # Define objective function
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        # Define bounds for all h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use Differential Evolution for global optimization
        def objective_for_de(h):
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        result = DifferentialEvolution(objective_for_de, bounds, strategy='best1bin',
                                      maxiter=50, popsize=20, mutation=0.9, recombination=0.5,
                                      tol=1e-6, disp=False)
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (1 / (np.sum(h_candidate) * dx))
            corr = correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n = n_points
                best_time = time.time()
    
    return best_h, best_c5, len(best_h)
