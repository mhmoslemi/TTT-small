from scipy.optimize import minimize, Bounds

    initial_h_scaled = np.clip(initial_h_scaled, 0, 1)

    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute the objective and constraints
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        sum_h = np.sum(h)
        return sum_h - 1
    
    # Define bounds for all h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to generate initial guesses
    def generate_initial_guess(n_sample):
        if n_sample == len(initial_h):
            return initial_h_scaled
        elif n_sample == 50:
            samples = np.linspace(0, 2, n_sample)
            h = np.zeros(n_sample)

                if 0.5 <= samples[i] <= 1.5:
                    h[i] = 1.0
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))
            return h
        else:
            return np.random.uniform(0, 1, n_sample)
    
    # Define a function to run optimization with different seeds and initial guesses
    def run_optimization(n_sample, seed):
        np.random.seed(seed)
        initial_guess = generate_initial_guess(n_sample)
        result = minimize(
            objective,
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=bounds,

        )
