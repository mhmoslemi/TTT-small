```python
import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Define initial patterns for diversity
    def create_initial_patterns(n_points):
        patterns = []
        # Linear rise and fall
        patterns.append(np.concatenate([np.linspace(0, 1, n_points // 2), np.linspace(1, 0, n_points // 2)]))
        # Sinusoidal pattern
        x = np.linspace(0, np.pi, n_points)
        patterns.append(np.sin(x) ** 2)
        # Checkerboard pattern
        patterns.append(np.concatenate([np.zeros(n_points // 2), np.ones(n_points // 2)]))
        # Random initial guess
        patterns.append(np.random.rand(n_points) * 2 - 1)
        # Symmetric step function
        patterns.append(np.concatenate([np.linspace(0, 1, n_points // 2), np.linspace(1, 0, n_points // 2)]))
        # Overlap of two functions
        patterns.append(np.concatenate([np.linspace(0, 1, n_points // 2), np.linspace(1, 1, n_points // 2)]))
        return patterns
    
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    best_time = 0
    
    for n_points in [100, 200, 300, 400, 500, 600]:
        if time.time() - best_time > budget_s - 1:
            break
        
        patterns = create_initial_patterns(n_points)
        for pattern in patterns:
            h_initial = np.clip(pattern, 0, 1)
            # Normalize to satisfy the integral constraint
            sum_h = np.sum(h_initial)
            scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
            h_initial = h_initial * scale
            h_initial = np.clip(h_initial, 0, 1)
            
            # Define the objective function
            def objective(h):
                h = np.clip(h, 0, 1)
                sum_h = np.sum(h)
                scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
                h_scaled = h * scale
                corr = correlate(h_scaled, 1 - h_scaled, mode='full')
                c5 = np.max(corr) * (2.0 / n_points)
                return c5
            
            # Bounds and constraints
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}
            
            # Use differential evolution for global optimization
            result = differential_evolution(
                fun=objective,
                bounds=bounds,
                constraints=[constraint],
                seed=seed,
                strategy='best2exp',
                popsize=20,
                maxiter=50,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.8,
                polish=True,
                vectorized=True
            )
            
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                sum_h_candidate = np.sum(h_candidate)
                scale = (n_points / 2) / sum_h_candidate if sum_h_candidate != 0 else 1.0
                h_candidate_scaled = h_candidate * scale
                corr = correlate(h_candidate_scaled, 1 - h_candidate_scaled, mode='full')
                c5_candidate = np.max(corr) * (2.0 / n_points)
                
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate_scaled
                    best_n = n_points
                    best_time = time.time()
    
    return best_h, best_c5, len(best_h)
```
