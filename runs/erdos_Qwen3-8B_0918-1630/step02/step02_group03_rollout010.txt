import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.optimize import Bounds
from functools import lru_cache

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    n_points = 400
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=False)

    # Initial guess: random binary-like with scaling
    h_init = np.random.rand(n_points)
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init / np.sum(h_init) * dx * n_points

    def obj_func(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / np.sum(h) * dx)
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5

    # Bounds on h
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization with constraints
    result = minimize(
        fun=obj_func,
        x0=h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1}
        ],
        options={'ftol': 1e-8, 'maxiter': 300}
    )

    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (1 / (np.sum(best_h) * dx))

    corr = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx

    return best_h, c5_bound, len(best_h)
