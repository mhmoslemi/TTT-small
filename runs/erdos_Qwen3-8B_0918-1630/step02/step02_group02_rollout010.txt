import numpy as np
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    h[:n_points//4] = 1.0
    h[n_points//2 - n_points//8 : n_points//2 + n_points//8] = 0.5
    h[-n_points//4:] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    # Try with different n_points
    n_points_list = [100, 200, 300, 400, 500, 600, 800, 1000]
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        
        # Use a more structured initial guess
        h_initial = np.zeros(n_points)
        h_initial[:n_points//4] = 1.0
        h_initial[n_points//2 - n_points//8 : n_points//2 + n_points//8] = 0.5
        h_initial[-n_points//4:] = 1.0
        h_initial = h_initial.tolist()
        
        def objective(h):
            h = np.array(h)
            h_1_minus = 1.0 - h
            correlate = np.correlate(h, h_1_minus, mode="full")
            overlap = correlate * dx
            return np.max(overlap)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        bounds = Bounds(0.0, 1.0)
        constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
        
        start_time = time.time()
        
        # Use SLSQP with a more sophisticated initial guess and more iterations
        result = minimize(objective, h_initial, method='SLSQP', bounds=bounds, constraints=[constraint], options={'ftol': 1e-10, 'maxiter': 300})
        
        elapsed = time.time() - start_time
        
        if elapsed > budget_s:
            continue
        
        h_values = result.x
        c5_bound = evaluate_erdos_solution(h_values, n_points)
        
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_values
            best_n_points = n_points
    
    return best_h, best_c5, best_n_points
