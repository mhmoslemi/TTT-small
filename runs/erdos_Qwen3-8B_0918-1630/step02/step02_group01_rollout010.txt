        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute the objective and constraints
    def objective(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = 1.0 / (sum_h * dx)
        h_scaled = h * scale
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5

        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        return sum_h - 1

    bounds = [(0, 1) for _ in range(n_points)]
    
    # Function to run the optimization
    def optimize(initial_guess):
        result = minimize(
            objective,
            x0=initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        sum_h = np.sum(best_h)

        corr = correlate(best_h_scaled, 1 - best_h_scaled, mode='full')
        c5_bound = np.max(corr) * dx
        return best_h_scaled, c5_bound
    
    # Run multiple optimizations with different initial guesses
    best_h = initial_h

    start_time = time.time()
