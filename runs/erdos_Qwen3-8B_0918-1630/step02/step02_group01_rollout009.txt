import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.optimize import Bounds
from scipy.optimize import dual_annealing
from scipy.optimize import minimize as minimize_scipy
from numpy.random import choice, uniform, randint
from collections import defaultdict
from functools import partial

# Pre-imported
initial_h_values = np.load('initial_h_values.npy')  # Assuming initial_h_values is a numpy array

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Base initial guess
    initial_h = initial_h_values[:n_points]
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (1 / (np.sum(initial_h) * dx))
    initial_h = np.clip(initial_h, 0, 1)

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Simple random binary initial guess
    def generate_random_binary(n):
        return np.random.choice([0, 1], size=n, p=[0.5, 0.5])
    
    # Generate a few initial guesses
    initial_guesses = [
        initial_h,
        generate_random_binary(n_points),
        np.random.rand(n_points) * 1.0,
        np.linspace(0, 1, n_points),
        np.zeros(n_points),
        np.ones(n_points),
        np.sin(np.linspace(0, np.pi, n_points)),
        np.cos(np.linspace(0, np.pi, n_points))
    ]
    
    initial_guesses = [np.clip(g, 0, 1) for g in initial_guesses]
    initial_guesses = [g * (1 / (np.sum(g) * dx)) for g in initial_guesses]
    initial_guesses = [np.clip(g, 0, 1) for g in initial_guesses]
    
    # Define the optimization function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        h = np.clip(h, 0, 1)
        h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
        sum_h = np.sum(h)
        return sum_h - 1
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define helper functions
    def optimize_with_slsqp(guess):
        result = minimize(
            objective,
            x0=guess,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 200, 'ftol': 1e-8, 'eps': 1e-8}
        )
        return result.x
    
    def optimize_with_dual_annealing(guess):
        result = dual_annealing(
            objective,
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint}],
            maxiter=200,
            x0=guess,
            seed=seed,
            no_increase=10
        )
        return result.x
    
    # Use both optimization methods
    all_results = []
    for guess in initial_guesses:
        # Use both optimization methods
        optimized1 = optimize_with_slsqp(guess.copy())
        optimized2 = optimize_with_dual_annealing(guess.copy())
        optimized3 = optimize_with_slsqp(optimized2.copy())
        all_results.append((optimized1, optimized2, optimized3))
    
    # Evaluate all results
    for result_set in all_results:
        for h in result_set:
            h = np.clip(h, 0, 1)
            h = h * (1 / (np.sum(h) * dx))  # Scale to maintain integral constraint
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
                best_n_points = n_points
    
    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (1 / (np.sum(best_h) * dx))  # Scale to maintain integral constraint
    corr = correlate(best_h, 1 - best_h, mode='full')
    c5_bound = np.max(corr) * dx
    return best_h, c5_bound, best_n_points
