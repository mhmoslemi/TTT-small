import numpy as np

from scipy.optimize import Bounds
from scipy.optimize import minimize

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values, dtype=np.float64)
    h_1_minus = 1.0 - h
    correlate = np.correlate(h, h_1_minus, mode="full")
    overlap = correlate * dx
    return np.max(overlap)

def initial_h_values(n_points):
    h = np.zeros(n_points)
    h[:n_points//4] = 1.0
    h[n_points//2 - n_points//8 : n_points//2 + n_points//8] = 0.5
    h[-n_points//4:] = 1.0
    return h.tolist()

def run(seed=42, budget_s=60, **kwargs):
    np.random.seed(seed)
    
    n_points = 500
    dx = 2.0 / n_points
    total_integral = 1.0
    
    def objective(h_values):
        h = np.array(h_values)
        h_1_minus = 1.0 - h
        correlate = np.correlate(h, h_1_minus, mode="full")
        overlap = correlate * dx
        return np.max(overlap)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0.0, 1.0)
    constraint = LinearConstraint(np.ones(n_points), lb= n_points / 2, ub= n_points / 2)
    
    # Initial guess: step function with three segments
    h_initial = np.zeros(n_points)
    h_initial[:n_points//3] = 1.0
    h_initial[n_points//3:n_points//3*2] = 0.5
    h_initial[-n_points//3:] = 1.0
    h_initial = h_initial.tolist()
    
    def surrogate_objective(h):
        return np.max(np.correlate(np.array(h), 1 - np.array(h), mode='full') * dx)
    
    result = minimize(surrogate_objective, h_initial, method='L-BFGS-B', bounds=bounds, constraints=[constraint], options={'ftol': 1e-12, 'maxiter': 500})
    
    if time.time() - time.time() > budget_s:
        return np.array(h_initial), evaluate_erdos_solution(h_initial, n_points), n_points
    
    h_values = result.x
    c5_bound = evaluate_erdos_solution(h_values, n_points)
    
    return h_values, c5_bound, n_points
