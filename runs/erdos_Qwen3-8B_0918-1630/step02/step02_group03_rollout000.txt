```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
from scipy.optimize import Bounds
from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=60):
    np.random.seed(seed)

    def evaluate(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        dx = 2.0 / len(h)
        scale = 1.0 / (sum_h * dx)
        h_scaled = h * scale
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return c5

    def objective(h):
        return evaluate(h)

    def initial_guess(n_points):
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.5
        return h

    def parallel_optimize(n_points):
        bounds = [(0, 1) for _ in range(n_points)]
        initial = initial_guess(n_points)
        result = minimize(objective, initial, bounds=bounds, method='L-BFGS-B', options={'maxiter': 200, 'ftol': 1e-8})
        return result.x

    n_points = 300
    best_h = initial_guess(n_points)
    best_value = evaluate(best_h)
    best_len = len(best_h)

    start_time = time.time()

    def run_parallel(args):
        seed, n_points = args
        np.random.seed(seed)
        h = np.random.uniform(0, 1, n_points)
        sum_h = np.sum(h)
        dx = 2.0 / n_points
        scale = 1.0 / (sum_h * dx)
        h_scaled = h * scale
        corr = correlate(h_scaled, 1 - h_scaled, mode='full')
        c5 = np.max(corr) * dx
        return (c5, h_scaled)

    pool = Pool(1)
    args = [(seed, n_points) for seed in range(10)]
    results = pool.map(run_parallel, args)
    pool.close()
    pool.join()

    best_result = min(results, key=lambda x: x[0])
    best_value = best_result[0]
    best_h = best_result[1]
    best_len = len(best_h)

    if time.time() - start_time < budget_s - 1:
        for n in [150, 200]:
            args = [(seed, n) for seed in range(5)]
            results = pool.map(run_parallel, args)
            best_result = min(results, key=lambda x: x[0])
            best_value = best_result[0]
            best_h = best_result[1]
            best_len = len(best_h)

    return best_h, best_value, best_len
```
