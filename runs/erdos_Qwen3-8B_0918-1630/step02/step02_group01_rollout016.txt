import numpy as np
import time
from scipy.optimize import minimize, Bounds
from scipy.signal import correlate
from scipy import stats

def run(seed=42, budget_s=60):
    np.random.seed(seed)
    
    # Initial guess from initial_h_values
    initial_h = initial_h_values
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h_initial = np.sum(initial_h)
    scale = 1.0 / (sum_h_initial * dx)
    initial_h_scaled = initial_h * scale
    initial_h_scaled = np.clip(initial_h_scaled, 0, 1)
    
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h == 0:
            return float('inf')
        scaled_h = h / (sum_h * dx)
        corr = correlate(scaled_h, 1 - scaled_h, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def objective(h):
        return evaluate_overlap(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint(h):
        sum_h = np.sum(h)
        return sum_h - 1
    
    # Main optimization with multiple initializations
    best_c5 = float('inf')
    best_h = initial_h_scaled
    
    # Run multiple random initializations to avoid local optima
    for _ in range(5):
        # Create perturbed initial guess
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        perturbed_initial = initial_h_scaled + perturbation
        perturbed_initial = np.clip(perturbed_initial, 0, 1)
        perturbed_initial = perturbed_initial * (1 / (np.sum(perturbed_initial) * dx))
        
        # Run optimization
        result = minimize(
            objective,
            x0=perturbed_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 50, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if not result.success:
            continue
        
        current_h = result.x
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (1 / (np.sum(current_h) * dx))
        c5 = evaluate_overlap(current_h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h
    
    return best_h, best_c5, len(best_h)
