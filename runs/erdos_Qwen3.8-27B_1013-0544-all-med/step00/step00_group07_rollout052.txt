import numpy as np
import time

def _compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(_compute_corr(h)) * dx)

def _compute_coefs(h):
    n = len(h)
    nc = 2 * n - 1
    coef = np.zeros((n, nc))
    for k in range(nc):
        kp = k - (n - 1)
        if kp >= 0:
            coef[kp:, k] += (1.0 - h[:n - kp])
        else:
            coef[:n + kp, k] += (1.0 - h[-kp:])
        if kp >= 0:
            coef[:n - kp, k] -= h[kp:]
        else:
            coef[-kp:, k] -= h[:n + kp]
    return coef

def _fix_sum(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    diff = target_sum - h.sum()
    n = len(h)
    for _ in range(n * 20):
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h.copy()
        total = room.sum()
        if total < 1e-15:
            break
        h += (diff / total) * room
        h = np.clip(h, 0.0, 1.0)
        diff = target_sum - h.sum()
    return h

def _pair_step(h, corr, coef, i, j, n_bisect=20):
    lo = max(-h[i], h[j] - 1.0)
    hi = min(1.0 - h[i], h[j])
    if lo >= hi - 1e-15:
        return 0.0
    c = coef[i] - coef[j]
    a, b = lo, hi
    for _ in range(n_bisect):
        mid = 0.5 * (a + b)
        vals = corr + mid * c
        mv = vals.max()
        active = vals >= (mv - 1e-13)
        if c[active].min() > 1e-11:
            b = mid
        elif c[active].max() < -1e-11:
            a = mid
        else:
            break
    return 0.5 * (a + b)

def _cd_phase(h, rng, time_limit, start_time, n_bisect=20):
    n = len(h)
    pairs = np.array([(i, j) for i in range(n) for j in range(i + 1, n)])
    n_pairs = len(pairs)
    while time.time() - start_time < time_limit:
        corr = _compute_corr(h)
        coef = _compute_coefs(h)
        order = rng.permutation(n_pairs)
        for idx in order:
            if time.time() - start_time > time_limit:
                break
            i, j = pairs[idx]
            delta = _pair_step(h, corr, coef, i, j, n_bisect)
            if abs(delta) < 1e-13:
                continue
            h[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h[j] = np.clip(h[j] - delta, 0.0, 1.0)
            corr += delta * (coef[i] - coef[j])
    return h

def _perturb_and_refine(h_best, rng, time_limit, start_time):
    n = len(h_best)
    target_sum = n / 2.0
    best = h_best.copy()
    best_c5 = _eval_c5(h_best)
    remaining = time_limit - (time.time() - start_time)
    n_restarts = max(1, int(remaining / 20))
    for r in range(n_restarts):
        if time.time() - start_time > time_limit:
            break
        h = h_best.copy()
        scale = 0.03 * (1.0 + r * 0.3)
        h += rng.randn(n) * scale
        h = _fix_sum(h, target_sum)
        sub_budget = min(15.0, 0.7 * (time_limit - (time.time() - start_time)))
        if sub_budget < 2:
            break
        h = _cd_phase(h, rng, time.time() + sub_budget, time.time(), n_bisect=25)
        c5 = _eval_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5
