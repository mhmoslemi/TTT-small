import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_objective(h, penalty, temperature):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temperature < 1e-12:
        sm = m
    else:
        shifted = np.clip((corr - m) / temperature, -500.0, 0.0)
        sm = m + temperature * np.log(np.sum(np.exp(shifted)) + 1e-300)
    cv = np.sum(h) - n * 0.5
    return sm + penalty * cv * cv

def project_constraint(h, target_sum):
    lo = -np.max(h) - 2.0
    hi = np.max(1.0 - h) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        p = np.clip(h + mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5*(lo+hi), 0.0, 1.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = np.ones(45) * 0.5
    
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    target_sum = n * 0.5
    
    # Ensure initial point satisfies constraint
    h0 = project_constraint(h0, target_sum)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Decreasing temperature optimization
    temperatures = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    penalties = [500.0, 1000.0, 2000.0, 5000.0, 10000.0, 20000.0, 50000.0]
    
    for i, temp in enumerate(temperatures):
        elapsed = time.time() - t0
        if elapsed > budget_s - 120:
            break
        pen = penalties[i]
        
        try:
            res = minimize(
                soft_max_objective, h0,
                args=(pen, temp),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 3000, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50}
            )
            h_trial = project_constraint(res.x, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
            h0 = h_trial.copy()
        except Exception:
            pass
    
    # Phase 2: Random restarts from best
    elapsed = time.time() - t0
    restarts = min(20, int((budget_s - elapsed - 60) / 30))
    for r in range(max(restarts, 1)):
        elapsed = time.time() - t0
        if elapsed > budget_s - 60:
            break

        h_pert = best_h + np.random.randn(n) * perturbation_scale
        h_pert = project_constraint(h_pert, target_sum)
        
        temp = 0.005
        pen = 10000.0
        try:

                soft_max_objective, h_pert,
                args=(pen, temp),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 2000, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50}
            )
            h_trial = project_constraint(res.x, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
        except Exception:
            pass
    
    # Final refinement
    elapsed = time.time() - t0
    if elapsed < budget_s - 30:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(h, penalty, temperature):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    if temperature < 1e-14:
        sm = m
    else:
        shifted = np.clip((corr - m) / temperature, -400.0, 0.0)
        sm = m + temperature * np.log(np.sum(np.exp(shifted)) + 1e-300)
    cv = np.sum(h) - 0.5 * n
    return sm + penalty * cv * cv

def project_constraint(h, target_sum):
    lo = -float(np.max(h)) - 3.0
    hi = float(np.max(1.0 - h)) + 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        p = np.clip(h + mid, 0.0, 1.0)
        if np.sum(p) > target_sum:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def try_optimize(h0, n, bounds, target_sum, penalty, temperature, maxiter, t_deadline):
    try:
        res = minimize(
            soft_max_obj, h0, args=(penalty, temperature),
            method='L-BFGS-B', bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50}
        )
        if time.time() > t_deadline:
            return None
        h_t = project_constraint(res.x, target_sum)
        return h_t
    except Exception:
        return None

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = np.ones(45) * 0.5

    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    target_sum = 0.5 * n
    h0 = project_constraint(h0, target_sum)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coarse-to-fine soft-max optimization
    temps = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    pens = [100.0, 500.0, 2000.0, 5000.0, 10000.0, 20000.0, 50000.0, 100000.0]

    for i, temp in enumerate(temps):
        if time.time() > deadline - 180:
            break
        pen = pens[i]
        h_t = try_optimize(h0, n, bounds, target_sum, pen, temp, 5000, deadline - 180)
        if h_t is not None:
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h_t.copy()
            h0 = h_t.copy()

    # Phase 2: Random restarts to escape local optima
    n_restarts = 15
    for r in range(n_restarts):
        if time.time() > deadline - 120:
            break
        scale = 0.05 + 0.15 * (r / max(n_restarts - 1, 1))
        h_pert = best_h + np.random.randn(n) * scale
        h_pert = project_constraint(h_pert, target_sum)

        temp = 0.003 + 0.01 * np.random.rand()
        pen = 50000.0

        if h_t is not None:
