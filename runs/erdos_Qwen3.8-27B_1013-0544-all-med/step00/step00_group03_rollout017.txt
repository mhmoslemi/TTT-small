```python
def soft_gradient(h, t, dx, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = corr.max()
    log_sum = np.log(np.sum(np.exp(t * (corr - cm))))
    weights = np.exp(t * (corr - cm) - log_sum)  # p_k for k=0..2n-2
    # weights sum to 1
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if weights[k] < 1e-10:
            continue
        L = k - (n - 1)
        w = weights[k] * dx
        j = np.arange(n)
        v1 = (j - L >= 0) & (j - L < n)
        grad[v1] += w * (1.0 - h[j[v1] - L])
        v2 = (j + L >= 0) & (j + L < n)
        grad[v2] += -w * h[j[v2] + L]
    return grad, (cm + log_sum / t) * dx
```
