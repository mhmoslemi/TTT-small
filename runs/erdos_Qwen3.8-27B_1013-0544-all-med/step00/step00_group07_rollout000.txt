def _smooth_max(v, beta):
    v = np.asarray(v, dtype=float)
    m = float(np.max(v))
    s = float(np.sum(np.exp(beta * (v - m))))
    if s == 0.0:
        return m
    return m + float(np.log(s)) / beta

def _objective(x, target, beta=8.0, lam=8.0):
    x = np.asarray(x, dtype=float)
    cross = np.correlate(x, 1.0 - x, 'full')
    m = _smooth_max(cross, beta)
    per = lam * (float(np.sum(x)) - target) ** 2
    return m * (2.0 / len(x)) + per

def _repair_sum(x, target):
    tol = 1e-9
    x = np.asarray(x, dtype=float).copy()
    for _ in range(400000):
        s = float(np.sum(x))
        diff = target - s
        if abs(diff) < tol:
            break
        eps = 1e-12
        if diff > 0:
            cand = np.where(x < 1.0 - eps)[0]
        else:
            cand = np.where(x > eps)[0]
        if cand.size == 0:
            break
        n = cand.size
        step = diff / n
        if diff > 0:
            add = np.minimum(step, 1.0 - x[cand])
            x[cand] += add
        else:
            sub = np.minimum(-step, x[cand])
            x[cand] -= sub
    s = float(np.sum(x))
    rem = target - s
    if abs(rem) > 1e-13:
        if rem > 0:
            room = 1.0 - x
            idx = int(np.argmax(room))
            x[idx] = min(1.0, x[idx] + rem)
        else:
            idx = int(np.argmax(x))
            x[idx] = max(0.0, x[idx] + rem)
    return x

def _evaluate_c5(x):
    x = np.asarray(x, dtype=float)
    dx = 2.0 / len(x)
    c = np.max(np.correlate(x, 1.0 - x, 'full'))
