def repair_h(h, target_sum):

import numpy as np
import time

def repair_h(h, target_sum):
    n = len(h)
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-14
            if not mask.any():
                break
            h[mask] += room[mask] * (diff / room[mask].sum())
        else:
            mask = h > 1e-14
            if not mask.any():
                break
            h[mask] -= h[mask] * ((-diff) / h[mask].sum())
    return np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def subgradient(h, lag_idx):
    n = len(h)
    m = abs(lag_idx - (n - 1))
    grad = np.zeros(n)
    if m >= n:
        return grad
    for j in range(m, n):
        grad[j] += (1.0 - h[j - m])
    for j in range(0, n - m):
        grad[j] -= h[j + m]
    return grad

def systematic_local_search(h, target_sum, max_rounds=100, init_step=0.05, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = repair_h(h.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = init_step
    stale = 0
    for r in range(max_rounds):
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        rng.shuffle(pairs)
        improved = False
        for (i, j) in pairs:
            for sign in [1, -1]:
                d = sign * step
                hi, hj = h[i] - d, h[j] + d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[j] = hi, hj
                    c5 = compute_c5(h)
                    if c5 < best_c5 - 1e-15:
                        best_c5 = c5
                        best_h = h.copy()
                        improved = True
                        break
                    else:
                        h[i], h[j] = hi + d, hj - d
            if improved:
                break
        if not improved:
            stale += 1
            step *= 0.5
            if step < 0.001:
                break
        else:
            stale = 0
    return best_h, best_c5

def stochastic_refine(h, target_sum, n_steps, seed, init_step=0.1, cool=0.9999):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = repair_h(h.copy(), target_sum)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = init_step
    for s in range(n_steps):
        step = max(0.002, init_step * (cool ** s))
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = rng.choice([-1.0, 1.0]) * step
        hi, hj = h[i] - d, h[j] + d
        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
            h[i], h[j] = hi, hj
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            else:
                h[i], h[j] = hi + d, hj - d
    return best_h, best_c5

def gradient_refine(h, target_sum, max_iters=3000, lr_init=0.02, seed=42):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = repair_h(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        grad = subgradient(h, k_star) * dx
        h_new = h - lr * grad
        h_new = repair_h(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        if it % 200 == 0 and it > 0:
            lr *= 0.9
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def tleft():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=np.float64)
    else:
        h0 = np.full(70, 0.5)

    best_h = None
    best_c5 = 1e9

    n0 = len(h0)
    ts0 = n0 / 2.0
    h0 = repair_h(h0, ts0)

    # Phase 1: Quick stochastic warm-up
    h1, c5_1 = stochastic_refine(h0, ts0, 30000, seed, 0.12)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()
    if tleft() < 20:
        return best_h, best_c5, len(best_h)

    # Phase 2: Systematic local search
    h2, c5_2 = systematic_local_search(h1, ts0, max_rounds=80, init_step=0.05, seed=seed+1)
    if c5_2 < best_c5:
        best_c5, best_h = c5_2, h2.copy()
    if tleft() < 30:
        return best_h, best_c5, len(best_h)

    # Phase 3: Gradient-based refinement
    h3, c5_3 = gradient_refine(h2, ts0, max_iters=2000, lr_init=0.015, seed=seed+2)
    if c5_3 < best_c5:
        best_c5, best_h = c5_3, h3.copy()
    if tleft() < 30:
        return best_h, best_c5, len(best_h)

    # Phase 4: Further stochastic polish
    h4, c5_4 = stochastic_refine(best_h, ts0, 80000, seed=seed+3, init_step=0.06, cool=0.99995)
    if c5_4 < best_c5:
        best_c5, best_h = c5_4, h4.copy()
    if tleft() < 60:
        return best_h, best_c5, len(best_h)

    # Phase 5: Try higher resolution
    for n_try in [n0 + 20, n0 + 40, 120, 160, 200]:
        if n_try > 250 or tleft() < 80:
            continue
        ts_try = n_try / 2.0
        h_try = np.interp(np.linspace(0, 1, n_try), np.linspace(0, 1, len(best_h)), best_h)
        h_try = repair_h(h_try, ts_try)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5, best_h = c5_try, h_try.copy()
        # Refine
        hr, cr = stochastic_refine(h_try, ts_try, min(40000, int(tleft() * 20)), seed=seed+n_try, init_step=0.08)
        if cr < best_c5:
            best_c5, best_h = cr, hr.copy()
        if tleft() > 60:
            hr2, cr2 = systematic_local_search(hr, ts_try, max_rounds=30, init_step=0.04, seed=seed+n_try+1)
            if cr2 < best_c5:
                best_c5, best_h = cr2, hr2.copy()

    # Phase 6: Final polish on best
    if tleft() > 20:
        nb = len(best_h)
        tsb = nb / 2.0
        nsteps = min(200000, max(10000, int(tleft() * 40)))
        hf, cf = stochastic_refine(best_h, tsb, nsteps, seed=seed+500, init_step=0.03, cool=0.99998)
        if cf < best_c5:
            best_c5, best_h = cf, hf.copy()
        if tleft() > 20:
            hf2, cf2 = systematic_local_search(best_h, tsb, max_rounds=30, init_step=0.02, seed=seed+600)
            if cf2 < best_c5:
                best_c5, best_h = cf2, hf2.copy()

    return best_h, best_c5, len(best_h)
