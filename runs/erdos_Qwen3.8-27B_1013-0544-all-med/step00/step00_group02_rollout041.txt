def compute_c5(h):

    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(c)

def project_onto_constraint(v, n):
    """Project v onto {h: sum(h) = n/2, 0 <= h <= 1}"""
    S = n / 2.0
    # Binary search for lambda
    lo, hi = -10.0, 10.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > S:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)
    return h

def smooth_c5_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    exp_c = np.exp(t * (c - max_c))
    sum_exp = np.sum(exp_c)
    obj = max_c + np.log(sum_exp) / t
    weights = exp_c / sum_exp

            if 0 <= j - delta < n:
                contrib -= h[j - delta]
            grad[j] += wk * contrib

    n = len(h0)
    h = h0.copy()
    
    if alpha is None:
        alpha = 1.0 / (t * 2.0)

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    exp_c = np.exp(t * (c - max_c))
    sum_exp = np.sum(exp_c)
    obj = max_c + np.log(sum_exp) / t
    weights = exp_c / sum_exp

    # weights[k] corresponds to delta = k - (n-1)
    # grad[j] = dx * sum_k weights[k] * ((1-h[j+delta_k]) - h[j-delta_k])
    
    # Vectorized: for each j, compute the sum
    # Let's build a matrix M where M[j, k] = (1-h[j+delta_k]) - h[j-delta_k]
    # Padded
    
    j_idx = np.arange(n)
    deltas = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # j + delta: shape (n, 2n-1)
    jd = j_idx[:, None] + deltas[None, :]
    jd_mask = (jd >= 0) & (jd < n)
    
    # j - delta
    jnd = j_idx[:, None] - deltas[None, :]
    jnd_mask = (jnd >= 0) & (jnd < n)
    
    # (1-h[j+delta]) where valid
    term1 = np.where(jd_mask, 1.0 - h[np.clip(jd, 0, n-1)], 0.0)
    # h[j-delta] where valid  
    term2 = np.where(jnd_mask, h[np.clip(jnd, 0, n-1)], 0.0)
    
    M = term1 - term2  # shape (n, 2n-1)
    grad = dx * (M @ weights)
    
    return obj, grad

def compute_c5(h):

def project_constraint(v, n):
    S = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:

def smooth_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    exp_c = np.exp(t * (c - max_c))
    sum_exp = np.sum(exp_c)
    obj = max_c + np.log(sum_exp) / t
    w = exp_c / sum_exp
    j_idx = np.arange(n)
    deltas = np.arange(-(n - 1), n)
    jd = j_idx[:, None] + deltas[None, :]

    grad = dx * ((term1 - term2) @ w)

    c = np.correlate(h, 1.0 - h, mode='full') * dx

        if s > S:
            lo = mid
        else:

    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(c)
    exp_c = np.exp(t * (c - max_c))

    obj = max_c + np.log(sum_exp) / t
    w = exp_c / sum_exp
    j_idx = np.arange(n)
    deltas = np.arange(-(n - 1), n)

    term2 = np.where(jp_mask, h[np.clip(jp, 0, n - 1)], 0.0)
    grad = dx * ((term1 - term2) @ w)
    return obj, grad
