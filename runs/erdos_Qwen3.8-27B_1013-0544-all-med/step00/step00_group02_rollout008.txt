            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Different resolutions
    for n_try in [n0 + 15, n0 + 30, n0 + 50, n0 + 75]:
        if time.time() - t0 > budget_s * 0.75 or n_try > 200:
            break
        x_old = np.linspace(0, 2, len(best_h), endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_opt, c5_opt = slsqp_optimize(h_interp, n_try, [50, 150, 500], 200)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Random restarts at n0
    n_restarts = 12
    for i in range(n_restarts):
        if time.time() - t0 > budget_s * 0.88:
            break
        if i == 0:
            h_rand = rng.random(n0)
        elif i == 1:
            h_rand = (rng.random(n0) > 0.5).astype(float)
        elif i == 2:
            blk = max(1, n0 // 5)
            h_rand = np.zeros(n0)
            for b in range(0, n0, blk):
                if rng.random() > 0.4:
                    h_rand[b:b + blk] = 1.0
        elif i == 3:
            freq = rng.uniform(0.5, 3.5)
            h_rand = 0.5 + 0.45 * np.sin(2 * np.pi * freq * np.arange(n0) / n0 * 2 + rng.uniform(0, 2 * np.pi))
        elif i == 4:
            freq1 = rng.uniform(1, 4)
            freq2 = rng.uniform(2, 6)
            h_rand = 0.5 + 0.3 * np.sin(2 * np.pi * freq1 * np.arange(n0) / n0 * 2) + 0.2 * np.cos(2 * np.pi * freq2 * np.arange(n0) / n0 * 2)
        elif i == 5:
            h_rand = np.zeros(n0)
            n_ones = n0 // 2
            positions = rng.choice(n0, n_ones, replace=False)
            h_rand[positions] = 1.0
        else:
            h_rand = rng.random(n0)
            if rng.random() > 0.5:
                h_rand = (h_rand > 0.4).astype(float)

        h_opt, c5_opt = slsqp_optimize(h_rand, n0, [50, 200], 120)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 4: Binary local search at even n
    n_bin = n0 if n0 % 2 == 0 else n0 + 1
    if time.time() - t0 < budget_s * 0.6:
        if len(best_h) != n_bin:
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_bin, endpoint=False)
            h_for_bin = np.interp(x_new, x_old, best_h)
        else:
            h_for_bin = best_h.copy()
        deadline = t0 + (budget_s - (time.time() - t0)) * 0.4
        h_bin, c5_bin = binary_local_search(h_for_bin, n_bin, 80000, rng, deadline)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 5: Final high-precision refinement
    if time.time() - t0 < budget_s * 0.9:
        n_best = len(best_h)
        h_opt, c5_opt = slsqp_optimize(best_h, n_best, [200, 800, 2000], 400)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    return best_h, best_c5, len(best_h)
