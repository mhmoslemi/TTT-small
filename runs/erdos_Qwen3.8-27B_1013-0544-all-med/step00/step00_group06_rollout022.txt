    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    """Project h onto {x: 0 <= x[i] <= 1, sum(x) = target_sum}."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = np.sum(h) - target_sum
        if abs(diff) < 1e-12:
            break
        # Adjust: if sum too high, reduce; if too low, increase
        # Only adjust coordinates that aren't at boundary
        if diff > 0:
            # Need to reduce sum by diff
            # Reduce from coordinates that are above some threshold
            # Simple approach: reduce all non-zero coordinates proportionally
            # Better: use the "soft" approach
            mask = h > 0
            if np.sum(mask) == 0:
                break
            # Reduce by diff/sum(h) proportionally where h > 0
            h[mask] -= diff * h[mask] / np.sum(h)
        else:
            # Need to increase sum by -diff
            mask = h < 1
            if np.sum(mask) == 0:
                break
            h[mask] += (-diff) * (1 - h[mask]) / np.sum(1 - h[mask])
        h = np.clip(h, 0.0, 1.0)
    return h

    """Compute subgradient of max_k c[k] with respect to h."""
    n = len(h)
    corr_vals = np.correlate(h, 1.0 - h, mode="full")

            grad[valid2] -= h[idx2]

Now, one concern: the subgradient of the max is not unique. I'm using a weighted average over active lags, which gives a specific subgradient. This should work for subgradient descent.

However, I realize that subgradient descent is known to be slow for non-smooth problems. The step size needs to decrease, and convergence is O(1/√T). For 50000 iterations, this might not be enough.

Let me think about alternative approaches:

**Approach: SOCP / MIQP**
The problem can be formulated as:
minimize T
subject to: sum_i h[i]*(1-h[i+k]) * dx ≤ T for all k
            0 ≤ h[i] ≤ 1
            sum(h) = n/2

The constraint sum_i h[i]*(1-h[i+k]) ≤ T/dx is a quadratic constraint (bilinear). This is a non-convex QCQP (quadratically constrained quadratic program).

We could try to relax it by:
- Replacing h[i]*(1-h[i+k]) with a linear upper bound
- Or using a Benders decomposition approach

But these are complex to implement. Let me stick with the heuristic approach but make it more effective.

**Better approach: Alternating minimization**

Notice that for a fixed k, the constraint c[k] ≤ T is:
sum_i (h[i] - h[i]*h[i+k]) ≤ T/dx

This is bilinear in h. If we fix all h[i] for i ≠ j, then for a given j, the objective becomes:
- For each k where j is involved, we get a linear function of h[j]
- The max of these linear functions is convex in h[j]
- So we can optimize h[j] for each j in turn (coordinate descent)

But we also have the integral constraint. Let me handle it with a Lagrange multiplier:
L = max_k c[k] + λ * (sum(h) - n/2)

For coordinate descent on h[j], we minimize:
max_k (c[k] + λ * (h[j] - ...)) 
subject to 0 ≤ h[j] ≤ 1

This is a 1D convex optimization problem! We can solve it exactly.

Actually, let me think about this more carefully. The Lagrangian for the problem:
min_h max_k c[k]
s.t. sum(h) = n/2, 0 ≤ h[i] ≤ 1

The Lagrangian is: L(h, λ, μ) = max_k c[k] + λ(sum(h) - n/2) + sum_i μ_i(1-h[i]) + sum_i ν_i(-h[i])

This is getting complex. Let me just use a practical approach:

**Practical approach: Coordinate descent with projected updates**

For each coordinate j:
1. Compute the current values of all c[k]
2. For each k, determine how c[k] changes when h[j] changes (linear in h[j])
3. The new objective (max_k c[k]) as a function of h[j] is the max of linear functions → piecewise linear and convex
4. Find the h[j] that minimizes this (subject to bounds and integral constraint)
5. Update the integral constraint by adjusting another coordinate (or using a slow λ update)

Actually, let me simplify further. Let me just do:
1. Coordinate descent ignoring the integral constraint (just bounds 0 ≤ h[j] ≤ 1)
2. After each full pass, project onto the integral constraint

This is a common trick and works well in practice.

For step 1, when optimizing h[j]:
- For each lag k, c[k] = A_k * h[j] + B_k where A_k and B_k don't depend on h[j]
  (because c[k] is either linear in h[j] or doesn't involve h[j] at all)
  
  Wait, c[k] = sum_i h[i](1-h[i+k]). For a fixed k:
  - If j is in the "h[i]" position (i.e., j+k < n): c[k] includes h[j](1-h[j+k])
  - If j is in the "h[i+k]" position (i.e., j-k ≥ 0): c[k] includes h[j-k](1-h[j])
  
  So c[k] = h[j](1-h[j+k]) + h[j-k](1-h[j]) + (terms not involving h[j])
         = h[j] - h[j]h[j+k] + h[j-k] - h[j-k]h[j] + (other terms)
         = h[j](1 - h[j+k] - h[j-k]) + h[j-k] + (other terms)
  
  This is indeed linear in h[j]! (The h[j]h[j+k] term has h[j+k] fixed, and h[j]h[j-k] has h[j-k] fixed.)

So for each k, c[k] = a_k * h[j] + b_k (linear in h[j]).

The objective max_k c[k] is the max of these linear functions, which is convex and piecewise linear in h[j].

To minimize it, we can:
- Find all breakpoints (where two different c[k] values cross)
- Evaluate the max at each breakpoint and at the boundaries 0 and 1
- Choose the minimum

The breakpoints are where a_k * h[j] + b_k = a_l * h[j] + b_l, i.e., h[j] = (b_l - b_k) / (a_k - a_l).

For n=95, there are 189 lags, so at most 189*188/2 ≈ 17766 breakpoints. But we only need to check those in [0,1]. This is feasible.

Actually, a simpler approach: since the function is convex and piecewise linear, we can use a ternary search or just evaluate at a fine grid.

But even simpler: the minimum of a convex piecewise linear function on [0,1] is either at a breakpoint or at the boundary. We can find it by:
1. Evaluating at h[j] = 0 and h[j] = 1
2. Finding the breakpoint where the function transitions from decreasing to increasing

Let me implement this:

```python
def optimize_coordinate(h, j, target_sum=None):
    """Find the best value of h[j] given all other coordinates fixed."""
    n = len(h)
    # Compute a_k and b_k for each lag
    # c[k] = a_k * h[j] + b_k
    
    # h[j] appears in c[k] when:
    # - As h[i] with i=j: needs j+k in [0,n), contributes h[j]*(1-h[j+k])
    # - As h[i+k] with i+k=j, i=j-k: needs j-k in [0,n), contributes h[j-k]*(1-h[j])
    
    corr_full = np.correlate(h, 1.0 - h, mode="full")  # 2n-1 values
    # But this includes the current h[j]. Let me compute a_k and b_k directly.
    
    a = np.zeros(2 * n - 1)
    b = np.zeros(2 * n - 1)
    
    for m in range(2 * n - 1):
        k = m - (n - 1)
        # c[k] = sum_i h[i](1-h[i+k]) for valid i
        # Terms involving h[j]:
        # 1. i = j (if j+k valid): h[j](1-h[j+k]) → a += (1-h[j+k]), b += 0
        # 2. i+k = j, i = j-k (if j-k valid): h[j-k](1-h[j]) → a += -h[j-k], b += h[j-k]
        # Other terms: b += h[i](1-h[i+k]) for i ≠ j and i+k ≠ j
        
        # Actually, it's easier to just compute a_k and b_k directly
        a_k = 0.0
        b_k = 0.0
        for i in range(n):
            i2 = i + k
            if 0 <= i2 < n:
                if i == j:
                    a_k += (1.0 - h[i2])
                elif i2 == j:
                    a_k += (-h[i])
                    b_k += h[i]
                else:
                    b_k += h[i] * (1.0 - h[i2])
        a[m] = a_k
        b[m] = b_k
    
    # Now minimize max_m (a[m]*x + b[m]) subject to 0 <= x <= 1
    # This is: minimize f(x) = max_m (a[m]*x + b[m])
    
    # Since f is convex, use ternary search or check breakpoints
    # For efficiency, let's use a simple approach:
    # The minimum is where the subgradient contains 0
    # Or: evaluate on a grid and refine
    
    # Simple approach: bisection on the subgradient
    # f'(x-) = max_m a[m] (for active constraints at x from left)
    # f'(x+) = min_m a[m] (for active constraints at x from right)
    # At optimal x: f'(x-) <= 0 <= f'(x+)
    
    # Even simpler: since n is small, just evaluate on a fine grid
    grid = np.linspace(0, 1, 101)
    vals = np.max(a[:, None] * grid[None, :] + b[:, None], axis=0)
    best_idx = np.argmin(vals)
    
    # Refine around best_idx
    lo = grid[max(0, best_idx - 1)]
    hi = grid[min(len(grid) - 1, best_idx + 1)]
    for _ in range(50):
        mid = (lo + hi) / 2
        f_lo = np.max(a * lo + b)
        f_hi = np.max(a * hi + b)
        # Check subgradient at mid
        active = a * mid + b == np.max(a * mid + b)
        if np.any(a[active] < -1e-10):
            hi = mid
        elif np.any(a[active] > 1e-10):
            lo = mid
        else:
            break
    
    return (lo + hi) / 2
```

Wait, this is O(n^2) per coordinate (due to the loop over i for each m), and O(n) coordinates, so O(n^3) per pass. For n=95, that's about 857,000 operations per pass. Fast enough.

But actually, I can vectorize the computation of a and b. Let me think...

For each lag k (where k ranges from -(n-1) to n-1):
- a_k = (1-h[j+k]) * [0 ≤ j+k < n] + (-h[j-k]) * [0 ≤ j-k < n]
- b_k = h[j-k] * [0 ≤ j-k < n] + sum_{i: i≠j, i+k≠j} h[i](1-h[i+k]) * [0 ≤ i+k < n]

The b_k computation is the tricky part. But note:
b_k = c[k] - a_k * h[j]  (where c[k] is the current correlation value)

So I can just compute c[k] once (using np.correlate) and then:
a_k = (1-h[j+k]) * [0 ≤ j+k < n] + (-h[j-k]) * [0 ≤ j-k < n]
b_k = c[k] - a_k * h[j]

This is O(n) per coordinate! Much faster.

Let me rewrite:

```python
def optimize_coordinate_fast(h, j, corr_full):
    """Optimize h[j] given current corr_full values."""
    n = len(h)
    # Compute a_k for all k
    # a_k = (1-h[j+k]) * [j+k valid] - h[j-k] * [j-k valid]
    a_vals = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        a_k = 0.0
        if 0 <= j + k < n:
            a_k += (1.0 - h[j + k])
        if 0 <= j - k < n:
            a_k += (-h[j - k])
        a_vals[m] = a_k
    
    # b_k = corr_full[m] - a_k * h[j]
    b_vals = corr_full - a_vals * h[j]
    
    # Minimize max_m (a_vals[m] * x + b_vals[m]) for x in [0,1]
    # Use bisection on subgradient
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = a_vals * mid + b_vals
        max_val = np.max(vals)
        active = vals >= max_val - 1e-15
        max_a = np.max(a_vals[active])
        min_a = np.min(a_vals[active])
        if min_a > 1e-10:
            # f is increasing at mid, go left
            hi = mid
        elif max_a < -1e-10:
            # f is decreasing at mid, go right
            lo = mid
        else:
            # 0 is in the subgradient, mid is optimal
            break
    
    return (lo + hi) / 2
```

This is O(n) per coordinate, O(n^2) per pass. Very fast!

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        diff = s - target_sum
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            if not np.any(mask):
                break
            h[mask] -= diff * (h[mask] / s)
        else:
            mask = h < 1.0 - 1e-15
            if not np.any(mask):
                break
            h[mask] += (-diff) * ((1.0 - h[mask]) / max(1e-15, n - s))
        h = np.clip(h, 0.0, 1.0)
    return h

def optimize_coordinate(h, j, corr_full):
    n = len(h)
    a_vals = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        a_k = 0.0
        if 0 <= j + k < n:
            a_k += (1.0 - h[j + k])
        if 0 <= j - k < n:
            a_k += (-h[j - k])
        a_vals[m] = a_k
    b_vals = corr_full - a_vals * h[j]
    
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        vals = a_vals * mid + b_vals
        max_val = np.max(vals)
        active = vals >= max_val - 1e-14
        if np.min(a_vals[active]) > 1e-12:
            hi = mid
        elif np.max(a_vals[active]) < -1e-12:
            lo = mid
        else:
            break
    return (lo + hi) / 2

def coordinate_descent_pass(h, target_sum):
    n = len(h)
    corr_full = np.correlate(h, 1.0 - h, mode="full")
    for j in range(n):
        new_val = optimize_coordinate(h, j, corr_full)
        h[j] = new_val
        # Update corr_full (only need to update lags involving j)
        # For simplicity, recompute
        corr_full = np.correlate(h, 1.0 - h, mode="full")
    h = project_feasible(h, target_sum)
    return h, corr_full

def subgradient(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = np.max(corr)
    tol = 0.01 * max_val + 1e-10
    active_mask = corr >= (max_val - tol)
    weights = active_mask.astype(float)
    w_sum = weights.sum()
    weights /= w_sum
    
    grad = np.zeros(n)
    j = np.arange(n)
    for m in np.where(active_mask)[0]:
        w = weights[m]
        k = m - (n - 1)
        v1 = (j >= k) & (j < n + k)
        v2 = (j >= -k) & (j < n - k)
        if np.any(v1):
            grad[v1] += w * (1.0 - h[j[v1] - k])
        if np.any(v2):
            grad[v2] -= w * h[j[v2] + k]
    return grad
