def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_grad(x, n, alpha):
    dx = 2.0 / n
    omx = 1.0 - x
    corr = np.correlate(x, omx, mode='full').astype(float) * dx
    m = np.max(corr)
    s = alpha * (corr - m)
    s = np.clip(s, -500, 0)
    e = np.exp(s)
    w = e / e.sum()
    val = m + np.log(e.sum()) / alpha
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        d = k - (n - 1)
        wk = w[k]
        if d >= 0:
            grad[:n - d] += wk * omx[d:]
            grad[d:] -= wk * x[:n - d]
        else:
            grad[-d:] += wk * omx[:n + d]
            grad[:n + d] -= wk * x[-d:]
    grad *= dx

def project_feasible(h, n):
    target = n / 2.0

        if p.sum() > target:
            lo = mid
        else:
            hi = mid
