def _c5_raw(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def _proj_box_sum(h, target):
    """Project h onto {x: 0<=x_i<=1, sum(x)=target} via bisection on offset."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def _soft_obj(x):
    alpha = _G['alpha']
    h = _proj_box_sum(x, _G['target'])
    corr = np.correlate(h, 1.0 - h, mode="full")
    s = alpha * corr
    mv = float(np.max(s))
    if mv > 500:
        return mv / alpha
    return (mv + np.log(float(np.sum(np.exp(s - mv))))) / alpha

def _sa_binary(h, n, dx, T_init, rng, deadline, cool=1e-5):
    """Simulated annealing on binary h; returns (best_h, best_c5)."""
    c5 = _c5_raw(h, dx)
    h_cur = h.copy()
    h_best_local = h.copy()
    c5_best_local = c5
    T = T_init
    while time.time() < deadline:
        j = rng.randint(n)
        h_cur[j] = 1.0 - h_cur[j]
        c5_new = _c5_raw(h_cur, dx)
        d = c5_new - c5
        if d <= 0 or rng.rand() < np.exp(-d / T):
            c5 = c5_new
            if c5 < c5_best_local:
                c5_best_local = c5
                h_best_local = h_cur.copy()
        else:
            h_cur[j] = 1.0 - h_cur[j]
        T = max(T * (1.0 - cool), 1e-6)
    return h_best_local, c5_best_local

def _make_binary(n, rng, target_int):
    h = (rng.rand(n) < 0.5).astype(float)
    cnt = int(np.sum(h))
    if cnt != target_int:
        idx = np.argsort(rng.rand(n))
        if cnt > target_int:
            h[idx[:cnt - target_int]] = 0.0
        else:
            h[idx[::-1][:target_int - cnt]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float)
    except Exception:
        h0 = None
    if h0 is None or len(h0) == 0:
        n = 70
        h0 = (rng.rand(n) < 0.5).astype(float)
    else:
        n = len(h0)

    dx = 2.0 / n
    target = n / 2.0
    target_int = int(target)

    h0 = _proj_box_sum(h0, target)
    c5_best = _c5_raw(h0, dx)
    h_best = h0.copy()

    # Phase 1: Binary SA from rounded initial
    sa_deadline = t0 + budget_s * 0.50
    h_b = (h0 >= 0.5).astype(float)
    cnt = int(np.sum(h_b))
    if cnt != target_int:
        idx = np.argsort(h0)
        if cnt > target_int:
            h_b[idx[:cnt - target_int]] = 0.0
        else:
            h_b[idx[::-1][:target_int - cnt]] = 1.0

    h_b, c5_b = _sa_binary(h_b, n, dx, 0.12, rng, sa_deadline, cool=8e-6)
    if c5_b < c5_best:
        c5_best = c5_b
        h_best = h_b.copy()

    # Phase 2: Binary SA from multiple random starts
    for r in range(8):
        if time.time() > sa_deadline:
            break
        h_r = _make_binary(n, rng, target_int)
        # Some diversity: try shifted/reversed variants
        if r % 3 == 1:
            h_r = np.roll(h_r, rng.randint(n))
            # Re-fix count after roll (roll preserves count, so fine)
        elif r % 3 == 2:
            h_r = 1.0 - h_r  # complement, then fix count
            cnt2 = int(np.sum(h_r))
            if cnt2 != target_int:
                idx2 = np.argsort(h_r)
                if cnt2 > target_int:
                    h_r[idx2[:cnt2 - target_int]] = 0.0
                else:
                    h_r[idx2[::-1][:target_int - cnt2]] = 1.0
        h_r, c5_r = _sa_binary(h_r, n, dx, 0.10, rng, min(sa_deadline, t0 + budget_s * 0.52), cool=1e-5)
        if c5_r < c5_best:
            c5_best = c5_r
            h_best = h_r.copy()

    # Phase 3: Continuous refinement via soft-max + L-BFGS-B
    _G['n'] = n
    _G['dx'] = dx
    _G['target'] = target
    cont_deadline = t0 + budget_s * 0.82

    for alpha in [80, 150, 300]:
        if time.time() > cont_deadline:
            break
        _G['alpha'] = alpha
        bounds = [(0.0, 1.0)] * n
        for restart in range(4):
            if time.time() > cont_deadline:
                break
            if restart == 0:
                x0 = h_best.copy()
            elif restart == 1:
                x0 = _proj_box_sum(h_best * 0.5 + 0.25, target)
            else:
                x0 = _proj_box_sum(h_best + rng.randn(n) * 0.04, target)
            try:
                res = minimize(_soft_obj, x0, method='L-BFGS-B',
                             bounds=bounds, maxiter=150,
                             options={'maxfun': 400, 'ftol': 1e-14})
                h_r = _proj_box_sum(res.x, target)
                c5_r = _c5_raw(h_r, dx)
                if c5_r < c5_best:
                    c5_best = c5_r
                    h_best = h_r
            except Exception:
                pass

    # Phase 4: Short binary SA polish on best continuous solution
    h_polish = (h_best >= 0.5).astype(float)
    cnt_p = int(np.sum(h_polish))
    if cnt_p != target_int:
        idx_p = np.argsort(h_best)
        if cnt_p > target_int:
            h_polish[idx_p[:cnt_p - target_int]] = 0.0
        else:
            h_polish[idx_p[::-1][:target_int - cnt_p]] = 1.0
    h_pol, c5_pol = _sa_binary(h_polish, n, dx, 0.05, rng,
                                min(t0 + budget_s * 0.88, cont_deadline + budget_s * 0.06), cool=2e-5)
    if c5_pol < c5_best:
        c5_best = c5_pol
        h_best = h_pol.copy()

    # Phase 5: Multi-size exploration
    for n_try in [50, 60, 80, 90, 100, 120, 140]:
        if time.time() > t0 + budget_s * 0.97:
            break
        dx_t = 2.0 / n_try
        tgt_t = n_try / 2.0
        ti_t = int(tgt_t)
        h_t = _make_binary(n_try, rng, ti_t)
        h_t, c5_t = _sa_binary(h_t, n_try, dx_t, 0.10, rng,
                                t0 + budget_s * 0.97, cool=1.2e-5)
        if c5_t < c5_best:
            c5_best = c5_t
            h_best = h_t
            n = n_try
            dx = dx_t

    # Final: ensure h_best is valid
    h_best = _proj_box_sum(h_best, target)
    c5_final = _c5_raw(h_best, dx)
    if c5_final < c5_best:
        c5_best = c5_final

    return (h_best, c5_best, n)
