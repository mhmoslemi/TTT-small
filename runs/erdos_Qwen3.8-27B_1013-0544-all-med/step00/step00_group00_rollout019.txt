import time

def compute_c5(h, dx):
    """Compute C5 = max_k of cross-correlation * dx."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_objective_and_grad(x, dx, n, alpha, penalty):
    """Compute soft-max objective and gradient."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_vals = corr * dx
    scaled = alpha * c_vals
    
    # Soft-max
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)
    sum_exp = np.sum(exp_vals)
    probs = exp_vals / sum_exp  # These are the soft-max weights
    
    # Objective with constraint penalty
    obj = (max_scaled + np.log(sum_exp)) / alpha
    integral = np.sum(x) * dx
    penalty_val = penalty * (integral - 1.0) ** 2
    total_obj = obj + penalty_val
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = probs[k] * dx
        if w < 1e-15:
            continue
        lo = max(0, n - 1 - k)
        hi = min(n - 1, 2 * n - 1 - k)
        indices = np.arange(lo, hi + 1)
        other = indices + k - (n - 1)
        grad[indices] += w * (1.0 - x[other])
        grad[other] += w * (-x[indices])
    
    # Constraint gradient
    penalty_grad = 2.0 * penalty * (integral - 1.0) * dx
    grad += penalty_grad
    
    return total_obj, grad

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_and_grad(x, dx, n, alpha, penalty):
    corr = np.correlate(x, 1.0 - x, mode='full')
    c_vals = corr * dx
    scaled = alpha * c_vals
    max_s = np.max(scaled)
    exps = np.exp(scaled - max_s)
    sum_e = np.sum(exps)
    probs = exps / sum_e
    soft_obj = (max_s + np.log(sum_e)) / alpha
    integral = np.sum(x) * dx
    pen_val = penalty * (integral - 1.0) ** 2
    total_obj = soft_obj + pen_val
    grad = np.zeros(n)
    num_lags = 2 * n - 1
    for k in range(num_lags):
        w = probs[k] * dx
        if w < 1e-14:
            continue
        lo1 = max(0, n - 1 - k)
        hi1 = min(n - 1, 2 * n - 2 - k)
        if hi1 >= lo1:
            idx = np.arange(lo1, hi1 + 1)
            other = idx + k - n + 1

            other2 = idx2 - k + n - 1
            grad[idx2] += w * (-x[other2])
    grad += 2.0 * penalty * (integral - 1.0) * dx
