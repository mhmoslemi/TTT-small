        # Gradient of lse = sum_k w_k * c_k
        # c_k = sum_i h[i]*(1-h[i+s]) where s = k-n+1
        # dc_k/dh[j] = (1-h[j+s])*1[0<=j+s<n] - h[j-s]*1[0<=j-s<n]
        g = np.zeros(n)
        for s in range(-(n - 1), n):
            k = s + n - 1
            wk = w[k]
            # (1-h[j+s]) for valid j: j from max(0,-s) to min(n-1, n-1-s)
            j_start = max(0, -s)
            j_end = min(n, n - s)
            if j_start < j_end:
                g[j_start:j_end] += wk * (1.0 - h[j_start + s:j_end + s])
            # -h[j-s] for valid j: j from max(0,s) to min(n-1, n-1+s)
            j2_start = max(0, s)
            j2_end = min(n, n + s)
            if j2_start < j2_end:
                g[j2_start:j2_end] -= wk * h[j2_start - s:j2_end - s]
        g *= temp * (2.0 / n)
        # Add penalty gradient
        g += 2.0 * pw * (np.sum(h) - n / 2.0)
        return g

    def repair(h, n):
        h = np.clip(h, 0.0, 1.0)
        target = n / 2.0
        s = np.sum(h)
        if s > 1e-15:
            h = h * (target / s)
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-14:
            if diff > 0:
                order = np.argsort(1.0 - h)
                for i in order:
                    a = min(diff, 1.0 - h[i])
                    h[i] += a
                    diff -= a
                    if diff <= 0:
                        break
            else:
                order = np.argsort(h)
                for i in order:
                    a = min(-diff, h[i])
                    h[i] -= a
                    diff += a
                    if diff >= 0:
                        break
        return h

    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float)
        c = eval_c5(h0)
        best_c5 = c
        best_h = h0.copy()
    except Exception:
        pass

    if best_h is None:
        n = 128
        best_h = np.zeros(n)
        best_h[::2] = 1.0
        best_c5 = eval_c5(best_h)

    n_candidates = sorted(set([len(best_h), 96, 128, 160, 192, 256]))

    for n in n_candidates:
        if time.time() - t0 > budget_s * 0.82:
            break
        target = n / 2.0
        bounds = [(0.0, 1.0)] * n

        inits = []

        # Resample best
        if len(best_h) != n:
            ox = np.linspace(0, 1, len(best_h), endpoint=False)
            nx = np.linspace(0, 1, n, endpoint=False)
            inits.append(repair(np.interp(nx, ox, best_h), n))
        else:
            inits.append(best_h.copy())

        # Alternating binary
        h_a = np.zeros(n)
        h_a[::2] = 1.0
        inits.append(h_a.copy())

        # Random binary
        inits.append(repair((rng.random(n) < 0.5).astype(float), n))

        # Two blocks
        h_b = np.zeros(n)
        q = n // 4
        h_b[:q] = 1.0
        h_b[n//2:n//2 + q] = 1.0
        inits.append(repair(h_b, n))

        # Three-level (0, 0.5, 1) with varying ratios
        for frac in [0.3, 0.5, 0.7]:
            h_3 = np.full(n, 0.5)
            n_one = int(round(n * frac * 0.5))
            n_zero = int(round(n * (1.0 - frac) * 0.5))
            idx = rng.permutation(n)
            h_3[idx[:n_one]] = 1.0
            h_3[idx[n_one:n_one + n_zero]] = 0.0
            inits.append(repair(h_3, n))

        # Smooth random
        r = rng.randn(n)
        kern = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
        r = np.convolve(r, kern, mode='same')
        h_r = np.clip(0.5 + 0.3 * r / (r.std() + 1e-10), 0, 1)
        inits.append(repair(h_r, n))

        # Optimize each init
        for hi in inits:
            if time.time() - t0 > budget_s * 0.88:
                break
            obj = lambda h: smooth_obj(h, n, 0.01, 500.0)
            res = minimize(obj, hi, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 300, 'gtol': 1e-4})
            h_opt = repair(res.x, n)
            c = eval_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt.copy()

        # Refinement with lower temperature
        if time.time() - t0 < budget_s * 0.92:
            if len(best_h) == n:
                start = best_h.copy()
            else:
                start = inits[0]
            obj2 = lambda h: smooth_obj(h, n, 0.003, 5000.0)
            res2 = minimize(obj2, start, method='L-BFGS-B', bounds=bounds,
                            options={'maxiter': 400, 'gtol': 1e-5})
            h_opt = repair(res2.x, n)
            c = eval_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt.copy()

    # SLSQP polish on best
    if time.time() - t0 < budget_s * 0.97:
        n = len(best_h)
        target = n / 2.0

        def obj_final(h):
            corr = np.correlate(h, 1.0 - h, mode="full")
            m = np.max(corr)
            return (m + 0.002 * np.log(np.sum(np.exp((corr - m) / 0.002)))) * (2.0 / n)

        res = minimize(obj_final, best_h, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                       options={'maxiter': 150, 'ftol': 1e-12})
        h_opt = repair(res.x, n)
        c = eval_c5(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()

    # Local perturbation search
    if time.time() - t0 < budget_s * 0.99:
        n = len(best_h)
        for _ in range(50):
            if time.time() - t0 > budget_s * 0.99:
                break
            h_pert = best_h + rng.randn(n) * 0.01
            h_pert = repair(h_pert, n)
            c = eval_c5(h_pert)
            if c < best_c5:
                best_c5 = c
                best_h = h_pert.copy()

    return best_h, best_c5, len(best_h)
