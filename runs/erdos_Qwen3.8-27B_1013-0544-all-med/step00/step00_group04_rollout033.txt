def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def local_search_1swap(h, time_limit, start_time):

def simulated_annealing(h, time_limit, start_time, T_init=0.004, cooling=0.99995):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = T_init
    while (time.time() - start_time) < start_time + time_limit and T > 1e-12:
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0

        delta = c - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-15)):
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= cooling

def upsample_smart(h, rng):
    n = len(h)
    h_new = np.zeros(2 * n)
    h_new[0::2] = h
    odd_pos = np.arange(1, 2 * n, 2)
    perm = rng.permutation(n)
    h_new[odd_pos[perm[:n // 2]]] = 1.0
    return h_new

def random_binary(n, rng):
    perm = rng.permutation(n)
    h = np.zeros(n)
    h[perm[:n // 2]] = 1.0
    return h
