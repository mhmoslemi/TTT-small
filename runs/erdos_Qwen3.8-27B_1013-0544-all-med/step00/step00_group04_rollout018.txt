def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_fast(h):
    """Faster version using the fact that for binary h, corr entries are integers."""
    n = len(h)
    dx = 2.0 / n

    # We can compute this more efficiently

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_int(h):
    """Return (max_corr_int, c5) for binary h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = int(np.max(corr))
    return m, m * dx

def local_search(h, max_rounds=2000, time_limit=None, start_time=None):
    n = len(h)
    h = h.astype(np.float64).copy()
    _, best_c5 = compute_c5_int(h)
    for _ in range(max_rounds):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h < 0.5)[0])
        if not ones or not zeros:
            break
        best_score = best_c5
        best_swap = None
        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                m, score = compute_c5_int(h)
                h[i] = 1.0
                h[j] = 0.0
                if score < best_score - 1e-12:
                    best_score = score
                    best_swap = (i, j)
        if best_swap is None:
            break
        i, j = best_swap
        h[i] = 0.0
        h[j] = 1.0
        best_c5 = best_score
    return h, best_c5

def local_search_sa(h, max_iter=50000, t_start=0.5, t_end=0.01, time_limit=None, start_time=None):

    best_h = h.copy()
    _, best_c5 = compute_c5_int(h)
    cur_c5 = best_c5
    for it in range(max_iter):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        T = t_start * (t_end / t_start) ** (it / max_iter)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        m, new_c5 = compute_c5_int(h)
        delta = new_c5 - cur_c5
        if delta < 0 or np.random.random() < np.exp(-delta / (T + 1e-15)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def paley_binary(n, seed=0):
    """Generate binary sequence initialized from Paley construction."""
    # Find a prime p >= n with p ≡ 3 (mod 4)
    p = n
    while p < n + 20:
        if p % 4 == 3:
            # Check prime
            is_p = True
            for d in range(2, int(np.sqrt(p)) + 1):
                if p % d == 0:
                    is_p = False
                    break
            if is_p:
                break
        p += 1
    # Legendre symbol
    leg = np.zeros(p)
    for i in range(1, p):
        leg[i] = 1.0 if pow(i, (p - 1) // 2, p) == 1 else 0.0
    leg[0] = 1.0
    # Pad or truncate to length n
    h = np.zeros(n)
    if n <= p:
        h[:] = leg[:n]
    else:
        h[:p] = leg
        # Fill rest randomly
        h[p:] = np.random.randint(0, 2, n - p).astype(np.float64)
    # Fix sum to n//2
    target = n // 2
    cur = int(np.round(h.sum()))
    if cur > target:
        idx = np.where(h > 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:cur - target]] = 0.0
    elif cur < target:
        idx = np.where(h < 0.5)[0]
        np.random.shuffle(idx)
        h[idx[:target - cur]] = 1.0
    return h

def upsample_binary(h, new_n):
    n = len(h)
    old_pos = np.arange(n) / n
    new_pos = np.arange(new_n) / new_n
    h_new = np.zeros(new_n)
    for i in range(new_n):
        d = np.abs(new_pos[i] - old_pos)
        h_new[i] = h[np.argmin(d)]
    target = new_n // 2
    cur = int(np.round(h_new.sum()))
    if cur > target:
        idx = np.where(h_new > 0.5)[0]
        np.random.shuffle(idx)
        h_new[idx[:cur - target]] = 0.0
    elif cur < target:
        idx = np.where(h_new < 0.5)[0]
        np.random.shuffle(idx)
        h_new[idx[:target - cur]] = 1.0
    return h_new

def random_binary(n, seed=None):
    if seed is not None:
        np.random.seed(seed)
    h = np.random.randint(0, 2, n).astype(np.float64)
    target = n // 2
    cur = int(h.sum())
    if cur > target:
        idx = np.where(h > 0.5)[0]

    elif cur < target:
        idx = np.where(h < 0.5)[0]
        np.random.shuffle(idx)

    return h
