    return float(np.max(corr) * dx)

def upsample(h, factor):

    n = len(h)
    new_n = n * factor
    x_old = np.arange(n)
    x_new = np.linspace(0, n - 1, new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_onto_constraint(h_new, new_n)

    lr = 0.01  # learning rate

import numpy as np
import time

def project_onto_constraint(g, n):
    """Project g onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(g - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_c5_and_grad(h, beta):
    """Compute smooth C5 approximation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(beta * (corr - m))
    weights = exp_vals / exp_vals.sum()
    
    # Gradient
    grad = np.zeros(n)
    n_full = 2 * n - 1
    for k in range(n_full):
        s = k - (n - 1)
        w = weights[k] * dx
        if abs(w) < 1e-18:
            continue
        # Part 1: [0<=j-s<n] * (1-h[j-s])
        js = max(0, s)
        je = min(n - 1, n - 1 + s)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad[idx] += w * (1.0 - h[idx - s])
        # Part 2: -[0<=j+s<n] * h[j+s]
        js = max(0, -s)
        je = min(n - 1, n - 1 - s)
        if js <= je:
            idx = np.arange(js, je + 1)
            grad[idx] -= w * h[idx + s]
    
    obj = m + np.log(exp_vals.sum()) / beta
    return obj, grad

def upsample(h, new_n):
    n = len(h)
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0, n - 1, new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_onto_constraint(h_new, new_n)

import numpy as np
import time

def project_onto_constraint(g, n):
    """Project g onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(g - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(h, beta):
    """Compute smooth objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m_max = np.max(corr)
    e = np.exp(beta * (corr - m_max))
    w = e / e.sum()
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        i_lo = max(0, m - n + 1)
        i_hi = min(n - 1, m)
        js = np.arange(i_lo, i_hi + 1)
        idxs = n - 1 - m + js
        grad[js] += wm * (1.0 - h[idxs])
        
        j2_lo = max(0, i_lo - m + n - 1)
        j2_hi = min(n - 1, i_hi - m + n - 1)
        if j2_lo <= j2_hi:
            js2 = np.arange(j2_lo, j2_hi + 1)
            idxs2 = js2 + m - n + 1
            grad[js2] -= wm * h[idxs2]
    
    obj = m_max + np.log(e.sum()) / beta
    return obj, grad

def upsample(h, new_n):
    n = len(h)
    if new_n == n:
        return h.copy()
    x_old = np.arange(n, dtype=float)
    x_new = np.linspace(0.0, n - 1.0, new_n)
    h_new = np.interp(x_new, x_old, h)
    return project_onto_constraint(h_new, new_n)

def optimize_grid(h0, n, time_budget, rng):
    """PGD with soft-max on fixed grid."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    beta = 5.0
    lr = 0.05
    stall = 0
    
    while time.time() - t0 < time_budget:
        obj, grad = smooth_obj_grad(h, beta)
        h_new = project_onto_constraint(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stall = 0
        else:
            stall += 1
            if stall > 500:
                noise = 0.05 * rng.standard_normal(n)
                h = project_onto_constraint(best_h + noise, n)
                lr = max(lr, 0.05)
                stall = 0
        
        h = h_new
        lr *= 0.999
        beta = min(beta + 0.01, 30.0)
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    
    h_init = np.array(initial_h_values, dtype=float)
