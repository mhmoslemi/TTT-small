For a fixed d, the gradient contribution to position m is:
(1-h[m-d]) if 0 ≤ m-d < N, minus h[m+d] if 0 ≤ m+d < N.

I can write this as:

def smooth_gradient_vec(h, N, beta, corr, cmax):
    exp_c = np.exp(beta * (corr - cmax))
    w = exp_c / exp_c.sum()
    grad = np.zeros(N)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        d = k - N + 1
        # (1 - h shifted right by d) - (h shifted left by d)
        # h[m-d] for m-d in [0,N-1] means m in [d, N-1+d]
        # h[m+d] for m+d in [0,N-1] means m in [-d, N-1-d]
        
        term1 = np.zeros(N)  # 1 - h[m-d] where valid
        if d >= 0:
            # m-d >= 0 means m >= d; m-d < N means m < N+d (always true since m < N and d < N)
            term1[d:] = 1.0 - h[:N-d]
        else:
            # m-d >= 0 means m >= d (d is negative, so m >= -|d|, always true for m >= 0)
            # m-d < N means m < N+d = N-|d|
            nd = -d
            term1[:N-nd] = 1.0 - h[nd:]
        
        term2 = np.zeros(N)  # h[m+d] where valid
        if d >= 0:
            # m+d < N means m < N-d
            term2[:N-d] = h[d:]
        else:
            # m+d >= 0 means m >= -d = |d|
            nd = -d
            term2[nd:] = h[:N-nd]
        
        grad += wk * (term1 - term2)
    return grad
