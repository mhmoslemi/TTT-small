def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def feasible_project(h, target_sum):
    """Project h onto {sum(h) = target_sum, 0 <= h <= 1}"""
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Simple uniform adjustment (not always optimal projection but good enough)
    # Try to adjust only where possible
    if diff > 0:
        # Need to increase sum
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        adj = diff / mask.sum()
        h[mask] = np.minimum(h[mask] + adj, 1.0)
    else:
        # Need to decrease sum
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        adj = -diff / mask.sum()
        h[mask] = np.maximum(h[mask] - adj, 0.0)
    return h
