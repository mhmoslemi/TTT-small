def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    h0 = np.array(initial_h_values, dtype=float)
    n = len(h0)
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n

    best_h = enforce_constraint(h0.copy(), target_sum)
    best_c5 = compute_c5(best_h)
    start_time = time.time()

    # Phase 1: SLSQP with equality constraint, increasing temperature
    for t in [15, 30, 60, 120, 300]:
        if time.time() - start_time > budget_s * 0.40:
            break
        h_init = best_h.copy()
        cons = [{'type': 'eq',
                 'fun': sum_eq_fun, 'args': (target_sum,),
                 'jac': sum_eq_jac, 'args': (target_sum,)}]
        try:
            result = minimize(obj_smooth, h_init, args=(n, target_sum, t, 0.0),
                             method='SLSQP', bounds=bounds, constraints=cons,
                             options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = enforce_constraint(result.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    # Phase 2: L-BFGS-B with penalty, multiple restarts
    for t in [40, 80, 200, 400]:
        if time.time() - start_time > budget_s * 0.65:
            break
        for restart in range(3):
            if time.time() - start_time > budget_s * 0.65:
                break
            if restart == 0:
                h_init = best_h.copy()
            elif restart == 1:
                h_init = best_h + 0.03 * np.random.randn(n)
                h_init = enforce_constraint(h_init, target_sum)
            else:
                h_init = best_h + 0.1 * np.random.randn(n)
                h_init = enforce_constraint(h_init, target_sum)
            try:
                result = minimize(obj_smooth, h_init, args=(n, target_sum, t, 2000.0),
                                 method='L-BFGS-B', bounds=bounds,
                                 options={'maxiter': 5000, 'ftol': 1e-15})
                h_new = enforce_constraint(result.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 3: Try doubling grid size
    if time.time() - start_time < budget_s * 0.72:
        n2 = min(2 * n, 180)
        if n2 > n:
            dx2 = 2.0 / n2
            target2 = n2 / 2.0
            x_old = np.arange(n) * (2.0 / n)
            x_new = np.arange(n2) * dx2
            fi = interp1d(x_old, best_h, kind='cubic', fill_value='extrapolate')
            h_large = fi(x_new)
            h_large = enforce_constraint(h_large, target2)
            bounds2 = [(0.0, 1.0)] * n2
            try:
                result = minimize(obj_smooth, h_large, args=(n2, target2, 50.0, 2000.0),
                                 method='L-BFGS-B', bounds=bounds2,
                                 options={'maxiter': 4000, 'ftol': 1e-15})
                h_new = enforce_constraint(result.x, target2)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 4: Final polish with very high temperature on current best
    if time.time() - start_time < budget_s * 0.85:
        n_f = len(best_h)
        target_f = n_f / 2.0
        bounds_f = [(0.0, 1.0)] * n_f
        for t in [500, 1000]:
            if time.time() - start_time > budget_s * 0.92:
                break
            try:
                result = minimize(obj_smooth, best_h, args=(n_f, target_f, t, 10000.0),
                                 method='L-BFGS-B', bounds=bounds_f,
                                 options={'maxiter': 5000, 'ftol': 1e-16})
                h_new = enforce_constraint(result.x, target_f)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 5: Small-step local search
    if time.time() - start_time < budget_s * 0.93:
        n_s = len(best_h)
        target_s = n_s / 2.0
        steps = [0.02, 0.01, 0.005]
        for ss in steps:
            if time.time() - start_time > budget_s * 0.96:
                break
            for i in range(n_s):
                if time.time() - start_time > budget_s * 0.96:
                    break
                for delta in [ss, -ss]:
                    nv = best_h[i] + delta
                    if 0.0 <= nv <= 1.0:
                        trial = best_h.copy()
                        trial[i] = nv
                        diff = target_s - np.sum(trial)
                        if abs(diff) < 2.0 * ss:
                            j = (i + 1) % n_s
                            trial[j] = best_h[j] + diff
                            if 0.0 <= trial[j] <= 1.0:
                                c5_t = compute_c5(trial)
                                if c5_t < best_c5:
                                    best_h = trial
                                    best_c5 = c5_t

    return best_h, best_c5, len(best_h)
