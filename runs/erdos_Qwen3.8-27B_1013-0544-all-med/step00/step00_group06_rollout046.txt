import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def smooth_obj_grad(h, alpha, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_corr = np.max(corr)
    shifted = alpha * (corr - max_corr)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    lse = max_corr + np.log(sum_exp) / alpha
    grad = np.zeros(n)
    active = w > 1e-14
    for m in range(n):
        s = 0.0
        for k_idx in np.where(active)[0]:
            k = k_idx - (n - 1)
            wi = w[k_idx]
            j1 = m + k
            j2 = m - k
            if 0 <= j1 < n:
                s += wi * (1.0 - h[j1])
            if 0 <= j2 < n:
                s -= wi * h[j2]
        grad[m] = dx * s
    diff = np.sum(h) - n * 0.5
    pen = penalty * diff * diff
    pg = 2.0 * penalty * diff
    return lse + pen, grad + pg

    n = len(h)
    target = n * 0.5
    s = np.sum(h)

    h_out = project_h(np.clip(res.x, 0.0, 1.0))
