def subgradient_descent(h, target_sum, time_limit, lr=0.01):
    n = len(h)
    h = h.copy()
    curr = compute_c5(h)
    best_h = h.copy()
    best = curr
    dx = 2.0 / n
    
    while time.time() < time_limit:
        corr = compute_corr(h)
        c5_val = np.max(corr) * dx
        
        # Find active shifts (within tolerance of max)
        tol = 0.01 * c5_val + 1e-6
        active = np.where(corr >= np.max(corr) - tol)[0]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)  # actual shift
            for j in range(n):
                # Contribution from h[j] as first factor
                js = j + s
                if 0 <= js < n:
                    grad[j] += (1 - h[js])
                # Contribution from h[j] as second factor
                jm = j - s
                if 0 <= jm < n:
                    grad[j] += (-h[jm])
        
        grad /= len(active)
        
        # Subgradient step
        alpha = lr / (1 + np.sqrt(time.time() - time_limit + 1))  # diminishing
        # Actually let me use a fixed small step
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target_sum)
        
        c5_try = compute_c5(h_try)
        if c5_try < curr:
            h = h_try
            curr = c5_try
            if curr < best:
                best = curr
                best_h = h.copy()
        else:
            lr *= 0.95
        
        if lr < 1e-8:
            break
    
    return best_h, best

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max(x, t):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(t * (x - m)))) / t

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        step = diff / mask.sum()
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def local_opt(h, target_sum, time_limit):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t_val in [30, 80, 150, 300]:
        if time.time() > time_limit:
            break
        def obj(p):
            hl = np.clip(p, 0.0, 1.0)
            c = compute_corr(hl)
            sm = smooth_max(c, t_val)
            pen = 500.0 * (np.sum(hl) - target_sum) ** 2
            return sm * dx + pen
        res = minimize(obj, best_h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-16})
        h_new = project_feasible(res.x, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    return best_h, best_c5

def subgrad_opt(h, target_sum, time_limit, rng):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    curr = compute_c5(h)
    best_h = h.copy()
    best = curr
    lr = 0.02
    it = 0
    while time.time() < time_limit:
        corr = compute_corr(h)
        max_corr = np.max(corr)
        tol = 0.02 * max_corr + 1e-7
        active = np.where(corr >= max_corr - tol)[0]
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            if s >= 0:
                grad[s:n] += (1.0 - h[:n-s])
                grad[:n-s] += (-h[s:n])
            else:
                sa = -s
                grad[:n-sa] += (1.0 - h[sa:n])
                grad[sa:n] += (-h[:n-sa])
        grad /= len(active)
        h_try = h - lr * grad
        h_try = project_feasible(h_try, target_sum)
        c5_try = compute_c5(h_try)
        if c5_try < curr:
            h = h_try
            curr = c5_try
            lr = min(lr * 1.05, 0.1)
            if curr < best:
                best = curr
                best_h = h.copy()
        else:
            lr *= 0.9
        if lr < 1e-9:
            break
        it += 1
    return best_h, best

def sa_optimize(h, target_sum, time_limit, rng):
    n = len(h)
    h = h.copy()
    curr = compute_c5(h)
    best_h = h.copy()
    best = curr
    T = 0.03
    T_min = 1e-9
    cool = 0.9995
    while time.time() < time_limit and T > T_min:
        r = rng.random()
        if r < 0.6:
            hi_mask = h > 0.4
            lo_mask = h < 0.6
            if hi_mask.sum() == 0 or lo_mask.sum() == 0:
                T *= cool
                continue
            i_hi = rng.choice(np.where(hi_mask)[0])
            i_lo = rng.choice(np.where(lo_mask)[0])
            if i_hi == i_lo:
                T *= cool
                continue
            delta = rng.uniform(0.02, 0.3)
            delta = min(delta, h[i_hi] - 0.01, 0.99 - h[i_lo])
            if delta < 1e-5:
                T *= cool
                continue
            h_try = h.copy()
            h_try[i_hi] -= delta
            h_try[i_lo] += delta
            c5_try = compute_c5(h_try)
        else:
            j = rng.randint(n)
            delta = rng.uniform(-0.4, 0.4)
            h_try = h.copy()
            h_try[j] = np.clip(h[j] + delta, 0.0, 1.0)
            h_try = project_feasible(h_try, target_sum)
            c5_try = compute_c5(h_try)
        dc = c5_try - curr
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-12)):
            h = h_try
            curr = c5_try
            if curr < best:
                best = curr
                best_h = h.copy()
        T *= cool
    return best_h, best

def coord_descent(h, target_sum, time_limit, max_passes=300):
    n = len(h)
    h = h.copy()
    curr = compute_c5(h)
    for p in range(max_passes):
        if time.time() > time_limit:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if h[j] > 0.5:
                h_try = h.copy()
                h_try[j] = 0.0
                h_try = project_feasible(h_try, target_sum)
            else:
                h_try = h.copy()
                h_try[j] = 1.0
                h_try = project_feasible(h_try, target_sum)
            c5_try = compute_c5(h_try)
            if c5_try < curr - 1e-12:
                h = h_try
                curr = c5_try
                improved = True
        if not improved:
            break
    return h, curr
