    h = np.clip(h, 0, 1)
    target = n / 2.0
    current = np.sum(h)
    if current == 0:
        h[:int(target)] = 1.0
        if target - int(target) > 0:
            h[int(target)] = target - int(target)
        return h
    # Scale to meet target
    h = h * (target / current)
    h = np.clip(h, 0, 1)
    # After clipping, might not meet target exactly; do iterative repair
    for _ in range(10):
        current = np.sum(h)
        diff = target - current
        if abs(diff) < 1e-10:

        if diff > 0:

        if scores[0] < best_score:

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def make_binary(h, n, rng):

    d = np.arange(-(n-1), n)  # lags
    delta = np.zeros(2*n-1)

    # Term 1: h[a+d]-1 for -a <= d <= n-1-a

    delta[mask1] += h[a + d[mask1]] - 1

    mask4 = (d >= b-n+1) & (d <= b)

def swap_delta(h, a, b, n):
    """Compute change in full correlation array when h[a]: 1->0, h[b]: 0->1."""
    # Full correlation has indices 0 to 2n-2
    # Index j corresponds to lag d = j - (n-1)
    # C[j] = sum_{valid i} h[i] * (1-h[i+d])
    
    d = np.arange(-(n-1), n)  # shape (2n-1,)
    delta = np.zeros(2*n-1)
    
    # h[a]: 1->0
    # In C, h[a] appears as:
    # (1) h[a]*(1-h[a+d]) when i=a, valid if 0<=a+d<n, i.e., -a<=d<n-a
    # (2) h[a-d]*(1-h[a]) when i=a-d, valid if 0<=a-d<n, i.e., a-n< d <= a (and d>=-(n-1))
    mask1 = (d >= -a) & (d < n-a)
    delta[mask1] += h[a + d[mask1]] - 1  # (h[a+d]-1) since we lose h[a]*(1-h[a+d])

    delta[mask2] += h[a - d[mask2]]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def swap_delta(h, a, b, n):
    d = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    m1 = (d >= -a) & (d < n-a)
    delta[m1] += h[a + d[m1]] - 1
    m2 = (d >= a-n+1) & (d <= a)
    delta[m2] += h[a - d[m2]]
    m3 = (d >= -b) & (d < n-b)
    delta[m3] += 1 - h[b + d[m3]]
    m4 = (d >= b-n+1) & (d <= b)
    delta[m4] += -h[b - d[m4]]
    return delta

def local_search(h, n, max_rounds=100, time_limit=30.0, start_time=0):
    C = np.correlate(h, 1.0 - h, mode="full").astype(float)
    best_c5 = float(np.max(C)) * (2.0/n)
    ones = set(int(x) for x in np.where(h == 1)[0])
    zeros = set(int(x) for x in np.where(h == 0)[0])
    
    for _ in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        best_new_max = float(np.max(C))
        best_swap = None
        best_delta = None
        for i in list(ones):
            for j in list(zeros):
                d = swap_delta(h, i, j, n)
                nm = float(np.max(C + d))
                if nm < best_new_max - 1e-12:
                    best_new_max = nm
                    best_swap = (i, j)
                    best_delta = d
        if best_swap is None:
            break
        i, j = best_swap
        C = C + best_delta
        h[i] = 0; h[j] = 1
        ones.discard(i); zeros.discard(j)
        ones.add(j); zeros.add(i)
        best_c5 = float(np.max(C)) * (2.0/n)
    return h, best_c5
