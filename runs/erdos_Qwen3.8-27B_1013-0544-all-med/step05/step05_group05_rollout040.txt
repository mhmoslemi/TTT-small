def compute_c5(h, n):

def project_feasible(h, n):

            s = h.sum()
            if s < 1e-14:

            h = h + diff / s * h

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def compute_all_g(h, n):
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def coord_sweep(h, n, g_full, max_sweeps=200):
    dx = 2.0 / n
    n1 = n - 1
    best_c = g_full.max()
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            c = np.zeros(2 * n - 1)
            L_lo1 = max(-j, -n1)
            L_hi1 = min(n1 - j, n1)
            if L_lo1 <= L_hi1:
                Ls = np.arange(L_lo1, L_hi1 + 1)
                c[Ls + n1] += 1.0 - h[j + Ls]
            L_lo2 = max(j - n1, -n1)
            L_hi2 = min(j, n1)
            if L_lo2 <= L_hi2:
                Ls = np.arange(L_lo2, L_hi2 + 1)
                c[Ls + n1] -= h[j - Ls]
            a = h[j]
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                d1, d2 = m1 - a, m2 - a
                v1 = np.max(g_full + d1 * dx * c)
                v1 = max(v1, g_full[n1] + d1 * dx * c[n1] - d1 * d1 * dx)
                v2 = np.max(g_full + d2 * dx * c)
                v2 = max(v2, g_full[n1] + d2 * dx * c[n1] - d2 * d2 * dx)
                if v1 < v2:
                    hi = m2
                else:
                    lo = m1
            b_opt = (lo + hi) / 2.0
            if abs(b_opt - a) > 1e-12:
                h[j] = b_opt
                d = b_opt - a
                g_full += d * dx * c
                g_full[n1] -= d * d * dx
                nc = g_full.max()
                if nc < best_c:
                    best_c = nc
                improved = True
        if not improved:
            break
    return h, g_full, best_c

def subgrad_polish(h0, n, deadline, max_iter=50000):
    best_h = h0.copy()
    best_c = compute_c5(best_h, n)
    lr = 0.05
    dx = 2.0 / n
    n1 = n - 1
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_all_g(best_h, n)
        m = g.max()
        alpha = 200.0
        shifted = (g - m) * alpha
        w = np.exp(shifted - shifted.max())
        w /= w.sum()
        grad = np.zeros(n)
        for j in range(n):
            s = 0.0
            L_lo1 = max(-j, -n1)
            L_hi1 = min(n1 - j, n1)
            if L_lo1 <= L_hi1:
                Ls = np.arange(L_lo1, L_hi1 + 1)
                s += np.sum(w[Ls + n1] * (1.0 - best_h[j + Ls]))
            L_lo2 = max(j - n1, -n1)
            L_hi2 = min(j, n1)
            if L_lo2 <= L_hi2:
                Ls = np.arange(L_lo2, L_hi2 + 1)
                s -= np.sum(w[Ls + n1] * best_h[j - Ls])
            grad[j] = dx * s
        ng = np.linalg.norm(grad)
        if ng < 1e-14:
            break
        grad /= ng
        found = False
        for f in [1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001]:
            step = lr * f
            ht = project_feasible(best_h - step * grad, n)
            ct = compute_c5(ht, n)
            if ct < m - 1e-13:
                best_h = ht
                if ct < best_c:
                    best_c = ct
                found = True
                break
        if not found:
            lr *= 0.7
            if lr < 1e-12:
                break
    return best_h, best_c

def binary_local_search(h0, n, deadline, max_sweeps=20000, seed=123):
    target = n // 2
    idx = np.argsort(h0)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    best_c = compute_c5(h, n)
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(zeros)
        improved = False
        for i in zeros:
            if time.time() > deadline:
                break
            best_j = -1
            best_cj = best_c
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                c = compute_c5(h, n)
                if c < best_cj:
                    best_cj = c
                    best_j = j
                h[i] = 0.0
                h[j] = 1.0
            if best_j >= 0:
                h[i] = 1.0
                h[best_j] = 0.0
                best_c = best_cj
                improved = True
                break
        if not improved:
            break
    return h, best_c
