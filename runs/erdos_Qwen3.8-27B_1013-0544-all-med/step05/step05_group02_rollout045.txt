def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k_idx):
    s = k_idx - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, s)
    hi1 = min(n - 1, n - 1 + s)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += (1.0 - h[lo1 - s:hi1 - s + 1])
    lo2 = max(0, -s)
    hi2 = min(n - 1, n - 1 - s)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] -= h[lo2 + s:hi2 + s + 1]
    return g

def smooth_max_and_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sh -= sh.max()
    exp_s = np.exp(sh)
    w = exp_s / exp_s.sum()
    f = m + np.log(exp_s.sum()) / alpha
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] > 1e-8:
            g += w[k] * dx * lag_gradient(h, n, k)
    return f, g

def slsqp_optimize(h, n, alpha, maxiter=3000):

        return f
    def obj_grad(x):
        hx = np.clip(x, 0, 1)
        _, g = smooth_max_and_grad(hx, n, alpha)
        return g
    bounds = [(0.0, 1.0)] * n

             'jac': lambda x: np.ones(n)}]

                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})

    except Exception:
        return h.copy()

    best_c = compute_c5(h, n)
    best_h = h.copy()
    c = get_corr(h, n)
    stall = 0
    step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    si = 0
    while time.time() < deadline:
        top_k = np.argsort(c)[-num_lags:]
        g = np.zeros(n)
        for k in top_k:

            c = new_c

        else:
