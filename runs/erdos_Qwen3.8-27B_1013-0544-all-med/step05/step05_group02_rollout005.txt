import time

    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):

    h = np.clip(h.copy(), 0.0, 1.0)

        if diff > 0:

            tot = room.sum()
            if tot < 1e-14:

        else:

            if tot < 1e-14:
                break
            h += (diff / tot) * h

def sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def p_norm_slsqp(h, n, p, maxiter=500):
    x0 = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': lambda x: np.ones(n), 'args': (n,)}]
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = get_corr(hx, n)
        return float(np.sum(c**p)) ** (1.0/p)
    def obj_grad(x):
        hx = np.clip(x, 0, 1)
        c = get_corr(hx, n)
        s = c**p
        S = s.sum()
        if S < 1e-30:
            return np.zeros(n)
        denom = S ** (1.0 - 1.0/p)
        w = (c**(p-1)) / denom
        dx = 2.0 / n

            lo = max(0, -s_lag)

            g[idx] += w[k] * (1.0 - hx[idx + s_lag]) * dx
            lo2 = max(0, lo + s_lag)
            hi2 = min(n, hi + s_lag)
            idx2 = np.arange(lo2, hi2)
            g[idx2] -= w[k] * hx[idx2 - s_lag] * dx
