def project_feasible(h, n):

    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):

        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break

        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h

def subgradient_at_max(h, n):
    """Returns gradient of c[k*] * dx where k* = argmax, and the max value."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    k_star = c.argmax()
    k = k_star - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] += -h[k:]
    else:
        grad[:n+k] += (1.0 - h[-k:])
        grad[-k:] += -h[:n+k]
    grad *= dx
    return grad, k_star, c * dx

def frank_wolfe(h, n, deadline, max_iter=5000):
    """Frank-Wolfe (conditional gradient) for the non-smooth problem."""

    best_h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    for t in range(max_iter):

            break
        grad, k_star, cdx = subgradient_at_max(h, n)
        # Linear minimization: find vertex v of {x in [0,1]^n, sum(x)=n/2}
        # that minimizes grad · v
        # This means set v[i]=1 for the n/2 indices with smallest grad[i]
        idx = np.argsort(grad)
        v = np.zeros(n)

        h_new = np.clip(h_new, 0.0, 1.0)

        if c_new < best_c:

            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

            min(n-1-i, i) = n-1-i
So s ranges from i-n+1 to n-1-i, giving 2(n-1-i)+1 values.

In any case, at most n values of s are affected. Good.

So for a swap (i,j, δ):
- Affected shifts due to h[i] changing: s in range_A(i) (at most n values)
- Affected shifts due to h[j] changing: s in range_A(j) (at most n values)
- Total affected: at most 2n values, but with overlap

For each affected shift s, c[s] changes by:
Δc[s] = δ*(1-h[i+s]) - δ*h[i-s] - δ*(1-h[j+s]) + δ*h[j-s] (terms that are valid)

This is linear in δ. So c[s](δ) = c[s](0) + a_s * δ where a_s is computable.

We want to minimize max_s c[s](δ) * dx.
This is a 1D piecewise-linear minimization! The answer is where the maximum of the lines {c[s](0) + a_s * δ} is minimized.

This can be solved by finding the intersection points and checking which one gives the minimum maximum.

But this is only valid if all other c[s] (for s not affected) stay below the new maximum. We need to check that.

This gives us an O(n) per pair operation (to compute a_s for affected s) and O(n log n) to find the optimal δ (sort the breakpoints). Total per sweep: O(n³) for all pairs, same as before.

Hmm, that's not an improvement. But we can be smarter: only try pairs where the affected shifts include the current bottleneck shift.

**Key insight**: The current bottleneck is at shift s* = argmax. To reduce C₅, we need to reduce c[s*]. The gradient at s* tells us which directions help. So we should only try swaps (i,j) where the subgradient has a useful structure.

Specifically, the subgradient at s* has components:
g[i] = (1-h[i+s*])*dx - h[i-s*]*dx (for valid i)

A swap from j to i (h[i]+=δ, h[j]-=δ) changes c[s*] by δ*(g[i] - g[j]) (approximately, ignoring the effect on other shifts).

So we want g[i] > g[j] for the swap to reduce c[s*].

This gives us a priority: try pairs (i,j) with g[i] - g[j] > 0, sorted by g[i] - g[j] descending.

def subgradient_at_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    k_star = c.argmax()
    k = k_star - (n - 1)
    grad = np.zeros(n)

        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] += -h[k:]
    else:
        grad[:n+k] += (1.0 - h[-k:])
        grad[-k:] += -h[:n+k]
    grad *= dx
    return grad, k_star, c * dx

        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e5 * (x.sum() - target) ** 2
        return val
    
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()

        else:
            s = h.sum()
            if s < 1e-14: break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def build_coeff_matrix(fixed, n):
    """Build coefficient matrix where row k gives coefficients for the free variable."""
    # c_k = sum_i free[i] * coeff[k, i]
    # where fixed is the other function
    m = 2 * n - 1
    coeff = np.zeros((m, n))
    for k_offset in range(m):
        k = k_offset - (n - 1)
        if k >= 0:
            coeff[k_offset, :n-k] = fixed[k:n]
        else:
            coeff[k_offset, -k:] = fixed[:n+k]
    return coeff

def solve_lp(fixed, n, method='highs'):
    """Solve: min t s.t. t >= dx * (coeff @ x) for all rows, 0<=x<=1, sum(x)=n/2."""
    dx = 2.0 / n
    m = 2 * n - 1
    coeff = build_coeff_matrix(fixed, n)
    
    # Variables: [x[0],...,x[n-1], t]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # A_ineq @ vars <= b_ineq
    # t - dx * (coeff[k] @ x) >= 0  →  dx * coeff[k] @ x - t <= 0
    A = np.zeros((m, n + 1))
    A[:, :n] = dx * coeff
    A[:, -1] = -1.0
    b = np.zeros(m)
    
    # Equality: sum(x) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ineq=A, b_ineq=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method=method)
        if res.success and res.x is not None:
            return np.clip(res.x[:n], 0, 1), float(res.x[-1] * dx)
    except Exception:
        pass
    return None, None

    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgradient_at_max(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    k_star = c.argmax()
    k = k_star - (n - 1)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] += -h[k:]
    else:
        grad[:n+k] += (1.0 - h[-k:])
        grad[-k:] += -h[:n+k]
    grad *= dx
    return grad, k_star, c * dx

def lbfgs_smooth(h, n, alpha, maxiter=500):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
        val += 1e5 * (x.sum() - target) ** 2
        return val
    def grad(x):
        c = np.correlate(x, 1 - x, mode='full') * dx
        m = c.max()
        sh = alpha * (c - m)
        sm = sh.max()
        w = np.exp(sh - sm)
        w /= w.sum()
        g = np.zeros(n)
        for j in range(2*n-1):
            if w[j] < 1e-8:
                continue
            k = j - (n-1)
            if k >= 0:
                g[k:] += w[j] * (1.0 - x[:n-k])
                g[:n-k] -= w[j] * x[k:]
            else:
                g[:n+k] += w[j] * (1.0 - x[-k:])
                g[-k:] -= w[j] * x[:n+k]
        g *= dx
        g += 2e5 * (x.sum() - target)
        return g
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0,1.0)]*n,
                       options={'maxiter':maxiter,'ftol':1e-18,'gtol':1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def build_coeff(fixed, n):
    m = 2*n - 1
    coeff = np.zeros((m, n))
    for ko in range(m):
        k = ko - (n-1)
        if k >= 0:
            coeff[ko, :n-k] = fixed[k:]
        else:
            coeff[ko, -k:] = fixed[:n+k]
    return coeff

def solve_lp_side(fixed, n):
    dx = 2.0 / n
    m = 2*n - 1
    coeff = build_coeff(fixed, n)
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    A = np.zeros((m, n+1))
    A[:, :n] = dx * coeff
    A[:, -1] = -1.0
    b = np.zeros(m)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    bounds = [(0,1)]*n + [(0, None)]
    try:
        res = linprog(c_obj, A_ineq=A, b_ineq=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return np.clip(res.x[:n], 0, 1)
    except Exception:
        pass
    return None

def alternating_lp(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alpha = 0.3
    stall = 0
    for it in range(10000):
        if time.time() > deadline:
            break
        h_new = solve_lp_side(1.0 - h, n)
        if h_new is not None:
            h = project_feasible((1-alpha)*h + alpha*h_new, n)
            c = compute_c5(h, n)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        v_new = solve_lp_side(h, n)
        if v_new is not None:
            h = project_feasible((1-alpha)*h + alpha*(1.0 - v_new), n)
            c = compute_c5(h, n)
            if c < best_c - 1e-12:
                best_c = c
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        if it % 50 == 0:
            alpha = min(0.7, 0.2 + 0.01*it/50)
        if stall > 200:
            alpha = max(0.1, alpha * 0.8)
            stall = 0
            h = project_feasible(best_h + 0.01*np.random.RandomState(it).randn(n), n)
            best_c = compute_c5(h, n)
    return best_h, best_c

def local_search(h, n, deadline):
    h = project_feasible(h.copy(), n)
    best_c = compute_c5(h, n)
    for it in range(50000):
        if time.time() > deadline:
            break
        grad, k_star, cdx = subgradient_at_max(h, n)
        i = int(grad.argmax())
        j = int(grad.argmin())
        if i == j or grad[i] - grad[j] < 1e-10:
            break
