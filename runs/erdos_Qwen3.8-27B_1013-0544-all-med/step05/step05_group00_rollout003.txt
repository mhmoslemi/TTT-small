                   #   from j: -= (1-h[j+(j-i)]) = -(1-h[2j-i]) (some value)
                   #            += h[j-(j-i)] = h[i] = 0 (h[i]=0)
                   # Actual change at shift j-i:
                   #   h[i]*(1-h[j]) goes 0→1: change +1
                   #   h[2j-i]*(1-h[2j-i+(j-i)])... nope, I'm overcomplicating.
                   
                   # The correct cross correction:
                   # At shift s where i+s=j (s=j-i): the product h[i]*(1-h[i+s]) = h[i]*(1-h[j])
                   # changes from 0*(1-1)=0 to 1*(1-0)=1, so +1.
                   # My linear formula gives: (1-h[j])*1 + h[i]*(-1) [from the two contributions]
                   # = (1-1) + 0*(-1) = 0. So I need to add +1.
                   
                   # At shift s where j+s=i (s=i-j): the product h[j]*(1-h[j+s]) = h[j]*(1-h[i])
                   # changes from 1*(1-0)=1 to 0*(1-1)=0, so -1.
                   # My linear formula gives: -(1-h[i])*(-1) + h[j-s]*1 [hmm I'm confusing myself]
                   
                   # Let me just add the two cross terms:
                   sj = j - i
                   si = i - j
                   k1 = n - 1 + sj
                   if 0 <= k1 < 2*n-1:
                       delta[k1] += 1.0  # h[i]*(1-h[j]) goes from 0 to 1
                   k2 = n - 1 + si
                   if 0 <= k2 < 2*n-1:
                       delta[k2] -= 1.0  # h[j]*(1-h[i]) goes from 1 to 0
                   
                   new_corr = corr + delta
                   nc = new_corr.max() * dx
                   if nc < cur_c - 1e-14:
                       h_bin[i], h_bin[j] = 1.0, 0.0
                       corr = new_corr
                       cur_c = nc
                       improved = True
                       if cur_c < best_c:
                           best_c = cur_c
                           best_h = h_bin.copy()
                       # Update ones/zeros
                       ones = np.where(h_bin == 1)[0]
                       zeros = np.where(h_bin == 0)[0]
                       break
               if improved:
                   break
           
           if not improved:
               # Try random swaps to escape local optima
               if rng.rand() < 0.1:
                   i = np.random.choice(zeros)
                   j = np.random.choice(ones)
                   h_bin[i], h_bin[j] = 1.0, 0.0
                   corr = np.correlate(h_bin, 1 - h_bin, mode='full').astype(float)
                   cur_c = corr.max() * dx
                   ones = np.where(h_bin == 1)[0]
                   zeros = np.where(h_bin == 0)[0]
               else:
                   break
       
       return best_h, best_c
