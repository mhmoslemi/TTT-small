def sa_binary(h, n, deadline, t_start=1.0, t_end=0.01, n_iters=None):
    """Simulated annealing on binary space with exactly n/2 ones."""
    if n_iters is None:
        n_iters = 1000000
    hb = h.copy()
    best_hb = hb.copy()
    best_c = compute_c5(hb, n)
    cur_c = best_c
    t = t_start
    
    # Precompute positions of ones and zeros
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    
    cooling = (t_end / t_start) ** (1.0 / n_iters)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        # Pick random swap
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]

        hb[i], hb[j] = 0.0, 1.0
        new_c = compute_c5(hb, n)
        delta = new_c - cur_c
        
        if delta < 0 or np.random.random() < np.exp(-delta / max(t, 1e-10)):
            # Accept
            cur_c = new_c
            ones = np.where(hb == 1)[0]
            zeros = np.where(hb == 0)[0]

                best_c = new_c
                best_hb = hb.copy()

            # Reject
            hb[i], hb[j] = 1.0, 0.0

        # FW step
        h_new = (1.0 - gamma) * h + gamma * b
        h_new = project_feasible(h_new, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c - 1e-10:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            gamma = min(gamma * 1.02, 0.9)
            stall = 0
        else:
            gamma *= 0.7
            stall += 1
            if gamma < 0.02:
                gamma = 0.4
    
    return best_h, best_c

def make_alt_initial(n, seed):
    """Alternating pattern with slight perturbation."""
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    h[0::2] = 1.0
    h[1::2] = 0.0
    # Perturb
    h += 0.1 * rng.randn(n)
    return project_feasible(h, n)
