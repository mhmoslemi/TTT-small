from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_round(h, n, alpha, maxiter=2000):
    dx = 2.0 / n
    def obj(x):
        h_x = np.clip(x, 0, 1)
        c = np.correlate(h_x, 1 - h_x, mode='full') * dx
        m = c.max()
        shifted = alpha * (c - m)
        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def worst_lag_gradient(h, n, k_star):
    s = k_star - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1)
        g[idx2] -= h[idx2 - s]
    return g

    dx = 2.0 / n
    c = get_corr(h, n)
    best_c = float(c.max())
    best_h = h.copy()
    steps = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

    while time.time() < deadline:

        g = worst_lag_gradient(h, n, k_star)

            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = steps[step_idx]

        if d < 1e-10:

            stall += 1

            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = get_corr(h, n)

            c = new_c

            stall += 1
            step_idx = min(step_idx + 1, len(steps) - 1)

    return best_h, best_c

    best_c = compute_c5(h, n)
    curr_h = h.copy()

        h_opt, c_opt = optimize_solution(h_init, n_init, t0 + alloc, seed=1)
        best_h, best_c = h_opt, c_opt
        x_init = np.linspace(0, 2, n_init)
        for n_try in [150, 200, 350]:
            if time.time() > deadline - 80:
                break
            remaining = deadline - time.time()
            alloc = min(100, remaining * 0.2)
            if alloc < 50:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_solution(h_interp, n_try, time.time() + alloc, seed=2)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 70:
        n = len(best_h)
        remaining = deadline - time.time() - 30
        n_restarts = max(4, int(remaining / 55))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.01 + 0.08 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(55, (deadline - time.time() - 30) * 0.35)
            if alloc < 25:
                break
            h_opt, c_opt = optimize_solution(hp, n, time.time() + alloc, seed=100+r)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = directed_polish(best_h, n, deadline - 2, num_lags=2)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        n = 250
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
