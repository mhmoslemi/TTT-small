from scipy.optimize import minimize
import time

def compute_c5_fast(h, n):
    dx = 2.0 / n
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(np.conj(H) * G, N)
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(x, N)
    G = np.fft.rfft(1.0 - x, N)
    c = np.fft.irfft(np.conj(H) * G, N) * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _cons_fun(x, n):
    return x.sum() - n / 2.0

def _cons_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
    try:
        res = minimize(smooth_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})

        return h.copy()

def pso_optimize(h_start, n, deadline, rng, n_particles=30, max_iter=100):
    """PSO with projection onto feasible set"""
    target = n / 2.0
    dx = 2.0 / n
    
    # Initialize particles
    pos = np.zeros((n_particles, n))
    vel = np.zeros((n_particles, n))
    pbest_pos = np.zeros((n_particles, n))
    pbest_fit = np.full(n_particles, np.inf)
    
    for i in range(n_particles):
        if i == 0:
            pos[i] = h_start.copy()
        else:
            pos[i] = h_start + rng.normal(0, 0.1, n)
            pos[i] = project_feasible(pos[i], n)
        vel[i] = rng.uniform(-0.05, 0.05, n)
        pbest_pos[i] = pos[i].copy()
        pbest_fit[i] = compute_c5_fast(pos[i], n)
    
    gbest = pbest_pos[np.argmin(pbest_fit)].copy()
    gbest_fit = pbest_fit.min()
    
    w = 0.9
    c1 = 1.8
    c2 = 2.0
    v_max = 0.3
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        # Decay inertia
        w = 0.9 - 0.4 * (it / max_iter)
        
        for i in range(n_particles):
            r1 = rng.rand(n)
            r2 = rng.rand(n)
            vel[i] = w * vel[i] + c1 * r1 * (pbest_pos[i] - pos[i]) + c2 * r2 * (gbest - pos[i])
            vel[i] = np.clip(vel[i], -v_max, v_max)
            new_pos = pos[i] + vel[i]
            new_pos = project_feasible(new_pos, n)
            
            fit = compute_c5_fast(new_pos, n)
            if fit < pbest_fit[i]:
                pbest_fit[i] = fit
                pbest_pos[i] = new_pos.copy()
                if fit < gbest_fit:
                    gbest_fit = fit
                    gbest = new_pos.copy()
            pos[i] = new_pos
    
    return gbest, gbest_fit

def pair_search_fast(h, n, deadline, deltas, radius=None):
    """Pair search with FFT-based evaluation"""
    best_c = compute_c5_fast(h, n)
    if radius is None:
        radius = n  # full search
    
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    # Try h[i] += d, h[j] -= d
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5_fast(h, n)

                        else:

                        h[j] += d

                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

    best_c = compute_c5_fast(h, n)
    best_h = h.copy()

    N = 1
    while N < 2 * n - 1:

        corr = np.fft.irfft(np.conj(H) * G, N)
        c_vals = corr * dx
        max_idx = np.argmax(c_vals)

        # c(k) = dx * sum_i h[i+k] * (1-h[i]) for lag k = max_idx - (n-1)

        # np.correlate(h, 1-h, mode='full')[max_idx]

        # c(lag) = dx * sum_i h[i+lag] * (1-h[i]) for lag >= 0
        #         = dx * sum_i h[i] * (1-h[i+lag]) for lag >= 0

        #               + dx * (1-h[i+lag]) * [from first term if i-lag valid]

        # For lag >= 0: i ranges from 0 to n-1-lag
        #               grad[i] (for i in [0, n-1-lag]) = dx * (1 - h[i+lag])

    c = np.fft.irfft(np.conj(H) * G, N) * dx
    m = c.max()
    sh = alpha * (c - m)
    mx = sh.max()
    return float(m + (mx + np.log(np.sum(np.exp(sh - mx)))) / alpha)

def _cons_fun(x, n):
    return float(x.sum() - n / 2.0)

def _cons_jac(x, n):
    return np.ones(n)

def slsqp_opt(h, n, alpha, maxiter=2000):

    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]

def pso_run(h_start, n, deadline, rng, n_par=30, max_it=80):

    vel = np.zeros((n_par, n))
    pb_pos = np.zeros((n_par, n))

    for i in range(n_par):
        if i == 0:
            pos[i] = h_start.copy()
        elif i == 1:

        else:

        pb_fit[i] = compute_c5_fast(pos[i], n)

    gb_fit = pb_fit.min()
    for it in range(max_it):
        if time.time() > deadline:

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5_fast(h, n)
    if radius is None:
        radius = n
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h, best_c
                lo, hi = max(0, i - radius), min(n, i + radius + 1)
                for j in range(lo, hi):
                    if i == j:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d; h[j] -= d
                        c = compute_c5_fast(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] -= d; h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d; h[j] += d
                        c = compute_c5_fast(h, n)
                        if c < best_c - 1e-15:
                            best_c = c; improved = True
                        else:
                            h[i] += d; h[j] -= d
    return h, best_c

def project(h, n):
    tgt = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15: break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15: break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def _so(x, n, a):
    dx = 2.0 / n
    N = _np2(2 * n - 1)
    c = np.fft.irfft(np.conj(np.fft.rfft(x, N)) * np.fft.rfft(1.0 - x, N), N) * dx
    m = c.max()
    s = a * (c - m)
    mx = s.max()
    return float(m + (mx + np.log(np.sum(np.exp(s - mx)))) / a)

    return float(x.sum() - n / 2.0)

    return np.ones(n)
