5. Repeat until convergence

Step 3 is the key: minimize g_{k}(h) = sum_i h[i]*(1-h[i+k])*dx subject to sum(h)=n/2, 0<=h<=1.

This is: dx * (sum_i h[i] - sum_i h[i]*h[i+k])
= dx * (n/2 - sum_i h[i]*h[i+k])  [since sum(h) = n/2 is a constraint]
= constant - dx * sum_i h[i]*h[i+k]

So we want to MAXIMIZE sum_i h[i]*h[i+k] subject to sum(h)=n/2, 0<=h<=1.

This is a maximization of a quadratic form! The quadratic form is:
sum_i h[i]*h[i+k] = h^T A h / 2 (approximately, where A is a circulant-like matrix with 1s at offset k)

Wait, let me be more careful:
sum_i h[i]*h[i+k] for i=0 to n-1-k (for k>0)
= h[0]*h[k] + h[1]*h[k+1] + ... + h[n-1-k]*h[n-1]

This is a quadratic form h^T B h where B has B[i,j] = 1/2 if j=i+k or i=j+k.

Maximizing h^T B h subject to 0<=h<=1, sum(h)=n/2.

If B is PSD, this is a convex maximization (non-convex problem). If B is NSD, it's a concave maximization (convex problem). 

For our case, B is a banded matrix with 1s on the k-th diagonal. Its eigenvalues are 2*cos(2πjk/n) for a circulant matrix. For k=1, the eigenvalues are 2*cos(2πj/n) which range from 2 to -2. So B is INDEFINITE. The problem is non-convex.

However, we can still use heuristics:
- Start from the current h
- Use gradient descent on the quadratic (which is easy to compute)
- Project back to feasible set

This is essentially the "active constraint" approach I described earlier!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    if sm > 500:
        return m + (sm + np.log(np.sum(np.exp(s - sm)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def _cf(x, n):
    return x.sum() - n / 2.0

def _cj(x, n):
    return np.ones(n)

def slsqp(h, n, alpha, deadline, maxiter=3000):
    if time.time() > deadline - 2:
        return h, compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cf, 'jac': _cj, 'args': (n,)}]
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new, compute_c5(h_new, n)
    except Exception:
        return h, compute_c5(h, n)

def active_grad_step(h, n, n_steps, deadline, lr0=0.8, decay=0.9995):
    """Projected subgradient on active constraint with momentum."""
    h_curr = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    lr = lr0
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h_curr, 1 - h_curr, mode='full') * dx
        c_max = corr.max()
        if c_max < best_c:
            best_c = c_max
            best_h = h_curr.copy()
        # Get top-3 active lags for stability
        top_k = np.argpartition(corr, -3)[-3:]
        t_lags = top_k - (n - 1)
        grad = np.zeros(n)
        for t in t_lags:
            tip = np.arange(max(0, t), min(n, n + t))
            if len(tip) > 0:
                grad[tip] += (1 - h_curr[tip + t])
            tim = np.arange(max(0, -t), min(n, n - t))
            if len(tim) > 0:
                grad[tim] -= h_curr[tim - t]
        grad_c = grad - np.mean(grad)
        v = 0.7 * v + 0.3 * grad_c
        h_new = h_curr - lr * v
        h_new = project_feasible(h_new, n)
        h_curr = h_new
        lr *= decay
    return best_h, best_c

def pair_search(h, n, deadline, deltas, radius=None):
    best_c = compute_c5(h, n)
    h = h.copy()
    if radius is None:
        radius = n
    for d in deltas:
        improved = True
        while improved and time.time() < deadline:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    break
                lo = max(0, i - radius)
                hi = min(n, i + radius + 1)
                for j in range(lo, hi):
                    if j == i:
                        continue
                    if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                        h[i] += d
                        h[j] -= d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] -= d
                            h[j] += d
                    if h[i] - d >= 0.0 and h[j] + d <= 1.0:
                        h[i] -= d
                        h[j] += d
                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            improved = True
                        else:
                            h[i] += d
                            h[j] -= d
    return h, best_c

def optimize_at_n(h_start, n, deadline, rng):
    h = project_feasible(h_start, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    # SLSQP with increasing alpha
    alphas = [5000, 25000, 100000, 500000, 2000000]
    for a in alphas:
        if time.time() > deadline - 20:
            break
        h_new, c_new = slsqp(h, n, a, deadline - 20, maxiter=2500)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        h = best_h
    # Active gradient steps
    if time.time() < deadline - 10:
        h_g, c_g = active_grad_step(best_h, n, 500, deadline - 10, lr0=0.5, decay=0.999)
        if c_g < best_c:
            best_c = c_g
            best_h = h_g
    # Pair search
    if time.time() < deadline - 5:
        if n <= 200:
            best_h, best_c = pair_search(best_h, n, deadline - 5,
                                         [0.01, 0.005, 0.002, 0.001, 0.0005])
        else:
            best_h, best_c = pair_search(best_h, n, deadline - 5,
                                         [0.01, 0.005, 0.002, 0.001], radius=60)
    # Final SLSQP
    if time.time() < deadline - 3:
        h2, c2 = slsqp(best_h, n, 5000000, deadline - 2, maxiter=2000)
        if c2 < best_c:
            best_c = c2
            best_h = h2
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)
