   ```python
           pop = [poisson_init(h0, 0.05, n) for _ in range(30)]
           pop = [repair_vector(p, n) for p in pop]
           pop += [np.random.rand(n) for _ in range(70)]
           pop = [repair_vector(p, n) for p in pop]
       else:
           pop = [np.random.rand(n) for _ in range(100)]
           pop = [repair_vector(p, n) for p in pop]
       return pop

   def genetic_algorithm():
       n = 250  # Fixed resolution to match parent context
       pop_size = 100
       h0 = np.full(n, 0.5) if "initial_h_values" not in kwargs else np.array(initial_h_values)
       
       pop = initialize_population(n, h0, pop_size)
       best_h = np.array(pop[0])
       best_c = compute_c5(best_h)
       
       # Algorithm parameters
       crossover_rate = 0.9
       mutation_rate = 0.1
       generations = 200
       
       for gen in range(generations):
           # Evaluate fitness
           fitness = [compute_c5(ind) for ind in pop]
            
           # Find best individual
           if any(f < best_c for f in fitness):
               best_idx = np.argmin(fitness)
               best_h = pop[best_idx]
               best_c = fitness[best_idx]
               
           # Elitism replacement
           new_pop = []
           sorted_idx = np.argsort(fitness)
           elitism = pop_size // 10  # Top 10%
           new_pop.extend([pop[i] for i in sorted_idx[:elitism]])
           
           # Crossover
           for i in range(elitism, pop_size):
               j = i + np.random.choice(pop_size - i)
               parent1 = pop[sorted_idx[i]]
               parent2 = pop[sorted_idx[j]]
               
               # Simulated binary crossover
               beta = 1.0 + (np.random.random() * (np.abs(parent1 - parent2) + 0.01))
               if np.random.random() < crossover_rate:
                   child = 0.5 * (parent1 + parent2) * abs(1 - beta) + 0.5 * (parent1 - parent2) * beta
                   child = repair_vector(child, n)
               else:
                   # Mutation: create new child with small random alterations
                   child = (1 - mutation_rate) * parent1 + mutation_rate * parent2
                   child = repair_vector(child, n)
                # Probability of mutation outside crossover
                if np.random.random() < mutation_rate * 0.5:
                    child = repair_vector(parent1 + 0.1 * np.random.randn(n), n)
                
               new_pop.append(child)
           pop = new_pop
           
       return best_h, best_c

   def local_search(h):
       # Nelder-Mead optimization
       from scipy.optimize import minimize
       from scipy.optimize import Bounds
       options = {'maxiter': 300, 'disp': False,
                  'bounds': Bounds(0.0, 1.0)}
       
       # Custom gradient (if we wanted, but Nelder-Mead doesn't need)
       res = minimize(lambda x: compute_c5(x), h, method='Nelder-Mead',
                      options=options, bounds=options['bounds'])
       return res.x

   def simulated_annealing(h, temperature=0.3, steps=500):
       # Improvement: avoid local minima
       n = len(h)
       new_h = h.copy()
       best = h.copy()
       
       for step in range(steps):
           candidate = new_h.copy()
           # Random mutation
           i = np.random.randint(n)
           candidate[i] += np.random.uniform(-0.2, 0.2)
           candidate = repair_vector(candidate, n)
           new_c5 = compute_c5(candidate)
           old_c5 = compute_c5(best)
           if new_c5 < old_c5:
               new_h = candidate
               best = candidate
               current_temp = temperature * (0.99 ** step)
           elif np.random.random() < np.exp(-(new_c5 - old_c5) / current_temp):
               new_h = candidate
       
       return best

   def run(seed=42, budget_s=1100, **kwargs):
       import time
       t0 = time.time()
       deadline = t0 + budget_s - 5
       
       # Set random seed
       np.random.seed(seed)
       
       # Initialize population
       h0 = np.array(kwargs.get('initial_h_values', np.random.rand(250)))
       n = 250  # Fixed resolution for consistency

       # Phases
       phases = [
           {'method': 'ga', 'time': 0.4, 'args': {}},
           {'method': 'local_search', 'time': 0.3, 'args': {}},
           {'method': 'sa', 'time': 0.3, 'args': {}}
       ]

       best_h = h0
       best_c = compute_c5(best_h)
       host_best = kwargs.get('initial_h_values')

       def cross_correlation(h):
           # Return all lags with appropriate scaling
           dx = 2.0 / n
           c = np.correlate(h, 1-h, mode='full')
           return float(np.max(c) * dx)

       for phase in phases:
           if time.time() >= deadline:
               break
           time_per_phase = deadline * phase['time'] / 1000
           best_cost = compute_c5(best_h)
           
           if phase['method'] == 'ga':
               # Perform GA for this phase
               new_best_h, new_c = genetic_algorithm()
               if new_c < best_cost:
                   best_h = new_best_h
                   best_c = new_c
                   host_best = new_best_h.tolist()  # Update host candidate if improved
           elif phase['method'] == 'local_search':
               for _ in range(10):
                   new_best = local_search(best_h)
                   if compute_c5(new_best) < best_c:
                       best_h = new_best
                       best_c = compute_c5(best_h)
           elif phase['method'] == 'sa':
               for _ in range(20):
                   new_best = simulated_annealing(best_h)
                   if compute_c5(new_best) < best_c:
                       best_h = new_best
                       best_c = compute_c5(best_h)
                       host_best = new_best.tolist()
           else:
               # Fallback to modular repair approach
               h_test = repair_vector(best_h, n)
               if compute_c5(h_test) < best_c:
                   best_h, best_c = h_test, compute_c5(h_test)
                      

       # Project to valid solution and enforce constraints
       final_h = project_feasible(best_h, n)
       final_c = compute_c5(final_h)  # Recompute to be safe
       return final_h, final_c, n
   ```
