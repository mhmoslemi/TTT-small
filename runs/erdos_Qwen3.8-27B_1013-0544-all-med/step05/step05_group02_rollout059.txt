Let me also think about the step size more carefully. The standard FW step size t_k = 2/(k+2) gives an O(1/k) convergence rate. For better convergence, we can use:
- Exact line search: t_k = argmin_{t∈[0,1]} f((1-t)h_k + t*v_k)
- Strong Wolfe conditions

For our problem, since the objective is the max of quadratics (non-convex), the line search might not be unimodal. But in practice, it often works well.

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lse_subgradient(h, n, alpha):
    dx = 2.0 / n
    c = get_corr(h, n)
    shifted = alpha * (c - c.max())
    max_s = shifted.max()
    w = np.exp(shifted - max_s)
    w /= w.sum()
    g = np.zeros(n)
    for k_idx in range(len(c)):
        wk = w[k_idx]
        if wk < 1e-12:
            continue
        s = k_idx - (n - 1)
        # First term: (1-h[j+s]) for valid j
        if s >= 0:
            lo, hi = 0, n - 1 - s
        else:
            lo, hi = -s, n - 1
        if lo <= hi:
            g[lo:hi+1] += wk * (1.0 - h[lo+s:hi+s+1]) * dx
        # Second term: -h[j-s] for valid j
        if s >= 0:
            lo2, hi2 = s, n - 1
        else:
            lo2, hi2 = 0, n - 1 + s
        if lo2 <= hi2:
            g[lo2:hi2+1] -= wk * h[lo2-s:hi2-s+1] * dx
    return g

def fw_line_search(h, v, n, max_c):
    """Find best t in [0,1] for h_new = (1-t)*h + t*v."""
    best_t = 0.0
    best_f = max_c
    # Coarse search
    for t in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9]:
        h_trial = (1.0 - t) * h + t * v
        h_trial = project_feasible(h_trial, n)
        f_trial = compute_c5(h_trial, n)
        if f_trial < best_f:
            best_f = f_trial
            best_t = t
    # Fine search around best_t
    if best_t > 0:
        for dt in [0.005, -0.005, 0.01, -0.01, 0.02, -0.02]:
            t2 = best_t + dt
            if 0 < t2 < 1:
                h_trial = (1.0 - t2) * h + t2 * v
                h_trial = project_feasible(h_trial, n)
                f_trial = compute_c5(h_trial, n)
                if f_trial < best_f:
                    best_f = f_trial
                    best_t = t2
    return best_t, best_f

def fw_phase(h, n, alpha_range, deadline, max_iter=500):
    """Frank-Wolfe phase with LSE smoothing."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for alpha in alpha_range:
        for it in range(max_iter):
            if time.time() > deadline:
                break
            g = lse_subgradient(h, n, alpha)
            # FW vertex: top n/2 indices get 1
            top_idx = np.argsort(g)[-n//2:]
            v = np.zeros(n)
            v[top_idx] = 1.0
            t, f_new = fw_line_search(h, v, n, best_c)
            if t > 0:
                h = (1.0 - t) * h + t * v
                h = project_feasible(h, n)
                if f_new < best_c:
                    best_c = f_new
                    best_h = h.copy()
            else:
                break  # No improvement possible
    return best_h, best_c

def subgrad_phase(h, n, step_schedule, deadline, max_iter=2000):
    """Subgradient descent phase."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        alpha = 1000.0  # fixed alpha for subgradient
        g = lse_subgradient(h, n, alpha)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-10:
            break
        eta = step_schedule(it)
        h_new = h - eta * g / g_norm
        h_new = project_feasible(h_new, n)
        f_new = compute_c5(h_new, n)
        if f_new < best_c:
            best_c = f_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Reduce step size
            eta *= 0.5
            h_new = h - eta * g / g_norm
            h_new = project_feasible(h_new, n)
            f_new = compute_c5(h_new, n)
            if f_new < compute_c5(h, n):
                h = h_new
    return best_h, best_c

def coordinate_polish(h, n, deadline):
    """Greedy coordinate descent polish."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    c = get_corr(h, n)
    for _ in range(100000):
        if time.time() > deadline:
            break
        k_star = int(np.argmax(c))
        s = k_star - (n - 1)
        # Compute gradient at worst lag
        g = np.zeros(n)
        if s >= 0:
            g[0:n-s] += (1.0 - h[s:n])
            g[s:n] -= h[0:n-s]
        else:
            g[-s:n] += (1.0 - h[0:n+s])
            g[0:n+s] -= h[-s:n]
        # Find best pair to swap
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst or g[i_src] - g[i_dst] < 1e-10:
            break
        # Try moving mass from src to dst
        best_move = 0.0
        best_f = best_c
        for frac in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            d = frac * h[i_src]
            if d <= 0 or d > 1 - h[i_dst]:
                continue
            h[i_src] -= d
            h[i_dst] += d
            f_new = compute_c5(h, n)
            if f_new < best_f:
                best_f = f_new
                best_move = d
            h[i_src] += d
            h[i_dst] -= d
        if best_move > 0:
            h[i_src] -= best_move
            h[i_dst] += best_move
            best_c = best_f
            best_h = h.copy()
            c = get_corr(h, n)
        else:
            # Try smaller moves
            for frac in [0.01, 0.02, 0.05]:
                d = frac * h[i_src]
                if d <= 0 or d > 1 - h[i_dst]:
                    continue
                h[i_src] -= d
                h[i_dst] += d
                f_new = compute_c5(h, n)
                if f_new < best_c:
                    best_c = f_new
                    best_h = h.copy()
                    c = get_corr(h, n)
                    break
                h[i_src] += d
                h[i_dst] -= d
            else:
                break
    return best_h, best_c

def project_onto_polytope(z, n):
    """Project z onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    # Bisection on lambda
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(z - mid, 0.0, 1.0)
        s = h_trial.sum()
        if abs(s - target) < 1e-12:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
