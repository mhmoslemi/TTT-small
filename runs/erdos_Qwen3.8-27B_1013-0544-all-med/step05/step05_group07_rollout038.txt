import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1][:target]
    h_bin = np.zeros(n)
    h_bin[idx] = 1.0
    return h_bin

def smooth_obj(h, n, alpha, pen):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
    if pen > 0:
        val += pen * (h.sum() - n / 2.0) ** 2
    return val

def smooth_grad(h, n, alpha, pen):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    sm = sh.max()
    w = np.exp(sh - sm)
    wsum = w.sum()
    g = np.zeros(n)
    for offset in range(2 * n - 1):
        if w[offset] < 1e-12 * wsum:
            continue
        wo = w[offset] / wsum
        k = offset - (n - 1)
        if k >= 0:
            g[:n-k] += wo * (1.0 - h[k:]) * dx
            g[k:] += wo * (-h[:n-k]) * dx
        else:
            k2 = -k
            g[k2:] += wo * (1.0 - h[:n-k2]) * dx
            g[:n-k2] += wo * (-h[k2:]) * dx
    if pen > 0:
        g += 2.0 * pen * (h.sum() - n / 2.0)
    return g

def lbfgs_smoothed(h, n, alpha, pen, maxiter, deadline):
    def obj(x):
        return smooth_obj(x, n, alpha, pen)
    def grad(x):
        return smooth_grad(x, n, alpha, pen)
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=grad,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        return project_feasible(res.x, n)
    except:
        return h

    best_c = compute_c5(h_bin, n)
    best_h = h_bin.copy()
    ones = list(np.where(h_bin == 1)[0])
    zeros = list(np.where(h_bin == 0)[0])
    temp = 0.002
    for it in range(500000):
        if time.time() > deadline:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h_bin[i], h_bin[j] = 1.0, 0.0
        c_new = compute_c5(h_bin, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_bin.copy()
            ones.remove(j); ones.append(i)
            zeros.remove(i); zeros.append(j)
            temp = max(temp * 0.999, 1e-8)
        else:
            if rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-9)):
                ones.remove(j); ones.append(i)
                zeros.remove(i); zeros.append(j)
            else:
                h_bin[i], h_bin[j] = 0.0, 1.0
        temp *= 0.9998
    return best_h, best_c

def coordinate_refine(h, n, deadline):

    best_c = compute_c5(h, n)

            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    idx = np.argsort(h)[::-1][:n // 2]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def _smooth_obj(x, n, alpha, pen):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    val = m + np.log(np.exp(sh).sum()) / alpha
    if pen > 0:
        val += pen * (x.sum() - n / 2.0) ** 2
    return val

def _smooth_grad(x, n, alpha, pen):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    sh = alpha * (c - m)
    w = np.exp(sh)
    wsum = w.sum()
    g = np.zeros(n)
    for off in range(2 * n - 1):
        if w[off] < 1e-14 * wsum:
            continue
        wo = w[off] / wsum * dx
        k = off - (n - 1)
        if k >= 0:
            g[k:] += wo * (1.0 - x[:n-k])
            g[:n-k] += wo * (-x[k:])
        else:
            k2 = -k
            g[:n-k2] += wo * (1.0 - x[k2:])
            g[k2:] += wo * (-x[:n-k2])
    if pen > 0:
        g += 2.0 * pen * (x.sum() - n / 2.0)
    return g

def lbfgs_opt(h, n, alpha, pen, maxiter):
    try:
        res = minimize(_smooth_obj, h, method='L-BFGS-B', jac=_smooth_grad,
                       args=(n, alpha, pen),
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except:
        return h

def binary_sa(h, n, deadline, rng):
    hb = to_binary(h, n)
    best_c = compute_c5(hb, n)
    best_hb = hb.copy()
    ones = list(np.where(hb == 1)[0])
    zeros = list(np.where(hb == 0)[0])
    temp = 0.003
    stall = 0
    while time.time() < deadline:
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        hb[i], hb[j] = 1.0, 0.0
        c_new = compute_c5(hb, n)
        if c_new < best_c:
            best_c = c_new
            best_hb = hb.copy()
            ones.remove(j); ones.append(i)
            zeros.remove(i); zeros.append(j)
            temp = max(temp * 0.999, 1e-8)
            stall = 0
        else:
            if rng.random() < np.exp(-(c_new - best_c) / max(temp, 1e-9)):
                ones.remove(j); ones.append(i)
                zeros.remove(i); zeros.append(j)
            else:
                hb[i], hb[j] = 0.0, 1.0
            stall += 1
        temp *= 0.9997
        if stall > 30000:
            for _ in range(10):
                i2 = zeros[rng.randint(len(zeros))]
                j2 = ones[rng.randint(len(ones))]
                hb[i2], hb[j2] = 1.0, 0.0
                ones.remove(j2); ones.append(i2)
                zeros.remove(i2); zeros.append(j2)
            temp = 0.003
            stall = 0
    return best_hb, best_c

def coord_refine(h, n, deadline):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for scale in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]:
        if time.time() > deadline:
            break
        improved_any = True
        while improved_any and time.time() < deadline:
            improved_any = False
            for i in range(n):
                if time.time() > deadline:
                    break
                cur = best_h[i]
                best_t = cur
                for s in (1, -1):
                    t = cur + s * scale
                    if 0 <= t <= 1 and t != cur:
                        best_h[i] = t
                        c = compute_c5(best_h, n)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_t = t
                            improved_any = True
                best_h[i] = best_t
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except:
        n = 250
        h0 = np.full(n, 0.5)
    best_h = project_feasible(h0.copy(), n)
    best_c = compute_c5(best_h, n)
    
    # Phase 1: Binary SA (15%)
    t1 = t0 + 0.15 * (budget_s - 5)
    hb, cb = binary_sa(best_h.copy(), n, t1, rng)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
    
    # Phase 2: Smoothed L-BFGS-B (45%)
    t2 = t0 + 0.60 * (budget_s - 5)
    alphas = [100, 500, 2000, 10000, 50000, 200000, 1000000, 5000000]
    for alpha in alphas:
        rem = t2 - time.time()
        if rem < 20:
            break
        mi = max(100, int(rem * 50))
        h_new = lbfgs_opt(best_h.copy(), n, alpha, 1e5, mi)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c, best_h = c_new, h_new.copy()
    
    # Phase 3: Coordinate refinement (20%)
    t3 = t0 + 0.80 * (budget_s - 5)
    h_new, c_new = coord_refine(best_h.copy(), n, t3)
    if c_new < best_c:
        best_c, best_h = c_new, h_new.copy()
    
    # Phase 4: Binary SA again (10%)
    t4 = t0 + 0.90 * (budget_s - 5)
    hb, cb = binary_sa(best_h.copy(), n, t4, rng)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()
