def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')

    idx = np.argsort(h, kind='stable')[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def greedy_random(hb, n, n_trials, deadline, seed=0):
    """Randomized greedy: try random swaps, accept if improves."""
    rng = np.random.RandomState(seed)
    c = compute_c5(hb, n)
    for _ in range(n_trials):

            break

        else:

    p1 = t0 + 0.08 * (budget_s - 3)
    hb = best_h.copy()
    hb, c = greedy_random(hb, n0, p1, seed)
    if c < best_c:
        best_c, best_h = c, hb.copy()

    # Phase 2: SA main (30%)
    p2 = t0 + 0.38 * (budget_s - 3)
    hb = best_h.copy()
    hb, c = sa_binary(hb, n0, p2, seed=seed + 1, T_start=0.005, T_end=0.00003)
    if c < best_c:
        best_c, best_h = c, hb.copy()

    # Phase 3: Greedy refine (7%)
    p3 = t0 + 0.45 * (budget_s - 3)
    hb = best_h.copy()
    hb, c = greedy_random(hb, n0, p3, seed=seed + 10)
    if c < best_c:
        best_c, best_h = c, hb.copy()

    # Phase 4: Iterated local search with kicks (25%)
    p4 = t0 + 0.70 * (budget_s - 3)
    n_kicks = 6
    per_kick = (p4 - time.time()) / n_kick
    for k in range(n_kicks):
        rdl = time.time() + per_kick
        if rdl > p4:
            rdl = p4
        if rdl < time.time() + 5:
            break
        start = best_h.copy()
        # Kick size increases for later kicks
        n_flip = rng.randint(4, 8 + k * 2)
        flip_idx = rng.choice(n0, size=min(n_flip, n0), replace=False)
        start[flip_idx] = 1.0 - start[flip_idx]
        start = fix_binary(start, n0)
        # SA then greedy
        hb, c = sa_binary(start, n0, rdl - 0.3 * per_kick, seed=seed + 100 + k,
                          T_start=0.003, T_end=0.00002)
        hb, c = greedy_random(hb, n0, rdl, seed=seed + 200 + k)
        if c < best_c:
            best_c, best_h = c, hb.copy()

    # Phase 5: Double-swap (12%)
    p5 = t0 + 0.82 * (budget_s - 3)
    hb = best_h.copy()
    hb, c = double_swap(hb, n0, p5, seed=seed + 300)
    if c < best_c:
        best_c, best_h = c, hb.copy()
    # Greedy after double-swap
    p5b = t0 + 0.87 * (budget_s - 3)
    hb, c = greedy_random(hb, n0, p5b, seed=seed + 350)
    if c < best_c:
        best_c, best_h = c, hb.copy()

    # Phase 6: Try other n values (10%)
    p6 = dl - 2
    for n_try in [200, 300, 350]:
        if time.time() > p6 - 25:
            break
        if n_try == n0:
            continue
        x_old = np.linspace(0, 2.0, n0, endpoint=False)
        x_new = np.linspace(0, 2.0, n_try, endpoint=False)
        h_ext = np.concatenate([h0, [h0[0]]])
        h_interp = np.interp(x_new, x_old, h_ext)
        hb_t = to_binary(h_interp, n_try)
        c_t = compute_c5(hb_t, n_try)
        # Quick SA
        hb_t, c_t = sa_binary(hb_t, n_try, min(time.time() + 15, p6 - 10),
                              seed=seed + n_try, T_start=0.004, T_end=0.00003)
        hb_t, c_t = greedy_random(hb_t, n_try, min(time.time() + 10, p6 - 3),
                                  seed=seed + n_try + 50)
        if c_t < best_c:
            best_c, best_h = c_t, hb_t.copy()

    # Final greedy polish
    if time.time() < dl - 1:
        hb, c = greedy_random(best_h.copy(), len(best_h), dl - 0.5, seed=seed + 999)
        if c < best_c:
            best_c, best_h = c, hb

    n_final = len(best_h)
    best_h = fix_binary(best_h, n_final)
    best_c = compute_c5(best_h, n_final)
    return best_h, best_c, n_final
