For the active shift k (the one achieving the max), the subgradient of c_k with respect to h is:

I can compute this in O(n). Then I take a step h ← h - η*g and project to feasibility.

- If the objective doesn't decrease, halve η
- If it decreases, accept the step

def subgradient_refine(h, n, deadline, max_iters=100000):
    h = h.copy()
    best_c = compute_c5(h, n)
    eta = 0.005
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        # Compute correlation and find active shift
        c = np.correlate(h, 1 - h, mode='full') * (2.0/n)
        c_max = c.max()
        # Use the single most active shift for subgradient
        k_idx = int(np.argmax(c))
        lag = k_idx - (n - 1)
        
        # Compute gradient of c_k w.r.t. h
        g = np.zeros(n)
        if lag >= 0:
            # c_k = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d/dh[m]: (1-h[m+lag]) if m in [0, n-1-lag]
            #         -h[m-lag] if m-lag in [0, n-1-lag], i.e., m in [lag, n-1]
            idx1 = np.arange(min(n - lag, n))
            g[idx1] += (1 - h[idx1 + lag])
            idx2 = np.arange(lag, n)
            g[idx2] -= h[idx2 - lag]
        else:
            L = -lag
            # c_{-L} = sum_{i=L}^{n-1} h[i]*(1-h[i-L])
            # d/dh[m]: (1-h[m-L]) if m in [L, n-1]
            #         -h[m+L] if m+L in [L, n-1], i.e., m in [0, n-1-L]
            idx1 = np.arange(L, n)
            g[idx1] += (1 - h[idx1 - L])
            idx2 = np.arange(0, n - L)
            g[idx2] -= h[idx2 + L]
        
        # Try step
        for _ls in range(20):
            h_try = h - eta * g
            h_try = project_feasible(h_try, n)
            c_try = compute_c5(h_try, n)
            if c_try < best_c - 1e-16:
                h = h_try
                best_c = c_try
                eta = min(eta * 1.05, 0.02)
                break
            eta *= 0.7
        if eta < 1e-8:
            break
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def soft_max_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    logsumexp = mx + np.log(np.sum(np.exp(shifted - mx)))
    return float(m + logsumexp / alpha)

def _sum_con(x, n):
    return float(np.sum(x) - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, alpha_list, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    for alpha in alpha_list:
        if time.time() > deadline - 5:
            break
        for attempt in range(2):
            if time.time() > deadline - 5:
                break
            try:
                res = minimize(soft_max_obj, h, args=(n, alpha),
                              method='SLSQP', bounds=bounds, constraints=cons,
                              options={'maxiter': 10000, 'ftol': 1e-18})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    h = h_new
                else:
                    break
            except Exception:
                break
    return h, best_c

def subgradient_refine(h, n, deadline, tol=0.002):

            gi = np.zeros(n)
            if lag >= 0:
                idx1 = np.arange(0, n - lag)
                gi[idx1] += (1.0 - h[idx1 + lag])
                idx2 = np.arange(lag, n)
                gi[idx2] -= h[idx2 - lag]
            else:
                L = -lag
                idx1 = np.arange(L, n)
                gi[idx1] += (1.0 - h[idx1 - L])
                idx2 = np.arange(0, n - L)
                gi[idx2] -= h[idx2 + L]
            g += gi
