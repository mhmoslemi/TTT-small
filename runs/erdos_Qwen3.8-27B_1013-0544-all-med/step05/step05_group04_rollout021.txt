    alloc1 = min(280, (deadline - t0) * 0.3)
    h_opt, c_opt = optimize_at_n(best_h, n_init, t0 + alloc1, rng)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt
    # Phase 2: try other n values via interpolation
    n_candidates = [120, 150, 200, 250, 300, 400]
    for n_try in n_candidates:
        if n_try == n_init:
            continue
        if time.time() > deadline - 120:
            break
        h_interp = interpolate_h(best_h, len(best_h), n_try)
        alloc = min(130, (deadline - time.time()) * 0.18)
        if alloc < 35:
            break
        h_opt, c_opt = optimize_at_n(h_interp, n_try, time.time() + alloc, rng)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
    # Phase 3: multi-restart
    n = len(best_h)
    if time.time() < deadline - 80:
        remaining = deadline - time.time() - 40
        n_restarts = max(2, int(remaining / 50))
        for r in range(n_restarts):
            if time.time() > deadline - 35:
                break
            scale = 0.02 + 0.08 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 25) * 0.4)
            if alloc < 15:
                break
            h_opt, c_opt = optimize_at_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    # Phase 4: final SLSQP with very high alpha
    n = len(best_h)
    if time.time() < deadline - 10:
        for alpha in [500000, 1000000, 2000000]:
            if time.time() > deadline - 5:
                break
            ht = slsqp_refine(best_h, n, alpha, maxiter=2500)
            c = compute_c5(ht, n)
            if c < best_c:
                best_c = c
                best_h = ht
    # Phase 5: final pair search
    if n <= 180 and time.time() < deadline - 3:
        best_h, best_c = full_pair_search(best_h, n, deadline - 1,
                                          [0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001])
    elif time.time() < deadline - 3:
        best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                           [0.005, 0.002, 0.001, 0.0005, 0.0002], radius=35)
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h, len(best_h))
    return best_h, best_c, len(best_h)
