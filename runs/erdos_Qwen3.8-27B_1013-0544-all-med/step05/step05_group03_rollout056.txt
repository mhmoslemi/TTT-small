from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14: break
            h += diff / s * room
        else:
            s = h.sum()
            if s < 1e-14: break
            h += diff / s * h
    return np.clip(h, 0.0, 1.0)

def paired_cd(h, n, deadline, max_passes=50):
    """Paired coordinate descent for the min-max problem."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    out = np.correlate(h, 1 - h, mode='full')
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline - 10:
                break
            for j in range(n):
                if i == j:
                    continue
                # Determine range of t
                t_min = max(-h[i], h[j] - 1.0)
                t_max = min(1.0 - h[i], h[j])
                if t_max - t_min < 1e-14:
                    continue
                
                # Compute slopes a[k] = d(out[k])/d(h[i]), b[k] = d(out[k])/d(h[j])
                # a[k] for lag = k - (n-1):
                lag = np.arange(2*n-1) - (n-1)
                # a[k] = (1-h[i-lag]) * [0<=i-lag<n] - h[i+lag] * [0<=i+lag<n]
                idx1 = i - lag  # position in (1-h[.]) term
                idx2 = i + lag  # position in h[.]) term
                a = np.zeros(2*n-1)
                mask1 = (idx1 >= 0) & (idx1 < n)
                mask2 = (idx2 >= 0) & (idx2 < n)
                a[mask1] = 1.0 - h[idx1[mask1]]
                a[mask2] -= h[idx2[mask2]]
                
                idx1_j = j - lag
                idx2_j = j + lag
                mask1_j = (idx1_j >= 0) & (idx1_j < n)
                mask2_j = (idx2_j >= 0) & (idx2_j < n)
                b = np.zeros(2*n-1)
                b[mask1_j] = 1.0 - h[idx1_j[mask1_j]]
                b[mask2_j] -= h[idx2_j[mask2_j]]
                
                slopes = a - b
                # f(t) = max_k (out[k] + t*slopes[k]) * dx
                # Minimize over [t_min, t_max]
                
                # Quick check: is current t=0 already optimal?
                # Subgradient at 0: check if 0 is in convex hull of active slopes
                active = np.abs(out + 0*slopes - out.max()) < 1e-12
                if active.any():
                    active_slopes = slopes[active]
                    if active_slopes.min() > 1e-10 or active_slopes.max() < -1e-10:
                        # Not optimal at t=0, need to move
                        pass
                    else:
                        continue  # Already optimal at t=0
                
                # Use minimize_scalar
                def f_1d(t):
                    return np.max(out + t * slopes)
                
                res = minimize_scalar(f_1d, bounds=(t_min, t_max), method='bounded',
                                     options={'xatol': 1e-10})
                if res.success and res.x != 0:
                    t_opt = res.x
                    h[i] += t_opt
                    h[j] -= t_opt
                    h = np.clip(h, 0, 1)
                    # Re-project to maintain sum
                    h = project_feasible(h, n)
                    out = np.correlate(h, 1 - h, mode='full')
                    c = np.max(out) * dx
                    if c < best_c - 1e-14:
                        best_c = c
                        best_h = h.copy()
                        improved = True
        
        if not improved:
            break
    
    return best_h, best_c

Actually, for the bisection, I need to know the sign of the subgradient at the current t for each j. The subgradient of max_k g_k(t) at t is the set {g_k'(t) : g_k(t) = max}. For the bisection, I need to know if this set contains 0.

For each j at a given t:
- g_j[k] = out[k] + t * slopes_all[k, j]
- max_val_j = max_k g_j[k]
- active_k for j: {k : g_j[k] ≈ max_val_j}
- subgradient for j: {slopes_all[k, j] : k active}
- If min(subgradient) > 0: t is too large, decrease
- If max(subgradient) < 0: t is too small, increase
- Otherwise: t is optimal for j

I can vectorize this:
```python
def evaluate_and_classify(t, out, slopes_all):
    """For all j, evaluate f_j(t) and determine subgradient sign."""
    vals = out[:, None] + t * slopes_all  # (2n-1, n)
    max_vals = vals.max(axis=0)  # (n,)
    # For bisection, we need: is the min active slope > 0? Is the max active slope < 0?
    # Active: vals[k, j] == max_vals[j] (within tolerance)
    active = np.abs(vals - max_vals[None, :]) < 1e-8  # (2n-1, n)
    active_sl = slopes_all * active
    min_active = active_sl.min(axis=0)  # (n,)
    max_active = active_sl.max(axis=0)  # (n,)
    # Classify:
    # increase: max_active < 0 (need larger t)
    # decrease: min_active > 0 (need smaller t)
    # optimal: min_active <= 0 <= max_active
    return max_vals, min_active, max_active
```
