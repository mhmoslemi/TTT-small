```python
    dx = 2.0 / n
    g = compute_g(h, n)
    F = compute_F(h, n)
    dm = np.arange(2 * n - 1) - (n - 1)
    
    # Find top K lags
    top_idx = np.argsort(g)[::-1][:K]
    
    best_val = g.max()
    best_i, best_j, best_delta = -1, -1, 0.0
    
    for idx in range(n):
        for jdx in range(idx+1, n):
            s = F[idx] - F[jdx]
            # Evaluate max(g + delta*s) at optimal delta
            # The optimal delta for minimizing max_m (g[m] + delta*s[m])
            # is where the two largest (after adding delta*s) lags are equal
            # For simplicity, evaluate at a few candidate deltas
            d_lo = max(-h[idx], h[jdx] - 1.0)
            d_hi = min(1.0 - h[idx], h[jdx])
            if d_hi <= d_lo:
                continue
            
            # Candidate deltas: 0, d_lo, d_hi, and crossings
            candidates = [0.0, d_lo, d_hi]
            for m1 in range(K):
                for m2 in range(m1+1, K):
                    ds = s[top_idx[m2]] - s[top_idx[m1]]
                    if abs(ds) > 1e-15:
                        dc = (g[top_idx[m1]] - g[top_idx[m2]]) / ds
                        if d_lo < dc < d_hi:
                            candidates.append(dc)

                d = np.clip(d, d_lo, d_hi)
                val = np.max(g + d * s)
                if val < best_val - 1e-14:

                    best_i, best_j, best_delta = idx, jdx, d

        h[best_j] -= best_delta
        h = project_feasible(h, n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_F(h, n):
    dm = np.arange(2 * n - 1) - (n - 1)
    F = np.zeros((n, 2 * n - 1))
    for m_idx in range(2 * n - 1):
        d = dm[m_idx]
        i_lo = max(0, d)
        i_hi = min(n, n + d)
        if i_lo < i_hi:
            F[i_lo:i_hi, m_idx] += (1.0 - h[i_lo - d:i_hi - d])
        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo < j_hi:
            F[j_lo:j_hi, m_idx] -= h[j_lo + d:j_hi + d]
    return F
```

- For any (k, m) where the formula for dc[m]/dh[k] involves h[i], F[k,m] changes.

From the formula:
dc[m]/dh[k] = (1-h[k-d_m])_valid - h[k+d_m]_valid

- F[k,m] changes by -δ if k+d_m = i (i.e., k = i - d_m), for valid k

- F[i-d_m, m] -= δ (if i-d_m is valid)

```python
            if s < 1e-14:

    F = np.zeros((n, 2 * n - 1))
```

```python
def pair_descent_fast(h, n, deadline, K=30, max_sweeps=200):
    """Fast pair descent using incremental F updates."""

    g = compute_g(h, n)
    F, dm = compute_F(h, n)
    best_c = float(g.max())
    best_h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = 0
        n_m = 2 * n - 1
        top_idx = np.argsort(g)[::-1][:K]
        current_max = float(g.max())
        
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-14:
                    continue
                # Find optimal delta
                candidates = [d_lo, d_hi]
                for a in range(K):
                    for b in range(a + 1, K):
                        m1, m2 = top_idx[a], top_idx[b]
                        ds = s[m2] - s[m1]
                        if abs(ds) > 1e-15:
                            dc = (g[m1] - g[m2]) / ds
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                best_delta = 0.0
                best_val = current_max
                for d in candidates:
                    val = float(np.max(g + d * s))
                    if val < best_val - 1e-14:
                        best_val = val
                        best_delta = d
                if best_delta != 0.0:

                    top_idx = np.argsort(g)[::-1][:K]

            best_c = c
            best_h = h.copy()
        if improved == 0:
            break
    
    return best_h, best_c
```
