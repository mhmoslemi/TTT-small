def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)

def project_feasible(h, n):

    w = np.exp(sh)
    wsum = w.sum()
    if wsum < 1e-30:
        return np.zeros(n)
    w /= wsum
    grad = np.zeros(n)
    corr_full = np.correlate(h, 1 - h, mode='full')
