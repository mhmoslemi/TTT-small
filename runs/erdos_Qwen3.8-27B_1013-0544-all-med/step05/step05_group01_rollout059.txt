For h[i] -= δ, h[j] += δ: δ*(grad[j] - grad[i])

So the best pair is the one with max |grad[i] - grad[j]|, and we move in the direction that reduces the objective.

One more consideration: the projection_feasible function. The current implementation does a "water-filling" to adjust the sum. This is correct but might not be optimal in terms of minimizing the objective. An alternative is to do the projection more carefully:

To project h onto {x: 0<=x_i<=1, sum(x)=target}:
1. Start with x = h
2. Clip to [0,1]
3. If sum > target: reduce the free variables proportionally (or use a threshold method)
4. If sum < target: increase the free variables proportionally

The "threshold" method (like the dual of the L1 projection) is:
- Find a threshold τ such that sum(clip(h_i + τ, 0, 1)) = target
- This can be found by bisection on τ

This is more principled and gives the closest point in L2 norm. Let me use this:

```python
def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    # Find τ such that sum(clip(h + τ, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    tau = (lo + hi) / 2
    return np.clip(h + tau, 0.0, 1.0)
```

import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    s -= s.max()
    e = np.exp(s)
    g = e / e.sum()
    obj = m + np.log(e.sum()) / alpha
    grad = np.zeros(n)
    for mm in range(2 * n - 1):
        lag = n - 1 - mm
        a = max(0, -lag)
        b = min(n, n - lag)
        if b > a:
            grad[a:b] += g[mm] * (1.0 - h[a + lag:b + lag])
        c2 = max(0, lag)
        if c2 < n:
            grad[c2:n] -= g[mm] * h[c2 - lag:n - lag]
    return obj, grad

def proj_grad_descent(h, n, alpha, n_iter, deadline):
    h = h.copy()
    lr = 0.05
    best_obj = None
    for it in range(n_iter):
        if time.time() > deadline:
            break
        obj, grad = soft_max_obj_grad(h, n, alpha)
        if best_obj is None or obj < best_obj - 1e-14:
            best_obj = obj
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step_dir = -grad / gnorm
        for _ in range(50):
            h_try = project_feasible(h + lr * step_dir, n)
            obj_try, _ = soft_max_obj_grad(h_try, n, alpha)
            if obj_try < obj - 1e-16:
                h = h_try
                lr = min(lr * 1.25, 0.5)
                break
            lr *= 0.5
            if lr < 1e-10:
                break
    return h, compute_c5(h, n)

def grad_pair_search(h, n, deadline, n_iter=200):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    for _it in range(n_iter):
        if time.time() > deadline:
            break
        _, grad = soft_max_obj_grad(h, n, 50000)
        order = np.argsort(grad)
        improved = False
        n_pairs = min(30, n // 3)
        for k in range(n_pairs):
            if time.time() > deadline:
                return h, best_c
            i_low = order[k]
            i_high = order[n - 1 - k]
            if i_low == i_high:
                continue
            for d in deltas:
                if h[i_high] < d or h[i_low] + d > 1.0:
                    continue
                h_try = h.copy()
                h_try[i_high] -= d
                h_try[i_low] += d
                h_try = project_feasible(h_try, n)
                c = compute_c5(h_try, n)
                if c < best_c - 1e-14:
                    h = h_try
                    best_c = c
                    improved = True
                    break
        if not improved:
            break
    return h, best_c

def coord_search(h, n, deadline):
    h = h.copy()
    best_c = compute_c5(h, n)
    deltas = [0.01, 0.005, 0.002, 0.001, 0.0005,

    for rnd in range(50):

        for j in range(n):

                h_try = h.copy()

                    best_c = c
                    imp = True

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start.copy(), n)
    best_c = compute_c5(h, n)
    alphas = [50, 200, 800, 3000, 10000, 50000, 200000, 1000000, 5000000]
    for a in alphas:
        if time.time() > deadline - 30:
            break
        h_new, c_new = proj_grad_descent(h, n, a, 600, time.time() + 45)
        if c_new < best_c - 1e-15:
            best_c = c_new
            h = h_new
    if time.time() < deadline - 15:
        h, best_c = grad_pair_search(h, n, min(deadline - 15, time.time() + 120))
    if time.time() < deadline - 5:
        h, best_c = coord_search(h, n, deadline - 2)
    h = project_feasible(h, n)
    best_c = compute_c5(h, n)
    return h, best_c

    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 3

    try:
