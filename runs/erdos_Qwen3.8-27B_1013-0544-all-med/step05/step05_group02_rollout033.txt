            if tot < 1e-14:

    return np.clip(h, 0.0, 1.0)

        mx = shifted.max()
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-13: break
        if d > 0:
            r = 1.0 - h; t = r.sum()
            if t < 1e-13: break
            h += (d/t) * r
        else:
            t = h.sum()
            if t < 1e-13: break
            h += (d/t) * h
    return np.clip(h, 0.0, 1.0)

def _sc(x, n): return float(x.sum() - n/2.0)
def _scj(x, n): return np.ones(n)

def slsqp(h, n, alpha, mi=2000):
    dx = 2.0/n
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1-hx, mode='full')*dx
        m = c.max()
        s = alpha*(c-m)
        mx = s.max()
        return m + (mx + np.log(np.sum(np.exp(s-mx))))/alpha
    try:
        r = minimize(obj, h, method='SLSQP', bounds=[(0,1)]*n,
                     constraints=[{'type':'eq','fun':_sc,'jac':_scj,'args':(n,)}],
                     options={'maxiter':mi,'ftol':1e-18})
        return project_feasible(r.x, n)
    except: return h.copy()

def lag_grad(h, n, k):
    s = int(k) - (n-1)
    if s == 0: return np.zeros(n)
    g = np.zeros(n)
    lo, hi = max(0,-s), min(n-1, n-1-s)

        g[idx] += (1.0 - h[idx+s])

        idx2 = np.arange(lo2, hi2+1)
        g[idx2] -= h[idx2-s]

        if abs(d) < 1e-13:

        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-13:
                break
            h += (d / t) * r
        else:
            t = h.sum()
            if t < 1e-13:
                break
            h += (d / t) * h
    return np.clip(h, 0.0, 1.0)
