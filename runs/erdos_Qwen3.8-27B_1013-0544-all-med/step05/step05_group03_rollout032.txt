def kick(hb, n, rng, num_flips=4):
    """Randomly flip some bits, maintaining the constraint."""
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    if len(ones) < num_flips or len(zeros) < num_flips:
        return hb
    n_flip = min(num_flips, len(ones), len(zeros))
    flip_from = rng.choice(ones, n_flip, replace=False)
    flip_to = rng.choice(zeros, n_flip, replace=False)
    hb[flip_from] = 0.0
    hb[flip_to] = 1.0
    return hb

def slsqp_polish(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sh -= sh.max()
        return m + np.log(np.exp(sh).sum()) / alpha
    def con_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def build_prof(hb, n):
    one_minus_hb = 1.0 - hb
    prof = np.zeros((n, 2 * n - 1))
    for p in range(n):
        k_start = max(0, n - 1 - p)
        k_end = min(2 * n - 2, 2 * n - 2 - p)
        if k_start <= k_end:
            q_start = p + k_start - (n - 1)
            prof[p, k_start:k_end + 1] = one_minus_hb[q_start:q_start + (k_end - k_start + 1)]
    return prof

def fast_binary_search(hb, n, deadline, max_sweeps=10000):
    dx = 2.0 / n
    C = np.correlate(hb, 1 - hb, mode='full').astype(float)
    cur_max = C.max()
    best_c = cur_max * dx
    best_hb = hb.copy()
    ones_list = np.where(hb == 1)[0]
    zeros_list = np.where(hb == 0)[0]
    prof = build_prof(hb, n)
    p_idx = np.arange(n)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if len(zeros_list) == 0 or len(ones_list) == 0:
            break
        Zi = zeros_list
        Oj = ones_list
        Zi_M = C[None, :] + prof[Zi]
        Oj_prof = prof[Oj]
        diff3d = Zi_M[:, None, :] - Oj_prof[None, :, :]
        new_maxes = diff3d.max(axis=2)
        min_val = new_maxes.min()
        if min_val >= cur_max - 1e-12:
            break
        min_idx = np.unravel_index(new_maxes.argmin(), new_maxes.shape)
        best_i = Zi[min_idx[0]]
        best_j = Oj[min_idx[1]]
        C = C + prof[best_i] - prof[best_j]
        hb[best_i] = 1.0
        hb[best_j] = 0.0
        cur_max = C.max()
        if cur_max * dx < best_c:
            best_c = cur_max * dx
            best_hb = hb.copy()
        k_i = n - 1 + best_i - p_idx
        k_j = n - 1 + best_j - p_idx
        vi = (k_i >= 0) & (k_i <= 2 * n - 2)
        vj = (k_j >= 0) & (k_j <= 2 * n - 2)
        prof[p_idx[vi], k_i[vi]] -= 1.0
        prof[p_idx[vj], k_j[vj]] += 1.0
        ones_list = np.setdiff1d(ones_list, [best_j])
        zeros_list = np.setdiff1d(zeros_list, [best_i])
        ones_list = np.concatenate([ones_list, [best_i]])
        zeros_list = np.concatenate([zeros_list, [best_j]])
        if sweep % 2000 == 1999:
            C = np.correlate(hb, 1 - hb, mode='full').astype(float)
            prof = build_prof(hb, n)
    return best_hb, best_c

def kick(hb, n, rng, num_flips):
    ones = np.where(hb == 1)[0]
    zeros = np.where(hb == 0)[0]
    if len(ones) < num_flips or len(zeros) < num_flips:
        return hb
    nf = min(num_flips, len(ones), len(zeros))
    ff = rng.choice(ones, nf, replace=False)
    ft = rng.choice(zeros, nf, replace=False)
    hb[ff] = 0.0
    hb[ft] = 1.0
    return hb

def slsqp_polish(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sh -= sh.max()
        return m + np.log(np.exp(sh).sum()) / alpha
    def con_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h
