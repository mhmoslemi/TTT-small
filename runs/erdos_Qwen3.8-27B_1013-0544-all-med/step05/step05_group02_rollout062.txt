import time

def compute_c5(h, n):

    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k):
    s = k - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        g[idx] += (1.0 - h[idx + s])
    lo2 = max(0, lo + s)
    hi2 = min(n - 1, hi + s)
    if lo2 <= hi2:
        idx2 = np.arange(lo2, hi2 + 1)
        valid = (idx2 - s) >= lo
        valid &= (idx2 - s) <= hi
        g[idx2[valid]] -= h[idx2[valid] - s]
    return g

            g += w[k] * lag_gradient(h, n, k)

    return float(np.sum(x) - n / 2.0)

    return np.ones(n)

            g += wi * lag_gradient(h, n, int(ki))
        pos_mask = g > 1e-12
        neg_mask = g < -1e-12
        if not pos_mask.any() or not neg_mask.any():
            break
        pos_idx = np.where(pos_mask)[0][np.argsort(g[pos_mask])[::-1][:K]]
        neg_idx = np.where(neg_mask)[0][np.argsort(g[neg_mask])[:K]]
        improved = False
        for i in pos_idx:
            for j in neg_idx:
                max_step = min(h[i], 1.0 - h[j])
                if max_step < 1e-12:
                    continue
                for frac in [0.5, 0.25, 0.1, 0.05, 0.02]:
                    s = max_step * frac
                    if s < 1e-12:
                        break
                    h[i] -= s
                    h[j] += s
                    val = compute_c5(h, n)
                    if val < best_c - 1e-15:
                        best_c = val
                        best_h = h.copy()
                        improved = True
                    else:
                        h[i] += s
                        h[j] -= s
                        break
        if not improved:
            break
    return best_h, best_c

def optimize_at_n(h_start, n, deadline):
    h = project_feasible(h_start, n)
    best_c = compute_c5(h, n)
    best_h = h.copy()
    alphas = [100, 300, 1000, 3000, 10000, 30000, 100000, 300000]
    for a in alphas:
        if time.time() > deadline - 10:
            break
        h_try = slsqp_round(best_h, n, a, maxiter=3000)
        c_try = compute_c5(h_try, n)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h_try
    if time.time() < deadline - 5:
        best_h, best_c = local_search(best_h, n, deadline - 3)
    if time.time() < deadline - 3:
        h_try = slsqp_round(best_h, n, 500000, maxiter=2000)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try
        if time.time() < deadline - 2:
            best_h, best_c = local_search(best_h, n, deadline - 1)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 250
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc1 = min(350, (deadline - t0) * 0.35)
        h_opt, c_opt = optimize_at_n(h_init, n_init, t0 + alloc1)
        best_h, best_c = h_opt, c_opt
        x_init = np.linspace(0, 2, n_init)
        for n_try in [200, 300, 350, 400, 500, 150]:
            if n_try == n_init:
                continue
            if time.time() > deadline - 80:
                break
            remaining = deadline - time.time() - 30
            alloc = min(100, remaining * 0.2)
            if alloc < 25:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, best_h)
            h_interp = project_feasible(h_interp, n_try)
            h_try, c_try = optimize_at_n(h_interp, n_try, time.time() + alloc)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
                x_init = np.linspace(0, 2, n_try)
    if best_h is not None and time.time() < deadline - 50:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(3, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.002 + 0.05 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(50, (deadline - time.time() - 25) * 0.3)
            if alloc < 15:
                break
            h_try, c_try = optimize_at_n(hp, n, time.time() + alloc)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        best_h, best_c = local_search(best_h, n, deadline - 2)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        n = 250
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
