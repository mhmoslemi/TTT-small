            lo2 = max(0, n - 1 - j)
            hi2 = min(L - 1, 2 * n - 2 - j)
            if lo2 <= hi2:
                ks2 = np.arange(lo2, hi2 + 1)
                B[jj, ks2] += h[ks2 - n + 1 + j]

        best_val = cur_max
        best_ii, best_jj = -1, -1
        for ii in range(nz):
            D = C + A[ii]
            vals = (D[None, :] + B).max(axis=1)
            jj = vals.argmin()
            if vals[jj] < best_val - 1e-14:
                best_val = vals[jj]
                best_ii, best_jj = ii, jj

        if best_ii < 0:
            break
        i = zeros[best_ii]
        j = ones[best_jj]
        h[i], h[j] = 1.0, 0.0
        C = C + A[best_ii] + B[best_jj]
        C[n - 1] -= 2.0
        C[i - j + n - 1] += 1.0
        C[j - i + n - 1] += 1.0
        cur_max = C.max()
        if cur_max < best_max:
            best_max = cur_max
            best_h = h.copy()

    return best_h, float(best_max * dx)

def binary_annealing(h, n, deadline, rng, T0=2.0, T_min=0.005, stall_limit=50000):
    dx = 2.0 / n
    h = h.copy().astype(float)
    C = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = C.max()
    best_max = cur_max
    best_h = h.copy()
    T = T0
    total_steps = 3000000
    cooling_rate = (T_min / T0) ** (1.0 / total_steps)
    stall = 0

    for step in range(total_steps):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(zeros[rng.randint(len(zeros))])
        j = int(ones[rng.randint(len(ones))])
        h[i], h[j] = 1.0, 0.0
        new_C = np.correlate(h, 1.0 - h, mode='full')
        new_max = new_C.max()
        delta = new_max - cur_max
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            C = new_C
            cur_max = new_max
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = 0.0, 1.0
            stall += 1
        T = max(T * cooling_rate, T_min)
        if stall > stall_limit:
            break

    return best_h, float(best_max * dx)

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n

    def obj(x):
        g = np.correlate(x, 1.0 - x, mode='full') * dx
        m = g.max()
        sh = alpha * (g - m)
        sm = sh.max()
        return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

    def cons_fun(x):
        return x.sum() - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        if time.time() > deadline:
            return h
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    dl = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Quick SLSQP (5%)
    p1 = t0 + 0.05 * (budget_s - 3)
    h = project_feasible(h0, n)
    for alpha in [5000, 50000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 300, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 2: Main binary local search (45%)
    p2 = t0 + 0.50 * (budget_s - 3)
    hb = make_binary(best_h, n, rng)
    hb, cb = binary_local_search(hb, n, p2, rng, max_sweeps=300000)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    # Phase 3: Annealing exploration (25%)
    p3 = t0 + 0.75 * (budget_s - 3)
    for restart in range(20):
        if time.time() > p3 - 5:
            break
        if restart == 0:
            ha = make_binary(best_h, n, rng)
        elif restart < 4:
            ha = make_binary(best_h + 0.3 * rng.randn(n), n, rng)
        else:
            ha = make_binary(rng.rand(n), n, rng)
        ha, ca = binary_annealing(ha, n, min(p3 - 2, time.time() + 30), rng,
                                  T0=1.5, T_min=0.003)
        if ca < best_c:
            best_c, best_h = ca, ha.copy()
        if time.time() < p3 - 10:
            hb2, cb2 = binary_local_search(ha, n, p3 - 5, rng, max_sweeps=30000)
            if cb2 < best_c:
                best_c, best_h = cb2, hb2.copy()

    # Phase 4: SLSQP polish (10%)
    p4 = t0 + 0.85 * (budget_s - 3)
    hp = project_feasible(best_h + 0.03 * rng.randn(n), n)
    for alpha in [100000, 500000, 2000000]:
        if time.time() > p4:
            break
        hp = slsqp_smooth(hp, n, alpha, 2000, p4)
        c = compute_c5(hp, n)
        if c < best_c:
            best_c, best_h = c, hp.copy()

    # Phase 5: Final local search (10%)
    hb = make_binary(best_h, n, rng)
    hb, cb = binary_local_search(hb, n, dl, rng, max_sweeps=500000)
    if cb < best_c:
        best_c, best_h = cb, hb.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
