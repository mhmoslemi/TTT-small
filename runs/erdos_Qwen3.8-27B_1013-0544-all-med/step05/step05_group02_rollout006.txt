```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def lag_gradient(h, n, k_lag):
    """Gradient of c[k_lag] with respect to h.
    k_lag is the index in the full correlation array."""
    s = k_lag - (n - 1)  # actual shift
    g = np.zeros(n)
    # c[idx] = sum_i h[i] * (1-h[i+s]) where i and i+s are both in [0, n-1]
    # d/dh[j] c[idx]:
    #   if j appears as i: contribution is (1-h[j+s])
    #   if j appears as i+s: contribution is -h[j-s]
    lo = max(0, -s)
    hi = min(n-1, n-1-s)
    for j in range(lo, hi+1):
        g[j] += (1.0 - h[j+s])  # j as the "i" position
    lo2 = max(0, s)
    hi2 = min(n-1, n-1+s)
    for j in range(lo2, hi2+1):
        g[j] -= h[j-s]  # j as the "i+s" position
    return g

def projected_subgradient(h, n, deadline, top_k=5, max_iter=50000):
    """Projected subgradient method with momentum."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_h = h.copy()
    vel = np.zeros(n)
    momentum = 0.9
    
    # Adaptive step size
    eta = 0.1
    eta_min = 1e-6
    
    c = get_corr(curr_h, n)
    curr_max = float(c.max())
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Find top k lags
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            g += lag_gradient(curr_h, n, int(k))
        g /= top_k
        
        # Nesterov-style momentum
        vel = momentum * vel + g
        step = eta / (np.linalg.norm(vel) + 1e-12)
        
        new_h = project_feasible(curr_h - step * vel, n)
        new_c = get_corr(new_h, n)
        new_max = float(new_c.max())
        
        if new_max < curr_max - 1e-15:
            curr_h = new_h
            curr_max = new_max
            c = new_c
            if curr_max < best_c:
                best_c = curr_max
                best_h = curr_h.copy()
            eta = min(eta * 1.01, 1.0)
        else:
            eta *= 0.98
            if eta < eta_min:
                break
        
        # Decay momentum slightly
        if it % 1000 == 0:
            momentum = max(0.5, momentum * 0.99)
    
    return best_h, best_c
```

```python
            cnt = mask.sum()
            if cnt == 0: break
            h[mask] += diff / cnt

def lag_gradient(h, n, k_idx):

    s = k_idx - (n - 1)
    g = np.zeros(n)
    lo = max(0, -s)
    hi = min(n-1, n-1-s)

    hi2 = min(n-1, n-1+s)
    idx2 = np.arange(lo2, hi2+1)
```

```python
def adam_optimize(h, n, deadline, top_k=3, max_iter=100000):
    """Adam optimizer for the non-convex C5 problem."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_h = h.copy()
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    lr = 0.05
    eps = 1e-8
    
    c = get_corr(curr_h, n)
    curr_max = float(c.max())
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute gradient (average of top k lags)
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            g += lag_gradient(curr_h, n, int(k))
        g /= top_k
        
        # Adam update
        m = beta1 * m + (1-beta1) * g
        v = beta2 * v + (1-beta2) * g**2
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        
        new_h = curr_h - lr * m_hat / (np.sqrt(v_hat) + eps)
        new_h = project_feasible(new_h, n)
        
        new_c = get_corr(new_h, n)
        new_max = float(new_c.max())
        
        if new_max <= curr_max + 1e-14:
            curr_h = new_h
            curr_max = new_max
            c = new_c
            if curr_max < best_c:
                best_c = curr_max
                best_h = curr_h.copy()
                # Don't change lr when improving
            else:
                lr = max(lr * 0.999, 1e-6)
        else:
            lr = max(lr * 0.95, 1e-7)
            if lr < 1e-7:
                break
    
    return best_h, best_c
```

```python
def pair_swap_polish(h, n, deadline, max_stall=500):
    """Try all pairs of (source, dest) and pick the best swap."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_h = h.copy()
    curr_c = compute_c5(curr_h, n)
    c = get_corr(curr_h, n)
    
    stall = 0
    step_sizes = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    step_idx = 0
    
    while time.time() < deadline:
        k_star = int(np.argmax(c))
        g = lag_gradient(curr_h, n, k_star)
        
        # Find best (source, dest) pair
        # Source: where g is most positive (decreasing h there reduces c_k_star)
        # Dest: where g is most negative (increasing h there reduces c_k_star)
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        
        if i_src == i_dst:
            break
        
        d = step_sizes[step_idx]
        d = min(d, curr_h[i_src] * 0.9, (1.0 - curr_h[i_dst]) * 0.9)
        
        if d < 1e-12:
            step_idx = min(step_idx + 1, len(step_sizes) - 1)
            if step_idx >= len(step_sizes) - 1:
                stall += 1
                if stall > max_stall:
                    break
            continue
        
        curr_h[i_src] -= d
        curr_h[i_dst] += d
        
        new_c = get_corr(curr_h, n)
        new_max = float(new_c.max())
        
        if new_max < curr_c - 1e-15:
            curr_c = new_max
            c = new_c
            stall = 0
            step_idx = max(0, step_idx - 2)
            if curr_c < best_c:
                best_c = curr_c
                best_h = curr_h.copy()
        else:
            curr_h[i_src] += d
            curr_h[i_dst] -= d
            step_idx = min(step_idx + 1, len(step_sizes) - 1)
            stall += 1
            if stall > max_stall:
                break
    
    return best_h, best_c
```

```python
import cvxpy as cp

def active_set_newton(h, n, deadline, max_iter=100):
    """Active set method: identify active lags, solve LP for direction, take step."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_h = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        c = get_corr(curr_h, n)
        curr_max = float(c.max())
        
        # Identify active set (lags within epsilon of max)
        threshold = curr_max - 0.01
        active = np.where(c >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        
        # Build LP: direction = dp (delta h)
        # min t
        # s.t. for k in active: sum_i grad_k[i] * dp[i] <= t - c[k]
        #      sum(dp) = 0
        #      -h[i] <= dp[i] <= 1-h[i]
        dp = cp.Variable(n)
        t = cp.Variable()
        
        constraints = [cp.sum(dp) == 0]
        constraints += [dp >= -curr_h]
        constraints += [dp <= 1 - curr_h]
        
        for k in active:
            g = lag_gradient(curr_h, n, int(k))
            constraints.append(cp.sum(g * dp) <= t - c[int(k)])
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.GLOP)  # or CLP
        except:
            try:
                prob.solve(solver=cp.CLP)
            except:
                break
        
        if dp.value is None:
            break
        
        step = dp.value
        # Limit step size
        max_step = np.abs(step).max()
        if max_step > 0.1:
            step = step * 0.1 / max_step
        
        new_h = project_feasible(curr_h + step, n)
        new_c = compute_c5(new_h, n)
        
        if new_c < curr_max - 1e-15:
            curr_h = new_h
            if new_c < best_c:
                best_c = new_c
                best_h = curr_h.copy()
        else:
            # Try smaller step
            for alpha in [0.5, 0.1, 0.01]:
                new_h = project_feasible(curr_h + alpha * step, n)
                new_c = compute_c5(new_h, n)
                if new_c < curr_max - 1e-15:
                    curr_h = new_h
                    if new_c < best_c:
                        best_c = new_c
                        best_h = curr_h.copy()
                    break
            else:
                break
    
    return best_h, best_c
```

```python
        c = get_corr(h, n)
        new_c = float(c.max())
        delta = new_c - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            curr_c = new_c
            pos_idx = np.where(h == 1.0)[0]
            neg_idx = np.where(h == 0.0)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        T *= 0.9997
    return best_h, best_c

def adam_optimize(h, n, deadline, top_k=3, lr_init=0.03):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = lr_init
    c = get_corr(curr_h, n)
    curr_max = float(c.max())
    stall = 0
    it = 0
    while time.time() < deadline:
        top_idx = np.argsort(c)[-top_k:]
        g = np.zeros(n)
        for k in top_idx:
            g += lag_gradient(curr_h, n, int(k))
        g /= top_k
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        new_h = project_feasible(curr_h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        new_c_arr = get_corr(new_h, n)
        new_max = float(new_c_arr.max())
        if new_max < curr_max - 1e-15:
            curr_h = new_h
            curr_max = new_max
            c = new_c_arr
            stall = 0
            if curr_max < best_c:
                best_c = curr_max
                best_h = curr_h.copy()
        else:
            stall += 1
            lr = max(lr * 0.97, 1e-9)
            if stall > 800:
                break
        it += 1
    return best_h, best_c

def active_set_lp(h, n, deadline, max_iter=200):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_h = h.copy()
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        c = get_corr(curr_h, n)
        curr_max = float(c.max())
        threshold = curr_max - 0.02
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        if len(active) > 8:
            active = np.argsort(c)[-8:]
        dp = cp.Variable(n)
        t = cp.Variable()
        cons = [cp.sum(dp) == 0, dp >= -curr_h + 1e-10, dp <= 1 - curr_h - 1e-10]
        for k in active:
            g = lag_gradient(curr_h, n, int(k))
            cons.append(cp.sum(g * dp) <= t - c[int(k)] + 1e-10)
        prob = cp.Problem(cp.Minimize(t), cons)
        try:
            prob.solve(solver=cp.GLOP)
        except Exception:
            try:
                prob.solve(solver=cp.CLARO or cp.SCS)
            except Exception:
                break
        if dp.value is None:
            break
        step = np.clip(dp.value, -curr_h + 1e-12, 1 - curr_h - 1e-12)
        max_step = np.abs(step).max()
        if max_step < 1e-12:
            break
        if max_step > 0.05:
            step = step * 0.05 / max_step
        found = False
        for alpha in [1.0, 0.5, 0.2, 0.1, 0.05]:
            new_h = project_feasible(curr_h + alpha * step, n)
            new_c = compute_c5(new_h, n)
            if new_c < curr_max - 1e-14:
                curr_h = new_h
                if new_c < best_c:
                    best_c = new_c
                    best_h = curr_h.copy()
                found = True
                break
        if not found:
            break
    return best_h, best_c

def pair_swap_polish(h, n, deadline, max_stall=400):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    curr_h = h.copy()
    curr_c = best_c
    c = get_corr(curr_h, n)
    stall = 0
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    si = 0
    while time.time() < deadline:
        k_star = int(np.argmax(c))
        g = lag_gradient(curr_h, n, k_star)
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = steps[si]
        d = min(d, curr_h[i_src] * 0.9, (1.0 - curr_h[i_dst]) * 0.9)
        if d < 1e-12:
            si = min(si + 1, len(steps) - 1)
            stall += 1
            if stall > max_stall:
                break
            continue
        curr_h[i_src] -= d
        curr_h[i_dst] += d
        new_c = get_corr(curr_h, n)
        new_max = float(new_c.max())
        if new_max < curr_c - 1e-15:
            curr_c = new_max
            c = new_c
            si = max(0, si - 2)
            stall = 0
            if curr_c < best_c:
                best_c = curr_c
                best_h = curr_h.copy()
        else:
            curr_h[i_src] += d
            curr_h[i_dst] -= d
            si = min(si + 1, len(steps) - 1)
            stall += 1
            if stall > max_stall:
                break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    except Exception:
        n = 250
        h_init = np.full(n, 0.5)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Adam from initial
    if time.time() < deadline - 300:
        h_a, c_a = adam_optimize(best_h, n, min(deadline - 250, t0 + 180), top_k=3, lr_init=0.03)
        if c_a < best_c:
            best_c = c_a
            best_h = h_a.copy()

    # Phase 2: Active set LP
    if time.time() < deadline - 200:
        h_lp, c_lp = active_set_lp(best_h, n, min(deadline - 150, t0 + 300))
        if c_lp < best_c:
            best_c = c_lp
            best_h = h_lp.copy()

    # Phase 3: Pair swap polish
    if time.time() < deadline - 80:
        h_ps, c_ps = pair_swap_polish(best_h, n, min(deadline - 50, t0 + 400))
        if c_ps < best_c:
            best_c = c_ps
            best_h = h_ps.copy()

    # Phase 4: Binary SA for diversification
    if time.time() < deadline - 100:
        h_bin, c_bin = binary_sa(best_h, n, min(deadline - 100, t0 + 450), seed=seed + 1)
        if c_bin < best_c + 0.02:
            h_relax = np.clip(h_bin + 0.05 * rng.randn(n), 0, 1)
            h_relax = project_feasible(h_relax, n)
            h_ad2, c_ad2 = adam_optimize(h_relax, n, min(deadline - 60, t0 + 550), top_k=3, lr_init=0.02)
            if c_ad2 < best_c:
                best_c = c_ad2
                best_h = h_ad2.copy()
            h_lp2, c_lp2 = active_set_lp(h_ad2, n, min(deadline - 40, t0 + 600))
            if c_lp2 < best_c:
                best_c = c_lp2
                best_h = h_lp2.copy()

    # Phase 5: Random restarts
    if time.time() < deadline - 30:
        n_restarts = max(2, int((deadline - time.time() - 30) / 80))
        for r in range(n_restarts):
            if time.time() > deadline - 30:
                break
            scale = 0.02 + 0.04 * r
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            h_ad3, c_ad3 = adam_optimize(hp, n, min(deadline - min(25, 10 * (n_restarts - r)), t0 + 700 + r * 30), top_k=3, lr_init=0.02)
            if c_ad3 < best_c:
                best_c = c_ad3
                best_h = h_ad3.copy()

    # Phase 6: Final polish
    if time.time() < deadline - 5:
        h_final, c_final = pair_swap_polish(best_h, n, deadline - 2)
        if c_final < best_c:
            best_c = c_final
            best_h = h_final.copy()
        h_lp3, c_lp3 = active_set_lp(best_h, n, deadline - 1)
        if c_lp3 < best_c:
            best_c = c_lp3
            best_h = h_lp3.copy()

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
```
