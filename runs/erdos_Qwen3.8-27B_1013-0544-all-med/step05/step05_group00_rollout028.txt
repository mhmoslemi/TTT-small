def sa_binary(h_init, n, deadline, seed, max_iter=100000):
    """Simulated annealing on binary solutions with informed moves."""
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0

    dx = 2.0 / n
    c_curr = float(corr.max() * dx)
    best_c = c_curr
    best_h = h.copy()
    
    T = 0.05
    T_min = 0.0005
    cooling = 0.9995
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Choose swap: 70% random, 30% informed
        if rng.random() < 0.3:
            # Informed: pick zero with most neighbors=1, and one with most neighbors=0
            # (This tends to reduce overlap)
            z_scores = np.zeros(len(zeros))
            o_scores = np.zeros(len(ones))
            for zi, z in enumerate(zeros):
                z_scores[zi] = h[max(0,z-5):z+6].sum()  # local density around z
            for oi, o in enumerate(ones):
                o_scores[oi] = 1.0 - h[max(0,o-5):o+6].sum() / 11.0  # want isolated ones
            i = zeros[np.argmax(z_scores)]
            j = ones[np.argmax(o_scores)]
        else:
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
        
        d = swap_delta_vec(h, i, j, n)
        new_corr = corr + d
        c_new = float(new_corr.max() * dx)
        delta_c = c_new - c_curr
        
        if delta_c < 0 or (T > T_min and rng.random() < np.exp(-delta_c / T)):
            h[i], h[j] = 1.0, 0.0
            corr = new_corr
            c_curr = c_new

                best_c = c_curr
                best_h = h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]

    #
    # Let's define W[k-n+1] = w[k] for k in -(n-1)..(n-1)
    # Then sum_k w[k] * x[i+k] = (W * x)[i] (cross-correlation)
    # = np.correlate(w_full, x, 'same') ... hmm, need to be careful
    #

    # This is np.convolve(x, W_full_reversed, 'same') or something
    #
    # Let me just use: term1[i] = sum_k w[k] * (1-x[i+k]) for valid i+k
    # = len_valid - sum_k w[k] * x[i+k] for valid i+k
    #
    # This is getting complicated. Let me just use the direct formula with numpy:

        # w[k] * x[i-k] for valid i-k: i in max(0,k) .. min(n-1, n-1+k)
        i_start1 = max(0, -k)
        i_end1 = min(n, n-k)  # exclusive

        i_end2 = min(n, n+k)  # exclusive

            grad[i_start2:i_end2] -= w[k_idx] * x[i_start2-k:i_end2-k] * dx
    
    return m + (a_max + np.log(w.sum())) / alpha, grad

def project_feasible(h, n):

            if s < 1e-14:

def swap_delta(h, i, j, n):

    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        a = alpha * (g - m)
        am = a.max()
        return m + (am + np.log(np.exp(a-am).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})

    except:
        return h

def subgradient_step(h, n, lr):
    dx = 2.0 / n
    corr = corr_full(h)
    k_star = corr.argmax() -(n-1)
    grad = np.zeros(n)

        if 0 <= i-k_star < n:
            g -= h[i-k_star]
        grad[i] = g * dx
    h_new = h - lr * grad
    return project_feasible(h_new, n), k_star

            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def greedy_binary(h_init, n, deadline, seed, max_sweeps=3000):
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    c_curr = compute_c5(h, n)
    best_c = c_curr
    best_h = h.copy()
    stall = 0
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        if stall > 25:
            break
        best_nc = c_curr
        best_pair = None
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        npairs = len(zeros) * len(ones)
        if npairs <= 40000:
            for i in zeros:
                for j in ones:
                    h[i], h[j] = 1.0, 0.0
                    nc = compute_c5(h, n)
                    h[i], h[j] = 0.0, 1.0
                    if nc < best_nc - 1e-14:
                        best_nc = nc
                        best_pair = (i, j)
        else:
            ns = min(40000, npairs)
            for _ in range(ns):
                i = zeros[rng.randint(len(zeros))]
                j = ones[rng.randint(len(ones))]
                h[i], h[j] = 1.0, 0.0
                nc = compute_c5(h, n)
                h[i], h[j] = 0.0, 1.0
                if nc < best_nc - 1e-14:
                    best_nc = nc
                    best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i], h[j] = 1.0, 0.0
            c_curr = best_nc
            if c_curr < best_c:
                best_c = c_curr
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
    return best_h, best_c

def sa_binary(h_init, n, deadline, seed, n_iter=500000):
    rng = np.random.RandomState(seed)
    target = n // 2
    idx = np.argsort(h_init)[::-1]
    h = np.zeros(n)
    h[idx[:target]] = 1.0
    dx = 2.0 / n
    c_curr = compute_c5(h, n)
    best_c = c_curr
    best_h = h.copy()
    T = 0.04
    T_min = 0.0005
    cool = 0.99995
    for it in range(n_iter):
        if time.time() > deadline:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i], h[j] = 1.0, 0.0
        c_new = compute_c5(h, n)
        delta = c_new - c_curr
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            c_curr = c_new
            if c_curr < best_c:
                best_c = c_curr
                best_h = h.copy()
        else:
            h[i], h[j] = 0.0, 1.0
        T *= cool
    return best_h, best_c

def slsqp_smooth(h, n, alpha, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        a = alpha * (g - m)
        am = a.max()
        return m + (am + np.log(np.exp(a - am).sum())) / alpha
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h
