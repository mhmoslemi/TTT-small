```python
def compute_weighted_gradient(h, w, n):
    """Compute gradient of weighted sum of g_k with respect to h."""
    dx = 2.0 / n
    d = np.arange(2*n-1) - (n-1)  # d ranges from -(n-1) to (n-1)
    grad = np.zeros(n)
    for j in range(n):
        # Sum over d where j+d is in [0,n): contribution -h[j+d] * w[d+n-1]
        valid_plus = (j + d >= 0) & (j + d < n)
        # Sum over d where j-d is in [0,n): contribution (1-h[j-d]) * w[d+n-1]
        valid_minus = (j - d >= 0) & (j - d < n)
        grad[j] = dx * (
            np.sum(w[valid_minus] * (1 - h[j - d[valid_minus]])) - 
            np.sum(w[valid_plus] * h[j + d[valid_plus]])
        )
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max(g, alpha):
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def slsqp_optimize(h, n, alpha, maxiter=200):
    dx = 2.0 / n
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max(g, alpha)
    def obj_grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w /= w.sum()
        # gradient of smooth max
        d = np.arange(2*n-1) - (n-1)
        grad = np.zeros(n)
        for j in range(n):
            vp = (j + d >= 0) & (j + d < n)
            vm = (j - d >= 0) & (j - d < n)
            grad[j] = dx * (np.sum(w[vm] * (1 - x[j - d[vm]])) - np.sum(w[vp] * x[j + d[vp]]))
        return grad
    def cons_fun(x):
        return x.sum() - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def weighted_subgradient(h, n, deadline, eta_init=0.5, lr_init=0.01):
    """Mirror descent with exponential weights on shifts."""
    dx = 2.0 / n
    m_total = 2 * n - 1
    w = np.ones(m_total) / m_total
    best_h = h.copy()
    best_c = compute_c5(h, n)
    d = np.arange(m_total) - (n - 1)
    eta = eta_init
    lr = lr_init
    t = 0
    while time.time() < deadline:
        t += 1
        g = compute_g(h, n)
        g_max = g.max()
        # Update weights (exponential weights / multiplicative weights)
        w *= np.exp(eta * g / max(g_max, 1e-10))
        w /= w.sum()
        # Compute weighted gradient
        grad = np.zeros(n)
        for j in range(n):
            vp = (j + d >= 0) & (j + d < n)
            vm = (j - d >= 0) & (j - d < n)
            grad[j] = dx * (np.sum(w[vm] * (1 - h[j - d[vm]])) - np.sum(w[vp] * h[j + d[vp]]))
        # Take gradient step
        h = h - lr * grad
        h = project_feasible(h, n)
        # Check improvement
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        # Decay learning rate and eta
        lr = lr_init / (1 + t * 0.01)
        eta = eta_init * (1 + 0.5 * np.log(1 + t / 100))
    return best_h, best_c
```

```python
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(g, alpha):
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def slsqp_step(h, n, alpha, maxiter=200):
    dx = 2.0 / n
    dm = np.arange(2*n-1) - (n-1)
    def obj(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        return smooth_max_obj(g, alpha)
    def grad(x):
        g = np.correlate(x, 1-x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w /= w.sum()
        out = np.zeros(n)
        for j in range(n):
            vp = (j + dm >= 0) & (j + dm < n)
            vm = (j - dm >= 0) & (j - dm < n)
            out[j] = dx * (np.sum(w[vm] * (1 - x[j - dm[vm]])) - np.sum(w[vp] * x[j + dm[vp]]))
        return out
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(g, alpha):
    m = g.max()
    shifted = alpha * (g - m)
    sm = shifted.max()
    return m + (sm + np.log(np.exp(shifted - sm).sum())) / alpha

def slsqp_step(h, n, alpha, maxiter=200):
    dx = 2.0 / n
    dm = np.arange(2 * n - 1) - (n - 1)
    def obj(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        return smooth_max_val(g, alpha)
    def grad(x):
        g = np.correlate(x, 1 - x, mode='full') * dx
        m = g.max()
        shifted = alpha * (g - m)
        sm = shifted.max()
        w = np.exp(shifted - sm)
        w /= w.sum()
        out = np.zeros(n)
        for j in range(n):
            vp = (j + dm >= 0) & (j + dm < n)
            vm = (j - dm >= 0) & (j - dm < n)
            out[j] = dx * (np.sum(w[vm] * (1 - x[j - dm[vm]])) - np.sum(w[vp] * x[j + dm[vp]]))
        return out
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n / 2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        return project_feasible(res.x, n)
    except Exception:
        return h

def pair_descent(h, n, deadline, K=16, recompute_every=200):
    dx = 2.0 / n
    m_total = 2 * n - 1
    dm = np.arange(m_total) - (n - 1)
    best_h = h.copy()
    best_c = compute_c5(h, n)
    stall = 0
    swaps_since_recompute = 0
    for sweep in range(10000):
        if time.time() > deadline:
            break
        g = compute_g(h, n)
        current_max = g.max()
        top_idx = np.argsort(g)[::-1][:K]
        F = np.zeros((n, m_total))
        for j in range(n):
            vp = (j + dm >= 0) & (j + dm < n)
            vm = (j - dm >= 0) & (j - dm < n)
            F[j, vm] = (1.0 - h[j - dm[vm]])
            F[j, vp] = -h[j + dm[vp]]
        improved = 0
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                s = F[i] - F[j]
                d_lo = max(-h[i], h[j] - 1.0)
                d_hi = min(1.0 - h[i], h[j])
                if d_hi - d_lo < 1e-13:
                    continue
                candidates = [d_lo, d_hi]
                for a in range(K):
                    for b in range(a + 1, K):
                        ta, tb = top_idx[a], top_idx[b]
                        ds = s[ta] - s[tb]
                        if abs(ds) > 1e-12:
                            dc = (g[ta] - g[tb]) / (dx * ds)
                            if d_lo < dc < d_hi:
                                candidates.append(dc)
                best_delta = 0.0
                best_val = current_max
                for dd in candidates:
                    dd = float(np.clip(dd, d_lo, d_hi))
                    val = float(np.max(g + dd * dx * s))
                    if val < best_val - 1e-12:
                        best_val = val
                        best_delta = dd
                if best_delta != 0.0:
                    h[i] += best_delta
                    h[j] -= best_delta
                    g += best_delta * dx * s
                    sign_i = 1
                    for pos_off in range(m_total):
                        d = dm[pos_off]
                        p1 = i + d
                        if 0 <= p1 < n:
                            F[p1, pos_off] -= sign_i * best_delta
                        p2 = i - d
                        if 0 <= p2 < n:
                            F[p2, pos_off] -= sign_i * best_delta
                        sign_j = -1
                        q1 = j + d
                        if 0 <= q1 < n:
                            F[q1, pos_off] -= sign_j * best_delta
                        q2 = j - d
                        if 0 <= q2 < n:
                            F[q2, pos_off] -= sign_j * best_delta
                    current_max = g.max()
                    top_idx = np.argsort(g)[::-1][:K]
                    improved += 1
                    swaps_since_recompute += 1
                    if swaps_since_recompute >= recompute_every:
                        F = np.zeros((n, m_total))
                        for jj in range(n):
                            vp = (jj + dm >= 0) & (jj + dm < n)
                            vm = (jj - dm >= 0) & (jj - dm < n)
                            F[jj, vm] = (1.0 - h[jj - dm[vm]])
                            F[jj, vp] = -h[jj + dm[vp]]
                        swaps_since_recompute = 0
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        if improved == 0:
            stall += 1
            if stall > 6:
                break
        else:
            stall = 0
    return best_h, best_c

    t0 = time.time()
    total = budget_s - 10
```
