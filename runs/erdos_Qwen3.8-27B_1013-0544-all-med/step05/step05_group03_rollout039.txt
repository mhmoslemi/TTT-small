from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def build_VW(hb, n):
    """Build V and W matrices for fast swap evaluation."""
    V = np.zeros((n, n-1))
    W = np.zeros((n, n-1))
    for p in range(n):
        L = n - 1 - p
        if L > 0:
            V[p, :L] = 1.0 - hb[p+1:p+1+L]
            W[p, :L] = hb[p+1:p+1+L]
    return V, W

def update_VW(V, W, hb, n, i, j):
    """Update V and W after swapping hb[i]=1, hb[j]=0."""
    # hb[i] went 0->1: affects V[p, i-p-1] and W[p, i-p-1] for p=0..i-1
    for p in range(i):
        k = i - p - 1
        V[p, k] -= 1.0
        W[p, k] += 1.0
    # hb[j] went 1->0: affects V[p, j-p-1] and W[p, j-p-1] for p=0..j-1
    for p in range(j):
        k = j - p - 1
        V[p, k] += 1.0
        W[p, k] -= 1.0

def steepest_descent_binary(hb, n, V, W, A_pos, A_neg, deadline, max_rounds=50000):

    ones = np.where(hb == 1)[0].tolist()
    zeros = np.where(hb == 0)[0].tolist()
    
    rounds = 0
    while time.time() < deadline and rounds < max_rounds:
        rounds += 1

        best_i = -1
        best_j = -1

            Wi = W[i]

                neg_max = (A_neg - Wi + W[j]).max()
                nm = max(pos_max, neg_max, 0.0)
                if nm < best_new_max:
                    best_new_max = nm
                    best_i = i
                    best_j = j

        hb[best_j] = 0.0
        update_VW(V, W, hb, n, best_i, best_j)
        
        # Update A_pos, A_neg (recompute from corr)

        A_pos = corr[n:]
        A_neg = corr[0:n-1][::-1]

        ones = np.where(hb == 1)[0].tolist()
        zeros = np.where(hb == 0)[0].tolist()
