    # up: shape (n, 2n-1)
    # zeros, ones: arrays of indices
    # Returns (best_i, best_j, new_max_value)
    
    up_zeros = up[zeros]  # shape (nz, 2n-1)
    up_ones = up[ones]    # shape (no, 2n-1)
    
    # delta[i, j, m] = up_zeros[i, m] - up_ones[j, m]
    # new_c[i, j, m] = c[m] + delta[i, j, m]
    # new_max[i, j] = max_m new_c[i, j, m]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_all(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return c * dx

def to_binary(h, n, rng=None):
    """Convert to binary with sum = n/2."""
    target = n // 2
    if rng is not None:
        # Break ties randomly
        h_noisy = h + rng.randn(n) * 1e-6
    else:
        h_noisy = h
    idx = np.argsort(h_noisy)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def compute_up_matrix(h, n):
    """Compute up[p, m_idx] for all positions p and lag indices m_idx."""
    m = np.arange(-(n-1), n)[None, :]  # (1, 2n-1)
    p = np.arange(n)[:, None]  # (n, 1)
    
    idx1 = p - m  # (n, 2n-1), represents position p-m
    valid1 = (idx1 >= 0) & (idx1 < n)
    
    idx2 = p + m  # (n, 2n-1), represents position p+m
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    up = np.zeros((n, 2*n-1))
    up[valid1] += 1.0 - h[idx1[valid1]]
    up[valid2] += -h[idx2[valid2]]
    
    return up

def hill_climb_binary(h, n, deadline, rng, max_stalls=200):
    """Binary hill climbing with vectorized best-swap search."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    best_c = float(c.max())
    best_h = h.copy()
    stalls = 0
    
    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Compute up matrix
        up = compute_up_matrix(h, n)
        
        # Get up for zeros and ones
        up_z = up[zeros]  # (nz, 2n-1)
        up_o = up[ones]   # (no, 2n-1)
        
        # Compute new max for all pairs
        # To save memory, do in chunks if needed
        # For n=250: nz=no=125, 2n-1=499, so 125*125*499 = 7.8M
        delta = up_z[:, None, :] - up_o[None, :, :]
        new_c = c[None, None, :] + delta
        new_max = new_c.max(axis=2)
        
        best_idx = new_max.argmin()
        bi, bj = np.unravel_index(best_idx, new_max.shape)
        new_best_c = float(new_max[bi, bj])
        
        if new_best_c < best_c - 1e-14:
            # Apply the swap
            new_h = h.copy()
            new_h[zeros[bi]] = 1.0
            new_h[ones[bj]] = 0.0
            h = new_h
            c = np.correlate(h, 1 - h, mode='full') * dx
            best_c = new_best_c
            best_h = h.copy()
            stalls = 0
        else:
            # Stuck - try simulated annealing step
            stalls += 1
            # Random swap with occasional acceptance
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            h_s = h.copy()
            h_s[i] = 1.0
            h_s[j] = 0.0
            c_s = float(np.max(np.correlate(h_s, 1 - h_s, mode='full')) * dx)
            
            if c_s < best_c or rng.random() < np.exp(-(c_s - best_c) / 0.001):
                h = h_s
                c = np.correlate(h, 1 - h, mode='full') * dx
            
            if stalls > max_stalls:
                # Restart with perturbation
                h = best_h.copy()
                for _ in range(10):
                    i = zeros[rng.randint(len(zeros))]
                    j = ones[rng.randint(len(ones))]
                    h[i], h[j] = 1.0 - h[i], h[i]
                # Re-ensure sum
                while h.sum() > n/2:
                    j = rng.choice(np.where(h == 1)[0])
                    h[j] = 0
                while h.sum() < n/2:
                    i = rng.choice(np.where(h == 0)[0])
                    h[i] = 1
                c = np.correlate(h, 1 - h, mode='full') * dx
                stalls = 0
    
    return best_h, best_c

        h_out, c_out = hill_climb_binary(h_block, n, p1_deadline, rng)
        if c_out < best_c:
            best_c, best_h = c_out, h_out.copy()

    if c_out < best_c:

    # Phase 3: Smooth optimization (10% time)
    if time.time() < deadline:
        h_start = best_h.copy()
        dx = 2.0 / n
        
        def obj(x):
            c = np.correlate(x, 1-x, mode='full') * dx
            m = c.max()
            alpha = 10000
            sh = alpha * (c - m)
            sm = sh.max()
            val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
            val += 1e4 * (x.sum() - n/2.0)**2
            return val
        
        def grad(x):
            c = np.correlate(x, 1-x, mode='full')
            m_star = int(np.argmax(c))
            k = m_star - (n-1)
            g = np.zeros(n)
            if k >= 0:

                g[:n-k] += -x[k:]
            else:
                g[:n+k] += (1.0 - x[-k:])
                g[-k:] += -x[:n+k]
            g *= dx
            g += 2e4 * (x.sum() - n/2.0)
            return g

            h_new = np.clip(res.x, 0, 1)
            # Project to feasible
            diff = n/2.0 - h_new.sum()
            h_new = h_new + diff / n
            h_new = np.clip(h_new, 0, 1)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c, best_h = c_new, h_new
        except Exception:
            pass
    
    # Final check
    best_h = np.clip(best_h, 0, 1)
    diff = n/2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        # Project
        for _ in range(100):
            h_t = best_h.copy()
            if diff > 0:
                mask = h_t < 1.0
                if mask.sum() == 0: break
                h_t[mask] += diff / mask.sum()
            else:
                mask = h_t > 0.0
                if mask.sum() == 0: break
                h_t[mask] += diff / mask.sum()
            h_t = np.clip(h_t, 0, 1)
            new_diff = n/2.0 - h_t.sum()
            if abs(new_diff) < 1e-12:
                best_h = h_t
                break
            best_h = h_t
            diff = new_diff
    
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full'))) * dx

def to_binary(h, n, rng=None):
    target = n // 2
    if rng is not None:
        h_noisy = h + rng.randn(n) * 1e-8
    else:
        h_noisy = h
    idx = np.argsort(h_noisy)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def hill_climb_binary(h, n, deadline, rng, max_stalls=300, sa_temp=0.002):
    dx = 2.0 / n
    c = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_c = float(c)
    best_h = h.copy()
    stalls = 0
    sa_count = 0

    while time.time() < deadline:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        best_improvement = best_c
        best_pair = None

        for i in zeros:
            for j in ones:
                h[i] = 1.0
                h[j] = 0.0
                c_new = float(np.max(np.correlate(h, 1 - h, mode='full'))) * dx
                h[i] = 0.0
                h[j] = 1.0
                if c_new < best_improvement:
                    best_improvement = c_new
                    best_pair = (i, j)

        if best_pair is not None:
            h[best_pair[0]] = 1.0
            h[best_pair[1]] = 0.0
            best_c = best_improvement
            best_h = h.copy()
            stalls = 0
        else:
            stalls += 1
            # Simulated annealing step
            i = int(rng.randint(len(zeros)))
            j = int(rng.randint(len(ones)))
            zi, oj = zeros[i], ones[j]
            h[zi] = 1.0
            h[oj] = 0.0
            c_s = float(np.max(np.correlate(h, 1 - h, mode='full'))) * dx
            if c_s < best_c or rng.random() < np.exp(-(c_s - best_c) / sa_temp):
                h[zi] = 1.0
                h[oj] = 0.0
                # Update tracking
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[zi] = 0.0
                h[oj] = 1.0
            sa_count += 1
            if sa_count % 50 == 0:
                sa_temp = max(sa_temp * 0.97, 1e-6)
            if stalls > max_stalls:
                h = best_h.copy()
                # Perturb: do 5-15 random swaps
                n_swaps = int(rng.randint(5, 16))
                for _ in range(n_swaps):
                    ones = np.where(h == 1)[0]
                    zeros = np.where(h == 0)[0]
                    if len(ones) == 0 or len(zeros) == 0:
                        break
                    a = int(rng.randint(len(zeros)))
                    b = int(rng.randint(len(ones)))
                    h[zeros[a]] = 1.0
                    h[ones[b]] = 0.0
                # Fix sum if needed
                target = n // 2
                while h.sum() > target:
                    h[np.where(h == 1)[0][int(rng.randint(len(ones)))] = 0.0
                    ones = np.where(h == 1)[0]
                while h.sum() < target:
                    h[np.where(h == 0)[0][int(rng.randint(n - len(ones)))] = 1.0
                    ones = np.where(h == 1)[0]
                stalls = 0
                sa_count = 0
                sa_temp = 0.002

    return best_h, best_c

def block_pattern(n, block_frac, offset, rng):
    h = np.zeros(n)
    blen = max(1, int(n * block_frac))
    start = int(offset * (n - blen))
    h[start:start + blen] = 1.0
    # Ensure sum = n/2
    target = n // 2
    if h.sum() < target:
        remaining = target - int(h.sum())
        avail = np.where(h == 0)[0]
        perm = rng.permutation(len(avail))
        h[avail[perm[:remaining]]] = 1.0
    elif h.sum() > target:
        excess = int(h.sum()) - target
        ones = np.where(h == 1)[0]
        perm = rng.permutation(len(ones))
        h[ones[perm[:excess]]] = 0.0
    return h

def smooth_refine(h, n, deadline):
    dx = 2.0 / n
    best_c = compute_c5(h, n)
    best_h = h.copy()

    for alpha in [5000, 50000, 500000]:
        if time.time() > deadline - 2:
            break

        def obj(x):
            c = np.correlate(x, 1 - x, mode='full') * dx
            m = c.max()
            sh = alpha * (c - m)
            sm = sh.max()
            val = m + (sm + np.log(np.exp(sh - sm).sum())) / alpha
            val += 1e4 * (x.sum() - n / 2.0) ** 2
            return val

        def grad(x):
            c_raw = np.correlate(x, 1 - x, mode='full')
            m_star = int(np.argmax(c_raw))
            k = m_star - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                g[k:] += (1.0 - x[:n - k])
                g[:n - k] += -x[k:]
            else:
                g[:n + k] += (1.0 - x[-k:])
                g[-k:] += -x[:n + k]
            g *= dx
            g += 2e4 * (x.sum() - n / 2.0)
            return g

        try:
            res = minimize(obj, best_h.copy(), method='L-BFGS-B', jac=grad,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 200, 'ftol': 1e-14})
            h_new = np.clip(res.x, 0.0, 1.0)
            diff = n / 2.0 - h_new.sum()
            mask = (h_new < 1.0) if diff > 0 else (h_new > 0.0)
            if mask.sum() > 0:
                h_new[mask] += diff / mask.sum()
            h_new = np.clip(h_new, 0.0, 1.0)
            c_new = compute_c5(h_new, n)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        except Exception:
            pass

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 250
        h0 = np.full(n, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n)

    # Phase 1: Hill climb from binarized initial (25% time)
    p1 = t0 + 0.25 * (budget_s - 3)
    h_b = to_binary(best_h, n, rng)
    h_out, c_out = hill_climb_binary(h_b, n, p1, rng)
    if c_out < best_c:
        best_c, best_h = c_out, h_out.copy()

    # Phase 2: Hill climb from random binary (20% time)
    p2 = t0 + 0.45 * (budget_s - 3)
    if time.time() < p2:
        idx = rng.permutation(n)
        h_rand = np.zeros(n)
        h_rand[idx[:n // 2]] = 1.0
        h_out, c_out = hill_climb_binary(h_rand, n, p2, rng)
        if c_out < best_c:
            best_c, best_h = c_out, h_out.copy()

    # Phase 3: Hill climb from block patterns (20% time)
    p3 = t0 + 0.65 * (budget_s - 3)
    if time.time() < p3:
        for bf in [0.25, 0.4, 0.5]:
            if time.time() > p3 - 30:
                break
            h_block = block_pattern(n, bf, rng.random(), rng)
            h_out, c_out = hill_climb_binary(h_block, n, min(p3, time.time() + 60), rng)
            if c_out < best_c:
                best_c, best_h = c_out, h_out.copy()

    # Phase 4: Continue hill climbing from best (20% time)
    p4 = t0 + 0.85 * (budget_s - 3)
    if time.time() < p4:

        # If already binary, continue from there
        if np.allclose(h_b2, best_h):

            h_start = h_b2

    best_h = np.clip(best_h, 0.0, 1.0)
    target = n / 2.0

        if abs(diff) < 1e-13:

        else:

        best_h[mask] += diff / mask.sum()
        best_h = np.clip(best_h, 0.0, 1.0)

    best_c = compute_c5(best_h, n)
