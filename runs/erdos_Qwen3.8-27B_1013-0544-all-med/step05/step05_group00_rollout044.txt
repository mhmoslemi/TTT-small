from scipy.optimize import minimize

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def make_delta_buf(n):

def compute_delta(i, j, h, n, buf):
    buf[:] = 0.0
    m_range = np.arange(-(n - 1), n)
    im = i + m_range
    imn = i - m_range
    jm = j + m_range
    jmn = j - m_range
    msk = (im >= 0) & (im < n)
    buf[msk] += 1.0 - h[im[msk]]
    msk = (imn >= 0) & (imn < n)
    buf[msk] -= h[imn[msk]]
    msk = (jm >= 0) & (jm < n)
    buf[msk] -= 1.0 - h[jm[msk]]
    msk = (jmn >= 0) & (jmn < n)
    buf[msk] += h[jmn[msk]]
    buf[n - 1] -= 2.0
    k1 = n - 1 + (j - i)
    if 0 <= k1 < 2 * n - 1:
        buf[k1] += 1.0
    k2 = n - 1 + (i - j)
    if 0 <= k2 < 2 * n - 1:
