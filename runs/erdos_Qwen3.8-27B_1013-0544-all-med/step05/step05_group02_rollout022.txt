```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(x.sum() - n / 2.0)

def _sum_con_jac(x, n):
    return np.ones(len(x))

def slsqp_opt(h, n, alpha, maxiter):
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * (2.0 / n)
        m = c.max()
        s = alpha * (c - m)
        sm = s.max()
        return m + np.log(np.sum(np.exp(s - sm))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except:
        return h.copy()

def lag_grad(h, n, k_idx):
    s = k_idx - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        g[:n - s] += 1.0
        g[:n - s] -= h[s:n]
        g[s:] -= h[:n - s]
    else:
        ss = -s
        g[ss:] += 1.0
        g[ss:] -= h[:n - ss]
        g[:n - ss] -= h[ss:]
    return g

def thue_morse_seq(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def directed_polish(h, n, deadline, max_stall=100):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005]
    si = 0
    stall = 0
    while time.time() < deadline:
        c = get_corr(h, n)
        curr = float(c.max())
        if curr < best_c - 1e-14:
            best_c = curr
            best_h = h.copy()
        k_star = int(np.argmax(c))
        g = lag_grad(h, n, k_star)
        if g.max() < 1e-12 or -g.min() < 1e-12:
            break
        i_src = int(np.argmax(g))
        i_dst = int(np.argmin(g))
        if i_src == i_dst:
            break
        d = steps[si]
        d = min(d, h[i_src] * 0.9, (1.0 - h[i_dst]) * 0.9)
        if d < 1e-11:
            si = min(si + 1, len(steps) - 1)
            stall += 1
            if stall > max_stall:
                break
            continue
        h[i_src] -= d
        h[i_dst] += d
        new_c = compute_c5(h, n)
        if new_c < curr - 1e-15:
            stall = 0
            si = max(0, si - 1)
        else:
            h[i_src] += d
            h[i_dst] -= d
            stall += 1
            si = min(si + 1, len(steps) - 1)
            if stall > max_stall:
                break
    return best_h, best_c
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def get_corr(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(10000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-14:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def _sum_con(x, n):
    return float(x.sum() - n / 2.0)

    return np.ones(len(x))

def slsqp_opt(h, n, alpha, maxiter):
    def obj(x):
        hx = np.clip(x, 0, 1)
        c = np.correlate(hx, 1 - hx, mode='full') * (2.0 / n)
        m = c.max()
        s = alpha * (c - m)
        sm = s.max()
        return m + np.log(np.sum(np.exp(s - sm))) / alpha
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_con_jac, 'args': (n,)}]

                      options={'maxiter': maxiter, 'ftol': 1e-18})
```
