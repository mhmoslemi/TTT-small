import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, ts):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = ts - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_lag(h, lag, n):
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            i = np.arange(m)
            g[i] += 1.0 - h[i + lag]
            g[i + lag] -= h[i]
    else:
        la = -lag
        m = n - la
        if m > 0:
            i = np.arange(m)
            g[i + la] += 1.0 - h[i]
            g[i] -= h[i + la]
    return g

    s = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(s))) / beta
