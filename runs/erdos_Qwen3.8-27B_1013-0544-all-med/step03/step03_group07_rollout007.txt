    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)

        return np.ones(n)

        return sm + penalty_term, g + grad_pen

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    # Clip for numerical stability
    shifted = np.maximum(shifted, -50.0 / beta)
    exps = np.exp(beta * shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        t = abs(s)
        if s >= 0:
            grad[:n-t] += wk * (1.0 - h[t:])
            grad[t:] += -wk * h[:n-t]
        else:
            grad[t:] += wk * (1.0 - h[:n-t])
            grad[:n-t] += -wk * h[t:]
    grad *= dx
    return val, grad

        return smoothmax_val_grad(x, beta)[0]

        return smoothmax_val_grad(x, beta)[1]

        return x.sum() - target

                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def lbfgs_refine(h, maxiter, beta, penalty):

        sm, g = smoothmax_val_grad(x, beta)
        pen = 0.5 * penalty * (x.sum() - target)**2
        return sm + pen
    def grad(x):
        sm, g = smoothmax_val_grad(x, beta)
        g = g + penalty * (x.sum() - target)
        return g

                       bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

        c = compute_c5(x_new)
        if c < best_val:
            best_val = c
            best_x = x_new.copy()
        x = x_new
    return best_x, best_val

        h_ld = lag_descent(h_best, 3, 200)
        c = compute_c5(h_ld)
        if c < best_c5: h_best, best_c5 = h_ld, c
    if tl() > 30:
        h_final = slsqp_refine(h_best, maxiter=400, beta=15000)
        c = compute_c5(h_final)
        if c < best_c5: h_best, best_c5 = h_final, c

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
