import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_max_val(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta


def softmax_weights(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    e = np.exp(x_shift)
    return e / np.sum(e)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_grad_core(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    obj_val = smooth_max_val(corr, beta) * dx
    w = softmax_weights(corr, beta)
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = w[idx]
        if wi < 1e-10:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            k_max = n - 1 - lag
            grad[0:k_max + 1] += wi * (1.0 - h[lag:n])
            if lag < n:
                grad[lag:n] -= wi * h[0:n - lag]
        else:
            m = -lag
            grad[m:n] += wi * (1.0 - h[0:n - m])
            grad[0:n - m] -= wi * h[m:n]
    grad *= dx
    return obj_val, grad


def _obj_pen(x, dx, beta, target_sum, mu):
    val, g = smooth_max_grad_core(x, dx, beta)
    s = x.sum() - target_sum
    return val + mu * s * s, g + 2.0 * mu * s


def _obj_only(x, dx, beta, target_sum, mu):
    return _obj_pen(x, dx, beta, target_sum, mu)[0]


def _grad_only(x, dx, beta, target_sum, mu):
    return _obj_pen(x, dx, beta, target_sum, mu)[1]


def run_lbfgs(h, dx, beta, target_sum, mu, maxiter=500, ftol=1e-15, gtol=1e-12):
    n = len(h)
    bnds = Bounds(np.zeros(n), np.ones(n))
    res = minimize(_obj_only, h, args=(dx, beta, target_sum, mu),
                   method='L-BFGS-B', jac=_grad_only,
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': gtol})
    return res.x


def _sg_step(h, target_sum, grad, steps):
    for step in steps:
        h_t = project_h(h - step * grad, target_sum)
        return h_t, step
    return h, 0


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(100, 0.5)

    best_c5 = compute_c5(best_h)

    # Phase 1: Analytical gradient L-BFGS-B at current resolution with increasing beta
    n = len(best_h)
    dx = 2.0 / n
    target_sum = n / 2.0

    for beta in [50, 100, 200, 500, 1000, 2000, 5000, 10000]:
        if time_left() < 8:
            break
        for mu in [50, 500, 5000]:
            if time_left() < 4:
                break
            try:
                h_new = run_lbfgs(best_h, dx, beta, target_sum, mu,
                                  maxiter=800, ftol=1e-16, gtol=1e-14)
                h_new = project_h(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 2: Multi-resolution upsampling and optimization
    resolutions = [150, 200, 250, 300, 400, 500]
    for n_new in resolutions:

            break
        if n_new <= len(best_h):
            continue

        target_new = n_new / 2.0
        x_old = np.linspace(0, 2, len(best_h))
        x_new_g = np.linspace(0, 2, n_new)
        h_interp = np.interp(x_new_g, x_old, best_h)
        h_interp = project_h(h_interp, target_new)

                    if c5_cand < cur_c5:

                        cur_c5 = c5_cand

            if time_left() < 3:
                break
            _, grad = smooth_max_grad_core(best_h, dx_fin, beta_sg)
            s = best_h.sum() - target_fin
            grad += 200.0 * s
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-13:
                break
            grad /= gnorm
            found = False
            for step in steps_list:
                h_trial = project_h(best_h - step * grad, target_fin)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-15:
                    best_h = h_trial
                    best_c5 = c5_trial
                    found = True
                    no_improve = 0
                    break
            if not found:
                no_improve += 1
                if no_improve > 80:
                    sc = 0.005 * (1.0 + 0.001 * no_improve)
                    pert = np.random.randn(n_fin) * sc
                    h_trial = project_h(best_h + pert, target_fin)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5:
                        best_h = h_trial
                        best_c5 = c5_trial
                        no_improve = 0
                if no_improve > 600:
                    break

    # Phase 4: Alternating local moves on active lags
    for iteration in range(30000):
        if time_left() < 2:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:4]
        improved = False
        for idx in top_idx:
            if time_left() < 1:
                break
            lag = int(idx) - (n_fin - 1)
            abs_lag = abs(lag)
            if abs_lag >= n_fin:
                continue
            n_valid = n_fin - abs_lag
            for _ in range(8):
                i = np.random.randint(n_valid)
                j = i + abs_lag if lag > 0 else i - lag
                for d in [0.03, -0.03, 0.015, -0.015, 0.007, -0.007]:
                    h_trial = best_h.copy()
                    h_trial[i] += d
                    h_trial[j] -= d
                    if 0 <= h_trial[i] <= 1 and 0 <= h_trial[j] <= 1:
                        h_trial = project_h(h_trial, target_fin)
                        c5_t = compute_c5(h_trial)
                        if c5_t < best_c5 - 1e-15:
                            best_h = h_trial
                            best_c5 = c5_t
                            improved = True
                            break
                if improved:
                    break
            if improved:
                break
        if not improved and iteration % 800 == 799:
            pert = np.random.randn(n_fin) * 0.004
            h_trial = project_h(best_h + pert, target_fin)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial

    # Phase 5: Final high-precision L-BFGS-B polish
    if time_left() > 8:
        n_pol = len(best_h)
        target_pol = n_pol / 2.0
        dx_pol = 2.0 / n_pol
        for beta_final in [5000, 10000, 20000]:
            if time_left() < 5:

                if time_left() < 3:

                                      maxiter=3000, ftol=1e-17, gtol=1e-15)
                    h_new = project_h(h_new, target_pol)

                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                except Exception:
                    pass

    return best_h, best_c5, len(best_h)
