**Another approach: Frank-Wolfe (conditional gradient) method**

The Frank-Wolfe method works well for problems with linear constraints and convex objectives. Our problem is:

2. Solve: min_{y ∈ P} g^T y (the linear oracle)
3. Update: h_new = (1-γ)h + γ*y, where γ is the step size (full, half, or line search)

- The optimal y has y_i = 1 for the smallest g_i, y_i = 0 for the largest g_i
- One component is fractional to satisfy the sum constraint
- This is O(n log n) per iteration!

This is MUCH faster than SLSQP per iteration! And Frank-Wolfe is known to work well for non-smooth convex objectives (using subgradients).

            if free.sum() > 0:
                h[free] += diff2 / free.sum()

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = m + np.log(sw) / beta
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] * dx
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += wk * (1.0 - h[s:])
            grad[s:] += wk * (-h[:n-s])
        elif s < 0:
            t = -s
            grad[t:] += wk * (1.0 - h[:n-t])
            grad[:n-t] += wk * (-h[t:])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    return val, grad

def fw_step(grad, n, target):
    """Frank-Wolfe linear oracle: min grad^T y s.t. 0<=y<=1, sum(y)=target"""
    order = np.argsort(grad)
    y = np.zeros(n)
    k = int(round(target))
    y[order[:k]] = 1.0
    return y

    best = h.copy()

        val, grad = smoothmax_val_grad(h, beta)

            d = y - h
            # Line search: minimize f(h + gamma * d) for gamma in [0,1]
            # Use golden section or simple bisection

            best = h_new.copy()

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    n = len(h)
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-12) & (h < 1.0 - 1e-12)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode="full")
    corr = corr_raw * dx
    m = np.max(corr)
    shifted = np.clip(corr - m, -60.0, 0.0)
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    if sw < 1e-30:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w = exps / sw
    val = m + np.log(sw) / beta
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] * dx
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += wk * (1.0 - h[s:])
            grad[s:] += wk * (-h[:n-s])
        elif s < 0:
            t = -s
            grad[t:] += wk * (1.0 - h[:n-t])
            grad[:n-t] += wk * (-h[t:])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    return val, grad

def lbfgs_optimize(h_init, beta, max_iter, penalty):
    n = len(h_init)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    ones = np.ones(n)
    def obj(h):
        v, _ = smoothmax_val_grad(h, beta)
        s = h.sum() - target
        return v + 0.5 * penalty * s * s
    def grad(h):
        _, g = smoothmax_val_grad(h, beta)
        s = h.sum() - target
        return g + penalty * s * ones
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h_init.copy(), target)

def nesterov_subgrad(h_init, max_iters, top_k, beta_w, lr0=0.03, decay=0.99995):
    n = len(h_init)
    dx = 2.0 / n
    target = n / 2.0
    x = h_init.copy().astype(float)
    x_prev = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        if t == 0:
            y = x.copy()
        else:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
        y = project_feasible(y, target)
        corr = np.correlate(y, 1.0 - y, mode="full") * dx
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        top_vals = corr[top_idx]
        w = np.exp(beta_w * (top_vals - top_vals.max()))
        w = w / w.sum()
        grad = np.zeros(n)
        for i in range(len(top_idx)):
            k = top_idx[i]
            s = k - (n - 1)
            wk = w[i] * dx
            if s >= 0:
                grad[:n-s] += wk * (1.0 - y[s:])
                grad[s:] += wk * (-y[:n-s])
            elif s < 0:
                t2 = -s
                grad[t2:] += wk * (1.0 - y[:n-t2])
                grad[:n-t2] += wk * (-y[t2:])
            else:
                grad[:] += wk * (1.0 - 2.0 * y)
        x_new = project_feasible(y - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

        if rem() < 40:
            break
        scale = 0.008 * (1.0 + 0.4 * trial)
        pert = np.random.randn(n) * scale
        h_pert = project_feasible(h_best + pert, n / 2.0)
        h_new = lbfgs_optimize(h_pert, beta=400, max_iter=200, penalty=1000.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
            if rem() > 30:
                h_new2 = lbfgs_optimize(h_best, beta=1500, max_iter=200, penalty=5000.0)
                c2 = compute_c5(h_new2)
                if c2 < best_c5:
                    h_best, best_c5 = h_new2, c2
    # Phase 5: Alternative starting points
    if rem() > 150:
        xg = (np.arange(n) + 0.5) * (2.0 / n)
        alts = [
            np.clip(0.5 + 0.35 * np.sin(2 * np.pi * xg), 0, 1),
            np.clip(0.5 + 0.3 * np.sin(4 * np.pi * xg), 0, 1),
            np.clip(0.5 + 0.25 * np.cos(2 * np.pi * xg + 0.7), 0, 1),
            np.clip(0.5 + 0.3 * np.sin(2 * np.pi * xg) + 0.15 * np.sin(4 * np.pi * xg), 0, 1),
        ]
        for h_alt in alts:
            if rem() < 50:
                break
            h_alt = project_feasible(h_alt, n / 2.0)
            h_new = lbfgs_optimize(h_alt, beta=80, max_iter=300, penalty=500.0)
            h_new = lbfgs_optimize(h_new, beta=400, max_iter=200, penalty=2000.0)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
    # Phase 6: Final subgradient + L-BFGS squeeze
    if rem() > 60:
        h_sg2, c_sg2 = nesterov_subgrad(h_best, 6000, top_k=32, beta_w=15, lr0=0.008, decay=0.99999)
        if c_sg2 < best_c5:
            h_best, best_c5 = h_sg2, c_sg2
    if rem() > 35:
        h_new = lbfgs_optimize(h_best, 3000, max_iter=300, penalty=10000.0)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
    # Phase 7: Grid refinement
    if rem() > 180 and len(h_best) <= 500:
        n_try = 600
        h_interp = interp_to_n(h_best, n_try)
        for beta in [100, 400, 1000, 2000]:
            if rem() < 40:
                break
            h_new = lbfgs_optimize(h_interp, beta, max_iter=250, penalty=3000.0)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c
                h_interp = h_best.copy()
    # Final projection
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
