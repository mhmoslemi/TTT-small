def subgradient_topk(h, top_k=5):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for j in top_idx:
        lag = j - (n - 1)
        # First term: for p where p-lag in [0, n-1], i.e., p in [lag, lag+n-1] ∩ [0, n-1]
        p_lo = max(0, lag)
        p_hi = min(n - 1, lag + n - 1)
        q_lo = p_lo - lag
        q_hi = p_hi - lag
        grad[p_lo:p_hi+1] += one_minus_h[q_lo:q_hi+1]
        
        # Second term: for p where p+lag in [0, n-1], i.e., p in [-lag, n-1-lag] ∩ [0, n-1]
        p_lo2 = max(0, -lag)
        p_hi2 = min(n - 1, n - 1 - lag)
        if p_lo2 <= p_hi2:
            r_lo = p_lo2 + lag
            r_hi = p_hi2 + lag
            grad[p_lo2:p_hi2+1] -= h[r_lo:r_hi+1]
    
    return grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_topk(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    omh = 1.0 - h
    for j in top_idx:
        lag = j - (n - 1)
        # First term: grad[p] += omh[p-lag] for valid p
        p_lo = max(0, lag)
        p_hi = min(n - 1, lag + n - 1)
        if p_lo <= p_hi:
            grad[p_lo:p_hi+1] += omh[p_lo - lag : p_hi - lag + 1]
        # Second term: grad[p] -= h[p+lag] for valid p
        p_lo2 = max(0, -lag)
        p_hi2 = min(n - 1, n - 1 - lag)
        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2+1] -= h[p_lo2 + lag : p_hi2 + lag + 1]
    return grad

def gradient_descent(h, target, t_limit, top_k, step_init):
    n = len(h)
    h = h.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    step = step_init
    momentum = np.zeros(n)
    mu = 0.85
    t_start = time.time()
    stagnant = 0
    
    while time.time() - t_start < t_limit:
        grad = subgrad_topk(h, top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad = -grad / gn
        momentum = mu * momentum + (1 - mu) * grad
        
        alpha = step
        improved = False
        for _ in range(8):
            h_trial = project_h(h + alpha * momentum, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-13:
                h = h_trial
                cur_c5 = c5_trial
                improved = True
                break
            alpha *= 0.5
        
        if improved:
            if cur_c5 < best_c5:
                best_h = h.copy()
                best_c5 = cur_c5
            stagnant = 0
            step = min(step * 1.02, 0.15)
        else:
            stagnant += 1
            step *= 0.6
            momentum *= 0.5
            if stagnant > 100:
                h = project_h(h + 0.02 * np.random.randn(n), target)
                cur_c5 = compute_c5(h)
                stagnant = 0
                step = step_init
                momentum = np.zeros(n)
    
    return best_h, best_c5

def try_binary_round(h, target):
    n = len(h)
    best_h, best_c5 = h.copy(), compute_c5(h)
    for t in np.linspace(0.2, 0.8, 25):
        hb = (h >= t).astype(float)
        deficit = target - hb.sum()
        if deficit > 1.0:
            zeros = np.where(hb == 0.0)[0]

            if c5 < best_c5:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_topk(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    omh = 1.0 - h
    for j in top_idx:
        lag = j - (n - 1)
        p_lo = max(0, lag)
        p_hi = min(n - 1, lag + n - 1)
        if p_lo <= p_hi:
            grad[p_lo:p_hi + 1] += omh[p_lo - lag:p_hi - lag + 1]
        p_lo2 = max(0, -lag)
        p_hi2 = min(n - 1, n - 1 - lag)
        if p_lo2 <= p_hi2:
            grad[p_lo2:p_hi2 + 1] -= h[p_lo2 + lag:p_hi2 + lag + 1]
    return grad

def gradient_descent(h, target, t_limit, top_k, step_init, noise_scale=0.0):

                improved = True
                break

        if improved:
            if cur_c5 < best_c5:
                best_h = h.copy()
                best_c5 = cur_c5

            step = min(step * 1.03, 0.15)
