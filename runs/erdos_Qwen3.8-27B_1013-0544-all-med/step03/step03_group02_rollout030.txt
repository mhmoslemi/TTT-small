    # Sort by value, keep top n/2 as 1s
    threshold = np.sort(h)[::-1][n//2 - 1] if n % 2 == 0 else np.sort(h)[::-1][n//2]
    h_bin = (h >= threshold).astype(float)

        # Remove excess ones: pick the smallest h values among the 1s

        excess = n_ones - n // 2
        remove = one_indices[np.argsort(h[one_indices])[:excess]]
        h_bin[remove] = 0
    elif n_ones < n // 2:
        zero_indices = np.where(h_bin == 0)[0]
        need = n // 2 - n_ones
        add = zero_indices[np.argsort(h[zero_indices])[::-1][:need]]
        h_bin[add] = 1

    zero_idx = np.where(h == 0)[0]

            b = zero_idx[rng.integers(len(zero_idx))]

            # Try the swap
            h[a] = 0
            h[b] = 1
            new_max = float(np.max(compute_lags(h)))

            if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):

                    best_max = new_max
                    best_h = h.copy()
            else:
                # Revert
                h[a] = 1
                h[b] = 0
    
    return best_h, best_max

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=5000, ftol=1e-15):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h0

import numpy as np
from scipy.optimize import minimize, Bounds, linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2

    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

    return np.sum(x_in) - target

        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
