def fft_correlate(h, n):
    """Compute correlation of h with 1-h using FFT."""
    a = 1.0 - h
    size = 2 * n  # Next power of 2 for efficiency
    size = int(2 ** np.ceil(np.log2(2 * n)))
    fa = np.fft.rfft(h, size)
    fb = np.fft.rfft(a, size)
    corr = np.fft.irfft(fa * np.conj(fb), size)
    # Full correlation: first n elements are direct, rest are reversed
    # Actually, np.correlate(a, b, mode='full') where a has length n, b has length n
    # gives 2n-1 elements
    # The FFT gives us the circular correlation of size `size`
    # We need to extract the linear correlation
    # For correlate(h, 1-h, 'full'):
    #   result[k] = sum_i h[i] * (1-h)[i+k-n+1] for k=0..2n-2
    # Using FFT: if we zero-pad both to size >= 2n-1, the linear correlation is in the first 2n-1 elements
    corr = np.fft.irfft(fa * np.conj(fb), size)
    return corr[:2*n-1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_c5(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            la = -lag
            grad[la:] += (1.0 - h[:n - la])
            grad[:n - la] -= h[la:]
    return grad, corr

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_desc(cur, target, n, n_iter, step, t0, budget, top_k=8, decay=0.999):
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        g, _ = subgradient_c5(cur, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        step *= decay
        mom = 0.85 * mom - step * g
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
        else:
            stall += 1
            mom = 0.5 * mom
            trial = project_h(cur + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_local - 1e-15:
                cur = trial
                best_local = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 300:
            break
    return cur, best_local

def coord_descent(cur, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        base = max(0.002, 0.04 * (0.65 ** sw))
        deltas = (base, -base, base*0.5, -base*0.5)
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                if 0 <= cur[j] + d <= 1:
                    trial = cur.copy()
                    trial[j] += d
                    trial = project_h(trial, target)
                    c_try = compute_c5(trial)
                    if c_try < best_c5 - 1e-15:
                        cur = trial
                        best_c5 = c_try
                        improved = True
                        break
        if not improved and sw > 3:
            break
    return cur, best_c5

from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_c5(h, top_k):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            la = -lag
            grad[la:] += (1.0 - h[:n - la])
            grad[:n - la] -= h[la:]
    return grad, corr

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sum_eq(x_in, target):
    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0.0, 2.0, n_new), np.linspace(0.0, 2.0, len(h)), h)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_desc(cur, target, n, n_iter, step, t0, budget, top_k=12, decay=0.9995):
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        g, _ = subgradient_c5(cur, top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        step *= decay
        if step < 1e-6:
            step = 1e-6
        mom = 0.85 * mom - step * g
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
        else:
            stall += 1
            mom = 0.5 * mom
            trial = project_h(cur + mom, target)
            c_try = compute_c5(trial)
            if c_try < best_local - 1e-15:
                cur = trial
                best_local = c_try
                stall = 0
            else:
                mom = np.zeros(n)
        if stall > 400:
            break
    return cur, best_local

def fine_coord_descent(cur, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(cur)
    for sw in range(sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        base = max(0.002, 0.03 * (0.7 ** sw))
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in (base, -base):
                if 0 <= cur[j] + d <= 1:
                    trial = cur.copy()
                    trial[j] += d
                    trial = project_h(trial, target)
                    c_try = compute_c5(trial)
                    if c_try < best_c5 - 1e-15:
                        cur = trial
                        best_c5 = c_try
                        improved = True
                        break
        if not improved and sw > 4:
            break
    return cur, best_c5

def anneal_coord(cur, target, n, t0, budget):
    """Simulated-annealing style coordinate descent."""
    best_c5 = compute_c5(cur)
    cur_c5 = best_c5
    step = 0.08
    for it in range(5000):
        if budget - (time.time() - t0) < 3:
            break
        step *= 0.998
        if step < 0.002:
            step = 0.002
        j = int(np.random.randint(n))
        d = np.random.uniform(-step, step)
        if 0 <= cur[j] + d <= 1:
            trial = cur.copy()
            trial[j] += d
            trial = project_h(trial, target)
            c_try = compute_c5(trial)
            delta = c_try - cur_c5
            if delta < -1e-15:
                cur = trial
                cur_c5 = c_try
                if c_try < best_c5 - 1e-15:
                    best_c5 = c_try
            elif np.random.random() < np.exp(-delta / (step * 0.1)):
                cur = trial
                cur_c5 = c_try
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(140, 0.5)

    n_base = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = n_base

    def optimize_at(h0, n_pts, phase):
        nonlocal best_h, best_c5, best_n
        target = n_pts / 2.0
        bounds = Bounds(np.zeros(n_pts), np.ones(n_pts))
        cur = project_h(h0, target)
        c = compute_c5(cur)
        improved = c < best_c5
        if improved:
            best_h, best_c5, best_n = cur.copy(), c, n_pts

        betas = [30, 100, 300, 1000, 3000] if phase == 'main' else [200, 800, 2000]
        maxit = 4000 if phase == 'main' else 2000
        for beta in betas:
            remaining = budget_s - (time.time() - t0)
            if remaining < 12:
                break
            cur = do_slsqp(cur, n_pts, target, beta, bounds, maxit, 1e-14)
            c = compute_c5(cur)
            if c < best_c5:
                best_h, best_c5, best_n = cur.copy(), c, n_pts

        remaining = budget_s - (time.time() - t0)
        if remaining > 20:
            nk = 16 if phase == 'main' else 12
            cur, c = subgrad_desc(cur, target, n_pts, 10000, 0.015, t0, budget_s, top_k=nk)
            if c < best_c5:
                best_h, best_c5, best_n = cur.copy(), c, n_pts

        remaining = budget_s - (time.time() - t0)
        if remaining > 12:
            cur, c = fine_coord_descent(cur, target, n_pts, 60, t0, budget_s)
            if c < best_c5:
                best_h, best_c5, best_n = cur.copy(), c, n_pts
        return cur

    # Phase 1: Main optimization at base resolution
    cur = optimize_at(best_h, n_base, 'main')

    # Phase 2: Multi-restart with structured perturbations
    remaining = budget_s - (time.time() - t0)
    n_restarts = min(8, max(3, int(remaining / 30)))
    for ri in range(n_restarts):
        remaining = budget_s - (time.time() - t0)
        if remaining < 25:
            break
        if ri == 0:
            hp = project_h(best_h + np.random.randn(n_base) * 0.025, n_base / 2.0)
        elif ri == 1:
            f = np.random.uniform(2, 12)
            ph = np.random.uniform(0, 6.28)
            pert = 0.025 * np.sin(2 * np.pi * f * np.arange(n_base) / n_base + ph)
            hp = project_h(best_h + pert, n_base / 2.0)
        elif ri == 2:
            f = np.random.uniform(1, 6)
            x = np.arange(n_base) / n_base
            pert = 0.02 * (2 * np.abs(2 * (x * f - np.floor(x * f + 0.5))) - 1)
            hp = project_h(best_h + pert, n_base / 2.0)
        else:
            np.random.seed(seed + ri * 1000)
            blk = max(4, n_base // 6)
            s = np.random.randint(0, n_base - blk)
            hp = best_h.copy()
            hp[s:s+blk] += np.random.uniform(-0.06, 0.06, size=blk)
            hp = project_h(hp, n_base / 2.0)

        hp = do_slsqp(hp, n_base, n_base/2.0, 500, Bounds(np.zeros(n_base), np.ones(n_base)), 1500, 1e-13)
        hp, c = subgrad_desc(hp, n_base/2.0, n_base, 4000, 0.01, t0, budget_s, top_k=16)
        hp, c = fine_coord_descent(hp, n_base/2.0, n_base, 30, t0, budget_s)
        if c < best_c5:
            best_h, best_c5 = hp.copy(), c

    # Phase 3: Try alternative resolutions
    remaining = budget_s - (time.time() - t0)
    for n_new in [256, 400, 480, 512, 640]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 40:
            break
        if n_new == n_base:
            continue
        cur2 = project_h(resample(best_h, n_new), n_new / 2.0)
        cur2 = optimize_at(cur2, n_new, 'sub')

    # Phase 4: Annealing at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        nf = len(best_h)
        ah, ac = anneal_coord(best_h, nf / 2.0, nf, t0, budget_s)
        if ac < best_c5:
            best_h, best_c5, best_n = ah.copy(), ac, nf

    # Phase 5: Final ultra-precise polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        nf = len(best_h)
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        for beta in [5000, 12000, 25000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 4:
                break
            best_h = do_slsqp(best_h, nf, tf, beta, bf, 8000, 1e-16)
            best_c5 = compute_c5(best_h)

    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        nf = len(best_h)
        tf = nf / 2.0
        best_h, best_c5 = subgrad_desc(best_h, tf, nf, 12000, 0.003, t0, budget_s, top_k=24)

    remaining = budget_s - (time.time() - t0)
    if remaining > 3:
        nf = len(best_h)
        tf = nf / 2.0
        best_h, best_c5 = fine_coord_descent(best_h, tf, nf, 120, t0, budget_s)

    # Ensure feasibility
    nf = len(best_h)
    best_h = project_h(best_h, nf / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, nf
