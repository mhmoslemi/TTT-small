                        best_state[1] = cur.copy()
                        best_state[2] = n
        v *= 0.95
    return cur, cur_c5

def coordinate_descent(h, n, target, dx, max_sweeps, deadline, best_state):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        for j in range(n):
            old_val = cur[j]
            best_trial = old_val
            best_c5_j = compute_c5(cur)
            for candidate in [0.0, 1.0, 0.5, old_val + 0.05, old_val - 0.05]:
                candidate = np.clip(candidate, 0.0, 1.0)
                if candidate == old_val:
                    continue
                cur[j] = candidate
                h_adj = project_h(cur, target)
                if np.abs(h_adj[j] - candidate) > 0.3:
                    cur[j] = old_val
                    continue
                c5_j = compute_c5(h_adj)
                if c5_j < best_c5_j - 1e-14:
                    best_c5_j = c5_j
                    best_trial = h_adj
            if best_trial != old_val:
                cur = best_trial
                cur_c5 = best_c5_j
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
            else:
                cur[j] = old_val
    return cur, cur_c5

def binary_improve(h, n, target, dx, deadline, best_state):
    h_bin = (h > 0.5).astype(float)
    h_bin = project_h(h_bin, target)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_state[0]:
        best_state[0] = c5_bin
        best_state[1] = h_bin.copy()
        best_state[2] = n
    h_bin, _ = nesterov_descent(h_bin, n, target, dx, 500, deadline, best_state)
    return h_bin

def slsqp_optimize(h, n, target, dx, beta, deadline, best_state):
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        return smooth_max_obj(x, dx, beta)
    cons = [{'type': 'eq', 'fun': lambda x: sum_constraint(x, target),
             'jac': lambda x: sum_constraint_jac(x)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 400, 'ftol': 1e-13})
        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)
        if c5_new < best_state[0]:
            best_state[0] = c5_new
            best_state[1] = h_new.copy()
            best_state[2] = n
        return h_new, c5_new
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)

    best_state = [compute_c5(h_parent), h_parent.copy(), n_parent]

    n_list = sorted(set([n_parent, 250, 300, 350]))
    n_list = [n for n in n_list if 100 <= n <= 500]

    for n in n_list:
        if time.time() > deadline - 100:
            break
        target = n / 2.0
        dx = 2.0 / n
        x = np.linspace(0, 2, n)

        inits = [project_h(resample(h_parent, n), target)]
        inits.append(project_h(resample(best_state[1], n), target))
        for mode in [1, 2, 3]:
            h_f = 0.5 + 0.28 * np.cos(mode * np.pi * x) + 0.08 * np.sin(2 * mode * np.pi * x)
            inits.append(project_h(h_f, target))
        h_r = np.clip(0.5 + 0.3 * np.random.randn(n), 0, 1)
        inits.append(project_h(h_r, target))
        h_b = (np.random.random(n) > 0.5).astype(float)
        inits.append(project_h(h_b, target))

        for h_init in inits:
            if time.time() > deadline - 50:
                break
            cur = project_h(h_init.copy(), target)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_state[0]:
                best_state[0] = cur_c5
                best_state[1] = cur.copy()
                best_state[2] = n

            for beta in [100, 300]:
                if time.time() > deadline - 30:
                    break
                cur, cur_c5 = slsqp_optimize(cur, n, target, dx, beta, deadline, best_state)

            cur, cur_c5 = nesterov_descent(cur, n, target, dx, 2500, deadline, best_state)

            if time.time() > deadline - 30:
                break
            cur, cur_c5 = coordinate_descent(cur, n, target, dx, 3, deadline, best_state)

            if time.time() > deadline - 20:
                break
            cur = binary_improve(cur, n, target, dx, deadline, best_state)

    if time.time() < deadline - 15:
        best_h = best_state[1].copy()
        best_c5 = best_state[0]
        n_f = len(best_h)
        target_f = n_f / 2.0
        dx_f = 2.0 / n_f

        for beta in [500, 1200]:
            if time.time() > deadline - 5:
                break
            best_h, best_c5 = slsqp_optimize(best_h, n_f, target_f, dx_f, beta, deadline, best_state)

        for it in range(8000):
            if time.time() > deadline - 3:
                break
            grad = weighted_gradient(best_h, n_f, dx_f, 3, 250.0)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            grad /= gnorm
            step = 0.002
            improved = False
            for _ in range(15):
                h_trial = project_h(best_h - step * grad, target_f)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-16:
                    best_h = h_trial
                    best_c5 = c5_trial
                    improved = True
                    break
                step *= 0.5
            if not improved and it % 50 == 49:
                T = 0.0008 * np.exp(-it / 3000.0)
                if T > 1e-7:
                    h_trial = project_h(best_h + np.random.normal(0, T, n_f), target_f)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5:
                        best_h = h_trial
                        best_c5 = c5_trial

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
