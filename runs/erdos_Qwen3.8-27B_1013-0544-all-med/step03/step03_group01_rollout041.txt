from scipy.signal import fftconvolve
import time

def compute_c(h, n):

    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(np.max(compute_c(h, n)))

def project(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(20000):
        diff = target - h.sum()
        if abs(diff) < 1e-16 * n:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-16: break
            h += diff / tot * room
        else:
            tot = h.sum()
            if tot < 1e-16: break
            h += diff / tot * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_value(c, alpha):
    m = c.max()
    return m + np.log(np.sum(np.exp(alpha * (c - m)))) / alpha

def get_weights(c, alpha):
    m = c.max()
    w = np.exp(alpha * (c - m))
    w /= w.sum()
    return w

def compute_grad(h, n, w):
    dx = 2.0 / n
    conv1 = fftconvolve(1 - h, w, mode='full')
    corr2 = np.correlate(h, w, mode='full')
    offset = n - 1
    return dx * (conv1[offset:offset+n] - corr2[offset:offset+n])

def pgd_optimize(h, n, alpha, lr, max_iters, deadline):
    """Projected gradient descent"""
    h = h.copy()
    c = compute_c(h, n)
    f = smooth_max_value(c, alpha)
    w = get_weights(c, alpha)
    best_h = h.copy()
    best_f = f
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad = compute_grad(h, n, w)
        # Backtracking line search
        for trial_lr in [lr, lr*0.5, lr*0.25, lr*0.1, lr*0.05]:
            h_new = h - trial_lr * grad
            h_new = project(h_new, n)
            c_new = compute_c(h_new, n)
            f_new = smooth_max_value(c_new, alpha)
            if f_new < f:
                h = h_new
                c = c_new
                f = f_new
                w = get_weights(c, alpha)
                if f < best_f:
                    best_f = f
                    best_h = h.copy()
                break
        else:

            break
    
    return best_h, best_f

def coordinate_descent(h, n, deadline, max_rounds=5):
    """Try pairwise coordinate moves to reduce C5"""
    h = h.copy()
    best_c = compute_c5(h, n)
    dx = 2.0 / n
    
    for round_num in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i+1, n):
                # Try to transfer mass from j to i or vice versa
                for delta in [0.01, 0.005, 0.001]:
                    for s in [1, -1]:
                        hi = h[i] + s * delta
                        hj = h[j] - s * delta
                        if 0 <= hi <= 1 and 0 <= hj <= 1:
                            h[i] = hi
                            h[j] = hj
                            c = compute_c5(h, n)
                            if c < best_c:

                                improved = True

                            h[i], h[j] = hi - s * d, hj + s * d
    return h, bc

def multi_phase_opt(h0, n, deadline, rng):
    best_h = project(h0, n)
    best_c = compute_c5(best_h, n)

    alphas = [50, 200, 1000, 5000, 20000, 100000, 500000]
    for alpha in alphas:
        if time.time() > deadline - 15:
            break
        lr = n / (4.0 * alpha + 1.0)
        mu = min(0.85, 1.0 - 2.0 / (alpha + 10))
        h2, c2 = pgd_momentum(best_h, n, alpha, lr, mu, 3000, deadline - 5)
        if c2 < best_c:
            best_c, best_h = c2, h2

    if time.time() < deadline - 25:
        for alpha in [10000, 100000, 1000000]:
            if time.time() > deadline - 10:
                break
            h2 = slsqp_refine(best_h, n, alpha, 5000, deadline)
            c2 = compute_c5(h2, n)
            if c2 < best_c - 1e-14:
                best_c, best_h = c2, h2

    if time.time() < deadline - 10:
        best_h, best_c = pair_search(best_h, n, deadline, [0.01, 0.005, 0.002, 0.001])

    return project(best_h, n), compute_c5(best_h, n)

def make_starts(n, rng):
    x = np.linspace(0, 2, n)
    starts = []
    starts.append(np.full(n, 0.5))
    for k in range(1, 10):
        starts.append(0.5 + 0.3 * np.sin(k * np.pi * x) + 0.1 * np.cos(2 * k * np.pi * x))
    for amp in [0.2, 0.3, 0.4, 0.5]:
        starts.append(0.5 + amp * np.sin(np.pi * x) + 0.1 * np.sin(3 * np.pi * x))
    for period in range(2, 9):
        base = 0.5 + 0.4 * np.sin(2 * np.pi * x / (4.0 / period))
        starts.append(base)
    for _ in range(4):
        starts.append(rng.uniform(0, 1, n))
    return starts

def optimize_n(h0, n, deadline, rng):
    best_h = project(h0, n)
    best_c = compute_c5(best_h, n)

    h_try, c_try = multi_phase_opt(best_h, n, deadline, rng)
    if c_try < best_c:
        best_c, best_h = c_try, h_try

    time_left = deadline - time.time()
    if time_left > 30:
        starts = make_starts(n, rng)
        n_starts = len(starts)
        per_start = max(5, (int(time_left) - 10) // n_starts)
        for s in starts[:8]:
            if time.time() > deadline - 10:
                break
            s = project(s, n)
            c_s = compute_c5(s, n)
            if c_s < best_c:
                best_c, best_h = c_s, s.copy()
            sub_dl = min(time.time() + per_start * 2, deadline - 5)
            h2, c2 = multi_phase_opt(s, n, sub_dl, rng)
            if c2 < best_c:
                best_c, best_h = c2, h2

    return project(best_h, n), compute_c5(best_h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 3
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        n_init = 100
        x = np.linspace(0, 2, n_init)
        h_init = project(0.5 + 0.3 * np.sin(np.pi * x), n_init)

    best_h_overall = h_init.copy()
    best_c_overall = compute_c5(best_h_overall, n_init)

    total_time = deadline - t0
    alloc_main = total_time * 0.50
    h2, c2 = optimize_n(h_init, n_init, t0 + alloc_main, rng)
    if c2 < best_c_overall:
        best_c_overall, best_h_overall = c2, h2

    remaining = deadline - time.time()
    if remaining > 40:
        n_tries = [n_init - 20, n_init + 20, n_init - 40, n_init + 40, 50, 80]
        n_tries = [x for x in n_tries if x >= 30 and x != n_init]
        seen = set()
        n_tries = [x for x in n_tries if x not in seen and not seen.add(x)]
        for n_try in n_tries[:5]:
            if time.time() > deadline - 25:
                break
            x = np.linspace(0, 2, n_try)
            h_try = project(0.5 + 0.3 * np.sin(np.pi * x), n_try)
            alloc = min(remaining * 0.2, 80)
            h3, c3 = optimize_n(h_try, n_try, time.time() + alloc, rng)
            if c3 < best_c_overall:
                best_c_overall, best_h_overall = c3, h3
            remaining = deadline - time.time()
            if remaining < 25:
                break

    if time.time() < deadline - 5:
        n_f = len(best_h_overall)
        best_h_overall, best_c_overall = pair_search(
            best_h_overall, n_f, min(time.time() + 15, deadline),
            [0.005, 0.002, 0.001, 0.0005])

    best_h_overall = project(best_h_overall, len(best_h_overall))
    best_c_overall = compute_c5(best_h_overall, len(best_h_overall))
    return best_h_overall, best_c_overall, len(best_h_overall)
