def project_feasible(h, target_sum):

def weighted_corr_grad(h, w):

    n = len(h)
    dx = 2.0 / n

            grad[t:] += wk * (-h[:n-t])
        else:

    grad *= dx

def smoothmax_c5(h, beta):
    """Compute smoothmax approximation of C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = weighted_corr_grad(h, w)
    return val, grad, w

def multiplicative_weights_optimize(h, max_iters, eta, lr_schedule):
    """Minimize max_k corr[k]*dx using multiplicative weights on the lags."""

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def weighted_corr_grad(h, w):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n-s])
            grad[:n-s] += wk * (-h[s:])
        elif s < 0:
            t = -s
            grad[:n-t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n-t])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return grad

def smoothmax_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -50, 50))
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = weighted_corr_grad(h, w)
    return val, grad, w

def adaptive_minmax(h, max_iters, beta0, beta1, lr0, decay, momentum=0.9):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    v = np.zeros(n)
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        frac = t / max(1, max_iters - 1)
        beta = beta0 + (beta1 - beta0) * frac
        lr = lr0 * (decay ** t)
        val, grad, w = smoothmax_obj_grad(x, beta)
        v = momentum * v + (1 - momentum) * grad
        x_new = project_feasible(x - lr * v, target)
        cv = compute_c5(x_new)
        if cv < best_v:
            best_v = cv
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

def topk_minmax(h, max_iters, top_k, lr0, decay):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        corr = get_corr(x)
        idx = np.argsort(corr)[-top_k:]
        w = np.zeros(len(corr))
        cmax = corr[idx].max()
        w[idx] = np.exp(100.0 * (corr[idx] - cmax))
        sw = w.sum()
        if sw < 1e-30:
            w = np.zeros(len(corr)); w[idx[0]] = 1.0
        else:
            w /= sw
        grad = weighted_corr_grad(x, w)
        x_new = project_feasible(x - lr * grad, target)
        cv = compute_c5(x_new)
        if cv < best_v:
            best_v = cv
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    def f(x):
        return smoothmax_obj_grad(x, beta)[0]
    def g(x):
        return smoothmax_obj_grad(x, beta)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def binary_project(h):
    n = len(h)
    target = n / 2.0
    h_bin = (h > 0.5).astype(float)
    diff = target - h_bin.sum()
    if diff > 0:
        fracs = np.sort(h[np.where(h_bin == 0)[0]])[::-1][:int(diff)]
        h_bin[np.where(h_bin == 0)[0][:int(diff)]] = 1.0
    elif diff < 0:
        fracs = np.sort(h[np.where(h_bin == 1)[0]])[:int(-diff)]
        h_bin[np.where(h_bin == 1)[0][:int(-diff)]] = 0.0
    return project_feasible(h_bin, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase 1: SLSQP progressive beta
    for beta in [80, 200, 500, 1000]:
        if tl() < 25:
            break
        h_new = slsqp_refine(h_best, maxiter=300, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: Adaptive minmax with momentum
    if tl() > 120:
        h2, c2 = adaptive_minmax(h_best, 8000, beta0=100, beta1=2000,
                                  lr0=0.05, decay=0.99995, momentum=0.85)
        if c2 < best_c5:
            h_best, best_c5 = h2, c2

    # Phase 3: Top-k focused optimization
    if tl() > 100:
        for tk in [5, 10, 20]:
            if tl() < 30:
                break
            h3, c3 = topk_minmax(h_best, 5000, top_k=tk, lr0=0.03, decay=0.99997)
            if c3 < best_c5:
                h_best, best_c5 = h3, c2 if c3 < c2 else c3
                h_best, best_c5 = h3, c3

    # Phase 4: SLSQP high-beta polish
    for beta in [800, 1500, 3000]:
        if tl() < 30:
            break
        h_new = slsqp_refine(h_best, maxiter=400, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 5: Binary projection and re-optimization
    if tl() > 150:
        for ntry in [n, 200, 300]:
            if tl() < 60:
                break
            h_proj = interp_to_n(h_best, ntry)
            h_bin = binary_project(h_proj)
            h_new = slsqp_refine(h_bin, maxiter=300, beta=200)
            h_new = adaptive_minmax(h_new, 4000, beta0=200, beta1=1500,
                                      lr0=0.04, decay=0.99996, momentum=0.85)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 6: Multiple perturbations
    for trial in range(10):
        if tl() < 50:
            break
        scale = 0.008 * (1.0 + 0.3 * trial)
        pert = np.random.randn(len(h_best)) * scale
        h_pert = project_feasible(h_best + pert, len(h_best) / 2.0)
        h_new = adaptive_minmax(h_pert, 3000, beta0=200, beta1=2000,
                                  lr0=0.04, decay=0.99995, momentum=0.85)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c
        if tl() < 30:
            break

    # Phase 7: Higher resolution attempt
    for ntry in [600, 800]:
        if tl() < 200:
            break
        h_interp = interp_to_n(h_best, ntry)
        h_interp = adaptive_minmax(h_interp, 4000, beta0=100, beta1=1000,
                                     lr0=0.04, decay=0.99997, momentum=0.85)
        h_interp = slsqp_refine(h_interp, maxiter=200, beta=500)
        c = compute_c5(h_interp)
        if c < best_c5:
            h_best, best_c5 = h_interp, c

    # Phase 8: Final refinement
    if tl() > 40:
        h_final = adaptive_minmax(h_best, 5000, beta0=500, beta1=5000,
                                    lr0=0.02, decay=0.99998, momentum=0.9)
        c = compute_c5(h_final)
        if c < best_c5:
            h_best, best_c5 = h_final, c
        if tl() > 20:
            h_final2 = slsqp_refine(h_best, maxiter=300, beta=3000)
            c2 = compute_c5(h_final2)
            if c2 < best_c5:
                h_best, best_c5 = h_final2, c2

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
