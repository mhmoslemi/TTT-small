import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h = np.clip(h + diff / len(h), 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(len(corr))
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        wk = w_sel[i]
        if wk < 1e-14:
            continue
        s = k - n + 1
        i_lo = max(0, s)
        i_hi = min(n, n + s)
        if i_hi > i_lo:
            grad[i_lo:i_hi] += wk * (1.0 - h[i_lo - s:i_hi - s])
        i_lo2 = max(0, -s)
        i_hi2 = min(n, n - s)
        if i_hi2 > i_lo2:
            grad[i_lo2:i_hi2] += wk * (-h[i_lo2 + s:i_hi2 + s])
    grad *= dx
    return val, grad


def slsqp_solve(h0, beta, maxiter, top_k=None):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]

    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq',
             'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h0, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0, target)


def lbfgsb_penalty(h0, beta, maxiter, penalty, top_k=None):
    n = len(h0)
    target = n / 2.0

    def obj(x):
        v, g = smoothmax_val_grad(x, beta, top_k)
        return v + penalty * (x.sum() - target) ** 2

    def objg(x):
        v, g = smoothmax_val_grad(x, beta, top_k)
        return g + 2.0 * penalty * (x.sum() - target)

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=objg, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)


def active_set_cycling(h, beta_list, n_cycles, top_k, t0, budget_s, seed):
    n = len(h)

            h = slsqp_solve(h, beta, 300, top_k=tk)
            h = project_feasible(h, target)
    return h


def basin_hop_diverse(h_best, best_c5, n_hops, beta_list, maxiter,

        elif use_structured and hop % 3 == 2:

def targeted_perturbation(h, rng, n_trials=5):
    """Try targeted perturbations on the worst lag."""
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best = h.copy()
    best_c = compute_c5(h)
    for trial in range(n_trials):
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst_idx = np.argmax(corr)
        s = worst_idx - n + 1
        # Perturb to reduce this lag: decrease h where h[j+s]*(1-h[j]) is large
        perturb = np.zeros(n)
        for j in range(n):
            if 0 <= j + s < n:
                perturb[j] += h[j + s] * (1 - h[j])
            if 0 <= j - s < n:
                perturb[j] -= h[j - s] * (1 - h[j])
        step = 0.02 * (1 + 0.5 * rng.rand())
        h_new = h + step * perturb
        h_new = project_feasible(h_new, target)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best = h_new.copy()
            h = h_new
    return best, best_c
