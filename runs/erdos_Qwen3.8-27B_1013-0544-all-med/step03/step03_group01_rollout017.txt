import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return c

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        # Distribute among non-bound elements
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_gradient(h, n, alpha):
    """Compute gradient of smooth max objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    max_c = c.max()
    exp_c = np.exp(alpha * (c - max_c))
    p = exp_c / exp_c.sum()
    
    # grad[j] = dx * sum_d p[d+n-1] * [(1-h[j+d])*1[valid] - h[j-d]*1[valid]]
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0
        for d in range(-(n-1), n):
            pd = p[d + n - 1]
            if pd < 1e-15:
                continue
            if 0 <= j + d < n:
                s1 += pd * (1.0 - h[j+d])
            if 0 <= j - d < n:
                s1 -= pd * h[j-d]
        grad[j] = dx * s1
    return grad

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    val = m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return float(val)

def projected_gradient_descend(h, n, deadline, alpha_start=100, alpha_end=100000, n_alphas=8):
    """Projected gradient descent with adaptive alpha."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    
    alphas = np.geomspace(alpha_start, alpha_end, n_alphas)
    
    for alpha in alphas:
        if time.time() > deadline:
            break
        h_cur = best_h.copy()
        c_cur = compute_c5(h_cur, n)
        
        # Gradient descent with backtracking
        step = 1.0 / (alpha * np.max(np.abs(compute_smooth_gradient(h_cur, n, alpha))) + 1e-10)
        
        for it in range(200):
            if time.time() > deadline:
                break
            grad = compute_smooth_gradient(h_cur, n, alpha)
            if np.max(np.abs(grad)) < 1e-12:
                break
            
            # Try step with backtracking
            for trial in range(20):
                h_new = project_feasible(h_cur - step * grad, n)
                obj_new = smooth_obj(h_new, n, alpha)
                obj_cur = smooth_obj(h_cur, n, alpha)
                if obj_new < obj_cur - 1e-15:
                    h_cur = h_new
                    break
                step *= 0.5
            
            # Update best
            c_check = compute_c5(h_cur, n)
            if c_check < best_c - 1e-14:
                best_c = c_check
                best_h = h_cur.copy()
    
    return best_h, best_c

def slsqp_optimize(h, n, alpha, maxiter=2000):
    def obj(x):
        return smooth_obj(x, n, alpha)
    
    def jac(x):
        return compute_smooth_gradient(x, n, alpha)
    
    # Add a small penalty for the sum constraint (since we project anyway)
    def obj_pen(x):
        return smooth_obj(x, n, alpha) + 1e6 * (np.sum(x) - n/2.0)**2
    
    def jac_pen(x):
        g = compute_smooth_gradient(x, n, alpha)
        g += 2e6 * (np.sum(x) - n/2.0)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_pen, h, jac=jac_pen, method='L-BFGS-B',
                      bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16})
        h_new = project_feasible(res.x, n)
        return h_new
    except Exception:
        return h

        return np.ones(n)

                        else:

                            h[j] += s * d

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_all_c(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_c(h, n)
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for d in range(-(n-1), n):
            pd = p[d + n - 1]
            if pd < 1e-14:
                continue
            if 0 <= j + d < n:
                s += pd * (1.0 - h[j + d])
            if 0 <= j - d < n:
                s -= pd * h[j - d]
        grad[j] = dx * s
    return grad

    def obj(x):

        return h

                        if c >= best_c:

                        else:

Let me also add one more idea: **coordinate descent with full correlation update**. At each step, for each coordinate i, try to优化 h[i] by finding the value that minimizes the max correlation. This is a 1D problem for each i.

For a fixed h[j] (j≠i), the correlation c_k as a function of h[i] is:
c_k(t) = Σ_{j≠i} h[j]*(1-h[j+d_k]) * dx + (terms involving h[i]) * dx

The terms involving h[i] in c_k are:
- From h[i]*(1-h[i+d_k]): contributes t*(1-h[i+d_k]) if 0≤i+d_k<n
- From h[i-d_k]*(1-h[i]): contributes h[i-d_k]*(1-t) if 0≤i-d_k<n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def compute_all_c(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h += (diff / s) * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    c = compute_all_c(h, n)
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def smooth_max_grad(h, n, alpha):
    dx = 2.0 / n
    c = compute_all_c(h, n)
    m = c.max()
    exp_c = np.exp(alpha * (c - m))
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        for d in range(-(n - 1), n):
            pd = p[d + n - 1]
            if pd < 1e-14:
                continue
            if 0 <= j + d < n:
                s += pd * (1.0 - h[j + d])
            if 0 <= j - d < n:
                s -= pd * h[j - d]
        grad[j] = dx * s
    return grad

def lbfgsb_opt(h, n, alpha, maxiter=400):
    def obj(x):
        return smooth_max_obj(x, n, alpha) + 1e4 * (x.sum() - n / 2.0) ** 2
    def jac(x):
        g = smooth_max_grad(x, n, alpha)
        g += 2e4 * (x.sum() - n / 2.0)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=jac, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_opt(h, n, alpha, maxiter=2000):
    def sfun(x):
        return x.sum() - n / 2.0
    def sjac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sfun, 'jac': sjac}]
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, n)
    except Exception:
        return h

def subgradient_step(h, n, step, deadline):
    """Direct subgradient method on the max correlation."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    for _ in range(500):
        if time.time() > deadline:
            break
        c = compute_all_c(h, n)
        k_star = int(np.argmax(c))
        d = k_star - (n - 1)
        dx = 2.0 / n
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            if 0 <= j + d < n:
                g += (1.0 - h[j + d])
            if 0 <= j - d < n:
                g -= h[j - d]
            grad[j] = dx * g
        h_new = project_feasible(h - step * grad, n)
        c_new = compute_c5(h_new, n)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        step *= 0.98
    return best_h, best_c

def pair_local(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                for s in (1.0, -1.0):
                    hi, hj = h[i] + s * d, h[j] - s * d
                    if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                        h[i], h[j] = hi, hj
                        c = compute_c5(h, n)
                        if c >= best_c:
                            h[i], h[j] = h[i] - s * d, h[j] + s * d
                        else:
                            best_c = c
    return h, best_c

def targeted_pair(h, n, deadline):
    best_c = compute_c5(h, n)
    deltas = [0.005, 0.002, 0.001, 0.0005, 0.0002]
    for d in deltas:
        for _ in range(3):
            if time.time() > deadline:
                break
            c = compute_all_c(h, n)
            k_star = int(np.argmax(c))
            d_lag = k_star - (n - 1)
            for i in range(n):
                if time.time() > deadline:
                    break
                j = i + d_lag
                if 0 <= j < n and j != i:
                    for s in (1.0, -1.0):
                        hi, hj = h[i] + s * d, h[j] - s * d
                        if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                            h[i], h[j] = hi, hj
                            c_new = compute_c5(h, n)
                            if c_new < best_c:
                                best_c = c_new
                            else:
                                h[i], h[j] = h[i] - s * d, h[j] + s * d
    return h, best_c

def make_starts(n, rng, h_init=None):

    starts = []
    if h_init is not None and len(h_init) == n:

    for _ in range(10):

    for period in [2, 3, 4, 5, 6, 7, 8]:
        half = period // 2
        pat = np.array([1.0 if (i % period) < half else 0.0 for i in range(n)])
        starts.append(project_feasible(pat, n))
    return starts

def optimize_for_n(h_start, n, deadline, rng):
    h = project_feasible(h_start.copy(), n)
    best_h = h.copy()
    best_c = compute_c5(h, n)

    def rem():
        return deadline - time.time()

    # Phase 1: Subgradient for quick improvement
    if rem() > 20:

        if ct < best_c:
            best_c, best_h = ct, ht

    # Phase 2: L-BFGS-B with increasing alpha
    for alpha in [100, 1000, 10000, 50000]:
