import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        h += diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-10:
            free = (h > 1e-6) & (h < 1.0 - 1e-6)
            if free.sum() > 0:
                h[free] += diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = corr - m
    shifted = np.clip(shifted, -50.0 / beta, 0.0)
    exps = np.exp(beta * shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s > 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] += wk * (-h[s:])
        elif s < 0:
            t = -s
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n - t])
        else:
            grad[:] += wk * (1.0 - 2.0 * h)
    grad *= dx
    return val, grad

def slsqp_refine(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    one_vec = np.ones(n)
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def jac(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: one_vec}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def proj_grad_descent(h, max_iters, beta, lr0, lr_decay):
    n = len(h)
    target = n / 2.0
    x = h.copy().astype(float)
    v_prev = compute_c5(x)
    best_x = x.copy()
    best_v = v_prev
    for t in range(max_iters):
        lr = lr0 * (lr_decay ** t)
        v, grad = smoothmax_val_grad(x, beta)
        x_new = project_feasible(x - lr * grad, target)
        v_new = compute_c5(x_new)
        if v_new < best_v:
            best_v = v_new
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def gradient_based_restart(h, n_restarts, scales, beta, maxiter, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for i in range(n_restarts):
        scale = scales[i % len(scales)]
        # Use gradient direction mixed with random
        _, grad = smoothmax_val_grad(best_h, beta)
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-14:
            gdir = grad / gnorm
        else:
            gdir = rng.randn(n)
            gdir /= np.linalg.norm(gdir)
        noise = rng.randn(n) * scale
        pert = 0.5 * scale * gdir + noise
        h_pert = project_feasible(best_h + pert, n / 2.0)
        h_new = slsqp_refine(h_pert, maxiter, beta)
        c = compute_c5(h_new)
        if c < best_c5:
            best_h, best_c5 = h_new, c
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)
    n = len(h_best)

    # Phase A: Progressive SLSQP polish with increasing beta
    for beta in [200, 500, 1000, 2000, 4000]:
        if tl() < 40:
            break
        h_new = slsqp_refine(h_best, maxiter=500, beta=beta)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase B: Gradient-based restarts (small scale)

    # Phase C: Medium-scale restarts

        h_best, best_c5 = gradient_based_restart(

            scales=[0.03, 0.05, 0.04, 0.06, 0.02],
            beta=800, maxiter=350, rng=rng)

    # Phase D: Projected gradient with high beta (annealing)

            # Follow up with SLSQP

            if c < best_c5:
