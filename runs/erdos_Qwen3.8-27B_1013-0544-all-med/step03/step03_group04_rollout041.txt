def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            # Try to adjust the closest to 0.5
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0, 1)
            continue
        per = diff / free.sum()
        h[free] += per
        h = np.clip(h, 0.0, 1.0)
    return h

def find_optimal_coord(h, i, n):
    """Find optimal value of h[i] given all other coordinates fixed."""
    # Slope of C[k] w.r.t. h[i]
    # C[k] = sum_j h[j]*(1-h[j+k])
    # dC[k]/dh[i] = (1-h[i+k])*[0<=i+k<n] - h[i-k]*[0<=i-k<n]
    slope_pos = np.zeros(2*n-1)  # indexed by k+(n-1)
    for k in range(-(n-1), n):
        idx = k + n - 1
        s = 0.0
        if 0 <= i+k < n:
            s += (1.0 - h[i+k])
        if 0 <= i-k < n:
            s -= h[i-k]
        slope_pos[idx] = s

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = free.sum()
        if cnt == 0:
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            continue
        h[free] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def coord_optimize(h, i, n):
    """Find optimal h[i] via bisection on 1D convex subproblem."""
    # Compute slope of C[k] w.r.t. h[i]
    si = np.zeros(2*n-1)
    # Vectorized: for each k, si[k+(n-1)] = (1-h[i+k])*[0<=i+k<n] - h[i-k]*[0<=i-k<n]
    # i+k ranges from i-(n-1) to i+(n-1); valid when 0 <= i+k < n
    lo_k = max(0, -i)  # minimum k such that i+k >= 0
    hi_k = min(n-1-i, n-1)  # maximum k such that i+k < n
    # For k in [lo_k, hi_k]: si[k+n-1] += (1-h[i+k])
    if lo_k <= hi_k:
        ks = np.arange(lo_k, hi_k+1)
        si[ks + n - 1] += (1.0 - h[i + ks])
    # i-k ranges from i-(n-1) to i+(n-1); valid when 0 <= i-k < n
    lo_k2 = max(0, i-(n-1))  # minimum k such that i-k < n, i.e., k > i-n
    hi_k2 = min(i, n-1)  # maximum k such that i-k >= 0
    if lo_k2 <= hi_k2:
        ks2 = np.arange(lo_k2, hi_k2+1)
        si[ks2 + n - 1] -= h[i - ks2]
    
    corr = np.correlate(h, 1.0 - h, mode="full")
    intercepts = corr - si * h[i]
    
    # Bisection
    lo, hi = 0.0, 1.0
    for _ in range(64):
        mid = (lo + hi) * 0.5
        vals = si * mid + intercepts
        mv = vals.max()
        active = vals >= mv - 1e-12
        mn = si[active].min()
        mx = si[active].max()
        if mn > 1e-9:
            hi = mid
        elif mx < -1e-9:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

    for sweep in range(n_sweeps):
        if time.time() - t0 > time_limit:
            break
        changed = False
        for i in range(n):

            if abs(x_opt - h[i]) > 1e-9:
                h[i] = x_opt
                h = project_h(h, target)
                changed = True
        if not changed:
            break
    return h

    si = np.zeros(2*n-1)
    # Positive contribution: (1-h[i+k]) for valid i+k
    a, b = max(0, -i), min(n-1-i, n-1)
    if a <= b:
        ks = np.arange(a, b+1)
        si[ks+n-1] += 1.0 - h[i+ks]
    # Negative contribution: -h[i-k] for valid i-k
    a2, b2 = max(0, i-(n-1)), min(i, n-1)
    if a2 <= b2:
        ks2 = np.arange(a2, b2+1)
        si[ks2+n-1] -= h[i-ks2]
    corr = np.correlate(h, 1.0-h, mode="full")
    ic = corr - si * h[i]
    lo, hi = 0.0, 1.0
    for _ in range(64):
        mid = (lo+hi)*0.5
        v = si*mid + ic
        mv = v.max()
        act = v >= mv - 1e-12
        if si[act].min() > 1e-9:
            hi = mid
        elif si[act].max() < -1e-9:
            lo = mid
        else:
            break
    return (lo+hi)*0.5

    for _ in range(n_sweeps):

            x = coord_opt(h, i, n)
            if abs(x-h[i]) > 1e-10:
                h[i] = x
                h = project_h(h, target)
    return h
