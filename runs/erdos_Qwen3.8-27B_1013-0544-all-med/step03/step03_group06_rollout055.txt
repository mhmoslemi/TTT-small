```python
def smooth_max_grad(h, beta):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    w = np.exp(x)
    w /= np.sum(w)
    
    grad = np.zeros(N)
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (N - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:N-lag])
            grad[:N-lag] -= wk * h[lag:]
        else:
            d = -lag
            grad[:N-d] += wk * (1.0 - h[d:])
            grad[d:] -= wk * h[:N-d]
    return grad * dx
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

            break
        try:
            bnds = Bounds(np.zeros(n_t), np.ones(n_t))
            def obj_fn(x):
                c = np.correlate(x, 1.0 - x, mode="full")
                mm = np.max(c)
                xx = np.clip(3000.0 * (c - mm), -500, 500)
                return (mm + np.log(np.sum(np.exp(xx))) / 3000.0) * (2.0 / n_t)
            def jac_fn(x):
                return smooth_max_grad(x, 3000)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt,
                     'jac': lambda x: np.ones(x.shape)}]
            res = minimize(obj_fn, h_cur, method='SLSQP', bounds=bnds,
                           constraints=cons, jac=jac_fn,
                           options={'maxiter': 2000, 'ftol': 1e-16})
            h_new = project_h(res.x, tgt)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                h_cur = h_new
                cur_c5 = c5_new
            if cur_c5 < best_c5:
                best_h = h_cur.copy()
                best_c5 = cur_c5
        except Exception:
            pass

    # Final phase: subgradient + pairwise at best resolution
    n_f = len(best_h)
    tgt = n_f / 2.0

    for it in range(15000):
        if tleft() < 5:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_n = min(8, max(4, n_f // 30))
        top_idx = np.argsort(corr)[::-1][:top_n]
        grad = np.zeros(n_f)
        for idx in top_idx:
            lag = int(idx) - (n_f - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * best_h)
            elif 0 < lag < n_f:
                grad[lag:] += (1.0 - best_h[:n_f - lag])
                grad[:n_f - lag] -= best_h[lag:]
            elif -n_f < lag < 0:
                d = -lag
                grad[:n_f - d] += (1.0 - best_h[d:])
                grad[d:] -= best_h[:n_f - d]
        gn = np.linalg.norm(grad)
        if gn < 1e-13:
            break
        grad /= gn
        improved = False
        for step in [0.05, 0.025, 0.012, 0.006, 0.003, 0.001]:
            h_t = project_h(best_h - step * grad, tgt)
            c5_t = compute_c5(h_t)
            if c5_t < best_c5:
                best_h = h_t
                best_c5 = c5_t
                improved = True
                break
        if not improved and it % 1500 == 1499:
            h_t = project_h(best_h + np.random.randn(n_f) * 0.006, tgt)
            if compute_c5(h_t) < best_c5:
                best_h = h_t
                best_c5 = compute_c5(h_t)

    # Pairwise coordinate moves on critical lags
    for it in range(50000):
        if tleft() < 3:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        top_idx = np.argsort(corr)[::-1][:4]
        imp = False
        for idx in top_idx:
            if tleft() < 1:
                break
            lag = int(idx) - (n_f - 1)
            if lag == 0:
                continue
            al = abs(lag)
            if al >= n_f:
                continue
            nv = n_f - al
            for _ in range(10):
                i = np.random.randint(nv)
                j = i + al if lag > 0 else i - lag
                for d in [0.035, -0.035, 0.018, -0.018, 0.009, -0.009, 0.004, -0.004]:
                    h_t = best_h.copy()
                    h_t[i] += d
                    h_t[j] -= d
                    if 0 <= h_t[i] <= 1 and 0 <= h_t[j] <= 1:
                        if abs(h_t.sum() - tgt) > 1e-10:
                            h_t = project_h(h_t, tgt)
                        c5_t = compute_c5(h_t)
                        if c5_t < best_c5:
                            best_h = h_t
                            best_c5 = c5_t
                            imp = True
                            break
                if imp:
                    break
            if imp:
                break
        if not imp and it % 3000 == 2999:
            h_t = project_h(best_h + np.random.randn(n_f) * 0.004, tgt)
            if compute_c5(h_t) < best_c5:
                best_h = h_t
                best_c5 = compute_c5(h_t)

    # Final ultra-fine SLSQP polish
    if tleft() > 10:
        try:
            bnds = Bounds(np.zeros(n_f), np.ones(n_f))
            big_beta = 8000
            def obj_f(x):
                c = np.correlate(x, 1.0 - x, mode="full")
                mm = np.max(c)
                xx = np.clip(big_beta * (c - mm), -500, 500)
                return (mm + np.log(np.sum(np.exp(xx))) / big_beta) * (2.0 / n_f)
            def jac_f(x):
                return smooth_max_grad(x, big_beta)
            cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - tgt,
                     'jac': lambda x: np.ones(x.shape)}]
            res = minimize(obj_f, best_h, method='SLSQP', bounds=bnds,
                           constraints=cons, jac=jac_f,
                           options={'maxiter': 5000, 'ftol': 1e-17})
            h_new = project_h(res.x, tgt)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except Exception:
            pass

    return best_h, best_c5, len(best_h)
```
