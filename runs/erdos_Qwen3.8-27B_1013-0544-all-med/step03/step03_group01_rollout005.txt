            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.005, 0.002, 0.001, 0.0005])
    else:
        if time.time() < deadline - 2:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.005, 0.002, 0.001], radius=30)

    if time.time() < deadline - 2:
        ht = slsqp_optimize(best_h, n, 5000000, maxiter=4000, ftol=1e-17)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht

    if n <= 150 and time.time() < deadline - 1:
        best_h, best_c = pair_search(best_h, n, deadline - 0.5, [0.002, 0.001, 0.0005, 0.0002])

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 100

    best_h = None
    best_c = float('inf')

    if h_init is not None:
        alloc1 = (deadline - t0) * 0.45
        h_opt, c_opt = optimize_n(h_init, n_init, t0 + alloc1, rng)
        best_h, best_c = h_opt, c_opt

        x_init = np.linspace(0, 2, n_init)
        for n_try in [80, 120, 150, 200]:
            if n_try == n_init or time.time() > deadline - 30:
                continue
            remaining = deadline - time.time()
            alloc = min(70, remaining * 0.15)
            if alloc < 15:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_n(h_interp, n_try, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 25:
        n = len(best_h)
        remaining = deadline - time.time() - 15
        n_restarts = max(3, int(remaining / 35))
        for r in range(n_restarts):
            if time.time() > deadline - 12:
                break
            scale = 0.02 + 0.12 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(45, (deadline - time.time() - 12) * 0.3)
            if alloc < 10:
                break
            h_opt, c_opt = optimize_n(hp, n, time.time() + alloc, rng)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt

    if best_h is not None and time.time() < deadline - 3:
        n = len(best_h)
        if n <= 150:
            best_h, best_c = pair_search(best_h, n, deadline - 1,
                                         [0.002, 0.001, 0.0005, 0.0002, 0.0001])
        else:
            best_h, best_c = near_pair_search(best_h, n, deadline - 1,
                                              [0.002, 0.001, 0.0005], radius=20)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)

    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)

    return best_h, best_c, len(best_h)
