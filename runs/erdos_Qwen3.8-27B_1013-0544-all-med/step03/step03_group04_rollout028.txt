= (sum_k w_k) * sum_j h[j] - sum_j sum_k w_k * h[j]*h[j+k]

The first term is linear in h. The second term is quadratic. So this is a QP!

We can solve this with scipy.optimize.minimize using L-BFGS-B (which handles bounds), or with a dedicated QP solver.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not m.any():
            m = np.ones(len(h), dtype=bool)
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_w(corr, beta):
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -400, 400)
    e = np.exp(x)
    return e / e.sum()

def grad_smooth(h, w, n):
    g = np.zeros(n)
    W0 = w[n-1]
    if abs(W0) > 1e-14:
        g += W0 * (1.0 - 2.0 * h)
    for l in range(1, n):
        Wp = w[n-1+l]
        Wm = w[n-1-l]
        nl = n - l
        if abs(Wp) > 1e-14:
            g[l:] += Wp * (1.0 - h[:nl])
            g[:nl] -= Wp * h[l:]
        if abs(Wm) > 1e-14:
            g[:nl] += Wm * (1.0 - h[l:])
            g[l:] -= Wm * h[:nl]
    return g

def pg_optimize(h_init, target, beta_start, beta_end, max_iters, step0, t0, tlimit):
    """Projected gradient with momentum and adaptive beta"""
    n = len(h_init)
    h = project_h(h_init.copy(), target)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros(n)
    mu = 0.85  # momentum coefficient
    stuck_count = 0
    
    for it in range(max_iters):
        if time.time() - t0 > tlimit:
            break
        
        # Adaptive beta: linear interpolation from beta_start to beta_end
        frac = min(it / max(max_iters - 1, 1), 1.0)
        beta = beta_start + (beta_end - beta_start) * frac
        
        corr = np.correlate(h, 1.0 - h, mode="full")
        w = smooth_max_w(corr, beta)
        g = grad_smooth(h, w, n)
        
        # Step size with decay
        step = step0 / (1.0 + 0.05 * it)
        
        # Heavy ball momentum
        momentum = mu * momentum - step * g
        h_new = project_h(h + momentum, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stuck_count = 0
            if cur_c5 < best_c5:
                best_h = h.copy()
                best_c5 = cur_c5
        else:
            # Halve step
            step *= 0.25
            momentum = mu * 0.5 * momentum - step * g
            h_new = project_h(h + momentum, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = cur_c5
            else:
                stuck_count += 1
                momentum *= 0.5
                if stuck_count > 50:
                    # Random perturbation
                    idx = np.random.choice(n, size=max(n//10, 1), replace=False)
                    h[idx] += np.random.uniform(-0.3, 0.3, size=len(idx))
                    h = project_h(h, target)
                    cur_c5 = compute_c5(h)
                    stuck_count = 0
    
    return best_h, best_c5

def binary_local_search(h, target, t0, tlimit, max_flips=5000):
    """Try flipping individual bits to improve"""
    n = len(h)
    h_b = (h >= 0.5).astype(float)
    deficit = target - h_b.sum()
    if deficit > 0:
        zeros = np.where(h_b == 0)[0]
        order = zeros[np.argsort(h[zeros])[::-1]]
        h_b[order[:int(np.ceil(deficit))]] = 1.0
    elif deficit < 0:
        ones = np.where(h_b == 1)[0]
        order = ones[np.argsort(h[ones])]
        h_b[order[:int(np.ceil(-deficit))]] = 0.0
    h_b = project_h(h_b, target)
    cur_c5 = compute_c5(h_b)
    best_h = h_b.copy()
    best_c5 = cur_c5
    
    for _ in range(max_flips):
        if time.time() - t0 > tlimit:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        if h_b[i] == h_b[j]:
            continue
        # Swap: set h_b[i] = 1, h_b[j] = 0 (or vice versa)
        h_b[i], h_b[j] = h_b[j], h_b[i]
        c5 = compute_c5(h_b)
        if c5 < best_c5 - 1e-14:
            best_h = h_b.copy()
            best_c5 = c5
        else:
            h_b[i], h_b[j] = h_b[j], h_b[i]  # revert
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target):
    h = np.clip(h, 0.0, 1.0).astype(float)
    for _ in range(80):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        if not m.any():
            m = np.ones(len(h), dtype=bool)
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_w(corr, beta):
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -400, 400)
    e = np.exp(x)
    s = e.sum()
    return e / s

def grad_smooth(h, w, n):
    g = np.zeros(n)
    W0 = w[n - 1]
    if abs(W0) > 1e-14:
        g += W0 * (1.0 - 2.0 * h)
    for l in range(1, n):
        Wp = w[n - 1 + l]
        Wm = w[n - 1 - l]
        nl = n - l
        if abs(Wp) > 1e-14:
            g[l:] += Wp * (1.0 - h[:nl])
            g[:nl] -= Wp * h[l:]
        if abs(Wm) > 1e-14:
            g[:nl] += Wm * (1.0 - h[l:])
            g[l:] -= Wm * h[:nl]
    return g

def pg_optimize(h_init, target, beta_start, beta_end, max_iters, step0, t0, tlimit):
    n = len(h_init)
    h = project_h(h_init.copy(), target)
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    momentum = np.zeros(n)
    mu = 0.82
    stuck = 0
    for it in range(max_iters):
        if time.time() - t0 > tlimit:
            break
        frac = min(it / max(max_iters - 1, 1), 1.0)
        beta = beta_start + (beta_end - beta_start) * frac
        corr = np.correlate(h, 1.0 - h, mode="full")
        w = smooth_max_w(corr, beta)
        g = grad_smooth(h, w, n)
        step = step0 / (1.0 + 0.04 * it)
        momentum = mu * momentum - step * g
        h_new = project_h(h + momentum, target)
        c5_new = compute_c5(h_new)
        if c5_new < cur_c5 - 1e-14:
            h = h_new
            cur_c5 = c5_new
            stuck = 0
            if cur_c5 < best_c5:
                best_h, best_c5 = h.copy(), cur_c5
        else:
            momentum = 0.5 * mu * momentum - 0.2 * step * g
            h_new = project_h(h + momentum, target)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5 - 1e-14:
                h = h_new
                cur_c5 = c5_new
                if cur_c5 < best_c5:
                    best_h, best_c5 = h.copy(), cur_c5
            else:
                stuck += 1
                momentum *= 0.4
                if stuck > 40:
                    idx = np.random.choice(n, size=max(n // 8, 1), replace=False)
                    h[idx] += np.random.uniform(-0.25, 0.25, size=len(idx))
                    h = project_h(h, target)
                    cur_c5 = compute_c5(h)
                    stuck = 0
    return best_h, best_c5

def lbfgs_optimize(h_init, target, beta, t0, tlimit, maxiter=400):
    n = len(h_init)
    dx = 2.0 / n
    lam = 5.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        w = smooth_max_w(corr, beta)
        return float(np.dot(w, corr) * dx + lam * (x.sum() - target) ** 2)
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        w = smooth_max_w(corr, beta)
        g = grad_smooth(x, w, n) * dx
        g += 2.0 * lam * (x.sum() - target)
        return g
    bounds = [(0, 1)] * n
    try:
        res = minimize(obj, h_init, method='L-BFGS-B', jac=grad, bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_init, compute_c5(h_init)

def binary_search(h, target, t0, tlimit, max_flips=3000):
    n = len(h)
    hb = (h >= 0.5).astype(float)
    deficit = target - hb.sum()
    if deficit > 0:
        zeros = np.where(hb == 0)[0]
        order = zeros[np.argsort(h[zeros])[::-1]]
        hb[order[:int(np.ceil(deficit))]] = 1.0
    elif deficit < 0:
        ones = np.where(hb == 1)[0]
        order = ones[np.argsort(h[ones])]
        hb[order[:int(np.ceil(-deficit))]] = 0.0
    hb = project_h(hb, target)
    cur_c5 = compute_c5(hb)
    best_h, best_c5 = hb.copy(), cur_c5
    flips = 0
    while flips < max_flips and time.time() - t0 < tlimit:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or hb[i] == hb[j]:
            flips += 1
            continue
        hb[i], hb[j] = hb[j], hb[i]
        c5 = compute_c5(hb)
        if c5 < best_c5 - 1e-14:
            best_h, best_c5 = hb.copy(), c5
        else:
            hb[i], hb[j] = hb[j], hb[i]
        flips += 1
    return best_h, best_c5

    if remaining > 25:
        n_b = len(best_h)
        hb, cb = binary_search(best_h, n_b / 2.0, t0, min(t0 + remaining * 0.2, budget_s - 30), max_flips=2500)
        if cb < best_c5:
            best_h, best_c5 = hb, cb
            # Polish binary solution
            remaining = budget_s - (time.time() - t0)
            if remaining > 15:
                hp, cp = pg_optimize(best_h, n_b / 2.0, 500, 8000, 2000, 0.04, t0, t0 + remaining)
                if cp < best_c5:
                    best_h, best_c5 = hp, cp

    # Phase 5: Final fine-tuning
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        n_f = len(best_h)
        hf, cf = pg_optimize(best_h, n_f / 2.0, 2000, 15000, 2000, 0.02, t0, t0 + remaining)
        if cf < best_c5:
            best_h, best_c5 = hf, cf

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, int(len(best_h))
```
