Let me also think about the correlation indexing more carefully.

`np.correlate(h, v, mode='full')[k]` where h and v both have length n:
Output has length 2n-1.
out[k] = Σ_{i} h[i] * v[i + (n-1) - k]

Let me define lag L = k - (n-1). So L ranges from -(n-1) to (n-1).
For L ≥ 0: out[n-1+L] = Σ_{i=0}^{n-1-L} h[i] * v[i+L]
For L < 0: out[n-1+L] = Σ_{i=-L}^{n-1} h[i] * v[i+L] = Σ_{j=0}^{n-1+L} h[j-L] * v[j]

In our case, v = 1-h, so:
For L ≥ 0: c[L] = Σ_{i=0}^{n-1-L} h[i] * (1-h[i+L])
For L < 0: c[L] = Σ_{j=0}^{n-1+L} h[j-L] * (1-h[j]) = Σ_{j=0}^{n-1+L} h[j-L] * h[j-1+1] ... 

Actually, by symmetry: c[-L] = c[L] if h is symmetric in a certain way. Let's not worry about that and just use L ≥ 0.

For L ≥ 0: c(L) = Σ_{i=0}^{n-1-L} h[i](1-h[i+L])

∂c(L)/∂h[j]:
- If j is in the "first" position (i=j): contributes (1-h[j+L]) if j ≤ n-1-L
- If j is in the "second" position (i+L=j, so i=j-L): contributes -h[j-L] if j ≥ L and j-L ≤ n-1-L, i.e., j ≤ n-1

So: ∂c(L)/∂h[j] = (1-h[j+L])·1{j ≤ n-1-L} - h[j-L]·1{j ≥ L}

For the coordinate descent on pair (j1, j2) with sum s:
h[j1] = t, h[j2] = s - t

Δc(L) = (t - h[j1]) · ∂c(L)/∂h[j1] + ((s-t) - h[j2]) · ∂c(L)/∂h[j2]
       = (t - h[j1]) · (∂c(L)/∂h[j1] - ∂c(L)/∂h[j2])  [since (s-t)-h[j2] = h[j1]-t]

So I just need d[L] = ∂c(L)/∂h[j1] - ∂c(L)/∂h[j2] for L = 0, ..., n-1.

Then new_c(L) = c(L) + (t - h[j1]) * d[L]

And I want to minimize max_L new_c(L) * dx over t ∈ [max(0,s-1), min(1,s)].

Since new_c(L) is linear in t, max_L new_c(L) is convex in t (max of linear functions). I can minimize it using ternary search or by finding the breakpoints.

For efficiency, I'll use a golden-section search (since it's convex and 1D).

Actually, even better: since it's piecewise linear and convex, the minimum occurs either at a breakpoint (where two lines cross) or at a boundary. The breakpoints are where new_c(L1)(t) = new_c(L2)(t) for some L1, L2, which gives:
c(L1) + (t-h1)*d[L1] = c(L2) + (t-h1)*d[L2]
(t-h1)*(d[L1]-d[L2]) = c(L2)-c(L1)
t = h1 + (c(L2)-c(L1))/(d[L1]-d[L2])

There are O(n²) breakpoints, but I can just evaluate at a fine grid (say 200 points) and then refine with a few bisection steps.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def compute_grad_diff(h, j1, j2, n):
    """Compute d[L] = dc(L)/dh[j1] - dc(L)/dh[j2] for all L."""
    L = np.arange(n)
    # dc(L)/dh[j] = (1-h[j+L]) * 1{j+L < n} - h[j-L] * 1{j >= L}
    # For j = j1:
    g1 = np.zeros(n)
    valid1_plus = j1 + L < n
    g1[valid1_plus] += (1.0 - h[j1 + L[valid1_plus]])
    valid1_minus = L <= j1
    g1[valid1_minus] -= h[j1 - L[valid1_minus]]
    
    g2 = np.zeros(n)
    valid2_plus = j2 + L < n
    g2[valid2_plus] += (1.0 - h[j2 + L[valid2_plus]])
    valid2_minus = L <= j2
    g2[valid2_minus] -= h[j2 - L[valid2_minus]]
    
    return g1 - g2

def pair_optimize(h, j1, j2, n, n_grid=200):
    """Optimize the pair (j1, j2) to minimize max correlation."""
    s = h[j1] + h[j2]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    
    if t_hi - t_lo < 1e-12:
        return h.copy(), False
    
    c = np.correlate(h, 1.0 - h, mode="full")
    # c is indexed from 0 to 2n-2, corresponding to lag -(n-1) to (n-1)
    # We only need L >= 0, which is c[n-1 : 2n-1]
    c_pos = c[n-1:]  # length n, c_pos[L] = correlation at lag L
    
    d = compute_grad_diff(h, j1, j2, n)
    h1 = h[j1]
    
    # new_c(L) = c_pos[L] + (t - h1) * d[L]
    # Minimize max_L new_c(L) over t ∈ [t_lo, t_hi]
    
    # Evaluate on grid
    t_grid = np.linspace(t_lo, t_hi, n_grid)
    # new_c matrix: shape (n_grid, n)
    # new_c[i, L] = c_pos[L] + (t_grid[i] - h1) * d[L]
    deltas = (t_grid - h1)[:, None] * d[None, :]
    new_c = c_pos[None, :] + deltas
    f_vals = np.max(new_c, axis=1)
    
    # Find best on grid
    best_idx = np.argmin(f_vals)
    t_best = t_grid[best_idx]
    
    # Refine with golden section in a small neighborhood
    if 0 < best_idx < n_grid - 1:
        lo = t_grid[best_idx - 1]
        hi = t_grid[best_idx + 1]
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(30):
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            c1 = np.max(c_pos + (t1 - h1) * d)
            c2 = np.max(c_pos + (t2 - h1) * d)
            if c1 < c2:
                hi = t2
            else:
                lo = t1
        t_best = (lo + hi) / 2
    
    h[j1] = t_best
    h[j2] = s - t_best
    return h, True

    best_idx = np.argmin(f_vals)

    # Golden section refinement
    if 0 < best_idx < n_grid - 1:
        lo = t_grid[best_idx - 1]
        hi = t_grid[best_idx + 1]
        gr = (np.sqrt(5) - 1) / 2
        for _ in range(40):
            t1 = hi - gr * (hi - lo)
            t2 = lo + gr * (hi - lo)
            f1 = np.max(c_pos + (t1 - h1) * d)
            f2 = np.max(c_pos + (t2 - h1) * d)
            if f1 < f2:
                hi = t2
            else:
                lo = t1
        t_best = (lo + hi) / 2
    
    return t_best

def _slsqp_obj(x):
    c = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(c)
    cs = np.clip(_G_beta * (c - m), -500, 500)
    return (m + np.log(np.sum(np.exp(cs)) / len(c)) / _G_beta) * _G_dx

def _slsqp_con_jac(x):
