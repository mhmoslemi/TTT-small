        n_active = active.sum()
        
        if n_active == 0:
            break
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            if not active[k]:
                continue
            wk = 1.0 / n_active
            s = k - (n - 1)
            js = max(0, -s)
            je = min(n - 1, n - 1 - s)
            if js <= je:
                grad[js:je+1] += wk * (1.0 - h[js+s:je+s+1])
            js2 = max(0, s)
            je2 = min(n - 1, n - 1 + s)
            if js2 <= je2:
                grad[js2:je2+1] -= wk * h[js2-s:je2-s+1]
        
        grad *= dx
        
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new, n)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.01, 2.0)
        else:
            lr *= 0.95
        
        if lr < 0.01:
            break
    
    return best_h, best_c
