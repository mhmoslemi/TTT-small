import numpy as np
from scipy.optimize import minimize, Bounds
from functools import partial
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h)
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] += -h[lag:]
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:])
        grad[l:] += -h[:n - l]
    return grad * dx

def obj_smooth_max(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * dx

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def subgradient_descent(h, n, target, dx, max_iters, min_time_fn, best_state, top_k_schedule=None):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    
    center = n - 1
    step = 0.05
    for it in range(max_iters):
        if min_time_fn() < 5:
            break
        
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        
        # Adaptive top-k
        if top_k_schedule is not None:
            k = top_k_schedule(it, max_iters)
        else:
            k = min(2 + it // 100, 5)
        
        top_idx = np.argpartition(corr_val, -k)[-k:]
        top_idx = top_idx[np.argsort(corr_val[top_idx])[::-1]]
        vals = corr_val[top_idx]
        
        # Use softmax weighting with temperature
        temp = 100.0 + 200.0 * (it / max(1, max_iters))
        weights = np.exp(-(vals[0] - vals) * temp)
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_for_lag(cur, lag, n, dx)
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm
        
        # Armijo backtracking
        improved = False
        for _ in range(30):
            h_trial = project_h(cur - step * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-10:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5
        
        if not improved:
            # SA escape
            T = 0.003 * np.exp(-it / 500.0)
            if T > 1e-5:
                noise = np.random.normal(0, T, n)
                h_trial = project_h(cur + noise, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 + T * 5:
                    cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_state[0]:
                        best_state[0] = cur_c5
                        best_state[1] = cur.copy()
                        best_state[2] = n
            step = 0.05
    return cur, cur_c5

def coordinate_polish(h, n, target, dx, max_sweeps, min_time_fn, best_state):
    """Cycle through coordinates and optimize each one."""
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    
    for sweep in range(max_sweeps):
        if min_time_fn() < 5:
            break
        for i in range(n):
            if min_time_fn() < 10 and i < n//2:
                break
            # Try different values of h[i]
            old_val = cur[i]
            candidates = np.concatenate([
                [0.0, 1.0],
                np.linspace(0, 1, 20),
                np.array([max(0, old_val + d) for d in [-0.1, -0.05, -0.02, 0.02, 0.05, 0.1]])
            ])
            candidates = np.clip(candidates, 0, 1)
            candidates = np.unique(candidates)
            
            best_i_val = old_val
            best_i_c5 = cur_c5
            
            for c in candidates:
                if abs(c - old_val) < 1e-6:
                    continue
                cur[i] = c
                # Adjust for sum constraint approximately
                # Actually, let's just compute C5 for this single change
                corr = np.correlate(cur, 1.0 - cur, mode="full")
                c5_val = float(np.max(corr) * dx)
                if c5_val < best_i_c5 - 1e-10:
                    best_i_c5 = c5_val
                    best_i_val = c
            cur[i] = best_i_val
            cur_c5 = best_i_c5
            
            # Re-project to maintain sum constraint
            cur = project_h(cur, target)
            cur_c5 = compute_c5(cur)
        
        if cur_c5 < best_state[0]:
            best_state[0] = cur_c5
            best_state[1] = cur.copy()
            best_state[2] = n
    
    return cur, cur_c5

        return budget_s - (time.time() - start)

The overlap at lag k is: ∫ h(x)(1-h(x+k)) dx = ∫h(x)dx - ∫h(x)h(x+k)dx = 1 - ∫h(x)h(x+k)dx

So minimizing the max overlap is equivalent to maximizing the min of ∫h(x)h(x+k)dx over k≠0!

Wait, that's not quite right. The overlap is:
∫₀^{2-k} h(x)(1-h(x+k)) dx for k > 0

But in the discrete version with the "full" correlate, we're computing:
corr[m] = sum_i h[i] * (1-h[i+m-(n-1)])

For m = n-1 (lag 0): sum_i h[i](1-h[i]) = sum_i h[i] - sum_i h[i]^2 = n/2 - sum_i h[i]^2
For m ≠ n-1: this is the cross-correlation of h with itself at the appropriate lag, minus the sum of h.

Actually let me be more careful. np.correlate(h, 1-h, mode="full") computes:
C[m] = sum_i h[i] * (1-h)[i + m - (n-1)] for valid i

Let me denote v = 1-h. Then C[m] = sum_i h[i] * v[i+m-(n-1)]

For m = n-1: C[n-1] = sum_i h[i] * v[i] = sum_i h[i](1-h[i]) = n/2 - sum_i h[i]^2

For m = n-1+k: C[n-1+k] = sum_i h[i] * v[i+k] = sum_i h[i](1-h[i+k])
= sum_i h[i] - sum_i h[i]h[i+k]
= n/2 - (h ⋆ h)[k]  (where ⋆ is cross-correlation)

So C[n-1+k] = n/2 - (h ⋆ h)[k] for k ≠ 0 (with appropriate boundary handling)

Therefore, minimizing max_k C[k]*dx is equivalent to minimizing max_k (n/2 - (h⋆h)[k])*dx, which means we want to MAXIMIZE the minimum of (h⋆h)[k] over k≠0!

This is a different perspective: we want h to have its autocorrelation as "flat" and as large as possible at all non-zero lags.

This is related to Barker sequences and low-autocorrelation sequences! The dual problem is to find a sequence with maximum minimum autocorrelation.

This is a useful reframing. It suggests that:
- At the optimum, the autocorrelation should be as flat as possible
- The "peak" lags of the overlap are the "trough" lags of the autocorrelation
- We should focus on maximizing the minimum autocorrelation

This doesn't change the algorithm much (we're still minimizing the max overlap), but it gives insight into what structure to look for.

For a sequence with flat autocorrelation, good candidates include:
- m-sequences (maximum length LFSR sequences) - but these are ±1 valued
- Barker sequences
- Cayley graph sequences

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_for_lag(h, lag, n, dx):
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h)
    elif lag > 0:
        grad[lag:] += (1.0 - h[:n - lag])
        grad[:n - lag] -= h[lag:]
    else:
        l = -lag
        grad[:n - l] += (1.0 - h[l:])
        grad[l:] -= h[:n - l]
    return grad * dx

def obj_smooth_max(h, dx, beta):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * dx

def make_con_fun(target):
    def con(x):
        return np.sum(x) - target
    return con

def make_con_jac(n):
    def jac(x):
        return np.ones(n)
    return jac

def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return np.interp(x_new, x_old, h_old)

def subgradient_descent(h, n, target, dx, max_iters, min_time_fn, best_state):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    center = n - 1
    step = 0.04
    stall = 0
    for it in range(max_iters):
        if min_time_fn() < 5:
            break
        corr_raw = np.correlate(cur, 1.0 - cur, mode="full")
        corr_val = corr_raw * dx
        # Adaptive k: start small, grow
        k = min(1 + it // 50, 5)
        top_idx = np.argpartition(corr_val, -k)[-k:]
        vals = corr_val[top_idx]
        # Temperature: high when we want focus, lower when we want spread
        temp = 80.0 + 300.0 * (it / max(1, max_iters))
        weights = np.exp(-(vals[0] - vals) * temp)
        weights /= weights.sum()
        grad = np.zeros(n)
        for t_idx, w in zip(top_idx, weights):
            lag = int(t_idx) - center
            grad += w * grad_for_lag(cur, lag, n, dx)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-13:
            break
        grad /= gnorm
        improved = False
        s = step
        for _ in range(30):
            h_trial = project_h(cur - s * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-11:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                stall = 0
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            s *= 0.5
        if not improved:
            stall += 1

                            best_state[1] = cur.copy()
                            best_state[2] = n
                    stall = 0
                else:
                    break
            step = 0.04
    return cur, cur_c5

def coordinate_polish(h, n, target, dx, max_sweeps, min_time_fn, best_state):

    for sweep in range(max_sweeps):

        if tleft() < 70:
            break
        target = n / 2.0
        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        x = np.linspace(0, 2, n)

        inits = []
        inits.append(project_h(resample(best_state[1], n), target))
        h_p = project_h(resample(best_state[1], n), target)
        h_p += np.random.normal(0, 0.06, n)
        inits.append(project_h(h_p, target))
        # Binary random
        s = np.random.choice([-1.0, 1.0], n)
        n_ones = int((s == 1).sum())
        n_wants = n // 2
        if n_ones > n_wants:
            idx = np.where(s == 1)[0]
            s[np.random.choice(idx, n_ones - n_wants, replace=False)] = -1
        elif n_ones < n_wants:
            idx = np.where(s == -1)[0]
            s[np.random.choice(idx, n_wants - n_ones, replace=False)] = 1
        inits.append(project_h((1.0 + s) / 2.0, target))
        # Cosine
        for freq in [1.5, 2.5]:
            inits.append(project_h(0.5 + 0.3 * np.cos(freq * np.pi * x), target))
        # Two bump
        h_b = 0.5 + 0.25 * (np.exp(-((x - 0.35) / 0.25)**2) + np.exp(-((x - 1.65) / 0.25)**2))
        inits.append(project_h(h_b, target))

        max_init = 6 if n > 250 else 7
        for h_init in inits[:max_init]:
            if tleft() < 25:
                break
            h_cur = project_h(h_init, target)
            c5_cur = compute_c5(h_cur)
            if c5_cur < best_state[0]:
                best_state[0] = c5_cur
                best_state[1] = h_cur.copy()
                best_state[2] = n

            h_cur, c5_cur = subgradient_descent(
                h_cur, n, target, dx, max_iters=1200,
                min_time_fn=tleft, best_state=best_state
            )

            if tleft() > 12:
                for beta in [300, 1000, 3000]:
                    if tleft() < 4:
                        break
                    try:
                        cf = partial(con_fun, target=target)
                        con = [{'type': 'eq', 'fun': cf, 'jac': con_jac}]
                        obj = partial(obj_smooth_max, dx=dx, beta=beta)
                        res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                                       constraints=con, options={'maxiter': 800, 'ftol': 1e-14})
                        h_new = project_h(res.x, target)
                        c5_new = compute_c5(h_new)
                        if c5_new < c5_cur:
                            h_cur = h_new
                            c5_cur = c5_new
                            if c5_cur < best_state[0]:
                                best_state[0] = c5_cur
                                best_state[1] = h_cur.copy()
                                best_state[2] = n
                    except Exception:
                        pass

            if tleft() > 50:
                h_cur, c5_cur = coordinate_polish(
                    h_cur, n, target, dx, max_sweeps=3,
                    min_time_fn=tleft, best_state=best_state
                )

    # Final high-precision polish
    if tleft() > 15:
        h_best = best_state[1].copy()
        n_best = best_state[2]
        target_b = n_best / 2.0
        dx_b = 2.0 / n_best
        bounds_b = Bounds(np.zeros(n_best), np.ones(n_best))
        for beta in [5000, 20000]:
            if tleft() < 5:
                break
            try:
                cf = partial(con_fun, target=target_b)
                con = [{'type': 'eq', 'fun': cf, 'jac': con_jac}]
                obj = partial(obj_smooth_max, dx=dx_b, beta=beta)
                res = minimize(obj, h_best, method='SLSQP', bounds=bounds_b,
                               constraints=con, options={'maxiter': 10000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_b)
                c5_new = compute_c5(h_new)
                if c5_new < best_state[0]:
                    h_best = h_new
                    best_state[0] = c5_new
                    best_state[1] = h_new.copy()
                    best_state[2] = n_best
            except Exception:
                pass

        if tleft() > 25:
            h_final, _ = coordinate_polish(
                best_state[1], best_state[2], best_state[2] / 2.0,
                2.0 / best_state[2], max_sweeps=2,
                min_time_fn=tleft, best_state=best_state
            )

    return best_state[1], best_state[0], best_state[2]
