            t = -s
            if n - t > 0:
                grad[:n-t] += wk * (1.0 - h[t:])
                grad[t:] += wk * (-h[:n-t])
    grad *= dx
    return val, grad

def do_slsqp(h, beta, maxiter):
    n = len(h)
    target = n / 2.0
    try:
        res = minimize(
            lambda x: smooth_max_grad(x, beta)[0],
            h, jac=lambda x: smooth_max_grad(x, beta)[1],
            method='SLSQP', bounds=[(0,1)]*n,
            constraints=[{'type':'eq','fun':lambda x: x.sum()-target,'jac':lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-18}
        )
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def do_trustconstr(h, beta, maxiter):
    from scipy.optimize import LinearConstraint, Bounds
    n = len(h)
    target = n / 2.0
    try:
        res = minimize(
            lambda x: smooth_max_grad(x, beta)[0],
            h, jac=lambda x: smooth_max_grad(x, beta)[1],
            method='trust-constr',
            bounds=Bounds(np.zeros(n), np.ones(n)),
            constraints=[LinearConstraint(np.ones(n), target, target)],
            options={'maxiter': maxiter, 'gtol': 1e-12, 'xtol': 1e-12}
        )
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def do_lbfgsb(h, beta, maxiter, cw):
    n = len(h)
    target = n / 2.0
    def obj(x):
        v, g = smooth_max_grad(x, beta)
        sv = x.sum() - target
        return v + 0.5 * cw * sv * sv, g + cw * sv
    try:
        res = minimize(obj, h, jac=True, method='L-BFGS-B',
                       bounds=[(0,1)]*n, options={'maxiter': maxiter, 'gtol': 1e-10})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def topk_subgrad(h, n_iters, lr0, decay, top_k):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(n_iters):
        lr_t = lr0 * (decay ** t)
        corr = np.correlate(x, 1.0 - x, mode="full")
        top_idx = np.argsort(corr)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            s = k - (n - 1)
            if s >= 0:
                grad[s:] += (1.0 - x[:n-s])
                grad[:n-s] += (-x[s:])
            else:
                t2 = -s
                grad[:n-t2] += (1.0 - x[t2:])
                grad[t2:] += (-x[:n-t2])
        grad /= top_k
        grad *= dx
        x_new = project_feasible(x - lr_t * grad, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x = x_new
    return best_x, best_v

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def legendre_start(n_pts, seed):
    p = 401
    legendre = np.zeros(p)
    for j in range(1, p):
        legendre[j] = 1.0 if pow(j, p-1, p) == 1 else -1.0
    legendre[0] = 0.0
    a = (legendre + 1.0) / 2.0
    if len(a) > n_pts:
        a = a[:n_pts]
    else:
        pad = np.zeros(n_pts - len(a))
        a = np.concatenate([a, pad])
    return project_feasible(a, n_pts / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(400, 0.5)
    n = len(h_best)
    h_best = project_feasible(h_best, n / 2.0)
    best_c5 = compute_c5(h_best)

    # Phase 1: Continuation with SLSQP
    for beta in [30, 60, 150, 300, 600, 1200]:
        if tl() < 45:
            break
        h_new = do_slsqp(h_best, beta, maxiter=350)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 2: L-BFGS-B with penalty (faster, different landscape)
    for beta in [200, 500, 1000, 2000]:
        if tl() < 45:
            break
        h_new = do_lbfgsb(h_best, beta, maxiter=300, cw=1e6)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 3: SLSQP high beta
    for beta in [1500, 3000, 5000]:
        if tl() < 40:
            break
        h_new = do_slsqp(h_best, beta, maxiter=400)
        c = compute_c5(h_new)
        if c < best_c5:
            h_best, best_c5 = h_new, c

    # Phase 4: Top-k subgradient (small k for focus)
    if tl() > 120:
        h_sg, c_sg = topk_subgrad(h_best, 8000, 0.015, 0.99995, top_k=3)
        if c_sg < best_c5:
            h_best, best_c5 = h_sg, c_sg
        if tl() > 40:
            h_new = do_slsqp(h_best, 4000, maxiter=500)
            c = compute_c5(h_new)
            if c < best_c5:
                h_best, best_c5 = h_new, c

    # Phase 5: Multi-restart exploration
    scales = [0.008, 0.015, 0.004, 0.025, 0.012, 0.006, 0.035, 0.018, 0.003, 0.022, 0.01, 0.04]
    for trial in range(len(scales)):
        if tl() < 55:
            break
        s = scales[trial]
        pert = np.random.randn(n) * s
        h_pert = project_feasible(h_best + pert, n / 2.0)
        h_new = do_lbfgsb(h_pert, 800, maxiter=150, cw=1e6)
        c = compute_c5(h_new)
        if c < best_c5:
            h_fin = do_slsqp(h_new, 3000, maxiter=300)
            c2 = compute_c5(h_fin)
            if c2 < best_c5:
                h_best, best_c5 = h_fin, c2
        elif c < best_c5 + 0.005 and tl() > 60:
            h_sg = topk_subgrad(h_new, 2000, 0.01, 0.9999, top_k=5)
            c3 = compute_c5(h_sg)
            if c3 < best_c5:
                h_fin = do_slsqp(h_sg, 3000, maxiter=300)
                c4 = compute_c5(h_fin)
                if c4 < best_c5:
                    h_best, best_c5 = h_fin, c4

    # Phase 6: Try trust-constr at high beta
    if tl() > 50:
        h_tc = do_trustconstr(h_best, 3000, maxiter=300)
        c = compute_c5(h_tc)
        if c < best_c5:
            h_best, best_c5 = h_tc, c

    # Phase 7: Final subgradient with adaptive top_k
    if tl() > 80:
        for tk in [3, 5, 8]:
            if tl() < 30:
                break
            h_sg, c_sg = topk_subgrad(h_best, 4000, 0.008, 0.99997, top_k=tk)
            if c_sg < best_c5:
                h_best, best_c5 = h_sg, c_sg

    # Phase 8: Final SLSQP polish
    if tl() > 25:
        h_final = do_slsqp(h_best, 8000, maxiter=600)
        c = compute_c5(h_final)
        if c < best_c5:
            h_best, best_c5 = h_final, c

    # Phase 9: Try higher resolution if time permits
    if tl() > 200 and n < 600:
        n_new = min(800, 2 * n)
        h_interp = interp_to_n(h_best, n_new)
        for beta in [200, 800, 3000]:
            if tl() < 50:
                break
            h_try = do_slsqp(h_interp, beta, maxiter=350)
            c = compute_c5(h_try)
            if c < best_c5:
                h_best, best_c5, n = h_try, c, n_new
                h_interp = h_best.copy()

    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)
    return h_best, best_c5, len(h_best)
