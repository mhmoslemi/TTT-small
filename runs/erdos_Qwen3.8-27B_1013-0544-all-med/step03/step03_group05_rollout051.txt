def compute_all_corr(h, dx):

    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

        grad[:] = (1.0 - 2.0 * h) * dx

        grad[lag:] += (1.0 - h[:n-lag]) * dx
        grad[:n-lag] -= h[lag:] * dx

                    if cur_c5 < best_state[0]:
                        best_state[0] = cur_c5
                        best_state[1] = cur.copy()
                        best_state[2] = n
    return cur, cur_c5

def phase_coord(h_init, n, target, dx, time_fn, time_limit, best_state, max_rounds=40):
    cur = h_init.copy()
    cur_c5 = compute_c5(cur)
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    for rnd in range(max_rounds):
        if time_fn() < time_limit:
            break
        grad_full, corr = subgrad_direction(cur, n, dx, 8, 120.0)
        order = np.random.permutation(n)
        for i in order:
            sg = grad_full[i]
            if abs(sg) < 1e-12:
                continue
            direction = -1.0 if sg > 0 else 1.0
            for step in [0.03, 0.015, 0.007]:
                h_trial = cur.copy()
                h_trial[i] = np.clip(h_trial[i] + direction * step, 0.0, 1.0)
                h_trial = project_h(h_trial, target)
                c5_trial = compute_c5(h_trial)
                if c5_trial < cur_c5 - 1e-15:
                    cur = h_trial
                    cur_c5 = c5_trial
                    if cur_c5 < best_state[0]:
                        best_state[0] = cur_c5
                        best_state[1] = cur.copy()
                        best_state[2] = n
                    break
    return cur, cur_c5

def phase_fine_subgrad(h_init, n, target, dx, time_fn, time_limit, best_state, max_iters=200000):
    cur = h_init.copy()
    cur_c5 = compute_c5(cur)
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    for it in range(max_iters):
        if time_fn() < time_limit:
            break
        grad, corr = subgrad_direction(cur, n, dx, 8, 200.0)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad /= gnorm
        step = 0.004 * max(0.05, 1.0 - it / max_iters)
        improved = False
        for f in [1.0, 0.5, 0.2, 0.1]:
            h_trial = project_h(cur - step * f * grad, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 - 1e-16:
                cur = h_trial
                cur_c5 = c5_trial
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
        if not improved and it % 100 == 0:
            T = 0.001 * np.exp(-it / 50000.0)
            noise = np.random.normal(0, T, n)
            h_trial = project_h(cur + noise, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < cur_c5 + T:
                cur = h_trial
                cur_c5 = c5_trial
    return cur, cur_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h_parent)
    else:
        n_parent = 250
        h_parent = np.full(n_parent, 0.5)

    best_state = [compute_c5(h_parent), h_parent.copy(), n_parent]

    n = n_parent
    target = n / 2.0
    dx = 2.0 / n

    # Phase 1: Main subgradient at parent resolution
    cur, _ = phase_subgradient(h_parent, n, target, dx, time_left, 60, best_state,
                               top_k=6, beta=100.0, mom=0.88)

    # Phase 2: Coordinate descent polish
    phase_coord(best_state[1], n, target, dx, time_left, 30, best_state, max_rounds=25)

    # Phase 3: Try a couple of other resolutions
    for n_try in [200, 300]:
        if time_left() < 80:
            break
        target_try = n_try / 2.0
        dx_try = 2.0 / n_try
        h_try = resample_to(best_state[1], n_try, target_try)
        phase_subgradient(h_try, n_try, target_try, dx_try, time_left, 25, best_state,
                          top_k=6, beta=100.0, mom=0.85)
        phase_coord(best_state[1], n_try, target_try, dx_try, time_left, 15, best_state,
                    max_rounds=15)

    # Phase 4: Fine subgradient on best solution found
    h_best = best_state[1].copy()
    n_b = len(h_best)
    tgt_b = n_b / 2.0
    dx_b = 2.0 / n_b
    phase_fine_subgrad(h_best, n_b, tgt_b, dx_b, time_left, 3, best_state, max_iters=300000)

    # Phase 5: Final coordinate polish
    h_fin = best_state[1].copy()
    n_f = len(h_fin)
    tgt_f = n_f / 2.0
    dx_f = 2.0 / n_f
    phase_coord(h_fin, n_f, tgt_f, dx_f, time_left, 1, best_state, max_rounds=10)

    # Final output
    h_out = project_h(best_state[1], len(best_state[1]) / 2.0)
    c5_out = compute_c5(h_out)
    if c5_out > best_state[0] + 1e-12:
        h_out = best_state[1]
        c5_out = best_state[0]
    return h_out, c5_out, len(h_out)
