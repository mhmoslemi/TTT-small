from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_overlap(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            if diff > 0:
                free = h < 1.0
            else:
                free = h > 0.0
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, target, rng):
    """Convert h to binary with exactly target ones."""
    n = len(h)
    # Sort by value, keep top 'target' as 1
    order = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[order[:int(target)]] = 1.0
    return hb

def swap_delta(h, i, j, n):
    """Compute delta in correlation when h[i]:1->0, h[j]:0->1. Vectorized."""
    delta = np.zeros(2 * n - 1)
    # h[i] 1->0: removes (1-h[i-k]) at lags where i-k valid, adds h[i+k] at lags where i+k valid
    # corr index r = n-1+k
    # Term 1: -(1-h[i-k]) for k = i - (n-1) .. i (where 0 <= i-k < n)
    lo1 = max(0, i - n + 1)
    hi1 = min(n - 1, i)
    if lo1 <= hi1:
        k1 = np.arange(lo1, hi1 + 1)
        r1 = n - 1 + k1
        p1 = i - k1
        delta[r1] -= (1.0 - h[p1])
    # Term 2: +h[i+k] for k = -i .. n-1-i (where 0 <= i+k < n)
    lo2 = max(0, -i)
    hi2 = min(n - 1, n - 1 - i)
    if lo2 <= hi2:
        k2 = np.arange(lo2, hi2 + 1)
        r2 = n - 1 + k2
        p2 = i + k2
        delta[r2] += h[p2]
    # h[j] 0->1: adds (1-h[j-k]) at lags where j-k valid, removes h[j+k] at lags where j+k valid
    lo3 = max(0, j - n + 1)
    hi3 = min(n - 1, j)
    if lo3 <= hi3:
        k3 = np.arange(lo3, hi3 + 1)
        r3 = n - 1 + k3
        p3 = j - k3
        delta[r3] += (1.0 - h[p3])
    lo4 = max(0, -j)
    hi4 = min(n - 1, n - 1 - j)
    if lo4 <= hi4:
        k4 = np.arange(lo4, hi4 + 1)
        r4 = n - 1 + k4
        p4 = j + k4
        delta[r4] -= h[p4]
    return delta

def binary_sa(h_init, n_iter, t_start, t_end, seed, time_limit):
    """Simulated annealing for binary sequence optimization."""
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    
    h = (h_init >= 0.5).astype(float)
    # Fix count
    ones_count = h.sum()
    if ones_count > target:
        ones_idx = np.where(h == 1)[0]
        to_flip = rng.choice(ones_idx, int(ones_count - target), replace=False)
        h[to_flip] = 0
    elif ones_count < target:
        zeros_idx = np.where(h == 0)[0]
        to_flip = rng.choice(zeros_idx, int(target - ones_count), replace=False)
        h[to_flip] = 1
    
    overlap = get_overlap(h)
    c5 = float(np.max(overlap) * dx)
    best_h = h.copy()
    best_c5 = c5
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    t0 = time.time()
    n_actual = min(n_iter, max(100, int((time_limit - (t0 - t0)) * 5000)))  # estimate
    # Actually let's just use n_iter and check time periodically
    
    for it in range(n_iter):
        if it % 5000 == 0 and time.time() - t0 > time_limit:
            break
        
        T = t_start * (t_end / t_start) ** (it / n_iter)
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        delta = swap_delta(h, i, j, n)
        new_overlap = overlap + delta
        new_c5 = float(np.max(new_overlap) * dx)
        
        if new_c5 <= c5 or rng.random() < np.exp(-(new_c5 - c5) / max(T, 1e-10)):
            h[i] = 0
            h[j] = 1

            c5 = new_c5
            ones.remove(i)
            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if c5 < best_c5:

    return best_h, best_c5

def continuous_refine(h_bin, budget_s):
    """Try to improve a binary solution by allowing small non-binary values."""
    n = len(h_bin)
    target = n / 2.0
    dx = 2.0 / n
    
    h = h_bin.copy()
    c5 = compute_c5(h)
    
    # Try modifying pairs of adjacent positions

    best_c5 = c5
    t0 = time.time()

    for i in range(n - 1):

                tc = compute_c5(trial)
                if tc < best_c5 - 1e-12:
                    best_c5 = tc
                    best_h = trial.copy()
    
    return best_h, best_c5

        target = n_pts / 2.0

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_overlap(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            if diff > 0:
                free = h < 1.0
            else:
                free = h > 0.0
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, target, rng):
    n = len(h)
    thresholds = np.linspace(0.3, 0.7, 7)
    best_hb = None
    best_c5 = np.inf
    for t in thresholds:
        hb = (h >= t).astype(float)
        deficit = int(target) - int(hb.sum())
        if deficit > 0:
            zeros = np.where(hb == 0)[0]
            order = zeros[np.argsort(h[zeros])[::-1]]
            if len(order) >= deficit:
                hb[order[:deficit]] = 1.0
        elif deficit < 0:
            ones = np.where(hb == 1)[0]
            order = ones[np.argsort(h[ones])]
            if len(order) >= -deficit:
                hb[order[:(-deficit)]] = 0.0
        if abs(hb.sum() - target) < 1e-6:
            c5 = compute_c5(hb)
            if c5 < best_c5:
                best_c5 = c5
                best_hb = hb.copy()
    if best_hb is None:
        order = np.argsort(h)[::-1]
        best_hb = np.zeros(len(h))
        best_hb[order[:int(target)]] = 1.0
    return best_hb

def swap_delta(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    lo1 = max(0, i - n + 1)
    hi1 = min(n - 1, i)
    if lo1 <= hi1:
        k1 = np.arange(lo1, hi1 + 1)
        r1 = n - 1 + k1
        delta[r1] -= (1.0 - h[i - k1])
    lo2 = max(0, -i)
    hi2 = min(n - 1, n - 1 - i)
    if lo2 <= hi2:
        k2 = np.arange(lo2, hi2 + 1)
        r2 = n - 1 + k2
        delta[r2] += h[i + k2]
    lo3 = max(0, j - n + 1)
    hi3 = min(n - 1, j)
    if lo3 <= hi3:
        k3 = np.arange(lo3, hi3 + 1)
        r3 = n - 1 + k3
        delta[r3] += (1.0 - h[j - k3])
    lo4 = max(0, -j)
    hi4 = min(n - 1, n - 1 - j)
    if lo4 <= hi4:
        k4 = np.arange(lo4, hi4 + 1)
        r4 = n - 1 + k4
        delta[r4] -= h[j + k4]
    return delta

def binary_sa(h_init, n_iter, t_start, t_end, seed, time_limit):
    rng = np.random.RandomState(seed)
    n = len(h_init)
    target = n // 2
    dx = 2.0 / n
    h = (h_init >= 0.5).astype(float)
    cnt = int(h.sum())
    if cnt > target:
        idx = rng.choice(np.where(h == 1)[0], cnt - target, replace=False)
        h[idx] = 0
    elif cnt < target:
        idx = rng.choice(np.where(h == 0)[0], target - cnt, replace=False)
        h[idx] = 1
    overlap = get_overlap(h)
    c5 = float(np.max(overlap) * dx)
    best_h = h.copy()
    best_c5 = c5
    t0 = time.time()
    ratio = t_end / max(t_start, 1e-15)
    for it in range(n_iter):
        if it % 2000 == 0 and (time.time() - t0) > time_limit:
            break
        T = t_start * (ratio ** (float(it) / float(n_iter)))
        ones_idx = np.flatnonzero(h)
        zeros_idx = np.flatnonzero(1 - h)
        i = int(ones_idx[rng.randint(len(ones_idx))])
        j = int(zeros_idx[rng.randint(len(zeros_idx))])
        delta = swap_delta(h, i, j, n)
        new_c5 = float(np.max(overlap + delta) * dx)
        if new_c5 <= c5 or rng.random() < np.exp(-(new_c5 - c5) / max(T, 1e-12)):
            h[i] = 0
            h[j] = 1
            overlap = overlap + delta
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def greedy_improve(h, time_limit=30, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    dx = 2.0 / n
    overlap = get_overlap(h)
    c5 = float(np.max(overlap) * dx)
    t0 = time.time()
    for _ in range(50000):
        if time.time() - t0 > time_limit:
            break
        best_new_c5 = c5
        best_pair = None
        best_delta = None
        for _ in range(80):
            i = int(rng.choice(np.flatnonzero(h)))
            j = int(rng.choice(np.flatnonzero(1 - h)))
            d = swap_delta(h, i, j, n)
            nc = float(np.max(overlap + d) * dx)
            if nc < best_new_c5 - 1e-14:
                best_new_c5 = nc
                best_pair = (i, j)
                best_delta = d
        if best_pair is None:
            break
        i, j = best_pair
        h[i] = 0
        h[j] = 1
        overlap = overlap + best_delta
        c5 = best_new_c5
    return h, c5

def continuous_pair_refine(h, time_limit=30):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    fracs = [0.2, 0.25, 0.3, 0.4, 0.5, 0.6, 0.7, 0.75, 0.8]
    for i in range(n - 1):
        if time.time() - t0 > time_limit:
            break
        if abs(h[i] - h[i + 1]) > 1e-6:
            for f in fracs:
                trial = best_h.copy()
                trial[i] = f
                trial[i + 1] = 1.0 - f
                tc = compute_c5(trial)
                if tc < best_c5 - 1e-13:
                    best_c5 = tc
                    best_h = trial.copy()
    return best_h, best_c5

def slsqp_polish(h_init, time_limit=30):
    n = len(h_init)
    target = n / 2.0
    dx = 2.0 / n
    t0 = time.time()
    for beta in [300, 800, 2000]:
        if time.time() - t0 > time_limit - 5:
            break
        h_cur = h_init.copy()
        c5_cur = compute_c5(h_cur)
        def obj_func(x):
            hp = project_h(x, target)
            corr = get_overlap(hp)
            m = np.max(corr)
            shifted = np.clip(beta * (corr - m), -500, 500)
            return (m + np.log(np.sum(np.exp(shifted)) / len(corr)) / beta) * dx
        def con_func(x):
            return np.sum(x) - target
        def con_jac(x):
            return np.ones(len(x))
        try:
            res = minimize(obj_func, h_cur, method='SLSQP',
                           bounds=Bounds(np.zeros(n), np.ones(n)),
                           constraints=[{'type': 'eq', 'fun': con_func, 'jac': con_jac}],
                           options={'maxiter': 1000, 'ftol': 1e-14})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                h_init = h_new
        except Exception:
            pass
    return h_init, compute_c5(h_init)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(280, 0.5)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Binary SA at parent resolution
    n_base = len(h_parent)
    h_bin = make_binary(h_parent, n_base / 2.0, np.random.RandomState(seed))
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin.copy(), c5_bin

    if remaining() > 100:
        n_iter = 800000
        time_alloc = min(remaining() * 0.4, 300)
        h_sa, c5_sa = binary_sa(best_h, n_iter, 0.04, 0.0003, seed, time_alloc)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa.copy(), c5_sa

    # Greedy improvement
    if remaining() > 80:
        h_g, c5_g = greedy_improve(best_h, min(remaining() * 0.15, 40), seed + 10)
        if c5_g < best_c5:
            best_h, best_c5 = h_g.copy(), c5_g

    # Phase 2: Multiple resolutions
    for n_try in [300, 320, 350, 400]:
        if remaining() < 80:
            break
        if n_try % 2 != 0:
            continue
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_interp = np.interp(x_new, x_old, best_h)
        h_bin_new = make_binary(h_interp, n_try / 2.0, np.random.RandomState(seed + n_try))
        c5_new = compute_c5(h_bin_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_bin_new.copy(), c5_new
        if remaining() < 60:
            break
        time_alloc = min(remaining() * 0.25, 120)
        n_iter = 400000
        h_sa2, c5_sa2 = binary_sa(best_h, n_iter, 0.03, 0.0003, seed + n_try, time_alloc)
        if c5_sa2 < best_c5:
            best_h, best_c5 = h_sa2.copy(), c5_sa2
        if remaining() > 30:
            h_g2, c5_g2 = greedy_improve(best_h, min(remaining() * 0.1, 20), seed + n_try + 1)
            if c5_g2 < best_c5:
                best_h, best_c5 = h_g2.copy(), c5_g2

    # Phase 3: Continuous pair refinement
    if remaining() > 40:
        h_cr, c5_cr = continuous_pair_refine(best_h, min(remaining() * 0.2, 30))
        if c5_cr < best_c5:
            best_h, best_c5 = h_cr.copy(), c5_cr

    # Phase 4: SLSQP polish
    if remaining() > 25:
        h_sp, c5_sp = slsqp_polish(best_h, min(remaining() * 0.4, 40))
        if c5_sp < best_c5:
            best_h, best_c5 = h_sp.copy(), c5_sp

    # Final feasibility check
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, int(len(best_h))
