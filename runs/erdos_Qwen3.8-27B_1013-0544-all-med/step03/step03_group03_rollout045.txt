import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    w = np.exp(shifted)
    w /= w.sum()
    return w

def obj_func(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx

def obj_jac(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    w = smooth_max_grad(corr, beta)
    grad = np.zeros(n)
    for k in range(n):
        wk = w[n - 1 + k]
        if wk > 1e-14:
            if k == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                m = n - k
                grad[:m] += wk * (1.0 - h[k:])
                grad[k:] += -wk * h[:m]
        if k > 0:
            wk2 = w[n - 1 - k]
            if wk2 > 1e-14:
                m = n - k
                grad[k:] += wk2 * (1.0 - h[:m])
                grad[:m] += -wk2 * h[k:]
    return grad * dx

def con_eq(h, target_sum):
    return h.sum() - target_sum

def con_jac_fn(h, target_sum):
    return np.ones_like(h)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    t0 = time.time()

        return budget_s - (time.time() - t0)

    n = len(h_init)
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac_fn, 'args': (target_sum,)}]

    best_h = project_feasible(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP continuation with analytic gradient
    betas_phase1 = [15, 40, 100, 300, 800, 2000]

        if remaining() < 120:
            break

            if remaining() < 60:

            x0 = best_h.copy()

                x0 += np.random.randn(n) * 0.004
                x0 = project_feasible(x0, target_sum)
            try:
                res = minimize(obj_func, x0, args=(dx, beta), jac=obj_jac,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 4000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5
            except Exception:
                pass

    # Phase 2: ADAM adaptive gradient with restarts

        h = best_h.copy()
        m_v = np.zeros(n)
        v_v = np.zeros(n)
        b1, b2, eps = 0.8, 0.999, 1e-8
        t_adam = 0
        lr = 0.003
        stagnant = 0
        beta_adam = 800
        for it in range(80000):

                m_v *= 0.95

                h += np.random.randn(n) * 0.01
                h = project_feasible(h, target_sum)

                v_v = np.zeros(n)
                t_adam = 0
                stagnant = 0

    if remaining() > 80:

            if remaining() < 40:

            for trial in range(2):
                if remaining() < 25:

                    h_new = project_feasible(res.x, target_sum)

                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                except Exception:
                    pass

    # Phase 4: Resolution exploration
    if remaining() > 100:
        res_list = [150, 180, 200, 240, 270, 300, 350, 400]
        for n_try in res_list:
            if remaining() < 50 or n_try == n:
                continue
            dx_t = 2.0 / n_try
            ts_t = n_try / 2.0
            x_old = np.linspace(0, 2, len(best_h), endpoint=False)
            x_new = np.linspace(0, 2, n_try, endpoint=False)
            h_t = np.interp(x_new, x_old, best_h)
            h_t = project_feasible(h_t, ts_t)

                    res = minimize(obj_func, h_t, args=(dx_t, beta_t), jac=obj_jac,
                                   method='SLSQP', bounds=bnds_t, constraints=cons_t,
                                   options={'maxiter': 3000, 'ftol': 1e-15})
                    h_new = project_feasible(res.x, ts_t)
                    c5 = compute_c5(h_new)
                    if c5 < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5
                        n = n_try
                        dx = dx_t
                        target_sum = ts_t
                        bounds = bnds_t
                        cons = cons_t
                except Exception:
                    pass

    # Phase 5: Final polish
    if remaining() > 20:

            if remaining() < 12:

                    best_c5 = c5
            except Exception:

            if remaining() < 5:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
