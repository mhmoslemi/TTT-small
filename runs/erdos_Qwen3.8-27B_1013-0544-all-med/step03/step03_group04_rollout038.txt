import numpy as np
from scipy.optimize import differential_evolution, minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            free = h < 1.0 - 1e-10
            if free.sum() == 0:
                break
            h[free] += diff / free.sum()
        else:
            free = h > 1e-10
            if free.sum() == 0:
                break
            h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_h(h, n_top):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_top]
    grad = np.zeros(n)
    for li in top_idx:
        lag = int(li) - (n - 1)
        j = np.arange(n)
        j_plus = j + lag
        j_minus = j - lag
        mask_plus = (j_plus >= 0) & (j_plus < n)
        mask_minus = (j_minus >= 0) & (j_minus < n)
        grad[mask_plus] += (1.0 - h[j_plus[mask_plus]])
        grad[mask_minus] -= h[j_minus[mask_minus]]
    return grad

def coord_descent(h, target, max_passes=5, use_full_recompute=True):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    corr = np.correlate(h_cur, 1.0 - h_cur, mode="full").astype(float)
    cur_c5 = float(np.max(corr) * dx)
    
    candidates = [0.0, 0.25, 0.5, 0.75, 1.0]
    
    for _pass in range(max_passes):
        improved = False
        for i in range(n):
            old_val = h_cur[i]
            best_cand = old_val
            best_c5 = cur_c5
            
            for c in candidates:
                if abs(c - old_val) < 1e-8:
                    continue
                h_trial = h_cur.copy()
                h_trial[i] = c
                h_trial = project_h(h_trial, target)
                c = np.correlate(h_trial, 1.0 - h_trial, mode="full")
                c5_trial = float(np.max(c) * dx)
                if c5_trial < best_c5 - 1e-12:
                    best_c5 = c5_trial
                    best_cand = c
            
            if best_cand != old_val:
                h_cur[i] = best_cand
                h_cur = project_h(h_cur, target)
                cur_c5 = best_c5
                corr = np.correlate(h_cur, 1.0 - h_cur, mode="full").astype(float)
                improved = True
        
        if not improved:
            break
    
    return h_cur, cur_c5

def nesterov_subgrad(h, target, max_iter=3000, n_top=10):
    n = len(h)
    h_cur = h.copy()
    h_prev = h.copy()
    cur_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    best_c5 = cur_c5
    
    for it in range(max_iter):
        alpha = 0.05 / (1.0 + 0.005 * it)
        beta_k = 2.0 / (it + 2)
        beta_next = 2.0 / (it + 3)
        
        # Nesterov extrapolation
        y = h_cur + (beta_k - 1.0) / beta_next * (h_cur - h_prev)
        y = project_h(y, target)
        
        # Subgradient
        g = subgradient_h(y, n_top)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g = g / gn
        
        h_new = project_h(y - alpha * g, target)
        c5_new = compute_c5(h_new)
        
        if c5_new < cur_c5:
            cur_c5 = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        h_prev = h_cur
        h_cur = h_new
    
    return best_h, best_c5

def try_binary_optimize(h, target, max_flips=100):
    """Try to improve by flipping individual values toward binary."""
    n = len(h)
    h_cur = h.copy()
    cur_c5 = compute_c5(h_cur)
    dx = 2.0 / n
    
    for _ in range(max_flips):
        best_flip = None
        best_c5 = cur_c5
        best_idx = -1
        best_dir = 0
        
        # Try flipping random non-binary values toward 0 or 1
        non_binary = np.where((h_cur > 0.01) & (h_cur < 0.99))[0]
        if len(non_binary) == 0:
            break
        
        # Try a few random ones
        n_try = min(20, len(non_binary))
        idxs = np.random.choice(non_binary, size=n_try, replace=False)
        
        for i in idxs:
            # Try moving toward 0
            h_trial = h_cur.copy()
            h_trial[i] = 0.0
            h_trial = project_h(h_trial, target)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-10:
                best_c5 = c5_t
                best_flip = h_trial
                best_idx = i
                best_dir = 0
            
            # Try moving toward 1
            h_trial = h_cur.copy()
            h_trial[i] = 1.0
            h_trial = project_h(h_trial, target)
            c5_t = compute_c5(h_trial)
            if c5_t < best_c5 - 1e-10:
                best_c5 = c5_t
                best_flip = h_trial
                best_idx = i
                best_dir = 1
        
        if best_flip is not None:
            h_cur = best_flip
            cur_c5 = best_c5
        else:

    return h_cur, cur_c5

def binary_coord_descent(h, target, max_passes=20):
    """Coordinate descent restricted to binary values."""
    n = len(h)
    dx = 2.0 / n
    # First, make it binary
    h_cur = (h >= 0.5).astype(float)
    # Fix sum
    deficit = target - h_cur.sum()
    if deficit > 0:
        zeros = np.where(h_cur == 0)[0]
        h_cur[zeros[:int(deficit)]] = 1.0
    elif deficit < 0:
        ones = np.where(h_cur == 1)[0]
        h_cur[ones[:int(-deficit)]] = 0.0
    
    cur_c5 = compute_c5(h_cur)
    
    for _pass in range(max_passes):
        improved = False
        # Try swapping: change a 1 to 0 and a 0 to 1
        ones = np.where(h_cur == 1)[0]
        zeros = np.where(h_cur == 0)[0]
        
        for i in ones:
            best_swap = None
            best_c5 = cur_c5
            for j in zeros:
                h_trial = h_cur.copy()
                h_trial[i] = 0.0
                h_trial[j] = 1.0
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-12:
                    best_c5 = c5_trial
                    best_swap = (i, j)
            if best_swap is not None:
                h_cur[best_swap[0]] = 0.0
                h_cur[best_swap[1]] = 1.0
                cur_c5 = best_c5
                improved = True
        if not improved:
            break

But this is O(n^2) per pass (n/2 ones * n/2 zeros = n^2/4 swaps to try), and each swap requires O(n^2) to compute correlation. So total is O(n^4) per pass. For n=280, that's 280^4/4 ≈ 6.1 * 10^9 operations per pass. Way too slow!

I need a smarter approach for binary optimization. One option: precompute the correlation, and for each swap (i→0, j→1), compute the change in correlation in O(1) or O(n).

When we swap: h[i] goes from 1 to 0, h[j] goes from 0 to 1.
The correlation at lag k (in full mode, index p = k + n - 1):
corr[p] = sum_{valid} h[m] * (1-h[m+k])

The change when h[i]: 1→0 and h[j]: 0→1:
- Terms where h[i] was the "1" that contributed: they lose contribution
- Terms where h[i] was the "0" (1-h): they gain contribution
- Similarly for h[j]

4. Repeat

def slsqp_smooth_max(h, target, max_iter=1000, beta=500):
    """SLSQP with smooth max approximation."""
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    cur_c5 = compute_c5(h_cur)
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(c)
        cs = np.clip(beta * (c - m), -500, 500)
        return (m + np.log(np.sum(np.exp(cs)) / len(c)) / beta) * dx
    
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode="full")
        m = np.max(c)
        cs = np.clip(beta * (c - m), -500, 500)
        exps = np.exp(cs)
        weights = exps / np.sum(exps)
        # Gradient of smooth max w.r.t. c
        # d/dc_k [m + log(sum(exp(beta*(c-m))))/beta] = weights[k]
        # (approximately, ignoring the derivative of m)
        # Actually, the exact gradient is:
        # g_k = weights[k] - (1/n_lags) * sum(weights) ... no
        # The exponential weighted gradient
        # d/dx of smooth_max = sum_k weights[k] * d(c_k)/dx
        # where weights[k] = exp(beta*(c_k - m)) / sum(exp(beta*(c_j - m)))
        
        # d(c_k)/dx: each c_k = sum_i x[i] * (1-x[i+lag_k])
        # d/dx[j] = (1-x[j+lag_k]) - x[j-lag_k] (with bounds)
        
        grad = np.zeros(n)
        for k_w, k_idx in enumerate(range(len(c))):
            w = weights[k_idx]
            if w < 1e-10:
                continue

            j = np.arange(n)
            jp = j + lag
            jm = j - lag
            mp = (jp >= 0) & (jp < n)
            mm = (jm >= 0) & (jm < n)
            grad[mp] += w * (1.0 - x[jp[mp]])
            grad[mm] -= w * x[jm[mm]]
        return grad * dx
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]

        res = minimize(obj, h_cur, jac=obj_grad, method='SLSQP', 
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_new = project_h(res.x, target)
        c5_new = compute_c5(h_new)
        return h_new, c5_new
    except:
        return h_cur, cur_c5

    best_c5 = compute_c5(best_h)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: Coordinate descent
    if remaining() > 40:
        h1, c1 = coord_descent(best_h, target, max_passes=5, t0=t0, time_limit=budget_s - 30)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: SLSQP with smooth max
    if remaining() > 80:
        for beta in [200, 500, 1500]:
            if remaining() < 20:
                break
            h_s, c_s = slsqp_optimize(best_h, target, beta=beta, max_iter=300, t0=t0, time_limit=budget_s - 10)
            if c_s < best_c5:
                best_h, best_c5 = h_s, c_s

    # Phase 3: Nesterov subgradient
    if remaining() > 60:
        h2, c2 = nesterov_subgrad(best_h, target, max_iter=8000, n_top=15, t0=t0, time_limit=budget_s - 50)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 4: Coordinate descent
    if remaining() > 30:
        h3, c3 = coord_descent(best_h, target, max_passes=5, t0=t0, time_limit=budget_s - 20)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 5: Binary local search
    if remaining() > 30:
        h4, c4 = binary_local_search(best_h, target, n_iters=8000, t0=t0, time_limit=budget_s - 20)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 6: Another SLSQP round with higher beta
    if remaining() > 40:
        for beta in [2000, 5000]:
            if remaining() < 15:
                break
            h6, c6 = slsqp_optimize(best_h, target, beta=beta, max_iter=200, t0=t0, time_limit=budget_s - 10)
            if c6 < best_c5:
                best_h, best_c5 = h6, c6

    # Phase 7: Nesterov with more lags
    if remaining() > 60:
        h7, c7 = nesterov_subgrad(best_h, target, max_iter=5000, n_top=25, t0=t0, time_limit=budget_s - 50)
        if c7 < best_c5:
            best_h, best_c5 = h7, c7
        if remaining() > 20:
            h7b, c7b = coord_descent(best_h, target, max_passes=3, t0=t0, time_limit=budget_s - 10)
            if c7b < best_c5:
                best_h, best_c5 = h7b, c7b

    # Phase 8: Try structured alternatives
    if remaining() > 150:
        alts = []
        a1 = np.zeros(n)
        a1[:int(n * 0.5)] = 1.0
        alts.append(project_h(a1, target))
        a2 = np.zeros(n)
        a2[:int(n * 0.25)] = 1.0
        a2[int(n * 0.25):int(n * 0.75)] = 0.5
        alts.append(project_h(a2, target))
        p = max(4, n // 10)
        a3 = np.zeros(n)
        for k in range(0, n, 2 * p):
            a3[k:min(k + p, n)] = 1.0
        alts.append(project_h(a3, target))
        tm_val = np.zeros(n)
        for i in range(n):
            tm_val[i] = float(bin(i).count('1') % 2)
        alts.append(project_h(tm_val, target))
        alts.append(project_h(1.0 - tm_val, target))

        for alt in alts:
            if remaining() < 40:
                break
            ha, ca = coord_descent(alt, target, max_passes=3, t0=t0, time_limit=budget_s - 35)
            if ca < best_c5:
                best_h, best_c5 = ha, ca
            hb2, cb2 = nesterov_subgrad(ha, target, max_iter=3000, n_top=15, t0=t0, time_limit=budget_s - 30)
            if cb2 < best_c5:
                best_h, best_c5 = hb2, cb2
            hc, cc = coord_descent(best_h, target, max_passes=3, t0=t0, time_limit=budget_s - 20)
            if cc < best_c5:
                best_h, best_c5 = hc, cc

    # Phase 9: Final polish
    if remaining() > 15:
        hf, cf = coord_descent(best_h, target, max_passes=3, t0=t0, time_limit=budget_s - 5)
        if cf < best_c5:
            best_h, best_c5 = hf, cf

    # Final: try small binary perturbation
    if remaining() > 10:
        hb_f, cb_f = binary_local_search(best_h, target, n_iters=3000, t0=t0, time_limit=budget_s - 5)
        if cb_f < best_c5:
            best_h, best_c5 = hb_f, cb_f

    return best_h, best_c5, int(len(best_h))
