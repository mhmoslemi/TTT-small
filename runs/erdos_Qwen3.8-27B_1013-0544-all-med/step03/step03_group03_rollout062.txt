import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_penalty(h, dx, beta, mu, target_sum):
    corr = np.correlate(h, 1.0 - h, mode='full')
    base = smooth_max(corr, beta) * dx
    viol = h.sum() - target_sum
    return base + mu * viol * viol

    return h.sum() - target_sum

def sum_con_jac(h, target_sum):
    return np.ones(len(h))
