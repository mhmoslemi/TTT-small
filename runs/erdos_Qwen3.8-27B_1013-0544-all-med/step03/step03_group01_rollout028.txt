def project_feasible(h, n):

        if abs(diff) < 1e-15:
            break

            room = 1.0 - h
            s = room.sum()
            if s < 1e-15:

        else:
            s = h.sum()
            if s < 1e-15:
                break
            h += (diff / s) * h
    return np.clip(h, 0.0, 1.0)

def smoothmax_obj_grad(h, n, alpha):
    c = get_c(h, n)
    m = c.max()
    a = alpha * (c - m)
    mx = a.max()
    ea = np.exp(a - mx)

    best_c = compute_c5(h, n)
    best_h = h.copy()
    lr = step_init
    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = get_c(h, n)
        if top_k and top_k < len(c):
            idx = np.argpartition(c, -top_k)[-top_k:]
            mask = np.zeros(len(c))
            mask[idx] = 1.0
        else:
            mask = np.ones(len(c))
        a = alpha * (c - c.max())
        mx = a.max()
        ea = np.exp(a - mx) * mask
        s = ea.sum()
        if s < 1e-30:
            break
        w = ea / s
        dx = 2.0 / n
        omh = 1.0 - h
        grad = np.zeros(n)
        for k in range(-(n - 1), n):
            m_idx = k + (n - 1)
            wv = w[m_idx]

                continue

                grad[:n - k] += wv * omh[k:]
                grad[k:] -= wv * h[:n - k]
            else:
                grad[-k:] += wv * omh[:n + k]
                grad[:n + k] -= wv * h[-k:]
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break

            break
