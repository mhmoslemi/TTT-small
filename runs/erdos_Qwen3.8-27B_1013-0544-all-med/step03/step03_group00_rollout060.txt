```python
def smoothmax_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = len(corr)  # 2n-1
    if top_k is not None and top_k < n_corr:
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(n_corr)
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for idx in top_idx:
        wk = w_sel[idx]
        if wk < 1e-14:
            continue
        lag = idx - (n - 1)
        # For each i, ∂c[lag]/∂h[i] = (1-h[i-lag])*1[0<=i-lag<n] - h[i+lag]*1[0<=i+lag<n]
        if lag >= 0:
            # i-lag in [0,n): i in [lag, n)
            # i+lag in [0,n): i in [0, n-lag)
            if lag == 0:
                # c[0] = sum_i h[i]*(1-h[i])
                # grad = 1 - 2*h[i]
                grad += wk * (1.0 - 2.0 * h)
            else:
                # First term: (1-h[i-lag]) for i in [lag, n)
                grad[lag:] += wk * (1.0 - h[:n-lag])
                # Second term: -h[i+lag] for i in [0, n-lag)
                grad[:n-lag] -= wk * h[lag:]
        else:
            # lag < 0, let L = -lag > 0
            L = -lag
            # i-(-L) = i+L in [0,n): i in [0, n-L)
            # i+(-L) = i-L in [0,n): i in [L, n)
            if L == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                # First term: (1-h[i+L]) for i in [0, n-L)
                grad[:n-L] += wk * (1.0 - h[L:])
                # Second term: -h[i-L] for i in [L, n)
                grad[L:] -= wk * h[:n-L]
    grad *= dx
    return val, grad
```

```python
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = len(corr)
    if top_k is not None and top_k < n_corr:
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(n_corr)
    c_sel = corr[top_idx]
    m = np.max(c_sel)

    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)

        wk = w_sel[idx]
        if wk < 1e-14:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:

            grad[:n - lag] -= wk * h[lag:]
        else:

            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    grad *= dx
    return val, grad

    n = len(h)
    target = n / 2.0
    ones = np.ones(n)
    
    def obj(x):

        return v + lam * (x.sum() - target) ** 2

        v, g = smoothmax_val_grad(x, beta, top_k)
        g = g + 2.0 * lam * (x.sum() - target) * ones
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def pgd_solve(h, beta, n_iter, top_k, lr=0.01):
    """Projected gradient descent with backtracking line search."""
    n = len(h)
    target = n / 2.0
    h_cur = h.copy()
    for it in range(n_iter):
        val, grad = smoothmax_val_grad(h_cur, beta, top_k)
        # Backtracking line search
        step = lr
        for _ in range(20):

            val_try, _ = smoothmax_val_grad(h_try, beta, top_k)
            if val_try < val:
                h_cur = h_try
                break
            step *= 0.5
        else:
            break
    return h_cur
```

```python
def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:

        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            # All at bounds, can't fix
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = len(corr)
    if top_k is not None and top_k < n_corr:
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(n_corr)
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for idx in top_idx:
        wk = w_sel[idx]
        if wk < 1e-15:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    grad *= dx
    return val, grad

def optimize_lbb(h, beta, maxiter, top_k=None, lam=1e5):
    n = len(h)
    target = n / 2.0
    ones = np.ones(n)
    
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        pen = (x.sum() - target) ** 2
        return v + lam * pen
    
    def objg(x):
        v, g = smoothmax_val_grad(x, beta, top_k)
        g = g + 2.0 * lam * (x.sum() - target) * ones
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14,
                                'maxls': 50})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)
```

```python
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    n_corr = len(corr)
    if top_k is not None and top_k < n_corr:
        top_idx = np.argsort(-corr)[:top_k]
    else:
        top_idx = np.arange(n_corr)
    c_sel = corr[top_idx]
    m = np.max(c_sel)
    exps = np.exp(beta * (c_sel - m))
    sw = exps.sum()
    w_sel = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for idx in top_idx:
        wk = w_sel[idx]
        if wk < 1e-15:
            continue
        lag = idx - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    grad *= dx
    return val, grad

def opt_lbb(h, beta, maxiter=1000, top_k=None, lam=1e4):
    n = len(h)
    target = n / 2.0
    ones = np.ones(n)
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta, top_k)
        return v + lam * (x.sum() - target) ** 2
    def objg(x):
        v, g = smoothmax_val_grad(x, beta, top_k)
        return g + 2.0 * lam * (x.sum() - target) * ones
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h, jac=objg, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-20, 'gtol': 1e-14, 'maxls': 100})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def opt_slsqp(h, beta, maxiter=300, top_k=None):
    n = len(h)
    target = n / 2.0
    def obj(x):
        return smoothmax_val_grad(x, beta, top_k)[0]
    def objg(x):
        return smoothmax_val_grad(x, beta, top_k)[1]
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = ({'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n)})
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
```
