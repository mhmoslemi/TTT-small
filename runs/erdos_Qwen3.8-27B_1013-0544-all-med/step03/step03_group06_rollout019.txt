import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_and_grad(h, beta, n_active=30):
    """Returns (smooth_max_value * dx, gradient) for the smooth max of corr."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    log_sum = m + np.log(exp_vals.sum()) / beta
    weights = exp_vals / exp_vals.sum()

    top_idx = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for idx in top_idx:
        w = weights[idx]
        if w < 1e-12:
            continue
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:n - lag])
            grad[:n - lag] -= w * h[lag:]
        elif -n < lag < 0:
            ma = -lag
            grad[:n - ma] += w * (1.0 - h[ma:])
            grad[ma:] -= w * h[:n - ma]
    grad *= dx
    return log_sum, grad


def subgradient_top(h, n_active=5):
    """Compute subgradient from top lags only."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(corr)[::-1][:n_active]
    grad = np.zeros(n)
    for idx in top_idx:
        lag = int(idx) - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
        elif -n < lag < 0:
            ma = -lag
            grad[:n - ma] += (1.0 - h[ma:])
            grad[ma:] -= h[:n - ma]
    return grad


def lbgfsb_optimize(h, target, beta, penalty, maxiter=3000):
    """L-BFGS-B optimization with penalty for sum constraint."""
    n = len(h)
    bounds = Bounds(np.zeros(n), np.ones(n))

    def obj(x):
        sm, _ = smooth_max_and_grad(x, beta, n_active=40)
        pen = penalty * (x.sum() - target) ** 2
        return sm + pen

    def grad(x):
        _, g = smooth_max_and_grad(x, beta, n_active=40)
        g += 2.0 * penalty * (x.sum() - target)
        return g

    try:
        res = minimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-18,
                                               'gtol': 1e-10, 'maxls': 40})
        return project_h(res.x, target)
    except Exception:
        return h
