    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-12:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad_matrix(h0):
    n = len(h0)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        s = k - (n - 1)
        if s == 0:
            G[k] = 1.0 - 2.0 * h0
        elif s > 0:
            G[k, s:] += 1.0 - h0[:n - s]
            G[k, :n - s] += -h0[s:]
        else:
            t = -s
            G[k, :n - t] += 1.0 - h0[t:]
            G[k, t:] += -h0[:n - t]
    G *= dx
    return G

def slp_step(h0):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    corr = compute_corr(h0)
    G = compute_grad_matrix(h0)
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    rhs = corr * dx - G @ h0
    cons = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target,
            t_var - G @ h_var >= rhs]
    prob = cp.Problem(cp.Minimize(t_var), cons)
    h_new = h0.copy()
    for solver in [cp.GLOP, cp.GLPK, cp.CBC]:
        try:
            prob.solve(solver=solver, verbose=False)
            if h_var.value is not None:
                h_new = project_feasible(np.array(h_var.value).flatten(), target)
            break
        except Exception:
            continue
    return h_new

def slsqp_polish(h, maxiter=300, beta=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    def sm_val(x):
        corr = compute_corr(x)
        m = np.max(corr)
        s = np.sum(np.exp(beta * (corr - m)))
        return (m + np.log(s) / beta) * dx
    
    def sm_grad(x):
        corr = compute_corr(x)
        m = np.max(corr)
        w = np.exp(beta * (corr - m))
        w /= w.sum()
        g = np.zeros(n)
        for k in range(len(corr)):
            if w[k] < 1e-12:
                continue
            s = k - (n - 1)
            if s == 0:
                g += w[k] * (1.0 - 2.0 * x)
            elif s > 0:
                g[s:] += w[k] * (1.0 - x[:n - s])
                g[:n - s] += -w[k] * x[s:]
            else:
                t = -s
                g[:n - t] += w[k] * (1.0 - x[t:])
                g[t:] += -w[k] * x[:n - t]
        g *= dx
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(sm_val, h, jac=sm_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate',
                 assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def subgrad_step(h, lr, top_k=10):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        if s == 0:
            grad += (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] += -h[s:]
        else:
            t = -s
            grad[:n - t] += (1.0 - h[t:])
            grad[t:] += -h[:n - t]
    grad *= dx / top_k
    h_new = project_feasible(h - lr * grad, target)
    return h_new
