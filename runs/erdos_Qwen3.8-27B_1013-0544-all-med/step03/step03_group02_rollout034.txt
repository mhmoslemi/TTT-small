def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    val = np.clip(beta * corr * dx, -500, 500)
    m = np.max(val)
    exp_val = np.exp(val - m)
    p = exp_val / np.sum(exp_val)
    obj = m + np.log(np.sum(exp_val) / len(exp_val))
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        pi = p[i]
        if pi < 1e-13:
            continue
        k = i - (n - 1)
        cnt = n - abs(k)
        if cnt <= 0:
            continue
        if k >= 0:
            grad[k:] += pi * (1.0 - h[:cnt])
            grad[:cnt] -= pi * h[k:k + cnt]
        else:
            grad[:cnt] += pi * (1.0 - h[-k:])
            grad[-k:] -= pi * h[:cnt]
    grad *= dx
    return obj, grad

    best_n = n_base

    resolutions = sorted(set([n_base, n_base - 40, n_base - 20, n_base + 20, n_base + 40]))
    resolutions = [r for r in resolutions if 240 <= r <= 500 and r % 2 == 0]
    if not resolutions:
        resolutions = [n_base]

    for n_pts in resolutions:
        remaining = budget_s - (time.time() - t0)
        if remaining < 45:
            break
        target = n_pts / 2.0
        cur = project_feasible(resample_h(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Multi-scale projected gradient descent
        for beta, n_iter, lr0 in [
            (20, 2000, 1.5), (50, 3000, 0.8), (100, 3000, 0.5),
            (200, 3000, 0.3), (500, 2500, 0.15), (1000, 2000, 0.08)
        ]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 8:
                break
            cur, cur_c5 = pgd_solve(cur, target, n_pts, beta, n_iter, lr0, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Coordinate descent
        remaining = budget_s - (time.time() - t0)
        if remaining > 15:
            cur, cur_c5 = coord_desc(cur, target, n_pts, t0, budget_s, 20)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Swap descent
        remaining = budget_s - (time.time() - t0)
        if remaining > 10:
            cur, cur_c5 = swap_descent(cur, target, n_pts, t0, budget_s, 10)
            if cur_c5 < best_c5:
                best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

        # Threshold check
        tc5, ht = try_threshold(cur, target)
        if tc5 < best_c5:
            best_h, best_c5, best_n = ht, tc5, n_pts

        # Random restarts
        remaining = budget_s - (time.time() - t0)
        if remaining > 30:
            for r in range(5):
                remaining = budget_s - (time.time() - t0)
                if remaining < 7:
                    break
                scale = 0.025 / (r + 1)
                hp = project_feasible(cur + np.random.randn(n_pts) * scale, target)
                hp, hp_c5 = pgd_solve(hp, target, n_pts, 200, 2000, 0.3, t0, budget_s)
                if hp_c5 < cur_c5:
                    cur, cur_c5 = hp, hp_c5
                if cur_c5 < best_c5:
                    best_h, best_c5, best_n = cur.copy(), cur_c5, n_pts

    # Final high-precision polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 12 and best_h is not None:
        nf = len(best_h)
        tf = nf / 2.0
        for beta, n_iter, lr0 in [(2000, 3000, 0.04), (5000, 2000, 0.02)]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 5:
                break
            best_h, best_c5 = pgd_solve(best_h, tf, nf, beta, n_iter, lr0, t0, budget_s)
        remaining = budget_s - (time.time() - t0)
        if remaining > 6:
            best_h, best_c5 = coord_desc(best_h, tf, nf, t0, budget_s, 30)
        remaining = budget_s - (time.time() - t0)
        if remaining > 4:
            best_h, best_c5 = swap_descent(best_h, tf, nf, t0, budget_s, 8)
        # Ensure feasibility
        best_h = project_feasible(best_h, tf)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, best_n
