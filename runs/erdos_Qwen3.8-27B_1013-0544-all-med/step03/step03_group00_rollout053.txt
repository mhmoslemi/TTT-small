def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:

        else:

def smoothmax_val_grad(h, beta, top_k=None):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = len(corr)

            mode_sched=mode_sched2, seed=seed + 100)

    # Phase 6: Fine basin hopping
    if (budget_s - (time.time() - t0)) > 50:
        mode_sched3 = ['gaussian', 'sinusoidal', 'gaussian']
        noise_scales3 = [0.02, 0.015, 0.01, 0.008, 0.006, 0.004]
        h_best, best_c5 = do_basin_hop(
            h_best, best_c5, n_hops=12, beta=600, maxiter=250,
            t0=t0, budget_s=budget_s, noise_scales=noise_scales3,
            mode_sched=mode_sched3, seed=seed + 200)

    # Phase 7: Final high-beta polish
    if (budget_s - (time.time() - t0)) > 30:
        h_fin = lbfgsb_optimize(h_best, 800, top_k=None,
                                maxiter=500, penalty=1e6)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5 = c_fin
            h_best = h_fin

    # Phase 8: Final micro-hops
    if (budget_s - (time.time() - t0)) > 25:
        mode_sched4 = ['gaussian']
        noise_scales4 = [0.008, 0.005, 0.003, 0.002]
        h_best, best_c5 = do_basin_hop(
            h_best, best_c5, n_hops=8, beta=800, maxiter=200,
            t0=t0, budget_s=budget_s, noise_scales=noise_scales4,
            mode_sched=mode_sched4, seed=seed + 300, min_time=20)

    # Final validation
    h_best = project_feasible(h_best, len(h_best) / 2.0)
    best_c5 = compute_c5(h_best)

    return h_best, best_c5, len(h_best)
