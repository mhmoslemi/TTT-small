def project_h(h, target_sum):

        if abs(diff) < 1e-12:

def sa_binary_swap(h, n, n_iter, T_start, T_end, t0, budget):
    """Simulated annealing with swap moves on binary sequence."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_c5 = best_c5
    accept_count = 0
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break
        T = T_start * (T_end / T_start) ** (it / n_iter)
        
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        # Swap
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        
        if new_c5 <= cur_c5 or np.random.rand() < np.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            # Update ones and zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            accept_count += 1
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5
