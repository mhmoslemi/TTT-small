import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
        diff2 = target_sum - h.sum()
        if abs(diff2) > 1e-14:
            free = (h > 1e-9) & (h < 1.0 - 1e-9)
            if free.sum() > 0:
                h[free] = h[free] + diff2 / free.sum()
                h = np.clip(h, 0.0, 1.0)
    return h

def smoothmax_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    m = np.max(corr)
    shifted = corr - m
    exps = np.exp(np.clip(beta * shifted, -600, 600))
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] += wk * (-h[s:])
        else:
            t = -s
            grad[:n - t] += wk * (1.0 - h[t:])
            grad[t:] += wk * (-h[:n - t])
    grad *= dx
    return val, grad

def slsqp_optimize(h, beta, maxiter, ftol=1e-16):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def jac(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    def cfun(x):
        return x.sum() - target
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h.copy(), target)

def top_lag_subgrad(h, max_iters, lr0, decay, top_k, target_c5=None):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    x = h.copy().astype(float)
    best_x = x.copy()
    best_c5 = compute_c5(x)
    for t in range(max_iters):
        lr_t = lr0 * (decay ** t)
        corr = compute_corr(x)
        top_idx = np.argsort(corr)[-top_k:]
        lags = top_idx - (n - 1)
        grad = np.zeros(n)
        for si in lags:
            s = int(si)
            if s >= 0:
                grad[s:] += (1.0 - x[:n - s])
                grad[:n - s] += (-x[s:])
            else:
                tt = -s
                grad[:n - tt] += (1.0 - x[tt:])
                grad[tt:] += (-x[:n - tt])
        grad = grad / top_k * dx
        x_new = project_feasible(x - lr_t * grad, target)
        c = compute_c5(x_new)
        if c < best_c5:
            best_c5 = c
            best_x = x_new.copy()
        x = x_new
    return best_x, best_c5

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate', assume_sorted=True)
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

def structured_perturb(h, scale, rng):
    n = len(h)
    target = n / 2.0
    perts = []
    for freq in [1, 2, 3, 5, 7]:
        phase = rng.uniform(0, 2 * np.pi)
        perts.append(np.sin(2 * np.pi * freq * np.arange(n) / n + phase))
    for src1 in range(0, n, max(1, n // 8)):
        src2 = (src1 + n // 2) % n
        p = np.zeros(n)
        p[src1] = 1.0
        p[src2] = -1.0
        perts.append(p)
    perts.append(rng.randn(n))
    pert = scale * sum(perts) / len(perts)
    return project_feasible(h + pert, target)
