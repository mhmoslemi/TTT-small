Now let me also think about using the "top-k" approach instead of the full smooth max. The top-k approach is:
1. Find the k largest correlation values
2. Compute the gradient as a weighted average of their individual gradients
3. The weights can be uniform (simple average) or softmax-based

The advantage of top-k is that it's faster (only compute gradients for k lags instead of all 2n-1). For k=3-5 and n=250, this is much faster.

Let me use the top-k approach with softmax weighting:

```python
def topk_grad(h, dx, k=5, beta=50):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    # Get top k
    idx = np.argpartition(c, -k)[-k:]
    idx = idx[np.argsort(c[idx])[::-1]]
    vals = c[idx]
    # Softmax weights
    shifted = beta * (vals - vals[0])
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    
    grad = np.zeros(n)
    center = n - 1
    for i in range(k):
        wk = w[i]
        lag = int(idx[i]) - center
        if lag >= 0:
            end = n - lag
            grad[:end] += wk * (1.0 - h[lag:])
            grad[lag:] += wk * (-h[:end])
        else:
            l = -lag
            grad[l:] += wk * (1.0 - h[:n-l])
            grad[:n-l] += wk * (-h[l:])
    return grad * dx
```

This is O(n*k) per gradient instead of O(n^2), which is much faster for large n.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.array(h, dtype=float).copy()
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
    return h

def resample(h, n_new):
    target = n_new / 2.0
    if len(h) == n_new:
        return project_feasible(h, target)
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h), target)

def topk_subgrad(h, n, dx, k, beta):
    """Compute weighted subgradient of max correlation using top-k lags."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    if k >= len(c):
        idx = np.argsort(c)[::-1]
    else:
        idx = np.argpartition(c, -k)[-k:]
        idx = idx[np.argsort(c[idx])[::-1]]
    vals = c[idx]
    # Softmax weights (beta controls sharpness)
    shifted = beta * (vals - vals[0])
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    
    grad = np.zeros(n)
    center = n - 1
    for i in range(len(idx)):
        wk = w[i]
        if wk < 1e-10:
            continue
        lag = int(idx[i]) - center
        if lag >= 0:
            end = n - lag
            if end > 0:
                grad[:end] += wk * (1.0 - h[lag:])
            grad[lag:n] += wk * (-h[:max(end,0)])
        else:
            l = -lag
            if n - l > 0:

                grad[:n-l] += wk * (-h[l:])
    return grad * dx

                  k_range=(3, 8), beta_range=(30, 300), max_iters=50000):
    """Main projected gradient loop with adaptive k and beta."""
    cur = project_feasible(h.copy(), target)
    cur_c5 = compute_c5(cur)
    if cur_c5 < best_state[0]:
        best_state[0] = cur_c5
        best_state[1] = cur.copy()
        best_state[2] = n
    
    alpha = 0.08
    for it in range(max_iters):
        if time_fn() < min_time:
            break
        
        # Adaptive k and beta
        progress = min(1.0, it / (max_iters * 0.5))
        k = int(k_range[0] + (k_range[1] - k_range[0]) * progress)
        k = max(k, 2)
        beta = beta_range[0] + (beta_range[1] - beta_range[0]) * progress
        
        g = topk_subgrad(cur, n, dx, k, beta)
        g = g - g.mean()  # Project onto hyperplane sum=0
        
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        
        direction = -g / gnorm
        
        # Backtracking line search
        improved = False
        for ls in range(50):
            step = alpha
            h_t = project_feasible(cur + step * direction, target)
            c5_t = compute_c5(h_t)
            if c5_t < cur_c5 - 1e-15:
                cur = h_t
                cur_c5 = c5_t
                alpha = min(step * 1.3, 0.5)
                improved = True
                if cur_c5 < best_state[0]:
                    best_state[0] = cur_c5
                    best_state[1] = cur.copy()
                    best_state[2] = n
                break
            step *= 0.5
            alpha = max(step * 0.7, 0.0005)

            if T > 1e-6:

                    cur = h_t

                        best_state[0] = cur_c5
                        best_state[1] = cur.copy()
                        best_state[2] = n

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    h = np.array(h, dtype=float).copy()
    for _ in range(300):
        h = np.clip(h, 0.0, 1.0)
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
    return h

def resample(h, n_new):
    target = n_new / 2.0
    if len(h) == n_new:
        return project_feasible(h, target)
    x_old = np.linspace(0, 2, len(h))
    x_new = np.linspace(0, 2, n_new)
    return project_feasible(np.interp(x_new, x_old, h), target)

def topk_subgrad(h, n, dx, k, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    if k >= len(c):
        idx = np.argsort(c)[::-1]
    else:
        idx = np.argpartition(c, -k)[-k:]
        idx = idx[np.argsort(c[idx])[::-1]]
    vals = c[idx]
    shifted = beta * (vals - vals[0])
    exp_v = np.exp(shifted)
    w = exp_v / exp_v.sum()
    grad = np.zeros(n)
    center = n - 1
    for i in range(len(idx)):
        wk = w[i]
        if wk < 1e-10:
            continue
        lag = int(idx[i]) - center
        if lag >= 0:
            end = n - lag
            if end > 0:
                grad[:end] += wk * (1.0 - h[lag:])
            grad[lag:] += wk * (-h[:end])
        else:
            l = -lag
            if n - l > 0:
                grad[l:] += wk * (1.0 - h[:n-l])
                grad[:n-l] += wk * (-h[l:])
    return grad * dx
