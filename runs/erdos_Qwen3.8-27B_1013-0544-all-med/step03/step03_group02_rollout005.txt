def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sml_obj(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * dx

def sml_grad(h_in, beta, dx):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    m = c.max()
    x_shift = np.clip(beta * (c - m), -500, 500)
    w = np.exp(x_shift)
    w /= w.sum()
    grad = np.zeros(len(h_in))
    n = len(h_in)
    for idx in range(len(c)):
        if w[idx] < 1e-12:
            continue
        lag = idx - (n - 1)
        if lag >= 0:
            grad[:n - lag] += w[idx] * (1.0 - h_in[lag:])
            grad[lag:] -= w[idx] * h_in[:n - lag]
        else:
            la = -lag
            grad[la:] += w[idx] * (1.0 - h_in[:n - la])
            grad[:n - la] -= w[idx] * h_in[la:]
    return grad * dx

    return np.sum(x_in) - target

def sum_eq_jac(x_in, target):
    return np.ones(x_in.shape)

                     args=(beta, 2.0/n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})

    except Exception:
        return h0

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def sum_eq(x, target):
    return np.sum(x) - target

def sum_eq_jac(x, target):
    return np.ones(x.shape)

def corr_grad_single(h, lag, n):
    """Gradient of corr[lag+(n-1)] wrt h"""
    g = np.zeros(n)
    if lag >= 0:
        g[:n - lag] += (1.0 - h[lag:])
        g[lag:] -= h[:n - lag]
    else:
        la = -lag
        g[la:] += (1.0 - h[:n - la])
        g[:n - la] -= h[la:]
    return g

def active_set_opt(h0, target, n, S, beta, maxiter, ftol, t0, budget):
    """Optimize smooth max over active set S only"""
    dx = 2.0 / n

        c = get_corr(x)
        active = c[S] * dx
        return smooth_max(active, beta)

        c = get_corr(x)
        active = c[S] * dx
        m = np.max(active)
        xs = np.clip(beta * (active - m), -500, 500)
        w = np.exp(xs)
        w /= w.sum()
        grad = np.zeros(n)
        dcorr = dx
        for i, k in enumerate(S):
            if w[i] < 1e-12:
                continue
            lag = k - (n - 1)
            grad += w[i] * dcorr * corr_grad_single(x, lag, n)
        return grad

def projected_grad(h0, target, n, n_iter, step_init, t0, budget, beta=300):
    cur = h0.copy()
    best_c5 = compute_c5(cur)
    step = step_init
    mom = np.zeros(n)
    dx = 2.0 / n
    dx_idx = n - 1
    for it in range(n_iter):
        if budget - (time.time() - t0) < 2:
            break

        m = c.max()
        xs = np.clip(beta * (c - m), -500, 500)
        w = np.exp(xs)
        w /= w.sum()
        grad = np.zeros(n)
        for idx in range(len(c)):
            if w[idx] < 1e-10:
                continue

            grad += w[idx] * dx * corr_grad_single(cur, lag, n)
        gn = np.linalg.norm(grad)

                best_c5 = c_try

    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(320, 0.5)
    
    n_base = len(h_parent)
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    
    # Try multiple resolutions
    resolutions = sorted(set([n_base, n_base-1, n_base+2, n_base-3, n_base+5]))
    resolutions = [r for r in resolutions if 120 <= r <= 450]
    
    for n_pts in resolutions:
        remaining = budget_s - (time.time() - t0)
        if remaining < 50:
            break
        target = n_pts / 2.0
        
        cur = project_h(resample(best_h, n_pts), target)
        cur_c5 = compute_c5(cur)
        if cur_c5 < best_c5:
            best_h, best_c5 = cur.copy(), cur_c5
        
        # Phase 1: Active-set optimization (iteratively identify and optimize top lags)
        for phase in range(6):
            remaining = budget_s - (time.time() - t0)
            if remaining < 15:
                break
            c = get_corr(cur) * (2.0 / n_pts)
            top_k = max(3, min(12, len(c) // 30))
            S = np.argsort(c)[::-1][:top_k]
            beta = 200 + phase * 300
            cur = active_set_opt(cur, target, n_pts, S, beta, 2000, 1e-14, t0, budget_s)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5
            # Check if new lags became active
            c2 = get_corr(cur) * (2.0 / n_pts)
            S2 = np.argsort(c2)[::-1][:top_k]
            if set(S2.tolist()) == set(S.tolist()):
                break
        
        # Phase 2: Full smooth max with high beta
        remaining = budget_s - (time.time() - t0)
        if remaining > 20:
            cur = full_sml_opt(cur, target, n_pts, 800, 3000, 1e-14, t0, budget_s)
            cur_c5 = compute_c5(cur)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5
        
        # Phase 3: Projected gradient
        remaining = budget_s - (time.time() - t0)
        if remaining > 30:
            cur, cur_c5 = projected_grad(cur, target, n_pts, 6000, 0.015, t0, budget_s)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5
        
        # Phase 4: Coordinate descent
        remaining = budget_s - (time.time() - t0)
        if remaining > 15:
            cur, cur_c5 = coord_descent(cur, target, n_pts, t0, budget_s, max_sweeps=35)
            if cur_c5 < best_c5:
                best_h, best_c5 = cur.copy(), cur_c5
        
        # Phase 5: Structured perturbations
        remaining = budget_s - (time.time() - t0)
        if remaining > 30:
            for att in range(3):
                remaining = budget_s - (time.time() - t0)
                if remaining < 12:
                    break
                scale = 0.012 / (att + 1)
                pert = np.random.randn(n_pts) * scale
                # Add low-frequency structure
                freq = np.random.choice([2, 3, 4, 5])
                pert += 0.006 * np.sin(2 * np.pi * freq * np.linspace(0, 1, n_pts))
                hp = project_h(cur + pert, target)
                c_hp = get_corr(hp) * (2.0 / n_pts)
                top = np.argsort(c_hp)[::-1][:8]
                hp = active_set_opt(hp, target, n_pts, top, 500, 1500, 1e-13, t0, budget_s)
                hp, cp = projected_grad(hp, target, n_pts, 2000, 0.01, t0, budget_s)
                if cp < best_c5:
                    best_h, best_c5 = hp.copy(), cp
                if cp < cur_c5:
                    cur, cur_c5 = hp, cp
    
    # Final polish at best resolution
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        nf = len(best_h)
        tf = nf / 2.0
        # High beta active set
        c_f = get_corr(best_h) * (2.0 / nf)
        top_f = np.argsort(c_f)[::-1][:10]
        best_h = active_set_opt(best_h, tf, nf, top_f, 3000, 5000, 1e-15, t0, budget_s)
        best_c5 = compute_c5(best_h)
        
        remaining = budget_s - (time.time() - t0)
        if remaining > 8:
            best_h, best_c5 = projected_grad(best_h, tf, nf, 4000, 0.005, t0, budget_s, beta=500)
        
        remaining = budget_s - (time.time() - t0)
        if remaining > 4:
            best_h, best_c5 = coord_descent(best_h, tf, nf, t0, budget_s, max_sweeps=30)
    
    return best_h, best_c5, len(best_h)
