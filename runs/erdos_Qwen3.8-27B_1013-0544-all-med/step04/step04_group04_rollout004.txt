def swap_delta_vec(h, a, b):
    n = len(h)
    delta = np.zeros(2 * n - 1)
    lags = np.arange(-(n-1), n)
    
    # h[a]: 1->0
    # First factor: valid when 0 <= a+lag < n, i.e., -a <= lag < n-a
    valid_a1 = (a + lags >= 0) & (a + lags < n)
    delta[valid_a1] -= (1 - h[a + lags[valid_a1]])
    
    # Second factor: valid when 0 <= a-lag < n, i.e., a-(n-1) <= lag <= a
    valid_a2 = (a - lags >= 0) & (a - lags < n)
    delta[valid_a2] += h[a - lags[valid_a2]]
    
    # h[b]: 0->1
    valid_b1 = (b + lags >= 0) & (b + lags < n)
    delta[valid_b1] += (1 - h[b + lags[valid_b1]])
    
    valid_b2 = (b - lags >= 0) & (b - lags < n)
    delta[valid_b2] -= h[b - lags[valid_b2]]
    
    return delta
