    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    m = np.max(x)
    x_shifted = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shifted)) / len(x)) / beta

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def obj_fn(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter=5000, ftol=1e-15):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(obj_fn, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def coord_desc(h, target, n, t0, budget, sweeps=15):
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(7)
    steps = (0.03, -0.03, 0.015, -0.015, 0.008, -0.008, 0.004, -0.004)
    for sw in range(sweeps):
        if time.time() - t0 > budget - 1:
            break
        improved = False
        for j in rng.permutation(n):
            old = h[j]
            for d in steps:
                nv = old + d
                if nv < 0 or nv > 1:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h, best_c5, improved = trial, c, True
                    break
        if not improved:
            break
    return h, best_c5

def pair_desc(h, target, n, count, t0, budget, seed=99):
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    for _ in range(count):
        if time.time() - t0 > budget - 1:
            break
        j, k = rng.choice(n, 2, replace=False)
        for d in (0.04, -0.04, 0.02, -0.02, 0.01, -0.01):
            trial = h.copy()
            trial[j] += d
            trial[k] -= d
            trial = project_h(trial, target)
            c = compute_c5(trial)
            if c < best_c5 - 1e-15:
                h, best_c5 = trial, c
                break
    return h, best_c5

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new, endpoint=False),
                     np.linspace(0, 2, len(h), endpoint=False), h)

def basin_hopping(h, target, n, bounds, t0, budget, n_restarts=8, sigma=0.04):
    best_h, best_c5 = h.copy(), compute_c5(h)
    rng = np.random.RandomState(55)
    for r in range(n_restarts):
        rem = budget - (time.time() - t0)
        if rem < 35:
            break
        perturbed = best_h + rng.normal(0, sigma, n)
        perturbed = project_h(perturbed, target)
        cur = do_slsqp(perturbed, n, target, 1500, bounds, 2000, 1e-13)
        c = compute_c5(cur)
        if c < best_c5:
            best_h, best_c5 = cur, c
            # Strengthen on success
            cur2 = do_slsqp(best_h, n, target, 8000, bounds, 3000, 1e-14)
            c2 = compute_c5(cur2)
            if c2 < best_c5:
                best_h, best_c5 = cur2, c2
    return best_h, best_c5

def binary_flip_search(h, target, n, t0, budget, n_flips=200):
    best = h.copy()
    best_c5 = compute_c5(best)
    rng = np.random.RandomState(33)
    for _ in range(n_flips):
        if time.time() - t0 > budget - 1:
            break
        j = rng.randint(n)
        trial = best.copy()
        trial[j] = 1.0 - trial[j]
        trial = project_h(trial, target)
        c = compute_c5(trial)
        if c < best_c5 - 1e-15:
            best, best_c5 = trial, c
    return best, best_c5
