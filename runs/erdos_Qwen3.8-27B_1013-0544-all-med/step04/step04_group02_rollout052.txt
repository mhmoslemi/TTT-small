import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_subgradient(h, corr, n, top_k, temperature):
    """Compute weighted subgradient using softmax over top lags."""
    # corr has length 2n-1, indexed 0 to 2n-2
    # lag for index k is k - (n-1)
    # Find top_k lags
    top_indices = np.argsort(corr)[-top_k:]
    weights = np.exp(temperature * (corr[top_indices] - corr[top_indices].max()))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for w, k in zip(weights, top_indices):
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            idx = np.arange(m)
            grad[idx] += w * (1.0 - h[idx + lag])
            grad[idx + lag] -= w * h[idx]
        else:
            la = -lag
            m = n - la
            idx = np.arange(m)
            grad[idx + la] += w * (1.0 - h[idx])
            grad[idx] -= w * h[idx + la]
    return grad

def accelerated_gradient(h, n, deadline, seed, max_iter=5000):
    """FISTA-like accelerated projected gradient with adaptive temperature."""
    np.random.seed(seed)
    ts = n / 2.0
    h_cur = h.copy()
    h_prev = h.copy()
    c5 = compute_c5(h_cur)
    t_k = 1.0
    momentum = 0.0  # We'll use FISTA momentum
    lr = 1.0 / (n * 2.0)  # initial learning rate
    
    stagnant = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        top_k = min(20, n)
        # Adaptive temperature: start moderate, increase over time
        temp = 5.0 + 50.0 * (it / max_iter)
        grad = weighted_subgradient(h_cur, corr, n, top_k, temp)
        
        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
        momentum = (t_k - 1.0) / t_new
        y = h_prev + momentum * (h_prev - h_cur)
        y = np.clip(y, 0.0, 1.0)
        
        # Gradient step on y
        h_new = project_feasible(y - lr * grad, ts)
        
        # Line search (backtracking)
        c5_new = compute_c5(h_new)
        attempts = 0
        while c5_new >= c5 - 1e-12 and attempts < 10:
            lr *= 0.5
            h_new = project_feasible(y - lr * grad, ts)
            c5_new = compute_c5(h_new)
            attempts += 1
        
        if c5_new < c5 - 1e-12:
            h_prev = h_cur
            h_cur = h_new
            c5 = c5_new
            t_k = t_new
            stagnant = 0
        else:

            lr *= 0.9
            momentum = 0.0
            t_k = 1.0

                h_prev = h_cur.copy()
                c5 = compute_c5(h_cur)
                stagnant = 0
                if lr < 1e-8:
                    break
    
    return h_cur, c5

def smooth_max_c5(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        diff = target_sum - s
        mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_lag(h, lag, n):
    """Subgradient of corr at a specific lag."""
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            i = np.arange(m)
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] -= h[i]
    else:
        la = -lag
        m = n - la
        if m > 0:
            i = np.arange(m)
            grad[i + la] += (1.0 - h[i])
            grad[i] -= h[i + la]
    return grad

def weighted_subgrad(h, corr, n, top_k, temp):
    """Weighted subgradient using softmax over top lags."""
    top_idx = np.argsort(corr)[-top_k:]
    w = np.exp(temp * (corr[top_idx] - corr[top_idx].max()))
    w /= w.sum()
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        grad += wi * subgrad_lag(h, k - (n - 1), n)
    return grad

    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac, 'args': (ts,)}]
    for beta in [100, 500, 2000, 10000, 50000, 200000]:
        if time.time() > deadline:
            break
        for attempt in range(2):
            if time.time() > deadline:
                break
            x0 = h.copy()
            if attempt == 1:
                x0 = project_feasible(h + 0.005 * np.random.randn(n), ts)
            try:
                res = minimize(_obj_smooth, x0, args=(dx, beta), method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 3000, 'ftol': 1e-15})
                h_new = project_feasible(res.x, ts)
                c5_new = compute_c5(h_new)
                if c5_new < c5:
                    h = h_new
                    c5 = c5_new
            except Exception:
                pass
    return h, c5

def micro_polish(h, n, deadline, seed):
    np.random.seed(seed)
    ts = n / 2.0
    c5 = compute_c5(h)
    for it in range(50000):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = np.argmax(corr)
        lag = k_max - (n - 1)
        grad = subgrad_lag(h, lag, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad = grad / gnorm
        step = 0.01 / n
        for _ in range(30):
            h_new = project_feasible(h - step * gnorm * grad, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-14:
                h = h_new
                c5 = c5_new
                step = min(step * 1.05, 1.0 / n)
                break
            step *= 0.5
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    parent = None
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=float).copy()
    if parent is None:
        parent = np.full(200, 0.5)
    pn = len(parent)

    # Phase 1: Multi-resolution gradient (50%)
    d1 = t0 + budget_s * 0.50
    resolutions = sorted(set([pn, max(150, pn - 50), pn + 50, max(200, pn + 100)]))
    resolutions = [r for r in resolutions if 80 <= r <= 600]
    for nr in resolutions:
        if time.time() > d1:
            break
        x_old = np.linspace(0, 2, pn, endpoint=False)
        x_new = np.linspace(0, 2, nr, endpoint=False)
        h0 = np.interp(x_new, x_old, parent)
        h0 = project_feasible(h0, nr / 2.0)
        h1, c1 = grad_optimize(h0, nr, d1, seed + nr)
        if c1 < best_c5:
            best_h, best_c5, best_n = h1.copy(), c1, nr

    # Phase 2: SLSQP polish (35%)
    d2 = t0 + budget_s * 0.83
    if best_h is not None and time.time() < d2:
        best_h, best_c5 = slsqp_polish(best_h, best_n, d2, seed + 333)

    # Phase 3: Additional gradient from slightly perturbed start (5%)
    d3 = t0 + budget_s * 0.90
    if best_h is not None and time.time() < d3:
        np.random.seed(seed + 777)
        h_pert = project_feasible(best_h + 0.01 * np.random.randn(best_n), best_n / 2.0)
        h3, c3 = grad_optimize(h_pert, best_n, d3, seed + 444, max_iter=5000)
        if c3 < best_c5:
            best_h, best_c5 = h3.copy(), c3

    # Phase 4: Micro polish on active lag (10%)
    d4 = t0 + budget_s * 0.98
    if best_h is not None and time.time() < d4:
        best_h, best_c5 = micro_polish(best_h, best_n, d4, seed + 666)

    if best_h is None:
        best_h, best_n = parent.copy(), pn
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
