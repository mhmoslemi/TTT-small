import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(h, beta, top_k=50):
    """Compute gradient of smooth_max(corr, beta) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted)

    # Only use top-k lags for efficiency
    top_idx = np.argsort(weights)[::-1][:top_k]
    weights = weights[top_idx]
    
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        if w < 1e-10:
            break
        lag = int(idx) - (n - 1)
        L = abs(lag)
        if L == 0:
            grad += w * (1 - 2 * h)
        else:

            if L < n:
                grad[L:] += w * (1 - h[:n-L])
                grad[:n-L] -= w * h[L:]
    
    return grad * dx, float(m * dx)

def projected_grad_descent(h, target, n, max_iter, t0, budget, beta=1000, lr=0.01, top_k=50):

    cur = h.copy()
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    momentum = np.zeros(n)
    mu = 0.9  # momentum coefficient

def smooth_max_grad(h, beta, top_k=64):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")

    w_sum = w_sel.sum()

def slsqp_step(h0, target, n, beta, maxiter=10000):

        m = np.max(c)
        s = np.clip(beta * (c - m), -400, 400)
        return (m + np.log(np.sum(np.exp(s)) / len(s)) / beta) * (2.0 / n)
    
    bounds = [(0.0, 1.0)] * n

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-13) & (h < 1.0 - 1e-13)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smax_grad(h, beta, top_k=64):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -400, 400)
    w = np.exp(sh)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n), 0.0
    w /= ws
    idx = np.argsort(w)[::-1][:top_k]
    sel = w[idx]
    ss = sel.sum()
    if ss < 1e-300:
        return np.zeros(n), 0.0
    sel /= ss
    grad = np.zeros(n)
    for i in range(len(idx)):
        wi = sel[i]
        if wi < 1e-12:
            continue
        lag = int(idx[i]) - (n - 1)
        L = abs(lag)
        if L == 0:
            grad += wi * (1.0 - 2.0 * h)
        elif L < n:
            grad[L:] += wi * (1.0 - h[:n - L])
            grad[:n - L] -= wi * h[L:]

    return best_h, best_c5, n
