            n_best, target_best = n_try, t_t
            best_h, best_c5 = subgrad_polish(best_h, target_best, 7, 500, t0, budget_s)

    # Phase 4: Structured restarts
    for rst in range(6):
        if tl() < 100:
            break
        if rst == 0:
            h_try = best_h.copy()
        elif rst == 1:
            h_try = project_sum(best_h + np.random.randn(n_best) * 0.04, target_best)
        elif rst == 2:
            sz = max(n_best // 8, 4)
            st = np.random.randint(0, n_best - sz)
            h_try = best_h.copy()
            h_try[st:st + sz] = 1.0 - best_h[st:st + sz]
            h_try = project_sum(h_try, target_best)
        elif rst == 3:
            h_try = project_sum(1.0 - best_h, target_best)
        elif rst == 4:
            h_try = sharpen(best_h, target_best, 0.3)
        else:
            fr = np.random.randint(3, 12)
            ph = np.random.random() * 2.0
            p = 0.03 * np.cos(np.pi * fr * np.arange(n_best) / n_best + ph)
            h_try = project_sum(best_h + p, target_best)

        h_new, c5_new = fw_optimize(h_try, target_best, 200, t0, budget_s)
        if tl() > 40 and c5_new <= best_c5 + 0.004:
            h_new = slsqp_run(h_new, target_best, 8000, 2000)
            c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 5: Final intensive polish
    best_h, best_c5 = subgrad_polish(best_h, target_best, 8, 10000, t0, budget_s)

    # Phase 6: Final SLSQP squeeze
    if tl() > 25:
        h_new = slsqp_run(best_h, target_best, 100000, 4000)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        best_h, best_c5 = subgrad_polish(best_h, target_best, 5, 3000, t0, budget_s)

    # Phase 7: Micro-polish with very small steps
    if tl() > 5:
        n_f = len(best_h)
        t_f = n_f / 2.0
        for it in range(3000):
            if tl() < 2:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            top_idx = np.argsort(corr)[::-1][:4]
            g = np.zeros(n_f)
            for idx in top_idx:
                lag = int(idx) - (n_f - 1)
                if lag == 0:
                    g += (1.0 - 2.0 * best_h)
                elif lag > 0:
                    g[lag:] += (1.0 - best_h[:-lag])
                    g[:n_f - lag] -= best_h[lag:]
                else:
                    m2 = -lag
                    g[:n_f - m2] += (1.0 - best_h[m2:])
                    g[m2:] -= best_h[:n_f - m2]
            gn = np.linalg.norm(g)
            if gn < 1e-15:
                break
            g /= gn
            for step in [0.005, 0.002, 0.001, 0.0004, 0.0001]:
                ht = project_sum(best_h - step * g, t_f)
                c5t = compute_c5(ht)
                if c5t < best_c5 - 1e-16:
                    best_h = ht
                    best_c5 = c5t
                    break

    return best_h, best_c5, len(best_h)
