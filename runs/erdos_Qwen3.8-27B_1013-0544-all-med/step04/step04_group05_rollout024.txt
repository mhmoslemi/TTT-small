    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(best)
    corr = np.correlate(best, 1.0 - best, mode="full")
    step = step0
    vel = np.zeros(n)
    
    for it in range(max_iters):
        if time_fn() < 2:
            break
        # Find active lags
        top = np.argsort(corr)[::-1][:n_active]
        lags = [int(i) - (n - 1) for i in top]
        
        # Average subgradient
        g = np.zeros(n)
        for lag in lags:
            g += subgrad_for_lag(best, lag)
        g /= len(lags)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        
        # Momentum
        vel = 0.85 * vel + 0.15 * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        
        # Line search
        improved = False
        for s in [step, step*0.5, step*0.25, step*0.125]:
            ht = project_h(best + s * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                corr = np.correlate(best, 1.0 - best, mode="full")
                improved = True
                break
        if not improved:
            step *= 0.5
            if step < step0 * 1e-4:
                # Perturb
                p = np.random.randn(n) * 0.003
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    corr = np.correlate(best, 1.0 - best, mode="full")
                step = step0
        else:
            step = min(step * 1.01, step0 * 2)
    
    return best, best_c5

def coord_descent(h, target, time_fn, max_passes, n_vals=5):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vals = np.linspace(0, 1, n_vals)  # candidate values for each coordinate
    
    for p in range(max_passes):
        if time_fn() < 3:
            break
        improved = False
        for i in range(n):
            old = best[i]
            best_c5_i = best_c5  # current max
            for v in vals:
                if abs(v - old) < 1e-10:
                    continue
                best[i] = v
                # Adjust sum: need to fix the constraint
                diff = target - best.sum()
                if abs(diff) < 1e-12:
                    c5t = compute_c5(best)
                else:
                    # Quick check: skip if diff is too large to fix easily
                    best[i] = old
                    break
                if c5t < best_c5 - 1e-14:
                    best_c5 = c5t
                    improved = True
                    break
            else:
                best[i] = old
        if not improved:
            break
    return best, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_for_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += (1.0 - h[:-lag])
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g

def smooth_max(x, beta):
    m = np.max(x)
    xs = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(xs)) / len(x)) / beta

def slsqp_phase(h_init, target, time_fn, betas, max_iter_each):
    n = len(h_init)
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    h_cur = h_init.copy()
    for beta in betas:
        if time_fn() < 10:
            break
        def obj(h_in):
            return smooth_max(np.correlate(h_in, 1.0 - h_in, mode="full"), beta) * dx
        def cfun(h_in):
            return h_in.sum() - target
        def cjac(h_in):
            return np.ones(n)
        try:
            res = minimize(obj, h_cur, method='SLSQP', bounds=bounds,
                          constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                          options={'maxiter': max_iter_each, 'ftol': 1e-18})
            h_new = project_h(res.x, target)
            if compute_c5(h_new) < compute_c5(h_cur):
                h_cur = h_new
        except Exception:
            pass
        h_cur = project_h(h_cur, target)
    return h_cur

    n = len(h_init)
    dx = 2.0 / n
    best = h_init.copy()

    corr = np.correlate(best, 1.0 - best, mode="full")
    step = step0
    vel = np.zeros(n)
    stall = 0
    for it in range(max_iters):
        if time_fn() < 2:
            break
        top = np.argsort(corr)[::-1][:n_active]
        lags = [int(i) - (n - 1) for i in top]
        g = np.zeros(n)
        for lag in lags:
            g += subgrad_for_lag(best, lag)
        g /= len(lags)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        improved = False
        for s in [step, step * 0.5, step * 0.25, step * 0.1]:
            ht = project_h(best + s * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-16:

        if improved:

        else:

    return best, best_c5

def coord_descent(h_init, target, time_fn, max_passes):
    n = len(h_init)
    best = h_init.copy()
    best_c5 = compute_c5(best)
    for p in range(max_passes):
