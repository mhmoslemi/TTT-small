from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def binary_snap(h, n):
    n_ones = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def smooth_max(x, beta):
    m = float(np.max(x))
    x_shift = np.clip(beta * (x - m), -500, 500)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def slsqp_obj(h_in, beta, n):
    c = np.correlate(h_in, 1.0 - h_in, mode="full")
    return smooth_max(c, beta) * (2.0 / n)

def sum_constraint(x_in, target):
    return np.sum(x_in) - target

def sum_jac(x_in, target):
    return np.ones(x_in.shape)

def do_slsqp(h0, n, target, beta, maxiter=8000, ftol=1e-16):
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jac, 'args': (target,)}]
    try:
        r = minimize(slsqp_obj, h0, args=(beta, n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=32, momentum=0.9):
    cur = h.copy()
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    stall = 0
    step = step_init
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0:
                if lag < n:
                    grad[:n - lag] += (1.0 - cur[lag:])
                    grad[lag:] -= cur[:n - lag]
            else:
                la = -lag
                if la < n:
                    grad[la:] += (1.0 - cur[:n - la])
                    grad[:n - la] -= cur[la:]
        grad /= top_k

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)

        diff = target_sum - h.sum()
        if abs(diff) < 1e-14: break

Key differences I want to make:
1. **More binary annealing iterations** with better temperature schedule
2. **Higher top_k in subgradient** (use more lags for the gradient)
3. **Adaptive step size** that increases more aggressively when making progress
4. **More perturbation restarts** with different scales

Actually, I realize the main issue might be that the previous program's coordinate descent is too slow (it evaluates compute_c5 for each trial, which is O(n) for the correlate). For n=384, that's 767 operations per evaluate, and we're doing it n * len(deltas) times per sweep.

Let me optimize the coordinate descent to be faster. Instead of computing the full correlation each time, I can update only the affected lags.

When I change h[j] by delta, the correlation at lag d changes by:
- For d = 0: delta * (1-2*h[j])
- For d > 0 (positive lag): delta * (1-h[j+d]) - h[j-d]*delta (for the relevant indices)
- etc.

This is complex to implement correctly. Let me just make the coordinate descent use fewer iterations but with a smaller step size.

Actually, let me think about this differently. The main bottleneck is compute_c5, which calls np.correlate. For n=384, the full correlation has 767 elements. This is fast (microseconds). The issue is the number of evaluations.

For coordinate descent with n=384 and 10 deltas per position, one sweep is 3840 evaluations. At maybe 10μs per evaluation, that's 38ms per sweep. So 400 sweeps would be about 15 seconds. That's reasonable.

Let me focus on making the algorithm more effective rather than faster.

Actually, here's another idea: **simulated annealing on the continuous space**. Instead of just binary annealing, do annealing on the continuous values. The idea is:
1. Pick a random position j
2. Perturb h[j] by a random amount
3. Accept if the new C₅ is lower, or with probability exp(-Δ/T)

This is similar to what subgradient descent does, but with random (not gradient-based) moves.

    n = len(h)
    return float(np.max(np.correlate(h, 1.0-h, mode="full")) * 2.0/n)

def compute_lags(h):
    n = len(h)
    return np.correlate(h, 1.0-h, mode="full") * 2.0/n

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14: break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff/len(h); h = np.clip(h, 0, 1)
        else:
            h[free] += diff/free.sum(); h = np.clip(h, 0, 1)
    return h

def binary_snap(h, n):
    h_bin = np.zeros(n)
    h_bin[np.argsort(h)[::-1][:n//2]] = 1.0

    m = float(np.max(x))
    return m + np.log(np.sum(np.exp(np.clip(beta*(x-m),-500,500)))/len(x))/beta

def slsqp_obj(h_in, beta, n):
    return smooth_max(np.correlate(h_in, 1.0-h_in, mode="full"), beta) * (2.0/n)
