            if s < 1e-14:

    if k >= 0:

        grad[valid_j2] -= h[:n-k]  # h[j-k] for j in valid_j2
    return dx * grad

import numpy as np

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def subgrad_gk(h, n, shift):
    dx = 2.0 / n
    k = abs(int(shift))
    if k == 0:
        return dx * (1.0 - 2.0 * h)
    if shift > 0:
        grad = np.zeros(n)
        grad[:n-k] += (1.0 - h[k:])
        grad[k:] -= h[:n-k]
    else:
        grad = np.zeros(n)
        grad[n-k:] += (1.0 - h[:k])
        grad[:n-k] -= h[n-k:]
    return dx * grad

def slsqp_smooth(h, n, alpha, maxiter, deadline):

        m = g.max()

        sm = sh.max()
        return m + np.log(np.exp(sh - sm).sum()) / alpha

    for alpha in [50, 100, 200, 500, 1000, 3000, 10000, 30000]:
        if time.time() > p1:
            break
        h = slsqp_smooth(h, n, alpha, 400, p1)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    p2 = t0 + 0.50 * (budget_s - 5)
    h, c = active_set_optimize(best_h.copy(), n, p2, max_rounds=8)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    p3 = t0 + 0.72 * (budget_s - 5)
    h, c = subgrad_polish(best_h.copy(), n, p3, max_iter=20000)
    if c < best_c:
        best_c = c
        best_h = h.copy()
    p4 = t0 + 0.87 * (budget_s - 5)
    if time.time() < p4 - 20:
        for nh in [240, 260, 300]:
            if time.time() > p4 - 15:
                break
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = project_feasible(np.interp(xn, xo, best_h, assume_sorted=True), nh)
            hh = slsqp_smooth(hh, nh, 1000, 300, p4 - 15)
            hh, c2 = subgrad_polish(hh, nh, p4 - 15, max_iter=3000)
            if c2 < best_c:
                best_c = c2
                best_h = hh.copy()
                n = nh
    if time.time() < deadline - 30:
        h, c = ternary_local_search(best_h.copy(), n, deadline - 30, max_sweeps=30)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if time.time() < deadline - 10:
        h = slsqp_smooth(best_h.copy(), n, 50000, 800, deadline - 10)
        c = compute_c5(h, n)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    if time.time() < deadline - 3:
        h, c = subgrad_polish(best_h.copy(), n, deadline, max_iter=5000)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
