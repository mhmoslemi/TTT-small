    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(80):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    x_shift = np.clip(beta * (x - m), -400, 400)
    return m + np.log(np.sum(np.exp(x_shift)) / len(x)) / beta

def compute_subgrad(h, lags):
    n = len(h)

            grad += (1.0 - 2.0 * h)

            grad[:n - lag] -= h[lag:]
        else:

            grad[:n - m] += (1.0 - h[m:])

    return grad

                best_c5 = c5x

    return best, best_c5

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        if abs(diff) < 1e-13:
            break

    m = np.max(x)

        h4 = project_h(np.interp(x_new, x_old, best_h), target4)
        if rem() > 80:
            bounds4 = Bounds(np.zeros(n4), np.ones(n4))
            for beta in [300, 1200]:
                if rem() < 30:
                    break
                def obj_fn4(h_in):
                    c = np.correlate(h_in, 1.0 - h_in, mode="full")
                    return smooth_max(c, beta) * dx4
                def con_fn4(h_in):
                    return h_in.sum() - target4
                def jac_fn4(h_in):
                    return np.ones(n4)
                try:
                    res = minimize(obj_fn4, h4, method='SLSQP', bounds=bounds4,
                                  constraints=[{'type': 'eq', 'fun': con_fn4, 'jac': jac_fn4}],
                                  options={'maxiter': 4000, 'ftol': 1e-20})
                    h4 = project_h(res.x, target4)
                except Exception:
                    pass
        if rem() > 50:
            h4, c5_4 = accel_subgrad(
                h4, target4, rem,
                max_iters=60000, threshold=0.003,
                step_init=0.005, momentum=0.93,
                perturb_every=500, perturb_scale=0.003)
            if c5_4 < best_c5:
                best_h = h4
                best_c5 = c5_4

    # Phase 5: Final SLSQP polish at best resolution
    nf = len(best_h)
    tf = nf / 2.0
    dxf = 2.0 / nf
    if rem() > 25:
        bounds_f = Bounds(np.zeros(nf), np.ones(nf))
        for beta in [5000, 15000]:
            if rem() < 12:
                break
            def obj_fnf(h_in):
                c = np.correlate(h_in, 1.0 - h_in, mode="full")
                return smooth_max(c, beta) * dxf
            def con_fnf(h_in):
                return h_in.sum() - tf
            def jac_fnf(h_in):
                return np.ones(nf)
            try:
                res = minimize(obj_fnf, best_h, method='SLSQP', bounds=bounds_f,
                              constraints=[{'type': 'eq', 'fun': con_fnf, 'jac': jac_fnf}],
                              options={'maxiter': 8000, 'ftol': 1e-20})
                h_new = project_h(res.x, tf)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_h = h_new
                    best_c5 = c5_new
            except Exception:
                pass

    # Phase 6: Final tight subgradient
    if rem() > 5:
        best_h, best_c5 = accel_subgrad(
            best_h, tf, rem,
            max_iters=40000, threshold=0.001,
            step_init=0.004, momentum=0.94,
            perturb_every=800, perturb_scale=0.002)

    return best_h, best_c5, len(best_h)
