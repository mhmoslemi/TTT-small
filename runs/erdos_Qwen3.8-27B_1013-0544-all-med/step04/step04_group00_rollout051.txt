import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-12:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-12:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def binary_sa(h, n, deadline, T0=0.01, T_end=0.0001, max_iter=200000):
    """Simulated annealing on binary solutions."""
    h = h.copy()
    c = compute_c5(h, n)
    best_h = h.copy()
    best_c = c
    T = T0
    n_ones = sum(h > 0.5)
    ones = set(np.where(h > 0.5)[0].tolist())
    zeros = set(np.where(h <= 0.5)[0].tolist())
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        T = T0 * (T_end / T0) ** (it / max(1, max_iter))

        i = next(iter(ones))
        j = next(iter(zeros))
        # Actually pick randomly
        i = list(ones)[np.random.randint(len(ones))]
        j = list(zeros)[np.random.randint(len(zeros))]
        
        # Swap
        h[i], h[j] = 0.0, 1.0
        c_new = compute_c5(h, n)
        
        if c_new < c or np.random.random() < np.exp(-(c_new - c) / max(T, 1e-10)):
            c = c_new
            ones.remove(i)
            ones.add(j)
            zeros.add(i)
