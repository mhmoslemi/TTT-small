```python
import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def resample_fn(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, len(h), endpoint=False)
    new_x = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def smooth_max_obj(h, args):
    beta, n, penalty, target = args
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    val = (m + np.log(np.sum(np.exp(x)) / len(corr)) / beta) * (2.0 / n)
    s = np.sum(h) - target
    val += penalty * s * s
    return val

def smooth_max_grad(h, args):
    beta, n, penalty, target = args
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x = np.clip(beta * (corr - m), -500, 500)
    w = np.exp(x)
    w /= w.sum()
    dx = 2.0 / n
    active = np.where(w > 1e-8)[0]
    h_grad = np.zeros(n)
    for k in active:
        wk = w[k] * dx
        lag = int(k) - (n - 1)
        if lag >= 0:
            if lag < n:
                h_grad[lag:] += wk * (1.0 - h[:n - lag])
                h_grad[:n - lag] += wk * (-h[lag:])
        else:
            la = -lag
            if la < n:
                h_grad[la:] += wk * (1.0 - h[:n - la])
                h_grad[:n - la] += wk * (-h[la:])
    h_grad += 2.0 * penalty * (np.sum(h) - target)
    return h_grad

def do_lbfgsb(h0, n, target, beta, penalty, maxiter=2000, t0=None, budget=0):
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = project_h(h0, target)
    bounds = Bounds(np.zeros(n), np.ones(n))
    args = (beta, n, penalty, target)
    try:
        r = minimize(smooth_max_obj, h0, jac=smooth_max_grad, args=args,
                     method='L-BFGS-B', bounds=bounds,
                     options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-10})
        h = project_h(r.x, target)
    except Exception:
        h = h0
    if t0 is not None and budget - (time.time() - t0) < 2:
        return h
    return project_h(h, target)

def do_slsqp(h0, n, target, beta, maxiter=5000, t0=None, budget=0):
    h0 = project_h(h0, target)
    bounds = Bounds(np.zeros(n), np.ones(n))
    args = (beta, n, n, target)
    rs = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        x = np.clip(beta * (corr - m), -500, 500)
        return (m + np.log(np.sum(np.exp(x)) / len(corr)) / beta) * rs
    def grad(h):
        corr = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(corr)
        x = np.clip(beta * (corr - m), -500, 500)
        w = np.exp(x)
        w /= w.sum()
        active = np.where(w > 1e-8)[0]
        hg = np.zeros(n)
        for k in active:
            wk = w[k] * rs
            lag = int(k) - (n - 1)
            if lag >= 0 and lag < n:
                hg[lag:] += wk * (1.0 - h[:n - lag])
                hg[:n - lag] += wk * (-h[lag:])
            elif lag < 0:
                la = -lag
                if la < n:
                    hg[la:] += wk * (1.0 - h[:n - la])
                    hg[:n - la] += wk * (-h[la:])
        return hg
    def cfun(h):
        return np.sum(h) - target
    def cjac(h):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        r = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        return project_h(r.x, target)
    except Exception:
        return h0

def binary_swap_anneal(h_bin, n, t0, budget, n_iter=500000, rng=None):
    if rng is None:
        rng = np.random.default_rng(123)
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur_c5
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        if it % 1000 == 0:
            # 2-for-2 swap
            a1, a2 = rng.choice(ones, size=2, replace=False)
            b1, b2 = rng.choice(zeros, size=2, replace=False)
            h[a1] = h[a2] = 0.0
            h[b1] = h[b2] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < cur_c5:
                cur_c5 = new_c5
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                frac = it / max(n_iter, 1)
                temp = 0.15 * (0.001) ** frac
                if rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-15)):
                    cur_c5 = new_c5
                else:
                    h[a1] = h[a2] = 1.0
                    h[b1] = h[b2] = 0.0
        else:
            # Single swap
            a = ones[rng.integers(len(ones))]
            b = zeros[rng.integers(len(zeros))]
            h[a] = 0.0
            h[b] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < cur_c5:
                cur_c5 = new_c5
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                frac = it / max(n_iter, 1)
                temp = 0.15 * (0.001) ** frac
                if rng.random() < np.exp(-(new_c5 - cur_c5) / max(temp, 1e-15)):
                    cur_c5 = new_c5
                else:
                    h[a] = 1.0
                    h[b] = 0.0
    return best_h, best_c5

def coord_descent(h, target, n, max_sweeps, t0, budget):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.05, -0.05, 0.025, -0.025, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        improved = False
        for j in range(n):
            if budget - (time.time() - t0) < 1:
                break
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c_try = compute_c5(trial)
                if c_try < best_c5 - 1e-14:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 5:
            break
    return cur, best_c5

def subgrad_desc(h, target, n, n_iter, step_init, t0, budget, top_k=20):
    cur = h.copy()
    mom = np.zeros(n)
    best_local = compute_c5(cur)
    stall = 0
    step = step_init
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag >= 0 and lag < n:
                grad[lag:] += (1.0 - cur[:n - lag])
                grad[:n - lag] += -cur[lag:]
            elif lag < 0:
                la = -lag
                if la < n:
                    grad[la:] += (1.0 - cur[:n - la])
                    grad[:n - la] += -cur[la:]
        grad /= top_k
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad /= gn
        mom = 0.9 * mom - step * grad
        trial = project_h(cur + mom, target)
        c_try = compute_c5(trial)
        if c_try < best_local - 1e-15:
            cur = trial
            best_local = c_try
            stall = 0
            step = min(step * 1.08, 0.12)
        else:
            stall += 1
            step *= 0.7
            mom *= 0.5
            if stall > 300:
                break
    return cur, best_local

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(384, 0.5)
        h_parent[:192] = 0.7
        h_parent[192:] = 0.3
        h_parent = project_h(h_parent, 192.0)

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    # Phase 1: Multi-resolution L-BFGS with penalty
    for n_res in [128, 192, 256, 384, 512]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 30:
            break
        target = n_res / 2.0
        cur = project_h(resample_fn(best_h, n_res), target)
        for beta in [200, 800, 3000, 10000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            penalty = 50.0
            cur = do_lbfgsb(cur, n_res, target, beta, penalty, maxiter=3000, t0=t0, budget=budget_s)
            c_try = compute_c5(cur)
            if c_try < best_c5:
                best_h = cur.copy()
                best_c5 = c_try
                best_n = n_res

    # Phase 2: SLSQP refinement at best resolution
    n_cur = best_n
    target = n_cur / 2.0
    for beta in [500, 2000, 8000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 15:
            break
        best_h = do_slsqp(best_h, n_cur, target, beta, maxiter=8000, t0=t0, budget=budget_s)
        best_c5 = compute_c5(best_h)

    # Phase 3: Binary annealing with 2-for-2 swaps
    remaining = budget_s - (time.time() - t0)
    if remaining > 40:
        h_bin = np.zeros(n_cur)
        h_bin[np.argsort(best_h)[::-1][:n_cur // 2]] = 1.0
        h_bin, bc5 = binary_swap_anneal(h_bin, n_cur, t0, budget_s, n_iter=300000, rng=rng)
        if bc5 < best_c5:
            best_c5 = bc5
            # Refine binary solution
            cur_ref = h_bin + 0.03 * rng.standard_normal(n_cur)
            cur_ref = project_h(cur_ref, target)
            for beta in [1000, 5000]:
                remaining = budget_s - (time.time() - t0)
                if remaining < 10:
                    break
                cur_ref = do_lbfgsb(cur_ref, n_cur, target, beta, 50.0, maxiter=3000, t0=t0, budget=budget_s)
                c_try = compute_c5(cur_ref)
                if c_try < best_c5:
                    best_h = cur_ref.copy()
                    best_c5 = c_try
            # SLSQP polish
            remaining = budget_s - (time.time() - t0)
            if remaining > 10:
                best_h = do_slsqp(best_h, n_cur, target, 5000, maxiter=6000, t0=t0, budget=budget_s)
                best_c5 = compute_c5(best_h)

    # Phase 4: Subgradient descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 30:
        best_h, best_c5 = subgrad_desc(best_h, target, n_cur, 15000, 0.03, t0, budget_s, top_k=20)

    # Phase 5: More L-BFGS after subgradient
    remaining = budget_s - (time.time() - t0)
    for beta in [3000, 15000, 50000]:
        remaining = budget_s - (time.time() - t0)
        if remaining < 10:
            break
        best_h = do_lbfgsb(best_h, n_cur, target, beta, 100.0, maxiter=4000, t0=t0, budget=budget_s)
        c_try = compute_c5(best_h)
        if c_try < best_c5:
            best_c5 = c_try

    # Phase 6: Try another resolution
    remaining = budget_s - (time.time() - t0)
    for n_alt in [448, 320]:
        if remaining < 40 or n_alt == n_cur:
            continue
        target_alt = n_alt / 2.0
        cur_alt = project_h(resample_fn(best_h, n_alt), target_alt)
        for beta in [500, 3000]:
            remaining = budget_s - (time.time() - t0)
            if remaining < 10:
                break
            cur_alt = do_lbfgsb(cur_alt, n_alt, target_alt, beta, 50.0, maxiter=3000, t0=t0, budget=budget_s)
            c_try = compute_c5(cur_alt)
            if c_try < best_c5:
                best_h = cur_alt.copy()
                best_c5 = c_try
                best_n = n_alt
                n_cur = n_alt
                target = target_alt

    # Phase 7: Coordinate descent polish
    remaining = budget_s - (time.time() - t0)
    if remaining > 10:
        best_h, best_c5 = coord_descent(best_h, target, n_cur, 150, t0, budget_s)

    # Phase 8: Diversified restarts
    remaining = budget_s - (time.time() - t0)
    for att in range(4):
        if remaining < 25:

        scale = [0.04, 0.02, 0.08, 0.015][att]
        hp = best_h + rng.standard_normal(n_cur) * scale
        hp = project_h(hp, target)
        hp = do_lbfgsb(hp, n_cur, target, 3000, 50.0, maxiter=3000, t0=t0, budget=budget_s)
        hp = do_slsqp(hp, n_cur, target, 5000, maxiter=4000, t0=t0, budget=budget_s)
        c_try = compute_c5(hp)
        if c_try < best_c5:
            best_h = hp.copy()
            best_c5 = c_try

    # Phase 9: Final ultra-precise SLSQP
    remaining = budget_s - (time.time() - t0)
    if remaining > 8:
        best_h = do_slsqp(best_h, n_cur, target, 50000, maxiter=10000, t0=t0, budget=budget_s)
        best_c5 = compute_c5(best_h)

    # Phase 10: Final coord descent
    remaining = budget_s - (time.time() - t0)
    if remaining > 5:
        best_h, best_c5 = coord_descent(best_h, target, n_cur, 200, t0, budget_s)

    return best_h, best_c5, best_n
```
