import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 0)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-8)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

def slsqp_opt(h, target, beta, maxiter, time_fn):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def _obj(x):
        v, _ = smooth_max_val_grad(x, beta)
        return v

    def _grd(x):
        _, g = smooth_max_val_grad(x, beta)
        return g

    def _coneq(x):
        return x.sum() - target

    def _conjac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': _coneq, 'jac': _conjac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       jac=_grd, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_h(res.x, target)
    except Exception:
        return h.copy()

def paired_swap_search(h, target, n_iterations, time_fn):
    """Try random paired swaps to escape local minima."""
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    for _ in range(n_iterations):
        if time_fn() < 3:
            break
        i, j = np.random.choice(n, 2, replace=False)
        if best[i] <= best[j]:
            i, j = j, i
        t_max = min(best[i] - best[j], 1.0 - best[j], best[i])
        if t_max < 1e-6:
            continue
        # Binary search for best t in [0, t_max]
        lo, hi = 0.0, t_max
        for _ in range(20):
            mid = 0.5 * (lo + hi)
            h_trial = best.copy()
            h_trial[i] += mid
            h_trial[j] -= mid
            h_trial = project_h(h_trial, target)
            c5_trial = compute_c5(h_trial)
            if c5_trial < compute_c5(best):
                lo = mid
                if c5_trial < best_c5:
                    best = h_trial
                    best_c5 = c5_trial
            else:
                hi = mid
    return best, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tleft():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in dir() and initial_h_values is not None:

    return best_h, best_c5, n
