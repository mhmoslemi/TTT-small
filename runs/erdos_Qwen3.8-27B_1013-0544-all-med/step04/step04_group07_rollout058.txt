def project_h(h, target_sum):

def fourier_encode(h, n_modes):

def binary_local_search(h0, target, t0, budget, n_iter=5000):

def spectral_optimize(h0, n, target, t0, budget, n_modes=16):
    """
    Optimize using Fourier parameterization.
    h(x) = c0 + sum_{k=1}^{K} (a_k cos(2πkx/n) + b_k sin(2πkx/n))

    x = np.arange(n)
    c0 = np.mean(h0)
    coeffs = np.zeros(1 + 2*n_modes)
    coeffs[0] = c0
    for k in range(1, n_modes+1):
        coeffs[2*k-1] = 2.0/n * np.sum(h0 * np.cos(2*np.pi*k*x/n))
        coeffs[2*k] = 2.0/n * np.sum(h0 * np.sin(2*np.pi*k*x/n))
    
    def obj(c):
        h = c[0] + np.zeros(n)
        for k in range(1, n_modes+1):
            h += c[2*k-1] * np.cos(2*np.pi*k*x/n)
            h += c[2*k] * np.sin(2*np.pi*k*x/n)
        h = np.clip(h, 0, 1)
        # Enforce sum
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            free = (h > 1e-10) & (h < 1.0 - 1e-10)
            if free.sum() > 0:
                h[free] += diff / free.sum()
                h = np.clip(h, 0, 1)
        return smooth_max(compute_corr_full(h), 100.0) * (2.0/n)
    
    # Optimize with L-BFGS-B
    bounds = [(0, 0)] + [(-0.5, 0.5)] * (2*n_modes)
    try:
        result = minimize(obj, coeffs, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 500, 'ftol': 1e-12})
        c_opt = result.x
        h_opt = c_opt[0] + np.zeros(n)
        for k in range(1, n_modes+1):
            h_opt += c_opt[2*k-1] * np.cos(2*np.pi*k*x/n)
            h_opt += c_opt[2*k] * np.sin(2*np.pi*k*x/n)
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = project_h(h_opt, target)
    except:

def targeted_slsqp(h0, n, target, t0, budget, beta=500):
    """
    SLSQP on smooth max of correlations, targeting the worst lags.
    """
    from scipy.optimize import minimize, Bounds
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    
    def objective(h_in):
        return smooth_max(compute_corr_full(h_in), beta) * (2.0/n)
    
    def sum_eq(x):
        return np.sum(x) - target
    
    def sum_jac(x):
        return np.ones(x.shape)
    
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac}]
    
    try:
        r = minimize(objective, h0, method='SLSQP', bounds=bounds,
                    constraints=cons, options={'maxiter': 3000, 'ftol': 1e-15})
        h_opt = project_h(r.x, target)
    except:
        h_opt = h0.copy()
    
    return h_opt, compute_c5(h_opt)

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new, endpoint=False),
                    np.linspace(0, 2, len(h), endpoint=False), h)

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def correct_subgrad(h, n, lags_to_use=30):
    """Correct subgradient for max correlation. Returns (gradient, corr_full)."""
    corr = compute_corr_full(h)
    mc = corr.max()
    thr = mc - 0.004 * (mc + 1e-10)
    idx = np.where(corr >= thr)[0]
    if len(idx) > lags_to_use:
        idx = idx[np.argsort(corr[idx])[::-1][:lags_to_use]]
    g = np.zeros(n)
    for m in idx:
        lag = m - (n - 1)
        if lag > 0:
            g[lag:] += (1.0 - h[:n - lag])
            g[:n - lag] -= h[lag:]
        elif lag < 0:
            la = -lag
            g[:n - la] += (1.0 - h[la:])
            g[la:] -= h[:n - la]
        # lag == 0: gradient is (1-2h) for each component
        else:
            g += (1.0 - 2.0 * h)
    return g / max(len(idx), 1), corr

def smooth_max(x, beta):
    m = x.max()
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500)))) / beta

def adaptive_lagrangian(h0, n, target, t0, budget, n_iter=4000):
    """Adaptive weights on lags, projected gradient on weighted sum."""
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    dx = 2.0 / n
    corr = compute_corr_full(h)
    w = np.ones(2 * n - 1)
    step = 0.02
    no_improve = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        # Select active lags
        wc = w * corr
        top_k = 25
        top_idx = np.argsort(wc)[::-1][:top_k]
        g = np.zeros(n)
        for m in top_idx:
            lag = m - (n - 1)
            wm = w[m]
            if lag > 0:
                g[lag:] += wm * (1.0 - h[:n - lag])
                g[:n - lag] -= wm * h[lag:]
            elif lag < 0:
                la = -lag
                g[:n - la] += wm * (1.0 - h[la:])
                g[la:] -= wm * h[:n - la]
            else:
                g += wm * (1.0 - 2.0 * h)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        # Backtracking line search
        s = step
        improved = False
        for _ in range(8):
            h_trial = project_h(h - s * g / gn, target)
            c_trial = compute_c5(h_trial)
            if c_trial < best_c5 - 1e-14:
                h = h_trial
                best_h, best_c5 = h.copy(), c_trial
                no_improve = 0
                improved = True
                break
            s *= 0.5
        if not improved:
            no_improve += 1
            if no_improve % 200 == 0:
                step *= 0.6
        # Update weights
        corr = compute_corr_full(h)
        mc = corr.max() + 1e-10
        w = np.exp(3.0 * corr / mc)
        w /= w.sum()
    return best_h, best_c5

def binary_swap_search(h0, target, t0, budget, n_iter=4000):
    """Swap 1s and 0s in binary pattern, with occasional perturbations."""
    n = len(h0)
    z = (h0 > 0.5).astype(float)
    diff = target - z.sum()
    if diff > 0:
        zeros = np.where(z == 0)[0]
        order = np.argsort(h0[zeros])[::-1][:int(np.ceil(diff))]
        z[zeros[order]] = 1.0
    elif diff < 0:
        ones = np.where(z == 1)[0]
        order = np.argsort(h0[ones])[:int(np.ceil(-diff))]
        z[ones[order]] = 0.0
    z = project_h(z, target)
    best_h, best_c5 = z.copy(), compute_c5(z)
    rng = np.random.RandomState(789)
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        ones = np.where(z > 0.99)[0]
        zeros = np.where(z < 0.01)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        trial = z.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        c = compute_c5(trial)
        if c < best_c5 - 1e-13:
            z, best_h, best_c5 = trial, trial.copy(), c
    return best_h, best_c5

def block_moves(h0, target, t0, budget, n_iter=3000):
    """Try shifting/swapping small blocks to rearrange structure."""
    n = len(h0)
    h = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    rng = np.random.RandomState(321)
    for it in range(n_iter):
        if time.time() - t0 > budget - 3:
            break
        bs = rng.choice([2, 3, 4, 5, 6, 8])
        i = rng.randint(0, max(1, n - bs))
        op = rng.randint(3)
        trial = h.copy()
        if op == 0 and i + 1 + bs <= n:
            # Shift right by 1
            trial[i:i+1+bs] = np.concatenate([trial[i+1:i+1+bs], trial[i:i+1]])
        elif op == 1 and i - 1 - bs >= 0:
            # Shift left by 1
            trial[i-1:i+bs] = np.concatenate([trial[i:i+bs], trial[i-1:i]])
        else:
            # Swap with random position
            j = rng.randint(0, max(1, n - bs))
            if j == i:
                continue
            trial[i:i+bs], trial[j:j+bs] = trial[j:j+bs].copy(), trial[i:i+bs].copy()
        trial = project_h(trial, target)
        c = compute_c5(trial)
        if c < best_c5 - 1e-13:
            h, best_h, best_c5 = trial, trial.copy(), c
    return best_h, best_c5

def slsqp_optimize(h0, n, target, t0, budget, beta=600, maxiter=4000):
    """SLSQP on smooth max."""
    bounds = Bounds(np.zeros(n), np.ones(n))
    dx = 2.0 / n
    def obj(x):
        return smooth_max(compute_corr_full(x), beta) * dx
    def eq_fun(x):
        return np.sum(x) - target
    def eq_jac(x):
        return np.ones(x.shape)
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    try:
        r = minimize(obj, h0, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_h(r.x, target), compute_c5(project_h(r.x, target))
    except:
        return h0, compute_c5(h0)

def fourier_refine(h0, n, target, t0, budget, n_modes=24):
    """Optimize in Fourier coefficient space, then decode."""
    x = np.arange(n)
    # Build basis matrix
    basis = np.ones((n_modes, n))
    for k in range(1, n_modes):
        basis[2*k-1] = np.cos(2 * np.pi * k * x / n)
        basis[2*k] = np.sin(2 * np.pi * k * x / n)
    # Initial coeffs: project h0 onto basis
    c0 = np.array([np.mean(h0)] + [2.0/n * np.dot(h0, basis[2*k-1]) for k in range(1, n_modes)] +
                  [2.0/n * np.dot(h0, basis[2*k]) for k in range(1, n_modes)])
    dx = 2.0 / n
    def obj(c):
        h = c[0] + basis[1:] @ c[1:]
        h = np.clip(h, 0, 1)
        diff = target - h.sum()
        if abs(diff) > 1e-10:
            free = (h > 1e-8) & (h < 1.0 - 1e-8)
            if free.sum() > 0:
                h[free] += diff / free.sum()
                h = np.clip(h, 0, 1)
        return smooth_max(compute_corr_full(h), 150.0) * dx
    bounds_c = [(0, 0)] + [(-0.5, 0.5)] * (2 * (n_modes - 1))
    try:
        r = minimize(obj, c0, method='L-BFGS-B', bounds=bounds_c,
                     options={'maxiter': 300, 'ftol': 1e-14})
        h_opt = r.x[0] + basis[1:] @ r.x[1:]
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = project_h(h_opt, target)
    except:
        h_opt = h0.copy()
    return h_opt, compute_c5(h_opt)

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    old_x = np.linspace(0, 2, len(h), endpoint=False)
    new_x = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(new_x, old_x, h)

def run(seed=42, budget_s=1100, **kwargs):
