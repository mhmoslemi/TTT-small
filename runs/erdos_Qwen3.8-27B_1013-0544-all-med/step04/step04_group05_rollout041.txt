import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, beta):
    m = np.max(c)
    return m + np.log(np.sum(np.exp(beta * (c - m)))) / beta

def compute_grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    c_shift = beta * (c - c.max())
    w = np.exp(c_shift)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(len(c)):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:n-k])
            grad[:n-k] -= wk * h[k:]
        else:
            m = -k
            grad[:n-m] += wk * (1.0 - h[m:])
            grad[m:] -= wk * h[:n-m]
    return grad * dx

def lbfgsb_phase(h, target, beta, max_iter, time_fn, mu=1.0):
    n = len(h)
    dx = 2.0 / n
    
    def obj(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")
        sm = smooth_max_val(c, beta) * dx
        pen = mu * (h_in.sum() - target) ** 2
        return sm + pen
    
    def jac(h_in):
        g = compute_grad_smooth_max(h_in, beta)
        g += 2.0 * mu * (h_in.sum() - target)
        return g
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    try:
        res = minimize(obj, h, method='L-BFGS-B', jac=jac, bounds=bounds,
                      options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def nesterov_phase(h, target, time_fn, max_iters, beta, lr_init=0.01, 
                   decay=0.999, min_lr=0.0001):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    x = best_h.copy()
    v = np.zeros(n)
    t = 1.0
    lr = lr_init
    
    for it in range(max_iters):
        if time_fn() < 3:
            break
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y = x + (t / t_new - 1) * v
        
        # Clip y to [0,1] for gradient computation stability
        y_clip = np.clip(y, 0, 1)
        g = compute_grad_smooth_max(y_clip, beta)
        
        # Nesterov update
        v = ((t / t_new) * v - lr * g)
        x_new = x + v
        
        # Project
        x_new = project_h(x_new, target)
        
        # Check improvement
        c5 = compute_c5(x_new)
        if c5 < best_c5:
            best_h = x_new.copy()
            best_c5 = c5
        
        x = x_new
        t = t_new
        lr = max(min_lr, lr * decay)
    
    return best_h, best_c5

def subgrad_refined(h, target, time_fn, max_iters, m_lags, step_list, 
                    perturb_every, perturb_scale, momentum):
    n = len(h)
    best = h.copy()
    best_c5 = compute_c5(best)
    vel = np.zeros(n)
    no_improve = 0
    
    for it in range(max_iters):
        if time_fn() < 2:
            break
        corr = np.correlate(best, 1.0 - best, mode="full")
        top = np.argsort(corr)[::-1][:m_lags]
        lags = [int(i) - (n - 1) for i in top]
        
        # Subgradient from top lags
        g = np.zeros(n)
        for lag in lags:
            if lag == 0:
                g += (1.0 - 2.0 * best)
            elif lag > 0:
                g[lag:] += (1.0 - best[:-lag])
                g[:n - lag] -= best[lag:]
            else:
                m = -lag
                g[:n - m] += (1.0 - best[m:])
                g[m:] -= best[:n - m]
        
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        vel = momentum * vel + (1.0 - momentum) * (g / gn)
        vn = np.linalg.norm(vel)
        if vn < 1e-14:
            break
        d = -vel / vn
        
        found = False
        for step in step_list:
            ht = project_h(best + step * d, target)
            c5t = compute_c5(ht)
            if c5t < best_c5 - 1e-15:
                best = ht
                best_c5 = c5t
                no_improve = 0
                found = True
                break
        
        if not found:
            no_improve += 1
            if no_improve >= perturb_every and it % perturb_every == perturb_every - 1:
                p = np.random.randn(n) * perturb_scale
                ht = project_h(best + p, target)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best = ht
                    best_c5 = c5t
                    vel *= 0.5
                    no_improve = 0
            if no_improve > perturb_every * 15:

    return best, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(c, beta):
    m = np.max(c)
    cs = np.clip(beta * (c - m), -500, 500)
    return m + np.log(np.sum(np.exp(cs))) / beta

def compute_grad_smooth_max(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    cs = np.clip(beta * (c - c.max()), -500, 500)
    w = np.exp(cs)
    w /= w.sum()
    grad = np.zeros(n)
    for idx in range(len(c)):
        wk = w[idx]
        if wk < 1e-14:
            continue
        k = idx - (n - 1)
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
        else:
            m = -k
            grad[:n - m] += wk * (1.0 - h[m:])
            grad[m:] -= wk * h[:n - m]
    return grad * dx

def lbfgsb_opt(h, target, beta, max_iter, time_fn, mu=100.0):
    n = len(h)
    dx = 2.0 / n

    def obj(h_in):
        c = np.correlate(h_in, 1.0 - h_in, mode="full")

    def jac(h_in):

        g = g + 2.0 * mu * (h_in.sum() - target)

    bounds = Bounds(np.zeros(n), np.ones(n))

                       options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxls': 50})
        h_new = project_h(res.x, target)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

    best_c5 = compute_c5(best_h)
    x = best_h.copy()
    v = np.zeros(n)
    t = 1.0
    lr = lr0
    no_imp = 0
    for it in range(max_iters):
        if time_fn() < 3:
            break
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = np.clip(x + (t / t_new - 1.0) * v, 0.0, 1.0)
        g = compute_grad_smooth_max(y, beta)
        v = (t / t_new) * v - lr * g
        x_new = project_h(x + v, target)
        c5 = compute_c5(x_new)
        if c5 < best_c5:
            best_h = x_new.copy()
            best_c5 = c5
            no_imp = 0
        else:

        x = x_new

        if it % 2000 == 1999:

    return best_h, best_c5

def subgrad_opt(h, target, time_fn, max_iters, m_lags, step_list,
