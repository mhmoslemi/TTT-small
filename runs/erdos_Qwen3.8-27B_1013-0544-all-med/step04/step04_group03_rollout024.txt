import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_sa(h_bin, n, t0, budget, n_iter_max, rng):
    """Simulated annealing for binary function."""
    cur = h_bin.copy()
    cur_max = float(np.max(np.correlate(cur, 1.0 - cur, mode="full")))
    best = cur.copy()
    best_max = cur_max
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    
    for it in range(n_iter_max):
        if budget - (time.time() - t0) < 5:
            break
        # Pick random swap
        ia = rng.integers(len(ones))
        ib = rng.integers(len(zeros))
        a = ones[ia]
        b = zeros[ib]
        
        # Do swap
        cur[a] = 0.0
        cur[b] = 1.0
        ones[ia] = b
        zeros[ib] = a
        ones = np.sort(ones)
        zeros = np.sort(zeros)
        
        new_corr = np.correlate(cur, 1.0 - cur, mode="full")
        new_max = float(np.max(new_corr))
        
        if new_max <= cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best = cur.copy()
        else:
            frac = it / max(n_iter_max, 1)
            temp = 2.0 * (0.001) ** frac  # Start at 2, cool to 0.002
            if rng.random() < np.exp(-(new_max - cur_max) / max(temp, 1e-10)):
                cur_max = new_max
            else:
                # Undo
                cur[a] = 1.0
                cur[b] = 0.0
                ones[ia] = a
                zeros[ib] = b
                ones = np.sort(ones)
                zeros = np.sort(zeros)
    
    return best, best_max

def refine_continuous(h_bin, n, t0, budget):
    """Refine binary solution to continuous."""
    target = n / 2.0
    cur = h_bin + 0.05 * np.random.randn(n)
    cur = project_h(cur, target)
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    
    # Top-lag subgradient descent
    for phase in range(5):
        top_k = [8, 12, 16, 20, 30][min(phase, 4)]
        step = 0.02 / (phase + 1)
        n_iter = 500
        for it in range(n_iter):
            if budget - (time.time() - t0) < 3:
                break
            corr = np.correlate(cur, 1.0 - cur, mode="full")
            top_idx = np.argsort(corr)[::-1][:top_k]
            grad = np.zeros(n)
            for idx in top_idx:
                lag = int(idx) - (n - 1)
                if 0 <= lag < n:
                    grad[:n-lag] += (1.0 - cur[lag:])
                    grad[lag:] -= cur[:n-lag]
                elif lag < 0:
                    la = -lag
                    grad[la:] += (1.0 - cur[:n-la])
                    grad[:n-la] -= cur[la:]
            grad /= top_k
            gn = np.linalg.norm(grad)
            if gn < 1e-12:
                break
            grad /= gn
            trial = project_h(cur - step * grad, target)
            c_try = compute_c5(trial)

                cur = trial

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_sa(h_init, n, t0, budget, max_iters, rng, t_init=3.0, t_final=0.01):
    """SA for binary function, minimizing max correlation."""
    cur = h_init.copy()
    cur_max = float(np.max(np.correlate(cur, 1.0 - cur, mode="full")))
    best = cur.copy()
    best_max = cur_max
    
    ones_indices = np.where(cur == 1)[0]
    zeros_indices = np.where(cur == 0)[0]
    
    n1 = len(ones_indices)
    n0 = len(zeros_indices)
    
    for it in range(max_iters):
        if budget - (time.time() - t0) < 3:
            break
        
        # Temperature: exponential cooling
        frac = it / max(max_iters, 1)
        temp = t_init * (t_final / t_init) ** frac
        
        ia = rng.integers(n1)
        ib = rng.integers(n0)
        a = ones_indices[ia]
        b = zeros_indices[ib]
        
        # Swap
        cur[a] = 0.0
        cur[b] = 1.0
        ones_indices[ia] = b
        zeros_indices[ib] = a
        
        new_corr = np.correlate(cur, 1.0 - cur, mode="full")
        new_max = float(np.max(new_corr))
        
        if new_max <= cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best = cur.copy()
        else:
            delta = new_max - cur_max
            if rng.random() < np.exp(-delta / max(temp, 1e-10)):
                cur_max = new_max
            else:
                cur[a] = 1.0
                cur[b] = 0.0
                ones_indices[ia] = a
                zeros_indices[ib] = b
    
    return best, best_max

def top_lag_subgrad(h, n, target, t0, budget, n_iter=10000, step_init=0.03):
    """Subgradient descent on top lags."""
    cur = h.copy()
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    step = step_init
    stall = 0
    
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        top_idx = np.argsort(corr)[::-1][:16]
        
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if 0 <= lag < n:
                grad[:n-lag] += (1.0 - cur[lag:])
                grad[lag:] -= cur[:n-lag]
            elif lag < 0:
                la = -lag
                grad[la:] += (1.0 - cur[:n-la])
                grad[:n-la] -= cur[la:]
        
        grad /= len(top_idx)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad /= gn
        
        trial = project_h(cur - step * grad, target)
        c_try = compute_c5(trial)
        if c_try < best_c5 - 1e-14:
            cur = trial
            best_c5 = c_try
            best_h = cur.copy()
            stall = 0
            step = min(step * 1.02, 0.1)
        else:
            stall += 1
            step *= 0.85
            if stall > 200:
                break
    
    return best_h, best_c5

def coord_polish(h, n, target, t0, budget, max_sweeps=50):
    """Coordinate descent polish."""
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.003, -0.003]

                trial[j] += d

                if c_try < best_c5 - 1e-13:
                    cur = trial
                    best_c5 = c_try
                    improved = True
                    break
        if not improved and sw > 2:
            break
    
    return cur, best_c5

            h = np.clip(h, 0.0, 1.0)
        else:
            h[free] += diff / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_sa(h_init, n, t0, budget, max_iters, rng, t_init=2.0, t_final=0.02):
    cur = h_init.copy()
    cur_max = float(np.max(np.correlate(cur, 1.0 - cur, mode="full")))
    best = cur.copy()
    best_max = cur_max
    ones = np.where(cur == 1)[0].copy()
    zeros = np.where(cur == 0)[0].copy()
    n1, n0 = len(ones), len(zeros)
    
    for it in range(max_iters):
        if budget - (time.time() - t0) < 3:
            break
        frac = it / max(max_iters, 1)
        temp = t_init * (t_final / t_init) ** frac
        
        ia = rng.integers(n1)
        ib = rng.integers(n0)
        a, b = ones[ia], zeros[ib]
        
        cur[a] = 0.0; cur[b] = 1.0
        ones[ia] = b; zeros[ib] = a
        
        new_max = float(np.max(np.correlate(cur, 1.0 - cur, mode="full")))
        
        if new_max <= cur_max:
            cur_max = new_max
            if new_max < best_max:
                best_max = new_max
                best = cur.copy()
        else:
            if temp > 1e-10 and rng.random() < np.exp(-(new_max - cur_max) / temp):
                cur_max = new_max
            else:
                cur[a] = 1.0; cur[b] = 0.0
                ones[ia] = a; zeros[ib] = b
    
    return best, best_max

def top_lag_subgrad(h, n, target, t0, budget, n_iter=5000, step_init=0.03, top_k=16):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    best_h = cur.copy()
    step = step_init
    stall = 0
    for it in range(n_iter):
        if budget - (time.time() - t0) < 3:
            break
        corr = np.correlate(cur, 1.0 - cur, mode="full")
        idx = np.argsort(corr)[::-1][:top_k]
        grad = np.zeros(n)
        for i in idx:
            lag = int(i) - (n - 1)
            if 0 <= lag < n:
                grad[:n-lag] += 1.0 - cur[lag:]
                grad[lag:] -= cur[:n-lag]
            elif lag < 0:
                la = -lag
                grad[la:] += 1.0 - cur[:n-la]
                grad[:n-la] -= cur[la:]
        grad /= len(idx)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad /= gn
        trial = project_h(cur - step * grad, target)
        c = compute_c5(trial)
        if c < best_c5 - 1e-14:
            cur, best_c5, best_h = trial, c, cur.copy()
            stall = 0
            step = min(step * 1.03, 0.15)
        else:
            stall += 1
            step *= 0.88
            if stall > 300:
                break
    return best_h, best_c5

def coord_polish(h, n, target, t0, budget, max_sweeps=100):
    cur = h.copy()
    best_c5 = compute_c5(cur)
    deltas = [0.015, -0.015, 0.008, -0.008, 0.004, -0.004, 0.002, -0.002]
    for sw in range(max_sweeps):
        if budget - (time.time() - t0) < 2:
            break
        imp = False
        for j in range(n):
            for d in deltas:
                trial = cur.copy()
                trial[j] += d
                trial = project_h(trial, target)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    cur, best_c5, imp = trial, c, True
                    break
        if not imp and sw > 3:
            break
    return cur, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.zeros(384)
        h_parent[:192] = 1.0
        h_parent = project_h(h_parent, 192.0)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    best_n = len(h_parent)
    
    # Multi-n binary SA
    n_list = [384, 256, 512, 320, 448]
    for n in n_list:
        if budget_s - (time.time() - t0) < 50:
            break
        target = n / 2.0
        # Resample parent
        old_x = np.linspace(0, 2, len(h_parent), endpoint=False)
        new_x = np.linspace(0, 2, n, endpoint=False)
        hr = np.interp(new_x, old_x, h_parent)
        hb = (hr > 0.5).astype(float)
        # Fix count
        cnt = int(hb.sum())
        if cnt > n // 2:
            oi = np.where(hb == 1)[0]
            hb[oi[rng.choice(len(oi), cnt - n//2, replace=False)]] = 0
        elif cnt < n // 2:
            oz = np.where(hb == 0)[0]
            hb[oz[rng.choice(len(oz), n//2 - cnt, replace=False)]] = 1
        
        # SA (longer for n=384)
        sa_time = 150 if n == 384 else 60
        iters = int(sa_time * 22000)
        h1, m1 = binary_sa(hb, n, t0, budget_s, iters, rng, t_init=2.0, t_final=0.02)
        
        # Random restart
        if budget_s - (time.time() - t0) > 30 and n == 384:
            hr2 = np.zeros(n)
            hr2[rng.choice(n, n//2, replace=False)] = 1.0
            h2, m2 = binary_sa(hr2, n, t0, budget_s, 3000000, rng, t_init=4.0, t_final=0.05)
            if m2 < m1:
                h1, m1 = h2, m2
        
        # Continuous refinement
        hc = h1 + 0.025 * rng.standard_normal(n)
        hc = project_h(hc, target)
        hc, c5c = top_lag_subgrad(hc, n, target, t0, budget_s, n_iter=4000, step_init=0.03)
        
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()
            best_n = n
    
    # Polish
    if budget_s - (time.time() - t0) > 10:
        best_h, best_c5 = coord_polish(best_h, best_n, best_n/2.0, t0, budget_s, max_sweeps=150)
    
    # Perturbation + re-optimize
    remaining = budget_s - (time.time() - t0)
    if remaining > 25:
        n_cur, target = best_n, best_n / 2.0
        for att in range(10):
            remaining = budget_s - (time.time() - t0)
            if remaining < 15:
                break
            s = [0.04, 0.02, 0.06, 0.01, 0.03, 0.05, 0.015, 0.025, 0.035, 0.05][att]
            hp = best_h + rng.standard_normal(n_cur) * s
            hp = project_h(hp, target)
            hp, c = top_lag_subgrad(hp, n_cur, target, t0, budget_s, n_iter=3000, step_init=0.02)
            if c < best_c5:
                best_c5 = c
                best_h = hp.copy()
    
    return best_h, best_c5, best_n
