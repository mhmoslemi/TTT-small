1. Compute all correlations
2. Find the worst lag k*

For step 3, given the worst lag k*, the shift is s = k* - (n-1).
c[k*] = sum_{valid i} h[i] * (1-h[i+s])

To reduce c[k*], we want to either:
- Decrease h[i] for i where h[i+s] is small (so 1-h[i+s] is large)
- Increase h[i] for i where h[i+s] is large (so 1-h[i+s] is small)
- Or increase h[i+s] for i where h[i] is large

The gradient of c[k*] with respect to h[i] is:
g[i] = (1-h[i+s]) * I(i valid) - h[i-s] * I(i-s valid)

So the most impactful single change is to decrease h at the position with the largest positive gradient, or increase h at the position with the largest negative gradient.

But we need to maintain the sum constraint, so we must transfer from one position to another.

The optimal transfer: decrease h[i] by δ (where g[i] > 0, largest) and increase h[j] by δ (where g[j] < 0, most negative). The change in c[k*] is approximately δ * (g[j] - g[i]) which is negative (reducing c[k*]).

But we also need to check that this doesn't increase other correlation values too much.

    """Polish by focusing on the worst lag and making directed moves."""
    dx = 2.0 / n

    best_c = c.max()
    best_h = h.copy()
    
    # Start with small steps
    steps = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]
    step_idx = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        k_star = np.argmax(c)
        s = k_star - (n - 1)
        
        # Compute gradient of c[k_star]
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        g = np.zeros(n)
        for i in range(lo, hi + 1):
            g[i] += (1 - h[i + s])
        lo2 = max(0, lo + s)
        hi2 = min(n - 1, hi + s)
        for i in range(lo2, hi2 + 1):
            g[i] -= h[i - s]
        
        # Find best transfer: decrease h at argmax(g) positive, increase at argmin(g) negative
        pos_mask = g > 0
        neg_mask = g < 0
        if not pos_mask.any() or not neg_mask.any():
            step_idx = min(step_idx + 1, len(steps) - 1)
            continue
        
        i_src = np.where(pos_mask)[0][np.argmax(g[pos_mask])]  # position with largest positive g
        i_dst = np.where(neg_mask)[0][np.argmin(g[neg_mask])]  # position with most negative g
        
        if i_src == i_dst:
            continue
        
        d = steps[step_idx]
        # Try the move
        h[i_src] -= d
        h[i_dst] += d
        
        if h[i_src] < 0 or h[i_dst] > 1:
            h[i_src] += d
            h[i_dst] -= d
            continue
        
        # Check improvement
        new_c = np.correlate(h, 1 - h, mode='full') * dx
        new_max = new_c.max()
        
        if new_max < best_c - 1e-15:
            best_c = new_max

        c = compute_c5(ht, n)
        if c < best_c - 1e-14:
            best_c = c
            best_h = ht
    if time.time() < deadline - 5:
        best_h, best_c = multi_lag_polish(best_h, n, min(deadline - 5, time.time() + 30), num_lags=3)
    if time.time() < deadline - 5:
        best_h, best_c = directed_polish(best_h, n, deadline - 5)
    if time.time() < deadline - 3:
        ht = slsqp_round(best_h, n, 2000000, maxiter=3000)
        c = compute_c5(ht, n)
        if c < best_c:
            best_c = c
            best_h = ht
    if time.time() < deadline - 2:
        best_h, best_c = directed_polish(best_h, n, deadline - 1)
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    deadline = t0 + budget_s - 5
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h_init)
    except Exception:
        h_init = None
        n_init = 200
    best_h = None
    best_c = float('inf')
    if h_init is not None:
        alloc = (deadline - t0) * 0.35
        h_opt, c_opt = optimize_solution(h_init, n_init, t0 + alloc)
        best_h, best_c = h_opt, c_opt
        x_init = np.linspace(0, 2, n_init)
        for n_try in [100, 150, 250, 300]:
            if n_try == n_init or time.time() > deadline - 80:
                continue
            remaining = deadline - time.time()
            alloc = min(120, remaining * 0.2)
            if alloc < 40:
                break
            x_new = np.linspace(0, 2, n_try)
            h_interp = np.interp(x_new, x_init, h_init)
            h_interp = project_feasible(h_interp, n_try)
            h_opt, c_opt = optimize_solution(h_interp, n_try, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 60:
        n = len(best_h)
        remaining = deadline - time.time() - 25
        n_restarts = max(4, int(remaining / 45))
        for r in range(n_restarts):
            if time.time() > deadline - 25:
                break
            scale = 0.005 + 0.06 * (r / max(1, n_restarts - 1))
            hp = best_h + rng.normal(0, scale, n)
            hp = project_feasible(hp, n)
            alloc = min(60, (deadline - time.time() - 25) * 0.35)
            if alloc < 20:
                break
            h_opt, c_opt = optimize_solution(hp, n, time.time() + alloc)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt
    if best_h is not None and time.time() < deadline - 5:
        n = len(best_h)
        best_h, best_c = directed_polish(best_h, n, deadline - 2)
        best_h = project_feasible(best_h, n)
        best_c = compute_c5(best_h, n)
    if best_h is None:
        n = n_init
        best_h = project_feasible(np.full(n, 0.5), n)
        best_c = compute_c5(best_h, n)
    return best_h, best_c, len(best_h)
