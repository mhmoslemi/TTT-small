def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def get_top_lags(h, num=20):
    corr = np.correlate(h, 1.0 - h, mode="full")
    idx = np.argsort(corr)[::-1][:num]
    return idx, corr

    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

    return np.sum(x) - t

    return np.ones(x.shape)

def nesterov(h0, n, target, t0, budget, n_iter=5000, step0=0.02):
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    stall = 0
    for it in range(n_iter):
        if time.time() - t0 > budget - 5:
            break
        corr = np.correlate(y, 1.0 - y, mode="full")
        mc = np.max(corr)
        thr = mc - 0.005 * (mc + 1e-10)
        idx = np.where(corr >= thr)[0]
        if len(idx) > 30:
            idx = idx[np.argsort(corr[idx])[::-1][:30]]
        g = np.zeros(n)
        for i in idx:
            lag = int(i) - (n - 1)
            if lag >= 0:
                g[:n-lag] += (1.0 - y[lag:])
                g[lag:] -= y[:n-lag]
            else:
                la = -lag
                g[la:] += (1.0 - y[:n-la])
                g[:n-la] -= y[la:]
        g /= max(len(idx), 1)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g /= gn
        h_new = project_h(y - step0 * g, target)
        mu = (1.0 - np.sqrt(1.0 + 4.0/((it+1)**2))) / 2.0

        c = compute_c5(h)

        else:
            stall += 1
            if stall % 200 == 0:
                step0 *= 0.8
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(np.clip(beta * (x - m), -500, 500))) / len(x)) / beta

def sml_obj(h, beta, dx):
    return smooth_max(np.correlate(h, 1.0 - h, mode="full"), beta) * dx

def sum_eq(x, t):
    return np.sum(x) - t

def sum_eq_jac(x, t):
    return np.ones(x.shape)

def do_slsqp(h0, n, target, beta, bounds, maxiter, ftol):
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target,)}]
    try:
        r = minimize(sml_obj, h0, args=(beta, 2.0 / n), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return project_h(r.x, target)
    except Exception:
        return h0

def resample(h, n_new):
    if len(h) == n_new:
        return h.copy()
    return np.interp(np.linspace(0, 2, n_new), np.linspace(0, 2, len(h)), h)

def binary_swap_search(h0, n, target, t0, budget, n_trials=10000):
    best_h = h0.copy()

    rng = np.random.RandomState(12345 + n)
    rejected_streak = 0

            break
        if rejected_streak > 6000:
            break
        ones = np.where(h0 > 0.25)[0]
        zeros = np.where(h0 < 0.75)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        amt = (h0[i] - h0[j]) * rng.uniform(0.3, 1.0)
        if abs(amt) < 1e-8:
            continue
        trial = h0.copy()
        trial[i] -= amt
        trial[j] += amt
        trial = np.clip(trial, 0.0, 1.0)
        trial = project_h(trial, target)
        c5 = compute_c5(trial)
        if c5 < best_c5 - 1e-14:
            best_h = trial.copy()
            best_c5 = c5
            h0 = trial.copy()
            rejected_streak = 0
        else:
            rejected_streak += 1
    return best_h, best_c5

    return best_h, best_c5

def nesterov_subgrad(h0, n, target, t0, deadline, n_iter=4000, step0=0.025):
    h = h0.copy()
    h_prev = h0.copy()
    y = h0.copy()
    best_h, best_c5 = h.copy(), compute_c5(h)
    stall = 0
    step = step0
    for it in range(n_iter):
        if time.time() - t0 > deadline - 5:
            break
        corr = np.correlate(y, 1.0 - y, mode="full")
        mc = np.max(corr)
        thr = mc - 0.004 * (mc + 1e-10)
        idx = np.where(corr >= thr)[0]
        if len(idx) > 25:
            idx = idx[np.argsort(corr[idx])[::-1][:25]]
        g = np.zeros(n)
        for ii in idx:
            lag = int(ii) - (n - 1)
            if lag >= 0:
                g[:n - lag] += (1.0 - y[lag:])
                g[lag:] -= y[:n - lag]
            else:
                la = -lag
                g[la:] += (1.0 - y[:n - la])
                g[:n - la] -= y[la:]
        g /= max(len(idx), 1)
        gn = np.linalg.norm(g)

        g /= gn
