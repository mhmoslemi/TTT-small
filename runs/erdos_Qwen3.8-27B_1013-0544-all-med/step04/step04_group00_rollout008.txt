    sh = alpha * (g - m)
    sm = sh.max()
    return m + (sm + np.log(np.exp(sh - sm).sum())) / alpha

def sum_constraint(x, n):
    return x.sum() - n / 2.0

def slsqp_optimize(h, n, alpha, maxiter, deadline):
    def obj(x):
        return slsqp_obj(x, n, alpha)
    def cons_fun(x):
        return sum_constraint(x, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h

def coord_descent(h, n, deadline, max_sweeps=200, grid_size=60):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    center = n - 1
    x_grid = np.linspace(0.0, 1.0, grid_size)
    quad_corr = np.zeros(2 * n - 1)
    quad_corr[center] = 1.0
    xg = x_grid
    xg2 = x_grid * (x_grid - 1.0)

    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        any_improve = False
        for i in range(n):
            if time.time() > deadline:
                break
            old_x = h[i]
            h[i] = 0.0
            g0 = np.correlate(h, 1 - h, mode='full').astype(np.float64)
            h[i] = 1.0
            g1 = np.correlate(h, 1 - h, mode='full').astype(np.float64)
            d = g1 - g0
            F = g0[:, None] + xg[None, :] * d[:, None] - quad_corr[:, None] * xg2[None, :]
            max_F = F.max(axis=0)
            best_x = x_g_min_index(max_F, x_grid)
            h[i] = best_x
            if abs(best_x - old_x) > 1e-5:
                any_improve = True
        h = project_feasible(h, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-12:
            best_c = c
            best_h = h.copy()
        if not any_improve:
            break
    return best_h, best_c

def x_g_min_index(max_F, x_grid):
    idx = int(np.argmin(max_F))
    return x_grid[idx]

def pair_local_search(h, n, deadline, rng, n_iter=50000):
    best_h = h.copy()
    best_c = compute_c5(h, n)
    target = n / 2.0
    for it in range(n_iter):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for sgn in (1.0, -1.0):
            for d in (0.05, 0.02, 0.01, 0.005):
                hi = h[i] + sgn * d
                hj = h[j] - sgn * d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h[i], h[j] = hi, hj
                    c = compute_c5(h, n)
                    if c < best_c - 1e-13:
                        best_c = c
                        best_h = h.copy()
                        break
                    else:
                        h[i], h[j] = hi - sgn * d, hj + sgn * d
    return best_h, best_c

def try_resolution(h_best, c_best, n_best, deadline, rng):
    for nh in [150, 200, 300, 350]:
        if time.time() > deadline:
            break
        if nh == n_best:
            continue
        xo = np.linspace(0, 2, n_best)
        xn = np.linspace(0, 2, nh)
        hh = np.interp(xn, xo, h_best)
        hh = project_feasible(hh, nh)
        c0 = compute_c5(hh, nh)
        if c0 > c_best + 0.005:
            continue
        hh = slsqp_optimize(hh, nh, 50000, 300, deadline)
        c1 = compute_c5(hh, nh)
        if c1 < c_best:
            hh, c1 = coord_descent(hh, nh, deadline, max_sweeps=50, grid_size=50)
        if c1 < c_best:
            c_best = c1
            h_best = hh.copy()
            n_best = nh
    return h_best, c_best, n_best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 200
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c = compute_c5(best_h, n0)
    n = n0

    # Phase 1: SLSQP with increasing alpha (35% time)
    p1_end = t0 + 0.35 * (budget_s - 5)
    alphas = [500, 5000, 50000, 200000, 1000000]
    for a in alphas:
        if time.time() > p1_end:
            break
        h_trial = slsqp_optimize(best_h.copy(), n, a, 500, p1_end)
        c = compute_c5(h_trial, n)
        if c < best_c:
            best_c = c
            best_h = h_trial.copy()

    # Phase 2: Coordinate descent (25% time)
    p2_end = t0 + 0.60 * (budget_s - 5)
    h_cd, c_cd = coord_descent(best_h.copy(), n, p2_end, max_sweeps=300, grid_size=60)
    if c_cd < best_c:
        best_c = c_cd
        best_h = h_cd.copy()

    # Phase 3: More SLSQP after coordinate descent (15% time)
    p3_end = t0 + 0.75 * (budget_s - 5)
    for a in [200000, 1000000, 5000000]:
        if time.time() > p3_end:
            break
        h_trial = slsqp_optimize(best_h.copy(), n, a, 400, p3_end)
        c = compute_c5(h_trial, n)
        if c < best_c:
            best_c = c
            best_h = h_trial.copy()

    # Phase 4: Try other resolutions (10% time)
    p4_end = t0 + 0.85 * (budget_s - 5)
    best_h, best_c, n = try_resolution(best_h, best_c, n, p4_end, rng)

    # Phase 5: Final coordinate descent + pair search (10% time)
    h_cd, c_cd = coord_descent(best_h.copy(), n, deadline, max_sweeps=200, grid_size=80)
    if c_cd < best_c:
        best_c = c_cd
        best_h = h_cd.copy()

    if time.time() < deadline - 10:
        hp, cp = pair_local_search(best_h.copy(), n, deadline, rng, n_iter=30000)
        if cp < best_c:
            best_c = cp
            best_h = hp.copy()

    # Final projection
    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
