import numpy as np
from scipy.optimize import minimize
import time

_beta_box = [1.0]
_target_box = [1.0]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(2000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    x_shift = np.clip(_beta_box[0] * (corr - m), -500, 500)
    e = np.exp(x_shift)
    return (m + np.log(np.sum(e)) / _beta_box[0]) * dx

def slsqp_grad(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode="full")
    m = np.max(corr)
    x_shift = np.clip(_beta_box[0] * (corr - m), -500, 500)
    e = np.exp(x_shift)
    s = np.sum(e)
    w = e / s
    grad = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        wi = w[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += wi * (1.0 - 2.0 * x)
        elif 0 < lag < n:
            grad[lag:] += wi * (1.0 - x[:-lag])
            grad[:n - lag] -= wi * x[lag:]
        elif -n < lag < 0:
            m2 = -lag
            grad[:n - m2] += wi * (1.0 - x[m2:])
            grad[m2:] -= wi * x[:n - m2]
    return grad

def slsqp_con(x):
    return x.sum() - _target_box[0]

def slsqp_con_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, betas, maxiter, time_fn):
    n = len(h0)
    target = n / 2.0
    _target_box[0] = target
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    for beta in betas:
        if time_fn() < 10:
            break
        _beta_box[0] = beta
        try:
            res = minimize(slsqp_obj, h_cur, method='SLSQP',
                          bounds=bounds, jac=slsqp_grad,
                          constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-15})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-15:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def coord_sweep(h, target, n_sweeps, time_fn, direction='forward'):

    nk = 2 * n - 1

    best_c5 = compute_c5(h)

    # valid_minus[i] = (i-s_vals >= 0) & (i-s_vals < n)
    # These are (n, nk) boolean arrays - for n=500, nk=999, that's ~500k booleans each
    
    for sweep in range(n_sweeps):

            break

        elif direction == 'backward':
            order = range(n - 1, -1, -1)
        else:  # random
            order = np.random.permutation(n)
        
        for i in order:
            old_hi = h[i]
            
            a = np.zeros(nk)
            vp = (i + s_vals >= 0) & (i + s_vals < n)
            vm = (i - s_vals >= 0) & (i - s_vals < n)
            a[vp] += (1.0 - h[i + s_vals[vp]])
