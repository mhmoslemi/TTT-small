import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-13:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-13:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def obj_soft(x, n, alpha):
    h = x
    s = h.sum()
    target = n / 2.0
    pen = 0.1 * (s - target) ** 2
    g = np.correlate(h, 1 - h, mode='full') * (2.0 / n)
    m = g.max()
    sh = alpha * (g - m)
    sm = sh.max()
    lse = sm + np.log(np.exp(sh - sm).sum())
    return m + lse / alpha + pen

def polish_continuous(h, n, deadline, alphas):
    best_h = h.copy()
    best_c = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    for alpha in alphas:
        if time.time() > deadline:
            break
        for attempt in range(2):
            if time.time() > deadline:
                break
            start = best_h + 0.01 * np.random.RandomState(attempt).randn(n) if attempt else best_h
            start = project_feasible(start, n)
            try:
                res = minimize(obj_soft, start, args=(n, alpha),
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 3000, 'ftol': 1e-20, 'gtol': 1e-15})
                h_new = project_feasible(res.x, n)
                c_new = compute_c5(h_new, n)
                if c_new < best_c:
                    best_c = c_new
                    best_h = h_new.copy()
            except Exception:
                pass
    return best_h, best_c

def snap_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:target]] = 1.0
    return hb

def binary_exhaustive(h, n, deadline, max_sweeps=2000):
    hb = snap_binary(h, n)
    best_c = compute_c5(hb, n)
    target = n // 2
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        ones = np.where(hb == 1)[0]
        zeros = np.where(hb == 0)[0]
        best_c_sweep = best_c
        best_pair = None
        # Full exhaustive: try all swaps
        for ii in range(len(zeros)):
            i = zeros[ii]
            for jj in range(len(ones)):
                j = ones[jj]
                hb[i] = 1.0
                hb[j] = 0.0
                c = compute_c5(hb, n)
                if c < best_c_sweep - 1e-15:
                    best_c_sweep = c
                    best_pair = (i, j)
                hb[i] = 0.0
                hb[j] = 1.0
        if best_pair is None:
            break
        i, j = best_pair
        hb[i] = 1.0
        hb[j] = 0.0
        best_c = best_c_sweep
    return hb, best_c

def binary_with_kicks(h, n, deadline, max_rounds=5):
    best_h = snap_binary(h, n)
    best_c = compute_c5(best_h, n)
    target = n // 2
    for r in range(max_rounds):
        if time.time() > deadline:
            break
        sub_dl = deadline - (deadline - time.time()) * 0.7
        sub_dl = min(sub_dl, deadline)
        hb, cb = binary_exhaustive(best_h, n, sub_dl, max_sweeps=500)
        if cb < best_c:
            best_c = cb
            best_h = hb.copy()
        # Kick: random perturbation
        if time.time() > deadline - 10:
            break
        rng_k = np.random.RandomState(1000 + r * 7)
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        n_kick = max(3, target // 25)
        for _ in range(n_kick):
            i = zeros[rng_k.randint(len(zeros))]
            j = ones[rng_k.randint(len(ones))]
            best_h[i] = 1.0
            best_h[j] = 0.0
    return best_h, best_c
