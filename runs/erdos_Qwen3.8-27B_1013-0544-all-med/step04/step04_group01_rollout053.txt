import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            tot = room.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * room
        else:
            tot = h.sum()
            if tot < 1e-15:
                break
            h += (diff / tot) * h
    return np.clip(h, 0.0, 1.0)

def coord_descent(h, n, num_sweeps=500, deadline=None, seed=42):
    h = h.copy().astype(float)
    h = np.clip(h, 0, 1)
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    rng = np.random.RandomState(seed)
    n_corr = 2 * n - 1
    ar = np.arange(n_corr)
    nm1 = n - 1

    prev_max = c.max()
    stagnant = 0

    for sweep in range(num_sweeps):
        if deadline is not None and time.time() > deadline:
            break

        order = rng.permutation(n)

        for i in order:
            old = h[i]

            idx_plus = i - nm1 + ar
            idx_minus = i + nm1 - ar

            valid_p = (idx_plus >= 0) & (idx_plus < n)
            valid_m = (idx_minus >= 0) & (idx_minus < n)

            B = np.zeros(n_corr)
            if valid_p.any():
                B[valid_p] = 1.0 - h[idx_plus[valid_p]]
            if valid_m.any():
                B[valid_m] -= h[idx_minus[valid_m]]

            d_lo = -old
            d_hi = 1.0 - old
            if d_hi - d_lo < 1e-14:
                continue

            # Quick check: if B all same sign, optimum is at boundary
            if np.all(B >= 0):
                d_star = d_lo
            elif np.all(B <= 0):
                d_star = d_hi
            else:
                a, b = d_lo, d_hi

                    m1 = a + step

                    if np.max(c + m1 * B) < np.max(c + m2 * B):
                        b = m2
                    else:
                        a = m1
                d_star = (a + b) * 0.5

            if abs(d_star) > 1e-15:
                h[i] = old + d_star
                c += d_star * B

            prev_max = cur_max

        else:
            stagnant += 1
        if stagnant > 80:
            break

    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')

def coord_descent_pair(h, n, num_sweeps=200, deadline=None, seed=42):
    """Pair-wise coordinate descent: update pairs (i,j) with constraint h[i]+h[j] fixed."""
    h = h.copy().astype(float)
    h = np.clip(h, 0, 1)
    c = np.correlate(h, 1.0 - h, mode='full').copy()
    rng = np.random.RandomState(seed)
    n_corr = 2 * n - 1
    ar = np.arange(n_corr)
    nm1 = n - 1

    for sweep in range(num_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for _ in range(n // 2):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            old_i, old_j = h[i], h[j]
            s = old_i + old_j

            # B^i and B^j
            ip = i - nm1 + ar
            im = i + nm1 - ar
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            Bi = np.zeros(n_corr)
            if vp.any():
                Bi[vp] = 1.0 - h[ip[vp]]
            if vm.any():
                Bi[vm] -= h[im[vm]]

            jp = j - nm1 + ar
            jm = j + nm1 - ar
            vp2 = (jp >= 0) & (jp < n)
            vm2 = (jm >= 0) & (jm < n)
            Bj = np.zeros(n_corr)
            if vp2.any():
                Bj[vp2] = 1.0 - h[jp[vp2]]
            if vm2.any():
                Bj[vm2] -= h[jm[vm2]]

            # c + d*Bi - d*Bj, d in [-old_i, s - old_i] intersect [-old_j, 1-old_j]... 
            # h[i] = old_i + d, h[j] = old_j - d
            # Need 0<=old_i+d<=1 and 0<=old_j-d<=1
            d_lo = max(-old_i, old_j - 1.0)
            d_hi = min(1.0 - old_i, old_j)
            if d_hi - d_lo < 1e-14:
                continue

            D = Bi - Bj
            if np.all(D >= 0):
                d_star = d_lo
            elif np.all(D <= 0):
                d_star = d_hi
            else:
                a, b = d_lo, d_hi
                for _ in range(25):
                    step = (b - a) / 3.0
                    m1 = a + step
                    m2 = b - step
                    if np.max(c + m1 * D) < np.max(c + m2 * D):
                        b = m2
                    else:
                        a = m1
                d_star = (a + b) * 0.5

            if abs(d_star) > 1e-15:
                h[i] = old_i + d_star
                h[j] = old_j - d_star
                c += d_star * D

    h = project_feasible(h, n)
    c = np.correlate(h, 1.0 - h, mode='full')
    return h, float(c.max() * 2.0 / n)
