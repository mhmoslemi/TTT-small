```python
def exact_coord_descent(h, target, n, sweeps, t0, budget):
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for sw in range(sweeps):
        if time.time() - t0 > budget - 2:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 1:
                break
            # Compute correlation at h[j]=0 and h[j]=1
            h_save = h[j]
            h[j] = 0.0
            corr_0 = np.correlate(h, 1.0 - h, mode="full")
            h[j] = 1.0
            corr_1 = np.correlate(h, 1.0 - h, mode="full")
            
            # C_k(t) = corr_0[k] + t * (corr_1[k] - corr_0[k])
            diff = corr_1 - corr_0
            # We want to minimize max_k (corr_0[k] + t * diff[k]) over t in [0,1]
            # This is a 1D convex piecewise linear minimization
            
            # Simple approach: use golden section or just check a few points
            # The minimum of max of linear functions occurs where two (or more) functions
            # are equal and all others are below. We can find it by checking all pairs.
            # But that's O(n^2). Instead, use a simpler approach:
            
            # Evaluate at a grid of points
            ts = np.linspace(0, 1, 50)
            vals = np.max(corr_0[None, :] + ts[:, None] * diff[None, :], axis=1) * dx
            t_best = ts[np.argmin(vals)]
            v_best = vals.min()
            
            # Refine around t_best
            if 0 < t_best < 1:
                ts_fine = np.linspace(max(0, t_best - 0.02), min(1, t_best + 0.02), 50)
                vals_fine = np.max(corr_0[None, :] + ts_fine[:, None] * diff[None, :], axis=1) * dx
                t_best = ts_fine[np.argmin(vals_fine)]
                v_best = vals_fine.min()
            
            h[j] = t_best
            c = compute_c5(h)
            if c < best_c5 - 1e-15:
                best_c5 = c
                improved = True
            else:
                h[j] = h_save  # revert if no improvement
        if not improved and sw > 3:
            break
    return h, best_c5
```

```python
def compute_sensitivity(h, n):
    """Compute ∂C_k/∂h[j] for all j, k."""
    # sensitivity[j, k] = ∂C_k/∂h[j] where k is the lag (-(n-1) to n-1)
    # We'll store it as sensitivity[j, idx] where idx = k + (n-1)
    sens = np.zeros((n, 2*n-1))
    for j in range(n):
        for k in range(n):  # positive lags 0 to n-1
            idx = n - 1 + k
            # h[j] as h[i+k] with i=j-k: valid if 0<=j-k<=n-1-k i.e. k<=j<=n-1
            if k <= j <= n-1:
                sens[j, idx] += (1 - h[j - k]) * (2.0/n)
            # h[j] as h[i] with i=j: valid if 0<=j<=n-1-k
            if 0 <= j <= n-1-k:
                sens[j, idx] -= h[j + k] * (2.0/n)
        for la in range(1, n):  # negative lags
```

```python
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        nf = len(best_h)
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        best_h = do_slsqp(best_h, nf, tf, 3000, bf, 5000, 1e-15)
        best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - t0)
    if rem > 20:
        nf = len(best_h)
        best_h, best_c5 = fast_coord_descent(best_h, nf, 50, t0, budget_s)

    rem = budget_s - (time.time() - t0)
    if rem > 20:
        nf = len(best_h)
        tf = nf / 2.0
        bf = Bounds(np.zeros(nf), np.ones(nf))
        best_h = do_slsqp(best_h, nf, tf, 5000, bf, 5000, 1e-15)
        best_c5 = compute_c5(best_h)

    rem = budget_s - (time.time() - t0)
    if rem > 15:
        nf = len(best_h)
        best_h, best_c5 = fast_coord_descent(best_h, nf, 40, t0, budget_s)

    rem = budget_s - (time.time() - t0)
    if rem > 10:
        nf = len(best_h)
        tf = nf / 2.0
        hb = snap_binary(best_h, tf)
        hb, cb = binary_local_search(hb, nf, 8000, t0, budget_s)
        if cb < best_c5:
            best_h, best_c5 = hb, cb

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
