import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            free = np.ones(len(h), dtype=bool)
        h[free] += diff / free.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_linearized_lp(h_star):
    n = len(h_star)
    n_var = n + 1
    A_rows = []
    b_rows = []
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        row = np.zeros(n_var)
        rhs = 0.0
        if lag == 0:
            row[:n] = 1.0 - 2.0 * h_star
            rhs = -np.sum(h_star * (1.0 - h_star))
        elif lag > 0:
            for j in range(n):
                c = 0.0
                if j <= n - 1 - lag:
                    c += 1.0 - h_star[j + lag]
                if j >= lag:
                    c -= h_star[j - lag]
                row[j] = c
            rhs = -np.sum(h_star[:n - lag] * (1.0 - h_star[lag:]))
        else:
            m = -lag
            for j in range(n):
                c = 0.0
                if j >= m:
                    c += 1.0 - h_star[j - m]
                if j <= n - 1 - m:
                    c -= h_star[j + m]
                row[j] = c
            rhs = -np.sum(h_star[m:] * (1.0 - h_star[:n - m]))
        row[-1] = -1.0
        A_rows.append(row)
        b_rows.append(rhs)
    A = np.array(A_rows)
    b = np.array(b_rows)
    A_eq = np.zeros((1, n_var))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    c_obj = np.zeros(n_var)
    c_obj[-1] = 1.0
    bounds = [(0, 1)] * n + [(None, None)]
    res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return project_h(res.x[:n], n / 2.0)
    return h_star.copy()

def linearization_method(h0, time_fn, max_iters=500):
    n = len(h0)
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    for it in range(max_iters):
        if time_fn() < 10:
            break
        h_new = solve_linearized_lp(h_cur)
        h_trial = 0.5 * h_cur + 0.5 * h_new
        h_trial = project_h(h_trial, n / 2.0)
        c5_trial = compute_c5(h_trial)
        if c5_trial < c5_cur - 1e-10:
            h_cur = h_trial
            c5_cur = c5_trial
        else:
            h_trial2 = 0.3 * h_cur + 0.7 * h_new
            h_trial2 = project_h(h_trial2, n / 2.0)
            c5_trial2 = compute_c5(h_trial2)
            if c5_trial2 < c5_cur - 1e-10:
                h_cur = h_trial2
                c5_cur = c5_trial2
            else:
                if it > 20:
                    break
    return h_cur, c5_cur

def smooth_max_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    x_shift = np.clip(beta * (corr - m), -500, 500)
    exp_vals = np.exp(x_shift)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    val = (m + np.log(sum_exp) / beta) * dx
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k in active:
        w = weights[k] * dx
        lag = int(k) - (n - 1)
        if lag == 0:
            grad += w * (1.0 - 2.0 * h)
        elif 0 < lag < n:
            grad[lag:] += w * (1.0 - h[:-lag])
            grad[:n - lag] -= w * h[lag:]
        elif lag < 0 and abs(lag) < n:
            m2 = -lag
            grad[:n - m2] += w * (1.0 - h[m2:])
            grad[m2:] -= w * h[:n - m2]
    return val, grad

_beta_box = [100.0]
_target_box = [100.0]

def _obj(x):
    return smooth_max_and_grad(x, _beta_box[0])[0]

def _grad(x):
    return smooth_max_and_grad(x, _beta_box[0])[1]

def _con(x):
    return x.sum() - _target_box[0]

def _con_jac(x):
    return np.ones(len(x))

def do_slsqp(h0, betas, maxiter_each, time_fn):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    _target_box[0] = target
    h_cur = h0.copy()
    c5_cur = compute_c5(h_cur)
    constraints = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
    for beta in betas:
        if time_fn() < 5:
            break
        _beta_box[0] = beta
        try:
            res = minimize(_obj, h_cur, method='SLSQP', bounds=bounds,
                           jac=_grad, constraints=constraints,
                           options={'maxiter': maxiter_each, 'ftol': 1e-16})
            h_new = project_h(res.x, target)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur - 1e-14:
                h_cur = h_new
                c5_cur = c5_new
        except Exception:
            pass
    return h_cur, c5_cur

def do_subgrad(h, time_fn, n_iter=50000, top_k=10):
    n = len(h)
    target = n / 2.0
    no_imp = 0
    for it in range(n_iter):
        if time_fn() < 3:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        nk = min(top_k, max(5, n // 30))
        top_idx = np.argsort(corr)[::-1][:nk]
        grad = np.zeros(n)
        for idx in top_idx:
            lag = int(idx) - (n - 1)
            if lag == 0:
                grad += (1.0 - 2.0 * h)
            elif 0 < lag < n:
                grad[lag:] += (1.0 - h[:-lag])
                grad[:n - lag] -= h[lag:]
            elif lag < 0 and abs(lag) < n:
                m2 = -lag
                grad[:n - m2] += (1.0 - h[m2:])
                grad[m2:] -= h[:n - m2]
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        grad /= gnorm
        c5_cur = compute_c5(h)
        found = False
        for step in [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005]:
            ht = project_h(h - step * grad, target)
            c5t = compute_c5(ht)
            if c5t < c5_cur - 1e-14:
                h = ht
                found = True
                no_imp = 0
                break
        if not found:
            no_imp += 1
            if no_imp % 150 == 149:
                sc = 0.004 * min(5, no_imp // 150 + 1)
                ht = project_h(h + np.random.randn(n) * sc, target)
                if compute_c5(ht) < c5_cur:
                    h = ht
                    no_imp = 0
            if no_imp > n_iter // 8:
                break
    return h, compute_c5(h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    def time_left():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(500, 0.5)
    best_c5 = compute_c5(best_h)

    if c5_lin < best_c5:
        best_h, best_c5 = h_lin, c5_lin

                          maxiter_each=3000, time_fn=time_left)
    if c5_s < best_c5:

    # Phase 3: More linearization rounds
    if time_left() > 60:
        h_lin2, c5_lin2 = linearization_method(best_h, time_left, max_iters=150)
        if c5_lin2 < best_c5:
            best_h, best_c5 = h_lin2, c5_lin2

    # Phase 4: Higher resolution
    n_high = 800
    if time_left() > 180:
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_high)
        h_high = np.interp(x_new, x_old, best_h)
        h_high = project_h(h_high, n_high / 2.0)
        h_high, c5_high = do_slsqp(h_high, [200, 1000, 5000, 20000, 60000],
                                    maxiter_each=2000, time_fn=time_left)
        if c5_high < best_c5:
            best_h, best_c5 = h_high, c5_high
        if time_left() > 60:
            h_high2, c5_high2 = linearization_method(best_h, time_left, max_iters=100)
            if c5_high2 < best_c5:
                best_h, best_c5 = h_high2, c5_high2
            if time_left() > 30:
                h_high3, c5_high3 = do_slsqp(best_h, [50000, 150000, 300000],
                                              maxiter_each=5000, time_fn=time_left)
                if c5_high3 < best_c5:
                    best_h, best_c5 = h_high3, c5_high3

    # Phase 5: Subgradient polish
    if time_left() > 20:
        best_h, best_c5 = do_subgrad(best_h, time_left, n_iter=40000, top_k=12)

    # Phase 6: Final SLSQP
    if time_left() > 15:
        n_f = len(best_h)
        h_f, c5_f = do_slsqp(best_h, [100000, 300000, 500000],
                              maxiter_each=8000, time_fn=time_left)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    # Phase 7: Tiny random perturbations + local search
    if time_left() > 10:
        n_f = len(best_h)
        for attempt in range(10):
            if time_left() < 5:
                break
            h_p = project_h(best_h + np.random.randn(n_f) * 0.002, n_f / 2.0)
            h_p, c5_p = do_slsqp(h_p, [50000, 200000],
                                  maxiter_each=3000, time_fn=time_left)
            if c5_p < best_c5:
                best_h, best_c5 = h_p, c5_p

    return best_h, best_c5, len(best_h)
