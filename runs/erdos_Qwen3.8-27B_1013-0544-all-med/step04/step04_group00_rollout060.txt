```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_all_g(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def compute_c5(h, n):
    return float(compute_all_g(h, n).max())

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            s = room.sum()
            if s < 1e-14:
                break
            h = h + diff / s * room
        else:
            s = h.sum()
            if s < 1e-14:
                break
            h = h + diff / s * h
    return np.clip(h, 0.0, 1.0)

def compute_smoothed_grad(h, n, alpha):
    """Compute gradient of smooth max surrogate using analytical formula."""
    g = compute_all_g(h, n)
    # Softmax weights
    g_shifted = g - g.max()
    p = np.exp(alpha * g_shifted)
    p /= p.sum()
    grad = compute_weighted_grad(h, n, p)
    return grad, g.max()

def compute_weighted_grad(h, n, p):
    """
    p: weights over lags, length 2n-1, summing to 1
    Returns gradient of sum_k p[k] * g(k) w.r.t. h
    """
    grad = np.ones(n).copy()
    n_half = n - 1
    # Only iterate over significant p
    thresh = 1e-12
    for m in range(2 * n - 1):
        w = p[m]
        if w < thresh:
            continue
        k = m - n_half
        if k >= 0:
            grad[:n - k] -= w * h[k:]
            grad[k:] -= w * h[:n - k]
        else:
            kk = -k
            grad[kk:] -= w * h[:n - kk]
            grad[:n - kk] -= w * h[kk:]
    return grad

def projected_grad_descent(h, n, deadline, alpha_init=100, alpha_max=100000, max_iter=50000):
    """Projected gradient descent with backtracking line search."""
    best_h = h.copy()
    best_c = compute_c5(h, n)
    alpha = alpha_init
    lr = 1.0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        g = compute_all_g(h, n)
        current_max = g.max()
        
        # Compute softmax weights
        g_shifted = g - current_max
        p = np.exp(alpha * g_shifted)
        p /= p.sum()
        
        # Compute gradient
        grad = compute_weighted_grad(h, n, p)
        
        # Project gradient direction: we want to minimize, so move in -grad direction
        # But also need to maintain the constraint sum(h) = n/2
        # The constraint gradient is ones, so project out the component along ones
        grad_mean = grad.mean()
        grad_centered = grad - grad_mean
        
        # Backtracking line search
        found = False
        for _ in range(30):
            new_h = project_feasible(h - lr * grad_centered, n)
            new_c = compute_c5(new_h, n)
            if new_c < current_max - 1e-15:
                h = new_h
                found = True
                break
            lr *= 0.5
        
        if found:
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            # Slowly increase alpha to focus more on the max
            alpha = min(alpha * 1.01, alpha_max)
        else:
            lr *= 0.5
            if lr < 1e-10:
```

```python
    # i as second (old): -h[::-1] at lags [i-n_half, i] → indices [i, i+n_half]
    delta[i : i + n] -= h[::-1]
    
    # j as first (old): -(1-h) at lags [-j, n-1-j] → indices [n_half-j, n_half+n-1-j]
    delta[n_half - j : n_half + n - j] -= (1.0 - h)
    
    # j as second (new): h_new[::-1] at lags [j-n_half, j] → indices [j, j+n_half]
    delta[j : j + n] += h_new[::-1]
    
    # Correction for double-counted pairs
    if j - i != 0:  # j+i is valid lag
        delta[n_half + (j - i)] -= 1.0
    if i - j != 0:
        delta[n_half + (i - j)] += 1.0
    
    return delta
```
