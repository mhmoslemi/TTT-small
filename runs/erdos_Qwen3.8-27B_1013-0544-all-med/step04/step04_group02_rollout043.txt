def compute_grad_weighted(h, weights, lags, n):
    """Compute gradient of weighted sum of correlations at given lags."""
    grad = np.zeros(n)
    for w, k in zip(weights, lags):
        # k is the index in the full correlation (0 to 2n-2)
        # actual lag is k - (n-1)
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            idx = np.arange(m)
            grad[idx] += w * (1.0 - h[idx + lag])
            grad[idx + lag] -= w * h[idx]
        else:
            la = -lag
            m = n - la
            idx = np.arange(m)
            grad[idx + la] += w * (1.0 - h[idx])
            grad[idx] -= w * h[idx + la]
    return grad

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def weighted_corr_grad(h, weights, corr_indices, n):
    """Gradient of sum_j weights[j] * corr[corr_indices[j]] with respect to h."""
    grad = np.zeros(n)
    for w, ci in zip(weights, corr_indices):
        lag = ci - (n - 1)
        if lag >= 0:
            m = n - lag
            idx = np.arange(m)
            grad[idx] += w * (1.0 - h[idx + lag])
            grad[idx + lag] -= w * h[idx]
        else:
            la = -lag
            m = n - la
            idx = np.arange(m)
            grad[idx + la] += w * (1.0 - h[idx])
            grad[idx] -= w * h[idx + la]
    return grad

def lbfgs_stage(h, n, deadline, seed, max_outer=50):
    """L-BFGS-B optimization with adaptive active-set focus."""
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = [(0.0, 1.0)] * n
    c5 = compute_c5(h)
    no_improve = 0
    
    for outer in range(max_outer):
        if time.time() > deadline:
            break
        
        # Identify active lags
        corr = np.correlate(h, 1.0 - h, mode='full')
        K = min(15, len(corr))
        top_idx = np.argsort(corr)[-K:]
        top_idx = sorted(top_idx)
        
        # Softmax weights (temperature adapts based on spread)
        top_vals = corr[top_idx]
        spread = top_vals.max() - top_vals.min() + 1e-10
        temp = max(spread * 0.3, 0.01)
        w = np.exp((top_vals - top_vals.max()) / temp)
        w /= w.sum()
        
        # Define penalized objective
        penalty = 100.0
        beta_sm = 50.0
        
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            active_c = c[top_idx]
            sm = smooth_max(active_c, beta_sm)
            return sm * dx + penalty * (x.sum() - ts) ** 2
        
        def grad(x):
            g = weighted_corr_grad(x, w, top_idx, n)
            g += 2.0 * penalty * (x.sum() - ts)
            return g
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 200, 'gtol': 1e-8})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5 - 1e-12:
                h = h_new
                c5 = c5_new
                no_improve = 0
            else:
                no_improve += 1

                if no_improve >= 3:
                    h = project_feasible(h + np.random.randn(n) * 0.01, ts)
                    no_improve = 0
        except Exception:
            no_improve += 1
    
    return h, c5

corr_j(h) = sum_m h[m] - sum_m h[m]*h[m+lag_j]

Let S1_j = sum_m h[m] (over valid m for this lag)
Let S2_j = sum_m h[m]*h[m+lag_j] (over valid m for this lag)

Then corr_j = S1_j - S2_j.

Now, if I change h[i] to x:
S1_j changes by: x - h[i] (if i is in the valid range for S1_j)
S2_j changes by: 
- If i is in valid range as first factor: x * h[i+lag_j] - h[i] * h[i+lag_j] = (x - h[i]) * h[i+lag_j]
- If i is in valid range as second factor (i.e., i - lag_j is valid): h[i-lag_j] * x - h[i-lag_j] * h[i] = h[i-lag_j] * (x - h[i])

So S2_j changes by: (x - h[i]) * (h[i+lag_j] * I1 + h[i-lag_j] * I2)

where I1 = 1 if i is valid as first factor for this lag, I2 = 1 if i-lag_j is valid (i.e., i is valid as second factor).

Therefore:
corr_j(x) = corr_j(h) + (x - h[i]) * (I1 - h[i+lag_j]*I1 - h[i-lag_j]*I2)
= corr_j(h) + (x - h[i]) * (I1*(1 - h[i+lag_j]) - h[i-lag_j]*I2)

So each correlation is LINEAR in x! That's great for coordinate descent.

Let me define for each coordinate i and each correlation j:

Then corr_j(x) = corr_j(h) + (x - h[i]) * a_ij

The smooth max objective becomes:

            I2 = (lag <= i <= n - 1)

def golden_section_min(f, a, b, tol=1e-10, maxiter=50):
    gr = (np.sqrt(5) - 1) / 2  # golden ratio
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(maxiter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2
```

    vals = corr_vals + (x - cur) * a_vec
    m = np.max(vals)
    e = np.exp(np.clip(beta * (vals - m), -500, 500))
    w = e / e.sum()
    return np.sum(w * a_vec)

def f1d_min(corr_vals, a_vec, cur, beta, tol=1e-10, maxiter=50):
    # Bisection for f'(x) = 0 on [0, 1]
    lo, hi = 0.0, 1.0
    dlo = f1d_deriv(lo, corr_vals, a_vec, cur, beta)
    dhi = f1d_deriv(hi, corr_vals, a_vec, cur, beta)
    if dlo >= 0:
        return lo
    if dhi <= 0:
        return hi
    for _ in range(maxiter):
        mid = (lo + hi) / 2
        dmid = f1d_deriv(mid, corr_vals, a_vec, cur, beta)
        if hi - lo < tol:
            break
        if dmid > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, n_new / 2.0)

def compute_a_vec(h, i, n):
    """Compute coefficient vector a_j for the 1D subproblem at coordinate i."""
    a = np.zeros(2 * n - 1)
    lags = np.arange(2 * n - 1) - (n - 1)
    
    # Positive lags (index n-1 to 2n-2)
    pos_mask = lags >= 0
    lags_p = lags[pos_mask]
    I1 = (i <= n - 1 - lags_p)
    I2 = (i >= lags_p)
    ap = np.zeros(len(lags_p))
    ap[I1] += (1.0 - h[i + lags_p[I1]])
    ap[I2] -= h[i - lags_p[I2]]
    a[pos_mask] = ap
    
    # Negative lags (index 0 to n-2)
    neg_mask = lags < 0
    lags_n = lags[neg_mask]
    la = -lags_n
    I1n = (i >= la)
    I2n = (i <= n - 1 - la)
    an = np.zeros(len(lags_n))
    an[I1n] += (1.0 - h[i + lags_n[I1n]])
    an[I2n] -= h[i - lags_n[I2n]]
    a[neg_mask] = an
    
    return a

def f1d_deriv(x, corr_vals, a_vec, cur, beta, penalty, sum_h, ts):
    """Derivative of the penalized objective at coordinate value x."""
    vals = corr_vals + (x - cur) * a_vec
    m = np.max(vals)
    e = np.exp(np.clip(beta * (vals - m), -500, 500))
    w = e / e.sum()
    d = np.sum(w * a_vec)
    d += 2.0 * penalty * (sum_h + (x - cur) - ts)
    return d

def f1d_minimize(corr_vals, a_vec, cur, beta, penalty, sum_h, ts):
    """Find x in [0,1] minimizing the 1D subproblem using bisection on derivative."""
    dlo = f1d_deriv(0.0, corr_vals, a_vec, cur, beta, penalty, sum_h, ts)
    dhi = f1d_deriv(1.0, corr_vals, a_vec, cur, beta, penalty, sum_h, ts)
    if dlo >= 0:
        return 0.0
    if dhi <= 0:
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        dmid = f1d_deriv(mid, corr_vals, a_vec, cur, beta, penalty, sum_h, ts)
        if hi - lo < 1e-12:
            break
        if dmid > 0:
            hi = mid
        else:
            lo = mid
    return (lo + hi) * 0.5

def coord_descent(h, n, deadline, seed, beta_start=50, beta_end=500, penalty=50.0, max_cycles=200000):
    np.random.seed(seed)
    ts = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    no_improve = 0
    total_cycles = max_cycles
    
    for cycle in range(total_cycles):
        if time.time() > deadline:
            break
        
        # Adaptive beta
        frac = min(cycle / (total_cycles * 0.5), 1.0)
        beta = beta_start + (beta_end - beta_start) * frac
        
        for i in range(n):
            cur = h[i]
            a_vec = compute_a_vec(h, i, n)
            x_new = f1d_minimize(corr, a_vec, cur, beta, penalty, h.sum(), ts)
            h[i] = x_new
            corr = np.correlate(h, 1.0 - h, mode='full')  # Update after each coordinate
        
        # Project to feasible set
        h = project_feasible(h, ts)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_new = float(np.max(corr) * dx)
        if c5_new < c5 - 1e-11:
            c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 15:
                h = project_feasible(h + np.random.randn(n) * 0.01, ts)
                corr = np.correlate(h, 1.0 - h, mode='full')
                c5 = float(np.max(corr) * dx)
                no_improve = 0
            if no_improve >= 50:
                break
    
    return h, c5
```

Wait, there's a problem with updating `corr` after each coordinate change. Computing the full correlation is O(n^2) (or O(n log n) with FFT). If I do this n times per cycle, it's O(n^3) per cycle. For n=350, that's 43 million operations per cycle. That's too slow if I want many cycles.

Better approach: update `corr` incrementally. When I change h[i] from cur to x_new, the change in correlation is:
corr_new = corr + (x_new - cur) * a_vec

where a_vec is the coefficient vector I already computed! So I don't need to recompute the full correlation after each coordinate update.

Let me fix this:

from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):

    an[I2n] -= h[i - lags_n[I2n]]

def coord_descent(h, n, deadline, seed, beta_lo=40, beta_hi=800, penalty=10.0):
    np.random.seed(seed)
    ts = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    no_improve = 0
    cycle = 0
    while time.time() < deadline:
        cycle += 1
        frac = min(cycle / 300.0, 1.0)

        penalty_now = penalty * (1.0 + 0.5 * frac)
        for i in range(n):

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def con_fun(h, ts):
    return h.sum() - ts

def con_jac(h, ts):
    return np.ones_like(h)

def slsqq_refine(h, n, deadline, seed):
    np.random.seed(seed)
    dx = 2.0 / n
    ts = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac, 'args': (ts,)}]
    c5 = compute_c5(h)
    for beta in [200, 1000, 5000]:
        if time.time() > deadline:
            break
        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return smooth_max(corr, beta) * dx
        def grad(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            m = np.max(corr)
            e = np.exp(np.clip(beta * (corr - m), -500, 500))
            w = e / e.sum()
            g = np.zeros(n)
            lags = np.arange(2 * n - 1) - (n - 1)
            for j in range(2 * n - 1):
                lag = lags[j]
                if abs(w[j]) < 1e-15:
                    continue
                if lag >= 0:
                    mp = n - lag
                    idx = np.arange(mp)
                    g[idx] += w[j] * (1.0 - x[idx + lag])
                    g[idx + lag] -= w[j] * x[idx]
                else:
                    la = -lag
                    mp = n - la
                    idx = np.arange(mp)
                    g[idx + la] += w[j] * (1.0 - x[idx])
                    g[idx] -= w[j] * x[idx + la]
            return g * dx
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=bounds, options={'maxiter': 300, 'gtol': 1e-9})
            h_new = project_feasible(res.x, ts)
            c5_new = compute_c5(h_new)
            if c5_new < c5:
                h = h_new
                c5 = c5_new
        except Exception:
            pass
    return h, c5
