                best_h = h2.copy()

    # Phase 3: SLSQP polish (20% budget)
    p3 = t0 + 0.90 * (budget_s - 3)
    if time.time() < p3 - 5:
        h_cur = best_h.copy()
        for alpha in [2000, 20000, 200000]:
            if time.time() > p3 - 5:
                break
            h_try = _slsqp_polish(h_cur, n, alpha, 3000, p3 - 5)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_c = c_try
                best_h = h_try
                h_cur = h_try

    # Phase 4: Try nearby resolutions (8% budget)
    p4 = dl - 8
    if time.time() < p4:
        for nh in [n - 25, n + 25]:
            if time.time() > p4:
                break
            if nh < 30 or nh > 1000:
                continue
            ev = np.arange(-(nh - 1), nh, dtype=np.int64)
            m0h = nh - 1
            xco = np.linspace(0.0, 1.0, 400)
            xo = np.linspace(0, 2, n)
            xn = np.linspace(0, 2, nh)
            hh = project_feasible(np.interp(xn, xo, best_h), nh)
            for _ in range(100):
                if time.time() > p4 - 3:
                    break
                _sweep(hh, nh, ev, m0h, xco, p4 - 3)
                hh = project_feasible(hh, nh)
            c_try = compute_c5(hh, nh)
            if c_try < best_c:
                best_c = c_try
                best_h = hh
                n = nh

    # Final polish
    if time.time() < dl - 1:
        h_try = _slsqp_polish(best_h, n, 500000, 2000, dl - 1)
        c_try = compute_c5(h_try, n)
        if c_try < best_c:
            best_c = c_try
            best_h = h_try

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
