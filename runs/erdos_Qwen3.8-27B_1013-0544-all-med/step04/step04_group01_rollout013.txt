def compute_grad_smooth(h, n, alpha):
    """Compute gradient of smooth max objective."""
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')  # shape (2n-1,)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            s = r.sum()
            if s < 1e-15: break
            h += (d / s) * r
        else:
            s = h.sum()
            if s < 1e-15: break
            h += (d / s) * h
    return np.clip(h, 0.0, 1.0)

def smooth_obj(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 400:
        return float(m + (mx + np.log(np.exp(0) + np.sum(np.exp(s - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(s))) / alpha)

def smooth_grad(x, n, alpha):
    dx = 2.0 / n
    c = np.correlate(x, 1 - x, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    sm = s.max()
    w = np.exp(s - sm)
    w /= w.sum()
    grad = np.zeros(n)
    for L in range(-(n-1), n):
        wL = w[n-1+L]
        if wL < 1e-14:
            continue
        i_lo = max(0, -L)
        i_hi = min(n-1, n-1-L)
        idx = np.arange(i_lo, i_hi+1)
        grad[idx] += wL * dx * (1.0 - x[idx+L])
        grad[idx+L] -= wL * dx * x[idx]
    return grad / alpha

def slsqp_opt(h, n, alpha, maxiter=2000):
    b = [(0.0, 1.0)] * n
    def f(x): return smooth_obj(x, n, alpha)
    def g(x): return smooth_grad(x, n, alpha)
    def feq(x): return float(x.sum() - n/2.0)
    def g_eq(x): return np.ones(n)
    try:
        r = minimize(f, h, jac=g, method='SLSQP',
                     bounds=b, constraints=[{'type':'eq','fun':feq,'jac':g_eq}],
                     options={'maxiter':maxiter, 'ftol':1e-16})
        return project_feasible(r.x, n)
    except:
        return h.copy()

def projected_grad_lbfgs(h, n, alpha, maxiter=2000, lr_init=0.01):
    """Custom L-BFGS with projection."""
    h = project_feasible(h, n).copy()
    f_best = compute_c5(h, n)
    h_best = h.copy()
    lr = lr_init
    for it in range(maxiter):

        # Project gradient: only move in directions that maintain feasibility
        # Simple approach: take step, then project
        h_new = h - lr * g
        h_new = project_feasible(h_new, n)
        f_new = smooth_obj(h_new, n, alpha)
        f_old = smooth_obj(h, n, alpha)
        if f_new < f_old:
            h = h_new
            lr *= 1.05
            c = compute_c5(h, n)
            if c < f_best:
                f_best = c
                h_best = h.copy()
        else:
            lr *= 0.5
            if lr < 1e-8:

    return h_best, f_best

def pair_search(h, n, deadline, deltas):
    best_c = compute_c5(h, n)
    for d in deltas:

            break

                        c = compute_c5(h, n)
                        if c < best_c - 1e-15:

            if not improved:

    return h, best_c
