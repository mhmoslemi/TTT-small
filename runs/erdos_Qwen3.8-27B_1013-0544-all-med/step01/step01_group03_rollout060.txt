import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

Let a_k = delta_k and b_k = c_k. We want to minimize max_k (b_k + a_k * delta).

This is a 1D LP:
minimize t
subject to: b_k + a_k * delta ≤ t for all k
           -h[i] ≤ delta ≤ 1-h[i]

The optimum delta* is where the upper envelope of the lines {b_k + a_k * delta} has its minimum.

Algorithm to solve:
1. Compute all "break points" delta_kj where b_k + a_k * delta = b_j + a_j * delta, i.e., delta = (b_j - b_k) / (a_k - a_j) for a_k ≠ a_j.
2. Filter to [-h[i], 1-h[i]].
3. Also add the endpoints -h[i] and 1-h[i].
4. Evaluate max_k (b_k + a_k * delta) at each candidate.
5. Return the best delta.

The number of break points is O(m²) where m = 2n-1, but many will be outside the interval. In practice, we can be smarter:

Since the function is convex, we can use the following:
- Start at delta = 0 (current value).
- Find the active k (the one with max b_k + a_k * 0 = b_k, i.e., argmax b_k).
- If a_{k*} < 0, the function is decreasing at delta=0, so we should increase delta.
- If a_{k*} > 0, decrease delta.
- If a_{k*} = 0, we're at the minimum (or a flat region).
- Move delta in the indicated direction until another line becomes active or we hit a boundary.

This is the "active set" method and takes at most m iterations.

def obj_and_grad(x, n, tau, penalty=50.0):
    target = n / 2.0
    corr = np.correlate(x, 1.0 - x, mode="full")
    
    # Smooth max
    tc = tau * corr
    tc_max = np.max(tc)
    exp_tc = np.exp(tc - tc_max)
    sum_exp = np.sum(exp_tc)
    sm = (tc_max + np.log(sum_exp)) / tau
    weights = tau * exp_tc / sum_exp  # shape (2n-1,)
    
    # Gradient of smooth max w.r.t. x
    # d(sm)/d(corr[j]) = weights[j]
    # d(corr[j])/d(x[m]) = (1-x[m+k])_valid - x[m-k]_valid where k = j-(n-1)
    
    grad_sm = np.zeros(n)
    lags = np.arange(2*n-1) - (n-1)
    for j in range(2*n-1):
        k = lags[j]
        w = weights[j]
        if w == 0:
            continue
        # (1-x[m+k]) for valid m: m ranges from max(0,-k) to min(n-1, n-1-k)
        m_start = max(0, -k)
        m_end = min(n-1, n-1-k)
        if m_start <= m_end:

        m_start2 = max(0, k)
        m_end2 = min(n-1, n-1+k)
        if m_start2 <= m_end2:
            grad_sm[m_start2:m_end2+1] -= w * x[m_start2-k:m_end2-k+1]
    
    # Sum constraint penalty
    sv = (x.sum() - target) ** 2
    grad_sv = 2 * penalty * (x.sum() - target) * np.ones(n)
    
    obj = sm + penalty * sv
    grad = grad_sm + grad_sv
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def fix_sum(h, target):
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0)
    d = target - h.sum()
    if abs(d) < 1e-13:
        return h
    msk = h < 0.9999 if d > 0 else h > 1e-4
    c = msk.sum()
    if c > 0:
        h[msk] += d / c
    return np.clip(h, 0.0, 1.0)

class LBFGSObj:
    def __init__(self, n, tau, penalty, active=None):
        self.n = n
        self.tau = tau
        self.penalty = penalty
        self.target = 0.5 * n
        self.active = active
    
    def f(self, x):
        return self._calc(x)[0]
    
    def g(self, x):
        return self._calc(x)[1]
    
    def _calc(self, x):
        n = self.n
        corr = np.correlate(x, 1.0 - x, mode="full")
        tc = self.tau * corr
        tc -= tc.max()
        e = np.exp(tc)
        se = e.sum()
        sm = (tc.max() + np.log(se)) / self.tau
        w = self.tau * e / se
        if self.active is not None:
            mask = np.ones(2*n-1, dtype=bool)
            mask[self.active] = False
            w[mask] *= 0.05
            w /= w.sum()
        grad = np.zeros(n)
        lags = np.arange(2*n-1) - (n-1)
        for j in range(2*n-1):
            k = lags[j]
            wj = w[j]
            if wj < 1e-14:
                continue
            ms = max(0, -k)
            me = min(n-1, n-1-k)
            if ms <= me:
                grad[ms:me+1] += wj * (1.0 - x[ms+k:me+k+1])
            ms2 = max(0, k)
            me2 = min(n-1, n-1+k)
            if ms2 <= me2:
                grad[ms2:me2+1] -= wj * x[ms2-k:me2-k+1]
        sv = (x.sum() - self.target) ** 2
        return sm + self.penalty * sv, grad + 2.0 * self.penalty * (x.sum() - self.target)

def try_optimize(h0, n, tau, penalty, maxiter, active=None):
    obj = LBFGSObj(n, tau, penalty, active)
    bnds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj.f, h0, jac=obj.g, method='L-BFGS-B',
                       bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        x = fix_sum(res.x, n / 2.0)
        return x, evaluate_c5(x)
    except Exception:
        return h0, None

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(120, 0.5)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    n0 = len(best_h)
    
    # Phase 1: Annealing
    taus_p1 = [500, 200, 100, 50, 20, 10, 5, 3, 2, 1.5, 1, 0.7]
    for tau in taus_p1:
        if rem() < 25:
            break
        n = len(best_h)
        h_new, c5_new = try_optimize(best_h, n, tau, 50.0, 400)
        if c5_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
    
    # Phase 2: Chase active lags
    for _ in range(6):
        if rem() < 25:
            break
        n = len(best_h)
        corr = np.correlate(best_h, 1.0 - best_h, mode="full")
        mx = corr.max()
        act = np.where(corr > mx - 0.3)[0]
        h_new, c5_new = try_optimize(best_h, n, 2.0, 50.0, 200, act)
        if c5_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
        h_new, c5_new = try_optimize(best_h, n, 1.0, 50.0, 200)
        if c5_new is not None and c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
    
    # Phase 3: Multi-start
    n_best = len(best_h)
    n_starts = 40
    for i in range(n_starts):
        if rem() < 15:
            break
        sc = 0.01 + 0.06 * np.random.random()
        hp = best_h + np.random.normal(0, sc, n_best)
        hp = fix_sum(hp, n_best / 2.0)
        for tau in [5, 2, 1]:
            if rem() < 8:
                break
            h_new, c5_new = try_optimize(hp, n_best, tau, 50.0, 150)
            if c5_new is not None and c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            hp = h_new
    
    # Phase 4: Different n
    for tn in [100, 140, 160, 200]:
        if rem() < 35:
            break
        nc = len(best_h)
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, nc), best_h)
        h_t = fix_sum(h_t, tn / 2.0)
        for tau in [50, 10, 3, 1]:
            if rem() < 12:
                break
            h_new, c5_new = try_optimize(h_t, tn, tau, 50.0, 200)
            if c5_new is not None and c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            h_t = h_new
    
    # Phase 5: Polish
    if rem() > 15:
        nf = len(best_h)
        for tau in [0.5, 0.3, 0.2]:
            if rem() < 8:
                break
            h_new, c5_new = try_optimize(best_h, nf, tau, 100.0, 300)
            if c5_new is not None and c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
    
    best_h = fix_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)

Let me also double-check the gradient computation. For the correlation:
corr[j] = sum_{i: valid} x[i] * (1-x[i+lag_j]) where lag_j = j-(n-1)

d(corr[j])/d(x[m]):
- If m is used as the "i" term: contributes (1-x[m+lag_j]) if 0 ≤ m+lag_j < n
- If m is used as the "i+lag_j" term (i.e., m = i+lag_j, so i = m-lag_j): contributes -x[m-lag_j] if 0 ≤ m-lag_j < n

In my code, k = lags[j]:
- First term: x[m+k] valid when 0 ≤ m+k < n, so m ∈ [-k, n-1-k] ∩ [0, n-1] = [max(0,-k), min(n-1, n-1-k)]
- Second term: x[m-k] valid when 0 ≤ m-k < n, so m ∈ [k, n-1+k] ∩ [0, n-1] = [max(0,k), min(n-1, n-1+k)]

This matches my code. ✓
