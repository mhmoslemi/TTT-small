        if val > target_sum:

        else:
            hi = mid

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(h, corr, tau):
    n = len(h)
    dx = 2.0 / n
    x = tau * corr
    x_max = np.max(x)
    p = np.exp(x - x_max)
    p /= p.sum()
    p[p < 1e-12] = 0.0
    grad = np.zeros(n)
    nm1 = n - 1
    for k in range(2 * n - 1):
        pk = p[k]
        if pk <= 0:
            continue
        s = k - nm1
        if s >= 0:
            if s < n:
                grad[:n - s] += pk * (1.0 - h[s:])
                grad[s:] -= pk * h[:n - s]
        else:
            sp = -s
            if sp < n:
                grad[sp:] += pk * (1.0 - h[:n - sp])
                grad[:n - sp] -= pk * h[sp:]
    return dx * grad
