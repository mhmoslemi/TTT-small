We want to maximize min_k autocorr_a(k) (to minimize max_k corr_k).

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.append(h_old, h_old[0])
    x_ext = np.append(x_old, 2.0 + 1.0 / n_old)
    return np.interp(x_new, x_ext, h_ext)

def coord_refine(h, target_sum, deadline, max_sweeps=50):
    n = len(h)
    best_c5 = compute_c5(h)
    h = h.copy()
    for _ in range(max_sweeps):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if h[i] < 0.05 and i != np.argmax(h):
                j = int(np.argmax(h))
                h_try = h.copy()
                delta = min(1.0 - h[i], h[j])
                h_try[i] += delta
                h_try[j] -= delta
                h_try = project_box_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
            elif h[i] > 0.95 and i != np.argmin(h):
                j = int(np.argmin(h))
                h_try = h.copy()
                delta = min(h[i], 1.0 - h[j])
                h_try[i] -= delta
                h_try[j] += delta
                h_try = project_box_sum(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-14:
                    h = h_try
                    best_c5 = c5_try
                    improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n = 200
        h0 = np.zeros(n)
        idx = rng.choice(n, n // 2, replace=False)
        h0[idx] = 1.0

    n = len(h0)
    target_sum = n / 2.0
    h0 = project_box_sum(h0, target_sum)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    best_n = n

    elapsed = time.time() - t0
    remaining = budget_s - elapsed

    # Phase 1: Projected gradient with topk=5
    if remaining > 40:
        dl = t0 + min(remaining * 0.3, 300.0)
        best_h, best_c5 = optimize_projected(best_h, best_n / 2.0, dl, rng, topk=5, alpha_init=35)
        best_n = len(best_h)

    # Phase 2: Try higher resolution
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 50:
        for n_try in [300, 400, 500]:
            if remaining < 50:
                break
            h_interp = interpolate_to_n(best_h, best_n, n_try)
            h_interp = project_box_sum(h_interp, n_try / 2.0)
            dl = time.time() + min(remaining * 0.25, 200.0)
            h_opt, c5_opt = optimize_projected(h_interp, n_try / 2.0, dl, rng, topk=4, alpha_init=30)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = n_try
            remaining = budget_s - (time.time() - t0)

    # Phase 3: Sigmoid optimization
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 80:
        dl = time.time() + min(remaining * 0.35, 300.0)
        h_sig, c5_sig = optimize_sigmoid(best_h, dl, rng, topk=5, alpha_init=50)
        if c5_sig < best_c5:
            best_h = h_sig
            best_c5 = c5_sig
            best_n = len(best_h)

    # Phase 4: Another projected gradient round with different params
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 30:
        dl = time.time() + min(remaining * 0.4, 250.0)
        best_h, best_c5 = optimize_projected(best_h, best_n / 2.0, dl, rng, topk=3, alpha_init=80)
        best_n = len(best_h)

    # Phase 5: Coordinate refinement
    elapsed = time.time() - t0
    remaining = budget_s - elapsed
    if remaining > 20:
        dl = t0 + budget_s - 10
        best_h, best_c5 = coord_refine(best_h, best_n / 2.0, dl, max_sweeps=30)

    # Final: short polish
    elapsed = time.time() - t0
    if budget_s - elapsed > 10:
        best_h, best_c5 = optimize_projected(best_h, best_n / 2.0, t0 + budget_s - 3, rng, topk=3, alpha_init=120)

    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
