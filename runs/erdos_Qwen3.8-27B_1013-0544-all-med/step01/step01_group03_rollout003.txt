from scipy.optimize import minimize, Bounds

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(corr, tau):
    x = tau * corr
    x_max = np.max(x)
    return float((x_max + np.log(np.sum(np.exp(x - x_max)))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float).copy()
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0 - 1e-12
    else:
        mask = h > 1e-12
    cnt = mask.sum()
    if cnt == 0:
        return np.clip(h, 0.0, 1.0)
    h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def make_binary(h, n):
    hb = np.where(h > 0.5, 1.0, 0.0)
    target = n / 2.0
    s = hb.sum()
    if s > target:
        ones = np.where(hb == 1.0)[0]
        for i in np.random.choice(ones, int(s - target), replace=False):
            hb[i] = 0.0
    elif s < target:
        zeros = np.where(hb == 0.0)[0]
        for i in np.random.choice(zeros, int(target - s), replace=False):
            hb[i] = 1.0
    return hb

        t = np.empty(p)
        t[:p-1] = x
        t[p-1] = target_p - np.sum(x)
