from scipy.optimize import minimize, Bounds
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_val(obj, tau):
    x = tau * obj
    x_max = np.max(x)
    return float(x_max / tau + np.log(np.sum(np.exp(x - x_max))) / tau)

def fix_sum(h, target):
    h = np.array(h, dtype=float)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

    """Convert h to binary with correct sum."""
    n = len(h)
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]

        h_bin[ones[target:]] = 0
    elif len(ones) < target:
        np.random.shuffle(zeros)
        h_bin[zeros[:target - len(ones)]] = 1
    return h_bin

    """SLSQP optimization with smooth max."""
    target = n_pts / 2.0
    bnds = Bounds(np.zeros(n_pts), np.ones(n_pts))
    
    def objfun(x):
        corr = np.correlate(x, 1.0 - x, mode="full")
        return smooth_max_val(corr, tau)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones(n_pts)}]

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        x = fix_sum(res.x, target)
        c5 = evaluate_c5(x)
        return x, c5
    except Exception:
        return h_start, None

    """Project v onto {h: 0<=h_i<=1, sum(h_i)=S} using bisection."""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    target = n / 2.0

        h_new = h - alpha * grad
        h_new = project_hypersimplex(h_new, target)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
        else:
            alpha_scale *= 0.95
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(60, 0.5)

    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Binary SA with multiple restarts and strategies
    h_bin = make_binary(h_init)
    sa_configs = [
        (0.15, 0.001, 400000),
        (0.08, 0.0005, 300000),
        (0.04, 0.0003, 250000),
    ]
    for restart in range(len(sa_configs)):
        if rem() < 120:
            break
        T_s, T_e, n_steps = sa_configs[restart]
        if restart == 0:
            h_start = h_bin.copy()
        else:
            h_start = make_binary(best_h)
            # Add random perturbation for diversity
            for _ in range(5):
                i = np.random.randint(n0)
                j = np.random.randint(n0)
                if h_start[i] != h_start[j]:
                    h_start[i], h_start[j] = h_start[j], h_start[i]
        h_new, c5_new = binary_sa(h_start, T_s, T_e, n_steps, rem)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 2: Continuous refinement with increasing tau
    h_cont = best_h.copy()
    if h_cont.max() - h_cont.min() < 0.1:
        h_cont = h_cont + 0.02 * np.random.randn(n0)
        h_cont = np.clip(h_cont, 0, 1)
        h_cont = fix_sum(h_cont, n0 / 2.0)

    tau_list = [50, 200, 1000, 5000, 20000, 50000]
    for tau in tau_list:
        if rem() < 25:
            break
        h_new, c5_new = optimize_continuous(h_cont, n0, tau, 800)
        if c5_new is not None and c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        h_cont = h_new if c5_new is not None else h_cont

    # Phase 3: Subgradient refinement
    if rem() > 80:
        h_new, c5_new = subgradient_refine(best_h, 10000, rem, alpha_scale=0.3)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Phase 4: Binary-continuous hybrid loop
    for iteration in range(5):
        if rem() < 60:
            break
        # Round to binary
        h_b = make_binary(best_h)
        # Short SA
        h_b, c5_b = binary_sa(h_b, 0.03, 0.001, 80000, rem)
        if c5_b < best_c5:
            best_h = h_b.copy()
            best_c5 = c5_b
        # Continuous from binary
        h_c = h_b + 0.01 * np.random.randn(len(h_b))
        h_c = np.clip(h_c, 0, 1)
        h_c = fix_sum(h_c, len(h_b) / 2.0)
        for tau in [500, 3000, 10000]:
            if rem() < 20:
                break
            h_new, c5_new = optimize_continuous(h_c, len(h_c), tau, 400)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_c = h_new if c5_new is not None else h_c

    # Phase 5: Try different resolutions
    for tn in [80, 100, 140, 160, 200, 240]:
        if rem() < 50:
            break
        if tn % 2 != 0:
            continue
        h_t = np.interp(np.linspace(0, 1, tn), np.linspace(0, 1, len(best_h)), best_h)
        h_t = fix_sum(h_t, tn / 2.0)
        h_t_bin = make_binary(h_t)
        h_t_bin, c5_t = binary_sa(h_t_bin, 0.05, 0.001, 150000, rem)
        if c5_t < best_c5:
            best_h = h_t_bin.copy()
            best_c5 = c5_t
        h_t2 = h_t_bin + 0.01 * np.random.randn(tn)
        h_t2 = np.clip(h_t2, 0, 1)
        h_t2 = fix_sum(h_t2, tn / 2.0)
        for tau in [200, 2000, 10000]:
            if rem() < 15:
                break
            h_new, c5_new = optimize_continuous(h_t2, tn, tau, 400)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_t2 = h_new if c5_new is not None else h_t2

    # Phase 6: Final high-precision refinement on best
    if rem() > 30:
        hn = len(best_h)
        h_fin = best_h + 0.005 * np.random.randn(hn)
        h_fin = np.clip(h_fin, 0, 1)
        h_fin = fix_sum(h_fin, hn / 2.0)
        for tau in [5000, 20000, 100000]:
            if rem() < 15:
                break
            h_new, c5_new = optimize_continuous(h_fin, hn, tau, 1000, ftol=1e-16)
            if c5_new is not None and c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
            h_fin = h_new if c5_new is not None else h_fin

    # Final constraint fix
    hn_final = len(best_h)
    best_h = fix_sum(best_h, hn_final / 2.0)
    best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = evaluate_c5(best_h)

    return best_h, best_c5, hn_final
