    #   (1-h[i-k])*I(0<=i-k<n) - h[i+k]*I(0<=i+k<n)
    # where k = m - (n-1)
    
    lags = np.arange(-(n-1), n)  # range of k values
    # For coordinate i:
    # a_k = (1-h[i-k])*I(0<=i-k<n) - h[i+k]*I(0<=i+k<n)
    
    # i-k is in [0,n) when k is in [i-n+1, i]
    # i+k is in [0,n) when k is in [-i, n-1-i]
    
    a = np.zeros(2*n-1)
    # For each k in lags:
    #   if 0 <= i-k < n: a[k] += (1-h[i-k])
    #   if 0 <= i+k < n: a[k] -= h[i+k]
    
    # Vectorized:
    # k ranges from -(n-1) to n-1, stored as index 0 to 2n-2
    # i-k in [0,n-1] means k in [i-(n-1), i]
    k_vals = lags  # shape (2n-1,)
    
    # Condition 1: 0 <= i-k < n
    cond1 = (i - k_vals >= 0) & (i - k_vals < n)
    a[cond1] += (1.0 - h[i - k_vals[cond1]])
    
    # Condition 2: 0 <= i+k < n
    cond2 = (i + k_vals >= 0) & (i + k_vals < n)
    a[cond2] -= h[i + k_vals[cond2]]
    
    # b_k = corr[k] - h[i] * a[k]
    b = corr - h[i] * a
    
    # Now solve: min_{t in [0,1]} max_k (a[k]*t + b[k])
    # Use ternary search
    lo, hi = 0.0, 1.0
    for _ in range(60):  # sufficient precision
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(a * m1 + b)
        f2 = np.max(a * m2 + b)
        if f1 > f2:
            lo = m1
        else:
            hi = m2
    
    return (lo + hi) / 2

from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = np.sum(result) - target_sum
    if abs(diff) > 1e-12:
        free = (result > 1e-8) & (result < 1.0 - 1e-8)
        if np.any(free):
            result[free] -= diff / np.sum(free)
            result = np.clip(result, 0.0, 1.0)
    return result

def get_a_vector(h, i, n):
    k_vals = np.arange(-(n - 1), n)
    a = np.zeros(2 * n - 1)
    cond1 = (i - k_vals >= 0) & (i - k_vals < n)
    a[cond1] += (1.0 - h[i - k_vals[cond1]])
    cond2 = (i + k_vals >= 0) & (i + k_vals < n)
    a[cond2] -= h[i + k_vals[cond2]]
    return a

def solve_1d(h, i, corr, n):

    a = get_a_vector(h, i, n)

        if m == idx0:
            continue
        C_k = l_vals[m]
        a_k = a[m]
        B = -(1.0 - a_k)
        C = -(C_0 - C_k)
        disc = B * B - 4.0 * C
        if disc >= 0.0:
            sq = np.sqrt(disc)
            t1 = (-B + sq) / 2.0

    best_t = 0.0

        f_val = max(q_val, np.max(l_at_t))
        if f_val < best_val:
            best_val = f_val
            best_t = t
    return best_t

def coord_descent(h, n, max_sweeps=500, rng=None):
    target_sum = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    for sweep in range(max_sweeps):
        total_change = 0.0
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        for i in order:
            a = get_a_vector(h, i, n)
            t_new = solve_1d(h, i, corr, n)
            delta = t_new - h[i]
            if abs(delta) > 1e-14:
                h[i] = t_new
                corr = corr + delta * a
                total_change += abs(delta)
        h = np.clip(h, 0.0, 1.0)
        h = project_feasible(h, target_sum)
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if total_change < 1e-12:
            break
    return best_h, best_c5

def softmax_objective(x, dx, T, target_sum):
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    cmax = np.max(corr)
    val = cmax + np.log(np.sum(np.exp(T * (corr - cmax)))) / T
    penalty = 100.0 * (np.sum(x) - target_sum) ** 2
    return val + penalty

def slsqp_polish(h_start, n, T=100.0, max_iter=200):
    dx = 2.0 / n
    target_sum = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraint_fn = lambda x: np.sum(x) - target_sum
    res = minimize(
        lambda x: softmax_objective(x, dx, T, target_sum),
        h_start, method='SLSQP', bounds=bounds,
        options={'maxiter': max_iter, 'ftol': 1e-14}
    )
    return project_feasible(res.x, target_sum)
