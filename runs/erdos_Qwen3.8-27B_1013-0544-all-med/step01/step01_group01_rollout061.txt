- If h[i] in (1-h[n]): n = i, valid if alag ≤ i ≤ n-1.
  Contribution: -h[n-alag] = -h[i-alag]

So for k < 0 (alag = -k):
∂c_k/∂h[i] = (1-h[i+alag]) * [0 ≤ i ≤ n-1-alag] - h[i-alag] * [alag ≤ i ≤ n-1]

Great, this matches what I had before. Let me implement this properly.

Actually, for efficiency, I can vectorize this. For each lag k, the gradient contribution is:
- For k ≥ 0: grad[0:n-k] += -h[k:n] (from the -h[i+k] term) and grad[k:n] += (1-h[0:n-k]) (from the (1-h[i-k]) term)
- For k < 0 (alag = -k): grad[0:n-alag] += (1-h[alag:n]) (from the (1-h[i+alag]) term) and grad[alag:n] += -h[0:n-alag] (from the -h[i-alag] term)

Let me verify:
For k ≥ 0:
- First term: -h[i+k] for i = 0 to n-1-k. So grad[0:n-k] += -h[k:n]. ✓
- Second term: (1-h[i-k]) for i = k to n-1. So grad[k:n] += (1-h[0:n-k]). ✓

For k < 0, alag = -k:
- First term: (1-h[i+alag]) for i = 0 to n-1-alag. So grad[0:n-alag] += (1-h[alag:n]). ✓
- Second term: -h[i-alag] for i = alag to n-1. So grad[alag:n] += -h[0:n-alag]. ✓

This is much more efficient! The total gradient computation is O(n²) (n laggs, O(n) work each), which is very fast.

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max_and_grad(h, beta, n_points):
    """Compute smooth max value and gradient."""
    dx = 2.0 / n_points
    n = n_points
    corr = get_corr(h)
    n_lags = len(corr)
    center = n - 1
    
    # Smooth max value
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted[shifted < -50] = -50
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    val = (m + np.log(sum_exp) / beta) * dx
    weights = exp_vals / sum_exp
    
    # Gradient
    grad = np.zeros(n)
    for k_idx in range(n_lags):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        lag = k_idx - center
        if lag >= 0:
            if lag < n:
                n_valid = n - lag
                grad[:n_valid] += w * (-h[lag:])
                grad[lag:] += w * (1.0 - h[:n_valid])
        else:
            alag = -lag
            if alag < n:
                n_valid = n - alag
                grad[:n_valid] += w * (1.0 - h[alag:])
                grad[alag:] += w * (-h[:n_valid])
    
    return val, grad * dx

def project_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        h = h + diff / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def custom_lbfgs(h0, beta, n_points, target_sum, max_iter=2000, ftol=1e-12):
    """Custom L-BFGS with projection."""
    h = h0.copy()
    n = n_points
    
    # Simple gradient descent with momentum
    lr = 0.01
    momentum = 0.9
    vel = np.zeros(n)
    best_val = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        val, grad = smooth_max_and_grad(h, beta, n_points)
        
        # Project gradient (accounting for sum constraint)
        # Remove the component in the direction of the constraint
        grad_proj = grad - np.mean(grad) * ... 
        # Actually, with the sum constraint sum(h) = target, the feasible directions
        # are those with sum(d) = 0. So we project grad onto this space:
        grad_proj = grad - np.sum(grad) / n  # This removes the mean, making sum = 0
        
        vel = momentum * vel - lr * grad_proj
        h_new = h + vel
        h_new = project_sum(h_new, target_sum)
        
        c5 = compute_c5(h_new)
        if c5 < best_val:
            best_val = c5
            best_h = h_new.copy()
        h = h_new
        lr *= 0.999
    
    return best_h, best_val

    if time_left() > 15:
        for beta in [500, 1000, 2000]:
            if time_left() < 10:
                break
            h_new, c5 = slsqp_refine(best_h, n_points, target_sum, beta, 3000, 1e-15)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Ensure final solution is feasible
    best_h = project_sum(best_h, target_sum)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, n_points
