import time

def compute_corr(h):
    return np.correlate(h, 1-h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0/n))

def project_h(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

def grad_fk(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0*h
    elif 0 < k < n:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    elif -n < k < 0:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(corr - m))) / alpha) * (2.0/n)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s

    for idx in active:
        k = idx - (n-1)
        grad += w[idx] * grad_fk(h, k)
    grad *= (2.0/n)
    return val, grad

    stall = 0

        y = np.clip(y, 0, 1)  # Soft clip for lookahead

But I'm also projecting h_new after the update, which means the Nesterov "lookahead" doesn't quite work the same way as in unconstrained optimization.

The gradient of the smoothed objective is also O(1) (after the 2/n scaling). So a learning rate of 0.01-0.1 should be reasonable.

One more thought: the problem has a lot of symmetry. If h is optimal, then 1-h might also be close to optimal (since the problem is somewhat symmetric). Let me check:

If h' = 1-h, then:
∫ h'(x)(1-h'(x+k)) dx = ∫ (1-h(x)) h(x+k) dx = ∫ h(x+k) dx - ∫ h(x)h(x+k) dx

And the original: ∫ h(x)(1-h(x+k)) dx = ∫ h(x) dx - ∫ h(x)h(x+k) dx = 1 - ∫ h(x)h(x+k) dx

So h'(x)(1-h'(x+k)) = ∫ h(x+k) dx - ∫ h(x)h(x+k) dx

For a piecewise constant function on [0,2], ∫ h(x+k) dx depends on k (it's not always 1). So h' = 1-h is NOT a symmetry of the problem.

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def project_h(h, target_sum):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def grad_fk(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < k < n:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    elif -n < k < 0:
        p = -k
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def smooth_obj_grad(h, alpha):
    n = len(h)
    corr = compute_corr(h)
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(corr - m))) / alpha) * (2.0 / n)
    w = np.exp(alpha * (corr - m))
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= s
    grad = np.zeros(n)
    active = np.where(w > 1e-10)[0]
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * grad_fk(h, k)
    grad *= (2.0 / n)
    return val, grad

def optimize_momentum(h0, target_sum, deadline, rng, alpha_init=40.0, alpha_max=250.0):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    mu = 0.85
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.008
    alpha = alpha_init
    stall = 0
    big_stall = 0

    while time.time() < deadline:
        obj, grad = smooth_obj_grad(h, alpha)
        grad_tan = grad - np.mean(grad)
        v = mu * v - lr * grad_tan
        h_new = project_h(h + v, target_sum)
        c5_new = compute_c5(h_new)

        if c5_new < best_c5 - 1e-13:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
            big_stall = 0
            lr = min(lr * 1.04, 0.08)
            alpha = min(alpha * 1.008, alpha_max)
        else:
            stall += 1
            lr *= 0.9
            if stall > 60:
                v = np.zeros(n)
                noise_scale = 0.01 + 0.005 * big_stall
                h_try = h + rng.randn(n) * noise_scale
                h_try = project_h(h_try, target_sum)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h.copy()
                lr = 0.005
                alpha = max(alpha_init, alpha * 0.6)
                stall = 0
                big_stall += 1
                if big_stall > 10:
                    break

    return best_h, best_c5

def interpolate_h(h_old, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_ext = np.concatenate([h_old, h_old[:1]])
    x_ext = np.concatenate([x_old, [2.0 + 2.0 / n_old]])
    return np.interp(x_new, x_ext, h_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        h0 = 0.5 + 0.3 * rng.randn(100)

    n_parent = len(h0)
    h0 = project_h(h0, n_parent / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Optimize at parent resolution
    dl1 = t0 + budget_s * 0.30
    best_h, best_c5 = optimize_momentum(best_h, len(best_h) / 2.0, dl1, rng,
                                         alpha_init=35.0, alpha_max=200.0)

    # Phase 2: Multi-resolution exploration with diverse starts
    elapsed = time.time() - t0
    remaining = budget_s - elapsed - 60
    n_best = len(best_h)
    candidates = []
    for n_try in [n_best, n_best * 2, n_best // 2, 60, 150]:
        if n_try < 30 or n_try > 800:
            continue
        if n_try not in candidates:
            candidates.append(n_try)

    per_trial = remaining / max(len(candidates), 1)
    for n_try in candidates:
        if time.time() > t0 + budget_s * 0.82:
            break
        budget_per = min(per_trial * 0.9, 120.0)
        h_interp = interpolate_h(best_h, n_best, n_try)
        h_interp = project_h(h_interp, n_try / 2.0)

        for trial in range(3):
            if time.time() > t0 + budget_s * 0.82:
                break
            scale = 0.008 * (trial + 1)
            h_try = h_interp + rng.randn(n_try) * scale
            h_try = project_h(h_try, n_try / 2.0)
            phase_dl = time.time() + budget_per / 3.0
            a_init = 30.0 + 15.0 * trial
            h_opt, c5_opt = optimize_momentum(h_try, n_try / 2.0, phase_dl, rng,
                                              alpha_init=a_init, alpha_max=200.0)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                n_best = n_try

    # Phase 3: Final polish with high alpha
    dl_final = t0 + budget_s - 5
    if time.time() < dl_final:
        best_h, best_c5 = optimize_momentum(best_h, len(best_h) / 2.0, dl_final,
                                            rng, alpha_init=100.0, alpha_max=400.0)

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
