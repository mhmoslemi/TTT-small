    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def smoothed_grad(h, beta, dx):
    """Compute gradient of (1/beta)*log(sum exp(beta*corr_k)) * dx"""
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Numerical stability
    c_max = np.max(corr)
    exp_vals = np.exp(beta * (corr - c_max))
    probs = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(N)
    for i in range(2 * N - 1):
        k = i - (N - 1)
        p = probs[i]
        if k >= 0:
            # First term: m in [k, N-1], contribution (1-h[m-k])
            m_lo, m_hi = k, N - 1
            grad[m_lo:m_hi+1] += p * (1.0 - h[m_lo - k:m_hi - k + 1])
            # Second term: m in [0, N-1-k], contribution -h[m+k]
            m_lo2, m_hi2 = 0, N - 1 - k
            if m_lo2 <= m_hi2:
                grad[m_lo2:m_hi2+1] -= p * h[m_lo2 + k:m_hi2 + k + 1]
        else:
            # k < 0
            # First term: m in [0, N-1+k], contribution (1-h[m-k]) = (1-h[m+|k|])
            m_lo, m_hi = 0, N - 1 + k
            if m_lo <= m_hi:
                grad[m_lo:m_hi+1] += p * (1.0 - h[m_lo - k:m_hi - k + 1])
            # Second term: m in [-k, N-1], contribution -h[m+k] = -h[m-|k|]
            m_lo2, m_hi2 = -k, N - 1
            grad[m_lo2:m_hi2+1] -= p * h[m_lo2 + k:m_hi2 + k + 1]
    
    return grad

def smoothed_obj(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(corr)
    return (np.log(np.sum(np.exp(beta * (corr - c_max)))) + beta * c_max) / beta

        if sum_clipped(mid) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)
