def project_feasible(x, target_sum):
    """Project x onto {y: 0<=y<=1, sum(y)=target_sum} using bisection on threshold."""
    lo = float(x.min()) - 1.5
    hi = float(x.max()) + 1.5
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.clip(x - mid, 0.0, 1.0).sum())

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_grad_lag(h, L):
    n = len(h)
    omh = 1.0 - h
    grad = np.zeros(n)
    if L == 0:
        grad[:] = omh[:] - h[:]
    elif L > 0:
        grad[L:] += omh[:n - L]
        grad[:n - L] -= h[L:]
    else:
        L2 = -L
        grad[:n - L2] += omh[L2:]
        grad[L2:] -= h[:n - L2]
    return grad

def active_set_gradient_step(h, dx, active_lags, step_size, momentum, mu, target_sum):

    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_parent = None

    best_h = None
    best_c5 = float('inf')
    best_n = 70

    def tleft():
        return start_time + budget_s - 5.0 - time.time()

    # Phase 1: Optimize parent at native resolution
    if h_parent is not None and tleft() > 60:
        n = len(h_parent)
        dx = 2.0 / n
        ts = n / 2.0
        h_parent = project_feasible(h_parent, ts)
        h_opt, c5_opt = local_opt(h_parent, dx, ts, min(45, tleft() * 0.2), ntop=14)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n

    # Phase 2: Upsample to finer grids and optimize
    for new_n in [100, 140, 180, 220]:
        if tleft() < 45:
            break
        dx = 2.0 / new_n
        ts = new_n / 2.0
        if h_parent is not None:
            h0 = upsample(h_parent, new_n)
        else:
            h0 = np.full(new_n, 0.5)
        h0 = project_feasible(h0, ts)
        alloc = min(55, tleft() * 0.3)
        h_opt, c5_opt = local_opt(h0, dx, ts, alloc, step0=0.004, ntop=14)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = new_n

    # Phase 3: Random starts at several resolutions
    for new_n in [best_n, max(50, best_n - 30), min(250, best_n + 50)]:
        if tleft() < 25:
            break
        dx = 2.0 / new_n
        ts = new_n / 2.0
        for s in range(4):
            if tleft() < 12:
                break
            h0 = rng.random(new_n)
            h0 = project_feasible(h0, ts)
            alloc = min(25, tleft() * 0.25)
            h_opt, c5_opt = local_opt(h0, dx, ts, alloc, step0=0.006, ntop=12)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = new_n

    # Phase 4: Intensive iterative local search on best
    if best_h is not None and tleft() > 40:
        n = best_n
        dx = 2.0 / n
        ts = n / 2.0
        h_cur = best_h.copy()
        c5_cur = best_c5
        p4_start = time.time()
        max_p4 = min(300, tleft() * 0.5)
        perturb_sigmas = [0.06, 0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001]
        pi = 0
        while time.time() - p4_start < max_p4:
            # Short local optimization from current
            remaining = max_p4 - (time.time() - p4_start)
            if remaining < 5:
                break
            t_alloc = min(30, remaining * 0.5)
            h_new, c5_new = local_opt(h_cur, dx, ts, t_alloc, step0=0.003, mu0=0.5, ntop=10)
            if c5_new < c5_cur:
                c5_cur = c5_new
                h_cur = h_new
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h_new.copy()
            # Perturb
            sigma = perturb_sigmas[min(pi, len(perturb_sigmas) - 1)]
            h_pert = h_cur + rng.normal(0, sigma, n)
            h_pert = project_feasible(h_pert, ts)
            c5_pert = compute_c5(h_pert, dx)
            if c5_pert < c5_cur or pi < 2:
                h_cur = h_pert
                c5_cur = c5_pert
            else:
                pi += 1
                if pi >= len(perturb_sigmas):
                    break

    # Phase 5: SLSQP polish
    if best_h is not None and tleft() > 20:
        n = best_n
        dx = 2.0 / n
        ts = n / 2.0
        gdx = dx
        gts = ts
        def _obj(x):
            return compute_c5(x, gdx)
        def _eqc(x):
            return np.sum(x) - gts
        try:
            res = minimize(_obj, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': _eqc}],
                          options={'maxiter': 3000, 'ftol': 1e-14})
            if res.x is not None:
                h_ref = project_feasible(res.x, gts)
                c5_ref = compute_c5(h_ref, gdx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = h_ref
        except Exception:
            pass

    # Phase 6: Final ultra-fine gradient
    if best_h is not None and tleft() > 5:
        n = best_n
        dx = 2.0 / n
        ts = n / 2.0
        h = best_h.copy()
        momentum = np.zeros(n)
        step = 0.0002
        mu = 0.3
        while tleft() > 1:
            grad, _ = grad_active(h, n_top=8)
            momentum = mu * momentum - step * grad
            h = project_feasible(h + momentum, ts)
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            else:
                step *= 0.93
                if step < 1e-12:
                    break

    # Final validation
    n = best_n
    dx = 2.0 / n
    ts = n / 2.0
    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h, dx)
    return (best_h, best_c5, n)
