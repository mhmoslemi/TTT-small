    return float(np.max(corr)) * dx

def compute_grad_at_lag(h, L):
    n = len(h)
    g = np.zeros(n)

        g[:] = 1.0 - 2.0 * h
    elif L > 0:
        g[:n-L] += (1.0 - h[L:])
        g[L:] -= h[:n-L]
    else:
        s = -L
        g[:n-s] += (1.0 - h[s:])

def lp_threshold(grad, target_sum):
    n = len(grad)
    order = np.argsort(grad)
    h_sol = np.zeros(n)
    remaining = target_sum

        if remaining >= 1.0:
            h_sol[idx] = 1.0
            remaining -= 1.0
