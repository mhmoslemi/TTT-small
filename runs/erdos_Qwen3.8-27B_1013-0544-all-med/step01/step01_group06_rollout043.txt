        g += grad_fk(h, k)
    g /= n_active

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_fk(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def subgradient_descent(h, target_sum, deadline, rng, lr0=0.005, momentum=0.9):
    n = len(h)
    h_curr = h.copy()
    velocity = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    k_iter = 0
    
    while time.time() < deadline:
        k_iter += 1
        lr = lr0 / np.sqrt(k_iter + 1)
        
        # Compute subgradient
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        max_val = np.max(corr)
        # Find active lags (within small tolerance of max)
        tol = 0.05
        active = np.where(corr >= max_val - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        # Limit to top 10 for efficiency
        if len(active) > 10:
            top_idx = np.argsort(corr[active])[-10:]
            active = active[top_idx]
        
        g = np.zeros(n)
        for idx in active:
            kk = idx - (n - 1)
            g += grad_fk(h_curr, kk)
        g /= len(active)
        g *= (2.0 / n)
        
        # Momentum update
        velocity = momentum * velocity - lr * g
        h_new = project_onto_simplex_box(h_curr + velocity, target_sum)
        h_curr = h_new
        
        c5 = compute_c5(h_curr)
        if c5 < best_c5:
            best_h = h_curr.copy()
            best_c5 = c5
        
        if k_iter % 500 == 0:
            # Periodic perturbation
            if rng.rand() < 0.3:
                h_curr = h_curr + rng.randn(n) * 0.05
                h_curr = project_onto_simplex_box(h_curr, target_sum)
                velocity = np.zeros(n)
    
    return best_h, best_c5

def binary_sa(h_init, deadline, rng, n=None):
    if n is None:
        n = len(h_init)
    # Start from rounded version or random
    h_int = np.round(h_init).astype(int)
    # Fix to have exactly n//2 ones
    target_ones = n // 2
    ones_count = np.sum(h_int)
    if ones_count > target_ones:
        ones_idx = np.where(h_int == 1)[0]
        rng.shuffle(ones_idx)
        for i in ones_idx[:ones_count - target_ones]:
            h_int[i] = 0
    elif ones_count < target_ones:
        zeros_idx = np.where(h_int == 0)[0]
        rng.shuffle(zeros_idx)
        for i in zeros_idx[:target_ones - ones_count]:
            h_int[i] = 1
    
    h_float = h_int.astype(float)
    best = h_float.copy()
    best_c5 = compute_c5(best)
    
    T = 0.3
    T_min = 0.005
    cool_rate = 0.9997
    
    while time.time() < deadline:
        ones_idx = np.where(h_int == 1)[0]
        zeros_idx = np.where(h_int == 0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h_int[i], h_int[j] = 0, 1

        new_c5 = compute_c5(h_float)

            if delta < 0:
                best = h_float.copy()
                best_c5 = new_c5
        else:
            h_int[i], h_int[j] = 1, 0
        
        T = max(T * cool_rate, T_min)
    
    return best, best_c5

    """Refine a (possibly binary) solution using gradient descent."""
    return subgradient_descent(h_init, target_sum, deadline, rng, lr0=0.003, momentum=0.85)

    x_ext = np.concatenate([x_old, [2.0 + 1.0/n_old]])
    return np.interp(x_new, x_ext, h_ext)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * (2.0 / n)

def project_onto_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def grad_fk(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n-k])
        if k > 0:
            g[:n-k] -= h[k:]
    else:
        p = -k
        g[:n-p] += (1.0 - h[p:])
        g[p:] -= h[:n-p]
    return g

def binary_sa(h_init, deadline, rng, n_pts=None):
    n = n_pts if n_pts else len(h_init)
    h_int = np.round(h_init[:n] if len(h_init) >= n else h_init).astype(int)
    target_ones = n // 2
    cnt = int(np.sum(h_int))
    if cnt > target_ones:
        idx = np.where(h_int == 1)[0]
        rng.shuffle(idx)
        for i in idx[:cnt - target_ones]:
            h_int[i] = 0
    elif cnt < target_ones:
        idx = np.where(h_int == 0)[0]
        rng.shuffle(idx)
        for i in idx[:target_ones - cnt]:
            h_int[i] = 1
    best = h_int.astype(float).copy()
    best_c5 = compute_c5(best)
    T = 0.25
    T_min = 0.003
    while time.time() < deadline:
        ones_idx = np.where(h_int == 1)[0]
        zeros_idx = np.where(h_int == 0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h_int[i], h_int[j] = 0, 1
        new_c5 = compute_c5(h_int.astype(float))
        delta = new_c5 - best_c5
        if delta < 0 or (T > T_min and rng.rand() < np.exp(-delta / T)):
            if delta < 0:
                best = h_int.astype(float).copy()
                best_c5 = new_c5
        else:
            h_int[i], h_int[j] = 1, 0
        T = max(T * 0.9998, T_min)
    return best, best_c5

def nesterov_pgd(h, target_sum, deadline, rng, lr0=0.006, momentum=0.92):
    n = len(h)
    h_curr = h.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    it = 0
    stall = 0
    while time.time() < deadline:
        it += 1
        lr = lr0 / np.sqrt(it + 1)
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 0.03)[0]
        if len(active) > 8:
            top = np.argsort(corr[active])[-8:]
            active = active[top]
        g = np.zeros(n)
        for idx in active:
            g += grad_fk(h_curr, idx - (n - 1))
        g /= len(active)
        g *= (2.0 / n)
        h_new = h_curr + momentum * (h_curr - h_prev) - lr * g
        h_new = project_onto_simplex_box(h_new, target_sum)
        h_prev = h_curr
        h_curr = h_new
        c5 = compute_c5(h_curr)
        if c5 < best_c5 - 1e-15:
            best_h = h_curr.copy()
            best_c5 = c5
            stall = 0
        else:
            stall += 1
            if stall > 200:
                h_curr = h_curr + rng.randn(n) * (0.02 + 0.01 * min(stall // 200, 5))
                h_curr = project_onto_simplex_box(h_curr, target_sum)
                h_prev = h_curr.copy()
                stall = 0
    return best_h, best_c5

def smooth_pgd(h, target_sum, deadline, rng, alpha=200.0, lr0=0.004):
    n = len(h)
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    it = 0
    while time.time() < deadline:
        it += 1
        lr = lr0 / np.sqrt(it + 1)
        corr = np.correlate(h_curr, 1 - h_curr, mode='full')
        m = np.max(corr)
        e = np.exp(corr - m)
        w = e / np.sum(e)
        g = np.zeros(n)
        for idx in np.where(w > 1e-12)[0]:
            g += w[idx] * grad_fk(h_curr, idx - (n - 1))
        g *= (2.0 / n) / alpha
        h_curr = project_onto_simplex_box(h_curr - lr * g, target_sum)
        c5 = compute_c5(h_curr)
        if c5 < best_c5:
            best_h = h_curr.copy()
            best_c5 = c5
    return best_h, best_c5

def interpolate_to_n(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_ext = np.concatenate([h_old, [h_old[0]]])
    x_ext = np.concatenate([x_old, [2.0 + 1.0 / n_old]])
    return np.interp(x_new, x_ext, h_ext)

    return best_h, best_c5, len(best_h)
