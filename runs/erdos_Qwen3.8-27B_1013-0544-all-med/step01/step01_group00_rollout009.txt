from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, target):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target}"""
    # Bisection on tau: find tau s.t. sum(clip(h-tau, 0, 1)) = target
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj_and_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full') * dx
    m = np.max(corr)
    e = np.exp((corr - m) / temp)
    s = np.sum(e)
    f = m + temp * np.log(s)
    w = e / s
    # Gradient via convolution
    grad = np.convolve(w[::-1], g, mode='full')[n-1:2*n-1] * dx
    return f, grad

    target = n * 0.5
    h = project_feasible(h0.copy(), target)
    y = h.copy()
    t_k = 1.0
    best_f = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        f, grad = lse_obj_and_grad(y, temp)
        # Backtracking line search for step size

        # Check descent
        f_new = lse_obj_and_grad(h_new, temp)[0]
        while f_new > f + 0.5 * alpha * np.dot(grad, h_new - y) and alpha > 1e-6:
            alpha /= 2.0

    for temp in [0.05, 0.02, 0.01, 0.005]:
        if time_left() < 120:
            break
        tl = min(60.0, time_left() * 0.15)
        h_new, c5_new = fista_optimize(h, temp, max_iter=8000, time_limit=tl)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 2: L-BFGS-B smooth refinement
    for temp in [0.005, 0.002, 0.001, 0.0005]:
        if time_left() < 100:
            break
        h_new, c5_new = lbfgs_smooth(h_best, temp, max_iter=3000)
        if c5_new < c5:
            h, c5 = h_new, c5_new
            h_best = h.copy()

    # Phase 3: SLSQP t-variable polish
    if time_left() > 250:
        h_new, c5_new = slsqp_tvar(h_best, max_iter=1000)
        if c5_new < c5:
            h_best, c5 = h_new, c5_new

    # Phase 4: Resolution upgrade
    for n_new in [360, 540]:
        if time_left() < 350:
            break
        h_interp = interp_h(h_best, n_new)
        h_cur = h_interp
        for temp in [0.03, 0.01, 0.005, 0.002]:
            if time_left() < 150:
                break
            tl = min(40.0, time_left() * 0.1)
            h_new, c5_new = fista_optimize(h_cur, temp, max_iter=5000, time_limit=tl)
            if c5_new < c5:
                h_cur, c5 = h_new, c5_new
                h_best = h_cur.copy()
        if time_left() > 150:
            h_new, c5_new = lbfgs_smooth(h_cur, 0.002, max_iter=2000)
            if c5_new < c5:
                h_best, c5 = h_new, c5_new
        if time_left() > 200:
            h_new, c5_new = slsqp_tvar(h_best, max_iter=800)
            if c5_new < c5:
                h_best, c5 = h_new, c5_new

    # Phase 5: Random perturbations at best resolution
    rng = np.random.RandomState(seed)
    n_cur = len(h_best)
    for i in range(20):
        if time_left() < 80:
            break
        scale = 0.008 + 0.004 * (i % 5)
        h_p = h_best + rng.randn(n_cur) * scale
        h_p = project_feasible(h_p, n_cur * 0.5)
        h_new, c5_new = lbfgs_smooth(h_p, 0.003, max_iter=1500)
        if c5_new < c5:
            h_best, c5 = h_new, c5_new

    # Final polish
    if time_left() > 60:
        h_new, c5_new = slsqp_tvar(h_best, max_iter=500)
        if c5_new < c5:
            h_best, c5 = h_new, c5_new

    h_best = project_feasible(h_best, len(h_best) * 0.5)
    c5_final = compute_c5(h_best)
    return h_best, c5_final, len(h_best)
