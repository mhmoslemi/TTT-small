def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = np.sum(h) - target
    if abs(diff) > 1e-14:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def local_search(h, n, dx, rng, time_limit, start_time):
    """Pairwise swap local search with adaptive step size."""
    best_c5 = compute_c5(h, dx)
    step = 0.1
    no_improve = 0
    
    while time.time() - start_time < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Feasible range
        dmax = min(1.0 - h[i], h[j])
        dmin = max(-h[i], h[j] - 1.0)
        if dmax <= dmin + 1e-10:
            continue
        
        # Try a few delta values
        improved = False
        for _ in range(3):
            delta = rng.uniform(max(dmin, -step), min(dmax, step))
            if abs(delta) < 1e-10:
                continue
            h[i] += delta
            h[j] -= delta
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                no_improve = 0
                improved = True
                break
            h[i] -= delta
            h[j] += delta
        
        if not improved:
            no_improve += 1
            if no_improve % 200 == 0:
                step *= 0.8
                no_improve = 0
            if step < 1e-6:
                break
    
    return h, best_c5

def annealing_search(h, n, dx, rng, time_limit, start_time):
    """Simulated annealing for the C5 problem."""
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    T = 0.05
    cool = 0.999
    
    while time.time() - start_time < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        dmax = min(1.0 - h[i], h[j])
        dmin = max(-h[i], h[j] - 1.0)
        if dmax <= dmin + 1e-10:
            continue
        delta = rng.uniform(dmin, dmax)
        if abs(delta) < 1e-10:
            continue
        h[i] += delta
        h[j] -= delta
        c5 = compute_c5(h, dx)
        delta_c5 = c5 - current_c5
        if delta_c5 < 0 or rng.rand() < np.exp(-delta_c5 / T):
            current_c5 = c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
        T *= cool
    
    return best_h, best_c5

def subgradient_search(h, n, dx, rng, time_limit, start_time):
    """Subgradient descent with projection."""
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    step = 0.01
    
    for _ in range(1000000):
        if time.time() - start_time >= time_limit:
            break
        # Compute all lag values
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        
        # Find active lags (top 3)
        top_idx = np.argsort(corr)[-3:]
        
        # Compute subgradient (average of active lag gradients)
        grad = np.zeros(n)
        for k in top_idx:
            # k is in [0, 2n-2], corresponding to lag = k - (n-1)
            lag = k - (n - 1)
            # ∂f_k/∂h[j] = (1-h[j+lag]) - h[j-lag] (with boundary handling)
            for j in range(n):
                # (1 - h[j+lag]) if j+lag in [0, n-1], else 0
                jp = j + lag
                jm = j - lag
                val = 0.0
                if 0 <= jp < n:
                    val += (1.0 - h[jp])
                if 0 <= jm < n:
                    val -= h[jm]
                grad[j] += val
        grad /= len(top_idx)
        
        # Projected subgradient step
        h_new = h - step * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new  # still move, but don't update best
        
        # Adapt step size
        step *= 0.999
    
    return best_h, best_c5

    grad[mask_m] -= h[jm[mask_m]]

def compute_subgradient(h, n, lags):
    """Compute average subgradient of max f_lag over given lags."""
    grad = np.zeros(n)
    j = np.arange(n)
    for lag in lags:
        jp = j + lag
        jm = j - lag
        mask_p = (jp >= 0) & (jp < n)
        mask_m = (jm >= 0) & (jm < n)
        grad[mask_p] += (1.0 - h[jp[mask_p]])
        grad[mask_m] -= h[jm[mask_m]]
    grad /= len(lags)
    return grad

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    for _ in range(200):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    h = np.clip(h, 0.0, 1.0)

        if np.any(free):
            h[free] -= diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_search(h, n, dx, rng, max_time):
    """Subgradient descent on max of correlation components."""
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    t = start = time.time()
    j = np.arange(n)
    
    while time.time() - t < max_time:
        # Compute all lags
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Top 3 lags
        top_k = np.argsort(corr)[-3:]
        
        # Subgradient
        grad = np.zeros(n)
        for k in top_k:
            lag = int(k) - (n - 1)
            jp = j + lag
            jm = j - lag
            mp = (jp >= 0) & (jp < n)
            mm = (jm >= 0) & (jm < n)
            grad[mp] += (1.0 - h[jp[mp]])
            grad[mm] -= h[jm[mm]]
        grad /= 3.0
        
        # Step size: c/sqrt(t)
        step = 0.5 / np.sqrt(time.time() - start + 1)
        
        h_new = project_feasible(h - step * grad, n)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def pair_search(h, n, dx, rng, max_time):
    """Random pairwise swap local search."""
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    start = time.time()
    step = 0.05
    
    while time.time() - start < max_time:
        i = rng.randint(n)
        j2 = rng.randint(n)
        if i == j2:
            continue

        dmin = max(-h[i], h[j2] - 1.0)
        if dmax <= dmin:
            continue

            d = rng.uniform(max(dmin, -step), min(dmax, step))
            if abs(d) < 1e-10:

            h[i] += d
            h[j2] -= d
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                break
            h[i] -= d
            h[j2] += d
        else:
            step *= 0.95
            if step < 1e-7:
                break
    
    return best_h, best_c5
