import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.count_nonzero(free) > 0:
            h[free] -= diff / np.count_nonzero(free)
        else:

        h = np.clip(h, 0.0, 1.0)

def all_correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_smoothed_gradient(h, T, n):
    """Compute gradient of T*log(sum(exp(c_m/T))) w.r.t. h"""
    c = all_correlations(h)
    # Softmax weights
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / np.sum(exp_c)
    # w has length 2n-1, corresponding to lags k = -(n-1) to n-1
    grad = np.zeros(n)
    for idx, k in enumerate(range(-(n-1), n)):
        if w[idx] < 1e-10:
            continue
        wk = w[idx]
        if k >= 0:
            grad[:n-k] += wk * (1.0 - h[k:])
            grad[k:] -= wk * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wk * (1.0 - h[:n-kk])
            grad[:n-kk] -= wk * h[kk:]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def all_c_values(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = np.count_nonzero(free)
        if cnt > 0:
            h[free] -= diff / cnt
        else:
            h -= diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothed_gradient(h, T, n):
    c = all_c_values(h)
    c_max = np.max(c)
    log_w = (c - c_max) / T
    log_w -= np.max(log_w)
    w = np.exp(log_w)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    # w has length 2n-1, index m corresponds to lag k = m-(n-1)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-12:
            continue
        k = m - (n - 1)
        if k >= 0:
            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:n-k]
        else:
            kk = -k
            grad[kk:] += wm * (1.0 - h[:n-kk])
            grad[:n-kk] -= wm * h[kk:]
    return grad

def subgradient_optimize(h0, n, dx, T_start, T_end, n_iters, verbose=False):
    h = h0.copy()
    v = np.zeros(n)
    beta = 0.9
    lr = 0.3
    
    for it in range(n_iters):
        progress = (it + 1) / n_iters
        T = T_start * (T_end / T_start) ** progress
        grad = smoothed_gradient(h, T, n)
        v = beta * v + grad
        h = h - lr * v
        h = project_feasible(h, n)
        lr *= 0.9995
    
    return h

def lbfgsb_optimize(h0, n, dx, T=0.01, max_iter=200):
    def obj(x):
        c = all_c_values(x)
        c_max = np.max(c)
        return T * (c_max + np.log(np.sum(np.exp((c - c_max) / T))))
    
    def grad(x):
        return smoothed_gradient(x, T, n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                    'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_feasible(res.x, n)

        max_change = 0

            # g_k = intercept_k + h[j] * slope_k
            # slope_k = (1-h[j+k])_valid - h[j-k]_valid

            slopes = np.zeros(2*n-1)
            for m in range(2*n-1):

```python
def compute_slopes(h, j, n):
    slopes = np.zeros(2*n-1)
    # For k >= 0: alpha = 1-h[j+k] (valid when j+k < n), beta = h[j-k] (valid when j-k >= 0)
    # For k < 0: alpha = 1-h[j+k] (valid when j+k >= 0), beta = h[j-k] (valid when j-k < n)
    
    k_range = np.arange(-(n-1), n)
    # alpha_j = 1 - h[j+k] where valid
    jk = j + k_range  # positions for alpha
    alpha = np.where((jk >= 0) & (jk < n), 1.0 - h[jk.clip(0, n-1)], 0.0)
    jmk = j - k_range  # positions for beta
    beta = np.where((jmk >= 0) & (jmk < n), h[jmk.clip(0, n-1)], 0.0)
    return alpha - beta
```
