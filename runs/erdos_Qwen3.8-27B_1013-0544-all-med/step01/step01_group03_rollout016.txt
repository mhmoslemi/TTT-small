        #   Valid when 0 <= i-m < n (i.e., -m <= i < n-m... just 0 <= i-m < n)
        #   Contribution when h[i]=v: v*(1-h[i-m])
        #   So delta[m] += (1-h[i-m]) if 0 <= i-m < n
        #   But wait, h[i-m] here is h_old[i-m] which is the current h (with h[i]=0)
        #   If i-m != i (i.e., m != 0), this is just h[i-m]
        
        # For the term (1-h[n]) to involve h[i]: n = i
        #   Valid when 0 <= i < n (always) and 0 <= i+m < n
        #   Contribution: h[i+m]*(1-v) vs h[i+m]*(1-0) = h[i+m]
        #   Change: -v*h[i+m]
        #   So delta[m] += -h[i+m] if 0 <= i+m < n
        
        # For m=0: both conditions give n=i, and the change is v - v²
        # (handled separately)
        
        for m in range(2*n-1):
            d = 0.0
            # First term: n = i-m, valid if 0 <= i-m < n
            if 0 <= i - m < n:
                idx1 = i - m
                d += (1.0 - h[idx1])
            # Second term: n = i, valid if 0 <= i+m < n
            if 0 <= i + m < n:
                idx2 = i + m
                d += -h[idx2]
            delta[m] = d
        
        # Now evaluate F(v) for v in grid
        # F(v) = max(max_{m!=0}(corr_zero[m] + v*delta[m]), corr_zero[0] + v - v²)
        # For m=0: corr_zero[0] + v - v²
        # Note: corr_zero[0] = sum_n h[n]*(1-h[n]) with h[i]=0
        #   = (sum of h[n]*(1-h[n]) for n!=i) + 0*(1-0)
        #   So the "true" corr[0] with h[i]=v is corr_zero[0] + v*(1-v)
        
        best_v = grid[np.argmin(
            np.maximum(
                np.max(corr_zero[1:] + grid[:,None] * delta[1:], axis=1),
                corr_zero[0] + grid - grid**2
            ) if n > 1 else np.zeros(n_grid)
        )]
        # Wait, I need to handle the max over all m including m=0 properly
        # Actually let me just compute the full max:
        # For each v in grid:
        #   corr_full = corr_zero + v * delta  (this is linear for all m)
        #   But for m=0, the actual value is corr_zero[0] + v - v², not corr_zero[0] + v*delta[0]
        #   So: corr_actual[0] = corr_zero[0] + v - v²
        #        corr_actual[m] = corr_zero[m] + v * delta[m] for m != 0
        
        # Let me compute delta[0]:
        # delta[0] = (1-h[i-0])*[0<=i-0<n] - h[i+0]*[0<=i+0<n]
        #          = (1-h[i]) - h[i] = 1 - 2*h[i]
        # But h[i]=0 at this point, so delta[0] = 1
        # And corr_actual[0] = corr_zero[0] + v - v²
        # While corr_zero[0] + v*delta[0] = corr_zero[0] + v
        # So the correction is: corr_actual[0] - (corr_zero[0] + v*delta[0]) = v - v² - v = -v²
        
        # So I can compute:
        # corr_linear = corr_zero + v * delta  (for all m)
        # Then fix m=0: corr_linear[0] -= v²
        # max over all m of corrected values
        
        # Vectorized:
        corr_linear_at_grid = corr_zero[None,:] + grid[:,None] * delta[None,:]
        corr_linear_at_grid[:,0] -= grid**2[:,None]
        F = np.max(corr_linear_at_grid, axis=1)
        h[i] = grid[np.argmin(F)]
    
    return h

Let me also think about the sum constraint. After each coordinate update, the sum changes. I should project back to the correct sum periodically (e.g., after each full pass).

The projection: if sum > target, reduce values that are above some threshold; if sum < target, increase values below threshold. A simple approach: add a uniform offset and clip. But this might not be exact. Better: add the correction to the coordinates that have the most "room" (closest to 0 or 1 depending on direction).

Actually, the simplest exact projection: 
- If sum - target > 0: distribute -(sum-target) among coordinates where h < 1 (proportional to room, or uniformly)
- If target - sum > 0: distribute (target-sum) among coordinates where h > 0

Uniform distribution among eligible coordinates is simplest and works well in practice.

                        np.linspace(0, 1, len(best_h)), best_h)
        h_t = fix_sum(h_t, target_t)
        h_t = coord_descent(h_t, max_passes=25, n_grid=250, seed=seed+tn,
                          time_limit=budget_s, t0=t0)
        h_t = fix_sum(h_t, target_t)
        c5_new = evaluate_c5(h_t)
        if c5_new < best_c5:
            best_h = h_t
            best_c5 = c5_new
    
    # Phase 3: Fine coordinate descent at best resolution
    if rem() > 40:
        hn = len(best_h)
        h_new = coord_descent(best_h, max_passes=40, n_grid=500, seed=seed+1,
                             time_limit=budget_s, t0=t0)
        h_new = fix_sum(h_new, hn / 2.0)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 4: SLSQP polish
    if rem() > 20:
        hn = len(best_h)
        h_new = slsqp_polish(best_h, maxiter=150, tau=5000)
        h_new = fix_sum(h_new, hn / 2.0)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 5: Even finer coordinate descent
    if rem() > 30:
        hn = len(best_h)
        h_new = coord_descent(best_h, max_passes=30, n_grid=1000, seed=seed+2,
                             time_limit=budget_s, t0=t0)
        h_new = fix_sum(h_new, hn / 2.0)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 6: SLSQP with lower smoothing
    if rem() > 15:
        hn = len(best_h)
        h_new = slsqp_polish(best_h, maxiter=100, tau=10000)
        h_new = fix_sum(h_new, hn / 2.0)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 7: Random restarts with coordinate descent
    if rem() > 30:
        hn = len(best_h)
        target_r = hn / 2.0
        for trial in range(8):
            if rem() < 15:
                break
            scale = 0.03 + 0.12 * np.random.random()
            hp = best_h + np.random.normal(0, scale, hn)
            hp = np.clip(hp, 0.0, 1.0)
            hp = fix_sum(hp, target_r)
            hp = coord_descent(hp, max_passes=20, n_grid=300, seed=seed+100+trial,
                              time_limit=budget_s, t0=t0)
            hp = fix_sum(hp, target_r)
            c5_new = evaluate_c5(hp)
            if c5_new < best_c5:
                best_h = hp
                best_c5 = c5_new
    
    # Phase 8: Swap-based local search
    if rem() > 25:
        hn = len(best_h)
        h_new = swap_refine(best_h, max_rounds=10, seed=seed+200,
                          time_limit=budget_s, t0=t0)
        h_new = fix_sum(h_new, hn / 2.0)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Phase 9: Final coordinate descent with very fine grid
    if rem() > 15:
        hn = len(best_h)
        h_new = coord_descent(best_h, max_passes=20, n_grid=2000, seed=seed+3,
                             time_limit=budget_s, t0=t0)
        h_new = fix_sum(h_new, hn / 2.0)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Final constraint fix and evaluation
    hn_final = len(best_h)
    best_h = fix_sum(best_h, hn_final / 2.0)
    best_c5 = evaluate_c5(best_h)
    
    return best_h, best_c5, hn_final
