def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)

            h[free] -= diff / np.sum(free)
        else:

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]

        res = minimize(lambda x: compute_c5(x, dx), h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,

            return project_feasible(res.x, n)
    except:
        pass
    return project_feasible(h_start, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float).copy()
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        diff = np.sum(h) - target
        if abs(diff) < 1e-15:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if np.any(free):
            h[free] -= diff / np.sum(free)
        else:
            h -= diff / n
    return np.clip(h, 0.0, 1.0)

def slsqp_polish(h_start, n, dx, target_sum, max_iter=400):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
    try:
        res = minimize(lambda x: compute_c5(x, dx), h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.fun < compute_c5(h_start, dx):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return project_feasible(h_start, n)

def de_search(h_init, n, dx, rng, time_limit, pop_size=60):

    for i in range(1, pop_size):
        scale = 0.08 + 0.12 * rng.rand()
        pop[i] = np.clip(h_init + rng.randn(n) * scale, 0, 1)
        pop[i] = project_feasible(pop[i], n)
        fit[i] = compute_c5(pop[i], dx)
    best_i = int(np.argmin(fit))
    gen = 0
    no_improve = 0

    while time.time() < time_limit:
        gen += 1
        t = min(1.0, gen / 400.0)
        F = 0.75 - 0.45 * t
        Cr = 0.8 - 0.3 * t

        for i in range(pop_size):
            if time.time() >= time_limit:
                return pop[best_i].copy(), fit[best_i]
            if rng.rand() < 0.4:
                a_i = best_i
                others = [j for j in range(pop_size) if j != i and j != a_i]
                b_i, c_i = int(rng.choice(others, 2, replace=False)[0]), int(rng.choice(others, 2, replace=False)[1])
            else:
                others = [j for j in range(pop_size) if j != i]
                picks = rng.choice(others, 3, replace=False)
                a_i, b_i, c_i = int(picks[0]), int(picks[1]), int(picks[2])
            v = pop[a_i] + F * (pop[b_i] - pop[c_i])
            cross = rng.rand(n) < Cr
            r = rng.randint(n)
            cross[r] = True
            u = np.where(cross, v, pop[i])
            u = project_feasible(u, n)
            fu = compute_c5(u, dx)
            if fu <= fit[i]:
                pop[i] = u
                fit[i] = fu
                if fu < fit[best_i]:
                    best_i = i
                    no_improve = 0
            else:
                no_improve += 1

                    pop[j] = project_feasible(pop[j], n)
                    fit[j] = compute_c5(pop[j], dx)
