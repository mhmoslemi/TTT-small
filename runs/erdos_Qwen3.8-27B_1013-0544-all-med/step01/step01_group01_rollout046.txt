            return smooth_max(corr_full(h), beta) * dx + penalty * (h.sum() - target_sum) ** 2

        def obj_pb_grad(h):
            return obj_smmax_grad(h, beta) * dx + 2.0 * penalty * (h.sum() - target_sum) * np.ones(n_points)

        try:
            res = minimize(obj_pb, best_h, method='L-BFGS-B', bounds=bounds,
                           jac=obj_pb_grad,
                           options={'maxiter': 800, 'ftol': 1e-16, 'gtol': 1e-12})
            h_new = project_sum_bound(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass
        penalty *= 3.0

    # Phase 2: Projected gradient with backtracking
    if time_left() > 30:
        alpha = 0.005
        beta_pg = 300
        stall = 0
        for it in range(500):
            if time_left() < 10:
                break
            grad = obj_smmax_grad(best_h, beta_pg) * dx
            h_new = best_h - alpha * grad
            h_new = project_sum_bound(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                best_h, best_c5 = h_new, c5_new
                stall = 0
                alpha = min(alpha * 1.1, 0.03)
            else:
                stall += 1
                alpha = max(alpha * 0.7, 0.0005)
                if stall > 50:
                    break

    # Phase 3: Diverse restarts with L-BFGS-B
    attempt = 0
    no_improve = 0
    while time_left() > 40 and no_improve < 30:
        attempt += 1
        h0 = best_h.copy()
        strategy = attempt % 6

        if strategy == 0:
            n_p = max(3, n_points // 6)
            idx = np.random.choice(n_points, n_p, replace=False)
            h0[idx] += np.random.randn(n_p) * 0.03
        elif strategy == 1:
            blen = np.random.randint(3, max(4, n_points // 8))
            bs = np.random.randint(0, max(1, n_points - blen))
            h0[bs:bs + blen] = np.clip(h0[bs:bs + blen] + np.random.randn(blen) * 0.1, 0, 1)
        elif strategy == 2:
            mask = np.random.random(n_points) < 0.1
            h0[mask] = np.clip(h0[mask] + np.random.randn(int(mask.sum())) * 0.15, 0, 1)
        elif strategy == 3:
            c = corr_full(h0)
            active = np.argmax(c)
            lag = active - (n_points - 1)
            for j in range(n_points):
                if 0 <= j + lag < n_points:
                    h0[j] -= 0.01 * (1.0 - h0[j + lag])
                if 0 <= j - lag < n_points:
                    h0[j] += 0.01 * h0[j - lag]
        elif strategy == 4:
            i, j = np.random.choice(n_points, 2, replace=False)
            h0[i], h0[j] = h0[j], h0[i]
        else:
            h0 += np.random.randn(n_points) * 0.01

        h0 = project_sum_bound(h0, target_sum)

        beta_r = [100, 300]
        for b in beta_r:
            if time_left() < 15:
                break

            def obj_r(h, bb=b):
                return smooth_max(corr_full(h), bb) * dx + 5.0 * (h.sum() - target_sum) ** 2

            def obj_r_grad(h, bb=b):
                return obj_smmax_grad(h, bb) * dx + 10.0 * (h.sum() - target_sum) * np.ones(n_points)

            try:
                res = minimize(obj_r, h0, method='L-BFGS-B', bounds=bounds,
                               jac=obj_r_grad,
                               options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-11})
                h_new = project_sum_bound(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_h, best_c5 = h_new.copy(), c5
                    no_improve = 0
                    break
            except Exception:
                pass
        else:
            no_improve += 1

    # Phase 4: Final SLSQP polish with analytical gradient
    if time_left() > 15:
        beta_f = 800

        def obj_f(h):
            return smooth_max(corr_full(h), beta_f) * dx

        def obj_f_grad(h):
            return obj_smmax_grad(h, beta_f) * dx

        def con_fun(x):
            return np.sum(x) - target_sum

        def con_jac(x):
            return np.ones(n_points)

        constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

        try:
            res = minimize(obj_f, best_h, method='SLSQP', bounds=bounds,
                           jac=obj_f_grad, constraints=constraints,
                           options={'maxiter': 2000, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_sum_bound(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    # Phase 5: Second SLSQP pass with even higher beta
    if time_left() > 10:
        beta_f2 = 1500

        def obj_f2(h):
            return smooth_max(corr_full(h), beta_f2) * dx

        def obj_f2_grad(h):
            return obj_smmax_grad(h, beta_f2) * dx

        try:
            res = minimize(obj_f2, best_h, method='SLSQP', bounds=bounds,
                           jac=obj_f2_grad, constraints=constraints,
                           options={'maxiter': 1500, 'ftol': 1e-15, 'iprint': 0})
            h_new = project_sum_bound(res.x, target_sum)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5
        except Exception:
            pass

    return best_h, best_c5, n_points
