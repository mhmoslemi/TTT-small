   ```python
       h_clipped = np.clip(h_np, 0, 1)
       diff = avg - np.sum(h_clipped)
       if abs(diff) < 1e-8:
           return h_clipped
       mask = (h_clipped > 1e-10) & (h_clipped < 1.0 - 1e-10)
       if len(mask) == 0:
           h_clipped[:] += diff / len(h)
           return h_clipped
       indices = np.arange(len(mask))[mask]
       adjustment = diff / len(indices)
       h_clipped[indices] += adjustment
       return h_clipped.clip(0, 1)
   ```
