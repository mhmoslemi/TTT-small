                # Evaluate candidate
                try:
                    cand_c5 = compute_c5(candidate)
                except:

                # Simulated annealing: accept if cand_c5 <= current_c5 or with probability exp((current_c5 - cand_c5)/T)

                        current_c5 = cand_c5
