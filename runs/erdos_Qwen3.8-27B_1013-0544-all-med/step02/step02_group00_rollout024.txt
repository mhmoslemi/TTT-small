            step_sz *= 0.75
    return h, c5

def pair_search(h, n, steps, tcheck, tlimit, seed_val):
    target = n / 2.0
    c5, _ = compute_c5(h)
    rng = np.random.RandomState(seed_val)
    for it in range(steps):
        if tcheck() > tlimit:
            break
        corr = np.correlate(h, 1.0 - h, mode="full")
        worst_idx = int(np.argmax(corr))
        worst_lag = worst_idx - (n - 1)
        if worst_lag == 0:
            # Lag 0: sum h[i](1-h[i]), minimize by making h more binary
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if h[i] < h[j]:
                h_try = h.copy()
                d = min(h[j] - h[i], 0.03)
                h_try[i] -= d
                h_try[j] += d
                h_try = project_h(h_try, target)
                c5_try, _ = compute_c5(h_try)
                if c5_try < c5 - 1e-15:
                    h = h_try
                    c5 = c5_try
            continue
        # For nonzero lag, find pairs at that lag
        if worst_lag > 0:
            lo = 0
            hi = n - 1 - worst_lag
        else:
            lo = -worst_lag
            hi = n - 1
        count = hi - lo + 1
        if count <= 0:
            continue
        # Pick a random index in the valid range
        i = lo + rng.randint(count)
        if worst_lag > 0:
            j = i + worst_lag
        else:
            j = i + worst_lag
        # Try to reduce corr at this lag: decrease h[i] or increase h[j]
        # corr = h[i](1-h[j]), to decrease: lower h[i] or raise h[j]
        for direction in [1, -1]:
            d = 0.02 * direction
            h_try = h.copy()
            h_try[i] += d
            h_try[j] -= d
            h_try = np.clip(h_try, 0, 1)
            h_try = project_h(h_try, target)
            c5_try, _ = compute_c5(h_try)
            if c5_try < c5 - 1e-15:
                h = h_try
                c5 = c5_try
                break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def tleft():
        return budget_s - elapsed()

    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_parent = len(h0)
    else:
        n_parent = 140
        h0 = np.full(n_parent, 0.5)

    n = n_parent
    target = n / 2.0
    best_h = project_h(h0.copy(), target)
    best_c5, _ = compute_c5(best_h)
    best_n = n

    # Phase 1: SLSQP with increasing beta at parent resolution
    h = best_h.copy()
    for beta in [15, 40, 100, 300, 800, 2000, 5000]:
        if tleft() < 25:
            break
        h = do_slsqp(h, n, beta, 1000, 1e-14)
        c5, _ = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 2: Subgradient with momentum
    if tleft() > 50:
        h, c5 = subgrad_search(best_h.copy(), n, 4000, elapsed, budget_s - 15)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 3: Pair search targeting worst lags
    if tleft() > 30:
        h, c5 = pair_search(best_h.copy(), n, 6000, elapsed, budget_s - 15, seed + 1)
        if c5 < best_c5:
            best_h, best_c5, best_n = h.copy(), c5, n

    # Phase 4: SLSQP polish after pair search
    if tleft() > 20:
        h = do_slsqp(best_h.copy(), n, 3000, 2000, 1e-15)
        c5, _ = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Phase 5: Try a slightly different resolution
    if tleft() > 50:
        n2 = n + 20 if n % 2 == 0 else n + 21
        if n2 <= 300:
            target2 = n2 / 2.0
            x1 = np.linspace(0, 2, n)
            x2 = np.linspace(0, 2, n2)
            h2 = project_h(np.interp(x2, x1, best_h), target2)
            for beta in [200, 1000, 4000]:
                if tleft() < 20:
                    break
                h2 = do_slsqp(h2, n2, beta, 600, 1e-14)
                c5_2, _ = compute_c5(h2)
                if c5_2 < best_c5:
                    best_h, best_c5, best_n = h2.copy(), c5_2, n2
            # Subgradient at new resolution
            if tleft() > 30:
                h2, c5_2 = subgrad_search(best_h.copy(), best_n, 2000, elapsed, budget_s - 10)
                if c5_2 < best_c5:
                    best_h, best_c5 = h2.copy(), c5_2

    # Phase 6: Another round of pair search + SLSQP at best resolution
    n = best_n
    target = n / 2.0
    if tleft() > 25:
        h, c5 = pair_search(best_h.copy(), n, 4000, elapsed, budget_s - 12, seed + 42)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5
        h = do_slsqp(best_h.copy(), n, 8000, 3000, 1e-15)
        c5, _ = compute_c5(h)
        if c5 < best_c5:
            best_h, best_c5 = h.copy(), c5

    # Final check
    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5, _ = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
