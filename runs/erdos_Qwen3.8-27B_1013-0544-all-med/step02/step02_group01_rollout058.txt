import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        n_free = free.sum()
        if n_free > 0 and abs(diff) < 0.9 * n_free:
            h[free] += diff / n_free
            h = np.clip(h, 0.0, 1.0)
            diff2 = target_sum - h.sum()
            if abs(diff2) > 1e-10:
                free2 = (h > 1e-8) & (h < 1.0 - 1e-8)
                if free2.sum() > 0:
                    h[free2] += diff2 / free2.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            diff2 = target_sum - h.sum()
            if abs(diff2) > 1e-10:
                free = (h > 1e-8) & (h < 1.0 - 1e-8)
                if free.sum() > 0:
                    h[free] += diff2 / free.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def nesterov_opt(h, target_sum, max_iters, lr0, decay, top_k, momentum_init=0.9):
    n = len(h)
    dx = 2.0 / n
    x = h.copy().astype(float)
    v = np.zeros(n)
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        # Momentum
        v = momentum_init * v - lr * _subgrad(x, n, dx, top_k, j)
        x_new = x + v
        x_new = project_feasible(x_new, target_sum)
        val = compute_c5(x_new)
        if val < best_v:
            best_v = val
            best_x = x_new.copy()
        x = x_new
    
    return best_x, best_v

def _subgrad(h, n, dx, top_k, j):
    corr = compute_corr_full(h)
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for s_idx in top_idx:
        s = s_idx - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += (-h[jp[mp]])
    return (grad / top_k) * dx

def slsqp_refine(h, target_sum, maxiter, beta):
    n = len(h)
    
    def objg(x):
        val, grad = _smoothmax(x, n, beta)
        return val, grad
    
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(objg, h, jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, target_sum)
    except:
        return project_feasible(h, target_sum)

def _smoothmax(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    shifted = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(shifted)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    
    j = np.arange(n)
    grad = np.zeros(n)
    active = np.where(w > 1e-6)[0]
    for k in active:
        wk = w[k]
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms_idx = (js >= 0) & (js < n)
        mp_idx = (jp >= 0) & (jp < n)

        if mp_idx.any():
            grad[mp_idx] += wk * (-h[jp[mp_idx]])
    grad *= dx
    return val, grad

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = f(x_new)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

    val = (m + np.log(sw) / beta) * dx

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr_full(h)) * dx)

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        nf = free.sum()
        if nf > 0 and abs(diff) < 0.9 * nf:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
            d2 = target_sum - h.sum()
            if abs(d2) > 1e-10:
                f2 = (h > 1e-8) & (h < 1.0 - 1e-8)
                if f2.sum() > 0:
                    h[f2] += d2 / f2.sum()
            h = np.clip(h, 0.0, 1.0)
        else:
            h += diff / len(h)
            h = np.clip(h, 0.0, 1.0)
            d2 = target_sum - h.sum()
            if abs(d2) > 1e-10:
                f2 = (h > 1e-8) & (h < 1.0 - 1e-8)
                if f2.sum() > 0:
                    h[f2] += d2 / f2.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def nesterov_run(h, target, max_iters, lr0, decay, top_k, mu0=0.9):
    n = len(h)
    dx = 2.0 / n
    x = h.copy().astype(float)
    v = np.zeros(n)
    best_x = x.copy()
    best_val = compute_c5(x)
    j = np.arange(n)
    for t in range(max_iters):
        lr = lr0 * (decay ** t)
        corr = compute_corr_full(x)
        top_idx = np.argsort(corr)[-top_k:]
        g = np.zeros(n)
        for s_idx in top_idx:
            s = s_idx - (n - 1)
            js = j - s
            jp = j + s
            ms = (js >= 0) & (js < n)
            mp = (jp >= 0) & (jp < n)
            if ms.any():
                g[ms] += (1.0 - x[js[ms]])
            if mp.any():
                g[mp] += (-x[jp[mp]])
        g = (g / top_k) * dx
        v = mu0 * v - lr * g
        x = project_feasible(x + v, target)
        if t % 200 == 199:
            val = compute_c5(x)
            if val < best_val:
                best_val = val
                best_x = x.copy()
    val = compute_c5(x)
    if val < best_val:
        best_val = val
        best_x = x.copy()
    return best_x, best_val

def smoothmax_obj_grad(h, n, beta):
    dx = 2.0 / n
    corr = compute_corr_full(h)
    m = np.max(corr)
    sh = np.clip(beta * (corr - m), -500, 500)
    exps = np.exp(sh)
    sw = np.sum(exps)
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    j = np.arange(n)
    grad = np.zeros(n)
    active = np.where(w > 1e-7)[0]
    for k in active:
        wk = w[k]
        s = k - (n - 1)
        js = j - s
        jp = j + s
        ms = (js >= 0) & (js < n)
        mp = (jp >= 0) & (jp < n)
        if ms.any():
            grad[ms] += wk * (1.0 - h[js[ms]])
        if mp.any():
            grad[mp] += wk * (-h[jp[mp]])
    grad *= dx
    return val, grad

def slsqp_refine(h, target, maxiter, beta):
    n = len(h)
    def fg(x):
        v, g = smoothmax_obj_grad(x, n, beta)
        return v, g
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fg, h, jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_feasible(h_new, n_new / 2.0)

    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:

        if len(h_start) != n_res:

        # Phase B: SLSQP with increasing beta

        # Phase D: Perturbations and re-optimize
