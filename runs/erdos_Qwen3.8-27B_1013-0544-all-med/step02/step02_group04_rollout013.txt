    m = np.max(x)
    if beta < 1:
        return m
    shifted = np.clip(beta * (x - m), -100, 100)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        if free.sum() == 0:
            break
        step = diff / free.sum()
        # Check if step would violate bounds
        max_step_up = (1.0 - h[free]).min() if diff > 0 else 0
        max_step_down = h[free].min() if diff < 0 else 0
        if diff > 0 and step > max_step_up:

        elif diff < 0 and -step > max_step_down:
            h[free] -= max_step_down
        else:
            h[free] += step
    return np.clip(h, 0.0, 1.0)

            if c5 < best_c5:
                best_h = h_new
                best_c5 = c5
        except Exception:
            pass

    # Phase 5: Fine perturbation search
    for attempt in range(1000):
        if time_left() < 5:
            break
        scale = 0.002 + 0.003 * np.exp(-attempt / 200.0)
        h_try = best_h + rng.randn(n_points) * scale
        h_try = project_h(h_try, target_sum)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_h = h_try
            best_c5 = c5_try
            # Quick SLSQP from this point
            if time_left() > 15:
                cons2 = [{'type': 'eq', 'fun': lambda x: slsqp_con_fun(x, target_sum),
                          'jac': lambda x: slsqp_con_jac(x, target_sum)}]
                try:
                    res = minimize(lambda x: slsqp_obj(x, 5000), h_try,
                                  jac=lambda x: slsqp_grad(x, 5000),
                                  method='SLSQP', bounds=bounds, constraints=cons2,
                                  options={'maxiter': 3000, 'ftol': 1e-16})
                    h_new2 = project_h(res.x, target_sum)
                    c5_2 = compute_c5(h_new2)
                    if c5_2 < best_c5:
                        best_h = h_new2
                        best_c5 = c5_2
                except Exception:
                    pass

    # Phase 6: Final high-precision polish
    if time_left() > 10:
        for beta in [50000, 100000]:
            if time_left() < 8:
                break
            h_res, _ = pgd_refine(best_h, beta, 20000, 0.005, 0.7, seed_counter)
            seed_counter += 1
            c5 = compute_c5(h_res)
            if c5 < best_c5:
                best_h = h_res
                best_c5 = c5

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
