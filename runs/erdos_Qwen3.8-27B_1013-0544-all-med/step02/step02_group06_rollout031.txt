            h = h.mean() + np.fft.irfft(H, n)
        elif strat == 3:
            # Block complement (flip a block: h -> 1-h in that region)
            blen = np.random.randint(4, max(5, n // 4))
            bs = np.random.randint(0, n - blen)
            h[bs:bs + blen] = 1.0 - h[bs:bs + blen]
        elif strat == 4:
            # Multi-lag subgradient
            corr = np.correlate(h, 1.0 - h, mode="full")
            top3 = np.argsort(corr)[::-1][:3]
            for kk in top3:
                k_shift = int(kk) - (n - 1)
                g = np.zeros(n)
                for j in range(n):
                    if 0 <= j + k_shift < n:
                        g[j] += (1.0 - h[j + k_shift])
                    if 0 <= j - k_shift < n:
                        g[j] -= h[j - k_shift]
                h -= 0.008 * g
            h += np.random.randn(n) * 0.003
        elif strat == 5:
            # Pair exchange
            nsw = np.random.randint(3, 8)
            for _ in range(nsw):
                i, j = np.random.choice(n, 2, replace=False)
                h[i], h[j] = h[j], h[i]
            h += np.random.randn(n) * 0.002
        else:
            # Antisymmetric half perturbation
            half = n // 2
            amp = 0.04
            h[:half] += amp * np.random.randn(half)
            h[half:] -= amp * np.random.randn(n - half)

        h = project_box_sum(h, ts)

        for beta_r in [1000, 2500]:
            if rem() < 15:
                break
            try:
                r = minimize(obj_lam, h, args=(n, ts, beta_r, beta_r * 1.5),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 350, 'ftol': 1e-15})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 6: Final ultra-high-precision polish
    if rem() > 12:
        for beta_f in [4000, 7000, 10000]:
            if rem() < 8:
                break
            try:
                r = minimize(obj_lam, best_h, args=(n, ts, beta_f, beta_f * 2.0),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-14})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Final SLSQP polish
    if rem() > 15:
        try:
            def final_obj(h):
                c = np.correlate(h, 1.0 - h, mode="full")
                vals = c[np.argsort(c)[::-1][:4]]
                return smooth_max(vals, 10000.0) * (2.0 / n)
            cons2 = [{'type': 'eq', 'fun': slsqp_eq, 'jac': slsqp_eq_jac}]
            r = minimize(final_obj, best_h, method='SLSQP', bounds=bnds,
                         constraints=cons2,
                         options={'maxiter': 3000, 'ftol': 1e-18})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
