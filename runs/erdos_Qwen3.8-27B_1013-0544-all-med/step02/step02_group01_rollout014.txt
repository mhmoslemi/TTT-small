    corr = np.correlate(h, 1.0 - h, mode="full")

        c = corr[idx]

    shifted = beta * (c - m)
    shifted = np.clip(shifted, -50, 0)
    exps = np.exp(shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)

        wk = w[kk]

        s = int(idx[kk]) - (n - 1)

def pgd_smoothmax(h, target, beta_start, beta_end, n_iters, alpha0=0.01, top_k=None):
    """Projected gradient descent with smooth max."""
    n = len(h)
    x = h.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    
    for t in range(n_iters):
        # Adaptive beta
        frac = t / max(1, n_iters - 1)
        beta = beta_start * (beta_end / beta_start) ** frac
        
        val, grad = smoothmax_val_grad(x, beta, top_k)
        
        # Backtracking line search
        alpha = alpha0
        f_curr = val
        gnorm = np.linalg.norm(grad)
        for bt in range(30):
            x_new = project_capped_simplex(x - alpha * grad, target)
            f_new, _ = smoothmax_val_grad(x_new, beta, top_k)
            if f_new <= f_curr + 1e-4 * alpha * gnorm * gnorm:
                break
            alpha *= 0.5
        
        x = x_new
        # Check actual C5
        v = compute_c5(x)
        if v < best_v:
            best_v = v
            best_x = x.copy()
    
    return best_x, best_v

def slsqp_refine(h, target, maxiter=500, beta=200):
    n = len(h)
    def obj(x):
        v, _ = smoothmax_val_grad(x, beta)
        return v
    def objg(x):
        _, g = smoothmax_val_grad(x, beta)
        return g
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_capped_simplex(res.x, target)
    except Exception:
        return project_capped_simplex(h, target)

    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)

    n = len(h)
    x = h.copy()
    x_prev = x.copy()
    y = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    j = np.arange(n)
    
    for t in range(n_iters):
        alpha = alpha0 * (decay ** t)
        if t > 0:
            mu = (t - 1) / (t + 2)
            y = x + mu * (x - x_prev)
            y = project_capped_simplex(y, target)
        
        # Find argmax lag
        corr = np.correlate(y, 1.0 - y, mode="full")
        k_max = np.argmax(corr)
        s = k_max - (n - 1)
        
        # Subgradient
        g = np.zeros(n)
        i1 = np.arange(max(0, s), n)
        if len(i1) > 0:
            g[i1] += (1.0 - y[i1 - s])
        i2_lo = max(0, -s)
        i2_hi = min(n, n - s)
        if i2_hi > i2_lo:
            g[i2_lo:i2_hi] += (-y[i2_lo:i2_hi + s])
        
        dx_val = 2.0 / n
        g *= dx_val
        
        x_new = project_capped_simplex(y - alpha * g, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    
    return best_x, best_v

where dx = 2/n.

To minimize C₅, we want to minimize max_s (transitions at offset s).

import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.interpolate import interp1d
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_capped_simplex(v, target):
    n = len(v)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def _grad_one_lag(y, s, n):
    g = np.zeros(n)
    i1 = np.arange(max(0, s), n)
    if len(i1) > 0:
        g[i1] += (1.0 - y[i1 - s])
    i2_lo = max(0, -s)
    i2_hi = min(n, n - s)
    if i2_hi > i2_lo:
        g[i2_lo:i2_hi] += (-y[i2_lo:i2_hi + s])
    return g


def smoothmax_val_grad(h, beta, top_k=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    nc = len(corr)
    if top_k is not None and top_k < nc:
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
    else:
        idx = np.arange(nc)
        c = corr
    m = c.max()
    shifted = np.clip(beta * (c - m), -40.0, 0.0)
    exps = np.exp(shifted)
    sw = exps.sum()
    w = exps / sw
    val = (m + np.log(sw) / beta) * dx
    grad = np.zeros(n)
    for kk in range(len(c)):
        wk = w[kk]
        if wk < 1e-12:
            continue
        s = int(idx[kk]) - (n - 1)
        grad += wk * _grad_one_lag(h, s, n)
    grad *= dx
    return val, grad


def pgd_smoothmax(h, target, beta_start, beta_end, n_iters, alpha0=0.02):
    n = len(h)
    x = h.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(n_iters):
        frac = (t + 1) / n_iters
        beta = beta_start * (beta_end / beta_start) ** frac
        val, grad = smoothmax_val_grad(x, beta)
        f_curr = val
        gnorm2 = np.dot(grad, grad)
        alpha = alpha0
        x_new = x
        for bt in range(20):
            cand = project_capped_simplex(x - alpha * grad, target)
            f_new, _ = smoothmax_val_grad(cand, beta)
            if f_new <= f_curr + 1e-4 * alpha * gnorm2:
                x_new = cand
                break
            alpha *= 0.5
        else:
            x_new = cand
        x = x_new
        v = compute_c5(x)
        if v < best_v:
            best_v = v
            best_x = x.copy()
    return best_x, best_v


def subgradient_nesterov(h, target, n_iters, alpha0=0.08, decay=0.9997, top_k=1):
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    x_prev = x.copy()
    y = x.copy()
    best_x = x.copy()
    best_v = compute_c5(x)
    for t in range(n_iters):
        alpha = alpha0 * (decay ** t)
        if t > 0:
            mu = (t - 1) / (t + 2.0)
            y = project_capped_simplex(x + mu * (x - x_prev), target)
        corr = np.correlate(y, 1.0 - y, mode="full")
        if top_k == 1:
            k_max = int(np.argmax(corr))
            s = k_max - (n - 1)
            g = _grad_one_lag(y, s, n) * dx
        else:
            idx = np.argsort(corr)[-top_k:]
            g = np.zeros(n)
            for kk in idx:
                s = int(kk) - (n - 1)
                g += _grad_one_lag(y, s, n)
            g = g / top_k * dx
        x_new = project_capped_simplex(y - alpha * g, target)
        v = compute_c5(x_new)
        if v < best_v:
            best_v = v
            best_x = x_new.copy()
        x_prev = x
        x = x_new
    return best_x, best_v


def slsqp_refine(h, maxiter=400, beta=300):
    n = len(h)
    target = n / 2.0

    def obj(x):
        return smoothmax_val_grad(x, beta)[0]

    def objg(x):
        return smoothmax_val_grad(x, beta)[1]

    bounds = Bounds(np.zeros(n), np.ones(n))
    ones_n = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: ones_n}]
    try:
        res = minimize(obj, h, jac=objg, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return project_capped_simplex(res.x, target)
    except Exception:
        return project_capped_simplex(h, target)


def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate')
    h_new = np.clip(f(x_new), 0.0, 1.0)
    return project_capped_simplex(h_new, n_new / 2.0)


def make_inits(n, seed):
    rng = np.random.RandomState(seed)
    inits = []
    inits.append(np.full(n, 0.5))
    h = np.zeros(n); h[:n // 2] = 1.0
    inits.append(project_capped_simplex(h, n / 2.0))
    h = np.zeros(n); h[::2] = 1.0
    inits.append(project_capped_simplex(h, n / 2.0))
    x = np.linspace(0, 2, n)
    inits.append(project_capped_simplex(0.5 + 0.3 * np.sin(np.pi * x), n / 2.0))
    inits.append(project_capped_simplex(rng.rand(n), n / 2.0))
    inits.append(project_capped_simplex(0.5 + 0.4 * np.cos(2 * np.pi * x), n / 2.0))
    return inits


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_best = np.array(initial_h_values, dtype=float).copy()
    else:
        h_best = np.full(200, 0.5)

    best_c5 = compute_c5(h_best)
    best_h = h_best.copy()
    n = len(best_h)
    target = n / 2.0

    # Phase 1: PGD with adaptive beta at base resolution
    if tl() > 80:
        h1, c1 = pgd_smoothmax(best_h, target, 40, 300, int(min(1500, tl() * 2)), 0.02)
        if c1 < best_c5:
            best_c5, best_h = c1, h1

    # Phase 2: Subgradient with Nesterov (exact max)
    if tl() > 60:
        h2, c2 = subgradient_nesterov(best_h, target, int(min(8000, tl() * 3)),
                                       alpha0=0.06, decay=0.9998, top_k=1)
        if c2 < best_c5:
            best_c5, best_h = c2, h2

    # Phase 3: SLSQP polish (only for n<=300)
    if tl() > 40 and n <= 300:
        h3 = slsqp_refine(best_h, maxiter=400, beta=400)
        c3 = compute_c5(h3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3

    # Phase 4: Try alternative initializations
    inits = make_inits(n, seed)
    for ini in inits:
        if tl() < 80:
            break
        h_r, c_r = pgd_smoothmax(ini, target, 30, 200, 300, 0.03)
        if c_r < best_c5:
            h_r2, c_r2 = subgradient_nesterov(h_r, target, 3000, 0.05, 0.9998, 1)
            if c_r2 < c_r:
                h_r, c_r = h_r2, c_r2
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r
                if tl() > 40 and n <= 300:
                    h_s = slsqp_refine(best_h, maxiter=300, beta=400)
                    c_s = compute_c5(h_s)
                    if c_s < best_c5:
                        best_c5, best_h = c_s, h_s

    # Phase 5: Higher resolution
    for n_new in [300, 500, 800]:
        if tl() < 100:
            break
        h_interp = interp_to_n(best_h, n_new)
        tgt_new = n_new / 2.0
        # PGD at new resolution
        if tl() > 80:
            h_a, c_a = pgd_smoothmax(h_interp, tgt_new, 50, 400,
                                      int(min(1000, tl() * 2)), 0.01)
            if c_a < best_c5:
                best_c5, best_h = c_a, h_a
        # Subgradient at new resolution
        if tl() > 50:
            h_b, c_b = subgradient_nesterov(best_h, tgt_new,
                                             int(min(5000, tl() * 2)),
                                             0.04, 0.9999, 1)
            if c_b < best_c5:
                best_c5, best_h = c_b, h_b
        # SLSQP only for smaller sizes
        if tl() > 40 and n_new <= 300:
            h_c = slsqp_refine(best_h, maxiter=300, beta=500)
            c_c = compute_c5(h_c)
            if c_c < best_c5:
                best_c5, best_h = c_c, h_c

    # Phase 6: Final refinement rounds on best
    n_fin = len(best_h)
    tgt_fin = n_fin / 2.0
    if tl() > 30:
        h_f, c_f = pgd_smoothmax(best_h, tgt_fin, 200, 1500,
                                  int(min(500, tl() * 2)), 0.005)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f
    if tl() > 20:
        h_g, c_g = subgradient_nesterov(best_h, tgt_fin,
                                         int(min(5000, tl() * 3)),
                                         0.03, 0.9999, 3)
        if c_g < best_c5:
            best_c5, best_h = c_g, h_g
    if tl() > 30 and n_fin <= 300:
        h_h = slsqp_refine(best_h, maxiter=500, beta=800)
        c_h = compute_c5(h_h)
        if c_h < best_c5:
            best_c5, best_h = c_h, h_h

    return best_h, best_c5, len(best_h)
