import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx


def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta


def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-13:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def obj_lse(h, n, target_sum, beta, lam):
    c = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(c, beta) * (2.0 / n) + lam * (h.sum() - target_sum) ** 2


def obj_worst(h, n, target_sum, beta, lam, worst_idx, wpen):
    c = np.correlate(h, 1.0 - h, mode="full")
    base = smooth_max(c, beta) * (2.0 / n)
    wp = 0.0
    for i in worst_idx:
        wp += c[i] * c[i]
    return base + wpen * wp + lam * (h.sum() - target_sum) ** 2


def get_worst(h, k=3):
    c = np.correlate(h, 1.0 - h, mode="full")
    return tuple(int(i) for i in np.argsort(c)[-k:])


def subgradient_step(h, n, k_shift, step_size):
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        i1 = j + k_shift
        i2 = j - k_shift
        if 0 <= i1 < n:
            val += (1.0 - h[i1])
        if 0 <= i2 < n:
            val -= h[i2]
        g[j] = val
    return h - step_size * g


def parametric_h(n, a1, w1, p1, a2, w2, p2):
    x = np.linspace(0, 2.0, n, endpoint=False) + 1.0 / n
    h = 0.5 + a1 * np.sin(w1 * x + p1) + a2 * np.sin(w2 * x + p2)
    return np.clip(h, 0.0, 1.0)


def time_left(t0, budget_s):
    return budget_s - (time.time() - t0)

            h_new = project_box_sum(r.x, ts)
