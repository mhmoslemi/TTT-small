import numpy as np
from scipy.optimize import minimize, Bounds
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = h.sum()
        if abs(s - target_sum) < 1e-13:
            break
        diff = target_sum - s
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta


def coord_opt_scalar(alpha, beta, lo, hi):
    """Solve min_{x in [lo,hi]} max_k alpha[k]*x + beta[k] via bisection on subgradient."""
    lo_b, hi_b = lo, hi
    for _ in range(50):
        mid = 0.5 * (lo_b + hi_b)
        vals = alpha * mid + beta
        mx = np.max(vals)
        active = np.where(vals >= mx - 1e-14)[0]
        sg = np.mean(alpha[active])
        if sg > 1e-14:
            hi_b = mid
        elif sg < -1e-14:
            lo_b = mid
        else:
            break
    return 0.5 * (lo_b + hi_b)


def coord_desc_step(h, i, target_sum):
    n = len(h)
    dx = 2.0 / n
    other_sum = h.sum() - h[i]
    lo = max(0.0, target_sum - other_sum)
    hi = min(1.0, target_sum - other_sum)

        return h[i]
    h_new = h.copy()
    h_new[i] = 0.0

    h_new[i] = 1.0
    vals1 = np.correlate(h_new, 1.0 - h_new, mode="full") * dx
    alpha = vals1 - vals0
    beta = vals0
    return coord_opt_scalar(alpha, beta, lo, hi)


def coordinate_descent(h, target_sum, max_sweeps=200, tol=1e-12):
    n = len(h)
    h = project_feasible(h.astype(float).copy(), target_sum)
    prev_c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        max_change = 0.0
        for i in range(n):
            x_opt = coord_desc_step(h, i, target_sum)
            change = abs(x_opt - h[i])
            if change > max_change:
                max_change = change
            h[i] = x_opt
        h = project_feasible(h, target_sum)
        c5 = compute_c5(h)
        if c5 < prev_c5:
            prev_c5 = c5
        if max_change < tol:
            break
    return h, prev_c5


def resample_h(h_old, n_new):
    x_old = np.linspace(0, 2, len(h_old), endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)


def smooth_max_obj(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return smooth_max(corr, beta) * dx


def con_fun(h, target_sum, n):
    return h.sum() - target_sum


def con_jac(h, target_sum, n):
    return np.ones(n)
