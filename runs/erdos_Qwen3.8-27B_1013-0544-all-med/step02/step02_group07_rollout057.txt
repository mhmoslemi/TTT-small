import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    if beta == 0:
        return m
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-14:
            break
        free = (h > 1e-12) & (h < 1.0 - 1e-12)
        n_free = free.sum()
        if n_free > 0:
            step = diff / n_free
            # Check if step would violate bounds
            max_up = (1.0 - h[free]).min() if diff > 0 else 0
            max_down = (-h[free]).min() if diff < 0 else 0
            step = np.clip(step, max_down, max_up) if n_free > 0 else 0
            h[free] += step
        else:
            break
    return np.clip(h, 0.0, 1.0)

def subgradient_corr(h, k, n):
    """Compute subgradient of corr[k] with respect to h."""
    # corr[k] = sum_i h[i] * (1 - h[i+k]) over valid i
    # For the full correlation, index n-1+k corresponds to lag k
    sg = np.zeros(n)
    if k >= 0:
        # i ranges from 0 to n-1-k
        for i in range(n - k):
            sg[i] += (1 - h[i + k])
            sg[i + k] += (-h[i])
    else:
        ka = -k
        for i in range(n - ka):
            sg[i + ka] += (1 - h[i])
            sg[i] += (-h[i + ka])
    return sg

# Vectorized version
def subgradient_corr_vec(h, k, n):
    sg = np.zeros(n)
    if k >= 0:
        valid = slice(0, n - k)
        sg[valid] += (1 - h[k:n])
        sg[k:] += (-h[0:n-k])
    else:
        ka = -k
        valid = slice(0, n - ka)
        sg[ka:] += (1 - h[valid])
        sg[valid] += (-h[ka:n])
    return sg
