                    adj_idx = np.where(mask)[0][0]
                    trial[adj_idx] += diff
                    trial = np.clip(trial, 0.0, 1.0)
                    if abs(trial.sum() - target_sum) > 1e-8:
                        continue
                c5 = compute_c5(trial)
                if c5 < cur_c5 - 1e-13:
                    cur = trial
                    cur_c5 = c5
                    old = frac
    return cur, cur_c5

def slsqp_solve(h0, beta, top_idx, pb, bounds, cons, dx, maxiter, ftol):
    if top_idx is not None:
        obj = partial(obj_active, top_idx=top_idx, pb=pb, dx=dx)
    else:
        obj = partial(obj_smooth, beta=beta, dx=dx)
    try:
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    t_deadline = t_start + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = None

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def try_update(h_new, n_pts):
        nonlocal best_h, best_c5, best_n
        h_new = project_h(h_new, n_pts / 2.0)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5, best_n = h_new.copy(), c5, n_pts
        return h_new, c5

    n_list = [70, 64, 80, 72, 84]
    time_alloc = [0.50, 0.15, 0.15, 0.10, 0.10]

    for ni, n_points in enumerate(n_list):
        if time.time() > t_deadline:
            break
        n_time = (budget_s - 10) * time_alloc[ni]
        t_n_start = time.time()
        t_n_end = t_n_start + n_time
        target_sum = n_points / 2.0
        dx = 2.0 / n_points
        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        cf = partial(con_fun, target_sum=target_sum)
        cj = partial(con_jac, target_sum=target_sum)
        cons = [{'type': 'eq', 'fun': cf, 'jac': cj}]

        # Build diverse starting points
        starts = []
        if h_init is not None:
            if len(h_init) == n_points:
                starts.append(project_h(h_init.copy(), target_sum))
            else:
                xo = np.linspace(0, 2, len(h_init), endpoint=False)
                xn = np.linspace(0, 2, n_points, endpoint=False)
                hi = np.interp(xn, xo, h_init, period=2.0)
                starts.append(project_h(hi, target_sum))

        # Structured starts
        h_alt = np.zeros(n_points)
        h_alt[0::2] = 1.0
        starts.append(project_h(h_alt, target_sum))

        # Block pattern: alternating blocks of 0s and 1s
        h_blk = np.zeros(n_points)
        block_size = max(2, n_points // 14)
        for b in range(0, n_points, block_size * 2):
            h_blk[b:b+block_size] = 1.0
        starts.append(project_h(h_blk, target_sum))

        # Sine-based
        x_ = np.linspace(0, 2, n_points, endpoint=False)
        h_sin = 0.5 + 0.4 * np.sin(2 * np.pi * x_ / 2.0 * 3)
        starts.append(project_h(h_sin, target_sum))

        # Random continuous
        h_rnd = np.random.random(n_points)
        starts.append(project_h(h_rnd, target_sum))

        # Random binary
        h_bins = np.zeros(n_points)
        pos = np.random.choice(n_points, size=int(target_sum), replace=False)
        h_bins[pos] = 1.0
        starts.append(h_bins)

        curr_h = None
        curr_c5 = float('inf')

        # Main optimization loop over starts
        max_starts = 5 if n_points == 70 else 3
        for si in range(min(len(starts), max_starts)):
            if time.time() > t_n_end or time.time() > t_deadline:
                break
            h0 = project_h(starts[si].copy(), target_sum)
            phase_time = (t_n_end - t_n_start) / max_starts

            # Phase 1: Graduated smooth_max SLSQP
            betas = [30, 60, 120, 250, 500, 1000] if si == 0 else [50, 150, 400]
            for beta in betas:
                if time.time() > t_n_end - phase_time * 0.3 or time.time() > t_deadline:
                    break
                xr = slsqp_solve(h0, beta, None, 0, bounds, cons, dx, 250, 1e-12)
                h_new, c5 = try_update(xr, n_points)
                if c5 < curr_c5:
                    curr_h, curr_c5 = h_new, c5
                    h0 = h_new.copy()

            # Phase 2: Active-set focused optimization
            if curr_h is not None and time.time() < t_n_end - 5:
                corr = np.correlate(curr_h, 1.0 - curr_h, mode="full")
                for top_k in [4, 8, 14]:
                    if time.time() > t_n_end - 5 or time.time() > t_deadline:
                        break
                    top_idx = np.argsort(corr)[::-1][:top_k]
                    for pb in [150, 300]:
                        xr = slsqp_solve(h0, 0, top_idx, pb, bounds, cons, dx, 200, 1e-13)
                        h_new, c5 = try_update(xr, n_points)
                        if c5 < curr_c5:
                            curr_h, curr_c5 = h_new, c5
                            h0 = h_new.copy()
                            corr = np.correlate(curr_h, 1.0 - curr_h, mode="full")

            # Phase 3: Coordinate descent
            if curr_h is not None and time.time() < t_n_end - 3:
                cd_end = min(time.time() + 15, t_n_end - 3, t_deadline)
                cd_h, cd_c5 = coord_descent(curr_h, target_sum, n_points,
                                            max_rounds=3, t_end=cd_end)
                if cd_c5 < curr_c5:
                    curr_h, curr_c5 = cd_h, cd_c5
                try_update(curr_h, n_points)

        # Phase 4: High-precision final polish on best for this n
        if curr_h is not None and time.time() < t_deadline:
            for beta in [500, 1000, 2000]:
                if time.time() > t_deadline:
                    break
                xr = slsqp_solve(curr_h, beta, None, 0, bounds, cons, dx, 400, 1e-14)
                h_new, c5 = try_update(xr, n_points)
                if c5 < curr_c5:
                    curr_h, curr_c5 = h_new, c5

            # Active polish with very high beta
            if time.time() < t_deadline - 3:
                corr = np.correlate(curr_h, 1.0 - curr_h, mode="full")
                top_idx = np.argsort(corr)[::-1][:6]
                xr = slsqp_solve(curr_h, 0, top_idx, 500, bounds, cons, dx, 300, 1e-14)
                h_new, c5 = try_update(xr, n_points)
                if c5 < curr_c5:
                    curr_h, curr_c5 = h_new, c5

    # Ultimate final polish
    if best_h is not None and time.time() < t_deadline:
        nf = best_n
        ts = nf / 2.0
        dxf = 2.0 / nf
        bf = Bounds(np.zeros(nf), np.ones(nf))
        cff = partial(con_fun, target_sum=ts)
        cjf = partial(con_jac, target_sum=ts)
        consf = [{'type': 'eq', 'fun': cff, 'jac': cjf}]
        for beta in [1500]:
            if time.time() > t_deadline - 1:
                break
            xr = slsqp_solve(best_h, beta, None, 0, bf, consf, dxf, 300, 1e-15)
            h_new, c5 = try_update(xr, best_n)

    return best_h, best_c5, best_n
