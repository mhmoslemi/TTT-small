def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-12 and s > 0:

        h = np.clip(h, 0.0, 1.0)
        # Second pass for tighter projection
        s2 = h.sum()
        h += (target_sum - s2) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize, Bounds, linprog
import time

_target_sum = 0.0
_n_pts = 1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target_sum) > 1e-14:
        h += (target_sum - s) / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, k):
    n = len(h)
    lag = k - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
    else:
        if n + lag > 0:
            grad[-lag:] += (1.0 - h[:n + lag])
            grad[:n + lag] -= h[-lag:]
    return grad

def active_set_lp(h, target_sum, tol, max_active):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_c = np.max(corr)
    active = np.where(corr >= max_c - tol)[0]
    if len(active) > max_active:
        active = np.argsort(corr)[::-1][:max_active]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    n_ineq = len(active) + 2 * n
    A_ub = np.zeros((n_ineq, n + 1))
    b_ub = np.zeros(n_ineq)
    for i, k in enumerate(active):
        g = lag_gradient(h, k)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = float(g @ h - corr[k])
    for j in range(n):
        A_ub[len(active) + j, j] = -1.0
        A_ub[len(active) + j + n, j] = 1.0
        b_ub[len(active) + j + n] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target_sum])
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, method='highs')
    if res.success:
        return res.x[:n]
    return None

def con_fun(x):
    return np.sum(x) - _target_sum

def con_jac(x):
    return np.ones(_n_pts)

def obj_smooth_max(h, dx, beta):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return smooth_max(corr, beta) * dx

def run(seed=42, budget_s=1100, **kwargs):
    global _target_sum, _n_pts
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = None

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def tl():
        return budget_s - (time.time() - start_time)

    n_list = [70, 80, 90, 100]
    time_frac = [0.35, 0.22, 0.22, 0.21]

    for ni, n in enumerate(n_list):
        if tl() < 100:
            break
        t_max = time_frac[ni] * budget_s
        target_sum = n / 2.0
        _target_sum = target_sum
        _n_pts = n
        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]

        starts = []
        if h_init is not None:
            if len(h_init) == n:
                starts.append(h_init.copy())
            else:
                xo = np.linspace(0, 2, len(h_init), endpoint=False)
                xn = np.linspace(0, 2, n, endpoint=False)
                hi = np.interp(xn, xo, h_init, period=2.0)
                starts.append(project_h(hi, target_sum))
        if best_h is not None and len(best_h) != n:

            xn = np.linspace(0, 2, n, endpoint=False)
            hi = np.interp(xn, xo, best_h, period=2.0)
            starts.append(project_h(hi, target_sum))
        starts.append(np.full(n, 0.5))

                        curr_h, curr_c5 = hnew.copy(), c5
                        base = hnew.copy()

                            best_h, best_c5, best_n = hnew.copy(), c5, n
                except Exception:
                    pass
            # Second round with smaller perturbations
            for _ in range(8):
                if tl() < 100:
                    break
