from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode="full")

def smooth_max(x, beta):
    m = np.max(x)
    shifted = beta * (x - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta

    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):

        if abs(d) < 1e-14:

    """Compute subgradient w.r.t. h for the smooth max of top correlations."""
    g = np.zeros(n)
    corr = get_corr(h)
    # Get weights for each lag based on soft assignment
    n_half = n - 1
    for k in top_lags:
        # Lag k means correlation at index n_half + k
        # corr[n_half + k] = sum_i h[i] * (1 - h[i+k]) for valid i
        # d(corr)/d(h[j]) = (1-h[j+k]) - h[j-k] for valid indices
        for i in range(n):
            contrib = 0.0
            if 0 <= i + k < n:
                contrib += (1.0 - h[i + k])
            if 0 <= i - k < n:
                contrib -= h[i - k]
            g[i] += contrib
    return g

def nesterov_smooth_max(h, n, ts, beta, steps, lr_init, lr_decay):
    """Nesterov accelerated gradient on smooth max with projection."""
    x = h.copy()
    y = h.copy()
    t = 1.0
    for step in range(steps):
        lr = lr_init / (1.0 + lr_decay * step)
        corr = get_corr(y)
        # Soft max gradient: exp(beta*(corr_k - max)) / sum(exp(...))
        m = np.max(corr)
        weights = np.exp(beta * (corr - m))
        weights /= weights.sum()
        
        # Gradient of smooth max w.r.t. h
        g = np.zeros(n)
        n_half = n - 1
        for k_idx in range(len(corr)):
            k = k_idx - n_half
            w = weights[k_idx]
            if w < 1e-10:

            # d(corr_{n_half+k})/d(h[j]) for each j
            # corr[n_half+k] = sum_i h[i]*(1-h[i+k])

            #           if j appears as i+k (i=j-k), contributes -h[j-k]
            for i in range(n):
                if 0 <= i + k < n:
                    g[i] += w * (1.0 - h[i + k])
                if 0 <= i - k < n:
                    g[i] -= w * h[i - k]
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        alpha = (1.0 - t) / t_new
        y = x + alpha * (y - x)  # Nesterov extrapolation
        x_new = y - lr * g
        x_new = project_box_sum(x_new, ts)
        t = t_new
        x = x_new
    
    return x

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        m = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] = np.clip(h[m] + d / cnt, 0.0, 1.0)
    return h

def obj_full(h, n, ts, beta, lam):
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.exp(beta * (corr - m))
    s = np.clip(s, 0, 1e300)
    val = (m + np.log(s.sum()) / beta) * (2.0 / n)
    return val + lam * (h.sum() - ts) ** 2

                g[i] += (1.0 - h[i + k])

    lags = get_top_lags(best_h, n, 4)
    for beta in [1500, 4000, 10000]:
        if tl() < 15:
            break
        try:
            r = minimize(obj_targeted, best_h, args=(n, ts, lags, beta, beta * 1.5),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 500, 'ftol': 1e-18})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
            lags = get_top_lags(h_new, n, 4)
        except Exception:
            pass

    # Phase 3: Binary projection + continuous re-relaxation
    b = binary_project(best_h, n)
    h_bin = b.copy()
    for beta in [500, 2000, 6000]:
        if tl() < 15:
            break
        try:
            r = minimize(obj_smoothmax, h_bin, args=(n, ts, beta, beta * 2),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 400, 'ftol': 1e-15})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 4: Parametric diverse starts
    for trial in range(6):
        if tl() < 40:
            break
        a1 = np.random.uniform(0.15, 0.45)
        w1 = np.random.uniform(1.5, 5.0) * np.pi
        p1 = np.random.uniform(0, 2 * np.pi)
        a2 = np.random.uniform(-0.25, 0.25)
        w2 = np.random.uniform(3.0, 9.0) * np.pi
        p2 = np.random.uniform(0, 2 * np.pi)
        h_p = parametric_h(n, a1, w1, p1, a2, w2, p2, ts)
        for beta in [1000, 4000]:
            if tl() < 20:
                break
            try:
                r = minimize(obj_smoothmax, h_p, args=(n, ts, beta, beta * 1.5),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 300, 'ftol': 1e-15})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 5: Diverse restarts with subgradient targeting
    for it in range(300):
        if tl() < 45:
            break

        h = best_h.copy()
        corr = np.correlate(h, 1.0 - h, mode="full")
        top3 = np.argsort(corr)[-3:]
        top5 = np.argsort(corr)[-5:]

        s = it % 8
        if s == 0:
            h += np.random.randn(n) * 0.012
        elif s == 1:
            h = subgrad_step(h, n, top3, 0.006)
            h += np.random.randn(n) * 0.003
        elif s == 2:
            blen = np.random.randint(3, max(4, n // 10))
            bs = np.random.randint(0, n - blen)
            h[bs:bs + blen] = 0.5 + 0.3 * np.random.randn(blen)
        elif s == 3:
            x = np.arange(n) * 2.0 / n
            w = np.random.uniform(1.5, 7.0) * np.pi
            ph = np.random.uniform(0, 2 * np.pi)
            h += 0.03 * np.sin(w * x + ph)
        elif s == 4:
            nf = np.random.randint(2, 5)
            flips = np.random.choice(n, nf, replace=False)
            h[flips] = 1.0 - h[flips]
        elif s == 5:
            for _ in range(np.random.randint(2, 6)):
                i, j = np.random.choice(n, 2, replace=False)
                h[i], h[j] = h[j], h[i]
            h += np.random.randn(n) * 0.004
        elif s == 6:
            h = subgrad_step(h, n, top5, 0.003)
            h += np.random.randn(n) * 0.005
        else:
            # Scale toward 0.5/1 extremes (bipolarization)
            h = 0.3 * h + 0.7 * np.where(h >= np.median(h), 1.0, 0.0)

        h = project_box_sum(h, ts)

        if tl() < 15:
            break
        for beta in [2000, 5000]:
            if tl() < 12:
                break
            try:
                r = minimize(obj_smoothmax, h, args=(n, ts, beta, beta * 1.5),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 150, 'ftol': 1e-15})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
                    break
            except Exception:
                pass

    # Phase 6: Final high-precision polish with very high beta
    if tl() > 15:
        for beta in [8000, 15000, 30000]:
            if tl() < 10:
                break
            try:
                r = minimize(obj_smoothmax, best_h, args=(n, ts, beta, beta * 2),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 3000, 'ftol': 1e-18, 'gtol': 1e-17})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 7: One more targeted pass on final top lags
    if tl() > 15:
        lags_final = get_top_lags(best_h, n, 3)
        try:
            r = minimize(obj_targeted, best_h, args=(n, ts, lags_final, 20000, 40000),
                         method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 2000, 'ftol': 1e-18})
            h_new = project_box_sum(r.x, ts)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
