            n_mask = max(1, int(mask.sum()))
            h_new[mask] += rng.randn(n_mask) * 0.1
        h_new = project_h(h_new, target_sum)
        c5_new = compute_c5(h_new)
        delta = c5_new - cur_c5
        if delta < 0 or rng.random() < np.exp(-min(150.0, abs(delta) / T)):
            cur, cur_c5 = h_new, c5_new
            if cur_c5 < best_c5:
                best, best_c5 = cur.copy(), cur_c5
                stag = 0
            else:
                stag += 1
        else:
            stag += 1
        T = max(T_min, T * cooling)
    return best, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random
    start_time = time.time()

    def tleft():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(70, 0.5)

    n_pts = len(h_init)
    target = n_pts / 2.0
    dx = 2.0 / n_pts
    h_init = project_h(h_init, target)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: SLSQP multi-beta
    t1_end = start_time + 0.20 * budget_s
    for beta in [80, 160, 320, 640]:
        if time.time() > t1_end or tleft() < 20:
            break
        h_new = slsqp_step(best_h, target, beta, dx, maxiter=3000)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new.copy(), c5

    # Phase 2: SA
    t2_end = start_time + 0.60 * budget_s
    if time.time() < t2_end and tleft() > 50:
        sa_budget = min(t2_end - time.time(), tleft() * 0.9)
        h_new, c5_new = sa_local(best_h, target, sa_budget, rng)
        if c5_new < best_c5:
            best_h, best_c5 = h_new.copy(), c5_new

    # Phase 3: SLSQP polish
    t3_end = start_time + 0.75 * budget_s
    if time.time() < t3_end and tleft() > 30:
        for beta in [400, 800, 1600, 3000]:
            if time.time() > t3_end or tleft() < 15:
                break
            h_new = slsqp_step(best_h, target, beta, dx, maxiter=4000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5

    # Phase 4: ILS with diverse perturbations
    if tleft() > 40:
        candidates = []
        for it in range(30):
            if tleft() < 25:
                break
            mode = it % 3
            if mode == 0:
                h_pert = best_h.copy()
                n_pert = rng.randint(5, 18)
                idx = rng.choice(n_pts, n_pert, replace=False)
                h_pert[idx] += rng.randn(n_pert) * rng.uniform(0.04, 0.12)
            elif mode == 1:
                h_pert = best_h.copy()
                corr = np.correlate(h_pert, 1.0 - h_pert, mode="full")
                k_max = int(np.argmax(corr))
                d = k_max - (n_pts - 1)
                n_adj = rng.randint(5, 15)
                if d > 0:
                    idx = rng.choice(n_pts - d, min(n_adj, n_pts - d), replace=False)
                    for i in idx:
                        if h_pert[i] < 0.5:
                            h_pert[i] += rng.uniform(0.03, 0.1)
                elif d < 0:
                    da = -d
                    idx = rng.choice(n_pts - da, min(n_adj, n_pts - da), replace=False)
                    for i in idx:
                        if h_pert[i + da] < 0.5:
                            h_pert[i + da] += rng.uniform(0.03, 0.1)
                else:
                    idx = rng.choice(n_pts, n_adj, replace=False)
                    for i in idx:
                        if h_pert[i] > 0.5:
                            h_pert[i] += rng.uniform(0.03, 0.1)
                        else:
                            h_pert[i] -= rng.uniform(0.03, 0.1)
            else:
                h_pert = best_h.copy()
                n_push = rng.randint(8, 25)
                idx = rng.choice(n_pts, n_push, replace=False)
                signs = rng.choice([-1, 1], n_push)
                mags = rng.uniform(0.15, 0.4, n_push)
                h_pert[idx] = np.clip(h_pert[idx] + signs * mags, 0, 1)

            h_pert = project_h(h_pert, target)
            h_s = slsqp_step(h_pert, target, 300, dx, maxiter=2000)
            c5_s = compute_c5(h_s)
            candidates.append((h_s, c5_s))
            if c5_s < best_c5:
                best_h, best_c5 = h_s.copy(), c5_s

        candidates.sort(key=lambda x: x[1])
        for h_cand, c5_cand in candidates[:4]:
            if tleft() < 15:
                break
            h_new = slsqp_step(h_cand, target, 600, dx, maxiter=3000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5

    # Phase 5: Explore different n with binary starts
    if tleft() > 50:
        for n_try in [64, 68, 72, 76, 80, 84]:
            if tleft() < 30:
                break
            t_target = n_try / 2.0
            dx_try = 2.0 / n_try
            for attempt in range(2):
                if tleft() < 20:
                    break
                h_bin = np.zeros(n_try)
                ones = rng.choice(n_try, int(t_target), replace=False)
                h_bin[ones] = 1.0
                h_bin = project_h(h_bin, t_target)
                h_opt = slsqp_step(h_bin, t_target, 200, dx_try, maxiter=2000)
                c5_try = compute_c5(h_opt)
                if tleft() > 15:
                    h_sa, c5_sa = sa_local(h_opt, t_target, min(6, tleft() * 0.2), rng)
                    if c5_sa < c5_try:
                        h_opt, c5_try = h_sa, c5_sa
                    h_opt = slsqp_step(h_opt, t_target, 500, dx_try, maxiter=2000)
                    c5_try = compute_c5(h_opt)
                if c5_try < best_c5:
                    best_h, best_c5 = h_opt.copy(), c5_try

    # Final polish
    if tleft() > 8:
        n_final = len(best_h)
        dx_final = 2.0 / n_final
        for beta in [500, 1000, 2000]:
            if tleft() < 6:
                break
            h_new = slsqp_step(best_h, n_final / 2.0, beta, dx_final, maxiter=5000)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new.copy(), c5

    best_h = project_h(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
