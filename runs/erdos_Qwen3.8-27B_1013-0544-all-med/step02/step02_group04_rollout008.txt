def optimize_one(h_init, n, rng, time_budget):
    target = n / 2.0
    dx = 2.0 / n
    h = project_h(h_init, target)
    c5 = compute_c5(h)
    # Phase A: gradient descent
    t_rem = time_budget - (time.time() - time_budget_start) if time_budget_start else time_budget
    t_rem = min(time_rem, time_budget * 0.4)
    if t_rem > 10:
        h, c5 = projected_grad_desc(h, target, dx, rng, t_rem)
    # Phase B: local search
    t_rem = time_budget - (time.time() - time_budget_start)
    t_rem = min(t_rem, time_budget * 0.35)
    if t_rem > 10:
        h, c5 = local_search(h, target, dx, rng, t_rem)
    # Phase C: SLSQP
    t_rem = time_budget - (time.time() - time_budget_start)
    if t_rem > 15:
        h, c5 = slsqp_polish(h, target, dx)
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    global time_budget_start
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    time_budget_start = time.time()

    def tl():
        return budget_s - (time.time() - start_time)

    h_best = None
    c5_best = float('inf')

    # Main: optimize from initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h70 = np.array(initial_h_values, dtype=float).copy()
        n70 = len(h70)
        target70 = n70 / 2.0
        dx70 = 2.0 / n70
        h70 = project_h(h70, target70)
        c5_70 = compute_c5(h70)
        c5_best = c5_70
        h_best = h70.copy()

        # Gradient descent from initial
        if tl() > 200:
            h70, c5_70 = projected_grad_desc(h70, target70, dx70, rng, 250)
            if c5_70 < c5_best:
                c5_best, h_best = c5_70, h70.copy()

        # Local search
        if tl() > 150:
            h70, c5_70 = local_search(h_best, target70, dx70, rng, min(tl() - 100, 200))
            if c5_70 < c5_best:
                c5_best, h_best = c5_70, h70.copy()

        # SLSQP
        if tl() > 60:
            h70, c5_70 = slsqp_polish(h_best, target70, dx70)
            if c5_70 < c5_best:
                c5_best, h_best = c5_70, h70.copy()

        # Another round of local search on improved solution
        if tl() > 80:
            h70, c5_70 = local_search(h_best, target70, dx70, rng, min(tl() - 40, 100))
            if c5_70 < c5_best:
                c5_best, h_best = c5_70, h70.copy()

    # Try other n values
    for n_try in [60, 66, 56, 80, 65]:
        if tl() < 80:
            break
        n_t = int(n_try)
        target_t = n_t / 2.0
        if target_t != int(target_t):
            continue
        target_t = int(target_t)
        dx_t = 2.0 / n_t
        h_t, c5_t = make_initial(n_t, rng)
        if c5_t < c5_best:
            c5_best, h_best = c5_t, h_t.copy()
        if tl() > 50:
            h_t, c5_t = projected_grad_desc(h_t, target_t, dx_t, rng, min(tl() - 30, 50))
            if c5_t < c5_best:
                c5_best, h_best = c5_t, h_t.copy()
        if tl() > 30:
            h_t, c5_t = local_search(h_best if c5_best < c5_t else h_t,
                                      target_t, dx_t, rng, min(tl() - 20, 25))
            if c5_t < c5_best:
                c5_best, h_best = c5_t, h_t.copy()

    # Final polish
    if tl() > 20:
        n_f = len(h_best)
        h_final, c5_final = slsqp_polish(h_best, n_f / 2.0, 2.0 / n_f)
        if c5_final < c5_best:
            h_best, c5_best = h_final, c5_final

    h_best = project_h(h_best, len(h_best) / 2.0)
    c5_best = compute_c5(h_best)
    return h_best, c5_best, len(h_best)
