**Key improvement idea**: Instead of optimizing the full h vector, notice that the problem has structure. The correlation at lag k is:
- For k >= 0: sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
- For k < 0: sum_{i=0}^{n-1-|k|} h[i+|k|] * (1-h[i])

So each correlation value is a quadratic function of h. The problem is:
minimize max_k f_k(h)
subject to 0 <= h <= 1, sum(h) = n/2

where f_k(h) = sum_i h[i]*(1-h[i+k]) (for appropriate ranges)

This is a convex-like problem (each f_k is not convex in h since it's quadratic with negative cross terms... actually h*(1-h) = h - h², so it's concave in each h individually, but the sum makes it more complex).

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad_weights(x, beta):
    m = np.max(x)
    shifted = np.clip(beta * (x - m), -700.0, 700.0)
    exps = np.exp(shifted)
    return exps / np.sum(exps)

def corr_grad_h(h, weights):
    """Compute gradient of weighted sum of correlations w.r.t. h."""
    n = len(h)
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    # weights has length 2n-1, indexed by k=0..2n-2, lag = k-(n-1)
    for k in range(len(weights)):
        w = weights[k]
        if w < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
