    if rem() > 40:
        for beta in [3000, 8000, 20000]:
            if rem() < 15:
                break
            try:
                r = minimize(obj_full, best_h, args=(n, ts, beta, 10000.0),
                             method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 8000, 'ftol': 1e-19, 'gtol': 1e-14})
                h_new = project_box_sum(r.x, ts)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 6: Light coordinate descent polish
    if rem() > 10:
        for _ in range(300):
            if rem() < 5:
                break
            improved = False
            order = np.random.permutation(n)
            for i in order:
                j = int(np.argmax(0.5 - np.abs(best_h - 0.5)))
                if j == i:
                    j = (i + 1) % n
                for delta in [0.002, -0.002, 0.005, -0.005, 0.001, -0.001]:
                    h2 = best_h.copy()
                    h2[i] = np.clip(h2[i] + delta, 0, 1)
                    h2[j] = np.clip(h2[j] - delta, 0, 1)
                    c2 = compute_c5(h2)
                    if c2 < best_c5:
                        best_h, best_c5 = h2, c2
                        improved = True
                        break
                if improved:
                    break
            if not improved:
                break

    best_h = project_box_sum(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
