            k = (i + 1) % n
        corr = compute_corr(best)
        idx_is = i - s_vals
        vis = (idx_is >= 0) & (idx_is < n)
        t1 = np.where(vis, 1.0 - best[np.clip(idx_is, 0, n - 1)], 0.0)
        idx_ip = i + s_vals
        vip = (idx_ip >= 0) & (idx_ip < n)
        t2 = np.where(vip, -best[np.clip(idx_ip, 0, n - 1)], 0.0)
        idx_ks = k - s_vals
        vks = (idx_ks >= 0) & (idx_ks < n)
        t3 = np.where(vks, -(1.0 - best[np.clip(idx_ks, 0, n - 1)]), 0.0)
        idx_kp = k + s_vals
        vkp = (idx_kp >= 0) & (idx_kp < n)
        t4 = np.where(vkp, best[np.clip(idx_kp, 0, n - 1)], 0.0)
        a = t1 + t2 + t3 + t4
        lo = max(-best[i], best[k] - 1.0)
        hi = min(1.0 - best[i], best[k])
        if lo >= hi - 1e-12:
            continue
        for _ in range(n_ternary):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            if np.max(corr + m1 * a) < np.max(corr + m2 * a):
                hi = m2
            else:
                lo = m1
        delta = (lo + hi) / 2.0
        new_h = best.copy()
        new_h[i] += delta
        new_h[k] -= delta
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = project_feasible(new_h, target)
        c = compute_c5(new_h)
        if c < best_c:
            best = new_h
            best_c = c
    return best, best_c

def slsqp(h, maxiter, beta):
    n = len(h)
    target = n / 2.0
    bounds = Bounds(np.zeros(n), np.ones(n))
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smoothmax_obj, h, jac=lambda x: smoothmax_grad(x, beta),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h, target)

def interp_to_n(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    f = interp1d(x_old, h, kind='cubic', fill_value='extrapolate',
                 assume_sorted=True)
    h_new = f(x_new)
    return project_feasible(np.clip(h_new, 0.0, 1.0), n_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(200, 0.5)

    best = h.copy()
    best_c = compute_c5(best)

    # Phase 1: Coordinate descent
    if rem() > 200:
        h1, c1 = coord_descend(h, 10000, rng)
        if c1 < best_c:
            best, best_c = h1, c1

    # Phase 2: SLSQP with increasing beta
    for beta in [80, 160, 320, 600]:
        if rem() < 40:
            break
        h2 = slsqp(best, 500, beta)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best, best_c = h2, c2

    # Phase 3: More coordinate descent
    if rem() > 120:
        h3, c3 = coord_descend(best, 15000, rng)
        if c3 < best_c:
            best, best_c = h3, c3

    # Phase 4: Multi-resolution search
    for n_try in [150, 100, 250, 300]:
        if rem() < 80:
            break
        hi = interp_to_n(best, n_try)
        h4, c4 = coord_descend(hi, 6000, rng)
        if c4 < best_c:
            best, best_c = h4, c4
        h5 = slsqp(hi, 400, 300)
        c5 = compute_c5(h5)
        if c5 < best_c:
            best, best_c = h5, c5

    # Phase 5: Perturbation and restart (multiple attempts)
    for pmag in [0.03, 0.015, 0.05, 0.01, 0.04]:
        if rem() < 60:
            break
        h_p = best + rng.randn(len(best)) * pmag
        h_p = project_feasible(h_p, len(best) / 2.0)
        h6, c6 = coord_descend(h_p, 6000, rng)
        if c6 < best_c:
            best, best_c = h6, c6
        h7 = slsqp(h_p, 300, 400)
        c7 = compute_c5(h7)
        if c7 < best_c:
            best, best_c = h7, c7

    # Phase 6: Final intensive polish
    if rem() > 50:
        h8, c8 = coord_descend(best, 15000, rng, n_ternary=35)
        if c8 < best_c:
            best, best_c = h8, c8
    if rem() > 30:
        h9 = slsqp(best, 800, 800)
        c9 = compute_c5(h9)
        if c9 < best_c:
            best, best_c = h9, c9
    if rem() > 15:
        h10, c10 = coord_descend(best, 8000, rng, n_ternary=30)
        if c10 < best_c:
            best, best_c = h10, c10

    return best, best_c, len(best)
