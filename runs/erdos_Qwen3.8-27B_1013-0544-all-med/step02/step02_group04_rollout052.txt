    if h_parent is not None:
        best_h = project_h(h_parent.copy(), n_parent / 2.0)
    else:
        best_h = np.full(n_parent, 0.5)
    best_c5 = compute_c5(best_h)
    best_n = n_parent
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Phase 1: Explore multiple n values
    n_candidates = sorted(set([n_parent, 48, 52, 56, 60, 64, 70, 72, 76, 80, 84, 88, 92, 96, 100, 108, 112, 120, 140]))
    n_restarts = [6, 5, 5, 4, 4, 4, 6, 5, 4, 4, 4, 4, 4, 4, 4, 4, 4, 3, 3]
    
    for idx, n in enumerate(n_candidates):
        if time_left() < 120:
            break
        target_sum = n / 2.0
        nr = n_restarts[idx] if idx < len(n_restarts) else 4
        
        for r in range(nr):
            if time_left() < 120:
                break
            h = make_initial(h_parent, n, target_sum, r, rng)
            h = gradient_descent(h, target_sum, n, 6000, rng)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                best_n = n
    
    # Phase 2: Intensive refinement on best
    if time_left() > 80:
        n = best_n
        target_sum = n / 2.0
        for r in range(30):
            if time_left() < 80:
                break
            scale = 0.008 * (1.0 + r / 10.0)
            h = best_h.copy() + rng.randn(n) * scale
            h = project_h(h, target_sum)
            h = gradient_descent(h, target_sum, n, 12000, rng)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Phase 3: SLSQP polish with increasing beta
    if time_left() > 40:
        n = best_n
        target_sum = n / 2.0
        dx = 2.0 / n
        bounds = Bounds(np.zeros(n), np.ones(n))
        
        for beta in [300, 800, 2000]:
            if time_left() < 25:
                break
            bx = beta
            dxl = dx
            tn = target_sum
            nn = n
            
            def obj_fn(x, _bx=bx, _dx=dxl, _tn=tn):
                corr = np.correlate(x, 1.0 - x, mode="full")
                m = np.max(corr)
                sh = np.clip(_bx * (corr - m), -60, 60)
                return (m + np.log(np.sum(np.exp(sh))) / _bx) * _dx
            
            def cfun(x, _tn=tn):
                return np.sum(x) - _tn
            
            def cjac(x, _nn=nn):
                return np.ones(_nn)
            
            constr = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
            try:
                res = minimize(obj_fn, best_h, method='SLSQP', bounds=bounds,
                              constraints=constr, options={'maxiter': 8000, 'ftol': 1e-16})
                h_new = project_h(res.x, target_sum)
                c5 = compute_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            except Exception:
                pass
    
    # Phase 4: Final targeted gradient steps
    if time_left() > 15:
        n = best_n
        target_sum = n / 2.0
        h = gradient_descent(best_h.copy(), target_sum, n, 20000, rng)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    best_h = project_h(best_h, best_n / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
