    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-10:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        nfree = free.sum()
        if nfree == 0:
            break
        h[free] += diff / nfree
    return np.clip(h, 0.0, 1.0)

def smooth_max_vals(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    s = exp_s.sum()
    weights = exp_s / s
    val = m + (max_s + np.log(s)) / beta
    return val, weights

    corr = np.correlate(h, 1.0 - h, mode="full")
    val, _ = smooth_max_vals(corr, beta)
    return val * dx

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-10:
            break
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        nf = free.sum()
        if nf == 0:
            break
        h[free] += diff / nf
    return np.clip(h, 0.0, 1.0)

def _smooth_max(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    ms = np.max(shifted)
    e = np.exp(shifted - ms)
    s = e.sum()
    weights = e / s
    val = m + (ms + np.log(s)) / beta
    return val, weights

def obj_fn(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    val, _ = _smooth_max(corr, beta)
    return val * dx

def grad_fn(h, beta, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    _, w = _smooth_max(corr, beta)
    n = len(h)
    g = np.zeros(n)
    mid = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = mid - k
        if lag >= 0:
            g[:n - lag] += wk * (1.0 - h[lag:])
            g[lag:] -= wk * h[:n - lag]
        else:
            ll = -lag
            g[ll:] += wk * (1.0 - h[:n - ll])
            g[:n - ll] -= wk * h[ll:]
    return g * dx

def c_sum(h, *_):
    return np.sum(h) - len(h) / 2.0

def c_jac(h, *_):
    return np.ones(len(h))
