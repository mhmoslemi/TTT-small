def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def project_box_sum(h, target):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h > 0.0) & (h < 1.0)
        if m.sum() == 0:
            break
        h[m] += d / m.sum()
        h = np.clip(h, 0.0, 1.0)
    return h


def grad_f_k(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        for j in range(n - k):
            g[j] += (1.0 - h[j + k])
        for j in range(k, n):
            g[j] -= h[j - k]
    else:
        kk = -k
        for j in range(n - kk):
            g[j] += (1.0 - h[j + kk])
        for j in range(kk, n):
            g[j] -= h[j - kk]
    return g


def smooth_obj(h, beta, lam, target, n):
    c = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
    m = np.max(c)
    sm = m + np.log(np.sum(np.exp(beta * (c - m)))) / beta
    return sm + lam * (h.sum() - target) ** 2
