    old = h[i]

    candidates.append(1.0 - old)  # h[i] -> 1

    crit = 0.5 - old
    if 0 <= -crit <= 1 - 0 and -crit >= -old and -crit <= 1-old:

import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    h = h.copy()
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        s = h.sum()
        if abs(s - target_sum) < 1e-14:
            break
        free = (h > 1e-14) & (h < 1.0 - 1e-14)
        if free.sum() > 0:
            h[free] += (target_sum - s) / free.sum()
        else:
            h += (target_sum - s) / len(h)
    return np.clip(h, 0.0, 1.0)

def compute_M(h):
    """Compute the coefficient matrix M[k, i] = d(c[k])/d(h[i]) linearized."""
    n = len(h)
    M = np.zeros((2*n-1, n))
    for i in range(n):
        for k in range(2*n-1):
            d = k - (n-1)
            coef = 0.0
            if 0 <= i - d < n:
                coef += (1.0 - h[i - d])
            if 0 <= i + d < n:
                coef -= h[i + d]
            M[k, i] = coef
    return M

def coord_descent_sweep(h, target_sum, max_sweeps=10):
    """Perform coordinate descent sweeps with exact-ish line search."""
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        M = compute_M(h)
        corr = np.correlate(h, 1.0 - h, mode="full")
        improved = False
        
        for i in range(n):
            # Find best delta for coordinate i
            old = h[i]
            candidates = [0.0, -old, 1.0 - old]
            
            # Add heuristic candidates based on worst lag direction
            k_star = np.argmax(corr)
            if M[k_star, i] > 0.01:
                candidates.extend([-0.05, -0.1, -0.2, -0.4])
            elif M[k_star, i] < -0.01:
                candidates.extend([0.05, 0.1, 0.2, 0.4])
            else:
                candidates.extend([-0.1, 0.1])
            
            best_obj = np.inf
            best_delta = 0.0
            
            for delta in candidates:
                if delta < -old - 1e-9 or delta > 1.0 - old + 1e-9:
                    continue
                new_h = h.copy()
                new_h[i] = old + delta
                new_h = project_h(new_h, target_sum)
                c = np.correlate(new_h, 1.0 - new_h, mode="full")
                obj = c.max()
                if obj < best_obj:
                    best_obj = obj
                    best_delta = delta
            
            if abs(best_delta) > 1e-10:
                h[i] = old + best_delta
                h = project_h(h, target_sum)
                corr = np.correlate(h, 1.0 - h, mode="full")
                M = compute_M(h)  # Update M
                new_c5 = corr.max() * dx
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    improved = True
        
        if not improved:
            break
    
    return h, best_c5
