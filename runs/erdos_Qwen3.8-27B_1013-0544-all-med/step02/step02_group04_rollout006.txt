import numpy as np
from scipy.optimize import minimize, Bounds
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def corr_arr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        diff = target_sum - s
        if abs(diff) < 1e-13:
            break
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if free.sum() == 0:
            break
        h[free] += diff / free.sum()
    return np.clip(h, 0.0, 1.0)

def smooth_max_val(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def smooth_max_weights(x, beta):
    e = np.exp(beta * (x - np.max(x)))
    return e / np.sum(e)

def grad_correlations(h, weights):
    n = len(h)
    j = np.arange(n)
    grad = np.zeros(n)
    for k in range(len(weights)):
        if weights[k] < 1e-8:
            continue
        idx = k - j
        valid = (idx >= 0) & (idx < n)
        if np.any(valid):
            grad[valid] += weights[k] * (1.0 - 2.0 * h[idx[valid]])
    return grad

def pgd_optimize(h, target_sum, n_iters, base_lr, top_k, time_fn):
    n = len(h)
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = base_lr
    momentum = np.zeros(n)
    beta_m = 0.85
    for it in range(n_iters):
        if time_fn() < 25:
            break
        corr = corr_arr(h)
        c5 = float(np.max(corr) * (2.0 / n))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        effective_k = max(1, top_k - it // max(1, n_iters // 3))
        top_idx = np.argsort(corr)[-effective_k:]
        w = np.zeros(len(corr))
        w[top_idx] = 1.0 / effective_k
        grad = grad_correlations(h, w)
        momentum = beta_m * momentum - lr * grad
        h_new = project_h(h + momentum, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new <= c5:
            h = h_new
            lr = min(lr * 1.05, 0.04)
        else:
            lr *= 0.7
            momentum *= 0.5
    return h, best_h, best_c5

def slsqp_refine(h, target_sum, n, beta, maxiter, time_fn):
    dx = 2.0 / n
    bounds = Bounds(np.zeros(n), np.ones(n))
    def obj(v):
        return smooth_max_val(corr_arr(v), beta) * dx
    def obj_grad(v):
        c = corr_arr(v)
        w = smooth_max_weights(c, beta)
        return grad_correlations(v, w) * dx
    cons = [{'type': 'eq', 'fun': lambda v: v.sum() - target_sum,
             'jac': lambda v: np.ones(n)}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_h(res.x, target_sum)
        return h_new
    except Exception:
        return h

def random_restart(h_base, n, target_sum, rng, strategy):
    h = h_base.copy()
    s = strategy % 5
    if s == 0:
        h += rng.randn(n) * 0.03
    elif s == 1:
        period = rng.randint(4, n // 3 + 1)
        t = np.arange(n)
        h = 0.5 + 0.2 * np.sin(2 * np.pi * (t + rng.uniform(0, period)) / period)
    elif s == 2:
        nb = rng.randint(3, 8)
        bs = n // nb
        for b in range(nb):
            st = b * bs
            en = st + bs if b < nb - 1 else n
            h[st:en] = 0.5 + rng.randn() * 0.15
    elif s == 3:
        h = h_base[::-1] + rng.randn(n) * 0.02
    else:
        idx = rng.choice(n, n // 3, replace=False)
        h[idx] += rng.randn(len(idx)) * 0.05
    return project_h(h, target_sum)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    def tl():
        return budget_s - (time.time() - start_time)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.full(n_points, 0.5)

    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    best_h = project_h(h_init, target_sum)
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick PGD from initial (60s)
    if tl() > 200:
        _, ph, pc5 = pgd_optimize(best_h, target_sum, 2000, 0.015, 5, tl)
        if pc5 < best_c5:
            best_h, best_c5 = ph, pc5

    # Phase 2: SLSQP with increasing beta (150s)
    for beta in [100, 300, 800, 2000, 5000]:
        if tl() < 100:
            break
        h_new = slsqp_refine(best_h, target_sum, n_points, beta, 3000, tl)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_h, best_c5 = h_new, c5

    # Phase 3: Multi-restart local search (250s)
    attempt = 0
    while tl() > 80:
        attempt += 1
        h0 = random_restart(best_h, n_points, target_sum, rng, attempt)
        # Quick PGD
        if tl() > 60:
            h0, ph, pc5 = pgd_optimize(h0, target_sum, 800, 0.01, 3, tl)
            if pc5 > best_c5 * 1.02:
                continue
            h0 = ph
        # SLSQP polish
        if tl() > 40:
            h_new = slsqp_refine(h0, target_sum, n_points, 2000, 2000, tl)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    # Phase 4: Fine-tune with exact gradient on worst lag (100s)
    if tl() > 30:
        h = best_h.copy()
        lr = 0.003
        momentum = np.zeros(n_points)
        for it in range(5000):
            if tl() < 15:
                break
            corr = corr_arr(h)
            c5 = float(np.max(corr) * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            k_star = np.argmax(corr)
            j = np.arange(n_points)
            idx = k_star - j
            valid = (idx >= 0) & (idx < n_points)
            g = np.zeros(n_points)
            g[valid] = 1.0 - 2.0 * h[idx[valid]]
            momentum = 0.8 * momentum - lr * g
            h_new = project_h(h + momentum, target_sum)
            if compute_c5(h_new) <= c5:
                h = h_new
                lr = min(lr * 1.02, 0.01)
            else:
                lr *= 0.8
                momentum *= 0.3

    # Phase 5: Final SLSQP polish
    if tl() > 15:
        for beta in [10000, 20000]:
            if tl() < 10:
                break
            h_new = slsqp_refine(best_h, target_sum, n_points, beta, 5000, tl)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_h, best_c5 = h_new, c5

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_points
