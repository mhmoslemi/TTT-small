                pop[i][st:en] = rng.uniform(0.15, 0.85)
        else:
            k = rng.randint(2, 6)
            t = np.linspace(0, 2 * np.pi * k, n)
            pop[i] = 0.5 + 0.2 * np.cos(t + rng.uniform(0, 6.28))
        pop[i] = project_h(pop[i], target_sum)
    return pop


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    dx = 2.0 / n
    target_sum = n / 2.0
    best_h = project_h(h_init.copy(), target_sum)
    best_c5 = compute_c5(best_h)

    def time_left():
        return budget_s - (time.time() - start_time)

    bounds = Bounds(np.zeros(n), np.ones(n))
    constraints = [{'type': 'eq', 'fun': lambda x: sum_constraint(x, target_sum),
                    'jac': lambda x: sum_jacobian(x)}]

    # Phase 1: Brief DE for exploration
    pop_size = 30
    pop = de_population(n, target_sum, best_h, pop_size, rng)
    F, CR = 0.5, 0.7
    fitness = np.array([compute_c5(pop[i]) for i in range(pop_size)])
    max_gen = 250
    for gen in range(max_gen):
        if time_left() < 120:
            break
        for i in range(pop_size):
            if time_left() < 120:
                break
            idxs = [x for x in range(pop_size) if x != i]
            a, b, c = rng.choice(idxs, 3, replace=False)
            v = pop[a] + F * (pop[b] - pop[c])
            mask = rng.random(n) <= CR
            mask[rng.randint(n)] = True
            u = np.where(mask, v, pop[i])
            u = project_h(u, target_sum)
            fu = compute_c5(u)
            if fu <= fitness[i]:
                pop[i] = u
                fitness[i] = fu
        bi = np.argmin(fitness)
        if fitness[bi] < best_c5:
            best_c5 = fitness[bi]
            best_h = pop[bi].copy()

    # Phase 2: SDCA-style dual weighting optimization
    if time_left() > 80:
        sdca_beta = 80.0
        for it in range(60):
            if time_left() < 80:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            alpha = np.exp(sdca_beta * (corr - corr.max()))
            alpha_sum = alpha.sum()
            if alpha_sum > 0:
                alpha = alpha / alpha_sum
            else:
                alpha = np.ones(len(corr)) / len(corr)
            try:
                res = minimize(weighted_corr_obj, best_h, args=(alpha, dx),
                               method='SLSQP', bounds=bounds, constraints=constraints,
                               options={'maxiter': 1500, 'ftol': 1e-14})
                h_new = project_h(res.x, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            except Exception:
                pass
            # Increase sdca_beta gradually
            sdca_beta *= 1.05

    # Phase 3: Active lag cycling
    if time_left() > 60:
        for cycle in range(200):
            if time_left() < 60:
                break
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            # Target the top few active lags
            top_idx = np.argsort(corr)[-3:]
            for ai in top_idx:
                if time_left() < 60:
                    break
                lag = ai - (n - 1)
                h_new = subgrad_step(best_h, lag, n, target_sum, 0.004)
                # Quick SLSQP polish
                try:
                    res = minimize(lambda x: smooth_max_obj(x, 200, dx), h_new,
                                   method='SLSQP', bounds=bounds, constraints=constraints,
                                   options={'maxiter': 500, 'ftol': 1e-13})
                    h_pol = project_h(res.x, target_sum)
                    c5_pol = compute_c5(h_pol)
                    if c5_pol < best_c5:
                        best_c5 = c5_pol
                        best_h = h_pol
                except Exception:
                    pass

    # Phase 4: SLSQP with increasing smooth max beta
    for beta in [100, 300, 800, 2000]:
        if time_left() < 40:
            break
        try:
            res = minimize(lambda x: smooth_max_obj(x, beta, dx), best_h,
                           method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 5000, 'ftol': 1e-15})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Phase 5: Iterated local search with diverse perturbations
    attempt = 0
    while time_left() > 30:
        attempt += 1
        h_s = best_h.copy()
        strat = attempt % 6
        if strat == 0:
            scale = 0.002 + 0.012 * np.exp(-attempt / 25.0)
            h_s += rng.randn(n) * scale
        elif strat == 1:
            corr = np.correlate(h_s, 1.0 - h_s, mode="full")
            active = np.argmax(corr)
            lag = active - (n - 1)
            h_s = subgrad_step(h_s, lag, n, target_sum, 0.006)
        elif strat == 2:
            i, j = rng.choice(n, 2, replace=False)
            d = 0.025 * rng.randn()
            h_s[i] += d
            h_s[j] -= d
        elif strat == 3:
            blen = rng.randint(3, 7)
            bs = rng.randint(0, max(1, n - blen))
            h_s[bs:bs + blen] += rng.randn(blen) * 0.012
        elif strat == 4:
            # Smoothing
            h_s = 0.9 * h_s + 0.1 * np.roll(h_s, 1)
        else:
            # Anti-smoothing (sharpen)
            h_s = 1.1 * h_s - 0.1 * np.roll(h_s, 1)
            h_s = project_h(h_s, target_sum)

        h_s = project_h(h_s, target_sum)
        beta_l = 400 if attempt % 2 == 0 else 150
        try:
            res = minimize(lambda x: smooth_max_obj(x, beta_l, dx), h_s,
                           method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 2000, 'ftol': 1e-14})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    # Final high-precision polish
    for beta in [3000, 8000]:
        if time_left() < 10:
            break
        try:
            res = minimize(lambda x: smooth_max_obj(x, beta, dx), best_h,
                           method='SLSQP', bounds=bounds, constraints=constraints,
                           options={'maxiter': 20000, 'ftol': 1e-16})
            h_new = project_h(res.x, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
        except Exception:
            pass

    best_h = project_h(best_h, target_sum)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
