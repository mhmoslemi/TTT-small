            g[:n - lag] -= w * h[lag:]
        else:
            llag = -lag
            g[:n - llag] += w * (1.0 - h[llag:])
            g[llag:] -= w * h[:n - llag]
    return g

def smooth_max_val(h, beta):

    m = np.max(c)
    e = beta * (c - m)
    return (m + np.log(np.sum(np.exp(e))) / beta) * (2.0 / len(h))

def project_feasible(h, target):
    """Project h onto {sum(h)=target, 0<=h<=1}."""
    h = h.astype(float).copy()
    n = len(h)
    # f(t) = sum(clip(h+t, 0, 1)) is monotonically increasing
    # Find t such that f(t) = target
    lo = target / n - h.max() - 0.1
    hi = target / n - h.min() + 0.1
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def corr_raw(h):
    return np.correlate(h, 1.0 - h, mode="full")


def smooth_max_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    c = corr_raw(h)
    m = np.max(c)
    e = beta * (c - m)
    e = np.exp(e - np.max(e))
    s = e.sum()
    p = e / s
    g = np.zeros(n)

        lag = i - (n - 1)
