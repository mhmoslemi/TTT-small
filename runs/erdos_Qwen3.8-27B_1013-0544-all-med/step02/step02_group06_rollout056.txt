    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-15:
            continue
        k = j - (n - 1)
        gw = w[j] * dx
        grad += gw * grad_lag(h, k)
    return grad

def prox_step(h, target, beta, lr):
    g = grad_smooth_max(h, beta)
    return project_feasible(h - lr * g, target)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n = len(h_init)
    else:
        n = 70
        h_init = np.full(n, 0.5)

    target = n / 2.0
    best_h = project_feasible(h_init, target)
    best_c5 = compute_c5(best_h)
    bnds = [(0.0, 1.0)] * n

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: Coarse-to-fine smooth max optimization
    for beta in [30, 80, 200, 500, 1000]:
        if tl() < 50:
            break
        def obj1(h):
            c = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(c, beta) * (2.0 / n) + 200.0 * (h.sum() - target)**2
        try:
            r = minimize(obj1, best_h, method='L-BFGS-B', bounds=bnds,
                         options={'maxiter': 400, 'ftol': 1e-15})
            h_new = project_feasible(r.x, target)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    # Phase 2: Proximal gradient with backtracking
    for beta in [200, 500, 1000]:
        if tl() < 30:
            break
        lr = 0.05
        for _ in range(100):
            if tl() < 15:
                break
            h_new = prox_step(best_h, target, beta, lr)
            f_new = smooth_max(np.correlate(h_new, 1.0 - h_new, mode="full"), beta) * (2.0/n)
            f_old = smooth_max(np.correlate(best_h, 1.0 - best_h, mode="full"), beta) * (2.0/n)
            if f_new < f_old - 1e-16:
                best_h = h_new
                c = compute_c5(best_h)
                if c < best_c5:
                    best_c5 = c
                lr = min(lr * 1.15, 0.4)
            else:
                lr *= 0.6

    # Phase 3: Iterative worst-lag minimization
    for beta in [300, 800]:
        if tl() < 40:
            break
        for _ in range(30):
            if tl() < 15:
                break
            corr = np.correlate(best_h, 1.0 - h_init if False else best_h, mode="full")
            # Actually recompute properly
            corr = np.correlate(best_h, 1.0 - best_h, mode="full")
            k_star = int(np.argmax(corr)) - (n - 1)
            # Minimize f_{k_star} directly
            def worst_lag_obj(h, ks=k_star):
                return float(grad_lag(h, ks).dot(h)) + h.sum()  # = sum h[i](1-h[i+ks])
            # Actually: f_k(h) = sum h[i] - sum h[i]h[i+k], gradient = grad_lag(h,k)
            # Let's just use the gradient in a local L-BFGS
            try:
                r = minimize(worst_lag_obj, best_h, method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 100, 'ftol': 1e-14})
                h_new = project_feasible(r.x, target)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 4: Diverse restarts
    n_restarts = 0
    while tl() > 80:
        n_restarts += 1
        h = best_h.copy()
        s = n_restarts % 7

        if s == 0:
            sc = max(0.003, 0.05 * np.exp(-n_restarts / 15.0))
            h += np.random.randn(n) * sc
        elif s == 1:
            # Target worst lag with subgradient
            corr = np.correlate(h, 1.0 - h, mode="full")
            k_star = int(np.argmax(corr)) - (n - 1)
            h -= 0.02 * grad_lag(h, k_star)
            h += np.random.randn(n) * 0.003
        elif s == 2:
            # Block perturbation
            bl = np.random.randint(4, max(5, n // 6))
            bs = np.random.randint(0, n - bl)
            h[bs:bs+bl] = 0.5 + 0.35 * np.random.randn(bl)
        elif s == 3:
            # Sinusoidal
            w = np.random.uniform(1, 8) * np.pi
            ph = np.random.uniform(0, 2 * np.pi)
            x = np.linspace(0, 2.0, n, endpoint=False)
            h += 0.025 * np.sin(w * x + ph)
        elif s == 4:
            # Multi-worst-lag subgradient
            corr = np.correlate(h, 1.0 - h, mode="full")
            top_idx = np.argsort(corr)[-4:]
            for kk in top_idx:
                ksh = int(kk) - (n - 1)
                h -= 0.008 * grad_lag(h, ksh)
        elif s == 5:
            # Complement symmetry: try 1-best_h
            h = 1.0 - best_h
            h += np.random.randn(n) * 0.02
        else:
            # Binary-pull: push values toward 0 or 1
            mask = h > 0.5
            h[mask] = np.minimum(1.0, h[mask] + 0.05)
            h[~mask] = np.maximum(0.0, h[~mask] - 0.05)

        h = project_feasible(h, target)

        # Quick polish
        for beta_p in [300, 800]:
            if tl() < 20:
                break
            def obj_p(h, bp=beta_p):
                c = np.correlate(h, 1.0 - h, mode="full")
                return smooth_max(c, bp) * (2.0 / n) + 500.0 * (h.sum() - target)**2
            try:
                r = minimize(obj_p, h, method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 300, 'ftol': 1e-15})
                h_new = project_feasible(r.x, target)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
                    h = h_new
            except Exception:
                pass

    # Phase 5: Final high-precision polish
    if tl() > 20:
        for beta in [1500, 3000, 5000, 8000]:
            if tl() < 10:
                break
            def obj_final(h, bp=beta):
                c = np.correlate(h, 1.0 - h, mode="full")
                return smooth_max(c, bp) * (2.0 / n) + 2000.0 * (h.sum() - target)**2
            try:
                r = minimize(obj_final, best_h, method='L-BFGS-B', bounds=bnds,
                             options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-12})
                h_new = project_feasible(r.x, target)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_h, best_c5 = h_new, c
            except Exception:
                pass

    # Phase 6: SLSQP with equality constraint for final squeeze
    if tl() > 15:
        def obj_sl(h):
            c = np.correlate(h, 1.0 - h, mode="full")
            return smooth_max(c, 3000.0) * (2.0 / n)
        cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                 'jac': lambda x: np.ones(n)}]
        try:
            r = minimize(obj_sl, best_h, method='SLSQP', bounds=bnds,
                         constraints=cons, options={'maxiter': 800, 'ftol': 1e-16})
            h_new = project_feasible(r.x, target)
            c = compute_c5(h_new)
            if c < best_c5:
                best_h, best_c5 = h_new, c
        except Exception:
            pass

    best_h = project_feasible(best_h, target)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
