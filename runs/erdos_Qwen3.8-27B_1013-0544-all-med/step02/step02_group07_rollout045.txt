def subgradient_c(h, n):
    """Compute subgradient of max_k c_k at h."""
    corr = np.correlate(h, 1.0 - h, mode="full")
    max_val = corr.max()
    active = (corr >= max_val - 1e-10)
    
    grad = np.zeros(n)
    for k_idx in np.where(active)[0]:
        d = k_idx - (n - 1)  # lag
        c = 1.0 / active.sum()  # convex combination weight
        for j in range(n):
            g = 0.0
            if d >= 0:
                if j < n - d:
                    g += (1.0 - h[j + d])
                if j >= d:
                    g -= h[j - d]
            else:
                ad = -d
                if j >= ad:
                    g += (1.0 - h[j - ad])
                if j + ad < n:
                    g -= h[j + ad]
            if d == 0:
                g = 1.0 - 2.0 * h[j]
            grad[j] += c * g
    return grad

        # Second index: h[j] for j = 0..n-ad-1: g[j] -= h[j+ad]
        g[:n-ad] -= h[ad:]
    return g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max(x, beta):
    m = np.max(x)
    return m + np.log(np.sum(np.exp(beta * (x - m)))) / beta

def project_h(h, target_sum):
    """Water-filling Euclidean projection onto {0<=x<=1, sum(x)=target_sum}."""
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(h - lam, 0.0, 1.0)
    diff = target_sum - x.sum()
    if abs(diff) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.sum() > 0:
            x[free] += diff / free.sum()
    return np.clip(x, 0.0, 1.0)
