```python
def eval_swap(h, c, i, j, n):
    """Evaluate the effect of setting h[i]=0 and h[j]=1 on the correlation vector."""
    # c[k] for k in -(n-1)..(n-1), stored as c_full[k+n-1]
    # Δc[k] = h[i+k] + h[i-k] - h[j+k] - h[j-k] (treating out-of-bounds as 0)
    delta = np.zeros(2*n - 1)
    # h[i+k] for k in -(n-1)..(n-1): valid when 0 <= i+k < n
    # h[i-k] for k in -(n-1)..(n-1): valid when 0 <= i-k < n, i.e., k in [i-n+1, i]
    # etc.
    
    # Let's index c_full with offset: c_full[k + n - 1] = c[k]
    # For h[i+k]: k ranges such that 0 <= i+k < n, i.e., -i <= k <= n-1-i
    # In c_full index: k + n - 1 ranges from n-1-i to 2n-2-i... 
    
    # This is getting complex. Let me just do it directly with numpy.
    shifts = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    # h[i+k]: valid for k where 0 <= i+k < n
    idx1 = (i + shifts)
    mask1 = (idx1 >= 0) & (idx1 < n)
    delta[mask1] += h[idx1[mask1]]
    
    # h[i-k]: valid for k where 0 <= i-k < n
    idx2 = (i - shifts)
    mask2 = (idx2 >= 0) & (idx2 < n)
    delta[mask2] += h[idx2[mask2]]
    
    # -h[j+k]
    idx3 = (j + shifts)
    mask3 = (idx3 >= 0) & (idx3 < n)
    delta[mask3] -= h[idx3[mask3]]
    
    # -h[j-k]
    idx4 = (j - shifts)
    mask4 = (idx4 >= 0) & (idx4 < n)
    delta[mask4] -= h[idx4[mask4]]
    
    new_c = c + delta
    return np.max(new_c), delta
```

```python
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr_full(h)) * 2.0 / n)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()

    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        ones = np.where(h == 1)[0]

    elif n_ones < target:

    shifts = np.arange(-(n-1), n)

    for i in range(n):
        a = i + shifts
        m = (a >= 0) & (a < n)
        b = i - shifts

        V[i, m] += h[a[m]]
        V[i, m2] += h[b[m2]]

    h = make_binary(h0, n)
    corr = compute_corr_full(h).astype(float)
    dx = 2.0 / n

    best_h, best_max = h.copy(), cur_max
    rng = np.random.RandomState(seed)

        frac = (time.time() - t0) / max_time
        T = max(1e-7, 0.01 * (1 - frac)**2)
        
        V = get_v_matrix(h, n)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

            for j in zeros:

            corr += V[best_i] - V[best_j]
            cur_max += best_gain
            stall = 0

    def objective(x):

        c = compute_corr_full(hx)
        # Smoothed max with log-sum-exp

        c_max = np.max(c)
        s = np.sum(np.exp((c - c_max) / beta))
        return (c_max + beta * np.log(s)) * dx + 0.01 * (np.sum(hx) - n/2)**2
```

Let me also think about the continuous refinement more carefully. The log-sum-exp approach:

f(h) = β * log(∑_k exp(c(k)/β)) * dx + penalty

As β → 0, this approaches max(c(k)) * dx = C₅.

For the gradient, I can use:
∂f/∂h[m] = dx * ∑_k p(k) * ∂c(k)/∂h[m]

where p(k) = exp(c(k)/β) / ∑_k' exp(c(k')/β)

And ∂c(k)/∂h[m] = (1-h[m-k]) * [0≤m-k<n] - h[m+k] * [0≤m+k<n]

This can be computed efficiently:
- For each m, the sum over k involves at most 2n-1 terms
- But most terms are zero (when m±k is out of bounds)
- For a given m, the non-zero terms are k such that 0≤m-k<n (i.e., k in [m-n+1, m]) and 0≤m+k<n (i.e., k in [-m, n-1-m])
- The overlap is k in [max(m-n+1, -m), min(m, n-1-m)]

For n=112 and m=56: k in [max(-55, -56), min(56, 55)] = [-55, 55], so 111 terms.

For each m, the gradient is a weighted sum of 111 values. Computing all 112 gradients is O(n * n) = O(n²) ≈ 12K. Very fast.

But implementing this analytically is complex. Let me just use numerical gradients (finite differences) for the continuous optimization. With n=112, a forward difference gradient requires 112 function evaluations, each O(n) = O(112). Total: O(n²) = O(12K) per gradient step. L-BFGS-B might do 100-500 gradient evaluations, so total O(1M - 6M). Very fast.

Actually, scipy's L-BFGS-B can compute numerical gradients internally. Let me just use that.

```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr_full(h)) * 2.0 / n)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h.astype(float)

def get_v_matrix(h, n):
    shifts = np.arange(-(n-1), n)
    V = np.zeros((n, 2*n-1))
    for i in range(n):
        a = i + shifts
        m1 = (a >= 0) & (a < n)
        b = i - shifts
        m2 = (b >= 0) & (b < n)
        if np.any(m1): V[i, m1] += h[a[m1]]
        if np.any(m2): V[i, m2] += h[b[m2]]
    return V

def binary_sa_search(h0, max_time, seed=42, double_swaps=True):
    n = len(h0)
    h = make_binary(h0, n)
    corr = compute_corr_full(h).astype(float)
    dx = 2.0 / n
    cur_max = float(np.max(corr))
    best_h, best_max = h.copy(), cur_max
    rng = np.random.RandomState(seed)
    t0 = time.time()
    stall = 0
    max_stall = 8
    
    while time.time() - t0 < max_time:
        frac = (time.time() - t0) / max_time
        T = max(1e-7, 0.012 * (1 - frac)**2.5)
        
        V = get_v_matrix(h, n)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Single swap search
        best_gain, best_i, best_j = -np.inf, -1, -1
        for i in ones:

        improved = False
        if best_gain > 1e-12:

            corr += V[best_i] - V[best_j]
            cur_max += best_gain
            stall = 0
            improved = True
            if cur_max < best_max:

        elif double_swaps and stall < 3 and frac < 0.7:

                nm = float(np.max(corr + d))
                g = cur_max - nm
                if g > 1e-12:
                    h[i1]=0; h[i2]=0; h[j1]=1; h[j2]=1
                    corr += d
                    cur_max += g
                    stall = 0
                    improved = True

                    break
        
        if not improved:
            stall += 1
            # Simulated annealing kick
            if stall >= max_stall:
                # Make multiple random moves
                for _ in range(rng.randint(2, 6)):
                    i = int(rng.choice(ones)); j = int(rng.choice(zeros))
                    h[i] = 0; h[j] = 1
                    V_new = get_v_matrix(h, n)  # This is wrong, we already changed h
                    # Actually just recompute
                corr = compute_corr_full(h).astype(float)
                cur_max = float(np.max(corr))
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                stall = 0
            elif T > 1e-6:
                i = int(rng.choice(ones)); j = int(rng.choice(zeros))
                d = V[i] - V[j]
                nm = float(np.max(corr + d))
                d_obj = nm - cur_max
                if d_obj < 0 or rng.random() < np.exp(-d_obj / max(T, 1e-10)):
                    h[i] = 0; h[j] = 1
                    corr += d
                    cur_max = nm
                    if cur_max < best_max:
                        best_max, best_h = cur_max, h.copy()
                    stall = 0
            else:
                break
    
    return best_h, best_max * dx

def continuous_refine(h0, max_time, seed=42):
    n = len(h0)
    h = np.asarray(h0, dtype=float).copy()
    dx = 2.0 / n
    t0 = time.time()
    
    beta = 0.005
    def objective(x):
        hx = np.clip(x, 0.0, 1.0)
        c = compute_corr_full(hx)
        c_max = np.max(c)
        s = np.sum(np.exp((c - c_max) / beta))
        pen = 0.05 * (np.sum(hx) - n / 2.0) ** 2
        return (c_max + beta * np.log(s)) * dx + pen
    
    results = []
    while time.time() - t0 < max_time:
        res = minimize(objective, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 100, 'ftol': 1e-10})
        h = np.clip(res.x, 0.0, 1.0)
        results.append(compute_c5(h))
        if res.fun - min(results) < 1e-10:
            break
        # Slightly perturb to restart
        h += 0.01 * np.random.RandomState(seed + len(results)).randn(n)
        h = np.clip(h, 0.0, 1.0)
    
    # Enforce sum constraint
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        total = np.sum(room)
        if total > 1e-10:
            h += room * diff / total
        h = np.clip(h, 0.0, 1.0)
    
    return h, compute_c5(h)

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    def rem():
        return max(0.0, budget_s - (time.time() - t0))
    
    h_init = np.array(initial_h_values, dtype=float)

    best_h, best_c5, best_n = h1.copy(), c5_1, n_init
    
    # Phase 2: Try different structured starts
    if rem() > 60:
        # Sinusoidal start
        x = np.linspace(0, 2, n_init, endpoint=False)
        h_sin = (0.5 + 0.5 * np.sin(2 * np.pi * 3 * x / 2.0)).astype(float)
        h_sin_b = make_binary(h_sin, n_init)
        hs, cs = binary_sa_search(h_sin_b, min(60.0, rem() * 0.1), seed=seed + 100)
        if cs < best_c5:
            best_h, best_c5, best_n = hs.copy(), cs, n_init
    
    # Phase 3: Multi-resolution
    for n_try in [96, 128, 160]:
        if rem() < 60:
            break
        h_interp = interpolate_h(best_h, n_try)
        ts = min(50.0, rem() * 0.1)
        hs, cs = binary_sa_search(h_interp, ts, seed=seed + 1000 + n_try)
        if cs < best_c5:
            best_h, best_c5, best_n = hs.copy(), cs, n_try
    
    # Phase 4: Continuous refinement
    if rem() > 40:
        hc, cc = continuous_refine(best_h, min(80.0, rem() * 0.15), seed=seed + 5000)
        if cc < best_c5:
            best_h, best_c5, best_n = hc.copy(), cc, len(hc)
    
    # Phase 5: Final binary refinement from best
    if rem() > 30:
        hb = make_binary(best_h, len(best_h))
        hsb, csb = binary_sa_search(hb, min(rem() * 0.5, 200.0), seed=seed + 9000)
        if csb < best_c5:
            best_h, best_c5, best_n = hsb.copy(), csb, len(hsb)
    
    # Final continuous polish
    if rem() > 15 and not np.allclose(best_h, np.round(best_h)):
        hc2, cc2 = continuous_refine(best_h, min(30.0, rem() * 0.4), seed=seed + 9999)
        if cc2 < best_c5:
            best_h, best_c5, best_n = hc2.copy(), cc2, len(hc2)
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
```
