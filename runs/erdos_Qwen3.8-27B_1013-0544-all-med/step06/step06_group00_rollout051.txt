import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            h += d / len(h)
            h = np.clip(h, 0.0, 1.0)
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.exp(temp * (corr - m))
    return (m + np.log(np.sum(s)) / temp) * dx

def slsqp_opt(h, temp, maxiter=200):
    n = len(h)
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
             'jac': lambda x: np.ones(n)}]
    res = minimize(soft_obj, h, args=(temp,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, ts)

def to_binary(h):
    n = len(h)
    ts = int(round(n / 2))
    idx = np.argsort(h)[::-1][:ts]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

    n = len(h_start)
    b = to_binary(h_start).copy()
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    t = t_init
    no_improve = 0
    patience = max_iters // 4
    for it in range(max_iters):
        i, j = np.random.randint(0, n, 2)
        if i == j or b[i] == b[j]:
            continue
        b[i], b[j] = b[j], b[i]
        new_c5 = compute_c5(b)
        delta = new_c5 - curr_c5
        if delta < 0 or (t > 1e-12 and np.random.random() < np.exp(-delta / t)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            b[i], b[j] = b[j], b[i]
            no_improve += 1
        t = max(t * t_decay, t_min)
        if no_improve > patience:
            break
    return best_b, best_c5

def continuous_sa(h_start, max_iters=30000, t_init=0.01, t_min=0.001, t_decay=0.9998):
    n = len(h_start)
    ts = n / 2.0
    h = h_start.copy()
    curr_c5 = compute_c5(h)
    best_c5 = curr_c5
    best_h = h.copy()
    t = t_init
    no_improve = 0
    patience = max_iters // 3
    for it in range(max_iters):
        j = np.random.randint(n)
        old_val = h[j]
        h[j] = np.random.random()
        h = project_feasible(h, ts)
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5
        if delta < 0 or (t > 1e-12 and np.random.random() < np.exp(-delta / t)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[j] = old_val
            h = project_feasible(h, ts)
            no_improve += 1
        t = max(t * t_decay, t_min)
        if no_improve > patience:
            break
    return best_h, best_c5

def coord_descent(h, n_levels=15, max_sweeps=15):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h.copy()
            best_c = c5
            best_h = orig.copy()
            for v in np.linspace(0.0, 1.0, n_levels):
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            if best_c < c5 - 1e-14:
                h = best_h
                c5 = best_c
                improved = True
            else:
                h = orig
        if not improved:
            break
    return h, c5
