    return h

def thue_morse(n, seed=0):
    h = np.zeros(n, dtype=float)
    for i in range(1, n):
        h[i] = h[i >> 1] ^ (i & 1)
    return make_binary(h, n, seed)

def random_binary(n, seed=42):
    rng = np.random.RandomState(seed)
    h = (rng.rand(n) < 0.5).astype(float)
    return make_binary(h, n, seed)

def interp_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h_old)

def local_search_binary(h0, max_time=60.0, seed=42, n_2swap=200):
    n = len(h0)
    h = make_binary(h0, n, seed)
    current_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    
    while time.time() - start < max_time:
        best_gain = 1e-14
        best_move = None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                nc = compute_c5(h)
                gain = current_c5 - nc
                if gain > best_gain:
                    best_gain = gain; best_move = (i, j)
                h[i] = 1.0; h[j] = 0.0
        
        if best_gain <= 1e-14 and n <= 160:
            no = len(ones); nz = len(zeros)
            for _ in range(n_2swap):
                if time.time() - start > max_time:
                    break
                a, b = rng.choice(no, 2, replace=False)
                c, d = rng.choice(nz, 2, replace=False)
                i1, i2 = ones[a], ones[b]
                j1, j2 = zeros[c], zeros[d]
                h[i1]=0; h[i2]=0; h[j1]=1; h[j2]=1
                nc = compute_c5(h)
                gain = current_c5 - nc
                if gain > best_gain:
                    best_gain = gain; best_move = (i1, i2, j1, j2)
                h[i1]=1; h[i2]=1; h[j1]=0; h[j2]=0
        
        if best_move is not None:
            if len(best_move) == 2:
                h[best_move[0]] = 0.0; h[best_move[1]] = 1.0
            else:
                h[best_move[0]]=0; h[best_move[1]]=0; h[best_move[2]]=1; h[best_move[3]]=1
            current_c5 = compute_c5(h)
        else:
            break
    return h, current_c5

def multi_sa(h0, max_time=120.0, seed=42, n_pos=4):
    n = len(h0)
    h = h0.astype(float).copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    
    while time.time() - start < max_time:
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = 0.15 * (1.0 - frac) ** 1.5 + 0.001
        k = n_pos if rng.random() < 0.5 else n_pos + 2
        
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) < k or len(zeros) < k:
            break
        oi = rng.choice(len(ones), k, replace=False)
        zi = rng.choice(len(zeros), k, replace=False)
        positions = np.concatenate([ones[oi], zeros[zi]])
        
        for p in positions:
            if h[p] > 0.5:
                h[p] = 0.0
            else:
                h[p] = 1.0
        h = make_binary(h, n, seed=int(rng.randint(100000)))
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5; best_h = h.copy()
        else:
            h = best_h.copy() if rng.random() < 0.3 else h
            current_c5 = compute_c5(h)
    return best_h, best_c5

def lns_search(h0, max_time=120.0, seed=42):
    n = len(h0)
    best_h = make_binary(h0, n, seed).copy()
    best_c5 = compute_c5(best_h)
    rng = np.random.RandomState(seed)
    start = time.time()
    pert = 8
    
    while time.time() - start < max_time:
        h_pert = best_h.copy()
        ones = np.where(h_pert == 1)[0]
        zeros = np.where(h_pert == 0)[0]
        for _ in range(pert):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_pert[i] = 0.0; h_pert[j] = 1.0
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
        
        t_alloc = min(8.0, (max_time - (time.time() - start)) * 0.3)
        if t_alloc < 1:
            break
        h_ls, c5_ls = local_search_binary(h_pert, max_time=t_alloc, seed=int(rng.randint(99999)), n_2swap=50)
        if c5_ls < best_c5 - 1e-12:
            best_c5 = c5_ls; best_h = h_ls.copy()
            pert = max(4, pert - 1)
        else:
            pert = min(n // 3, pert + 2)
    return best_h, best_c5

def cont_refine(h0, max_time=40.0, seed=42):
    n = len(h0)
    h = normalize_continuous(h0.astype(float).copy(), n)
    current_c5 = compute_c5(h)
    best_h = h.copy(); best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    lr = 0.05
    
    while time.time() - start < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        lag = max_idx - (n - 1)
        grad = np.zeros(n)
        for i in range(n):
            if 0 <= i + lag < n:
                grad[i] += (1.0 - h[i + lag])
            if 0 <= i - lag < n:
                grad[i] += h[i - lag]
        h_new = np.clip(h - lr * grad, 0, 1)
        h_new = normalize_continuous(h_new, n)
        nc = compute_c5(h_new)
        if nc < current_c5 - 1e-15:
            current_c5 = nc; h = h_new
            if nc < best_c5:
                best_c5 = nc; best_h = h.copy()
            lr = max(lr * 0.97, 0.001)
        else:
            lr *= 0.7
            if lr < 1e-5:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    
    candidates = []
    candidates.append(make_binary(h_init, n_init, 0))
    candidates.append(thue_morse(128, 1))
    candidates.append(thue_morse(112, 2))
    candidates.append(thue_morse(96, 3))
    rng_m = np.random.RandomState(seed)
    for n_try in [100, 112, 120, 128, 140, 156]:
        if n_try % 2 == 0:
            candidates.append(random_binary(n_try, int(rng_m.randint(99999))))
    
    best_c5 = float('inf')
    best_h = None
    
    # Phase 1: local search each candidate
    for ci in range(len(candidates)):
        if budget_s - (time.time() - t0) < 25:
            break
        h_c = candidates[ci]
        t_a = min(15.0, (budget_s - (time.time() - t0)) * 0.08)
        if t_a < 2:
            break
        h_ls, c5_ls = local_search_binary(h_c, max_time=t_a, seed=seed + ci * 7, n_2swap=100)
        if c5_ls < best_c5:
            best_c5 = c5_ls; best_h = h_ls.copy()
    
    if best_h is None:
        best_h = make_binary(h_init, n_init, 0)
        best_c5 = compute_c5(best_h)
    
    # Phase 2: SA with multi-position moves
    if budget_s - (time.time() - t0) > 50:
        t_sa = min((budget_s - (time.time() - t0)) * 0.25, 180.0)
        h_sa, c5_sa = multi_sa(best_h, max_time=t_sa, seed=seed + 1000, n_pos=4)
        if c5_sa < best_c5:
            best_c5 = c5_sa; best_h = h_sa.copy()
        # local refine after SA
        if budget_s - (time.time() - t0) > 20:
            h_r, c5_r = local_search_binary(best_h, max_time=min(15.0, (budget_s - (time.time() - t0)) * 0.1), seed=seed + 2000, n_2swap=80)
            if c5_r < best_c5:
                best_c5 = c5_r; best_h = h_r.copy()
    
    # Phase 3: LNS
    if budget_s - (time.time() - t0) > 50:
        t_lns = min((budget_s - (time.time() - t0)) * 0.25, 150.0)
        h_lns, c5_lns = lns_search(best_h, max_time=t_lns, seed=seed + 3000)
        if c5_lns < best_c5:
            best_c5 = c5_lns; best_h = h_lns.copy()
    
    # Phase 4: Multi-resolution
    if budget_s - (time.time() - t0) > 40:
        n_best = len(best_h)
        tried = set()
        for n_try in sorted([n_best-8, n_best+8, n_best-16, n_best+16, 96, 112, 120, 128, 136, 144, 160, 176]):
            if budget_s - (time.time() - t0) < 25:
                break
            if n_try in tried or n_try < 48 or n_try > 256 or n_try % 2 != 0:
                continue
            tried.add(n_try)
            h_int = interp_h(best_h, n_try)
            h_bin = make_binary(h_int, n_try, seed + n_try)
            t_a = min(25.0, (budget_s - (time.time() - t0)) * 0.12)
            if t_a < 3:
                break
            h_ls, c5_ls = local_search_binary(h_bin, max_time=t_a, seed=seed + 5000 + n_try, n_2swap=120)
            if c5_ls < best_c5:
                best_c5 = c5_ls; best_h = h_ls.copy()
    
    # Phase 5: Continuous refinement + round + binary search
    if budget_s - (time.time() - t0) > 30:
        t_c = min(40.0, (budget_s - (time.time() - t0)) * 0.2)
        h_cont, c5_cont = cont_refine(best_h, max_time=t_c, seed=seed + 8000)
        if c5_cont < best_c5:
            best_c5 = c5_cont; best_h = h_cont.copy()
        if budget_s - (time.time() - t0) > 20:
            h_round = make_binary(h_cont, len(h_cont), seed + 8500)
            t_b = min(20.0, (budget_s - (time.time() - t0)) * 0.25)
            h_lb, c5_lb = local_search_binary(h_round, max_time=t_b, seed=seed + 8600, n_2swap=100)
            if c5_lb < best_c5:
                best_c5 = c5_lb; best_h = h_lb.copy()
    
    # Phase 6: Final intensive local search
    if budget_s - (time.time() - t0) > 15:
        t_f = (budget_s - (time.time() - t0)) * 0.9
        h_f, c5_f = local_search_binary(best_h, max_time=t_f, seed=seed + 99999, n_2swap=200)
        if c5_f < best_c5:
            best_c5 = c5_f; best_h = h_f.copy()
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
