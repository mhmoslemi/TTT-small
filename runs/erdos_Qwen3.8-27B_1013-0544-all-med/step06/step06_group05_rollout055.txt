import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            room = 1.0 - h
            tr = room.sum()
            if tr > 1e-10:
                h += room * (diff / tr)
            else:
                h += diff / n
        else:
            mass = h.copy()
            tm = mass.sum()
            if tm > 1e-10:
                h -= mass * (-diff / tm)
            else:
                h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 5e-10:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def compute_grad(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    active = np.where(corr >= mx - 1e-10)[0]
    g = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            g[0:n-s] += 1.0 - h[s:n]
            g[s:n] += -h[0:n-s]
        else:
            ss = -s
            g[ss:n] += 1.0 - h[0:n-ss]
            g[0:n-ss] += -h[ss:n]
    return g, mx, corr

def optimize_subgrad(h0, max_time, seed=42):
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    alpha = 0.4
    mu = 0.85
    momentum = np.zeros(n)
    stall = 0
    no_improve = 0

    while time.time() - start < max_time:
        g, mx, corr = compute_grad(h, n)
        g -= np.mean(g)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        momentum = mu * momentum + (alpha / gn) * g
        h_new = normalize(h + momentum, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stall = 0
            no_improve = 0
            alpha = min(alpha * 1.03, 1.0)
        elif c5_new < compute_c5(h) + 0.001:
            h = h_new
            stall += 1
            no_improve += 1
            alpha *= 0.97
        else:
            stall += 1
            no_improve += 1
            alpha *= 0.93
            momentum *= 0.5
            if stall > 60:
                h = best_h + rng.normal(0, 0.08, n)
                h = normalize(h, n)
                momentum = np.zeros(n)
                alpha = 0.4
                stall = 0
            if no_improve > 500:
                h = best_h + rng.normal(0, 0.15, n)
                h = normalize(h, n)
                momentum = np.zeros(n)
                alpha = 0.5
                stall = 0
                no_improve = 0

    return best_h, best_c5

def optimize_local(h0, max_time, seed=42):
    n = len(h0)
    h = h0.astype(float).copy()
    best_c5 = compute_c5(h)
    start = time.time()

    while time.time() - start < max_time:
        g, mx, corr = compute_grad(h, n)
        g -= np.mean(g)
        order = np.argsort(-g)
        improved = False
        max_tries = min(30, n)
        for a in range(max_tries):
            i = int(order[a])
            if g[i] < 0.005:
                break
            for b in range(max_tries):
                j = int(order[n-1-b])
                if i == j or (g[i] - g[j]) < 0.005:
                    continue
                for eps in [0.08, 0.04, 0.15, 0.02, 0.2]:
                    if h[i] >= eps - 1e-9 and h[j] + eps <= 1.0 + 1e-9:
                        h[i] -= eps
                        h[j] += eps
                        new_c5 = compute_c5(h)
                        if new_c5 < best_c5 - 1e-12:
                            best_c5 = new_c5
                            improved = True
                            break
                        h[i] += eps
                        h[j] -= eps
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c5

def optimize_binary_local(h0, max_time, seed=42):
    n = len(h0)
    h = np.round(h0).astype(float)
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        ones = np.where(h == 1)[0]
        h[ones[target:]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target - n_ones]] = 1
    best_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)

    while time.time() - start < max_time:
        g, mx, corr = compute_grad(h, n)
        g -= np.mean(g)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        improved = False
        # Try best single flips guided by gradient
        best_gain = 0
        best_move = None
        n_try = min(200, len(ones) * len(zeros))
        for _ in range(n_try):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0; h[j] = 1
            nc = compute_c5(h)
            h[i] = 1; h[j] = 0
            if best_c5 - nc > best_gain:
                best_gain = best_c5 - nc
                best_move = (int(i), int(j))
        if best_move is not None:
            h[best_move[0]] = 0; h[best_move[1]] = 1
            best_c5 = compute_c5(h)
            improved = True
        if not improved:
            # Try double flips
            n_ones = min(50, len(ones))
            n_zeros = min(50, len(zeros))
            idx_o = rng.choice(len(ones), n_ones, replace=False)
            idx_z = rng.choice(len(zeros), n_zeros, replace=False)
            best_gain = 0
            best_move = None
            for a in range(min(30, n_ones)):
                for b in range(min(30, n_zeros)):
                    i1, i2 = ones[idx_o[a]], ones[idx_o[(a+1)%n_ones]] if a+1 < n_ones else None
                    if i2 is None or i1 == i2:
                        i1, i2 = ones[idx_o[a]], ones[idx_o[(a+7)%n_ones]]
                        if i1 == i2:
                            continue
                    j1, j2 = zeros[idx_z[b]], zeros[idx_z[(b+1)%n_zeros]] if b+1 < n_zeros else None
                    if j2 is None or j1 == j2:
                        j1, j2 = zeros[idx_z[b]], zeros[idx_z[(b+7)%n_zeros]]
                        if j1 == j2:
                            continue
                    h[i1]=0; h[i2]=0; h[j1]=1; h[j2]=1
                    nc = compute_c5(h)
                    h[i1]=1; h[i2]=1; h[j1]=0; h[j2]=0
                    if best_c5 - nc > best_gain:
                        best_gain = best_c5 - nc
                        best_move = (int(i1), int(i2), int(j1), int(j2))
            if best_move is not None:
                h[best_move[0]]=0; h[best_move[1]]=0
                h[best_move[2]]=1; h[best_move[3]]=1
                best_c5 = compute_c5(h)
            else:
                break
    return h, best_c5
