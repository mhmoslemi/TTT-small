def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5

        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0

    if abs(diff) > 1e-10:

        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_lag(h, shift):
    N = len(h)
    g = np.zeros(N)
    s1 = max(0, shift)
    e1 = min(N, N + shift)
    if s1 < e1:

    s2 = max(0, -shift)
    e2 = min(N, N - shift)
    if s2 < e2:
        g[s2:e2] -= h[s2 + shift:e2 + shift]
    return g

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_lag(h, shift):
    N = len(h)
    g = np.zeros(N)
    s1 = max(0, shift)
    e1 = min(N, N + shift)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - shift:e1 - shift]
    s2 = max(0, -shift)
    e2 = min(N, N - shift)
    if s2 < e2:
        g[s2:e2] -= h[s2 + shift:e2 + shift]
    return g

def soft_grad_topk(h, corr, dx, alpha, topk):
    N = len(h)
    top_indices = np.argsort(corr)[-topk:]
    weights = np.exp(alpha * (corr[top_indices] - corr.max()))
    wsum = weights.sum()
    if wsum < 1e-300:
        return grad_lag(h, int(np.argmax(corr)) - (N - 1)) * dx
    weights /= wsum
    g = np.zeros(N)
    for j, w in zip(top_indices, weights):
        if w > 1e-10:
            g += w * grad_lag(h, j - (N - 1))
    return g * dx

def interpolate_to_n(h_src, n_dst):
    x0 = np.linspace(0.0, 1.0, len(h_src))
    x1 = np.linspace(0.0, 1.0, n_dst)
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def local_optimize(h, N, rng, t_end, lr_init, max_iter):
    val, corr, dx = obj_and_corr(h)
    v = np.zeros(N)
    mu = 0.8
    lr = lr_init
    alpha = 6.0
    stagnant = 0
    best_val = val
    best_h = h.copy()
    for it in range(max_iter):
        if time.time() > t_end:
            break
        alpha = min(6.0 + 0.03 * it, 80.0)
        topk = 5 if alpha > 30 else 7
        g = soft_grad_topk(h, corr, dx, alpha, topk)
        v = mu * v - lr * g
        h_new = project_feasible(h + v, N)
        val_new, corr_new, _ = obj_and_corr(h_new)
        if val_new < val - 1e-14:
            h = h_new
            val = val_new
            corr = corr_new
            stagnant = 0
            lr = min(lr * 1.008, 0.04)
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            stagnant += 1
            v *= 0.3
            lr *= 0.85
            if stagnant > 25:
                break
    return best_h, best_val

def update_best(best, h, val):
    if val < best['val']:
        best['val'] = val
        best['h'] = h.copy()

def make_perturbation(best_h, N, rng, strategy):
    if strategy == 0:
        scale = 0.02 + 0.05 * rng.random()
        return project_feasible(best_h + rng.randn(N) * scale, N)
    elif strategy == 1:
        bs = rng.randint(2, max(3, N // 5))
        s = rng.randint(0, N - bs)
        h_p = best_h.copy()
        h_p[s:s+bs] = 1.0 - h_p[s:s+bs]
        return project_feasible(h_p, N)
    elif strategy == 2:
        pert = rng.randn(N) * 0.035
        pert = (pert + pert[::-1]) / 2.0
        return project_feasible(best_h + pert, N)
    elif strategy == 3:
        h_rand = project_feasible(rng.uniform(0.0, 1.0, N), N)
        a = 0.05 + 0.15 * rng.random()
        return project_feasible(a * h_rand + (1.0 - a) * best_h, N)
    elif strategy == 4:
        h_p = best_h.copy()
        idx = rng.choice(N, size=max(2, N // 8), replace=False)
        h_p[idx] = 1.0 - h_p[idx]
        return project_feasible(h_p, N)
    else:
        scale = 0.04 + 0.03 * rng.random()
        return project_feasible(best_h + rng.randn(N) * scale, N)
