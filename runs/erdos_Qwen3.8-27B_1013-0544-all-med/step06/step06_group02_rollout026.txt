For m = N-1: output[N-1] = sum_n h[n] * (1-h[n]) ✓
For m = N-1+1 = N: output[N] = sum_n h[n+1] * (1-h[n]) (for n=0 to N-2)
For m = N-1-1 = N-2: output[N-2] = sum_n h[n-1] * (1-h[n]) (for n=1 to N-1)

So:
output[N-1+p] = sum_{n=0}^{N-1-p} h[n+p] * (1-h[n])  for p = 0, ..., N-1
output[N-1-p] = sum_{n=p}^{N-1} h[n-p] * (1-h[n])  for p = 1, ..., N-1

Let's verify with N=4:
output[3] = h[0](1-h[0]) + h[1](1-h[1]) + h[2](1-h[2]) + h[3](1-h[3]) ✓
output[4] = h[1](1-h[0]) + h[2](1-h[1]) + h[3](1-h[2]) ✓ (p=1)
output[2] = h[0](1-h[1]) + h[1](1-h[2]) + h[2](1-h[3]) ✓ (p=1, from the other side)

So output[N-1+1] = h[1](1-h[0]) + h[2](1-h[1]) + h[3](1-h[2])
And output[N-1-1] = h[0](1-h[1]) + h[1](1-h[2]) + h[2](1-h[3])

These ARE different (not symmetric in general). The full correlation is NOT symmetric when a ≠ v.

So I need to handle positive and negative lags separately. The gradient for output[N-1+p] (p > 0):
f_p = sum_{n=0}^{N-1-p} h[n+p](1-h[n])
∂f_p/∂h[j]:
- j appears in h[n+p] when n = j-p, valid if 0 ≤ j-p ≤ N-1-p, i.e., p ≤ j ≤ N-1
  contribution: (1-h[j-p])
- j appears in h[n] when n = j, valid if 0 ≤ j ≤ N-1-p
  contribution: -h[j+p]

So:
- j ∈ [0, p-1]: only second term (j-p < 0): ∂f_p/∂h[j] = -h[j+p] (valid since j ≤ p-1 < N-1-p for p < N/2)
- j ∈ [p, N-1-p]: both terms: ∂f_p/∂h[j] = (1-h[j-p]) - h[j+p]
- j ∈ [N-p, N-1]: only first term (j > N-1-p): ∂f_p/∂h[j] = 1-h[j-p]

For output[N-1-p] (p > 0):
f_{-p} = sum_{n=p}^{N-1} h[n-p](1-h[n])
Let m = n-p, so n = m+p:
f_{-p} = sum_{m=0}^{N-1-p} h[m](1-h[m+p])
∂f_{-p}/∂h[j]:
- j appears in h[m] when m = j, valid if 0 ≤ j ≤ N-1-p:
  contribution: -(1-h[j+p]) = h[j+p] - 1
- j appears in h[m+p] when m = j-p, valid if 0 ≤ j-p ≤ N-1-p, i.e., p ≤ j ≤ N-1:
  contribution: -h[j-p]

So:
- j ∈ [0, p-1]: only first term: ∂f_{-p}/∂h[j] = h[j+p] - 1
- j ∈ [p, N-1-p]: both terms: ∂f_{-p}/∂h[j] = (h[j+p] - 1) + (-h[j-p]) = h[j+p] - 1 - h[j-p]
- j ∈ [N-p, N-1]: only second term: ∂f_{-p}/∂h[j] = -h[j-p]

Interesting! So the gradients for +p and -p are NOT the same.

Now, the objective is max_k output[k] * dx over ALL k (both positive and negative lags). So I need to consider all 2N-1 lags.

But wait, output[N-1+p] and output[N-1-p] are different. However, if h is "balanced" in some sense, they might be close.

For the top-K approach, I should consider all 2N-1 lags and pick the top K.

Let me simplify: just compute the gradient for a specific output index m.

Given m (0 ≤ m ≤ 2N-2), the lag is k = m - (N-1) (can be negative, zero, or positive).

If k = 0: f = sum h[i](1-h[i]), gradient = 1 - 2h
If k > 0: f = sum_{i=0}^{N-1-k} h[i+k](1-h[i]), gradient as derived above
If k < 0: let p = -k > 0, f = sum_{i=p}^{N-1} h[i-p](1-h[i]) = sum_{m=0}^{N-1-p} h[m](1-h[m+p]), gradient as derived above

```python
def grad_for_output(h, m):
    """Gradient of output[m] w.r.t. h, where output = np.correlate(h, 1-h, mode='full')."""
    N = len(h)
    k = m - (N - 1)
    g = np.zeros(N)
    if k == 0:
        return 1.0 - 2.0 * h
    elif k > 0:
        p = k
        # j in [0, p-1]: -h[j+p]
        hi1 = min(p, N)
        g[0:hi1] += -h[p:p+hi1]
        # j in [p, N-1-p]: (1-h[j-p]) - h[j+p]
        lo2, hi2 = p, N - p
        if lo2 < hi2:
            g[lo2:hi2] += (1.0 - h[0:hi2-lo2]) - h[lo2+p:hi2+p]
        # j in [N-p, N-1]: 1-h[j-p]
        lo3 = N - p
        if lo3 < N:
            g[lo3:N] += 1.0 - h[lo3-p:N-p]
    else:
        p = -k
        # j in [0, p-1]: h[j+p] - 1
        hi1 = min(p, N)
        g[0:hi1] += h[p:p+hi1] - 1.0
        # j in [p, N-1-p]: h[j+p] - 1 - h[j-p]
        lo2, hi2 = p, N - p
        if lo2 < hi2:
            g[lo2:hi2] += h[lo2+p:hi2+p] - 1.0 - h[0:hi2-lo2]
        # j in [N-p, N-1]: -h[j-p]
        lo3 = N - p
        if lo3 < N:
            g[lo3:N] += -h[lo3-p:N-p]
    return g
```

def project_feasible(v, N):
    target = N / 2.0
    lo = v.min() - 0.1
    hi = v.max() + 0.1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo = v.min() - 0.1
    hi = v.max() + 0.1
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_correlate(h):
    return np.correlate(h, 1.0 - h, mode='full')

def obj_from_corr(corr, N):
    return corr.max() * 2.0 / N

def grad_for_output(h, m):
    N = len(h)
    k = m - (N - 1)
    g = np.zeros(N)
    if k == 0:
        return 1.0 - 2.0 * h
    elif k > 0:
        p = k
        hi1 = min(p, N)
        if hi1 > 0:
            g[0:hi1] += -h[p:p+hi1]
        lo2, hi2 = p, N - p
        if lo2 < hi2:
            g[lo2:hi2] += (1.0 - h[0:hi2-lo2]) - h[lo2+p:hi2+p]
        lo3 = N - p
        if lo3 < N:
            g[lo3:N] += 1.0 - h[lo3-p:N-p]
    else:
        p = -k
        hi1 = min(p, N)
        if hi1 > 0:
            g[0:hi1] += h[p:p+hi1] - 1.0
        lo2, hi2 = p, N - p
        if lo2 < hi2:
            g[lo2:hi2] += h[lo2+p:hi2+p] - 1.0 - h[0:hi2-lo2]
        lo3 = N - p
        if lo3 < N:
            g[lo3:N] += -h[lo3-p:N-p]
    return g

def topk_gradient(h, corr, topk=5):
    N = len(h)
    top_indices = np.argsort(corr)[-topk:][::-1]
    g = np.zeros(N)
    for m in top_indices:
        g += grad_for_output(h, int(m))
    return g / topk

def interpolate_to_n(h_src, n_dst):
    x0 = np.linspace(0, 1, len(h_src))
    x1 = np.linspace(0, 1, n_dst)
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def grad_descent(h, N, topk, lr_init, beta, max_iter, t_end, rng):
    corr = compute_correlate(h)
    val = obj_from_corr(corr, N)
    best_h = h.copy()
    best_val = val
    momentum = np.zeros(N)
    lr = lr_init
    stagnant = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = topk_gradient(h, corr, topk)
        momentum = beta * momentum + (1.0 - beta) * g
        h_new = project_feasible(h - lr * momentum, N)
        corr_new = compute_correlate(h_new)
        val_new = obj_from_corr(corr_new, N)
        if val_new < val - 1e-15:
            h = h_new
            corr = corr_new
            val = val_new
            lr = min(lr * 1.012, lr_init * 4.0)
            stagnant = 0
            if val < best_val:
                best_val = val
                best_h = h.copy()
        else:
            stagnant += 1
            lr *= 0.92
            if stagnant > 80:
                scale = 0.02 + 0.02 * min(stagnant / 200.0, 2.0)
                h = project_feasible(h + rng.randn(N) * scale, N)
                corr = compute_correlate(h)
                val = obj_from_corr(corr, N)
                lr = lr_init
                momentum = np.zeros(N)
                stagnant = 0
    return best_h, best_val

def push_phase(h, N, t_end, rng, max_iter=30000):
    best_h = h.copy()
    corr = compute_correlate(h)
    best_val = obj_from_corr(corr, N)
    hcurr = h.copy()
    for it in range(max_iter):

            h_try[i] = 1.0

        h_try = project_feasible(h_try, N)

        if val_try < best_val - 1e-13:
            hcurr = h_try
            best_val = val_try
            best_h = hcurr.copy()
    return best_h, best_val
