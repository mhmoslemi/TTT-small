import numpy as np
import time
import warnings

warnings.simplefilter("ignore")


def _project(h, target):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)


def _cvals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, "full") * (2.0 / n)


def _c5(h):
    return float(np.max(_cvals(h)))


def _grad(h, idxs):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    for idx in idxs:
        lag = int(idx) - (n - 1)
        if lag == 0:
            g += 1.0 - 2.0 * h
        elif lag > 0:
            g[lag:] += 1.0 - h[:n - lag]
            g[:n - lag] -= h[lag:]
        else:
            al = -lag
            g[:n - al] += 1.0 - h[al:]
            g[al:] -= h[:n - al]
    return g * dx


def _resample(h0, n_new):
    n0 = len(h0)
    if n0 == n_new:
        return _project(h0.copy(), n_new / 2.0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return _project(np.interp(x1, x0, h0), n_new / 2.0)


def _optimize(h0, deadline, rng):
    n = len(h0)
    target = n / 2.0
    h = _project(h0.copy(), target)
    best_h, best_c = h.copy(), _c5(h)
    alpha = 0.06
    top_k = 2
    beta = 0.7
    v = np.zeros(n)
    stall = 0
    use_momentum = True

    while time.time() < deadline - 15:
        c = _cvals(h)
        C = float(c.max())

        if C < best_c - 1e-14:
            best_c, best_h, stall = C, h.copy(), 0
        else:
            stall += 1

        top_idx = np.argsort(c)[::-1][:top_k]
        g = _grad(h, top_idx)
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break

        if use_momentum and stall < 10:
            v = beta * v + (1.0 - beta) * g
            direction = v
        else:
            direction = g
            v = np.zeros(n)

        a = alpha
        ok = False
        for _ in range(12):
            h_new = _project(h - a * direction, target)
            if _c5(h_new) < C - 1e-13:
                h = h_new
                ok = True
                break
            a *= 0.45
            if a < 1e-11:
                break

        if ok:
            alpha = min(a * 1.25, 0.18)
            if stall == 0 and top_k < 5 and best_c < 0.3815:
                top_k += 1
            use_momentum = True
        else:
            alpha = max(a * 0.6, 0.003)
            if stall > 25:
                s = 0.004 + 0.006 * min((stall - 25) // 25, 4)
                h = _project(h + rng.normal(0, s, n), target)
                v = np.zeros(n)
                stall = 0
                top_k = 2
                use_momentum = False
            elif stall > 8:
                use_momentum = False
                top_k = 1
            if stall > 12 and top_k > 1:
                top_k = max(1, top_k - 1)

    return best_h, best_c


def _perturb(h, rng, scale):
    n = len(h)
    return _project(h + rng.normal(0, scale, n), n / 2.0)
