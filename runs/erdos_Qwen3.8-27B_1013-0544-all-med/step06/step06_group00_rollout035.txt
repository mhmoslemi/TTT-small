def project_box_sum(h, target):
    """Project h onto {x: 0≤x≤1, sum(x)=target}."""
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0, 1)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fine-tune
    d = target - proj.sum()
    if abs(d) > 1e-14:
        mask = (proj > 0) & (proj < 1)
        if mask.sum() > 0:
            proj[mask] += d / mask.sum()
            proj = np.clip(proj, 0, 1)
    return proj

                    best_c5 = c5_r2
                    best_h = h_r

    # Phase 4: Fine-tune with lower temperature
    if remaining() > 30:
        h_fin, c5_fin = optimize_pg(best_h, n0, 0.5, 2000, mu=0.5, alpha0=0.02, t_limit=t0)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Phase 5: Coordinate descent polish
    if remaining() > 20:
        h_cd, c5_cd = coord_descend(best_h, n_levels=9, max_sweeps=10, t_limit=t0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Final feasibility check
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
