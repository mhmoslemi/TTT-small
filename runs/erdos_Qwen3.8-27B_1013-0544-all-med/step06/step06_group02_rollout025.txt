    best_local = val
    best_local_h = h.copy()
    lr = 0.04
    eta_b = 0.08
    alpha = 12.0
    stagnant = 0
    method = 'grad'
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        
        if mode == 'adaptive':
            g = soft_grad(h, corr, dx, alpha)
        elif mode == 'lp':
            p = 8.0
            g = lp_grad(h, corr, dx, p)
        else:
            g = soft_grad(h, corr, dx, 40.0)
        
        if method == 'grad':
            h_new = project_feasible(h - lr * g, N)
        elif method == 'bregman':
            h_new = bregman_step(h, g, eta_b, N)
        else:
            i, j = rng.randint(N, 2)
            h_new = h.copy()
            h_new[i], h_new[j] = h[j], h[i]
        
        val_new, corr_new = obj_corr(h_new)
        
        if val_new < val - 1e-14:
            h = h_new
            val = val_new
            corr = corr_new
            stagnant = 0
            if val < best_local:
                best_local = val
                best_local_h = h.copy()
            if method == 'grad':
                lr = min(lr * 1.015, 0.12)
            elif method == 'bregman':
                eta_b = min(eta_b * 1.015, 0.4)
        else:
            stagnant += 1
            if method == 'grad':
                lr *= 0.75
            elif method == 'bregman':
                eta_b *= 0.75
            if stagnant > 40:
                method = 'bregman' if method == 'grad' else ('swap' if method == 'bregman' else 'grad')
                stagnant = 0
                lr = 0.04
                eta_b = 0.08
            if stagnant > 150:
                h = project_feasible(h + rng.randn(N) * 0.08, N)
                val, corr = obj_corr(h)
                stagnant = 0
                lr = 0.04
                eta_b = 0.08
        
        if it % 3000 == 0 and mode == 'adaptive':
            alpha = min(12.0 + it * 0.008, 45.0)
    
    return best_local_h, best_local

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 10
    
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None
    
    best_h = None
    best_val = float('inf')
    
    def try_update(h, val):
        nonlocal best_h, best_val
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    if h_init is not None:
        try_update(h_init, obj_corr(h_init)[0])
    
    if best_h is None:
        best_h = 0.5 * np.ones(64)
        best_val = obj_corr(best_h)[0]
    
    N = len(best_h)
    
    # Phase 1: Main refinement at initial resolution
    if time.time() < t_end - 100:
        t1 = min(time.time() + 300, t_end - 100)
        h, v = refine(best_h.copy(), N, rng, t1, mode='adaptive')
        try_update(h, v)
    
    # Phase 2: Bregman-focused refinement
    if time.time() < t_end - 80:
        t2 = min(time.time() + 150, t_end - 80)
        h = refine(best_h.copy(), N, rng, t2, mode='lp')[0]
        val = obj_corr(h)[0]
        try_update(h, val)
    
    # Phase 3: Multi-resolution exploration
    n_list = [N + 20, N + 40, N + 80]
    for n_t in n_list:
        if time.time() > t_end - 50:
            break
        h_start = interpolate_to_n(best_h, n_t)
        t3 = min(time.time() + 50, t_end - 50)
        h, v = refine(h_start, n_t, rng, t3, mode='adaptive')
        try_update(h, v)
    
    # Phase 4: Aggressive local search with swaps on best
    if time.time() < t_end - 20:
        N_best = len(best_h)
        h = best_h.copy()
        val, corr = obj_corr(h)
        dx = 2.0 / N_best
        best_local = val
        best_local_h = h.copy()
        for it in range(100000):
            if time.time() > t_end - 20:
                break
            k_star = np.argmax(corr)
            n_lag = k_star - (N_best - 1)
            g = grad_lag(h, n_lag)
            # Find best swap: pair (i,j) where g[i] is most positive and g[j] most negative
            idx_pos = np.argmax(g)
            idx_neg = np.argmin(g)
            if g[idx_pos] > 0 and g[idx_neg] < 0:
                h_new = h.copy()
                h_new[idx_pos], h_new[idx_neg] = h[idx_neg], h[idx_pos]
                val_new, corr_new = obj_corr(h_new)
                if val_new < best_local - 1e-14:
                    h = h_new
                    val = val_new
                    corr = corr_new
                    best_local = val_new
                    best_local_h = h.copy()
                    try_update(h, val)
                    continue
            # Also try random swaps
            i, j = rng.randint(N_best, 2)
            h_new = h.copy()
            h_new[i], h_new[j] = h[j], h[i]
            val_new, corr_new = obj_corr(h_new)
            if val_new < val - 1e-14:
                h = h_new
                val = val_new
                corr = corr_new
                if val < best_local:
                    best_local = val
                    best_local_h = h.copy()
                    try_update(h, val)
    
    # Phase 5: Final micro-polish
    if best_h is not None and time.time() < t_end:
        N_best = len(best_h)
        h = best_h.copy()
        val, corr = obj_corr(h)
        dx = 2.0 / N_best
        lr = 0.003
        for it in range(200000):
            if time.time() > t_end:
                break
            g = soft_grad(h, corr, dx, 50.0)
            h_new = project_feasible(h - lr * g, N_best)
            val_new, corr_new = obj_corr(h_new)
            if val_new < val - 1e-15:
                h = h_new
                val = val_new
                corr = corr_new
                try_update(h, val)
                lr = min(lr * 1.01, 0.01)
            else:
                lr *= 0.93
    
    return best_h, best_val, len(best_h)
