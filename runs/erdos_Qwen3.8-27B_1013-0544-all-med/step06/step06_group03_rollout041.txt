def solve_1d_subproblem(A, B, n_iters=40):

    A and B are arrays of length m (number of shifts)."""

    # Lower bound: min of max at t=0 and t=1
    max_at_0 = np.max(A)
    max_at_1 = np.max(A + B)
    lo_V = min(max_at_0, max_at_1)  # Actually this isn't a valid lower bound in general
    # Better lower bound: for each s, A[s] + B[s]*t >= min(A[s], A[s]+B[s]) (since t in [0,1])
    # So max_s >= max_s min(A[s], A[s]+B[s])
    lo_V = np.max(np.minimum(A, A + B))
    hi_V = np.max(np.maximum(A, A + B))
    
    # Handle B == 0
    # For B_s == 0: need A_s <= V, so V >= max(A_s where B_s==0)
    mask_zero = (B == 0)
    if np.any(mask_zero):
        lo_V = max(lo_V, np.max(A[mask_zero]))
    
    # Binary search on V
    for _ in range(n_iters):
        mid_V = (lo_V + hi_V) / 2.0
        # Check if feasible: exists t in [0,1] with A_s + B_s*t <= mid_V for all s
        lo_t = 0.0
        hi_t = 1.0
        
        mask_pos = B > 1e-15
        mask_neg = B < -1e-15
        
        if np.any(mask_pos):
            # A_s + B_s*t <= mid_V => t <= (mid_V - A_s)/B_s
            vals = (mid_V - A[mask_pos]) / B[mask_pos]
            hi_t = min(hi_t, np.min(vals))
        
        if np.any(mask_neg):
            # A_s + B_s*t <= mid_V => t >= (mid_V - A_s)/B_s (flip since B_s < 0)
            vals = (mid_V - A[mask_neg]) / B[mask_neg]
            lo_t = max(lo_t, np.max(vals))
        
        if lo_t <= hi_t:
            hi_V = mid_V
        else:
            lo_V = mid_V
    
    # Now find the actual t at V = hi_V (or lo_V)
    V = (lo_V + hi_V) / 2.0
    lo_t = 0.0
    hi_t = 1.0
    mask_pos = B > 1e-15
    mask_neg = B < -1e-15
    if np.any(mask_pos):
        hi_t = min(hi_t, np.min((V - A[mask_pos]) / B[mask_pos]))
    if np.any(mask_neg):
        lo_t = max(lo_t, np.max((V - A[mask_neg]) / B[mask_neg]))
    
    if lo_t > hi_t:
        # Infeasible at this V, use t that minimizes the max directly
        # Just evaluate at 0, 1, and a few points
        best_t = 0.0
        best_val = np.max(A)
        for t_cand in [0.0, 1.0, 0.5, 0.25, 0.75]:
            val = np.max(A + B * t_cand)
            if val < best_val:
                best_val = val
                best_t = t_cand
        return best_t
    
    return (lo_t + hi_t) / 2.0

def solve_1d(A, B, n_iter=40):
    lo_V = float(np.max(np.minimum(A, A + B)))
    hi_V = float(np.max(np.maximum(A, A + B)))
    if hi_V - lo_V < 1e-15:
        return 0.5
    mp = B > 1e-18
    mn = B < -1e-18
    mz = ~(mp | mn)
    if np.any(mz):
        lo_V = max(lo_V, float(np.max(A[mz])))
    for _ in range(n_iter):
        mid_V = (lo_V + hi_V) * 0.5
        lo_t = 0.0
        hi_t = 1.0
        if np.any(mp):
            hi_t = min(hi_t, float(np.min((mid_V - A[mp]) / B[mp])))
        if np.any(mn):
            lo_t = max(lo_t, float(np.max((mid_V - A[mn]) / B[mn])))
        if lo_t <= hi_t:
            hi_V = mid_V
        else:
            lo_V = mid_V
    V = (lo_V + hi_V) * 0.5
    lo_t = 0.0
    hi_t = 1.0
    if np.any(mp):
        hi_t = min(hi_t, float(np.min((V - A[mp]) / B[mp])))
    if np.any(mn):
        lo_t = max(lo_t, float(np.max((V - A[mn]) / B[mn])))
    if lo_t > hi_t + 1e-15:
        best_t = 0.0
        best_v = float(np.max(A))
        for tc in (0.0, 0.25, 0.5, 0.75, 1.0):
            v = float(np.max(A + B * tc))
            if v < best_v:
                best_v = v
                best_t = tc
        return best_t
    return max(0.0, min(1.0, (lo_t + hi_t) * 0.5))

def coord_descent(h, n_sweeps, t0, budget_s, proj_every):
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    best_c5 = float(np.max(conv) * dx)
    best_h = h.copy()
    for sweep in range(n_sweeps):
        if time.time() - t0 > budget_s * 0.88:
            break
        for i in range(n):
            old = h[i]
            B = np.zeros(2 * n - 1)
            i1s = n - 1 - i
            i1e = 2 * n - 2 - i
            B[i1s:i1e + 1] += dx * (1.0 - h[0:n - i])
            i2s = i
            i2e = i + n - 1
            B[i2s:i2e + 1] -= dx * h[::-1]
            A = conv - B * old
            t = solve_1d(A, B)
            d = t - old
            if abs(d) > 1e-14:
                conv[i1s:i1e + 1] += dx * d * (1.0 - h[0:n - i])
                conv[i2s:i2e + 1] -= dx * d * h[::-1]
                h[i] = t
        if (sweep + 1) % proj_every == 0:
            h = project_feasible(h, n)
            conv = np.correlate(h, 1.0 - h, mode="full")
            c5 = float(np.max(conv) * dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    h = project_feasible(h, n)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h
    return best_h, best_c5

def do_perturb(h, rng, t0, budget_s, n_pert):
    n = len(h)
    h2 = h.copy()
    idx = rng.choice(n, size=min(n_pert, n), replace=False)
    for i in idx:
        h2[i] = rng.uniform(0.0, 1.0)
    h2 = project_feasible(h2, n)
    return coord_descent(h2, 80, t0, budget_s, 5)
