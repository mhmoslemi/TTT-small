    # i is at position i in h, j is at position j
    # corr[k] = sum_m h[m] * (1-h[m + k - (n-1)])
    # Let l = k - (n-1) be the lag

    # - Direct term: -(1-h[i+l]) for valid i+l
    # - Indirect term (in 1-h[...]): +h[i-l] for valid i-l (where 1-h[i] becomes 1-h[i]=1)

    #   So terms where m+l = i, i.e., m = i-l, gain h[m] * (1-0) - h[m]*(1-1) = h[m]

    # - Direct term: +(1-h[j+l]) for valid j+l
    # - Indirect term: 1-h[j] was 1 (since h[j]=0), now becomes 0
    #   So terms where m+l = j, i.e., m = j-l, lose h[m] * 1

        if 0 <= i + l < n:

        # Indirect from j (1-h[j] changes from 1 to 0)
        if 0 <= j - l < n:
            d -= h[j-l]
