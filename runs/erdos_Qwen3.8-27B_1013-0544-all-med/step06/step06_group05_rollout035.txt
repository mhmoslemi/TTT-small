            h_s = interp_to_n(best_h, n_t)
            h_s = rng.rand(n_t)
            h_s = make_binary(h_s, n_t)
        t_s = min(40.0, rem() * 0.15)
        if t_s < 5:
            break
        h_r, c5_r = binary_local_search(h_s, max_time=t_s, seed=seed + n_t * 7, double=True)
        if c5_r < best_c5:
            best_c5 = c5_r; best_h = h_r.copy()
    # Phase 3: continuous optimization
    if rem() > 60:
        t3 = min(200.0, rem() * 0.35)
        h_cont = best_h.astype(float).copy()
        h_cont = project_constraint(h_cont, len(h_cont))
        h_c, c5_c = softmax_gd(h_cont, max_time=t3, seed=seed + 5000)
        if c5_c < best_c5:
            best_c5 = c5_c; best_h = h_c.copy()
        # also try continuous at different n
        for n_try in [len(best_h)+8, len(best_h)-8]:
            if rem() < 80 or n_try < 40 or n_try > 256 or n_try % 2 != 0:
                continue
            h_int = interp_to_n(best_h, n_try)
            t_c = min(60.0, rem() * 0.2)
            h_c2, c5_c2 = softmax_gd(h_int, max_time=t_c, seed=seed + 6000 + n_try)
            if c5_c2 < best_c5:
                best_c5 = c5_c2; best_h = h_c2.copy()
    # Phase 4: SA on best
    if rem() > 40:
        t4 = min(150.0, rem() * 0.3)
        h_sa, c5_sa = simulated_anneal(best_h, max_time=t4, seed=seed + 9000)
        if c5_sa < best_c5:
            best_c5 = c5_sa; best_h = h_sa.copy()
    # Phase 5: final local search
    if rem() > 20:
        h_bin = make_binary(best_h, len(best_h))
        t5 = rem() * 0.85
        h_f, c5_f = binary_local_search(h_bin, max_time=t5, seed=seed + 12000, double=True)
        if c5_f < best_c5:
            best_c5 = c5_f; best_h = h_f.copy()
        # Also keep continuous if better
        final_c5 = compute_c5(best_h)
        if final_c5 >= best_c5:
            final_c5 = best_c5
    return best_h, min(compute_c5(best_h), best_c5), len(best_h)
