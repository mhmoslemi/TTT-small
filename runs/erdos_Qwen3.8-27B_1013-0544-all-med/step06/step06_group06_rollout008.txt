    h = project_sum(h, n)
    c = np.correlate(h, 1 - h, mode='full') * (2.0/n)
    return smooth_max(c, alpha)

def coord_descent_ap(h_ap, n, max_rounds=500, rng=None):
    n_half = n // 2
    best = compute_c5(antiper_to_full(h_ap, n), n)
    for _ in range(max_rounds):
        improved = False
        for j in range(n_half):
            for delta in [0.002, -0.002, 0.005, -0.005, 0.01, -0.01, 0.02, -0.02]:
                h_ap[j] += delta
                if h_ap[j] < 0 or h_ap[j] > 1:
                    h_ap[j] -= delta
                    continue
                val = compute_c5(antiper_to_full(h_ap, n), n)
                if val < best - 1e-13:
                    best = val
                    improved = True
                    break
                else:
                    h_ap[j] -= delta
        if not improved:
            break
    return h_ap, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h0)
    except Exception:
        h0 = None
        n_init = 80

    def time_left():
        return budget_s - (time.time() - t0)

    # Phase 1: Anti-periodic L-BFGS-B optimization
    n_list = [40, 60, 80, 100, 120, 160, 200, 240]
    if n_init not in n_list:
        n_list.insert(0, n_init)
    
    for n in n_list:
        if time_left() < 20:
            break
        n_half = n // 2
        bounds = [(0.0, 1.0)] * n_half

        # Starting points
        starts = []
        if n == n_init and h0 is not None:
            starts.append(h0[:n_half].copy())
        # Cosine-based starts
        for freq in [1, 2, 3, 4]:
            t = np.linspace(0, 2*np.pi, n_half)
            starts.append(np.clip(0.5 + 0.35*np.cos(freq*t + rng.random()*0.5), 0.01, 0.99))
        # Random
        starts.append(np.clip(0.5 + 0.3*rng.randn(n_half), 0.01, 0.99))

        for h_start in starts:
            if time_left() < 10:
                break
            h_cur = h_start.copy()
            for alpha in [30, 100, 300, 1000, 3000, 10000]:
                if time_left() < 8:
                    break
                try:
                    res = minimize(obj_ap, h_cur, args=(n, alpha), jac=grad_ap,
                                  method='L-BFGS-B', bounds=bounds,
                                  options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-10})
                    h_opt = np.clip(res.x, 0, 1)
                    hf = antiper_to_full(h_opt, n)
                    c5 = compute_c5(hf, n)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hf.copy()
                        best_n = n
                    h_cur = h_opt
                except Exception:
                    pass

    # Phase 2: Full-space optimization (no anti-periodic constraint)
    full_ns = []
    if h0 is not None:
        full_ns.append(n_init)
    full_ns.extend([100, 160, 200])
    full_ns = sorted(set(full_ns))

    for n in full_ns:
        if time_left() < 20:
            break
        if h0 is not None:
            if n == n_init:
                h_start = h0.copy()
            else:
                x_old = np.linspace(0, 2, len(h0))
                x_new = np.linspace(0, 2, n)
                h_start = np.interp(x_new, x_old, h0)
        else:
            h_start = np.full(n, 0.5)
        h_start = project_sum(h_start, n)
        bounds = [(0.0, 1.0)] * n

        for alpha in [100, 300, 1000]:
            if time_left() < 10:
                break
            try:
                res = minimize(obj_full, h_start, args=(n, alpha),
                             method='L-BFGS-B', bounds=bounds,
                             options={'maxiter': 2000, 'ftol': 1e-15})
                h_opt = project_sum(res.x, n)
                c5 = compute_c5(h_opt, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    best_n = n
                h_start = h_opt
            except Exception:
                pass

    # Phase 3: Coordinate descent on anti-periodic
    if best_h is not None and time_left() > 20:
        n = best_n
        n_half = n // 2
        h_ap = best_h[:n_half].copy()
        diff_ap = np.max(np.abs(best_h[n_half:] - (1.0 - h_ap)))
        if diff_ap < 0.2:
            h_ap, c5 = coord_descent_ap(h_ap, n, max_rounds=300, rng=rng)
            hf = antiper_to_full(h_ap, n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hf.copy()
        else:
            # Try to project to anti-periodic and optimize
            h_ap = (best_h[:n_half] + (1.0 - best_h[n_half:])) / 2.0
            h_ap = np.clip(h_ap, 0, 1)
            if time_left() > 10:
                bounds = [(0.0, 1.0)] * n_half
                try:
                    res = minimize(obj_ap, h_ap, args=(n, 3000), jac=grad_ap,
                                  method='L-BFGS-B', bounds=bounds,
                                  options={'maxiter': 2000, 'ftol': 1e-15})
                    h_ap2 = np.clip(res.x, 0, 1)
                    hf = antiper_to_full(h_ap2, n)
                    c5 = compute_c5(hf, n)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hf.copy()
                        h_ap = h_ap2
                except Exception:
                    pass
                if time_left() > 10:
                    h_ap, c5 = coord_descent_ap(h_ap, n, max_rounds=300, rng=rng)
                    hf = antiper_to_full(h_ap, n)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = hf.copy()

    # Phase 4: Fine coordinate descent on best solution
    if best_h is not None and time_left() > 15:
        n = best_n
        h = best_h.copy()
        best = best_c5
        n_half = n // 2
        is_ap = np.max(np.abs(h[n_half:] - (1.0 - h[:n_half]))) < 0.05
        if is_ap:
            h_ap = h[:n_half].copy()
            for it in range(20000):
                if time_left() < 5:
                    break
                j = rng.randint(n_half)
                for delta in [0.001, -0.001, 0.003, -0.003, 0.007, -0.007]:
                    h_ap[j] += delta
                    if h_ap[j] < 0 or h_ap[j] > 1:
                        h_ap[j] -= delta
                        continue
                    val = compute_c5(antiper_to_full(h_ap, n), n)
                    if val < best - 1e-13:
                        best = val
                        break
                    else:
                        h_ap[j] -= delta
            h = antiper_to_full(h_ap, n)
            if best < best_c5:
                best_c5 = best
                best_h = h.copy()
        else:
            for it in range(30000):
                if time_left() < 5:
                    break
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                for delta in [0.002, -0.002, 0.005, -0.005]:
                    h[i] += delta
                    h[j] -= delta
                    if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
                        h[i] -= delta
                        h[j] += delta
                        continue
                    val = compute_c5(h, n)
                    if val < best - 1e-13:
                        best = val
                        break
                    else:
                        h[i] -= delta
                        h[j] += delta
            if best < best_c5:
                best_c5 = best
                best_h = h.copy()

    # Final SLSQP polish
    if best_h is not None and time_left() > 8:
        n = best_n
        n_half = n // 2
        h_ap = best_h[:n_half].copy()
        if np.max(np.abs(best_h[n_half:] - (1.0 - h_ap))) < 0.05:
            bounds = [(0.0, 1.0)] * n_half
            try:
                res = minimize(obj_ap, h_ap, args=(n, 5000), jac=grad_ap,
                              method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-12})
                h_ap = np.clip(res.x, 0, 1)
                hf = antiper_to_full(h_ap, n)
                c5 = compute_c5(hf, n)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hf.copy()
            except Exception:
                pass

    # Ensure feasibility
    if best_h is None:
        best_h = np.full(80, 0.5)
        best_n = 80
    best_h = project_sum(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
