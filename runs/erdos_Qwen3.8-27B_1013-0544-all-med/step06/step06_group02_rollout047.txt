Actually: corr[k] = h[j]*(1-h[j+d]) - h[j-d]*h[j] + C_k = h[j]*(1 - h[j+d] - h[j-d]) + C_k

This is LINEAR in h[j]! (Since h[j+d] and h[j-d] are fixed when varying h[j].)

So for each k, corr[k] is affine in h[j]. The max of affine functions is convex in h[j]. So for coordinate descent, the optimal h[j] for a given direction is at a boundary (0 or 1) or at a kink point where two corr values are equal.

This is interesting! It means we can do exact coordinate descent:
- For each j, the objective (max of affine functions) is convex in h[j]
- The minimum over h[j] ∈ [0,1] (with the constraint that the sum stays at N/2) is at h[j]=0, h[j]=1, or at a point where two corr values are equal.

But the constraint sum(h) = N/2 complicates things. If we fix all other coordinates, changing h[j] by δ changes the sum by δ, so we'd need to change another coordinate by -δ. This makes it a 2-coordinate problem at minimum.

Actually, for a simpler approach: we can do "swap" moves. Pick two indices i, j with h[i] < h[j], and try transferring some mass from i to j (or vice versa). This maintains the sum constraint.

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, lag_idx):
    N = len(h)
    d = lag_idx - (N - 1)
    g = np.zeros(N)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[:N - d] = 1.0 - h[d:]
        g[d:] -= h[:N - d]
    else:
        d2 = -d
        g[d2:] += 1.0 - h[:N - d2]
        g[:N - d2] -= h[d2:]
    return g

def obj_val(h):
    N = len(h)
    dx = 2.0 / N
    corr = compute_corr(h)
    return corr.max() * dx, corr, dx

def top_k_grad(h, corr, k=5):
    N = len(h)
    # Find top-k lags by correlation value
    top_indices = np.argsort(corr)[-k:]
    g = np.zeros(N)
    for idx in top_indices:
        g += grad_lag(h, idx)
    return g / k

def soft_top_grad(h, corr, alpha=20.0, k=10):
    N = len(h)
    # Use softmax over top-k lags
    c = corr - corr.max()
    w = np.exp(alpha * c)
    # Only keep top k
    top_indices = np.argsort(c)[-k:]
    w_masked = np.zeros_like(w)
    w_masked[top_indices] = w[top_indices]
    wsum = w_masked.sum()
    if wsum < 1e-300:
        return grad_lag(h, int(np.argmax(corr)))
    w_masked /= wsum
    g = np.zeros(N)
    for idx in top_indices:
        if w_masked[idx] > 1e-10:
            g += w_masked[idx] * grad_lag(h, idx)
    return g

def interpolate_n(h_src, n_dst):
    x0 = np.linspace(0.0, 1.0, len(h_src))
    x1 = np.linspace(0.0, 1.0, n_dst)
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

    val, corr, dx = obj_val(h)
    try_update(h, val)
    best_val = val

    v = np.zeros(N)
    lr = 0.04
    stagnant = 0

    for it in range(max_iter):

            break

            val = val_new

def optimize_fine(h, N, rng, t_end, try_update, max_iter=100000):

    val0, _, _ = obj_val(h)

        h2 = h.copy()
        h2[start:start + block_len] = 1.0 - h2[start:start + block_len]

            return h2, v2
    # Try small shift

    h2 = np.roll(h, shift)
    h2 = project_feasible(h2, N)
    v2, _, _ = obj_val(h2)
