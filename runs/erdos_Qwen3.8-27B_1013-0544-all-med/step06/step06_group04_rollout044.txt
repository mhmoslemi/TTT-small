import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import logsumexp

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x, t, n, pen):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    sse = logsumexp(t * corr) / t
    return sse + pen * (np.sum(x) - 0.5 * n) ** 2

def optimize_smooth(h, t, max_iter=500, pen=1e6):
    n = len(h)
    h0 = h.copy()
    res = minimize(lambda x: smooth_obj(x, t, n, pen), h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
    h = project_feasible(res.x, 0.5 * n)
    return h

def power_mean_obj(x, p, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    return np.sum(corr ** p) / len(corr) * (len(corr)) ** (1.0 / p)

def optimize_power(h, p=20, max_iter=300):
    n = len(h)
    res = minimize(lambda x: power_mean_obj(x, p, n) + 1e6 * (np.sum(x) - 0.5*n)**2,
                   h.copy(), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return project_feasible(res.x, 0.5 * n)
