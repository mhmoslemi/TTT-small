        g += wc * compute_grad_lag(h, k, N)
    return g

def optimize(h, N, rng, t_end, alpha=10.0, k_top=4, max_iter=50000, lr_init=0.04):
    val, corr, dx = obj_and_corr(h)
    best_local = val
    best_local_h = h.copy()
    lr = lr_init
    stagnant = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        if stagnant > 70:
            break
        g = soft_grad_top_k(h, corr, alpha=alpha, k_top=k_top)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            stagnant += 5
            continue
        best_trial_val = val
        best_trial_h = None
        for lr_try in [lr, lr * 0.5, lr * 2.0]:
            h_new = project_feasible(h - lr_try * g * dx, N)
            val_new, _, _ = obj_and_corr(h_new)
            if val_new < best_trial_val - 1e-14:
                best_trial_val = val_new
                best_trial_h = h_new
                if val_new < val - 1e-14:
                    break
        if best_trial_h is not None:
            h = best_trial_h
            val = best_trial_val
            _, corr, _ = obj_and_corr(h)
        if val < best_local - 1e-14:
            best_local = val
            best_local_h = h.copy()
            stagnant = 0
            lr = min(lr * 1.05, 0.15)
        else:
            stagnant += 1
            lr *= 0.82
        if stagnant > 30:
            scale = 0.05 + 0.15 * min(stagnant / 70.0, 1.0)
            h = project_feasible(h + rng.randn(N) * scale, N)
            val, corr, _ = obj_and_corr(h)
            if val < best_local:
                best_local = val
                best_local_h = h.copy()
            lr = lr_init
            stagnant = 0
    return best_local_h, best_local

def interpolate_to_n(h_src, n_dst):
    x0 = np.linspace(0.0, 1.0, len(h_src))
    x1 = np.linspace(0.0, 1.0, n_dst)
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 12
    best_h = None
    best_val = float('inf')
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None
    if h_init is not None:
        v_init = obj_and_corr(h_init)[0]
        best_h = h_init.copy()
        best_val = v_init
    # Phase 1: Initial refinement with adaptive alpha
    if best_h is not None and time.time() < t_end - 200:
        N0 = len(best_h)
        # Sub-phase 1a: smooth phase (low alpha)
        t1 = min(time.time() + 80, t_end - 180)
        h1a, v1a = optimize(best_h.copy(), N0, rng, t1, alpha=5.0, k_top=6, max_iter=60000, lr_init=0.06)
        if v1a < best_val:
            best_val = v1a
            best_h = h1a
        # Sub-phase 1b: focused phase (higher alpha)
        if time.time() < t_end - 120:
            t2 = min(time.time() + 100, t_end - 100)
            h1b, v1b = optimize(best_h.copy(), N0, rng, t2, alpha=15.0, k_top=3, max_iter=80000, lr_init=0.03)
            if v1b < best_val:
                best_val = v1b
                best_h = h1b
    # Phase 2: Multi-resolution exploration
    if best_h is not None and time.time() < t_end - 80:
        n_candidates = [50, 70, 90, 110, 130, 150, 180, 240]
        for n_t in n_candidates:
            if time.time() > t_end - 50:
                break
            h_start = interpolate_to_n(best_h, n_t)
            t_phase = min(time.time() + 55, t_end - 50)
            h_t, v_t = optimize(h_start, n_t, rng, t_phase, alpha=10.0, k_top=4, max_iter=30000, lr_init=0.05)
            if v_t < best_val:
                best_val = v_t
                best_h = h_t
    # Phase 3: Diverse restarts at best resolution
    if best_h is not None and time.time() < t_end - 30:
        N_best = len(best_h)
        n_restarts = 0
        while time.time() < t_end - 15 and n_restarts < 60:
            n_restarts += 1
            # Vary perturbation scale and type
            rtype = n_restarts % 4
            if rtype == 0:
                scale = 0.03
                h_start = project_feasible(best_h + rng.randn(N_best) * scale, N_best)
            elif rtype == 1:
                scale = 0.08
                h_start = project_feasible(best_h + rng.randn(N_best) * scale, N_best)
            elif rtype == 2:
                # Structured perturbation: shift-like
                shift = rng.randint(2, max(3, N_best // 10))
                h_start = project_feasible(np.roll(best_h, shift), N_best)
                h_start = project_feasible(h_start + rng.randn(N_best) * 0.03, N_best)
            else:
                # Block perturbation
                nb = N_best // 4
                h_start = best_h.copy()
                h_start[rng.randint(0, N_best - nb, size=1).tolist()][0]:h_start.size * 0 + nb] = 1.0 - best_h[nb:]
                h_start = project_feasible(h_start + rng.randn(N_best) * 0.02, N_best)
            t_phase = min(time.time() + 18, t_end - 15)
            h_r, v_r = optimize(h_start, N_best, rng, t_phase, alpha=12.0, k_top=3, max_iter=10000, lr_init=0.04)
            if v_r < best_val:
                best_val = v_r
                best_h = h_r
    # Phase 4: Final fine-tuning
    if best_h is not None and time.time() < t_end - 2:
        N_best = len(best_h)
        h = best_h.copy()
        val, corr, dx = obj_and_corr(h)
        lr = 0.006
        for it in range(40000):
            if time.time() > t_end:
                break
            g = soft_grad_top_k(h, corr, alpha=30.0, k_top=3)
            gn = np.linalg.norm(g)
            if gn < 1e-12:
                break
            h_new = project_feasible(h - lr * g * dx, N_best)
            val_new, corr_new, _ = obj_and_corr(h_new)
            if val_new < val - 1e-14:
                h = h_new
                val = val_new
                corr = corr_new
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
                lr = min(lr * 1.01, 0.03)
            else:
                lr *= 0.9
    # Phase 5: One more pass at a different resolution if time permits
    if best_h is not None and time.time() < t_end - 20:
        N_alt = len(best_h) + (len(best_h) // 3)
        if N_alt <= 400:
            h_alt = interpolate_to_n(best_h, N_alt)
            t_alt = time.time() + min(40, max(5, t_end - time.time() - 5))
            h_a, v_a = optimize(h_alt, N_alt, rng, t_alt, alpha=10.0, k_top=4, max_iter=20000, lr_init=0.05)
            if v_a < best_val:
                best_val = v_a
                best_h = h_a
    if best_h is None:
        best_h = 0.5 * np.ones(120)
        best_val = obj_and_corr(best_h)[0]
    return best_h, best_val, len(best_h)
