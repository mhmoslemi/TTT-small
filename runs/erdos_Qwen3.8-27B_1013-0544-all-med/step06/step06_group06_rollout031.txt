        # S2[i] += wi * h[i+k] if 0<=i+k<n
        if k >= 0:

            # i-k < n means i < n+k (always true since i < n and k >= 0)
            S1[k:] += wi * (1-h[:n-k])
            # i+k < n means i < n-k
            S2[:n-k] += wi * h[k:]
        else:

            # i-k < n means i < n+k = n-kk
            S1[:n-kk] += wi * (1-h[kk:])
            # i+k < n means i < n-k = n+kk, always true
            # i+k >= 0 means i >= -k = kk
            S2[kk:] += wi * h[:n-kk]
    
    return dx * (S1 - S2)

    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def corr_values(h, n):
    return np.correlate(h, 1 - h, mode='full') * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        d = target - h.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            r = 1.0 - h
            t = r.sum()
            if t < 1e-14:
                break
            h += d * r / t
        else:
            t = h.sum()
            if t < 1e-14:
                break
            h += d * h / t
    return np.clip(h, 0.0, 1.0)

def smax(c, alpha):
    m = c.max()
    x = (c - m) * alpha
    return m + (x.max() + np.log(np.sum(np.exp(x - x.max())))) / alpha

def smax_grad(h, n, alpha):
    c = corr_values(h, n)
    m = c.max()
    x = (c - m) * alpha
    w = np.exp(x - x.max())
    w /= w.sum()
    dx = 2.0 / n
    S1 = np.zeros(n)

            kk = -k
            S1[:n - kk] += wi * (1.0 - h[kk:])
            S2[kk:] += wi * h[:n - kk]
