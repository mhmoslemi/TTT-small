    n = x.size

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(60):
        diff = target - float(x.sum())
        if abs(diff) <= 1e-14:
            break
        if diff > 0.0:
            slack = 1.0 - x
            mask = slack > 1e-14
            if not np.any(mask):
                break
            x[mask] += diff * slack[mask] / float(slack[mask].sum())
        else:
            mask = x > 1e-14
            if not np.any(mask):
                break
            x[mask] += diff * x[mask] / float(x[mask].sum())
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - float(x.sum())
    if abs(diff) > 1e-10:
        k = int(np.floor(target))
        x[:] = 0.0
        if k > 0:
            x[:k] = 1.0
        rem = target - float(k)
        if rem > 0.0 and k < n:
            x[k] = rem
    return x

def c_values(h):
    n = h.size
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))

def exact_c(h):
    return float(np.max(c_values(h)))

def active_grad(x, active):
    n = x.size
    dx = 2.0 / float(n)

    for k in active:
        lag = int(k) - (n - 1)
        L = abs(lag)
        if L == 0:
            grad += (1.0 - 2.0 * x)
        else:
            g = np.zeros(n)
            if lag > 0:
                g[:n - L] = 1.0
            else:
                g[L:] = 1.0
            g[:n - L] -= x[L:n]
            g[L:] -= x[:n - L]
            grad += g
    return grad * dx / max(1, len(active))

def smooth_obj(x, tau, topk):
    c = c_values(x)
    cm = float(np.max(c))
    m = c.size
    if tau <= 0:
        w = np.zeros(m)
        w[int(np.argmax(c))] = 1.0
        return cm, w
    z = (c - cm) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(m)
        w[int(np.argmax(c))] = 1.0
        return cm, w
    w = e / s
    if topk is not None and 0 < topk < m:
        idx = np.argsort(w)[::-1][:topk]
        w2 = np.zeros(m)
        w2[idx] = w[idx]
        ss = w2.sum()
        if ss > 0:
            w = w2 / ss
    return cm + tau * np.log(s), w

def smooth_grad(x, w):
    n = x.size
    dx = 2.0 / float(n)
    grad = np.zeros(n)
    for k in np.nonzero(w > 1e-14)[0]:
        wk = float(w[k])
        lag = int(k) - (n - 1)
        L = abs(lag)
        if L == 0:
            grad += wk * (1.0 - 2.0 * x)
        else:
            g = np.zeros(n)
            if lag > 0:
                g[:n - L] = 1.0
            else:
                g[L:] = 1.0
            g[:n - L] -= x[L:n]
            g[L:] -= x[:n - L]
            grad += wk * g
    return grad * dx

    n = x0.size

        v, _ = smooth_obj(x, tau, topk)
        return v

        return smooth_grad(x, w)
