import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def compute_c5_array(h):
    """Compute all correlation values for analysis."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def project_feasible(h, n):
    """Project onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_grad(h, t):
    """Soft-max objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    obj = t * (m / t + np.log(np.sum(exps))) * dx
    
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        # grad of conv[idx] w.r.t. h
        if shift >= 0:
            if shift < n:
                grad[shift:] += w * (1.0 - h[:n - shift])
            if shift > 0:
                grad[:n - shift] -= w * h[shift:]
        else:
            s = -shift
            if s < n:
                grad[:n - s] += w * (1.0 - h[s:])
            if s > 0:
                grad[s:] -= w * h[:n - s]
    return obj, grad * dx

def nesterov_pgd(h_init, budget_s, t_start, t_end, t_decay):

    n = len(h_init)
    h = h_init.astype(float).copy()

    best_h = h.copy()

    t0 = time.time()
    
    k = 1.0
    lr = 0.01
    momentum = 0.9
    
    while time.time() - t0 < budget_s:
        t = max(t_end, t_start * (t_end / t_start) ** (k / (budget_s * 100)))
        # Actually, let's use a simpler decay
        t = max(t_end, t_start * np.exp(-3.0 * k / (budget_s * 50)))

        y_new = h + momentum * (y - h)
        h_new = project_feasible(y_new - lr * grad, n)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Adaptive step size

            lr *= 0.5
            if lr < 1e-10:
                lr = 1e-10
        else:
            lr = min(lr * 1.05, 0.1)
        
        h = h_new
        y = y_new
        k += 1
    
    return best_h, best_c5

Another trick: **coordinate-wise optimization**. Instead of updating all coordinates simultaneously, update them one at a time (or in small groups). For each coordinate h[i], the optimal value (given all other coordinates) can be found in closed form because the objective is a max of linear functions in h[i].

Actually, for a fixed shift s, the contribution to the correlation from h[i] is:
- If i is the "first" index: h[i] * (1-h[i+s]) → linear in h[i]
- If i is the "second" index: h[i-s] * (1-h[i]) → linear in h[i]

So for a fixed shift s, the contribution is linear in h[i]. The max over shifts is a max of linear functions, which is convex. So for each h[i], the optimal value (minimizing the max) is where the "active" constraints balance.

This gives a **block coordinate descent** approach:
- For each i, fix all other coordinates.
- The objective becomes max_s (a_s * h[i] + b_s) where a_s and b_s depend on the other coordinates.
- The optimal h[i] is the value that minimizes this max, which is where the maximum of the lines a_s * h[i] + b_s is minimized.
- This is a 1D convex optimization problem (minimize max of linear functions).

The solution is: h[i]* = -b* / a* where (a*, b*) is the "balance point" of the lines. Specifically, we want to find h[i] such that at least two lines achieve the same maximum value, and all other lines are below.

This is essentially a 1D linear programming problem, which can be solved in O(n) time (by finding the upper envelope of lines).
