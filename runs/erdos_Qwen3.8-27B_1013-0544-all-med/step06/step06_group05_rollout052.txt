import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def binarize(h, n, rng):
    h = np.round(h).astype(float).copy()
    t = n // 2
    s = int(h.sum())
    if s > t:
        idx = rng.choice(np.flatnonzero(h==1), s-t, replace=False)
        h[idx] = 0
    elif s < t:
        idx = rng.choice(np.flatnonzero(h==0), t-s, replace=False)
        h[idx] = 1
    return h
