def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_max_val(corr, temp):
    m = np.max(corr)
    if temp < 1e-10:
        return m
    return m + np.log(np.sum(np.exp(temp * (corr - m)))) / temp

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            if d > 0:
                idx = np.argmin(h)
                h[idx] = min(1.0, h[idx] + d)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + d)
        else:
            step = d / mask.sum()
            h[mask] += step
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, corr, temp):
    n = len(h)
    dx = 2.0 / n
    m = np.max(corr)
    ex = np.exp(temp * (corr - m))
    weights = ex / ex.sum()
    c_rev = 1.0 - h[::-1]
    T1 = np.correlate(weights, c_rev, 'full')[:n]
    T2 = np.correlate(weights, h, 'full')[n-1::-1][:n]
    return dx * (T1 - T2)

def gradient_descent(h, temp, n_steps=500, lr=0.01, push_binary=0.0):
    n = len(h)
    target = n / 2.0
    corr = compute_corr(h)
    for step in range(n_steps):
        grad = compute_gradient(h, corr, temp)
        h = h - lr * grad
        # Push toward binary
        if push_binary > 0:
            h = h + push_binary * (np.where(h > 0.5, 0.5, -0.5))
        h = project_feasible(h, target)
        corr = compute_corr(h)
    return h
