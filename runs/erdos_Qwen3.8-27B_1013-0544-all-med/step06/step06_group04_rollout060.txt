```python
    target = n / 2.0

    for _ in range(100):

        if diff > 0:

            amount = min(diff, np.sum(h[mask]))

    h = np.clip(h, 0.0, 1.0)

    diff = np.sum(h) - target
```

```python
def compute_grad(h, weights):
    """Compute gradient of weighted sum of correlations.
    weights: array of length 2n-1 (softmax weights for each lag)
    Returns: gradient array of length n
    """
    n = len(h)
    grad = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k]
        if w < 1e-10:
            continue
        L = k - (n - 1)
        if L >= 0:
            # grad[m] gets (1-h[m-L]) for m in [L, n-1]
            grad[L:] += w * (1.0 - h[:n-L])
            # grad[m] gets -h[m+L] for m in [0, n-1-L]
            grad[:n-L] -= w * h[L:]
        else:
            Lp = -L
            # grad[m] gets (1-h[m+Lp]) for m in [0, n-1-Lp]
            grad[:n-Lp] += w * (1.0 - h[Lp:])
            # grad[m] gets -h[m-Lp] for m in [Lp, n-1]
            grad[Lp:] -= w * h[:n-Lp]
    return grad
```

```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(cor))

def project_onto_constraint(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    return h

def upsample(h, new_n):
    n = len(h)
    dx_old = 2.0 / n
    dx_new = 2.0 / new_n
    x_old = (np.arange(n) + 0.5) * dx_old
    x_new = (np.arange(new_n) + 0.5) * dx_new
    new_h = np.interp(x_new, x_old, h)
    return project_onto_constraint(new_h, new_n)

def compute_grad_fast(h, weights):
    n = len(h)
    conv1 = np.convolve(h, weights, 'full')
    B = conv1[n-1:2*n-1]
    W_rev = weights[::-1]
    conv2 = np.convolve(h, W_rev, 'full')
    C = conv2[n-1:2*n-1]
    cs = np.concatenate([[0.0], np.cumsum(weights)])
    A = cs[n:2*n] - cs[0:n]
    return A - B - C

def gradient_descent(h, n, time_limit, lr_init=0.4, beta=40.0, max_iters=50000):
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    lr = lr_init
    stuck = 0
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_c = np.max(corr)
        exp_vals = np.exp(beta * (corr - max_c))
        w_sum = np.sum(exp_vals)
        if w_sum < 1e-300:
            break
        weights = exp_vals / w_sum
        grad = compute_grad_fast(h, weights) * dx
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stuck = 0
        else:
            h = h_new
            stuck += 1
            if stuck % 20 == 0:
                lr *= 0.92
            if stuck > 300:
                break
    return best_h, best_c5

def binary_hill_climb(h, n, time_limit, rng, max_rounds=500000, all_pairs=True):
    h = (h > 0.5).astype(float)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_move_c5 = best_c5
        best_move = None
        if all_pairs and n <= 100:
            for i in ones:
                for j in zeros:
                    h[i] = 0.0
                    h[j] = 1.0
                    c5_t = evaluate_c5(h)
                    if c5_t < best_move_c5 - 1e-14:
                        best_move_c5 = c5_t
                        best_move = (i, j)
                    h[i] = 1.0
                    h[j] = 0.0
        else:
            n_trials = min(300, len(ones) * len(zeros))
            for _ in range(n_trials):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                c5_t = evaluate_c5(h)
                if c5_t < best_move_c5 - 1e-14:
                    best_move_c5 = c5_t
                    best_move = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        if best_move is not None:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
            best_c5 = best_move_c5
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                # Random restart perturbation
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                c5_t = evaluate_c5(h)
                if c5_t > best_c5 + 0.05:
                    h[i] = 1.0
                    h[j] = 0.0
                no_improve = 0
                if rnd > 20000:
                    break
    return best_h, best_c5
```
