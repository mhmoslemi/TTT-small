Let me also think about a different approach to the gradient computation. Instead of looping over active shifts, I can compute the gradient more efficiently.

The gradient of the soft-max objective at h is:
∇f(h) = ∑_k w_k ∇c_k(h)

where ∇c_k(h) is the gradient of the k-th correlation.

Now, ∇c_k(h) can be written as:
(∇c_k(h))[j] = (1-h[j-k]) * 1[0≤j-k<n] - h[j+k] * 1[0≤j+k<n]

So ∇f(h)[j] = ∑_k w_k [(1-h[j-k]) * 1[0≤j-k<n] - h[j+k] * 1[0≤j+k<n]]
= ∑_k w_k (1-h[j-k]) * 1[0≤j-k<n] - ∑_k w_k h[j+k] * 1[0≤j+k<n]

Let me define:
A[j] = ∑_k w_k (1-h[j-k]) * 1[0≤j-k<n]
B[j] = ∑_k w_k h[j+k] * 1[0≤j+k<n]

Then ∇f(h)[j] = A[j] - B[j].

A[j] = ∑_{k: 0≤j-k<n} w_k - ∑_{k: 0≤j-k<n} w_k h[j-k]
= ∑_{k: j-n<k≤j} w_k - ∑_{k: j-n<k≤j} w_k h[j-k]

Let m = j-k, so k = j-m:
A[j] = ∑_{m: 0≤m<n} w_{j-m} - ∑_{m: 0≤m<n} w_{j-m} h[m]

The first term is a convolution of w with 1's (truncated). The second term is a convolution of w and h.

Similarly:
B[j] = ∑_{k: 0≤j+k<n} w_k h[j+k]
Let m = j+k, so k = m-j:
B[j] = ∑_{m: j≤m<n} w_{m-j} h[m]

This is also a convolution-like operation.

So ∇f(h) = (convolution of w with 1's, truncated) - (convolution of w and h, truncated) - (convolution of w^R and h, truncated)

where w^R is the reversed weight vector.

This can be computed efficiently using FFT! The overall complexity is O(n log n) instead of O(n * m).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode="full")) * (2.0 / n)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0, 1)

def grad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        m = n - k
        if m > 0:
            g[k:k+m] += 1.0 - h[:m]
            g[:m] -= h[k:k+m]
    else:
        s = -k
        m = n - s
        if m > 0:
            g[s:s+m] += 1.0 - h[:m]
            g[:m] -= h[s:s+m]
    return g

def softmax_obj_grad(h, t, n_act=None):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    m = conv.max()
    exps = np.exp((conv - m) / t)
    ws = exps.sum()
    obj = m + t * np.log(ws)
    if n_act is not None and n_act < len(conv):
        top = np.argsort(conv)[-n_act:]
    else:
        top = np.arange(len(conv))
    w = exps[top]
    wn = w / w.sum()
    g = np.zeros(n)
    for j in range(len(top)):
        if wn[j] > 1e-10:
            g += wn[j] * grad_shift(h, top[j] - (n - 1))
    return obj, g * dx

def fista(h0, n, t, max_iter, n_act=30, base_step=None):
    h = h0.copy()
    hp = h0.copy()
    tk = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    if base_step is None:
        base_step = t / (2.0 * n)
    for it in range(max_iter):
        tnew = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = h + ((tk - 1.0) / tnew) * (h - hp)
        y = np.clip(y, 0, 1)
        obj, g = softmax_obj_grad(y, t, n_act)
        h_new = project_feasible(y - base_step * g, n)
        hp = h
        h = h_new
        tk = tnew
        if (it + 1) % 200 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def nesterov_sg(h0, n, max_iter, n_act=5, step_scale=0.5):
    h = h0.copy()
    hp = h0.copy()
    tk = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    for it in range(max_iter):
        tnew = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        y = h + ((tk - 1.0) / tnew) * (h - hp)
        y = np.clip(y, 0, 1)
        conv = np.correlate(y, 1.0 - y, mode="full")
        top = np.argsort(conv)[-n_act:]
        g = np.zeros(n)
        for idx in top:
            g += grad_shift(y, idx - (n - 1))
        g *= dx / n_act
        step = step_scale / np.sqrt(it + 1.0)
        h_new = project_feasible(y - step * g, n)
        hp = h
        h = h_new
        tk = tnew
        if (it + 1) % 500 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def binary_sa(h0, n, max_iter=50000, seed=42):
    rng = np.random.RandomState(seed)
    h = (h0 > 0.5).astype(float)
    nt = n // 2
    while h.sum() > nt:
        h[rng.choice(np.where(h == 1)[0])] = 0
    while h.sum() < nt:
        h[rng.choice(np.where(h == 0)[0])] = 1
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    for it in range(max_iter):
        t = 1.0 * (0.01) ** (it / max_iter)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i], h[j] = 0.0, 1.0
        nc5 = compute_c5(h)
        if nc5 < cur_c5 or rng.random() < np.exp(-(nc5 - cur_c5) / max(t, 1e-6)):
            cur_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:

    best_c5 = compute_c5(best_h)
    
    def tl():

    for t_val in [0.05, 0.02]:
        if tl() < 60:
            break
        def obj_fn(h, tv=t_val, nv=n0):
            o, _ = softmax_obj_grad(h, tv)

        def grad_fn(h, tv=t_val, nv=n0):

            ht = project_feasible(res.x, n0)

        ht, c5t = fista(best_h, n0, t_val, max_it, n_act=40)
