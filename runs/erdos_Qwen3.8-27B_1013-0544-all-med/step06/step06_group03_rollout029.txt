from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        if shift > 0:
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def soft_max_obj(h, t):
    n = len(h)

    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    # Use log-sum-exp trick
    shifted = (conv - m) / t
    max_s = np.max(shifted)
    return t * (m / t + max_s + np.log(np.sum(np.exp(shifted - max_s)))) * dx

    grad = np.zeros(n)

        # d(k) = d_i(k) - d_j(k)

        d_vec = np.zeros(2*n - 1)
        for k in range(2*n - 1):

            if 0 <= i + s < n:

            if 0 <= i - s < n:

            dj = 0.0
            if 0 <= j + s < n:
                dj += (1.0 - h[j + s])
            if 0 <= j - s < n:
                dj -= h[j - s]
            d_vec[k] = di - dj

        # subject to h[i]+δ ∈ [0,1], h[j]-δ ∈ [0,1]

            m2 = hi - (hi - lo) / 3
            f1 = np.max(conv + m1 * d_vec)
            f2 = np.max(conv + m2 * d_vec)
            if f1 < f2:
                hi = m2
            else:

            h[j] -= delta_opt
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
    
    return h, best_c5

min t
s.t. f_k(h) ≤ t for all k

     0 ≤ h ≤ 1

where f_k(h) = ∑_i h_i (1 - h_{i+k})

This is a bilinear (not convex) program. However, we can linearize it using the substitution: let z_{i,k} = h_i * h_{i+k}. Then:
f_k(h) = ∑_i h_i - ∑_i z_{i,k} = n/2 - ∑_i z_{i,k}

Constraints:
- z_{i,k} = h_i * h_{i+k} (bilinear, not linear)
- 0 ≤ h_i ≤ 1
- sum(h) = n/2

The bilinear constraint makes this non-convex.

But we can use the McCormick envelopes: for 0 ≤ a ≤ 1, 0 ≤ b ≤ 1:
max(a+b-1, 0) ≤ ab ≤ min(a, b)

So we can write:
z_{i,k} ≤ h_i
z_{i,k} ≤ h_{i+k}
z_{i,k} ≥ h_i + h_{i+k} - 1
z_{i,k} ≥ 0

    x = h_init.copy()
    best_c5 = compute_c5(x)
    best_h = x.copy()
    lr = lr_init

def pair_cd(h_init, n_rounds=500, seed=42):
    n = len(h_init)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    
    for r in range(n_rounds):
        i, j = rng.choice(n, 2, replace=False)
        conv = np.correlate(h, 1.0 - h, mode="full")

            di = 0.0
            if 0 <= i + s < n:

            m2 = hi - (hi - lo) / 3.0
            if np.max(conv + m1 * d_vec) < np.max(conv + m2 * d_vec):

            h[j] -= delta

Wait, I need to double-check the pair_cd function. The d_vec computation is correct: when we change h[i] by +δ and h[j] by -δ, the change in conv[k] is δ * (contribution of h[i] to conv[k] - contribution of h[j] to conv[k]).

The contribution of h[i] to conv[k] (where s = k-(n-1)):
- As the "first factor": h[i] * (1-h[i+s]) → derivative w.r.t. h[i] is (1-h[i+s]) if 0≤i+s<n

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        if shift > 0:
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    shifted = (conv - m) / t
    ms = np.max(shifted)
    exps = np.exp(shifted - ms)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return grad * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def fista_softmax(h_init, max_iter, t, L):
    n = len(h_init)
    x = h_init.copy()
    y = h_init.copy()
    best_c5 = compute_c5(x)
    best_h = x.copy()
    for it in range(max_iter):
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        grad = soft_max_grad(y, t)
        z = project_feasible(y - grad / L, n)
        if (it + 1) % 200 == 0:
            c5 = compute_c5(z)
            if c5 < best_c5:
                best_c5 = c5
                best_h = z.copy()
        new_y = z + ((beta - 1.0) / (beta + 1.0)) * (z - x)
        new_y = np.clip(new_y, -1.0, 2.0)
        x = z.copy()
        y = new_y

            if it > 5000:

    c5 = compute_c5(x)
    if c5 < best_c5:
        best_c5 = c5
        best_h = x.copy()
    return best_h, best_c5

def lfbgsm(h_init, n, t_val, max_iter=3000):
    def obj(h):
        conv = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(conv)
        shifted = (conv - m) / t_val
        ms = np.max(shifted)
        return t_val * (m / t_val + ms + np.log(np.sum(np.exp(shifted - ms)))) * (2.0 / n) + 100.0 * (np.sum(h) - n / 2.0) ** 2
    def grad(h):
        gx = soft_max_grad(h, t_val)
        return gx + 200.0 * (np.sum(h) - n / 2.0)
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
        ht = project_feasible(res.x, n)
        return ht
    except Exception:
        return h_init.copy()

            best_h, best_c5 = ht, c5t
    if tl() > 15:
        ht = lfbgsm(best_h, len(best_h), 0.001, max_iter=2000)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
