    g = np.zeros(N)
    for j in idx:
        n_lag = j - offset
        g += grad_lag(h, n_lag)
    return g * dx / top_k

def compute_grad_soft(h, corr, dx, alpha=20.0):
    N = len(h)
    offset = N - 1
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        return grad_lag(h, k_star - offset) * dx
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-8:
            continue
        g += w[j] * grad_lag(h, j - offset)
    return g * dx

def interpolate_to_n(h_src, n_dst):
    x0 = np.arange(len(h_src)) / len(h_src)
    x1 = np.arange(n_dst) / n_dst
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def optimize_h(h, N, rng, t_end, try_update, max_iter=50000):
    val, corr, dx = obj_and_corr(h)
    try_update(h, val)
    best_local = val
    best_local_h = h.copy()
    lr = 0.025
    momentum = 0.8
    v_mom = np.zeros(N)
    stagnant = 0
    top_k = 3

    for it in range(max_iter):
        if time.time() > t_end:
            break
        if stagnant % 50 == 0 and stagnant > 0:
            top_k = min(3 + stagnant // 50, 10)
        g = compute_grad(h, corr, dx, top_k)
        v_mom = momentum * v_mom - lr * g
        h_new = project_feasible(h + v_mom, N)
        val_new, corr_new, _ = obj_and_corr(h_new)
        if val_new < val - 1e-14:
            h = h_new
            val = val_new
            corr = corr_new
            stagnant = 0
            lr = min(lr * 1.008, 0.12)
            if val < best_local:
                best_local = val
                best_local_h = h.copy()
        else:
            stagnant += 1
            v_mom *= 0.5
            lr *= 0.82
            if stagnant > 40:
                r = stagnant % 5
                if r == 0:
                    scale = 0.02 + 0.04 * min((stagnant - 40) / 80.0, 1.5)
                    h = project_feasible(h + rng.randn(N) * scale, N)
                elif r == 1:
                    k_star = np.argmax(corr)
                    lag = k_star - (N - 1)
                    freq = abs(lag) / N * 2.0 * np.pi
                    pert = 0.08 * np.sin(np.arange(N) * freq + rng.uniform(0, 6.28))
                    h = project_feasible(h + pert, N)
                elif r == 2:
                    n_flip = max(2, N // 12)
                    idx = rng.choice(N, n_flip, replace=False)
                    h_try = h.copy()
                    h_try[idx] = 1.0 - h_try[idx]
                    h = project_feasible(h_try, N)
                elif r == 3:
                    block = max(2, N // 8)
                    start = rng.randint(0, N - block)
                    pert = np.zeros(N)
                    pert[start:start + block] = rng.choice([-1, 1]) * 0.12
                    h = project_feasible(h + pert, N)
                else:
                    h_binary = (h + rng.randn(N) * 0.25 > 0.5).astype(float)
                    h = project_feasible(h_binary, N)
                v_mom = np.zeros(N)
                val, corr, _ = obj_and_corr(h)
                lr = 0.025
                stagnant = 0
                if val < best_local:
                    best_local = val
                    best_local_h = h.copy()

    try_update(best_local_h, best_local)
    return best_local_h, best_local

def polish(h, N, t_end, try_update):
    val, corr, dx = obj_and_corr(h)
    try_update(h, val)
    lr = 0.003
    for it in range(50000):
        if time.time() > t_end:
            break
        g = compute_grad(h, corr, dx, top_k=3)
        h_new = project_feasible(h - lr * g, N)
        val_new, corr_new, _ = obj_and_corr(h_new)
        if val_new < val - 1e-15:
            h = h_new
            val = val_new
            corr = corr_new
            try_update(h, val)
        else:
            lr *= 0.92
            if lr < 1e-7:
                break
    return h, val

def soft_polish(h, N, t_end, try_update):
    val, corr, dx = obj_and_corr(h)
    try_update(h, val)
    lr = 0.005
    for it in range(30000):
        if time.time() > t_end:
            break
        g = compute_grad_soft(h, corr, dx, alpha=25.0)
        h_new = project_feasible(h - lr * g, N)
        val_new, corr_new, _ = obj_and_corr(h_new)
        if val_new < val - 1e-15:
            h = h_new
            val = val_new
            corr = corr_new
            try_update(h, val)
        else:
            lr *= 0.93
            if lr < 1e-7:
                break
    return h, val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    best_h = None
    best_val = float('inf')

    def try_update(h, val):
        nonlocal best_h, best_val
        if val < best_val:
            best_val = val
            best_h = h.copy()

    if h_init is not None:
        try_update(h_init.copy(), obj_and_corr(h_init)[0])

    # Phase 1: Refine initial at its native resolution
    if h_init is not None and time.time() < t_end - 200:
        N_init = len(h_init)
        t_phase1 = min(time.time() + 180, t_end - 200)
        h, v = optimize_h(h_init.copy(), N_init, rng, t_phase1, try_update, max_iter=80000)
        try_update(h, v)
        t_pol = min(time.time() + 30, t_end - 180)
        h, v = polish(h, N_init, t_pol, try_update)
        try_update(h, v)

    # Phase 2: Try multiple resolutions
    n_list = [50, 70, 90, 110, 130, 150]
    for n_t in n_list:
        if time.time() > t_end - 60:
            break
        if best_h is not None:
            h_start = interpolate_to_n(best_h, n_t)
        elif h_init is not None:
            h_start = interpolate_to_n(h_init, n_t)
        else:
            h_start = 0.5 * np.ones(n_t)
        t_phase = min(time.time() + 70, t_end - 30)
        h, v = optimize_h(h_start, n_t, rng, t_phase, try_update, max_iter=40000)
        try_update(h, v)

    # Phase 3: Diverse restarts on best resolution
    if best_h is not None and time.time() < t_end - 40:
        N_best = len(best_h)
        for restart in range(25):
            if time.time() > t_end - 20:
                break
            strategy = restart % 5
            if strategy == 0:
                scale = 0.03 + 0.02 * (restart % 10)
                h_start = project_feasible(best_h + rng.randn(N_best) * scale, N_best)
            elif strategy == 1:
                h_start = best_h.copy()
                n_flip = N_best // 10
                idx = rng.choice(N_best, n_flip, replace=False)
                h_start[idx] = 1.0 - h_start[idx]
                h_start = project_feasible(h_start, N_best)
            elif strategy == 2:
                freq = rng.choice([2, 3, 4, 5, 7])
                phase = rng.uniform(0, 2 * np.pi)
                pert = 0.06 * np.sin(freq * 2 * np.pi * np.arange(N_best) / N_best + phase)
                h_start = project_feasible(best_h + pert, N_best)
            elif strategy == 3:
                h_noisy = best_h + rng.randn(N_best) * 0.2
                h_bin = (h_noisy > 0.5).astype(float)
                h_start = project_feasible(h_bin + rng.randn(N_best) * 0.03, N_best)
            else:
                # Interpolate from a different resolution
                n_alt = N_best + rng.choice([-20, 20, -40, 40])
                if n_alt < 30:
                    n_alt = N_best
                h_interp = interpolate_to_n(best_h, n_alt)
                h_start = interpolate_to_n(h_interp, N_best)
                h_start = project_feasible(h_start + rng.randn(N_best) * 0.03, N_best)

            t_phase = min(time.time() + 35, t_end - 10)
            h, v = optimize_h(h_start, N_best, rng, t_phase, try_update, max_iter=20000)
            try_update(h, v)

    # Phase 4: Try lower resolutions for potential better local optima
    if best_h is not None and time.time() < t_end - 30:
        for n_low in [40, 50, 60]:
            if time.time() > t_end - 10:
                break
            h_low = interpolate_to_n(best_h, n_low)
            t_phase = min(time.time() + 30, t_end - 10)
            h, v = optimize_h(h_low, n_low, rng, t_phase, try_update, max_iter=25000)
            try_update(h, v)

    # Phase 5: Final polish
    if best_h is not None and time.time() < t_end - 2:
        N_best = len(best_h)
        h, v = polish(best_h.copy(), N_best, t_end, try_update)
        try_update(h, v)
        if time.time() < t_end - 2:
            h, v = soft_polish(h, N_best, t_end, try_update)
            try_update(h, v)

    if best_h is None:
        best_h = 0.5 * np.ones(120)
        best_val = obj_and_corr(best_h)[0]

    return best_h, best_val, len(best_h)
