import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def binarize(h, rng=None):
    N = len(h)
    target = N // 2
    if rng is not None:
        h_noisy = h + rng.uniform(-0.02, 0.02, N)
    else:
        h_noisy = h
    sorted_idx = np.argsort(h_noisy)[::-1]
    b = np.zeros(N)
    b[sorted_idx[:target]] = 1.0
    return b

def local_search_all_swaps(h, N, time_limit):
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = (corr.max() * 2.0 / N)
    for it in range(200):
        if time.time() > time_limit:
            break
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        best_delta = 0
        best_pair = None
        best_corr = None
        current_max = corr.max()
        for i in ones:
            h[i] = 0
            for j in zeros:
                h[j] = 1
                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = new_corr.max()
                delta = new_max - current_max
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (i, j)
                    best_corr = new_corr
                h[j] = 0
            h[i] = 1
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0
            h[j] = 1
            corr = best_corr
            c5 = corr.max() * 2.0 / N
            if c5 < best_c5:
                best_c5 = c5
        else:
            break
    return h, best_c5

def simulated_annealing(h, N, t_end, rng, t0_temp=3.0, cooling=0.99995):
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = corr.max() * 2.0 / N
    best_c5 = current_c5
    best_h = h.copy()
    T = t0_temp
    while time.time() < t_end:
        T = max(T * cooling, 1e-7)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0
        h[j] = 1
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = new_corr.max() * 2.0 / N
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            corr = new_corr
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    return best_h, best_c5

def interpolate_to_n(h_src, n_dst):
    N_src = len(h_src)
    x0 = np.arange(N_src) / N_src
    x1 = np.arange(n_dst) / n_dst
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def random_binary(N, rng):
    h = np.zeros(N)
    h[rng.choice(N, N // 2, replace=False)] = 1.0
    return h

def perturbed_binary(h_best, N, rng, num_swaps):
    h = h_best.copy()
    for _ in range(num_swaps):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        h[ones[rng.randint(len(ones))]] = 0
        h[zeros[rng.randint(len(zeros))]] = 1
    return h

def continuous_refine(h, N, t_end):
    val, corr, dx = obj_and_corr(h)
    lr = 0.02
    for it in range(20000):
        if time.time() > t_end:
            break
        k_max = np.argmax(corr) - (N - 1)
        g = np.zeros(N)
        if k_max >= 0:
            for i in range(N - k_max):
                g[i] += (1.0 - h[i + k_max])
                g[i + k_max] -= h[i]
        else:
            k_abs = -k_max
            for i in range(k_abs, N):
                g[i] += (1.0 - h[i - k_abs])
                g[i - k_abs] -= h[i]
        h_new = project_feasible(h - lr * g, N)
        val_new, corr_new, _ = obj_and_corr(h_new)
        if val_new < val - 1e-15:
            h = h_new
            val = val_new
            corr = corr_new
        else:
            lr *= 0.92
            if lr < 1e-7:
                break
    return h, val

        h_b = binarize(h_interp, rng)
        # Quick local search
