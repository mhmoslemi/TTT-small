def repair(h):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    # Bisection for water-filling
    for _ in range(60):
        s = x.sum()
        if abs(s - target) < 1e-14:
            break
        x += (target - s) / n
        np.clip(x, 0.0, 1.0, out=x)
    # Final fix
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        # Distribute diff to elements with most room
        if diff > 0:
            room = 1.0 - x
            order = np.argsort(room)[::-1]
        else:
            order = np.argsort(x)[::-1]
        for idx in order:
            if abs(diff) < 1e-15:
                break
            if diff > 0:
                add = min(diff, 1.0 - x[idx])
            else:
                add = max(diff, -x[idx])
            x[idx] += add
            diff -= add
    return x

def compute_c(h):
    """Compute all C(k) values"""
    n = h.size
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return c

def exact_c(h):
    return float(np.max(compute_c(h)))

def smooth_objective(h, tau):

    cm = np.max(c)
    z = (c - cm) / tau
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    return cm + tau * np.log(s), c, w

def grad_smooth(h, w):

    n = h.size
    dx = 2.0 / n

            continue

                         bounds=bounds, constraints=cons,
                         options={'maxiter': maxiter, 'ftol': 1e-12})

    except:
        pass
    return h0

def project(h, target):
    n = h.size
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) * 0.5
    return np.clip(h - lam, 0.0, 1.0)


def repair(h):
    x = np.asarray(h, dtype=float)
    n = x.size
    if n == 0:
        return x
    return project(x, 0.5 * n)


def compute_c(h):
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def exact_c(h):
    return float(np.max(compute_c(h)))


def smooth_obj(h, tau):
    c = compute_c(h)
    cm = float(np.max(c))
    z = (c - cm) / tau
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e / s
    return cm + tau * np.log(s), c, w


def grad_smooth(h, w):
    n = h.size
    dx = 2.0 / n
    grad = np.zeros(n)
    m = w.size
    for k in range(m):
        wk = w[k]
        if wk < 1e-14:
            continue
        wk = wk * dx
        shift = k - (n - 1)
        if shift >= 0:
            jlo, jhi = shift, n - 1
            grad[jlo:jhi + 1] += wk * (1.0 - h[jlo - shift:jhi + 1 - shift])
            jlo2, jhi2 = 0, n - 1 - shift
            grad[jlo2:jhi2 + 1] -= wk * h[jlo2 + shift:jhi2 + 1 + shift]
        else:

            jlo, jhi = 0, n - 1 - s
            grad[jlo:jhi + 1] += wk * (1.0 - h[jlo + s:jhi + 1 + s])
            jlo2, jhi2 = s, n - 1
            grad[jlo2:jhi2 + 1] -= wk * h[jlo2 - s:jhi2 + 1 - s]
    return grad


def subgrad_direction(h):
    c = compute_c(h)
    C = float(np.max(c))
    tol = max(1e-9, 0.03 * C)
    active = np.where(c >= C - tol)[0]
    if len(active) > 6:
        active = np.argsort(c)[::-1][:6]
    n = h.size
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active:
        shift = k - (n - 1)
        if shift >= 0:
            jlo, jhi = shift, n - 1
            grad[jlo:jhi + 1] += (1.0 - h[jlo - shift:jhi + 1 - shift])
            jlo2, jhi2 = 0, n - 1 - shift
            grad[jlo2:jhi2 + 1] -= h[jlo2 + shift:jhi2 + 1 + shift]
        else:
            s = -shift
            jlo, jhi = 0, n - 1 - s

            jlo2, jhi2 = s, n - 1
            grad[jlo2:jhi2 + 1] -= h[jlo2 - s:jhi2 + 1 - s]
    grad /= len(active)
    return grad * dx


def run_slsqp(h0, tau, maxiter):
    n = h0.size
    bounds = [(0.0, 1.0)] * n

    def fun(x):
        v, _, _ = smooth_obj(x, tau)
        return v

    def jac(x):
        _, _, w = smooth_obj(x, tau)
        return grad_smooth(x, w)

    def cons_fun(x):
        return np.array([x.sum() - 0.5 * n])

    def cons_jac(x):
        return np.ones((1, n))

    try:
        res = so.minimize(fun, h0, method="SLSQP", jac=jac,
                          bounds=bounds,
                          constraints=[{"type": "eq", "fun": cons_fun, "jac": cons_jac}],
                          options={"maxiter": maxiter, "ftol": 1e-13})
        if res.x is not None:
            return repair(res.x)
    except Exception:
        pass
    return h0
