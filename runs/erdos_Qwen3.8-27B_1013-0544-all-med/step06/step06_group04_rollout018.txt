import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, factor=2):
    n = len(h)
    n_new = n * factor
    # Linear interpolation
    old_idx = np.linspace(0, n-1, n_new)
    idx0 = np.floor(old_idx).astype(int)
    idx1 = np.minimum(idx0 + 1, n - 1)
    frac = old_idx - idx0
    h_new = h[idx0] * (1 - frac) + h[idx1] * frac
    return project_feasible(h_new, n_new)

Given h with values in [0,1] and sum != n/2:
- If sum > n/2: need to decrease sum by (sum - n/2). Spread the decrease among all h_i > 0.
- If sum < n/2: need to increase sum by (n/2 - sum). Spread the increase among all h_i < 1.

The iterative approach:
1. Clip to [0,1]
2. Compute diff = target - sum
3. If diff > 0: distribute +diff/n among those with h<1, clip
4. If diff < 0: distribute diff/n among those with h>0, clip
5. Repeat until converged

- Start with h = [2, 0, 0, ..., 0]. After clip: [1, 0, 0, ..., 0]. Sum = 1. Need n/2.
