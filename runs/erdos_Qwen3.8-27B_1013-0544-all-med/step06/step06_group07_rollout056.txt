```python
        j = rng.randint(n)

                best_x = y.copy()

            else:
                no_improve += 1
        else:

def subgrad_descent(h0, time_limit, seed):

    x = repair(h0.copy())
    best_x = x.copy()
    best_c = exact_c(x)
    n = x.size
    rng = np.random.RandomState(seed)
    deadline = time.time() + time_limit

    no_improve = 0
    momentum = np.zeros(n)
    beta = 0.7
    
    while time.time() < deadline:
        g = subgrad_top(x, top_k=3)
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        
        # Normalized step
        step = lr / gn
        momentum = beta * momentum - step * g
        y = repair(x + momentum)
        
        cy = exact_c(y)
        if cy < best_c:
            best_c = cy
            best_x = y.copy()
            no_improve = 0
        elif cy < exact_c(x) + 1e-8:
            no_improve += 1
        else:
            no_improve += 1
        
        x = y
        lr *= 0.999
        
        if no_improve > 200:
            # Restart
            x = best_x + rng.normal(0, 0.005, n)
            x = repair(x)
            momentum = np.zeros(n)
            lr = 0.03
            no_improve = 0
        
        if no_improve > 500:
            break
    
    return best_x, best_c
```

```python
from scipy.optimize import linprog

def solve_lp(h0, active_set, n):
    """Solve min t s.t. c_k(h) ≈ c_k(h0) + grad_k(h0)·(h-h0) ≤ t"""
    dx = 2.0 / n
    C_k = {}
    grad_k = {}
    for k in active_set:
        c_k = c_values(h0)[k]
        g_k = subgrad_k(h0, k)
        C_k[k] = c_k
        grad_k[k] = g_k
    
    # Variables: h[0], ..., h[n-1], t (index n is t)
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: grad_k·h + c_k(h0) - grad_k·h0 - t ≤ 0
    # i.e., grad_k·h - t ≤ -c_k(h0) + grad_k·h0
    A_ub = []
    b_ub = []
    for k in active_set:
        row = np.zeros(n + 1)
        row[:n] = grad_k[k]
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(-C_k[k] + float(grad_k[k] @ h0))
    
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])
    
    # Bounds: 0 ≤ h[i] ≤ 1, t free (or t ≥ 0)
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return repair(res.x[:n])
    return h0
```

```python
def repair(h):

    n = x.size

        return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)

        if abs(diff) <= 1e-13:
            break

            mask = slack > 1e-14
            if not np.any(mask):
                break
            x[mask] += diff * slack[mask] / slack[mask].sum()
        else:
            mask = x > 1e-14
            if not np.any(mask):
                break
            x[mask] += diff * x[mask] / x[mask].sum()

    if abs(diff) > 1e-10:
        k = int(np.floor(target))
        x[:] = 0.0
        if k > 0:
            x[:k] = 1.0

        if rem > 0 and k < n:

def c_values(h):
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(c_values(h)))

def subgrad_k(h, k_idx):
    n = h.size
    dx = 2.0 / n
    s = int(k_idx) - (n - 1)
    g = np.zeros(n)
    lo = max(0, s)
    hi = min(n - 1, s + n - 1)
    if lo <= hi:
        g[lo:hi+1] += (1.0 - h[lo-s:hi+1-s]) * dx
    lo2 = max(0, -s)
    hi2 = min(n - 1, n - 1 - s)
    if lo2 <= hi2:
        g[lo2:hi2+1] -= h[lo2+s:hi2+1+s] * dx
    return g

def lp_solve(h0, active_indices, n):
    """Solve min t s.t. linearized c_k(h) <= t, sum(h)=n/2, 0<=h<=1"""
    dx = 2.0 / n
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub_list = []
    b_ub_list = []
    
    for k in active_indices:
        g = subgrad_k(h0, k)
        c_k = c_values(h0)[k]
        # c_k(h) ≈ c_k + g·(h - h0) = c_k - g·h0 + g·h
        # Constraint: c_k - g·h0 + g·h - t <= 0
        # i.e., g·h - t <= g·h0 - c_k
        row = np.zeros(n + 1)
        row[:n] = g
        row[n] = -1.0
        A_ub_list.append(row)
        b_ub_list.append(float(g @ h0) - c_k)
    
    A_ub = np.array(A_ub_list)
    b_ub = np.array(b_ub_list)

    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])
    
    bounds = [(0, 1)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        pass
    return repair(h0), None

        else:
```

```python
            if not np.any(mask):

    n = h.size

def subgrad_k(h, k_idx):
    n = h.size
    dx = 2.0 / n
    s = int(k_idx) - (n - 1)
    g = np.zeros(n)
    lo = max(0, s)
    hi = min(n - 1, s + n - 1)
    if lo <= hi:
        g[lo:hi + 1] += (1.0 - h[lo - s:hi + 1 - s]) * dx
    lo2 = max(0, -s)
    hi2 = min(n - 1, n - 1 - s)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] -= h[lo2 + s:hi2 + 1 + s] * dx
    return g

        A_ub[ii, n] = -1.0
```
