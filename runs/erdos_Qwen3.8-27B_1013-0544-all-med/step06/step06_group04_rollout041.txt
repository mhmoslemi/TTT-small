Hmm, let me just use the lag k = m - (n-1) formulation:
c(k) = sum_n h[n+k] * (1-h[n]) for n where 0≤n<n and 0≤n+k<n.

∂c(k)/∂h[i]:
- From h[n+k] = h[i] when n = i-k: contributes (1-h[i-k]) if 0≤i-k<n and 0≤i<n
- From (1-h[n]) = (1-h[i]) when n = i: contributes -h[i+k] if 0≤i<n and 0≤i+k<n

So:
∂c(k)/∂h[i] = (1-h[i-k]) * 1_{0≤i-k<n} * 1_{0≤i<n} + (-h[i+k]) * 1_{0≤i<n} * 1_{0≤i+k<n}

Since we always have 0≤i<n (i is our index), this simplifies to:
∂c(k)/∂h[i] = (1-h[i-k]) * 1_{0≤i-k<n} - h[i+k] * 1_{0≤i+k<n}

```python
def gradient_of_lag(h, k, n):
    """Gradient of c(k) = sum h[n+k]*(1-h[n]) with respect to h."""
    grad = np.zeros(n)
    # First term: (1-h[i-k]) for i where 0 ≤ i-k < n
    # i ranges from max(0,k) to min(n-1, n-1+k)
    lo = max(0, k)
    hi = min(n, n + k)
    if lo < hi:
        grad[lo:hi] += (1 - h[lo-k:hi-k])
    # Second term: -h[i+k] for i where 0 ≤ i+k < n
    # i ranges from max(0,-k) to min(n-1, n-1-k)
    lo2 = max(0, -k)
    hi2 = min(n, n - k)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2+k:hi2+k]
    return grad
```

```python
    if h_init is not None:
        h_init = project_feasible(h_init, n_init / 2.0)
        v = c5_exact(h_init)
        if v < best_val:
            best_val = v
            best_h = h_init.copy()
    
    # Candidate n values (must be even for n/2 to be integer)
    n_candidates = sorted(set(
        [n_init] + [n for n in [20, 30, 40, 50, 56, 60, 70, 80, 90, 100, 120, 140, 160, 200] if n % 2 == 0]
    ))
    
    beta_list = [2, 5, 10, 30, 80, 200]
    
    for n in n_candidates:
        if budget_remaining() < 30:
            break
        target_sum = n / 2.0
        
        # Build starting points
        starts = []
        if h_init is not None:
            if n == n_init:
                starts.append(h_init.copy())
            else:
                starts.append(interpolate_to_n(h_init, n))
        starts.append(np.full(n, 0.5))
        
        # Random continuous
        hr = np.clip(0.5 + 0.35 * rng.randn(n), 0.0, 1.0)
        starts.append(project_feasible(hr, target_sum))
        
        # Binary
        hb = np.zeros(n)
        hb[rng.choice(n, n // 2, replace=False)] = 1.0
        starts.append(hb)
        
        for start in starts:
            if budget_remaining() < 10:
                break
            start = project_feasible(start, target_sum)
            
            # Smooth optimization
            h_opt = optimize_one_start(start, n, target_sum, beta_list, 200, budget_remaining)
            
            # Subgradient polish
            if budget_remaining() > 5:
                n_sg = min(3000, max(500, int(budget_remaining() * 50)))
                h_sg = subgradient_optimize(h_opt, n, target_sum, n_sg, 0.08, 0.9995)
                if c5_exact(h_sg) < c5_exact(h_opt):
                    h_opt = h_sg
            
            # Local search
            if budget_remaining() > 3:
                n_ls = min(5000, max(500, int(budget_remaining() * 30)))
                h_ls, _ = local_search(h_opt, n, target_sum, n_ls, rng)
                if c5_exact(h_ls) < c5_exact(h_opt):
                    h_opt = h_ls
            
            # Binary round
            h_bin = binary_round(h_opt, n)
            if c5_exact(h_bin) < c5_exact(h_opt):
                h_opt = h_bin
            
            # Update best
            v = c5_exact(h_opt)
            if v < best_val:
                best_val = v
                best_h = h_opt.copy()
    
    # Final polish on best
    if best_h is not None and budget_remaining() > 10:
        n_best = len(best_h)
        h_final = subgradient_optimize(best_h, n_best, n_best / 2.0,
                                       int(min(10000, budget_remaining() * 80)), 0.03, 0.9998)
        if c5_exact(h_final) < best_val:
            best_val = c5_exact(h_final)
            best_h = h_final
        h_ls, _ = local_search(best_h, len(best_h), len(best_h) / 2.0,
                               int(min(5000, budget_remaining() * 20)), rng)
        if c5_exact(h_ls) < best_val:
            best_val = c5_exact(h_ls)
            best_h = h_ls
    
    if best_h is None:
        best_h = np.full(n_init, 0.5)
        best_val = c5_exact(best_h)
    
    return best_h, best_val, len(best_h)
```

Wait, I'm using `budget_remaining` as a closure (it captures `t0` and `budget_s`). Let me check the rules again: "Make all helper functions top level, no closures or lambdas."

The function `budget_remaining` is defined inside `run`, so it's a closure. I need to avoid this.

I can pass `t0` and `budget_s` explicitly:

```python
def budget_remaining(t0, budget_s):
    return budget_s - (time.time() - t0)
```

Then call it as `budget_remaining(t0, budget_s)` everywhere.

Similarly, `optimize_one_start` takes `budget_fn` as a parameter, but that's a function reference, not a closure. Actually, if I pass `budget_remaining` (which is now a top-level function) with its arguments... hmm, I can't partially apply without a closure.

Let me restructure: instead of passing a budget function, I'll check the budget at each major step in the main loop and break out.

import numpy as np
from scipy.optimize import minimize
import time


def project_feasible(h, target_sum):
    n = len(h)
    if n == 0:
        return h.copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)


def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def smooth_obj(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(np.clip(corr - m, -50.0, 50.0)))) / beta


def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    exp_c = np.exp(np.clip(corr - m, -50.0, 50.0))
    Z = np.sum(exp_c)
    w = exp_c / Z
    grad = np.zeros(n)
    for ki in range(2 * n - 1):
        k = ki - (n - 1)
        wki = w[ki]
        if wki < 1e-12:
            continue
        f = dx * wki
        lo1 = max(0, k)
        hi1 = min(n, n + k)
        if lo1 < hi1:
            grad[lo1:hi1] += f * (1.0 - h[lo1 - k:hi1 - k])
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= f * h[lo2 + k:hi2 + k]
    return grad


def subgradient_optimize(h, n, target_sum, n_iters, lr_init, decay):
    dx = 2.0 / n
    h = h.copy()
    lr = lr_init
    for _ in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode="full")
        ki = int(np.argmax(corr))
        k_star = ki - (n - 1)
        grad = np.zeros(n)
        lo1 = max(0, k_star)
        hi1 = min(n, n + k_star)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1 - k_star:hi1 - k_star])
        lo2 = max(0, -k_star)
        hi2 = min(n, n - k_star)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2 + k_star:hi2 + k_star]
        grad *= dx
        h = h - lr * grad
        h = project_feasible(h, target_sum)
        lr *= decay
    return h


def interpolate_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old, dtype=float) / n_old
    x_new = np.arange(n_new, dtype=float) / n_new
    h_interp = np.interp(x_new, x_old, h, period=1.0)
    return project_feasible(h_interp, n_new / 2.0)

    h_bin[indices] = 1.0
    return h_bin
