```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_val_grad(h, temp, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
    else:
        idx = np.arange(len(corr))
    corr_sub = corr[idx]
    m = np.max(corr_sub)
    shifted = np.clip(temp * (corr_sub - m), -500, 500)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    val = (m + np.log(exps.sum()) / temp) * dx
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for ki, w in zip(idx, weights):
        d = ki - (n - 1)
        if d >= 0:
            if n > d:
                grad[d:] += w * one_minus_h[:n-d]
                grad[:n-d] -= w * h[d:]
        else:
            e = -d
            if n > e:
                grad[:n-e] += w * one_minus_h[e:]
                grad[e:] -= w * h[:n-e]
    grad *= dx
    return val, grad

def optimize_slsqp(h_start, temp, top_k, maxiter, n):
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n

    def func(x):
        return smooth_val_grad(x, temp, top_k)[0]

    def func_grad(x):
        return smooth_val_grad(x, temp, top_k)

    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts,
             'jac': lambda x: np.ones(n)}]
    res = minimize(func_grad, h_start, method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return project_feasible(res.x, ts)

def optimize_active_set(h_start, n, budget_remaining, top_k_init=60, top_k_min=10):
    """Iteratively optimize against active set, updating it each round."""
    ts = n / 2.0
    h = h_start.copy()
    temps = [200.0, 100.0, 50.0, 25.0, 10.0, 5.0, 2.0, 1.0]
    for ti, temp in enumerate(temps):
        if budget_remaining() < 30:
            break
        top_k = max(top_k_min, top_k_init - ti * 5)
        for round_i in range(3):
            if budget_remaining() < 15:
                break
            h = optimize_slsqp(h, temp, top_k, maxiter=300, n=n)
            c = compute_c5(h)
    return h

def make_block_pattern(n, blocks=8):
    """Create a block pattern: alternating high/low blocks."""
    h = np.zeros(n)
    block_size = n // (2 * blocks)
    for i in range(blocks):
        start = i * 2 * block_size
        end = min(start + block_size, n)
        h[start:end] = 1.0
    return project_feasible(h, n / 2.0)

def make_sinusoidal(n, freq=3, amp=0.5):
    x = np.linspace(0, 2 * np.pi * freq, n)
    h = 0.5 + amp * 0.5 * np.sin(x)
    return project_feasible(h, n / 2.0)

def make_staircase(n, steps=6):
    h = np.zeros(n)
    for i in range(n):
        h[i] = ((i * steps) // n) % 2
    return project_feasible(h, n / 2.0)

def coord_descent_fine(h, n_levels=20, max_sweeps=10):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h[j]
            best_c = c5
            best_v = orig
            for v in np.linspace(0.0, 1.0, n_levels):
                if abs(v - orig) < 1e-10:
                    continue
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = h.copy()
            if best_c < c5 - 1e-13:
                h = best_v
                c5 = best_c
                improved = True
            else:
                h[j] = orig
        if not improved:
            break
    return h, c5

    h0 = initial_h_values.astype(float).copy()

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    h1 = optimize_active_set(h0, n0, remaining, top_k_init=80, top_k_min=15)
    c1 = compute_c5(h1)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1.copy()

    # Phase 2: Fine coordinate descent
    if remaining() > 60:
        h2, c2 = coord_descent_fine(best_h, n_levels=20, max_sweeps=12)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 3: Structured perturbations + optimize
    if remaining() > 100:
        perturbations = []
        # Block flip perturbation
        hpf = best_h.copy()
        flip_start = np.random.randint(0, n0 // 3)

        hpf[flip_start:flip_start + flip_len] = 1.0 - hpf[flip_start:flip_start + flip_len]
        hpf = project_feasible(hpf, n0 / 2.0)

        # Shift perturbation
```
