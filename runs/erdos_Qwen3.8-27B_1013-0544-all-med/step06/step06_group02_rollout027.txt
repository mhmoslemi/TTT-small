```python
def local_search_binary(h, N, max_passes=1000):
    """Local search: repeatedly find the best improving swap."""
    dx = 2.0 / N
    improved = True
    passes = 0
    while improved and passes < max_passes:
        improved = False
        passes += 1
        corr = np.correlate(h, 1.0 - h, 'full')
        current_max = corr.max()
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        best_gain = 0
        best_swap = None
        
        for i in zeros:
            for j in ones:
                if i == j:
                    continue
                h[i], h[j] = 1.0, 0.0
                new_corr = np.correlate(h, 1.0 - h, 'full')
                new_max = new_corr.max()
                h[i], h[j] = 0.0, 1.0
                
                if new_max < current_max - 1e-12:
                    gain = current_max - new_max
                    if gain > best_gain:
                        best_gain = gain
                        best_swap = (i, j)
        
        if best_swap is not None:
            i, j = best_swap
            h[i], h[j] = 1.0, 0.0
            improved = True
    
    return h, corr.max() * dx
```
