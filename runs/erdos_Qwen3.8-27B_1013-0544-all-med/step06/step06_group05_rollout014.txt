                    g -= weights[k] * h[i + m]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(h > 0.5, 1.0, 0.0)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(ones)

    elif n_ones < target:

        rng = np.random.RandomState(0)
        rng.shuffle(zeros)
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def softmax_obj_and_grad(x, beta, dx, n):
    """Compute softmax approximation of max and its gradient"""
    c5_vals = np.correlate(x, 1.0 - x, mode='full') * dx
    max_val = np.max(c5_vals)
    shifted = c5_vals - max_val
    exp_vals = np.exp(beta * shifted)
    sum_exp = np.sum(exp_vals)
    obj_val = max_val + np.log(sum_exp) / beta
    weights = exp_vals / sum_exp
    
    # Gradient
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        for k in range(2*n - 1):
            m = k - (n - 1)
            if 0 <= i - m < n:
                g += weights[k] * (1.0 - x[i - m])
            if 0 <= i + m < n:
                g -= weights[k] * x[i + m]
        grad[i] = g * dx
    return obj_val, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(h > 0.5, 1.0, 0.0)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(ones)

        rng.shuffle(zeros)
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def softmax_obj(x, beta, dx):
    c5_vals = np.correlate(x, 1.0 - x, mode='full') * dx
    mx = np.max(c5_vals)
    shifted = c5_vals - mx
    return mx + np.log(np.sum(np.exp(beta * shifted))) / beta

def softmax_grad(x, beta, dx):
    n = len(x)
    c5_vals = np.correlate(x, 1.0 - x, mode='full') * dx
    mx = np.max(c5_vals)
    shifted = c5_vals - mx
    exp_vals = np.exp(beta * shifted)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for i in range(n):
        g = 0.0
        for k in range(2 * n - 1):
            m = k - (n - 1)
            if 0 <= i - m < n:
                g += weights[k] * (1.0 - x[i - m])
            if 0 <= i + m < n:
                g -= weights[k] * x[i + m]
        grad[i] = g * dx
    return grad

def optimize_smoothed(h0, beta, penalty, maxiter=200):

    dx = 2.0 / n
    target = n / 2.0

        return softmax_obj(x, beta, dx) + penalty * (np.sum(x) - target) ** 2

        return g

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    h_new = np.clip(h_new + (target - np.sum(h_new)) / n_new, 0, 1)
    return h_new
