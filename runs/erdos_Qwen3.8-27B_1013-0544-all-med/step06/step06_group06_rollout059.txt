    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')  # unscaled
    max_c = c.max()
    
    # Compute d_k for the pair (i, j): delta from j to i means h[i]+=δ, h[j]-=δ
    # d_k = (h[j-m_k] - h[i-m_k]) + (h[j+m_k] - h[i+m_k]) with boundary handling
    # where m_k = k - (n-1)
    
    m = np.arange(len(c)) - (n - 1)  # shifts for each correlation index
    
    def get_h(idx):
        """Get h[idx] with boundary handling (0 if out of range)."""
        if 0 <= idx < n:
            return h[idx]
        return 0.0
    
    d = np.zeros(len(c))
    for k, mk in enumerate(m):
        d[k] = (get_h(j - mk) - get_h(i - mk)) + (get_h(j + mk) - get_h(i + mk))
    
    # c_k(δ) = c[k] + δ * d[k]
    # Minimize max_k (c[k] + δ * d[k]) over valid δ range

    # → -h[i] ≤ δ ≤ 1-h[i] and h[j]-1 ≤ δ ≤ h[j]
    delta_min = max(-h[i], h[j] - 1)
    delta_max = min(1 - h[i], h[j])
    
    if delta_min >= delta_max:
        return 0, max_c
    
    # Binary search for optimal δ (the max of linear functions is convex)
    lo, hi = delta_min, delta_max
    for _ in range(50):
        mid = (lo + hi) / 2
        # Compute max at mid
        vals = c + mid * d
        max_val = vals.max()
        # Find where the max is achieved

        # If d[argmax] > 0, decreasing δ helps → move hi down
