import numpy as np

import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.argsort(1.0 - h)[-int(np.ceil(abs(d2)))]
                h[idx] += d2 / len(idx)
            else:
                idx = np.argsort(h)[-int(np.ceil(abs(d2)))]
                h[idx] += d2 / len(idx)

def _obj(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    m = np.max(alpha * C)
    es = np.exp(alpha * C - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha
    f += beta * (np.sum(h) - 0.5 * n) ** 2
    return f

def _coord_descent(h, n, beta, n_sweeps=50, grid_size=200):
    """Coordinate descent with sum constraint via penalty."""
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    m = 2 * n - 1
    target = 0.5 * n
    x_grid = np.linspace(0, 1, grid_size)
    
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            # Compute a[k] for coordinate i
            km = np.arange(m) - (n - 1)  # shifts from -(n-1) to n-1
            valid_p = (i + km >= 0) & (i + km < n)
            valid_m = (i - km >= 0) & (i - km < n)
            hp = np.where(valid_p, h[np.clip(i + km, 0, n-1)], 0.0)
            hm = np.where(valid_m, h[np.clip(i - km, 0, n-1)], 0.0)
            a = (1.0 - hp - hm) * dx
            
            # b[k] = C[k] - a[k] * h[i]
            b = C - a * h[i]
            
            # Solve 1D: min_x max_k(a[k]*x + b[k]) + beta*(x + sum_others - target)^2
            sum_others = np.sum(h) - h[i]
            
            # Evaluate on grid
            C_grid = a[:, None] * x_grid[None, :] + b[:, None]  # (m, grid_size)
            max_vals = np.max(C_grid, axis=0)
            pen = beta * (x_grid + sum_others - target) ** 2
            obj_vals = max_vals + pen
            
            best_idx = np.argmin(obj_vals)
            best_x = x_grid[best_idx]
            
            # Refine around best
            if 0 < best_idx < grid_size - 1:
                lo = x_grid[max(0, best_idx - 5)]
                hi = x_grid[min(grid_size - 1, best_idx + 5)]
                x_fine = np.linspace(lo, hi, 50)
                C_fine = a[:, None] * x_fine[None, :] + b[:, None]
                obj_fine = np.max(C_fine, axis=0) + beta * (x_fine + sum_others - target) ** 2
                best_x = x_fine[np.argmin(obj_fine)]
            
            delta = best_x - h[i]
            if abs(delta) > 1e-10:
                h[i] = best_x
                C += a * delta
                improved = True
        
        if not improved:
            break
    
    return h, C

def _lbgfs_opt(h, n, alpha, beta, maxiter=500):
    """L-BFGS-B optimization with soft-max objective."""
    dx = 2.0 / n
    
    def obj_grad(hv):
        C = _c5_array(hv)
        m = np.max(alpha * C)
        ac = alpha * C - m
        es = np.exp(ac)
        se = np.sum(es)
        f = (m + np.log(se)) / alpha + beta * (np.sum(hv) - 0.5 * n) ** 2
        w = es / se
        
        # Gradient: d/dh[i] of C[k]
        # C[k] = dx * sum_j h[j]*(1-h[j+m_k])
        # dC[k]/dh[i] = dx * [(1-h[i+m_k])*[valid] - h[i-m_k]*[valid]]
        km = np.arange(2*n-1) - (n-1)
        valid_p = (i + km >= 0) & (i + km < n)  # This is per-i, can't precompute
        # Actually, let me compute gradient differently
        # grad[i] = dx * sum_k w[k] * dC[k]/dh[i]
        # dC[k]/dh[i] = dx * [(1-h[i+m_k])*[0<=i+m_k<n] - h[i-m_k]*[0<=i-m_k<n]]
        
        grad = np.zeros(n)
        for i in range(n):
            vp = (i + km >= 0) & (i + km < n)
            vm = (i - km >= 0) & (i - km < n)
            term_p = np.where(vp, 1.0 - h[np.clip(i + km, 0, n-1)], 0.0)
            term_m = np.where(vm, h[np.clip(i - km, 0, n-1)], 0.0)
            grad[i] = dx * np.dot(w, term_p - term_m)
        
        grad += 2.0 * beta * (np.sum(hv) - 0.5 * n)
        return f, grad
    
    res = minimize(obj_grad, h, jac=True, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-15})
    if res.x is not None:
        return _fix(res.x, n)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-12:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-10:
            k = max(1, int(np.ceil(abs(d2))))
            if d2 > 0:
                idx = np.argsort(1.0 - h)[:k]
            else:
                idx = np.argsort(h)[:k]
            h[idx] += d2 / k
            h = np.clip(h, 0.0, 1.0)
    return h

def _lbgfs(h, n, alpha, beta, maxiter=300):
    dx = 2.0 / n
    km = np.arange(2 * n - 1) - (n - 1)

    def fg(hv):
        C = _c5_array(hv)
        ac = alpha * C
        m = np.max(ac)
        es = np.exp(ac - m)
        se = np.sum(es)
        f = (m + np.log(se)) / alpha + beta * (np.sum(hv) - 0.5 * n) ** 2
        w = es / se
        grad = np.zeros(n)
        for i in range(n):
            vp = (i + km >= 0) & (i + km < n)
            vm = (i - km >= 0) & (i - km < n)
            tp = np.where(vp, 1.0 - hv[np.clip(i + km, 0, n - 1)], 0.0)
            tm = np.where(vm, hv[np.clip(i - km, 0, n - 1)], 0.0)
            grad[i] = dx * np.dot(w, tp - tm)
        grad += 2.0 * beta * (np.sum(hv) - 0.5 * n)
        return f, grad

    res = minimize(fg, h, jac=True, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return _fix(res.x, n) if res.x is not None else h

def _coord_desc(h, n, beta, time_limit, grid=200):
    t_start = time.time()
    dx = 2.0 / n
    m = 2 * n - 1
    C = _c5_array(h)
    target = 0.5 * n
    xg = np.linspace(0, 1, grid)
    km = np.arange(m) - (n - 1)
    tot = np.sum(h)

    while time.time() - t_start < time_limit:
        improved = False
        for i in range(n):
            if time.time() - t_start > time_limit:
                break
            vp = (i + km >= 0) & (i + km < n)
            vm = (i - km >= 0) & (i - km < n)
            hp = np.where(vp, h[np.clip(i + km, 0, n - 1)], 0.0)
            hm = np.where(vm, h[np.clip(i - km, 0, n - 1)], 0.0)
            a = (1.0 - hp - hm) * dx
            b = C - a * h[i]
            so = tot - h[i]
            vals = np.max(a[:, None] * xg[None, :] + b[:, None], axis=0)
            vals += beta * (xg + so - target) ** 2
            bi = np.argmin(vals)
            bx = xg[bi]
            if 0 < bi < grid - 1:
                lo = xg[max(0, bi - 3)]
                hi = xg[min(grid - 1, bi + 3)]
                xf = np.linspace(lo, hi, 40)
                vf = np.max(a[:, None] * xf[None, :] + b[:, None], axis=0)
                vf += beta * (xf + so - target) ** 2
                bx = xf[np.argmin(vf)]
            d = bx - h[i]
            if abs(d) > 1e-10:
                h[i] = bx
                C += a * d
                tot += d
                improved = True
        if not improved:
            break
    return h, C, np.max(C)

def _binary_check(h, n):
    hb = (h >= 0.5).astype(float)
    n1 = int(np.sum(hb))
    tgt = n // 2
    if n1 > tgt:
        idx = np.argsort(h)[n - n1:n - (n1 - tgt)]
        hb[idx] = 0.0
    elif n1 < tgt:
        idx = np.argsort(-h)[:tgt - n1]
        hb[idx] = 1.0
    return _fix(hb, n)
