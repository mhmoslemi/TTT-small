import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
            diff2 = target - h.sum()
            if abs(diff2) > 1e-12:
                mask2 = (h > 1e-10) & (h < 1.0 - 1e-10)
                if mask2.sum() > 0:
                    h[mask2] += diff2 / mask2.sum()
                    h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, d):
    N = len(h)
    g = np.zeros(N)
    if d >= 0:
        m = N - d
        if m > 0:
            g[:m] += (1.0 - h[d:])
            g[d:] -= h[:m]
    else:
        m = N + d
        if m > 0:
            g[-d:] += (1.0 - h[:m])
            g[:m] -= h[-d:]
    return g

def obj_value(h):
    N = len(h)
    return compute_corr(h).max() * (2.0 / N)

def binary_from_continuous(h, N):
    """Convert continuous h to binary with exactly N/2 ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = h_bin.sum()
    target = N // 2
    if n_ones > target:
        # Need to flip some 1s to 0s
        indices_1 = np.where(h_bin == 1)[0]
        # Flip those with smallest h values
        order = np.argsort(h[indices_1])
        for i in order[:int(n_ones - target)]:
            h_bin[indices_1[i]] = 0
    elif n_ones < target:
        indices_0 = np.where(h_bin == 0)[0]
        order = np.argsort(-h[indices_0])
        for i in order[:int(target - n_ones)]:
            h_bin[indices_0[i]] = 1
    return h_bin

def binary_search(h_bin, N, rng, t_end, try_update, max_iter=100000):

    zeros = np.where(h == 0)[0]

    try_update(h, val)

            h[j] = 0

def continuous_refine(h0, N, rng, t_end, try_update, max_iter=100000):
    """Gradient descent with momentum on continuous h."""
    h = h0.copy()
    val = obj_value(h)
    try_update(h, val)
    best_val = val
    best_h = h.copy()
    
    v = np.zeros(N)  # momentum

    momentum = 0.7

        if time.time() > t_end:
            break

        new_val = obj_value(h_new)

            h = h_new
            val = new_val
            stagnant = 0

                best_val = val
                best_h = h.copy()

                v = np.zeros(N)
                lr = 0.01
                stagnant = 0

                    best_val = val
                    best_h = h.copy()

    h = np.interp(x1, x0, h_src)
