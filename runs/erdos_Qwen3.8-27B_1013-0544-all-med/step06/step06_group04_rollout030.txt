from scipy.optimize import minimize
import time

_g_n = 0
_g_eps = 0.02
_rng_ls = np.random.RandomState(123)

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _obj(x):
    dx = 2.0 / _g_n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    return float(m + _g_eps * np.log(np.sum(np.exp((corr - m) / _g_eps))))

def _obj_grad(x):
    n = _g_n
    dx = 2.0 / n
    N2 = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp((corr - m) / _g_eps)
    w_sum = np.sum(exp_vals)
    weights = exp_vals / w_sum
    grad = np.zeros(n)
    for k in range(N2):
        w = weights[k]
        if w < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            i_s, i_e = 0, n - lag
        else:
            i_s, i_e = -lag, n
        idx = np.arange(i_s, i_e)
        grad[idx + lag] += (1.0 - x[idx]) * w
        grad[idx] += -x[idx + lag] * w
    return grad

    return np.sum(x) - _g_n / 2.0

    return np.ones(_g_n)

            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x

def quantize_h(h, n):
    k_target = n // 2
    h_sorted_desc = np.sort(h)[::-1]
    threshold = h_sorted_desc[k_target - 1]
    h_q = (h >= threshold).astype(np.float64)
    count = int(np.sum(h_q))
    if count != k_target:
        if count > k_target:
            ones_idx = np.where(h_q == 1.0)[0]
            order = np.argsort(h[ones_idx])
            for i in range(count - k_target):
                h_q[ones_idx[order[i]]] = 0.0
        else:
            zeros_idx = np.where(h_q == 0.0)[0]
            order = np.argsort(-h[zeros_idx])
            for i in range(k_target - count):
                h_q[zeros_idx[order[i]]] = 1.0
    return h_q

    h = h.copy()

            h[i], h[j] = h[j], h[i]
