import numpy as np
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full'))) * dx

def c5_corr(h):
    """Return the full correlation array and the C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return corr * dx, np.max(corr) * dx

def make_binary(h, n, rng):
    """Convert h to binary with exactly n//2 ones."""
    h = np.array(h, dtype=float)
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        remove = rng.choice(one_idx, n_ones - target, replace=False)
        h_bin[remove] = 0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        add = rng.choice(zero_idx, target - n_ones, replace=False)
        h_bin[add] = 1
    return h_bin

def local_search_binary(h, n, rng, max_time=60):
    """Best-improvement local search for binary h."""
    start = time.time()
    current_c5 = c5_value(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    
    improved = True
    while improved and time.time() - start < max_time:
        improved = False
        best_delta = 0
        best_pair = None
        current_c5 = c5_value(h)
        # Try all swaps
        for i in range(len(ones)):
            for j in range(len(zeros)):
                a, b = ones[i], zeros[j]
                h[a], h[b] = h[b], h[a]
                new_c5 = c5_value(h)
                delta = current_c5 - new_c5
                if delta > best_delta:
                    best_delta = delta
                    best_pair = (a, b)
                h[a], h[b] = h[b], h[a]
        if best_pair is not None:
            a, b = best_pair
            h[a], h[b] = h[b], h[a]
            ones.remove(a)
            ones.append(b)
            zeros.remove(b)
            zeros.append(a)
            current_c5 = c5_value(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            improved = True
    
    return best_h, best_c5

def sa_binary(h, n, rng, T_start=0.01, T_end=0.00001, max_time=60):
    """Simulated annealing for binary h."""
    start = time.time()
    current_c5 = c5_value(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    it = 0
    total_iters = max(100000, int(max_time * 200000))  # estimate

        T = T_start * (T_end / T_start) ** (it / total_iters)

        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        new_c5 = c5_value(h)
        delta = current_c5 - new_c5  # positive = improvement

            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5

def best_improve(h, n, rng, max_time=60):
    start = time.time()
    best_c5 = c5_value(h)
    best_h = h.copy()
    cur_c5 = best_c5
    ones = set(int(i) for i in np.where(h == 1)[0])
    zeros = set(int(i) for i in np.where(h == 0)[0])
    while time.time() - start < max_time:
        improved = False
        best_delta = 0
        best_i, best_j = -1, -1
        for i in ones:
            for j in zeros:
                h[i], h[j] = h[j], h[i]
                nc = c5_value(h)
                h[i], h[j] = h[j], h[i]
