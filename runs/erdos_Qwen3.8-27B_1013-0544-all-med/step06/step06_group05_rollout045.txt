def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_continuous(h, n):
    """Project h onto [0,1]^n ∩ {sum = n/2}"""
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    # Bisection on offset
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_trial = np.clip(h + mid, 0, 1)
        s = np.sum(h_trial)
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0, 1)
    # Fix residual
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:

        free = (h > 1e-10) & (h < 1.0 - 1e-10)

    return h

def subgradient_step(h, alpha, k_star):
    """
    Compute subgradient of f_{k_star} and take a step.
    k_star is the index into the full correlation output.
    """
    n = len(h)
    # The full correlation has length 2n-1
    # corr[m] = sum_i h[i] * (1-h[i+m-n+1]) for m = 0..2n-2
    # Actually let me be more careful.
    # np.correlate(a, v, 'full')[k] = sum_i a[i+k] * v[i] for valid i
    # where k ranges from -(len(v)-1) to len(a)-1

    # For our case: a = h, v = 1-h
    # output[m] = sum_i h[i] * (1-h[i+m-n+1])
    
    # Let lag = m - (n-1). Then:
    # output[m] = sum_i h[i] * (1-h[i+lag])
    
    # For lag >= 0: output[n-1+lag] = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
    # For lag < 0: output[n-1+lag] = sum_{i=-lag}^{n-1} h[i] * (1-h[i+lag])
    
    # Gradient of output[m] w.r.t. h[p]:
    # = sum_i [δ_{i,p}*(1-h[i+lag]) + h[i]*(-δ_{i+lag,p})]
    # = (1-h[p+lag]) * 1[p+lag < n, p >= 0, valid range]
    #   - h[p-lag] * 1[p-lag >= 0, valid range]
    
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    
    if lag >= 0:
        # f = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
        # ∂f/∂h[p] = (1-h[p+lag]) for p = 0..n-1-lag
        #            - h[p-lag] for p = lag..n-1
        for p in range(n - lag):
            grad[p] += (1.0 - h[p + lag])
        for p in range(lag, n):
            grad[p] -= h[p - lag]
    else:
        # lag < 0, let ll = -lag > 0
        # f = sum_{i=ll}^{n-1} h[i] * (1-h[i-ll])
        # = sum_{j=0}^{n-1-ll} h[j+ll] * (1-h[j])
        # ∂f/∂h[p] = (1-h[p-ll]) for p = ll..n-1  (from first term where j+ll=p, j=p-ll)
        #            - h[p+ll] for p = 0..n-1-ll  (from second term where j=p)
        ll = -lag
        for p in range(ll, n):
            grad[p] += (1.0 - h[p - ll])
        for p in range(n - ll):
            grad[p] -= h[p + ll]
    
    # Step
    h_new = h - alpha * grad
    return h_new

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_continuous(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h + mid, 0, 1)
        if np.sum(h_trial) < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0, 1)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0, 1)
        else:
            idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))
            h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def compute_subgradient(h, k_star):
    n = len(h)
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        p = np.arange(n - lag)
        grad[p] += (1.0 - h[p + lag])
        p2 = np.arange(lag, n)
        grad[p2] -= h[p2 - lag]
    else:
        ll = -lag
        p = np.arange(ll, n)
        grad[p] += (1.0 - h[p - ll])
        p2 = np.arange(n - ll)
        grad[p2] -= h[p2 + ll]
    return grad

def projected_subgradient(h, max_time, seed=42):
    n = len(h)
    h = normalize_continuous(h.copy(), n)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    alpha = 0.03
    patience = 0
    no_improve = 0

    while (time.time() - start) < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        grad = compute_subgradient(h, k_star)
        gn = np.linalg.norm(grad)
        if gn < 1e-10:
            break
        grad_unit = grad / gn

        best_step_h = None
        best_step_c5 = current_c5
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 1.5, alpha * 2.0]:
            h_trial = normalize_continuous(h - a * grad_unit, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_step_c5 - 1e-12:
                best_step_c5 = c5_trial
                best_step_h = h_trial

        if best_step_h is not None:
            h = best_step_h
            current_c5 = best_step_c5
            if current_c5 < best_c5 - 1e-10:
                best_c5 = current_c5
                best_h = h.copy()
                patience = 0
                no_improve = 0
                alpha = min(alpha * 1.05, 0.15)
            else:
                no_improve += 1
                if no_improve > 10:
                    alpha = max(alpha * 0.85, 0.002)
                    no_improve = 0
                    patience += 1
                if patience > 30:
                    break
        else:
            alpha = max(alpha * 0.7, 0.002)
            patience += 1
            if patience > 40:
                break

    return best_h, best_c5

def directed_local_search(h0, max_time=60.0, seed=42):
    n = len(h0)
    h = normalize_continuous(h0.copy(), n)
    current_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    stall = 0

    while (time.time() - start) < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        grad = compute_subgradient(h, k_star)

        # Identify positions to decrease (positive grad) and increase (negative grad)
        dec_candidates = np.where(grad > 0.01)[0]
        inc_candidates = np.where(grad < -0.01)[0]

        if len(dec_candidates) == 0 or len(inc_candidates) == 0:
            stall += 1
            if stall > 5:
                break
            # Perturb
            idx = rng.randint(n)
            h[idx] = np.clip(h[idx] + rng.uniform(-0.05, 0.05), 0, 1)
            h = normalize_continuous(h, n)
            continue

        best_gain = 1e-12
        best_move = None
        max_trials = min(len(dec_candidates) * len(inc_candidates), 3000)
        trials = 0

        for _ in range(max_trials):
            i = dec_candidates[rng.randint(len(dec_candidates))]
            j = inc_candidates[rng.randint(len(inc_candidates))]
            if i == j:
                continue
            eps = rng.uniform(0.02, 0.15)
            if h[i] < eps or h[j] > 1.0 - eps:
                continue
            h[i] -= eps
            h[j] += eps
            h = normalize_continuous(h, n)
            new_c5 = compute_c5(h)
            h[i] += eps
            h[j] -= eps
            h = normalize_continuous(h, n)
            gain = current_c5 - new_c5
            if gain > best_gain:
                best_gain = gain
                best_move = (i, j, eps)
            trials += 1
            if (time.time() - start) > max_time:
                break

        if best_move is not None:
            i, j, eps = best_move
            h[i] -= eps
            h[j] += eps
            h = normalize_continuous(h, n)
            current_c5 = compute_c5(h)
            stall = 0
        else:
            stall += 1
            if stall > 8:
                break

    return h, current_c5

def binary_local_search(h0, max_time=60.0, seed=42):
    n = len(h0)
    h = np.round(h0).astype(float)
    target = n // 2
    n_ones = int(np.sum(h))

            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    current_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)

    while (time.time() - start) < max_time:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_gain = 1e-14
        best_move = None

        for i in ones:
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j, None, None)
                h[i] = 1.0
                h[j] = 0.0

        if best_move is None or best_gain <= 1e-14:
            n_o, n_z = len(ones), len(zeros)
            if n_o >= 2 and n_z >= 2:
                for _ in range(8000):
                    if (time.time() - start) > max_time:
                        break
                    ia, ib = rng.choice(n_o, 2, replace=False)
                    ja, jb = rng.choice(n_z, 2, replace=False)
                    i1, i2 = ones[ia], ones[ib]
                    j1, j2 = zeros[ja], zeros[jb]
                    h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                    new_c5 = compute_c5(h)
                    gain = current_c5 - new_c5
                    if gain > best_gain:
                        best_gain = gain
                        best_move = (i1, i2, j1, j2)
                    h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0

        if best_move is not None and best_gain > 1e-14:
            if best_move[2] is None:
                i, j, _, _ = best_move
                h[i] = 0.0; h[j] = 1.0
            else:
                i1, i2, j1, j2 = best_move
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
            current_c5 = compute_c5(h)
            improved = True
        if not improved:
            break

    return h, current_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old

    h_new = np.interp(x_new, x_old, h_old)
    return normalize_continuous(h_new, n_new)
