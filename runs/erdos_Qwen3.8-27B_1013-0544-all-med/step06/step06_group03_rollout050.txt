    h = h.copy()

        if s > target:
            hi = mid
        else:
            lo = mid

def annealed_momentum_pgd(h0, n_iter, t_init=0.05, t_end=0.001, lr_init=0.008, lr_end=0.0001):

    best_h = h.copy()

        t = t_init * (t_end / t_init) ** f

        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def exact_pgd(h0, n_iter, lr_init=0.002):
    """PGD with exact subgradient (argmax) and backtracking."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(n_iter):
        conv = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(conv) * dx
        idx = np.argmax(conv)
        shift = idx - (n - 1)
        grad = conv_grad_shift(h, shift) * dx
        # Backtracking
        step_lr = lr
        for _ in range(10):
            h_new = project_feasible(h - step_lr * grad, n)
            c5_new = compute_c5(h_new)
            if c5_new <= c5 + 1e-14:
                break
            step_lr *= 0.5
        h = h_new
        c5 = c5_new
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr = min(step_lr * 1.05, 0.02)
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def conv_grad_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    if 0 <= shift < n:
        g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    elif -n < shift < 0:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def soft_max_og(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    s = np.sum(exps)
