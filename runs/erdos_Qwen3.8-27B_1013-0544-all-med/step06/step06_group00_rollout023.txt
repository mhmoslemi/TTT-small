    # T2[j] = sum_m p[m] * h[j-m+(n-1)] where 0 <= j-m+(n-1) < n

    # Both are convolutions/correlations!
    # T1[j] = sum_i h[i] * p[i-(j-(n-1))] = (h * p_rev)[j] ... 
    
    # Let me just use matrix multiplication for clarity (O(n²) but vectorized)

    # Then T1 = P @ h
    # And Q[j, i] = p[j-i+(n-1)] if valid, else 0
    # Then T2 = Q @ h
    
    # Actually, P and Q are related: Q[j,i] = P[i,j] (by symmetry of the index formula)
    # P[j,i] = p[i-j+n-1], Q[j,i] = p[j-i+n-1]
    # So Q = P^T!
    
    # Build P: P[j,i] = p[i-j+n-1] if 0 <= i-j+n-1 <= 2n-2, else 0
    j_idx = np.arange(n)[:, None]  # n x 1
    i_idx = np.arange(n)[None, :]  # 1 x n

    P = np.where(valid, p[m_idx], 0.0)

    n = len(h)

    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target_sum - h.sum()
        if abs(d) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            mask = (h > 0) & (h < 1)
            if mask.sum() == 0:
                break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")

    T1 = P @ h
    T2 = P.T @ h
    grad = dx * (1.0 - T1 - T2)
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            mask = (h > 0) & (h < 1)
            if mask.sum() == 0:
                break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def precompute_grad_struct(n):
    j_idx = np.arange(n)[:, None]
    i_idx = np.arange(n)[None, :]
    m_idx = (i_idx - j_idx + (n - 1)).astype(np.intp)
    valid = (m_idx >= 0) & (m_idx <= 2 * n - 2)
    return m_idx, valid

def smooth_obj_grad(h, beta, m_idx, valid, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    mc = corr.max()
    ev = np.exp(beta * (corr - mc))
    p = ev / ev.sum()
    obj = (mc + np.log(ev.sum()) / beta) * dx
    P = np.where(valid, p[m_idx], 0.0)
    T1 = P.dot(h)
    T2 = P.T.dot(h)
    grad = dx * (1.0 - T1 - T2)
    return obj, grad

def pgd_smooth(h, beta, m_idx, valid, dx, max_iter=3000):
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    obj_prev, _ = smooth_obj_grad(h, beta, m_idx, valid, dx)
    lr = 2.0 / (beta * dx + 1e-8)
    for it in range(max_iter):
        obj, grad = smooth_obj_grad(h, beta, m_idx, valid, dx)
        step = lr
        improved = False
        for _ in range(40):
            h_new = project_feasible(h - step * grad, target)
            obj_new, _ = smooth_obj_grad(h_new, beta, m_idx, valid, dx)
            if obj_new < obj - 1e-15:
                h = h_new
                obj_prev = obj_new
                improved = True
                break
            step *= 0.5
        if not improved:
            lr *= 0.8
            if lr < 1e-12:
                break
        if it > 10 and abs(obj - obj_prev) < 1e-12:
            break
    return h

def coord_descent(h, max_sweeps=10):
    n = len(h)
    target = n / 2.0
    c5 = compute_c5(h)
    offsets = np.array([-0.15, -0.08, -0.03, 0.03, 0.08, 0.15])
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h.copy()
            best_c = c5
            best_state = orig.copy()
            ov = h[j]
            for v in np.unique(np.clip(ov + offsets, 0.0, 1.0)):
                if abs(v - ov) < 1e-10:
                    continue
                h_try = orig.copy()
                h_try[j] = v
                h_try = project_feasible(h_try, target)
                c = compute_c5(h_try)
                if c < best_c - 1e-14:
                    best_c = c
                    best_state = h_try.copy()
            if best_c < c5 - 1e-14:
                h = best_state
                c5 = best_c
                improved = True
            else:
                h = orig
        if not improved:
            break
    return h, c5

def bottleneck_lag_optimize(h, m_idx, valid, dx, n_iter=200):
    """Focus gradient on the worst lag to reduce it specifically."""
    n = len(h)
    target = n / 2.0
    h = project_feasible(h, target)
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode="full")
        k_star = np.argmax(corr) - (n - 1)
        # Subgradient at bottleneck lag
        sg = np.zeros(n)
        if k_star >= 0:
            for j in range(n - k_star):
                sg[j] += (1.0 - h[j + k_star])
            for j in range(k_star, n):
                sg[j] -= h[j - k_star]
        else:
            ka = -k_star
            for j in range(n - ka):
                sg[j] += (1.0 - h[j + ka])
            for j in range(ka, n):
                sg[j] -= h[j - ka]
        h_new = project_feasible(h - 0.01 * sg, target)
        c_new = compute_c5(h_new)
        c_cur = compute_c5(h)
        if c_new < c_cur:
            h = h_new
        else:
            break
    return h
