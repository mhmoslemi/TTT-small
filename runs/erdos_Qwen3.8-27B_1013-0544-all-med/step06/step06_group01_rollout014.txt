    h = np.clip(h, 0.0, 1.0)

    # Distribute deficit

            else:
                h += d / n

            else:
                h += d / n

            rng = np.random.RandomState(sw)
        for _ in range(n):
            i, j = rng.randint(n, size=2)
            if i == j:
                continue
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-12:
                continue
            h[i] = lo
            h[j] = s - lo
            C0 = np.correlate(h, 1.0 - h, mode='full') * dx
            h[i] = hi
            h[j] = s - hi
            C1 = np.correlate(h, 1.0 - h, mode='full') * dx
            t = _opt_t(C0, (C1 - C0) / (hi - lo), lo, hi)
            h[i] = t
            h[j] = s - t
        h = _fix(h, n)
        c = _c5(h)
        if c < best - 1e-13:
            best = c
            stall = 0
        else:
            stall += 1
        if stall >= 6:
            break
    return h, best

def _resample(h_old, n_old, n_new):
    x = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xnew = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(xnew, x, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n0 = len(initial_h_values)
    hb = _fix(initial_h_values.astype(float).copy(), n0)
    cb = _c5(hb)
    nb = n0

    # Phase 1: coordinate descent at initial n
    if elapsed() < budget_s * 0.2:
        hb, cb = _cd(hb, nb, nswp=500, rng=rng)

    # Phase 2: explore different n values
    n_cands = sorted(set(list(range(max(50, nb - 15), min(200, nb + 25), 5)) +
                         [60, 65, 70, 75, 80, 85, 90, 95, 100, 105, 110, 115, 120]))
    n_cands = [n for n in n_cands if 50 <= n <= 200 and n != nb]

    for n in n_cands:
        if elapsed() > budget_s * 0.55:
            break
        ht = _fix(_resample(hb, nb, n), n)
        ht, cn = _cd(ht, n, nswp=400, rng=rng)
        if cn < cb - 1e-10:
            cb, hb, nb = cn, ht, n

    # Phase 3: paired CD for more refinement
    if elapsed() < budget_s * 0.65:
        ht, cn = _paired_cd(hb, nb, nswp=300, rng=rng)
        if cn < cb:
            cb, hb = cn, ht

    # Phase 4: random restarts at best n
    max_restarts = 50
    for r in range(max_restarts):
        if elapsed() > budget_s * 0.85:
            break
        scale = 0.03 + 0.06 * rng.rand()
        ht = hb + rng.randn(nb) * scale
        ht = _fix(ht, nb)
        ht, cn = _cd(ht, nb, nswp=250, rng=rng)
        if cn < cb - 1e-10:
            cb, hb = cn, ht
            # also try changing n around the new best
            for dn in [-5, 5]:
                nn = nb + dn
                if 50 <= nn <= 200 and elapsed() < budget_s * 0.87:
                    ht2 = _fix(_resample(hb, nb, nn), nn)
                    ht2, cn2 = _cd(ht2, nn, nswp=200, rng=rng)
                    if cn2 < cb - 1e-10:
                        cb, hb, nb = cn2, ht2, nn

    # Phase 5: fine n search around best
    if elapsed() < budget_s * 0.92:
        for n in range(max(50, nb - 8), nb + 9):
            if elapsed() > budget_s * 0.93:
                break
            if n == nb:
                continue
            ht = _fix(_resample(hb, nb, n), n)
            ht, cn = _cd(ht, n, nswp=300, rng=rng)
            if cn < cb - 1e-10:
                cb, hb, nb = cn, ht, n

    # Phase 6: final polish with many sweeps
    if elapsed() < budget_s * 0.97:
        hb, cb = _cd(hb, nb, nswp=2000, rng=None)

    # Final: paired CD polish
    if elapsed() < budget_s * 0.99:
        ht, cn = _paired_cd(hb, nb, nswp=500, rng=rng)
        if cn < cb:
            cb, hb = cn, ht

    hb = _fix(hb, nb)
    cb = _c5(hb)
    return (hb, cb, nb)
