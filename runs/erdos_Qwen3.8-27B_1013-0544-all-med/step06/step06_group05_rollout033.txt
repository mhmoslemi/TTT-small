Where all terms are 0 if the index is out of bounds.

```python
def fast_swap_c5(h, S, i, j):
    """Compute new max correlation after swapping h[i]:1->0, h[j]:0->1.
    S is the current full correlation vector.
    Returns the new max correlation value (not multiplied by dx)."""
    n = len(h)
    # ΔS(k) = h[i-k] + h[i+k] - h[j+k] - h[j-k] (with bounds checking)
    # k ranges from -(n-1) to n-1, stored as index k+(n-1) in S
    # S has length 2n-1, index 0 corresponds to k=-(n-1), index n-1 to k=0, etc.
    
    delta = np.zeros(2*n-1)
    # For each index t in S (t = k + (n-1), so k = t - (n-1)):
    # delta[t] = h[i-k] + h[i+k] - h[j+k] - h[j-k]
    # where h[x] = 0 if x < 0 or x >= n
    
    k = np.arange(-(n-1), n)  # all lags
    # h[i-k] valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i
    # h[i+k] valid when 0 <= i+k < n, i.e., -i <= k <= n-1-i
    # etc.
    
    # Vectorized approach: create index arrays and mask
    idx_ik = i - k  # index for h[i-k]
    idx_ikp = i + k  # index for h[i+k]
    idx_jk = j + k   # index for h[j+k]
    idx_jkm = j - k  # index for h[j-k]
    
    mask_ik = (idx_ik >= 0) & (idx_ik < n)
    mask_ikp = (idx_ikp >= 0) & (idx_ikp < n)
    mask_jk = (idx_jk >= 0) & (idx_jk < n)
    mask_jkm = (idx_jkm >= 0) & (idx_jkm < n)
    
    delta = (h[idx_ik] * mask_ik + h[idx_ikp] * mask_ikp 
           - h[idx_jk] * mask_jk - h[idx_jkm] * mask_jkm)
    
    new_S = S + delta
    return float(np.max(new_S))

def greedy_best_swap(h, S, A):
    """Find the best single swap to minimize max correlation.
    h: binary array of length n
    S: current full correlation (length 2n-1)
    A: precomputed shift matrix (n x (2n-1)), A[i,k] = change in S[k] when h[i] goes 1->0
    Returns: (i, j, new_max) or None if no improving swap exists
    """
    n = len(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    current_max = S.max()
    
    # For each i in ones, compute new_max for all j in zeros
    # new_max(i,j) = max_k(S[k] + A[i,k] - A[j,k])
    
    best_overall_max = current_max
    best_i, best_j = None, None
    
    for i in ones:
        # new_maxes[j] = max_k(S[k] + A[i,k] - A[j,k]) for j in zeros
        # = max_k((S + A[i])[k] - A[j][k])
        C_i = S + A[i]  # (2n-1,)
        # For all j: max_k(C_i[k] - A[j][k])
        diff = C_i[None, :] - A  # (n, 2n-1)
        new_maxes = diff.max(axis=1)  # (n,)
        
        # Only consider j in zeros
        for j in zeros:
            if new_maxes[j] < best_overall_max:
                best_overall_max = new_maxes[j]
                best_i, best_j = i, j
    
    if best_i is None:
        return None
    return best_i, best_j, best_overall_max
```
