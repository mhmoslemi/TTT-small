import numpy as np

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_lag(h, n_lag):
    N = len(h)
    g = np.zeros(N)
    s1 = max(0, n_lag)
    e1 = min(N, N + n_lag)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - n_lag:e1 - n_lag]
    s2 = max(0, -n_lag)
    e2 = min(N, N - n_lag)
    if s2 < e2:
        g[s2:e2] -= h[s2 + n_lag:e2 - n_lag]
    return g

def top_k_subgrad(h, corr, dx, k=3):
    N = len(h)
    idx = np.argsort(corr)[-k:]
    g = np.zeros(N)
    for i in idx:
        g += grad_lag(h, i - (N - 1))
    return (g / k) * dx
