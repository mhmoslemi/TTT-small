For a binary sequence, if we swap position i (currently 1) to 0 and position j (currently 0) to 1:
- For shift k, the old contribution at position p is h[p]*(1-h[p+k])
- After the swap:

  - At p=j: was 0*(1-h[j+k]), now 1*(1-h[j+k]). Change: +(1-h[j+k])
  - At p where p+k=i (i.e., p=i-k): was h[i-k]*(1-1)=0, now h[i-k]*(1-0)=h[i-k]. Change: +h[i-k]
  - At p where p+k=j (i.e., p=j-k): was h[j-k]*(1-0)=h[j-k], now h[j-k]*(1-1)=0. Change: -h[j-k]

So for each shift k, the change in the sum is:
-(1-h[i+k]) + (1-h[j+k]) + h[i-k] - h[j-k]
= -1 + h[i+k] + 1 - h[j+k] + h[i-k] - h[j-k]
= h[i+k] - h[j+k] + h[i-k] - h[j-k]

(where we use h values before the swap, and out-of-bounds indices contribute 0)

So the new correlation at shift k is:
corr_new[k] = corr_old[k] + h[i+k] - h[j+k] + h[i-k] - h[j-k]

This is O(1) per shift! So we can evaluate the effect of a swap in O(n) time (over all shifts) instead of O(n²).

import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()

    n_ones = int(np.sum(h))
    target = n // 2

            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def normalize_continuous(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
            else:
                h += diff / n
        else:
            total = np.sum(h)
            if total > 1e-12:
                h = h * (1.0 + diff / total)
            else:
                h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmax(1.0 - h)) if diff > 0 else int(np.argmax(h))

    x_new = (np.arange(n_new) + 0.5) / n_new

    dx = 2.0 / n
    h = h0.astype(float).copy()

    current_c5 = float(np.max(corr) * dx)
    start = time.time()
    s = np.arange(-(n-1), n)
    stall_count = 0
    
    while time.time() - start < timeout:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        n1, n0 = len(ones), len(zeros)
        
        if n1 == 0 or n0 == 0:
            break
        
        # Precompute F and B matrices
        # F[p, k] = h[p + s[k]] if 0 <= p+s[k] < n else 0
        # B[p, k] = h[p - s[k]] if 0 <= p-s[k] < n else 0
        F = np.zeros((n, len(s)))
        B = np.zeros((n, len(s)))
        for p in range(n):
            idx_f = p + s
            valid = (idx_f >= 0) & (idx_f < n)
            F[p, valid] = h[idx_f[valid]]
            idx_b = p - s
            valid = (idx_b >= 0) & (idx_b < n)
            B[p, valid] = h[idx_b[valid]]
        
        AB_ones = (F[ones] + B[ones])  # shape (n1, 2n-1)
        AB_zeros = (F[zeros] + B[zeros])  # shape (n0, 2n-1)
        
        # new_corr[i, j, k] = corr[k] + AB_ones[i, k] - AB_zeros[j, k]
        new_corr_3d = corr[None, None, :] + AB_ones[:, None, :] - AB_zeros[None, :, :]
        new_c5_matrix = new_corr_3d.max(axis=2) * dx  # shape (n1, n0)
        
        best_idx = np.unravel_index(new_c5_matrix.argmin(), new_c5_matrix.shape)
        best_c5 = new_c5_matrix[best_idx]
        
        if best_c5 < current_c5 - 1e-12:
            i = ones[best_idx[0]]
            j = zeros[best_idx[1]]
            h[i] = 0.0
            h[j] = 1.0
            corr = compute_corr(h)
            current_c5 = float(np.max(corr) * dx)
            stall_count = 0
        else:
            stall_count += 1
            if stall_count >= max_stall:
                break
    
    return h, current_c5
