    dx = 2.0 / n
    conv = np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")
    best_c5 = np.max(conv) * dx
    best_h = h.copy()
    
    if rng is None:
        rng = np.random.RandomState(42)
    
    for it in range(max_iters):
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_delta_max = 0  # We want to minimize max, so we want negative delta at the max position
        best_swap = None
        
        for _ in range(n_trials):
            i = rng.choice(zeros)
            j = rng.choice(ones)
            
            # Compute delta for all affected m
            # Affected m: those where m-i in [0,n-1] or m-j in [0,n-1] or m-i-n in [0,n-1] or m-j-n in [0,n-1]
            # Actually m ranges from 0 to 2n-2
            # m-i in [0,n-1] means m in [i, i+n-1]
            # m-j in [0,n-1] means m in [j, j+n-1]
            
            m_lo = max(0, min(i, j))
            m_hi = min(2*n-2, max(i, j) + n - 1)
            
            # For each m in [m_lo, m_hi], compute delta
            # delta[m] = f(m-i) - g(m-j) - f2(m-i) + g2(m-j)
            # where these are defined as above
            
            a = np.arange(m_lo, m_hi+1) - i  # = m - i for each m
            b = np.arange(m_lo, m_hi+1) - j  # = m - j for each m
            
            # f(a) = (1-new_h[a]) if valid
            f = np.zeros_like(a, dtype=float)
            valid_a = (a >= 0) & (a < n)
            # For valid a: new_h[a] = 0 if a==j, 1 if a==i, h[a] otherwise
            f[valid_a] = 1.0 - h[a[valid_a]]
            f[valid_a & (a == j)] = 1.0  # new_h[j]=0, so 1-new_h[j]=1
            f[valid_a & (a == i)] = 0.0  # new_h[i]=1, so 1-new_h[i]=0
            
            # g(b) = (1-h[b]) if valid
            g = np.zeros_like(b, dtype=float)
            valid_b = (b >= 0) & (b < n)
            g[valid_b] = 1.0 - h[b[valid_b]]
            
            # f2(a) = h[a] if valid and a not in {i,j}
            f2 = h[a[valid_a]].copy()
            f2[(a[valid_a] == i) | (a[valid_a] == j)] = 0
            f2_full = np.zeros_like(a, dtype=float)
            f2_full[valid_a] = f2
            
            # g2(b) = h[b] if valid and b not in {i,j}
            g2 = h[b[valid_b]].copy()
            g2[(b[valid_b] == i) | (b[valid_b] == j)] = 0
            g2_full = np.zeros_like(b, dtype=float)
            g2_full[valid_b] = g2
            
            delta = f - g - f2_full + g2_full
            
            # New conv for affected positions
            affected_m = np.arange(m_lo, m_hi+1)
            new_conv_affected = conv[affected_m] + delta
            
            # Find the new max
            # The overall max is max(max of affected, max of unaffected)
            # Unaffected max: we can precompute, but for simplicity, just compute new max
            new_conv = conv.copy()
            new_conv[affected_m] = new_conv_affected
            new_max = np.max(new_conv)
            
            if new_max < best_delta_max:
                best_delta_max = new_max
                best_swap = (i, j, affected_m, delta)
        
        if best_swap is not None and best_delta_max < np.max(conv) - 1e-12:
            i, j, affected_m, delta = best_swap
            h[i] = 1
            h[j] = 0
            conv[affected_m] += delta
            if np.max(conv) * dx < best_c5:
                best_c5 = np.max(conv) * dx
                best_h = h.copy()
        elif it % 1000 == 0:
            # Periodically try random swaps even if not improving (simulated annealing style)
            pass
    
    return best_h.astype(float), best_c5

- m-j is a position with h=0 (and m-j ≠ i, m-j ≠ j)

So: i = m - p where h[p]=1, and j = m - q where h[q]=0, with the constraint that p ≠ j, q ≠ i, etc.

This gives a structured way to find good swaps for each bottleneck shift!

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj(h, t, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    return t * (m / t + np.log(np.sum(exps))) * dx + 500.0 * (np.sum(h) - n/2.0)**2

def soft_max_grad(h, t, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-12:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            if shift < n:
                grad[shift:] += (1.0 - h[:n - shift])
                grad[:n - shift] -= h[shift:]
        else:
            s = -shift
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
    grad *= w
    return grad * dx + 1000.0 * (np.sum(h) - n/2.0)

The function ∫ h(x)h(x+k) dx is bilinear (quadratic but not convex or concave in general). So the problem is non-convex.

For a swap (i,j) where h[i]: 0→1, h[j]: 1→0:
The change in conv[m] for each m is:
delta[m] = (1 - new_h[m-i]) * [0≤m-i<n] - (1-h[m-j]) * [0≤m-j<n] - h[m-i] * [0≤m-i<n, m-i∉{i,j}] + h[m-j] * [0≤m-j<n, m-j∉{i,j}]

For binary h:
- (1 - new_h[m-i]): if m-i==j, new_h[j]=0 so this is 1. If m-i==i, new_h[i]=1 so this is 0. Otherwise, 1-h[m-i].
- The formula simplifies for binary h to:
  delta[m] = h[m-j] - h[m-i] (when m-i and m-j are both in [0,n) and not equal to i or j)
  With boundary corrections.

Let me implement the fast version:

```python
def eval_swap(h, conv, i, j, n):
    """Evaluate the change in conv when swapping h[i]:0->1, h[j]:1->0."""
    # Affected m range
    m_lo = max(0, min(i, j))
    m_hi = min(2*n-2, max(i, j) + n - 1)
    
    m = np.arange(m_lo, m_hi+1)
    a = m - i  # = m - i
    b = m - j  # = m - j
    
    # delta[m] for binary h:
    # = (1-new_h[a]) * valid_a - (1-h[b]) * valid_b - h[a]*valid_a*(a∉{i,j}) + h[b]*valid_b*(b∉{i,j})
    # For binary: 1-h[x] = 1 when h[x]=0
    # new_h[a]: if a==i then 1, if a==j then 0, else h[a]
    
    # Simple version: just compute the vectors
    delta = np.zeros(len(m))
    
    # Term 1: (1 - new_h[a]) where valid
    va = (a >= 0) & (a < n)
    t1 = np.zeros(len(m))
    t1[va] = 1.0 - h[a[va]]
    t1[va & (a == j)] = 1.0  # new_h[j] = 0
    t1[va & (a == i)] = 0.0   # new_h[i] = 1
    delta += t1
    
    # Term 2: -(1 - h[b]) where valid
    vb = (b >= 0) & (b < n)
    t2 = np.zeros(len(m))
    t2[vb] = -(1.0 - h[b[vb]])
    delta += t2
    
    # Term 3: -h[a] where valid and a not in {i,j}
    t3 = np.zeros(len(m))
    mask_a = va & (a != i) & (a != j)
    t3[mask_a] = -h[a[mask_a]]
    delta += t3
    
    # Term 4: +h[b] where valid and b not in {i,j}
    t4 = np.zeros(len(m))
    mask_b = vb & (b != i) & (b != j)
    t4[mask_b] = h[b[mask_b]]
    delta += t4
    
    return m, delta
```

This is O(n) per swap evaluation (vectorized). Much faster than O(n²) for full correlate.

        if s > target: hi = mid
        else: lo = mid

def _cg(h, shift, n):
    """Gradient of conv at a given shift."""
    g = np.zeros(n)
    if 0 <= shift < n:
        g[shift:] += 1.0 - h[:n-shift]
        g[:n-shift] -= h[shift:]
    elif -n < shift < 0:
        s = -shift
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def sm_grad(h, t, n):
    dx = 2.0/n
    conv = np.correlate(h, 1-h, mode="full")
    m = conv.max()
    w = np.exp((conv-m)/t)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-11: continue
        g += w[k] * _cg(h, k-(n-1), n)
    return (g*dx + 2e4*(h.sum()-n/2.0), conv)

def sm_obj(h, t, n):
    dx = 2.0/n
