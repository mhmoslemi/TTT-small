import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1 - h, mode='full')) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
            t = room.sum()
            if t < 1e-15:
                break
            h += (diff / t) * room
        else:
            t = h.sum()
            if t < 1e-15:
                break
            h -= (diff / t) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 500:
        v = (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    else:
        v = np.log(np.sum(np.exp(s))) / alpha
    return float(m + v)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

_ell_cache = {}

def get_ell(n):
    if n not in _ell_cache:
        _ell_cache[n] = np.arange(-(n - 1), n)
    return _ell_cache[n]

def compute_a(h, n, i):
    ell = get_ell(n)
    a = np.zeros(2 * n - 1)
    j1 = i + ell
    m1 = (j1 >= 0) & (j1 < n)
    a[m1] += (1.0 - h[j1[m1]])
    j2 = i - ell
    m2 = (j2 >= 0) & (j2 < n)
    a[m2] -= h[j2[m2]]
    return a

def pairwise_cd(h, n, rng, time_limit, max_rounds=500):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full').copy()
    best = float(np.max(c) * dx)
    count = 0
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        n_pairs = 300 if rnd < 50 else 200
        for _ in range(n_pairs):
            i, j = rng.choice(n, size=2, replace=False)
            b = compute_a(h, n, int(i)) - compute_a(h, n, int(j))
            lo = max(-h[i], h[j] - 1.0)
            hi = min(1.0 - h[i], h[j])
            if lo >= hi - 1e-12:
                continue
            ns = 25
            d = np.linspace(lo, hi, ns)
            fv = np.max(c + d[:, None] * b[None, :], axis=1)
            bk = int(np.argmin(fv))
            cur_max = float(np.max(c) * dx)
            if fv[bk] >= cur_max - 1e-14:
                continue
            lo2 = d[max(0, bk - 1)]
            hi2 = d[min(ns - 1, bk + 1)]
            d2 = np.linspace(lo2, hi2, 25)
            fv2 = np.max(c + d2[:, None] * b[None, :], axis=1)
            bk2 = int(np.argmin(fv2))
            h[i] += d2[bk2]
            h[j] -= d2[bk2]
            c += d2[bk2] * b
            improved = True
            count += 1
            cur = float(np.max(c) * dx)
            if cur < best:
                best = cur
            if count % 500 == 0:
                c = np.correlate(h, 1 - h, mode='full')
                best = min(best, float(np.max(c) * dx))
        if not improved:
            break
    c = np.correlate(h, 1 - h, mode='full')
    best = float(np.max(c) * dx)
    return h, best

def slsqp_round(h, n, alphas, cons, bounds, time_limit):
    for alpha in alphas:
        if time.time() > time_limit:
            break
        try:
            res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-18})
            h_new = project_feasible(res.x, n)
            c_new = compute_c5(h_new, n)
            c_old = compute_c5(h, n)
            if c_new <= c_old + 1e-15:
                h = h_new
        except Exception:
            pass
    return h

def perturbed_restart(h, n, rng, scale, cons, bounds, time_limit):
    h_pert = project_feasible(h + rng.normal(0, scale, size=n), n)
    if time.time() > time_limit:
        return h_pert, compute_c5(h_pert, n)
    h_out = slsqp_round(h_pert, n, [500, 2000, 10000], cons, bounds, time_limit)
    c_out = compute_c5(h_out, n)
    return h_out, c_out
