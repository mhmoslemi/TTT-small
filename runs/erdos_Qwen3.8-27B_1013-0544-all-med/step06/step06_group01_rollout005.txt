```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    """Project to feasible set: 0<=h<=1, sum(h)=n/2"""
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-14:
        return h
    # Distribute d among coordinates with room
    if d > 0:
        # Need to add mass; add to coordinates with most room (lowest h)
        order = np.argsort(h)
        for idx in order:
            add = min(d, 1.0 - h[idx])
            h[idx] += add
            d -= add
            if d < 1e-14:
                break
    else:
        # Need to remove mass; remove from coordinates with most mass (highest h)
        d = -d
        order = np.argsort(-h)
        for idx in order:
            rem = min(d, h[idx])
            h[idx] -= rem
            d -= rem
            if d < 1e-14:
                break
    return h

def _soft_obj(h, alpha, beta):
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + 0.5 * beta * (np.sum(h) - 0.5 * len(h)) ** 2
    return f

def _soft_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    w = es / se
    f = (m + np.log(se)) / alpha + 0.5 * beta * (np.sum(h) - 0.5 * n) ** 2
    j_arr = np.arange(nc)
    grad = np.zeros(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _optimize_soft(h, alpha, beta, maxiter=500):
    res = minimize(_soft_grad, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * len(h),
                  options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-15})
    if res.x is not None:

    return h
```

Let me also think about the "balanced gradient" idea. At the KKT point:
∇f(h*) = λ * 1 (for the active constraint)

Where f(h) = C_{k*} for the active lag k*. So:
∇C_{k*}(h*) should be parallel to 1 (all components equal).

If it's not, we can project the gradient onto the orthogonal complement of 1 to get the "unbalanced" part, and use that to guide the search.

Actually, a simpler approach: the component of the gradient along the all-ones direction is "wasted" (it just changes the sum). The useful direction is the projection onto the orthogonal complement. So:
g_useful = g - (mean(g)) * 1

This is already handled by the constraint term in the objective, but we can be more explicit.

Let me write a cleaner version:

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) < 1e-12:
        return h
    if d > 0:
        order = np.argsort(h)
        for idx in order:
            add = min(d, 1.0 - h[idx])
            h[idx] += add
            d -= add
            if abs(d) < 1e-12:
                break
    else:
        d = -d
        order = np.argsort(-h)
        for idx in order:
            rem = min(d, h[idx])
            h[idx] -= rem
            d -= rem
            if abs(d) < 1e-12:
                break
    return h

def _soft_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    w = es / se
    f = (m + np.log(se)) / alpha + 0.5 * beta * (np.sum(h) - 0.5 * n) ** 2
    j_arr = np.arange(nc)
    grad = np.zeros(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _optimize(h, alpha, beta, maxiter=500):
    n = len(h)
    res = minimize(_soft_grad, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-15})
    if res.x is not None:
        return _fix(res.x, n)
    return h

def _coord_refine(h, n, num_sweeps=3):
    """Fast coordinate descent to polish the solution."""
    best_c5 = _c5_max(h)
    h = h.copy()
    for _ in range(num_sweeps):
        improved = False
        for i in range(n):
            for d in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                new_val = h[i] + d
                if new_val < 0 or new_val > 1:
                    continue
                old_val = h[i]
                h[i] = new_val
                c5 = _c5_max(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    improved = True
                else:
                    h[i] = old_val
        if not improved:
            break
    return _fix(h, n), best_c5
```

Wait, this coordinate descent is still O(n * 6 * n²) per sweep = O(6n³) per sweep. For n=95, that's about 5M per sweep. With 3 sweeps, 15M. That's fine.

But I can be even smarter: instead of trying fixed deltas, I can try to move h[i] to optimize the objective for a fixed other coordinates. This is a 1D optimization problem for each coordinate.

For fixed h[0..i-1, i+1..n-1], the objective (soft-max) as a function of h[i]=t is:
C_k(t) = C_k_base + dx * t * (1 - h[i+k]) - dx * t * h[i-k] (approximately)

Actually it's not quite linear in t because t appears in both factors of the product. Let me think...

C_k = ∑_j h[j](1-h[j+k]) * dx

If h[i] = t, then:
- Terms where j = i: t * (1 - h[i+k]) * dx
- Terms where j+k = i, i.e., j = i-k: h[i-k] * (1-t) * dx

So C_k(t) = C_k_base + dx * [t*(1-h[i+k]) - h[i-k]*(1-t)] (for valid indices)
= C_k_base + dx * [t - t*h[i+k] - h[i-k] + t*h[i-k]]
= C_k_base + dx * [t*(1 - h[i+k] + h[i-k]) - h[i-k]]

This IS linear in t! (For a single coordinate change, the correlation is linear in that coordinate.)

So the soft-max objective as a function of t is:
f(t) = (max_k (C_k_base + dx * t * a_k - b_k) + log(∑exp(α(C_k_base + dx*t*a_k - b_k - m'))))/α + penalty(t)

where a_k and b_k are the coefficients for coordinate i at lag k.

This is a convex function of t (sum of max of linear functions plus log-sum-exp of linear functions plus quadratic). So we can find the optimal t in [0,1] using ternary search or golden section!

This is a great insight. For each coordinate, the optimal value can be found in O(log(1/ε)) evaluations of the objective, each of which is O(n). So total per sweep is O(n * log(1/ε) * n) = O(n² * log(1/ε)).

For n=95, that's about 9000 * 20 = 180K per sweep. Very fast!

Let me implement this:

```python
def _coord_optimize(h, n, tol=1e-6, max_iter=50):
    """For each coordinate, find optimal value using golden section search."""
    best_c5 = _c5_max(h)
    h = h.copy()
    
    # Precompute correlation base
    C_base = _c5_array(h)
    
    for sweep in range(3):
        improved = False
        for i in range(n):
            # Compute coefficients a_k and b_k for coordinate i
            # C_k(t) = C_k + dx * t * a_k - dx * b_k (where b_k = h[i-k] for valid k)
            # Actually: C_k(t) = C_k_base - dx*old_h_i*(1-h[i+k]) + dx*t*(1-h[i+k])
            #                              - dx*h[i-k]*(1-old_h_i) + dx*h[i-k]*(1-t)
            # = C_k_base + dx*(t - old_h_i)*(1-h[i+k]) - dx*h[i-k]*(t - old_h_i)
            # = C_k_base + dx*(t - old_h_i)*(1-h[i+k] - h[i-k])
            
            old = h[i]
            dx = 2.0 / n
            # coefficient of (t - old) in C_k is dx * (1 - h[i+k] - h[i-k]) for valid indices
            # Let's compute the change in C for a given t:
            # ΔC_k = dx * (t - old) * (1 - h[i+k_valid] - h[i-k_valid])
            # where h[i+k_valid] is 0 if i+k >= n, and h[i-k_valid] is 0 if i-k < 0
            
            # For golden section, we need to evaluate f(t) for t in [0,1]
            # f(t) = softmax(C(t)) where C(t)_k = C_base_k + dx*(t-old)*(coeff_k)
            
            # Compute coeff_k for all k
            nc = 2 * n - 1
            j_arr = np.arange(nc)
            # When h[i] changes, which correlation values change?
            # c[k] = sum_j h[j+k-(n-1)] * (1-h[j])
            # h[i] appears as:
            #   h[j+k-(n-1)] = h[i] when j+k-(n-1) = i => k = i-j+n-1
            #     contribution: (1-h[j])
            #   (1-h[j]) where j=i:
            #     c[k] includes h[i+k-(n-1)] * (1-h[i])
            #     contribution: -h[i+k-(n-1)]
            
            # For a change δ = t - old in h[i]:
            # Δc[k] = δ * (1-h[j]) where j = k - i + n - 1 (if valid)
            #          - δ * h[i+k-(n-1)] where the term with j=i exists
            # 
            # More carefully: c[k] = sum_j h[j+k-(n-1)] * (1-h[j])
            # ∂c[k]/∂h[i]:
            #   from h[j+k-(n-1)] where j+k-(n-1)=i, i.e., j = k-i+n-1:
            #     (1-h[j]) = (1-h[k-i+n-1]) if 0 <= k-i+n-1 < n
            #   from (1-h[j]) where j=i:
            #     -h[i+k-(n-1)] if 0 <= i+k-(n-1) < n
            # 
            # So ∂c[k]/∂h[i] = (1-h[k-i+n-1])_valid - h[i+k-n+1]_valid
            
            # Let's compute this vector
            k_arr = j_arr  # 0 to 2n-2
            # First term: 1-h[k-i+n-1] for valid k-i+n-1
            idx1 = k_arr - i + n - 1
            valid1 = (idx1 >= 0) & (idx1 < n)
            term1 = np.where(valid1, 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
            # Second term: -h[i+k-n+1] for valid i+k-n+1
            idx2 = i + k_arr - n + 1
            valid2 = (idx2 >= 0) & (idx2 < n)
            term2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
            
            coeff = dx * (term1 - term2)  # ΔC_k = (t-old) * coeff[k]
            
            # Now find t in [0,1] that minimizes softmax(C_base + (t-old)*coeff)
            # Using golden section
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            a, b = 0.0, 1.0
            c = b - gr * (b - a)
            d_pt = a + gr * (b - a)
            
            def eval_f(t_val):
                C_t = C_base + (t_val - old) * coeff
                m = np.max(C_t)
                es = np.exp(100 * (C_t - m))  # use alpha=100 for coordinate opt
                return m + np.log(np.sum(es)) / 100
            
            f_c = eval_f(c)
            f_d = eval_f(d_pt)
            
            for _ in range(40):
                if f_c < f_d:
                    b = d_pt
                    d_pt = c
                    f_d = f_c
                    c = b - gr * (b - a)
                    f_c = eval_f(c)
                else:
                    a = c
                    c = d_pt
                    f_c = f_d
                    d_pt = a + gr * (b - a)
                    f_d = eval_f(d_pt)
            
            t_opt = (a + b) / 2
            h[i] = t_opt
            C_base = _c5_array(h)  # Update base
            new_c5 = _c5_max(h)
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                improved = True
        
        if not improved:
            break
    
    return _fix(h, n), best_c5
```

```python
import time

        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _optimize_soft(h, alpha, beta, maxiter=500):
    n = len(h)
    res = minimize(_soft_grad, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-15})
    if res.x is not None:
        return _fix(res.x, n)
    return h

def _coord_golden(h, n, alpha_cd=200.0, sweeps=5):
    """Golden-section coordinate descent using soft-max for 1D subproblems."""
    h = h.copy()
    best_c5 = _c5_max(h)
    dx = 2.0 / n
    nc = 2 * n - 1
    j_arr = np.arange(nc)
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    for _ in range(sweeps):
        improved = False
        C_base = _c5_array(h)
        for i in range(n):
            old = h[i]
            idx1 = j_arr - i + n - 1
            v1 = (idx1 >= 0) & (idx1 < n)
            t1 = np.where(v1, 1.0 - h[np.clip(idx1, 0, n - 1)], 0.0)
            idx2 = i + j_arr - n + 1
            v2 = (idx2 >= 0) & (idx2 < n)
            t2 = np.where(v2, h[np.clip(idx2, 0, n - 1)], 0.0)
            coeff = dx * (t1 - t2)

            def _eval_cd(tv):
                C_t = C_base + (tv - old) * coeff
                m2 = np.max(C_t)
                es2 = np.exp(alpha_cd * (C_t - m2))
                return m2 + np.log(np.sum(es2)) / alpha_cd

            a_lo, b_hi = 0.0, 1.0
            c_val = b_hi - gr * (b_hi - a_lo)
            d_val = a_lo + gr * (b_hi - a_lo)
            f_c = _eval_cd(c_val)
            f_d = _eval_cd(d_val)
            for _ in range(35):
                if f_c < f_d:
                    b_hi = d_val
                    d_val = c_val
                    f_d = f_c
                    c_val = b_hi - gr * (b_hi - a_lo)
                    f_c = _eval_cd(c_val)
                else:
                    a_lo = c_val
                    c_val = d_val
                    f_c = f_d
                    d_val = a_lo + gr * (b_hi - a_lo)
                    f_d = _eval_cd(d_val)
            t_opt = (a_lo + b_hi) * 0.5
            h[i] = t_opt
            C_base = _c5_array(h)
            if _c5_max(h) < best_c5 - 1e-11:
                best_c5 = _c5_max(h)
                improved = True
        if not improved:
            break
    return _fix(h, n), best_c5

def _try_resolution(hb, nb, cb, n_new, budget_check, budget_frac):
    """Interpolate to new resolution and optimize."""
    x_old = np.linspace(0, 2, nb, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    ht = _fix(np.interp(x_new, x_old, hb), n_new)
    cn = _c5_max(ht)
    for alpha in [20, 50, 100, 200, 500]:
        if budget_check() > budget_frac:
            break
        beta = max(1.0, 0.03 * alpha)
        ht = _optimize_soft(ht, alpha, beta, maxiter=400)
        cn = _c5_max(ht)
        if cn < cb - 1e-9:
            hb, cb, nb = ht, cn, n_new
    if budget_check() < budget_frac * 0.95:
        hb, cb = _coord_golden(hb, nb, alpha_cd=500.0, sweeps=3)
    return hb, cb, nb
```
