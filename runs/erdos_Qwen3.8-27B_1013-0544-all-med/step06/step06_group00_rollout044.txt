from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        mc = mask.sum()
        if mc == 0:
            break
        h[mask] += d / mc
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_softmax(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    scaled = temp * (corr - m)
    scaled[scaled > 500] = 500
    w = np.exp(scaled)
    ws = w.sum()
    w = w / ws
    val = (m + np.log(ws) / temp) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        else:
            ka = abs(lag)
            grad[ka:] += wk * (1.0 - h[:n - ka])
            grad[:n - ka] -= wk * h[ka:]
    return grad * dx, val

def pg_solve(h, n, temp, max_iter, lr, momentum=0.85):
    target = n / 2.0
    h = project_feasible(h, target)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iter):
        grad, _ = grad_softmax(h, n, temp)
        v = momentum * v + (1.0 - momentum) * grad
        h_new = h - lr * v
        h = project_feasible(h_new, target)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if it % 200 == 199 and c5 > best_c5 + 1e-6:
            lr *= 0.8
    return best_h, best_c5

def coord_descent(h, n_levels=12, max_sweeps=15):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    levels = np.linspace(0.0, 1.0, n_levels)
    for _ in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h.copy()
            best_c = c5
            best_v = h[j]
            for v in levels:
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            h = orig.copy()
            if best_c < c5 - 1e-14:
                h[j] = best_v
                h = project_feasible(h, ts)
                c5 = best_c
                improved = True
        if not improved:
            break
    return h, c5

def binarize_smooth(h, n, sigma=1.0):
    ts = n / 2.0
    binary = (h > 0.5).astype(float)
    if binary.sum() > ts:
        order = np.argsort(h)[::-1]
        for i in order:
            if binary.sum() <= ts:
                break
            binary[i] = 0.0
    elif binary.sum() < ts:
        order = np.argsort(h)
        for i in order:
            if binary.sum() >= ts:
                break
            binary[i] = 1.0
    kernel = np.fft.ifft(np.abs(np.fft.fft(np.exp(-0.5 * (np.arange(n) - n//2)**2 / sigma**2))))
    kernel /= kernel.sum()
    smoothed = np.convolve(binary, kernel, mode='wrap')
    return project_feasible(smoothed, ts)
