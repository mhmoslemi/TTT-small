import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def grad_for_shift(h, shift, n, dx):
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[:n - shift] += (1.0 - h[shift:])
            g[shift:] -= h[:n - shift]
    else:
        s = -shift
        if s < n:
            g[s:] += (1.0 - h[:n - s])
            g[:n - s] -= h[s:]
    return g * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    s = np.sum(exps)
    obj = (m + t * np.log(s)) * dx
    weights = exps / s
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-6:
            continue
        shift = idx - (n - 1)
        grad += w * grad_for_shift(h, shift, n, dx)
    return obj, grad

def hard_max_subgrad(h):

    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    thresh = np.max(conv) - 1e-9
    active = np.where(conv >= thresh)[0]
    grad = np.zeros(n)
    for idx in active:
        shift = idx - (n - 1)
        grad += grad_for_shift(h, shift, n, dx)
    grad /= len(active)
    return c5, grad

    z = x.copy()
    t_param = 1.0
    best_x = x.copy()
    best_val = compute_c5(x)
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        obj, grad = soft_max_obj_grad(z, t)
        x_new = project_feasible(z - lr * grad, n)
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        z = x_new + ((t_param - 1.0) / t_next) * (x_new - x)
        z = project_feasible(z, n)
        x = x_new
        t_param = t_next
        val = compute_c5(x)
        if val < best_val:
            best_val = val
            best_x = x.copy()
    return best_x, best_val

def momentum_subgrad(h0, n, max_iter, lr, momentum, time_limit=None, t_start=None):
