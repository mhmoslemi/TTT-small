    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:

            total = room.sum()

            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

    return np.sum(h) - n / 2.0

    return np.ones(n)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def subgrad_momentum(h, n, n_steps=300, lr_init=0.05, beta=0.85):

        c = np.correlate(h, 1 - h, mode='full') * dx

        grad = np.zeros(n)
        for idx in active:

                grad += (1.0 - 2.0 * h)
            elif k > 0:

        if c_new < best_c:

            best_c = c_new

def coord_descent(h, n, max_sweeps=100):
    dx = 2.0 / n
    h = h.copy()
    best_c = compute_c5(h, n)
    for sweep in range(max_sweeps):
        c_full = np.correlate(h, 1 - h, mode='full') * dx
        any_imp = False
        for i in range(n):
            a = np.zeros(2 * n - 1)
            for idx in range(2 * n - 1):
                k = int(idx - (n - 1))
                if k == 0:
                    a[idx] = 1.0 - 2.0 * h[i]
                elif k > 0:
                    coeff = 0.0
                    if i <= n - 1 - k:
                        coeff += (1.0 - h[i + k])
                    if i >= k:
                        coeff += -h[i - k]
                    a[idx] = coeff
                else:
                    kk = -k
                    coeff = 0.0
                    if i >= kk:
                        coeff += (1.0 - h[i - kk])
                    if i <= n - 1 - kk:
                        coeff += -h[i + kk]
                    a[idx] = coeff
            b = c_full - a * h[i]
            # Bisection for 1D min-max
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) * 0.5
                vals = a * mid + b
                amax = vals.max()

                a_min = a[active_ks].min()
                a_max = a[active_ks].max()

                any_imp = True

        c_now = compute_c5(h, n)
        if c_now < best_c:
            best_c = c_now
        if not any_imp:

    return h, best_c

        h = project_feasible(h, n)  # 142

        if c_now < best_c:  # 144
            best_c = c_now  # 145
        if not any_imp:  # 146
            break  # 147
    return h, best_c  # 148
