def fast_coord_descent(h, n_cands=11, max_sweeps=30, time_limit=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    
    for sweep in range(max_sweeps):
        improved = False
        for j in range(n):
            # Compute effect vector for position j
            # Forward: m in [max(0, n-1-j), 2n-2-j], effect = 1 - h[j+m-n+1]
            # Backward: m in [j, min(2n-2, j+n-1)], effect = -h[j-m+n-1]
            
            e = np.zeros(2*n-1)
            
            # Forward part
            m_start = max(0, n-1-j)
            m_end = min(2*n-2, 2*n-2-j) + 1
            if m_start < m_end:
                h_indices = j + np.arange(m_start, m_end) - n + 1
                e[m_start:m_end] += 1.0 - h[h_indices]
            
            # Backward part
            m_start_b = j
            m_end_b = min(2*n-2, j+n-1) + 1
            if m_start_b < m_end_b:
                h_indices_b = j - np.arange(m_start_b, m_end_b) + n - 1
                e[m_start_b:m_end_b] -= h[h_indices_b]
            
            # Try candidate values
            orig = h[j]
            candidates = np.unique(np.concatenate([[0, 1, orig], 
                                                   np.linspace(0, 1, n_cands)]))
            candidates = candidates[(candidates >= 0) & (candidates <= 1)]
            
            best_c = c5
            best_v = orig
            
            for v in candidates:
                if v == orig:
                    continue
                new_corr = corr + (v - orig) * e
                c = np.max(new_corr) * dx
                if c < best_c:
                    best_c = c
                    best_v = v
            
            if best_v != orig:
                corr += (best_v - orig) * e
                h[j] = best_v
                c5 = best_c
                improved = True
        
        if not improved:
            break
    
    return h, c5

    n = len(h)
    L = 2 * n - 1
    e = np.zeros(L)
    ms = max(0, n - 1 - j)
    me = min(L - 1, L - 1 - j)
    if ms <= me:

        e[ms:me + 1] += 1.0 - h[idx]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def compute_effect(h, j):
    n = len(h)
    L = 2 * n - 1
    e = np.zeros(L)
    ms = max(0, n - 1 - j)
    me = min(L - 1, L - 1 - j)
    if ms <= me:
        idx = j + np.arange(ms, me + 1) - n + 1
        e[ms:me + 1] += 1.0 - h[idx]
    ms2 = j
    me2 = min(L - 1, j + n - 1)
    if ms2 <= me2:
        ar = np.arange(ms2, me2 + 1)
        idx2 = j - ar + n - 1
        v = (idx2 >= 0) & (idx2 < n)
        if v.any():
            e[ms2:me2 + 1][v] -= h[idx2[v]]
    return e

def build_E(h):
    n = len(h)
    L = 2 * n - 1
    E = np.zeros((n, L))
    for j in range(n):
        E[j] = compute_effect(h, j)
    return E

def swap_optimize(h, n_iters=5000, t0=None, time_limit=None, seed=42,
                  structured=False, n_deltas=7, update_every=150):
    n = len(h)
    dx = 2.0 / n
    L = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    c5 = float(np.max(corr) * dx)
    rng = np.random.RandomState(seed)
    E = build_E(h)

    for it in range(n_iters):
        if t0 is not None and time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        if structured:
            j = it % n
            k = (j + 1 + rng.randint(n - 1)) % n
        else:
            j = rng.randint(n)
            k = rng.randint(n)
        if j == k:
            continue
        dmin = max(-h[j], h[k] - 1.0)
        dmax = min(1.0 - h[j], h[k])
        if dmax - dmin < 1e-10:
            continue
        diff = E[j] - E[k]
        best_c = c5
        best_d = 0.0
        best_nc = None
        frac = np.linspace(0.0, 1.0, n_deltas)
        d_cands = dmin + frac * (dmax - dmin)
        for d in d_cands:
            if abs(d) < 1e-14:
                continue
            nc = corr + d * diff
            c_val = float(nc.max() * dx)
            if c_val < best_c - 1e-15:
                best_c = c_val
                best_d = d
                best_nc = nc
        if best_nc is not None:
            h[j] += best_d
            h[k] -= best_d
            corr = best_nc
            c5 = best_c
            if (it + 1) % update_every == 0:
                E[j] = compute_effect(h, j)
                E[k] = compute_effect(h, k)
    h = project_feasible(h, n / 2.0)
    c5 = compute_c5(h)
    return h, c5

def binary_swap_optimize(h_bin, n_iters=5000, t0=None, time_limit=None, seed=42):
    """Optimize a binary sequence via 0->1, 1->0 swaps"""
    n = len(h_bin)
    dx = 2.0 / n
    L = 2 * n - 1
    h_bin = h_bin.copy().astype(float)
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full').astype(float)
    c5 = float(np.max(corr) * dx)
    rng = np.random.RandomState(seed)
    ones = np.where(h_bin > 0.5)[0]
    zeros = np.where(h_bin < 0.5)[0]

    for it in range(n_iters):
        if t0 is not None and time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        if len(ones) == 0 or len(zeros) == 0:
            break
        k = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if j == k:
            continue
        diff = compute_effect(h_bin, j) - compute_effect(h_bin, k)
        nc = corr + diff
        c_val = float(nc.max() * dx)
        if c_val < c5 - 1e-15:
            h_bin[j] = 1.0
            h_bin[k] = 0.0
            corr = nc
            c5 = c_val
            ones = np.where(h_bin > 0.5)[0]
            zeros = np.where(h_bin < 0.5)[0]
    return h_bin, c5
