from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def logsumexp_smooth_max(values, t):
    # Smooth max approximation
    m = np.max(values)
    if m > 700 / t:  # avoid overflow
        m = 700 / t
    return m + np.log(np.sum(np.exp(t * (values - m)))) / t

def objective(h_flat, t):
    n = len(h_flat)
    dx = 2.0 / n
    h = h_flat
    # Compute all correlation values
    corr = np.correlate(h, 1 - h, mode="full") * dx
    # Smooth max
    return logsumexp_smooth_max(corr, t)

def project(h, target_sum):
    # Project h onto {0 ≤ h ≤ 1, sum(h) = target_sum}
    # Use bisection on the threshold
    h = np.clip(h, 0, 1)
    # If sum is already close, return
    s = np.sum(h)
    if abs(s - target_sum) < 1e-10:
        return h
    # Bisection for the water level
    lo, hi = 0.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)
