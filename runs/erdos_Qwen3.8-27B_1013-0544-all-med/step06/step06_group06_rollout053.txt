        h2 = project_feasible(res.x, n)
        return h2, compute_c5(h2, n)
    except Exception:
        return h, compute_c5(h, n)

def interp_to(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def bang_bang(rng, n):
    h = np.zeros(n)
    h[rng.choice(n, size=n // 2, replace=False)] = 1.0
    return h

def structured_init(n, mode=0):
    x = np.arange(n) / n
    if mode == 0:
        h = np.where(np.sin(2 * np.pi * x) > 0, 1.0, 0.0)
    elif mode == 1:
        h = np.where(np.sin(4 * np.pi * x) > 0.3, 1.0, 0.0)
    elif mode == 2:
        h = np.where((x % 0.25) < 0.125, 1.0, 0.0)
    elif mode == 3:
        h = np.where(np.cos(2 * np.pi * 3 * x) > -0.2, 1.0, 0.0)
    else:
        h = np.where(x < 0.5, 1.0, 0.0)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 80
        h0 = np.full(n0, 0.5)
    h0 = project_feasible(h0, n0)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)

    def remaining():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP continuation on n0
    for alpha in [50, 200, 1000, 3000, 8000]:
        if remaining() < 40:
            break
        h_t, c = slsqp_run(best_h, n0, alpha, maxiter=3000)
        if c < best_c5:
            best_h, best_c5 = h_t, c

    # Phase 2: Coordinate descent polish on n0
    if remaining() > 40:
        hd = best_h.copy()
        hd, c = coord_descent(hd, n0, max_passes=8000, deadline=t0 + remaining() - 10)
        if c < best_c5:
            best_h, best_c5 = hd, c

    # Phase 3: Multi-resolution
    n_parent = n0
    h_parent = best_h.copy()
    for n_try in [n0 * 2, n0 + 80, n0 * 3, n0 * 4]:
        if n_try > 480 or n_try <= n0 or remaining() < 100:
            continue
        h_interp = interp_to(h_parent, n_parent, n_try)
        # SLSQP on interpolated
        if remaining() > 60:
            h_t, c = slsqp_run(h_interp, n_try, 5000, maxiter=2000)
            if c < best_c5:
                best_h, best_c5 = h_t, c
                n_parent, h_parent = n_try, h_t
        # Coordinate descent
        if remaining() > 30:
            hd, c = coord_descent(best_h if len(best_h) == n_try else h_interp,
                                  n_try, max_passes=4000, deadline=t0 + remaining() - 10)
            if c < best_c5:
                best_h, best_c5 = hd, c
                n_parent, h_parent = n_try, hd

    # Phase 4: Structured and random restarts with coordinate descent
    n_best = len(best_h)
    restart_count = 0
    while remaining() > 25 and restart_count < 30:
        restart_count += 1
        mode = restart_count % 5
        if mode == 4:
            h_start = best_h + rng.normal(0, 0.08, size=n_best)
        elif mode == 3:
            h_start = bang_bang(rng, n_best)
        else:
            h_start = structured_init(n_best, mode=mode % 4)
        hd, c = coord_descent(h_start, n_best, max_passes=2000,
                              deadline=t0 + remaining() - 8)
        if c < best_c5:
            best_h, best_c5 = hd, c
            n_best = len(best_h)

    # Phase 5: Try different n with random restarts
    for n_alt in [n0, n0 + 40, n0 * 2]:
        if remaining() < 30 or n_alt == n_best:
            continue
        for trial in range(3):
            if remaining() < 15:
                break
            h_s = structured_init(n_alt, mode=trial) if trial < 3 else bang_bang(rng, n_alt)
            hd, c = coord_descent(h_s, n_alt, max_passes=1500,
                                  deadline=t0 + remaining() - 8)
            if c < best_c5:
                best_h, best_c5 = hd, c
                n_best = n_alt

    # Phase 6: Final CD polish
    if remaining() > 5:
        hd, c = coord_descent(best_h, len(best_h), max_passes=5000,
                              deadline=t0 + remaining() - 2)
        if c < best_c5:
            best_h, best_c5 = hd, c

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h, len(best_h))
    return best_h, best_c5, len(best_h)
