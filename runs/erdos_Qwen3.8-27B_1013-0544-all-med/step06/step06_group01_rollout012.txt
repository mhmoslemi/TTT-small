    """Project h onto {x: 0<=x[i]<=1, sum(x)=n/2}"""
    target = 0.5 * n
    # Binary search for threshold
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fix small discrepancy
    diff = target - np.sum(proj)
    if abs(diff) > 1e-10:
        if diff > 0:
            # Add to elements where proj < 1
            candidates = np.where(proj < 1.0)[0]
            # Add proportionally
            for idx in candidates:
                add = min(diff, 1.0 - proj[idx])
                proj[idx] += add
                diff -= add
                if diff <= 0:
                    break
        else:
            candidates = np.where(proj > 0.0)[0]
            for idx in candidates:
                rem = min(-diff, proj[idx])
                proj[idx] -= rem
                diff += rem
                if diff >= 0:
                    break
    return proj

    return np.max(_c5_array(h))

def _project(h, n):

    for _ in range(80):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

    h[idx[:half]] = 1.0
    if n % 2 == 1:
        h[idx[half]] = 0.5
    return _project(h, n)

def _make_sine(n, rng):
    freq = rng.uniform(1.0, 6.0)
    amp = rng.uniform(0.15, 0.48)
    phase = rng.uniform(0, 2 * np.pi)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
    return _project(h + rng.randn(n) * 0.03, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _rem():
        return budget_s - (time.time() - t0)

    n0 = len(initial_h_values)
    h_best = _project(initial_h_values.astype(float).copy(), n0)
    c5_best = _c5_max(h_best)
    n_best = n0

    # Phase 1: Progressive alpha optimization on initial
    h_cur = h_best.copy()
    for alpha in [10, 30, 100, 300, 1000]:
        if _rem() < 30:
            break
        beta = max(1.0, 0.02 * alpha)
        h_new = _opt_lbfgsb(h_cur, n_best, alpha, beta, maxiter=400)
        c5_new = _c5_max(h_new)
        if c5_new < c5_best - 1e-12:
            c5_best, h_best, h_cur = c5_new, h_new, h_new

    # Phase 2: Coordinate descent polish
    if _rem() > 60:
        h_cd = _coord_descent(h_best, n_best, max_sweeps=20)
        c5_cd = _c5_max(h_cd)
        if c5_cd < c5_best - 1e-12:
            c5_best, h_best = c5_cd, h_cd

    # Phase 3: Try nearby resolutions via interpolation
    for n_try in [n0 - 5, n0 + 5, n0 + 10, n0 - 10, 90, 100, 110]:
        if _rem() < 60:
            break
        if n_try <= 0 or n_try == n_best or abs(n_try - n_best) < 3:
            continue
        x_old = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_int = _project(np.interp(x_new, x_old, h_best), n_try)
        for alpha in [50, 200, 500]:
            if _rem() < 30:
                break
            beta = max(1.0, 0.02 * alpha)
            h_new = _opt_lbfgsb(h_int, n_try, alpha, beta, maxiter=300)
            c5_new = _c5_max(h_new)
            if c5_new < c5_best - 1e-12:
                c5_best, h_best, n_best = c5_new, h_new, n_try
                h_int = h_new

    # Phase 4: Diverse random restarts
    n_attempts = [n_best, n_best - 5, n_best + 5, 90, 100]
    n_attempts = [max(50, min(150, x)) for x in n_attempts]
    for n_try in n_attempts:
        if _rem() < 80:
            break
        for r in range(15):
            if _rem() < 25:
                break
            strat = (r + int(n_try)) % 4
            if strat == 0:
                h_init = _make_binary(n_try, rng)
            elif strat == 1:
                h_init = _make_sine(n_try, rng)
            elif strat == 2:
                # Two-frequency sine
                f1, f2 = rng.uniform(1, 4), rng.uniform(3, 8)
                a1, a2 = rng.uniform(0.1, 0.3), rng.uniform(0.1, 0.2)
                p1, p2 = rng.uniform(0, 6.28), rng.uniform(0, 6.28)
                hh = 0.5 + a1 * np.sin(2*np.pi*f1*np.arange(n_try)/n_try + p1) + \
                     a2 * np.sin(2*np.pi*f2*np.arange(n_try)/n_try + p2)
                h_init = _project(hh, n_try)
            else:
                # Perturbed best at this resolution
                if n_try == n_best:
                    h_init = _project(h_best + rng.randn(n_best) * 0.08, n_best)
                else:
                    xo = np.linspace(0, 2, n_best, endpoint=False)
                    xn = np.linspace(0, 2, n_try, endpoint=False)
                    h_init = _project(np.interp(xn, xo, h_best) + rng.randn(n_try) * 0.05, n_try)
            for alpha in [50, 200]:
                if _rem() < 20:
                    break
                beta = max(1.0, 0.02 * alpha)
                h_opt = _opt_lbfgsb(h_init, n_try, alpha, beta, maxiter=250)
                c5_val = _c5_max(h_opt)
                if c5_val < c5_best - 1e-12:
                    c5_best, h_best, n_best = c5_val, h_opt, n_try

    # Phase 5: Final polish - high alpha + coord descent
    if _rem() > 40:
        for alpha in [1000, 3000]:
            if _rem() < 15:
                break
            beta = max(5.0, 0.02 * alpha)
            h_pol = _opt_lbfgsb(h_best, n_best, alpha, beta, maxiter=500)
            if _c5_max(h_pol) < c5_best - 1e-12:
                h_best = h_pol
                c5_best = _c5_max(h_best)
    if _rem() > 20:
        h_final = _coord_descent(h_best, n_best, max_sweeps=30)
        cf = _c5_max(h_final)
        if cf < c5_best - 1e-12:
            h_best, c5_best = h_final, cf

    h_best = _project(h_best, n_best)
    c5_best = _c5_max(h_best)
    return (h_best, c5_best, n_best)
