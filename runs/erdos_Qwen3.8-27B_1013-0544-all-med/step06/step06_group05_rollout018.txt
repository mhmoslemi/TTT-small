```python
import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def local_search_binary_fast(h0, max_time=60.0, seed=42):
    n = len(h0)
    h = make_binary(h0, n)
    current_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    
    while (time.time() - start) < max_time:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # Try all single swaps (randomized order for diversification)
        swaps = []
        for i in ones:
            for j in zeros:
                swaps.append((i, j))
        rng.shuffle(swaps)
        
        best_gain = 1e-14
        best_move = None
        for i, j in swaps:
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            gain = current_c5 - new_c5
            if gain > best_gain:
                best_gain = gain
                best_move = (i, j)
            h[i] = 1.0
            h[j] = 0.0
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0.0
            h[j] = 1.0
            current_c5 = compute_c5(h)
            improved = True
        else:
            break
    
    return h, current_c5
```

```python
    n = len(h0)

    while time.time() - start < max_time:

            i = ones[idx // len(zeros)]
            j = zeros[idx % len(zeros)]
            
            # Compute new max quickly
```

This is a useful criterion! We can use it to guide our search.

**Algorithm: Directed local search**

1. Find the shift k* with maximum correlation.
2. Among all 1-positions i, compute score_i = h[i+k*] + h[i-k*].
3. Among all 0-positions j, compute score_j = h[j-k*] + h[j+k*].
4. Try swaps where score_j > score_i (these reduce cv[k*]).
5. Among those, pick the one that best reduces the overall max.

This should be much more efficient than random swaps!

Actually, even better: we can compute the effect of a swap on ALL correlations simultaneously in O(n) time using the formula above. Then we can find the best swap in O(n²) time (trying all i,j pairs) but each evaluation is O(1) if we precompute the scores.

Wait, the effect on cv[k] for a specific k is h[i+k] + h[i-k] - h[j-k] - h[j+k]. But we need the effect on ALL k to know the new max. So it's still O(n) per swap.

However, we can precompute for each i: vector v_i[k] = h[i+k] + h[i-k] for all k.
And for each j: vector w_j[k] = h[j-k] + h[j+k] for all k.

Then the change in cv[k] for swap (i,j) is v_i[k] - w_j[k].
The new cv[k] = old_cv[k] + v_i[k] - w_j[k].
The new max = max_k (old_cv[k] + v_i[k] - w_j[k]).

We can precompute v_i and w_j for all i, j in O(n²) time (each is O(n) to compute, and there are n of them). Then trying all n²/4 swaps costs O(n² * n) = O(n³) total (since each max computation is O(n)).

For n=112: 112³ ≈ 1.4M. This is fast!

```python
def directed_local_search(h0, max_time, seed=42):

    h = make_binary(h0, n)
    corr = np.correlate(h.astype(float), 1.0-h, mode='full')
    current_max = float(corr.max())
    start = time.time()
    rng = np.random.RandomState(seed)
    
    dx = 2.0 / n
    
    while time.time() - start < max_time:
        if current_max * dx < 0.3808:
            break
        
        # Find the worst shift
        worst_idx = int(np.argmax(corr))
        # Convert to lag: lag = worst_idx - (n-1)
        worst_lag = worst_idx - (n - 1)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_gain = 1e-10
        best_i, best_j = -1, -1
        
        # Try swaps that target the worst shift
        for i in ones:
            for j in zeros:
                # Effect on cv at worst_lag
                # cv[lag] changes by: h[i+lag] + h[i-lag] - h[j-lag] - h[j+lag]
                # (with h out of bounds = 0)
                
                # Quick check: does this reduce the worst shift's correlation?
                vi = (h[i+worst_lag] if 0 <= i+worst_lag < n else 0) + \
                     (h[i-worst_lag] if 0 <= i-worst_lag < n else 0)
                vj = (h[j-worst_lag] if 0 <= j-worst_lag < n else 0) + \
                     (h[j+worst_lag] if 0 <= j+worst_lag < n else 0)
                
                if vj <= vi:
                    continue  # This swap wouldn't reduce the worst shift
                
                # Full evaluation
                h[i] = 0; h[j] = 1
                new_corr = np.correlate(h.astype(float), 1.0-h, mode='full')
                new_max = float(new_corr.max())
                h[i] = 1; h[j] = 0
                
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i] = 0; h[best_j] = 1
            corr = np.correlate(h.astype(float), 1.0-h, mode='full')
            current_max = float(corr.max())
        else:
            # Try double swaps
            break
    
    return h, current_max * dx
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy().round()
    h = np.clip(h, 0, 1)
    n_ones = int(h.sum())
    target = n // 2

        h[idx[:target-n_ones]] = 1
    return h.astype(float)
