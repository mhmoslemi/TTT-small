    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n

                if total_room < 1e-15: break
                add = min(d, total_room)
                h[mask] += add * room / total_room

                avail = h[mask]
                total_avail = avail.sum()
                if total_avail < 1e-15: break

                h[mask] -= sub * avail / total_avail

    m = 2 * n - 1  # number of lags
    rng = np.random.RandomState(seed)
    best_c5 = np.max(overlap)
    
    for sweep in range(max_sweeps):

        improved = False

def _l-bfgs_smooth(h, n, alpha, beta, maxiter=500):
    # Smooth max objective with gradient
    dx = 2.0 / n
    m = 2 * n - 1
    
    def obj_grad(x):
        x = np.clip(x, 0, 1)
        C = np.correlate(x, 1.0 - x, mode='full') * dx
        sh = alpha * C
        mx = np.max(sh)
        es = np.exp(sh - mx)
        se = es.sum()
        f = (mx + np.log(se)) / alpha + beta * (np.sum(x) - 0.5 * n) ** 2
        w = es / se
        grad = np.zeros(n)
        for i in range(n):
            ks = np.arange(m)
            j1 = i - ks + (n - 1)
            j2 = i + ks - (n - 1)
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            ta = np.zeros(m)
            tb = np.zeros(m)
            ta[v1] = 1.0 - x[j1[v1]]
            tb[v2] = x[j2[v2]]
            grad[i] = dx * np.dot(w, ta - tb)
        grad += 2.0 * beta * (np.sum(x) - 0.5 * n)
        return f, grad

                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix_sum(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = 0.5 * n
    for _ in range(500):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
            if not np.any(mask):
                break
            room = (1.0 - h)[mask]
            tr = room.sum()
            if tr < 1e-14:
                break
            h[mask] = np.minimum(h[mask] + (d / tr) * room, 1.0)
        else:
            mask = h > 1e-14
            if not np.any(mask):
                break
            avail = h[mask]
            ta = avail.sum()
            if ta < 1e-14:
                break
            h[mask] = np.maximum(h[mask] - (-d / ta) * avail, 0.0)
    return h

def _coord_descent(h, n, max_sweeps=300, seed=42, tol=1e-11):

    m = 2 * n - 1

    h = h.copy()
    overlap = _c5_array(h)
    best = np.max(overlap)
    for sweep in range(max_sweeps):
        if sweep > 0 and sweep % 200 == 0:
            overlap = _c5_array(h)
        order = rng.permutation(n)
        any_improved = False
        for i in order:
            ks = np.arange(m)
            j1 = i - ks + (n - 1)
            j2 = i + ks - (n - 1)
            v1 = (j1 >= 0) & (j1 < n)
            v2 = (j2 >= 0) & (j2 < n)
            g = np.zeros(m)
            if np.any(v1):
                g[v1] = 1.0 - h[j1[v1]]
            if np.any(v2):
                g[v2] -= h[j2[v2]]

                overlap += t * g
                best = new_max

    h = _fix_sum(h, n)

def _smooth_obj_grad(x, alpha, beta, n):
    dx = 2.0 / n
    C = np.correlate(x, 1.0 - x, mode='full') * dx
    sh = alpha * C
    mx = np.max(sh)
    es = np.exp(sh - mx)
    se = es.sum()
    f = (mx + np.log(se)) / alpha + beta * (np.sum(x) - 0.5 * n) ** 2
    w = es / se
    m = len(C)
    grad = np.zeros(n)
    for i in range(n):
        ks = np.arange(m)
        j1 = i - ks + (n - 1)
        j2 = i + ks - (n - 1)
        v1 = (j1 >= 0) & (j1 < n)
        v2 = (j2 >= 0) & (j2 < n)
        ta = np.zeros(m)
        tb = np.zeros(m)
        ta[v1] = 1.0 - x[j1[v1]]
        tb[v2] = x[j2[v2]]
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(x) - 0.5 * n)
    return f, grad

def _l_bfgs_opt(h, n, alpha, beta, maxiter=300):
    res = minimize(_smooth_obj_grad, h, args=(alpha, beta, n), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
    if res.x is None:
        return h
    return _fix_sum(res.x, n)
