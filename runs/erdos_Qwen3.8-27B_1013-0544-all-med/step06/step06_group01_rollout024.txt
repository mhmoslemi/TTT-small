```python
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _project_sum(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection."""
    h = h.copy()
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(h - lam, 0.0, 1.0)
    # Fix tiny residual
    d = target_sum - np.sum(h)
    if abs(d) > 1e-15:
        h = h + d / len(h)
        h = np.clip(h, 0.0, 1.0)
    return h

def _smoothmax_obj(h, alpha, n):
    """Negative log-sum-exp style smooth max objective."""
    C = _c5_array(h)
    m = np.max(C)
    if alpha * m > 700:
        # Overflow protection
        return alpha * m
    es = np.exp(alpha * (C - m))
    val = m + np.log(np.sum(es)) / alpha
    return val

def _smoothmax_grad(h, alpha, n):
    """Gradient of smooth max."""
    C = _c5_array(h)
    dx = 2.0 / n
    m = np.max(C)
    sh = alpha * (C - m)
    es = np.exp(sh)
    se = np.sum(es)
    w = es / se
    nc = len(C)

    # C[k] = dx * sum_j h[j]*(1-h[j+lag_k]) where lag_k = k-(n-1)
    # dC[k]/dh[i] = dx * [(1-h[i+lag_k])*I(0<=i+lag_k<n) - h[i-lag_k]*I(0<=i-lag_k<n)]
    
    # Vectorized approach: for each i, sum over k of w[k] * (t1[k,i] - t2[k,i])
    # This is O(n*nc) = O(n^2)
    
    i_arr = np.arange(n)
    grad = np.zeros(n)
    
    # For efficiency, process in chunks or use the structure
    # Actually for n~100, the O(n^2) loop is fine
    for k in range(nc):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        ip = i_arr + lag
        im = i_arr - lag
        t1 = np.where((ip >= 0) & (ip < n), 1.0 - h[np.clip(ip, 0, n-1)], 0.0)
        t2 = np.where((im >= 0) & (im < n), h[np.clip(im, 0, n-1)], 0.0)
        grad += wk * dx * (t1 - t2)
    
    return grad

    return _smoothmax_obj(h, alpha, n), _smoothmax_grad(h, alpha, n)

    h = h.copy()

                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _project_sum(h, target_sum):
    h = h.copy()
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(h - lam, 0.0, 1.0)
    d = target_sum - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target_sum - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.argmax(1.0 - h)
                h[idx] = min(1.0, h[idx] + d2)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + d2)
            h = np.clip(h, 0.0, 1.0)
    return h

def _smooth_obj_grad(h, alpha, n):
    C = _c5_array(h)
    dx = 2.0 / n
    m = np.max(C)
    sh = alpha * (C - m)
    # Clip to avoid overflow
    sh = np.clip(sh, -500, 500)
    es = np.exp(sh)
    se = np.sum(es)
    f = m + np.log(se) / alpha
    w = es / se
    nc = len(C)
    i_arr = np.arange(n)
    grad = np.zeros(n)
    for k in range(nc):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        ip = i_arr + lag
        im = i_arr - lag
        t1 = np.where((ip >= 0) & (ip < n), 1.0 - h[np.clip(ip, 0, n-1)], 0.0)
        t2 = np.where((im >= 0) & (im < n), h[np.clip(im, 0, n-1)], 0.0)
        grad += wk * dx * (t1 - t2)
    return f, grad

def _optimize_smooth(h, n, alpha, maxiter=500):
    h = h.copy()
    res = minimize(_smooth_obj_grad, h, args=(alpha, n), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 20, 'ftol': 1e-18, 'gtol': 1e-12})
    if res.x is not None:
        h = _project_sum(res.x, 0.5 * n)
    else:
        h = _project_sum(h, 0.5 * n)
    return h

def _subgrad_polish(h, n, iters=300, top_k=5):
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_max(h)
    dx = 2.0 / n
    i_arr = np.arange(n)
    
    for it in range(iters):
        C = _c5_array(h)
        current_max = np.max(C)
        top_indices = np.argsort(C)[-top_k:]
        g = np.zeros(n)
        for k in top_indices:
            lag = k - (n - 1)
            ip = i_arr + lag
            im = i_arr - lag
            t1 = np.where((ip >= 0) & (ip < n), 1.0 - h[np.clip(ip, 0, n-1)], 0.0)
            t2 = np.where((im >= 0) & (im < n), h[np.clip(im, 0, n-1)], 0.0)
            g += dx * (t1 - t2)
        
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        
        # Adaptive step size
        step = 0.1 / (1.0 + 0.01 * it)
        
        h_new = _project_sum(h - step * g, 0.5 * n)
        c5_new = _c5_max(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
        else:
            # Try smaller step
            for factor in [0.5, 0.25, 0.1]:
                h_new = _project_sum(h - step * factor * g, 0.5 * n)
                c5_new = _c5_max(h_new)
                if c5_new < current_max - 1e-13:
                    h = h_new
                    if c5_new < best_c5:
                        best_h = h_new.copy()
                        best_c5 = c5_new
                    break
    
    return best_h, best_c5

def _local_search(h, n, max_rounds=200, rng=None):
    h = h.copy()
    best_c5 = _c5_max(h)
    if rng is None:
        rng = np.random.RandomState(0)
    
    for rd in range(max_rounds):
        improved = False
        # Try random pairs
        for _ in range(min(500, n * 5)):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for delta in [0.002, 0.005, 0.01, 0.02]:
                for di, dj in [(delta, -delta), (-delta, delta)]:
                    h2 = h.copy()
                    h2[i] += di
                    h2[j] += dj
                    if h2[i] < 0 or h2[i] > 1 or h2[j] < 0 or h2[j] > 1:
                        continue
                    c5 = _c5_max(h2)
                    if c5 < best_c5 - 1e-11:
                        h = h2
                        best_c5 = c5
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c5
```

Let me also add one more idea: **coordinate descent with exact line search**. For each coordinate i, find the best value of h[i] (given all others fixed) that minimizes the max C_k. Since the max of linear functions is piecewise linear, the optimal h[i] is at a "kink" point or at a boundary.

For a fixed h[i] = v, C_k is linear in v (with slope depending on whether h[i] appears in C_k or not). The max of these linear functions is convex in v, so we can find the minimum by bisection or by checking the kink points.

The slope of C_k with respect to h[i] is:
- dx * (1 - h[i+lag_k]) if i+lag_k is valid (h[i] appears as h[j])
- -dx * h[i-lag_k] if i-lag_k is valid (h[i] appears as h[j+lag])
- Both if both are valid

So the slope for C_k w.r.t. h[i] is:
s_k = dx * [(1-h[i+lag_k])*I(valid) - h[i-lag_k]*I(valid)]

And C_k(v) = C_k(current) + s_k * (v - h[i])

The max_k C_k(v) is the upper envelope of these lines. The minimum is where the slope changes sign (from negative to positive).

This is a 1D convex optimization problem! We can solve it exactly.

```python
def _coordinate_descent(h, n, sweeps=50):
    h = h.copy()
    best_c5 = _c5_max(h)
    dx = 2.0 / n
    nc = 2 * n - 1
    i_arr = np.arange(n)
    
    for sweep in range(sweeps):
        improved = False
        for i in range(n):
            old_val = h[i]
            # Compute slope of each C_k w.r.t. h[i]
            slopes = np.zeros(nc)
            for k in range(nc):
                lag = k - (n - 1)
                s = 0.0
                ip = i + lag
                im = i - lag
                if 0 <= ip < n:
                    s += (1.0 - h[ip])
                if 0 <= im < n:
                    s -= h[im]
                slopes[k] = dx * s
            
            # Current C_k values
            C = _c5_array(h)
            # C_k(v) = C_k + slopes[k] * (v - old_val)
            # Find v in [0,1] that minimizes max_k (C_k + slopes[k]*(v - old_val))
            # This is a 1D convex minimization
            
            # The minimum is where the max subgradient contains 0
            # i.e., where there exist active k+ (slopes > 0) and k- (slopes < 0)
            # such that max is achieved by both
            
            # Simple approach: bisection on the subgradient
            # Subgradient at v: {sum of slopes of active k at v}
            # For minimize: we want subgradient to contain 0
            
            # Actually, let's just do a golden section or ternary search
            # Since it's convex and unimodal on [0,1]
            
            def f1d(v):
                return np.max(C + slopes * (v - old_val))
            
            # Golden section search on [0, 1]
            gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate
            a, b = 0.0, 1.0
            c_ = b - gr * (b - a)
            d_ = a + gr * (b - a)
            fc = f1d(c_)
            fd = f1d(d_)
            for _ in range(50):
                if fc < fd:
                    b = d_
                    d_ = c_
                    fd = fc
                    c_ = b - gr * (b - a)
                    fc = f1d(c_)
                else:
                    a = c_
                    c_ = d_
                    fc = fd
                    d_ = a + gr * (b - a)
                    fd = f1d(d_)
            
            v_opt = (a + b) / 2.0
            h[i] = v_opt
            c5_new = _c5_max(h)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                improved = True
        
        if not improved:
            break
    
    return h, best_c5
```

```python
def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _project_sum(h, target_sum):

        else:
            hi = mid
```
