def add_lag_grad(x, grad, k, scale):
    if scale == 0.0:
        return
    n = x.size
    s = int(k) - (n - 1)
    # s is the lag
    # For the first term: (1-x[j+s]) for j where j+s is valid
    #   j+s in [0,n-1] => j in [-s, n-1-s]
    #   clamped to [0, n-1]
    lo = max(0, -s)
    hi = min(n-1, n-1-s)
    if lo <= hi:
        grad[lo:hi+1] += scale * (1.0 - x[lo+s:hi+1+s])
    # For the second term: -x[j-s] for j where j-s is valid
    #   j-s in [0,n-1] => j in [s, s+n-1]
    #   clamped to [0, n-1]
    lo = max(0, s)
    hi = min(n-1, s+n-1)
    if lo <= hi:
        grad[lo:hi+1] -= scale * x[lo-s:hi+1-s]

    """Project x onto {y : 0 <= y_i <= 1, sum(y) = target}."""
    # Use bisection on the water-filling parameter
    lo, hi = -1.0, 1.0  # range of the multiplier
    for _ in range(60):

        y = np.clip(x - mid, 0.0, 1.0)

        if s > target:
            lo = mid  # need to subtract more

            hi = mid  # need to subtract less

import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")

_T = [0.01]


def _tl(d):
    return max(0.0, d - time.time())


def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    tgt = 0.5 * n
    np.clip(x, 0, 1, out=x)
    for _ in range(30):
        d = tgt - x.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = 1 - x > 1e-14
        else:
            m = x > 1e-14
        if m.any():
            if d > 0:
                x[m] += d * (1 - x[m]) / (1 - x[m]).sum()
            else:
                x[m] += d * x[m] / x[m].sum()
        np.clip(x, 0, 1, out=x)
    d = tgt - x.sum()
    if abs(d) > 1e-10:
        k = int(tgt)
        x[:] = 0.0
        x[:k] = 1.0
        if k < n:
            x[k] = tgt - k
    return x


def cvals(h):
    return np.correlate(h, 1 - h, mode="full") * (2.0 / h.size)


def ec(h):
    return float(cvals(h).max())


def _f(x):
    c = cvals(x)
    cm = c.max()
    t = _T[0]
    if t < 1e-6:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return cm, c, w
    z = (c - cm) / t
    z -= z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return cm, c, w
    return cm + t * np.log(s), c, e / s


def _fun(x):
    return _f(x)[0]


def _jac(x):
    n = x.size
    dx = 2.0 / n
    _, _, w = _f(x)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-13:
            continue
        lag = k - (n - 1)
        a = max(0, -lag)
        b = min(n - 1, n - 1 - lag)
        if a <= b:
            g[a:b + 1] += wk * dx * (1 - x[a + lag:b + lag + 1])
        a = max(0, lag)
        b = min(n - 1, lag + n - 1)
        if a <= b:
            g[a:b + 1] -= wk * dx * x[a - lag:b - lag + 1]
    return g


def _cf(x):
    return np.array([x.sum() - 0.5 * x.size])


def _cj(x):
    return np.ones((1, x.size))


def slsqp(x0, mi):
    n = x0.size
    try:
        r = so.minimize(_fun, x0, method="SLSQP", jac=_jac,
                        bounds=[(0.0, 1.0)] * n,
                        constraints=[{"type": "eq", "fun": _cf, "jac": _cj}],
                        options={"maxiter": mi, "ftol": 1e-14})
        return repair(r.x)
    except Exception:
        return x0


def lsearch(x, tl):
    x = x.copy()
    bx, bc = x.copy(), ec(x)
    st = time.time()
    step = 0.06
    ni = 0
    while time.time() - st < tl:
        c = cvals(x)
        C = c.max()
        if C < bc - 1e-14:
            bc, bx, ni = C, x.copy(), 0
        else:
            ni += 1
            if ni > 20:
                break
        tol = max(1e-9, 1e-5 * C)
        act = np.where(c >= C - tol)[0]
        if act.size > 8:
            act = np.argsort(c)[::-1][:8]
        w = np.zeros(c.size)
        w[act] = 1.0 / act.size
        n = x.size
        dx = 2.0 / n
        g = np.zeros(n)
        for k in act:
            lag = int(k) - (n - 1)
            a = max(0, -lag)
            b = min(n - 1, n - 1 - lag)
            if a <= b:
                g[a:b + 1] += dx * (1 - x[a + lag:b + lag + 1])
            a = max(0, lag)
            b = min(n - 1, lag + n - 1)
            if a <= b:
                g[a:b + 1] -= dx * x[a - lag:b - lag + 1]
        g /= act.size
        if np.linalg.norm(g) < 1e-14:
            break
        done = False
        for _ in range(12):
            y = repair(x - step * g)
            if ec(y) < C - 1e-13:
                x = y
                done = True
                break
            step *= 0.5
            if step < 1e-12:
                break
        if not done:
            ni += 1
    return bx, bc


def resample(h0, nn):
    if h0.size == nn:
        return repair(h0)
    x0 = (np.arange(h0.size) + 0.5) * (2.0 / h0.size)
    x1 = (np.arange(nn) + 0.5) * (2.0 / nn)
    return repair(np.interp(x1, x0, h0))


def mk_start(h0, n, seed, freq, amp):
    t = (np.arange(n) + 0.5) * (2.0 / n)
    rng = np.random.RandomState(seed)
    h = h0.copy()
    h += amp * np.sin(2 * np.pi * freq * t + rng.uniform(0, 6.283))
    h += 0.02 * rng.normal(0, 1, n)
    return repair(h)


def opt_res(h0, tl, seed):
    x = repair(h0)
    bx, bc = x.copy(), ec(x)
    st = time.time()
    n = x.size
    sf = 1.0 if n <= 100 else (0.75 if n <= 160 else 0.5)
    for tau, mi in [(0.02, 150), (0.008, 200), (0.003, 250), (0.001, 250)]:
        if time.time() - st > tl * 0.7:
            break
        _T[0] = tau
        xr = slsqp(x, int(mi * sf))
        c = ec(xr)
        if c < bc:
            bc, bx = c, xr.copy()
        x = bx if c < bc else xr
    rem = tl - (time.time() - st)
    if rem > 2:
        xp, cp = lsearch(bx, min(rem * 0.4, 12))
        if cp < bc:
            bc, bx = cp, xp
    return bx, bc


def run(seed=42, budget_s=1100, **kwargs):
    dl = time.time() + max(1.0, float(budget_s))
    sv = int(seed)
    raw = globals().get("initial_h_values", None)
    init = np.asarray(raw, dtype=float) if raw is not None else repair(np.full(70, 0.5))
    best_x = init.copy()
    best_c = ec(init)

    res_list = [100, 140, 200, 280]
    weights = [0.20, 0.30, 0.28, 0.22]

    for ri, (n_r, wt) in enumerate(zip(res_list, weights)):
        if _tl(dl) < 35:
            break
        alloc = wt * budget_s * 0.85
        alloc = min(alloc, _tl(dl) * 0.6)
        alloc = max(12.0, alloc)

        base = resample(init, n_r)
        starts = [base]
        for j, (fq, am) in enumerate([(1, 0.04), (2, 0.06), (3, 0.08)]):
            starts.append(mk_start(base, n_r, sv + ri * 100 + j, fq, am))

        ns = min(len(starts), 3 if n_r <= 160 else 2)
        ps = alloc / ns

        for si in range(ns):
            if _tl(dl) < 12:
                break
            x, c = opt_res(starts[si], ps, sv + si)
            if c < best_c:
                best_c, best_x = c, x

    best_x = repair(best_x)
    best_c = ec(best_x)
    return best_x, best_c, int(best_x.size)

    if s <= 0 or not np.isfinite(s):
        w = np.zeros(c.size)
        w[c.argmax()] = 1.0
        return cm, c, w
    return cm + t * np.log(s), c, e / s


def _fun(x):
    return _smw(x)[0]


def _grad(x, w):
    n = x.size
    dx = 2.0 / n
    g = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if abs(wk) < 1e-13:
            continue
        lag = k - (n - 1)
        a = max(0, -lag)
        b = min(n - 1, n - 1 - lag)
        if a <= b:
            g[a:b + 1] += wk * dx * (1 - x[a + lag:b + lag + 1])
        a = max(0, lag)
        b = min(n - 1, lag + n - 1)
        if a <= b:
            g[a:b + 1] -= wk * dx * x[a - lag:b - lag + 1]
    return g


def _jac(x):
    return _grad(x, _smw(x)[2])


def _cf(x):
    return np.array([x.sum() - 0.5 * x.size])


def _cj(x):
    return np.ones((1, x.size))


def slsqp(x0, mi):
    n = x0.size
    try:
        r = so.minimize(_fun, x0, method="SLSQP", jac=_jac,
                        bounds=[(0.0, 1.0)] * n,
                        constraints=[{"type": "eq", "fun": _cf, "jac": _cj}],
                        options={"maxiter": mi, "ftol": 1e-14})
        return repair(r.x)
    except Exception:
        return x0


def lsearch(x, tl):
    x = x.copy()
    bx, bc = x.copy(), ec(x)
    st = time.time()
    step = 0.06
    ni = 0
    while time.time() - st < tl:
        c = cvals(x)
        C = c.max()
        if C < bc - 1e-14:
            bc, bx, ni = C, x.copy(), 0
        else:
            ni += 1
            if ni > 20:
                break
        tol = max(1e-9, 1e-5 * C)
        act = np.where(c >= C - tol)[0]
        if act.size > 8:
            act = np.argsort(c)[::-1][:8]
        w = np.zeros(c.size)
        w[act] = 1.0 / act.size
        g = _grad(x, w)
        if np.linalg.norm(g) < 1e-14:
            break
        done = False
        for _ in range(12):
            y = repair(x - step * g)
            if ec(y) < C - 1e-13:
                x = y
                done = True
                break
            step *= 0.5
            if step < 1e-12:
                break
        if not done:
            ni += 1
    return bx, bc
