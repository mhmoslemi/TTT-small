    grad = np.zeros(n)
    for j in top_indices:

        # ∂corr[j]/∂h[m] = (1-h[m-lag]) * 1_{0≤m-lag<n} - h[m+lag] * 1_{0≤m+lag<n}
        # For the first term: m-lag in [0, n-1] → m in [lag, lag+n-1], intersect [0, n-1]
        # For the second term: m+lag in [0, n-1] → m in [-lag, n-1-lag], intersect [0, n-1]
        m_lo1 = max(0, lag)
        m_hi1 = min(n - 1, n - 1 + lag)
        if m_lo1 <= m_hi1:
            m_range = np.arange(m_lo1, m_hi1 + 1)
            grad[m_range] += (1 - h[m_range - lag])

        if m_lo2 <= m_hi2:
            m_range = np.arange(m_lo2, m_hi2 + 1)
            grad[m_range] -= h[m_range + lag]
    return grad
