def soft_max_grad_fast(h, t, top_m=20):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    Z = np.sum(exps)
    weights = exps / Z
    
    # Find top M lags by weight
    top_idx = np.argsort(weights)[::-1][:top_m]
    top_idx = top_idx[weights[top_idx] > 1e-10]
    
    obj = (m + t * np.log(Z)) * dx
    grad = np.zeros(n)
    for idx in top_idx:
        w = weights[idx]
        k = idx - (n - 1)  # lag
        if k >= 0:
            # df/dh[i] = (1-h[i-k]) * 1[k<=i<n] + (-h[i+k]) * 1[0<=i<=n-1-k]
            if k == 0:
                grad[:] += w * ((1.0 - h) - h)  # (1-h[0..n-1]) - h[0..n-1] ... 
                # Wait: for k=0: (1-h[i]) * 1[0<=i<n] + (-h[i]) * 1[0<=i<=n-1]
                # = (1-h[i]) - h[i] = 1 - 2h[i] for all i
                # That's not right either. Let me recompute.
                # For k=0: f(0) = sum_{i=0}^{n-1} h[i]*(1-h[i])
                # df/dh[i] = (1-h[i]) - h[i] = 1 - 2h[i]
                grad[:] += w * (1.0 - 2.0 * h)
            else:
                # (1-h[i-k]) for i in [k, n-1]
                grad[k:] += w * (1.0 - h[:n-k])
                # -h[i+k] for i in [0, n-1-k]
                grad[:n-k] -= w * h[k:]
        else:
            r = -k
            # (1-h[i+r]) for i in [0, n-1-r]
            grad[:n-r] += w * (1.0 - h[r:])
            # -h[i-r] for i in [r, n-1]
            grad[r:] -= w * h[:n-r]
    
    return obj, grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def _grad_single_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        r = -k
        g[:n - r] += (1.0 - h[r:])
        g[r:] -= h[:n - r]
    return g

def soft_max_grad_fast(h, t, top_m=30):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(conv))
    exps = np.exp((conv - m) / t)
    Z = float(np.sum(exps))
    obj = (m + t * np.log(Z)) * dx
    weights = exps / Z
    top_idx = np.argsort(weights)[::-1][:top_m]
    top_idx = top_idx[weights[top_idx] > 1e-12]
    grad = np.zeros(n)
    for idx in top_idx:
        w = weights[idx]
        k = idx - (n - 1)
        grad += w * _grad_single_shift(h, k)
    return obj, grad * dx

def subgrad_at_max(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    c5 = float(conv[idx] * dx)
    k = idx - (n - 1)
    return c5, _grad_single_shift(h, k) * dx

def pgd_smooth_anneal(h_init, t_start, t_end, max_iter, lr_init):
    n = len(h_init)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    stuck = 0
    for it in range(max_iter):
        frac = it / max_iter
        t = t_start * (t_end / t_start) ** frac
        _, grad = soft_max_grad_fast(h, t)
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new <= compute_c5(h) + 1e-13:
            h = h_new
            lr = min(lr * 1.03, 0.08)
            stuck = 0
        else:
            lr *= 0.6
            stuck += 1
            if stuck > 80 or lr < 1e-16:
                break
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    return best_h, best_c5

def pgd_hard(h_init, max_iter, lr_init):

    stuck = 0

                break
        c = compute_c5(h)
        if c < best_c5:
            best_c5 = c
            best_h = h.copy()
    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    xo = np.arange(n_old) * (2.0 / n_old)
    xn = np.arange(n_new) * (2.0 / n_new)
    return np.interp(xn, xo, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(100, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Optimize at current resolution with annealing
    if tl() > 100:
        ht, c5t = pgd_smooth_anneal(best_h, 0.04, 0.004, 6000, 0.012)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        ht, c5t = pgd_hard(best_h, 30000, 0.006)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2: Try multiple resolutions
    res_list = [48, 64, 80, 96, 120, 144, 160, 192, 224, 256]
    if n0 not in res_list and n0 <= 400:
        res_list.append(n0)
    res_list = sorted(set(r for r in res_list if r <= 400))

    for n_try in res_list:
        if tl() < 150:
            break
        hi = project_feasible(interpolate_h(best_h, n_try), n_try)
        ht, c5t = pgd_smooth_anneal(hi, 0.03, 0.003, 5000, 0.01)
        ht, c5t2 = pgd_hard(ht, 15000, 0.003)
        if c5t2 < best_c5:
            best_h, best_c5 = ht, c5t2

    # Phase 3: Diversified restarts at best resolution
    n_curr = len(best_h)
    n_restarts = 50
    for r in range(n_restarts):
        if tl() < 80:
            break
        mode = r % 5
        if mode == 0:
            hr = best_h + rng.normal(0, 0.15, n_curr)
        elif mode == 1:
            hr = best_h.copy()
            st = rng.randint(0, n_curr // 2)
            ln = rng.randint(4, max(5, n_curr // 3))
            hr[st:st + ln] = 1.0 - hr[st:st + ln]
        elif mode == 2:
            hr = 0.6 * best_h + 0.4 * rng.uniform(0, 1, n_curr)
        elif mode == 3:
            hr = np.roll(best_h, rng.randint(1, max(2, n_curr // 5)))
        else:
            hr = best_h.copy()
            hr[rng.choice(n_curr, size=max(3, n_curr // 8), replace=False)] = 1.0 - hr[
                rng.choice(n_curr, size=max(3, n_curr // 8), replace=False)]
        hr = project_feasible(hr, n_curr)
        ht, c5t = pgd_smooth_anneal(hr, 0.03, 0.005, 2000, 0.015)
        if c5t < best_c5 + 0.001:
            ht, c5t = pgd_hard(ht, 10000, 0.004)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Try L-BFGS-B at reduced resolution for potential escape
    if tl() > 120:
        n_lb = min(120, n_curr)
        hi = project_feasible(interpolate_h(best_h, n_lb), n_lb)

        def obj_lnb(h, nv=n_lb):
            return soft_max_grad_fast(h, 0.01)[0] + 100.0 * (np.sum(h) - nv / 2.0) ** 2

        def grad_lnb(h, nv=n_lb):
            _, g = soft_max_grad_fast(h, 0.01)
            return g + 200.0 * (np.sum(h) - nv / 2.0)

        try:
            res = minimize(obj_lnb, hi, jac=grad_lnb, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n_lb,
                           options={'maxiter': 3000, 'ftol': 1e-15, 'gtol': 1e-10})
            ht = project_feasible(res.x, n_lb)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                ht, c5t = pgd_hard(ht, 20000, 0.003)
                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t
        except Exception:
            pass

    # Phase 5: Final intensive polish at multiple resolutions
    if tl() > 60:
        for n_fin in [len(best_h), max(48, len(best_h) // 2), min(320, len(best_h) * 2)]:
            if tl() < 40:
                break
            hi = project_feasible(interpolate_h(best_h, n_fin), n_fin)
            ht, c5t = pgd_smooth_anneal(hi, 0.01, 0.002, 4000, 0.006)
            ht, c5t2 = pgd_hard(ht, 40000, 0.002)
            if c5t2 < best_c5:
                best_h, best_c5 = ht, c5t2

    # Final: one more hard PGD at best resolution
    if tl() > 20:
        ht, c5t = pgd_hard(best_h, 60000, 0.001)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
