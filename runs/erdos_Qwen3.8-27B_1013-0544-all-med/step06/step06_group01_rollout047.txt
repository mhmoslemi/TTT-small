    m = np.max(C)
    ex = np.exp(alpha * (C - m))
    se = np.sum(ex)
    f = (m + np.log(se) / alpha) + beta * (np.sum(h) - 0.5 * n) ** 2
    w = ex / se
    lags = np.arange(-(n - 1), n)
    grad = np.zeros(n)
    for i in range(n):
        idx1 = i + lags
        valid1 = (idx1 >= 0) & (idx1 < n)
        t1 = np.where(valid1, -h[np.clip(idx1, 0, n - 1)], 0.0)
        idx2 = i - lags
        valid2 = (idx2 >= 0) & (idx2 < n)
        t2 = np.where(valid2, 1.0 - h[np.clip(idx2, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, t1 + t2)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-10:
            if d2 > 0:
                idx = np.argsort(1.0 - h)
                for j in idx:
                    add = min(d2, 1.0 - h[j])
                    h[j] += add
                    d2 -= add
                    if d2 <= 1e-10:
                        break
            else:
                idx = np.argsort(h)
                for j in idx:
                    sub = min(-d2, h[j])
                    h[j] -= sub
                    d2 += sub
                    if abs(d2) <= 1e-10:
                        break
    return h

def _optimize(h, n, alpha, beta, maxiter=500):
    res = minimize(_obj_grad, h.copy(), args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 25, 'ftol': 1e-15})
    if res.x is None:
        return h
    return _fix_sum(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    hb = _fix_sum(initial_h_values.astype(float).copy(), n0)
    cb = _c5_max(hb)
    nb = n0

    # Phase 1: Progressive alpha at original n
    for alpha in [20, 50, 100, 300, 1000, 3000, 8000]:
        if _el() > budget_s * 0.22:
            break
        beta = max(1.0, 0.04 * alpha)
        hn = _optimize(hb, nb, alpha, beta, maxiter=2000)
        cn = _c5_max(hn)
        if cn < cb - 1e-9:
            cb, hb = cn, hn

    # Phase 2: Explore different n values
    n_candidates = list(range(55, 230, 2))
    n_candidates = [n for n in n_candidates if n != nb]
    n_candidates.sort(key=lambda x: abs(x - nb))

    for n in n_candidates:
        if _el() > budget_s * 0.52:
            break
        x_old = np.linspace(0, 2, nb, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        ht = _fix_sum(np.interp(x_new, x_old, hb), n)
        ht = _optimize(ht, n, 200, 5.0, maxiter=400)
        cn = _c5_max(ht)
        if cn < cb - 1e-8:
            for alpha in [500, 2000, 5000]:
                if _el() > budget_s * 0.55:
                    break
                beta = max(1.0, 0.04 * alpha)
                ht = _optimize(ht, n, alpha, beta, maxiter=600)
                cn = _c5_max(ht)
                if cn < cb - 1e-9:
                    cb, hb, nb = cn, ht, n

    # Phase 3: Diverse random restarts
    for restart in range(60):
        if _el() > budget_s * 0.80:
            break
        r = rng.rand()
        if r < 0.30:
            ht = _fix_sum(hb + rng.randn(nb) * (0.01 + 0.04 * rng.rand()), nb)
        elif r < 0.60:
            ht = np.zeros(nb)
            ht[np.random.choice(nb, nb // 2, replace=False)] = 1.0
            ht = _fix_sum(ht, nb)
        elif r < 0.80:
            ht = np.zeros(nb)
            for _ in range(nb // 4):
                blk = rng.randint(2, max(3, nb // 6))
                st = rng.randint(0, max(1, nb - blk))
                ht[st:st + blk] = 1.0
            ht = _fix_sum(ht, nb)
        else:
            ht = np.clip(0.5 + 0.3 * rng.randn(nb), 0, 1)
            ht = _fix_sum(ht, nb)
        for alpha in [300, 1500]:
            if _el() > budget_s * 0.82:
                break
            beta = max(1.0, 0.04 * alpha)
            ht = _optimize(ht, nb, alpha, beta, maxiter=500)
            cn = _c5_max(ht)
            if cn < cb - 1e-9:
                cb, hb = cn, ht

    # Phase 4: Very high alpha polish
    if _el() < budget_s * 0.92:
        for alpha in [5000, 15000, 30000]:
            if _el() > budget_s * 0.94:
                break
            beta = max(10.0, 0.04 * alpha)
            hn = _optimize(hb, nb, alpha, beta, maxiter=3000)
            cn = _c5_max(hn)
            if cn < cb:
                cb, hb = cn, hn

    # Phase 5: Simulated annealing local search
    if _el() < budget_s * 0.99:
        hb = hb.copy()
        t_sa = 0.004
        for step in range(30000):
            if _el() > budget_s * 0.99:
                break
            if step % 800 == 0 and step > 0:
                t_sa *= 0.65
            i = rng.randint(nb)
            delta = rng.uniform(-0.02, 0.02)
            h_try = hb.copy()
            h_try[i] = np.clip(hb[i] + delta, 0.0, 1.0)
            h_try = _fix_sum(h_try, nb)
            c5_new = _c5_max(h_try)
            if c5_new < cb:
                hb = h_try
                cb = c5_new
            elif rng.rand() < np.exp(-(c5_new - cb) / max(t_sa, 1e-12)):
                hb = h_try

    # Final cleanup
    hb = _fix_sum(hb, nb)
    cb = _c5_max(hb)
    return (hb, cb, nb)
