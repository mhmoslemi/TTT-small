                if 0 <= idx < n:

        try:
            prob.solve(solver=solver)
        except:
            try:
                prob.solve(solver='ECOS')
            except:

            break
        h_new = np.clip(h_var.value, 0, 1)
        if np.max(np.abs(h_new - h)) < 1e-8:
            h = h_new
            break
        h = h_new

            best_c5 = actual_c5
        if not improved:
            break
    return h, _c5_max(h)

import numpy as np
from scipy.optimize import linprog
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    for _ in range(30):
        d = target - np.sum(h)
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += d / cnt
        else:
            mask = h > 1e-14
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _delta_for_i(h, n, i):
    m = np.arange(2 * n - 1)
    idx_a = i + m - n + 1
    idx_b = i - m + n - 1
    va = (idx_a >= 0) & (idx_a < n)
    vb = (idx_b >= 0) & (idx_b < n)
    delta = np.zeros(2 * n - 1)
    if np.any(va):
        delta[va] = 1.0 - h[np.clip(idx_a, 0, n - 1)[va]]
    if np.any(vb):
        delta[vb] -= h[np.clip(idx_b, 0, n - 1)[vb]]
    return delta

def _coord_descent(h, n, max_sweeps=200, tol=1e-10, b_iters=25):
    dx = 2.0 / n
    h = h.copy()
    C = _c5_array(h).copy()
    best_c5 = C.max()
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            old = h[i]
            delta = _delta_for_i(h, n, i)
            lo, hi = 0.0, 1.0
            for _ in range(b_iters):
                mid = (lo + hi) * 0.5
                C_try = C + (mid - old) * dx * delta
                m_star = C_try.argmax()
                if delta[m_star] < 0:
                    lo = mid
                elif delta[m_star] > 0:
                    hi = mid
                else:
                    break
            best_v = (lo + hi) * 0.5
            C_new = C + (best_v - old) * dx * delta
            new_c5 = C_new.max()
            if new_c5 < best_c5 - tol:
                h[i] = best_v
                C = C_new
                best_c5 = new_c5
                improved = True
        h = _fix(h, n)
        C = _c5_array(h)
        actual = C.max()
        best_c5 = min(best_c5, actual)
        if not improved:
            break
    return h, _c5_max(h)

def _alt_lp(h, n, max_iter=80, relax=1.0):
    dx = 2.0 / n
    h = h.copy()
    nc = 2 * n - 1
    indices = np.zeros((nc, n), dtype=np.int32)
    valid = np.zeros((nc, n), dtype=bool)
    for k in range(nc):
        for j in range(n):
            idx = j + k - (n - 1)
            if 0 <= idx < n:
                indices[k, j] = idx
                valid[k, j] = True
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.5 * n])
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    best_h = h.copy()
    best_c5 = _c5_max(h)
    for it in range(max_iter):
        g = 1.0 - h
        A_ub = np.zeros((nc, n + 1))
        A_ub[valid] = g[indices[valid]] * dx
        A_ub[:, -1] = -1.0
        b_ub = np.zeros(nc)
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status != 0 or res.x is None:
            break
        h_new = np.clip(res.x[:n], 0, 1)
        h_cand = relax * h_new + (1.0 - relax) * h
        h_cand = _fix(h_cand, n)
        c5_cand = _c5_max(h_cand)
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
        diff = np.max(np.abs(h_new - h))
        h = h_cand
        if diff < 1e-7:
            break
    return best_h, best_c5

    ht, cn = _coord_descent(ht, n_new, max_sweeps=max_cd_sweeps)
    return ht, cn
