import numpy as np
import time

def project_feasible(v, N):
    """Project v onto {h : 0<=h<=1, sum(h)=N/2}"""
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    # Fix any residual
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        nmask = mask.sum()
        if nmask > 0:
            h[mask] += diff / nmask
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    """Compute full correlation"""
    return np.correlate(h, 1.0 - h, mode='full')

def obj(h):
    """Compute C5 value"""
    N = len(h)
    dx = 2.0 / N
    return compute_corr(h).max() * dx

def grad_single_lag(h, lag):
    """Gradient of f_lag(h) = sum_i h[i](1-h[i+lag]) w.r.t. h"""
    N = len(h)
    g = np.zeros(N)
    # For positive lag k: f = sum_{i=0}^{N-1-k} h[i](1-h[i+k])
    # df/dh[i] for i < N-k: (1-h[i+k]) + sum of contributions from other terms
    # Actually: f = sum_i h[i](1-h[i+k])
    # df/dh[i] = (1-h[i+k]) [from term i] + (-h[i-k]) [from term i-k, if i-k>=0 and i-k+k=i]
    # Wait: term j contributes h[j](1-h[j+k]). This involves h[j] and h[j+k].
    # So df/dh[i] gets contributions from:
    #   - term j=i (if i < N-k): derivative of h[i](1-h[i+k]) w.r.t. h[i] = (1-h[i+k])
    #   - term j=i-k (if i-k >= 0 and i-k < N-k, i.e., i < N): derivative of h[i-k](1-h[i]) w.r.t. h[i] = -h[i-k]
    
    if lag >= 0:
        # lag = k >= 0
        k = lag
        if k == 0:
            # f = sum h[i](1-h[i]) = sum h[i] - sum h[i]^2
            # df/dh[i] = 1 - 2h[i]
            g = 1.0 - 2.0 * h
        else:
            # For i in [0, N-k): contribution from term i: (1-h[i+k])
            # For i in [k, N): contribution from term i-k: -h[i-k]
            g[:N-k] += 1.0 - h[k:]
            g[k:] -= h[:N-k]
    else:
        # lag = -k where k > 0
        # f = sum_{i=0}^{N-1+k} h[i-k](1-h[i]) ... 
        # Actually for negative lag -k: f = sum_i h[i](1-h[i-k]) where i-k >= 0, so i >= k
        # f = sum_{i=k}^{N-1} h[i](1-h[i-k])
        k = -lag
        # For i in [k, N): derivative of h[i](1-h[i-k]) w.r.t. h[i] = (1-h[i-k])
        # For i in [0, N-k): derivative of h[i+k](1-h[i]) w.r.t. h[i] = -h[i+k]
        g[k:] += 1.0 - h[:N-k]
        g[:N-k] -= h[k:]
    return g

def soft_gradient(h, corr, dx, alpha=50.0, top_m=5):
    """Compute soft gradient using top-m lags"""
    N = len(h)
    # Find top m lags
    top_indices = np.argsort(corr)[-top_m:][::-1]  # descending
    # Weight by softmax over top m
    c_top = corr[top_indices]
    w = np.exp(alpha * (c_top - c_top[0]))
    w /= w.sum()
    
    g = np.zeros(N)
    for idx, wi in zip(top_indices, w):
        lag = int(idx - (N - 1))
        g += wi * grad_single_lag(h, lag)
    return g * dx

def nesterov_optimize(h, N, t_end, try_update, alpha=80.0, top_m=3, lr=0.02, momentum=0.9):
    """Nesterov accelerated gradient with soft max"""
    val = obj(h)
    try_update(h, val)
    
    v = np.zeros(N)  # momentum
    best_h = h.copy()
    best_val = val
    x_old = h.copy()
    
    it = 0
    while time.time() < t_end:
        # Nesterov lookahead
        y = h + momentum * v
        y = np.clip(y, 0.0, 1.0)  # tentative clip
        
        # Compute gradient at y (but we need corr at y)
        # Actually, to save time, compute gradient at h and use it
        corr = compute_corr(h)
        g = soft_gradient(h, corr, dx=2.0/N, alpha=alpha, top_m=top_m)
        
        # Update
        v = momentum * v - lr * g
        h_new = project_feasible(h + v, N)
        
        val_new = obj(h_new)
        if val_new < val:
            h = h_new
            val = val_new
            if val < best_val:
                best_val = val
                best_h = h.copy()
                try_update(h, val)
            x_old = h - v  # for Nesterov
        else:
            # Retry with smaller step
            lr *= 0.7
            if lr < 1e-5:
                # Stuck, add noise
                h = project_feasible(h + np.random.randn(N) * 0.03, N)
                val = obj(h)
                v = np.zeros(N)
                lr = 0.02
                it += 100  # skip some iterations
                continue
        
        it += 1
        # Decay momentum slightly
        momentum = min(momentum * 1.001, 0.95)
    
    return best_h, best_val

        g[:N-k] += 1.0 - h[k:]
        g[k:] -= h[:N-k]
    else:
        m = -k
        g[m:] += 1.0 - h[:N-m]
        g[:N-m] -= h[m:]
    return g

def soft_grad(h, corr, dx, alpha, m_top):
    N = len(h)
    idx = np.argsort(corr)[-m_top:]
    c_sub = corr[idx]
    w = np.exp(alpha * (c_sub - c_sub.max()))
    w /= w.sum()
    g = np.zeros(N)
    for ix, wi in zip(idx, w):
        lag = int(ix) - (N - 1)
        g += wi * grad_one_lag(h, lag)
    return g * dx

def try_update(best, h, val):
    if val < best[1]:
        best[1] = val
        best[0] = h.copy()

def grad_desc(h, N, t_end, best, alpha=80, m_top=5, lr_init=0.04, rng=None):
    dx = 2.0 / N
    val = get_obj(h)
    try_update(best, h, val)
    lr = lr_init
    stalled = 0
    while time.time() < t_end:
        corr = get_corr(h)
        g = soft_grad(h, corr, dx, alpha, m_top)
        h_new = h
        v_new = val
        try_lr = lr
        for _ in range(8):
            cand = project_feasible(h - try_lr * g, N)
            v = get_obj(cand)
            if v < v_new - 1e-14:
                h_new = cand
                v_new = v
            try_lr *= 0.5
            if try_lr < 1e-7:
                break
        if v_new < val - 1e-14:
            h = h_new
            val = v_new
            try_update(best, h, val)
            lr = min(lr * 1.04, 0.15)
            stalled = 0
        else:
            stalled += 1
            lr *= 0.75
            if stalled > 35:
                scale = 0.01 + 0.01 * min(stalled / 50.0, 1.0)
                h = project_feasible(h + rng.randn(N) * scale, N)
                val = get_obj(h)
                try_update(best, h, val)
                lr = lr_init
                stalled = 0
    return h, val

def pair_optimize(h, i, j, n_eval):
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return
    best_ti = h[i]
    best_tj = h[j]
    best_v = get_obj(h)
    pts = np.linspace(lo, hi, n_eval)
    for t in pts:
        h[i] = t
        h[j] = s - t
        v = get_obj(h)
        if v < best_v:
            best_v = v
            best_ti = t
            best_tj = s - t
    h[i] = best_ti
    h[j] = best_tj

def coord_sweep(h, N, t_end, best, n_eval=8, rng=None):
    val = get_obj(h)
    try_update(best, h, val)
    sweep_no_improve = 0
    while time.time() < t_end:
        improved = False
        order = list(range(N))
        pairs = []
        for i in range(N):
            for j in range(i + 1, N):
                pairs.append((i, j))
        rng.shuffle(pairs)
        for i, j in pairs:
            if time.time() > t_end:
                break
            s = h[i] + h[j]
            if s < 1e-12 or s > 2.0 - 1e-12:
                continue
            pair_optimize(h, i, j, n_eval)
            v = get_obj(h)
            if v < val - 1e-14:
                val = v
                improved = True
                try_update(best, h, val)
        if improved:
            sweep_no_improve = 0
        else:
            sweep_no_improve += 1
            if sweep_no_improve >= 2:
                break
    return h, val

def fine_tune(h, N, t_end, best, rng=None):
    dx = 2.0 / N
    lr = 0.008
    stalled = 0
    val = get_obj(h)
    try_update(best, h, val)
    while time.time() < t_end:
        corr = get_corr(h)
        g = soft_grad(h, corr, dx, 120, 3)
        h_new = project_feasible(h - lr * g, N)
        v_new = get_obj(h_new)
        if v_new < val - 1e-15:
            h = h_new
            val = v_new
            try_update(best, h, val)
            lr = min(lr * 1.02, 0.02)
            stalled = 0
        else:
            stalled += 1
            lr *= 0.9
            if stalled > 80:
                h = project_feasible(h + rng.randn(N) * 0.005, N)
                val = get_obj(h)
                try_update(best, h, val)
                lr = 0.008
                stalled = 0
    return h, val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    best = [None, float('inf')]

    if h_init is not None:
        try_update(best, h_init, get_obj(h_init))
    else:
        h0 = project_feasible(0.5 + 0.3 * rng.randn(120), 120)
        try_update(best, h0, get_obj(h0))

    N = len(best[0])

    # Phase 1: Gradient descent (35% of time)
    t1 = t0 + 0.35 * budget_s
    h_work = best[0].copy()
    grad_desc(h_work, N, min(t1, t_end), best, alpha=80, m_top=5, rng=rng)

    # Phase 2: Coordinate descent (50% of time)
    t2 = t0 + 0.85 * budget_s
    h_work = best[0].copy()
    coord_sweep(h_work, N, min(t2, t_end), best, n_eval=8, rng=rng)

    # Phase 3: Fine-tune (remaining time)
    if time.time() < t_end:
        h_work = best[0].copy()
        fine_tune(h_work, N, t_end, best, rng=rng)

    # Phase 4: Try a second round of coord descent if time allows
    if time.time() < t_end - 30:
        h_work = best[0].copy()
        coord_sweep(h_work, N, min(time.time() + 60, t_end - 5), best, n_eval=12, rng=rng)

    return best[0], best[1], len(best[0])
