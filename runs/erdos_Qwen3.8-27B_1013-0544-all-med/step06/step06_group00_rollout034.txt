But wait, I can make this faster. Instead of computing the full correlation for each trial swap, I can compute the change in correlation analytically.

When swapping h[i]: 1→0 and h[j]: 0→1:
- The change in C[m] (correlation at index m) is:
  - Effect of h[i] going 1→0:
    - As h[x+k]: position i appears as h[i] where i = x+k, so x = i-k. Contribution was (1-h[i-k]), now 0. Delta: -(1-h[i-k])
    - In (1-h[x]): position i appears as h[i] where x = i. Term was h[i+k]*(1-1) = 0, now h[i+k]*(1-0) = h[i+k]. Delta: +h[i+k]
  - Effect of h[j] going 0→1:
    - As h[x+k]: position j appears as h[j] where j = x+k, so x = j-k. Contribution was 0*(1-h[j-k]) = 0, now 1*(1-h[j-k]) = 1-h[j-k]. Delta: +(1-h[j-k])
    - In (1-h[x]): position j appears as h[j] where x = j. Term was h[j+k]*(1-0) = h[j+k], now h[j+k]*(1-1) = 0. Delta: -h[j+k]

Total delta_C[m] = -(1-h[i-k_m]) + h[i+k_m] + (1-h[j-k_m]) - h[j+k_m]
where k_m = m - (n-1), and we only include terms where indices are valid.

This is O(1) per m, so O(n) per swap evaluation. For 2000 trials, that's 2000*n = 2000*767 ≈ 1.5M operations. In numpy, this should be very fast (maybe 1-5ms per iteration).

    corr = compute_corr(h)
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    t0 = time.time()
    lr = 0.5
    j_arr = np.arange(n)

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        m_star = np.argmax(corr)
        k_star = m_star - (n - 1)
        grad = np.zeros(n)
        v1 = (j_arr - k_star >= 0) & (j_arr - k_star < n)
        grad[v1] = 1.0 - h[j_arr[v1] - k_star]
        v2 = (j_arr + k_star >= 0) & (j_arr + k_star < n)
        grad[v2] -= h[j_arr[v2] + k_star]
        h_new = project_continuous(h - lr * grad, ts)
        corr_new = compute_corr(h_new)
        c5_new = np.max(corr_new) * dx
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new; best_h = h_new.copy()
            h = h_new; corr = corr_new
            lr = min(lr * 1.03, 3.0)
        else:
            lr *= 0.93
            if lr < 0.02:
                lr = 0.3

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()

    def remaining():
        return budget_s - (time.time() - t_start)

    h0 = initial_h_values.astype(float).copy()
    n0 = len(h0)
    h0 = project_continuous(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = np.max(compute_corr(h0)) * (2.0 / n0)

    # Phase 1: Tabu search on binarized initial
    if remaining() > 120:
        h_bin = binarize(h0)
        h_tab, c5_tab = tabu_search(h_bin, time_limit=min(250, remaining() - 40),
                                     tabu_size=40, n_samples=5000)
        if c5_tab < best_c5:
            best_c5 = c5_tab; best_h = h_tab.copy()

    # Phase 2: Subgradient polish
    if remaining() > 100:
        h_sg, c5_sg = subgradient_descent(best_h, max_iter=30000,
                                           time_limit=min(180, remaining() - 30))
        if c5_sg < best_c5:
            best_c5 = c5_sg; best_h = h_sg.copy()

    # Phase 3: Tabu on re-binarized
    if remaining() > 100:
        h_bin = binarize(best_h)
        h_tab, c5_tab = tabu_search(h_bin, time_limit=min(250, remaining() - 40),
                                     tabu_size=35, n_samples=6000)
        if c5_tab < best_c5:
            best_c5 = c5_tab; best_h = h_tab.copy()

    # Phase 4: Subgradient again
    if remaining() > 80:
        h_sg, c5_sg = subgradient_descent(best_h, max_iter=30000,
                                           time_limit=min(150, remaining() - 25))
        if c5_sg < best_c5:
            best_c5 = c5_sg; best_h = h_sg.copy()

    # Phase 5: Multi-restart tabu
    for s in range(4):
        if remaining() < 80:
            break
        h_pert = best_h + np.random.randn(n0) * 0.25
        h_bin = binarize(h_pert)
        h_tab, c5_tab = tabu_search(h_bin, time_limit=min(120, remaining() - 30),
                                     tabu_size=30, n_samples=4000)
        if c5_tab < best_c5:
            best_c5 = c5_tab; best_h = h_tab.copy()

    # Phase 6: Final subgradient
    if remaining() > 30:
        h_sg, c5_sg = subgradient_descent(best_h, max_iter=30000,
                                           time_limit=min(60, remaining() - 5))
        if c5_sg < best_c5:
            best_c5 = c5_sg; best_h = h_sg.copy()

    best_h = project_continuous(best_h, len(best_h) / 2.0)
    best_c5 = np.max(compute_corr(best_h)) * (2.0 / len(best_h))
    return best_h, best_c5, len(best_h)
