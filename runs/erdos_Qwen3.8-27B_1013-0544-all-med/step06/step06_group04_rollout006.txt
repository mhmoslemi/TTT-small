import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    result = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(result) * dx)

def soft_c5(h, t=0.005):
    n = len(h)
    dx = 2.0 / n
    result = np.correlate(h, 1.0 - h, mode="full") * dx
    rmax = np.max(result)
    return rmax + t * np.log(np.sum(np.exp((result - rmax) / t)))

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def nelder_mead_step(h, n, max_iter=200):
    """Run Nelder-Mead with soft objective and projection"""
    def objective(x):
        x = make_feasible(x, n)
        return soft_c5(x, t=0.005)
    
    res = minimize(objective, h, method='Nelder-Mead',
                   options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-10})
    best = make_feasible(res.x, n)
    return best

def simulated_annealing(h, n, n_iter=5000, rng=None, start_time=None, budget_s=1100):
    """Simulated annealing with sum-preserving moves"""
    if rng is None:
        rng = np.random.default_rng()
    
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    T = 0.05
    T_min = 0.001
    cooling = (T_min / T) ** (1.0 / max(n_iter, 1))
    
    for i in range(n_iter):
        if start_time is not None and time.time() - start_time > budget_s * 0.85:

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        
        T *= cooling

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    result = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(result) * dx)

def soft_c5(h, t):
    n = len(h)
    dx = 2.0 / n
    result = np.correlate(h, 1.0 - h, mode="full") * dx
    rmax = np.max(result)
    return float(rmax + t * np.log(np.sum(np.exp((result - rmax) / t))))

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def nm_optimize(h0, n, max_iter, deadline):
    t_soft = 0.004
    best = h0.copy()
    best_val = compute_c5(h0)
    try:
        res = minimize(
            lambda x: soft_c5(make_feasible(x, n), t_soft),
            h0, method='Nelder-Mead',
            options={'maxiter': max_iter, 'xatol': 1e-9, 'fatol': 1e-11,
                     'adaptive': True, 'initial_simplex_size': 0.05}
        )
        if time.time() < deadline:
            cand = make_feasible(res.x, n)
            val = compute_c5(cand)
            if val < best_val:
                best, best_val = cand, val
    except Exception:
        pass
    return best, best_val

def sa_optimize(h0, n, max_iters, rng, deadline):
    h = h0.copy()
    best_h = h.copy()
    curr_val = compute_c5(h)
    best_val = curr_val
    T = 0.015
    T_min = 0.0002
    cool = (T_min / T) ** (1.0 / max(max_iters, 1))
    for _ in range(max_iters):
        if time.time() > deadline:
            break
        mt = rng.integers(0, 4)
        if mt == 0:
            i = rng.integers(0, n)
            h[i] = np.clip(h[i] + rng.normal(0, 0.1), 0, 1)
        elif mt == 1:
            i, j = rng.integers(0, n, 2)
            if i != j:
                amt = rng.normal(0, 0.06)
                h[i] = np.clip(h[i] + amt, 0, 1)
                h[j] = np.clip(h[j] - amt, 0, 1)
        elif mt == 2:
            bs = int(rng.integers(3, min(12, n)))
            st = int(rng.integers(0, n - bs + 1))
            h[st:st+bs] = np.clip(h[st:st+bs] + rng.normal(0, 0.04, bs), 0, 1)
        else:
            i = rng.integers(0, n)
            h[i] = rng.uniform(0, 1)
        h = make_feasible(h, n)
        new_val = compute_c5(h)
        d = new_val - curr_val
        if d < 0 or (d > 0 and rng.random() < np.exp(-d / max(T, 1e-12))):
            curr_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_h = h.copy()
        T *= cool
    return best_h, best_val

def coordinate_descent(h, n, iterations, deadline):
    h = h.copy()
    best_val = compute_c5(h)
    for _ in range(iterations):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            orig = h[i]
            candidates = [0.0, 1.0, orig,
                          max(0, orig - 0.1), min(1, orig + 0.1),
                          max(0, orig - 0.3), min(1, orig + 0.3)]
            best_cand = orig
            best_v = compute_c5(h)
            for cv in candidates:
                if abs(cv - orig) < 1e-12:
                    continue
                h[i] = cv
                h = make_feasible(h, n)
                v = compute_c5(h)
                if v < best_v:
                    best_v = v
                    best_cand = cv
            h[i] = best_cand
            h = make_feasible(h, n)
            if best_cand != orig:
                improved = True
            if best_v < best_val:
                best_val = best_v
        if not improved:
            break
    return h, best_val
