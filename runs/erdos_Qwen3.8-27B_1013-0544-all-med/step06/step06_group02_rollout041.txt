import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def grad_lag(h, lag, N):
    g = np.zeros(N)
    s1 = max(0, lag)
    e1 = min(N, N + lag)
    if s1 < e1:
        g[s1:e1] += 1.0 - h[s1 - lag:e1 - lag]
    s2 = max(0, -lag)
    e2 = min(N, N - lag)
    if s2 < e2:
        g[s2:e2] -= h[s2 + lag:e2 + lag]
    return g

def soft_grad(h, corr, dx, alpha):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        return grad_lag(h, k_star - (N - 1), N) * dx
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-6:
            continue
        g += w[j] * grad_lag(h, j - (N - 1), N)
    return g * dx

def binary_local_search(h, rng, t_limit, max_iter=100000):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    best_h = h.copy()
    temp = 0.05
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    stagnant = 0
    for it in range(max_iter):
        if time.time() > t_limit:
            break
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        improved = False
        n_try = 30
        for _ in range(n_try):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            if a == b:
                continue
            h[a], h[b] = h[b], h[a]
            corr_new = np.correlate(h, 1.0 - h, mode='full')
            val_new = corr_new.max() * dx
            if val_new < best_val - 1e-15 or (val_new <= best_val + 1e-15 and rng.rand() < np.exp((best_val - val_new) / max(temp, 1e-8))):
                best_val = val_new
                best_h = h.copy()
                corr = corr_new
                improved = True
                stagnant = 0
                break
            else:
                h[a], h[b] = h[b], h[a]
        if not improved:
            stagnant += 1
            if stagnant > 20:
                temp *= 0.8
                if temp < 0.005:
                    break
        else:
            temp = min(temp * 1.01, 0.05)
        ones = np.where(h > 0.5)[0]

    val, corr, dx = compute_c5(h)
