def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)

        idx = np.argsort(np.abs(h - 0.5))[::-1]  # prefer coordinates far from boundaries

            adj = min(room_up, abs(d)) if d > 0 else max(-room_up, d)
            h[i] += d  # this might not work perfectly

            break

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float)
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    s = np.exp(temp * (corr - m))
    return (m + np.log(s.sum()) / temp) * dx

def slsqp_step(h_start, n, temp, maxiter=300):
    bnds = [(0.0, 1.0)] * n
    ts = n / 2.0
    ones = np.ones(n)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - ts, 'jac': lambda x: ones}]
    res = minimize(smooth_obj, h_start, args=(temp,), method='SLSQP',
                   bounds=bnds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, ts)

def subgradient_step(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    top_k = np.argsort(corr)[-3:]
    g = np.zeros(n)
    for k in top_k:
        L = int(k) - (n - 1)
        for i in range(n):
            val = 1.0
            if 0 <= i - L < n:
                val -= h[i - L]
            if 0 <= i + L < n:
                val -= h[i + L]
            g[i] += val
    g /= len(top_k)
    return g, corr

            best_h = h_new.copy()
        h = h_new

def coord_descent(h, n_levels=13, max_sweeps=25, focus_top=7):

    ts = n / 2.0
    c5 = compute_c5(h)
    for sweep in range(max_sweeps):
        improved = 0
        corr = np.correlate(h, 1.0 - h, mode="full")
        top_k = np.argsort(corr)[-focus_top:]
        importance = np.zeros(n)
        for k in top_k:
            L = int(k) - (n - 1)
            for i in range(n):
                v = 1.0
                if 0 <= i - L < n:
                    v -= h[i - L]
                if 0 <= i + L < n:
                    v -= h[i + L]
                importance[i] += abs(v)
        order = np.argsort(-importance)
        for j in order:
            orig = h[j]
            best_c = c5
            best_v = orig
            for v in np.linspace(0.0, 1.0, n_levels):

                h = project_feasible(h, ts)
                c5 = best_c
                improved += 1
