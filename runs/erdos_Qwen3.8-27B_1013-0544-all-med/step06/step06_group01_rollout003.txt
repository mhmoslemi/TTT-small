import numpy as np
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.argmax(1.0 - h)
                h[idx] = min(1.0, h[idx] + d2)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + d2)
    return h

def _coord_descent(h, n, n_passes=50, n_trials=15, t0=None, budget_frac=None, budget_s=None):
    best = _c5_max(h)
    h = h.copy()
    for p in range(n_passes):
        if t0 is not None and time.time() - t0 > budget_s * budget_frac:
            break
        improved = False
        for i in range(n):
            old_val = h[i]
            best_local = best
            best_trial = old_val
            if abs(old_val) < 0.01:
                trials = np.array([0.0, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0])
            elif old_val > 0.99:
                trials = np.array([0.0, 0.3, 0.5, 0.7, 0.8, 0.9, 0.95, 1.0])
            else:
                trials = np.linspace(0, 1, n_trials)
            for trial in trials:
                h_try = h.copy()
                h_try[i] = trial
                h_try = _fix_sum(h_try, n)
                c5_try = _c5_max(h_try)
                if c5_try < best_local - 1e-14:
                    best_local = c5_try
                    best_trial = trial
            if best_trial != old_val:
                h[i] = best_trial
                h = _fix_sum(h, n)
                if best_local < best - 1e-14:
                    best = best_local
                    improved = True
        if not improved:
            break
    return h, best

def _sa_search(h, n, n_iter, t0, budget_s, frac, seed):

    best_h = h.copy()
    best_c5 = _c5_max(h)
    cur_h = h.copy()
    cur_c5 = best_c5

    T_min = 1e-7

        elif move_type == 1:
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.uniform(0.02, 0.12) * rng.choice([-1, 1])
            h_try[i] = np.clip(cur_h[i] + delta, 0, 1)
            h_try[j] = np.clip(cur_h[j] - delta, 0, 1)
        elif move_type == 2:

            start = rng.randint(0, n - length)
            shift = rng.choice([-2, -1, 1, 2])
            end = start + length

            cur_c5 = c5_try
            if c5_try < best_c5:

def _binary_projection(h, n):
    """Project to nearest binary-like solution preserving sum."""
    h_bin = (h > 0.5).astype(float)
    target = 0.5 * n
    excess = np.sum(h_bin) - target
    if excess > 0:
        # Need to flip some 1s to 0s
        ones = np.where(h_bin == 1)[0]
        order = ones[np.argsort(h[ones])]
        for i in range(int(np.ceil(excess))):
            h_bin[order[i]] = 0.0
    elif excess < 0:
        zeros = np.where(h_bin == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        for i in range(int(np.ceil(-excess))):
            h_bin[order[i]] = 1.0
    # Handle fractional remainder
    diff = target - np.sum(h_bin)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h_bin)
            h_bin[idx] = min(1.0, h_bin[idx] + diff)
        else:
            idx = np.argmax(h_bin)
            h_bin[idx] = max(0.0, h_bin[idx] + diff)
    return h_bin

def _gradient_step(h, n, alpha, beta, maxiter=300):
    """Quick gradient polish using L-BFGS-B."""
    from scipy.optimize import minimize

    def obj(hv):
        c_arr = _c5_array(hv)
        m = np.max(c_arr)
        s = np.sum(np.exp(alpha * c_arr - alpha * m))
        return (m + np.log(s) / alpha) + beta * (np.sum(hv) - 0.5 * n) ** 2

    res = minimize(obj, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    if res.fun < 1e10:
        return _fix_sum(res.x, n)
    return h
