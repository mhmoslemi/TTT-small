def evaluate_c5(h, dx):
    """Compute C5 = max_k sum_i h[i]*(1-h[i+k]) * dx"""
    rev = 1 - h
    corr = np.correlate(h, rev, mode='full')
    return np.max(corr) * dx

def repair_sum(h, target_sum):
    """Adjust h to have the target sum while maintaining 0<=h<=1"""
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    n = len(h)
    # Distribute diff evenly, but respect bounds
    # Simple approach: add diff/n to all, then clip and redistribute
    h = h.copy()
    h += diff / n
    # If any are out of bounds, fix iteratively
    for _ in range(10):
        over = h > 1
        under = h < 0
        if not over.any() and not under.any():
            break
        if over.any():
            excess = np.sum(h[over] - 1)
            h[over] = 1
            free = ~over
            if free.any():
                h[free] -= excess / free.sum()
        if under.any():
            deficit = np.sum(-h[under])
            h[under] = 0
            free = ~under
            if free.any():
                h[free] += deficit / free.sum()
    return np.clip(h, 0, 1)

def upscale(h, new_n):
    """Upscale h from current resolution to new_n points"""
    old_n = len(h)
    old_dx = 2.0 / old_n
    new_dx = 2.0 / new_n
    # Interpolate
    old_x = np.arange(old_n) * old_dx
    new_x = (np.arange(new_n) + 0.5) * new_dx

    # Fix sum
    target_sum = new_n / 2
    h_new = repair_sum(h_new, target_sum)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_sum(h, target_sum):
    h = h.copy()
    for _ in range(20):
        diff = target_sum - np.sum(h)
        if abs(diff) < 1e-12:
            break
        n = len(h)
        h += diff / n
        h = np.clip(h, 0, 1)
    return h

def project(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    h = h.copy()
    # Bisection on lambda
    lo, hi = -10, 10
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(h + mid, 0, 1)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def projected_subgradient(h, max_steps=500, lr=0.05):
    """Projected subgradient method for minimizing C5"""
    n = len(h)
    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for step in range(max_steps):
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find active constraint (the lag that gives max)
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr) - (n - 1)  # Convert to signed lag
        # Actually np.argmax gives index in [0, 2n-2], lag = index - (n-1)
        
        # Compute subgradient of f_{k_star}
        g = np.zeros(n)
        # f_k = sum_i h[i]*(1-h[i+k]) * dx for valid i
        # ∂f_k/∂h[j] = (1-h[j+k])*dx - h[j-k]*dx (when valid)
        for j in range(n):
            term1 = 0.0
            term2 = 0.0
            if j + k_star >= 0 and j + k_star < n:
                term1 = (1 - h[j + k_star]) * dx
            if j - k_star >= 0 and j - k_star < n:
                term2 = h[j - k_star] * dx
            g[j] = term1 - term2
        
        # Projected gradient step
        lr_t = lr / (1 + step * 0.01)
        h_new = np.clip(h - lr_t * g, 0, 1)
        h_new = repair_sum(h_new, target_sum)
        
        # Check if it improved

        if new_c5 < c5:
            h = h_new

            h_new = repair_sum(h_new, target_sum)
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < c5:
                h = h_new
            else:
                break
    
    return best_h, best_c5

def project_polytope(v, target_sum):
    lo = -float(np.max(v)) - 1.0
    hi = 1.0 - float(np.min(v)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(v + mid, 0.0, 1.0)) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)

def nelder_mead(h, max_iters=200):
    n = len(h)
    ts = n / 2.0
    dx = 2.0 / n
    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        pen = (np.sum(x) - ts) ** 2 * 100.0
        return evaluate_c5(x, dx) + pen
    res = minimize(obj, h, method='Nelder-Mead',
                   options={'maxiter': max_iters, 'xatol': 1e-9, 'fatol': 1e-10})
    return np.clip(res.x, 0.0, 1.0)

def subgradient(h, max_steps=500, lr=0.03):
    n = len(h)
    ts = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    for step in range(max_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = int(np.argmax(corr))
        k = idx - (n - 1)
        g = np.zeros(n)
        for j in range(n):
            if 0 <= j + k < n:
                g[j] += (1.0 - h[j + k]) * dx
            if 0 <= j - k < n:
                g[j] -= h[j - k] * dx
        alpha = lr / (1.0 + 0.005 * step)
        h_new = project_polytope(h - alpha * g, ts)
        c5_new = evaluate_c5(h_new, dx)
        if c5_new <= best_c5 + 1e-12:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        elif step > 100:
            break
    return best_h, best_c5

def sa(h, max_iter=50000, t0=0.02, t1=0.0005):
    n = len(h)
    dx = 2.0 / n
    cur_c5 = evaluate_c5(h, dx)
    best_c5 = cur_c5
    best_h = h.copy()
    temp = t0
    cr = (t1 / t0) ** (1.0 / max_iter)
    for i in range(max_iter):
        if temp < t1:
            break
        i1 = np.random.randint(0, n)
        i2 = np.random.randint(0, n)
        if i1 == i2:
            continue
        d = np.random.uniform(-0.05, 0.05)
        nh = h.copy()
        nh[i1] += d
        nh[i2] -= d
        if nh[i1] < 0.0 or nh[i1] > 1.0 or nh[i2] < 0.0 or nh[i2] > 1.0:
            temp *= cr
            continue
        nc5 = evaluate_c5(nh, dx)
        if nc5 < cur_c5:
            h = nh
            cur_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_h = h.copy()
        else:
            r = (nc5 - cur_c5) / temp
            if r < 400.0 and np.random.random() < np.exp(-r):
                h = nh
                cur_c5 = nc5
        temp *= cr
    return best_h, best_c5

def upscale(h, new_n):
    on = len(h)
    ox = (np.arange(on) + 0.5) * (2.0 / on)
    nx = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(nx, ox, h)
    return repair_sum(h_new, new_n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    deadline = t_start + budget_s

    try:
        h0 = np.array(initial_h_values, dtype=float)
    except NameError:
        h0 = np.random.uniform(0.0, 1.0, 70)
        h0 = repair_sum(h0, 35.0)

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0, 2.0 / len(h0))
    n = len(h0)

    # Phase 1: Refine at initial resolution
    h_c = h0.copy()
    h_c = nelder_mead(h_c, 150)
    h_c = repair_sum(h_c, n / 2.0)
    c5 = evaluate_c5(h_c, 2.0 / n)
    if c5 < best_c5:
        best_c5, best_h = c5, h_c.copy()

    if time.time() < deadline - 250:
        h_c, c5 = subgradient(h_c, 500, 0.03)
        if c5 < best_c5:
            best_c5, best_h = c5, h_c.copy()

    if time.time() < deadline - 120:
        h_c, c5 = sa(h_c, 80000, 0.02, 0.001)
        if c5 < best_c5:
            best_c5, best_h = c5, h_c.copy()

    # Phase 2: Higher resolutions
    for new_n in [90, 105, 140, 200, 280, 350]:
        if time.time() > deadline - 250:
            break
        h_u = upscale(best_h, new_n)
        h_u = nelder_mead(h_u, 80)
        h_u = repair_sum(h_u, new_n / 2.0)
        c5 = evaluate_c5(h_u, 2.0 / new_n)
        if c5 < best_c5:
            best_c5, best_h = c5, h_u.copy()
        if time.time() < deadline - 180:
            h_u, c5 = subgradient(h_u, 300, 0.02)
            if c5 < best_c5:
                best_c5, best_h = c5, h_u.copy()
        if time.time() < deadline - 100:
            h_u, c5 = sa(h_u, 40000, 0.01, 0.0005)
            if c5 < best_c5:
                best_c5, best_h = c5, h_u.copy()

    # Phase 3: Alternative starts + random restarts
    # Try alternating pattern
    n_alt = n
    h_alt = np.zeros(n_alt)
    h_alt[::2] = 1.0
    h_alt = repair_sum(h_alt, n_alt / 2.0)
    h_alt = nelder_mead(h_alt, 80)
    h_alt = repair_sum(h_alt, n_alt / 2.0)
    c5 = evaluate_c5(h_alt, 2.0 / n_alt)
    if c5 < best_c5:
        best_c5, best_h = c5, h_alt.copy()
    if time.time() < deadline - 60:
        h_alt, c5 = sa(h_alt, 30000, 0.02, 0.001)
        if c5 < best_c5:
            best_c5, best_h = c5, h_alt.copy()

    # Random restarts from best
    while time.time() < deadline - 20:
        n_try = len(best_h)
        sigma = 0.03 + 0.02 * np.random.random()
        h_try = best_h + np.random.normal(0.0, sigma, n_try)
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = repair_sum(h_try, n_try / 2.0)
        h_try = nelder_mead(h_try, 40)
        h_try = repair_sum(h_try, n_try / 2.0)
        c5 = evaluate_c5(h_try, 2.0 / n_try)
        if c5 < best_c5:
            best_c5, best_h = c5, h_try.copy()
        if time.time() < deadline - 20:
            h_try, c5 = sa(h_try, 20000, 0.01, 0.0005)
            if c5 < best_c5:
                best_c5, best_h = c5, h_try.copy()

    # Final validation
    best_h = repair_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h, 2.0 / len(best_h))
    return (best_h, best_c5, len(best_h))
