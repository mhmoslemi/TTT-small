from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def soft_max_val(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    # Use log-sum-exp for numerical stability
    return t * np.log(np.sum(np.exp((conv - m) / t))) * dx + m * dx

        if w < 1e-12:
            continue
        s = k_idx - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += w * (1.0 - h[:n - s])
                grad[:n - s] -= w * h[s:]
        else:
            s2 = -s
            if s2 < n:
                grad[:n - s2] += w * (1.0 - h[s2:])
                grad[s2:] -= w * h[:n - s2]
    return grad * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def pgd_momentum(h_init, max_iter, lr_init, t_init, t_decay, t_min, mu=0.8):
    n = len(h_init)
    h = h_init.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    t = t_init
    stall = 0
    for it in range(max_iter):
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        grad = soft_max_grad(h, t, top_k=40)
        v = mu * v - lr * grad
        h_trial = project_feasible(h + v, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial > c5 + 1e-13:
            v = mu * 0.5 * v - lr * 0.3 * grad
            h_trial = project_feasible(h + v, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial > c5 + 1e-13:
                v = np.zeros(n)
                h_trial = project_feasible(h - lr * 0.05 * grad, n)
                c5_trial = compute_c5(h_trial)
        h = h_trial
        t = max(t * t_decay, t_min)
        if c5_trial < c5 - 1e-12:
            lr = min(lr * 1.02, 0.05)
        elif stall > 200:
            lr *= 0.9
            stall = 0
    return best_h, best_c5

def _obj_pen(h, t_val, nsz):
    return soft_max_val(h, t_val) + 500.0 * (np.sum(h) - nsz / 2.0) ** 2

def _grad_pen(h, t_val, nsz):
    return soft_max_grad(h, t_val, top_k=50) + 1000.0 * (np.sum(h) - nsz / 2.0)

def _cons_pen(h, t_val, nsz):
    penalty = soft_max_val(h, t_val) + 500.0 * (np.sum(h) - nsz / 2.0) ** 2
    return penalty

def _cons_grad(h, t_val, nsz):
    return soft_max_grad(h, t_val, top_k=50) + 1000.0 * (np.sum(h) - nsz / 2.0) * np.ones(nsz)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(64, 0.5)
    n = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Quick L-BFGS-B cleanup
    for t_val in [0.05, 0.01]:
        if tl() < 60:
            break
        try:
            res = minimize(_obj_pen, best_h, args=(t_val, n), jac=_grad_pen,
                           method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 2000, 'ftol': 1e-18, 'gtol': 1e-14})
            ht = project_feasible(res.x, n)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
        except Exception:
            pass

    # Phase 2: Coarse-to-fine multi-scale
    for n_scale in [16, 24, 32, 48, 64, 96, 128, 192]:
        if tl() < 180:
            break
        if n_scale >= n:
            continue
        idx = np.linspace(0, n - 1, n_scale).astype(int)
        h_s = best_h[idx].copy()
        h_s = project_feasible(h_s, n_scale)
        ht_s, c5t_s = pgd_momentum(h_s, max_iter=20000, lr_init=0.02,
                                    t_init=0.1, t_decay=0.999, t_min=0.003, mu=0.7)
        xo = np.linspace(0, 2.0, n_scale, endpoint=False)
        xn = np.linspace(0, 2.0, n, endpoint=False)
        h_up = np.interp(xn, xo, ht_s)
        h_up = project_feasible(h_up, n)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_h, best_c5 = h_up, c5_up

    # Phase 3: Main PGD with momentum at full resolution
    if tl() > 80:
        n_iter = min(60000, int(tl() * 45))
        ht, c5t = pgd_momentum(best_h, max_iter=n_iter, lr_init=0.008,
                                t_init=0.03, t_decay=0.9996, t_min=0.0005, mu=0.85)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Fine-tune with small t
    if tl() > 60:
        n_iter = min(80000, int(tl() * 40))
        ht, c5t = pgd_momentum(best_h, max_iter=n_iter, lr_init=0.003,
                                t_init=0.005, t_decay=0.9999, t_min=0.0001, mu=0.5)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Structured random restarts
    n_restarts = min(25, int(tl() / 25))
    for i in range(n_restarts):
        if tl() < 30:
            break
        mode = i % 4
        if mode == 0:
            h_pert = best_h + 0.04 * rng.randn(n)
        elif mode == 1:
            s = rng.randint(1, max(2, n // 20))
            h_pert = 0.65 * best_h + 0.35 * np.roll(best_h, s)
        elif mode == 2:
            freq = rng.randint(2, 8)
            amp = 0.03
            h_pert = best_h + amp * np.sin(2 * np.pi * freq * np.arange(n) / n + rng.uniform(0, 6.28))
        else:
            idx_swap = rng.choice(n, size=n // 10, replace=False)
            h_pert = best_h.copy()
            h_pert[idx_swap] = np.clip(best_h[idx_swap] + 0.1 * rng.randn(len(idx_swap)), 0, 1)
            h_pert = project_feasible(h_pert, n)
            continue
        h_pert = project_feasible(h_pert, n)
        ht, c5t = pgd_momentum(h_pert, max_iter=8000, lr_init=0.01,
                                t_init=0.03, t_decay=0.998, t_min=0.003, mu=0.7)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Final ultra-fine polish
    if tl() > 20:
        n_iter = min(80000, int(tl() * 45))
        ht, c5t = pgd_momentum(best_h, max_iter=n_iter, lr_init=0.0008,
                                t_init=0.002, t_decay=0.99995, t_min=0.00005, mu=0.3)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
