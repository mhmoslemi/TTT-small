import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_lag(h, lag):
    N = len(h)
    g = np.zeros(N)
    s1 = max(0, -lag)
    e1 = min(N, N - lag)
    s2 = max(0, lag)
    e2 = min(N, N + lag)
    if s1 < e1:
        g[s1:e1] += (1.0 - h[s1+lag:e1+lag])
    if s2 < e2:
        g[s2:e2] -= h[s2-lag:e2-lag]
    return g

def soft_grad(h, corr, alpha, dx):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        m_star = np.argmax(corr)
        return grad_lag(h, m_star - (N - 1)) * dx
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 5e-5:
            continue
        g += w[j] * grad_lag(h, j - (N - 1))
    return g * dx

def interpolate_h(h_src, n_dst):
    x0 = np.linspace(0, 1, len(h_src), endpoint=False)
    x1 = np.linspace(0, 1, n_dst, endpoint=False)

def thue_morse(N):
    t = np.zeros(N)
    for i in range(1, N):
        t[i] = t[i >> 1] ^ (i & 1)
    return t
