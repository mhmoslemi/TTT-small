        g[0:m] += dx * (-h[kn:kn+m])
    return g

This is a great approach because:
- Each coordinate update is exact (or near-exact with a 1D solver)
- It's guaranteed to not increase the objective
- It converges (for convex problems; here we're non-convex but it often works)

def coordinate_update(h, n, i):
    """Find optimal h[i] given all other h[j], j≠i."""
    dx = 2.0 / n
    # For each lag k, g_k as function of t = h[i]:
    # g_k(t) = A_k * t + B_k * t² + C_k (quadratic in t)
    # where A_k, B_k, C_k depend on which other h[j] appear with h[i]
    
    # Actually, g_k = dx * sum_{valid j} h[j] * (1-h[j+k'])
    # h[i] appears when i is in the "first" position or "second" position for lag k
    
    # Let me compute A_k and B_k for each lag:
    # g_k(t) = dx * [terms not involving h[i]] + dx * [terms involving h[i]]
    # Terms involving h[i]:
    #   If h[i] is in the "h" part: dx * t * (1 - h[i+k']) 
    #   If h[i] is in the "(1-h)" part: dx * h[i-k'] * (1 - t)
    
    # So g_k(t) = C_k + dx * t * (1 - h[i+k']) + dx * h[i-k'] * (1-t) [for valid indices]
    #            = C_k + dx * [t - t*h[i+k'] + h[i-k'] - h[i-k']*t]
    #            = C_k + dx * [(1 - h[i-k']) + t * (h[i-k'] - h[i+k'] - 1 + h[i-k'] - h[i-k'])]
    
    # Hmm, let me be more careful. Let me define:
    # a_k = coefficient of t (linear)
    # b_k = coefficient of t² (should be 0 since g_k is linear in each h[j]!)
    
    # Wait! g_k = dx * sum h[j](1-h[j+k]) is BILINEAR in h, not quadratic!
    # If I fix all h[j] for j≠i, then g_k as a function of h[i]=t is LINEAR!
    # Because h[i] appears at most once in each product h[j](1-h[j+k]).
    
    # Actually, h[i] can appear as h[j] in one term AND as h[j+k] in another term of the same sum.
    # E.g., if i = j for one term and i = j'+k for another term j', then:
    # term1: h[i](1-h[i+k]) = t(1-h[i+k])
    # term2: h[i-k](1-h[i]) = h[i-k](1-t)
    # Together: t - t*h[i+k] + h[i-k] - h[i-k]*t = h[i-k] + t(1 - h[i-k] - h[i+k])
    # Still linear in t!
    
    # So g_k(t) = A_k * t + B_k for each k, where A_k, B_k are constants (given other h[j]).
    # The objective is max_k (A_k * t + B_k), which is the max of linear functions = convex piecewise linear.
    # Minimizing this over t ∈ [0,1] is a 1D linear program!
    
    # The minimum is at t=0, t=1, or at a point where two lines cross (and the value is less than at t=0 and t=1).
    
    # So I just need to find the "lower envelope minimum" of the max of lines.
    # This can be done efficiently.
    
    return t_opt

- This is the minimum of a convex piecewise-linear function
- The minimum occurs at a breakpoint or at the boundary

For the max of linear functions f(t) = max_k (a_k*t + b_k), the minimum over [0,1] occurs at:
- t = 0, t = 1, or at a point where two lines cross and the value is below both f(0) and f(1)

To find the optimal t:
1. Compute f(0) = max_k b_k and f(1) = max_k (a_k + b_k)
2. Find all breakpoints in [0,1] where the argmax changes
3. At each breakpoint, compute the value
4. Take the minimum

But actually, there's an even simpler approach. The minimum of max_k (a_k*t + b_k) over t ∈ [0,1] can be found by solving:

This is a 2-variable LP! We can solve it with a simple algorithm or even by enumeration:
- The optimal solution is at a vertex of the feasible region
- The vertices are: (t=0, z=max b_k), (t=1, z=max(a_k+b_k)), or (t*, z*) where t* is where two constraints are tight

t = (b_k2 - b_k1) / (a_k1 - a_k2)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def one_sweep(h, n):
    dx = 2.0 / n
    for i in range(n):
        old_val = h[i]
        h[i] = 0.0
        base = np.correlate(h, 1 - h, mode='full') * dx
        h[i] = 1.0
        top = np.correlate(h, 1 - h, mode='full') * dx
        h[i] = old_val
        slope = top - base
        val_0 = float(np.max(base))
        val_1 = float(np.max(base + slope))
        lo, hi = 0.0, 1.0
        for _ in range(30):
            m1 = lo + (hi - lo) / 3.0
            m2 = hi - (hi - lo) / 3.0
            f1 = np.max(base + slope * m1)
            f2 = np.max(base + slope * m2)
            if f1 < f2:
                hi = m2
            else:
                lo = m1
        t_opt = (lo + hi) / 2.0
        val_opt = float(np.max(base + slope * t_opt))
        best_val = min(val_0, val_1, val_opt)
        if val_0 == best_val:
            h[i] = 0.0
        elif val_1 == best_val:
            h[i] = 1.0
        else:
            h[i] = t_opt

def multi_sweeps(h, n, n_sweeps):
    for _ in range(n_sweeps):
        one_sweep(h, n)
    return h
