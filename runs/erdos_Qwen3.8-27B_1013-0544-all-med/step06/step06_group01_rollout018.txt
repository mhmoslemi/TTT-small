    h = np.array(h, dtype=float)

        g[:n - t] -= h[t:]
    return g * dx

def _subgrad_refine(h, n, iters, topk=3, lr0=0.08):
    h = h.copy().astype(float)
    best_c5 = _c5_max(h)
    best_h = h.copy()
    lr = lr0
    for it in range(iters):
        C = _c5_array(h)
        if C.max() < best_c5 - 1e-12:
            best_c5 = C.max()
            best_h = h.copy()
        top = np.argsort(C)[::-1][:topk]
        grad = np.zeros(n)
        for k_idx in top:
            tau = int(k_idx) - (n - 1)
            grad += _grad_tau(h, tau, 2.0 / n)
        grad /= topk
        h = _project_feasible(h - lr * grad, n)
        lr *= 0.9997
        if lr < 1e-6:
            break
    return best_h, best_c5

def _lbgbs_obj(hv, n, alpha, beta):
    dx = 2.0 / n
    C = _c5_array(hv)
    m = np.max(C)
    sh = alpha * (C - m)
    es = np.exp(sh)
    se = np.sum(es)
    f = m + np.log(se) / alpha + beta * (np.sum(hv) - 0.5 * n) ** 2
    w = es / se
    nw = w > 1e-12
    grad = np.zeros(n)
    for k in np.where(nw)[0]:
        wk = w[k]
        if wk < 1e-12:
            continue
        tau = k - (n - 1)
        if tau >= 0:
            grad[:n - tau] += wk * (1.0 - hv[tau:])
            grad[tau:] -= wk * hv[:n - tau]
        else:
            t = -tau
            grad[t:] += wk * (1.0 - hv[:n - t])
            grad[:n - t] -= wk * hv[t:]
    grad *= dx
    grad += 2.0 * beta * (np.sum(hv) - 0.5 * n)
    return f, grad

def _lbgbs_optimize(h, n, alpha, beta, maxiter=600):
    res = minimize(_lbgbs_obj, h, args=(n, alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 12,
                            'ftol': 1e-15, 'gtol': 1e-12})
    return _project_feasible(res.x, n)

def _resample(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return _project_feasible(np.interp(x_new, x_old, h_old), n_new)

def _try_sinusoidal(n, rng):
    m = 1 + int(rng.randint(0, 5))
    a = 0.1 + 0.3 * rng.rand()
    ph = rng.rand() * 2 * np.pi
    i = np.arange(n, dtype=float)
    h = 0.5 + a * np.sin(2 * np.pi * m * i / n + ph)
    return _project_feasible(h, n)

def _try_sc_step(n, rng):
    a = 0.5 + 3.0 * rng.rand()
    c = 0.2 + 0.6 * rng.rand()
    i = np.arange(n, dtype=float)
    h = 0.5 + 0.5 * np.tanh(a * (i / n - c))
    return _project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    n = len(initial_h_values)
    h = _project_feasible(initial_h_values.astype(float).copy(), n)
    h_best = h.copy()
    c5_best = _c5_max(h)

    # Phase 1: L-BFGS-B progressive alpha
    for alpha in [30, 80, 200, 500, 1500, 4000]:
        if elapsed() > budget_s * 0.22:
            break
        beta = max(5.0, alpha * 0.04)
        hn = _lbgbs_optimize(h, n, alpha, beta, maxiter=1000)
        cn = _c5_max(hn)
        if cn < c5_best - 1e-11:
            h_best, c5_best, h = hn, cn, hn

    # Phase 2: Subgradient refinement (fast, many steps)
    if elapsed() < budget_s * 0.38:
        iters = min(50000, int((budget_s * 0.38 - elapsed()) * 100))
        for tk in [3, 5]:
            if elapsed() > budget_s * 0.36:
                break
            hn, cn = _subgrad_refine(h, n, iters=iters, topk=tk, lr0=0.06)
            if cn < c5_best - 1e-11:
                h_best, c5_best, h = hn, cn, hn

    # Phase 3: L-BFGS-B again (benefits from subgrad refinement)
    if elapsed() < budget_s * 0.48:
        for alpha in [500, 2000, 8000]:
            if elapsed() > budget_s * 0.47:
                break
            beta = max(10.0, alpha * 0.04)
            hn = _lbgbs_optimize(h, n, alpha, beta, maxiter=800)
            cn = _c5_max(hn)
            if cn < c5_best - 1e-11:
                h_best, c5_best, h = hn, cn, hn

    # Phase 4: Try different n values
    n_candidates = sorted(set([n - 5, n - 10, n + 5, n + 10, 80, 85, 100, 110,
                               120, 130, 140, 150, 160]), key=lambda x: abs(x - n))
    n_candidates = [x for x in n_candidates if 60 <= x <= 200 and x != n]
    for nt in n_candidates[:8]:
        if elapsed() > budget_s * 0.68:
            break
        h_try = _resample(h_best, len(h_best), nt)
        for alpha in [100, 500, 2000]:
            if elapsed() > budget_s * 0.67:
                break
            beta = max(5.0, alpha * 0.04)
            hn = _lbgbs_optimize(h_try, nt, alpha, beta, maxiter=600)
            cn = _c5_max(hn)
            if cn < c5_best - 1e-11:
                h_best, c5_best, n = hn, cn, nt
                h_try = hn
        if elapsed() < budget_s * 0.68:
            hn, cn = _subgrad_refine(h_try, nt, iters=20000, topk=3, lr0=0.06)
            if cn < c5_best - 1e-11:
                h_best, c5_best, n = hn, cn, nt

    # Phase 5: Structured random restarts
    n_cur = len(h_best)
    for attempt in range(60):
        if elapsed() > budget_s * 0.85:
            break
        r = rng.rand()
        if r < 0.4:
            h_try = _try_sinusoidal(n_cur, rng)
        elif r < 0.7:
            h_try = _try_sc_step(n_cur, rng)
        else:
            scale = 0.02 + 0.06 * rng.rand()
            h_try = _project_feasible(h_best + rng.randn(n_cur) * scale, n_cur)
        for alpha in [200, 1000]:
            if elapsed() > budget_s * 0.84:
                break
            beta = max(5.0, alpha * 0.04)
            hn = _lbgbs_optimize(h_try, n_cur, alpha, beta, maxiter=500)
            cn = _c5_max(hn)
            if cn < c5_best - 1e-11:
                h_best, c5_best = hn, cn
                h_try = hn
        if elapsed() < budget_s * 0.85:
            hn, cn = _subgrad_refine(h_try, n_cur, iters=15000, topk=3, lr0=0.05)
            if cn < c5_best - 1e-11:
                h_best, c5_best = hn, cn

    # Phase 6: Final subgradient polish
    if elapsed() < budget_s * 0.96:
        remaining = budget_s * 0.96 - elapsed()
        iters = int(remaining * 120)
        iters = max(iters, 5000)
        hn, cn = _subgrad_refine(h_best, len(h_best), iters=iters, topk=3, lr0=0.03)
        if cn < c5_best - 1e-11:
            h_best, c5_best = hn, cn

    # Phase 7: Final L-BFGS-B squeeze with very high alpha
    if elapsed() < budget_s * 0.99:
        hn = _lbgbs_optimize(h_best, len(h_best), 10000, 200.0, maxiter=500)
        cn = _c5_max(hn)
        if cn < c5_best:
            h_best, c5_best = hn, cn

    h_best = _project_feasible(h_best, len(h_best))
    c5_best = _c5_max(h_best)
    return (h_best, c5_best, len(h_best))
