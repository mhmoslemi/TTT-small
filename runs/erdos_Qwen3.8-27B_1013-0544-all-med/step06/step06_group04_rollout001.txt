import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def repair(h, n):
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if not mask.any():
                break
            room = np.sum(1.0 - h[mask])
            frac = min(1.0, diff / room) if room > 0 else 0.0
            h[mask] += frac * (1.0 - h[mask])
        else:
            mask = h > 0.0
            if not mask.any():
                break
            frac = min(1.0, -diff / s) if s > 0 else 0.0
            h[mask] -= frac * h[mask]
    h = np.clip(h, 0.0, 1.0)
    return h

def nm_optimize(h0, n, maxiter, fatol):
    def obj(h):
        hc = np.clip(h, 0.0, 1.0)
        c5 = evaluate_c5(hc)
        pen = 200.0 * (np.sum(hc) - n / 2.0) ** 2
        return c5 + pen
    res = minimize(obj, h0, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-11, 'fatol': fatol, 'adaptive': True})
    return repair(res.x, n), evaluate_c5(repair(res.x, n))

def interpolate_h(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return repair(h_new, n_new)

    # Phase 2: Random restarts with decreasing scale
