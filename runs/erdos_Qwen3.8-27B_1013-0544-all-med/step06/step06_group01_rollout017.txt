            hi = mid
        elif sr < -1e-13:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

def _coord_desc(h, n, lam, max_passes=40):
    """Coordinate descent: min max_k C_k + lam*sum(h)."""
    dx = 2.0 / n
    nc = 2 * n - 1
    mv = np.arange(nc)
    h = h.copy()
    for p in range(max_passes):
        C = _c5_array(h)
        max_chg = 0.0
        for i in range(n):
            si = i + mv - (n - 1)
            fi = i - mv + (n - 1)
            omhs = np.zeros(nc)
            vs = (si >= 0) & (si < n)
            omhs[vs] = 1.0 - h[si[vs]]
            hf = np.zeros(nc)
            vf = (fi >= 0) & (fi < n)
            hf[vf] = h[fi[vf]]
            B = dx * (omhs - hf)
            t = _solve_1d(C, B, 0.0, 1.0, has_lam=True, lam=lam)
            chg = abs(t - h[i])
            if chg > max_chg:
                max_chg = chg
            h[i] = t
        if max_chg < 1e-12:
            break
    return h

def _optimize_lambda(h, n, tol=1e-7):
    """Bisect lambda so that sum(h) = n/2 after coord descent."""
    target = 0.5 * n
    lam_lo, lam_hi = -1.0, 1.0
    for _ in range(12):
        ht = _coord_desc(h, n, lam_hi, max_passes=8)
        if np.sum(ht) <= target:
            break
        lam_hi *= 2
    for _ in range(12):
        ht = _coord_desc(h, n, lam_lo, max_passes=8)
        if np.sum(ht) >= target:
            break
        lam_lo *= 2
    best_h, best_c = h.copy(), _c5_max(h)
    h_cur = h.copy()
    for _ in range(20):
        lam = 0.5 * (lam_lo + lam_hi)
        h_new = _coord_desc(h_cur, n, lam, max_passes=25)
        s = np.sum(h_new)
        if s > target:
            lam_lo = lam
        else:
            lam_hi = lam
        h_cur = h_new
        if abs(s - target) / target < tol:
            h_new = _fix_sum(h_new, n)
            c = _c5_max(h_new)
            if c < best_c:
                best_h, best_c = h_new, c
            break
    else:
        lam = 0.5 * (lam_lo + lam_hi)
        h_new = _coord_desc(h_cur, n, lam, max_passes=40)
        h_new = _fix_sum(h_new, n)
        c = _c5_max(h_new)
        if c < best_c:
            best_h, best_c = h_new, c
    return best_h, best_c

def _local_search(h, n, max_rounds=3, n_pairs=None):
    """Pairwise swap local search."""
    dx = 2.0 / n
    nc = 2 * n - 1
    mv = np.arange(nc)
    best_c = _c5_max(h)
    h = h.copy()
    if n_pairs is None:
        n_pairs = n
    for rnd in range(max_rounds):
        improved = False
        C = _c5_array(h)
        effects = np.zeros((n, nc))
        for i in range(n):
            si = i + mv - (n - 1)
            fi = i - mv + (n - 1)
            omhs = np.zeros(nc)
            vs = (si >= 0) & (si < n)
            omhs[vs] = 1.0 - h[si[vs]]
            hf = np.zeros(nc)
            vf = (fi >= 0) & (fi < n)
            hf[vf] = h[fi[vf]]
            effects[i] = dx * (omhs - hf)
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        np.random.shuffle(pairs)
        for i, j in pairs[:n_pairs]:
            D = effects[i] - effects[j]
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-12:
                continue
            delta = _solve_1d(C, D, d_lo, d_hi)
            if abs(delta) < 1e-12:
                continue
            h_try = h.copy()
            h_try[i] = np.clip(h[i] + delta, 0.0, 1.0)
            h_try[j] = np.clip(h[j] - delta, 0.0, 1.0)
            c_try = _c5_max(h_try)
            if c_try < best_c - 1e-11:
                h = h_try
                best_c = c_try
                improved = True
        if not improved:
            break
    return h, best_c

def _interpolate(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return np.interp(x_new, x_old, h_old)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n_init = len(initial_h_values)
    hb = _fix_sum(initial_h_values.astype(float).copy(), n_init)
    cb = _c5_max(hb)
    nb = n_init

    # Phase 1: Coordinate descent with Lagrange multiplier at original n
    if _el() < budget_s * 0.30:
        hn, cn = _optimize_lambda(hb, nb)
        if cn < cb:
            cb, hb, nb = cn, hn, nb

    # Phase 2: Local search
    if _el() < budget_s * 0.45:
        hb, cb = _local_search(hb, nb, max_rounds=3, n_pairs=nb)

    # Phase 3: Try different resolutions
    n_list = [80, 90, 100, 110, 120, 130, 150, 180, 200, 250, 300]
    n_list = [n for n in n_list if n != nb and 50 <= n <= 400]
    for n_try in n_list:
        if _el() > budget_s * 0.70:
            break
        ht = _fix_sum(_interpolate(hb, nb, n_try), n_try)
        ht2, ct = _optimize_lambda(ht, n_try)
        if ct < cb - 1e-9:
            cb, hb, nb = ct, ht2, n_try
            hb, cb = _local_search(hb, nb, max_rounds=2, n_pairs=nb)

    # Phase 4: Random restarts from best
    for _ in range(30):
        if _el() > budget_s * 0.88:
            break
        scale = 0.02 + 0.04 * rng.rand()
        ht = hb + rng.randn(nb) * scale
        ht = _fix_sum(ht, nb)
        ht2, ct = _optimize_lambda(ht, nb)
        if ct < cb - 1e-9:
            cb, hb = ct, ht2
            hb, cb = _local_search(hb, nb, max_rounds=2, n_pairs=nb)

    # Phase 5: Final local search polish
    if _el() < budget_s * 0.97:
        hb, cb = _local_search(hb, nb, max_rounds=5, n_pairs=nb * 2)
        hb = _fix_sum(hb, nb)
        cb = _c5_max(hb)

    return (hb, cb, nb)
