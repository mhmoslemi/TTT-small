import numpy as np

import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def weighted_grad_fft(h, lam):
    n = len(h)
    dx = 2.0 / n
    N = 4 * n
    H_hat = np.fft.rfft(h, N)
    L_hat = np.fft.rfft(lam, N)
    one_minus_H = np.fft.irfft(np.fft.rfft(1.0 - h, N) * L_hat, N)
    H_conv = np.fft.irfft(H_hat * L_hat, N)
    A = one_minus_H[n-1:2*n-1]
    B = H_conv[n-1:2*n-1]
    return dx * (A - B)

def exact_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    k_star = np.argmax(corr)
    m = k_star - (n - 1)
    sg = np.zeros(n)
    if m >= 0:
        sg[:n-m] -= h[m:]
        sg[m:] += (1.0 - h[:n-m])
    else:
        ma = -m
        sg[ma:] -= h[:n-ma]
        sg[:n-ma] += (1.0 - h[ma:])
    sg *= dx
    return sg, k_star, m

def dual_descent(h, target, n_iters, lr_base=0.5, sharpness=20.0, noise_scale=0.0):
    n = len(h)
    num_lags = 2 * n - 1
    lam = np.ones(num_lags) / num_lags
    h_curr = h.copy()
    
    for it in range(n_iters):
        grad = weighted_grad_fft(h_curr, lam)
        lr = lr_base / (1.0 + 0.01 * it)
        h_new = h_curr - lr * grad
        if noise_scale > 0:
            h_new += noise_scale * np.random.randn(n)
        h_new = project_feasible(h_new, target)
        
        # Update lam
        corr = compute_corr(h_new) * (2.0 / n)
        lam = np.exp(sharpness * (corr - np.max(corr)))
        lam_sum = lam.sum()
        if lam_sum > 0:
            lam /= lam_sum
        else:
            lam = np.ones(num_lags) / num_lags
        
        h_curr = h_new
    
    return h_curr

Actually, let me also think about whether the dual descent is really the right approach. The key issue is:

The weighted objective Σ_k lam[k] * f_k(h) is a "smoothed" version of max_k f_k(h). As sharpness → ∞, it approaches the max. But at finite sharpness, the gradient of the weighted objective points in a direction that balances all lags, which might not be the most efficient direction to reduce the max.

The exact subgradient points in the direction of the argmax lag only. This is more targeted but can be "jumpy" (the argmax can change abruptly).

A good compromise: use the dual descent to converge (it naturally increases sharpness over time), then switch to exact subgradient for pol
</think>

from scipy.optimize import minimize_scalar
import time

    return np.max(corr) * dx

        if abs(d) < 1e-13:

def weighted_grad_fft(h, lam):
    n = len(h)
    dx = 2.0 / n
    N = 4 * n
    L_hat = np.fft.rfft(lam, N)
    conv_A = np.fft.irfft(np.fft.rfft(1.0 - h, N) * L_hat, N)
    conv_B = np.fft.irfft(np.fft.rfft(h, N) * L_hat, N)
    A = conv_A[n - 1:2 * n - 1]
    B = conv_B[n - 1:2 * n - 1]
    return dx * (A - B)

def exact_subgradient(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    k_star = int(np.argmax(corr))
    m = k_star - (n - 1)
    sg = np.zeros(n)

        sg[:n - m] -= h[m:]
        sg[m:] += (1.0 - h[:n - m])
    else:
        ma = -m
        sg[ma:] -= h[:n - ma]
        sg[:n - ma] += (1.0 - h[ma:])
    sg *= dx
    return sg, k_star, m

def multi_lag_subgradient(h, top_k=3):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    top_indices = np.argsort(corr)[::-1][:top_k]
    sg = np.zeros(n)
    for k_star in top_indices:
        m = k_star - (n - 1)
        if m >= 0:
            sg[:n - m] -= h[m:]
            sg[m:] += (1.0 - h[:n - m])
        else:
            ma = -m
            sg[ma:] -= h[:n - ma]
            sg[:n - ma] += (1.0 - h[ma:])
    sg *= dx / top_k
    return sg

def dual_descent(h, target, n_iters, lr_base=0.3, sharpness=15.0, decay=0.999):
    n = len(h)
    num_lags = 2 * n - 1
    lam = np.ones(num_lags) / num_lags
    h_curr = h.copy()
    for it in range(n_iters):
        grad = weighted_grad_fft(h_curr, lam)
        lr = lr_base * (decay ** it)
        h_new = h_curr - lr * grad
        h_new = project_feasible(h_new, target)
        corr = compute_corr(h_new) * (2.0 / n)
        lam = np.exp(sharpness * (corr - np.max(corr)))
        s = lam.sum()
        if s > 1e-30:
            lam /= s
        else:
            lam = np.ones(num_lags) / num_lags
        h_curr = h_new
    return h_curr

def targeted_line_search(h, target, top_n=30, n_trials=12):
    sg, _, _ = exact_subgradient(h)
    idx = np.argsort(np.abs(sg))[::-1][:top_n]
    improved = False
    for j in idx:
        orig = h.copy()
        orig_c = compute_c5(h)
        best_v = h[j]
        best_c = orig_c
        for v in np.linspace(0.0, 1.0, n_trials):
            if abs(v - h[j]) < 1e-8:

            h_trial = orig.copy()
            h_trial[j] = v
            h_trial = project_feasible(h_trial, target)
            c = compute_c5(h_trial)
            if c < best_c:
                best_c = c
                best_v = v
        if best_c < orig_c - 1e-12:
            h[j] = best_v
            h = project_feasible(h, target)
            improved = True
        else:
            h = orig
    return h, improved
