    dx = 2.0 / n

    nc = len(C)

    grad = np.zeros(n)

        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n-1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n-1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)

    return f, grad

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = 0.5 * n
    for _ in range(40):
        d = target - np.sum(h)
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += d / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h
