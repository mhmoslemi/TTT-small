import numpy as np
import time
import scipy.optimize as so
import warnings
warnings.simplefilter("ignore")

def _time_left(deadline):
    return max(0.0, deadline - time.time())

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(50):
        diff = target - float(x.sum())
        if abs(diff) <= 1e-14:
            break
        if diff > 0:
            slack = 1.0 - x
            m = slack > 1e-14
            if not np.any(m):
                break
            x[m] += diff * slack[m] / float(slack[m].sum())
        else:
            m = x > 1e-14
            if not np.any(m):
                break
            x[m] += diff * x[m] / float(x[m].sum())
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - float(x.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(1.0 - x)[::-1]
            for idx in order:
                if diff <= 1e-15: break
                add = min(diff, 1.0 - x[idx])
                x[idx] += add; diff -= add
        else:
            order = np.argsort(x)[::-1]
            for idx in order:
                if diff >= -1e-15: break
                add = max(diff, -x[idx])
                x[idx] += add; diff -= add
    return x

def c_vals(h):
    n = h.size
    return np.correlate(h, 1.0-h, mode="full") * (2.0/n)

def exact_c(h):
    return float(np.max(c_vals(h)))

def soft_max(c, tau):
    cm = c.max()
    if tau <= 0:
        return cm
    z = (c - cm) / tau
    z -= z.max()
    e = np.exp(z)
    return cm + tau * float(np.log(e.sum()))

def smooth_obj(x, tau, penalty=0.0):
    c = c_vals(x)
    val = soft_max(c, tau)
    if penalty > 0:
        val += penalty * (x.sum() - 0.5*x.size)**2
    return val

def smooth_grad(x, tau, penalty=0.0):
    # Gradient of soft_max(c, tau) w.r.t. x
    c = c_vals(x)
    cm = c.max()
    if tau <= 0:
        # Just use gradient of max
        k = int(np.argmax(c))
        w = np.zeros_like(c)
        w[k] = 1.0
    else:
        z = (c - cm) / tau
        z -= z.max()
        e = np.exp(z)
        w = e / e.sum()
    # w[k] is the weight for each c_k
    # Now compute gradient
    n = x.size
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(c)):
        if w[k] < 1e-15:
            continue
        s = k - (n - 1)  # actual lag
        # c_k = sum_i x[i] * (1-x[i+s]) for valid i
        # d c_k / d x[i] = (1-x[i+s]) - x[i-s] (when both valid)
        lo = max(0, -s)
        hi = min(n-1, n-1-s)
        if lo <= hi:
            # from first factor: (1 - x[i+s])
            grad[lo:hi+1] += w[k] * (1.0 - x[lo+s:hi+1+s])
        lo2 = max(0, s)
        hi2 = min(n-1, n-1+s)
        if lo2 <= hi2:
            # from second factor: -x[i-s]
            grad[lo2:hi2+1] += w[k] * (-x[lo2-s:hi2+1-s])
    grad *= dx
    if penalty > 0:
        grad += 2 * penalty * (x.sum() - 0.5*x.size)
    return grad

But actually, I can vectorize this better. Let me think...

For each k (correlate index), s = k - (n-1) is the lag.
The gradient contribution from c_k to x[j] is:
- (1 - x[j+s]) if 0 ≤ j ≤ n-1 and 0 ≤ j+s ≤ n-1 [j is in the first factor]
- -x[j-s] if 0 ≤ j ≤ n-1 and 0 ≤ j-s ≤ n-1 [j is in the second factor, j = i+s so i = j-s]

So for a given s:
- First factor: valid j are max(0,-s) to min(n-1, n-1-s), contribution (1-x[j+s])
- Second factor: valid j are max(0,s) to min(n-1, n-1+s), contribution -x[j-s]

This is what I had. The loop over k is the slow part, but for n=70, there are 139 values of k, and each does O(n) work, so total is O(n²) ≈ 10000. Fine.

Actually, I can vectorize over k using the fact that for each j, the sum over k of w[k] * grad_contribution(k, j) can be computed more efficiently. But for n=70, it doesn't matter.

The penalty approach: if penalty is large enough, the sum constraint is satisfied to high precision. The downside is that it might not be exact.

**Approach 2: Projected gradient with Nesterov momentum**

For the constraint sum(h) = n/2 and 0 ≤ h ≤ 1:
- The projection onto this set is: project onto [0,1]^n first, then adjust to satisfy the sum
- The "projection" onto {h : 0≤h≤1, sum(h)=n/2} is:
  1. Clip to [0,1]
  2. If sum > n/2: reduce the largest values proportionally (water-filling from top)
  3. If sum < n/2: increase the smallest values proportionally

This is exactly what `repair` does!

So the projected gradient step is:
1. Compute gradient g of the smooth objective
2. Take a step: y = x - α * g
3. Project: x_new = repair(y)
4. (Nesterov: use extrapolated point for gradient evaluation)

This is clean and efficient. Let me implement this.

**Approach 3: Alternating between smooth and exact objective**

1. Use smooth objective with L-BFGS-B to get close to optimum
2. Switch to exact objective with subgradient + line search for final polish
3. Use coordinate descent for the very last improvement

import numpy as np
import time
import scipy.optimize as so
import warnings
warnings.simplefilter("ignore")

def _tl(deadline):
    return max(0.0, deadline - time.time())

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0: return x
    t = 0.5 * n
    np.clip(x, 0, 1, out=x)
    for _ in range(50):
        d = t - x.sum()
        if abs(d) < 1e-14: break
        if d > 0:
            s = 1 - x; m = s > 1e-14
            if not m.any(): break
            x[m] += d * s[m] / s[m].sum()
        else:
            m = x > 1e-14
            if not m.any(): break
            x[m] += d * x[m] / x[m].sum()
        np.clip(x, 0, 1, out=x)
    d = t - x.sum()
    if abs(d) > 1e-10:
        if d > 0:
            for i in np.argsort(1-x)[::-1]:
                if d < 1e-15: break
                a = min(d, 1-x[i]); x[i] += a; d -= a
        else:
            for i in np.argsort(x)[::-1]:
                if d > -1e-15: break
                a = max(d, -x[i]); x[i] += a; d -= a
    return x

def cv(h):
    return np.correlate(h, 1-h, mode="full") * (2.0/h.size)

def ec(h):
    return float(np.max(cv(h)))

def sobj(x, tau):
    c = cv(x)
    m = c.max()
    if tau < 1e-10: return m
    z = (c - m) / tau
    z -= z.max()
    e = np.exp(z)
    return m + tau * np.log(e.sum())

def sgrad(x, tau):
    c = cv(x)
    m = c.max()
    n = x.size
    dx = 2.0/n
    if tau < 1e-10:
        w = np.zeros_like(c); w[c.argmax()] = 1
    else:
        z = (c-m)/tau; z -= z.max()
        e = np.exp(z); w = e/e.sum()
    g = np.zeros(n)
    for k in range(2*n-1):
        wk = w[k]
        if wk < 1e-14: continue
        s = k - (n-1)
        lo = max(0, -s); hi = min(n-1, n-1-s)
        if lo <= hi: g[lo:hi+1] += wk * (1.0 - x[lo+s:hi+1+s])
        lo2 = max(0, s); hi2 = min(n-1, n-1+s)
        if lo2 <= hi2: g[lo2:hi2+1] -= wk * x[lo2-s:hi2+1-s]
    return g * dx

def opt_lbfgsb(x0, time_limit, tau=0.01, penalty=200.0):
    n = x0.size
    bounds = [(0,1)]*n
    def f(x): return sobj(x, tau) + penalty*(x.sum()-0.5*n)**2
    def jf(x): return sgrad(x, tau) + 2*penalty*(x.sum()-0.5*n)
    start = time.time()
    res = so.minimize(f, x0, jac=jf, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter':1000, 'ftol':1e-15, 'gtol':1e-10, 'maxfun':20000})
    xr = repair(res.x)
    return xr, ec(xr)

def opt_momentum(x0, time_limit, seed=0):
    """Nesterov projected gradient with momentum"""
    n = x0.size
    x = x0.copy()
    v = np.zeros(n)
    best_x = x.copy()
    best_c = ec(x)
    alpha = 0.1
    beta = 0.9
    rng = np.random.RandomState(seed)
    deadline = time.time() + time_limit
    no_imp = 0
    
    while _tl(deadline) > 0.1:
        y = x + beta * v  # Nesterov extrapolation
        g = sgrad(y, 0.005)
        v = beta * v - alpha * g
        x_new = repair(x + v)
        c_new = ec(x_new)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_x = x_new.copy()
            no_imp = 0
            alpha = min(0.3, alpha * 1.05)
        else:
            no_imp += 1
            alpha *= 0.8
            if no_imp > 20:
                # Perturb
                x_new = repair(x + rng.normal(0, 0.02, n))
                c_new = ec(x_new)
                if c_new < best_c:
                    best_c = c_new
                    best_x = x_new.copy()
                    no_imp = 0
                else:
                    break
        x = x_new
        if no_imp > 30:
            break
    
    return best_x, best_c

def upsample(h, n_new):
    n = h.size
    t_old = (np.arange(n) + 0.5) * 2.0/n
    t_new = (np.arange(n_new) + 0.5) * 2.0/n_new
    return repair(np.interp(t_new, t_old, h))

def binary_start(n, seed, nblocks=2):
    """Create a binary-like starting point with nblocks high regions"""
    rng = np.random.RandomState(seed)
    x = np.zeros(n)
    # Create nblocks intervals where h=1
    widths = np.sort(rng.dirichlet(np.ones(nblocks) * 2) * n // 2)
    pos = np.sort(rng.choice(n, nblocks, replace=False))
    for p, w in zip(pos, widths):
        x[p:min(n, p+int(w))] = 1.0
    # Add some fractional values
    for _ in range(n // 5):
        i = rng.randint(n)
        x[i] = rng.uniform(0, 1)
    return repair(x)

def sine_start(n, seed, freq):
    rng = np.random.RandomState(seed)
    t = (np.arange(n) + 0.5) * 2.0/n
    phase = rng.uniform(0, 2*np.pi)
    amp = 0.4 + 0.2 * rng.rand()
    h = 0.5 + amp * np.sin(2*np.pi*freq*t/2.0 + phase)
    h += 0.03 * rng.normal(0, 1, n)
    return repair(h)

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    s0 = int(seed)
    
    # Get initial values
    init = globals().get("initial_h_values", None)
    h_init = repair(np.asarray(init, dtype=float)) if init is not None else None
    n_init = h_init.size if h_init is not None else 70
    
    best_h = h_init if h_init is not None else repair(np.full(70, 0.5))
    best_c = ec(best_h)
    
    # Phase 1: Optimize initial at its resolution
    if _tl(deadline) > 60:
        h1, c1 = opt_lbfgsb(best_h, min(40, _tl(deadline)*0.3), tau=0.02, penalty=500)
        if c1 < best_c: best_h, best_c = h1, c1
        h2, c2 = opt_lbfgsb(best_h, min(30, _tl(deadline)*0.25), tau=0.005, penalty=500)
        if c2 < best_c: best_h, best_c = h2, c2
        h3, c3 = opt_momentum(best_h, min(20, _tl(deadline)*0.2), seed=s0)
        if c3 < best_c: best_h, best_c = h3, c3
    
    # Phase 2: Try different resolutions
    resolutions = []
    if n_init >= 50:
        resolutions = [n_init, n_init + 20, n_init + 40, n_init + 60]
    else:
        resolutions = [n_init, n_init + 30, n_init + 60, n_init + 100]
    resolutions = [r for r in resolutions if 40 <= r <= 300]
    
    for nr in resolutions:
        if _tl(deadline) < 50: break
        if nr == n_init: continue  # Already optimized
        
        h_up = upsample(best_h, nr)
        c_up = ec(h_up)
        
        # Optimize at new resolution
        h_r, c_r = opt_lbfgsb(h_up, min(40, _tl(deadline)*0.3), tau=0.02, penalty=500)
        if c_r < best_c: best_h, best_c = h_r, c_r
        
        h_r2, c_r2 = opt_lbfgsb(best_h, min(25, _tl(deadline)*0.2), tau=0.005, penalty=500)
        if c_r2 < best_c: best_h, best_c = h_r2, c_r2
        
        if _tl(deadline) > 20:
            h_r3, c_r3 = opt_momentum(best_h, min(15, _tl(deadline)*0.15), seed=s0+nr)
            if c_r3 < best_c: best_h, best_c = h_r3, c_r3
    
    # Phase 3: Try different starting topologies at best resolution
    n_best = best_h.size
    starts = []
    for freq in [1, 2, 3, 5]:
        starts.append(sine_start(n_best, s0 + freq*100, freq))
    for nb in [2, 3, 4]:
        starts.append(binary_start(n_best, s0 + nb*200, nb))
    
    for h_start in starts:
        if _tl(deadline) < 30: break
        c_s = ec(h_start)
        if c_s > best_c + 0.01: continue  # Skip if too far
        h_opt, c_opt = opt_lbfgsb(h_start, min(25, _tl(deadline)*0.2), tau=0.02, penalty=500)
        if c_opt < best_c: best_h, best_c = h_opt, c_opt
        h_opt2, c_opt2 = opt_lbfgsb(best_h, min(20, _tl(deadline)*0.15), tau=0.003, penalty=800)
        if c_opt2 < best_c: best_h, best_c = h_opt2, c_opt2
    
    # Phase 4: Final polish
    if _tl(deadline) > 10:
        h_f, c_f = opt_momentum(best_h, min(20, _tl(deadline)*0.5), seed=s0+999)
        if c_f < best_c: best_h, best_c = h_f, c_f

        if c_f2 < best_c: best_h, best_c = h_f2, c_f2
    
    best_h = repair(best_h)
    best_c = ec(best_h)
    return best_h, best_c, int(best_h.size)

def _tl(dl):
    return max(0.0, dl - time.time())

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    t = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(60):
        d = t - float(x.sum())
        if abs(d) < 1e-14:
            break
        if d > 0:
            s = 1.0 - x
            m = s > 1e-14
            if not np.any(m):
                break
            x[m] += d * s[m] / float(s[m].sum())
        else:
            m = x > 1e-14
            if not np.any(m):
                break
            x[m] += d * x[m] / float(x[m].sum())
        np.clip(x, 0.0, 1.0, out=x)
    d = t - float(x.sum())
    if abs(d) > 1e-10:
        if d > 0:
            for i in np.argsort(1.0 - x)[::-1]:
                if d < 1e-15: break
                a = min(d, 1.0 - x[i]); x[i] += a; d -= a
        else:
            for i in np.argsort(x)[::-1]:
                if d > -1e-15: break
                a = max(d, -x[i]); x[i] += a; d -= a
    return x

def cv(h):
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / h.size)

def ec(h):
    return float(np.max(cv(h)))

def sobj(x, tau):
    c = cv(x)
    m = float(c.max())
    if tau < 1e-12:
        return m
    z = (c - m) / tau
    z -= z.max()
    e = np.exp(z)
    return m + tau * float(np.log(e.sum()))

def sgrad(x, tau):
    c = cv(x)
    m = float(c.max())
    n = x.size
    dx = 2.0 / n
    if tau < 1e-12:
        w = np.zeros(2*n-1)
        w[int(np.argmax(c))] = 1.0
    else:
        z = (c - m) / tau
        z -= z.max()
        e = np.exp(z)
        w = e / e.sum()
    g = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        lo = max(0, -s)
        hi = min(n - 1, n - 1 - s)
        if lo <= hi:
            g[lo:hi+1] += wk * (1.0 - x[lo+s:hi+1+s])
        lo2 = max(0, s)
        hi2 = min(n - 1, n - 1 + s)
        if lo2 <= hi2:
            g[lo2:hi2+1] -= wk * x[lo2-s:hi2+1-s]
    return g * dx

def opt_lbfgsb(x0, time_limit, tau=0.01, penalty=800.0):
    n = x0.size
    bounds = [(0.0, 1.0)] * n
    def f(x):
        return sobj(x, tau) + penalty * (x.sum() - 0.5*n)**2
    def jf(x):
        return sgrad(x, tau) + 2.0 * penalty * (x.sum() - 0.5*n)
    res = so.minimize(f, x0, jac=jf, method='L-BFGS-B', bounds=bounds,
                      options={'maxiter': 2000, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': 50000})
    xr = repair(res.x)
    return xr, ec(xr)

def coord_polish(h, time_limit, seed=0):

def upsample(h, n_new):
    n = h.size
    t_old = (np.arange(n) + 0.5) * 2.0 / n
    t_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    return repair(np.interp(t_new, t_old, h))

    rng = np.random.RandomState(seed)
    t = (np.arange(n) + 0.5) * 2.0 / n
    phase = rng.uniform(0, 2*np.pi)

    h = 0.5 + amp * np.sin(2*np.pi*freq*t/2.0 + phase)
    h += 0.03 * rng.normal(0, 1, n)
    return repair(h)

def binary_start(n, seed, nblocks=3):
    rng = np.random.RandomState(seed)
    x = np.full(n, 0.3)
    total_high = n // 2
    per_block = max(1, total_high // nblocks)
    positions = np.sort(rng.choice(n, nblocks, replace=False))
    for p in positions:
        x[p:min(n, p + per_block)] = 1.0
    for _ in range(n // 4):
        x[rng.randint(n)] = rng.uniform(0, 1)
    return repair(x)

def jitter(h, scale, seed):
    rng = np.random.RandomState(seed)
    return repair(h + rng.normal(0, scale, h.size))
