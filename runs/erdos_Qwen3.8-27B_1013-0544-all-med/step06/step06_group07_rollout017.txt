def coordinate_descent(x0, time_limit, seed):

            # Try a few values for x[j]

For coordinate descent on a single coordinate j, I only need to evaluate the change in c_values when x[j] changes. The only lags that change are those k where j is involved, i.e., all k from -(n-1) to (n-1) (since j can appear as either the first or second index in the correlation sum for any lag).

Actually, for a change in x[j], the correlation at lag k changes if:
- k ≥ 0: j ≤ n-1-k (j appears as first index) OR j ≥ k (j appears as second index)
- k < 0: similar

So actually all lags are affected by a change in x[j]. That means I need to recompute all correlations.

However, I can do it more efficiently. The correlation at lag k is:
f_k = ∑_{i=0}^{n-1-k} x[i]*(1-x[i+k])

If I change x[j] to x[j]+δ:
- For k ≥ 0:
  - If j ≤ n-1-k: f_k changes by δ*(1-x[j+k])
  - If j ≥ k: f_k changes by -δ*x[j-k]... wait no.
  
  Actually, x[j] appears in f_k in these terms:
  - As x[i] when i=j (if j ≤ n-1-k): contributes x[j]*(1-x[j+k])
  - As x[i+k] when i+k=j, i.e., i=j-k (if j-k ≥ 0 and j-k ≤ n-1-k, i.e., k≤j≤n-1): contributes x[j-k]*(1-x[j])
  
  So the change in f_k when x[j] → x[j]+δ is:
  δ*(1-x[j+k]) [if j ≤ n-1-k] + (-δ)*x[j-k] [if k≤j]
  
  = δ * [(1-x[j+k]) (if j≤n-1-k) - x[j-k] (if j≥k)]

This means I can update all f_k values in O(n) time for a single coordinate change.

import numpy as np
import time
import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(80):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            slack = 1.0 - x
            m = slack > 1e-14
            if not m.any():
                x[:] = 0.5
                break
            x[m] += diff * slack[m] / slack[m].sum()
        else:
            m = x > 1e-14
            if not m.any():
                break
            x[m] += diff * x[m] / x[m].sum()
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        x[:] = 0.5
    elif abs(diff) > 1e-14:
        if diff > 0:
            m = 1.0 - x > 1e-14
            x[m] += diff * (1.0 - x[m]) / (1.0 - x[m]).sum()
        else:
            m = x > 1e-14
            x[m] += diff * x[m] / x[m].sum()
        np.clip(x, 0.0, 1.0, out=x)
    return x

def c_values(h):
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(c_values(h)))

def subgrad_max(x, cvals, n, top_k=6):
    C = float(cvals.max())
    tol = max(1e-9, 5e-4 * C)
    active = np.where(cvals >= C - tol)[0]
    if len(active) > top_k:
        active = np.argsort(cvals)[::-1][:top_k]
    grad = np.zeros(n)
    dx = 2.0 / n
    for k in active:
        k = int(k)
        if k >= 0:
            if n - k > 0:
                grad[0:n-k] += (1.0 - x[k:n])
                grad[k:n] -= x[0:n-k]
        else:
            kp = -k
            if n - kp > 0:
                grad[kp:n] += (1.0 - x[0:n-kp])
                grad[0:n-kp] -= x[kp:n]
    return grad * dx / len(active)

def momentum_opt(x0, time_limit, seed, lr=0.08, mom=0.85):

    x = np.array(x0, dtype=float)
    n = x.size
    v = np.zeros(n)
    best_x = x.copy()
    best_c = exact_c(x)
    deadline = time.time() + time_limit
    step = lr
    stall = 0

    while time.time() < deadline:
        cvals = c_values(x)
        C = float(cvals.max())
        if C < best_c - 1e-15:
            best_c = C
            best_x = x.copy()
            stall = 0
        else:
            stall += 1

        if stall > 80:
            x = repair(x + rng.normal(0, 0.008, n))
            v *= 0.3
            stall = 0
            step = lr
            continue

        g = subgrad_max(x, cvals, n)
        v = mom * v - step * g
        x = repair(x + v)
        step = max(0.002, step * 0.9995)
        if rng.random() < 0.05:
            step = lr * (0.5 + rng.random())

    return best_x, best_c

def quantized_sa(x0, time_limit, seed, n_levels=4):

    T = 0.015

    evals = 0

    while time.time() < deadline:

        old = q[j]
        q[j] = rng.randint(n_levels)
        if q[j] == old:

        c_new = exact_c(x_new)
        evals += 1
        if c_new <= c_cur + 1e-15:
            x = x_new
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_x = x.copy()

            if rng.random() < accept:
                x = x_new
                c_cur = c_new
            else:
                q[j] = old
        T *= 0.99985
        if T < 0.0008:
            T = 0.0008
    return best_x, best_c

def resample_h(h0, n_new):
    n0 = h0.size
    if n0 == n_new:
        return repair(h0)
    if n0 == 0:
        return repair(np.full(n_new, 0.5))

def stop_and_go(x0, time_limit, seed):
    """Alternates between gradient descent and random restarts."""
    rng = np.random.RandomState(seed)
    x = np.array(x0, dtype=float)

            v = np.zeros(n)

    return best_x, best_c
