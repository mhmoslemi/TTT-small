            if s < 1e-13:

def grad_single_lag(h, n, shift):
    grad = np.zeros(n)
    lo1 = max(0, shift)
    hi1 = min(n, n + shift)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1 - shift:hi1 - shift])

                break
            h_rand = np.zeros(n)
            idx = rng.permutation(n)
            h_rand[idx[:n // 2]] = 1.0
            hb, cb = binary_local_search(h_rand, n, max_moves=8000, rng=rng)
            if cb < best_c5:
                best_h, best_c5 = hb, cb

    # Phase 5: More subgradient from current best
    if remaining() > 60:
        h5, c5 = subgradient_optimize(best_h, n, max_iter=40000, base_lr=0.03)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    # Phase 6: SLSQP refinement with higher alpha
    for alpha in [5000, 15000]:
        if remaining() < 20:
            break
        h6, c6 = slsqp_refine(best_h, n, alpha=alpha, maxiter=5000)
        if c6 < best_c5:
            best_h, best_c5 = h6, c6

    # Phase 7: Final smooth gradient polish
    if remaining() > 20:
        h7, c7 = smooth_grad_descent(best_h, n, 2000, max_iter=3000, lr=0.001, momentum=0.95)
        if c7 < best_c5:
            best_h, best_c5 = h7, c7

    # Final SLSQP
    if remaining() > 10:
        h8, c8 = slsqp_refine(best_h, n, alpha=20000, maxiter=8000)
        if c8 < best_c5:
            best_h, best_c5 = h8, c8

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
