    # So dC_k/dh_i = dx * [(1-h[i+lag_k]) * 1{0<=i+lag_k<n} + (-h[i-lag_k]) * 1{0<=i-lag_k<n}]
    # where lag_k = k - (n-1)
    
    lags = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    # For each i, we need sum over k of w[k] * dx * [(1-h[i+lag_k]) * valid + (-h[i-lag_k]) * valid]
    
    # This is O(n * 2n) = O(n²) which is fine for n~100
    grad = np.zeros(n)
    for i in range(n):
        # Terms where h[i] is the "first" factor: lag such that i+lag is in [0,n)
        # i.e., lag in [-i, n-1-i]
        # Terms where h[i] is the "second" factor (as h[n+lag]): n+lag=i, n=i-lag
        # i.e., lag in [i-n+1, i]
        
        # First type: contribution (1-h[i+lag]) for valid lag
        # Second type: contribution -h[i-lag] for valid lag
        
        # Combined: for each k (index into C), lag_k = lags[k]
        # If 0 <= i+lag_k < n: add (1-h[i+lag_k])
        # If 0 <= i-lag_k < n: add (-h[i-lag_k])
        
        # Vectorized:
        plus_lag = lags + i  # = i + lag_k, need 0 <= this < n
        minus_lag = i - lags  # = i - lag_k, need 0 <= this < n
        
        mask1 = (plus_lag >= 0) & (plus_lag < n)
        mask2 = (minus_lag >= 0) & (minus_lag < n)
        
        contrib1 = np.where(mask1, 1.0 - h[np.clip(plus_lag, 0, n-1)], 0.0)
        contrib2 = np.where(mask2, -h[np.clip(minus_lag, 0, n-1)], 0.0)
        
        grad[i] = dx * np.dot(w, contrib1 + contrib2)
    
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return grad

def _grad_func(h, alpha, beta):
    return _smooth_obj(h, alpha, beta), _smooth_grad(h, alpha, beta)

def _optimize(h, n, alpha, beta, maxiter=500):
    res = minimize(_grad_func, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)]*n,
                   options={'maxiter': maxiter, 'maxfun': maxiter*20, 'ftol': 1e-16, 'gtol': 1e-12})
    return _project(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _project(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = 0.5 * n - np.sum(h)
    if abs(diff) > 1e-14:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = 0.5 * n - np.sum(h)
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                order = np.argsort(1.0 - h)
            else:
                order = np.argsort(h)
            for i in order:
                if diff2 > 0:
                    add = min(diff2, 1.0 - h[i])
                    h[i] += add
                    diff2 -= add
                else:
                    sub = min(-diff2, h[i])
                    h[i] -= sub
                    diff2 += sub
                if abs(diff2) < 1e-14:
                    break
    return h

def _obj_and_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(C)
    e = np.exp(alpha * (C - m))
    s = np.sum(e)
    f = m + np.log(s) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = e / s
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        pl = lags + i
        ml = i - lags
        m1 = (pl >= 0) & (pl < n)
        m2 = (ml >= 0) & (ml < n)
        c1 = np.where(m1, 1.0 - h[np.clip(pl, 0, n - 1)], 0.0)
        c2 = np.where(m2, -h[np.clip(ml, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, c1 + c2)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_max(h)
    cur_c5 = best_c5
    for it in range(n_iter):
        T = t_start * (t_end / t_start) ** (it / max(1, n_iter - 1))
        move_type = rng.randint(3)
        if move_type == 0:
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.uniform(0.02, 0.15) * (1 if rng.rand() < 0.5 else -1)
            h_try = h.copy()
            h_try[i] = np.clip(h[i] + delta, 0, 1)
            h_try[j] = np.clip(h[j] - delta, 0, 1)
        elif move_type == 1:
            i = rng.randint(n)
            h_try = h.copy()
            h_try[i] = rng.uniform(0, 1)
        else:
            i, j = rng.choice(n, 2, replace=False)
            h_try = h.copy()
            h_try[i], h_try[j] = h[j], h[i]
        h_try = _project(h_try, n)
        new_c5 = _c5_max(h_try)

        if delta_f < 0 or rng.rand() < np.exp(-delta_f / max(T, 1e-8)):
            h = h_try
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
    return best_h, best_c5
