```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _project_sum(h, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _softmax_obj(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    f = (m + np.log(se) / alpha) + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    nc = len(C)
    grad = np.zeros(n)
    for k in range(nc):
        if w[k] < 1e-10:
            continue
        s = k - (n - 1)
        wk_dx = w[k] * dx
        if s >= 0:
            grad[:n-s] += wk_dx * (1.0 - h[s:])
            grad[s:] -= wk_dx * h[:n-s]
        else:
            ss = -s
            grad[ss:] += wk_dx * (1.0 - h[:n-ss])
            grad[:n-ss] -= wk_dx * h[ss:]
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _optimize_lbfgsb(h, alpha, beta, maxiter=500):
    n = len(h)
    res = minimize(_softmax_obj, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-14})
    if res.x is not None:
        h = _project_sum(res.x, n)
    return h

def _coord_descent(h, n, time_limit):
    """Coordinate descent with 1D min-max optimization per coordinate."""
    dx = 2.0 / n
    n_c = 2 * n - 1
    h = h.copy()
    best_c5 = _c5_max(h)
    t_start = time.time()
    sweep = 0
    
    while time.time() - t_start < time_limit:
        sweep += 1
        for i in range(n):
            # Compute a_k and b_k for all k
            C_old = _c5_array(h)
            a = np.zeros(n_c)
            b = np.zeros(n_c)
            for k in range(n_c):
                s = k - (n - 1)
                a_k = 0.0
                if 0 <= i + s < n:
                    a_k += (1.0 - h[i + s])
                if 0 <= i - s < n:
                    a_k -= h[i - s]
                a[k] = a_k * dx
                b[k] = C_old[k] - a[k] * h[i]
            
            # Find t in [0,1] minimizing max_k(a[k]*t + b[k])
            candidates = np.array([0.0, 1.0])
            # Intersections
            for k1 in range(n_c):
                for k2 in range(k1 + 1, n_c):
                    da = a[k1] - a[k2]
                    if abs(da) > 1e-14:
                        t_star = (b[k2] - b[k1]) / da
                        if -0.01 <= t_star <= 1.01:
                            candidates = np.append(candidates, t_star)
            
            # Clip and unique
            candidates = np.clip(np.unique(candidates), 0.0, 1.0)
            
            # Evaluate
            best_t = h[i]
            best_val = np.max(a * h[i] + b)
            for t in candidates:
                val = np.max(a * t + b)
                if val < best_val - 1e-12:
                    best_val = val
                    best_t = t
            
            h[i] = best_t
        
        h = _project_sum(h, n)
        c5 = _c5_max(h)
        if c5 < best_c5:
            best_c5 = c5
    
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _project_sum(h, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _softmax_fg(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    m = np.max(C)
    at = alpha * (C - m)
    at[at > 500] = 500  # clip for numerical stability
    es = np.exp(at)
    se = np.sum(es)
    f = m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    n_c = len(C)
    s_arr = np.arange(-(n-1), n)
    j = np.arange(n)
    grad = np.zeros(n)
    for k in range(n_c):
        wk = w[k]
        if wk < 1e-12:
            continue
        sk = s_arr[k]
        jp = j + sk
        jm = j - sk
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        grad[vp] += wk * dx * (1.0 - h[jp[vp]])
        grad[vm] -= wk * dx * h[jm[vm]]
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgsb(h, alpha, beta, maxiter=400):
    n = len(h)
    res = minimize(_softmax_fg, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-14})
    if res.x is not None:
        return _project_sum(res.x, n)
    return h

def _coord_descent(h, n, time_limit, top_frac=1.0):
    dx = 2.0 / n
    n_c = 2 * n - 1
    s_arr = np.arange(-(n-1), n)
    j = np.arange(n)
    h = h.copy()
    best_c5 = _c5_max(h)
    t_start = time.time()
    stall_count = 0
    prev_c5 = best_c5
    
    while time.time() - t_start < time_limit:
        C_all = _c5_array(h)
        # Optional: only use top fraction of lags
        if top_frac < 1.0:
            n_active = max(5, int(n_c * top_frac))
            thresh = np.sort(C_all)[-n_active]
            active = np.where(C_all >= thresh)[0]
        else:
            active = np.arange(n_c)
        
        for i in range(n):
            # Compute a and b for active lags
            a = np.zeros(len(active))
            b = np.zeros(len(active))
            for idx, k in enumerate(active):
                sk = s_arr[k]
                ip = i + sk
                im = i - sk
                a_k = 0.0
                if 0 <= ip < n:
                    a_k += (1.0 - h[ip])
                if 0 <= im < n:
                    a_k -= h[im]
                a[idx] = a_k * dx
                b[idx] = C_all[k] - a[idx] * h[i]
            
            # 1D optimization: min_t max(a*t + b) over t in [0,1]
            na = len(active)
            if na == 1:
                # Single line: minimum at boundary
                if a[0] > 0:
                    best_t = 0.0
                elif a[0] < 0:
                    best_t = 1.0
                else:
                    best_t = h[i]
            else:
                # Find all intersection points in [0,1]
                A_mat = a[np.newaxis, :] - a[:, np.newaxis]
                B_mat = b[np.newaxis, :] - b[:, np.newaxis]
                mask = np.abs(A_mat) > 1e-14
                T_mat = np.where(mask, B_mat / np.where(mask, A_mat, 1.0), 999.0)
                T_valid = T_mat[mask]
                T_valid = T_valid[(T_valid >= 0.0) & (T_valid <= 1.0)]
                candidates = np.unique(np.concatenate(([0.0, 1.0], T_valid)))
                
                # Evaluate
                vals = np.max(a[np.newaxis, :] * candidates[:, np.newaxis] + b[np.newaxis, :], axis=1)
                best_t = candidates[np.argmin(vals)]
            
            h[i] = best_t
        
        h = _project_sum(h, n)
        c5 = _c5_max(h)
        if c5 < best_c5:
            best_c5 = c5
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 200:
                break
    
    return h, best_c5

def _local_search(h, n, time_limit):
    """Gradient-informed local search."""
    dx = 2.0 / n
    h = h.copy()
    best_c5 = _c5_max(h)
    t_start = time.time()
    rng = np.random.RandomState(42)
    
    while time.time() - t_start < time_limit:
        C = _c5_array(h)
        # Find the max lag
        k_max = np.argmax(C)
        s_max = k_max - (n - 1)
        
        # Gradient of C[k_max]
        j = np.arange(n)
        jp = j + s_max
        jm = j - s_max
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        grad = np.zeros(n)
        grad[vp] = dx * (1.0 - h[jp[vp]])
        grad[vm] -= dx * h[jm[vm]]
        
        # Find coordinate with most negative gradient (best to increase)
        # and most positive gradient (best to decrease)
        i_dec = np.argmax(grad)  # decreasing this helps most
        i_inc = np.argmin(grad)  # increasing this helps most
        
        if i_dec == i_inc:
            # Try random move
            i_dec = rng.randint(n)
            i_inc = rng.randint(n)
            if i_dec == i_inc:
                continue
        
        # Try moving mass from i_dec to i_inc
        delta = min(h[i_dec], 1.0 - h[i_inc]) * 0.1
        if delta < 1e-6:

        h2 = h.copy()
        h2[i_dec] -= delta
        h2[i_inc] += delta
        c52 = _c5_max(h2)
        if c52 < best_c5 - 1e-12:
            h = h2
            best_c5 = c52
    
    return h, best_c5
```

```python
from scipy.optimize import minimize

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _project_sum(h, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _softmax_fg(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    m = np.max(C)
    at = alpha * (C - m)
    at = np.minimum(at, 500.0)
    es = np.exp(at)
    se = np.sum(es)
    f = m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    n_c = len(C)
    s_arr = np.arange(-(n - 1), n)
    j = np.arange(n)
    grad = np.zeros(n)
    for k in range(n_c):
        wk = w[k]
        if wk < 1e-12:
            continue
        sk = s_arr[k]
        jp = j + sk
        jm = j - sk
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        grad[vp] += wk * dx * (1.0 - h[jp[vp]])
        grad[vm] -= wk * dx * h[jm[vm]]
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgsb(h, alpha, beta, maxiter=400):
    n = len(h)
    res = minimize(_softmax_fg, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-14})

        for i in range(n):

            if top_frac < 1.0:

                thresh = np.sort(C_cur)[-n_act]

            else:

            na = len(active)

                ip = i + sk
```
