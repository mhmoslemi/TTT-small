    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h -= (-diff / total) * h
    return np.clip(h, 0.0, 1.0)

def topk_smooth_max(c, alpha, k=20):
    """Smooth max using only top k values."""
    m = c.max()
    # Get top k values
    if len(c) > k:
        top_k_idx = np.argpartition(-c, k)[:k]
        c_use = c[top_k_idx]
    else:
        c_use = c
    shifted = alpha * (c_use - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def c5_all(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h -= (-diff / total) * h
    return np.clip(h, 0.0, 1.0)

def topk_smooth_max(c, alpha, k):
    m = c.max()
    if len(c) > k:
        idx = np.argpartition(-c, min(k, len(c)) - 1)[:k]
        cu = c[idx]
    else:
        cu = c
    shifted = alpha * (cu - m)
    mx = shifted.max()
    if mx > 500:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

def obj_topk(h, n, alpha, k):
    c = c5_all(h, n)
    return topk_smooth_max(c, alpha, k)

def obj_full(h, n, alpha):
    c = c5_all(h, n)
    return topk_smooth_max(c, alpha, 2 * n - 1)

def make_sum_con(n):
    def fun(x):
        return np.sum(x) - n / 2.0
    def jac(x):
        return np.ones(n)
    return [{'type': 'eq', 'fun': fun, 'jac': jac}]

def slsqp_refine(h, n, alpha, k, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = make_sum_con(n)
    res = minimize(obj_topk, h.copy(), args=(n, alpha, k),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, n)

def nelder_mead_free(h, n, maxiter):
    mask = (h > 0.02) & (h < 0.98)
    free_idx = np.where(mask)[0]
    if len(free_idx) < 3:
        return h.copy()
    x0 = h[free_idx].copy()
    n_free = len(free_idx)
    
    def nm_obj(x):
        hf = h.copy()
        hf[free_idx] = x
        hf = project_feasible(hf, n)
        c = c5_all(hf, n)
        return topk_smooth_max(c, 5000, 15)
    
    res = minimize(nm_obj, x0, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12})
    h_new = h.copy()
    h_new[free_idx] = res.x
    return project_feasible(h_new, n)

def pair_swap_search(h, n, rng, n_trials):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    deltas = [0.001, 0.002, 0.005, 0.01, 0.003]
    count = 0
    while count < n_trials:
        count += 1
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for d in deltas:
            for s in [1, -1]:
                hi = best_h[i] + s * d
                hj = best_h[j] - s * d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h_try = best_h.copy()
                    h_try[i] = hi
                    h_try[j] = hj
                    h_try = project_feasible(h_try, n)
                    c = compute_c5(h_try, n)
                    if c < best_c5:
                        best_h = h_try
                        best_c5 = c
                        break
    return best_h, best_c5

def coord_descent(h, n, epochs):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    for ep in range(epochs):
        improved = False
        order = np.random.permutation(n)
        for i in order:
            old_val = best_h[i]
            
            def coord_obj(v):
                hf = best_h.copy()
                hf[i] = v
                hf = project_feasible(hf, n)
                return compute_c5(hf, n)
            
            res = minimize_scalar(coord_obj, bounds=(0.0, 1.0), method='bounded',
                                  options={'maxiter': 30, 'xatol': 1e-8})
            if res.fun < best_c5 - 1e-12:
                best_h[i] = res.x
                best_h = project_feasible(best_h, n)
                best_c5 = res.fun
                improved = True
    return best_h, best_c5
