def _fix(h, n):
    """Repair h: clip to [0,1] and adjust sum to n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)

    #   - When i appears as j+s, i.e., j=i-s: h[i-s] * (-1) * [0<=i-s<n] ...

For a binary sequence b of length n (circular, for simplicity), the autocorrelation at lag s is:
R(s) = sum_i b[i] * b[i+s] (circular)

This is related to the Fourier transform: R(s) is the inverse DFT of |B[k]|².

We want to minimize max_s (n/2 - R(s)) * (2/n) = 2/n * max_s (n/2 - R(s))

Equivalently, maximize min_s R(s).

For a "random" binary sequence, R(s) ≈ n/4 for s ≠ 0, and R(0) = n/2.
So max_s (n/2 - R(s)) ≈ n/2 - n/4 = n/4 (at s ≠ 0).
And C₅ ≈ 2/n * n/4 = 1/2.

We need C₅ < 0.38, so we need max_s (n/2 - R(s)) < 0.38 * n/2 = 0.19n.
This means min_s R(s) > n/2 - 0.19n = 0.31n.

For a "perfect" sequence, R(s) would be as close to n/4 as possible for all s ≠ 0 (since the average of R(s) over s is n/4 by Parseval).

Wait, the average of R(s) over all s is:
(1/n) * sum_s R(s) = (1/n) * R(0) * n = R(0) = n/2 (circular case)

No wait. sum_s R(s) = sum_s sum_i b[i]b[i+s] = sum_i b[i] * sum_s b[i+s] = sum_i b[i] * (n/2) = (n/2) * (n/2) = n²/4.
So average R(s) = n/4.

But R(0) = n/2 (sum of b[i]² = sum b[i] = n/2).
So the average over s ≠ 0 is: (n²/4 - n/2) / (n-1) ≈ n/4 for large n.

So we want R(s) to be as close to n/4 as possible for all s ≠ 0. The deviation from n/4 is what we want to minimize.

import time

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(1.0 - h)
            for i in idx:

                add = min(room, d)
                h[i] += add
                d -= add

            for i in idx[::-1]:
                take = min(h[i], -d)
                h[i] -= take
                d += take
                if d >= -1e-12:
                    break
    return h

def _soft_max_obj(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sc = alpha * C
    m = np.max(sc)
    ec = np.exp(sc - m)
    se = np.sum(ec)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5*n)**2
    w = ec / se
    grad = np.zeros(n)
    s_arr = np.arange(nc) - (n - 1)
    for i in range(n):
        vi = (i + s_arr >= 0) & (i + s_arr < n)
        vj = (i - s_arr >= 0) & (i - s_arr < n)
        grad[i] = dx * (np.dot(w[vi], 1.0 - h[i + s_arr[vi]]) - np.dot(w[vj], h[i - s_arr[vj]]))
    grad += 2.0 * beta * (np.sum(h) - 0.5*n)
    return f, grad

def _grad_optimize(h, n, alpha, beta, maxiter=500):
    res = minimize(_soft_max_obj, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)]*n,
                  options={'maxiter': maxiter, 'maxfun': maxiter*10, 'ftol': 1e-14})
    if res.x is not None:
        return _fix(res.x, n)
    return h

def _binarize_even(h, n):

    # Try different thresholds
    best_b = None
    best_c5 = np.inf
    sorted_h = np.sort(h)

    candidates = [sorted_h[n//2 - 1] + 1e-10, sorted_h[n//2] - 1e-10,
                  (sorted_h[n//2-1] + sorted_h[n//2]) / 2]

            b = np.zeros(n)
            b[idx[n//2:]] = 1.0
        c5 = _c5_max(b)
        if c5 < best_c5:
            best_c5 = c5
            best_b = b.copy()
    return best_b, best_c5

def _sa_binary(n, init_b, t_start=1.0, t_end=0.001, n_iter=50000):
    """Simulated annealing on binary sequences of length n with n/2 ones."""
    dx = 2.0 / n
    b = init_b.copy()
    best_b = b.copy()
    best_c5 = _c5_max(b)
    cur_c5 = best_c5
    
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    rng = np.random.RandomState(42)
    
    for it in range(n_iter):
        t = t_start * (t_end / t_start) ** (it / n_iter)
        
        # Pick a random 1 and a random 0, swap them
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        new_c5 = _c5_max(b)

                best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]  # revert
    
    return best_b, best_c5

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-13:
        if d > 0:
            idx = np.argsort(1.0 - h)
            for i in idx:
                room = 1.0 - h[i]
                add = min(room, d)
                h[i] += add
                d -= add
                if d <= 1e-13:
                    break
        else:
            idx = np.argsort(h)
            for i in idx[::-1]:
                take = min(h[i], -d)
                h[i] -= take
                d += take
                if d >= -1e-13:
                    break
    return h

def _obj(h, alpha, beta):
    n = len(h)
    C = _c5_array(h)
    sc = alpha * C
    m = np.max(sc)
    ec = np.exp(sc - m)
    se = np.sum(ec)
    return (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5*n)**2

def _obj_grad(h, alpha, beta):
    f = _obj(h, alpha, beta)
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sc = alpha * C
    m = np.max(sc)
    w = np.exp(sc - m) / np.sum(np.exp(sc - m))
    s_arr = np.arange(nc) - (n - 1)
    grad = np.zeros(n)
    for i in range(n):
        vi = (i + s_arr >= 0) & (i + s_arr < n)
        vj = (i - s_arr >= 0) & (i - s_arr < n)
        grad[i] = dx * (np.dot(w[vi], 1.0 - h[i + s_arr[vi]]) - np.dot(w[vj], h[i - s_arr[vj]]))
    grad += 2.0 * beta * (np.sum(h) - 0.5*n)
    return f, grad

                  method='L-BFGS-B', bounds=[(0.0, 1.0)]*n,
                  options={'maxiter': maxiter, 'maxfun': maxiter*20, 'ftol': 1e-15})
