            h[i] = 0.0; s -= 1
    elif s < target:
        idx = np.where(h == 0.0)[0].copy()
        rng.shuffle(idx)
        for i in idx:
            if s == target: break
            h[i] = 1.0; s += 1
    return h

def optimize(h0, n, dur, seed=42):
    """Optimize binary h via swap-based local search."""
    rng = np.random.RandomState(seed)
    h = h0.copy()
    t0 = time.time()
    c5 = compute_c5(h, n)
    best_c5 = c5
    best_h = h.copy()
    
    while time.time() - t0 < dur:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        # Single swap: best improvement
        best_i, best_j, best_val = -1, -1, c5
        for i in ones:
            if time.time() - t0 >= dur: break
            for j in zeros:
                h[i] = 0.0; h[j] = 1.0
                v = compute_c5(h, n)
                h[i] = 1.0; h[j] = 0.0
                if v < best_val - 1e-14:
                    best_val = v; best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] = 0.0; h[best_j] = 1.0
            c5 = best_val; improved = True
            if c5 < best_c5:
                best_c5 = c5; best_h = h.copy()
            continue
        
        # Double swap: random sampling
        n_try = max(50, int(2 * (dur - (time.time() - t0)) * 10))
        n_try = min(n_try, 5000)
        for _ in range(n_try):
            if time.time() - t0 >= dur: break
            if len(ones) < 2 or len(zeros) < 2: break
            i1, i2 = rng.choice(ones, 2, replace=False)
            j1, j2 = rng.choice(zeros, 2, replace=False)
            h[i1] = 0.0; h[i2] = 0.0; h[j1] = 1.0; h[j2] = 1.0
            v = compute_c5(h, n)
