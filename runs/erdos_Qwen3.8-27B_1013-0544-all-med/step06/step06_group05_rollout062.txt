def project_bounded_simplex(h, target):

    n = len(h)

        else:

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

                lr *= 0.5
            else:
                stadium += 1
                if stadium > 40:
                    break
            if stadium > 0:
                h = project_bh(bh + rng.normal(0, 0.05, n), tgt)
                lr = 0.3
                stadium = 0
    return bh, bc

def interp_h(h0, n_new):
    n0 = len(h0)
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return project_bh(np.interp(x1, x0, h0), n_new / 2.0)

def binary_ls(h0, t_budget, seed=42):
    n = len(h0)
    h = np.clip(np.round(h0), 0, 1).astype(float)
    tgt = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > tgt:
        for i in ones[tgt:]:
            h[i] = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    elif len(ones) < tgt:
        for i in zeros[:tgt - len(ones)]:
            h[i] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    bc = compute_c5(h)
    bh = h.copy()
    rng = np.random.RandomState(seed)
    t0 = time.time()
    while time.time() - t0 < t_budget and len(ones) > 0 and len(zeros) > 0:
        improved = False
        n_trials = min(2000, max(200, int(t_budget * 20)))
        for _ in range(n_trials):
            if time.time() - t0 > t_budget:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c2 = compute_c5(h)
            if c2 < bc - 1e-14:
                improved = True
                bc = c2
                bh = h.copy()
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            break
    return bh, bc

    t0 = time.time()
    np.random.seed(seed)

        return budget_s - (time.time() - t0)

    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Smooth optimization at native resolution
    t1 = min(250.0, rem() * 0.25)
    if t1 > 10:
        h1, c5_1 = solve_smooth(h_init, t1, seed)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1

    if rem() > 40:
        t2 = min(80.0, rem() * 0.12)
        h2, c5_2 = binary_ls(best_h, t2, seed + 100)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    for mult in [2, 3]:
        if rem() < 120:
            break
