```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def compute_grad_soft(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)

        if w > 1e-12:
```

```python
def compute_grad_soft(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    
    # Only process shifts with significant weight
    significant = np.where(weights > 1e-10)[0]
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for idx in significant:
        w = weights[idx]
        s = idx - (n - 1)
        if s >= 0:
            grad[:n-s] += w * one_minus_h[s:]
            grad[s:] -= w * h[:n-s]
        else:
            ss = -s
            grad[ss:] += w * one_minus_h[:n-ss]
            grad[:n-ss] -= w * h[ss:]
    return grad * dx
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def compute_grad_soft(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    significant = np.where(weights > 1e-10)[0]
    for idx in significant:
        w = weights[idx]
        s = idx - (n - 1)
        if s >= 0:
            grad[:n-s] += w * one_minus_h[s:]
            grad[s:] -= w * h[:n-s]
        else:
            ss = -s
            grad[ss:] += w * one_minus_h[:n-ss]
            grad[:n-ss] -= w * h[ss:]
    return grad * dx

def compute_grad_hard(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    idx = int(np.argmax(conv))
    s = idx - (n - 1)
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    if s >= 0:
        grad[:n-s] += one_minus_h[s:]
        grad[s:] -= h[:n-s]
    else:
        ss = -s
        grad[ss:] += one_minus_h[:n-ss]
        grad[:n-ss] -= h[ss:]
    return grad * dx

def compute_grad_topk(h, k=30, t_k=0.005):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    top_idx = np.argsort(conv)[::-1][:k]
    max_val = conv[top_idx[0]]
    exps = np.exp((conv[top_idx] - max_val) / t_k)
    weights = exps / np.sum(exps)
    one_minus_h = 1.0 - h
    grad = np.zeros(n)
    for j in range(len(top_idx)):
        idx = top_idx[j]
        w = weights[j]
        s = idx - (n - 1)
        if s >= 0:
            grad[:n-s] += w * one_minus_h[s:]
            grad[s:] -= w * h[:n-s]
        else:
            ss = -s
            grad[ss:] += w * one_minus_h[:n-ss]
            grad[:n-ss] -= w * h[ss:]
    return grad * dx

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    s = h.sum()
    if abs(s - target) < 1e-10:
        return h
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def adam_optimize(h_init, max_iter, t_schedule, lr=0.03, beta1=0.9, beta2=0.999,
                  eps=1e-8, patience=400, rng=None, use_topk=False):
    n = len(h_init)
    h = h_init.copy()
    m_vec = np.zeros(n)
    v_vec = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stagnation = 0
    for it in range(max_iter):
        frac = (it + 1) / max_iter
        if len(t_schedule) == 1:
            t = t_schedule[0]
        else:
            seg = min(int(frac * (len(t_schedule) - 1)), len(t_schedule) - 2)
            sf = frac * (len(t_schedule) - 1) - seg
            t = t_schedule[seg] * (1 - sf) + t_schedule[seg + 1] * sf
        if t < 1e-6:
            if use_topk:
                grad = compute_grad_topk(h, k=30)
            else:
                grad = compute_grad_hard(h)
        else:
            if use_topk and t < 0.01:
                grad = compute_grad_topk(h, k=30, t_k=t)
            else:
                grad = compute_grad_soft(h, t)
        hard_c5 = compute_c5(h)
        if hard_c5 < best_c5 - 1e-12:
            best_c5 = hard_c5
            best_h = h.copy()
            stagnation = 0
        else:
            stagnation += 1
        m_vec = beta1 * m_vec + (1 - beta1) * grad
        v_vec = beta2 * v_vec + (1 - beta2) * grad * grad
        m_hat = m_vec / (1 - beta1 ** (it + 1))
        v_hat = v_vec / (1 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h - step, n)
        if stagnation > patience and rng is not None:
            scale = 0.03 * (stagnation / patience)
            h = project_feasible(best_h + rng.normal(0, scale, n), n)
            m_vec = np.zeros(n)
            v_vec = np.zeros(n)
            stagnation = 0
    return best_h, best_c5

def pgd_polish(h_init, max_iter, lr_init=0.001, t=0.0):
    n = len(h_init)
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        if t < 1e-6:
            grad = compute_grad_hard(h)
        else:
            grad = compute_grad_soft(h, t)
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-14:
            lr *= 0.5
            if lr < 1e-16:
                break
        else:
            h = h_new
            if it % 200 == 0:
                lr = min(lr * 1.04, 0.02)
    return best_h, best_c5

def make_sine_init(n, freq, phase):
    x = np.linspace(0, 2.0, n, endpoint=False)
    h = 0.5 + 0.45 * np.cos(np.pi * freq * x + phase)
    return project_feasible(h, n)

def make_triangle_init(n, freq):
    x = np.linspace(0, 2.0, n, endpoint=False)
    t = (freq * x / 2.0) % 1.0
    h = 1.0 - np.abs(2.0 * t - 1.0)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return max(0, budget_s - (time.time() - t0))

    if 'initial_h_values' in globals() and initial_h_values is not None:
```
