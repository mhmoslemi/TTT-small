    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, h)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Quick greedy on initial (binary)
    h_bin0 = make_binary(h_init, n_init)
    h1, c5_1 = greedy_ls(h_bin0, max_time=min(15.0, rem() * 0.05))
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1

    # Phase 2: ILS from the best binary (main workhorse)
    if rem() > 60:
        t_ils = min(rem() * 0.35, 250.0)
        h2, c5_2 = ils(h1, max_time=t_ils, seed=seed)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2

    # Phase 3: Try golden ratio and random starts at n_init
    for start_type in ['golden', 'random']:
        if rem() < 45:
            break
        if start_type == 'golden':
            h_start = golden_ratio_h(n_init)
        else:
            h_start = random_h(n_init, seed=seed + 7)
        h_b = make_binary(h_start, n_init)
        t_ils = min(40.0, rem() * 0.25)
        if t_ils < 5:
            break
        h3, c5_3 = ils(h_b, max_time=t_ils, seed=seed + 100)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3

    # Phase 4: Try different resolutions
    n_trials = [96, 84, 100, 128, 140, 168, 192]
    for n_t in n_trials:
        if rem() < 50:
            break
        h_interp = interpolate_to_n(best_h, n_t)
        h_b = make_binary(h_interp, n_t)
        t_g = min(12.0, rem() * 0.3)
        if t_g < 3:
            break
        h_g, c5_g = greedy_ls(h_b, max_time=t_g)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g
        if rem() > 40:
            h_i, c5_i = ils(h_b, max_time=min(25.0, rem() * 0.3), seed=seed + n_t)
            if c5_i < best_c5:
                best_c5 = c5_i
                best_h = h_i

    # Phase 5: Continuous refinement
    if rem() > 20:
        h_c, c5_c = continuous_refine(best_h, max_time=min(40.0, rem() * 0.3), seed=seed + 500)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c

    # Phase 6: Final long ILS on best binary
    if rem() > 30:
        n_f = len(best_h)
        h_f = make_binary(best_h, n_f)
        t_final = rem() * 0.8
        h_f2, c5_f = ils(h_f, max_time=t_final, seed=seed + 999)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f2

    # Final evaluation
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
