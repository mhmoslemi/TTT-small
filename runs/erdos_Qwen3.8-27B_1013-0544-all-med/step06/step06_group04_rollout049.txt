        h280 = upsample(best_h, 280)
        h280, c5 = pgm_optimize(h280, 400000, t_rem, start_time, step=0.5)
        if c5 < best_c5:
            best_h, best_c5 = h280, c5
        hr, c5r = try_rounding(best_h)
        if c5r < best_c5:
            best_h, best_c5 = hr, c5r
    
    # Final polish at best resolution if time remains
    t_rem = budget_s * 0.98 - (time.time() - start_time)
    if t_rem > 5 and len(best_h) > 70:
        h, c5 = pgm_optimize(best_h, 100000, t_rem, start_time, step=0.3)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Verify
    n_f = len(best_h)
    dx = 2.0 / n_f
    c = np.correlate(best_h, 1.0 - best_h, mode='full') * dx
    best_c5 = float(np.max(c))
    return (best_h, best_c5, n_f)
