            break
        n_try = N_best if attempt % 3 == 0 else [40, 48, 56, 64, 72, 80][attempt % 6]
        h_start = structured_init(n_try, rng, pattern_type=attempt % 5)
        t_phase = min(time.time() + 40, t_end - 15)
        h, v = optimize(h_start, n_try, rng, t_phase, try_update, max_iter=25000, top_k=4, alpha=20.0)
        try_update(h, v)

    # Phase 4: Fine-tuning with different alpha values
    if best_h is not None and time.time() < t_end - 20:
        N_bt = len(best_h)
        for alpha_try in [15.0, 30.0, 50.0]:
            if time.time() > t_end - 10:
                break
            h_start = best_h.copy()
            t_phase = min(time.time() + 30, t_end - 5)
            h, v = optimize(h_start, N_bt, rng, t_phase, try_update, max_iter=30000, top_k=3, alpha=alpha_try)
            try_update(h, v)

    # Phase 5: Small perturbations around best
    if best_h is not None and time.time() < t_end:
        N_bt = len(best_h)
        for pert in range(20):
            if time.time() > t_end:
                break
            scale = 0.02 + 0.03 * (pert % 4)
            h_start = project_feasible(best_h + rng.randn(N_bt) * scale, N_bt)
            t_phase = min(time.time() + 20, t_end)
            h, v = optimize(h_start, N_bt, rng, t_phase, try_update, max_iter=20000, top_k=3, alpha=30.0)
            try_update(h, v)

    if best_h is None:
        best_h = 0.5 * np.ones(48)
        best_val = obj_and_corr(best_h)[0]

    return best_h, best_val, len(best_h)
