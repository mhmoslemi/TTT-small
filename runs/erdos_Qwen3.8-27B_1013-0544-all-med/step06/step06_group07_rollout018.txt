    rng = np.random.RandomState(seed)
    bc = exact_c(x)
    for _ in range(max_iter):
        cv = c_vals(x)
        C = cv.max()
        if C < bc:
            bc = C
        kw = int(cv.argmax())
        sh = kw - (n - 1)
        cand1 = [i for i in range(n) if 0 <= i + sh < n and x[i] > 0.4 and x[i + sh] < 0.6]
        cand0 = [i for i in range(n) if 0 <= i + sh < n and x[i] < 0.6 and x[i + sh] > 0.4]
        if not cand1 or not cand0:
            ones = np.where(x > 0.5)[0]
            zeros = np.where(x < 0.5)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            i1 = int(rng.choice(ones))
            i0 = int(rng.choice(zeros))
        else:
            i1 = int(rng.choice(cand1))
            i0 = int(rng.choice(cand0))
        if i1 == i0:
            continue
        x[i1], x[i0] = x[i0], x[i1]
        if exact_c(x) > bc + 1e-6:
            x[i1], x[i0] = x[i0], x[i1]
    return x, bc

def exact_polish(x0, tl, seed):
    x = repair(x0)
    bx = x.copy()
    bc = exact_c(x)
    rng = np.random.RandomState(seed)
    dl = time.time() + max(0.5, tl)
    step = 0.12
    ni = 0
    while time.time() < dl:
        cv = c_vals(x)
        C = cv.max()
        if C < bc - 1e-15:
            bc = C
            bx = x.copy()
            ni = 0
        else:
            ni += 1
        if ni > 30:
            imp = False
            for sg in (0.05, 0.02, 0.008, 0.003, 0.001):
                y = repair(x + rng.normal(0, sg, len(x)))
                cy = exact_c(y)
                if cy < bc:
                    bc = cy
                    bx = y.copy()
                    x = y
                    ni = 0
                    imp = True
                    break
            if not imp:
                break
            continue
        n_act = min(8, max(3, int((cv >= C - 1e-6).sum())))
        act = np.argsort(cv)[::-1][:n_act]
        g = agrad(x, act)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        acc = False
        for _ in range(30):
            y = repair(x - step * g)
            cy = exact_c(y)
            if cy < C - 1e-13:
                x = y
                acc = True
                break
            step *= 0.5
            if step < 1e-12:
                break
        if acc:
            ni = 0
        else:
            ni += 1
            step = min(step * 1.15, 0.35)
        if rng.rand() < 0.1:
            y = repair(x + rng.normal(0, 0.003, len(x)))
            if exact_c(y) < bc:
                bc = exact_c(y)
                bx = y.copy()
                x = y
                ni = 0
    return bx, bc

def opt_start(x0, tl, seed):
    x = repair(x0)
    bx = x.copy()
    bc = exact_c(x)
    st = time.time()
    n = len(x)
    sc = 1.0 if n <= 80 else (0.85 if n <= 120 else 0.65)
    stages = [
        (0.06, int(100 * sc)),
        (0.025, int(180 * sc)),
        (0.01, int(220 * sc)),
        (0.004, int(220 * sc)),
        (0.0015, int(200 * sc)),
        (0.0005, int(150 * sc)),
    ]
    for tau, mi in stages:
        if time.time() - st > tl * 0.68:
            break
        set_opt(tau, None)
        r = slsqp(s_fun, s_jac, x, mi)
        if r is not None:
            xr = repair(r)
            c = exact_c(xr)
            if c < bc:
                bc = c
                bx = xr.copy()
            x = xr if c <= bc + 5e-4 else bx
    if time.time() - st < tl * 0.8:
        set_opt(0.002, 4)
        r = slsqp(s_fun, s_jac, bx, int(60 * sc))
        if r is not None:
            xr = repair(r)
            c = exact_c(xr)
            if c < bc:
                bc = c
                bx = xr.copy()
    rem = tl - (time.time() - st)
    if rem > 2.0:
        xp, cp = exact_polish(bx, min(15.0, max(1.0, rem * 0.6)), seed)
        if cp < bc:
            bc = cp
            bx = xp
    return bx, bc

def run(seed=42, budget_s=1100, **kwargs):
    dl = time.time() + max(1.0, float(budget_s))
    sv = int(seed)
    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)
    bn = int(init.size) if init is not None and init.size >= 10 else 70

    ns = []
    for n in (bn, bn + 5, bn - 5, bn + 10, 80, 90, 100, 110, 120):
        n = int(n)
        if 30 <= n <= 130 and n not in ns:
            ns.append(n)
    ns = ns[:5]

    bx, bc = None, 1e9
    for ni, n in enumerate(ns):
        if time_left(dl) < 55:
            break
        per = min([220, 130, 90, 70, 60][ni] if ni < 5 else 60, time_left(dl) * 0.38)
        per = max(15.0, per)
        starts = []
        if init is not None and len(init) > 0:
            starts.append(resample(init, n))
        rng1 = np.random.RandomState(sv + n + 1)
        starts.append(repair(rnd_bin(n, sv + n) + 0.12 * rng1.randn(n) * 0.5))
        rng2 = np.random.RandomState(sv + n + 501)
        starts.append(repair(rnd_bin(n, sv + n + 500) + 0.08 * rng2.randn(n) * 0.5))
        if init is not None and len(init) > 0:
            b = resample(init, n)
            rng3 = np.random.RandomState(sv + n + 900)
            starts.append(repair(b + 0.06 * rng3.randn(n)))
        mx = 5 if ni == 0 else 3
        for ii in range(min(mx, len(starts))):
            if time_left(dl) < 18:
                break
            x, c = opt_start(starts[ii], per, sv + 7 * n + ii * 13)
            if c < bc:
                bc = c
                bx = x

    if bx is None:
        bx = repair(np.full(50, 0.5))
        bc = exact_c(bx)
    bx = repair(bx)
    bc = exact_c(bx)
    return bx, bc, len(bx)
