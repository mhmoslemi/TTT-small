```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-12:
            idx = np.argmin(h) if d2 > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + d2, 0.0, 1.0)
    return h

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgs(h, n, alpha, beta, maxiter=300):
    res = minimize(_obj_grad, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-14})
    return _fix(res.x if res.x is not None else h, n)
```

```python
def _coord_optimal(h, i, n):
    """Find optimal value for coordinate i given all others fixed."""
    # Save current value
    old = h[i]
    # Set to 0 to compute b_k (constant part)
    h[i] = 0.0
    C0 = _c5_array(h)  # C[k] when h[i] = 0
    # Set to 1 to compute a_k (coefficient)
    h[i] = 1.0
    C1 = _c5_array(h)  # C[k] when h[i] = 1
    h[i] = old  # Restore
    
    a = C1 - C0  # Coefficient of h[i] in C[k]
    b = C0       # Constant part
    
    # Find minimizer of max_k (a[k]*v + b[k]) over v in [0,1]
    # This is the 1D "minimax of linear functions" problem
    # The optimal v is where the upper envelope has its minimum
    
    # Try: binary search on the value, or check all breakpoints
    # For efficiency, just use a fine grid + refinement
    nc = len(C0)
    
    # Find all breakpoints
    breakpoints = [0.0, 1.0]
    for k1 in range(nc):
        for k2 in range(k1+1, nc):
            if abs(a[k1] - a[k2]) > 1e-15:
                v = (b[k2] - b[k1]) / (a[k1] - a[k2])
                if 0.0 < v < 1.0:
                    breakpoints.append(v)

    # Evaluate at all breakpoints
    best_v = old
    best_c5 = np.inf
    for v in breakpoints:
        C = a * v + b
        c5 = np.max(C)
        if c5 < best_c5:
            best_c5 = c5
            best_v = v
    
    return best_v, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-12:
            idx = np.argmin(h) if d2 > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + d2, 0.0, 1.0)
    return h

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgs(h, n, alpha, beta, maxiter=300):
    res = minimize(_obj_grad, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15})
    return _fix(res.x if res.x is not None else h, n)

def _coord_descent(h, n, sweeps=50, n_grid=100, t0=None, time_limit=None):
    """Coordinate descent exploiting linearity in each coordinate."""
    h = h.copy()
    dx = 2.0 / n
    nc = 2 * n - 1
    m_arr = np.arange(nc)
    C = _c5_array(h)
    best_c5 = float(np.max(C))
    
    for sweep in range(sweeps):
        if t0 is not None and time_limit is not None and (time.time() - t0) > time_limit:
            break
        improved = False
        for i in range(n):
            # Compute a[m] and b[m] such that C[m] = a[m]*h[i] + b[m]
            valid1 = (m_arr <= i)
            valid2 = (m_arr <= n - 1 - i)
            
            idx1 = i - m_arr[valid1]
            idx2 = i + m_arr[valid2]
            
            a = np.zeros(nc)
            a[valid1] = dx * (1.0 - h[idx1])
            a[valid2] -= dx * h[idx2]
            
            b = C - h[i] * a
            
            # Minimize max(b + v*a) over v in [0,1]
            grid = np.linspace(0.0, 1.0, n_grid)
            best_val = h[i]
            best_c5_i = best_c5
            
            vals = b + grid[:, None] * a[None, :]
            c5_vals = np.max(vals, axis=1)
            
            # Find best grid point
            best_idx = np.argmin(c5_vals)
            if c5_vals[best_idx] < best_c5_i - 1e-13:
                best_val = grid[best_idx]
                best_c5_i = c5_vals[best_idx]
            
            # Refine: check if we can do better with a finer local grid
            if 0 < best_idx < n_grid - 1:
                lo = grid[best_idx - 1]
                hi = grid[best_idx + 1]
                fine_grid = np.linspace(lo, hi, 20)
                vals_f = b + fine_grid[:, None] * a[None, :]
                c5_f = np.max(vals_f, axis=1)
                fi = np.argmin(c5_f)
                if c5_f[fi] < best_c5_i - 1e-14:
                    best_val = fine_grid[fi]
                    best_c5_i = c5_f[fi]
            
            if best_val != h[i]:
                h[i] = best_val
                C = b + best_val * a
                best_c5 = best_c5_i
                improved = True
        
        if not improved:
            break
    
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-12:
            idx = np.argmin(h) if d2 > 0 else np.argmax(h)
            h[idx] = np.clip(h[idx] + d2, 0.0, 1.0)
    return h

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgs(h, n, alpha, beta, maxiter=300):
    res = minimize(_obj_grad, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15})
    return _fix(res.x if res.x is not None else h, n)

def _coord_descent(h, n, sweeps=50, n_grid=200, t0=None, time_limit=None):
    h = h.copy()
    dx = 2.0 / n
    nc = 2 * n - 1
    m_arr = np.arange(nc)
    C = _c5_array(h)
    best_c5 = float(np.max(C))
    for sweep in range(sweeps):
        if t0 is not None and time_limit is not None and (time.time() - t0) > time_limit:
            break
        improved = False
        for i in range(n):
            v1 = (m_arr >= i) & (m_arr <= i + n - 1)
            v2 = (m_arr >= n - 1 - i) & (m_arr <= 2 * n - 2 - i)
            k1 = i - m_arr[v1] + n - 1
            k2p = i + m_arr[v2] - n + 1
            a = np.zeros(nc)
            a[v1] = dx * (1.0 - h[k1])
            a[v2] -= dx * h[k2p]
            b = C - h[i] * a
            grid = np.linspace(0.0, 1.0, n_grid)
            vals = b + grid[:, None] * a[None, :]
            c5_vals = np.max(vals, axis=1)
            best_idx = int(np.argmin(c5_vals))
            if c5_vals[best_idx] < best_c5 - 1e-12:
                best_val = grid[best_idx]
                if 0 < best_idx < n_grid - 1:
                    lo = grid[best_idx - 1]
                    hi = grid[best_idx + 1]
                    fg = np.linspace(lo, hi, 21)
                    fv = b + fg[:, None] * a[None, :]
                    fc = np.max(fv, axis=1)
                    fi = int(np.argmin(fc))
                    if fc[fi] < c5_vals[best_idx]:
                        best_val = fg[fi]
                        c5_vals[best_idx] = fc[fi]
                h[i] = best_val
                C = b + best_val * a
                best_c5 = c5_vals[best_idx]
                improved = True
        if not improved:
            break
    return h, best_c5

def _pair_shuffle(h, n, rng, n_trials=2000, t0=None, time_limit=None):
    h = h.copy()
    best_c5 = _c5_max(h)
    for _ in range(n_trials):
        if t0 is not None and time_limit is not None and (time.time() - t0) > time_limit:
            break
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        delta = (rng.rand() - 0.5) * 0.4
        ht = h.copy()
        ht[i] = np.clip(h[i] + delta, 0, 1)
        ht[j] = np.clip(h[j] - delta, 0, 1)
        c5 = _c5_max(ht)
        if c5 < best_c5 - 1e-10:
            h = ht
            best_c5 = c5
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    hb = _fix(initial_h_values.astype(float).copy(), n0)
    cb = _c5_max(hb)
    nb = n0

    # Phase 1: Heavy coordinate descent
    hb, cb = _coord_descent(hb, nb, sweeps=100, n_grid=200, t0=t0, time_limit=budget_s * 0.25)

    # Phase 2: L-BFGS-B + coord descent interleaved
    for alpha in [30, 80, 150, 300, 600]:
        if _el() > budget_s * 0.45:
            break
        beta = 0.05 * alpha
        hn = _lbfgs(hb, nb, alpha, beta, maxiter=300)
        cn = _c5_max(hn)
        if cn < cb - 1e-10:
            cb, hb = cn, hn
        hb, cb2 = _coord_descent(hb, nb, sweeps=40, n_grid=150, t0=t0, time_limit=budget_s * 0.48)
        if cb2 < cb:
            cb = cb2

    # Phase 3: Try other n values
    n_candidates = sorted(set([nb - 5, nb + 5, nb + 10, nb + 15, nb + 25,
                               80, 90, 100, 110, 120, 130, 150, 180]))
    n_candidates = [x for x in n_candidates if 50 <= x <= 250 and x != nb]
    for n in n_candidates:
        if _el() > budget_s * 0.68:
            break
        x_old = np.linspace(0, 2, nb, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        ht = _fix(np.interp(x_new, x_old, hb, assume_sorted=True), n)
        ht, cn = _coord_descent(ht, n, sweeps=40, n_grid=100, t0=t0, time_limit=budget_s * 0.68)
        if cn < cb - 1e-10:
            cb, hb, nb = cn, ht, n
        else:
            # Try L-BFGS at this n too
            if _el() < budget_s * 0.68:
                hn2 = _lbfgs(ht, n, 100, 5.0, maxiter=200)
                cn2 = _c5_max(hn2)
                hn2, cn2b = _coord_descent(hn2, n, sweeps=20, n_grid=80, t0=t0, time_limit=budget_s * 0.68)
                if cn2b < cb - 1e-10:
                    cb, hb, nb = cn2b, hn2, n

    # Phase 4: Randomized restarts with coord descent
    for _ in range(200):
        if _el() > budget_s * 0.85:
            break
        scale = 0.02 + 0.06 * rng.rand()
        ht = _fix(hb + rng.randn(nb) * scale, nb)
        ht, cn = _coord_descent(ht, nb, sweeps=20, n_grid=80, t0=t0, time_limit=budget_s * 0.85)
        if cn < cb - 1e-10:
            cb, hb = cn, ht

    # Phase 5: Pair shuffle to escape local optima
    if _el() < budget_s * 0.92:
        hb, cb2 = _pair_shuffle(hb, nb, rng, n_trials=3000, t0=t0, time_limit=budget_s * 0.92)
        if cb2 < cb:
            cb = cb2
        hb, cb3 = _coord_descent(hb, nb, sweeps=30, n_grid=150, t0=t0, time_limit=budget_s * 0.95)
        if cb3 < cb:
            cb = cb3

    # Phase 6: Final high-alpha polish
    for alpha in [500, 1000, 2000]:
        if _el() > budget_s * 0.97:
            break
        beta = 0.05 * alpha
        hn = _lbfgs(hb, nb, alpha, beta, maxiter=500)
        cn = _c5_max(hn)
        if cn < cb - 1e-10:
            cb, hb = cn, hn

    if _el() < budget_s * 0.995:
        hb, cb2 = _coord_descent(hb, nb, sweeps=50, n_grid=200, t0=t0, time_limit=budget_s * 0.995)
        if cb2 < cb:
            cb = cb2

    hb = _fix(hb, nb)
    cb = _c5_max(hb)
    return (hb, cb, nb)
```
