                break
    return best_h, best_c5

def local_search(h, rng, n_tries, time_limit):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    for trial in range(n_tries):
        if time.time() - t_start > time_limit:
            break
        h_try = h.copy()
        n_perturb = rng.randint(1, 5)
        for _ in range(n_perturb):
            idx = rng.randint(n)
            h_try[idx] = rng.uniform(0.0, 1.0)
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            h = h_try
            best_c5 = c5_try
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return max(0.0, budget_s - (time.time() - t0))

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(320, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent at base resolution
    if tl() > 60:
        h_cd = coord_descent(best_h, n_sweeps=25, M=20, time_limit=tl()-50)
        h_cd = project_feasible(h_cd, n0)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 2: L-BFGS-B with soft-max
    for t_val in [0.03, 0.01, 0.003]:
        if tl() < 45:
            break
        n = len(best_h)
        target = n / 2.0
        try:
            res = minimize(soft_max_obj, best_h,
                           args=(t_val, n, target),
                           jac=soft_max_grad,
                           method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 2000, 'ftol': 1e-15, 'gtol': 1e-10})
            ht = project_feasible(res.x, n)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
        except Exception:
            pass

    # Phase 3: Multi-shift PGD
    if tl() > 50:
        ht, c5t = multi_shift_pgd(best_h, 100000, 0.003, tl()-40, K=3, temp=0.005)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Single-shift PGD (fine polish)
    if tl() > 40:
        ht, c5t = pgd_optimize(best_h, 150000, 0.001, tl()-30)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Higher resolution
    for n_try in [480, 640, 800]:
        if tl() < 200:
            break
        nc = len(best_h)
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)

        # Coordinate descent at higher res
        h_cd = coord_descent(hi, n_sweeps=15, M=15, time_limit=tl()-180)
        h_cd = project_feasible(h_cd, n_try)

        # Multi-shift PGD
        if tl() > 120:
            ht, c5t = multi_shift_pgd(h_cd, 50000, 0.001, tl()-100, K=3, temp=0.005)
        else:
            ht, c5t = h_cd, compute_c5(h_cd)

        ht = project_feasible(ht, n_try)
        c5t = compute_c5(ht)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Local search
    if tl() > 40:
        ht, c5t = local_search(best_h, rng, 2000, tl()-30)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 7: Final coordinate descent polish
    if tl() > 30:
        n = len(best_h)
        h_cd = coord_descent(best_h, n_sweeps=20, M=20, time_limit=tl()-20)
        h_cd = project_feasible(h_cd, n)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 8: Final PGD
    if tl() > 20:
        ht, c5t = pgd_optimize(best_h, 100000, 0.0005, tl()-10)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
