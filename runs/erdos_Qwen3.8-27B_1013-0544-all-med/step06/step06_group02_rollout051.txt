        g[idx] += (1.0 - h[idx + lag])

Wait, I need to double-check the `soft_gradient` function. The issue is that `grad_for_lag` computes the gradient of f_lag with respect to h, and we need to be careful about what exactly f_lag is.

The correlation output at index m (0 to 2N-2) is:
corr[m] = ∑_i h[i] * (1-h)[i + (m - (N-1))]

Let lag = m - (N-1). Then:
corr[m] = ∑_i h[i] * (1 - h[i+lag]) for valid i

The gradient with respect to h[j]:
- If j appears as h[i] (i.e., j = i for some valid i): contribution is (1 - h[j+lag])
- If j appears as h[i+lag] (i.e., j = i+lag, so i = j-lag): contribution is -h[j-lag]

So:
∂corr[m]/∂h[j] = (1 - h[j+lag]) * 1[j-lag valid and j valid as i] + (-h[j-lag]) * 1[j+lag valid and j-lag valid as i]

For lag >= 0:
- Valid i: 0 to N-1-lag
- j appears as h[i] when j in [0, N-1-lag]: contribution (1-h[j+lag])
- j appears as h[i+lag] when j-lag in [0, N-1-lag], i.e., j in [lag, N-1]: contribution -h[j-lag]

For lag < 0 (let l = -lag):
- corr[m] = ∑_i h[i] * (1-h[i-l]) for valid i, where i-l >= 0, so i in [l, N-1]
- j appears as h[i] when j in [l, N-1]: contribution (1-h[j-l])
- j appears as h[i-l] when j = i-l, i = j+l, valid when j+l in [l, N-1], i.e., j in [0, N-1-l]: contribution -h[j+l]

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_for_lag(h, lag, N):
    g = np.zeros(N)
    if lag >= 0:
        idx = np.arange(N - lag)
        g[idx] += (1.0 - h[idx + lag])
        g[idx + lag] -= h[idx]
    else:
        l = -lag
        idx = np.arange(l, N)
        g[idx] += (1.0 - h[idx - l])
        g[idx - l] -= h[idx]
    return g

def soft_gradient_full(h, corr, dx, alpha, top_k=10):
    N = len(h)
    c_max = corr.max()
    # Use top-k lags for efficiency
    top_idx = np.argsort(corr)[::-1][:top_k]
    w = np.zeros(2 * N - 1)
    for idx in top_idx:
        w[idx] = np.exp(alpha * (corr[idx] - c_max))
    wsum = w.sum()
    if wsum < 1e-300:
        return grad_for_lag(h, top_idx[0] - (N - 1), N) * dx
    w /= wsum
    g = np.zeros(N)
    for idx in top_idx:
        if w[idx] < 1e-7:
            continue
        lag = idx - (N - 1)
        g += w[idx] * grad_for_lag(h, lag, N)
    return g * dx

def optimize_anneal(h, N, rng, t_end, try_update,
                    alpha_start=2.0, alpha_end=60.0, max_iter=40000):
    val, corr, dx = obj_and_corr(h)
    try_update(h, val)
    best_h = h.copy()
    best_val = val
    lr = 0.04
    stagnant = 0
    momentum = np.zeros(N)
    beta = 0.7
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        progress = min(1.0, it / max(1, max_iter * 0.8))
        alpha = alpha_start + (alpha_end - alpha_start) * progress ** 1.5
        top_k = max(5, int(3 * np.sqrt(N)))
        
        g = soft_gradient_full(h, corr, dx, alpha, top_k)
        momentum = beta * momentum + (1 - beta) * g
        
        # Backtracking line search
        armijo_c = 1e-4
        improved = False
        step = lr
        delta = h - project_feasible(h - step * momentum, N)
        slope = np.dot(momentum, delta)
        for _ in range(8):
            h_new = project_feasible(h - step * momentum, N)
            val_new, corr_new, _ = obj_and_corr(h_new)
            if val_new <= val - armijo_c * step * max(abs(slope), 1e-10):
                h = h_new
                val = val_new
                corr = corr_new
                improved = True
                break
            step *= 0.5
        
        if improved:
            stagnant = 0
            lr = min(lr * 1.01, 0.1)
            if val < best_val:
                best_val = val
                best_h = h.copy()
                try_update(best_h, best_val)
        else:
            stagnant += 1
            lr *= 0.8
            beta = max(0.3, beta - 0.01)
            if stagnant > 25:
                scale = 0.08 + 0.12 * min(stagnant / 80.0, 1.0)
                h = project_feasible(h + rng.randn(N) * scale, N)
                val, corr, _ = obj_and_corr(h)
                lr = 0.04
                beta = 0.7
                momentum = np.zeros(N)
                stagnant = 0
    
    try_update(best_h, best_val)
    return best_h, best_val

def interpolate_to_n(h_src, n_dst):
    x0 = np.arange(len(h_src)) / len(h_src)
    x1 = np.arange(n_dst) / n_dst
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def binary_round(h, N, rng):
    N_ones = N // 2
    scores = h + rng.randn(N) * 0.15
    idx = np.argsort(scores)[::-1]
    h_bin = np.zeros(N)
    h_bin[idx[:N_ones]] = 1.0
    return h_bin

def structured_binary(N, rng):
    h = np.zeros(N)
    half = N // 2
    pattern = rng.randint(5)
    if pattern == 0:
        block_size = rng.randint(2, max(3, N // 5))
        for i in range(0, N, 2 * block_size):
            h[i:i + block_size] = 1.0
        n_ones = int(h.sum())
        if n_ones > half:
            idx = np.where(h == 1)[0]
            to_remove = rng.permutation(len(idx))[:n_ones - half]
            h[idx[to_remove]] = 0
        elif n_ones < half:
            idx = np.where(h == 0)[0]
            to_add = rng.permutation(len(idx))[:half - n_ones]
            h[idx[to_add]] = 1
    elif pattern == 1:
        h[rng.choice(N, half, replace=False)] = 1.0
    elif pattern == 2:
        freq = rng.uniform(1.5, N / 3)
        phase = rng.uniform(0, 2 * np.pi)
        vals = np.sin(2 * np.pi * freq * np.arange(N) / N + phase)
        h[np.argsort(vals)[::-1][:half]] = 1.0
    elif pattern == 3:
        phi = (1 + np.sqrt(5)) / 2
        positions = np.unique(np.floor(np.arange(half) * phi + rng.uniform(0, 1)).astype(int) % N)
        h[positions] = 1.0
        if h.sum() < half:
            remaining = np.where(h == 0)[0]
            h[rng.choice(remaining, int(half - h.sum()), replace=False)] = 1.0
    else:
        # Block with random phase
        p1, p2 = sorted(rng.randint(0, N, 2))
        h[p1:p2] = 1.0
        n_ones = int(h.sum())
        if n_ones > half:
            idx = np.where(h == 1)[0]
            h[idx[rng.permutation(len(idx))[:n_ones - half]]] = 0
        elif n_ones < half:
            idx = np.where(h == 0)[0]
            h[idx[rng.permutation(len(idx))[:half - n_ones]]] = 1
    return h
