import numpy as np
import time

def repair(h):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    x = np.asarray(h, dtype=float).copy()
    n = x.size
    target = 0.5 * n
    np.clip(x, 0, 1, out=x)
    for _ in range(50):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            slack = 1.0 - x
            mask = slack > 1e-14
            if not mask.any():
                break
            x[mask] += diff * slack[mask] / slack[mask].sum()
        else:
            mask = x > 1e-14
            if not mask.any():
                break
            x[mask] += diff * x[mask] / x[mask].sum()
        np.clip(x, 0, 1, out=x)
    # Final exact fix
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1-x)[::-1]
            for i in idx:
                add = min(diff, 1-x[i])
                x[i] += add
                diff -= add
                if diff <= 1e-15: break
        else:
            idx = np.argsort(x)[::-1]
            for i in idx:
                sub = min(-diff, x[i])
                x[i] -= sub
                diff += sub
                if diff >= -1e-15: break
    return x

    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1-h, mode="full") * dx

    return float(np.max(c_values(h)))

def compute_subgrad(h, active):
    """Compute subgradient of max_k c_k at active shifts"""
    n = h.size
    dx = 2.0 / n
    m = len(active)
    if m == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    for k in active:
        # k is index in the full correlation output (0 to 2n-2)
        # shift = k - (n-1)
        shift = k - (n-1)
        # dc/dh[j] = (1-h[j-shift]) * 1[0<=j-shift<n] - h[j+shift] * 1[0<=j+shift<n]
        if shift >= 0:
            # j-shift valid when j >= shift
            grad[shift:] += 1.0 - h[:n-shift]
            # j+shift valid when j < n-shift
            grad[:n-shift] -= h[shift:]
        else:
            s = -shift
            # j-shift = j+s valid when j+s < n, i.e., j < n-s
            grad[:n-s] += 1.0 - h[s:]
            # j+shift = j-s valid when j >= s
            grad[s:] -= h[:n-s]
    return grad * dx / m

def repair(h):
    x = np.asarray(h, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(60):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            slack = 1.0 - x
            mask = slack > 1e-14
            if not mask.any():
                break
            x[mask] += diff * slack[mask] / slack[mask].sum()
        else:
            mask = x > 1e-14
            if not mask.any():
                break
            x[mask] += diff * x[mask] / x[mask].sum()
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - x)[::-1]
            for idx in order:
                if diff <= 1e-15:
                    break
                add = min(diff, 1.0 - x[idx])
                x[idx] += add
                diff -= add
        else:
            order = np.argsort(x)[::-1]
            for idx in order:
                if diff >= -1e-15:
                    break
                sub = min(-diff, x[idx])
                x[idx] -= sub
                diff += sub
    return x

def c_values(h):
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(c_values(h)))

def coef_kj(h, shift, j):

    ni = j - shift

def solve_1d_minimax(b, a):

        return 0.5

def smooth_eval(x, tau, topk):
    n = x.size
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode="full") * dx
    cm = float(np.max(c))
    if tau <= 0:
        w = np.zeros(c.size)
        w[int(np.argmax(c))] = 1.0
        return cm, w
    z = (c - cm) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(c.size)
        w[int(np.argmax(c))] = 1.0
        return cm, w
    w = e / s
    if topk is not None and 0 < topk < c.size:
        idx = np.argsort(w)[::-1][:topk]
        w2 = np.zeros(c.size)
        w2[idx] = w[idx]
        ss = float(w2.sum())
        if ss > 0:
            w = w2 / ss
    return cm + tau * float(np.log(s)), w

def weighted_grad(x, w):
    n = x.size
    grad = np.zeros(n)
    idxs = np.nonzero(w > 1e-13)[0]
    for k in idxs:
        shift = int(k) - (n - 1)
        if shift >= 0:
            grad[shift:] += (1.0 - x[:n - shift])
            grad[:n - shift] -= x[shift:]
        else:
            s = -shift
            grad[:n - s] += (1.0 - x[s:])
            grad[s:] -= x[:n - s]
        grad *= float(w[k])
    grad /= float(w[idxs].sum()) if np.any(w > 1e-13) else 1.0
    return grad * (2.0 / n)

def slsqp_polish(h0, time_limit, seed):
    h = repair(h0)
    n = h.size
    best_x = h.copy()
    best_c = exact_c(h)
    start = time.time()

    for tau, mi in stages:
        if time.time() - start > time_limit * 0.8:
            break
        def fun(x):
            v, _ = smooth_eval(x, tau, None)
            return v
        def jac(x):
            _, w = smooth_eval(x, tau, None)
            return weighted_grad(x, w)
        def cf(x):
            return np.array([x.sum() - 0.5 * x.size])
        def cj(x):
            return np.ones((1, x.size))
        bounds = [(0.0, 1.0)] * n
        try:
            res = so.minimize(fun, h, method="SLSQP", jac=jac,
                              bounds=bounds,
                              constraints=[{"type": "eq", "fun": cf, "jac": cj}],
                              options={"maxiter": mi, "ftol": 1e-12})
            if res is not None:
                xr = repair(res.x)
                c = exact_c(xr)
                if c < best_c:
                    best_c = c
                    best_x = xr.copy()
                if c <= best_c + 1e-4:
                    h = xr
        except Exception:
            pass
    return best_x, best_c

def resample(h, n_new):

    # Phase 3: multi-resolution search
    base_n = int(best_x.size)
    n_list = []
    for n in [base_n, base_n + 10, base_n - 10, 60, 80, 100, 50, 90, 110, 75, 85, 95]:
        n = int(n)
        if 30 <= n <= 150 and n not in n_list:
            n_list.append(n)
    n_list = n_list[:8]

    for n in n_list:
        if time.time() > deadline - 60:
            break
        h_start = resample(best_x, n)
        per_cd = min(25.0, (deadline - time.time()) * 0.06)
        per_sq = min(15.0, (deadline - time.time()) * 0.04)
        if per_cd < 3 and per_sq < 2:
            break

        for scale in [0.0, 0.03, 0.08]:
            if time.time() > deadline - 30:
                break
            if scale == 0:
                h_try = h_start.copy()
            else:
                rng = np.random.RandomState(seed + n * 31 + int(scale * 1000))
                h_try = repair(h_start + rng.normal(0, scale, n))
            if per_cd > 3:
                hx, cx = coord_descent(h_try, per_cd, seed + n * 7 + int(scale * 50), k_act=5)
                if cx < best_c:
                    best_c = cx
                    best_x = repair(hx)
                    h_start = resample(best_x, n)
            if per_sq > 2 and time.time() < deadline - 25:
                hx, cx = slsqp_polish(h_try, per_sq, seed + n * 11)
                if cx < best_c:
                    best_c = cx
                    best_x = repair(hx)

    # Phase 4: final intensive polish
    tl = max(0.0, (deadline - time.time()) - 5.0)
    if tl > 10:
        hx, cx = coord_descent(best_x, tl * 0.6, seed + 7777, k_act=3)
        if cx < best_c:
            best_c = cx
            best_x = repair(hx)
        tl2 = max(0.0, (deadline - time.time()) - 3.0)
        if tl2 > 5:
            hx, cx = slsqp_polish(best_x, tl2, seed + 8888)
            if cx < best_c:
                best_c = cx
                best_x = repair(hx)

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
