```python
            # So h[m+j] = h_pad[n + m + j]
            
            d1 = h_pad[n + m_vals + j]  # h[m+j]
            d2 = h_pad[n + m_vals + i]  # h[m+i]
            d3 = h_pad[n + j - m_vals]  # h[j-m]
            d4 = h_pad[n + i - m_vals]  # h[i-m]
            
            d = d1 - d2 + d3 - d4
            
            # Delta bounds
            dmin = max(-h[i], h[j] - 1.0)
            dmax = min(1.0 - h[i], h[j])
            if dmax - dmin < 1e-10:
                continue
            
            deltas = np.linspace(dmin + 1e-8, dmax - 1e-8, n_delta)
            # new_c5 for each delta:
            # max_k (corr[k] + delta * d[k]) * 2/n
            # Vectorized:
            new_vals = corr + deltas[:, None] * d[None, :]
            new_c5s = np.max(new_vals, axis=1) * 2.0 / n
            
            best_i = int(np.argmin(new_c5s))
            if new_c5s[best_i] < current_c5 - 1e-14:
                best_delta = deltas[best_i]
                h[i] = float(np.clip(h[i] + best_delta, 0, 1))
                h[j] = float(np.clip(h[j] - best_delta, 0, 1))
                corr = corr + best_delta * d
                current_c5 = float(new_c5s[best_i])
                found = True
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        
        step += 1
        
        if not found:
            no_improve_count += 1
            if no_improve_count >= 5:
                # Perturbation
                n_p = rng.integers(2, 6)
                for _ in range(n_p):
                    p1 = rng.randint(n)
                    p2 = rng.randint(n)
                    if p1 != p2:
                        eps = rng.uniform(-0.15, 0.15)
                        h[p1] = float(np.clip(h[p1] + eps, 0, 1))
                        h[p2] = float(np.clip(h[p2] - eps, 0, 1))
                corr = compute_corr(h)
                current_c5 = float(np.max(corr)) * 2.0 / n
                no_improve_count = 0
    
    return best_h, best_c5
```
