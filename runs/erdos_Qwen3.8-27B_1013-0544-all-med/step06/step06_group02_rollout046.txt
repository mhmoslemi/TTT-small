Actually, np.correlate is O(N²) in general (it's a direct computation). For N=120, that's 14400 operations, done in C, so it's fast (~microseconds).

The main bottleneck is the Python loop in soft_subgrad. Let me limit it to top M lags.

Let me rewrite with this optimization:

```python
def soft_subgrad_top(h, corr, alpha, top_m=5):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    # Get top M indices
    top_idx = np.argpartition(w, -top_m)[-top_m:]
```

This is O(M*N) per call, much faster.

4. Try Thue-Morse and other structured initializations

6. Final fine-tuning

```python
import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        frac = (h > 1e-8) & (h < 1.0 - 1e-8)
        if frac.sum() > 0:
            h[frac] += diff / frac.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def obj_val(h):
    N = len(h)
    dx = 2.0 / N
    corr = compute_corr(h)
    return corr.max() * dx, corr, dx

def subgrad_at_lag(h, k):
    N = len(h)
    shift = k - (N - 1)
    g = np.zeros(N)
    if shift >= 0:
        n_valid = N - shift
        g[:n_valid] += (1.0 - h[shift:N])
        g[shift:] -= h[:n_valid]
    else:
        s = -shift
        n_valid = N - s
        g[s:] += (1.0 - h[:n_valid])
        g[:n_valid] -= h[s:]
    return g

def soft_subgrad_top(h, corr, alpha, top_m=8):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    if top_m < len(w):
        top_idx = np.argpartition(w, -top_m)[-top_m:]
    else:
        top_idx = np.arange(len(w))
    w_top = w[top_idx]
    wsum = w_top.sum()
    if wsum < 1e-300:
        g = subgrad_at_lag(h, int(np.argmax(corr)))
        return g
    w_top /= wsum
    g = np.zeros(N)
    for idx, wt in zip(top_idx, w_top):
        if wt < 1e-6:
            continue
        g += wt * subgrad_at_lag(h, int(idx))
    return g

def thue_morse(N):
    t = np.zeros(N)
    for i in range(1, N):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def interpolate_to_n(h_src, n_dst):
    x0 = np.arange(len(h_src)) / len(h_src)
    x1 = np.arange(n_dst) / n_dst
    return project_feasible(np.interp(x1, x0, h_src), n_dst)

def adam_optimize(h, N, rng, t_phase_end, max_iter, top_m=8, alpha0=12.0):
    val, corr, dx = obj_val(h)
    m = np.zeros(N)
    v = np.zeros(N)
    beta1, beta2, eps = 0.75, 0.95, 1e-8
    lr = 0.04
    t_step = 0
    best_local = val
    best_local_h = h.copy()
    stagnant = 0
    alpha = alpha0

    for it in range(max_iter):
        if time.time() > t_phase_end:
            break

        g = soft_subgrad_top(h, corr, alpha, top_m) * dx

        t_step += 1
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)

        h_new = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), N)
        val_new, corr_new, _ = obj_val(h_new)

        if val_new < val - 1e-14:
            h = h_new
            val = val_new
            corr = corr_new
            stagnant = 0
            lr = min(lr * 1.015, 0.12)
            if val < best_local:
                best_local = val
                best_local_h = h.copy()
        else:
            stagnant += 1
            if stagnant == 20:
                alpha = max(2.0, alpha * 0.7)
            elif stagnant > 50:
                scale = 0.03 + 0.08 * min((stagnant - 50) / 80.0, 1.0)
                h = project_feasible(h + rng.randn(N) * scale, N)
                val, corr, _ = obj_val(h)
                m = np.zeros(N)
                v = np.zeros(N)
                t_step = 0
                lr = 0.04
                stagnant = 0
                alpha = alpha0

    return best_local_h, best_local

def coord_descent(h, N, rng, t_phase_end):
    val, corr, dx = obj_val(h)
    improved_total = True
    sweep = 0
    while improved_total and time.time() < t_phase_end:
        improved_total = False
        sweep += 1
        order = rng.permutation(N)
        for i in order:
            if time.time() > t_phase_end:
                return h, val
            old = h[i]
            candidates = [0.0, 1.0, (1.0 - old)]
            # Also try small perturbations
            if 0.0 < old < 1.0:
                candidates.append(old * 0.5)
                candidates.append(0.5 * (old + 1.0))
            for nv in candidates:
                if abs(nv - old) < 1e-6:
                    continue
                h[i] = nv
                h = project_feasible(h, N)
                v_new = obj_val(h)[0]
                if v_new < val - 1e-12:
                    val = v_new
                    improved_total = True
                else:
                    h[i] = old
                    # Re-project to restore
                    h = project_feasible(h, N)
            h[i] = old
            h = project_feasible(h, N)
    return h, val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 12

    best_h = None
    best_val = float('inf')

    def try_update(h, val):
        nonlocal best_h, best_val
        if val < best_val:
            best_val = val
            best_h = h.copy()

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    if h_init is not None:
        v0 = obj_val(h_init)[0]
        try_update(h_init.copy(), v0)

    # Phase 1: Refine initial with Adam
    if h_init is not None and time.time() < t_end - 200:
        h1, v1 = adam_optimize(h_init.copy(), len(h_init), rng, t_end - 200, max_iter=300000, top_m=8, alpha0=15.0)
        try_update(h1, v1)

    # Phase 2: Thue-Morse starts at various N
    for N_tm in [64, 80, 96, 128, 160]:
        if time.time() > t_end - 120:
            break
        h_tm = thue_morse(N_tm).astype(float)
        h_tm = project_feasible(h_tm, N_tm)
        t_phase = min(time.time() + 40, t_end - 60)
        h_r, v_r = adam_optimize(h_tm, N_tm, rng, t_phase, max_iter=60000, top_m=6, alpha0=10.0)
        try_update(h_r, v_r)

    # Phase 3: Interpolate best to other resolutions and refine
    if best_h is not None and time.time() < t_end - 80:
        N_best = len(best_h)
        n_list = sorted(set([max(48, N_best - 32), N_best - 16, N_best, N_best + 16, N_best + 32, N_best + 48]))
        for N_t in n_list:
            if N_t < 40 or N_t > 200 or time.time() > t_end - 40:
                continue
            if N_t == N_best:
                h_start = best_h.copy()
            else:
                h_start = interpolate_to_n(best_h, N_t)
            t_phase = min(time.time() + 35, t_end - 30)
            h_r, v_r = adam_optimize(h_start, N_t, rng, t_phase, max_iter=50000, top_m=8, alpha0=12.0)
            try_update(h_r, v_r)

    # Phase 4: Coordinate descent on best
    if best_h is not None and time.time() < t_end - 40:
        h_cd, v_cd = coord_descent(best_h.copy(), len(best_h), rng, min(time.time() + 80, t_end - 30))
        try_update(h_cd, v_cd)

    # Phase 5: More Adam from best with different hyperparams
    if best_h is not None and time.time() < t_end - 30:
        for alpha_try in [8.0, 20.0]:
            if time.time() > t_end - 20:
                break
            h_start = project_feasible(best_h + rng.randn(len(best_h)) * 0.02, len(best_h))
            t_phase = min(time.time() + 30, t_end - 15)
            h_r, v_r = adam_optimize(h_start, len(best_h), rng, t_phase, max_iter=50000, top_m=10, alpha0=alpha_try)
            try_update(h_r, v_r)

    # Phase 6: Final coordinate descent + fine Adam
    if best_h is not None and time.time() < t_end - 5:
        h_cd2, v_cd2 = coord_descent(best_h.copy(), len(best_h), rng, min(time.time() + 20, t_end - 5))
        try_update(h_cd2, v_cd2)
        if time.time() < t_end:
            h_fin, v_fin = adam_optimize(best_h.copy(), len(best_h), rng, t_end, max_iter=50000, top_m=12, alpha0=25.0)
            try_update(h_fin, v_fin)

    if best_h is None:
        best_h = project_feasible(thue_morse(120).astype(float), 120)
        best_val = obj_val(best_h)[0]

    return best_h, best_val, len(best_h)
```
