def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break

        else:

            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

    A_j = np.zeros(2 * n - 1)
    A_m = np.zeros(2 * n - 1)
    for k in range(2 * n - 1):
        kk = k - (n - 1)  # actual shift
        # A_k[j]: contribution of h[j] to c(kk)

        # h[j] appears in (1-h[i]) when i = j, contributing -h[j+kk]
        a = 0.0
        if 0 <= j - kk < n:
            a += (1.0 - h[j - kk])
        if 0 <= j + kk < n:
            a -= h[j + kk]
        A_j[k] = a
        b = 0.0
        if 0 <= m - kk < n:
            b += (1.0 - h[m - kk])
        if 0 <= m + kk < n:
            b -= h[m + kk]
        A_m[k] = b
    B = A_j - A_m
    # f(d) = max_k (c[k] + d * B[k])
    # Convex in d. Minimize over feasible range.
    d_lo = max(-h[j], -(1.0 - h[m]))
    d_hi = min(1.0 - h[j], h[m])
    if d_hi <= d_lo:
        return 0.0
    # Ternary search (works for convex functions)
    # Actually for piecewise linear, use subgradient bisection
    lo = d_lo
    hi = d_hi
    for _ in range(n_iter):
        mid = 0.5 * (lo + hi)
        # Subgradient at mid: any B[k] where c[k] + mid*B[k] = max
        vals = c + mid * B
        imax = np.argmax(vals)
        # For convex PWL, if B[imax] > 0, minimum is to the left
        # If B[imax] < 0, minimum is to the right
        # More robust: check if 0 is in subgradient interval
        active = np.where(np.isclose(vals, vals.max(), atol=1e-10))[0]
        sg_lo = B[active].min()
        sg_hi = B[active].max()
        if sg_lo > 1e-12:
            hi = mid
        elif sg_hi < -1e-12:
            lo = mid
        else:
            break
    return 0.5 * (lo + hi)

                        improved = True

    rng = np.random.RandomState(seed)
    half = n // 2
    # Initialize population
    pop = []
    for _ in range(pop_size):
        seq = np.zeros(n, dtype=int)
        ones = rng.choice(n, half, replace=False)
        seq[ones] = 1
        pop.append(seq)
    
    def fitness(seq):
        h = seq.astype(float)
        return compute_c5(h, n)
    
    # Evaluate initial
    fits = [fitness(s) for s in pop]
    
    for gen in range(generations):
        # Sort by fitness
        order = np.argsort(fits)
        pop = [pop[i] for i in order]
        fits = [fits[i] for i in order]
        
        # Elitism: keep top 10
        new_pop = pop[:10]
        
        # Generate rest
        while len(new_pop) < pop_size:
            # Tournament selection
            p1 = pop[rng.choice(5)]
            p2 = pop[rng.choice(5)]
            # Crossover
            if rng.rand() < 0.7:
                point = rng.randint(1, n-1)
                child = np.concatenate([p1[:point], p2[point:]])
            else:
                child = p1.copy()
            # Mutation: flip 1-3 bits (maintain balance)
            for _ in range(rng.randint(1, 4)):
                if child.sum() > half:
                    # Flip a 1 to 0
                    ones = np.where(child == 1)[0]
                    child[ones[rng.randint(len(ones))]] = 0
                elif child.sum() < half:
                    zeros = np.where(child == 0)[0]
                    child[zeros[rng.randint(len(zeros))]] = 1
                else:
                    # Equal: swap a 1 and a 0
                    ones = np.where(child == 1)[0]
                    zeros = np.where(child == 0)[0]
                    child[ones[rng.randint(len(ones))]] = 0
                    child[zeros[rng.randint(len(zeros))]] = 1
            new_pop.append(child)
        
        pop = new_pop
        fits = [fitness(s) for s in pop]
    
    return pop[np.argmin(fits)], min(fits)
