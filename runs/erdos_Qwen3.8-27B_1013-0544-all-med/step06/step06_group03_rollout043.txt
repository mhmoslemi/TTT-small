def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:

        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _cgs(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n-shift])
            g[:n-shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
    return g

def sm_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    s = exps.sum()
    if s < 1e-300:
        return _cgs(h, int(np.argmax(conv)) - n + 1) * dx
    w = exps / s
    grad = np.zeros(n)
    for i in range(2*n-1):
        if w[i] < 1e-11:
            continue
        grad += w[i] * _cgs(h, i - n + 1)
    return grad * dx

def proj(h, n):
    h = h.copy()
    tgt = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > tgt:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo+hi)*0.5, 0, 1)

def mom_pgd(h0, niter=30000, lr0=0.006, t0=0.05, t1=0.002, mu0=0.85):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best = compute_c5(h)
    bh = h.copy()
    lr = lr0
    mu = mu0
    ne = 8
    pe = max(1, niter // ne)
    for e in range(ne):
        f = e / max(1, ne-1)
        t = t0 * (t1/t0)**f
        for _ in range(pe):
            c = compute_c5(h)
            if c < best - 1e-14:
                best = c
                bh = h.copy()
            g = sm_grad(h, t)
            v = mu*v - lr*g
            hn = proj(h+v, n)
            cn = compute_c5(hn)
            if cn > c*1.02 + 1e-10:
                lr *= 0.5
                v *= 0.3
                if lr < 1e-15:
                    return bh, best
            else:
                h = hn
                lr = min(lr*1.02, 0.02)
        mu *= 0.94
    return bh, best

def sg_pgd(h0, niter=30000, lr0=0.002):
    n = len(h0)
    h = h0.copy()
    best = compute_c5(h)
    bh = h.copy()
    lr = lr0

        conv = np.correlate(h, 1.0-h, mode="full")

        sh = int(conv.argmax()) - n + 1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _cgs(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n-shift])
            g[:n-shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
    return g

def sm_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    return t * (m / t + np.log(np.exp((conv - m) / t).sum())) * dx

def sm_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    s = exps.sum()
    if s < 1e-300:
        return _cgs(h, int(np.argmax(conv)) - n + 1) * dx
    w = exps / s
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        if w[i] < 1e-11:
            continue
        grad += w[i] * _cgs(h, i - n + 1)
    return grad * dx

def proj(h, n):
    h = h.copy()
    tgt = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        if np.clip(h + mid, 0, 1).sum() > tgt:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) * 0.5, 0, 1)

def mom_pgd(h0, niter=30000, lr0=0.006, t0=0.05, t1=0.002, mu0=0.85):
    n = len(h0)
    h = h0.copy()
    v = np.zeros(n)
    best = compute_c5(h)
    bh = h.copy()
    lr = lr0
    mu = mu0
    ne = 8
    pe = max(1, niter // ne)
    for e in range(ne):
        f = e / max(1, ne - 1)
        t = t0 * (t1 / t0) ** f

                best = c
                bh = h.copy()
            g = sm_grad(h, t)

            hn = proj(h + v, n)
            cn = compute_c5(hn)

def sg_pgd(h0, niter=30000, lr0=0.002):
    n = len(h0)
    h = h0.copy()
    best = compute_c5(h)
    bh = h.copy()

    for _ in range(niter):
        conv = np.correlate(h, 1.0 - h, mode="full")
        c = conv.max() * (2.0 / n)

            best = c
            bh = h.copy()
