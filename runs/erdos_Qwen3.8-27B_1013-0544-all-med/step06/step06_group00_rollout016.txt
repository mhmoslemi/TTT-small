```python
    grad = np.zeros(n)
    for k in active_indices:
        m = k - (n - 1)
        if m >= 0:
            # First: grad[0:n-m] += (1-h[m:n])
```

```python
    h = h.copy()
```

```python
def evaluate_swap(corr, h, n, i, j):
    """Evaluate C5 after swapping h[i]: 1->0, h[j]: 0->1 (binary)."""
    corr_new = corr.copy()
    # h[i]: 1->0
    k_start = n - 1 - i
    k_end = 2 * n - 2 - i
    ks = np.arange(k_start, k_end + 1)
    idx1 = ks + (n - 1) - i  # = i + k - (n-1)
    idx2 = i - (ks - (n - 1))  # = i - k + (n-1) = i + (n-1) - k
    # Wait: i+k-(n-1) and i-k+(n-1) = i+(n-1)-k
    # Both should be in [0, n-1] for valid k in [n-1-i, 2n-2-i]
    delta1 = -(1.0 - h[idx1]) + h[idx2]
    corr_new[ks] += delta1
    
    # h[j]: 0->1
    k_start2 = n - 1 - j
    k_end2 = 2 * n - 2 - j
    ks2 = np.arange(k_start2, k_end2 + 1)
    idx1_2 = ks2 + (n - 1) - j
    idx2_2 = j + (n - 1) - ks2
    delta2 = (1.0 - h[idx1_2]) - h[idx2_2]
    corr_new[ks2] += delta2
    
    return np.max(corr_new) * (2.0 / n)
```

This is O(n) per evaluation (two array operations of size n). In numpy, each takes ~1μs. So per swap: ~2μs. For 36864 swaps: ~74ms. That's much better!
