    h = h.copy()

        if s > target:

        else:

def make_beatty(n, alpha):
    """Create a binary vector using Beatty sequence"""
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (int((i+1)*alpha) - int(i*alpha)) == 1 else 0.0
    return h

def binary_local_search(h, rng, max_rounds=50, attempts_per_round=50):
    """Local search: swap 1s and 0s to reduce C5"""
    n = len(h)

Actually, here's another idea: **use the correlation array to prioritize swaps**. If the max is at lag s*, then we want to reduce corr[s*]. We can find which (i, j) pairs would most reduce corr[s*] and try those first.

For lag s*, corr[s*] = sum_i h[i]*(1-h[i-s*]) (with valid range).
A swap (i: 1→0, j: 0→1) reduces corr[s*] by:
- (1-h[i-s*]) if i is in the valid range (i.e., the term h[i]*(1-h[i-s*]) was contributing)
- minus (1-h[j-s*]) if j is in the valid range (the new term h[j]*(1-h[j-s*]) is added)

Net change at lag s*: -(1-h[i-s*]) + (1-h[j-s*]) = h[i-s*] - h[j-s*]

So we want h[i-s*] > h[j-s*], i.e., the position i-s* should be 1 (or close to 1) and j-s* should be 0 (or close to 0).

This gives us a way to prioritize: for the active lag s*, try swaps where i-s* is a 1 and j-s* is a 0.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def conv_grad_single(h, shift, n):
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n-shift])
            g[:n-shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
    return g

def smooth_grad(h, t, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = conv.max()
    exps = np.exp((conv - m) / t)
    w = exps / exps.sum()
    grad = np.zeros(n)
    active = w > 1e-10
    for idx in np.where(active)[0]:
        shift = idx - (n - 1)
        grad += w[idx] * conv_grad_single(h, shift, n)
    return grad * dx

def pgd(h_init, max_iter, lr_init, t=0.0):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        if t > 0:
            grad = smooth_grad(h, t, n)
        else:
            conv = np.correlate(h, 1.0 - h, mode="full")
            idx = np.argmax(conv)
            shift = idx - (n - 1)
            grad = conv_grad_single(h, shift, n) * dx
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-12:
            lr *= 0.5
            if lr < 1e-15:
                break
        else:
            h = h_new
            lr = min(lr * 1.06, 0.05)
    return best_h, best_c5

def blocal(h, rng, rounds, attempts):
    """Binary local search with incremental correlation update"""
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    best_max = corr.max()
    ones = np.where(h > 0.99)[0]
    zeros = np.where(h < 0.01)[0]
    stall = 0
    for r in range(rounds):
        if stall > 30:
            break
        improved = False
        for _ in range(attempts):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            # Delta computation
            delta = np.zeros(2 * n - 1)
            # h[i]: 1->0
            plo, phi = i, min(i + n - 1, 2 * n - 2)
            if plo <= phi:
                pv = np.arange(plo, phi + 1)
                hi_ = i + n - 1 - pv
                delta[pv] -= (1.0 - h[hi_])
            # h[j]: 0->1
            plo2, phi2 = j, min(j + n - 1, 2 * n - 2)
            if plo2 <= phi2:
                pv2 = np.arange(plo2, phi2 + 1)
                hj_ = j + n - 1 - pv2
                delta[pv2] += (1.0 - h[hj_])
            nm = (corr + delta).max()
            if nm < best_max:
                h[i] = 0.0; h[j] = 1.0
                corr += delta
                best_max = nm
                improved = True
                oi = np.where(ones == i)[0]
                oj = np.where(ones == j)[0]
                oz = np.where(zeros == j)[0]
                oz2 = np.where(zeros == i)[0]
                if len(oi) > 0:
                    ones = np.delete(ones, oi[0])
                else:
                    ones = np.append(ones, j)
                if len(oz) > 0:
                    zeros = np.delete(zeros, oz[0])
                else:
                    zeros = np.append(zeros, i)
        if not improved:
            stall += 1
    return h, best_max * dx

def make_structured(n, seed):
    rng = np.random.RandomState(seed)
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    x = np.arange(n, dtype=float)
    h = (np.mod(x * phi, 1.0) < 0.5).astype(float)
    target = n // 2
    if h.sum() > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:int(h.sum() - target)]] = 0.0
    elif h.sum() < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:int(target - h.sum())]] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):

    rng = np.random.RandomState(seed)
    
    def tl():
        return budget_s - (time.time() - t0)
    
    # Initial
    if 'initial_h_values' in globals() and initial_h_values is not None:

    else:

    hb = (best_h > 0.5).astype(float)
