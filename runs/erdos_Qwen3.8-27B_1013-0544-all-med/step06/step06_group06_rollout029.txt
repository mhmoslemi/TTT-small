                res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 1500, 'ftol': 1e-16})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5:
                    best_h, best_c5 = h_try, c
        except Exception:
            pass

    if time.time() < t0 + budget_s - 10:
        best_h, best_val = greedy_polish(binarize(best_h), n, t0 + budget_s - 10)
        c5b = best_val * dx
        if c5b < best_c5:
            best_h, best_c5 = best_h, c5b

    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
