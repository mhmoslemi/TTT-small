def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        h += d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:

            else:

                h[idx] += d2 / len(idx) if len(idx) > 0 else d2

def _coord_descent(h, n, steps=100):
    """Coordinate descent: for each i, find best h[i] given others fixed."""
    dx = 2.0 / n
    best_c5 = _c5_max(h)
    for step in range(steps):
        improved = False
        for i in range(n):

            for val in [0.0, 1.0, h[i] - 0.1, h[i] + 0.1, 0.5]:
                if val < 0 or val > 1:

                h_old = h[i]

        h_bin, _ = _binary_local_search(h_bin, nb, max_passes=20)
        ht = _opt(h_bin, 2000, 100, 1000)
        cn = _c5_max(ht)
        if cn < cb:
            cb, hb = cn, ht

    hb = _fix(hb, nb)
    cb = _c5_max(hb)
    return (hb, cb, nb)
