```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(y, n):
    """Project y onto {0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        x = np.clip(y - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    x = np.clip(y - lam, 0.0, 1.0)
    # Fix residual
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-9) & (x < 1.0 - 1e-9)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
    return np.clip(x, 0.0, 1.0)

def grad_softmax(h, weights, n):
    """Compute soft-max subgradient."""
    grad = np.zeros(n)
    sig = np.where(weights > 1e-10)[0]
    for m in sig:
        w = weights[m]
        k = m - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[:n-k] += w * (1.0 - h[k:])
            if k < n:
                grad[k:] -= w * h[:n-k]
        else:

            if kk < n:
                grad[kk:] += w * (1.0 - h[:n-kk])
            if n - kk > 0:
                grad[:n-kk] -= w * h[kk:]
```

```python
def grad_softmax(h, weights, n):
    grad = np.zeros(n)
    sig = np.where(weights > 1e-10)[0]
    for m in sig:
        w = weights[m]
        k = m - (n - 1)
        # Term 1: w * (1-h[j-k]) for valid j-k (i.e., 0 <= j-k < n)
        # j-k in [0, n) means j in [k, k+n), intersected with [0, n) gives j in [max(0,k), min(n, k+n))
        if k >= 0:
            # j in [k, n): h[j-k] = h[0:n-k]
            grad[k:] += w * (1.0 - h[:n-k])
        else:
            # j-k = j+|k| in [0, n) means j in [0, n-|k|)
            kk = -k
            grad[:n-kk] += w * (1.0 - h[kk:])
        
        # Term 2: -w * h[j+k] for valid j+k (i.e., 0 <= j+k < n)
        if k >= 0:
            # j+k in [0, n) means j in [-k, n-k), intersected with [0,n) gives [0, n-k)
            grad[:n-k] -= w * h[k:]
        else:
            # j+k = j-|k| in [0, n) means j in [|k|, n)
            kk = -k
            grad[kk:] -= w * h[:n-kk]
    return grad
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(y - mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0, 1)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        mask = (x > 1e-9) & (x < 1 - 1e-9)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
    return np.clip(x, 0, 1)

def grad_sm(h, w, n):
    g = np.zeros(n)
    sig = np.where(w > 1e-10)[0]
    for m in sig:
        wi = w[m]
        k = m - (n - 1)
        if k >= 0:
            if n - k > 0:
                g[k:] += wi * (1.0 - h[:n-k])
            if k < n and n - k > 0:

            kk = -k
```
