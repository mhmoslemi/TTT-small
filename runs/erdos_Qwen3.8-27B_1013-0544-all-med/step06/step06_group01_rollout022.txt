import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _project(h, target):
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _grad_active(h, n, shifts):
    dx = 2.0 / n
    j = np.arange(n)
    grad = np.zeros(n)
    for s in shifts:
        ip = j + s
        vp = (ip >= 0) & (ip < n)
        im = j - s
        vm = (im >= 0) & (im < n)
        grad += np.where(vp, 1.0 - h[np.clip(ip, 0, n - 1)], 0.0)
        grad -= np.where(vm, h[np.clip(im, 0, n - 1)], 0.0)
    return dx * grad

def _obj_and_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    m = C.max()
    exp_C = np.exp(alpha * (C - m))
    se = exp_C.sum()
    f = (m + np.log(se)) / alpha + beta * (h.sum() - 0.5 * n) ** 2
    w = exp_C / se
    j = np.arange(n)
    nc = len(C)
    grad = np.zeros(n)
    for idx in range(nc):
        wi = w[idx]
        if wi < 1e-14:
            continue
        s = idx - (n - 1)
        ip = j + s
        vp = (ip >= 0) & (ip < n)
        im = j - s
        vm = (im >= 0) & (im < n)
        grad += wi * np.where(vp, 1.0 - h[np.clip(ip, 0, n - 1)], 0.0)
        grad -= wi * np.where(vm, h[np.clip(im, 0, n - 1)], 0.0)
    grad *= dx * alpha
    grad += 2.0 * beta * (h.sum() - 0.5 * n)
    return f, grad

def _active_set_loop(h, n, max_iter=10000, tol=1e-12):
    h = _project(h.copy(), n * 0.5)
    target = n * 0.5
    best_c5 = _c5_max(h)
    shifts_all = np.arange(-(n - 1), n)
    for it in range(max_iter):
        C = _c5_array(h)
        cur_max = C.max()
        # Adaptive threshold: tighten as we converge
        frac = max(0.002, 0.02 * (1.0 - it / max_iter))
        thr = max(0.003, cur_max * frac)
        active_mask = C >= cur_max - thr
        active_shifts = shifts_all[active_mask]
        if len(active_shifts) == 0:
            active_shifts = shifts_all[np.argmax(C):np.argmax(C)+1]
        grad = _grad_active(h, n, active_shifts)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        base_lr = 0.15 / (gnorm + 1e-10)
        best_h = h
        best_val = cur_max
        for lr in [base_lr, base_lr * 0.5, base_lr * 0.25, base_lr * 0.1,
                   0.05 / gnorm, 0.01 / gnorm]:
            h_try = _project(h - lr * grad, target)
            c5_try = _c5_max(h_try)
            if c5_try < best_val:
                best_val = c5_try
                best_h = h_try
        if best_val >= cur_max - tol:
            break
        h = best_h
        if best_val < best_c5:
            best_c5 = best_val
    return _project(h, target), best_c5

def _lse_opt(h, n, alpha, beta, maxiter=200):
    res = minimize(_obj_and_grad, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
    if res.x is None:
        return h
    return _project(res.x, n * 0.5)

def _interp_h(h_old, n_old, n_new):
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return _project(np.interp(x_new, x_old, h_old), n_new * 0.5)
