def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):

        if abs(d) < 1e-14:

            h = np.clip(h, 0.0, 1.0)
        else:
            h[mask] += d / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    shifted = temp * (corr - m)
    # For numerical stability, clip
    shifted = np.clip(shifted, -500, 500)
    s = np.sum(np.exp(shifted))
    return (m + np.log(s) / temp) * dx

    n = len(h)

            if best_c < c5 - 1e-14:

                improved = True
        if not improved:

def smooth_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    
    # Compute softmax weights
    m = np.max(corr)
    shifted = beta * (corr - m)
    shifted = np.clip(shifted, -500, 0)  # clip for stability
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)  # length 2n-1
    
    # w[p] corresponds to lag k = p - (n-1)
    # grad[j] = dx * sum_{p=0}^{2n-2} w[p] * [(1-h[j-k_p])*[0<=j-k_p<n] - h[j+k_p]*[0<=j+k_p<n]]
    # where k_p = p - (n-1)
    
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-15:
            continue
        k = p - (n - 1)  # lag
        # Term 1: (1-h[j-k]) for 0 <= j-k < n, i.e., k <= j < k+n
        # Term 2: -h[j+k] for 0 <= j+k < n, i.e., -k <= j < n-k
        if k >= 0:
            # Term 1: j from k to n-1
            grad[k:n] += w[p] * (1.0 - h[0:n-k])
            # Term 2: j from 0 to n-1-k
            grad[0:n-k] += w[p] * (-h[k:n])
        else:
            # k < 0, let k' = -k
            kprime = -k
            # Term 1: (1-h[j+k']) for 0 <= j+k' < n, i.e., 0 <= j < n-k'
            #   h[j-k] = h[j+k'], valid when 0 <= j+k' < n
            grad[0:n-kprime] += w[p] * (1.0 - h[kprime:n])
            # Term 2: -h[j-k'] for 0 <= j-k' < n, i.e., k' <= j < n
            grad[kprime:n] += w[p] * (-h[0:n-kprime])
    
    return grad * dx
