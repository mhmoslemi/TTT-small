from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_objective(h, alpha=50.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha

    n = len(h)
    target = n / 2.0
    obj = smooth_objective(h, alpha)
    # Box penalty
    box = np.sum(np.maximum(0.0, h - 1.0)**2) + np.sum(np.maximum(0.0, -h)**2)
    # Sum penalty
    s = np.sum(h) - target
    return obj + 1000.0 * (box + s * s)

    """Project h onto the feasible set: 0 <= h <= 1, sum(h) = n/2"""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_nm(h, alpha):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    bp = np.sum(np.maximum(0.0, h - 1.0)**2) + np.sum(np.maximum(0.0, -h)**2)
    sp = (np.sum(h) - target)**2
    return sm + 200.0 * (bp + sp)

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        c = mask.sum()
        if c > 0:
            h[mask] += diff / c
            h = np.clip(h, 0.0, 1.0)

    res = minimize(lambda x: obj_nm(x, alpha), h, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True})
    return project(res.x)

    n = len(h)
    cur = compute_c5(h)
    t0 = time.time()
    ds = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]
    for _ in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        bc, bm = cur, None
        for i in range(n):
            for j in range(i + 1, n):
                for d in ds:
                    if h[i]+d < 0 or h[i]+d > 1 or h[j]-d < 0 or h[j]-d > 1:
                        continue
                    h[i] += d; h[j] -= d
                    c = compute_c5(h)
                    h[i] -= d; h[j] += d
                    if c < bc:
                        bc, bm = c, (i, j, d)
        if bm is None:
            break
        i, j, d = bm
        h[i] += d; h[j] -= d
        cur = bc
    return h, cur

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def obj_nm(h, alpha):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    sm = m + np.log(np.sum(np.exp(alpha * (corr - m)))) / alpha
    bp = np.sum(np.maximum(0.0, h - 1.0)**2) + np.sum(np.maximum(0.0, -h)**2)
    sp = (np.sum(h) - target)**2
    return sm + 200.0 * (bp + sp)

def project(h):
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        c = mask.sum()
        if c > 0:
            h[mask] += diff / c
            h = np.clip(h, 0.0, 1.0)
    return h

def nm_opt(h, alpha=60.0, maxiter=300):
    res = minimize(lambda x: obj_nm(x, alpha), h, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-9, 'fatol': 1e-11, 'adaptive': True})
    return project(res.x)

def cd_first(h, max_rounds=3000, time_limit=30.0):
    n = len(h)
    cur = compute_c5(h)
    t0 = time.time()
    ds = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]
    for _ in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        imp = False
        for i in range(n):
            for j in range(i + 1, n):
                for d in ds:
                    if h[i]+d < 0 or h[i]+d > 1 or h[j]-d < 0 or h[j]-d > 1:
                        continue
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h)
                    if c < cur - 1e-13:
                        cur = c
                        imp = True
                        break
                    h[i] -= d
                    h[j] += d
                if imp:
                    break
            if imp:
                break
        if not imp:
            break
    return h, cur

def cd_best(h, max_rounds=100, time_limit=30.0):
    n = len(h)
    cur = compute_c5(h)
    t0 = time.time()
    ds = [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]
    for _ in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        bc, bm = cur, None
        for i in range(n):
            for j in range(i + 1, n):
                for d in ds:
                    if h[i]+d < 0 or h[i]+d > 1 or h[j]-d < 0 or h[j]-d > 1:
                        continue
                    h[i] += d
                    h[j] -= d
                    c = compute_c5(h)
                    h[i] -= d
                    h[j] += d
                    if c < bc:
                        bc, bm = c, (i, j, d)
        if bm is None:
            break
        i, j, d = bm
        h[i] += d
        h[j] -= d
        cur = bc
    return h, cur

        c = mask.sum()
        if c > 0:
