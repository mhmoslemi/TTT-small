def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_for_lag(h, lag):
    N = len(h)
    g = np.zeros(N)
    if lag >= 0:
        if lag < N:
            g[lag:] += 1.0 - h[:N-lag]
            g[:N-lag] -= h[lag:]
    else:
        p = -lag
        if p < N:
            g[:N-p] += 1.0 - h[p:]
            g[p:] -= h[:N-p]
    return g

def soft_gradient(h, corr, dx, alpha):

    return project_feasible(h, n_dst)

def binary_local_search(h, N, rng, t_end, max_sweeps=50):
    best_h = h.copy()
    best_val, _, _ = evaluate(best_h)
    for _ in range(max_sweeps):
        if time.time() > t_end:
            break
        improved = False
        ones = np.where(best_h > 0.5)[0]
        zeros = np.where(best_h <= 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for i in range(len(ones)):
            if time.time() > t_end:
                return best_h, best_val
            i_idx = ones[rng.randint(len(ones))]
            j_idx = zeros[rng.randint(len(zeros))]
            trial = best_h.copy()
            trial[i_idx] = 0.0
            trial[j_idx] = 1.0
            val_t, _, _ = evaluate(trial)
            if val_t < best_val - 1e-12:
                best_h = trial
                best_val = val_t
                improved = True
        if not improved:
            break
    return best_h, best_val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 15
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None
    best_box = {'h': None, 'val': float('inf')}
    def update_best(h, val):
        if val < best_box['val']:
            best_box['val'] = val
            best_box['h'] = h.copy()
    if h_init is not None:
        update_best(h_init, evaluate(h_init)[0])
    N_parent = len(h_init) if h_init is not None else 120
    # Phase 1: Optimize at parent resolution
    if h_init is not None and time.time() < t_end - 80:
        t_phase = min(time.time() + 100, t_end - 60)
        h, v = continuous_optimize(h_init.copy(), N_parent, rng, t_phase, update_best)
        update_best(h, v)
    # Phase 2: Try multiple resolutions
    n_values = [60, 80, 100, 120, 140, 160, 200]
    for n_t in n_values:
        if time.time() > t_end - 50:
            break
        if best_box['h'] is not None:
            h_start = interpolate_to_n(best_box['h'], n_t)
        else:
            h_start = project_feasible(0.5 + 0.1 * rng.randn(n_t), n_t)
        t_phase = min(time.time() + 45, t_end - 35)
        h, v = continuous_optimize(h_start, n_t, rng, t_phase, update_best)
        update_best(h, v)
    # Phase 3: Binary local search on best even-length solution
    if best_box['h'] is not None and time.time() < t_end - 20:
        N_best = len(best_box['h'])
        if N_best % 2 == 0:
            h_bin = (best_box['h'] > 0.5).astype(float)
            if abs(h_bin.sum() - N_best / 2.0) < 0.01:
                t_bl = min(time.time() + 25, t_end - 10)
                h_bl, v_bl = binary_local_search(h_bin, N_best, rng, t_bl, max_sweeps=80)
                update_best(h_bl, v_bl)
    # Phase 4: Fine-tuning with high alpha
    if best_box['h'] is not None and time.time() < t_end:
        N_best = len(best_box['h'])
        h = best_box['h'].copy()
        val, corr, dx = evaluate(h)
        lr = 0.0015
        for it in range(300000):
            if time.time() > t_end:
                break
            alpha = 50.0 + 30.0 * min(float(it) / 50000.0, 1.0)
            g = soft_gradient(h, corr, dx, alpha)
            h_new = project_feasible(h - lr * g, N_best)
            val_new, corr_new, _ = evaluate(h_new)
            if val_new < val - 1e-15:
                h = h_new
                val = val_new
                corr = corr_new
                update_best(h, val)
                lr = min(lr * 1.005, 0.006)
            else:
                lr *= 0.94
    # Phase 5: Multi-restart on best N with varied structure
    if best_box['h'] is not None and time.time() < t_end - 10:
        N_best = len(best_box['h'])
        for r in range(5):
            if time.time() > t_end - 5:
                break
            if r % 2 == 0:
                sc = 0.04 + 0.02 * r
                h_s = project_feasible(best_box['h'] + sc * rng.randn(N_best), N_best)
            else:
                freq = 2 + r
                h_s = project_feasible(best_box['h'] + 0.03 * np.sin(2 * np.pi * freq * np.arange(N_best) / N_best), N_best)
            t_p = min(time.time() + 20, t_end - 5)
            h, v = continuous_optimize(h_s, N_best, rng, t_p, update_best, max_iter=30000)
            update_best(h, v)
    if best_box['h'] is None:
        best_box['h'] = project_feasible(0.5 * np.ones(N_parent), N_parent)
        best_box['val'] = evaluate(best_box['h'])[0]
    h_final = project_feasible(best_box['h'], len(best_box['h']))
    val_final = evaluate(h_final)[0]
    return h_final, val_final, len(h_final)
