Let me be more careful about the Nesterov implementation. The standard Nesterov accelerated gradient is:
- v_{t+1} = beta * v_t
- x_{t+1} = x_t + v_{t+1} - alpha_t * grad(f, x_t + v_{t+1})

Or equivalently:
- y_t = x_t + (beta_t - 1) * (x_t - x_{t-1})
- x_{t+1} = y_t - alpha_t * grad(f, y_t)

Let me use the second form. With projection onto the feasible set.

Actually, for constrained problems, the "projected" Nesterov method is:
- y_t = P(x_t + (t-1)/(t+beta-1) * (x_t - x_{t-1}))
- x_{t+1} = P(y_t - (1/(L*t)) * grad(f, y_t))

where P is the projection and L is the Lipschitz constant.

But this requires knowing L, which is hard here. Let me just use a simpler momentum approach with adaptive step size.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def grad_at_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    return t * (m / t + np.log(np.sum(exps))) * dx

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w > 1e-10:
            shift = idx - (n - 1)
            grad += w * grad_at_shift(h, shift)
    return grad * dx

def nesterov_opt(h_init, n, t_init=0.03, t_final=0.002, max_iter=30000, lr_init=0.02):
    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    t = t_init
    beta = 0.9
    
    for it in range(max_iter):
        momentum = (it + 1) / (it + 1 + beta)
        y = h + (momentum - 1) * (h - h_prev)
        y = project_feasible(y, n)
        
        conv = np.correlate(y, 1.0 - y, mode="full")
        m = np.max(conv)
        exps = np.exp((conv - m) / t)
        weights = exps / np.sum(exps)
        grad = np.zeros(n)
        dx = 2.0 / n
        for idx in range(len(conv)):
            w = weights[idx]
            if w > 1e-10:
                shift = idx - (n - 1)
                grad += w * grad_at_shift(y, shift)
        grad *= dx
        
        h_prev = h.copy()
        h_new = y - lr * grad
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        c5_y = compute_c5(y)
        
        if c5_new <= c5_y + 1e-12:
            h = h_new
        else:
            lr *= 0.5
            if lr < 1e-8:
                lr = 1e-8
                h = project_feasible(y - lr * grad, n)
            else:
                h = y
        
        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            best_h = h.copy()
        
        t = max(t * 0.9997, t_final)
        lr = min(lr * 1.005, 0.05) if not (c5_new > c5_y + 1e-12) else lr
    
    return best_h, best_c5

def local_perturb(h, n, n_trials=500, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        idx = rng.choice(n, 2, replace=False)
        i, j = int(idx[0]), int(idx[1])
        delta = rng.uniform(0.01, 0.15)
        if rng.rand() < 0.5:
            delta = -delta
        if h[i] + delta < 0 or h[i] + delta > 1 or h[j] - delta < 0 or h[j] - delta > 1:
            continue
        h_test = h.copy()
        h_test[i] += delta
        h_test[j] -= delta
        c5_test = compute_c5(h_test)
        if c5_test < best_c5 - 1e-12:
            best_h = h_test
            best_c5 = c5_test
            h = h_test
    return best_h, best_c5

def coordinate_refine(h, n, max_passes=50):
    h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for _ in range(max_passes):
        conv = np.correlate(h, 1.0 - h, mode="full")
        idx_max = np.argmax(conv)
        k_star = idx_max - (n - 1)
        g = grad_at_shift(h, k_star)
        improved = False
        order = np.argsort(g)
        for pos in range(n):
            j = order[pos]
            if g[j] > 0.01:
                break
            for pos2 in range(n - 1 - pos, -1, -1):
                i = order[n - 1 - pos2]
                if g[i] < -0.01:
                    break
                max_d = min(1.0 - h[j], h[i])
                if max_d < 0.005:
                    continue
                for frac in [0.3, 0.7, 1.0]:
                    d = max_d * frac
                    h_t = h.copy()
                    h_t[j] += d
                    h_t[i] -= d
                    c5_t = compute_c5(h_t)
                    if c5_t < best_c5 - 1e-12:
                        h = h_t
                        best_c5 = c5_t
                        improved = True
                        break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - t0)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(50, 0.5)
    
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Nesterov at original resolution with multiple t schedules
    for t_init, t_fin, lr_i in [(0.05, 0.005, 0.015), (0.03, 0.003, 0.02), (0.02, 0.002, 0.025)]:
        if time_left() < 60:
            break
        ht, c5t = nesterov_opt(best_h, n0, t_init=t_init, t_final=t_fin, max_iter=25000, lr_init=lr_i)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    
    # Phase 2: Coordinate refinement on active shift
    if time_left() > 30:
        ht, c5t = coordinate_refine(best_h, n0, max_passes=80)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    
    # Phase 3: Local perturbation search
    if time_left() > 30:
        ht, c5t = local_perturb(best_h, n0, n_trials=2000, rng=rng)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        ht, c5t = local_perturb(best_h, n0, n_trials=2000, rng=rng)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    
    # Phase 4: Increase resolution
    for n_try in [400, 480, 600, 800, 1000]:
        if time_left() < 120:
            break
        nc = len(best_h)
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        
        ht, c5t = nesterov_opt(hi, n_try, t_init=0.03, t_final=0.002, max_iter=20000, lr_init=0.015)
        if time_left() > 60:
            ht2, c5t2 = coordinate_refine(ht, n_try, max_passes=30)
            if c5t2 < c5t:
                ht, c5t = ht2, c5t2
        if time_left() > 30:
            ht3, c5t3 = local_perturb(ht, n_try, n_trials=1000, rng=rng)
            if c5t3 < c5t:
                ht, c5t = ht3, c5t3
        
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    
    # Phase 5: Final Nesterov polish at best resolution
    if time_left() > 30:
        n_best = len(best_h)
        ht, c5t = nesterov_opt(best_h, n_best, t_init=0.01, t_final=0.001, max_iter=30000, lr_init=0.01)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        ht, c5t = coordinate_refine(best_h, n_best, max_passes=50)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    
    # Phase 6: One more local perturbation pass
    if time_left() > 10:
        ht, c5t = local_perturb(best_h, len(best_h), n_trials=3000, rng=rng)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    
    return best_h, best_c5, len(best_h)
