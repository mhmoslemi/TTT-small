One more thought: the problem has a specific structure. The overlap at lag k involves pairs (i, i+k). This means the problem has a "banded" structure - each coordinate h[i] interacts with h[i±k] for all k. This is different from, say, an Ising model where only nearest neighbors interact.

For the gradient, this means each coordinate's gradient depends on all other coordinates (through the different lags). So the problem is fully coupled, which makes it hard to solve.

However, for the binary case, the local search (swap-based) naturally handles this because a swap only affects a limited number of lags (those involving the two swapped positions).

import numpy as np
import time
import scipy.optimize as so
import warnings
warnings.simplefilter("ignore")

def _repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * n
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(60):
        diff = target - x.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            slack = 1.0 - x
            mask = slack > 1e-14
            if not mask.any():
                break
            x[mask] += diff * slack[mask] / slack[mask].sum()
        else:
            mask = x > 1e-14
            if not mask.any():
                break
            x[mask] += diff * x[mask] / x[mask].sum()
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        x[:] = 0.0
        k = int(target)
        x[:k] = 1.0
        rem = target - k
        if rem > 0 and k < n:
            x[k] = min(rem, 1.0)
    return x

def _compute_c(h):
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def _exact_c(h):
    return float(np.max(_compute_c(h)))

def _grad_lag(h, k):
    n = h.size
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        for j in range(k, n):
            grad[j] += dx * (1.0 - h[j - k])
        for j in range(0, n - k):
            grad[j] -= dx * h[j + k]
    else:
        kk = -k
        for j in range(0, n - kk):
            grad[j] += dx * (1.0 - h[j + kk])
        for j in range(kk, n):
            grad[j] -= dx * h[j - kk]
    return grad

def _binary_local_search(h, iters=500):
    x = np.round(h)
    n = x.size
    x = _repair(x)
    best_c = _exact_c(x)
    best_x = x.copy()
    ones = np.where(x == 1)[0]
    zeros = np.where(x == 0)[0]
    for _ in range(iters):
        improved = False
        rng = np.random.RandomState(_)
        for _ in range(min(200, len(ones) * len(zeros))):
            if len(ones) == 0 or len(zeros) == 0:
                break
            p = ones[rng.randint(len(ones))]
            q = zeros[rng.randint(len(zeros))]
            x[p] = 0.0
            x[q] = 1.0
            c = _exact_c(x)
            if c < best_c - 1e-14:
                best_c = c
                best_x = x.copy()
                ones = np.where(x == 1)[0]
                zeros = np.where(x == 0)[0]
                improved = True
            else:
                x[p] = 1.0
                x[q] = 0.0
        if not improved:
            break
    return best_x, best_c

def _proj_grad_descent(h0, time_limit, seed=42):
    x = _repair(h0)
    n = x.size
    best_x = x.copy()
    best_c = _exact_c(x)
    deadline = time.time() + max(0.1, time_limit)
    rng = np.random.RandomState(seed)
    step = 0.15
    no_imp = 0
    kick_cd = 0
    while time.time() < deadline:
        cvals = _compute_c(h=x)
        C = float(np.max(cvals))
        if C < best_c - 1e-14:
            best_c = C
            best_x = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 30:
            kick_cd += 1
            if kick_cd > 5:
                break
            scale = 0.03 * (1 + kick_cd * 0.5)
            x = _repair(x + rng.normal(0, scale, n))
            no_imp = 0
            continue
        active_idx = int(np.argmax(cvals))
        k_shift = active_idx - (n - 1)
        grad = _grad_lag(x, k_shift)
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        accepted = False
        for _ in range(20):
            y = _repair(x - step * grad / gn)
            cy = _exact_c(y)
            if cy < C - 1e-13:
                x = y
                accepted = True
                break
            step *= 0.6
            if step < 1e-10:
                break
        if not accepted:
            no_imp += 1
            if rng.rand() < 0.3:
                step = min(step * 1.5, 0.3)
        else:
            step = min(step * 1.05, 0.3)
        if rng.rand() < 0.08:
            y = _repair(x + rng.normal(0, 0.005, n))
            cy = _exact_c(y)
            if cy < best_c:
                best_c = cy
                best_x = y.copy()
                x = y
                no_imp = 0
    return best_x, best_c

def _simulated_annealing(h0, time_limit, seed=42):
    x = _repair(h0)
    n = x.size
    best_x = x.copy()
    best_c = _exact_c(x)
    deadline = time.time() + max(0.1, time_limit)
    rng = np.random.RandomState(seed)
    T = 0.05
    cool = 0.998
    cur_c = best_c
    while time.time() < deadline:
        delta = rng.normal(0, 0.04, n)
        y = _repair(x + delta)
        cy = _exact_c(y)
        diff = cy - cur_c
        if diff < 0 or rng.rand() < np.exp(-diff / max(T, 1e-8)):
            x = y
            cur_c = cy
            if cy < best_c:
                best_c = cy
                best_x = y.copy()
        T *= cool
        if T < 0.001:
            break
    return best_x, best_c

def _structured_start(n, seed, freq=2, amp=0.35):
    rng = np.random.RandomState(seed)
    t = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    for f in (freq, 2 * freq, 3 * freq):
        h += (amp / f) * 0.5 * np.sin(2 * np.pi * f * t / 2.0 + rng.uniform(0, 2 * np.pi))
    h += 0.02 * rng.normal(0, 1, n)
    return _repair(h)

def _resample(h0, n_new):
    n0 = h0.size
    if n0 == n_new:
        return _repair(h0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return _repair(np.interp(x1, x0, h0))

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    sv = int(seed)
    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)
    
    if init is not None and init.size >= 10:
        base_n = int(init.size)
    else:
        base_n = 70
    
    best_x = None
    best_c = 1e9
    
    # Build list of (n, initial) pairs
    candidates = []
    if init is not None:
        candidates.append((base_n, _repair(init)))
    
    # Try nearby n values
    for n_try in [base_n, base_n - 10, base_n + 10, 50, 60, 80, 90, 100]:
        if n_try < 20 or n_try > 200 or n_try % 2 != 0:
            continue
        if (n_try, None) in [(c[0], None) for c in candidates]:
            continue
        if init is not None:
            candidates.append((n_try, _resample(init, n_try)))
        else:
            candidates.append((n_try, _structured_start(n_try, sv + n_try, 2)))
    if len(candidates) > 7:
        candidates = candidates[:7]
    
    # Phase 1: Quick gradient descent on all candidates (30% budget)
    t1 = (deadline - time.time()) * 0.30
    per1 = max(2.0, t1 / len(candidates))
    for i, (n, h0) in enumerate(candidates):
        rem = deadline - time.time()
        if rem < 5:
            break
        alloc = min(per1, rem * 0.8 / max(1, len(candidates) - i))
        x, c = _proj_grad_descent(h0, alloc, sv + 100 * i + n)
        if c < best_c:
            best_c = c
            best_x = x
    
    # Phase 2: Binary local search + continuous polish (35% budget)
    t2_end = deadline - (deadline - time.time()) * 0.65
    x_work = best_x.copy()
    while time.time() < t2_end:
        # Binarize and do binary local search
        bx, bc = _binary_local_search(x_work, iters=300)
        if bc < best_c:
            best_c = bc
            best_x = bx
        # Continuous polish from binary
        x2, c2 = _proj_grad_descent(bx, min(3.0, (t2_end - time.time()) * 0.3), sv + 999)
        if c2 < best_c:
            best_c = c2
            best_x = x2
        x_work = best_x
        # Small perturbation to escape
        rng = np.random.RandomState(int(time.time() * 1000) % 100000)
        x_work = _repair(best_x + rng.normal(0, 0.02, best_x.size))
    
    # Phase 3: Simulated annealing from best (20% budget)
    t3_end = deadline - (deadline - time.time()) * 0.20
    x3, c3 = _simulated_annealing(best_x, min(15.0, max(1.0, t3_end - time.time())), sv + 777)
    if c3 < best_c:
        best_c = c3
        best_x = x3
    # Polish after SA
    x4, c4 = _proj_grad_descent(x3, min(8.0, max(1.0, t3_end - time.time())), sv + 888)
    if c4 < best_c:
        best_c = c4
        best_x = x4
    
    # Phase 4: Final multi-kick local search (remaining budget)
    while time.time() < deadline - 2.0:
        rem = deadline - time.time()
        if rem < 3:
            break
        # Try several kicks
        improved = False
        for kick in range(5):
            rng = np.random.RandomState(sv + int(time.time() * 100) % 99999 + kick)
            scale = 0.01 + 0.02 * kick
            xk = _repair(best_x + rng.normal(0, scale, best_x.size))
            xk, ck = _proj_grad_descent(xk, min(4.0, rem * 0.15), sv + kick * 31)
            if ck < best_c - 1e-12:
                best_c = ck
                best_x = xk
                improved = True
                break
        if not improved:
            # Try binary search
            bx, bc = _binary_local_search(best_x, iters=200)
            if bc < best_c - 1e-12:
                best_c = bc
                best_x = bx
                x2, c2 = _proj_grad_descent(bx, min(3.0, rem * 0.2), sv + 555)
                if c2 < best_c:
                    best_c = c2
                    best_x = x2
            else:
                break
    
    best_x = _repair(best_x)
    best_c = _exact_c(best_x)
    return best_x, best_c, int(best_x.size)
