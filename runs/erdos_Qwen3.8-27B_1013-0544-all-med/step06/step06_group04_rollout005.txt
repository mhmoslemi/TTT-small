def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def fix_sum(h, target):
    h = h.copy()
    np.clip(h, 0, 1, out=h)
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        mask = h < 1.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    else:
        mask = h > 0.0
        if mask.sum() == 0:
            return h
        h[mask] += diff / mask.sum()
    np.clip(h, 0, 1, out=h)
    # Final fix
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = h < 1.0
            idx = np.where(mask)[0]
            h[idx[0]] += min(residual, 1.0 - h[idx[0]])
            remaining = residual - (h[idx[0]] - min(h[idx[0]], h[idx[0]]))
        else:
            mask = h > 0.0
            idx = np.where(mask)[0]
            h[idx[0]] += max(residual, -h[idx[0]])
    return h

def project_simplex_box(v, target):
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(v - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, "full") * dx
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

        return base + 1e6 * sum_err * sum_err

Actually the more precise statement: if x_1 is the max and the rest are much smaller, then log(sum(exp(βx_i)))/β ≈ x_1 + (1/β)log(1 + sum_{i>1} exp(β(x_i-x_1))) ≈ x_1 + (1/β)log(1) = x_1 when the gap is large.

The issue is when multiple terms are close to the max. In that case, the smooth approximation is the average, which is what we want - it's a smooth surrogate that's close to the max.

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: L-BFGS-B at original resolution
    if rem() > 80:
        h_opt, c5v = lbgfgs_opt(best_h, max_iter=4000, beta=50.0)
        if c5v < best_c5:
            best_h, best_c5 = h_opt, c5v

    # Phase 2: Subgradient refinement at original resolution
    if rem() > 80:
        h_opt, c5v = subgrad_opt(best_h, max_iter=8000)
        if c5v < best_c5:
            best_h, best_c5 = h_opt, c5v

    # Phase 3: Upsample to 2x, optimize
    if rem() > 150:
        h_up = upsample(best_h, 2)
        n_up = len(h_up)
        h_up = fix_sum(h_up, n_up // 2)
        h_opt, c5v = lbgfgs_opt(h_up, max_iter=2500, beta=45.0)
        if c5v < best_c5:
            best_h, best_c5 = h_opt, c5v
        if rem() > 80:
            h_opt, c5v = subgrad_opt(best_h, max_iter=6000)
            if c5v < best_c5:
                best_h, best_c5 = h_opt, c5v

    # Phase 4: Interpolated upsample to 2x, optimize
    if rem() > 120:
        h_interp = np.interp(np.arange(n0 * 2) * (1.0 / n0), np.arange(n0) * (2.0 / n0), best_h, left=0, right=0)
        h_interp = fix_sum(h_interp, n0)
        h_opt, c5v = lbgfgs_opt(h_interp, max_iter=2000, beta=40.0)
        if c5v < best_c5:
            best_h, best_c5 = h_opt, c5v

    # Phase 5: Upsample to 4x original (or 2x of current)
    if rem() > 100:
        factor = 2
        h_up = upsample(best_h, factor)
        n_up = len(h_up)
        h_up = fix_sum(h_up, n_up // 2)
        h_opt, c5v = lbgfgs_opt(h_up, max_iter=1500, beta=35.0)
        if c5v < best_c5:
            best_h, best_c5 = h_opt, c5v
        if rem() > 40:
            h_opt, c5v = subgrad_opt(best_h, max_iter=4000)
            if c5v < best_c5:
                best_h, best_c5 = h_opt, c5v

    # Phase 6: Random restarts at original resolution
    n_restarts = 8
    for _ in range(n_restarts):
        if rem() < 30:
            break
        h_rand = random_h(n0, rng)
        h_opt, c5v = lbgfgs_opt(h_rand, max_iter=2000, beta=50.0)
        if c5v < best_c5:
            best_h, best_c5 = h_opt, c5v
            if rem() > 60:
                h_opt2, c5v2 = subgrad_opt(best_h, max_iter=5000)
                if c5v2 < best_c5:
                    best_h, best_c5 = h_opt2, c5v2

    # Phase 7: Final polish with subgradient at best resolution
    if rem() > 30:
        h_opt, c5v = subgrad_opt(best_h, max_iter=5000)
        if c5v < best_c5:
            best_h, best_c5 = h_opt, c5v

    return best_h, best_c5, len(best_h)
