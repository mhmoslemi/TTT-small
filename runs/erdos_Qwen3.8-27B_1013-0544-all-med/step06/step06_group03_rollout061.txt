    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    f = conv * dx  # shape (2n-1,)

def _conv_grad_single(h, shift):

    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        if shift < n:

        g[s:] -= h[:n - s]

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def dual_ascent_optimize(h0, n, max_iter, lr_init, beta_init, momentum=0.9):
    """Primal-dual optimization with exponential weight updates."""
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)  # momentum
    
    # Initialize weights uniformly over shifts
    n_shifts = 2 * n - 1
    
    for it in range(max_iter):

        dx = 2.0 / n
        conv = np.correlate(h, 1.0 - h, mode="full")
        f = conv * dx  # shape (2n-1,)

        if c5 < best_c5 - 1e-14:

            best_h = h.copy()

        # Use a stable version: subtract max for numerical stability
        f_max = f.max()
        beta = beta_init * (0.999 ** (it // 100))  # decay beta
        p = np.exp((f - f_max) / beta)
        p /= p.sum()
        
        # Compute weighted gradient

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def conv_all(h):
    """Compute all correlation values (not just the max)."""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        if shift < n:
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def dual_ascent(h0, n, max_iter, lr_init, beta, momentum=0.9, time_limit=None, t0=None):
    """Dual-ascent (primal-dual) optimization."""
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    dx = 2.0 / n
    n_shifts = 2 * n - 1
    
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        
        # Compute all correlations
        conv = np.correlate(h, 1.0 - h, mode="full")
        f = conv * dx
        c5 = f.max()
        
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        
        # Update weights (exponential tilting)
        f_shifted = f - f.max()
        p = np.exp(f_shifted / beta)
        p_sum = p.sum()
        p /= p_sum
        
        # Compute weighted gradient (only active shifts)
        grad = np.zeros(n)
        # Sort by weight and take top ones
        threshold = 0.01 / p_sum  # top ~1% or so
        active_mask = p > threshold
        for idx in np.where(active_mask)[0]:
            shift = idx - (n - 1)
            grad += p[idx] * _conv_grad_single(h, shift)
        grad *= dx
        
        # Nesterov momentum update
        lr = lr_init * (0.9995 ** it)
        v = momentum * v + grad
        h_new = project_feasible(h - lr * v, n)
        
        # Check
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 100 * lr * np.max(np.abs(grad)):
            # Step too large, reduce
            h_new = project_feasible(h - 0.5 * lr * v, n)
            c5_new = compute_c5(h_new)
            if c5_new > c5:
                v *= 0.5
                h_new = h
        h = h_new
    
    return best_h, best_c5

def best_shift_pgd(h0, n, max_iter=50000, lr_init=0.005, time_limit=None, t0=None):
    """PGD using only the best (worst) shift at each step."""
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time_limit is not None and (time.time() - t0) > time_limit:
            break
        conv = np.correlate(h, 1.0 - h, mode="full")
        c5 = conv.max() * dx
        idx = np.argmax(conv)
        shift = idx - (n - 1)
        grad = _conv_grad_single(h, shift) * dx
        
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        
        lr = lr_init * (0.9998 ** it)
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new <= c5 + 1e-15:
            h = h_new
        else:
            # Try multiple shifts (top 3)
            top3 = np.argsort(conv)[-3:]
            improved = False
            for idx2 in top3:
                shift2 = idx2 - (n - 1)
                grad2 = _conv_grad_single(h, shift2) * dx
                h_try = project_feasible(h - lr * grad2, n)
                if compute_c5(h_try) < c5:
                    h = h_try
                    improved = True
                    break
            if not improved:
                lr *= 0.5
                if lr < 1e-10:
                    break
    
    return best_h, best_c5
