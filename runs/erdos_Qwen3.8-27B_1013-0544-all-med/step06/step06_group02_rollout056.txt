        g[:N-lag] += (1.0 - h[lag:N])
        g[lag:] -= h[:N-lag]
    else:
        k = -lag
        g[k:] += (1.0 - h[:N-k])
        g[:N-k] -= h[k:]
    return g

def active_set_grad(h, corr, dx, eps=1e-3):
    """Gradient using uniform average over active set (top lags within eps of max)"""
    N = len(h)
    max_c = corr.max()
    active = np.where(corr > max_c - eps)[0]
    g = np.zeros(N)
    for j in active:
        lag = j - (N - 1)
        g += grad_at_lag(h, lag, N)
    g /= len(active)
    return g * dx

def softmax_grad(h, corr, dx, alpha):
    N = len(h)
    max_c = corr.max()
    w = np.exp(alpha * (corr - max_c))
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        return grad_at_lag(h, k_star - (N-1), N) * dx
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-6:
            continue
        lag = j - (N - 1)
        g += w[j] * grad_at_lag(h, lag, N)
    return g * dx

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def obj_val(h):
    N = len(h)
    corr = compute_corr(h)
    return corr.max() * 2.0 / N

def grad_at_lag(h, lag, N):
    g = np.zeros(N)
    if lag >= 0:
        g[:N-lag] += (1.0 - h[lag:N])
        g[lag:] -= h[:N-lag]
    else:
        k = -lag
        g[k:] += (1.0 - h[:N-k])
        g[:N-k] -= h[k:]
    return g

def weighted_grad(h, corr, dx, alpha):
    N = len(h)
    max_c = corr.max()
    c = corr - max_c
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        return grad_at_lag(h, k_star - (N-1), N) * dx
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-7:
            continue
        lag = j - (N - 1)
        g += w[j] * grad_at_lag(h, lag, N)
    return g * dx

def paired_cd_val(h, m, m2, delta):
    h[m] += delta
    h[m2] -= delta
    val = obj_val(h)
    h[m] -= delta
    h[m2] += delta
    return val

def paired_cd_optimize(h, N, rng, t_end, try_update, max_sweeps=500):
    best_val = obj_val(h)
    best_h = h.copy()
    try_update(best_h, best_val)
    dx = 2.0 / N
    
    for sweep in range(max_sweeps):
        if time.time() > t_end:
            break
        n_pairs = min(500, N * 4)
        for _ in range(n_pairs):
            if time.time() > t_end:
                break
            m = rng.randint(N)
            m2 = rng.randint(N)
            if m == m2:
                continue
            lo_d = -min(h[m2], 1.0 - h[m])
            hi_d = min(h[m2], 1.0 - h[m])
            if hi_d - lo_d < 1e-12:

            a, b = lo_d, hi_d
            for _ in range(25):

                m2_ = b - (b - a) / 3.0
                v1 = paired_cd_val(h, m, m2, m1)
                v2 = paired_cd_val(h, m, m2, m2_)
                if v1 < v2:

                else:
                    a = m1
            d_opt = (a + b) / 2.0

                try_update(best_h, best_val)
    return best_h, best_val

def gradient_descent(h, N, rng, t0, t_end, try_update, max_iter=100000):
    val = obj_val(h)
    corr = compute_corr(h)
    dx = 2.0 / N
    try_update(h.copy(), val)
    best_local = val
    best_local_h = h.copy()
    lr = 0.04
    alpha = 12.0
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = weighted_grad(h, corr, dx, alpha)
        
        step = lr
        improved = False
        for _ in range(12):
            h_new = project_feasible(h - step * g, N)
            corr_new = compute_corr(h_new)
            val_new = corr_new.max() * dx
            if val_new < val - 1e-15:
                h = h_new
                val = val_new
                corr = corr_new
                improved = True
                break
            step *= 0.5
        
        if improved:
            stagnant = 0
            lr = min(lr * 1.04, 0.15)
            alpha = max(alpha * 0.97, 5.0)
            if val < best_local:
                best_local = val
                best_local_h = h.copy()
        else:
            stagnant += 1
            lr *= 0.7
            alpha = min(alpha * 1.25, 80.0)
            if stagnant > 25:
                scale = 0.02 + 0.06 * min(stagnant / 80.0, 1.0)
                h = project_feasible(h + rng.randn(N) * scale, N)
                corr = compute_corr(h)
                val = corr.max() * dx
                lr = 0.04
                alpha = 12.0
                stagnant = 0
                if val < best_local:
                    best_local = val
                    best_local_h = h.copy()
    
    try_update(best_local_h, best_local)
    return best_local_h, best_local

def interpolate_to_n(h_src, n_dst):
    x0 = np.arange(len(h_src)) / len(h_src)
    x1 = np.arange(n_dst) / n_dst
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def fine_polish(h, N, t_end, try_update):
    val = obj_val(h)
    corr = compute_corr(h)
    dx = 2.0 / N
    try_update(h.copy(), val)
    lr = 0.005
    alpha = 40.0
    for it in range(100000):
        if time.time() > t_end:
            break
        g = weighted_grad(h, corr, dx, alpha)
        h_new = project_feasible(h - lr * g, N)
        corr_new = compute_corr(h_new)
        val_new = corr_new.max() * dx
        if val_new < val - 1e-16:
            h = h_new
            val = val_new
            corr = corr_new
            try_update(h.copy(), val)
        else:
            lr *= 0.92
            if lr < 1e-6:
                break
    return h, val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 15
    
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None
    
    best_h = None
    best_val = float('inf')
    
    def try_update(h, val):
        nonlocal best_h, best_val
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    if h_init is not None:
        try_update(h_init.copy(), obj_val(h_init))
    
    # Phase 1: Gradient descent on initial (n=120)
    if h_init is not None and time.time() < t_end - 200:
        t_phase = min(time.time() + 250, t_end - 200)
        h, v = gradient_descent(h_init.copy(), len(h_init), rng, t0, t_phase, try_update, max_iter=80000)
        try_update(h, v)
    
    # Phase 2: Paired coordinate descent polishing
    if best_h is not None and time.time() < t_end - 150:
        N_best = len(best_h)
        t_phase = min(time.time() + 180, t_end - 150)
        h, v = paired_cd_optimize(best_h.copy(), N_best, rng, t_phase, try_update, max_sweeps=200)
        try_update(h, v)
    
    # Phase 3: Try other n values
    n_list = [96, 144, 160, 200]
    for n_t in n_list:
        if time.time() > t_end - 80:
            break
        if best_h is not None:
            h_start = interpolate_to_n(best_h, n_t)
        elif h_init is not None:
            h_start = interpolate_to_n(h_init, n_t)
        else:
            h_start = project_feasible(0.5 + 0.1 * rng.randn(n_t), n_t)
        t_phase = min(time.time() + 60, t_end - 80)
        h, v = gradient_descent(h_start, n_t, rng, t0, t_phase, try_update, max_iter=30000)
        try_update(h, v)
    
    # Phase 4: More paired CD on best
    if best_h is not None and time.time() < t_end - 60:
        N_best = len(best_h)
        t_phase = min(time.time() + 80, t_end - 60)
        h, v = paired_cd_optimize(best_h.copy(), N_best, rng, t_phase, try_update, max_sweeps=300)
        try_update(h, v)
    
    # Phase 5: Random restarts from best
    if best_h is not None and time.time() < t_end - 40:
        N_best = len(best_h)
        for restart in range(15):
            if time.time() > t_end - 40:
                break
            scale = 0.03 + 0.02 * restart
            h_start = project_feasible(best_h + rng.randn(N_best) * scale, N_best)
            t_phase = min(time.time() + 30, t_end - 40)
            h, v = gradient_descent(h_start, N_best, rng, t0, t_phase, try_update, max_iter=20000)
            try_update(h, v)
    
    # Phase 6: Final fine polish
    if best_h is not None and time.time() < t_end:
        N_best = len(best_h)
        h, v = fine_polish(best_h.copy(), N_best, t_end, try_update)
        try_update(h, v)
    
    # Phase 7: One more paired CD
    if best_h is not None and time.time() < t_end - 5:
        N_best = len(best_h)
        h, v = paired_cd_optimize(best_h.copy(), N_best, rng, t_end, try_update, max_sweeps=100)
        try_update(h, v)
    
    if best_h is None:
        best_h = project_feasible(0.5 + 0.1 * rng.randn(47), 47)
        best_val = obj_val(best_h)
    
    return best_h, best_val, len(best_h)
