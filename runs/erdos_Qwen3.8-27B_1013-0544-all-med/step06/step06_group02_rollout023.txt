import numpy as np

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_obj(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr.max() * dx, corr, dx)

def grad_for_lag(h, s):
    N = len(h)
    g = np.zeros(N)
    if s >= 0:
        m = N - s
        g[:m] += 1.0 - h[s:s+m]
        if s > 0:
            g[s:] -= h[:m]
    else:
        t = -s
        g[t:] += 1.0 - h[:N-t]
        g[:N-t] -= h[t:]
    return g

def top_grad(h, corr, dx, m=8):
    N = len(h)
    top = np.argsort(corr)[-m:]
    max_c = corr[top].max()
    w = np.exp(30.0 * (corr[top] - max_c))
    w /= w.sum()
    g = np.zeros(N)
    for j in range(m):
        if w[j] < 1e-8:
            continue
        s = top[j] - (N - 1)
        g += w[j] * grad_for_lag(h, s)
    return g * dx

            coeff[mask1] += 1.0 - h[idx1]
            mask2 = (ks >= i) & (ks <= i+N-1)

                corr += best_x * coeff

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_obj(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr.max() * dx, corr, dx)

def coord_descent_pass(h, N, rng, grid=40):
    """One pass of coordinate descent with 1D minimization per coordinate."""
    val, corr, dx = compute_obj(h)
    ks = np.arange(2 * N - 1)
    order = rng.permutation(N)
    for i in order:
        old = h[i]
        n_vals = ks - (N - 1)
        idx1 = i - n_vals
        mask1 = (idx1 >= 0) & (idx1 < N)
        idx2 = i + n_vals
        mask2 = (idx2 >= 0) & (idx2 < N)
        coeff = np.zeros(2 * N - 1)
        coeff[mask1] += 1.0 - h[idx1[mask1]]
        coeff[mask2] -= h[idx2[mask2]]
        lo_x = -old
        hi_x = 1.0 - old
        if hi_x - lo_x < 1e-12:
            continue
        xs = np.linspace(lo_x, hi_x, grid)
        vals = np.empty(grid)
        for xi in range(grid):
            vals[xi] = (corr + xs[xi] * coeff).max()
        best_xi = np.argmin(vals)
        if 0 < best_xi < grid - 1 and vals[best_xi] < vals[0] and vals[best_xi] < vals[-1]:
            x_fine = np.linspace(xs[best_xi - 1], xs[best_xi + 1], 30)
            vf = np.empty(30)
            for xi in range(30):
                vf[xi] = (corr + x_fine[xi] * coeff).max()
            best_xi2 = np.argmin(vf)
            best_x = x_fine[best_xi2]
            new_max = vf[best_xi2]
        else:
            best_x = xs[best_xi]
            new_max = vals[best_xi]
        if new_max < corr.max() - 1e-13:
            h[i] = old + best_x
            corr += best_x * coeff
    return h, corr

def coord_optimize(h, N, rng, t_end, try_update, max_passes=50):
    """Multiple passes of coordinate descent with periodic re-projection."""
    h = h.copy()
    val, corr, dx = compute_obj(h)
    try_update(h, val)
    best_h = h.copy()
    best_val = val
    for p in range(max_passes):
        if time.time() > t_end:
            break
        h, corr = coord_descent_pass(h, N, rng)
        if (p + 1) % 5 == 0:
            h = project_feasible(h, N)
            val, corr, dx = compute_obj(h)
            if val < best_val:
                best_val = val
                best_h = h.copy()
                try_update(h, val)
        else:
            val, _, _ = (corr.max() * dx, 0, 0)
            if val < best_val:
                val_exact, _, _ = compute_obj(h)
                if val_exact < best_val:
                    best_val = val_exact
                    best_h = h.copy()
                    try_update(h, val_exact)
    h = project_feasible(h, N)
    val, _, _ = compute_obj(h)
    if val < best_val:
        best_h = h.copy()
        best_val = val
        try_update(h, val)
    return best_h, best_val

def grad_step(h, corr, dx, N, top_m=10):
    """Compute gradient using only top-m lags."""
    top = np.argsort(corr)[-top_m:]
    max_c = corr.max()
    w = np.exp(40.0 * (corr[top] - max_c))
    w /= w.sum()
    g = np.zeros(N)
    for j in range(top_m):
        if w[j] < 1e-8:
            continue
        s = top[j] - (N - 1)
        if s >= 0:
            m = N - s
            g[:m] += w[j] * (1.0 - h[s:s + m])
            if s > 0:
                g[s:] -= w[j] * h[:m]
        else:
            t = -s
            g[t:] += w[j] * (1.0 - h[:N - t])
            g[:N - t] -= w[j] * h[t:]
    return g * dx

def grad_optimize(h, N, rng, t_end, try_update, top_m=8):
    """Gradient descent with momentum and adaptive step."""
    val, corr, dx = compute_obj(h)
    try_update(h, val)
    lr = 0.015
    mom = np.zeros(N)
    beta = 0.85
    best_val = val
    best_h = h.copy()
    stag = 0
    while time.time() < t_end:
        g = grad_step(h, corr, dx, N, top_m)
        mom = beta * mom + (1 - beta) * g
        h_new = project_feasible(h - lr * mom, N)
        val_new, corr_new, _ = compute_obj(h_new)
        if val_new < val - 1e-14:
            h = h_new
            val = val_new
            corr = corr_new
            stag = 0
            lr = min(lr * 1.015, 0.06)
            if val < best_val:
                best_val = val
                best_h = h.copy()
                try_update(h, val)
        else:
            stag += 1
            lr *= 0.7
            if stag > 35:
                scale = 0.04 + 0.06 * min(stag / 100.0, 1.0)
                h = project_feasible(h + rng.randn(N) * scale, N)
                val, corr, _ = compute_obj(h)
                lr = 0.015
                mom = np.zeros(N)
                stag = 0
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
                    try_update(h, val)
    try_update(best_h, best_val)
    return best_h, best_val

def interp_to_n(h_src, n_dst):
    x0 = np.linspace(0, 1, len(h_src))
    x1 = np.linspace(0, 1, n_dst)
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)
