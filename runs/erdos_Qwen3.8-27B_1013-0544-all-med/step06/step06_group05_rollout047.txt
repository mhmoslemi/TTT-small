import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_grad_smooth(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr_max = np.max(corr)
    exp_corr = np.exp(beta * (corr - corr_max))
    p = exp_corr / np.sum(exp_corr)
    s = np.arange(2*n-1) - (n-1)
    grad = np.zeros(n)
    for t in range(2*n-1):
        if p[t] < 1e-10:
            continue
        st = s[t]
        # Add (1-h[i+st]) where i+st is in [0,n)
        lo1 = max(0, -st)
        hi1 = min(n, n-st)
        if lo1 < hi1:
            grad[lo1:hi1] += p[t] * (1.0 - h[lo1+st:hi1+st])
        # Subtract h[i-st] where i-st is in [0,n)
        lo2 = max(0, st)
        hi2 = min(n, n+st)
        if lo2 < hi2:
            grad[lo2:hi2] -= p[t] * h[lo2-st:hi2-st]
    grad *= dx
    return grad

def project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
        else:
            total = np.sum(h)
            if total > 1e-12:
                h -= (h / total) * (-diff)
    h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def gradient_descent(h0, max_time, beta_start=20.0, beta_end=200.0, seed=42):
    n = len(h0)
    h = h0.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    
    # Adaptive step size
    lr = 0.1
    momentum = 0.9
    v = np.zeros(n)
    
    n_iter = 0
    while (time.time() - start) < max_time:

        beta = beta_start + (beta_end - beta_start) * elapsed
        beta = max(beta, 1.0)
        
        grad = compute_c5_grad_smooth(h, beta)
        
        v = momentum * v - lr * grad
        h_new = h + v
        h_new = project(h_new, n)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5 + 1e-10:
            h = h_new

            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
            v *= 0.5
        
        n_iter += 1
        if n_iter % 1000 == 0 and (time.time() - start) > max_time * 0.8:
            break
    
    return best_h, best_c5

def compute_c5(h):

def c5_grad(h, beta):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = np.max(corr)
    ec = np.exp(beta * (corr - cm))
    s = ec.sum()
    if s < 1e-300:
        return np.zeros(n)
    p = ec / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pt = p[k]
        if pt < 1e-11:
            continue
        sh = k - (n - 1)
        lo1, hi1 = max(0, -sh), min(n, n - sh)
        if lo1 < hi1:
            grad[lo1:hi1] += pt * (1.0 - h[lo1 + sh:hi1 + sh])
        lo2, hi2 = max(0, sh), min(n, n + sh)
        if lo2 < hi2:
            grad[lo2:hi2] -= pt * h[lo2 - sh:hi2 - sh]
    return grad * dx

        if d > 0:
            r = 1.0 - h
            sr = r.sum()
            if sr > 1e-14:
                h = h + r * (d / sr)
        else:
            sh = h.sum()
            if sh > 1e-14:

    return best_h, best_c5

def binary_local_search(h0, t_budget, seed=42):
    n = len(h0)
    h = np.round(h0).clip(0, 1)
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        ones = np.where(h == 1)[0]
        h[ones[target:]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        h[zeros[:target - n_ones]] = 1
    c5 = compute_c5(h)
    best_h, best_c5 = h.copy(), c5
    rng = np.random.RandomState(seed)
    t_start = time.time()
    while time.time() - t_start < t_budget:
        ones = list(np.where(h == 1)[0])
        zeros = list(np.where(h == 0)[0])
        best_gain = 1e-13
        best_move = None
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                c5_new = compute_c5(h)
                gain = c5 - c5_new
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j)
                h[i] = 1; h[j] = 0
        if best_move is not None:
            i, j = best_move
            h[i] = 0; h[j] = 1
            c5 -= best_gain
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
        else:
            best_gain2 = 1e-13
            best_move2 = None
            for _ in range(4000):
                a = rng.randint(len(ones))
                b = rng.randint(len(ones))
                c = rng.randint(len(zeros))
                d = rng.randint(len(zeros))
                if a == b or c == d:
                    continue
                i1, i2 = ones[a], ones[b]
                j1, j2 = zeros[c], zeros[d]
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                c5_new = compute_c5(h)
                gain = c5 - c5_new
                if gain > best_gain2:
                    best_gain2 = gain
                    best_move2 = (i1, i2, j1, j2)
                h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
            if best_move2 is not None:
                i1, i2, j1, j2 = best_move2
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                c5 -= best_gain2
                if c5 < best_c5:
                    best_c5, best_h = c5, h.copy()
            else:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Gradient descent at n_init
    t1 = min(150.0, rem() * 0.25)
    h1, c5_1 = grad_desc(h_init, t1, [15.0, 80.0, 400.0], seed=seed)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Try higher resolutions
    for n_try in [168, 224, 280, 392]:
        if rem() < 80:
            break
        h_int = interp_to(h1, n_try)
        t2 = min(90.0, rem() * 0.18)
        h_opt, c5_opt = grad_desc(h_int, t2, [20.0, 100.0, 500.0], seed=seed + n_try)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 3: Binary local search at best resolution
    if rem() > 50:
        n_b = len(best_h)
        t3 = min(rem() * 0.3, 180.0)
        h_bin, c5_bin = binary_local_search(best_h, t3, seed=seed + 700)
        if c5_bin < best_c5:
            best_c5, best_h = c5_bin, h_bin.copy()

    # Phase 4: Gradient polish on best
    if rem() > 30:
        t4 = min(rem() * 0.5, 120.0)
        h4, c5_4 = grad_desc(best_h, t4, [100.0, 500.0, 2000.0], seed=seed + 1500)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Try another resolution if time
    if rem() > 60:
        n_alt = len(best_h) + 56
        if n_alt % 2 == 0:
            h_alt = interp_to(best_h, n_alt)
            t5 = min(rem() * 0.4, 100.0)
            h5, c5_5 = grad_desc(h_alt, t5, [50.0, 300.0, 1500.0], seed=seed + 2500)
            if c5_5 < best_c5:
                best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Final binary refinement
    if rem() > 20:
        t6 = rem() * 0.8
        h6, c5_6 = binary_local_search(best_h, t6, seed=seed + 3000)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
