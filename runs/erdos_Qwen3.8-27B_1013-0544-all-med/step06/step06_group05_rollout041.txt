def normalize_continuous(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:

            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
        else:
            total = np.sum(h)
            if total > 1e-12:

        h = np.clip(h, 0.0, 1.0)

    """Compute the gradient of (1/t) * log(sum_k exp(t * f_k)) where f_k are correlation values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # length 2n-1
    # f_k values (all lags)
    # For numerical stability, subtract max
    max_val = np.max(corr)
    exp_vals = np.exp(t * (corr - max_val))
    weights = exp_vals / np.sum(exp_vals)  # softmax weights, length 2n-1
    
    # Now compute gradient
    # For lag k (where output index = n-1+k):
    # ∂f_k/∂h[j] = (1-h[j+k]) * 1_{0 <= j < n-k} - h[j-k] * 1_{k <= j < n} for k >= 0
    # For lag -k (output index = n-1-k):
    # ∂f_{-k}/∂h[j] = (1-h[j-k]) * 1_{k <= j < n} - h[j+k] * 1_{0 <= j < n-k}
    
    grad = np.zeros(n)
    for k in range(1, n):
        # Positive lag k: weight = weights[n-1+k]
        w_pos = weights[n-1+k]
        # ∂f_k/∂h[j] = (1-h[j+k]) for j < n-k, minus h[j-k] for j >= k
        # First term: for j = 0 to n-k-1, add w_pos * (1-h[j+k])
        if n-k > 0:
            grad[0:n-k] += w_pos * (1.0 - h[k:n])
        # Second term: for j = k to n-1, subtract w_pos * h[j-k]
        if k < n:
            grad[k:n] -= w_pos * h[0:n-k]
        
        # Negative lag -k: weight = weights[n-1-k]
        w_neg = weights[n-1-k]
        # ∂f_{-k}/∂h[j] = (1-h[j-k]) for j >= k, minus h[j+k] for j < n-k
        if k < n:
            grad[k:n] += w_neg * (1.0 - h[0:n-k])
        if n-k > 0:
            grad[0:n-k] -= w_neg * h[k:n]
    
    return grad / t  # gradient of (1/t)*log(sum exp(t*f))

def gradient_descent(h0, t, lr, n_iters, seed=42):
    """Run projected gradient descent on smooth approximation."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    
    for it in range(n_iters):
        grad = smooth_max_gradient(h, t)
        # Adaptive learning rate
        h_new = h - lr * grad
        h_new = project_onto_constraint(h_new, n)
        
        # Line search (simple)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new
        
        # Decrease learning rate
        lr *= 0.999
    
    return best_h, best_c5

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def normalize_continuous(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            room = 1.0 - h
            total_room = np.sum(room)
            if total_room > 1e-12:
                h += room * (diff / total_room)
        else:
            total = np.sum(h)
            if total > 1e-12:
                h *= (1.0 + diff / total)
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-9:
            if diff > 0:
                idx = int(np.argmax(1.0 - h))
                h[idx] = min(1.0, h[idx] + diff)
            else:
                idx = int(np.argmax(h))
                h[idx] = max(0.0, h[idx] + diff)
    return h

def project(h, n):

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

def binary_local_search(h0, max_time, seed=42):
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    current_c5 = float(np.max(corr))
    start = time.time()
    rng = np.random.RandomState(seed)
    
    improved = True
    while improved and (time.time() - start) < max_time:
        improved = False
        best_gain = 1e-15
        best_move = None
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        for i in ones:
            for j in zeros:
                h[i] = 0
                h[j] = 1
                new_corr = np.correlate(h, 1.0 - h, mode='full') * dx
                new_c5 = float(np.max(new_corr))
                gain = current_c5 - new_c5
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j)
                h[i] = 1
                h[j] = 0
        
        if best_move is not None:
            i, j = best_move
            h[i] = 0
            h[j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            current_c5 = float(np.max(corr))
            improved = True
    
    return h, current_c5

    h = np.round(h).astype(int)
    target = n // 2
    n_ones = int(h.sum())

        h[ones[target:]] = 0

def project_constrained(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mx = np.max(corr)
    w = np.exp(t * (corr - mx))
    w /= w.sum()
    grad = np.zeros(n)
    for k in range(1, n):
        wp = w[n - 1 + k]
        wn = w[n - 1 - k]
        if wp > 1e-30:
            if n - k > 0:
                grad[0:n-k] += wp * (1.0 - h[k:n])
            grad[k:n] -= wp * h[0:n-k]
        if wn > 1e-30:
            if n - k > 0:
                grad[k:n] += wn * (1.0 - h[0:n-k])
                grad[0:n-k] -= wn * h[k:n]
    return grad / t

def smooth_optimize(h0, max_time, t=30.0, lr=0.05, seed=42):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    while time.time() - start < max_time:
        g = smooth_grad(h, t)
        h = project_constrained(h - lr * g, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr *= 0.9995
        if lr < 0.001:
            break
    return best_h, best_c5

def guided_hill_climb(h0, max_time, seed=42):
