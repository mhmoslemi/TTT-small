import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.argmax(1.0 - h)
                h[idx] += d2
            else:
                idx = np.argmax(h)
                h[idx] += d2
            h = np.clip(h, 0.0, 1.0)
    return h

def _project(h, n):
    """Project onto {0<=h<=1, sum h = n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def _subgrad_C(h, n, m):
    """Gradient of C[m] w.r.t. h (C from np.correlate)."""
    dx = 2.0 / n
    j = np.arange(n)
    idx1 = j + m - n + 1
    idx2 = j - m + n - 1
    g = np.zeros(n)
    v1 = (idx1 >= 0) & (idx1 < n)
    v2 = (idx2 >= 0) & (idx2 < n)
    g[np.where(v1)] = 1.0 - h[idx1[v1]]
    g[np.where(v2)] -= h[idx2[v2]]
    return dx * g

def _obj_grad(h, alpha, beta):
    """Soft-max objective with gradient."""
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n-1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n-1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _weighted_obj_grad(h, weights, n, beta):
    """Weighted sum of C_k with gradient."""
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    f = np.dot(weights, C) + beta * (np.sum(h) - 0.5*n)**2
    j_arr = np.arange(nc)
    grad = np.zeros(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n-1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n-1)], 0.0)
        grad[i] = dx * np.dot(weights, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5*n)
    return f, grad

def _lbfgs(h, n, obj_fun, args, maxiter=500):
    res = minimize(obj_fun, h, args=args, jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)]*n,
                  options={'maxiter': maxiter, 'maxfun': maxiter*10, 'ftol': 1e-14})
    return _fix(res.x, n) if res.x is not None else h

def _subgrad_descent(h, n, steps=5000):
    """Subgradient method for min max_k C_k."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_max(h)
    for t in range(steps):
        C = _c5_array(h)
        c5 = np.max(C)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        # Choose a random subgradient from the top lags
        top_frac = max(1, int(0.1 * len(C)))
        top_idx = np.argsort(C)[-top_frac:]
        k_star = top_idx[np.random.randint(len(top_idx))]
        g = _subgrad_C(h, n, k_star)
        eta = 0.5 / (1.0 + 0.005 * t)
        h = h - eta * g
        h = _project(h, n)
    return best_h, best_c5
