def _subgrad_momentum(h, n, time_limit, t0):

    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = _c5_max(h)
    v = np.zeros(n)  # momentum
    lr = 0.02
    mu = 0.9  # momentum coefficient
    
    for it in range(10000):
        if time.time() - t0 > time_limit:
            break
        C = _c5_array(h)
        k_star = int(np.argmax(C))
        c5_now = float(C[k_star])
        
        if c5_now < best_c5 - 1e-11:
            best_c5 = c5_now
            best_h = h.copy()
        
        shift = k_star - (n - 1)
        g = _subgrad_at_shift(h, shift, n, dx)
        v = mu * v + (1.0 - mu) * g
        h_new = _project(h - lr * v, n)
        c5_new = _c5_max(h_new)
        
        if c5_new < c5_now - 1e-13:
            h = h_new
            lr = min(lr * 1.02, 0.5)
        else:
            lr *= 0.8
            if lr < 1e-7:
                break
    
    return best_h, best_c5

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _subgrad_at_shift(h, shift, n, dx):
    g = np.zeros(n)
    if shift == 0:
        g[:] = dx * (1.0 - 2.0 * h)

        g[:n - shift] += dx * (1.0 - h[shift:])
