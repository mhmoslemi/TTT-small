import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            mask = np.ones(len(h), dtype=bool)
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_gradient(h, temp, lam, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(temp * (corr - m))
    w = exp_vals / exp_vals.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        shift = k - (n - 1)
        js = max(0, -shift)
        je = min(n, n - shift)
        if je > js:
            grad[js:je] += wk * omh[js + shift:je + shift]
        js2 = max(0, shift)
        je2 = min(n, n + shift)
        if je2 > js2:
            grad[js2:je2] -= wk * h[js2 - shift:je2 - shift]
    grad *= dx
    grad += 2.0 * lam * (h.sum() - target_sum)
    return grad

def compute_obj(h, temp, lam, target_sum):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(corr)
    exp_vals = np.exp(temp * (corr - m))
    return (m + np.log(exp_vals.sum()) / temp) * dx + lam * (h.sum() - target_sum) ** 2

def l_bfgsb_run(h0, temp, lam, maxiter=300):
    n = len(h0)
    target = n / 2.0
    bnds = [(0.0, 1.0)] * n
    def obj(z):
        return compute_obj(z, temp, lam, target)
    def grad(z):
        return compute_gradient(z, temp, lam, target)
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bnds, options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
    return res.x

def coord_descent(h, n_levels=15, max_sweeps=10):
    n = len(h)
    ts = n / 2.0
    c5 = compute_c5(h)
    for _ in range(max_sweeps):
        improved = False
        for j in range(n):
            orig = h[j]
            best_c = c5
            best_h = h.copy()
            for v in np.linspace(0.0, 1.0, n_levels):
                h[j] = v
                h = project_feasible(h, ts)
                c = compute_c5(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_h = h.copy()
            h = best_h
            c5 = best_c
            if abs(h[j] - orig) > 1e-12:
                improved = True
        if not improved:
            break
    return h, c5

def interpolate_h(h, n_to):
    x_old = np.arange(len(h), dtype=float)
    x_new = np.arange(n_to, dtype=float) * (len(h) - 1) / (n_to - 1)
    h_new = np.interp(x_new, x_old, h)
    return h_new

def generate_restart(best_h, rng):
    n = len(best_h)
    r = rng.random()
    if r < 0.25:
        block = max(2, n // rng.randint(3, 12))
        nb = n // block
        idx = rng.permutation(nb)
        cur = np.zeros(n)
        for i in range(nb):
            cur[i*block:(i+1)*block] = best_h[idx[i]*block:(idx[i]+1)*block]
        if n % block:
            cur[nb*block:] = best_h[nb*block:]
    elif r < 0.45:
        s = rng.randint(1, n)
        cur = np.roll(best_h, s)
    elif r < 0.6:
        cur = best_h[::-1].copy()
        cur += rng.randn(n) * rng.uniform(0.05, 0.15)
    elif r < 0.8:
        sigma = rng.uniform(0.05, 0.35)
        cur = best_h + rng.randn(n) * sigma
    else:
        cur = best_h + rng.randn(n) * rng.uniform(0.1, 0.5)
        # Block-level perturbation

        bw = n // nb
        for b in range(nb):
            cur[b*bw:(b+1)*bw] *= rng.uniform(0.5, 1.5)
    return cur
