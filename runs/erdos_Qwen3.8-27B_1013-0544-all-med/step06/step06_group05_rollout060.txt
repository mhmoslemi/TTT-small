```python
def subgradient_step(h, learning_rate):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find active lags (within some tolerance of max)
    active = np.where(corr >= max_val - 1e-9)[0]
    
    # Compute subgradient (average of gradients of active lags)
    grad = np.zeros(n)
    for k in active:
        j = k - (n - 1)  # lag = j (can be negative, zero, or positive)
        if j == 0:
            grad += 1.0 - 2.0 * h
        elif j > 0:
            # f_j: corr[n-1+j]
            for p in range(n):
                if p <= n - 1 - j:
                    grad[p] += (1.0 - h[p+j])
                if p >= j:
                    grad[p] -= h[p-j]
        else:
            # g_{-j}: corr[n-1-j] where j>0, so lag = -j
            jj = -j
            for p in range(n):
                if p >= jj:
                    grad[p] += (1.0 - h[p-jj])
                if p <= n - 1 - jj:
                    grad[p] -= h[p+jj]
    
    grad /= len(active)
    return h - learning_rate * grad, max_val
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.round(np.asarray(h, dtype=float))
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        rng = np.random.RandomState(0)
        ones = np.where(h == 1)[0]
        remove = rng.choice(ones, n_ones - target, replace=False)
        h[remove] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(0)
        add = rng.choice(zeros, target - n_ones, replace=False)
        h[add] = 1
    return h

def precompute_PQ(h):
    """Precompute P_i and Q_j for all i, j."""
    n = len(h)
    length = 2 * n - 1
    k = np.arange(length)
    
    # P_i[k] = h[i+n-1-k] - (1-h)[i+k-n+1]
    P = np.zeros((n, length))
    Q = np.zeros((n, length))
    
    for i in range(n):
        idx1 = i + n - 1 - k
        m1 = (idx1 >= 0) & (idx1 < n)
        P[i, m1] += h[idx1[m1]]
        idx3 = i + k - n + 1
        m3 = (idx3 >= 0) & (idx3 < n)
        P[i, m3] -= (1.0 - h[idx3[m3]])
    
    for j in range(n):
        idx2 = j + n - 1 - k
        m2 = (idx2 >= 0) & (idx2 < n)
        Q[j, m2] -= h[idx2[m2]]
        idx4 = j + k - n + 1
        m4 = (idx4 >= 0) & (idx4 < n)
        Q[j, m4] += (1.0 - h[idx4[m4]])
    
    return P, Q

def local_search_fast(h, max_time=60.0, seed=42):
    """Binary local search with O(n) per evaluation using precomputed P, Q."""
    n = len(h)
    h = make_binary(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(corr)
    best_h = h.copy()
    best_max = current_max
    length = 2 * n - 1
    
    start = time.time()
    rng = np.random.RandomState(seed)
    
    while (time.time() - start) < max_time:
        # Precompute P and Q
        P, Q = precompute_PQ(h)
        
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        best_gain = 0
        best_i, best_j = -1, -1
        
        for i in ones:
            for j in zeros:
                # swap_effect = P[i] + Q[j] + R
                se = P[i] + Q[j]
                # Add R terms
                se[n-1] -= 2.0
                p1 = n - 1 + j - i
                if 0 <= p1 < length:
                    se[p1] += 1.0
                p2 = n - 1 + i - j
                if 0 <= p2 < length:
                    se[p2] += 1.0
                
                new_max = np.max(corr + se)
                gain = current_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_i, best_j = i, j
        
        if best_i >= 0:
            h[best_i] = 0
            h[best_j] = 1
            corr = corr + P[best_i] + Q[best_j]
            corr[n-1] -= 2.0
            p1 = n - 1 + best_j - best_i
            if 0 <= p1 < length:
                corr[p1] += 1.0
            p2 = n - 1 + best_i - best_j
            if 0 <= p2 < length:
                corr[p2] += 1.0
            current_max = np.max(corr)
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
            improved = True
        else:
            break
    
    return best_h, best_max
```
