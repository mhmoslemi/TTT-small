    dx = 2.0 / n

    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:

    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    m = np.max(C)
    e = np.exp(alpha * (C - m))
    w = e / np.sum(e)
    f = (m + np.log(np.sum(e)) / alpha) + beta * (np.sum(h) - 0.5 * n) ** 2

    grad = np.zeros(n)

from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    return h

def _coord_step(h, i, n):
    """Optimize h[i] given all other values fixed."""
    dx = 2.0 / n
    nc = 2 * n - 1
    others_sum = np.sum(h) - h[i]
    lo = max(0.0, 0.5 * n - others_sum)
    hi = min(1.0, 0.5 * n - others_sum)
    if hi - lo < 1e-12:
        return h[i]
    
    # Compute a (linear coefficient) for each lag
    a = np.zeros(nc)
    idx1 = i + np.arange(nc) - n + 1
    v1 = (idx1 >= 0) & (idx1 < n)
    a[v1] += dx * (1.0 - h[idx1[v1]])
    idx2 = i - np.arange(nc) + n - 1
    v2 = (idx2 >= 0) & (idx2 < n)
    a[v2] -= dx * h[idx2[v2]]
    
    C = _c5_array(h)
    b = C - h[i] * a
    
    # Ternary search for min of max(a*t + b) on [lo, hi]
    for _ in range(60):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if np.max(a * m1 + b) < np.max(a * m2 + b):
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def _coord_descent(h, n, sweeps=3, rng=None):
    h = h.copy()
    best_c5 = _c5_max(h)
    for s in range(sweeps):
        if rng is not None:
            order = rng.permutation(n)
        else:
            order = np.arange(n)
        improved = False
        for i in order:
            new_val = _coord_step(h, i, n)
            if abs(new_val - h[i]) > 1e-10:
                h[i] = new_val
                h = _fix(h, n)
                c5 = _c5_max(h)
                if c5 < best_c5:
                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return h, best_c5

def _soft_obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    m = np.max(C)
    e = np.exp(alpha * (C - m))
    se = np.sum(e)
    w = e / se
    f = m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(n):
        idx1 = i + np.arange(nc) - n + 1
        v1 = (idx1 >= 0) & (idx1 < n)
        s1 = np.dot(w[v1], omh[idx1[v1]])
        idx2 = i - np.arange(nc) + n - 1
        v2 = (idx2 >= 0) & (idx2 < n)
        s2 = np.dot(w[v2], h[idx2[v2]])
        grad[i] = dx * (s1 - s2)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgs(h, alpha, beta, maxiter=200):

                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * len(h),
                   options={'maxiter': maxiter, 'maxfun': maxiter * 20, 'gtol': 1e-8})
    return _fix(res.x if res.x is not None else h, len(h))

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    return h

    se = float(np.sum(e))
    w = e / se
    f = m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    grad = np.zeros(n)
    omh = 1.0 - h
    for i in range(n):
        idx1 = i + ar - n + 1
        v1 = (idx1 >= 0) & (idx1 < n)

        grad[i] = dx * (s1 - s2)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
