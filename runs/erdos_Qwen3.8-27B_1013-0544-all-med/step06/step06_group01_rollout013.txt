    corr2 = np.correlate(v, h, mode='full')
    grad = dx * (corr1[0:n][::-1] - corr2[0:n][::-1])
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _optimize_smooth(h, alpha, beta, maxiter=2000, time_limit=60):
    t0 = time.time()
    n = len(h)
    res = minimize(_smooth_obj_grad, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'maxfun': maxiter * 20,
                           'ftol': 1e-16, 'gtol': 1e-14})
    return _fix(res.x, n)

def _coord_descent(h, n, passes=10, grid_size=50, time_limit=120):
    t0 = time.time()
    h = h.copy()
    dx = 2.0 / n
    best_c5 = _c5_max(h)
    grid = np.linspace(0.0, 1.0, grid_size)
    for p in range(passes):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            old = h[i]
            old_c5 = best_c5
            best_cv = old
            for cv in grid:
                h[i] = cv
                c5 = _c5_max(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_cv = cv
            h[i] = best_cv
            h = _fix(h, n)
            if best_c5 < old_c5 - 1e-12:
                improved = True
        if not improved and p > 0:
            break
    return h, best_c5

def _swap_polish(h, n, time_limit=30):
    t0 = time.time()
    h = h.copy()
    best_c5 = _c5_max(h)
    dx = 2.0 / n
    deltas = [0.005, 0.01, 0.02, 0.002]
    while time.time() - t0 < time_limit:
        improved = False
        for i in range(n):
            for j in range(i + 1, n):
                if h[i] <= 0 or h[j] >= 1:
                    continue
                for d in deltas:
                    if h[i] < d:
                        continue
                    h[i] -= d
                    h[j] = min(h[j] + d, 1.0)
                    c5 = _c5_max(h)
                    if c5 < best_c5 - 1e-11:
                        best_c5 = c5
                        improved = True
                        break
                    h[i] += d
                    h[j] -= d
                    if c5 < best_c5 - 1e-11:
                        best_c5 = c5
                        improved = True
                        break
                    h[j] += d
        if not improved:
            break
    h = _fix(h, n)
    return h, _c5_max(h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    hb = _fix(initial_h_values.astype(float).copy(), n0)
    cb = _c5_max(hb)
    nb = n0

    # Phase 1: Progressive smooth optimization
    for alpha in [10, 30, 80, 200, 500, 1000]:
        if _el() > budget_s * 0.22:
            break
        beta = max(1.0, 0.08 * alpha)
        tl = min(40, budget_s * 0.035)
        hn = _optimize_smooth(hb, alpha, beta, maxiter=1500, time_limit=tl)
        cn = _c5_max(hn)
        if cn < cb - 1e-9:
            cb, hb, nb = cn, hn, len(hn)

    # Phase 2: Coordinate descent
    if _el() < budget_s * 0.50:
        tl = budget_s * 0.50 - _el()
        passes = min(30, max(5, int(tl / max(1, nb * 50 * 2e-6))))
        hb, cb = _coord_descent(hb, nb, passes=passes, grid_size=50, time_limit=tl)

    # Phase 3: Try nearby n values
    n_try = sorted(set([nb - 5, nb - 10, nb + 5, nb + 10, nb + 15,
                        85, 95, 105, 115, 125, 135]))
    n_try = [n for n in n_try if 60 <= n <= 250 and n != nb]
    for n in n_try:
        if _el() > budget_s * 0.72:
            break
        x_old = np.linspace(0, 2, nb, endpoint=False)
        x_new = np.linspace(0, 2, n, endpoint=False)
        ht = _fix(np.interp(x_new, x_old, hb), n)
        for alpha in [50, 200]:
            if _el() > budget_s * 0.73:
                break
            ht = _optimize_smooth(ht, alpha, max(1.0, 0.08 * alpha),
                                  maxiter=800, time_limit=25)
            cn = _c5_max(ht)
            if cn < cb - 1e-9:
                cb, hb, nb = cn, ht, n
                break

    # Phase 4: Coordinate descent again
    if _el() < budget_s * 0.85:
        tl = budget_s * 0.85 - _el()
        passes = min(25, max(5, int(tl / max(1, nb * 50 * 2e-6))))
        hb, cb = _coord_descent(hb, nb, passes=passes, grid_size=60, time_limit=tl)

    # Phase 5: Swap polish
    if _el() < budget_s * 0.93:
        tl = min(40, budget_s * 0.93 - _el())
        hb, cb = _swap_polish(hb, nb, time_limit=tl)

    # Phase 6: Final high-alpha polish
    if _el() < budget_s * 0.97:
        for alpha in [500, 1500]:
            if _el() > budget_s * 0.97:
                break
            hn = _optimize_smooth(hb, alpha, max(10.0, 0.08 * alpha),
                                  maxiter=1000, time_limit=15)
            cn = _c5_max(hn)
            if cn < cb:
                cb, hb = cn, hn

    # Final coordinate descent
    if _el() < budget_s * 0.995:
        hb, cb = _coord_descent(hb, nb, passes=3, grid_size=40, time_limit=5)

    hb = _fix(hb, nb)
    cb = _c5_max(hb)
    return (hb, cb, nb)
