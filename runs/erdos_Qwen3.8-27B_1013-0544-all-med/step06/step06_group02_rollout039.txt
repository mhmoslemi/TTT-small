```python
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)

def compute_obj(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def compute_soft_grad(h, corr, dx, alpha):
    """Compute soft gradient using weighted average of subgradients."""
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        lag = k_star - (N - 1)
        # Direct subgradient
        g = np.zeros(N)
        # grad of corr[N-1+lag] w.r.t. h[i] = (1-h[i-lag]) - h[i+lag]
        if lag >= 0:
            g[:N-lag] += 1.0 - h[lag:N]
            g[lag:] -= h[:N-lag]
        else:
            g[-lag:] += 1.0 - h[:N+lag]
            g[:N+lag] -= h[-lag:]
        return g * dx
    
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-10:

        lag = j - (N - 1)
        if lag >= 0:

            g[lag:] -= w[j] * h[:N-lag]
```

```python
from scipy.optimize import linprog

def lp_descent_step(h0, N, dx):
    """One step of LP-based descent.
    Returns a new h that minimizes the linearized objective."""
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    k_star = np.argmax(corr0)
    
    # We want to minimize max_k (corr0[k] + grad_k · (h - h0))
    # = minimize t s.t. corr0[k] + grad_k · h - grad_k · h0 ≤ t for all k
    # = minimize t s.t. grad_k · h - t ≤ grad_k · h0 - corr0[k] for all k
    
    # Decision variables: x = [h[0], ..., h[N-1], t]
    # Objective: minimize t → c = [0, ..., 0, 1]
    
    n_vars = N + 1
    c = np.zeros(n_vars)
    c[N] = 1  # minimize t
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each k (2N-1 of them):
    #   grad_k[0]*h[0] + ... + grad_k[N-1]*h[N-1] - t ≤ grad_k·h0 - corr0[k]
    #   → [grad_k, -1] · x ≤ grad_k·h0 - corr0[k]
    
    A_rows = []
    b_vals = []
    for k in range(2*N - 1):
        g = grad_at_lag(h0, k - (N-1))
        row = np.zeros(n_vars)
        row[:N] = g
        row[N] = -1.0
        rhs = g @ h0 - corr0[k]
        A_rows.append(row)
        b_vals.append(rhs)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_vals)
    
    # Equality constraint: sum(h) = N/2
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :N] = 1.0
    b_eq = np.array([N / 2.0])
    
    # Bounds: 0 ≤ h[i] ≤ 1, t is free
    bounds = [(0, 1)] * N + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
```

```python
    if ws < 1e-300:
        k_star = int(np.argmax(corr))
        return grad_at_lag(h, k_star - (N - 1)) * dx
    w /= ws
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-10:
            continue
        g += w[j] * grad_at_lag(h, j - (N - 1))
    return g * dx

def interpolate_h(h_src, n_dst):
    x0 = np.linspace(0.0, 1.0, len(h_src), endpoint=False)
    x1 = np.linspace(0.0, 1.0, n_dst, endpoint=False)
    h = np.interp(x1, x0, h_src, period=1)
    return project_feasible(h, n_dst)

def optimize_band(h, N, rng, t_end, best, max_iter=3000, top_k_override=None):
    val, corr, dx = compute_obj(h)
    if val < best[1]:
        best[1] = val
        best[0] = h.copy()
    top_k = top_k_override if top_k_override else min(60, 2 * N - 1)
    stagnant = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        corr_cur = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr_cur)[-top_k:]
        n_vars = N + 1
        c = np.zeros(n_vars)
        c[N] = 1.0
        A_rows = np.zeros((len(top_idx), n_vars))
        b_vals = np.zeros(len(top_idx))
        for ii, k in enumerate(top_idx):
            g = grad_at_lag(h, int(k) - (N - 1))
            A_rows[ii, :N] = g
            A_rows[ii, N] = -1.0
            b_vals[ii] = g @ h - corr_cur[k]
        A_eq = np.zeros((1, n_vars))
        A_eq[0, :N] = 1.0
        b_eq = np.array([N / 2.0])
        bounds = [(0.0, 1.0)] * N + [(None, None)]
        res = linprog(c, A_ub=A_rows, b_ub=b_vals, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            h = project_feasible(h + rng.randn(N) * 0.05, N)
            val, corr, _ = compute_obj(h)
            if val < best[1]:
                best[1] = val
                best[0] = h.copy()
            stagnant += 1
            continue
        h_lp = res.x[:N]
        found = False
        for alpha_ls in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
            h_trial = project_feasible((1.0 - alpha_ls) * h + alpha_ls * h_lp, N)
            val_trial, corr_trial, _ = compute_obj(h_trial)
            if val_trial < val - 1e-14:
                h = h_trial
                val = val_trial
                corr = corr_trial
                stagnant = 0
                found = True
                if val < best[1]:
                    best[1] = val
                    best[0] = h.copy()
                break
        if not found:
            stagnant += 1
            if stagnant > 12:
                scale = 0.03 + 0.02 * min(stagnant / 10.0, 1.0)
                h = project_feasible(h + rng.randn(N) * scale, N)
                val, corr, _ = compute_obj(h)
                if val < best[1]:
                    best[1] = val
                    best[0] = h.copy()
                stagnant = 0
    return h, val

def optimize_grad(h, N, rng, t_end, best, max_iter=80000):
    val, corr, dx = compute_obj(h)
    if val < best[1]:
        best[1] = val
        best[0] = h.copy()
    beta = 0.85
    v = np.zeros(N)
    lr = 0.05
    stagnant = 0
    for it in range(max_iter):
        if time.time() > t_end:
            break
        progress = min(it / 3000.0, 1.0)
        alpha_cur = 5.0 + 50.0 * progress
        g = soft_grad(h, corr, dx, alpha_cur)
        v = beta * v - lr * g
        h_new = project_feasible(h + v, N)
        val_new, corr_new, _ = compute_obj(h_new)
        if val_new < val - 1e-15:
            h = h_new
            val = val_new
            corr = corr_new
            stagnant = 0
            if it % 200 == 0:
                lr = min(lr * 1.005, 0.12)
            if val < best[1]:
                best[1] = val
                best[0] = h.copy()
        else:
            v *= beta
            stagnant += 1
            lr *= 0.93
            if stagnant > 25:
                v = np.zeros(N)
                h = project_feasible(h + rng.randn(N) * 0.03, N)
                val, corr, _ = compute_obj(h)
                lr = 0.05
                stagnant = 0
                if val < best[1]:
                    best[1] = val
                    best[0] = h.copy()
    return h, val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 15
    best = [None, float('inf')]
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None
    # Phase 1: LP refinement of parent at native resolution
    if h_init is not None:
        v0 = compute_obj(h_init)[0]
        if v0 < best[1]:
            best[1] = v0
            best[0] = h_init.copy()
        N_par = len(h_init)
        t_p1 = min(t0 + 280, t_end - 200)
        h, v = optimize_band(h_init.copy(), N_par, rng, t_p1, best, max_iter=2000)
    # Phase 2: Try multiple resolutions with LP
    n_list = [64, 80, 100, 120, 144]
    for n_t in n_list:
        if time.time() > t_end - 80:
            break
        if best[0] is not None:
            h_start = interpolate_h(best[0], n_t)
        elif h_init is not None:
            h_start = interpolate_h(h_init, n_t)
        else:
            h_start = project_feasible(rng.rand(n_t) * 1.2, n_t)
        t_phase = min(time.time() + 130, t_end - 50)
        h, v = optimize_band(h_start, n_t, rng, t_phase, best, max_iter=1200)
    # Phase 3: Gradient refinement at best resolution
    if best[0] is not None and time.time() < t_end - 40:
        N_best = len(best[0])
        t_p3 = min(time.time() + 100, t_end - 25)
        h, v = optimize_grad(best[0].copy(), N_best, rng, t_p3, best, max_iter=60000)
    # Phase 4: Additional LP passes at best and nearby resolutions
    if best[0] is not None and time.time() < t_end - 20:
        N_best = len(best[0])
        t_p4 = min(time.time() + 60, t_end - 10)
        h, v = optimize_band(best[0].copy(), N_best, rng, t_p4, best, max_iter=800, top_k_override=80)
    # Phase 5: Final gradient polish
    if best[0] is not None and time.time() < t_end:
        N_best = len(best[0])
        h, v = optimize_grad(best[0].copy(), N_best, rng, t_end, best, max_iter=30000)
    if best[0] is None:
        best[0] = project_feasible(0.5 * np.ones(64), 64)
        best[1] = compute_obj(best[0])[0]
    return best[0], best[1], len(best[0])
```
