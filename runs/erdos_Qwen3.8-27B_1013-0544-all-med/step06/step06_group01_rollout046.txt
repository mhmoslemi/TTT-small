    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            idx = np.argmax(1.0 - h) if d2 > 0 else np.argmax(h)
            h[idx] += d2
            h = np.clip(h, 0.0, 1.0)
    return h

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se

    grad = np.empty(n)

        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)

    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _optimize(h, n, alpha, beta, maxiter=500):

import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            idx = np.argmax(1.0 - h) if d2 > 0 else np.argmax(h)
            h[idx] += d2
            h = np.clip(h, 0.0, 1.0)
    return h

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _optimize(h, n, alpha, beta, maxiter=400):
    res = minimize(_obj_grad, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-14})
    if res.x is None:
        return h
    return _fix(res.x, n)

def _binarize(h, n):

    hb = np.zeros(n)
    n_ones = int(round(n / 2))
    for i in range(n_ones):
        hb[h_sorted[i]] = 1.0
    return hb

                cb, hb = cn, ht
                nb = n
    
    # Phase 3: Push toward binary then gradient refine
    if _el() < budget_s * 0.45:
        for strength in [10, 30, 100]:
            if _el() > budget_s * 0.45:
                break
            ht = _push_to_binary(hb, nb, strength)
            ht = _optimize(ht, nb, 300, 10.0, maxiter=200)
            cn = _c5_max(ht)
            if cn < cb - 1e-9:
                cb, hb = cn, ht
    
    # Phase 4: Binary local search at current and other resolutions
    search_ns = sorted(set([nb, 100, 150, 200]))
    search_ns = [x for x in search_ns if 60 <= x <= 350]
    
    for n in search_ns:
        if _el() > budget_s * 0.78:
            break
        ht = _interpolate(hb, nb, n)
        hb_n = _binarize(ht, n)
        limit = min(budget_s * 0.82, _el() + 80.0)
        hn, cn = _binary_local_search(hb_n, n, max_passes=300, time_limit=limit, t0=t0, rng=rng)
        if cn < cb - 1e-9:
            cb, hb = cn, hn
            nb = n
    
    # Phase 5: Random binary starts + local search
    for n in [nb, 120, 180]:
        if _el() > budget_s * 0.90:
            break
        if n != nb:
            ht = _interpolate(hb, nb, n)
        else:
            ht = hb.copy()
        for attempt in range(3):
            if _el() > budget_s * 0.92:
                break
            hb_rand = _binarize(ht + rng.randn(n) * 0.3, n)
            limit = min(budget_s * 0.94, _el() + 40.0)
            hn, cn = _binary_local_search(hb_rand, n, max_passes=200, time_limit=limit, t0=t0, rng=rng)
            if cn < cb - 1e-9:
                cb, hb = cn, hn
                nb = n
    
    # Phase 6: Final gradient polish on best solution
    if _el() < budget_s * 0.96:
        for alpha in [500, 1000]:
            if _el() > budget_s * 0.97:
                break
            hn = _optimize(hb, nb, alpha, max(10.0, 0.02 * alpha), maxiter=300)
            cn = _c5_max(hn)
            if cn < cb:
                cb, hb = cn, hn
    
    # Phase 7: Final binary polish
    if _el() < budget_s * 0.985:
        hb_bin = _binarize(hb, nb)
        hb2, cb2 = _binary_local_search(hb_bin, nb, max_passes=200,
                                         time_limit=budget_s * 0.99, t0=t0, rng=rng)
        if cb2 < cb:
            cb, hb = cb2, hb2
    
    hb = _fix(hb, nb)
    cb = _c5_max(hb)
    return (hb, cb, nb)
