import numpy as np
from scipy.optimize import minimize
import time

_OPT = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _lag_grad(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[:n-shift] += (1.0 - h[shift:])
            g[shift:] -= h[:n-shift]
    else:
        s = -shift
        if s < n:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
    return g

def softmax_obj(h):
    t = _OPT['t']
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    shifted = np.clip((conv - m) / t, -500, 0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    return t * (m / t + np.log(s)) * dx

def softmax_grad_fn(h):
    t = _OPT['t']
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    shifted = np.clip((conv - m) / t, -500, 0)
    exps = np.exp(shifted)
    s = np.sum(exps)
    weights = exps / s
    top_k = _OPT.get('top_k', 20)
    order = np.argsort(weights)[::-1][:top_k]
    grad = np.zeros(n)
    for idx_k in order:
        w = weights[idx_k]
        if w < 1e-10:
            continue
        shift = idx_k - (n - 1)
        grad += w * _lag_grad(h, shift)
    return grad * dx

def penalty_obj(h):
    n = _OPT['n']
    return softmax_obj(h) + 500.0 * (np.sum(h) - n / 2.0) ** 2

def penalty_grad(h):
    n = _OPT['n']
    return softmax_grad_fn(h) + 1000.0 * (np.sum(h) - n / 2.0)

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def lbfgs_optimize(h, n, t, maxiter=2000):
    _OPT['t'] = t
    _OPT['n'] = n
    _OPT['top_k'] = 30
    try:
        res = minimize(penalty_obj, h, jac=penalty_grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-14})
        return project_feasible(res.x, n)
    except Exception:
        return h.copy()

def adam_optimize(h, n, t, max_iter=20000, lr=0.001, top_k=20):
    _OPT['t'] = t
    _OPT['top_k'] = top_k
    h = h.copy()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    b1, b2, eps = 0.9, 0.999, 1e-8
    for it in range(max_iter):
        grad = softmax_grad_fn(h)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        it1 = it + 1
        m_v = b1 * m_v + (1 - b1) * grad
        v_v = b2 * v_v + (1 - b2) * grad ** 2
        m_hat = m_v / (1 - b1 ** it1)
        v_hat = v_v / (1 - b2 ** it1)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        if np.max(np.abs(h_new - h)) < 1e-14:
            break
        h = h_new
    return best_h, best_c5

def pgd_optimize(h, n, max_iter=50000, lr=0.001):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        conv = np.correlate(h, 1.0 - h, mode="full")
        dx = 2.0 / n
        c5 = np.max(conv) * dx
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        idx = np.argmax(conv)
        shift = idx - (n - 1)
        grad = _lag_grad(h, shift) * dx
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-13:
            lr *= 0.6
            if lr < 1e-15:
                break
        else:
            h = h_new
            lr = min(lr * 1.01, 0.01)
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)

    n0 = len(h0)

    for t_val in [0.1, 0.05, 0.02, 0.005, 0.002]:

    # Phase 2: Adam with soft-max

        if c5t < best_c5:

    # Phase 3: PGD hard max
    if time_left() > 50:

        if c5t < best_c5:

    # Phase 4: Higher resolution attempts
    for n_try in [480, 640]:
        if time_left() < 250:
            break
        nc = len(best_h)
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)

        ht = lbfgs_optimize(hi, n_try, 0.01, maxiter=1000)
        ht, c5t = adam_optimize(ht, n_try, 0.003, max_iter=10000, lr=0.0003, top_k=30)
        ht, c5t2 = pgd_optimize(ht, n_try, max_iter=30000, lr=0.0003)
        c5t = min(c5t, c5t2)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Random restarts
    for rep in range(80):
        if time_left() < 30:
            break
        r = rng.random()
        if r < 0.25:
            hr = rng.uniform(0.0, 1.0, n0)
        elif r < 0.5:
            nb = max(1, n0 // 10)
            blocks = rng.uniform(0.1, 0.9, n0 // nb + 1)
            hr = np.repeat(blocks, nb + 1)[:n0]
        elif r < 0.75:

        hr = project_feasible(hr, n0)
        ht, c5t = adam_optimize(hr, n0, 0.03, max_iter=5000, lr=0.002, top_k=15)

        if c5t < best_c5:

    return best_h, best_c5, len(best_h)
