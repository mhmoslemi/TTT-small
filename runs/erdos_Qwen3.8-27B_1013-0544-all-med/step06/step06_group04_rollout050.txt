import numpy as np
from scipy.optimize import minimize
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_onto_feasible(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} via bisection on lambda."""
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def subgradient(h, n):
    """Compute gradient of the active constraint (max correlation)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argmax(corr)
    lag = idx - n + 1
    
    grad = np.zeros(n)
    if lag >= 0:
        # corr = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
        # d/dh[p] = (1-h[p-lag]) for p in [lag, n-1]
        #         + (-h[p+lag]) for p in [0, n-1-lag]
        grad[lag:] += (1.0 - h[:n-lag])
        grad[:n-lag] += -h[lag:]
    else:
        la = -lag
        # corr = sum_{j=0}^{n-1-la} h[j] * (1-h[j+la])
        # d/dh[p] = (1-h[p+la]) for p in [0, n-1-la]
        #         + (-h[p-la]) for p in [la, n-1]
        grad[:n-la] += (1.0 - h[la:])
        grad[la:] += -h[:n-la]
    
    return grad * dx

def subgradient_smooth(h, n, temperature=0.01):
    """Compute smoothed subgradient using log-sum-exp weights."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    weights = np.exp((corr - m) / temperature)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if weights[idx] < 1e-8:
            continue
        lag = idx - n + 1
        if lag >= 0:
            grad[lag:] += weights[idx] * (1.0 - h[:n-lag])
            grad[:n-lag] += -weights[idx] * h[lag:]
        else:
            la = -lag
            grad[:n-la] += weights[idx] * (1.0 - h[la:])
            grad[la:] += -weights[idx] * h[:n-la]
    
    return grad * dx

def pgd_step(h, grad, target_sum, lr):
    """One projected gradient descent step."""
    h_new = h - lr * grad
    return project_onto_feasible(h_new, target_sum)

def backtracking_line_search(h, grad, target_sum, initial_lr, c5_current):
    """Backtracking line search for projected gradient."""
    n = len(h)
    dx = 2.0 / n
    c = 1e-4
    rho = 0.5
    lr = initial_lr
    
    for _ in range(30):
        h_new = pgd_step(h, grad, target_sum, lr)
        c5_new = compute_c5(h_new)
        # Armijo condition: f(x - lr*grad) <= f(x) - c*lr*||grad||^2
        # For non-smooth, we use: f_new <= f_old - c*lr*||grad||^2
        if c5_new <= c5_current - c * lr * np.dot(grad, grad) + 1e-12:
            return h_new, c5_new, lr
        lr *= rho
    
    return h_new, c5_new, lr

def upsample(h, factor=2):
    """Upsample h by repeating each element 'factor' times, preserving integral."""
    h_up = np.repeat(h, factor)
    n_old = len(h)
    n_new = len(h_up)

    # But we need sum(h_up) = n_new/2
    # sum(h_up) = factor * sum(h) = factor * n_old/2 = n_new/2 ✓
    return h_up

    rem = budget_s - elapsed

    # Phase 3: upsample again
    if rem > 30:
        alloc3 = min(budget_s * 0.20, rem - 10)
        h3 = upsample(best_h, 2)
        h3, c53 = optimize_pgd(h3, alloc3, seed + 3)
        if c53 < best_c5:
            best_h, best_c5, best_n = h3.copy(), c53, len(h3)
        print(f"P3 n={len(h3)} C5={c53:.6f}")

    elapsed = time.time() - t0
    rem = budget_s - elapsed

    # Phase 4: refine best at its resolution
    if rem > 20:
        alloc4 = min(budget_s * 0.20, rem - 5)
        h4, c54 = optimize_pgd(best_h, alloc4, seed + 10)
        if c54 < best_c5:
            best_h, best_c5, best_n = h4.copy(), c54, len(h4)
        print(f"P4 n={len(best_h)} C5={c54:.6f}")

    elapsed = time.time() - t0
    rem = budget_s - elapsed

    # Phase 5: one more upsample if time allows
    if rem > 40:
        alloc5 = min(budget_s * 0.05, rem - 5)
        h5 = upsample(best_h, 2)
        h5, c55 = optimize_pgd(h5, alloc5, seed + 20)
        if c55 < best_c5:
            best_h, best_c5, best_n = h5.copy(), c55, len(h5)
        print(f"P5 n={len(h5)} C5={c55:.6f}")

    # Final fixup: ensure exact feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    # Try rounding and re-optimizing if close to binary
    h_rounded = np.round(best_h).astype(np.float64)
    h_rounded = project_feasible(h_rounded, len(h_rounded) / 2.0)
    c5_rounded = compute_c5(h_rounded)
    if c5_rounded < best_c5 and rem > 10:
        t_rem = time.time() - t0
        if budget_s - t_rem > 15:
            h_rounded, c5_rounded = optimize_binary_sa(h_rounded, budget_s - t_rem - 2, seed + 99)
    if c5_rounded < best_c5:
        best_h = h_rounded
        best_c5 = c5_rounded

    final_c5 = compute_c5(best_h)
    print(f"FINAL n={len(best_h)} C5={final_c5:.6f}")
    return best_h, final_c5, len(best_h)
