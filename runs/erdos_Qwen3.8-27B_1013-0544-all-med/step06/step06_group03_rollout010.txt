def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def c5_subgrad(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    c5 = np.max(conv) * dx
    idx = np.argmax(conv)
    shift = idx - (n - 1)
    grad = _conv_grad_single(h, shift)
    return c5, grad * dx

def soft_max_obj(h, t, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    return t * (m / t + np.log(np.sum(exps))) * dx

def soft_max_grad(h, t, n):

- If s < 0: a ranges from 0 to n-1+s, b = a-s

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")

    idx = np.argmax(conv)
    shift = idx - (n - 1)
    grad = _conv_grad_single(h, shift)
    return c5, grad * dx

        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def pgd_optimize(h_init, n, max_iter, lr_init, t_smooth, use_smooth, momentum_init):
    h = h_init.copy()
    z = h.copy()
    mom = momentum_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(max_iter):
        if use_smooth:
            c5 = compute_c5(z)
            grad = soft_max_grad(z, t_smooth, n)
        else:
            c5, grad = c5_subgrad(z)
        h_new = project_feasible(z - lr * grad, n)
        z = h_new + mom * (h_new - h)
        z = np.clip(z, 0.0, 1.0)
        h = h_new
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        mom = max(0.0, mom - 0.002)
    return best_h, best_c5

def _delta_conv(h, i, j, n):
    d = np.zeros(2 * n - 1)
    lo_i = max(i - n + 1, -(n - 1))
    hi_i = min(i, n - 1)
    for s in range(lo_i, hi_i + 1):
        d[s + n - 1] += h[i - s] - 1.0
    lo_i2 = max(-i, -(n - 1))
    hi_i2 = min(n - 1 - i, n - 1)
    for s in range(lo_i2, hi_i2 + 1):
        d[s + n - 1] += h[i + s]
    lo_j = max(j - n + 1, -(n - 1))
    hi_j = min(j, n - 1)
    for s in range(lo_j, hi_j + 1):
        d[s + n - 1] += 1.0 - h[j - s]
    lo_j2 = max(-j, -(n - 1))
    hi_j2 = min(n - 1 - j, n - 1)
    for s in range(lo_j2, hi_j2 + 1):
        d[s + n - 1] -= h[j + s]
    return d

def binary_sa(h, n, time_budget, T_init=0.3, T_final=0.005, seed=0):
    rng = np.random.RandomState(seed)
    if np.allclose(h, np.round(h)) and abs(h.sum() - n / 2.0) < 1.0:
        h_bin = h.copy()
    else:
        idx = np.argsort(h)[::-1]
        h_bin = np.zeros(n)
        h_bin[idx[:n // 2]] = 1.0
    conv = np.correlate(h_bin, 1.0 - h_bin, mode="full")
    dx = 2.0 / n
    c5 = np.max(conv) * dx
    best_h = h_bin.copy()
    best_c5 = c5
    best_conv = conv.copy()
    t_start = time.time()
    while time.time() - t_start < time_budget:
        frac = min(1.0, (time.time() - t_start) / max(time_budget, 1e-10))
        T = T_init * (T_final / T_init) ** frac
        ones = np.where(h_bin == 1.0)[0]
        zeros = np.where(h_bin == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        d = _delta_conv(h_bin, i, j, n)
        new_c5 = np.max(conv + d) * dx
        delta_c5 = new_c5 - c5
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / max(T, 1e-12)):
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            conv = conv + d
            c5 = new_c5
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_bin.copy()
                best_conv = conv.copy()
    return best_h, best_c5, best_conv

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(128, 0.5)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B soft-max at original resolution
    for t_val in [0.05, 0.02, 0.01, 0.005]:
        if tl() < 40:
            break
        try:
            res = minimize(
                lambda h: soft_max_obj(h, t_val, n0) + 50.0 * (np.sum(h) - n0 / 2.0) ** 2,
                best_h,
                jac=lambda h: soft_max_grad(h, t_val, n0) + 100.0 * (np.sum(h) - n0 / 2.0),
                method='L-BFGS-B', bounds=[(0.0, 1.0)] * n0,
                options={'maxiter': 2000, 'ftol': 1e-16})
            ht = project_feasible(res.x, n0)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
        except Exception:
            pass

    # Phase 2: Nesterov PGD smooth
    if tl() > 80:
        ht, c5t = pgd_optimize(best_h, n0, 30000, 0.003, 0.01, True, 0.7)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: Binary SA at original resolution
    if tl() > 250:
        sa_time = min(200, tl() - 80)
        hb, c5b, _ = binary_sa(best_h, n0, sa_time, seed=seed)
        if c5b < best_c5:
            best_h, best_c5 = hb, c5b
        if tl() > 60:
            ht, c5t = pgd_optimize(best_h, n0, 20000, 0.001, 0.005, False, 0.4)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 4: Multi-resolution with binary SA
    for n_try in [80, 112, 160, 200, 256, 320, 400, 480, 640]:
        if tl() < 180:
            break
        if n_try <= n0:
            continue
        nc = len(best_h)
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        # L-BFGS-B
        if tl() > 80:
            try:
                res = minimize(
                    lambda h: soft_max_obj(h, 0.02, n_try) + 50.0 * (np.sum(h) - n_try / 2.0) ** 2,
                    hi,
                    jac=lambda h: soft_max_grad(h, 0.02, n_try) + 100.0 * (np.sum(h) - n_try / 2.0),
                    method='L-BFGS-B', bounds=[(0.0, 1.0)] * n_try,
                    options={'maxiter': 1500, 'ftol': 1e-15})
                hi = project_feasible(res.x, n_try)
            except Exception:
                pass
        # Binary SA
        if tl() > 120:
            sa_time = min(80, tl() - 60)
            hb, c5b, _ = binary_sa(hi, n_try, sa_time, seed=seed + n_try)
            if c5b < best_c5:
                best_h, best_c5 = hb, c5b
        # PGD polish
        if tl() > 60:
            ht, c5t = pgd_optimize(best_h, len(best_h), 15000, 0.001, 0.005, False, 0.3)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 5: Random restarts (binary)
    for r in range(20):
        if tl() < 30:
            break
        hr = rng.uniform(0.0, 1.0, n0)
        hr = project_feasible(hr, n0)
        ht, c5t, _ = binary_sa(hr, n0, min(20, tl() - 30), T_init=1.0, T_final=0.1, seed=seed + 100 + r)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Final continuous polish
    if tl() > 30:
        ht, c5t = pgd_optimize(best_h, len(best_h), 40000, 0.0005, 0.001, False, 0.2)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
