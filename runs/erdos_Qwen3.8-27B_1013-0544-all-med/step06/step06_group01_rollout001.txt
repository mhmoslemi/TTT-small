import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.argsort(1.0 - h)
                for k in range(n):
                    room = 1.0 - h[idx[k]]
                    add = min(room, d2)
                    h[idx[k]] += add
                    d2 -= add
                    if d2 <= 1e-15:
                        break
            else:
                idx = np.argsort(h)
                d2 = -d2
                for k in range(n):
                    room = h[idx[k]]
                    rem = min(room, d2)
                    h[idx[k]] -= rem
                    d2 -= rem
                    if d2 <= 1e-15:
                        break
    return h

def _smooth_obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _active_obj_grad(h, active_lags, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    active_C = C[active_lags]
    f = np.mean(active_C) + beta * (np.sum(h) - 0.5 * n) ** 2
    j_arr = np.array(active_lags)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * (np.mean(ta) - np.mean(tb))
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _opt_smooth(h, n, alpha, beta, maxiter=400):
    def f(h):
        return _smooth_obj_grad(h, alpha, beta)[0]
    def g(h):
        return _smooth_obj_grad(h, alpha, beta)[1]
    res = minimize(f, h, jac=g, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-15})
    return _fix(res.x if res.x is not None else h, n)

def _opt_active(h, n, eps, beta, maxiter=300):
    C = _c5_array(h)
    cm = np.max(C)
    active = np.where(C >= cm - eps)[0]
    if len(active) < 2:
        active = np.where(C >= cm - eps * 3)[0]
    if len(active) < 1:
        return h
    def f(h):
        return _active_obj_grad(h, active, beta)[0]
    def g(h):
        return _active_obj_grad(h, active, beta)[1]
    res = minimize(f, h, jac=g, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 5, 'ftol': 1e-15})
    return _fix(res.x if res.x is not None else h, n)

def _structured_restart(h, n, rng, mode):
    if mode == 0:
        x = np.linspace(0, 2, n, endpoint=False)
        freq = rng.uniform(1.0, 6.0)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.04, 0.12)
        h_new = h + amp * np.sin(2 * np.pi * freq * x + phase)
    elif mode == 1:
        block_size = max(3, n // rng.randint(8, 15))
        n_blocks = n // block_size
        offsets = rng.randn(n_blocks) * rng.uniform(0.03, 0.08)
        h_new = h.copy()
        for b in range(n_blocks):
            s, e = b * block_size, min((b + 1) * block_size, n)
            h_new[s:e] += offsets[b]
    elif mode == 2:
        h_new = h + rng.randn(n) * rng.uniform(0.02, 0.05)
    else:
        noise = rng.randn(n)
        k = np.ones(5) / 5.0
        noise = np.convolve(noise, k, mode='same')
        h_new = h + 0.06 * noise
    return _fix(h_new, n)
