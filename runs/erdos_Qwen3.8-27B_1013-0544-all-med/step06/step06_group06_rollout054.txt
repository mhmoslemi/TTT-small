            break
        if d > 0:
            m = h < 1.0
            cnt = m.sum()
            if cnt == 0:
                break
            h[m] = np.minimum(1.0, h[m] + d / cnt)
        else:
            m = h > 0.0
            cnt = m.sum()
            if cnt == 0:
                break
            h[m] = np.maximum(0.0, h[m] + d / cnt)
    return h

def smooth_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    s = alpha * (c - m)
    mx = s.max()
    if mx > 400:
        return m + (mx + np.log(np.sum(np.exp(s - mx)))) / alpha
    return m + np.log(np.sum(np.exp(s))) / alpha

def sum_eq(h, n):
    return h.sum() - n / 2.0

def sum_jac(h, n):
    return np.ones(n)

def thue_morse_init(n):
    h = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    return project_feasible(h, n)

def local_search_exact(h, n, max_iters=5000):
    best = h.copy()
    best_c = compute_c5(best, n)
    for _ in range(max_iters):
        improved = False
        for i in range(n):
            for j in [(i+1)%n, (i+n//2)%n, (i+3*n//5)%n, (i+n//3)%n]:
                if i == j:
                    continue
                for d in [0.002, 0.005, 0.01, 0.02]:
                    for s in [1, -1]:
                        hi, hj = best[i] + s*d, best[j] - s*d
                        if 0 <= hi <= 1 and 0 <= hj <= 1:
                            h_try = best.copy()
                            h_try[i], h_try[j] = hi, hj
                            c = compute_c5(h_try, n)
                            if c < best_c - 1e-14:
                                best, best_c = h_try, c
                                improved = True
                                break
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = thue_morse_init(n)

    best_h = project_feasible(h0, n)
    best_c5 = compute_c5(best_h, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_jac, 'args': (n,)}]

    def rem():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP continuation - fine alpha schedule
    alphas = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000]
    for alpha in alphas:
        if rem() < 30:
            break
        try:
            r = minimize(smooth_obj, best_h, args=(n, alpha), method='SLSQP',
                        bounds=bounds, constraints=cons,
                        options={'maxiter': 10000, 'ftol': 1e-20})
            ht = project_feasible(r.x, n)
            c = compute_c5(ht, n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = ht, c
        except Exception:
            pass

    # Intermediate local search
    if rem() > 60:
        best_h, best_c5 = local_search_exact(best_h, n, max_iters=2000)

    # Phase 2: Multi-restart exploration
    n_restarts = min(25, max(5, int(rem() // 25)))
    for r_idx in range(n_restarts):
        if rem() < 50:
            break
        scale = 0.03 + 0.15 * (r_idx % 4) * (0.85 ** (r_idx // 4))
        h_p = best_h + rng.normal(0, scale, size=n)
        h_p = project_feasible(h_p, n)
        try:
            r = minimize(smooth_obj, h_p, args=(n, 5000), method='SLSQP',
                        bounds=bounds, constraints=cons,
                        options={'maxiter': 5000, 'ftol': 1e-18})
            ht = project_feasible(r.x, n)
            c = compute_c5(ht, n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = ht, c
        except Exception:
            pass

    # Phase 3: Final polish - very high alpha SLSQP
    if rem() > 20:
        try:
            r = minimize(smooth_obj, best_h, args=(n, 300000), method='SLSQP',
                        bounds=bounds, constraints=cons,
                        options={'maxiter': 20000, 'ftol': 1e-20})
            ht = project_feasible(r.x, n)
            c = compute_c5(ht, n)
            if c < best_c5:
                best_h, best_c5 = ht, c
        except Exception:
            pass

    # Phase 4: Final exact local search
    if rem() > 15:
        best_h, best_c5 = local_search_exact(best_h, n, max_iters=3000)

    # Final feasibility check
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)
    return best_h, best_c5, n
