            break
        
        # Compute influence for all positions
        inf_ones = np.zeros((n_o, n_total))
        for idx, pos in enumerate(ones):
            i1 = pos - shift
            i2 = pos + shift
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            inf_ones[idx, v1] += 1 - h[i1[v1]]
            inf_ones[idx, v2] -= h[i2[v2]]
        
        inf_zeros = np.zeros((n_z, n_total))
        for idx, pos in enumerate(zeros):
            i1 = pos - shift
            i2 = pos + shift
            v1 = (i1 >= 0) & (i1 < n)
            v2 = (i2 >= 0) & (i2 < n)
            inf_zeros[idx, v1] += 1 - h[i1[v1]]
            inf_zeros[idx, v2] -= h[i2[v2]]
        
        # Evaluate all possible swaps
        # For each ii: base = corr - inf_ones[ii]
        # new_max[jj] = max_k(base[k] + inf_zeros[jj, k])
        # gain = current_max - new_max[jj]
        
        best_gain = -np.inf
        best_ii = 0
        best_jj = 0
        all_gains = np.zeros((n_o, n_z))
        
        for ii in range(n_o):
            base = corr - inf_ones[ii]
            new_maxes = np.max(base[np.newaxis, :] + inf_zeros, axis=1)
            all_gains[ii] = current_max - new_maxes
            jj_best = int(np.argmax(all_gains[ii]))
            if all_gains[ii, jj_best] > best_gain:
                best_gain = all_gains[ii, jj_best]
                best_ii = ii
                best_jj = jj_best
        
        # Accept/reject
        if best_gain > 0 or rng.random() < np.exp(best_gain / max(T, 1e-10)):
            h[ones[best_ii]] = 0
            h[zeros[best_jj]] = 1
            corr = compute_corr(h)
            current_max = float(np.max(corr))
            if current_max * dx < best_c5:
                best_c5 = current_max * dx
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        T *= cooling
        if T < 0.001:
            T = 0.001
    
    return best_h, best_c5

import numpy as np

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * 2.0 / n)

def make_binary(h, n, rng=None):
    h = np.round(np.asarray(h, dtype=float)).copy()
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        ones = np.where(h == 1)[0]
        if rng is not None:
            idx = rng.choice(n_ones, n_ones - target, replace=False)
        else:
            idx = np.arange(n_ones - target)
        h[ones[idx]] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        if rng is not None:
            idx = rng.choice(n - n_ones, target - n_ones, replace=False)
        else:
            idx = np.arange(target - n_ones)
        h[zeros[idx]] = 1
    return h

def interpolate_to_n(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.astype(float).copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x_new, x_old, np.asarray(h, dtype=float))
