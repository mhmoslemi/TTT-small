def active_grad(x, k_list):
    n = x.size
    dx = 2.0/n

            grad[lag:] -= x[:n-lag]
        else:
            L = -lag
            grad[L:] += (1.0 - x[:n-L])
            grad[:n-L] -= x[L:]
    return grad * dx / len(k_list)

        if time.time() > deadline:
            return 1e9
        v, _ = smooth_obj(x, tau, topk)
        return v
    def g(x):
        if time.time() > deadline:
            return np.zeros(n)
        _, gr = smooth_obj(x, tau, topk)
        return gr
    try:
        res = so.minimize(f, x0, method='SLSQP', jac=g,
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - 0.5 * n,
                         'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.fun < 1e8:
            return repair(res.x)
    except Exception:
        pass
    return repair(x0)

def sg_momentum(x0, deadline, topk=5, momentum=0.7):
    x = repair(x0)
    n = x.size
    best_x, best_c = x.copy(), exact_c(x)
    v = np.zeros(n)
    step = 0.03
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        c = c_vals(x)
        C = c.max()
        if C < best_c - 1e-15:
            best_c, best_x, stall = C, x.copy(), 0
        else:
            stall += 1
            if stall > 50:
                break
        tol = max(1e-10, 5e-5 * C)
        act = np.where(c >= C - tol)[0]
        if act.size > topk:
            act = np.argsort(c)[::-1][:topk]
        offset = n - 1
        lags = (act - offset).astype(int)
        grad = grad_for_lags(x, lags) * (2.0 / n) / len(lags)
        v = momentum * v - step * grad
        y = repair(x + v)
        cy = exact_c(y)
        if cy < C - 1e-15:
            x = y
        else:
            v *= 0.5
            step *= 0.8
            if step < 1e-6:
                step = 0.03
                v = -step * grad
            y = repair(x + v)
            cy = exact_c(y)
            if cy < C - 1e-15:
                x = y
            else:
                stall += 2
    return best_x, best_c

def fine_polish(x0, deadline, seed=0):
    x = repair(x0)
    n = x.size
    best_x, best_c = x.copy(), exact_c(x)
    rng = np.random.RandomState(seed)
    stall = 0
    it = 0
    while time.time() < deadline:
        it += 1
        c = c_vals(x)
        C = c.max()
        if C < best_c - 1e-15:
            best_c, best_x, stall = C, x.copy(), 0
        else:
            stall += 1
        if stall > 20:
            improved = False
            for sc in [0.004, 0.002, 0.001, 0.0005]:
                y = repair(x + rng.normal(0, sc, n))
                if exact_c(y) < best_c:
                    best_c, best_x, x, stall, improved = exact_c(y), y.copy(), y, 0, True
                    break
            if not improved:
                break
            continue
        tol = max(1e-10, 3e-5 * C)
        act = np.where(c >= C - tol)[0]
        if act.size > 6:
            act = np.argsort(c)[::-1][:6]
        offset = n - 1
        lags = (act - offset).astype(int)
        grad = grad_for_lags(x, lags) * (2.0 / n) / len(lags)
        ok = False
        s = 0.02
        for _ in range(16):
            y = repair(x - s * grad)
            if exact_c(y) < C - 1e-14:
                x = y
                ok = True
                break
            s *= 0.5
            if s < 1e-13:
                break
        if not ok:
            stall += 2
    return best_x, best_c

def resample(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    if n0 == n_new:
        return repair(h0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return repair(np.interp(x1, x0, h0))

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + budget_s
    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size >= 10:
        h0 = repair(np.asarray(raw, dtype=float))
    else:
        h0 = repair(np.full(70, 0.5))

    best_x, best_c = h0.copy(), exact_c(h0)

    # Phase 1: Multi-stage SLSQP
    stages = [
        (0.05, 10, 180), (0.03, 8, 200), (0.015, 6, 220),
        (0.008, 5, 250), (0.004, 4, 250), (0.002, 3, 280),
    ]
    for tau, topk, mi in stages:
        if time.time() > deadline - 180:
            break
        x = slsqp_run(best_x, tau, topk, mi, deadline - 40)
        c = exact_c(x)
        if c < best_c:
            best_c, best_x = c, x

    # Phase 2: Subgradient with momentum
    if time.time() < deadline - 80:
        x, c = sg_momentum(best_x, deadline - 50, topk=5)
        if c < best_c:
            best_c, best_x = c, x

    # Phase 3: Fine polish
    if time.time() < deadline - 20:
        x, c = fine_polish(best_x, deadline - 3, seed=int(seed))
        if c < best_c:
            best_c, best_x = c, x

    # Phase 4: Restart attempts with varied perturbations
    rng = np.random.RandomState(int(seed) + 999)
    for i in range(10):
        if time.time() > deadline - 10:
            break
        sc = 0.004 + 0.004 * i
        x0 = repair(best_x + rng.normal(0, sc, best_x.size))
        tl = max(3, min(15, deadline - time.time() - 3))
        x, c = fine_polish(x0, time.time() + tl, seed=int(seed) + 500 + i)
        if c < best_c:
            best_c, best_x = c, x

    # Phase 5: Try slightly different resolution
    for n_alt in [best_x.size + 5, best_x.size + 10, best_x.size - 5]:
        if time.time() > deadline - 40:
            break
        if n_alt < 30 or n_alt > 200:
            continue
        h_alt = resample(best_x, n_alt)
        c_alt = exact_c(h_alt)
        if c_alt < best_c:
            x = slsqp_run(h_alt, 0.005, 4, 150, deadline - 10)
            c2 = exact_c(x)
            if c2 < best_c:
                best_c, best_x = c2, x

    # Phase 6: Final SLSQP squeeze
    if time.time() < deadline - 5:
        x = slsqp_run(best_x, 0.002, 3, 120, deadline)
        c = exact_c(x)
        if c < best_c:
            best_c, best_x = c, x

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
