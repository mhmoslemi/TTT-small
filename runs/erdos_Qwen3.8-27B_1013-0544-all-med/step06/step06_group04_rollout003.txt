def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_findy(h, n):
    """Project onto {0<=h<=1, sum=h_target} via bisection."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def gradient_c5(h, beta=30.0):
    """Compute C5 and smooth subgradient using softmax over active lags."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    max_val = float(np.max(corr))
    weights = np.exp(beta * (corr - max_val))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for j in range(len(corr)):
        w = weights[j]
        if w < 1e-10:
            continue

        i_lo = max(0, -shift)
        i_hi = min(n - 1, n - 1 - shift)
        if i_lo > i_hi:
            continue
        i_idx = np.arange(i_lo, i_hi + 1)
        m_idx = i_idx + shift
        grad[m_idx] += (1.0 - h[i_idx]) * dx * w
        grad[i_idx] += -h[m_idx] * dx * w
    return max_val, grad
