def get_worst_grad(h):
    """Compute subgradient of max_k c_k at the worst lag."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = int(np.argmax(corr))
    l = k_max - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        if j - l >= 0:
            g += (1.0 - h[j - l])

            g -= h[j + l]

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr_and_lag(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = int(np.argmax(corr))
    lag = k_max - (n - 1)
    return corr, k_max, lag, float(corr[k_max])

def compute_grad_single(h, lag):
    n = len(h)
    k = lag
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += (1.0 - h[:n-k])
        grad[:n-k] -= h[k:]
    else:
        grad[:n+k] += (1.0 - h[-k:])
        grad[-k:] -= h[:n+k]
    return grad

def compute_grad_soft(h, beta):
    """Compute gradient of soft-max approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Soft max weights
    shifted = corr - corr.max()
    w = np.exp(beta * shifted)
    w /= w.sum()
    # Only use top contributing lags
    threshold = w.max() * 0.01
    active = w > threshold
    if not np.any(active):
        active[np.argmax(w)] = True
    grad = np.zeros(n)
    for idx in range(len(corr)):
        if active[idx]:
            lag = idx - (n - 1)
            grad += w[idx] * compute_grad_single(h, lag)
    return grad, float(corr.max())

def normalize_h(h, n):
    h = np.clip(h, 0, 1).astype(float)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h = h * (1.0 + diff / total)
        h = np.clip(h, 0, 1)
    return h

def interpolate_to(h_src, n_new):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_src, h_src)
    return normalize_h(h_new, n_new)

def optimize_sg(h0, max_time, seed, beta=50.0):
    """Subgradient optimization with soft-max gradient."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    start = time.time()
    
    step = 0.02
    min_step = 1e-5
    stall = 0
    
    while (time.time() - start) < max_time:
        grad, c5_cur = compute_grad_soft(h, beta)
        
        h_new = h - step * grad
        h_new = normalize_h(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            step = min(step * 1.05, 0.1)
        elif c5_new < c5_cur:
            stall = max(0, stall - 1)
            step = min(step * 1.02, 0.1)
        else:
            h = h  # reject
            stall += 1
            step = max(step * 0.8, min_step)
            if stall > 30:
                # Perturbation
                n_pert = rng.randint(3, max(4, n // 5))
                idx = rng.choice(n, n_pert, replace=False)
                h[idx] += rng.normal(0, 0.05, n_pert)
                h = normalize_h(h, n)
                step = 0.01
                stall = 0
                beta = max(beta * 0.8, 10.0)
            else:
                h = h_new
                step = max(step * 0.95, min_step)
        
        # Occasionally increase beta (make gradient sharper)
        if stall % 20 == 0:
            beta = min(beta * 1.2, 500.0)
    
    return best_h, best_c5

def optimize_sa(h0, max_time, seed):
    """Simulated annealing with subgradient-informed moves."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    start = time.time()
    
    T0 = 0.03
    T_end = 1e-7
    
    while (time.time() - start) < max_time:
        elapsed = time.time() - start
        frac = elapsed / max_time
        T = max(T0 * (1 - frac)**2, T_end)
        
        # Get gradient direction
        grad, c5_cur = compute_grad_soft(h, 30.0)
        
        # Generate moves

        best_move_c5 = c5_cur
        
        # Move 1: Gradient step
        step = T * 2.0
        h1 = normalize_h(h - step * grad, n)
        c1 = compute_c5(h1)
        if c1 < best_move_c5:
            best_move_c5 = c1; best_move = h1
        
        # Move 2: Random pair
        i, j = rng.choice(n, 2, replace=False)
        delta = rng.normal(0, np.sqrt(T) * 0.08)
        h2 = h.copy()
        h2[i] += delta; h2[j] -= delta
        h2 = normalize_h(h2, n)
        c2 = compute_c5(h2)
        if c2 < best_move_c5:
            best_move_c5 = c2; best_move = h2
        
        # Move 3: Local smooth perturbation
        center = rng.randint(n)
        width = rng.randint(2, max(3, n // 15))
        amp = rng.normal(0, np.sqrt(T) * 0.15)
        h3 = h.copy()
        for d in range(-width, width + 1):
            idx = (center + d) % n
            h3[idx] += amp * np.exp(-3.0 * (d / float(width))**2)
        h3 = normalize_h(h3, n)
        c3 = compute_c5(h3)
        if c3 < best_move_c5:
            best_move_c5 = c3; best_move = h3
        
        # Accept best move (always accept if better, sometimes accept worse)
        if best_move is not None:
            delta_c5 = best_move_c5 - c5_cur
            if delta_c5 < 0 or (T > T_end * 100 and rng.random() < np.exp(-delta_c5 / T)):
                h = best_move
                if c5_cur < best_c5:
                    best_c5 = c5_cur; best_h = h.copy()

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad_soft(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corr - corr.max()
    w = np.exp(beta * shifted)
    w /= w.sum()
    threshold = w.max() * 0.005
    active_idx = np.where(w > threshold)[0]
    if len(active_idx) == 0:
        active_idx = np.array([np.argmax(w)])
    grad = np.zeros(n)
    for idx in active_idx:
        lag = idx - (n - 1)
        k = lag
        if k >= 0:
            grad[k:] += w[idx] * (1.0 - h[:n-k])
            grad[:n-k] -= w[idx] * h[k:]
        else:
            nn = n + k
            grad[:nn] += w[idx] * (1.0 - h[-k:])
            grad[-k:] -= w[idx] * h[:nn]
    return grad, float(corr.max())

def normalize_h(h, n):
    h = np.clip(h, 0, 1).astype(float)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += room * (diff / total)
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h = h * (1.0 + diff / total)
        h = np.clip(h, 0, 1)
    return h

def interpolate_to(h_src, n_new):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_src, h_src)
    return normalize_h(h_new, n_new)

    step = 0.02

    stall = 0

        h_new = h - step * grad
        h_new = normalize_h(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            step = min(step * 1.03, 0.08)
            h = h_new
        elif c5_new < c5_cur + 1e-12:
