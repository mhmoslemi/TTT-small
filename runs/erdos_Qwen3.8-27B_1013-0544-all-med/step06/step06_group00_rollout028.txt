import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_for_lag(h, k):
    """Gradient of corr at lag k (k is the actual lag, can be negative)"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    ka = abs(k)
    if ka == 0:
        # corr at lag 0 = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
        # grad[m] = 1 - 2*h[m]
        grad[:] = (1.0 - 2.0 * h) * dx
    elif k > 0:
        grad[k:] += (1.0 - h[:n-ka])
        grad[:n-ka] -= h[ka:]
    else:
        grad[:n-ka] += (1.0 - h[ka:])
        grad[ka:] -= h[:n-ka]
    return grad * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        d = target_sum - h.sum()
        if abs(d) < 1e-12:
            break
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() == 0:
            # Need to adjust boundary values
            if d > 0:
                # Increase some values
                idx = np.argsort(h)[-1]
                h[idx] = min(1.0, h[idx] + d)
            else:
                idx = np.argsort(h)[0]
                h[idx] = max(0.0, h[idx] + d)
            continue
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_descent(h, n_steps, alpha0, alpha_decay, top_k=1):
    """Projected subgradient descent minimizing max_k corr(k)"""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for step in range(n_steps):
        alpha = alpha0 * (alpha_decay ** step)
        if alpha < 1e-6:
            break
        
        corr = compute_corr(best_h)
        # Get top_k largest correlations
        top_idx = np.argsort(corr)[-top_k:]
        # Average gradient over top lags
        grad_sum = np.zeros(n)
        for idx in top_idx:
            lag = idx - (n - 1)
            grad_sum += grad_for_lag(best_h, lag)
        grad_avg = grad_sum / len(top_idx)
        
        h_new = best_h - alpha * grad_avg
        h_new = project_feasible(h_new, target)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
        best_h = h_new
    
    return best_h, best_c5

def coord_descent_linesearch(h, max_sweeps=10):
    """Coordinate descent with binary search line search"""
    n = len(h)
    target = n / 2.0
    c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        improved = False
        # Process coordinates in random order
        order = np.random.permutation(n)
        for j in order:
            best_val = h[j]
            best_c = c5
            
            # Try a few values via binary search
            # The objective as a function of h[j] (with other coords fixed)
            # is: max_k corr_k, where each corr_k is quadratic in h[j]
            # For a specific k, corr_k = (sum of terms not involving h[j]) + h[j]*(1-h[j+k]) - ... 
            # It's a concave quadratic in h[j] for each k, so max of concave functions
            # is not necessarily concave, but we can still do a discrete search
            
            # Try golden section or just a few points
            lo, hi = 0.0, 1.0
            # Evaluate at a few points
            candidates = [best_val, 0.0, 1.0, (lo+hi)/2, best_val/2, best_val+0.5, best_val-0.5]
            candidates = [max(0.0, min(1.0, c)) for c in candidates]

    n = len(h)
    dx = 2.0 / n
    ka = abs(k)
    if ka == 0:

    grad = np.zeros(n)

        grad[k:] += (1.0 - h[:n-ka])
        grad[:n-ka] -= h[ka:]

        grad[:n-ka] += (1.0 - h[ka:])
        grad[ka:] -= h[:n-ka]
