    val = obj_and_corr(h)[0]
    for sweep in range(max_sweeps):
        if time_limit is not None and time.time() > time_limit:
            break
        improved = False
        for i in range(N):
            j = (i + N // 3) % N
            if j == i:
                j = (i + 1 + N // 2) % N
            s = h[i] + h[j]
            lo = max(0.0, s - 1.0)
            hi = min(1.0, s)
            if hi - lo < 1e-7:
                continue
            best_hi = h[i]
            best_cv = val
            for frac in (0.15, 0.35, 0.5, 0.65, 0.85):
                hi_val = lo + frac * (hi - lo)
                h[i] = hi_val
                h[j] = s - hi_val
                v = obj_and_corr(h)[0]
                if v < best_cv:
                    best_cv = v
                    best_hi = hi_val
            h[i] = best_hi
            h[j] = s - best_hi
            if best_cv < val - 1e-11:
                val = best_cv
                improved = True
        if not improved:
            break
    return h, val

def interpolate_to_n(h_src, n_dst):
    x0 = np.arange(len(h_src)) / len(h_src)
    x1 = np.arange(n_dst) / n_dst
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None

    best_h = None
    best_val = float('inf')

    def update(h, val):
        nonlocal best_h, best_val
        if val < best_val:
            best_val = val
            best_h = h.copy()

    if h_init is not None:
        v0 = obj_and_corr(h_init)[0]
        update(h_init, v0)

    # Phase 1: FISTA on initial at its native resolution
    if h_init is not None and time.time() < t_end - 180:
        h1, v1 = fista_opt(h_init.copy(), len(h_init), rng, min(t0 + 200, t_end - 180), max_iter=80000)
        update(h1, v1)

    # Phase 2: Coordinate descent polish
    if best_h is not None and time.time() < t_end - 150:
        h2, v2 = coord_hash(best_h.copy(), len(best_h), max_sweeps=4, time_limit=t_end - 150)
        update(h2, v2)

    # Phase 3: Try other resolutions
    if best_h is not None:
        n_list = [80, 100, 140, 160, 200, 240]
        for n_t in n_list:
            if time.time() > t_end - 60:
                break
            h_start = interpolate_to_n(best_h, n_t)
            rem = t_end - 40 - time.time()
            if rem < 15:
                break
            h3, v3 = fista_opt(h_start, n_t, rng, t0 + time.time() - t0 + min(rem * 0.5, 50), max_iter=20000)
            update(h3, v3)
            if time.time() < t_end - 30:
                h3c, v3c = coord_hash(h3.copy(), n_t, max_sweeps=2, time_limit=min(t_end - 30, time.time() + 20))
                update(h3c, v3c)

    # Phase 4: Symmetry transforms
    if best_h is not None and time.time() < t_end - 50:
        N_b = len(best_h)
        transforms = [1.0 - best_h, best_h[::-1].copy()]
        for tf in transforms:
            if time.time() > t_end - 25:
                break
            tf = project_feasible(tf, N_b)
            v_tf = obj_and_corr(tf)[0]
            update(tf, v_tf)
            if time.time() < t_end - 25:
                h4, v4 = fista_opt(tf, N_b, rng, min(time.time() + 25, t_end - 20), max_iter=15000)
                update(h4, v4)

    # Phase 5: Multiple restarts from best with varying perturbations
    if best_h is not None and time.time() < t_end - 40:
        N_b = len(best_h)
        for restart in range(20):
            if time.time() > t_end - 15:
                break
            scale = 0.02 + 0.015 * (restart % 5) + 0.01 * (restart // 5)
            h_start = project_feasible(best_h + rng.randn(N_b) * scale, N_b)
            rem = t_end - 10 - time.time()
            if rem < 10:
                break
            h5, v5 = fista_opt(h_start, N_b, rng, time.time() + min(rem * 0.4, 20), max_iter=8000)
            update(h5, v5)

    # Phase 6: Final coordinate descent + FISTA polish
    if best_h is not None and time.time() < t_end - 5:
        h6, v6 = coord_hash(best_h.copy(), len(best_h), max_sweeps=5, time_limit=t_end - 5)
        update(h6, v6)
        if time.time() < t_end - 5:
            h7, v7 = fista_opt(h6, len(h6), rng, t_end, max_iter=30000)
            update(h7, v7)

    # Phase 7: Try binary rounding followed by optimization
    if best_h is not None and time.time() < t_end - 20:
        N_b = len(best_h)
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_feasible(h_bin, N_b)
        v_bin = obj_and_corr(h_bin)[0]
        update(h_bin, v_bin)
        if v_bin < best_val + 0.02 and time.time() < t_end - 10:
            h8, v8 = fista_opt(h_bin, N_b, rng, t_end, max_iter=20000)
            update(h8, v8)

    if best_h is None:
        best_h = 0.5 * np.ones(47)
        best_val = obj_and_corr(best_h)[0]

    return best_h, best_val, len(best_h)
