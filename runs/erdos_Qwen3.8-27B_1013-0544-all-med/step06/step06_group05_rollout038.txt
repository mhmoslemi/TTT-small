    """Precompute P[p] = corr(h, e_p) - corr(e_p, 1-h) for all p."""
    n = len(h)
    g = 1.0 - h
    P = np.zeros((n, 2*n-1))
    I = np.eye(n)
    for p in range(n):
        P[p] = np.correlate(h, I[p], mode='full') - np.correlate(I[p], g, mode='full')
    return P

def update_P(P, h, i, j):
    """Update P after swap h[i]: 1->0, h[j]: 0->1.
    ΔP_p = -corr(e_i, e_p) + corr(e_j, e_p) - corr(e_p, e_i) + corr(e_p, e_j)
    Each corr(e_a, e_b) is a delta vector: 1 at position (n-1)+(b-a) if valid."""
    n = len(h)
    # corr(e_i, e_p): 1 at position (n-1) + (p-i) if 0 <= (n-1)+(p-i) < 2n-1
    # corr(e_j, e_p): 1 at position (n-1) + (p-j)
    # corr(e_p, e_i): 1 at position (n-1) + (i-p)
    # corr(e_p, e_j): 1 at position (n-1) + (j-p)
    
    # For each p, the updates are at specific positions
    # ΔP_p[(n-1)+(p-i)] -= 1  (if valid)
    # ΔP_p[(n-1)+(p-j)] += 1  (if valid)
    # ΔP_p[(n-1)+(i-p)] -= 1  (if valid)
    # ΔP_p[(n-1)+(j-p)] += 1  (if valid)
    
    n1 = n - 1
    positions = np.arange(n)
    
    # Vectorized update:
    # For term -corr(e_i, e_p): position in P[p] is n1 + (p-i), valid if 0 <= n1+p-i < 2n-1
    pos1 = n1 + (positions - i)  # shape (n,)
    mask1 = (pos1 >= 0) & (pos1 < 2*n-1)
    P[positions[mask1], pos1[mask1]] -= 1
    
    pos2 = n1 + (positions - j)
    mask2 = (pos2 >= 0) & (pos2 < 2*n-1)
    P[positions[mask2], pos2[mask2]] += 1
    
    pos3 = n1 + (i - positions)
    mask3 = (pos3 >= 0) & (pos3 < 2*n-1)
    P[positions[mask3], pos3[mask3]] -= 1
    
    pos4 = n1 + (j - positions)
    mask4 = (pos4 >= 0) & (pos4 < 2*n-1)
    P[positions[mask4], pos4[mask4]] += 1
    
    return P

    Returns (best_i, best_j, predicted_c5)."""

    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_o = len(ones)
    n_z = len(zeros)

    best_b = best_flat % n_z

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def interp_resample(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return h_new
