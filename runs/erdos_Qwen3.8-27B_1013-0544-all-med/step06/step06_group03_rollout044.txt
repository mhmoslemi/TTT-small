    def time_left():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float)
        n_parent = len(h_parent)
    else:
        h_parent = np.full(64, 0.5)
        n_parent = 64

    best_h = h_parent.copy()
    best_c5 = compute_c5(best_h)
    parent = h_parent.copy()
    n_parent = len(parent)

    resolutions = [40, 48, 56, 64, 80, 96, 112, 128, 160, 192, 224, 256, 320]
    resolutions = [r for r in resolutions if r <= 400]

    for n in resolutions:
        if time_left() < 80:
            break

        if n == n_parent:
            h_start = parent.copy()
        elif n > n_parent:
            h_start = upsample(parent, n)
        else:
            x_old = np.linspace(0, 2.0, n_parent, endpoint=False)
            x_new = np.linspace(0, 2.0, n, endpoint=False)
            h_start = np.interp(x_new, x_old, parent)
            h_start = project(h_start, n)

        c5_before = compute_c5(h_start)
        local_best_h = h_start.copy()
        local_best_c5 = c5_before

        n_starts = max(4, min(12, int(time_left() / 40)))
        for si in range(n_starts):
            if time_left() < 40:
                break

            if si == 0:
                hs = h_start.copy()
            elif si == 1:
                hs = h_start + 0.08 * rng.randn(n)
                hs = project(hs, n)
            elif si == 2:
                hs = make_sine_init(n, seed + si * 1000)
            elif si == 3:
                hs = make_step_init(n, seed + si * 1000)
            elif si == 4:
                hs = h_start
                hs[rng.choice(n, size=n // 4, replace=False)] = rng.uniform(0, 1, size=n // 4)
                hs = project(hs, n)
            else:
                hs = rng.uniform(0, 1, n)
                hs = project(hs, n)

            # Coarse phase
            h1, c1 = nesterov_opt(hs, n, iters=1500, lr=0.012, t0=0.15, t1=0.03, mom0=0.3, mom1=0.85)
            if time_left() < 20:
                break
            # Medium phase
            h2, c2 = nesterov_opt(h1, n, iters=2500, lr=0.004, t0=0.03, t1=0.005, mom0=0.85, mom1=0.97)
            if time_left() < 20:
                break
            # Fine phase
            h3, c3 = nesterov_opt(h2, n, iters=2000, lr=0.001, t0=0.005, t1=0.0005, mom0=0.97, mom1=0.995)
            # Polish with exact subgradient
            h4, c4 = nesterov_opt(h3, n, iters=1000, lr=0.0003, t0=0.0005, t1=0.0001, mom0=0.99, mom1=0.999)

            if c4 < local_best_c5:
                local_best_c5 = c4
                local_best_h = h4.copy()
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4.copy()

        # Update parent if improved
        if local_best_c5 < compute_c5(parent, n_parent) or n == n_parent:
            if n == n_parent:
                parent = local_best_h.copy()
            else:
                # Interpolate back to parent resolution for cross-resolution use
                if n > n_parent:
                    x_new = np.linspace(0, 2.0, n_parent, endpoint=False)
                    x_old = np.linspace(0, 2.0, n, endpoint=False)
                    parent = project(np.interp(x_new, x_old, local_best_h), n_parent)
                elif n < n_parent:
                    parent = upsample(local_best_h, n_parent)

    # Final polish at best resolution
    if time_left() > 30:
        nf = len(best_h)
        hf, cf = nesterov_opt(best_h, nf, iters=3000, lr=0.0005, t0=0.003, t1=0.0002, mom0=0.95, mom1=0.998)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf

    return best_h, best_c5, len(best_h)
