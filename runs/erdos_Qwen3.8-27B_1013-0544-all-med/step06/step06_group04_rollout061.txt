def project_to_cell(x, target_sum):
    lo, hi = float(x.min()) - 2.0, float(x.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    e = np.exp((corr - m) / tau)
    p = e / e.sum()
    g = np.zeros(n)

        else:
            g[:n - d] += pk * (1.0 - h[d:])
            g[d:] -= pk * h[:n - d]

import numpy as np
import time

def project_to_cell(x, target_sum):
    lo, hi = float(x.min()) - 2.0, float(x.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def get_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    e = np.exp((corr - m) / tau)
    p = e / e.sum()
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        d = abs(k - (n - 1))
        if d >= n:
            continue
        if d == 0:
            g += pk * (1.0 - 2.0 * h)
        elif k >= n - 1:
            g[d:] += pk * (1.0 - h[:n - d])
            g[:n - d] -= pk * h[d:]
        else:
            g[:n - d] += pk * (1.0 - h[d:])
            g[d:] -= pk * h[:n - d]
    return dx * g

def do_opt(h_init, target, max_iter, tau0, lr0, deadline):
    h_c = h_init.copy()
    n = len(h_c)
    v = np.zeros(n)
    beta = 0.9
    tau, lr = tau0, lr0
    best_c5 = compute_c5(h_c)
    best_h = h_c.copy()
    patience = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = get_grad(h_c, tau)
        v = beta * v + (1.0 - beta) * g
        h_c = project_to_cell(h_c - lr * v, target)
        c5 = compute_c5(h_c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_c.copy()
            patience = 0
        else:
            patience += 1
        if patience > 300:
            h_c = best_h + np.random.RandomState(it).normal(0, 0.03, n)
            h_c = project_to_cell(h_c, target)
            v = np.zeros(n)
            patience = 0
        if it % 200 == 199:
            tau = max(tau * 0.8, 0.003)
            lr = max(lr * 0.93, 0.001)
    return best_h, best_c5
