    cm = float(np.max(c))
    z = (c - cm) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros(len(c)); w[int(np.argmax(c))] = 1.0
        return cm, w
    w = e / s
    return cm + tau * np.log(s), w

def resample(h0, nn):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    nn = int(nn)
    if n0 == nn:
        return repair(h0)
    if n0 == 0:
        return repair(np.full(nn, 0.5))
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(nn) + 0.5) * (2.0 / nn)
    return repair(np.interp(x1, x0, h0))

def start_sin(n, freq, amp, phase, seed):
    rng = np.random.RandomState(seed)
    t = (np.arange(n) + 0.5) * (2.0 / n)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * t / 2.0 + phase)
    h += 0.01 * rng.normal(0, 1, n)
    return repair(h)

def start_block(n, period, duty, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    for i in range(n):
        pos = (i % period)
        h[i] = 1.0 if pos < int(period * duty) else 0.0
    h += 0.03 * rng.normal(0, 1, n)
    return repair(h)

def start_comb(n, period, width, seed):
    rng = np.random.RandomState(seed)
    h = np.full(n, 0.3)
    for i in range(0, n, period):
        end = min(i + width, n)
        h[i:end] = 0.9
    h += 0.02 * rng.normal(0, 1, n)
    return repair(h)

def start_random(n, seed, freq):
    rng = np.random.RandomState(seed)
    t = (np.arange(n) + 0.5) * (2.0 / n)
    amp = 0.3 + 0.2 * rng.rand()
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * t / 2.0 + rng.uniform(0, 6.28))
    h += 0.06 * rng.normal(0, 1, n)
    return repair(h)

def optimize_one(x0, tlim, seed):
    x = repair(x0)
    best_x = x.copy()
    best_c = exact_c(x)
    t0 = time.time()

    stages = [(0.04, 200), (0.015, 300), (0.006, 350), (0.002, 400), (0.0008, 400)]
    for tau, mi in stages:
        if time.time() - t0 > tlim * 0.75:
            break
        def fun(x):
            return sg_c(x, tau)[0]
        def jac(x):
            return sg_c(x, tau)[1]
        xr = slsqp(fun, jac, x, mi)
        if xr is not None:
            xr = repair(xr)
            c = exact_c(xr)
            if c < best_c:
                best_c = c
                best_x = xr.copy()
            if c <= best_c + 5e-5:
                x = xr
            else:
                x = best_x

    if time.time() - t0 < tlim * 0.88:
        for lam in (10.0, 100.0):
            if time.time() - t0 > tlim * 0.92:
                break
            xr = lbfgs_smooth(x, 0.001, lam, 300)
            if xr is not None:
                c = exact_c(xr)
                if c < best_c:
                    best_c = c
                    best_x = xr.copy()
                if c <= best_c + 5e-5:
                    x = xr

    tl = tlim - (time.time() - t0)
    if tl > 2.0:
        xp, cp = polish(best_x, min(6.0, tl * 0.8), seed)
        if cp < best_c:
            best_c = cp
            best_x = xp
    return best_x, best_c

def polish(x0, tlim, seed):
    x = repair(x0).copy()
    best_x = x.copy()
    best_c = exact_c(x)
    rng = np.random.RandomState(seed)
    deadline = time.time() + tlim
    step = 0.4
    stuck = 0

    while time.time() < deadline:
        c = cvals(x)
        C = float(np.max(c))
        if C < best_c - 1e-15:
            best_c = C
            best_x = x.copy()
            stuck = 0
        else:
            stuck += 1
        if stuck > 20:
            improved = False
            for sg in (0.01, 0.005, 0.002, 0.001):
                y = repair(x + rng.normal(0, sg, x.size))
                cy = exact_c(y)
                if cy < best_c:
                    best_c = cy
                    best_x = y.copy()
                    x = y
                    stuck = 0
                    improved = True
                    break
            if not improved:
                break
            continue

        tol = max(1e-9, 5e-6 * C)
        act = np.where(c >= C - tol)[0]
        if act.size > 4:
            act = np.argsort(c)[::-1][:4]
        n = x.size
        g = np.zeros(n)
        dx = 2.0 / n
        for k in act:
            s = int(k) - (n - 1)
            if s >= 0:
                v = np.arange(n - s)
                g[v] += (1.0 - x[v + s])
                g[v + s] += -x[v]
            else:
                t = -s
                g[:n - t] += (1.0 - x[t:])
                g[t:] += -x[:n - t]
        g *= dx / max(1, act.size)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        acc = False
        for _ in range(20):
            y = repair(x - step * g)
            cy = exact_c(y)
            if cy < C - 1e-13:
                x = y
                acc = True
                break
            step *= 0.5
            if step < 1e-11:
                break
        if acc:
            stuck = 0
            step = min(step * 1.2, 0.5)
        else:
            stuck += 1
        if rng.rand() < 0.1:
            y = repair(x + rng.normal(0, 0.003, x.size))
            cy = exact_c(y)
            if cy < best_c:
                best_c = cy
                best_x = y.copy()
                x = y
                stuck = 0
    return best_x, best_c

def coordinate_polish(x0, tlim, seed):
    x = repair(x0).copy()
    best_c = exact_c(x)
    n = x.size
    rng = np.random.RandomState(seed)
    deadline = time.time() + tlim
    deltas = [0.05, 0.02, 0.01, 0.005, 0.002]
    while time.time() < deadline:
        improved = False
        order = rng.permutation(n)
        for i in order:
            if time.time() > deadline:
                break
            for d in deltas:
                for sign in (1, -1):
                    y = x.copy()
                    y[i] += sign * d
                    y = repair(y)
                    cy = exact_c(y)
                    if cy < best_c - 1e-14:
                        x = y
                        best_c = cy
                        improved = True
                        break
                if improved:
                    break
    return x, best_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    sv = int(seed)
    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)

    n = 70
    n2 = 80

    starts = []
    if init is not None and init.size >= 10:
        starts.append(resample(init, n))
        starts.append(resample(init, n2))
    starts.append(start_sin(n, 1, 0.38, 0.5, sv))
    starts.append(start_sin(n, 2, 0.32, 1.2, sv + 1))
    starts.append(start_sin(n, 3, 0.28, 2.1, sv + 2))
    starts.append(start_block(n, 7, 0.5, sv + 3))
    starts.append(start_block(n, 11, 0.45, sv + 4))
    starts.append(start_comb(n, 5, 2, sv + 5))
    starts.append(start_comb(n, 8, 3, sv + 6))
    starts.append(start_random(n, sv + 10, 2))
    starts.append(start_random(n, sv + 20, 3))

    best_x = None
    best_c = 1e9
    t0 = time.time()
    total_time = budget_s
    per_start = total_time * 0.22
    max_starts = max(3, int(total_time * 0.22 / 40) + 2)

    for i, x0 in enumerate(starts[:max_starts]):
        rem = deadline - time.time()
        if rem < 80:
            break
        alloc = min(per_start, rem * 0.9)
        x, c = optimize_one(x0, alloc, sv + i * 7919)
        if c < best_c:
            best_c = c
            best_x = x

    if best_x is None:
        best_x = repair(np.full(n, 0.5))
        best_c = exact_c(best_x)

    rem = deadline - time.time()
    if rem > 15:
        best_x, best_c = polish(best_x, rem * 0.5, sv + 99999)

    rem = deadline - time.time()
    if rem > 5:
        cx, cc = coordinate_polish(best_x, min(rem * 0.7, 60.0), sv + 88888)
        if cc < best_c:
            best_c = cc
            best_x = cx

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
