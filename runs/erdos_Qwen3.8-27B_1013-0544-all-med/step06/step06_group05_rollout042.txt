import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_and_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), np.argmax(corr)

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.round(h)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(ones)
        for idx in ones[:n_ones - target]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        rng = np.random.RandomState(0)
        rng.shuffle(zeros)
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def swap_eval(h, i, j, n):
    """Evaluate C5 after swapping positions i (1->0) and j (0->1)"""
    h[i] = 0.0
    h[j] = 1.0
    c5 = compute_c5(h)
    h[i] = 1.0
    h[j] = 0.0
    return c5

def local_search_binary(h0, max_time=60.0, seed=42):
    n = len(h0)
    h = h0.copy()
    current_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    
    while (time.time() - start) < max_time:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_gain = 1e-14
        best_move = None
        count = 0
        
        # Generate random subset of swaps
        n_swaps = min(len(ones) * len(zeros), 3000)
        for _ in range(n_swaps):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            new_c5 = swap_eval(h, i, j, n)
            gain = current_c5 - new_c5
            count += 1
            if gain > best_gain:
                best_gain = gain
                best_move = (i, j)
        
        if best_move is not None:
            h[best_move[0]] = 0.0
            h[best_move[1]] = 1.0
            current_c5 = compute_c5(h)
        else:

            found = False
            n_dswaps = min(1000, len(ones)*len(ones)//2 * len(zeros)*len(zeros)//2)
            for _ in range(n_dswaps):
                i1 = ones[rng.randint(len(ones))]
                i2 = ones[rng.randint(len(ones))]
                j1 = zeros[rng.randint(len(zeros))]
                j2 = zeros[rng.randint(len(zeros))]
                if i1 == i2 or j1 == j2:

                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                new_c5 = compute_c5(h)

                    current_c5 = new_c5
                    found = True
                    break
                h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_binary(h, n, seed=0):
    """Ensure h is binary with exactly n/2 ones"""
    h = np.round(h).astype(float)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        if seed >= 0:
            rng = np.random.RandomState(seed)
            permutations = rng.permutation(idx)
            h[permutations[:n_ones - target]] = 0
        else:
            h[idx[n_ones-target:]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        if seed >= 0:
            rng = np.random.RandomState(seed)
            permutations = rng.permutation(idx)
            h[permutations[:target - n_ones]] = 1
        else:
            h[idx[:target-n_ones]] = 1
    return h

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_binary(h, n, seed=0):
    h = np.round(h).astype(float)
    n_ones = int(h.sum())
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones > target:
        idx = np.where(h == 1)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:n_ones - target]]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        perm = rng.permutation(len(idx))
        h[idx[perm[:target - n_ones]]] = 1
    return h

def exhaustive_ls(h0, max_time, use_double=True):
    n = len(h0)
    h = h0.copy()
    cur = compute_c5(h)
    t0s = time.time()
    stall = 0
    while time.time() - t0s < max_time and stall < 20:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_gain = 0
        best_mv = None
        for i in ones:
            for j in zeros:
                h[i] = 0; h[j] = 1
                nc = compute_c5(h)
                h[i] = 1; h[j] = 0
                g = cur - nc
                if g > best_gain:
                    best_gain = g
                    best_mv = ('s', i, j)
        if best_mv is None and use_double:
            no = len(ones); nz = len(zeros)
            ntry = min(5000, no * (no - 1) * nz * (nz - 1) // 4)
            for _ in range(ntry):
                i1 = ones[np.random.randint(no)]
                i2 = ones[np.random.randint(no)]
                j1 = zeros[np.random.randint(nz)]
                j2 = zeros[np.random.randint(nz)]
                if i1 == i2 or j1 == j2:
                    continue
                h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
                nc = compute_c5(h)
                h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
