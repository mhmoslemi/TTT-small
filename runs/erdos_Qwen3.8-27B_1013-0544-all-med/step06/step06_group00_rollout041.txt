import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def soft_obj_and_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = np.max(corr)
    shifted = temp * (corr - m)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    S = exp_s.sum()
    w = exp_s / S
    obj = (m + (max_s + np.log(S)) / temp) * dx
    
    g = (1.0 - h)[::-1]  # reversed
    grad = np.zeros(n)
    for j in range(n):
        first = np.dot(w[j:j+n], g)
        second = np.dot(w[n-1-j:2*n-1-j], h)
        grad[j] = dx * (first - second)
    
    return obj, grad

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

    ts = n / 2.0
    
    def obj(x):
        val, _ = soft_obj_and_grad(x, temp)
        return val

        _, g = soft_obj_and_grad(x, temp)
        return g

import numpy as np
from scipy.optimize import minimize
import time

_pen_temp = 30.0
_pen_target = 0.0
_pen_weight = 1.0e6

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def soft_obj_and_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    m = float(np.max(corr))
    shifted = temp * (corr - m)
    max_s = float(np.max(shifted))
    exp_s = np.exp(shifted - max_s)
    S = float(exp_s.sum())
    w = exp_s / S
    obj = (m + (max_s + np.log(S)) / temp) * dx
    g_rev = (1.0 - h)[::-1]
    grad = np.empty(n)
    for j in range(n):
        grad[j] = float(np.dot(w[j:j+n], g_rev)) - float(np.dot(w[n-1-j:2*n-1-j], h))
    grad *= dx
    return obj, grad

def project_feasible(h, target_sum):
    h = np.clip(h, 0.0, 1.0).astype(float).copy()
    for _ in range(2000):
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h
