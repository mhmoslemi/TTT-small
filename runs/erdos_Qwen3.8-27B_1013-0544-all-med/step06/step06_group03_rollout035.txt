import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def soft_max_val_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    shifted = (conv - m) / t
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    sum_exps = np.sum(exps)
    value = (m + t * np.log(sum_exps)) * dx
    weights = exps / sum_exps
    
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-14:
            continue
        s = idx - (n - 1)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            grad[:n-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n-s]
        else:
            s2 = -s
            grad[s2:] += w * (1.0 - h[:n-s2])
            grad[:n-s2] -= w * h[s2:]
    
    return value, grad * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.clip(h + mid, 0, 1).sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2, 0.0, 1.0)

def fista_optimize(h_init, t_start, t_end, max_iter, time_limit, t0):
    n = len(h_init)
    h = h_init.copy().astype(float)
    y = h.copy()
    t_mom = 1.0
    L = 5.0 / n  # initial Lipschitz estimate
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for k in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        # Anneal temperature
        frac = k / max(1, max_iter - 1)
        t = t_start * (t_end / t_start) ** frac
        
        # Gradient at y
        val_y, grad_y = soft_max_val_grad(y, t)
        
        # FISTA step
        x_new = project_feasible(y - grad_y / L, n)
        
        # Backtracking (limited)
        for _ in range(5):
            val_new = soft_max_val_grad(x_new, t)[0]
            if val_new <= val_y + np.dot(grad_y, x_new - y) + 0.5 * L * np.sum((x_new - y)**2) + 1e-10:
                break
            L = min(L * 1.5, 100.0 / n * 100)
            x_new = project_feasible(y - grad_y / L, n)
        
        # Decay L slowly
        L = max(L * 0.999, 1.0 / n)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2)) / 2.0

        y = np.clip(y, 0.0, 1.0)
        h = x_new
        t_mom = t_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def soft_max_val_grad(h, t):
    n = len(h)
    dx = 2.0 / n

    shifted = (conv - m) / t
    shifted -= np.max(shifted)
    exps = np.exp(shifted)
    sum_exps = np.sum(exps)
    value = (m + t * np.log(sum_exps)) * dx
    weights = exps / sum_exps
