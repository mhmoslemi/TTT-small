    for _ in range(50):
        d = t - x.sum()
        if abs(d) < 1e-14:
            break
        if d > 0:
            s = 1 - x
            m = s > 1e-12
            if not m.any():
                break
            x[m] += d * s[m] / s[m].sum()
        else:
            m = x > 1e-12
            if not m.any():
                break
            x[m] += d * x[m] / x[m].sum()
        np.clip(x, 0, 1, out=x)
    return x

def soft_c(h, tau):
    c = np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
    if tau < 1e-8:
        return float(np.max(c))
    z = c / tau
    z -= z.max()
    return tau * np.log(np.exp(z).sum())

def soft_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    z = c / tau
    z -= z.max()
    w = np.exp(z)
    w /= w.sum()
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-12:
            continue
        m = k - (n - 1)
        lo1 = max(0, m)
        hi1 = min(n - 1, n - 1 + m)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += dx * w[k] * (1 - h[lo1 - m:hi1 - m + 1])
        lo2 = max(0, -m)
        hi2 = min(n - 1, n - 1 - m)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= dx * w[k] * h[lo2 + m:hi2 + m + 1]
    return g

def act_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    C = c.max()
    tol = max(1e-8, 1e-5 * C)
    act = np.where(c >= C - tol)[0]
    if len(act) > 8:
        act = np.argsort(c)[::-1][:8]
    g = np.zeros(n)
    for k in act:
        m = k - (n - 1)
        lo1 = max(0, m)
        hi1 = min(n - 1, n - 1 + m)
        if lo1 <= hi1:
            g[lo1:hi1 + 1] += dx * (1 - h[lo1 - m:hi1 - m + 1])
        lo2 = max(0, -m)
        hi2 = min(n - 1, n - 1 - m)
        if lo2 <= hi2:
            g[lo2:hi2 + 1] -= dx * h[lo2 + m:hi2 + m + 1]
    return g / len(act)

def per_h(f):
    return np.tile(f, _G['L'])

def per_obj(f):
    return soft_c(per_h(f), _G['tau'])

def per_obj_g(f):
    p = len(f)
    eps = 1e-6
    g = np.zeros(p)
    for i in range(p):
        fp = f.copy()
        fp[i] += eps
        fm = f.copy()
        fm[i] -= eps
        g[i] = (per_obj(fp) - per_obj(fm)) / (2 * eps)
    return g

def per_obj_max(f):
    return c5(per_h(f))

def per_obj_max_g(f):
    h = per_h(f)
    n = len(h)
    p = len(f)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    C = c.max()
    tol = max(1e-8, 1e-5 * C)
    act = np.where(c >= C - tol)[0]
    if len(act) > 6:
        act = np.argsort(c)[::-1][:6]
    g = np.zeros(p)
    for k in act:
        m = k - (n - 1)
        lo1 = max(0, m)
        hi1 = min(n - 1, n - 1 + m)
        if lo1 <= hi1:
            for idx in range(hi1 - lo1 + 1):
                g[(lo1 + idx) % p] += dx * (1 - h[lo1 + idx - m])
        lo2 = max(0, -m)
        hi2 = min(n - 1, n - 1 - m)
        if lo2 <= hi2:
            for idx in range(hi2 - lo2 + 1):
                g[(lo2 + idx) % p] -= dx * h[lo2 + idx + m]
    return g / len(act)

def full_obj(x):
    return soft_c(x, _G['tau'])

def full_obj_g(x):
    return soft_grad(x, _G['tau'])

def full_obj_max(x):
    return c5(x)

def full_obj_max_g(x):
    return act_grad(x)

def eq_fun(x):
    return np.array([x.sum() - 0.5 * len(x)])

def eq_jac(x):
    return np.ones((1, len(x)))

def eq_fun_f(f):
    return np.array([f.sum() - 0.5 * len(f)])

def eq_jac_f(f):
    return np.ones((1, len(f)))

def resample(h0, n_new):
    n0 = len(h0)
    if n0 == n_new:
        return repair(h0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return repair(np.interp(x1, x0, h0))

def slsqp_per(p, L, tlim, seed):
    global _G
    _G['L'] = L
    start_t = time.time()
    rng = np.random.RandomState(seed)
    best_f = None
    best_c = 1e9
    starts = [np.full(p, 0.5)]
    for _ in range(3):
        starts.append(0.5 + 0.4 * rng.randn(p))
    for _ in range(2):
        f = np.zeros(p)
        f[rng.permutation(p)[:p // 2]] = 1
        starts.append(f)
    for fr in range(1, min(p, 4)):
        starts.append(0.5 + 0.3 * np.sin(2 * np.pi * fr * np.arange(p) / p + rng.uniform(0, 6.28)))
    bounds = [(0, 1)] * p
    cons = [{'type': 'eq', 'fun': eq_fun_f, 'jac': eq_jac_f}]
    for f0 in starts[:8]:
        if time.time() - start_t > tlim:
            break
        x = repair_f(f0)
        for tau in [0.05, 0.01, 0.003, 0.0]:
            if time.time() - start_t > tlim:
                break
            _G['tau'] = tau
            if tau > 0:
                res = so.minimize(per_obj, x, jac=per_obj_g, method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 80, 'ftol': 1e-12})
            else:
                res = so.minimize(per_obj_max, x, jac=per_obj_max_g, method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 50, 'ftol': 1e-12})
            if res is not None and res.x is not None:
                fn = repair_f(res.x)
                cn = c5(per_h(fn))
                if cn < best_c:
                    best_c = cn
                    best_f = fn.copy()
                x = fn
    if best_f is not None:
        return per_h(best_f), best_c
    return None, 1e9

def slsqp_full(x0, tlim, seed):
    global _G
    x = repair(x0)
    best_x = x.copy()
    best_c = c5(x)
    start_t = time.time()
    n = len(x)
    scale = 1.0 if n <= 100 else 0.6
    stages = [(0.04, 180), (0.015, 220), (0.005, 250), (0.0015, 250), (0.0004, 200), (0.0, 120)]
    stages = [(t, max(15, int(m * scale))) for t, m in stages]
    bounds = [(0, 1)] * n
    cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
    for tau, mi in stages:
        if time.time() - start_t > tlim:
            break
        _G['tau'] = tau
        if tau > 0:
            res = so.minimize(full_obj, x, jac=full_obj_g, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': mi, 'ftol': 1e-12})
        else:
            res = so.minimize(full_obj_max, x, jac=full_obj_max_g, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': mi, 'ftol': 1e-12})
        if res is not None and res.x is not None:
            xr = repair(res.x)
            cr = c5(xr)
            if cr < best_c:
                best_c = cr
                best_x = xr.copy()
            x = xr if cr < best_c + 2e-4 else best_x
    return best_x, best_c

def polish(x0, tlim, seed):
    x = repair(x0)
    best = x.copy()
    bc = c5(x)
    start_t = time.time()
    rng = np.random.RandomState(seed)
    step = 0.25
    noi = 0
    while time.time() - start_t < tlim:
        C = c5(x)
        if C < bc - 1e-14:
            bc = C
            best = x.copy()
            noi = 0
        else:
            noi += 1
        if noi > 25:
            improved = False
            for sg in [0.01, 0.005, 0.002, 0.001]:
                y = repair(x + rng.normal(0, sg, len(x)))
                if c5(y) < bc:
                    bc = c5(y)
                    best = y.copy()
                    x = y
                    noi = 0
                    improved = True
                    break
            if not improved:
                break
            continue
        g = act_grad(x)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = min(step * 1.05, 0.5)
        acc = False
        for _ in range(14):
            y = repair(x - step * g)
            if c5(y) < C - 1e-13:
                x = y
                acc = True
                break
            step *= 0.5
            if step < 1e-10:
                break
        if not acc:
            noi += 1
        if rng.rand() < 0.15:
            y = repair(x + rng.normal(0, 0.003, len(x)))
            if c5(y) < bc:
                bc = c5(y)
                best = y.copy()
                x = y
                noi = 0
    return best, bc

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    sv = int(seed)
    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)
    best_x = None
    best_c = 1e9

    # Phase 1: Periodic optimization for diverse basins
    if time.time() < deadline - 120:
        for p in [3, 4, 5, 6, 7, 8, 10]:
            if time.time() > deadline - 120:
                break
            for L in [10, 16, 25]:
                n = p * L
                if n < 20 or n > 200:
                    continue
                if time.time() > deadline - 100:
                    break
                h, cc = slsqp_per(p, L, 12.0, sv + p * 100 + L)
                if h is not None and cc < best_c:
                    best_c = cc
                    best_x = h

    # Phase 2: Full optimization
    starts = []
    if init is not None and len(init) >= 10:
        starts.append(resample(init, min(len(init), 120)))
    if best_x is not None:
        starts.append(resample(best_x, 100))
    rng = np.random.RandomState(sv)
    n_start = 100
    if starts:
        base = starts[0]
    else:
        base = repair(np.full(n_start, 0.5))
        starts.append(base)
    for s in [0.02, 0.06, 0.12]:
        starts.append(repair(base + rng.normal(0, s, len(base))))
    t = (np.arange(len(base)) + 0.5) * (2.0 / len(base))
    for fr in [1, 2, 3]:
        starts.append(repair(0.5 + 0.35 * np.sin(2 * np.pi * fr * t / 2 + rng.uniform(0, 6.28))))
    starts = starts[:7]

    per_t = max(12.0, min(35.0, (deadline - time.time()) / max(1, len(starts)) / 1.8))
    for i, x0 in enumerate(starts):
        if time.time() > deadline - 40:
            break
        x, cc = slsqp_full(x0, per_t, sv + 5000 * i + len(x0))
        if cc < best_c:
            best_c = cc
            best_x = x

    # Phase 3: Polish
    if best_x is not None and time.time() < deadline - 15:
        tl = min(35.0, deadline - time.time() - 8)
        best_x, best_c = polish(best_x, tl, sv + 7777)

    if best_x is None:
        best_x = repair(np.full(60, 0.5))
        best_c = c5(best_x)

    best_x = repair(best_x)
    best_c = c5(best_x)
    return best_x, best_c, len(best_x)
