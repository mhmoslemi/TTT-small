    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def compute_c5_vec(h, n):
    dx = 2.0 / n
    return np.correlate(h, 1 - h, mode='full') * dx

def project_feasible(h, n):
    """Project onto {0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection for the sum constraint
    if h.sum() == target:
        return h
    # If sum is too high, we need to reduce; if too low, increase
    # General approach: find t such that sum(clip(y - t, 0, 1)) = target
    # where y is the "unconstrained" point
    
    # Simpler iterative approach (works well in practice)
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h = h + (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h = h + (diff / total) * h
    return np.clip(h, 0.0, 1.0)
