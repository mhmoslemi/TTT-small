def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_binary(h, n):
    """Ensure h is binary with exactly n//2 ones"""
    h = np.round(h).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

            total_room = np.sum(room)
            if total_room > 1e-10:
                h += room * (diff / total_room)
            else:
                h += diff / n
        else:

            if total > 1e-10:

    current_c5 = float(np.max(corr) * dx)

    start = time.time()
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    no_stall = 0
    while (time.time() - start) < max_time:

        zeros = np.where(h == 0)[0]
        n_o = len(ones)
        n_z = len(zeros)

        F = np.zeros((n_o, 2*n-1))

            bi = rng.randint(n_z)

            else:

        else:

        else:

                b = zero_idx[rng.randint(len(zero_idx))]

        while (time.time() - start) < max_time:

            zeros = np.where(h == 0)[0]

                h[zeros[pos[1]]] = 1.0

def normalize_binary(h, n):
    h = np.round(h).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
