```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def project_onto_feasible(h, target_sum):
    """Project h onto {x: 0<=x[i]<=1, sum(x)=target_sum}"""
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        if proj.sum() > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix tiny numerical errors
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-12:
        free = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def compute_subgradient(h, n):
    """Compute subgradient of max_k overlap_k at h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = np.argmax(corr)
    lag = (n - 1) - k_max
    
    grad = np.zeros(n)
    if lag >= 0:
        # Valid i: 0 <= i <= n-1-lag
        # For j as first index (j=i): valid if 0<=j<=n-1-lag, contrib: (1-h[j+lag])*dx
        # For j as second index (j=i+lag): valid if lag<=j<=n-1, contrib: -h[j-lag]*dx
        max_i = n - 1 - lag
        # j ranges 0..max_i for first index
        grad[:max_i + 1] += (1.0 - h[lag:max_i + 1 + lag]) * dx
        # j ranges lag..n-1 for second index  
        grad[lag:] -= h[:max_i + 1] * dx
    else:
        lag2 = -lag
        # Valid i: lag2 <= i <= n-1
        # For j as first index (j=i): valid if lag2<=j<=n-1, contrib: (1-h[j-lag2])*dx
        # For j as second index (j=i-lag2): valid if 0<=j<=n-1-lag2, contrib: -h[j+lag2]*dx
        max_i = n - 1 - lag2
        grad[lag2:] += (1.0 - h[:max_i + 1]) * dx
        grad[:max_i + 1] -= h[lag2:] * dx
    
    return grad

def upsample(h, n_old, n_new):
    """Upsample h from n_old to n_new points, preserving integral"""
    # Use linear interpolation
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h)
    # Fix integral
    target_sum = n_new / 2
    h_new = project_onto_feasible(h_new, target_sum)
    return h_new

def local_optimize(h, n, rng, time_budget):
    """Projected subgradient optimization"""
    start = time.time()
    target_sum = n / 2
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    # Adaptive step size
    alpha = 0.5
    min_alpha = 0.01
    momentum = 0.9
    velocity = np.zeros(n)
    
    while time.time() - start < time_budget * 0.95:
        # Compute subgradient
        grad = compute_subgradient(h, n)
        
        # Momentum step
        velocity = momentum * velocity - alpha * grad
        h_new = h + velocity
        
        # Project
        h_new = project_onto_feasible(h_new, target_sum)
        
        # Evaluate
        c5_new = compute_c5(h_new, n)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            # Maybe increase alpha slightly
            alpha = min(alpha * 1.05, 1.0)
        else:
            h = best_h.copy()
            # Decrease alpha
            alpha = max(alpha * 0.8, min_alpha)
            # Reduce momentum
            momentum = max(momentum * 0.99, 0.5)
    
    return best_h, best_c5
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_onto_feasible(h, target_sum):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    proj = np.clip(h - t, 0.0, 1.0)
    diff = target_sum - proj.sum()
    if abs(diff) > 1e-14:
        free = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        if free.sum() > 0:
            proj[free] += diff / free.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def compute_subgradient(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_max = int(np.argmax(corr))
    lag = (n - 1) - k_max
    grad = np.zeros(n)
    if lag >= 0:
        m = n - 1 - lag
        grad[:m + 1] += (1.0 - h[lag:lag + m + 1]) * dx
        grad[lag:] -= h[:m + 1] * dx
    else:
        lag2 = -lag
        m = n - 1 - lag2
        grad[lag2:] += (1.0 - h[:m + 1]) * dx
        grad[:m + 1] -= h[lag2:lag2 + m + 1] * dx
    return grad

def upsample(h, n_old, n_new):
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h)
    return project_onto_feasible(h_new, n_new / 2.0)

def to_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    half = n // 2
    num_ones = int(h_bin.sum())
    if num_ones > half:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for idx in order[:num_ones - half]:
            h_bin[ones_idx[idx]] = 0.0
    elif num_ones < half:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for idx in order[:half - num_ones]:
            h_bin[zeros_idx[idx]] = 1.0
    return h_bin

def binary_local_search(h, n, rng, time_budget):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    c5 = best_c5
    no_improve = 0
    max_no_improve = 8000
    temp = 0.02
    temp_min = 0.001
    
    while time.time() - start < time_budget * 0.95 and no_improve < max_no_improve:
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h, n)
        delta = new_c5 - c5
        if delta < -1e-15 or (temp > temp_min and rng.random() < np.exp(-delta / temp)):
            c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_h = h.copy()
                best_c5 = new_c5
                no_improve = 0
            else:
                no_improve += 1
            temp = max(temp * 0.9997, temp_min)
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
            if no_improve % 500 == 0:
                temp = max(temp * 0.95, temp_min)
    
    return best_h, best_c5

def subgradient_optimize(h, n, rng, time_budget):
    start = time.time()
    target_sum = n / 2.0
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    alpha = 0.15
    min_alpha = 0.002
    momentum = 0.5
    velocity = np.zeros(n)
    patience = 0
    max_patience = 500
    
    while time.time() - start < time_budget * 0.9 and patience < max_patience:
        grad = compute_subgradient(h, n)
        velocity = momentum * velocity - alpha * grad
        h_new = project_onto_feasible(h + velocity, target_sum)
        c5_new = compute_c5(h_new, n)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            patience = 0
            alpha = min(alpha * 1.02, 0.6)
        else:
            patience += 1
            alpha = max(alpha * 0.93, min_alpha)
            momentum = max(momentum * 0.998, 0.2)
            h = best_h.copy()
    
    return best_h, best_c5
```
