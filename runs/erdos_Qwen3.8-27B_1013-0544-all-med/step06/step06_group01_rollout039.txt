```python
import time

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _project(h, n):
    """Project onto {0 ≤ h ≤ 1, sum(h) = n/2}"""
    target = n / 2.0
    # Binary search for threshold
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_try = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_try)
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(h - tau, 0.0, 1.0)

def _grad_active(h, active_m, n, dx):
    """Compute gradient of sum of active C[m] with respect to h."""
    grad = np.zeros(n)
    for m in active_m:
        d = m - n + 1
        if d >= 0:
            if d < n:
                grad[:n-d] += (1.0 - h[d:])
                grad[d:] += -h[:n-d]
        else:
            d2 = -d
            if d2 < n:
                grad[d2:] += (1.0 - h[:n-d2])
                grad[:n-d2] += -h[d2:]
    return dx * grad

def _subgrad_optimize(h, n, max_iter=5000, tol=0.005):
    """Subgradient method with adaptive step size."""
    dx = 2.0 / n
    best_c5 = _c5_max(h)
    best_h = h.copy()
    lr = 0.1
    momentum = 0.9
    v = np.zeros(n)
    
    for it in range(max_iter):
        C = _c5_array(h)
        C_max = np.max(C)
        
        if C_max < best_c5 - 1e-10:
            best_c5 = C_max
            best_h = h.copy()
            lr = min(lr * 1.05, 1.0)
        else:
            lr = max(lr * 0.95, 0.001)
        
        # Find active lags
        active = np.where(C > C_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(C)])
        
        # Compute gradient
        G = _grad_active(h, active, n, dx)
        
        # Momentum update
        v = momentum * v - lr * G
        h = h + v
        h = _project(h, n)
    
    return best_h, best_c5
```

**Alternative: Frank-Wolfe (conditional gradient) method**

For the problem min C(h) s.t. h ∈ S where S = {0 ≤ h ≤ 1, sum(h) = n/2}:
- The feasible set S is the intersection of the hypercube [0,1]^n and the hyperplane sum = n/2
- The Linear Minimization Oracle (LMO) for S: given direction g, find h_S = argmin_{h ∈ S} g·h

The LMO for S: 
- First, ignore the sum constraint: h*_i = 0 if g_i > 0, 1 if g_i < 0
- Then adjust to satisfy the sum constraint: this is the water-filling problem again

Actually, the LMO for the product of a box and a simplex-like constraint is:
h_S = clip(y, 0, 1) where y is chosen so that sum = n/2.
Specifically: h_S = argmin_{h ∈ S} g·h

This is equivalent to: find τ such that sum(clip(-g - τ, 0, 1)) = n/2, then h_S = clip(-g - τ, 0, 1).

Wait, that's the same as the projection! The LMO for the constraint set is the same as the projection when the "unconstrained" optimum is the current point. No wait, the LMO is different.

For Frank-Wolfe:
- At point h, compute subgradient g of C at h
- Solve: h_S = argmin_{z ∈ S} g·z
- Update: h ← (1-γ)h + γ*h_S where γ is the step size

The LMO: h_S = argmin_{z ∈ {0≤z≤1, sum(z)=n/2}} g·z
This is: sort g, assign z_i = 1 to the smallest g_i values, z_i = 0 to the largest, and use a fractional value for one element to exactly hit the sum constraint.

This is the same as the projection of -g! So the LMO and the projection are the same here (which makes sense because the constraint set is an intersection of a box and a hyperplane, and the LMO for such sets coincides with the projection in this special case).

The optimal solution is: sort g_1 ≤ g_2 ≤ ... ≤ g_n.

```python
def _lmo(g, n):
    """Linear minimization oracle for {0 ≤ z ≤ 1, sum(z) = n/2}"""
    target = n / 2.0
    order = np.argsort(g)
    z = np.zeros(n)
    # Assign 1 to smallest g values
    num_ones = int(target)  # 47 for n=95
    z[order[:num_ones]] = 1.0
    remainder = target - num_ones  # 0.5
    if remainder > 0:
        z[order[num_ones]] = remainder
    return z
```

**Better local search: Gradient-based coordinate descent**

For each coordinate i in random order:
1. Compute the current C5
2. Try moving h[i] slightly up and slightly down
3. Keep the move that reduces C5

Actually, updating C5 when h[i] changes by δ:

ΔC = dx * [(1-h[i+d_m])*I[valid] - h[i-d_m])*I[valid]] for each m

Then C_new[m] = C_old[m] + δ * ΔC[m]

And I just need to check if max(C_new) < max(C_old).

This makes each coordinate update O(n), and a full pass O(n²). For n=95, that's ~9000 per pass, which is very fast.

```python
def _coord_optimize_1d(h, i, n, C_old, dx):
    """Find optimal h[i] given others fixed."""
    # C[m] = a_m * t + b_m
    # a_m = dx * ((1-h[i+d_m])*I[i+d_m<n] - h[i-d_m]*I[i-d_m>=0])
    # b_m = C_old[m] - a_m * h[i]  (since C_old was computed with old h[i])
    
    n_c = 2*n - 1
    d_all = np.arange(n_c) - n + 1  # shifts for each m
    
    # Compute a_m for each m
    i_d = i + d_all  # i + d_m
    i_minus_d = i - d_all  # i - d_m
    
    I1 = (i_d >= 0) & (i_d < n)
    I2 = (i_minus_d >= 0) & (i_minus_d < n)
    
    a = np.zeros(n_c)
    a[I1] = (1.0 - h[i_d[I1]])
    a[I2] -= h[i_minus_d[I2]]
    a *= dx
    
    # b_m = C_old[m] - a_m * h[i]
    b = C_old - a * h[i]
    
    # Minimize max_m (a_m * t + b_m) over t in [0, 1]
    # Try a grid
    ts = np.linspace(0, 1, 20)
    best_t = 0
    best_val = np.inf
    for t in ts:
        vals = a * t + b
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_t = t
    return best_t, best_val
```

```python
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _project(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _smooth_obj(h, alpha, beta):
    n = len(h)
    C = _c5_array(h)
    m = np.max(C)
    f = m / alpha + np.log(np.sum(np.exp(alpha * C - alpha * m))) / alpha
    f += beta * (np.sum(h) - 0.5 * n) ** 2
    return f

def _smooth_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    m = np.max(C)
    es = np.exp(alpha * C - alpha * m)
    se = np.sum(es)
    w = es / se
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a_idx = i - j_arr + n - 1
        b_idx = i + j_arr - n + 1
        va = (a_idx >= 0) & (a_idx < n)
        vb = (b_idx >= 0) & (b_idx < n)
        ta = np.where(va, 1.0 - h[np.clip(a_idx, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b_idx, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return grad

def _lbgfgs_optimize(h, alpha, beta, maxiter=500):
    n = len(h)
    res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, beta),
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    if res.x is None:
        return h
    return _project(res.x, n)

    best_c5 = np.max(C)

            old_val = h[i]

            I2 = (i_md >= 0) & (i_md < n)
```

```python
    target = n / 2.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    C = _c5_array(h)
    n = len(h)
    m = np.max(C)
    f = (m + np.log(np.sum(np.exp(alpha * (C - m)))) / alpha)
    f += beta * (np.sum(h) - 0.5 * n) ** 2
    return f

    dx = 2.0 / n
    C = _c5_array(h)
```

```python
    for i in range(n):
        a_idx = i - j_arr + n - 1
        b_idx = i + j_arr - n + 1
        va = (a_idx >= 0) & (a_idx < n)
        vb = (b_idx >= 0) & (b_idx < n)
        ta = np.where(va, 1.0 - h[np.clip(a_idx, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b_idx, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return grad

def _lbgfgs(h, alpha, beta, maxiter=400):
    n = len(h)
    res = minimize(_smooth_obj, h, jac=_smooth_grad, args=(alpha, beta),
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    if res.x is None:
        return h
    return _project(res.x, n)

def _coord_sweep(h, n, max_sweeps=500, grid=25):
    dx = 2.0 / n
    C = _c5_array(h)
    best_c5 = np.max(C)
    n_c = 2 * n - 1
    d_all = np.arange(n_c) - (n - 1)
    ts = np.linspace(0.0, 1.0, grid)
    for _ in range(max_sweeps):
        improved = False
        for i in range(n):
            ov = h[i]
            i_d = i + d_all
            i_md = i - d_all
            I1 = (i_d >= 0) & (i_d < n)
            I2 = (i_md >= 0) & (i_md < n)
            a = np.zeros(n_c)
            a[I1] = 1.0 - h[i_d[I1]]
            a[I2] -= h[i_md[I2]]
            a *= dx
            b = C - a * ov
            best_t = ov
            best_v = np.max(a * ov + b)
            for t in ts:
                if abs(t - ov) < 1e-9:
                    continue
                v = np.max(a * t + b)
                if v < best_v - 1e-14:
                    best_v = v
                    best_t = t
            if best_t != ov:
                h[i] = best_t
                C = a * best_t + b
                c5 = np.max(C)
                if c5 < best_c5:
                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return h, best_c5

def _subgrad(h, n, max_iter=3000, tol=0.003):
    dx = 2.0 / n
    best_c5 = _c5_max(h)
    best_h = h.copy()
    lr = 0.5
    mom = 0.7
    v = np.zeros(n)
    n_c = 2 * n - 1
    d_all = np.arange(n_c) - (n - 1)
    idx = np.arange(n)
    for _ in range(max_iter):
        C = _c5_array(h)
        C_max = np.max(C)
        if C_max < best_c5 - 1e-12:
            best_c5 = C_max
            best_h = h.copy()
            lr = min(lr * 1.01, 2.0)
        else:
            lr = max(lr * 0.98, 0.005)
        active = np.where(C > C_max - tol)[0]
        if len(active) == 0:
            active = np.array([np.argmax(C)])
        G = np.zeros(n)
        for m in active:
            d = d_all[m]
            i_d = idx + d
            i_md = idx - d
            I1 = (i_d >= 0) & (i_d < n)
            I2 = (i_md >= 0) & (i_md < n)
            G[I1] += 1.0 - h[i_d[I1]]
            G[I2] -= h[i_md[I2]]
        G *= dx / len(active)
        v = mom * v - lr * G
        h = _project(h + v, n)
    return best_h, best_c5

def _try_n(h_best, n_best, n_try, t0, budget_s):
    if time.time() - t0 > budget_s * 0.5:
        return None, None
    x_old = np.linspace(0, 2, n_best, endpoint=False)
    x_new = np.linspace(0, 2, n_try, endpoint=False)
    ht = _project(np.interp(x_new, x_old, h_best), n_try)
    ht, ct = _coord_sweep(ht, n_try, 300, 20)
    if time.time() - t0 < budget_s * 0.6:
        hn = _lbgfgs(ht, 200, 10.0, 300)
        cn = _c5_max(hn)
        if cn < ct:
            ht, ct = hn, cn
        ht, ct = _coord_sweep(ht, n_try, 200, 25)
    return ht, ct

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    hb = _project(initial_h_values.astype(float).copy(), n0)
    cb = _c5_max(hb)
    nb = n0

    # Phase 1: L-BFGS-B progressive alpha
    for alpha in [20, 50, 100, 300, 500, 1000]:
        if _el() > budget_s * 0.15:
            break
        beta = max(1.0, 0.05 * alpha)
        hn = _lbgfgs(hb, alpha, beta, 400)
        cn = _c5_max(hn)
        if cn < cb - 1e-10:
            cb, hb = cn, hn

    # Phase 2: Heavy coordinate sweep
    if _el() < budget_s * 0.40:
        hb, cb = _coord_sweep(hb, nb, 800, 25)

    # Phase 3: High-alpha L-BFGS-B polish
    for alpha in [500, 1000, 2000, 5000]:
        if _el() > budget_s * 0.50:
            break
        beta = max(5.0, 0.05 * alpha)
        hn = _lbgfgs(hb, alpha, beta, 500)
        cn = _c5_max(hn)
        if cn < cb - 1e-10:
            cb, hb = cn, hn
        hb, cb = _coord_sweep(hb, nb, 200, 20)

    # Phase 4: Subgradient method
    if _el() < budget_s * 0.65:
        hn, cn = _subgrad(hb, nb, 5000, 0.002)
        if cn < cb:
            cb, hb = cn, hn
        hb, cb = _coord_sweep(hb, nb, 400, 25)

    # Phase 5: Try nearby n values
    for n_try in [nb - 2, nb - 1, nb + 1, nb + 2, nb + 5]:
        if _el() > budget_s * 0.80:
            break
        if n_try < 50 or n_try > 200 or n_try == nb:
            continue
        ht, ct = _try_n(hb, nb, n_try, t0, budget_s)
        if ct is not None and ct < cb - 1e-10:
            cb, hb, nb = ct, ht, n_try

    # Phase 6: Random restarts with coordinate sweep
    for _ in range(80):
        if _el() > budget_s * 0.94:
            break
        scale = 0.015 + 0.05 * rng.rand()
        ht = _project(hb + rng.randn(nb) * scale, nb)
        ht, ct = _coord_sweep(ht, nb, 200, 20)
        if _el() < budget_s * 0.95:
            hn2 = _lbgfgs(ht, 300, 15.0, 250)
            cn2 = _c5_max(hn2)
            if cn2 < ct:
                ht, ct = hn2, cn2
            ht, ct = _coord_sweep(ht, nb, 100, 20)
        if ct < cb - 1e-10:
            cb, hb = ct, ht

    # Phase 7: Final coordinate sweep with fine grid
    if _el() < budget_s * 0.99:
        hb, cb = _coord_sweep(hb, nb, 500, 35)

    return (hb, cb, nb)
```
