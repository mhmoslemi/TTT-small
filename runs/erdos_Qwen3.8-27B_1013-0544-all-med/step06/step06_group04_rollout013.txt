    ts = time.time()

            return 1e10

    res = minimize(obj, h, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12, 'adaptive': True})
    return res.x

from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_sum(h, n):
    target = n * 0.5
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def resample_h(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = np.arange(n_old) / float(n_old)
    x_new = np.arange(n_new) / float(n_new)
    h_new = np.interp(x_new, x_old, h)
    return fix_sum(h_new, n_new)

def opt_nelder(h, n, maxiter, tlimit):
    dx = 2.0 / n
    ts = time.time()
    def obj(x):
        if time.time() - ts > tlimit:
            return 1e10
        h_l = np.clip(x, 0.0, 1.0)
        pen = 1000.0 * (np.sum(h_l) - n * 0.5) ** 2
        return evaluate_c5(h_l, dx) + pen
    res = minimize(obj, h, method='Nelder-Mead',
                   options={'maxiter': maxiter, 'xatol': 1e-10, 'fatol': 1e-12, 'adaptive': True})
    return res.x, res.fun

def coord_descent(h, n, sweeps, tlimit):

            best_c5 = c5

                v = np.clip(v, 0.0, 1.0)

                h = fix_sum(h, n)

                if c < best_c5:
                    best_c5 = c
                    best_v = v
            h[i] = best_v
            h = fix_sum(h, n)
            c5 = best_c5
    return h, c5
