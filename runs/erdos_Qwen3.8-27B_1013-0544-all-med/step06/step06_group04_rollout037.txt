```python
def compute_c5(h, dx):

    corr = np.correlate(h, comp, mode="full")
    return float(np.max(corr) * dx)

def delta_c5_swap(h, i, j, dx):
    """Compute change in C5 when swapping h[i] 1->0 and h[j] 0->1."""
    # Just recompute (fast for small n)
    h[i] = 0.0
    h[j] = 1.0
    new_val = compute_c5(h, dx)
    h[i] = 1.0
    h[j] = 0.0
    return new_val

def local_search_2swap(h, dx, max_rounds=2000, delta=0.0):
    """Greedily improve by 2-swaps."""
    h = h.astype(float).copy()
    n = len(h)
    current = compute_c5(h, dx)
    for rnd in range(max_rounds):
        best_imp = delta
        best_pair = None
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                old_hi, old_hj = h[i], h[j]
                h[i], h[j] = 0.0, 1.0
                new_val = compute_c5(h, dx)
                h[i], h[j] = old_hi, old_hj
                imp = current - new_val
                if imp > best_imp:
                    best_imp = imp
                    best_pair = (i, j, new_val)
        if best_pair is None:
            break
        i, j, new_val = best_pair
        h[i], h[j] = 0.0, 1.0
        current = new_val
    return h, current

def simulated_annealing(h, dx, t_start=0.1, t_stop=0.001, cooling=0.999, max_rounds=5000):
    """Simulated annealing on binary 2-swaps."""
    h = h.astype(float).copy()
    n = len(h)
    current = compute_c5(h, dx)
    best_h = h.copy()
    best_val = current

    while t > t_stop and max_rounds > 0:

        zeros = np.where(h < 0.5)[0]

        j = rng.choice(zeros)
        if i == j:

        delta = new_val - current

            current = new_val
            if current < best_val:
                best_val = current
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
        t *= cooling
        max_rounds -= 1
    
    return best_h, best_val
```

```python
def ls_2swap_fast(h, dx, max_rounds=10000):
    h = h.astype(float).copy()
    n = len(h)
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full").astype(float)
    cur = float(np.max(corr) * dx)
    
    for _ in range(max_rounds):
        best_imp = 1e-12
        best = None
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                # Compute delta corr
                delta = np.zeros(2 * n - 1)
                for k in range(2 * n - 1):
                    # k corresponds to lag m = k - (n-1)
                    # corr[k] = sum_{valid n} h[n+m] * (1-h[n])
                    # where m = k - (n-1)
                    m = k - (n - 1)
                    # h[i]: 1 -> 0
                    #   if i = n+m (i.e., n = i-m = i - k + n - 1): term was 1*(1-h[i-m]), now 0
                    #     valid if 0 <= i-m < n, i.e., 0 <= i - m
                    #   if i = n (i.e., the "n" index = i): term was h[i+m]*(1-1)=0, now h[i+m]*(1-0)=h[i+m]
                    #     valid if 0 <= i+m < n
                    # h[j]: 0 -> 1
                    #   if j = n+m: term was 0, now 1*(1-h[j-m]), valid if 0 <= j-m < n
                    #   if j = n: term was h[j+m]*(1-0)=h[j+m], now 0, valid if 0 <= j+m < n
                    
                    dk = 0.0
                    # i as "h[n+m]" position: n = i - m
                    ni = i - m
                    if 0 <= ni < n:
                        dk += -(1.0 - h[ni])  # was 1*(1-h[ni]), now 0
                    # i as "n" position: n = i, so h[i+m]
                    ni2 = i + m
                    if 0 <= ni2 < n:
                        dk += h[ni2]  # was h[i+m]*(1-1)=0, now h[i+m]*(1-0)
                    # j as "h[n+m]" position: n = j - m
                    nj = j - m
                    if 0 <= nj < n:
                        dk += (1.0 - h[nj])  # was 0, now 1*(1-h[j-m])
                    # j as "n" position: n = j, so h[j+m]
                    nj2 = j + m
                    if 0 <= nj2 < n:
                        dk += -h[nj2]  # was h[j+m]*(1-0), now 0

                new_max = float(np.max(new_corr) * dx)

                    best = (i, j, new_max, delta)

        i, j, new_max, delta = best
        h[i] = 0.0
        h[j] = 1.0
        corr = corr + delta
        cur = new_max
    
    return h, cur
```

```python
def compute_G(h, n):
    """Compute gradient matrix G where G[p, k] = d corr[k] / d h[p]."""
    n_corr = 2 * n - 1
    lags = np.arange(n_corr) - (n - 1)
    G = np.zeros((n, n_corr))
    for p in range(n):
        # G[p, k] = (1-h[p-m]) * I[0<=p-m<n] - h[p+m] * I[0<=p+m<n]
        # where m = lags[k]
        pm = p - lags  # p - m, shape (n_corr,)
        pp = p + lags  # p + m, shape (n_corr,)
        mask1 = (pm >= 0) & (pm < n)
        mask2 = (pp >= 0) & (pp < n)
        vals1 = np.zeros(n_corr)
        vals2 = np.zeros(n_corr)
        vals1[mask1] = 1.0 - h[pm[mask1]]
        vals2[mask2] = h[pp[mask2]]
        G[p, :] = vals1 - vals2
    return G

def ls_2swap_matrix(h, dx, max_rounds=10000):
    h = h.astype(float).copy()
    n = len(h)
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full").astype(float)
    cur = float(np.max(corr) * dx)
    
    G = compute_G(h, n)
    
    for _ in range(max_rounds):
        best_imp = 1e-12
        best_i = -1
        best_j = -1
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        
        for i in ones:
            # For all j in zeros, compute delta = G[j,:] - G[i,:]
            Deltas = G[zeros, :] - G[i, :]
            NewMaxs = (corr[None, :] + Deltas).max(axis=1)
            imps = cur - NewMaxs * dx
            
            best_local = np.argmax(imps)
            if imps[best_local] > best_imp:
                best_imp = imps[best_local]
                best_i = i
                best_j = zeros[best_local]
        
        if best_i < 0:
            break
        
        i, j = best_i, best_j
        # Update
        h[i] = 0.0
        h[j] = 1.0
        corr += G[j, :] - G[i, :]
        cur += best_imp
        # Update G (since h changed)
        G = compute_G(h, n)
    
    return h, cur
```

Wait, but G depends on h, and h changes each round. So I need to recompute G each round. That's O(n²) per round (n × 2n matrix, each entry computed in O(1) but with numpy overhead). For n=70: 70 * 139 = 9730 entries. The computation involves array operations of size 139, so ~70 * 10 * 1μs = 700μs per round. Over 10000 rounds: 7 seconds. Acceptable.

Actually wait, I can be smarter. When h[i] changes from 1 to 0, G only changes for rows affected by h[i] and h[j]. Specifically:
- G[p, k] depends on h[p±m] where m = lags[k].
- If h[i] changes, then G[p, k] changes when p-m = i (i.e., the "first index" is i) or p+m = i (the "second index" is i).
- p-m = i → k = p - i - (n-1) (one specific k for each p)
- p+m = i → k = i - p - (n-1) (one specific k for each p)

So changing h[i] affects at most 2 entries in each row of G. That's 2n entries total. I could update G incrementally, but it's probably not worth the complexity for n=70.

Let me just recompute G each round. Total cost: recompute G (O(n²) with numpy, ~1ms) + search (O(n * 2n) with numpy, ~1ms) = ~2ms per round. 10000 rounds = 20 seconds. 

Actually, let me reconsider. The `compute_G` function as written has a Python loop over n=70, and inside it does numpy ops on arrays of size 139. The Python loop overhead is 70 * ~1μs = 70μs, and the numpy ops are 70 * ~5μs = 350μs. Total: ~400μs per round. Very fast.

OK so my revised estimate: ~2ms per round including G recomputation and search. 10000 rounds = 20 seconds. 

But wait, I realize there's an issue. When I update `corr += G[j,:] - G[i,:]`, this is correct for the first swap. But after the swap, G changes (because h changed), so the next round's G is different. That's fine because I recompute G at the start of each round.

But there's a subtlety: `corr` should always be consistent with the current `h`. Let me verify:
- Initially: corr = correlate(h, 1-h, full). ✓
- After swap: corr_new = corr + G[j,:] - G[i,:] where G was computed with the OLD h.
  
  Is this correct? G[p, k] = ∂corr[k]/∂h[p] evaluated at the current h.
  After changing h[i] from 1 to 0 and h[j] from 0 to 1:
  corr_new[k] = corr[k] + (∂corr[k]/∂h[i]) * (0-1) + (∂corr[k]/∂h[j]) * (1-0)
               = corr[k] - G[i,k] + G[j,k]
               = corr[k] + G[j,k] - G[i,k]
  
  But wait! This is only an approximation if corr is a nonlinear function of h. But corr IS linear in each h[p] separately (it's a sum of products h[a]*(1-h[b]), which is linear in h[a] for fixed h[b]). So the partial derivative gives the EXACT change when only one variable is changed.
  
  But we're changing TWO variables simultaneously. The cross term is: the change in h[i] affects the terms where h[j] appears, and vice versa.
  
  Specifically, corr[k] = sum h[n+m]*(1-h[n]). If both h[i] and h[j] are in the same product (e.g., h[i] is the "first" factor and h[j] is the "second" factor for the same n), then there's a cross term.
  
  h[a]*(1-h[b]) where a and b are both changed:
  Before: h[a]*(1-h[b]) = 1*1 = 1 (if h[a]=1, h[b]=0)
  After: h[a]*(1-h[b]) = 0*0 = 0
  
  Change: -1.
  
  Using partial derivatives:
  ∂/∂h[a] of h[a]*(1-h[b]) = (1-h[b]) = 1 (evaluated at old values)
  ∂/∂h[b] of h[a]*(1-h[b]) = -h[a] = -1 (evaluated at old values)
  Linear approximation: 1*(-1) + (-1)*(1) = -2. But actual change is -1!
  
  So the linear approximation is WRONG when both variables appear in the same product term!
  
  The exact change for a product h[a]*(1-h[b]) when h[a]: δa and h[b]: δb:
  (h[a]+δa)*(1-h[b]-δb) - h[a]*(1-h[b])
  = h[a] - h[a]*δb + δa - δa*δb - h[a] + h[a]
  
  Wait: = h[a]*(1-h[b]) + δa*(1-h[b]) - h[a]*δb - δa*δb - h[a]*(1-h[b])
  = δa*(1-h[b]) - h[a]*δb - δa*δb
  
  The linear approximation (ignoring δa*δb) gives: δa*(1-h[b]) - h[a]*δb
  
  The exact formula has an additional term -δa*δb.
  
  In our case, δa = -1 (h[i] from 1 to 0) and δb = +1 (h[j] from 0 to 1).
  Cross term: -(-1)*(1) = +1.
  
  So the exact change = linear approximation + 1 (for terms where i appears as "first" and j as "second" in the same product, or vice versa).
  
  Specifically, the cross term -δa*δb = -(-1)(1) = 1 appears for each (n, k) where {n, n+m} = {i, j}, i.e., where i and j are paired in the same product term.
  
  This happens when:
  - i = n and j = n+m (i is first, j is second): m = j-i, and we need 0≤i<n, 0≤j<n
  - j = n and i = n+m (j is first, i is second): m = i-j, and we need 0≤j<n, 0≤i<n
  
  So there are at most 2 lags where the cross term appears (m = j-i and m = i-j, i.e., the positive and negative of the same offset).
  
  For each such lag k (where m = j-i or m = i-j), the correction is +1.
  
  So: exact corr_new[k] = corr[k] + G[j,k] - G[i,k] + cross_term[k]
  where cross_term[k] = 1 if the lag m_k = k-(n-1) equals j-i or i-j (and the indices are valid), else 0.
  
  More precisely:
  - If m = j-i (so i is the "first" index n=i, and j = n+m = i+(j-i) = j): cross term = -δa*δb = 1
    Valid if 0 ≤ i < n and 0 ≤ j < n (which is always true since i,j are positions in h).
    Also need i+m = j < n, which is true since j < n.
  - If m = i-j (so j is the "first" index n=j, and i = n+m = j+(i-j) = i): cross term = -δa*δb = 1
    Valid if 0 ≤ j < n and 0 ≤ i < n.
    
  So: cross_term[k] = 1 if lags[k] == j-i, plus 1 if lags[k] == i-j.
  Since lags contains both positive and negative values, and j-i and i-j have opposite signs, these are two different k values (unless i=j, which we exclude).
  
  So the correction is: add 1 to corr at the two positions k = j-i+(n-1) and k = i-j+(n-1).
