            mask = x > 1e-14
            if np.any(mask):
                x[mask] += diff * x[mask] / float(x[mask].sum())
            else:
                break
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - float(x.sum())
    if abs(diff) > 1e-9:
        k = int(np.floor(target))
        x[:] = 0.0
        if k > 0:
            x[:k] = 1.0
        rem = target - float(k)
        if rem > 0.0 and k < n:
            x[k] = min(rem, 1.0)
    elif diff > 1e-13:
        order = np.argsort(1.0 - x)[::-1]
        for idx in order:
            if diff <= 1e-15:
                break
            room = 1.0 - x[idx]
            if room > 0:
                add = min(diff, room)
                x[idx] += add
                diff -= add
    elif diff < -1e-13:
        order = np.argsort(x)[::-1]
        for idx in order:
            if diff >= -1e-15:
                break
            add = max(diff, -x[idx])
            x[idx] += add
            diff -= add
    return x

def c_values(h):
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(c_values(h)))

def subgradient(h, active):
    n = h.size
    dx = 2.0 / float(n)
    grad = np.zeros(n)
    j = np.arange(n)
    for k in active:
        k = int(k)
        m1 = (j - k >= 0) & (j - k < n)
        m2 = (j + k >= 0) & (j + k < n)
        if np.any(m1):
            grad[m1] += (1.0 - h[j[m1] - k])
        if np.any(m2):
            grad[m2] -= h[j[m2] + k]
    grad *= dx / float(len(active))
    return grad

def smooth_obj(h, tau, topk):
    c = c_values(h)
    cm = float(np.max(c))
    m = c.size
    z = (c - cm) / tau
    z = z - np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        w = np.zeros(m)
        w[int(np.argmax(c))] = 1.0
        return cm, c, w
    w = e / s
    if topk is not None and 0 < int(topk) < m:
        idx = np.argsort(w)[::-1][:int(topk)]
        w2 = np.zeros(m)
        w2[idx] = w[idx]
        ss = float(w2.sum())
        if ss > 0:
            w = w2 / ss
    return cm + tau * np.log(s), c, w

def smooth_grad(h, tau, topk):
    val, c, w = smooth_obj(h, tau, topk)
    n = h.size
    j = np.arange(n)
    grad = np.zeros(n)
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-14:
            continue
        m1 = (j - k >= 0) & (j - k < n)
        m2 = (j + k >= 0) & (j + k < n)
        if np.any(m1):
            grad[m1] += wk * (1.0 - h[j[m1] - k])
        if np.any(m2):
            grad[m2] -= wk * h[j[m2] + k]
    grad *= (2.0 / float(n))
    return grad

def slsqp_smooth(h0, tau, topk, maxiter):
    n = h0.size
    bounds = [(0.0, 1.0)] * n
    def fun(x):
        v, _, _ = smooth_obj(x, tau, topk)
        return v
    def jac(x):
        return smooth_grad(x, tau, topk)
    def cfun(x):
        return x.sum() - 0.5 * n
    def cjac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = so.minimize(fun, h0, method='SLSQP', jac=jac, bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return repair(res.x)
    except Exception:
        return repair(h0)

def bb_subgrad(h0, time_limit, seed, tol_active=1e-5, max_active=6):
    rng = np.random.RandomState(seed)
    h = repair(h0.copy())
    best_h = h.copy()
    best_c = exact_c(h)
    deadline = time.time() + max(0.0, time_limit)
    prev_h = None
    prev_g = None
    alpha = 0.02
    no_imp = 0
    stall = 0

    while time.time() < deadline:
        c = c_values(h)
        C = float(np.max(c))
        if C < best_c - 1e-14:
            best_c = C
            best_h = h.copy()
            no_imp = 0
            stall = 0
        else:
            no_imp += 1

        tol = max(1e-8, float(tol_active) * C)
        active = np.where(c >= C - tol)[0]
        if active.size > max_active:
            active = np.argsort(c)[::-1][:max_active]

        g = subgradient(h, active)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break

        if prev_g is not None and prev_h is not None:
            s_vec = h - prev_h
            y_vec = g - prev_g
            sy = float(np.dot(s_vec, y_vec))
            yy = float(np.dot(y_vec, y_vec))
            ss = float(np.dot(s_vec, s_vec))
            if yy > 1e-14 and ss > 1e-14:
                if sy > 1e-14:
                    alpha_bb = ss / yy
                else:
                    alpha_bb = ss / (yy + 1e-10)
                alpha_bb = np.clip(alpha_bb, 1e-5, 0.5)
                alpha = 0.4 * alpha + 0.6 * alpha_bb

        accepted = False
        for _ in range(25):
            y = repair(h - alpha * g)
            cy = exact_c(y)
            if cy < C - 1e-13:
                h = y
                accepted = True
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break

        if not accepted:
            stall += 1
            if stall > 8:
                improved = False
                for sg in (0.015, 0.008, 0.004, 0.002):
                    y = repair(h + rng.normal(0.0, sg, h.size))
                    cy = exact_c(y)
                    if cy < best_c:
                        best_c = cy
                        best_h = y.copy()
                        h = y
                        no_imp = 0
                        stall = 0
                        alpha = 0.02
                        improved = True
                        break
                if not improved and no_imp > 40:
                    break
        else:
            stall = max(0, stall - 1)

        prev_h = h.copy()
        prev_g = g.copy()

        if rng.rand() < 0.03 and no_imp < 15:
            y = repair(h + rng.normal(0.0, 0.003, h.size))
            cy = exact_c(y)
            if cy < best_c:
                best_c = cy
                best_h = y
                h = y
                no_imp = 0

    return best_h, best_c

def coord_descent(h0, time_limit, seed):
    rng = np.random.RandomState(seed)
    h = repair(h0.copy())
    best_h = h.copy()
    best_c = exact_c(h)
    deadline = time.time() + max(0.0, time_limit)
    n = h.size
    no_imp = 0

    while time.time() < deadline:
        if no_imp > 500:
            break
        idx = rng.randint(n)
        found = False
        for delta in (0.008, -0.008, 0.004, -0.004, 0.012, -0.012, 0.002, -0.002):
            y = h.copy()
            y[idx] = np.clip(y[idx] + delta, 0.0, 1.0)
            y = repair(y)
            cy = exact_c(y)
            if cy < best_c - 1e-14:
                best_c = cy
                best_h = y
                h = y
                no_imp = 0
                found = True
                break
        if not found:
            no_imp += 1
            if no_imp % 10 == 0:
                i2 = rng.randint(n)
                if i2 != idx:
                    for di, dj in ((0.008, -0.008), (-0.008, 0.008), (0.004, -0.004)):
                        y = h.copy()
                        y[idx] = np.clip(y[idx] + di, 0.0, 1.0)
                        y[i2] = np.clip(y[i2] + dj, 0.0, 1.0)
                        y = repair(y)
                        cy = exact_c(y)
                        if cy < best_c - 1e-14:
                            best_c = cy
                            best_h = y
                            h = y
                            no_imp = 0
                            break

    return best_h, best_c

def resample_to(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    if n0 == n_new:
        return repair(h0)
    dt0 = 2.0 / float(n0)
    x0 = (np.arange(n0) + 0.5) * dt0
    x1 = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x1, x0, h0)
    return repair(y)

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)

    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size >= 10:
        init = repair(np.asarray(raw, dtype=float))
    else:
        init = repair(np.full(70, 0.5))

    best_x = init.copy()
    best_c = exact_c(init)

    # Phase 1: SLSQP smooth (15% budget)
    p1_end = time.time() + 0.15 * budget_s
    for tau, topk in ((0.04, 5), (0.015, 4), (0.006, 3)):
        if time.time() > p1_end or time_left(deadline) < 80:
            break
        x = slsqp_smooth(best_x, tau, topk, 400)
        c = exact_c(x)
        if c < best_c:
            best_c = c
            best_x = x

    # Phase 2: BB subgradient main (45% budget)
    p2_time = 0.45 * budget_s
    if time_left(deadline) > 30:
        t_avail = min(p2_time, time_left(deadline) - 30)
        x, c = bb_subgrad(best_x, t_avail, seed_val, 1e-5, 6)
        if c < best_c:
            best_c = c
            best_x = x

    # Phase 2b: Restart with perturbation
    if time_left(deadline) > 40:
        rng = np.random.RandomState(seed_val + 777)
        pert = repair(best_x + rng.normal(0.0, 0.012, best_x.size))
        t_avail = min(60.0, time_left(deadline) - 20)
        x, c = bb_subgrad(pert, t_avail, seed_val + 888, 1e-5, 5)
        if c < best_c:
            best_c = c
            best_x = x

    # Phase 3: Tighter SLSQP (15%)
    p3_time = 0.15 * budget_s
    for tau, topk in ((0.003, 3), (0.001, None)):
        if time_left(deadline) < 25:
            break
        x = slsqp_smooth(best_x, tau, topk, 350)
        c = exact_c(x)
        if c < best_c:
            best_c = c
            best_x = x

    # Phase 4: BB subgradient refinement (10%)
    if time_left(deadline) > 25:
        t_avail = min(0.10 * budget_s, time_left(deadline) - 15)
        x, c = bb_subgrad(best_x, t_avail, seed_val + 2000, 5e-6, 4)
        if c < best_c:
            best_c = c
            best_x = x

    # Phase 5: Coordinate descent (10%)
    if time_left(deadline) > 15:
        t_avail = min(0.10 * budget_s, time_left(deadline) - 5)
        x, c = coord_descent(best_x, t_avail, seed_val + 3000)
        if c < best_c:
            best_c = c
            best_x = x

    # Final SLSQP polish
    if time_left(deadline) > 8:
        x = slsqp_smooth(best_x, 0.0005, None, 200)
        c = exact_c(x)
        if c < best_c:
            best_c = c
            best_x = x

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
