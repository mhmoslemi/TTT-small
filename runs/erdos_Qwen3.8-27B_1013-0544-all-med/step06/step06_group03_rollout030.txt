```python
import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def slp_step(h0, n, max_lp_iter=50):
    """One SLP iteration: linearize and solve LP."""
    dx = 2.0 / n
    target = n / 2.0
    
    # Build linearized constraints
    # For each lag m (0 to 2n-2), compute the linearized correlation
    # c[m] ≈ A_m @ h + const_m
    # We want A_m @ h + const_m <= t/dx for all m
    
    # Build matrix A (2n-1 x n) and vector const (2n-1)
    A = np.zeros((2*n-1, n))
    const = np.zeros(2*n-1)
    
    for m in range(2*n-1):
        s = m - (n-1)  # shift
        # valid i: 0 <= i < n and 0 <= i+s < n
        i_lo = max(0, -s)
        i_hi = min(n, n-s)  # exclusive
        for i in range(i_lo, i_hi):
            j = i + s
            # h[i]*(1-h[j]) linearized:
            # h[i]*(1-h0[j]) - h0[i]*h[j] + h0[i]*h0[j]
            A[m, i] += (1.0 - h0[j])
            A[m, j] += (-h0[i])
            const[m] += h0[i] * h0[j]
    
    # LP: minimize t subject to A@h + const <= t/dx, 0<=h<=1, sum(h)=target
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        A @ h_var + const <= t_var / dx
    ]
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iter=10000)
        if h_var.value is not None:
            h_new = np.clip(h_var.value, 0, 1)
            h_new = project_feasible(h_new, n)
            return h_new
    except Exception:
        pass
    
    return h0.copy()
```

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def slp_solve(h0, n):
    """Solve one SLP LP subproblem."""
    dx = 2.0 / n
    target = n / 2.0
    m = 2 * n - 1
    A = np.zeros((m, n))
    const = np.zeros(m)
    
    for idx in range(m):
        s = idx - (n - 1)
        i_lo = max(0, -s)
        i_hi = min(n, n - s)
        for i in range(i_lo, i_hi):
            j = i + s
            A[idx, i] += (1.0 - h0[j])
            A[idx, j] += (-h0[i])
            const[idx] += h0[i] * h0[j]
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == target,
        A @ h_var + const <= t_var / dx
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iter=5000)
        if h_var.value is not None and t_var.value is not None:
            return np.clip(h_var.value, 0, 1), t_var.value
    except Exception:
        pass
    try:
        prob.solve(solver=cp.SCS, max_iters=10000)
        if h_var.value is not None:
            return np.clip(h_var.value, 0, 1), cp.value(t_var)
    except Exception:
        pass
    return h0.copy(), None

def pgd_optimize(h_init, max_iter=20000, lr_init=0.005, t_smooth=0.01):
    n = len(h_init)
    dx = 2.0 / n
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(max_iter):
        conv = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(conv) * dx
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
        
        # Soft-max gradient
        cm = np.max(conv)
        exps = np.exp((conv - cm) / t_smooth)
        weights = exps / np.sum(exps)
        grad = np.zeros(n)
        for idx in range(2 * n - 1):
            w = weights[idx]
            if w < 1e-14:
                continue
            s = idx - (n - 1)
            i_lo = max(0, -s)
            i_hi = min(n, n - s)
            for i in range(i_lo, i_hi):
                j = i + s
                grad[i] += w * (1.0 - h[j])
                grad[j] += w * (-h[i])
        grad *= dx
        
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-15:
            lr *= 0.5
            if lr < 1e-15:
                break
        else:
            h = h_new
            lr = min(lr * 1.1, 0.05)
    
    return best_h, best_c5

def binary_local_search(h_init, n, max_iters=5000, rng=None):
    """2-flip local search on binary h."""
    if rng is None:
        rng = np.random.RandomState(42)
    dx = 2.0 / n
    
    # Binarize: pick the top n/2 values
    h = h_init.copy()
    target = n // 2
    thresh = np.sort(h)[::-1][target - 1] if target > 0 else 0
    h_bin = (h >= thresh).astype(float)
    # Fix if needed
    while h_bin.sum() > target:
        idx = np.where(h_bin == 1)[0]
        h_bin[idx[rng.randint(len(idx))]] = 0
    while h_bin.sum() < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[rng.randint(len(idx))]] = 1
    
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    
    for it in range(max_iters):
        # Try random 2-flips
        best_gain = 1e-15
        best_flip = None
        for _ in range(50):
            zeros = np.where(h_bin == 0)[0]
            ones = np.where(h_bin == 1)[0]
            if len(zeros) == 0 or len(ones) == 0:
                break
            a = zeros[rng.randint(len(zeros))]
            b = ones[rng.randint(len(ones))]
            if a == b:
                continue
            # Compute change in all c[m]
            h_bin[a] = 1.0
            h_bin[b] = 0.0
            c5_trial = compute_c5(h_bin)
            h_bin[a] = 0.0
            h_bin[b] = 1.0
            if best_c5 - c5_trial > best_gain:
                best_gain = best_c5 - c5_trial
                best_flip = (a, b)
        
        if best_flip is not None:
            a, b = best_flip
            h_bin[a] = 1.0
            h_bin[b] = 0.0
            best_c5 = compute_c5(h_bin)
            best_h = h_bin.copy()
        else:
            break
    
    return best_h, best_c5
