```python
def _coord_descent_fast(h, n, n_passes=20, grid_size=50):
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    best_c5 = np.max(C)
    t_grid = np.linspace(0, 1, grid_size)
    k0 = n - 1  # index of zero lag in full correlation
    
    for p in range(n_passes):
        improved = False
        for i in range(n):
            v = h[i]
            # Compute delta vector for position i
            k = np.arange(2*n-1)
            a = i - k + n - 1  # index in h for the "first" term
            b = i + k - n + 1  # index in h for the "second" term
            valid_a = (a >= 0) & (a < n)
            valid_b = (b >= 0) & (b < n)
            delta = np.zeros(2*n-1)
            delta[valid_a] = 1.0 - h[np.clip(a[valid_a], 0, n-1)]
            delta[valid_b] -= h[np.clip(b[valid_b], 0, n-1)]
            # At k=k0, the linear formula overcounts (both a and b equal i)
            # The correct formula at k0 is: C[k0] + (t(1-t) - v(1-v))
            # The linear formula gives: C[k0] + (t-v)*(1-h[i]-h[i]) = C[k0] + (t-v)*(1-2v)
            # Difference: (t(1-t)-v(1-v)) - (t-v)(1-2v) = (t-v)(1-t-v) - (t-v)(1-2v) = (t-v)(-t+v) = -(t-v)²
            # So: C_actual[k0] = C[k0] + (t-v)*(1-2v) - (t-v)²
            #     = C[k0] + (t(1-t) - v(1-v))  ✓
            
            best_t = v
            best_local = best_c5
            for t in t_grid:
                dt = t - v
                if abs(dt) < 1e-8:
                    continue
                # Compute new max
                C_new = C + dt * delta
                C_new[k0] = C[k0] + (t*(1-t) - v*(1-v))
                local_max = np.max(C_new)
                if local_max < best_local - 1e-12:
                    best_local = local_max
                    best_t = t
            
            if best_t != v:
                dt = best_t - v
                C += dt * delta
                C[k0] += (best_t*(1-best_t) - v*(1-v))
                h[i] = best_t
                best_c5 = best_local
                improved = True
        
        if not improved:
            break
    
    return h, best_c5
```

```python
def _best_t_for_i(h, n, i, C, v):
    """Find best t for h[i] given current C array and C[k0]."""
    dx = 2.0 / n
    k0 = n - 1
    nc = 2 * n - 1
    k = np.arange(nc)
    a = i - k + n - 1
    b = i + k - n + 1
    valid_a = (a >= 0) & (a < n)
    valid_b = (b >= 0) & (b < n)
    delta = np.zeros(nc)
    if np.any(valid_a):
        delta[valid_a] = 1.0 - h[a[valid_a]]
    if np.any(valid_b):
        delta[valid_b] -= h[b[valid_b]]
    
    def obj(t):
        dt = t - v
        C_new = C + dt * delta
        C_new[k0] = C[k0] + (t*(1-t) - v*(1-v))
        return np.max(C_new)
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    a_lo, a_hi = 0.0, 1.0
    c = a_hi - gr * (a_hi - a_lo)
    d = a_lo + gr * (a_hi - a_lo)
    fc, fd = obj(c), obj(d)
    for _ in range(50):  # ~50 iterations gives precision ~ 0.618^50 ≈ 1e-11
        if fc < fd:
            a_hi = d
            d = c
            fd = fc
            c = a_hi - gr * (a_hi - a_lo)
            fc = obj(c)
        else:
            a_lo = c
            c = d
            fc = fd
            d = a_lo + gr * (a_hi - a_lo)
            fd = obj(d)
    
    return (a_lo + a_hi) / 2
```

```python
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0/n
    return float(np.max(np.correlate(h, 1.0-h, mode='full') * dx))

def _fix(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(100):
        d = 0.5*n - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            m = h < 1.0
            cnt = m.sum()
            if cnt == 0: break
            h[m] += d/cnt
        else:
            m = h > 0.0
            cnt = m.sum()
            if cnt == 0: break
            h[m] += d/cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _eval_t(C, delta, k0, v, t):
    dt = t - v
    if abs(dt) < 1e-15:
        return float(np.max(C))
    tmp = C + dt * delta
    tmp[k0] = C[k0] + t*(1.0-t) - v*(1.0-v)
    return float(np.max(tmp))

def _golden(C, delta, k0, v, nit=50):
    gr = 0.6180339887
    lo, hi = 0.0, 1.0
    c = hi - gr*(hi-lo)
    d = lo + gr*(hi-lo)
    hc = _eval_t(C, delta, k0, v, c)
    hd = _eval_t(C, delta, k0, v, d)
    for _ in range(nit):
        if hc <= hd:
            hi = d; d = c; hd = hc
            c = hi - gr*(hi-lo); hc = _eval_t(C, delta, k0, v, c)
        else:
            lo = c; c = d; hc = hd
            d = lo + gr*(hi-lo); hd = _eval_t(C, delta, k0, v, d)
    t_star = (lo+hi)*0.5
    f_star = _eval_t(C, delta, k0, v, t_star)
    return t_star, f_star

def _coord_desc(h, n, passes=50):
    dx = 2.0/n
    k0 = n-1
    nc = 2*n-1
    C = np.correlate(h, 1.0-h, mode='full') * dx
    best = float(np.max(C))
    
    for p in range(passes):
        improved = False
        for i in range(n):
            v = h[i]
            k = np.arange(nc)
            a = i - k + n - 1
            b = i + k - n + 1
            va = (a >= 0) & (a < n)
            vb = (b >= 0) & (b < n)
            delta = np.zeros(nc)
            if va.any():
                delta[va] = 1.0 - h[a[va]]
            if vb.any():
                delta[vb] -= h[b[vb]]
            t_star, f_star = _golden(C, delta, k0, v)
            if f_star < best - 1e-12:
                dt = t_star - v
                C += dt * delta
                C[k0] += t_star*(1.0-t_star) - v*(1.0-v)
                h[i] = t_star
                best = f_star
                improved = True
        if not improved:
            break
        if (p+1) % 10 == 0:
            C = np.correlate(h, 1.0-h, mode='full') * dx
            best = float(np.max(C))
    return h, best

def _lbfgs_obj(h, alpha, beta):
    n = len(h)
    dx = 2.0/n
    C = np.correlate(h, 1.0-h, mode='full') * dx
    m = np.max(alpha * C)
    e = np.exp(alpha * C - m)
    se = np.sum(e)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5*n)**2
    w = e / se
    nc = len(C)
    grad = np.zeros(n)
    for i in range(n):
        k = np.arange(nc)
        a = i - k + n - 1
        b = i + k - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        da = np.where(va, 1.0 - h[np.clip(a, 0, n-1)], 0.0)
        db = np.where(vb, h[np.clip(b, 0, n-1)], 0.0)
        grad[i] = dx * np.dot(w, da - db)
    grad += 2.0 * beta * (np.sum(h) - 0.5*n)
    return f, grad

def _lbfgs(h, n, alpha=100, beta=10, maxiter=500):
    res = minimize(_lbfgs_obj, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)]*n,
                  options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    return _fix(res.x, n)
```
