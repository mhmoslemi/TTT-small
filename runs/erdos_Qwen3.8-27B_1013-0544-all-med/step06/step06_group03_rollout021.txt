def soft_max_c5(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    return t * (m / t + np.log(np.sum(exps))) * dx

def soft_max_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            if shift < n:
                grad[shift:] += (1.0 - h[:n - shift]) * w
                grad[:n - shift] -= h[shift:] * w
        else:
            s = -shift
            grad[:n - s] += (1.0 - h[s:]) * w
            grad[s:] -= h[:n - s] * w
    return grad * dx

                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t
            except Exception:
                pass

    # Phase 6: Final targeted local search with many iterations
    if remaining() > 20:
        lh, lc5 = targeted_local_search(best_h, max_iter=15000, seed=seed + 9999)
        if lc5 < best_c5:
            best_h, best_c5 = lh, lc5

    # Final feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
