    return float(np.max(corr) * dx)

def softmax_val_grad(x, tau, lam):
    """Objective and gradient for L-BFGS-B."""
    n = len(x)
    dx = 2.0 / n
    one_minus_x = 1.0 - x
    corr = np.correlate(x, one_minus_x, mode='full') * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fix_binary(h, n):
    h = np.round(h).astype(float)
    target = n // 2
    no = int(h.sum())
    if no > target:
        idx = np.where(h == 1)[0][target:]
        h[idx] = 0.0
    elif no < target:
        idx = np.where(h == 0)[0][:target - no]
        idx = idx[np.random.choice(len(idx), target - no, replace=False)] if len(idx) > target - no else idx
        h[idx] = 1.0
    return h

def sm_objective(x, tau, lam):
    n = len(x)
    dx = 2.0 / n
    omx = 1.0 - x
    corr = np.correlate(x, omx, mode='full') * dx
    sv = np.sum(x) - n / 2.0
    tc = tau * corr
    m = tc.max()
    ec = np.exp(tc - m)
    se = ec.sum()
    obj = (m + np.log(se)) / tau + lam * sv * sv
    w = ec / se
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:

        lag = abs(k - (n - 1))

        else:
            tmp = np.zeros(n)
            tmp[:n - lag] += omx[lag:]
            tmp[lag:] -= x[:n - lag]
            grad += wk * tmp * dx
    grad += 2.0 * lam * sv
    return obj, grad
