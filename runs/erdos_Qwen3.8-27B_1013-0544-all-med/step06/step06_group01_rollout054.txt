        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                h[np.argmax(1.0 - h)] += d2
            else:
                h[np.argmax(h)] += d2
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, n, step_size, tol=0.01):
    dx = 2.0 / n
    C = c5_array(h)
    max_c = np.max(C)
    active = C >= max_c - tol
    n_active = np.sum(active)
    if n_active == 0:
        return h
    
    inv_na = 1.0 / n_active
    grad = np.zeros(n)
    for s in range(len(C)):
        if not active[s]:
            continue
        w = inv_na * dx
        # Term 1: (1-h[i-s+n-1]) for i in [max(0,s-n+1), min(n-1,s)]
        i_lo1 = max(0, s - n + 1)
        i_hi1 = min(n-1, s)
        if i_lo1 <= i_hi1:
            partners = np.arange(i_lo1, i_hi1+1) - s + n - 1
            grad[i_lo1:i_hi1+1] += w * (1.0 - h[partners])
        # Term 2: -h[i+s-n+1] for i in [max(0,n-1-s), min(n-1,2n-2-s)]
        i_lo2 = max(0, n - 1 - s)
        i_hi2 = min(n-1, 2*n - 2 - s)
        if i_lo2 <= i_hi2:
            partners = np.arange(i_lo2, i_hi2+1) + s - n + 1
            grad[i_lo2:i_hi2+1] -= w * h[partners]
    
    h_new = h - step_size * grad
    return fix(h_new, n)

def coord_descent(h, n, max_rounds=20, deltas=None):
    if deltas is None:
        deltas = [0.005, 0.01, 0.02, 0.05]
    best_c = c5_max(h)
    h = h.copy()
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            for d in deltas:
                for sign in [1, -1]:
                    h_try = h.copy()
                    h_try[i] = np.clip(h[i] + sign * d, 0, 1)
                    h_try = fix(h_try, n)
                    c = c5_max(h_try)
                    if c < best_c - 1e-12:
                        h = h_try
                        best_c = c
                        improved = True
        if not improved:
            break
    return h, best_c

def softmax_obj(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    f = (m + np.log(np.sum(es)) / alpha) + beta * (np.sum(h) - 0.5 * n) ** 2
    return f

def softmax_obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    f = (m + np.log(se) / alpha) + beta * (np.sum(h) - 0.5 * n) ** 2
    w = alpha * es / se
    
    grad = np.zeros(n)
    for s in range(len(C)):
        ws = w[s]
        if abs(ws) < 1e-15:
            continue
        i_lo1 = max(0, s - n + 1)
        i_hi1 = min(n-1, s)
        if i_lo1 <= i_hi1:
            partners = np.arange(i_lo1, i_hi1+1) - s + n - 1
            grad[i_lo1:i_hi1+1] += ws * dx * (1.0 - h[partners])
        i_lo2 = max(0, n - 1 - s)
        i_hi2 = min(n-1, 2*n - 2 - s)
        if i_lo2 <= i_hi2:
            partners = np.arange(i_lo2, i_hi2+1) + s - n + 1
            grad[i_lo2:i_hi2+1] -= ws * dx * h[partners]
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def lbfgs_optimize(h, n, alpha, beta, maxiter=500):
    res = minimize(softmax_obj_grad, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'maxfun': maxiter * 20, 'ftol': 1e-15})
    if res.x is not None:
        return fix(res.x, n)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def c5_max(h):
    return np.max(c5_array(h))

def fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                order = np.argsort(1.0 - h)
                for idx in order:
                    add = min(d2, 1.0 - h[idx])
                    if add > 0:
                        h[idx] += add
                        d2 -= add
                    if abs(d2) < 1e-12:
                        break
            else:
                order = np.argsort(h)
                for idx in order:
                    rem = min(-d2, h[idx])
                    if rem > 0:
                        h[idx] -= rem
                        d2 += rem
                    if abs(d2) < 1e-12:
                        break
    return h

def _grad_c5_components(h, n, weights):
    dx = 2.0 / n
    nc = 2 * n - 1
    grad = np.zeros(n)
    for s in range(nc):
        w = weights[s]
        if abs(w) < 1e-15:
            continue
        wdx = w * dx
        i_lo1 = max(0, s - n + 1)
        i_hi1 = min(n - 1, s)
        if i_lo1 <= i_hi1:
            partners = np.arange(i_lo1, i_hi1 + 1) - s + n - 1
            grad[i_lo1:i_hi1 + 1] += wdx * (1.0 - h[partners])
        i_lo2 = max(0, n - 1 - s)
        i_hi2 = min(n - 1, 2 * n - 2 - s)
        if i_lo2 <= i_hi2:
            partners = np.arange(i_lo2, i_hi2 + 1) + s - n + 1
            grad[i_lo2:i_hi2 + 1] -= wdx * h[partners]
    return grad

def _softmax_obj(h, alpha, beta):
    n = len(h)
    C = c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    return m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2

def _softmax_grad(h, alpha, beta):
    n = len(h)
    C = c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    f = m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    weights = alpha * es / se
    grad = _grad_c5_components(h, n, weights)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgs(h, n, alpha, beta, maxiter):
    res = minimize(_softmax_grad, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 20, 'ftol': 1e-15})
    if res.x is not None:
        return fix(res.x, n)
    return h

def _subgrad_step(h, n, step, tol):
    C = c5_array(h)
    max_c = np.max(C)
    active = (C >= max_c - tol).astype(float)
    n_act = np.sum(active)
    if n_act == 0:
        return h
    weights = active / n_act
    grad = _grad_c5_components(h, n, weights)
    return fix(h - step * grad, n)

def _coord_descent(h, n, rounds, deltas):
    best_c = c5_max(h)
    h = h.copy()
    for _ in range(rounds):
        improved = False
        for i in range(n):
            for d in deltas:
                for sgn in [1, -1]:
                    h_try = h.copy()
                    h_try[i] = np.clip(h[i] + sgn * d, 0.0, 1.0)
                    h_try = fix(h_try, n)
                    c = c5_max(h_try)
                    if c < best_c - 1e-12:
                        h = h_try
                        best_c = c
                        improved = True
        if not improved:
            break
    return h, best_c
