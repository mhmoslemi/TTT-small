    """Find optimal delta for moving mass from i to j.
    C: current correlation values (length 2n-1)
    d: change in correlation per unit delta (length 2n-1)
    h_i: current value at i
    h_j: current value at j
    Returns optimal delta in [0, min(h_i, 1-h_j)]
    """
    lo = 0.0
    hi = min(h_i, 1.0 - h_j)
    if hi <= 0:
        return 0.0
    
    # f(delta) = max_k (C[k] + delta * d[k])
    # subgradient at delta: weighted avg of d[k] for k in argmax
    
    def subgrad(delta):
        vals = C + delta * d
        m = vals.max()
        mask = np.isclose(vals, m, atol=1e-10)
        return d[mask].mean()
    
    sg_lo = subgrad(lo)
    sg_hi = subgrad(hi)
    
    if sg_lo >= 0:
        return lo
    if sg_hi <= 0:
        return hi
    
    # Bisection
    for _ in range(40):
        mid = (lo + hi) / 2
        sg = subgrad(mid)
        if sg > 0:
            hi = mid
        else:
            lo = mid
    
    return (lo + hi) / 2

    n = len(h)

    return grad

from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        if d > 0:
            free = 1.0 - h
            idx = np.argsort(-free)
            rem = d
            for i in idx:
                add = min(rem, free[i])
                h[i] += add
                rem -= add
                if rem < 1e-12:
                    break
        else:
            rem = -d
            idx = np.argsort(-h)
            for i in idx:
                take = min(rem, h[i])
                h[i] -= take
                rem -= take
                if rem < 1e-12:
                    break
    return h

def _grad(h, alpha):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sh = alpha * C
    m = sh.max()
    es = np.exp(sh - m)
    w = es / es.sum()
    grad = np.zeros(n)
    for i in range(n):
        a = i - np.arange(nc) + n - 1
        b = i + np.arange(nc) - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n-1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n-1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    return grad

def _pair_d(h, i, j):
    n = len(h)
    nc = 2 * n - 1
    idx = np.arange(nc)
    bb = j + idx - n + 1
    ba = j - idx + n - 1
    ab = i + idx - n + 1
    aa = i - idx + n - 1
    d = np.zeros(nc)
    m1 = (bb >= 0) & (bb < n)
    d[m1] += 1.0 - h[bb[m1]]
    m2 = (ba >= 0) & (ba < n)
    d[m2] -= h[ba[m2]]
    m3 = (ab >= 0) & (ab < n)
