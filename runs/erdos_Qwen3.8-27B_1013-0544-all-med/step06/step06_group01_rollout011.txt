            h[j] = np.clip(h[j] - delta, 0.0, 1.0)
            C = np.correlate(h, 1.0 - h, mode='full')
            if C.max() < best - 1e-14:
                best = C.max()
                updates += 1
        if it > 0 and it % 2000 == 0 and updates == 0:
            break
    h = _fix(h, n)
    return h, _c5_max(h)

def _structured_init(n, seed):
    rng = np.random.RandomState(seed)
    candidates = []
    h1 = np.zeros(n)
    h1[:n // 2] = 1.0
    candidates.append(_fix(h1, n))
    h2 = np.zeros(n)
    h2[::2] = 1.0
    h2 = np.clip(h2, 0, 1)
    candidates.append(_fix(h2, n))
    h3 = 0.5 + 0.4 * np.sin(2 * np.pi * np.arange(n) / n * 3 + rng.rand() * 2)
    candidates.append(_fix(h3, n))
    h4 = rng.rand(n)
    h4 = (h4 > 0.5).astype(float)
    candidates.append(_fix(h4, n))
    h5 = np.zeros(n)
    block = max(2, n // 4)
    for s in range(0, n, 2 * block):
        h5[s:s + block] = 1.0
    candidates.append(_fix(h5, n))
    best = None
    best_c = 1e9
    for c in candidates:
        cv = _c5_max(c)
        if cv < best_c:
            best_c = cv
            best = c
    return best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    n0 = len(initial_h_values)
    hb = _fix(initial_h_values.astype(float).copy(), n0)
    cb = _c5_max(hb)
    nb = n0

    def el():
        return time.time() - t0

    # Phase 1: L-BFGS from initial (warm start)
    for alpha in [30, 100, 300]:
        if el() > budget_s * 0.12:
            break
        hn = _lbfgs(hb, nb, alpha, max(1.0, 0.05 * alpha), maxiter=300)
        cn = _c5_max(hn)
        if cn < cb:
            cb, hb = cn, hn

    # Phase 2: Pair CD (main improvement driver)
    if el() < budget_s * 0.45:
        steps = max(3000, int((budget_s * 0.45 - el()) * 1500))
        hb, cb = _pair_cd(hb, nb, n_iter=steps, t0=t0, budget_s=budget_s)

    # Phase 3: L-BFGS polish with high alpha
    if el() < budget_s * 0.55:
        for alpha in [500, 1500]:
            if el() > budget_s * 0.56:
                break
            hn = _lbfgs(hb, nb, alpha, max(10.0, 0.05 * alpha), maxiter=600)
            cn = _c5_max(hn)
            if cn < cb:
                cb, hb = cn, hn

    # Phase 4: More pair CD
    if el() < budget_s * 0.75:
        steps = int((budget_s * 0.75 - el()) * 1200)
        hb, cb = _pair_cd(hb, nb, n_iter=max(2000, steps), t0=t0, budget_s=budget_s)

    # Phase 5: Try structured inits at same n
    if el() < budget_s * 0.82:
        si = _structured_init(nb, seed + 1)
        si = _lbfgs(si, nb, 100, 5.0, maxiter=200)
        si, sc = _pair_cd(si, nb, n_iter=3000, t0=t0, budget_s=budget_s)
        if sc < cb:
            cb, hb = sc, si

    # Phase 6: Try nearby resolutions
    for dn in [-5, 5, 10, -10]:
        n_new = nb + dn
        if n_new < 60 or n_new > 200 or el() > budget_s * 0.92:
            continue
        x_old = np.linspace(0, 2, nb, endpoint=False)
        x_new = np.linspace(0, 2, n_new, endpoint=False)
        ht = _fix(np.interp(x_new, x_old, hb), n_new)
        ht = _lbfgs(ht, n_new, 100, 5.0, maxiter=200)
        ht, cn = _pair_cd(ht, n_new, n_iter=3000, t0=t0, budget_s=budget_s)
        if cn < cb:
            cb, hb, nb = cn, ht, n_new

    # Phase 7: Final pair CD polish
    if el() < budget_s * 0.985:
        steps = int((budget_s * 0.985 - el()) * 1500)
        hb, cb = _pair_cd(hb, nb, n_iter=max(1000, steps), t0=t0, budget_s=budget_s)

    hb = _fix(hb, nb)
    cb = _c5_max(hb)
    return (hb, cb, nb)
