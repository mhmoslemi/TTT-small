    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_max_val(x, tau):
    c = c_values(x)
    cm = float(np.max(c))
    if tau <= 0: return cm
    z = (c - cm) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s): return cm
    return cm + tau * np.log(s)

def smooth_max_obj(x, tau):
    return smooth_max_val(x, tau)

def add_lag_contrib(x, grad, k, scale):
    n = x.size
    if k >= 0:
        lo = 0; hi = n - 1 - k
        if lo <= hi:
            grad[lo:hi+1] += scale * (1.0 - x[lo+k:hi+1+k])
        lo2 = k; hi2 = n - 1
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= scale * x[lo2-k:hi2+1-k]
    else:
        kp = -k
        lo = kp; hi = n - 1
        if lo <= hi:
            grad[lo:hi+1] += scale * (1.0 - x[lo-kp:hi+1-kp])
        lo2 = 0; hi2 = n - 1 - kp
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= scale * x[lo2+kp:hi2+1+kp]

def weighted_gradient(x, w):

    dx = 2.0 / float(n)
    grad = np.zeros(n)
    m = 2 * n - 1
    for ki in range(m):
        wi = float(w[ki])
        if wi < 1e-13: continue
        k = ki - (n - 1)
        add_lag_contrib(x, grad, k, wi * dx)
    return grad

def smooth_max_grad(x, tau):
    c = c_values(x)
    cm = float(np.max(c))
    if tau <= 0:
        # Use top-5 subgradient
        idx = np.argsort(c)[::-1][:5]
        w = np.zeros(c.size)
        w[idx] = 1.0 / len(idx)
    else:

        z -= np.max(z)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0 or not np.isfinite(s):
            idx = int(np.argmax(c))
            w = np.zeros(c.size)
            w[idx] = 1.0
        else:
            w = e / s
    return weighted_gradient(x, w)

            dn = rng.choice(remaining, half, replace=False)
            for i in up:
                y[i] = min(1.0, y[i] + T * 1.5)
            for i in dn:
                y[i] = max(0.0, y[i] - T * 1.5)
        else:
            y = x.copy()
            i, j = rng.choice(n, 2, replace=False)
            y[i], y[j] = y[j], y[i]
        y = repair(y)
        cy = exact_c(y)
        evals += 1
        delta = cy - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            x = y; cur_c = cy
            if cy < best_c:
                best_c = cy; best_x = y.copy(); no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        T = max(0.0008, T * cool)
        if no_improve > 800:
            T = max(T, 0.015)
            no_improve = 0
    return best_x, best_c


def pattern_search(x0, time_limit, seed):
    rng = np.random.RandomState(seed)
    x = repair(x0)
    best_x = x.copy()
    best_c = exact_c(x)
    n = x.size
    deadline = time.time() + max(0.5, time_limit)
    step = 0.04
    stagnant = 0

    while time.time() < deadline:
        improved = False
        order = rng.permutation(n)
        for i in order:
            for d in (1, -1):
                y = x.copy()
                y[i] = min(1.0, max(0.0, y[i] + d * step))
                y = repair(y)
                cy = exact_c(y)
                if cy < best_c - 1e-12:
                    x = y; best_x = y.copy(); best_c = cy
                    improved = True
                    break
            if improved:
                break
        if improved:
            step = min(0.08, step * 1.15)
            stagnant = 0
        else:
            step *= 0.65
            stagnant += 1
            if step < 0.002:
                y = repair(x + rng.normal(0, 0.015, n))
                cy = exact_c(y)
                if cy < best_c:
                    best_x = y.copy(); best_c = cy; x = y
                    step = 0.03; stagnant = 0
                elif stagnant > 8:
                    break
    return best_x, best_c


def resample_to(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    n_new = int(n_new)
    if n0 == n_new:
        return repair(h0)
    if n0 == 0:
        return repair(np.full(n_new, 0.5))
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    return repair(np.interp(x1, x0, h0))


def multi_start_slqp(x0, time_limit, seed):
    rng = np.random.RandomState(seed)
    x = repair(x0)
    best_x = x.copy()
    best_c = exact_c(x)
    deadline = time.time() + max(1.0, time_limit)
    n = x.size

    tau_seq = [0.008, 0.004, 0.002, 0.001, 0.0005]
    for tau in tau_seq:
        if time.time() >= deadline - 30:
            break
        maxiter = max(40, int(120 * 70.0 / max(50, n)))
        res_x = run_slsqp_smooth(x, tau, maxiter)
        if res_x is not None:
            res_x = repair(res_x)
            rc = exact_c(res_x)
            if rc < best_c:
                best_c = rc; best_x = res_x
            if rc < exact_c(x) + 0.002:
                x = res_x
            else:
                x = best_x
        else:
            x = best_x
    return best_x, best_c


def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)

    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size >= 10:
        init = repair(np.asarray(raw, dtype=float))
    else:
        init = repair(np.full(70, 0.5))

    best_x = init.copy()
    best_c = exact_c(init)

    # Phase 1: Simulated Annealing (global exploration)
    sa_budget = min(0.32 * float(budget_s), (deadline - time.time()) * 0.38)
    if sa_budget > 5:
        sa_x, sa_c = simulated_annealing(init, sa_budget, seed_val)
        if sa_c < best_c:
            best_c = sa_c; best_x = sa_x

    # Phase 2: SLSQP refinement from best
    slqp_budget = min(0.38 * float(budget_s), (deadline - time.time()) * 0.45)
    if slqp_budget > 10:
        s1_x, s1_c = multi_start_slqp(best_x, slqp_budget, seed_val + 100)
        if s1_c < best_c:
            best_c = s1_c; best_x = s1_x

    # Phase 3: Additional SLSQP from perturbed starts
    if time.time() < deadline - 80:
        rng = np.random.RandomState(seed_val + 700)
        for restart in range(4):
            if time.time() >= deadline - 60:
                break
            scale = 0.015 + 0.02 * restart
            x0 = repair(best_x + rng.normal(0, scale, best_x.size))
            xc = exact_c(x0)
            if xc < best_c:
                best_c = xc; best_x = x0
            t_rem = min(60.0, (deadline - time.time()) * 0.2)
            s2_x, s2_c = multi_start_slqp(x0, t_rem, seed_val + 200 + restart)
            if s2_c < best_c:
                best_c = s2_c; best_x = s2_x

    # Phase 4: Pattern search polishing
    if time.time() < deadline - 15:
        ps_time = min(0.15 * float(budget_s), (deadline - time.time()) * 0.6)
        if ps_time > 3:
            ps_x, ps_c = pattern_search(best_x, ps_time, seed_val + 500)
            if ps_c < best_c:
                best_c = ps_c; best_x = ps_x

    # Phase 5: Final tiny SA refinement
    if time.time() < deadline - 5:
        tiny_sa = min(8.0, (deadline - time.time()) * 0.5)
        if tiny_sa > 1:
            sa2_x, sa2_c = simulated_annealing(best_x, tiny_sa, seed_val + 999)
            if sa2_c < best_c:
                best_c = sa2_c; best_x = sa2_x

    # Phase 6: Try resampling at slightly different n
    if time.time() < deadline - 20:
        n_orig = best_x.size
        for n_try in (n_orig - 5, n_orig + 5):
            if n_try < 30 or n_try > 200:
                continue
            if time.time() >= deadline - 10:
                break
            x_r = resample_to(best_x, n_try)
            cr = exact_c(x_r)
            if cr < best_c:
                best_c = cr; best_x = x_r
            t_rem = min(25.0, (deadline - time.time()) * 0.3)
            if t_rem > 3:
                sx, sc = multi_start_slqp(x_r, t_rem, seed_val + 800 + n_try)
                if sc < best_c:
                    best_c = sc; best_x = sx

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
