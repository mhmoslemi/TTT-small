    if n0 == n_new:
        return repair(h0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return repair(np.interp(x1, x0, h0))

def run_slsqp(h0, maxiter, deadline):
    n = h0.size
    def cf(x):
        return np.array([x.sum() - 0.5 * n])
    def cj(x):
        return np.ones((1, n))
    c = lags(h0)
    C = c.max()
    tol = max(1e-9, 5e-5 * C)
    active = np.where(c >= C - tol)[0]
    if active.size > 8:
        active = np.argsort(c)[::-1][:8]
    def mf(x):
        cv = lags(x)
        return cv[active].mean()
    def mj(x):
        g = np.zeros(n)
        dx = 2.0 / n
        for k in active:
            s = k - (n - 1)
            lo = max(0, -s)
            hi = min(n - 1, n - 1 - s)
            if lo > hi:
                continue
            g[lo:hi + 1] += (1.0 - h0[lo + s:hi + s + 1])
            lo2 = max(0, lo + s)
            hi2 = min(n - 1, hi + s)
            if lo2 <= hi2:
                g[lo2:hi2 + 1] += (-h0[lo2 - s:hi2 - s + 1])
        return g * dx
    try:
        res = so.minimize(mf, h0.copy(), jac=mj, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': cf, 'jac': cj}],
                          options={'maxiter': maxiter, 'ftol': 1e-14})
        xr = repair(res.x)
        return xr, obj_val(xr)
    except Exception:
        return h0, obj_val(h0)

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    sv = int(seed)
    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size >= 10:
        h_init = repair(np.asarray(raw, dtype=float))
    else:
        h_init = repair(np.full(70, 0.5))
    best_h = h_init.copy()
    best_c = obj_val(best_h)
    n = best_h.size
    rng = np.random.RandomState(sv)

    # Phase 1: L-BFGS-B from initial with decreasing tau
    for tau in (0.05, 0.03, 0.02, 0.01, 0.005, 0.002):
        if time.time() > deadline - 60:
            break
        h_new, c_new = run_lbfgsb(best_h, tau, 50.0, 400, deadline)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new

    # Phase 2: Basin hopping
    hop_scale = 0.04
    n_hops = 0
    while time.time() < deadline - 120:
        sigma = hop_scale * (0.5 + rng.rand())
        h_pert = repair(best_h + sigma * rng.randn(n))
        c_pert = obj_val(h_pert)
        # Quick L-BFGS-B from perturbed point
        h_opt, c_opt = run_lbfgsb(h_pert, 0.01, 50.0, 80, deadline)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt
            hop_scale = max(0.005, hop_scale * 0.92)
        else:
            hop_scale = min(0.15, hop_scale * 1.03)
        n_hops += 1
        if n_hops % 10 == 0 and rng.rand() < 0.3:
            h_ad, c_ad = run_adam(best_h, 0.02, 50.0, 500, 0.005,
                                   min(deadline - 60, time.time() + 30), sv + n_hops)
            if c_ad < best_c:
                best_c = c_ad
                best_h = h_ad

    # Phase 3: Adam fine-tuning
    if time.time() < deadline - 60:
        for lr in (0.003, 0.001, 0.0005):
            if time.time() > deadline - 30:
                break
            h_ad, c_ad = run_adam(best_h, 0.003, 100.0, 2000, lr,
                                   deadline - 20, sv + 9999)
            if c_ad < best_c:
                best_c = c_ad
                best_h = h_ad

    # Phase 4: Try higher resolution
    if time.time() < deadline - 80:
        for n_new in (n + 20, n + 40):
            if time.time() > deadline - 40:
                break
            h_hi = resample(best_h, n_new)
            h_hi, c_hi = run_lbfgsb(h_hi, 0.01, 50.0, 300, deadline - 20)
            if c_hi < best_c:
                best_c = c_hi
                best_h = h_hi

    # Phase 5: SLSQP polish
    if time.time() < deadline - 10:
        h_sl, c_sl = run_slsqp(best_h, 200, deadline)
        if c_sl < best_c:
            best_c = c_sl
            best_h = h_sl
        # Second SLSQP with slightly different active set
        c_vec = lags(best_h)
        C_v = c_vec.max()
        near = np.where(c_vec >= C_v - 2e-4)[0]
        if near.size > 0 and near.size <= 15:
            _OPT['tau'] = 0.001
            _OPT['mu'] = 200.0
            try:
                res = so.minimize(_f, best_h.copy(), jac=_fprime, method='L-BFGS-B',
                                  bounds=[(0.0, 1.0)] * best_h.size,
                                  options={'maxiter': 200, 'ftol': 1e-15})
                xr = repair(res.x)
                if obj_val(xr) < best_c:
                    best_c = obj_val(xr)
                    best_h = xr
            except Exception:
                pass

    best_h = repair(best_h)
    best_c = obj_val(best_h)
    return best_h, best_c, int(best_h.size)
