    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(c_values(h)))

def swap_perturb(h, rng, scale):
    x = h.copy()
    n = x.size
    n_swaps = 1 + int(rng.rand() * 3)
    for _ in range(n_swaps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if x[i] < x[j]:
            i, j = j, i
        max_t = min(x[i], 1.0 - x[j])
        if max_t <= 0:
            continue
        delta = rng.uniform(-1, 1) * scale * max_t
        if delta > 0:
            delta = min(delta, max_t)
            x[i] -= delta
            x[j] += delta
        else:
            max_b = min(x[j], 1.0 - x[i])
            delta = max(delta, -max_b)
            x[i] -= delta
            x[j] += delta
    np.clip(x, 0.0, 1.0, out=x)
    target = 0.5 * float(n)
    d = target - float(x.sum())
    if abs(d) > 1e-15:
        if d > 0:
            mask = x < 1.0 - 1e-14
        else:
            mask = x > 1e-14
        if np.any(mask):
            x[mask] += d / float(mask.sum())
    np.clip(x, 0.0, 1.0, out=x)
    return x

def sa_optimize(h0, time_limit, seed):
    rng = np.random.RandomState(seed)
    x = h0.copy()
    best_x = x.copy()
    best_c = exact_c(x)
    cur_c = best_c
    deadline = time.time() + time_limit
    T = 0.012
    alpha = 0.99985
    scale = 0.12
    no_imp = 0
    while time.time() < deadline:
        new_x = swap_perturb(x, rng, scale)
        new_c = exact_c(new_x)
        if new_c < best_c:
            best_c = new_c
            best_x = new_x.copy()
            x = new_x
            cur_c = new_c
            no_imp = 0
        else:
            delta_c = new_c - cur_c
            if new_c < cur_c or rng.rand() < np.exp(-delta_c / T):
                x = new_x
                cur_c = new_c
            no_imp += 1
        T *= alpha
        if T < 0.0008:
            T = 0.0008
        if no_imp > 8000:
            T = 0.006 + 0.006 * rng.rand()
            scale = 0.06 + 0.08 * rng.rand()
            no_imp = 0
    return best_x, best_c

def active_subgrad(x, c, C, topk):
    n = x.size
    dx = 2.0 / float(n)
    grad = np.zeros(n)
    tol = max(1e-9, 1e-4 * C)
    active = np.where(c >= C - tol)[0]
    if active.size > topk:
        active = np.argsort(c)[::-1][:topk]
    m = 2 * n - 1
    for k in active:
        offset = int(k) - (n - 1)
        i_min = max(0, -offset)
        i_max = min(n, n - offset)
        for i in range(i_min, i_max):
            j = i + offset
            grad[i] += (1.0 - x[j])
            grad[j] += -x[i]
    grad *= dx
    if active.size > 0:
        grad /= float(active.size)
    return grad

def subgrad_optimize(h0, time_limit, seed):
    rng = np.random.RandomState(seed)
    x = h0.copy()
    n = x.size
    best_x = x.copy()
    best_c = exact_c(x)
    deadline = time.time() + time_limit
    steps = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
    si = 0
    no_imp = 0
    while time.time() < deadline:
        c = c_values(x)
        C = float(np.max(c))
        if C < best_c - 1e-12:
            best_c = C
            best_x = x.copy()
            no_imp = 0
        else:
            no_imp += 1
        if no_imp > 300:
            if si < len(steps) - 1:
                si += 1
            else:
                x = repair(best_x + rng.normal(0, 0.008, n))
            no_imp = 0
        grad = active_subgrad(x, c, C, 4)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break
        step = steps[si]
        accepted = False
        for _ in range(10):
            x_new = repair(x - step * grad)
            c_new = exact_c(x_new)
            if c_new < C - 1e-13:
                x = x_new
                accepted = True
                break
            step *= 0.5
            if step < 1e-12:
                break
        if accepted:
            no_imp = 0
        else:
            no_imp += 1
        if rng.rand() < 0.08:
            x_try = repair(x + rng.normal(0, 0.003, n))
            c_try = exact_c(x_try)
            if c_try < best_c:
                best_c = c_try
                best_x = x_try.copy()
                x = x_try
                no_imp = 0
    return best_x, best_c

def coord_descent(h0, time_limit, seed):
    rng = np.random.RandomState(seed)
    x = h0.copy()
    n = x.size
    best_x = x.copy()
    best_c = exact_c(x)
    deadline = time.time() + time_limit
    no_imp_rounds = 0
    while time.time() < deadline:
        order = rng.permutation(n)
        improved = False
        for idx in order:
            for delta in (0.03, -0.03, 0.015, -0.015, 0.007, -0.007, 0.003, -0.003):
                trial = x.copy()
                trial[idx] += delta
                trial = repair(trial)
                c_trial = exact_c(trial)
                if c_trial < best_c - 1e-12:
                    best_c = c_trial
                    best_x = trial
                    x = trial
                    improved = True
                    break
        if improved:
            no_imp_rounds = 0
        else:
            no_imp_rounds += 1
            if no_imp_rounds >= 4:
                break
    return best_x, best_c

def resample_to(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    if n0 == n_new:
        return repair(h0)
    dt0 = 2.0 / float(n0)
    x0 = (np.arange(n0) + 0.5) * dt0
    x1 = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    return repair(np.interp(x1, x0, h0))

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)
    raw = globals().get("initial_h_values", None)
    if raw is not None:
        init = repair(np.asarray(raw, dtype=float))
    else:
        init = repair(np.full(70, 0.5))
    best_x = init.copy()
    best_c = exact_c(init)
    # Phase 1: SA at base resolution
    if time_left(deadline) > 120:
        t = min(350, time_left(deadline) * 0.32)
        x1, c1 = sa_optimize(best_x, t, seed_val)
        if c1 < best_c:
            best_c, best_x = c1, x1
    # Phase 2: Subgradient at base resolution
    if time_left(deadline) > 100:
        t = min(200, time_left(deadline) * 0.22)
        x2, c2 = subgrad_optimize(best_x, t, seed_val + 200)
        if c2 < best_c:
            best_c, best_x = c2, x2
    # Phase 3: Multi-resolution
    for n_t in (50, 80, 100, 140):
        if time_left(deadline) < 100:
            break
        h_r = resample_to(best_x, n_t)
        c_r = exact_c(h_r)
        if time_left(deadline) > 120:
            t = min(180, time_left(deadline) * 0.2)
            xr, cr = sa_optimize(h_r, t, seed_val + n_t)
            if cr < c_r:
                c_r, h_r = cr, xr
        if time_left(deadline) > 60:
            t = min(100, time_left(deadline) * 0.15)
            xr, cr = subgrad_optimize(h_r, t, seed_val + 500 + n_t)
            if cr < c_r:
                c_r, h_r = cr, xr
        if c_r < best_c:
            best_c, best_x = c_r, h_r
    # Phase 4: Coordinate descent polish
    if time_left(deadline) > 40:
        t = min(80, time_left(deadline) * 0.3)
        x4, c4 = coord_descent(best_x, t, seed_val + 999)
        if c4 < best_c:
            best_c, best_x = c4, x4
    # Phase 5: Final SA burst
    if time_left(deadline) > 30:
        t = min(60, time_left(deadline) * 0.5)
        x5, c5 = sa_optimize(best_x, t, seed_val + 7777)
        if c5 < best_c:
            best_c, best_x = c5, x5
    # Final subgradient if time remains
    if time_left(deadline) > 15:
        t = min(30, time_left(deadline) * 0.6)
        x6, c6 = subgrad_optimize(best_x, t, seed_val + 8888)
        if c6 < best_c:
            best_c, best_x = c6, x6
    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
