import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def soft_max_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    shifted = (conv - m) / t
    exps = np.exp(shifted)
    wsum = exps.sum()
    weights = exps / wsum
    obj = t * (m / t + np.log(wsum)) * dx
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        grad += w * conv_grad_single(h, shift)
    grad *= dx
    return obj, grad

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

_sm_t = [1.0]

def _sm_objective(h):
    o, _ = soft_max_obj_grad(h, _sm_t[0])
    return o

def _sm_gradient(h):
    _, g = soft_max_obj_grad(h, _sm_t[0])
    return g

def lbfgs_smooth(h, n, t, maxiter=1500):
    _sm_t[0] = t
    try:
        res = minimize(_sm_objective, h, jac=_sm_gradient, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14,
                               'maxcor': 15, 'maxls': 30})
        return project_feasible(res.x, n)
    except Exception:
        return h

def active_set_pgd(h, max_iter=20000, lr_init=0.001, tol_frac=0.005):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    for it in range(max_iter):
        conv = np.correlate(h, 1.0 - h, mode="full")
        max_val = conv.max()
        c5_val = max_val * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
        threshold = max_val * (1.0 - tol_frac)
        active = np.where(conv >= threshold)[0]
        grad = np.zeros(n)
        for idx in active:
            shift = idx - (n - 1)
            grad += conv_grad_single(h, shift)
        grad /= len(active)
        grad *= dx
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new <= c5_val + 1e-15:
            h = h_new
            lr = min(lr * 1.02, 0.02)
        else:
            lr *= 0.5
            if lr < 1e-15:
                break
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n)

def make_perturbation(best_h, bn, rng, ptype):
    h_pert = best_h.copy()
    if ptype == 0:
        f = rng.randint(1, 10)
        h_pert += 0.03 * rng.uniform(-1, 1) * np.sin(2 * np.pi * f * np.arange(bn) / bn)
    elif ptype == 1:
        for i in range(4):
            h_pert += 0.02 * rng.uniform(-1, 1) * np.cos(2 * np.pi * (i + 1) * np.arange(bn) / bn)
    elif ptype == 2:
        noise = rng.normal(0, 0.015, bn)
        h_pert += np.convolve(noise, np.ones(7) / 7.0, mode='same')
    elif ptype == 3:
        bs = max(1, bn // 20)
        for b in range(bn // bs):
            h_pert[b * bs:(b + 1) * bs] += rng.uniform(-0.02, 0.02)
    else:
        shift_amt = rng.randint(1, max(2, bn // 50))
        h_pert = np.roll(h_pert, shift_amt)
    return project_feasible(h_pert, bn)

    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with decreasing temperature
    for t_val in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() - t0 > budget_s - 80:
            break
        h_opt = lbfgs_smooth(best_h, n0, t_val, maxiter=1200)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 2: Active set PGD with two tolerance levels
    if time.time() - t0 < budget_s - 80:
        h_opt, c5_opt = active_set_pgd(best_h, max_iter=20000, lr_init=0.002, tol_frac=0.02)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
    if time.time() - t0 < budget_s - 60:
        h_opt, c5_opt = active_set_pgd(best_h, max_iter=30000, lr_init=0.001, tol_frac=0.003)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Phase 3: Higher resolutions
    for n_try in [400, 500, 640, 800, 1000, 1280, 1600, 2000]:
        if time.time() - t0 > budget_s - 200:
            break
        if n_try <= len(best_h):
            continue
        h_up = upsample(best_h, n_try)
        if n_try <= 500:
            for t_val in [0.02, 0.005]:
                if time.time() - t0 > budget_s - 100:
                    break
                h_opt = lbfgs_smooth(h_up, n_try, t_val, maxiter=800)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_h, best_c5 = h_opt, c5_opt
                    h_up = best_h
                else:
                    h_up = h_opt
        if time.time() - t0 < budget_s - 80:
            max_it = min(20000, int((budget_s - time.time() - 100) * 1.5))
            if n_try > 800:
                max_it = min(max_it, 8000)
            if n_try > 1500:
                max_it = min(max_it, 4000)
            h_opt, c5_opt = active_set_pgd(h_up, max_iter=max_it, lr_init=0.001, tol_frac=0.005)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt

    # Phase 4: Perturbation-based escape at best resolution
    bn = len(best_h)
    for trial in range(60):
        if time.time() - t0 > budget_s - 80:
            break
        h_pert = make_perturbation(best_h, bn, rng, trial % 5)
        h_opt = lbfgs_smooth(h_pert, bn, 0.01, maxiter=400)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt
        if time.time() - t0 < budget_s - 50:
            h_opt2, c5_opt2 = active_set_pgd(h_opt, max_iter=5000, lr_init=0.001, tol_frac=0.003)
            if c5_opt2 < best_c5:
                best_h, best_c5 = h_opt2, c5_opt2

    # Phase 5: Random restarts
    for trial in range(30):
        if time.time() - t0 > budget_s - 60:
            break
        hr = rng.uniform(0.0, 1.0, bn)
        hr = project_feasible(hr, bn)
        h_opt = lbfgs_smooth(hr, bn, 0.05, maxiter=300)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    # Final polish
    if time.time() - t0 < budget_s - 40:
        h_opt, c5_opt = active_set_pgd(best_h, max_iter=40000, lr_init=0.0005, tol_frac=0.002)
        if c5_opt < best_c5:

        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_h, best_c5 = h_opt, c5_opt

    return best_h, best_c5, len(best_h)
