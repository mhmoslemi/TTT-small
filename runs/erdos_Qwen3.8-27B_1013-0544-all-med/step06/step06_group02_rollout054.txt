                stagnant = 0
    return best_h, best_val

def update_best(h, val, state):
    if val < state[1]:
        state[0] = h.copy()
        state[1] = val
    return state

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 10
    state = [None, float('inf')]
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except NameError:
        h_init = None
    if h_init is not None:
        val0, _, _ = obj_and_corr(h_init)
        state = update_best(h_init.copy(), val0, state)
        h1, v1 = pair_sweep(h_init.copy(), len(h_init), M_lags=40, max_rounds=60, t_end=t_end - 180)
        state = update_best(h1, v1, state)
    if state[0] is not None and time.time() < t_end - 80:
        N_b = len(state[0])
        h2, v2 = gradient_optimize(state[0].copy(), N_b, rng, t_end - 80, max_iter=15000)
        state = update_best(h2, v2, state)
        h3, v3 = pair_sweep(state[0].copy(), N_b, M_lags=50, max_rounds=40, t_end=t_end - 60)
        state = update_best(h3, v3, state)
    if state[0] is not None:
        for n_t in [96, 144, 160, 200]:
            if time.time() > t_end - 50:
                break
            h_start = interpolate_to_n(state[0], n_t)
            t_phase = min(time.time() + 70, t_end - 25)
            h_m, v_m = pair_sweep(h_start, n_t, M_lags=35, max_rounds=35, t_end=t_phase)
            state = update_best(h_m, v_m, state)
            if time.time() < t_end - 30:
                h_g, v_g = gradient_optimize(h_m.copy(), n_t, rng, t_end - 30, max_iter=8000)
                state = update_best(h_g, v_g, state)
    if time.time() < t_end - 30:
        for n_bin in [20, 28, 32]:
            if time.time() > t_end - 20:
                break
            for trial in range(4):
                if time.time() > t_end - 15:
                    break
                h_b = np.zeros(n_bin)
                idx = rng.choice(n_bin, size=n_bin // 2, replace=False)
                h_b[idx] = 1.0
                h_b = project_feasible(h_b, n_bin)
                h_b2, v_b2 = pair_sweep(h_b, n_bin, M_lags=min(18, 2 * n_bin - 1), max_rounds=20, t_end=t_end - 10)
                if state[0] is not None:
                    N_up = len(state[0])
                    h_up = interpolate_to_n(h_b2, N_up)
                else:
                    N_up = 120
                    h_up = project_feasible(0.5 * np.ones(N_up), N_up)
                t_ph = min(time.time() + 35, t_end - 10)
                h_up2, v_up2 = pair_sweep(h_up, N_up, M_lags=30, max_rounds=25, t_end=t_ph)
                state = update_best(h_up2, v_up2, state)
    if state[0] is not None and time.time() < t_end - 5:
        N_f = len(state[0])
        h_f, v_f = pair_sweep(state[0].copy(), N_f, M_lags=50, max_rounds=80, t_end=t_end - 5)
        state = update_best(h_f, v_f, state)
        if time.time() < t_end - 2:
            h_f2, v_f2 = gradient_optimize(h_f.copy(), N_f, rng, t_end, max_iter=8000)
            state = update_best(h_f2, v_f2, state)
    if state[0] is None:
        state[0] = 0.5 * np.ones(47)
        state[1] = obj_and_corr(state[0])[0]
    return state[0], state[1], len(state[0])
