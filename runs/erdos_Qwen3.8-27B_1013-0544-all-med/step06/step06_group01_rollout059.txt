                        hj = h[j] - sign * delta
                        if 0 <= hi <= 1 and 0 <= hj <= 1:
                            h_save_i, h_save_j = h[i], h[j]
                            h[i], h[j] = hi, hj
                            c5t = _c5_max(h)
                            if c5t < c5 - 1e-12:
                                c5 = c5t
                                improved = True
                                break
                            h[i], h[j] = h_save_i, h_save_j
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n = len(initial_h_values)
    h = _fix(initial_h_values.astype(float).copy(), n)
    c5 = _c5_max(h)

    # Phase 1: Progressive L-BFGS-B
    for alpha in [20, 50, 100, 200, 500]:
        if _el() > budget_s * 0.2:
            break
        h = _lbfgs(h, alpha, alpha * 0.1, maxiter=800)
        c5 = _c5_max(h)

    # Phase 2: Coordinate descent polish
    if _el() < budget_s * 0.35:
        h, c5 = _coord_descent(h, n, sweeps=50)

    # Phase 3: Try different n values
    n_candidates = sorted(set([n, n + 5, n + 10, n + 20, n + 40, 110, 130, 150, 200]))
    n_candidates = [x for x in n_candidates if 60 <= x <= 250]
    for n_try in n_candidates:
        if _el() > budget_s * 0.55:
            break
        if n_try == n:
            continue
        n_cur = len(h)
        x_old = np.linspace(0, 2, n_cur, endpoint=False) + 1.0 / n_cur
        x_new = np.linspace(0, 2, n_try, endpoint=False) + 1.0 / n_try
        ht = np.interp(x_new, x_old, h)
        ht = _fix(ht, n_try)
        for alpha in [50, 200]:
            if _el() > budget_s * 0.58:
                break
            ht = _lbfgs(ht, alpha, alpha * 0.1, maxiter=600)
            ht, c5t = _coord_descent(ht, n_try, sweeps=40)
            if c5t < c5 - 1e-10:
                h, c5, n = ht, c5t, n_try

    # Phase 4: Binarize and binary local search
    if _el() < budget_s * 0.65:
        hb = _binarize(h, n)
        c5b = _c5_max(hb)
        if c5b < c5:
            hb, c5 = _binary_local_search(hb, n, max_iters=200)
            h = hb.copy()

    # Phase 5: Random restarts with coordinate descent
    for r in range(40):
        if _el() > budget_s * 0.85:
            break
        scale = 0.02 + 0.05 * rng.rand()
        ht = _fix(h + rng.randn(n) * scale, n)
        ht = _lbfgs(ht, 150, 15, maxiter=500)
        ht, c5t = _coord_descent(ht, n, sweeps=25)
        if c5t < c5 - 1e-10:
            h, c5 = ht, c5t

    # Phase 6: High-alpha polish + final coordinate descent
    if _el() < budget_s * 0.95:
        for alpha in [500, 1000, 2000]:
            if _el() > budget_s * 0.96:
                break
            h = _lbfgs(h, alpha, alpha * 0.1, maxiter=1500)
            c5 = _c5_max(h)
        h, c5 = _coord_descent(h, n, sweeps=80)

    # Phase 7: Pair swap polish on best solution
    if _el() < budget_s * 0.99:
        h, c5 = _pair_swap(h, n, max_rounds=200)
        h = _fix(h, n)
        c5 = _c5_max(h)

    return (h, c5, n)
