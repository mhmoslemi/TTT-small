Actually, let me reconsider. The issue with very high alpha is that the log-sum-exp becomes numerically unstable. The formula is:
m + (mx + log(sum(exp(shifted - mx)))) / alpha

where shifted = alpha * (c - m) and mx = max(shifted).

    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    iters = 0
    step_scale = 0.015
    while remaining_fn() > 20 and iters < max_iters:
        iters += 1
        if iters % 5000 == 0:
            step_scale = max(0.001, step_scale * 0.85)
        i = rng.randint(n)
        j = rng.randint(n)
        while j == i:
            j = rng.randint(n)
        steps = [step_scale, step_scale * 0.5, step_scale * 0.1, step_scale * 2.0, step_scale * 3.0]
        improved = False
        for d in steps:
            for s in [1, -1]:
                hi = best_h[i] + s * d
                hj = best_h[j] - s * d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h_try = best_h.copy()
                    h_try[i] = hi
                    h_try[j] = hj
                    c = compute_c5(h_try, n)
                    if c < best_c5:
                        best_h = h_try
                        best_c5 = c
                        improved = True
                        break
            if improved:
                break
    return best_h, best_c5

def slsqp_phase(h, n, alphas, remaining_fn, bounds, cons):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    for alpha in alphas:
        if remaining_fn() < 10:
            break
        for _ in range(2):
            if remaining_fn() < 8:
                break
            try:
                res = minimize(obj_smooth, best_h, args=(n, alpha), method='SLSQP',
                             bounds=bounds, constraints=cons,
                             options={'maxiter': 10000, 'ftol': 1e-18})
                h_try = project_feasible(res.x, n)
                c = compute_c5(h_try, n)
                if c < best_c5:
                    best_h, best_c5 = h_try, c
            except Exception:
                pass
    return best_h, best_c5

def upsample(h, n_old, n_new):
    old_indices = np.linspace(0, n_old, n_new, endpoint=False).astype(int)
    h_new = h[old_indices]
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n = len(h0)
    except Exception:
        n = 80
        h0 = np.full(n, 0.5)

    h0 = project_feasible(h0, n)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n)

    def remaining():
        return budget_s - (time.time() - t0)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n,)}]

    # Phase 1: SLSQP continuation
    best_h, best_c5 = slsqp_phase(best_h, n, [10, 50, 200, 1000, 5000], remaining, bounds, cons)

    # Phase 2: Subgradient descent on true max
    if remaining() > 80:
        best_h, best_c5 = subgradient_phase(best_h, n, remaining)

    # Phase 3: Swap search
    if remaining() > 60:
        best_h, best_c5 = swap_search(best_h, n, remaining, seed)

    # Phase 4: High-alpha SLSQP
    best_h, best_c5 = slsqp_phase(best_h, n, [5000, 50000, 200000], remaining, bounds, cons)

    # Phase 5: More subgradient + swaps
    if remaining() > 80:
        best_h, best_c5 = subgradient_phase(best_h, n, remaining)
    if remaining() > 40:
        best_h, best_c5 = swap_search(best_h, n, remaining, seed + 50)

    # Phase 6: Try higher resolution
    if remaining() > 200:
        n_up = n * 2
        bounds_up = [(0.0, 1.0)] * n_up
        cons_up = [{'type': 'eq', 'fun': sum_con, 'jac': sum_con_jac, 'args': (n_up,)}]
        h_up = upsample(best_h, n, n_up)
        c_up = compute_c5(h_up, n_up)
        best_h_up = h_up.copy()
        best_c5_up = c_up
        best_h_up, best_c5_up = slsqp_phase(h_up, n_up, [100, 1000, 10000, 100000], remaining, bounds_up, cons_up)
        if remaining() > 60:
            best_h_up, best_c5_up = swap_search(best_h_up, n_up, remaining, seed + 200)
        if remaining() > 30:
            best_h_up, best_c5_up = slsqp_phase(best_h_up, n_up, [500000], remaining, bounds_up, cons_up)
        if best_c5_up < best_c5:
            best_h, best_c5, n = best_h_up, best_c5_up, n_up
            bounds = bounds_up
            cons = cons_up

    # Phase 7: Final refinement
    if remaining() > 30:
        best_h, best_c5 = subgradient_phase(best_h, n, remaining)
    if remaining() > 30:
        best_h, best_c5 = swap_search(best_h, n, remaining, seed + 500)
    if remaining() > 10:
        best_h, best_c5 = slsqp_phase(best_h, n, [100000, 500000], remaining, bounds, cons)

    # Final fine swaps
    if remaining() > 10:
        best_h, best_c5 = swap_search(best_h, n, remaining, seed + 999)

    # Final feasibility
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h, n)

    return best_h, best_c5, n
