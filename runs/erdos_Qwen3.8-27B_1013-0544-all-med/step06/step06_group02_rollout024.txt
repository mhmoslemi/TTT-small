  Contribution: h[i+k-(N-1)] * (1-h[i])
  Change when h[i] → h[i]+δ: -h[i+k-(N-1)] * δ

Total change in output[k]:
- If k ∈ [i, i+N-1]: δ*(1-h[i-k+(N-1)])
- If k ∈ [N-1-i, 2N-2-i]: -δ*h[i+k-(N-1)]

These ranges can overlap. In the overlap, both terms apply.

Now, the coefficient of δ in output[k] is:
coeff[k] = (1-h[i-k+(N-1)])_caseA - h[i+k-(N-1)]_caseB

where _caseA is 1 if k ∈ [i, i+N-1] else 0, and _caseB is 1 if k ∈ [N-1-i, 2N-2-i] else 0.

This matches what I had before. Good.

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_value(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def solve_1d_minimax(base, coeffs, lam):
    lo, hi = 0.0, 1.0
    for _ in range(70):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(base + m1 * coeffs) + lam * m1
        f2 = np.max(base + m2 * coeffs) + lam * m2
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coord_descend(h, lam, max_sweeps=200, tol=1e-12, shuffle_order=None):
    N = len(h)
    h = h.copy()
    corr = compute_corr(h)
    order = shuffle_order if shuffle_order is not None else range(N)
    for sweep in range(max_sweeps):
        max_change = 0.0
        for i in order:
            old = h[i]
            coeffs = np.zeros(2 * N - 1)
            k_s = N - 1 - i
            k_e = 2 * N - 2 - i
            for k in range(k_s, k_e + 1):
                coeffs[k] += (1.0 - h[i + k - (N - 1)])
            k_s2 = i
            k_e2 = min(i + N - 1, 2 * N - 2)
            for k in range(k_s2, k_e2 + 1):
                coeffs[k] -= h[i + (N - 1) - k]
            base = corr - old * coeffs
            t_opt = solve_1d_minimax(base, coeffs, lam)
            change = abs(t_opt - old)
            if change > max_change:
                max_change = change
            h[i] = t_opt
            corr = base + t_opt * coeffs
        if max_change < tol:
            break
        if sweep % 50 == 49:
            corr = compute_corr(h)
    return h

def interpolate_h(h_src, n_dst):
    n_src = len(h_src)
    if n_src == n_dst:
        return h_src.copy()
    x0 = np.arange(n_src) / n_src
    x1 = np.arange(n_dst) / n_dst
    h = np.interp(x1, x0, h_src)
    return project_feasible(h, n_dst)
