    return repair(y)

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(10.0, float(budget_s))
    rng = np.random.RandomState(int(seed))

    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size >= 10:
        h = repair(np.asarray(raw, dtype=float))
    else:
        h = repair(np.full(70, 0.5))

    h = coord_descent(h, 10, rng)
    best_h = h.copy()
    best_c = exact_c(h)

    stagnation = 0
    scale = 0.08
    cycle = 0

    while time.time() < deadline - 25:
        cycle += 1
        h = coord_descent(h, 3, rng)
        c = exact_c(h)

        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            stagnation = 0
        else:
            h = best_h.copy()
            stagnation += 1

        if stagnation > 1:
            improved = False

            # Strategy 1: multi-coordinate perturbation
            for _ in range(25):
                if time.time() > deadline - 30:
                    break
                h_try = best_h.copy()
                npert = rng.randint(2, 7)
                idxs = rng.choice(h_try.size, npert, replace=False)
                h_try[idxs] = np.clip(h_try[idxs] + rng.normal(0, scale, npert), 0, 1)
                h_try = repair(h_try)
                h_try = coord_descent(h_try, 4, rng)
                c_try = exact_c(h_try)
                if c_try < best_c - 1e-13:
                    best_c = c_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    stagnation = 0
                    improved = True
                    break

            # Strategy 2: binary rounding + noise
            if not improved and time.time() < deadline - 50:
                n = best_h.size
                k = n // 2
                for _ in range(10):
                    idx_top = np.argsort(best_h)[::-1][:k]
                    h_bin = np.zeros(n)
                    h_bin[idx_top] = 1.0
                    h_try = h_bin + rng.normal(0, 0.06 + 0.04 * rng.rand(), n)
                    h_try = repair(h_try)
                    h_try = coord_descent(h_try, 4, rng)
                    c_try = exact_c(h_try)
                    if c_try < best_c - 1e-13:
                        best_c = c_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        stagnation = 0
                        improved = True
                        break

            # Strategy 3: swap-based perturbation
            if not improved and time.time() < deadline - 50:
                for _ in range(15):
                    h_try = best_h.copy()
                    nswaps = rng.randint(2, 5)
                    for _s in range(nswaps):
                        i, j = rng.choice(h_try.size, 2, replace=False)
                        h_try[i], h_try[j] = h_try[j], h_try[i]
                    h_try = repair(h_try)
                    h_try = coord_descent(h_try, 4, rng)
                    c_try = exact_c(h_try)
                    if c_try < best_c - 1e-13:
                        best_c = c_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        stagnation = 0
                        improved = True
                        break

            # Strategy 4: resolution change
            if not improved and time.time() < deadline - 70:
                n_list = [x for x in [60, 80, 100, 120, 50, 90, 140, 110, 130, 85]
                          if x != best_h.size and 40 <= x <= 160]
                rng.shuffle(n_list)
                for n_try in n_list[:4]:
                    if time.time() > deadline - 35:
                        break
                    h_try = resample(best_h, n_try)
                    h_try = coord_descent(h_try, 8, rng)
                    c_try = exact_c(h_try)
                    if c_try < best_c - 1e-13:
                        best_c = c_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        stagnation = 0
                        improved = True
                        break

            if not improved:
                scale = min(0.3, scale * 1.25)
                if scale > 0.28:
                    scale = 0.07
                h = best_h.copy()

    # Final polish with many passes
    if time.time() < deadline - 3:
        h_pol = coord_descent(best_h, 20, rng)
        c_pol = exact_c(h_pol)
        if c_pol < best_c:
            best_c = c_pol
            best_h = h_pol

    best_h = repair(best_h)
    best_c = exact_c(best_h)
    return best_h, best_c, int(best_h.size)
