def project_feasible_fast(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} in one shot."""
    h = np.clip(h, 0.0, 1.0)
    diff = n/2.0 - h.sum()
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        # Need to add mass, add to elements with most room (highest 1-h)
        room = 1.0 - h
        # Add proportionally to room
        total_room = room.sum()
        if total_room < 1e-15:
            return h
        h = h + diff * room / total_room
    else:
        # Need to remove mass, remove proportionally to h
        total = h.sum()
        if total < 1e-15:
            return h
        h = h + diff * h / total
    return np.clip(h, 0.0, 1.0)

  So c[k] increases by h[i-k] for each valid k where i is the second index.

Similarly for h[j]: 0→1
- As first factor: h[j] * (1-h[j+k]). Before: 0. After: 1*(1-h[j+k]).
  So c[k] increases by (1-h[j+k]) for each valid k.
  
- As second factor: h[j-k] * (1-h[j]). Before: h[j-k]*1. After: h[j-k]*0.
  So c[k] decreases by h[j-k] for each valid k.

So the total change to c[k] is:
Δc[k] = -（1-h[i+k]）[i+k valid] + h[i-k][i-k valid] + (1-h[j+k])[j+k valid] - h[j-k][j-k valid]

This is O(1) per lag k (just checking a few indices). So I can compute the new max correlation in O(n) time (scanning all lags), or even faster if I maintain the correlation array incrementally.

This makes simulated annealing very fast! Each move is O(n) to evaluate, and I can do millions of moves in the budget.

def sa_binary(h, n, rng, deadline):
    """Simulated annealing on binary solutions."""
    dx = 2.0 / n
    target = n // 2  # number of 1s
    
    # Ensure binary
    h = (h > 0.5).astype(float)
    if h.sum() != target:
        # Fix the count
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if h.sum() > target:

                h[i] = 0
        else:

                h[i] = 1

    corr = np.correlate(h, 1-h, mode='full').astype(float)
    max_c5 = corr.max() * dx
    
    ones_set = set(np.where(h == 1)[0].tolist())
    zeros_list = np.where(h == 0)[0].tolist()
    
    T = 0.01  # initial temperature
    T_min = 1e-6
    cooling = 0.999
    
    while time.time() < deadline:
        # Propose a swap
        if not ones_set or not zeros_list:
            break
        i = rng.choice(list(ones_set))
        j = rng.choice(zeros_list)
        
        # Compute delta
        old_corr = corr.copy()
        
        # Compute changes
        # For i: 1 -> 0
        # For j: 0 -> 1
        
        # Update correlation
        # c[k] = sum_m h[m]*(1-h[m+k])
        # Using the full correlation indexing:
        # corr[m] = sum_i h[i] * (1-h[i+m-(n-1)])
        # That's the numpy convention for mode='full'
        
        # Let me just recompute for simplicity (n is small)
        h_swap = h.copy()
        h_swap[i] = 0
        h_swap[j] = 1
        new_corr = np.correlate(h_swap, 1-h_swap, mode='full')
        new_max = new_corr.max() * dx
        
        # Metropolis acceptance
        delta = new_max - max_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            h = h_swap
            corr = new_corr
            max_c5 = new_max
            ones_set.remove(i)
            ones_set.add(j)
            zeros_list.remove(j)  # O(n) but n is small
            zeros_list.append(i)
        
        T = max(T_min, T * cooling)
    
    return h, max_c5

The change to corr_full[m] comes from terms where either index is i or j:
- Terms with h[i] as first factor: h[i] * g[i + m - (n-1)]

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(300):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()

                break
            h += (diff / total) * room

            total = h.sum()
            if total < 1e-15:

    if m > 700:
