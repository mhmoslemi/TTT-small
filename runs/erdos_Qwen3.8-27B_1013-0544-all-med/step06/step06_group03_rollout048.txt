3. For each of these shifts, try to reduce the overlap by adjusting h
4. The adjustment for shift s: we want to minimize sum_{i valid} h[i](1-h[i+s])
   This is like: for pairs (i, i+s) where i is valid, we want h[i] and h[i+s] to be "similar" (both 0 or both 1)
   So the adjustment is: make h[i] and h[i+s] more similar

This suggests a "smoothing" operation at specific shifts. We can implement this as:
- For each active shift s: for each valid i, move h[i] and h[i+s] toward their mean
- This reduces the product h[i](1-h[i+s])

This is like a "heat equation" step at specific shifts! Let me formalize:

For shift s, the "energy" is E_s = sum_{i valid} h[i](1-h[i+s])
To minimize E_s, we want h[i] ≈ h[i+s] for all valid i.
A gradient step in this direction: h[i] += η * (h[i+s] - h[i]) / 2 for the "first" factor
And: h[i+s] += η * (h[i] - h[i+s]) / 2 for the "second" factor

Combined: h[i] += η/2 * (sum of neighbors - h[i] * degree)

This is like diffusion! We can use this as a pre-conditioner.

Actually, this is essentially what the gradient descent already does, since the gradient of E_s pushes h[i] and h[i+s] toward each other. So the gradient approach already captures this.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def conv_for_shift(h, s):
    n = len(h)
    i_min = max(0, -s)
    i_max = min(n - 1, n - 1 - s)
    if i_min > i_max:
        return 0.0
    return np.sum(h[i_min:i_max+1] * (1.0 - h[i_min+s:i_max+s+1]))

def conv_grad_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    i_min = max(0, -s)
    i_max = min(n - 1, n - 1 - s)
    if i_min <= i_max:
        grad[i_min:i_max+1] += (1.0 - h[i_min+s:i_max+s+1])
    p_min = max(0, s)
    p_max = min(n - 1, n - 1 + s)
    if p_min <= p_max:
        grad[p_min:p_max+1] -= h[p_min-s:p_max-s+1]
    return grad

def soft_max_val_grad(h, t, top_k=30):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    idx = np.argsort(conv)[-top_k:]
    shift_vals = (conv[idx] - m) / t
    shift_vals = np.clip(shift_vals, -60, 60)
    exps = np.exp(shift_vals)
    weights = exps / np.sum(exps)
    val = (m / t + np.log(np.sum(exps))) * t * dx
    grad = np.zeros(n)
    for w, j in zip(weights, idx):
        if w < 1e-14:
            continue
        s = int(j) - (n - 1)
        grad += w * conv_grad_shift(h, s)
    return val, grad * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def pgd_optimize(h, temps, max_iter_per_temp=15000, top_k=30, time_limit=float('inf')):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    lr = 0.01
    for t in temps:
        lr = min(lr, 0.05)
        for it in range(max_iter_per_temp):
            if time.time() - t_start > time_limit:
                return best_h
            _, grad = soft_max_val_grad(h, t, top_k)
            step = lr
            h_new = project_feasible(h - step * grad, n)
            c5_new = compute_c5(h_new)
            for _ in range(12):
                if c5_new <= compute_c5(h) + 1e-14:
                    break
                step *= 0.5
                h_new = project_feasible(h - step * grad, n)
                c5_new = compute_c5(h_new)
            if c5_new <= compute_c5(h) + 1e-14:
                h = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                lr = min(lr * 1.05, 0.2)
            else:
                lr *= 0.5
                if lr < 1e-10:
                    return best_h
    return best_h

def make_periodic(n, period, fill_frac, rng):
    h = np.zeros(n)
    for i in range(n):
        pos = (i % period) / period
        center = rng.uniform(0, 1)
        if abs(pos - center) < fill_frac / 2.0 or abs(pos - center - 1) < fill_frac / 2.0:
            h[i] = 1.0
    return project_feasible(h, n)
