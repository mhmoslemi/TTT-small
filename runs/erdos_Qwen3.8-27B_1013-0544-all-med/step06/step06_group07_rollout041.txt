        if abs(d) < 1e-13:
            break
        if d > 0:
            s = 1.0 - x
            m = s > 1e-15
            if m.any():
                x[m] += d * s[m] / s[m].sum()
        else:
            m = x > 1e-15
            if m.any():
                x[m] += d * x[m] / x[m].sum()
            else:
                break
        np.clip(x, 0.0, 1.0, out=x)
    d = target - x.sum()
    if abs(d) > 1e-12:
        order = np.argsort(1.0 - x)[::-1] if d > 0 else np.argsort(x)[::-1]
        for idx in order:
            if abs(d) < 1e-15:
                break
            if d > 0:
                a = min(d, 1.0 - x[idx])
                x[idx] += a
                d -= a
            else:
                a = max(d, -x[idx])
                x[idx] += a
                d -= a
    return x


def subgradient(h, dx, active_lags):
    n = h.size
    G = np.zeros(n)
    for lag in active_lags:
        lag = int(lag)
        lo1, hi1 = max(0, lag), min(n-1, lag+n-1)
        if lo1 <= hi1:
            G[lo1:hi1+1] += 1.0 - h[lo1-lag:hi1-lag+1]
        lo2, hi2 = max(0, -lag), min(n-1, n-1-lag)
        if lo2 <= hi2:
            G[lo2:hi2+1] -= h[lo2+lag:hi2+lag+1]
    G *= dx / len(active_lags)
    return G


def binary_local_search(h0, tlimit, seed=42):
    h = h0.copy().astype(float)
    n = h.size
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    c, cvec = compute_c5(h)
    best_h, best_c = h.copy(), c
    deadline = time.time() + tlimit
    no_imp = 0
    temp = 0.004
    
    while time.time() < deadline - 0.2:
        tol = max(1e-10, 2e-4 * c)
        active = np.where(cvec >= c - tol)[0] - (n - 1)
        G = subgradient(h, dx, active)
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) < 1 or len(zeros) < 1:
            break
        
        # Try top-K swaps
        i_top = ones[np.argsort(G[ones])[::-1][:8]]
        j_bot = zeros[np.argsort(G[zeros])[:8]]
        
        best_delta, best_pair, best_c_new, best_cv_new = 0, None, c, cvec
        for ii in i_top:
            for jj in j_bot:
                if ii == jj: continue
                h[ii], h[jj] = 0.0, 1.0
                nc, ncv = compute_c5(h)
                delta = nc - c
                if delta < best_delta:
                    best_delta, best_pair = delta, (ii, jj)
                h[ii], h[jj] = 1.0, 0.0
        
        if best_pair is not None:
            ii, jj = best_pair
            h[ii], h[jj] = 0.0, 1.0
            c, cvec = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
                no_imp = 0
            else:
                no_imp += 1
            # With small probability, keep the non-improving move (diversity)
            if no_imp > 0 and rng.rand() < 0.02:
                pass  # keep it
            else:
                if best_delta >= 0:
                    h[ii], h[jj] = 1.0, 0.0
                    c, cvec = compute_c5(h)
        else:
            no_imp += 1
            # Random move
            if no_imp > 5:
                ii, jj = rng.choice(ones), rng.choice(zeros)
                h[ii], h[jj] = 0.0, 1.0
                nc, ncv = compute_c5(h)
                if nc < best_c:
                    best_c, best_h = nc, h.copy()
                    c, cvec = nc, ncv
                    no_imp = 0
                elif rng.rand() < np.exp(-(nc - c) / max(temp, 1e-7)):
                    c, cvec = nc, ncv
                    no_imp += 1
                else:
                    h[ii], h[jj] = 1.0, 0.0
                    no_imp += 1
                temp = max(1e-5, temp * 0.995)
        
        if no_imp > 80:
            h = best_h.copy()
            c, cvec = compute_c5(h)
            no_imp = 0
            temp = 0.003
    
    return best_h, best_c


def round_binary(h):
    n = h.size
    order = np.argsort(h)[::-1]
    x = np.zeros(n)
    x[order[:n//2]] = 1.0
    return x
