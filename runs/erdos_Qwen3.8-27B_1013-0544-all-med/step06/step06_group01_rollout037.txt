def _compute_subgradient(h, n, active_lags):
    dx = 2.0 / n
    S = np.zeros(n)
    for lag in active_lags:
        offset = lag - (n - 1)
        # First term: (1 - h[j + offset]) for valid j
        # j + offset in [0, n-1] => j in [-offset, n-1-offset]
        j_min = max(0, -offset)
        j_max = min(n-1, n-1-offset)
        if j_min <= j_max:
            S[j_min:j_max+1] += (1.0 - h[j_min+offset : j_max+offset+1])

        if j_min2 <= j_max2:

For convex optimization with subgradient g_i at step i, the standard step size is:
- lr_i = c / sqrt(i) (diminishing)
- Or Polyak: lr_i = (f(x_i) - f*) / ||g_i||²

Actually, let me try a different approach. Since n is small (95), I can afford to be more aggressive.

**Key idea: Use the fact that the objective is convex in each coordinate**

For a fixed h with only h[i] varying, the objective f(h) = max_k c[k] is a convex PL function of h[i]. Its minimum can be found exactly.

Here's the algorithm:
1. For coordinate i:
   a. Compute c[k] for all k (current values)
   b. Compute a[k] = ∂c[k]/∂h[i] for all k
   c. The function t → max_k (c[k] + a[k]*(t - h[i])) is convex PL
   d. Find t* = argmin_{t ∈ [0,1]} max_k (c[k] + a[k]*(t - h[i]))
   e. Set h[i] = t*
   f. Adjust sum constraint

For step (d): the minimum of max_k (a[k]*t + c[k] - a[k]*h[i]) over t ∈ [0,1].
Let b[k] = c[k] - a[k]*h[i]. Then we want min_{t∈[0,1]} max_k (a[k]*t + b[k]).

This is a 1D convex PL optimization. The minimum occurs at:
- t = 0
- t = 1
- Or at a point where the upper envelope changes (intersection of two lines that form the max)

The number of "breakpoints" is at most n (one for each pair of active lines). We can find them all and check.

But this is O(n²) per coordinate (to find all breakpoints), and O(n) coordinates, so O(n³) per pass. For n=95, that's ~857K operations per pass. Very fast!

But there's a complication: the sum constraint. If I change h[i], the sum changes. I need to compensate.

One approach: use a penalty for the sum constraint and ignore it in the coordinate update. Then periodically re-project.

Another approach: when updating h[i], also update h[j] (some other coordinate) to maintain the sum. This makes it a 2D optimization, which is still tractable.

Let me go with the penalty approach:

Objective: max_k c[k] + β * (sum(h) - n/2)²

For coordinate i:
- The max part: min_t max_k (a[k]*t + b[k]) as before
- The penalty part: β * (sum - n/2 + t - h[i])² = β * (S + t - h[i])² where S = sum(h) - n/2

So the full 1D objective for coordinate i is:
f_i(t) = max_k (a[k]*t + b[k]) + β * (S + t - h[i])²

This is convex PL + quadratic. Its minimum can be found by:
1. Finding the breakpoints of the PL part
2. On each linear piece, the function is linear + quadratic = quadratic
3. Finding the minimum of each quadratic on its interval
4. Taking the overall minimum

The PL part has at most n+1 pieces (bounded by the number of lines in the max).

2. If s[i] > 0, try to decrease h[i]; if s[i] < 0, try to increase
3. Do a line search: try several values and pick the best

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _project(p, n):
    """Project onto {x in [0,1]^n : sum(x) = n/2}"""
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(p - mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(p - lam, 0.0, 1.0)

def _subgradient(h, active_lags):
    n = len(h)
    dx = 2.0 / n
    S = np.zeros(n)
    for k in active_lags:
        offset = k - (n - 1)
        jlo = max(0, -offset)
        jhi = min(n - 1, n - 1 - offset)
        if jlo <= jhi:
            S[jlo:jhi+1] += 1.0 - h[jlo+offset : jhi+offset+1]
        jlo2 = max(0, offset)
        jhi2 = min(n - 1, n - 1 + offset)
        if jlo2 <= jhi2:
            S[jlo2:jhi2+1] -= h[jlo2-offset : jhi2-offset+1]
    return S * dx / len(active_lags)

def _sgd_optimize(h, n, max_iter, check_time=None, time_limit=0.95):
    """Subgradient optimization with adaptive step size."""
    h = h.copy()
    best_c5 = _c5_max(h)
    best_h = h.copy()
    
    # Momentum
    m = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t = 0
    
    for it in range(max_iter):
        if check_time is not None and check_time() > time_limit:
            break
        C = _c5_array(h)
        Cmax = np.max(C)
        if Cmax < best_c5 - 1e-10:
            best_c5 = Cmax
            best_h = h.copy()
        
        # Active set: shifts within 1% of max
        eps = max(1e-4, 0.005 * Cmax)
        active = np.where(C >= Cmax - eps)[0]
        
        # Subgradient
        S = _subgradient(h, active)
        
        t += 1
        m = beta1 * m + (1 - beta1) * S
        m_hat = m / (1 - beta1**t)
        lr = 0.3 / (np.sqrt(t + 1) + 1.0)
        
        h = h - lr * m_hat
        h = _project(h, n)
    
    return best_h, best_c5

def _coord_descent(h, n, passes=10):
    """Coordinate descent: for each i, find best h[i] in [0,1]."""
    h = h.copy()
    best_c5 = _c5_max(h)
    
    for _ in range(passes):
        improved = False
        for i in range(n):
            # Try several values for h[i]
            best_val = h[i]
            best_c = best_c5
            # Candidate values
            candidates = [0.0, 1.0, h[i] - 0.1, h[i] + 0.1, h[i] - 0.05, h[i] + 0.05]
            candidates = [c for c in candidates if 0 <= c <= 1]
            for cv in candidates:
                if abs(cv - h[i]) < 1e-6:

                h[i] = cv
                h = _project(h, n)
                c = _c5_max(h)
                if c < best_c - 1e-10:
                    best_c = c
                    best_val = cv
                    improved = True
                h[i] = cv  # keep for next candidate (project might change others)
            h[i] = best_val
            h = _project(h, n)
            if best_c < best_c5:
                best_c5 = best_c
        if not improved:
            break
    
    return h, best_c5

def _binary_round(h, n):
    """Round to binary and fix sum."""
    h_bin = (h > 0.5).astype(float)
    # Fix sum
    target = n // 2
    s = int(np.sum(h_bin))
    if s > target:

        indices = np.argsort(h_bin)  # zeros first
        # Actually, we want to flip ones to zeros
        ones = np.where(h_bin == 1)[0]
        # Flip the ones that are "least important" (lowest h[i] was)
        order = np.argsort(h[ones])  # originally closest to 0

        for idx in order[:target - s]:

Actually, for a convex PL function of one variable, the minimum is at a breakpoint. The breakpoints occur where two of the line segments cross. For the k-th correlation value:
c_k = a_k * h[i] + b_k (linear in h[i])

The minimum of max_k c_k is at the point where the "upper envelope" is minimized, which is at a crossing of two lines (or at 0 or 1).

Let me compute the breakpoints more carefully:

For coordinate i, given all other coordinates fixed:
- a[k] = ∂c[k]/∂h[i] (the slope)
- b[k] = c[k] - a[k] * h[i] (the intercept, using current h[i])

The minimum of max_k (a[k]*t + b[k]) over t ∈ [0,1] can be found by:
1. For each pair (k1, k2) with a[k1] ≠ a[k2], compute the crossing point t* = (b[k2] - b[k1]) / (a[k1] - a[k2])
2. If 0 ≤ t* ≤ 1, evaluate max_j (a[j]*t* + b[j]) at t*

import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _project(p, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(p - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)

def _subgradient(h, active_lags):
    n = len(h)
    dx = 2.0 / n
    S = np.zeros(n)
    for k in active_lags:
        offset = k - (n - 1)
        jlo = max(0, -offset)
        jhi = min(n - 1, n - 1 - offset)
        if jlo <= jhi:
            S[jlo:jhi+1] += 1.0 - h[jlo+offset : jhi+offset+1]
        jlo2 = max(0, offset)
        jhi2 = min(n - 1, n - 1 + offset)
        if jlo2 <= jhi2:
            S[jlo2:jhi2+1] -= h[jlo2-offset : jhi2-offset+1]
    return S * dx / len(active_lags)

def _sgd(h, n, t0, deadline, lr_scale=1.0, momentum=0.9):
    h = h.copy()
    best_c5 = _c5_max(h)
    best_h = h.copy()
    m = np.zeros(n)
    t = 0
    for it in range(100000):
        if time.time() > deadline:
            break
        C = _c5_array(h)
        Cmax = np.max(C)
        if Cmax < best_c5 - 1e-11:
            best_c5 = Cmax
            best_h = h.copy()
        eps = max(1e-4, 0.01 * Cmax)
        active = np.where(C >= Cmax - eps)[0]
        S = _subgradient(h, active)
        t += 1
        m = momentum * m + (1.0 - momentum) * S
        lr = lr_scale / (np.sqrt(t) + 1.0)
        h = h - lr * m
        h = _project(h, n)
    return best_h, best_c5

def _coord_desc(h, n, t0, deadline):
    h = h.copy()
    best_c5 = _c5_max(h)
    # Precompute slopes for speed
    while True:
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            # Compute a[k] = slope of c[k] w.r.t. h[i]
            # a[k] = dx * [(1-h[i+k-(n-1)]) if valid - h[i-k+(n-1)] if valid]
            C = _c5_array(h)
            n_c = len(C)
            a = np.zeros(n_c)
            for k in range(n_c):
                offset = k - (n - 1)
                i1 = i + offset
                i2 = i - offset
                val = 0.0
                if 0 <= i1 < n:
                    val += (1.0 - h[i1])
                if 0 <= i2 < n:
                    val -= h[i2]
                a[k] = val * (2.0 / n)
            
            b = C - a * h[i]
            # Minimize max_k (a[k]*t + b[k]) for t in [0,1]
            # Use bisection on subdifferential
            lo_t, hi_t = 0.0, 1.0
            best_t = h[i]
            best_f = np.max(a * h[i] + b)
            for _ in range(50):
                mid = (lo_t + hi_t) / 2.0
                vals = a * mid + b
                v_max = np.max(vals)
                active_k = np.where(vals >= v_max - 1e-12)[0]
                min_s = np.min(a[active_k])
                max_s = np.max(a[active_k])
                if min_s <= 0.0 <= max_s:
                    lo_t = mid
                    hi_t = mid + 0  # converged
                    break
                elif max_s < 0:
                    lo_t = mid
                else:
                    hi_t = mid
                if v_max < best_f:
                    best_f = v_max
                    best_t = mid
            # Evaluate at boundaries too
            for t_cand in [0.0, 1.0, best_t]:
                f_c = np.max(a * t_cand + b)
                if f_c < best_f:
                    best_f = f_c
                    best_t = t_cand
            
            h[i] = best_t
            h = _project(h, n)
            c_new = _c5_max(h)
            if c_new < best_c5 - 1e-10:
                best_c5 = c_new
                improved = True
        if not improved:
            break
    return h, best_c5

    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(p - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(p - (lo + hi) / 2.0, 0.0, 1.0)

def _subgradient(h, active_lags):
    n = len(h)
    dx = 2.0 / n
    S = np.zeros(n)
    for k in active_lags:
        offset = k - (n - 1)
        jlo = max(0, -offset)
        jhi = min(n - 1, n - 1 - offset)
        if jlo <= jhi:
            S[jlo:jhi + 1] += 1.0 - h[jlo + offset:jhi + offset + 1]
        jlo2 = max(0, offset)
        jhi2 = min(n - 1, n - 1 + offset)
        if jlo2 <= jhi2:
            S[jlo2:jhi2 + 1] -= h[jlo2 - offset:jhi2 - offset + 1]
    return S * dx / len(active_lags)

def _sgd(h, n, deadline, lr_scale=1.0, momentum=0.9):
    h = h.copy()
    best_c5 = _c5_max(h)
    best_h = h.copy()
    m = np.zeros(n)
    t = 0
    for it in range(200000):
        if time.time() > deadline:
            break
        C = _c5_array(h)
        Cmax = np.max(C)
        if Cmax < best_c5 - 1e-11:
            best_c5 = Cmax
            best_h = h.copy()
        eps = max(1e-4, 0.01 * Cmax)
        active = np.where(C >= Cmax - eps)[0]
        S = _subgradient(h, active)
        t += 1
        m = momentum * m + (1.0 - momentum) * S
        lr = lr_scale / (np.sqrt(t) + 1.0)
        h = h - lr * m
        h = _project(h, n)
    return best_h, best_c5

def _coord_desc(h, n, deadline):
    h = h.copy()
    best_c5 = _c5_max(h)
    for sweep in range(200):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            C = _c5_array(h)
            nc = len(C)
            dx = 2.0 / n
            # Compute slope of each c[k] w.r.t. h[i]
            a = np.zeros(nc)
            for k in range(nc):
                offset = k - (n - 1)
                i1 = i + offset
                i2 = i - offset
                val = 0.0
                if 0 <= i1 < n:
                    val += (1.0 - h[i1])
                if 0 <= i2 < n:
                    val -= h[i2]
                a[k] = val * dx
            b = C - a * h[i]
            # Bisection to find min of max_k(a[k]*t + b[k]) on [0,1]
            lo_t, hi_t = 0.0, 1.0
            best_t = h[i]
            best_f = np.max(a * h[i] + b)
            for _ in range(50):
                mid = (lo_t + hi_t) / 2.0
                vals = a * mid + b
                v_max = np.max(vals)
                idx = np.where(vals >= v_max - 1e-12)[0]
                min_s = a[idx.min()]
                max_s = a[idx.max()]
                if min_s <= 0.0 <= max_s:
                    lo_t = mid
                    hi_t = mid
                    break
                elif max_s < 0:
                    lo_t = mid
                else:
                    hi_t = mid
                if v_max < best_f:
                    best_f = v_max
                    best_t = mid
            for tc in [0.0, 1.0, best_t]:
                fc = np.max(a * tc + b)
                if fc < best_f:
                    best_f = fc
                    best_t = tc
            h[i] = best_t
            h = _project(h, n)
            c_new = _c5_max(h)
            if c_new < best_c5 - 1e-10:
                best_c5 = c_new
                improved = True
        if not improved:
            break
    return h, best_c5

def _binary_local_search(h, n, deadline, max_passes=500):
    """Local search on binary solutions: swap 1 and 0 to reduce max correlation."""
    h = h.copy()
    best_c5 = _c5_max(h)
    for _ in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        for i in ones:
            if time.time() > deadline:
                break
            for j in zeros:
                h[i] = 0.0
                h[j] = 1.0
                c = _c5_max(h)
                if c < best_c5 - 1e-12:
                    best_c5 = c
