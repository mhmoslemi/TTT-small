import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def corr_vector(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def sigmoid_fn(a):
    return 1.0 / (1.0 + np.exp(-np.clip(a, -50, 50)))

def sigmoid_obj(a, t, lam):
    n = len(a)
    h = sigmoid_fn(a)
    dx = 2.0 / n
    conv = corr_vector(h)
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    sm = t * (m / t + np.log(np.sum(exps))) * dx
    s = np.sum(h) - n / 2.0
    return sm + lam * s * s

def sigmoid_grad(a, t, lam):
    n = len(a)
    h = sigmoid_fn(a)
    hd = h * (1.0 - h)
    dx = 2.0 / n
    conv = corr_vector(h)
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad_h = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-13:
            continue
        shift = idx - (n - 1)
        grad_h += w * _conv_grad_single(h, shift)
    grad_h *= dx
    s = np.sum(h) - n / 2.0
    return grad_h * hd + 2.0 * lam * s * hd

def nesterov_pgd(h_init, max_iter, t_start, t_end, lr_init=0.003, momentum=0.9):
    n = len(h_init)
    h = h_init.copy()
    h_prev = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init

    for it in range(max_iter):
        frac = (it + 1) / max_iter
        t = t_start * (t_end / t_start) ** frac

        y = h + momentum * (h - h_prev)

        conv = corr_vector(y)
        m = np.max(conv)
        exps = np.exp((conv - m) / t)
        weights = exps / np.sum(exps)

        grad = np.zeros(n)
        for idx in range(len(conv)):
            w = weights[idx]
            if w < 1e-13:
                continue
            shift = idx - (n - 1)
            grad += w * _conv_grad_single(y, shift)
        grad *= 2.0 / n

        h_new = project_feasible(y - lr * grad, n)

        if compute_c5(h_new) > compute_c5(y) + 1e-13:
            lr *= 0.5
            if lr < 1e-13:
                lr = lr_init * 0.05
            h_new = project_feasible(y - lr * grad, n)
        else:
            lr = min(lr * 1.04, 0.04)

        h_prev = h
        h = h_new

        c5_h = compute_c5(h)
        if c5_h < best_c5 - 1e-14:
            best_c5 = c5_h
            best_h = h.copy()

    return best_h, best_c5

def coarse_to_fine(h_coarse, n_fine):
    nc = len(h_coarse)
    xo = np.linspace(0, 2.0, nc, endpoint=False)
    xn = np.linspace(0, 2.0, n_fine, endpoint=False)
    hi = np.interp(xn, xo, h_coarse)
    return project_feasible(hi, n_fine)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(45, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Sigmoid parameterization with L-BFGS-B (smooth landscape)
    if remaining() > 100:
        h_clip = np.clip(best_h, 1e-6, 1 - 1e-6)
        a0 = np.log(h_clip / (1.0 - h_clip))
        a0 = np.clip(a0, -12, 12)
        for t_val, lam in [(0.04, 30.0), (0.015, 100.0), (0.005, 400.0), (0.002, 1000.0)]:
            if remaining() < 25:
                break
            try:
                res = minimize(sigmoid_obj, a0, args=(t_val, lam),
                              jac=sigmoid_grad, method='L-BFGS-B',
                              maxiter=3000, ftol=1e-16, gtol=1e-11)
                ht = sigmoid_fn(res.x)
                ht = project_feasible(ht, n0)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t
                a0 = np.clip(res.x, -15, 15)
            except Exception:
                pass

    # Phase 2: Nesterov PGD with geometric annealing
    if remaining() > 80:
        n_iter = min(15000, int(remaining() * 80))
        ht, c5t = nesterov_pgd(best_h, n_iter, 0.04, 0.001, lr_init=0.003)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: Coarse-to-fine multi-resolution
    for n_coarse in [30, 40, 60]:
        if remaining() < 100:
            break
        nc = len(best_h)
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xc = np.linspace(0, 2.0, n_coarse, endpoint=False)
        hc = np.interp(xc, xo, best_h)
        hc = project_feasible(hc, n_coarse)
        for _ in range(3):
            hc, _ = nesterov_pgd(hc, 3000, 0.03, 0.005, lr_init=0.01)
        hf = coarse_to_fine(hc, n0)
        if remaining() < 30:
            break
        ht, c5t = nesterov_pgd(hf, 8000, 0.02, 0.001, lr_init=0.003)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Diverse restarts
    x = np.linspace(0, 2, n0, endpoint=False)

    # Binary-like initializations
    for _ in range(3):
        if remaining() < 20:
            break
        hr = (rng.uniform(0, 1, n0) > 0.5).astype(float)
        hr = project_feasible(hr, n0)
        ht, c5t = nesterov_pgd(hr, 5000, 0.03, 0.001, lr_init=0.006)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Structured patterns
    for freq, amp in [(2, 0.4), (3, 0.45), (5, 0.35), (1, 0.5)]:
        if remaining() < 20:
            break
        hs = 0.5 + amp * np.sin(np.pi * freq * x + rng.uniform(0, 2 * np.pi))
        hs = project_feasible(hs, n0)
        ht, c5t = nesterov_pgd(hs, 4000, 0.03, 0.001, lr_init=0.006)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Perturbed parent
    for frac in [0.15, 0.35]:
        if remaining() < 20:
            break
        hp = best_h.copy()
        idxs = rng.choice(n0, int(n0 * frac), replace=False)
        hp[idxs] = rng.uniform(0, 1, len(idxs))
        hp = project_feasible(hp, n0)
        ht, c5t = nesterov_pgd(hp, 6000, 0.03, 0.001, lr_init=0.005)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Try different target resolutions
    for n_try in [200, 256, 400, 512]:
        if remaining() < 80:
            break
        if n_try == n0:
            continue
        hi = coarse_to_fine(best_h, n_try)
        ht, c5t = nesterov_pgd(hi, 6000, 0.03, 0.001, lr_init=0.003)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Final high-precision polish
    if remaining() > 30:
        n_polish = min(25000, int(remaining() * 100))
        ht, c5t = nesterov_pgd(best_h, n_polish, 0.001, 0.0003, lr_init=0.0008, momentum=0.85)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
