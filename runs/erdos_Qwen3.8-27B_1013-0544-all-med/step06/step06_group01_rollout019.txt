    return np.correlate(h, 1.0 - h, mode='full') * dx


def _c5_max(h):
    return float(np.max(_c5_array(h)))


def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                idx = np.argmax(1.0 - h)
                h[idx] = min(1.0, h[idx] + diff2)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + diff2)
    return h


def _vec_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    nc = 2 * n - 1
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = es.sum()
    w = es / se
    km = np.arange(nc)
    k0 = km[:n]
    k1 = km[n:]
    t0 = np.clip(k0 - np.arange(n)[:, None], 0, n - 1).astype(int)
    t1 = np.clip(np.arange(n)[:, None] + k1, 0, n - 1).astype(int)
    ind0 = (np.arange(n)[:, None] <= k0[None, :])
    ind1 = (np.arange(n)[:, None] >= k1[None, :] - n + 1)
    A = (1.0 - h[t0]) * ind0
    B = h[t1] * ind1
    grad = (A @ w[:n] - B @ w[n:]) * dx
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return grad


def _vec_obj(h, alpha, beta):
    n = len(h)
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    f = (m + np.log(es.sum())) / alpha
    f += beta * (np.sum(h) - 0.5 * n) ** 2
    return f


def _optimize(h, alpha, beta, maxiter=300):
    n = len(h)
    res = minimize(_vec_obj, h, args=(alpha, beta), jac=_vec_grad,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-14})
    if res.x is not None:
        return _fix(res.x, n)
    return h


def _thue_morse(n):
    t = np.zeros(n, dtype=np.float64)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t


def _make_initializations(n, seed):
    rng = np.random.RandomState(seed)
    inits = []
    inits.append(_fix(_thue_morse(n), n))
    for s in range(6):
        h = _thue_morse(n).copy()
        for _ in range(50):
            i, j = rng.randint(n), rng.randint(n)
            if h[i] != h[j]:
                h[i], h[j] = h[j], h[i]
        inits.append(_fix(h, n))
    for s in range(6):
        x = np.linspace(0, 2, n, endpoint=False)
        f1, f2 = rng.uniform(0.5, 4.0), rng.uniform(0.5, 4.0)
        p1, p2 = rng.uniform(0, 6.28), rng.uniform(0, 6.28)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * f1 * x + p1) + 0.2 * np.sin(2 * np.pi * f2 * x + p2)
        inits.append(_fix(h, n))
    for s in range(4):
        h = rng.binomial(1, 0.5, n).astype(float)
        inits.append(_fix(h, n))
    return inits


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    hb = _fix(initial_h_values.astype(float).copy(), n0)
    cb = _c5_max(hb)
    nb = n0

    # Phase 1: quick L-BFGS on parent
    for alpha in [20, 50, 100]:
        if _el() > budget_s * 0.10:
            break
        hn = _optimize(hb, alpha, max(1.0, 0.1 * alpha), maxiter=300)
        cn = _c5_max(hn)
        if cn < cb - 1e-10:
            cb, hb, nb = cn, hn, n0

    # Phase 2: diverse initializations across multiple n
    n_candidates = sorted(set([n0, max(50, n0 - 10), n0 + 10, n0 + 20, n0 + 40,
                               80, 100, 120, 160, 200, 250]))
    n_candidates = [n for n in n_candidates if 50 <= n <= 300]

    best_list = [(cb, hb.copy(), nb)]

    for n in n_candidates:
        if _el() > budget_s * 0.55:
            break
        inits = _make_initializations(n, seed)
        for init in inits:
            if _el() > budget_s * 0.55:
                break
            for alpha in [30, 80]:
                if _el() > budget_s * 0.55:
                    break
                hn = _optimize(init.copy(), alpha, max(1.0, 0.1 * alpha), maxiter=200)
                cn = _c5_max(hn)
                if cn < cb - 1e-10:
                    cb, hb, nb = cn, hn, n
                best_list.append((cn, hn.copy(), n))

    best_list.sort(key=lambda x: x[0])
    cb, hb, nb = best_list[0]

    # Phase 3: high-alpha polish on top candidates
    for i in range(min(4, len(best_list))):
        if _el() > budget_s * 0.82:
            break
        _, hc, nc = best_list[i]
        for alpha in [200, 500]:
            if _el() > budget_s * 0.82:
                break
            hn = _optimize(hc.copy(), alpha, max(5.0, 0.1 * alpha), maxiter=500)
            cn = _c5_max(hn)
            if cn < cb - 1e-10:
                cb, hb, nb = cn, hn, nc

    # Phase 4: block coordinate descent
    if _el() < budget_s * 0.93:
        n = nb
        h = hb.copy()
        dx = 2.0 / n
        B = 10
        for rnd in range(80):
            if _el() > budget_s * 0.93:
                break
            improved = False
            for start in range(0, n, B):
                end = min(start + B, n)
                if _el() > budget_s * 0.93:
                    break
                sub = h[start:end].copy()
                res = minimize(_vec_obj, sub, args=(500, 10.0),
                               jac=lambda x, *a: _vec_grad(
                                   np.concatenate([h[:start], x, h[end:]]), 500, 10.0)[start:end],
                               method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * (end - start),
                               options={'maxiter': 50, 'ftol': 1e-14})
                if res.x is not None:
                    h2 = h.copy()
                    h2[start:end] = res.x
                    h2 = _fix(h2, n)
                    c2 = _c5_max(h2)
                    if c2 < _c5_max(h) - 1e-12:
                        h = h2
                        improved = True
            if not improved:
                break
        if _c5_max(h) < cb:
            cb, hb, nb = _c5_max(h), h, n

    # Phase 5: local swap refinement
    if _el() < budget_s * 0.985:
        h = hb.copy()
        n = nb
        best_c = cb
        for _ in range(300):
            if _el() > budget_s * 0.985:
                break
            b = _c5_array(h)
            k_active = np.where(b >= best_c - 0.003)[0]
            found = False
            for ka in k_active:
                k_signed = int(ka) - (n - 1)
                if k_signed == 0:
                    continue
                same = set()
                for ca in k_active:
                    ks = int(ca) - (n - 1)
                    if ks == -k_signed:
                        same.add(ca)
                same.discard(ka)
                for i in range(n):
                    if k_signed > 0 and i + k_signed >= n:
                        continue
                    if k_signed < 0 and i + k_signed < 0:
                        continue
                    j = i + k_signed
                    r2 = 1.0 - h[i] - h[j]
                    matched = all(abs(int(ca) - (n - 1)) == k_signed for ca in same)
                    if matched and r2 > 1e-9 and h[i] < 1.0 and h[j] < 1.0:
                        delta = min(h[i], h[j], (1.0 - h[i]) * 0.5, (1.0 - h[j]) * 0.5, 0.1)
                        if delta > 1e-6:
                            h[i] = min(1.0, h[i] + delta)
                            h[j] = min(1.0, h[j] + delta)
                            h = _fix(h, n)
                            nc = _c5_max(h)
                            if nc < best_c:
                                best_c = nc
                                found = True
                    elif not matched and r2 < -1e-9 and h[i] > 0.0 and h[j] > 0.0:
                        delta = min(h[i], h[j], 0.05)
                        if delta > 1e-6:
                            h[i] = max(0.0, h[i] + delta)
                            h[j] = max(0.0, h[j] + delta)
                            h = _fix(h, n)
                            nc = _c5_max(h)
                            if nc < best_c:
                                best_c = nc
                                found = True
            if not found:
                break
        if best_c < cb:
            cb, hb = best_c, h

    return (hb, cb, nb)
