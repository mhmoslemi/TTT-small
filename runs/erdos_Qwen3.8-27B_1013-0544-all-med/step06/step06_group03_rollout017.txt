    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def nesterov_pgd(h_init, n_iter, t0, t_end, lr0, time_limit=None, t0_time=0):
    """Nesterov PGD with exponential temperature annealing."""
    n = len(h_init)
    h = h_init.copy()
    y = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for k in range(n_iter):
        if time_limit is not None and time.time() - t0_time > time_limit:
            break
        
        # Temperature schedule: exponential decay
        frac = k / max(n_iter - 1, 1)
        t = t0 * (t_end / t0) ** frac
        
        # Learning rate: polynomial decay
        lr = lr0 * (1.0 - 0.5 * frac)
        
        # Nesterov momentum
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (k + 1) ** 2)) / (2.0 * (k + 1))
        
        grad = soft_max_grad(y, t)
        y_old = y.copy()
        h_new = project_feasible(y - lr * grad, n)
        y = h_new + beta * (h_new - y_old)
        y = np.clip(y, -0.1, 1.1)  # Keep y in reasonable range
        h = h_new
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def upsample(h, n_new):
    """Upsample h to n_new points using linear interpolation."""
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

def lbfgs_refine(h_init, n, t_val, max_iter=500, time_limit=None, t0_time=0):
    """L-BFGS-B refinement with soft-max objective."""
    def obj(h):
        if time_limit is not None and time.time() - t0_time > time_limit:
            raise TimeoutError()
        conv = np.correlate(h, 1.0 - h, mode="full")
        m = np.max(conv)
        shifted = np.clip((conv - m) / t_val, -30, 30)
        return t_val * (m / t_val + np.log(np.sum(np.exp(shifted)))) * (2.0 / n)
    
    def grad(h):
        if time_limit is not None and time.time() - t0_time > time_limit:
            raise TimeoutError()
        return soft_max_grad(h, t_val)
    
    try:
        res = minimize(obj, h_init, jac=grad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
        return project_feasible(res.x, n)
    except Exception:
        return h_init
