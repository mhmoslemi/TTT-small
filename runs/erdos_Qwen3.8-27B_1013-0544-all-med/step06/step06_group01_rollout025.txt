import time

    return np.correlate(h, 1.0 - h, mode='full') * dx

    d = 0.5 * n - np.sum(h)

                    room = 1.0 - h[idx[k]]
                    add = min(d2, room)
                    h[idx[k]] += add
                    d2 -= add
                else:
                    rem = h[idx[k]]
                    sub = min(-d2, rem)
                    h[idx[k]] -= sub
                    d2 += sub
                if abs(d2) < 1e-12:
                    break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.argsort(-(1.0 - h))
            else:
                idx = np.argsort(h)
            for k in range(n):
                if d2 > 0:
                    room = 1.0 - h[idx[k]]
                    add = min(d2, room)
                    h[idx[k]] += add
                    d2 -= add
                else:
                    rem = h[idx[k]]
                    sub = min(-d2, rem)
                    h[idx[k]] -= sub
                    d2 += sub
                if abs(d2) < 1e-12:
                    break
    return h

def _soft_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    w = es / np.sum(es)
    j_arr = np.arange(nc)
    ell = j_arr - (n - 1)
    grad = np.empty(n)
    for i in range(n):
        a = i - ell
        b = i + ell
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(vb, 1.0 - h[np.clip(b, 0, n - 1)], 0.0)
        tb = np.where(va, h[np.clip(a, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return grad

def _adam(h0, alpha, beta, lr0=0.01, maxiter=3000, decay=0.999):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    for t in range(1, maxiter + 1):
        g = _soft_grad(h, alpha, beta)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h = h - lr * mh / (np.sqrt(vh) + eps)
        h = _fix(h, n)
        lr *= decay
    return h

def _lgbfgs(h0, alpha, beta, maxiter=500):
    n = len(h0)
    def obj(h):
        C = _c5_array(h)
        sh = alpha * C
        m = np.max(sh)
        f = m / alpha + np.log(np.sum(np.exp(sh - m))) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
        return f, _soft_grad(h, alpha, beta)
    res = minimize(obj, h0, jac=True, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None:
        return _fix(res.x, n)
    return h0

def _binary_local_search(h_start, n, max_iters=50000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h_start.copy()
    best_c5 = _c5_max(h)
    for _ in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] <= h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5 = _c5_max(h)
        if c5 > best_c5 + 1e-12:
            h[i], h[j] = h[j], h[i]
        else:
            best_c5 = c5
    return h, best_c5

def _coord_descent(h, n, max_rounds=200, step=0.01):
    """Coordinate descent: try small changes to individual coordinates."""
    best_c5 = _c5_max(h)
    h = h.copy()
    for rnd in range(max_rounds):
        improved = False
        for i in range(n):
            for delta in [step, -step, step*2, -step*2]:
                new_val = h[i] + delta
                if new_val < 0 or new_val > 1:
                    continue
                # Need to adjust another coordinate to maintain sum
                # Find j with most room in the needed direction
                h_try = h.copy()
                h_try[i] = new_val
                d = 0.5 * n - np.sum(h_try)
                if abs(d) < 1e-10:
                    continue
                if d > 0:
                    j = np.argmax(1.0 - h_try)
                else:
                    j = np.argmax(h_try)
                if j == i:
                    continue
                h_try[j] = np.clip(h_try[j] + d, 0, 1)
                d2 = 0.5 * n - np.sum(h_try)
                if abs(d2) > 1e-6:
                    continue
                h_try = _fix(h_try, n)
                c5 = _c5_max(h_try)
                if c5 < best_c5 - 1e-10:
                    h = h_try
                    best_c5 = c5
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.argsort(-(1.0 - h))
            else:
                idx = np.argsort(h)
            for k in range(n):
                if d2 > 0:
                    room = 1.0 - h[idx[k]]
                    add = min(d2, room)
                    h[idx[k]] += add
                    d2 -= add
                else:
                    rem = h[idx[k]]
                    sub = min(-d2, rem)
                    h[idx[k]] -= sub
                    d2 += sub
                if abs(d2) < 1e-12:
                    break
    return h

def _soft_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    w = es / np.sum(es)
    j_arr = np.arange(nc)
    ell = j_arr - (n - 1)
    grad = np.empty(n)
    for i in range(n):
        a = i - ell
        b = i + ell
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(vb, 1.0 - h[np.clip(b, 0, n - 1)], 0.0)
        tb = np.where(va, h[np.clip(a, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return grad

def _adam(h0, alpha, beta, lr0=0.01, maxiter=3000, decay=0.9995):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = _c5_max(h)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    lr = lr0
    for t in range(1, maxiter + 1):
        g = _soft_grad(h, alpha, beta)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h_new = _fix(h_new, n)
        c5 = _c5_max(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        lr *= decay
    return best_h

def _lgbfgs(h0, alpha, beta, maxiter=400):
    n = len(h0)
    def obj(h):
        C = _c5_array(h)
        sh = alpha * C
        m = np.max(sh)
        f = m / alpha + np.log(np.sum(np.exp(sh - m))) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
        return f, _soft_grad(h, alpha, beta)
    res = minimize(obj, h0, jac=True, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None:
        return _fix(res.x, n)
    return h0

def _targeted_swap(h, n, max_iters=20000, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    best_c5 = _c5_max(h)
    C = _c5_array(h)
    for it in range(max_iters):
        k_star = int(np.argmax(C))
        lag = k_star - (n - 1)
        try_improved = False
        for _ in range(20):
            if lag >= 0:
                valid = list(range(n - lag))
            else:
                valid = list(range(-lag, n))
            if not valid:
                break
            i = valid[rng.randint(len(valid))]
            if lag >= 0:
                j = i + lag
            else:
                j = i + lag
            if j < 0 or j >= n or j == i:
                continue
            delta = rng.choice([0.02, 0.05, 0.1, 0.15])
            for (src, dst) in [(i, j), (j, i)]:
                if h[src] < delta or h[dst] > 1.0 - delta:
                    continue
                h_try = h.copy()
                h_try[src] -= delta
                h_try[dst] += delta
                h_try = _fix(h_try, n)
                c5 = _c5_max(h_try)
                if c5 < best_c5 - 1e-10:
                    h = h_try
                    best_c5 = c5
                    C = _c5_array(h)
                    try_improved = True
                    break
            if try_improved:
                break
        if not try_improved:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.choice([0.02, 0.05, 0.1])
            for (src, dst) in [(i, j), (j, i)]:
                if h[src] < delta or h[dst] > 1.0 - delta:
                    continue
                h_try = h.copy()
                h_try[src] -= delta
                h_try[dst] += delta
                h_try = _fix(h_try, n)
                c5 = _c5_max(h_try)
                if c5 < best_c5 - 1e-10:
                    h = h_try
                    best_c5 = c5
                    C = _c5_array(h)
                    break
    return h, best_c5

def _binary_project(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    if n % 2 == 1:
        h_bin[idx[n // 2]] = 0.5
    return h_bin

def _binary_ls(h_bin, n, max_iters=30000, rng=None):
    if rng is None:
        rng = np.random.RandomState(1)
    h = h_bin.copy()
    best_c5 = _c5_max(h)
    for _ in range(max_iters):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h[i] <= h[j]:
            continue
        h[i], h[j] = h[j], h[i]
        c5 = _c5_max(h)
        if c5 > best_c5 + 1e-12:
            h[i], h[j] = h[j], h[i]
        else:
            best_c5 = c5
    return h, best_c5
