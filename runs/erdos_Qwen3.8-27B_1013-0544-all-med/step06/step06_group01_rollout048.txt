        if abs(d2) > 1e-12:
            if d2 > 0:
                h[np.argmax(1.0 - h)] = min(1.0, h[np.argmax(1.0 - h)] + d2)
            else:
                h[np.argmax(h)] = max(0.0, h[np.argmax(h)] + d2)
    return h

def _slopes(h, n, i):
    dx = 2.0 / n
    lags = np.arange(2 * n - 1) - (n - 1)
    fwd = i + lags
    bwd = i - lags
    fv = (fwd >= 0) & (fwd < n)
    bv = (bwd >= 0) & (bwd < n)
    s = np.zeros(2 * n - 1)
    if fv.any():
        s[fv] += 1.0 - h[fwd[fv]]
    if bv.any():
        s[bv] -= h[bwd[bv]]
    return s * dx

def _bisection_1d(sl, ic):
    lo, hi = 0.0, 1.0
    for _ in range(35):
        mid = 0.5 * (lo + hi)
        m = np.argmax(sl * mid + ic)
        if sl[m] < -1e-14:
            lo = mid
        elif sl[m] > 1e-14:
            hi = mid
        else:
            break
    return 0.5 * (lo + hi)

def _coord_descent(h0, n, max_sweeps=500, seed=42):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    C = _c5_arr(h)
    best_c = C.max()
    for sweep in range(max_sweeps):
        order = rng.permutation(n)
        total_change = 0.0
        for i in order:
            sl = _slopes(h, n, i)
            ic = C - sl * h[i]
            t = _bisection_1d(sl, ic)
            delta = t - h[i]
            if abs(delta) > 1e-12:
                h[i] = t
                C += sl * delta
                total_change += abs(delta)
        h = _fix(h, n)
        C = _c5_arr(h)
        c = C.max()
        if c < best_c:
            best_c = c
        if total_change < 1e-10:
            break
    return h, best_c

def _subgrad_polish(h0, n, n_iter=3000):
    h = h0.copy()
    dx = 2.0 / n
    best_c = _c5(h)
    for it in range(n_iter):
        C = _c5_arr(h)
        k_star = int(np.argmax(C))
        k = k_star - (n - 1)
        fwd = np.arange(n) + k
        bwd = np.arange(n) - k
        g = np.zeros(n)
        fv = (fwd >= 0) & (fwd < n)
        bv = (bwd >= 0) & (bwd < n)
        if fv.any():
            g[fv] = 1.0 - h[fwd[fv]]
        if bv.any():
            g[bv] -= h[bwd[bv]]
        g *= dx
        best_step = 0.0
        for eta in [0.005, 0.02, 0.05, 0.1, 0.3]:
            h_try = _fix(np.clip(h - eta * g, 0.0, 1.0), n)
            c = _c5(h_try)
            if c < best_c - 1e-12:
                best_c = c
                best_step = eta
                h = h_try
                break
    return h, best_c

def _binary_local_search(h0, n, max_iter=2000):
    h = h0.copy()
    binary = (h > 0.5).astype(float)
    # Ensure sum constraint
    excess = binary.sum() - 0.5 * n
    if excess > 0:
        ones = np.where(binary == 1)[0]
        for idx in ones[:int(excess)]:
            binary[idx] = 0.0
    elif excess < 0:
        zeros = np.where(binary == 0)[0]
        for idx in zeros[:int(-excess)]:
            binary[idx] = 1.0
    h = _fix(binary, n)
    best_c = _c5(h)
    C = _c5_arr(h)
    for it in range(max_iter):
        improved = False
        for i in range(n):
            for j in range(n):
                if i == j:
                    continue
                if h[i] > 0.5 and h[j] < 0.5:
                    h_try = h.copy()
                    h_try[i] = 0.0
                    h_try[j] = 1.0
                    h_try = _fix(h_try, n)
                    c = _c5(h_try)
                    if c < best_c - 1e-12:
                        h = h_try
                        best_c = c
                        C = _c5_arr(h)
                        improved = True
                        break
            if improved:
                break
        if not improved:
            break
    return h, best_c

def _softmax_obj(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_arr(h)
    nc = len(C)
    sh = alpha * C
    m = sh.max()
    es = np.exp(sh - m)
    se = es.sum()
    f = (m + np.log(se)) / alpha + beta * (h.sum() - 0.5 * n) ** 2
    w = es / se
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (h.sum() - 0.5 * n)
    return f, grad

def _softmax_opt(h0, n, alpha, beta, maxiter=300):
    res = minimize(_softmax_obj, h0, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is None:
        return h0
    return _fix(res.x, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _el():
        return time.time() - t0

    n0 = len(initial_h_values)
    h0 = _fix(initial_h_values.astype(float).copy(), n0)
    best_h = h0.copy()
    best_c = _c5(h0)
    best_n = n0

    # Phase 1: Coordinate descent from initial (most important)
    if _el() < 350:
        hn, cn = _coord_descent(h0, n0, max_sweeps=400, seed=seed)
        if cn < best_c:
            best_h, best_c = hn, cn

    # Phase 2: Subgradient polish
    if _el() < 550:
        hn, cn = _subgrad_polish(best_h, best_n, n_iter=3000)
        if cn < best_c:
            best_h, best_c = hn, cn
        # Coordinate descent again after subgrad
        hn, cn = _coord_descent(best_h, best_n, max_sweeps=200, seed=seed+1)
        if cn < best_c:
            best_h, best_c = hn, cn

    # Phase 3: Binary local search
    if _el() < 700:
        hn, cn = _binary_local_search(best_h, best_n, max_iter=500)
        if cn < best_c:
            best_h, best_c = hn, cn
        # Then continuous coordinate descent from binary
        hn, cn = _coord_descent(best_h, best_n, max_sweeps=200, seed=seed+2)
        if cn < best_c:
            best_h, best_c = hn, cn

    # Phase 4: Soft-max L-BFGS-B
    if _el() < 900:
        for alpha in [30, 80, 200]:
            if _el() > 900:
                break
            hn = _softmax_opt(best_h, best_n, alpha, 10.0, maxiter=200)
            cn = _c5(hn)
            if cn < best_c:
                best_h, best_c = hn, cn
            # Follow with coordinate descent
            hn, cn = _coord_descent(best_h, best_n, max_sweeps=100, seed=seed+3)
            if cn < best_c:
                best_h, best_c = hn, cn

    # Phase 5: Try different n values
    if _el() < 1020:
        x_old = np.linspace(0, 2, best_n, endpoint=False)
        n_candidates = sorted(set([max(40, best_n - 15), best_n - 5,
                                    best_n + 5, max(40, best_n - 25),
                                    best_n + 10, 64, 80, 128]))
        for n in n_candidates:
            if _el() > 1020:
                break
            if n == best_n or n < 30 or n > 250:
                continue
            x_new = np.linspace(0, 2, n, endpoint=False)
            ht = _fix(np.interp(x_new, x_old, best_h), n)
            ht, ct = _coord_descent(ht, n, max_sweeps=150, seed=seed+n)
            if ct < best_c:
                best_h, best_c, best_n = ht, ct, n
                x_old = np.linspace(0, 2, best_n, endpoint=False)

    # Phase 6: Final polish
    if _el() < 1080:
        best_h, best_c = _coord_descent(best_h, best_n, max_sweeps=200, seed=seed+100)
        hn, cn = _subgrad_polish(best_h, best_n, n_iter=1000)
        if cn < best_c:
            best_h, best_c = hn, cn
        best_h, best_c = _coord_descent(best_h, best_n, max_sweeps=100, seed=seed+200)
        best_h = _fix(best_h, best_n)
        best_c = _c5(best_h)

    return (best_h, best_c, best_n)
