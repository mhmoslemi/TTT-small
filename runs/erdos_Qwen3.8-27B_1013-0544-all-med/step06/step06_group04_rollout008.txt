        x_n = np.linspace(0, 2, n_try, endpoint=False)
        h_int = repair_h(np.interp(x_n, x_o, best_h), n_try)
        c5_int = evaluate_c5(h_int)
        if c5_int < best_c5:
            best_c5 = c5_int
            best_h = h_int.copy()
        h_opt = h_int.copy()
        while time.time() < t3:
            h_new = nm_optimize(h_opt, n_try, maxiter=6000, deadline=t3)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_opt = repair_h(best_h + rng.randn(n_try) * 0.008, n_try)
            else:
                h_opt = repair_h(best_h + rng.randn(n_try) * 0.1, n_try)
            if time.time() > t3:
                break

    # Final polish on best solution
    if time.time() < t0 + budget_s - 3:
        n_f = len(best_h)
        h_pol = nm_optimize(best_h, n_f, maxiter=10000, deadline=t0 + budget_s - 2)
        c5_pol = evaluate_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    best_h = repair_h(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return best_h, best_c5, len(best_h)
