def soft_max_grad(h, t, top_k):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full") * dx
    if top_k < len(conv):
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
        cv = conv[top_idx]
    else:
        top_idx = np.arange(len(conv))
        cv = conv
    m = cv.max()
    exps = np.exp((cv - m) / t)
    w = exps / exps.sum()
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        if w[i] < 1e-14:
            continue
        shift = top_idx[i] - (n - 1)
        grad += w[i] * _grad_shift(h, shift)
    return grad * dx

              binary_every=0, binary_tau=0.03, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimization with time limit."""
    n = len(h0)
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    step = 0
    check_interval = 50
    
    while time.time() < t_end:
        step += 1
        progress = min(step / 5000.0, 1.0)  # Normalize over ~5000 steps
        t_curr = t_init * (t_final / t_init) ** progress
        tk = max(3, int(top_k_init + (top_k_final - top_k_init) * progress))
        
        g = soft_max_grad(h, t_curr, tk)
        
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * g * g
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        if binary_every > 0 and step % binary_every == 0:
            h_new = 1.0 / (1.0 + np.exp(-(h_new - 0.5) / binary_tau))
            h_new = project_feasible(h_new, n)
        
        h = h_new
        if step % check_interval == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _grad_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += 1.0 - h[:n-shift]
        if shift > 0:
            g[:n-shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n-s] += 1.0 - h[s:]
            g[s:] -= h[:n-s]
    return g

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()

            hi = mid
        else:
