import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.max(c))

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def smooth_max_obj(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def compute_smooth_grad(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    mc = c.max()
    shifted = alpha * (c - mc)
    mx = shifted.max()
    w = np.exp(shifted - mx)
    w /= w.sum()
    result = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-8:
            continue
        s = k - (n - 1)
        a = max(0, -s)
        b = min(n - 1, n - 1 - s)
        if a > b:
            continue
        ww = w[k] * dx
        result[a:b+1] += ww * (1.0 - h[a+s:b+1+s])
        result[a+s:b+1+s] -= ww * h[a:b+1]
    return result

def proximal_gradient_step(h, n, alpha, lr, momentum_buf, beta):
    grad = compute_smooth_grad(h, n, alpha)
    momentum_buf = beta * momentum_buf - lr * grad
    h_new = project_feasible(h + momentum_buf, n)
    return h_new, momentum_buf

def slsqp_optimize(h, n, alpha, bounds, cons, maxiter=2000, ftol=1e-16):
    try:
        res = minimize(smooth_max_obj, h, args=(n, alpha), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': ftol})
        return project_feasible(res.x, n)
    except Exception:
        return h
