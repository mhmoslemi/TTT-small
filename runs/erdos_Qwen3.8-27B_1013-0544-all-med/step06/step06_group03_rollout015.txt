def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def softmax_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = conv.max()

        shift = idx - (n - 1)
        grad += weights[idx] * _conv_grad_single(h, shift)
    return grad * dx

        # Try the step with backtracking
        x_new = project_feasible(z - lr * grad, n)
        c5_new = compute_c5(x_new)
        
        # Backtracking: if objective increased, reduce step size
        while c5_new > prev_c5 + 1e-13 and lr > 1e-10:
            lr *= 0.5
            x_new = project_feasible(z - lr * grad, n)
            c5_new = compute_c5(x_new)
        
        # Occasionally increase step size if we've been successful
        if c5_new < best_c5:
            stall = 0
            lr = min(lr * 1.05, 50.0 / n)
        else:
            stall += 1
        
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = x_new.copy()
        
        if stall > 2000:
            break
        
        # Momentum update
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
        z = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        x = x_new
        t_k = t_new
        prev_c5 = c5_new
    
    return best_h, best_c5
