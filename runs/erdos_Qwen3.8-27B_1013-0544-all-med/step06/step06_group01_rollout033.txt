```python
def _coord_min_hj(h, j, alpha, beta, dx, n):
    """Find optimal h[j] given all other h fixed."""
    # Precompute C_base (h[j]=0) and the coefficients for v
    h0 = h.copy()
    h0[j] = 0
    C_const = np.correlate(h0, 1 - h0, mode='full') * dx  # length 2n-1
    nc = 2*n - 1
    
    # Coefficients a_m, b_m, c_m such that C[m] = a_m*v^2 + b_m*v + C_const[m]
    a_coef = np.zeros(nc)
    b_coef = np.zeros(nc)
    
    for ell in range(-(n-1), n):
        m = ell + (n-1)
        jp = j + ell
        jm = j - ell
        # From i=j: dx*v*(1-h[jp]) if jp valid
        if 0 <= jp < n:
            if jp == j:  # ell=0
                a_coef[m] += -dx  # -dx*v^2 term (from v*(1-v))
                b_coef[m] += dx   # +dx*v term
            else:
                b_coef[m] += dx * (1 - h[jp])
        # From i=jm (i+ell=j): dx*h[jm]*(1-v) if jm valid, ell!=0
        if ell != 0 and 0 <= jm < n:
            b_coef[m] += -dx * h[jm]
    
    # Now C[m] = C_const[m] + a_coef[m]*v^2 + b_coef[m]*v
    # Minimize f(v) = (max + log(sum(exp(alpha*C - max))))/alpha + beta*(sum_h - n/2)^2
    # where sum_h = sum(h) - h[j] + v
    
    sum_rest = np.sum(h) - h[j]
    
    from scipy.optimize import minimize_scalar
    def obj(v):
        C = C_const + a_coef*v*v + b_coef*v
        m = np.max(C)
        E = np.exp(alpha*(C - m))
        se = np.sum(E)
        f = (m + np.log(se))/alpha
        s = sum_rest + v
        f += beta * (s - 0.5*n)**2
        return f
    
    res = minimize_scalar(obj, bounds=(0, 1), method='bounded', options={'xatol': 1e-10})
    return res.x
```

```python
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        if d > 0:
            # Need to add mass; add to smallest values
            idx = np.argsort(h)
            for i in idx:
                add = min(d, 1.0 - h[i])
                h[i] += add
                d -= add
```

```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(h)
            for i in idx:
                add = min(d, 1.0 - h[i])
                h[i] += add
                d -= add
                if d <= 1e-14:
                    break
        else:
            idx = np.argsort(-h)
            for i in idx:
                rem = min(-d, h[i])
                h[i] -= rem
                d += rem
                if d >= -1e-14:
                    break
    return h

def _smoothed_max_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    nc = len(C)
    m = np.max(C)
    E = np.exp(alpha * (C - m))
    se = np.sum(E)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = E / se
    grad = np.zeros(n)
    for j in range(n):
        s1 = 0.0
        s2 = 0.0
        for mi in range(nc):
            ell = mi - (n - 1)
            jp = j + ell
            jm = j - ell
            if 0 <= jp < n:
                s1 += w[mi] * (1.0 - h[jp])
            if 0 <= jm < n:
                s2 += w[mi] * h[jm]
        grad[j] = (dx / alpha) * (s1 - s2)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbgfs(h, alpha, beta, maxiter=500):
    n = len(h)
    res = minimize(_smoothed_max_grad, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-15})
    if res.x is None:
        return h
    return _fix(res.x, n)

    mval = np.max(C)
    Ev = np.exp(alpha * (C - mval))
    return (mval + np.log(np.sum(Ev))) / alpha + beta * (sum_rest + v - 0.5 * n) ** 2

    dx = 2.0 / n

            res = minimize_scalar(_coord_scalar_obj,
```
