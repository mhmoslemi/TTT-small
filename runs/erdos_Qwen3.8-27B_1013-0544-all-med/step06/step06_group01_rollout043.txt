from scipy.optimize import minimize
import time

def _c5_max(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _project(h, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def _soft_min_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, h, mode='full')[n - 1:] * dx
    s = -beta * R
    s -= s.max()
    e = np.exp(s)
    p = e / e.sum()
    idx = np.abs(np.arange(n)[:, None] - np.arange(n)[None, :])
    grad = dx * (p[0] * h + p[idx].dot(h))
    return grad

def _nesterov(h0, n, betas, total_iters, t0, deadline):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = _c5_max(h0)
    v = np.zeros(n)
    mu = 0.9
    it = 0
    
    for beta in betas:
        n_inner = total_iters // len(betas)
        step0 = 50.0 / (beta * np.sqrt(n))
        for i in range(n_inner):

            if time.time() > deadline:
                return best_h, best_c5
            grad = _soft_min_grad(h, beta)
            v = mu * v + (1.0 - mu) * grad
            step = step0 / (1.0 + 0.1 * i)
            h_new = h - step * v
            h_new = _project(h_new, n)
            c5 = _c5_max(h_new)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h_new.copy()
            h = h_new
    return best_h, best_c5

def _symmetrize(h):
    return 0.5 * (h + h[::-1].copy())

        m = alpha * C
        m -= m.max()
        return m.max() + np.log(np.exp(m - m.max()).sum()) / alpha + beta_con * (np.sum(h_in) - 0.5 * n) ** 2

        m = np.max(sh)

import numpy as np
from scipy.optimize import minimize
import time

def _c5_max(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def _project(h, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

def _sym(h):
    return 0.5 * (h + h[::-1])

def _soft_min_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, h, mode='full')[n - 1:] * dx
    s = -beta * R
    s -= s.max()
    e = np.exp(s)
    p = e / e.sum()
    idx = np.abs(np.arange(n)[:, None] - np.arange(n)[None, :])
    return dx * (p[0] * h + p[idx].dot(h))

def _nesterov(h0, n, betas, iters_per_beta, t0, deadline):
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = _c5_max(h0)
    v = np.zeros(n)
    mu = 0.9
    for beta in betas:
        base_step = n / 200.0 / np.sqrt(1.0 + beta / 100.0)
        step = base_step
        for i in range(iters_per_beta):
            if time.time() > deadline:
                return best_h, best_c5
            grad = _soft_min_grad(h, beta)
            v = mu * v + (1.0 - mu) * grad
            h_new = h - step * v
            h_new = _sym(_project(h_new, n))
            c5 = _c5_max(h_new)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h_new.copy()
            elif c5 > best_c5 + 0.02:
                step = max(0.005, step * 0.5)
                h_new = h - step * v
                h_new = _sym(_project(h_new, n))
                c5 = _c5_max(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
            h = h_new
            step = base_step / (1.0 + 0.005 * i)
    return best_h, best_c5

def _resample(h_old, n_old, n_new):
    if n_old == n_new:
        return h_old.copy()
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    return _sym(_project(np.interp(x_new, x_old, h_old), n_new))

def _lbgfs_obj(h, n, alpha, beta_c):
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    sh = alpha * C
    m = sh.max()
    es = np.exp(sh - m)
    return (m + np.log(es.sum())) / alpha + beta_c * (h.sum() - 0.5 * n) ** 2

def _lbgfs(h, n, alpha, beta_c, maxiter):
    res = minimize(_lbgfs_obj, h, args=(n, alpha, beta_c), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    if res.x is not None:
        return _project(res.x, n)
    return h

def _local_search(h, n, deadline, max_passes=10):
    best_c5 = _c5_max(h)
    for _ in range(max_passes):
        if time.time() > deadline:

        improved = False

                        c5 = _c5_max(h_try)
                        if c5 < best_c5 - 1e-10:

                    if h[j] >= delta and h[i] <= 1.0 - delta:
                        h_try = h.copy()
                        h_try[j] -= delta
                        h_try[i] += delta
                        c5 = _c5_max(h_try)
                        if c5 < best_c5 - 1e-10:
                            h, best_c5 = h_try, c5
                            improved = True
                            break
        if not improved:

    return h, best_c5
