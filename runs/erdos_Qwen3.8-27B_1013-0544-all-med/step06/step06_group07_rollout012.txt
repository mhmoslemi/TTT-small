import numpy as np
import time

def time_left(deadline):
    return max(0.0, deadline - time.time())

def c_values(h):
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(c_values(h)))

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(50):
        diff = target - float(x.sum())
        if abs(diff) <= 1e-12:
            break
        if diff > 0:
            slack = 1.0 - x
            mask = slack > 1e-14
            if not np.any(mask):
                break
            x[mask] += diff * slack[mask] / float(slack[mask].sum())
        else:
            mask = x > 1e-14
            if not np.any(mask):
                break
            x[mask] += diff * x[mask] / float(x[mask].sum())
        np.clip(x, 0.0, 1.0, out=x)
    # Final exact fix
    diff = target - float(x.sum())
    if abs(diff) > 1e-10:
        k = int(np.floor(target))
        x[:] = 0.0
        if k > 0:
            x[:k] = 1.0
        rem = target - float(k)
        if rem > 0.0 and k < n:
            x[k] = min(rem, 1.0)
    return x

def grad_c_k(x, k, n):
    """Gradient of C_k with respect to x, where k is the lag index into the full correlation."""
    g = np.zeros(n, dtype=float)
    s = int(k) - (n - 1)  # actual shift
    # (1 - h[j+s]) for valid j: j+s in [0,n-1] means j in [-s, n-1-s]
    lo = max(0, -s)
    hi = min(n-1, n-1-s)
    if lo <= hi:
        g[lo:hi+1] += (1.0 - x[lo+s:hi+1+s])
    # -h[j-s] for valid j: j-s in [0,n-1] means j in [s, s+n-1]
    lo2 = max(0, s)
    hi2 = min(n-1, s+n-1)
    if lo2 <= hi2:
        g[lo2:hi2+1] -= x[lo2-s:hi2+1-s]
    return g

def grad_smooth(x, tau, topk=None):
    """Gradient of smooth max approximation."""
    c = c_values(x)
    cm = float(np.max(c))
    m = c.size
    n = x.size
    
    if tau <= 0:
        active = np.array([np.argmax(c)], dtype=int)
    else:
        z = (c - cm) / tau
        z -= np.max(z)
        e = np.exp(z)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros(m)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        if topk is not None and 0 < topk < m:
            idx = np.argsort(w)[::-1][:topk]
            w2 = np.zeros(m)
            w2[idx] = w[idx]
            ss = float(w2.sum())
            if ss > 0:
                w = w2 / ss
    
    g = np.zeros(n, dtype=float)
    for k_idx in np.nonzero(w > 1e-13)[0]:
        g += grad_c_k(x, int(k_idx), n) * w[k_idx]
    return g * (2.0 / float(n))

def smooth_obj(x, tau, topk=None):
    c = c_values(x)
    cm = float(np.max(c))
    m = c.size
    if tau <= 0:
        return cm
    z = (c - cm) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0:
        return cm
    if topk is not None and 0 < topk < m:
        idx = np.argsort(e)[::-1][:topk]
        s = float(e[idx].sum())
    return cm + tau * float(np.log(s))

def project(v, target):
    """Project v onto {h: 0<=h_i<=1, sum(h)=target}."""
    n = v.size
    lo = -target  # minimum lambda
    hi = 1.0 + target  # maximum lambda
    # h_i = clip(v_i - lambda, 0, 1), sum = target
    # As lambda increases, sum decreases
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        h = np.clip(v - mid, 0.0, 1.0)
        s = float(h.sum())
        if abs(s - target) < 1e-14:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)
    # Fix exact
    diff = target - float(h.sum())
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-14
            if mask.any():
                h[mask] += diff * (1.0 - h[mask]) / float((1.0 - h[mask])[mask].sum())
        else:
            mask = h > 1e-14
            if mask.any():
                h[mask] += diff * h[mask] / float(h[mask][mask].sum())
        np.clip(h, 0.0, 1.0, out=h)
    return h

def pg_optimize(x0, time_limit, tau, topk, momentum, seed):

    x = repair(x0.astype(float))
    n = x.size
    target = 0.5 * n
    best_x = x.copy()
    best_c = exact_c(x)

    alpha = 1.0 / (n * 1.0)  # initial step size
    rng = np.random.RandomState(seed)
    deadline = time.time() + time_limit
    
    # Nesterov momentum
    beta = momentum
    
    while time.time() < deadline:
        g = grad_smooth(x, tau, topk)
        # Nesterov
        v = beta * v + (1 - beta) * g
        # Try step
        accepted = False
        for _ in range(20):
            x_new = project(x - alpha * v, target)
            c_new = exact_c(x_new)
            c_curr = exact_c(x)
            if c_new < c_curr - 1e-14:
                x = x_new
                accepted = True
                break
            alpha *= 0.5
            if alpha < 1e-12:
                break
        if not accepted:
            alpha = max(alpha * 2.0, 1e-12)
        
        c_now = exact_c(x)
        if c_now < best_c:
            best_c = c_now
            best_x = x.copy()

        if rng.rand() < 0.05:
            x_pert = repair(x + rng.normal(0, 0.01, n))
            if exact_c(x_pert) < best_c:
                best_c = exact_c(x_pert)
                best_x = x_pert.copy()
                x = x_pert
    
    return best_x, best_c

def coord_descent(x0, time_limit, seed):
    """Cyclic coordinate descent."""
    x = repair(x0.astype(float))
    n = x.size
    target = 0.5 * n
    best_x = x.copy()
    best_c = exact_c(x)
    rng = np.random.RandomState(seed)
    deadline = time.time() + time_limit
    order = np.arange(n)
    
    while time.time() < deadline:
        rng.shuffle(order)
        for idx in order:
            if time.time() > deadline:
                break
            # Try moving mass from idx
            old_val = x[idx]
            for delta in (-0.02, -0.01, 0.01, 0.02):
                new_val = old_val + delta
                if 0 <= new_val <= 1:
                    x[idx] = new_val
                    # Redistribute
                    diff = target - float(x.sum())
                    # Add diff to a random other position
                    other = rng.randint(n)
                    if other != idx:
                        x[other] += diff
                        x = repair(x)
                        if exact_c(x) < best_c:
                            best_c = exact_c(x)
                            best_x = x.copy()
                        else:
                            x[idx] = old_val
                            x[other] -= diff
                    else:
                        x[idx] = old_val
            x[idx] = old_val if x[idx] != new_val else x[idx]
    
    return best_x, best_c

        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.sin(np.pi * f * t + phase)
    return repair(h)

def binary_pattern(n, seed, blocks):
    """Create a start with block structure."""
    rng = np.random.RandomState(seed)
    n = int(n)
    h = np.full(n, 0.5)
    if blocks <= 1:
        return h
    boundaries = np.sort(rng.uniform(0, n, blocks - 1))
    boundaries = np.concatenate([[0], boundaries, [n]])
    for i in range(blocks - 1):
        lo = int(boundaries[i])
        hi = int(boundaries[i+1])
        if hi > lo:
            h[lo:hi] = 0.5 + rng.uniform(-0.2, 0.2)
    return repair(h)

import scipy.optimize as so

def _tl(deadline):
    return max(0.0, deadline - time.time())


def _cvals(h):
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx


def _cx(h):
    return float(np.max(_cvals(h)))


def _repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(50):
        d = target - float(x.sum())
        if abs(d) <= 1e-13:
            break
        if d > 0:
            slack = 1.0 - x
            m = slack > 1e-14
            if not np.any(m):
                break
            x[m] += d * slack[m] / float(slack[m].sum())
        else:
            m = x > 1e-14
            if not np.any(m):
                break
            x[m] += d * x[m] / float(x[m].sum())
        np.clip(x, 0.0, 1.0, out=x)
    d = target - float(x.sum())
    if abs(d) > 1e-10:
        k = int(np.floor(target))
        x[:] = 0.0
        if k > 0:
            x[:k] = 1.0
        rem = target - float(k)
        if rem > 0.0 and k < n:
            x[k] = min(rem, 1.0)
    return x


def _grad_k(x, k_idx, n):
    g = np.zeros(n, dtype=float)
    s = int(k_idx) - (n - 1)
    lo = max(0, -s)
    hi = min(n - 1, n - 1 - s)
    if lo <= hi:
        g[lo:hi + 1] += (1.0 - x[lo + s:hi + 1 + s])
    lo2 = max(0, s)
    hi2 = min(n - 1, s + n - 1)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] -= x[lo2 - s:hi2 + 1 - s]
    return g


def _grad_smooth(x, tau, topk):
    c = _cvals(x)
    cm = float(np.max(c))
    m = c.size
    n = x.size
    if tau <= 0:
        w = np.zeros(m)
        w[int(np.argmax(c))] = 1.0
    else:
        z = (c - cm) / tau
        z -= np.max(z)
        e = np.exp(z)
        s = float(e.sum())
        if not np.isfinite(s) or s <= 0:
            w = np.zeros(m)
            w[int(np.argmax(c))] = 1.0
        else:
            w = e / s
        if topk is not None and 0 < topk < m:

            w2 = np.zeros(m)
            w2[idx] = w[idx]
            ss = float(w2.sum())
            if ss > 0:
                w = w2 / ss
    g = np.zeros(n, dtype=float)
    act = np.nonzero(w > 1e-13)[0]
    for ki in act:
        g += _grad_k(x, int(ki), n) * w[ki]
    return g * (2.0 / float(n))

    cm = float(np.max(c))
    m = c.size
    if tau <= 0:
        return cm
    z = (c - cm) / tau
    z -= np.max(z)
