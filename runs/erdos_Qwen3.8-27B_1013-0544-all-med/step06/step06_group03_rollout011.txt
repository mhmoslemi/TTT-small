    m = np.max(conv)
    return t * (m + np.log(np.sum(np.exp((conv - m) / t)))) * dx

    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        if shift > 0:

    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
        if s > 0:
            g[s:] -= h[:n - s]
    return g

def lse_grad(h, t, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    w = exps / np.sum(exps)

        if w[p] < 1e-10:
            continue
        shift = p - (n - 1)
        grad += w[p] * conv_grad_single(h, shift)
    return grad * dx

            return lse_obj(h, t, n)

            return lse_grad(h, t, n)

            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,

            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_c5 = c5t
                best_h = ht.copy()
            h = ht
        except Exception:
            pass

        bh, bc = binary_local_search(best_h, max_iter=20000, n_top=3, n_try=30, seed=seed, t_deadline=min(deadline, t0 + 450))
        if bc < best_c5:
            best_h, best_c5 = bh, bc
    
    # Phase 3: Smooth from binary result
    if time.time() < deadline - 80:
        temps2 = [0.02, 0.01, 0.005, 0.002]
        d3 = min(deadline, t0 + 600)
        temp_h = best_h.copy()
        # Blend binary with continuous
        temp_h = 0.5 * (best_h + (best_h >= 0.5).astype(float))
        temp_h = project_feasible(temp_h, n0)
        bh, bc = lse_optimize(temp_h, n0, temps2, d3)
        if bc < best_c5:
            best_h, best_c5 = bh, bc
    
    # Phase 4: Try different n values
    for n_try in [160, 200, 256, 400, 512]:
        if time.time() > deadline - 120:
            break
        nc = len(best_h)
        if n_try == nc:
            continue
        xo = np.linspace(0, 2.0, nc, endpoint=False) + 1.0 / nc
        xn = np.linspace(0, 2.0, n_try, endpoint=False) + 1.0 / n_try
        hi = np.interp(xn, xo, best_h, period=2.0)
        hi = project_feasible(hi, n_try)
        
        # Quick smooth optimization
        temps3 = [0.03, 0.01, 0.005]
        d4 = min(deadline, t0 + max(300, 600 - (n_try - nc)))
        bh, bc = lse_optimize(hi, n_try, temps3, d4)
        
        # Binary local search at this n
        if time.time() < deadline - 60:
            bh2, bc2 = binary_local_search(bh, max_iter=8000, n_top=3, n_try=25, seed=seed + n_try, t_deadline=min(deadline, d4 + 80))
            if bc2 < bc:
                bh, bc = bh2, bc2
        
        if bc < best_c5:
            best_h, best_c5 = bh, bc
    
    # Phase 5: PGD polish on best solution
    if time.time() < deadline - 20:
        d5 = deadline - 5
        bh, bc = pgd_refine(best_h, max_iter=100000, lr_init=0.001, t_deadline=d5)
        if bc < best_c5:
            best_h, best_c5 = bh, bc
    
    # Phase 6: Final binary local search
    if time.time() < deadline - 10:
        bh, bc = binary_local_search(best_h, max_iter=10000, n_top=4, n_try=30, seed=seed + 999, t_deadline=deadline - 2)
        if bc < best_c5:
            best_h, best_c5 = bh, bc
    
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
