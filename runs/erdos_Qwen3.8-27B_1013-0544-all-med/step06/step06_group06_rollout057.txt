Is M positive semidefinite? Not necessarily. The weights p are a probability vector, and the structure of M depends on the specific p.

def dual_ascent(h, n, n_iter=100, alpha_dual=100, bounds=None, cons=None):
    """Optimize using primal-dual (mirror descent) approach."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, n)
    n_lags = 2 * n - 1
    
    for it in range(n_iter):
        # Compute current correlations
        dx = 2.0 / n
        c = np.correlate(h, 1 - h, mode='full') * dx
        
        # Check improvement
        c_val = c.max()
        if c_val < best_c5 - 1e-15:
            best_c5 = c_val
            best_h = h.copy()
        
        # Update dual weights (softmax)
        shifted = alpha_dual * (c - c.max())
        mx = shifted.max()
        if mx > 700:
            p = np.exp(shifted - mx)
        else:
            p = np.exp(shifted)
        p /= p.sum()
        
        # Solve inner QP: minimize sum_k p_k * c_k = const - dx * h^T M h
        # Which is equivalent to maximizing h^T M h
        # Or: minimize -h^T M h + linear terms
        # For SLSQP, we just use the weighted sum as objective
        
        def inner_obj(h_in):
            c_in = np.correlate(h_in, 1 - h_in, mode='full') * dx
            return float(np.sum(p * c_in))
        
        try:
            res = minimize(inner_obj, h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 2000, 'ftol': 1e-18})
            h = project_feasible(res.x, n)
        except Exception:
            pass
        
        # Gradually increase alpha_dual
        alpha_dual *= 1.1
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def obj_weighted(h, n, weights):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    return float(np.dot(weights, c))

def sum_con(h, n):
    return np.sum(h) - n / 2.0

def sum_con_jac(h, n):
    return np.ones(n)

def get_subgradient(h, n, n_active=5):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    top_idx = np.argsort(c)[-n_active:]
    w = np.exp(200 * (c[top_idx] - c[top_idx].max()))
    w /= w.sum()
    m = np.arange(n)
    g = np.zeros(n)
    for i, k in enumerate(top_idx):
        idx1 = m + k - n + 1
        idx2 = m - k + n - 1
        v1 = (idx1 >= 0) & (idx1 < n)
        v2 = (idx2 >= 0) & (idx2 < n)
        g[v1] += w[i] * (1.0 - h[idx1[v1]])
        g[v2] -= w[i] * h[idx2[v2]]
    return g, c

def slsqp_step(h, n, alpha, maxiter=3000, bounds=None, cons=None):
    try:
        res = minimize(obj_smooth, h, args=(n, alpha), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        return project_feasible(res.x, n)
    except Exception:
        return h

def slsqp_weighted(h, n, weights, maxiter=2000, bounds=None, cons=None):
    try:
        res = minimize(obj_weighted, h, args=(n, weights), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-18})
        return project_feasible(res.x, n)
    except Exception:
        return h
