from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp(np.clip((conv - m) / t, -50, 0))
    return t * (m + np.log(np.sum(exps))) * dx

def smooth_max_grad(h, t, top_k=None):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp(np.clip((conv - m) / t, -50, 0))
    weights = exps / np.sum(exps)
    if top_k is not None and top_k < len(weights):
        top_indices = np.argsort(weights)[::-1][:top_k]
        top_weights = weights[top_indices]
    else:
        top_indices = np.arange(len(weights))
        top_weights = weights
    grad = np.zeros(n)
    for i in range(len(top_indices)):
        idx = top_indices[i]
        w = top_weights[i]
        if w < 1e-14:
            continue
        s = idx - (n - 1)
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
        else:
            ss = -s
            grad[:n - ss] += w * (1.0 - h[ss:])
            grad[ss:] -= w * h[:n - ss]
    return grad * dx
