Let me think about another approach: **subgradient method with step size decay**.

At each iteration:
1. Compute C = correlate(h, 1-h, 'full') * dx
2. Find k* = argmax C
3. Compute the subgradient g = ∂C[k*]/∂h (which is just the i-th row of the gradient matrix for that k*)
4. Take a step: h -= lr * g
5. Project to [0,1]^n
6. Fix integral constraint

The subgradient for a specific k* is:

where d* = k* - (n-1).

The issue with subgradient is slow convergence. But it's very fast per iteration, so we can do many iterations.

Let me combine: use L-BFGS-B for the main optimization (with the fast gradient), then use subgradient or coordinate descent for polishing.

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    nc = len(C)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    w_rev = w[::-1]
    c1 = np.convolve(1.0 - h, w, mode='full')
    c2 = np.convolve(h, w_rev, mode='full')
    grad = dx * (c1[n-1:2*n-1] - c2[n-1:2*n-1])
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-12:
            if d2 > 0:
                idx = np.argsort(1.0 - h)
                h[idx[0]] += d2
            else:
                idx = np.argsort(h)
                h[idx[0]] += d2
            h = np.clip(h, 0.0, 1.0)
    return h

def _optimize_lbfgsb(h, alpha, beta, maxiter=500, gtol=1e-14):
    n = len(h)
    res = minimize(_obj_grad, h, args=(alpha, beta), jac=True,
                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'gtol': gtol, 'ftol': 1e-16,
                           'maxls': 50, 'maxfun': maxiter * 20})
    if res.x is not None:
        return _fix(res.x, n)
    return h

def _coord_descent(h, alpha, n_sweeps=10):
    """Coordinate descent: optimize one coordinate at a time."""
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    
    for sweep in range(n_sweeps):
        np.random.shuffle(range(n))  # Hmm, can't use closure... let me use a different approach
        for i in range(n):
            # Compute current C
            C = np.correlate(h, 1.0 - h, mode='full') * dx
            # The objective as a function of h[i] is:
            # (1/alpha) * log(sum_k exp(alpha * (a_k * h[i] + b_k)))
            # where a_k and b_k are the coefficients and constants.
            # C[k] = C[k]_current + (h[i]_new - h[i]_old) * (contribution of h[i] to C[k])
            # The contribution of h[i] to C[k] is:
            #   dx * [(1-h[i+d_k]) * 1{valid} - h[i-d_k] * 1{valid}] where d_k = k-(n-1)
            # But this depends on h[i] itself in the (1-h[i+d_k]) term when i+d_k = i, i.e., d_k=0.
            # And in the h[i-d_k] term when i-d_k = i, i.e., d_k=0.
            # So for k = n-1 (d=0): C[n-1] has a term h[i]*(1-h[i]) which is quadratic.
            # For other k: C[k] is linear in h[i].
            
            # For simplicity, let me just use a small 1D search.
            # Evaluate the log-sum-exp at a few points.
            
            # Actually, let me just do a golden section search on [0, 1].
            def obj_1d(val):
                h_new = h.copy()
                h_new[i] = val
                C_new = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
                sh = alpha * C_new
                m = np.max(sh)
                return (m + np.log(np.sum(np.exp(sh - m)))) / alpha
            
            res = minimize_scalar(obj_1d, bounds=(0.0, 1.0), method='bounded',
                                  options={'xatol': 1e-8})
            if res.fun < obj_1d(h[i]) - 1e-12:
                h[i] = res.x
    
    return _fix(h, n)

def _subgradient_polish(h, n_iters=5000, lr0=0.01):
    """Subgradient method with adaptive step size."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c5 = _c5_max(h)
    best_h = h.copy()
    
    for it in range(n_iters):
        C = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = np.argmax(C)
        d = k_star - (n - 1)
        
        # Subgradient
        g = np.zeros(n)
        # (1-h[i-d]) for valid i-d
        valid_a = (np.arange(n) - d >= 0) & (np.arange(n) - d < n)
        g[valid_a] += dx * (1.0 - h[np.arange(n)[valid_a] - d])
        # -h[i+d] for valid i+d
        valid_b = (np.arange(n) + d >= 0) & (np.arange(n) + d < n)
        g[valid_b] -= dx * h[np.arange(n)[valid_b] + d]
        g += 2.0 * 10.0 * (np.sum(h) - 0.5 * n)  # constraint penalty
        
        # Adaptive step size
        lr = lr0 / (1.0 + 0.01 * it)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = _fix(h_new, n)
        
        c5_new = _c5_max(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

But wait, there's still an issue. When I compute lins for coordinate i, I use the current values of h (other than h[i]). But h[i] itself affects lins when d=0 (the term (1-2h[i])*dx). Since I'm about to change h[i], the lins should be computed with the current h[i], which is what I'm doing. Good.

However, there's another subtlety: when I update C_base using lins and quads, I'm using the lins computed with the OLD h[i]. But the actual change in C due to changing h[i] from old_val to new_val is:
ΔC[k] = lin_k * δ + quad_k * δ²

where lin_k and quad_k depend on the other h values (not h[i] itself, except for the d=0 case where lin depends on h[i]).

For d ≠ 0: lin_k = [(1-h[i+d]) - h[i-d]]*dx (not depending on h[i] itself, assuming i+d ≠ i and i-d ≠ i, which is true for d≠0). So these are correct regardless of h[i].

For d = 0: The term is h[i]*(1-h[i])*dx. Change = (1-2h[i])_*old_*δ - δ². But I computed lins[n-1] = (1-2*h[i])*dx using the OLD h[i], and quads[n-1] = -dx. So ΔC[n-1] = (1-2h[i]_old)*δ*dx - δ²*dx. Let me verify:
(h[i]+δ)*(1-h[i]-δ)*dx - h[i]*(1-h[i])*dx
= (h[i] - h[i]² + δ - 2h[i]δ - δ² - δ² + ... )*dx 

Wait: (h+δ)(1-h-δ) = h(1-h) + δ(1-h) - hδ - δ² = h(1-h) + δ - 2hδ - δ²
So change = (δ - 2hδ - δ²)*dx = (1-2h)*δ*dx - δ²*dx ✓

Great, so my formula is correct.

from scipy.optimize import minimize, minimize_scalar
import time

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return np.max(_c5_array(h))

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    w_rev = w[::-1]
    c1 = np.convolve(1.0 - h, w, mode='full')
    c2 = np.convolve(h, w_rev, mode='full')
    grad = dx * (c1[n - 1:2 * n - 1] - c2[n - 1:2 * n - 1])
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-14:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = 0.5 * n - np.sum(h)
        if abs(d2) > 1e-10:
            if d2 > 0:
                h[np.argmax(1.0 - h)] += d2
            else:
                h[np.argmax(h)] += d2
            h = np.clip(h, 0.0, 1.0)
    return h

                  method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                  options={'maxiter': maxiter, 'gtol': 1e-15, 'ftol': 1e-16,
                           'maxls': 50, 'maxfun': maxiter * 30})
    if res.x is not None:

    return h

def _coord_eval(val, i, h, C_base, lins, quads, alpha):
    delta = val - h[i]
    C = C_base + lins * delta + quads * delta * delta
    sh = alpha * C
    m = np.max(sh)
    return (m + np.log(np.sum(np.exp(sh - m)))) / alpha

                    lins[k] = c

                                  args=(i, h, C_base, lins, quads, alpha),
                                  bounds=(0.0, 1.0), method='bounded',

            if res.x is not None:
