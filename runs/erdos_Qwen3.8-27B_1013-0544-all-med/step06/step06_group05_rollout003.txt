    n = len(h)
    # lag ranges from -(n-1) to n-1, stored at index lag + n - 1
    lags = np.arange(-(n-1), n)

Let me also think about a completely different approach: what if I use a subgradient method on the continuous relaxation?

The objective is f(h) = max_k g_k(h) where g_k(h) = Σ_l h[l] * (1-h[l+k]) * dx

The subgradient at h (when the argmax is unique at k*):
∂f/∂h[i] = dx * [(1-h[i+k*]) - h[i-k*]]  (with 0 for out of bounds)

So the negative subgradient direction is: -[(1-h[i+k*]) - h[i-k*]] = [h[i-k*] - (1-h[i+k*])] = h[i-k*] + h[i+k*] - 1

To minimize f, I want to move h in the direction:
Δh[i] = -α * (h[i-k*] + h[i+k*] - 1) = α * (1 - h[i-k*] - h[i+k*])

This means: increase h[i] when both neighbors at distance k* are 0, decrease h[i] when both are 1.

This makes intuitive sense! If at the worst lag k*, we have many 01 transitions, we want to reduce them by either:
- Making the 0s into 1s (where possible)
- Making the 1s into 0s (where possible)

The constraint Σ h[i] = n/2 must be maintained. So I need to project onto this constraint.

def subgradient_step(h, alpha):
    n = len(h)
    # Find the worst lag
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)  # index in [0, 2n-2]
    lag = k_star - (n - 1)  # actual lag
    
    # Compute subgradient
    sg = np.zeros(n)
    for i in range(n):
        val = 0.0
        if 0 <= i - lag < n:
            val += h[i - lag]
        if 0 <= i + lag < n:
            val += h[i + lag]
        sg[i] = alpha * (1.0 - val)
    
    # Update
    h_new = h - sg  # negative subgradient direction... wait
    # Actually sg[i] = alpha * (1 - h[i-lag] - h[i+lag])
    # This is the NEGATIVE of the subgradient (since subgradient = h[i-lag] + h[i+lag] - 1)
    # So h_new = h + sg moves in the negative subgradient direction
    # Wait no: subgradient of f at h is (1-h[i+k*]) - h[i-k*] = 1 - h[i+lag] - h[i-lag]
    # Negative subgradient = -(1 - h[i+lag] - h[i-lag]) = h[i+lag] + h[i-lag] - 1
    # So h_new[i] = h[i] - alpha * (h[i+lag] + h[i-lag] - 1)
    #               = h[i] + alpha * (1 - h[i+lag] - h[i-lag])
    
    h_new = h + alpha * (1.0 - sg_sum)  # where sg_sum[i] = h[i+lag] + h[i-lag]
    
    # Project onto 0 <= h <= 1 and sum = n/2
    h_new = np.clip(h_new, 0, 1)
    # Adjust sum
    target = n / 2.0
    diff = target - np.sum(h_new)
    if diff > 0:
        room = 1.0 - h_new
        h_new += diff / np.sum(room) * room
    else:
        h_new += diff / np.sum(h_new) * h_new
    h_new = np.clip(h_new, 0, 1)
    
    return h_new

def rng_pick_pairs(arr, n_samples, seed):

def subgradient_optimize(h, max_time=30.0, seed=42):
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    dx = 2.0 / n
    target = n / 2.0
    h = np.clip(h, 0, 1)
    h *= target / max(np.sum(h), 1e-12)
    h = np.clip(h, 0, 1)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    rng = np.random.RandomState(seed)
    start = time.time()
    alpha = 0.3

    while time.time() - start < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)

        grad = np.zeros(n)
        for i in range(n):
            val = 0.0
            if 0 <= i - lag < n:
                val += (1.0 - h[i + lag]) if (0 <= i + lag < n) else 0
            if 0 <= i + lag < n:
                val += (1.0 - h[i - lag]) if (0 <= i - lag < n) else 0
            # subgradient of g_k at h[i] = (1-h[i+k]) - h[i-k]
            sg = 0.0
            if 0 <= i + lag < n:
                sg += (1.0 - h[i + lag])
            if 0 <= i - lag < n:
                sg -= h[i - lag]
            grad[i] = sg

        # Projected gradient step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        diff = target - np.sum(h_new)
        if diff > 1e-9:
            room = 1.0 - h_new
            total = np.sum(room)
            if total > 1e-12:
                h_new += room * (diff / total)
            h_new = np.clip(h_new, 0, 1)
        elif diff < -1e-9:
            total = np.sum(h_new)
            if total > 1e-12:
                h_new *= (target / total)
            h_new = np.clip(h_new, 0, 1)

        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            alpha *= 0.95
            if alpha < 0.001:
                break

    return best_h, best_c5

def heuristic_init(n):

    h = np.zeros(n)
    # Use a block pattern: alternate blocks of 1s and 0s with varying sizes

        if (i + 1) % block == 0:
            val = 1 - val
    # Adjust to have exactly n//2 ones
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones = np.where(h == 1)[0]
        for idx in ones[target:]:
            h[idx] = 0
    elif n_ones < target:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:target - n_ones]:
            h[idx] = 1
    return h

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old)
    # Renormalize
    target = n_new / 2.0
    s = np.sum(h_new)
    if s > 1e-12:
        h_new *= target / s
    h_new = np.clip(h_new, 0, 1)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return max(0, budget_s - (time.time() - t0))

    # Get initial
    h_init = np.array(initial_h_values, dtype=float).copy()
    n_init = len(h_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n_init
    best_is_binary = np.allclose(h_init, np.round(h_init))

    # Phase 1: Fast binary local search on initial
    h_b = make_binary(h_init, n_init)
    t1 = min(180.0, rem() * 0.35)
    if t1 > 10:
        h1, c5_1 = fast_binary_ls(h_b, max_time=t1, use_double=True, double_interval=30)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
            best_n = n_init
            best_is_binary = True

    # Phase 2: Subgradient refinement
    if rem() > 40:
        t2 = min(60.0, rem() * 0.15)
        h2, c5_2 = subgradient_optimize(best_h, max_time=t2, seed=seed + 100)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            best_n = n_init
            best_is_binary = False

    # Phase 3: Try different resolutions
    n_candidates = []
    for n in [n_init - 4, n_init - 2, n_init + 2, n_init + 4, n_init + 8, n_init + 16, n_init + 32]:
        if n >= 40 and n % 2 == 0 and n != n_init and n < 1000:
            n_candidates.append(n)

    for nc in n_candidates:
        if rem() < 80:
            break
        h_interp = interpolate_h(best_h, nc)
        h_nb = make_binary(h_interp, nc)
        t_n = min(40.0, rem() - 80)
        if t_n < 5:
            break
        h_opt, c5_opt = fast_binary_ls(h_nb, max_time=t_n, use_double=True, double_interval=20)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = nc
            best_is_binary = True

    # Phase 4: Multi-start with heuristics
    for n_try in [n_init, n_init + 2, n_init + 4]:
        if rem() < 60:
            break
        h_heur = heuristic_init(n_try)
        t_h = min(30.0, rem() - 60)
        if t_h < 5:
            break
        h_opt, c5_opt = fast_binary_ls(h_heur, max_time=t_h, use_double=True, double_interval=15)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n_try
            best_is_binary = True

    # Phase 5: More subgradient from best
    if rem() > 30:
        t5 = min(50.0, rem() * 0.25)
        h5, c5_5 = subgradient_optimize(best_h, max_time=t5, seed=seed + 500)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
            best_n = len(h5)
            best_is_binary = False

    # Phase 6: Final fast binary polish if binary
    if rem() > 20 and best_is_binary:
        t6 = rem() * 0.85
        h6, c5_6 = fast_binary_ls(best_h, max_time=t6, use_double=True, double_interval=10)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
            best_n = len(h6)

    # Final evaluation
    final_c5 = compute_c5(best_h)
    n_final = len(best_h)
    return best_h, final_c5, n_final
