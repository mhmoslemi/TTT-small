                for b in range(a + 1, n_o):
                    for c in range(n_z):
                        for d in range(c + 1, n_z):
                            if count >= max_try:
                                break
                            h[ones[a]]=0; h[ones[b]]=0; h[zeros[c]]=1; h[zeros[d]]=1
                            new_c5 = compute_c5(h)
                            gain = current_c5 - new_c5
                            if gain > best_gain2:
                                best_gain2 = gain
                                best_move2 = (ones[a], ones[b], zeros[c], zeros[d])
                            h[ones[a]]=1; h[ones[b]]=1; h[zeros[c]]=0; h[zeros[d]]=0
                            count += 1
                        if count >= max_try: break
                    if count >= max_try: break
                if count >= max_try: break

            if best_move2 is not None:
                i1, i2, j1, j2 = best_move2
                h[i1]=0; h[i2]=0; h[j1]=1; h[j2]=1
                current_c5 = compute_c5(h)
                improved = True

        if not improved:
            break
    return h, current_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new)

def continuation_optimize(h_init, n, max_time, seed=42):
    """Multi-stage continuation: smooth to sharp"""
    h_curr = project_feasible(h_init.copy())
    betas = [3.0, 8.0, 20.0, 50.0, 120.0, 300.0, 800.0]
    start = time.time()
    for beta in betas:
        if time.time() - start > max_time * 0.8:
            break
        h_opt = grad_optimize(h_curr, beta, max_iter=200)
        c5v = compute_c5(h_opt)
        if c5v < compute_c5(h_curr) - 1e-10 or True:
            h_curr = h_opt
    return h_curr

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    best_h = project_feasible(h_init.copy())
    best_c5 = compute_c5(best_h)

    # Phase 1: Continuation gradient optimization at n_init
    if rem() > 60:
        t_alloc = min(rem() * 0.25, 200.0)
        h_cont = continuation_optimize(best_h, n_init, max_time=t_alloc, seed=seed)
        c5v = compute_c5(h_cont)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_cont.copy()

    # Phase 2: Binary local search from best continuous
    if rem() > 40:
        h_bin = make_binary(best_h)
        h_bs, c5_bs = binary_local_search(h_bin, max_time=min(rem() * 0.2, 150.0), seed=seed)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()

    # Phase 3: Polish continuous from binary solution
    if rem() > 50:
        h_cont2 = project_feasible(best_h.astype(float))
        for beta in [30.0, 100.0, 300.0]:
            if rem() < 30:
                break
            h_opt = grad_optimize_penalized(h_cont2, beta, penalty=100.0, max_iter=200)
            c5v = compute_c5(h_opt)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_opt.copy()
            h_cont2 = h_opt

    # Phase 4: Try other resolutions
    n_candidates = [96, 104, 108, 116, 120, 128, 90, 100, 140, 84]
    for n_try in n_candidates:
        if rem() < 80:
            break
        if n_try == n_init or n_try % 2 != 0:
            continue
        h_interp = interpolate_h(best_h, n_try)
        # Gradient optimize
        t_g = min(rem() * 0.15, 80.0)
        h_opt_n = continuation_optimize(h_interp, n_try, max_time=t_g, seed=seed + n_try)
        c5v = compute_c5(h_opt_n)
        if c5v < best_c5:
            best_c5 = c5v
            best_h = h_opt_n.copy()
        # Binary search
        if rem() > 40:
            h_b = make_binary(h_opt_n)
            h_bs, c5_bs = binary_local_search(h_b, max_time=min(rem() * 0.15, 50.0), seed=seed + n_try)
            if c5_bs < best_c5:
                best_c5 = c5_bs
                best_h = h_bs.copy()

    # Phase 5: Multi-start from random initializations at n_init
    if rem() > 60:
        rng = np.random.RandomState(seed + 7777)
        for trial in range(3):
            if rem() < 30:
                break
            # Random binary with n_init/2 ones
            h_rand = np.zeros(n_init)
            idx = rng.choice(n_init, n_init // 2, replace=False)
            h_rand[idx] = 1.0
            h_b, c5_b = binary_local_search(h_rand, max_time=min(rem() * 0.1, 40.0), seed=seed + 5000 + trial)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()

    # Phase 6: Final binary polish
    if rem() > 15:
        h_final = make_binary(best_h)
        h_f, c5_f = binary_local_search(h_final, max_time=min(rem() * 0.8, 100.0), seed=seed + 9999)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
