    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]

                s = frac.sum()
                if s > 1e-12:

                else:
                    h[idx[0]] = 1.0

                s = h[idx].sum()
                if s > 1e-12:
                    h[idx] += (diff / s) * h[idx]
                else:
                    h[idx[0]] = 0.0
        h = np.clip(h, 0.0, 1.0)
        # Final fix
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:

                i = int(np.argmin(h))
                h[i] = min(1.0, h[i] + diff)

                i = int(np.argmax(h))
                h[i] = max(0.0, h[i] + diff)

    h_best = h_cur.copy()
    c5_best = c5_cur

        zeros_idx = np.where(h_cur == 0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        h_new = h_cur.copy()
        h_new[i] = 0
        h_new[j] = 1
        c5_new = evaluate_c5(h_new)

            h_cur = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
    
    return h_best, c5_best

                break

        h_cur[i] = 0.0
        h_cur[j] = 1.0
        c5_new = evaluate_c5(h_cur)
        delta = c5_new - c5_cur
        if delta <= 0 or rng.random() < np.exp(-delta / max(t, 1e-15)):
            c5_cur = c5_new
            if c5_new < c5_best:
                h_best = h_cur.copy()
                c5_best = c5_new
        else:
            h_cur[i] = 1.0
            h_cur[j] = 0.0
    return h_best, c5_best

def subgradient_opt(h, n_steps, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h)
    h_cur = h.copy()
    c5_cur = evaluate_c5(h_cur)
    h_best = h_cur.copy()
    c5_best = c5_cur
    lr = 0.02
    for step in range(n_steps):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        max_val = np.max(corr)
        active = np.where(corr >= max_val - 1e-9)[0]
        k = active[rng.randint(len(active))]
        s = k - n + 1
        g = np.zeros(n)
        js = np.arange(n)
        fwd = js - s
        fwd_ok = (fwd >= 0) & (fwd < n)
        g[fwd_ok] += (1.0 - h_cur[fwd[fwd_ok]])
        bwd = js + s
        bwd_ok = (bwd >= 0) & (bwd < n)
        g[bwd_ok] -= h_cur[bwd[bwd_ok]]
        lr_cur = lr * (1.0 + 0.5 * np.sin(0.01 * step))
        h_new = h_cur - lr_cur * g
        h_new = repair(h_new)
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_cur:
            h_cur = h_new
            c5_cur = c5_new
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
        else:
            lr = lr * 0.95
    return h_best, c5_best
