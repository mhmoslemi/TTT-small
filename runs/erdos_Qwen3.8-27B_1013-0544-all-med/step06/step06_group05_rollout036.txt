def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_continuous(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection for the threshold
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

    """
    Compute gradient of g_lag(h) where lag = lag_offset - (n-1)
    g_lag(h) = sum_i h[i] * (1-h[i+lag]) for valid pairs

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_continuous(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:

            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    return d_lags

    """Find optimal t for h[i]+=t, h[j]-=t."""
    n = len(h)
    c = h[i] + h[j]
    t_min = max(0.0, c - 1.0) - h[i]
    t_max = min(1.0, c) - h[i]
    
    if t_max - t_min < 1e-12:
        return 0.0
    
    d_lags = compute_all_gradients(h, i, j)
    
    # f(t) = max_k (corr0[k] + t * d_lags[k])
    # Binary search for minimum
    lo = t_min
    hi = t_max
    
    # Check if minimum is at boundary
    f_lo = np.max(corr0 + lo * d_lags)
    f_hi = np.max(corr0 + hi * d_lags)
    
    # Subgradient check at lo
    vals_lo = corr0 + lo * d_lags
    max_lo = np.max(vals_lo)
    active_lo = d_lags[np.abs(vals_lo - max_lo) < 1e-9]
    
    # Subgradient check at hi
    vals_hi = corr0 + hi * d_lags
    max_hi = np.max(vals_hi)
    active_hi = d_lags[np.abs(vals_hi - max_hi) < 1e-9]
    
    if len(active_lo) > 0 and np.min(active_lo) > 1e-10:
        return lo  # Increasing at lo, so lo is minimum
    if len(active_hi) > 0 and np.max(active_hi) < -1e-10:
        return hi  # Decreasing at hi, so hi is minimum
    
    # Bisection
    for _ in range(40):
        mid = (lo + hi) / 2.0
        vals = corr0 + mid * d_lags
        f_mid = np.max(vals)
        active = d_lags[np.abs(vals - f_mid) < 1e-9]
        
        if len(active) == 0:
            hi = mid
        elif np.min(active) > 1e-10:
            hi = mid
        elif np.max(active) < -1e-10:
            lo = mid
        else:
            break
    
    best_t = (lo + hi) / 2.0
    return best_t

def pair_descent(h, max_time=60.0, seed=42, n_passes=100):
    """Coordinate descent on pairs."""
    n = len(h)
    h = normalize_continuous(h, n).copy()
    current_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    start = time.time()
    
    # First, do greedy passes
    for passe in range(n_passes):
        if (time.time() - start) > max_time:
            break
        # Find active lag and gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        active_idx = int(np.argmax(corr))
        lag = active_idx - (n - 1)
        
        # Compute gradient of active lag
        s = np.zeros(n)
        if lag >= 0:
            for i in range(n - lag):
                s[i] += (1.0 - h[i + lag])
            for i in range(lag, n):
                s[i] -= h[i - lag]
        else:
            m = -lag
            for i in range(m, n):
                s[i] += (1.0 - h[i - m])
            for i in range(0, n - m):
                s[i] -= h[i + m]
        
        # Best pair: i = argmin(s), j = argmax(s)
        i_best = int(np.argmin(s))
        j_best = int(np.argmax(s))
        
        if s[j_best] - s[i_best] < 1e-10:
            # Try random pairs
            improved_this_pass = False
            for _ in range(50):
                i = rng.randint(n)
                j = rng.randint(n)
                if i == j:
                    continue
                t = optimize_pair_1d(h, i, j, corr)
                h[i] = np.clip(h[i] + t, 0, 1)
                h[j] = np.clip(h[j] - t, 0, 1)
                new_c5 = compute_c5(h)
                if new_c5 < current_c5 - 1e-12:
                    current_c5 = new_c5
                    improved_this_pass = True
                    break
            if not improved_this_pass:
                break
        else:
            t = optimize_pair_1d(h, i_best, j_best, corr)
            h[i_best] = np.clip(h[i_best] + t, 0, 1)
            h[j_best] = np.clip(h[j_best] - t, 0, 1)
            h = normalize_continuous(h, n)
            new_c5 = compute_c5(h)
            if new_c5 >= current_c5 - 1e-12:
                # Revert
                h[i_best] = np.clip(h[i_best] - t, 0, 1)
                h[j_best] = np.clip(h[j_best] + t, 0, 1)
                h = normalize_continuous(h, n)
                break
            current_c5 = new_c5
    
    return h, compute_c5(h)

def multi_resolution_descent(h0, n, max_time=60.0, seed=42):

    h = np.interp(x_new, x_old, h0)
    h = normalize_continuous(h, n)

            d[k_off_neg] = (p1i + p2i) - (p1j + p2j)

min_t max_k (a_k + b_k * t)

This can be rewritten as:
min_t, y  y
subject to a_k + b_k * t ≤ y for all k

This is a 1D LP with 2n-1 constraints! I can solve it in O(n log n) using the fact that it's 1D.

The optimal t is at a vertex of the feasible region, which is at the intersection of two constraint lines. So I need to find the pair (k1, k2) that gives the minimum maximum.

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_continuous(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def compute_d_lags(h, i, j):
    """Compute d_lags[k] = ∂g_k/∂h[i] - ∂g_k/∂h[j]."""
    n = len(h)
    d = np.zeros(2 * n - 1)
    for ell in range(n):
        # Positive lag ell (offset = n-1+ell)
        k_off = n - 1 + ell
        di = (1.0 - h[i + ell]) if (i + ell < n) else 0.0
        di += -h[i - ell] if (i >= ell) else 0.0
        dj = (1.0 - h[j + ell]) if (j + ell < n) else 0.0
        dj += -h[j - ell] if (j >= ell) else 0.0
        d[k_off] = di - dj
        if ell > 0:
            # Negative lag -ell (offset = n-1-ell)
            k_off_neg = n - 1 - ell
            di_neg = (1.0 - h[i - ell]) if (i >= ell) else 0.0
            di_neg += -h[i + ell] if (i + ell < n) else 0.0
            dj_neg = (1.0 - h[j - ell]) if (j >= ell) else 0.0
            dj_neg += -h[j + ell] if (j + ell < n) else 0.0
            d[k_off_neg] = di_neg - dj_neg
    return d

def optimize_pair(h, i, j):
    """Find optimal t: h[i]+=t, h[j]-=t. Returns (t, new_c5)."""
    c = h[i] + h[j]
    t_lo = max(0.0, c - 1.0) - h[i]
    t_hi = min(1.0, c) - h[i]
    if t_hi - t_lo < 1e-12:
        return 0.0
    
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    d_lags = compute_d_lags(h, i, j)
    
    # Ternary search on f(t) = max(corr0 + t * d_lags)
    a, b = t_lo, t_hi
    for _ in range(30):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(corr0 + m1 * d_lags)
        f2 = np.max(corr0 + m2 * d_lags)
        if f1 <= f2:
            b = m2
        else:
            a = m1
    
    t = (a + b) / 2.0
    return t

def pair_descent(h0, max_time=120.0, seed=42):

            j_best = int(np.argmax(s))

But wait, if clipping occurs (h[i]+t > 1 or h[j]-t < 0), then the sum changes. In that case, I need to normalize.

Actually, the ternary search already constrains t to [t_lo, t_hi] which ensures both values stay in [0,1]. So clipping shouldn't occur. Let me verify:
- t_lo = max(0, c-1) - h[i], where c = h[i]+h[j]
- t_hi = min(1, c) - h[i]
- h[i] + t_lo = max(0, c-1) ≥ 0 ✓
- h[i] + t_hi = min(1, c) ≤ 1 ✓
- h[j] - t_lo = h[j] - max(0,c-1) + h[i] = c - max(0,c-1) = min(c, 1) ≤ 1 ✓
- h[j] - t_hi = h[j] - min(1,c) + h[i] = c - min(1,c) = max(c-1, 0) ≥ 0 ✓

So the bounds are guaranteed. No clipping needed, and the sum is preserved.

This means I can remove the `normalize_continuous` call in the inner loop! That saves a lot of time.

Let me also think about whether the ternary search is the best approach. Since f(t) = max_k (a_k + b_k * t) is convex and piecewise linear, the minimum is at a "knee point" where the active line changes. Ternary search will converge to this point, but it might take a while to pin down the exact knee.

Alternative: Since the function is piecewise linear with at most 2n-1 pieces, I could find all the breakpoints and evaluate f at each. But that's O(n) which is the same as one evaluation.

Actually, a better approach: I can find the minimum exactly by solving the 1D LP:
min_t y
s.t. a_k + b_k * t ≤ y for all k
     t_lo ≤ t ≤ t_hi

The optimal solution is at a vertex, which is either t_lo, t_hi, or the intersection of two constraint lines a_k1 + b_k1*t = a_k2 + b_k2*t.

To find the optimal t without enumerating all pairs:
1. If the function is decreasing at t_lo, the minimum is to the right
2. If increasing at t_hi, the minimum is to the left
3. Otherwise, bisect (as before)

The subgradient at t is the set {b_k : a_k + b_k*t = max_j (a_j + b_j*t)}.
- If min of active b_k > 0: we're past the minimum, go left
- If max of active b_k < 0: we're before the minimum, go right
- If 0 is in [min, max] of active b_k: we're at the minimum

This is what I had before. The bisection converges in O(log(1/ε)) steps. With 40 steps and the interval being at most 1, we get precision ~1e-12. That's more than enough.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def normalize_continuous(h, n):
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def compute_d_lags(h, i, j):
    n = len(h)
    d = np.zeros(2 * n - 1)
    for ell in range(n):
        k_off = n - 1 + ell
        di = (1.0 - h[i + ell]) if (i + ell < n) else 0.0
        di += -h[i - ell] if (i >= ell) else 0.0
        dj = (1.0 - h[j + ell]) if (j + ell < n) else 0.0
        dj += -h[j - ell] if (j >= ell) else 0.0
        d[k_off] = di - dj

            djn = (1.0 - h[j - ell]) if (j >= ell) else 0.0
            djn += -h[j + ell] if (j + ell < n) else 0.0
            d[k_neg] = din - djn
    return d

def optimize_pair(h, i, j):

    t_hi = min(1.0, c) - h[i]

            s = np.zeros(n)

            ib, jb = int(np.argmin(s)), int(np.argmax(s))
            if s[jb] - s[ib] < 1e-8:
                use_greedy = False
        
        improved = False
        n_try = n  # Try n pairs per sweep
        
        if use_greedy:

        else:

            for _ in range(n_try):

                jj = rng.randint(n)
                if ii != jj:

                jj = rng.randint(n)
                if ii != jj:
                    pairs_list.append((ii, jj))

            if abs(t) < 1e-10:

            h[jj] -= t

                improved = True

                    best_c5 = cur
                    best_h = h.copy()
            else:
                h[ii] -= t
                h[jj] += t
        
        if not improved:
            stall += 1
        else:
            stall = 0
