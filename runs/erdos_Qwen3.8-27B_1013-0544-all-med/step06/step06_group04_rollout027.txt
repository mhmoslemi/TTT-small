def compute_subgradient(h, tol=0.01):
    """Compute subgradient of max correlation at h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    max_val = np.max(corr)
    active = corr >= max_val - tol * max_val

            # Contribution from h[j] in (1-h[n]) where n = j
            if 0 <= j + k < n:
                grad[j] += dx * (-h[j+k])
    
    grad /= len(active_indices)

import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def compute_max_corr(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx, np.max(corr) * dx

def project_feasible(v, target_sum):
    n = len(v)
    def sum_at(tau):
        return np.sum(np.clip(v - tau, 0.0, 1.0))
    
    # If already in [0,1] and sum is correct, return
    s = sum_at(0.0)
    if s >= target_sum and v.min() >= 0 and v.max() <= 1:
        # Need to decrease some values
        pass
    
    lo = -2.0
    hi = 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        if sum_at(mid) > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(v - tau, 0.0, 1.0)

def subgradient_step(h, step_size, tol=0.05):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    active = corr >= max_c * (1.0 - tol)
    n_active = np.sum(active)
    
    grad = np.zeros(n)
    active_indices = np.where(active)[0]
    
    for idx in active_indices:
        k = idx - (n - 1)
        # For this lag k: g(k) = sum_n h[n+k]*(1-h[n])*dx
        # d/dh[j] = dx*(1-h[j-k])*[0<=j-k<n] - dx*h[j+k]*[0<=j+k<n]
        
        # First term: j where 0 <= j-k < n, i.e., k <= j < n+k
        j_lo = max(0, k)
        j_hi = min(n, n + k)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += dx * (1.0 - h[j_lo - k : j_hi - k])
        
        # Second term: j where 0 <= j+k < n, i.e., -k <= j < n-k
        j_lo2 = max(0, -k)
        j_hi2 = min(n, n - k)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] += -dx * h[j_lo2 + k : j_hi2 + k]
    
    grad /= n_active
    return grad, max_c * dx

def make_valid(h, target_sum):
    """Ensure 0<=h<=1 and sum(h) == target_sum exactly."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - h.sum()

        if room > 0:

            h[free] += (1.0 - h[free]) * scale

        room = np.sum(h[free])
        if room > 0:

            h[free] -= h[free] * scale
        remaining = h.sum() - target_sum
        if remaining > 1e-10:

            h[ones] = max(0.0, h[ones] - remaining)
    
    # Final correction: adjust to make sum exact
    # This might require more careful handling
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:

        if diff > 0:

            idx = np.argsort(1.0 - h)
            for i in idx:
                if h[i] < 1.0:
                    amt = min(diff, 1.0 - h[i])
                    h[i] += amt
                    diff -= amt
                    if abs(diff) < 1e-10:

                    amt = min(-diff, h[i])
                    h[i] -= amt
                    diff += amt
                    if abs(diff) < 1e-10:

def optimize_subgradient(h, n_iter=500, alpha=0.1, tol=0.05):

    n = len(h)
    target_sum = n / 2.0

def local_refine(h, n_iters=1000):
    """Local refinement by trying small perturbations."""
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.default_rng(42)
    
    for it in range(n_iters):
        # Pick a random index
        j = rng.integers(n)
        # Try small perturbation
        delta = rng.choice([0.1, 0.05, 0.01]) * rng.choice([-1, 1])
        h_trial = h.copy()

        if 0 <= h[j] + delta <= 1:
            h_trial[j] += delta
            # Find compensation index
            if delta > 0:

                candidates = np.where(h_trial > delta + 1e-10)[0]
                candidates = candidates[candidates != j]
            else:
                # Need to increase somewhere else
                candidates = np.where(h_trial < 1.0 + delta + 1e-10)[0]
                candidates = candidates[candidates != j]

                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5:
