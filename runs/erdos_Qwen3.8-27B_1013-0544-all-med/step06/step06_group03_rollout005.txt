import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def compute_conv(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def bottleneck_grad(h, K=50):
    """Gradient using top-K shifts (bottleneck approach)"""
    n = len(h)
    dx = 2.0 / n
    conv = compute_conv(h)
    # Get top K shifts
    topk_idx = np.argsort(conv)[-K:]
    weights = np.zeros(len(conv))
    # Use linear weights: heaviest weight on the max
    for i, idx in enumerate(topk_idx):
        weights[idx] = i + 1
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for idx in topk_idx:
        w = weights[idx]
        if w < 1e-14:
            continue
        shift = idx - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    return grad * dx, np.max(conv) * dx

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def nesterov_optimize(h_init, max_iter=50000, lr_init=0.005, K=50):
    """Nesterov-accelerated projected gradient with bottleneck gradient"""
    n = len(h_init)
    h = h_init.copy()
    h_prev = h.copy()
    t_val = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(max_iter):
        t_prev = t_val
        t_val = (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev)) / 2.0
        momentum = (t_prev - 1.0) / t_val
        
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        grad, c5_y = bottleneck_grad(y, K)
        
        h_new = project_feasible(y - lr * grad, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if c5_new > c5_y + 0.5 * lr * np.sum(grad * (h_new - y)):
            lr *= 0.7
            if lr < 1e-15:
                break
        else:
            lr = min(lr * 1.02, 0.05)
        
        h_prev = h
        h = h_new
    
    return best_h, best_c5

    dx = 2.0 / n

    best_c5 = np.max(conv) * dx
    best_h = h.copy()
    best_conv = conv.copy()
    
    rng = np.random.RandomState(42)

        # Annealing schedule
        frac = it / max_iter

        i, j = rng.choice(n, 2, replace=False)

- From flip_effect(h, j, -d): type 2 gives d*h[i]

Total linear at shift i-j: -d*h[j] - d*(1-h[i]) ✓

        if s > target:

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

    m = np.max(conv)

def soft_max_grad(h, t):

def flip_effect(h, i, delta):

        t = 0
        x = i
        while x:
            t ^= 1
            x &= x - 1
        h[i] = 1.0 - t  # or t, both work by symmetry

                current_max = np.max(conv)
                c5_now = current_max * dx
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
                break
    return best_h, best_c5

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        t = bin(i).count('1') % 2
        h[i] = 1.0 if t == 0 else 0.0
    return project_feasible(h, n)

def block_init(n, block_size):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i // block_size) % 2 == 0 else 0.0
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = thue_morse_init(64)
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov refinement
    if tl() > 80:
        ht, c5t = nesterov_pgd(best_h, max_iter=20000, lr_init=0.01, t_smooth=0.01)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 2: Pair flip search (coarse)
    if tl() > 60:
        ht, c5t = pair_flip_search(best_h, max_iter=80000, delta_start=0.08, delta_end=0.01, rng=rng)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 3: Nesterov (finer smooth)
    if tl() > 60:
        ht, c5t = nesterov_pgd(best_h, max_iter=20000, lr_init=0.005, t_smooth=0.005)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Pair flip search (fine)
    if tl() > 50:
        ht, c5t = pair_flip_search(best_h, max_iter=100000, delta_start=0.04, delta_end=0.002, rng=rng)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Nesterov final polish
    if tl() > 40:
        ht, c5t = nesterov_pgd(best_h, max_iter=15000, lr_init=0.002, t_smooth=0.003)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 6: Try different resolutions
    cur_n = len(best_h)
    for n_try in [48, 64, 96, 128, 192, 256, 400, 480, 640]:
        if tl() < 100:
            break
        if n_try == cur_n:
            continue
        xo = np.linspace(0, 2.0, cur_n, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        if tl() > 50:
            ht, c5t = nesterov_pgd(hi, max_iter=12000, lr_init=0.01, t_smooth=0.01)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
                cur_n = n_try
        if tl() > 40:
            ht, c5t = pair_flip_search(best_h, max_iter=40000, delta_start=0.06, delta_end=0.005, rng=rng)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
                cur_n = n_try

    # Phase 7: Structured inits
    if tl() > 80:
        for n_s in [64, 128, 256, 320]:
            if tl() < 40:
                break
            for init_fn in [thue_morse_init, lambda n: block_init(n, 4), lambda n: block_init(n, 8)]:
                if tl() < 30:
                    break
                hs = init_fn(n_s)
                ht, c5t = nesterov_pgd(hs, max_iter=8000, lr_init=0.01, t_smooth=0.02)
                if tl() > 20:
                    ht, c5t = pair_flip_search(ht, max_iter=30000, delta_start=0.08, delta_end=0.01, rng=rng)
                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t
                    cur_n = n_s

    # Phase 8: Final pair flip polish
    if tl() > 30:
        ht, c5t = pair_flip_search(best_h, max_iter=60000, delta_start=0.03, delta_end=0.001, rng=rng)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 9: Very fine Nesterov
    if tl() > 20:
        ht, c5t = nesterov_pgd(best_h, max_iter=10000, lr_init=0.001, t_smooth=0.002)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        final_c5 = best_c5
    return best_h, final_c5, len(best_h)
