                    break
    return best_h, best_c5

def targeted_pair_search(h, n, rng, max_iters=50000):
    best_h = h.copy()
    best_c5 = compute_c5(best_h, n)
    center = n - 1
    for _ in range(max_iters):
        c = np.correlate(best_h, 1 - best_h, mode='full')
        peak_idx = int(np.argmax(c))
        k = peak_idx - center
        ak = abs(k)
        if ak == 0:
            pairs = np.arange(n)
            contribs = best_h * (1 - best_h)
        elif k > 0:
            pairs = np.arange(n - ak)
            contribs = best_h[pairs + ak] * (1 - best_h[pairs])
        else:
            pairs = np.arange(n - ak)
            contribs = best_h[pairs] * (1 - best_h[pairs + ak])
        if len(contribs) == 0:
            continue
        top = np.argsort(contribs)[::-1][:8]
        pi = top[rng.randint(min(8, len(top)))]
        i1 = pairs[pi]
        i2 = i1 + ak
        if i1 == i2:
            continue
        for d in [0.01, 0.03, 0.05, 0.1]:
            for sgn in [1, -1]:
                hi1 = best_h[i1] + sgn * d
                hi2 = best_h[i2] - sgn * d
                if 0.0 <= hi1 <= 1.0 and 0.0 <= hi2 <= 1.0:
                    new_h = best_h.copy()
                    new_h[i1] = hi1
                    new_h[i2] = hi2
                    nc = compute_c5(new_h, n)
                    if nc < best_c5 - 1e-13:
                        best_h = new_h
                        best_c5 = nc
                        break
            else:
                continue
            break
    return best_h, best_c5

def interpolate_to_n(h_src, n_src, n_dst):
    xp = np.linspace(0, 2, n_src, endpoint=False)
    xp_ext = np.concatenate([xp - 2.0, xp, xp + 2.0])
    fp_ext = np.concatenate([h_src, h_src, h_src])
    x_new = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(x_new, xp_ext, fp_ext)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 80
        h0 = np.full(n0, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h, n0)
    best_n = n0

    # Phase 1: SLSQP continuation at original resolution
    for alpha in [30, 100, 300, 1000, 3000, 8000]:
        if budget_s - (time.time() - t0) < 25:
            break
        h_try = slsqp_optimize(best_h, best_n, alpha)
        c = compute_c5(h_try, best_n)
        if c < best_c5 - 1e-14:
            best_h, best_c5 = h_try, c

    # Phase 2: L-BFGS-B with penalty (different optimizer, may escape local opt)
    if budget_s - (time.time() - t0) > 40:
        h_try = best_h.copy()
        for alpha in [200, 1000, 3000]:
            if budget_s - (time.time() - t0) < 20:
                break
            h_try = lbfgsb_optimize(h_try, best_n, alpha, pen=1000.0)
            c = compute_c5(h_try, best_n)
            if c < best_c5 - 1e-14:
                best_h, best_c5 = h_try, c

    # Phase 3: Try higher resolutions
    for n_high in [100, 120, 160, 200]:
        if budget_s - (time.time() - t0) < 70:
            break
        h_hi = interpolate_to_n(h0, n0, n_high)
        h_hi = project_feasible(h_hi, n_high)
        c5_hi = compute_c5(h_hi, n_high)

        for alpha in [100, 500, 2000, 5000]:
            if budget_s - (time.time() - t0) < 35:
                break
            h_hi = slsqp_optimize(h_hi, n_high, alpha)
            c5_hi = compute_c5(h_hi, n_high)

        if budget_s - (time.time() - t0) > 30:
            h_hi = lbfgsb_optimize(h_hi, n_high, 3000, pen=500.0)
            c5_hi = compute_c5(h_hi, n_high)

        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_high

    # Phase 4: Targeted pair search on best solution
    if budget_s - (time.time() - t0) > 30:
        best_h, c_try = targeted_pair_search(best_h, best_n, rng, max_iters=80000)
        if c_try < best_c5:
            best_c5 = c_try

    # Phase 5: Binary push search
    if budget_s - (time.time() - t0) > 25:
        h_bin, c_bin = binary_push_local_search(best_h, best_n, rng, max_iters=60000)
        if c_bin < best_c5:
            best_h, best_c5 = h_bin, c_bin

    # Phase 6: Final SLSQP + L-BFGS-B polish
    if budget_s - (time.time() - t0) > 15:
        h_try = slsqp_optimize(best_h, best_n, 15000, maxiter=5000)
        c = compute_c5(h_try, best_n)
        if c < best_c5:
            best_h, best_c5 = h_try, c

    if budget_s - (time.time() - t0) > 10:
        h_try = lbfgsb_optimize(best_h, best_n, 10000, pen=2000.0, maxiter=5000)
        c = compute_c5(h_try, best_n)
        if c < best_c5:
            best_h, best_c5 = h_try, c

    # Final feasibility
    best_h = project_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h, best_n)

    return best_h, best_c5, best_n
