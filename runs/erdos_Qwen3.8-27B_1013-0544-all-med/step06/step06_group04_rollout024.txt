import numpy as np
from scipy.optimize import minimize
import time

def project_h(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum}"""
    # Use bisection on the Lagrange multiplier
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2, 0, 1)
    return h_proj

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def c5_with_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    
    # Find active constraint
    best_k = np.argmax(corr)
    shift = best_k - (n - 1)
    
    # Compute gradient of the active constraint
    grad = np.zeros(n)
    s = abs(shift)
    if shift >= 0:
        # g = sum_{j=0}^{n-s-1} h[j]*(1-h[j+s])
        valid_j = np.arange(n - s)
        grad[valid_j] += 1 - h[valid_j + s]
        valid_i = np.arange(s, n)
        grad[valid_i] -= h[valid_i - s]
    else:

        valid_j = np.arange(s, n)
        grad[valid_j] += 1 - h[valid_j - s]
        valid_i = np.arange(n - s)
        grad[valid_i] -= h[valid_i + s]
    
    return c5, grad * dx  # gradient of c5 (scaling by dx)

import numpy as np
import time

def project_h(h, target_sum):
    """Project h onto {0<=h<=1, sum(h)=target_sum} using bisection."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    """Compute C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def smooth_max_grad(h, beta=50.0):
    """Compute smooth max of correlation values and its gradient."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode="full") * dx
    
    # Log-sum-exp for stability
    max_val = np.max(corr)
    shifted = corr - max_val
    exp_shifted = np.exp(beta * shifted)
    weights = exp_shifted / np.sum(exp_shifted)
    
    smooth_val = max_val + np.log(np.sum(exp_shifted)) / beta
    
    # Gradient
    grad = np.zeros(n)
    n_corr = len(corr)  # 2n-1
    for k in range(n_corr):
        w = weights[k]
        if w < 1e-8:
            continue
        shift = k - (n - 1)
        s = abs(shift)
        if shift >= 0:
            j_range = np.arange(n - s)
            grad[j_range] += w * (1.0 - h[j_range + s])
            i_range = np.arange(s, n)
            grad[i_range] -= w * h[i_range - s]
        else:
            j_range = np.arange(s, n)
            grad[j_range] += w * (1.0 - h[j_range - s])
            i_range = np.arange(n - s)
            grad[i_range] -= w * h[i_range + s]
    
    return smooth_val, grad

def optimize_from_start(h_start, n, target_sum, max_iter=5000, beta=50.0):

    step_size = 0.05

        val = val * 1.0  # already scaled by dx in smooth_max_grad

                best_h = h_new.copy()
            if c5_new < val * 1.05:  # accept if not too bad

                break
        else:

        if c5_new < val:
            step_size = min(step_size * 1.1, 0.2)
        else:
