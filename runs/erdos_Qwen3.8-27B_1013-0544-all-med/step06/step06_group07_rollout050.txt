This is a 1D problem that can be solved exactly! The optimal t is where the active constraints intersect, or at a boundary.

    dx = 2.0 / n
    c = c_values(x)
    cm = float(np.max(c))

        idx = np.argmax(c)
        w = np.zeros(c.size); w[idx] = 1.0
        val = cm
    else:
        z = (c - cm) / tau
        z -= z.max()
        e = np.exp(z)
        s = e.sum()
        if s < 1e-300:
            w = np.zeros(c.size); w[np.argmax(c)] = 1.0
            val = cm
        else:
            w = e / s
            val = cm + tau * np.log(s)
    # Gradient

            continue
        L = ki - (n - 1)
