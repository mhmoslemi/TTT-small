Let me also think about the alpha parameter. The log-sum-exp approximation:
f_α(h) = (1/α) * log(sum_k exp(α * C_k))

For large α, this approximates max_k C_k. The gradient is:
∇f_α = (1/α) * (sum_k exp(α*C_k) * α * ∇C_k) / (sum_k exp(α*C_k))
      = sum_k w_k * ∇C_k where w_k = exp(α*C_k) / sum_j exp(α*C_j)

For α → ∞, w_k → indicator of the argmax. So the gradient becomes the gradient of the active (maximum) constraint. This is what we want for a constrained optimization.

The issue with very large α is numerical instability (overflow in exp). The standard fix is to subtract the max:
f_α = (1/α) * (m + log(sum_k exp(α*(C_k - m)))) where m = max C_k

This is stable for any α. Good.

def _c5_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-10:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        idx = np.argmax(1.0 - h) if d2 > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + d2, 0.0, 1.0)
    return h

def _smax_obj(h, alpha, beta, n):
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    f = m + np.log(se) / alpha
    f += beta * (np.sum(h) - 0.5 * n) ** 2
    return f

def _smax_grad(h, alpha, beta, n):
    C = _c5_array(h)
    nc = len(C)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    w = es / se
    f = m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    dx = 2.0 / n
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n-1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n-1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgs(h, n, alpha, beta, maxiter=400):
    res = minimize(_smax_grad, h, args=(alpha, beta, n), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    if res.x is not None:
        return _fix(res.x, n)
    return h

        return h, _c5_max(h)
    t0 = time.time()
    hb = (h > 0.5).astype(float)
    n1 = int(np.sum(hb))
    tgt = n // 2
    if n1 > tgt:
        idx = np.where(hb == 1)[0]
        hb[idx[:n1-tgt]] = 0
    elif n1 < tgt:
        idx = np.where(hb == 0)[0]
        hb[idx[:tgt-n1]] = 1
    best = _c5_max(hb)
    while time.time() - t0 < time_limit:
        imp = False
        oi = np.where(hb == 1)[0]
        zi = np.where(hb == 0)[0]
        for i in oi:
            for j in zi:
                hb[i], hb[j] = 0.0, 1.0
                c = _c5_max(hb)
                if c < best - 1e-14:
                    best = c
                    imp = True
                    break
                hb[i], hb[j] = 1.0, 0.0
            if imp:
                break
        if not imp:
            break
    return hb, best

def _swap_ls(h, n, time_limit=10.0):
    t0 = time.time()
    h = h.copy()
    best = _c5_max(h)
    step = 0.05

            if imp:

    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5_max(h):
    return float(np.max(_c5_array(h)))

def _fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    d = 0.5 * n - np.sum(h)
    if abs(d) > 1e-10:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
    d2 = 0.5 * n - np.sum(h)
    if abs(d2) > 1e-12:
        idx = np.argmax(1.0 - h) if d2 > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + d2, 0.0, 1.0)
    return h

def _smax_grad(h, alpha, beta, n):
    C = _c5_array(h)
    nc = len(C)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    w = es / se
    f = m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    dx = 2.0 / n
    j_arr = np.arange(nc)
    grad = np.empty(n)
    for i in range(n):
        a = i - j_arr + n - 1
        b = i + j_arr - n + 1
        va = (a >= 0) & (a < n)
        vb = (b >= 0) & (b < n)
        ta = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
        tb = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
        grad[i] = dx * np.dot(w, ta - tb)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgs(h, n, alpha, beta, maxiter=400):
    res = minimize(_smax_grad, h, args=(alpha, beta, n), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
    if res.x is not None:
        return _fix(res.x, n)
    return h

def _binary_ls(h, n, time_limit=5.0):
    if n % 2 != 0:
        return h, _c5_max(h)
    t0 = time.time()
    hb = (h > 0.5).astype(float)
    n1 = int(np.sum(hb))
    tgt = n // 2
    if n1 > tgt:
        idx = np.where(hb == 1)[0]
        hb[idx[:n1 - tgt]] = 0
    elif n1 < tgt:
        idx = np.where(hb == 0)[0]
        hb[idx[:tgt - n1]] = 1
    best = _c5_max(hb)
    while time.time() - t0 < time_limit:
        imp = False
        oi = list(np.where(hb == 1)[0])
        zi = list(np.where(hb == 0)[0])
        for i in oi:
            for j in zi:
                hb[i] = 0.0
                hb[j] = 1.0
                c = _c5_max(hb)
                if c < best - 1e-14:
                    best = c
                    imp = True
                    break
                hb[i] = 1.0
                hb[j] = 0.0
            if imp:
                break
        if not imp:
            break
    return hb, best

def _binary_anneal(n, time_limit=10.0, seed=0):
    """Try multiple random binary starting points with local search."""
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_h = None
    best_c = 1.0
    attempts = 0
    while time.time() - t0 < time_limit:
        attempts += 1
        # Generate random binary with n/2 ones
        pos = np.arange(n)
        rng.shuffle(pos)
        hb = np.zeros(n)
        hb[pos[:n // 2]] = 1.0
        # Refine with local search (short)
        t_sub = min(0.5, (time_limit - (time.time() - t0)) * 0.8)
        hb, c = _binary_ls(hb, n, time_limit=t_sub)
        if c < best_c:
            best_c = c
            best_h = hb.copy()
    return best_h if best_h is not None else (np.zeros(n)), best_c

def _swap_ls(h, n, time_limit=15.0):
    t0 = time.time()
    h = h.copy()
    best = _c5_max(h)
    step = 0.1
    while time.time() - t0 < time_limit:
        imp = False
        order = np.argsort(h)
        lo = order[:n // 2]
        hi = order[n // 2:]
        for i in lo:
            for j in hi:
                if h[j] - h[i] < step * 0.3:
                    continue
                si, sj = h[i], h[j]
                h[i] = min(1.0, si + step)
                h[j] = max(0.0, sj - step)
                c = _c5_max(h)
                if c < best - 1e-13:
                    best = c
                    imp = True
                    break
                h[i], h[j] = si, sj
            if imp:
                break
        if not imp:
            step *= 0.4
            if step < 0.003:
                break
    return h, best

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def el():
        return time.time() - t0

    n0 = len(initial_h_values)
    hb = _fix(initial_h_values.astype(float).copy(), n0)
    cb = _c5_max(hb)
    nb = n0
    x_ref = np.linspace(0, 2, nb, endpoint=False)

    # Phase 1: Try various resolutions, binary search + L-BFGS-B
    n_list = [94, 96, 98, 100, 102, 104, 106, 108, 110, 112, 114, 118, 120, 126, 132, 144, 160, 192]
    for n in n_list:
        if el() > budget_s * 0.35:
            break
        x_new = np.linspace(0, 2, n, endpoint=False)
        ht = _fix(np.interp(x_new, x_ref, hb), n)
        # Binary local search
        if n % 2 == 0:
            ht_b, cb_b = _binary_ls(ht, n, time_limit=2.0)
            if cb_b < cb - 1e-9:
                cb, hb, nb = cb_b, ht_b, n
                x_ref = np.linspace(0, 2, nb, endpoint=False)
        # L-BFGS-B refinement
        ht2 = _lbfgs(ht, n, 200, 2.0, maxiter=250)
        cn = _c5_max(ht2)
        if cn < cb - 1e-9:
            cb, hb, nb = cn, ht2, n
            x_ref = np.linspace(0, 2, nb, endpoint=False)

    # Phase 2: Binary annealing at several n values
    for n in [nb, nb + 2, nb + 4, 96, 100, 128]:
        if el() > budget_s * 0.50:
            break
        if n % 2 != 0 or n < 50:
            continue
        bh, bc = _binary_anneal(n, time_limit=3.0, seed=seed + n)
        if bc < cb - 1e-9:
            cb, hb, nb = bc, bh, n
            x_ref = np.linspace(0, 2, nb, endpoint=False)
        # Refine binary result with L-BFGS-B
        if el() < budget_s * 0.52:
            ht = _lbfgs(bh, n, 500, 5.0, maxiter=400)
            cn = _c5_max(ht)
            if cn < cb - 1e-9:
                cb, hb, nb = cn, ht, n
                x_ref = np.linspace(0, 2, nb, endpoint=False)

    # Phase 3: Deep continuous optimization
    for alpha in [200, 500, 1000, 2000, 5000, 10000, 20000]:
        if el() > budget_s * 0.72:
            break
        beta = max(5.0, 0.005 * alpha)
        hb = _lbfgs(hb, nb, alpha, beta, maxiter=500)
        cb = _c5_max(hb)

    # Phase 4: Swap local search
    if el() < budget_s * 0.88:
        hb, cb = _swap_ls(hb, nb, time_limit=min(25.0, budget_s * 0.10))

    # Phase 5: Final L-BFGS-B + swap
    if el() < budget_s * 0.94:
        hb = _lbfgs(hb, nb, 10000, 20.0, maxiter=300)
        cb = _c5_max(hb)
    if el() < budget_s * 0.97:
        hb, cb = _swap_ls(hb, nb, time_limit=3.0)

    # Phase 6: Random restarts
    for _ in range(25):
        if el() > budget_s * 0.99:
            break
        s = 0.02 + 0.08 * rng.rand()
        ht = _fix(hb + rng.randn(nb) * s, nb)
        ht = _lbfgs(ht, nb, 1000, 10.0, maxiter=200)
        cn = _c5_max(ht)
        if cn < cb - 1e-9:
            cb, hb = cn, ht

    hb = _fix(hb, nb)
    cb = _c5_max(hb)
    return (hb, cb, nb)
