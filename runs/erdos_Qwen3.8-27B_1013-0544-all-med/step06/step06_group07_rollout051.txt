        return repair(h0)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return repair(np.interp(x1, x0, h0))

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)

    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)

    if init is not None and init.size >= 10:
        base_n = int(init.size)
        if base_n > 200:
            base_n = 200
    else:
        base_n = 70

    n_list = []
    for n in (base_n, base_n - 5, base_n + 5, base_n - 10, base_n + 10,
              50, 60, 80, 100, 120, 140):
        n = int(n)
        if 30 <= n <= 180 and n not in n_list:
            n_list.append(n)
    n_list = n_list[:8]

    best_x = None
    best_c = 1e9

    for n in n_list:
        tl = deadline - time.time()
        if tl < 40:
            break

        if init is not None and init.size > 0:
            base = resample(init, n)
        else:
            base = repair(np.full(n, 0.5))

        starts = [base]
        rng = np.random.RandomState(seed_val + n)
        for s in (0.01, 0.03, 0.06):
            starts.append(repair(base + rng.normal(0, s, n)))
        t = (np.arange(n) + 0.5) * (2.0 / n)
        for f in (2, 3, 4):
            amp = 0.08 + 0.04 * f
            starts.append(repair(base + amp * np.sin(2 * np.pi * f * t / 2.0 + rng.uniform(0, 6.28))))

        max_start = 5 if n == base_n else 4
        starts = starts[:max_start]

        per_time = min(30.0, max(10.0, tl * 0.12))

        for ii, h0 in enumerate(starts):
            tl = deadline - time.time()
            if tl < 15:
                break
            t_slsqp = min(per_time * 0.5, tl * 0.4)
            t_nest = min(per_time * 0.3, tl * 0.3)
            t_cd = min(per_time * 0.2, tl * 0.2)

            x_s, c_s = slsqp_opt(h0, t_slsqp, seed_val + n * 100 + ii)
            if c_s < best_c:
                best_c = c_s
                best_x = x_s.copy()

            x_n, c_n = nesterov_pg(x_s, t_nest, seed_val + 5000 + n + ii)
            if c_n < best_c:
                best_c = c_n
                best_x = x_n.copy()

            x_c, c_c = coord_descent(x_n, t_cd, seed_val + 9000 + n + ii)
            if c_c < best_c:
                best_c = c_c
                best_x = x_c.copy()

    if best_x is None:
        best_x = repair(np.full(70, 0.5))
        best_c = exact_c(best_x)

    tl = deadline - time.time()
    if tl > 20:
        for attempt in range(3):
            if time.time() > deadline - 10:
                break
            x2, c2 = nesterov_pg(best_x, min(20.0, tl * 0.3), seed_val + 7777 + attempt)
            if c2 < best_c:
                best_c = c2
                best_x = x2
            x3, c3 = coord_descent(best_x, min(15.0, tl * 0.2), seed_val + 8888 + attempt)
            if c3 < best_c:
                best_c = c3
                best_x = x3

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
