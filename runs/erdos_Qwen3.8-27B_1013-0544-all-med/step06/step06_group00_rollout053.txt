def project_feasible(h, target_sum):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max(h, corr, T):
    n = len(h)
    dx = 2.0 / n
    cmax = np.max(corr)
    exp_T = np.exp(T * (corr - cmax))
    s = exp_T.sum()
    if s < 1e-300:
        return np.zeros(n)
    p = exp_T / s
    
    g = np.zeros(n)
    # Use top terms for efficiency
    n_terms = min(300, len(p))
    idx_order = np.argsort(-p)[:n_terms]
    
    for idx in idx_order:
        wk = p[idx]
        if wk < 1e-12:
            break
        k = idx - (n - 1)
        # First term: (1-h[j-k]) for valid j-k
        js = max(0, k)
        je = min(n, n + k)
        if js < je:
            g[js:je] += wk * (1.0 - h[js - k:je - k])
        # Second term: -h[j+k] for valid j+k
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            g[js2:je2] -= wk * h[js2 + k:je2 + k]
    
    g *= dx
    return g

def local_search_binary_fast(h_bin, max_sweeps=300, temperature=0.0):

    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode="full").astype(float)
    c5 = np.max(corr) * dx
    best_h = h_bin.copy()
    best_c5 = c5
    stagnant = 0
    
    # Precompute shift array
    shifts = np.arange(2 * n - 1) - (n - 1)
    
    for sweep in range(max_sweeps):
        # Compute change matrix: change[j, m] = delta in corr[m] if we flip h[j] (0->1)
        J = np.arange(n)[:, None]  # (n, 1)
        S = shifts[None, :]  # (1, 2n-1)
        idx1 = J - S  # j - shift
        idx2 = J + S  # j + shift
        valid1 = (idx1 >= 0) & (idx1 < n)
        valid2 = (idx2 >= 0) & (idx2 < n)
        idx1c = np.clip(idx1, 0, n - 1)
        idx2c = np.clip(idx2, 0, n - 1)
        change = (1.0 - h_bin[idx1c]) * valid1.astype(float) - h_bin[idx2c] * valid2.astype(float)
        
        # New corr for each possible flip
        # If h[j]=0: new_corr = corr + change[j]
        # If h[j]=1: new_corr = corr - change[j]
        sign = np.where(h_bin == 0, 1.0, -1.0)
        corr_new = corr[None, :] + sign[:, None] * change

Let me review this for correctness:

1. `compute_c5`: Correct. Computes max correlation * dx.

2. `project_feasible`: Clips to [0,1] and adjusts sum. The iteration might not converge perfectly if all values are at bounds, but that's an edge case.

3. `grad_smooth_max`: Computes gradient of smooth max approximation. The formula:
   g[j] = dx * Σ_k p[k] * [(1-h[j-k])*[valid] - h[j+k]*[valid]]
   
   Let me double-check. corr[m] = Σ_i h[i + (m-(n-1))] * (1-h[i]) for valid i.
   
   Let shift = m - (n-1). Then corr[m] = Σ_i h[i+shift] * (1-h[i]).
   
   ∂corr[m]/∂h[j]:
   - h[j] appears as h[i+shift] when i = j-shift: contributes (1-h[j-shift])
   - h[j] appears as h[i] when i = j: contributes -h[j+shift]
   
   So ∂corr[m]/∂h[j] = (1-h[j-shift])*[0≤j-shift<n] - h[j+shift]*[0≤j+shift<n]
   
   And g[j] = dx * Σ_m p[m] * ∂corr[m]/∂h[j]
   = dx * Σ_m p[m] * [(1-h[j-shift_m])*valid - h[j+shift_m]*valid]
   
   where shift_m = m - (n-1).
   
   In my code: k = idx - (n-1) where idx is the index into corr (which is m). So k = m - (n-1) = shift_m. ✓
   
   First term: (1-h[j-k]) for j where 0 ≤ j-k < n, i.e., k ≤ j < k+n, clipped to [0, n).
   js = max(0, k), je = min(n, n+k). Then j ranges from js to je-1, and j-k ranges from js-k to je-1-k.
   h[js-k : je-k] gives h[j-k] for j in [js, je). ✓
   
   Second term: -h[j+k] for j where 0 ≤ j+k < n, i.e., -k ≤ j < n-k, clipped to [0, n).
   js2 = max(0, -k), je2 = min(n, n-k). Then j ranges from js2 to je2-1, and j+k ranges from js2+k to je2-1+k.
   h[js2+k : je2+k] gives h[j+k] for j in [js2, je2). ✓
   
   Looks correct!

4. `local_search_binary_fast`: The change matrix computation:
   - change[j, m] = (1-h[j-shift_m])*[valid] - h[j+shift_m]*[valid] where shift_m = m-(n-1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx, corr

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_smooth_max(h, corr, T):
    n = len(h)
    dx = 2.0 / n
    cmax = np.max(corr)
    exp_T = np.exp(T * (corr - cmax))
    s = exp_T.sum()
    if s < 1e-300:
        return np.zeros(n)
    p = exp_T / s
    g = np.zeros(n)
    n_terms = min(400, len(p))
    idx_order = np.argsort(-p)[:n_terms]
    for idx in idx_order:
        wk = p[idx]
        if wk < 1e-12:
            break
        k = idx - (n - 1)
        js = max(0, k)
        je = min(n, n + k)
        if js < je:
            g[js:je] += wk * (1.0 - h[js - k:je - k])
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            g[js2:je2] -= wk * h[js2 + k:je2 + k]
    g *= dx
    return g
