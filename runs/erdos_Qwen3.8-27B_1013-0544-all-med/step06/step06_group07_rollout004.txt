            if no_improve % 5 == 0:
                step *= 0.93
            if step < 0.001:

        # Update bossing weights
        # Constraints with high C(k) get more weight
        cnorm = cvals - cvals.min()

        x_new = repair(x_new)

                    x = x_tr
                    step = trial_step
                    break

                # Add small random perturbation

                if exact_c(x_pert) < best_c:
                    best_c = exact_c(x_pert)
                    best_x = x_pert.copy()
                    x = x_pert
                    break

import numpy as np
import time
import warnings

warnings.simplefilter("ignore")

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(30):
        diff = target - float(x.sum())
        if abs(diff) <= 1e-13:
            break
        if diff > 0:
            slack = 1.0 - x
            mask = slack > 1e-14
            if not np.any(mask):
                break
            x[mask] += diff * slack[mask] / float(slack[mask].sum())
        else:
            mask = x > 1e-14
            if not np.any(mask):
                break
            x[mask] += diff * x[mask] / float(x[mask].sum())
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - float(x.sum())
    if abs(diff) > 1e-10:
        k = int(np.floor(target))
        x[:] = 0.0
        if k > 0:
            x[:k] = 1.0
        rem = target - float(k)
        if rem > 0 and k < n:
            x[k] = min(rem, 1.0)
    return x

def c_values(h):
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(c_values(h)))

def grad_shift(x, k):
    n = x.size
    dx = 2.0 / float(n)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[k:] += dx * (1.0 - x[:n - k])
            g[:n - k] -= dx * x[k:]
    else:
        k2 = -k
        if k2 < n:
            g[:n - k2] += dx * (1.0 - x[k2:])
            g[k2:] -= dx * x[:n - k2]
    return g

def topm_grad(x, topm):
    cvals = c_values(x)
    n = x.size
    m = 2 * n - 1
    topm = min(topm, m)
    idx = np.argpartition(cvals, -topm)[-topm:]
    g = np.zeros(n)
    for i in idx:
        k = int(i) - (n - 1)
        g += grad_shift(x, k)
    return g / float(topm), cvals

def resample_to(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    n_new = int(n_new)

        return repair(h0)
    if n0 == 0:
        return repair(np.full(n_new, 0.5))
    dt0 = 2.0 / float(n0)
    x0 = (np.arange(n0) + 0.5) * dt0
    x1 = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x1, x0, h0)
    return repair(y)

    target = int(round(0.5 * n))
    order = np.argsort(h)[::-1]
    x = np.zeros(n)
    x[order[:target]] = 1.0
    return x
