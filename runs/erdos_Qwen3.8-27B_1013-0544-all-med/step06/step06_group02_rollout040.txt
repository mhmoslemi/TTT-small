import numpy as np

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):

import numpy as np

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def lse_gradient(h, corr, dx, t):
    N = len(h)
    c = corr * dx
    c_max = c.max()
    w = np.exp(t * (c - c_max))
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        w = np.zeros(len(corr))
        w[k_star] = 1.0
    else:
        w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        wj = w[j]
        if wj < 1e-12:
            continue
        m = N - 1 - j
        s1 = max(0, -m)
        e1 = min(N, N - m)
        if s1 < e1:
            g[s1:e1] += wj * dx * (1.0 - h[s1 + m:e1 + m])
        s2 = max(0, m)
        e2 = min(N, N + m)
        if s2 < e2:
            g[s2:e2] -= wj * dx * h[s2 - m:e2 - m]
    return g
