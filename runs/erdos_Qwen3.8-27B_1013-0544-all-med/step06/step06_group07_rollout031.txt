def solve_1d_max(A, D, t_min, t_max):
    """Minimize max_k (A[k] + t * D[k]) over t in [t_min, t_max]"""
    # This is convex piecewise-linear in t
    # Find the optimal t
    
    # Approach: the minimum is at a breakpoint or boundary
    # Breakpoints: t = (A[i] - A[j]) / (D[j] - D[i]) for D[i] != D[j]
    
    # Simpler: use the fact that at the optimum, either:
    # - t is at a boundary
    # - the subgradient contains 0: sum of D[k] * w[k] = 0 where w[k] are the
    #   weights (probabilities) on the active set
    
    # For a robust solution, let's use the following:
    # The function f(t) = max_k (A[k] + t*D[k]) is convex.
    # Use ternary search (works for convex functions)
    
    best_t = t_min
    best_val = float('inf')
    
    # Evaluate at boundaries and some interior points
    # Then refine with ternary search
    for _ in range(60):  # enough for machine precision
        mid1 = t_min + (t_max - t_min) / 3
        mid2 = t_max - (t_max - t_min) / 3
        f1 = max(A + mid1 * D)
        f2 = max(A + mid2 * D)
        if f1 < f2:
            t_max = mid2
        else:
            t_min = mid1
        if f1 < best_val:
            best_val = f1
            best_t = mid1
        if f2 < best_val:
            best_val = f2
            best_t = mid2
    
    return best_t, best_val

    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return repair(np.interp(x1, x0, h0))

def binary_start(n, seed):
    rng = np.random.RandomState(seed)
    target = n / 2.0
    h = np.zeros(n)
    rng.shuffle(h)
    k = int(target)
    h[:k] = 1.0
    rem = target - k
    if rem > 0 and k < n:
        h[k] = rem
    rng.shuffle(h)
    return repair(h)

def structured_start(n, seed, freq):
    rng = np.random.RandomState(seed)
    t = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    for f in (freq, 2 * freq, 3 * freq):
        h += (0.45 / f) * np.sin(2.0 * np.pi * f * t / 2.0 + rng.uniform(0, 2 * np.pi))
    h += 0.02 * rng.normal(0, 1, n)
    return repair(h)

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    sv = int(seed)
    raw = globals().get("initial_h_values", None)
    init = None if raw is None else np.asarray(raw, dtype=float)

    best_h = None
    best_c = 1e9

    n_list = [70]
    if init is not None and init.size > 0:
        n_list = [int(init.size), 70, 80, 90, 100]
    else:
        n_list = [70, 80, 100]
    n_list = list(dict.fromkeys(n_list))
    n_list = [n for n in n_list if 40 <= n <= 110][:5]

    for n in n_list:
        if time.time() > deadline - 40:
            break
        time_for_n = min(200.0, (deadline - time.time()) * 0.6)

        if init is not None and init.size > 0:
            base = resample(init, n)
        else:
            base = repair(np.full(n, 0.5))

        starts = [base]
        for s in range(3):
            starts.append(binary_start(n, sv + 100 * n + s))
        for f in (1, 2, 3):
            starts.append(structured_start(n, sv + 500 * n + f, f))

        per_time = min(30.0, max(8.0, time_for_n / len(starts)))

        for si, h0 in enumerate(starts):
            if time.time() > deadline - 30:
                break
            t_lim = min(per_time, deadline - time.time() - 25)
            if t_lim < 3.0:
                break
            h_opt, c_opt = pair_exchange(h0, max_sweeps=300, seed=sv + 1000 * n + si, deadline=deadline)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
            # Try escape and re-optimize if time permits
            if time.time() < deadline - 50 and c_opt < 0.385:
                rng = np.random.RandomState(sv + 7000 + si)
                for esc in range(3):
                    if time.time() > deadline - 40:
                        break
                    scale = 0.04 + 0.03 * esc
                    h_esc = repair(best_h + rng.normal(0, scale, n) if best_h.size == n else h0 + rng.normal(0, scale, n))
                    h2, c2 = pair_exchange(h_esc, max_sweeps=200, seed=sv + 2000 * n + si + esc, deadline=deadline)
                    if c2 < best_c:
                        best_c = c2
                        best_h = h2.copy()

    # Final extended polish on the best
    if best_h is not None and time.time() < deadline - 10:
        h_f, c_f = pair_exchange(best_h, max_sweeps=1000, seed=sv + 99999, deadline=deadline)
        if c_f < best_c:
            best_c = c_f
            best_h = h_f

    if best_h is None:
        best_h = repair(np.full(70, 0.5))
        best_c = c5_eval(best_h)

    best_h = repair(best_h)
    best_c = c5_eval(best_h)
    return best_h, best_c, int(best_h.size)
