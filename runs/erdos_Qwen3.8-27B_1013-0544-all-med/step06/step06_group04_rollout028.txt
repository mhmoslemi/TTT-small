        if time.time() - start_time > budget_s * 0.88:
            break
        h_rand = np.zeros(n)
        h_rand[np.random.choice(n, n // 2, replace=False)] = 1.0
        h_sa, c5_sa = simulated_anneal(h_rand, n, 0.15, 0.001, 0.9995,
                                        200000, start_time, budget_s, 0.88)
        if c5_sa < best_c5:
            best_h, best_c5 = h_sa, c5_sa
        h_ls, c5_ls = deterministic_ls(best_h.copy(), n, 5000, start_time, budget_s, 0.92)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    # Phase 5: Final polish with extended deterministic search
    if time.time() - start_time < budget_s * 0.97:
        h_ls, c5_ls = deterministic_ls(best_h.copy(), n, 15000, start_time, budget_s, 1.0)
        if c5_ls < best_c5:
            best_h, best_c5 = h_ls, c5_ls

    best_h = repair_h(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
