import numpy as np
import time
import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")

def project(x, target):
    n = x.size
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    h = np.clip(x - t, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            m = h < 1.0
            if m.sum() > 0:
                h[m] += diff / m.sum()
        else:
            m = h > 0.0
            if m.sum() > 0:
                h[m] += diff / m.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def c_values(h):
    n = h.size
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(c_values(h)))

def compute_grad(h, w):
    n = h.size
    dx = 2.0 / n
    o_h = 1.0 - h
    w_rev = w[::-1]
    A_full = np.correlate(o_h, w, mode="full")
    B_full = np.correlate(h, w_rev, mode="full")
    A = A_full[:n][::-1]
    B = B_full[:n][::-1]
    return dx * (A - B)

def smooth_weights(c, beta):
    m = c.size
    z = c * beta
    z -= np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(m)
        w[np.argmax(c)] = 1.0
        return w
    return e / s

def resample_to(h0, n_new):
    n0 = h0.size
    if n0 == 0:
        return project(np.full(n_new, 0.5), 0.5 * n_new)
    if n0 == n_new:
        return project(h0.astype(float), 0.5 * n_new)
    dt0 = 2.0 / n0
    x0 = (np.arange(n0) + 0.5) * dt0
    x1 = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x1, x0, h0)
    return project(y, 0.5 * n_new)

def adam_anneal(h0, time_limit, seed, n_max_beta=3000):
    n = h0.size
    x = project(h0.copy(), 0.5 * n).astype(float)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_c = exact_c(x)
    best_x = x.copy()
    deadline = time.time() + max(0.0, time_limit)

    beta_start = 15.0
    beta_end = float(n_max_beta)
    lr_start = 0.008
    lr_end = 0.0003
    b1, b2, eps = 0.7, 0.999, 1e-8
    m_sets = 2 * n - 1

    n_iters_max = 8000
    it = 0
    while time.time() < deadline and it < n_iters_max:
        frac = it / float(n_iters_max)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        c = c_values(x)
        w = smooth_weights(c, beta)
        grad = compute_grad(x, w)

        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)

            sub_budget = min(15.0, t_rem * 0.3)
            yr, cr = adam_anneal(y0, sub_budget, seed + 900 + restart, n_max_beta=2000)
            if cr < best_c:
                best_c = cr
                best_x = yr
            t_rem = deadline - time.time()
            if t_rem < 5:
                break
            yr2, cr2 = local_search(best_x, min(5.0, t_rem * 0.3), seed + 950 + restart)
            if cr2 < best_c:
                best_c = cr2
                best_x = yr2
            t_rem = deadline - time.time()

    best_x = project(best_x, 0.5 * n_work)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
