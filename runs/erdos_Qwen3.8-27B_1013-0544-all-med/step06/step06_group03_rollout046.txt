import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def softmax_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / tau)
    return tau * (m / tau + np.log(np.sum(exps))) * dx

def softmax_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / tau)
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        if w[idx] < 1e-11:
            continue
        shift = idx - (n - 1)
        grad += w[idx] * conv_grad_single(h, shift)
    return grad * dx

def fista_smooth(h0, n, tau, max_iter, lr):
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    best_c5 = compute_c5(h)
    best_h = h0.copy()
    for k in range(max_iter):
        grad = softmax_grad(y, tau)
        h_new = project_feasible(y - lr * grad, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + (t / t_new - 1.0) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def subgrad_polish(h, n, max_iter, lr):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iter):
        conv = np.correlate(h, 1.0 - h, mode="full")
        c5 = np.max(conv) * (2.0 / n)
        idx = int(np.argmax(conv))
        shift = idx - (n - 1)
        grad = conv_grad_single(h, shift) * (2.0 / n)
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new > c5 + 1e-15:
            lr *= 0.5
            if lr < 1e-16:
                break
        else:
            lr = min(lr * 1.03, 0.01)
        h = h_new
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        best_h = np.array(initial_h_values, dtype=float).copy()
    else:
        best_h = np.full(320, 0.5)

    n0 = len(best_h)
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with progressive temperature reduction
    for tau in [0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.002, 0.001]:
        if tl() < 60:
            break
        def obj_pm(h, tv=tau):
            return softmax_obj(h, tv)
        def grad_pm(h, tv=tau):
            return softmax_grad(h, tv)
        try:
            res = minimize(obj_pm, best_h, jac=grad_pm, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n0,
                          options={'maxiter': 5000, 'ftol': 1e-18, 'gtol': 1e-14, 'maxcor': 30})
            ht = project_feasible(res.x, n0)
            c5t = compute_c5(ht)
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t
        except Exception:
            pass

    # Phase 2: Active-set boosting
    for rnd in range(6):
        if tl() < 120:
            break
        conv = np.correlate(best_h, 1.0 - best_h, mode="full")
        c5v = np.max(conv)
        threshold = c5v * 0.98
        active = np.where(conv >= threshold)[0]
        if len(active) == 0:
            break
        for tau in [0.005, 0.002]:
            if tl() < 30:
                break
            def obj_ab(h, act=active, tv=tau, boost=0.02 * c5v):
                n_ = len(h)
                dx_ = 2.0 / n_
                cv = np.correlate(h, 1.0 - h, mode="full")
                cv_b = cv.copy()
                cv_b[act] += boost
                m = np.max(cv_b)
                exps = np.exp((cv_b - m) / tv)
                return tv * (m / tv + np.log(np.sum(exps))) * dx_
            def grad_ab(h, act=active, tv=tau, boost=0.02 * c5v):
                n_ = len(h)
                dx_ = 2.0 / n_
                cv = np.correlate(h, 1.0 - h, mode="full")
                cv_b = cv.copy()
                cv_b[act] += boost
                m = np.max(cv_b)
                exps = np.exp((cv_b - m) / tv)
                w = exps / np.sum(exps)
                g = np.zeros(n_)
                for idx in range(len(cv)):
                    if w[idx] < 1e-11:
                        continue
                    shift = idx - (n_ - 1)
                    g += w[idx] * conv_grad_single(h, shift)
                return g * dx_
            try:
                res = minimize(obj_ab, best_h, jac=grad_ab, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n0,
                              options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-13})
                ht = project_feasible(res.x, n0)
                c5t = compute_c5(ht)
                if c5t < best_c5:
                    best_h, best_c5 = ht, c5t
            except Exception:
                pass

    # Phase 3: FISTA with very low temperature
    if tl() > 80:
        ht, c5t = fista_smooth(best_h, n0, tau=0.003, max_iter=20000, lr=0.01)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
        ht, c5t = fista_smooth(best_h, n0, tau=0.001, max_iter=20000, lr=0.005)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 4: Subgradient polishing
    if tl() > 80:
        ht, c5t = subgrad_polish(best_h, n0, max_iter=100000, lr=0.003)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    # Phase 5: Multi-resolution
    for n_try in [240, 360, 420, 480]:
        if tl() < 180:
            break
        nc = len(best_h)
        xo = np.linspace(0, 2.0, nc, endpoint=False)
        xn = np.linspace(0, 2.0, n_try, endpoint=False)
        hi = np.interp(xn, xo, best_h)
        hi = project_feasible(hi, n_try)
        for tau in [0.02, 0.005, 0.002]:
            if tl() < 40:
                break
            def obj_mr(h, tv=tau):
                return softmax_obj(h, tv)
            def grad_mr(h, tv=tau):
                return softmax_grad(h, tv)
            try:
                res = minimize(obj_mr, hi, jac=grad_mr, method='L-BFGS-B',
                              bounds=[(0.0, 1.0)] * n_try,
                              options={'maxiter': 3000, 'ftol': 1e-16, 'gtol': 1e-13})
                ht = project_feasible(res.x, n_try)
            except Exception:
                ht = hi
            c5t = compute_c5(ht)
            ht, c5t = fista_smooth(ht, n_try, tau=0.003, max_iter=10000, lr=0.008)
            ht, c5t2 = subgrad_polish(ht, n_try, max_iter=50000, lr=0.002)
            if c5t2 < c5t:
                c5t = c5t2
            if c5t < best_c5:
                best_h, best_c5 = ht, c5t

    # Phase 6: Structured perturbations
    for trial in range(50):
        if tl() < 25:
            break
        freq = rng.choice([1, 2, 3, 4, 5, 7, 8, 10, 12, 15])
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.01, 0.06)
        nc = len(best_h)
        x = np.linspace(0, 2.0, nc, endpoint=False)
        pert = amp * np.sin(2 * np.pi * freq * x + phase)
        if rng.random() < 0.3:
            pert += amp * 0.5 * np.sin(2 * np.pi * 2 * freq * x + rng.uniform(0, 2 * np.pi))
        hp = project_feasible(best_h + pert, nc)
        ht, c5t = fista_smooth(hp, nc, tau=0.01, max_iter=5000, lr=0.01)
        ht, c5t2 = subgrad_polish(ht, nc, max_iter=20000, lr=0.003)
        if c5t2 < best_c5:
            best_h, best_c5 = ht, c5t2

    # Phase 7: Final intensive polish
    if tl() > 40:
        ht, c5t = fista_smooth(best_h, n0, tau=0.001, max_iter=30000, lr=0.003)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t
    if tl() > 30:
        ht, c5t = subgrad_polish(best_h, n0, max_iter=150000, lr=0.001)
        if c5t < best_c5:
            best_h, best_c5 = ht, c5t

    return best_h, best_c5, len(best_h)
