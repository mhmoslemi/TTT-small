import warnings

    return float(np.max(c_values(h)))


def lag_grad(x, k):
    """Gradient of c(k) w.r.t. x, where k is shift index in [0, 2n-2]."""
    n = x.size
    dx = 2.0 / n
    s = int(k) - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        lo, hi = s, min(s + n - 1, n - 1)
        if lo <= hi:
            grad[lo:hi + 1] += (1.0 - x[lo - s:hi + 1 - s])
        lo2, hi2 = 0, min(n - 1 - s, n - 1)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= x[lo2 + s:hi2 + 1 + s]
    else:
        lo, hi = 0, min(n - 1 + s, n - 1)
        if lo <= hi:
            grad[lo:hi + 1] += (1.0 - x[lo - s:hi + 1 - s])
        lo2, hi2 = -s, n - 1
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= x[lo2 + s:hi2 + 1 + s]
    return grad * dx


def subgrad_max(x, topk=5):
    c = c_values(x)
    C = np.max(c)
    tol = max(1e-9, 1e-5 * C)
    active = np.where(c >= C - tol)[0]
    if len(active) > topk:
        active = np.argsort(c)[::-1][:topk]
    grad = sum(lag_grad(x, k) for k in active)
    return grad / len(active)


def smooth_max(x, tau, topk=None):
    n = x.size
    c = c_values(x)
    C = np.max(c)
    if tau <= 0:
        return C, np.zeros_like(c)
    z = (c - C) / tau
    z -= np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    if topk is not None and 0 < topk < len(c):
        idx = np.argsort(w)[::-1][:topk]
        w2 = np.zeros_like(w)
        w2[idx] = w[idx]
        ss = w2.sum()
        if ss > 1e-300:
            w = w2 / ss
    return C + tau * np.log(max(s, 1e-300)), w


def smooth_grad(x, tau, topk=None):
    _, w = smooth_max(x, tau, topk)
    n = x.size
    idxs = np.where(w > 1e-12)[0]
    grad = sum(w[i] * lag_grad(x, i) for i in idxs)
    return grad


def sa_optimize(x0, budget, seed, block_sizes=None):

    best_x = x.copy()
    best_c = exact_c(x)
    rng = np.random.RandomState(seed)
    n = x.size
    deadline = time.time() + budget
    if block_sizes is None:
        block_sizes = [1, 2, 3, 5, 8]
    T = 0.015
    cool = 0.997
    steps = 0
    while time.time() < deadline:
        if steps % 200 == 0:
            T *= cool
        # Block move: pick contiguous block, shift values
        bs = block_sizes[rng.randint(len(block_sizes))]
        bs = min(bs, n - 1)
        start = rng.randint(0, n - bs)
        delta = rng.normal(0, 0.03 * (1.0 + T * 50))
        old = x[start:start + bs].copy()
        new = np.clip(old + delta, 0, 1)
        x[start:start + bs] = new
        x = repair(x)
        c_new = exact_c(x)
        if c_new < best_c:
            best_c = c_new
            best_x = x.copy()
        elif rng.rand() > np.exp(-(c_new - best_c) / max(T, 1e-8)):
            x = repair(best_x.copy())
        steps += 1
    return best_x, best_c


def slsqp_optimize(x0, budget, seed):
    x = repair(x0.copy())
    best_x = x.copy()
    best_c = exact_c(x)
    deadline = time.time() + budget
    n = x.size
    bounds = [(0.0, 1.0)] * n

    def cons_fun(v):
        return np.array([v.sum() - 0.5 * n])

    def cons_jac(v):
        return np.ones((1, n))

    stages = [(0.03, 200), (0.01, 250), (0.004, 300), (0.001, 250)]
    for tau, mk in stages:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        if remaining < 5:
            break

        def fun(v):
            val, _ = smooth_max(v, tau)
            return val

        def jac(v):
            _, w = smooth_max(v, tau)
            idxs = np.where(w > 1e-12)[0]
            g = sum(w[i] * lag_grad(v, i) for i in idxs)
            return g

        try:
            res = so.minimize(fun, x, method="SLSQP", jac=jac,
                              bounds=bounds,
                              constraints=[{"type": "eq", "fun": cons_fun, "jac": cons_jac}],
                              options={"maxiter": mk, "ftol": 1e-12})
            if res is not None:
                xr = repair(res.x)
                cr = exact_c(xr)

def resample_to(h0, n_new):
