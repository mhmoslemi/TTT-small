import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2.0, 0.0, 1.0)

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def soft_gradient(h, corr, dx, alpha):
    N = len(h)
    g = np.zeros(N)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        idx = np.argmax(corr)
        w = np.zeros(len(corr))
        w[idx] = 1.0
    else:
        w /= wsum
    for j in range(len(w)):
        if w[j] < 1e-7:
            continue
        k = j - (N - 1)
        wk = w[j]
        if k >= 0:
            N1k = N - k
            g[0:N1k] += wk * (1.0 - h[k:N])
            g[k:N] -= wk * h[0:N1k]
        else:
            kp = -k
            g[kp:N] += wk * (1.0 - h[0:N-kp])
            g[0:N-kp] -= wk * h[kp:N]
    return g * dx

def interpolate_to_n(h_src, n_dst):
    x0 = np.arange(len(h_src)) * 2.0 / len(h_src)
    x1 = np.arange(n_dst) * 2.0 / n_dst
    return project_feasible(np.interp(x1, x0, h_src), n_dst)

def adam_opt(h, N, rng, t_end, try_update, max_iter, alpha0, alpha_min, lr0, b1, b2):
    m = np.zeros(N)
    v = np.zeros(N)
    t = 0
    alpha = alpha0
    val, corr, dx = obj_and_corr(h)
    try_update(h, val)
    best_val, best_h = val, h.copy()
    lr = lr0
    stag = 0
    for _ in range(max_iter):
        if time.time() > t_end:
            break
        t += 1
        alpha = max(alpha * 0.9997, alpha_min)
        g = soft_gradient(h, corr, dx, alpha)
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mhat = m / (1.0 - b1 ** t)
        vhat = v / (1.0 - b2 ** t)
        h_new = project_feasible(h - lr * mhat / (np.sqrt(vhat) + 1e-8), N)
        vn, cn, _ = obj_and_corr(h_new)
        if vn < val - 1e-16:
            h, val, corr = h_new, vn, cn
            stag = 0
            if val < best_val:
                best_val, best_h = val, h.copy()
                try_update(best_h, best_val)
        else:
            stag += 1
            lr *= 0.94
            if stag > 40:
                sc = 0.03 + 0.04 * min(stag / 150.0, 1.0)
                h = project_feasible(h + rng.randn(N) * sc, N)
                val, corr, _ = obj_and_corr(h)
                m, v, t, lr, stag = np.zeros(N), np.zeros(N), 0, lr0, 0
    try_update(best_h, best_val)
    return best_h, best_val
