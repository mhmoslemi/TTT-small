import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx), corr * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def get_subgradient(h, k_star):
    n = len(h)
    shift = k_star - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        g[shift:] = (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[s:] = (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    return g

def projected_subgradient(h0, target_sum, max_iter=5000, alpha0=0.02, 
                          decay=0.9999, momentum=0.7, top_k=1):
    n = len(h0)
    h = project_feasible(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    v = np.zeros(n)
    
    for it in range(max_iter):
        c5, corr_vals = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Get top-k subgradients
        if top_k == 1:
            k_star = np.argmax(corr_vals)
            g = get_subgradient(h, k_star)
        else:
            top_indices = np.argsort(corr_vals)[-top_k:]
            g = np.zeros(n)
            for ki in top_indices:
                g += get_subgradient(h, ki)
            g /= top_k
        
        v = momentum * v + (1 - momentum) * g
        h = project_feasible(h - alpha0 * v, target_sum)
        alpha0 *= decay
    
    return best_h, best_c5

def smooth_max_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = temp * (corr - m)
    return m + np.log(np.sum(np.exp(np.clip(shifted, -500, 0)))) / temp

def optimize_slsqp(h_start, temp, maxiter=500, ftol=1e-15):
    n = len(h_start)
    bnds = [(0.0, 1.0)] * n
    ts = n / 2.0
    res = minimize(smooth_max_obj, h_start, args=(temp,), method='SLSQP',
                   bounds=bnds,
                   constraints=[{'type': 'eq', 'fun': lambda x: x.sum() - ts,
                                  'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return project_feasible(res.x, ts)

def try_binary_threshold(h):
    """Try rounding to binary and see if it helps."""
    n = len(h)
    # Sort by value, take top n/2 as 1s
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx, corr * dx

def project_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    for _ in range(200):
        d = target_sum - h.sum()
        if abs(d) < 1e-15:
            break
        mask = (h > 0) & (h < 1)
        if mask.sum() == 0:
            break
        h[mask] += d / mask.sum()
        np.clip(h, 0.0, 1.0, out=h)
    return h

def get_subgradient(h, k_star):
    n = len(h)
    shift = k_star - (n - 1)
    g = np.zeros(n)
    if shift >= 0:
        g[shift:] = (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] = (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def projected_subgradient(h0, target_sum, max_iter=5000, alpha0=0.02,
                          decay=0.9999, momentum=0.5, top_k=1, noise=0.0, seed=0):
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = project_feasible(h0.copy(), target_sum)
    best_h = h.copy()
    best_c5, _ = compute_c5(h)
    v = np.zeros(n)
    a = alpha0
    for it in range(max_iter):
        c5, corr_vals = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        if top_k == 1:
            g = get_subgradient(h, np.argmax(corr_vals))
        else:
            top_idx = np.argsort(corr_vals)[-top_k:]
            g = sum(get_subgradient(h, ki) for ki in top_idx) / top_k
        if noise > 0:
            g += noise * rng.randn(n)
        v = momentum * v + (1.0 - momentum) * g
        h = project_feasible(h - a * v, target_sum)
        a *= decay
    return best_h, best_c5

def smooth_max_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    m = np.max(corr)
    shifted = temp * (corr - m)
    return m + np.log(np.sum(np.exp(np.clip(shifted, -500, 0)))) / temp

def optimize_slsqp(h_start, temp, maxiter=500):
    n = len(h_start)
    ts = n / 2.0
    bnds = [(0.0, 1.0)] * n
    res = minimize(smooth_max_obj, h_start, args=(temp,), method='SLSQP',
                   bounds=bnds,
                   constraints=[{'type': 'eq',
                                 'fun': lambda x: x.sum() - ts,
                                 'jac': lambda x: np.ones(n)}],
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return project_feasible(res.x, ts)

def try_binary_threshold(h):
    n = len(h)
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin
