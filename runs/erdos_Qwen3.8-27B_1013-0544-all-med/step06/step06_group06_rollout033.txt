import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, n):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def compute_slopes(h, j, n):
    """Compute slope of each correlation c[m] with respect to h[j]."""
    slopes = np.zeros(2 * n - 1)
    # m ranges from j to min(j+n-1, 2n-2)
    m_lo = j
    m_hi = min(j + n - 1, 2 * n - 2)
    for m in range(m_lo, m_hi + 1):
        d = m - (n - 1)
        s = 0.0
        # h[j] as first factor: contributes (1 - h[j+d]) if valid
        jd = j + d
        if 0 <= jd < n:
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo <= j <= i_hi:
                s += (1.0 - h[jd])
        # h[j] as second factor: contributes -h[j-d] if valid
        jd2 = j - d
        if 0 <= jd2 < n:
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo <= jd2 <= i_hi:
                s += (-h[jd2])
        slopes[m] = s
    return slopes

def minimize_1d_max(a, b):
    """Find t in [0,1] minimizing max_i (a[i] + b[i]*t)."""
    # Ternary search (works for convex functions)
    lo, hi = 0.0, 1.0
    for _ in range(80):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        f1 = np.max(a + b * m1)
        f2 = np.max(a + b * m2)
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    return (lo + hi) / 2.0

def coordinate_descent(h, n, n_sweeps=100, seed=42, time_limit=None, t0=None):
    """Run coordinate descent. Returns improved h."""
    rng = np.random.RandomState(seed)
    h = h.copy()
    c = np.correlate(h, 1 - h, mode='full')
    
    for sweep in range(n_sweeps):
        if time_limit is not None and time.time() - t0 > time_limit:
            break
        order = rng.permutation(n)
        for j in order:
            slopes = compute_slopes(h, j, n)
            a = c - slopes * h[j]
            b = slopes
            t_opt = minimize_1d_max(a, b)
            c += slopes * (t_opt - h[j])
            h[j] = t_opt
        h = project_feasible(h, n)
        c = np.correlate(h, 1 - h, mode='full')
    
    return h

def obj_smooth(h, n, alpha):
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def sum_con_jac(x, n):
    return np.ones_like(x)

def coordinate_update(h, c, j, n):
    """Update h[j] to optimal value. c is current correlation (without dx)."""
    # Non-zero slope range for m
    m_lo = j
    m_hi = min(j + n - 1, 2 * n - 2)
    
    # Max of inactive correlations (m outside [m_lo, m_hi])
    if m_lo > 0:
        max_inactive_left = c[:m_lo].max()
    else:
        max_inactive_left = -np.inf
    if m_hi < 2 * n - 2:
        max_inactive_right = c[m_hi+1:].max()
    else:
        max_inactive_right = -np.inf
    max_inactive = max(max_inactive_left, max_inactive_right)
    
    # Compute slopes for active m range
    slopes = np.zeros(m_hi - m_lo + 1)
    for idx, m in enumerate(range(m_lo, m_hi + 1)):
        d = m - (n - 1)
        s = 0.0
        jd = j + d
        if 0 <= jd < n:
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo <= j <= i_hi:
                s += (1.0 - h[jd])
        jd2 = j - d
        if 0 <= jd2 < n:
            i_lo = max(0, -d)
            i_hi = min(n - 1, n - 1 - d)
            if i_lo <= jd2 <= i_hi:
                s += (-h[jd2])
        slopes[idx] = s
    
    # For active m: c[m] as function of t = (c[m] - slopes[m]*h[j]) + slopes[m]*t
    a = c[m_lo:m_hi+1] - slopes * h[j]
    b = slopes
    
    # Minimize max(max_inactive, max_i(a[i] + b[i]*t)) over t in [0,1]
    # = minimize max(max_inactive, g(t)) where g(t) = max_i(a[i] + b[i]*t)
    # If max_inactive ≥ max(g(t)) for all t, then the max is just max_inactive (constant)
    # In that case, any t works (but we still want to minimize g(t) for future steps)
    
    t_opt = minimize_1d_max_full(a, b, max_inactive)
    
    # Update c
    c[m_lo:m_hi+1] += slopes * (t_opt - h[j])
    h[j] = t_opt
    
    return h, c

    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = np.clip(h.copy(), 0.0, 1.0)
    for _ in range(200):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def compute_slopes(h, j, n):
    slopes = np.zeros(2 * n - 1)
    m_lo = j
    m_hi = min(j + n - 1, 2 * n - 2)
    for m in range(m_lo, m_hi + 1):
        d = m - (n - 1)
        s = 0.0

            if i_lo <= jd2 <= i_hi:
                s += -h[jd2]
        slopes[m] = s

        m2 = hi - (hi - lo) / 3.0
        if np.max(a + b * m1) < np.max(a + b * m2):
            hi = m2
        else:
