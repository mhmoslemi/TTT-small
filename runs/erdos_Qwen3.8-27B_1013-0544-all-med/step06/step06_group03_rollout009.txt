            best_h = h_new.copy()

`np.correlate(a, v, mode='full')` computes:
c[k] = ∑_n a[n+k] * conj(v[n])

For real arrays: c[k] = ∑_n a[n+k] * v[n]

So `np.correlate(h, 1-h, mode='full')` gives:
c[k] = ∑_n h[n+k] * (1-h[n])

The index k ranges from -(n-1) to n-1.

For k=0: c[0] = ∑_n h[n] * (1-h[n]) = ∑_n (h[n] - h[n]²) = sum(h) - sum(h²)

For k>0: c[k] = ∑_{n=0}^{n-1-k} h[n+k] * (1-h[n])

So the overlap at "shift k" in the problem is c[k]*dx for k in the full correlation.

The maximum of this over all k is our C₅.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(conv) * dx)

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        if shift > 0:
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def soft_max_and_grad(h, t, min_w=1e-12):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    logsum = np.log(np.sum(np.exp((conv - m) / t)))
    obj = (m + t * logsum) * dx
    weights = np.exp((conv - m) / t - logsum)
    grad = np.zeros(n)
    active = weights > min_w
    idxs = np.where(active)[0]
    for i in idxs:
        w = weights[i]
        shift = i - (n - 1)
        grad += w * _conv_grad_single(h, shift)
    grad *= dx
    return obj, grad

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def nesterov_pgd(h_init, max_iter, t_soft, lr_init):
    n = len(h_init)
    h = h_init.copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_mom = 1.0
    lr = lr_init
    t = t_soft
    for it in range(max_iter):
        beta = (t_mom - 1.0) / (t_mom + 1.0)
        y = h + beta * (h - h_prev)
        y = project_feasible(y, n)
        _, grad = soft_max_and_grad(y, t)
        h_prev = h.copy()
        h_new = project_feasible(y - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        c5_y = compute_c5(y)
        if c5_new > c5_y + 1e-14:
            lr *= 0.5
            h = y.copy()
            if lr < 1e-16:
                break
        else:
            h = h_new
            lr = min(lr * 1.01, 0.008)
        if it % 2000 == 1999 and t > 0.001:
            t *= 0.8
        t_mom = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom**2))
    return best_h, best_c5
