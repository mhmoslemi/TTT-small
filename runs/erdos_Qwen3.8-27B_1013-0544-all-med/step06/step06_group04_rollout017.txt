    n = _cur_n
    c5s = smooth_c5(h)
    sv = float(np.sum(h)) - n / 2.0
    return c5s + 10000.0 * sv * sv

    return float(np.sum(x) - _cur_n / 2.0)

def fix_sum(h, n):
    target = n / 2.0
    h = h.copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:

        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    return np.interp(x_new, x_old, h, left=h[0], right=h[-1])
