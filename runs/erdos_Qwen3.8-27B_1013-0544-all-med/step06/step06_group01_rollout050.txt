            else:

    n = len(h)
    dx = 2.0 / n
    C = _circ_corr(h)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5*n)**2

    H = np.fft.fft(h)
    W = np.fft.fft(w)
    G = np.zeros(n, dtype=complex)

    G[0] += 2.0 * beta * (np.sum(h) - 0.5*n) * n

                   options={'maxiter': maxit, 'ftol': 1e-15, 'gtol': 1e-10})
    return _fix_sum(res.x if res.x is not None else h, n)

    C = np.correlate(h, 1.0 - h, mode='full') * dx

    thresh = max_c * (1 - tol_frac)

        top = np.array([np.argmax(C)])
    if len(top) > num_top:
        top = top[np.argsort(-C[top])][:num_top]

                grad[i + L] += -dx * h[i]

            for i in range(M, n):
                grad[i] += dx * (1.0 - h[i - M])
                grad[i - M] += -dx * h[i]

import numpy as np
from scipy.optimize import minimize
import time

def _c5_max(h):
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(C))

def _fix_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n
    d = target - np.sum(h)
    if abs(d) > 1e-12:
        h = h + d / n
        h = np.clip(h, 0.0, 1.0)
        d2 = target - np.sum(h)
        if abs(d2) > 1e-10:
            if d2 > 0:
                idx = np.argmax(1.0 - h)
                h[idx] = min(1.0, h[idx] + d2)
            else:
                idx = np.argmax(h)
                h[idx] = max(0.0, h[idx] + d2)
    return h

def _circ_corr(h):
    n = len(h)
    dx = 2.0 / n
    H = np.fft.fft(h)
    return dx * np.sum(h) - np.fft.ifft(np.abs(H) ** 2).real * dx

def _sm_obj(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _circ_corr(h)
    sh = alpha * C
    m = np.max(sh)
    es = np.exp(sh - m)
    se = np.sum(es)
    f = (m + np.log(se)) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    H = np.fft.fft(h)
    W = np.fft.fft(w)
    G = np.zeros(n, dtype=complex)
    G -= dx * (W + np.conj(W)) * H
    G[0] += 2.0 * beta * (np.sum(h) - 0.5 * n) * n
    g = np.fft.ifft(G).real
    return f, g

def _opt_fft(h, alpha, beta, maxit=3000):
    n = len(h)
    res = minimize(_sm_obj, h.astype(float).copy(), args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxit, 'ftol': 1e-16, 'gtol': 1e-12})
    if res.x is not None:
        return _fix_sum(res.x, n)
    return _fix_sum(h, n)

def _top_lag_subgrad(h, n, num_top=5, tol_frac=0.03):
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = float(np.max(C))
    thresh = max_c * (1.0 - tol_frac)
    top = np.where(C >= thresh)[0]
    if len(top) == 0:
        top = np.array([int(np.argmax(C))])
    if len(top) > num_top:
        top = top[np.argsort(-C[top])][:num_top]
    grad = np.zeros(n)
    for k in top:
        L = int(k) - (n - 1)
        if L >= 0:
            idx1 = np.arange(n - L)
            grad[idx1] += dx * (1.0 - h[idx1 + L])
            idx2 = np.arange(L, n)
            grad[idx2] += -dx * h[idx2 - L]
        else:
            M = -L
            idx1 = np.arange(M, n)
            grad[idx1] += dx * (1.0 - h[idx1 - M])
            idx2 = np.arange(n - M)
            grad[idx2] += -dx * h[idx2 + M]
    grad /= float(len(top))
    return grad, max_c

def _subgrad_momentum(h, n, maxit=10000, lr_init=0.03, decay=0.9998, gamma=0.9):
    h = h.astype(float).copy()
    v = np.zeros(n)
    best = _c5_max(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(maxit):
        grad, cur = _top_lag_subgrad(h, n, num_top=5, tol_frac=0.03)
        v = gamma * v - lr * grad
        h_new = _fix_sum(h + v, n)
        c_new = _c5_max(h_new)
        if c_new < best:
            best = c_new
            best_h = h_new.copy()
            h = h_new
        lr *= decay
    return best_h, best

def _swap_polish(h, n, rng, max_iters=5000):
    h = h.astype(float).copy()
    best = _c5_max(h)
    for _ in range(max_iters):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        if i == j:
            continue
        for delta in (0.01, 0.02, 0.05, 0.1, 0.2):
            h_try = h.copy()
            hi = h_try[i] + delta
            hj = h_try[j] - delta
            if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                h_try[i] = hi
                h_try[j] = hj
                c_try = _c5_max(h_try)
                if c_try < best - 1e-11:
                    h = h_try
                    best = c_try
                    break
    return h, best
