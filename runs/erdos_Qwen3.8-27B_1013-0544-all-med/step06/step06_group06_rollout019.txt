    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(h, n):
    target_sum = n / 2.0
    h = h.copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        diff = target_sum - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-14:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-14:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def coordinate_sweep(h, n, n_grid=200, max_sweeps=100, tol=1e-12):

    ks = np.arange(2 * n - 1)
    ms = ks - (n - 1)
    k0 = n - 1
    
    prev_c = compute_c5(h, n)
    for sweep in range(max_sweeps):

        for j in range(n):

            h[j] = 0.0
            c_base = np.correlate(h, 1 - h, mode='full') * dx
            
            coeff = np.zeros(2 * n - 1)
            valid1 = (j + ms >= 0) & (j + ms < n)
            coeff[valid1] = (1.0 - h[j + ms[valid1]])
            valid2 = (j - ms >= 0) & (j - ms < n)
            coeff[valid2] -= h[j - ms[valid2]]
            
            x_vals = np.linspace(0.0, 1.0, n_grid)
            c_vals = c_base[None, :] + x_vals[:, None] * coeff[None, :]
            c_vals[:, k0] -= x_vals ** 2
            
            c_max = c_vals.max(axis=1)

            best_c = c_max[best_idx]

        if not improved or (prev_c - cur_c) < tol:
            prev_c = cur_c
            if not improved:
                break
        prev_c = cur_c
    return h, compute_c5(h, n)

    c = np.correlate(h, 1 - h, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 700:
        return m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha
    return m + np.log(np.sum(np.exp(shifted))) / alpha

    return np.sum(h) - n / 2.0

    return np.ones(n)

    """Interpolate h from n_src points to n_dst points."""
    x_src = np.linspace(0, 2, n_src, endpoint=False) + 1.0/n_src
    x_dst = np.linspace(0, 2, n_dst, endpoint=False) + 1.0/n_dst
    # For step functions, use midpoint values
    x_src_edges = np.linspace(0, 2, n_src + 1)
    x_dst_edges = np.linspace(0, 2, n_dst + 1)
    h_dst = np.zeros(n_dst)
    for i in range(n_dst):
        # Find which source intervals overlap with destination interval i
        lo = x_dst_edges[i]
        hi = x_dst_edges[i+1]
        # Sum of h_src * overlap_length for overlapping source intervals
        total = 0.0
        overlap_len = 0.0
        for j in range(n_src):

            if o > 0:

                overlap_len += o
        h_dst[i] = total / overlap_len if overlap_len > 0 else 0.5
    return h_dst

def coord_sweep(h, n, n_grid, max_sweeps, tol=1e-13):
    dx = 2.0 / n
    ks = np.arange(2 * n - 1)
    ms = ks - (n - 1)
    k0 = n - 1
    prev_c = compute_c5(h, n)
    for _ in range(max_sweeps):
        for j in range(n):
            h[j] = 0.0
            c_base = np.correlate(h, 1 - h, mode='full') * dx
            coeff = np.zeros(2 * n - 1)
            v1 = (j + ms >= 0) & (j + ms < n)
            coeff[v1] = 1.0 - h[j + ms[v1]]
            v2 = (j - ms >= 0) & (j - ms < n)
            coeff[v2] -= h[j - ms[v2]]
            xg = np.linspace(0.0, 1.0, n_grid)
            cv = c_base[None, :] + xg[:, None] * coeff[None, :]
            cv[:, k0] -= xg * xg
            cm = cv.max(axis=1)
            h[j] = xg[cm.argmin()]
        h = project_feasible(h, n)
