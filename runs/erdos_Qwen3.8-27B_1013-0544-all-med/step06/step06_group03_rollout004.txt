import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def conv_grad_at(h, shift, n):
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
            g[:n - shift] -= h[shift:]
    else:
        s = -shift
        if s < n:
            g[:n - s] += (1.0 - h[s:])
            g[s:] -= h[:n - s]
    return g

def topk_obj_grad(h, k):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    kk = min(k, len(conv))
    idx = np.argpartition(conv, -kk)[-kk:]
    grad = np.zeros(n)
    for i in idx:
        shift = i - (n - 1)
        grad += conv_grad_at(h, shift, n)
    val = np.mean(conv[idx]) * dx
    grad = grad * dx / kk
    return val, grad

def softmax_obj_grad(h, temp):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / temp)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    top_idx = np.argsort(weights)[-max(20, n):]
    for i in top_idx:
        if weights[i] < 1e-10:
            continue
        shift = i - (n - 1)
        grad += weights[i] * conv_grad_at(h, shift, n)
    val = temp * (m / temp + np.log(np.sum(exps))) * dx
    grad *= dx
    return val, grad

def adam_optimize(h_init, max_iter, lr_init, k, b1=0.9, b2=0.999, eps=1e-8,
                  lr_decay=0.9995, time_limit=300, t0=0.0, use_softmax=False, temp=0.01):
    n = len(h_init)
    h = h_init.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stalled = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if use_softmax:
            val, grad = softmax_obj_grad(h, temp)
        else:
            val, grad = topk_obj_grad(h, k)
        t += 1
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad ** 2)
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        h_new = project_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
        h = h_new
        if stalled > 1000:
            lr *= 0.85
        elif stalled > 200:
            lr *= 0.98
        else:
            lr = min(lr * lr_decay, 0.1)
        if lr < 1e-6:
            break
    return best_h, best_c5

def subgrad_optimize(h_init, max_iter, lr_init, time_limit=300, t0=0.0, decay=0.999):
    n = len(h_init)
    h = h_init.copy()
    dx = 2.0 / n
    lr = lr_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stalled = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        conv = np.correlate(h, 1.0 - h, mode="full")
        idx = np.argmax(conv)
        shift = idx - (n - 1)
        grad = conv_grad_at(h, shift, n) * dx
        h_new = project_feasible(h - lr * grad, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_new.copy()
            stalled = 0
        else:
            stalled += 1
        h = h_new
        if stalled > 200:
            lr *= 0.9
        else:
            lr = max(lr * decay, 1e-7)
    return best_h, best_c5

def coord_descent_polish(h, max_passes=20, time_limit=60, t0=0.0):
    n = len(h)
    best_c5 = compute_c5(h)
    for p in range(max_passes):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - t0 > time_limit:
                break
            old = h[i]
            for trial in [0.0, 1.0, 0.25, 0.5, 0.75]:
                if abs(trial - old) < 1e-4:
                    continue
                h[i] = trial
                s = h.sum()
                target = n / 2.0
                if s != target:
                    diff = target - s
                    if -1e-10 <= diff <= 1e-10:
                        pass
                    else:
                        j = (i + 1) % n
                        adj = diff / (n - 1) if n > 1 else 0
                        h2 = h.copy()
                        h2[j] = np.clip(h2[j] + adj, 0.0, 1.0)
                        h2[i] = trial
                        h2 = project_feasible(h2, n)
                else:
                    h2 = h.copy()
                c5_trial = compute_c5(h2)
                if c5_trial < best_c5 - 1e-10:
                    h = h2.copy()
                    best_c5 = c5_trial
                    improved = True
                    break
            h[i] = old
        if not improved:
            break
    return h, best_c5
