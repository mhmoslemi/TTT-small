def project_onto_constrained(v, n):
    """Project v onto {h : 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo = -1.0
    hi = max(v) + 1.0
    for _ in range(100):

        s = np.sum(_candidate)
        if s > target:
            lo = mid
        else:
            hi = mid
    _candidate = np.clip(v - (lo+hi)/2.0, 0.0, 1.0)
    # Fix rounding
    diff = target - np.sum(_candidate)
    if abs(diff) > 1e-10:
        # Add to elements with most room
        if diff > 0:
            room = 1.0 - _candidate
            order = np.argsort(-room)
            for idx in order:
                add = min(diff, room[idx])
                _candidate[idx] += add
                diff -= add
                if diff <= 1e-12:
                    break
        else:
            order = np.argsort(_candidate)
            for idx in order:
                sub = min(-diff, _candidate[idx])
                _candidate[idx] -= sub
                diff += sub
                if diff >= -1e-12:
                    break
    return _candidate

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    # Soft-max weights
    c_max = np.max(c)
    weights = np.exp((c - c_max) / temperature)
    weights /= np.sum(weights)
    # Compute subgradient
    # For each lag k (stored at index k + n - 1 in the full correlation),
    # ∂C[k]/∂h[m] = (1-h[m-k])*I[0<=m-k<n] - h[m+k]*I[0<=m+k<n]

        if m_lo2 <= m_hi2:
            m = np.arange(m_lo2, m_hi2+1)
            grad[m] -= w * h[m + k]
    return dx * grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(v, n):
    target = n / 2.0
    lo, hi = v.min() - 2.0, v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argmax(1.0 - h)
            h[idx] = min(h[idx] + diff, 1.0)
        else:
            idx = np.argmin(h)
            h[idx] = max(h[idx] + diff, 0.0)
    return h

def compute_subgrad(h, temp=0.02):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    w = np.exp((c - c.max()) / temp)
    ws = w.sum()
    if ws < 1e-30:
        return np.zeros(n)
    w /= ws
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        if w[idx] < 1e-10:
            continue
        k = idx - (n - 1)
        wi = w[idx]
        if k >= 0:
            m = np.arange(k, n)
            grad[m] += wi * (1.0 - h[m - k])
            m2 = np.arange(0, n - k)
            grad[m2] -= wi * h[m2 + k]
        else:
            kp = -k
            m = np.arange(0, n - kp)
            grad[m] += wi * (1.0 - h[m + kp])
            m2 = np.arange(kp, n)
            grad[m2] -= wi * h[m2 - kp]
    return dx * grad

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_h(h_new, n_new)

    h = project_h(h0.astype(float), n)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    rng = np.random.RandomState(seed)
    lr = lr_init
    temp = 0.03
    no_improve = 0
    max_no_improve = 500

        for _ in range(batch):
            if time.time() - start >= max_time:
                break
            g = compute_subgrad(h, temp)
            h_new = project_h(h - lr * g, n)
            new_c5 = compute_c5(h_new)
            if new_c5 < current_c5:

                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                no_improve = 0
            else:

                if rng.random() < np.exp(-(new_c5 - current_c5) / max(0.02 * np.exp(-no_improve / 100), 1e-8)):

            no_improve = 0
            lr = max(lr, 0.02)
            temp = max(temp, 0.02)

                break

    return best_h, best_c5

def local_refine(h0, max_time, seed=42):
    n = len(h0)
    h = project_h(h0.astype(float), n)
    current_c5 = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(seed)
    stall = 0

    while time.time() - start < max_time:
        g = compute_subgrad(h, temp=0.005)
        best_gain = 1e-14
        best_move = None
        neg_idx = np.argsort(g)[:15]
        pos_idx = np.argsort(-g)[:15]
        fracs = [0.02, 0.05, 0.1, 0.2, 0.4]

        for i in neg_idx:
            if h[i] >= 0.99:
                continue
            for j in pos_idx:
                if h[j] <= 0.01 or i == j:
                    continue
                max_amt = min(1.0 - h[i], h[j])
                for frac in fracs:
                    amt = frac * max_amt
                    if amt < 1e-6:
                        continue
                    h[i] += amt
                    h[j] -= amt
                    new_c5 = compute_c5(h)
                    gain = current_c5 - new_c5
                    if gain > best_gain:
                        best_gain = gain
                        best_move = (i, j, amt)
                    h[i] -= amt
                    h[j] += amt

        if best_move is not None:
            i, j, amt = best_move
            h[i] = min(h[i] + amt, 1.0)
            h[j] = max(h[j] - amt, 0.0)
            current_c5 = compute_c5(h)
            stall = 0
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j:
                amt = rng.uniform(0.01, 0.3)
                amt = min(amt, 1.0 - h[i], h[j])
                if amt > 1e-5:
                    h[i] += amt
                    h[j] -= amt
                    new_c5 = compute_c5(h)
                    if new_c5 < current_c5:
                        current_c5 = new_c5
                        stall = 0
                    else:
                        h[i] -= amt
                        h[j] += amt
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1
            if stall > 30:
                break

    return h, current_c5

def simulated_anneal_cf(h0, max_time, seed=42):
    n = len(h0)
    h = h0.astype(float).copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start = time.time()
    rng = np.random.RandomState(seed)
    T0 = 0.03
    Tend = 0.0005

    while time.time() - start < max_time:
        t_frac = (time.time() - start) / max_time
        T = T0 * (Tend / T0) ** t_frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        amt = rng.uniform(0.0, 0.4)
        if rng.random() < 0.5:
            amt = min(amt, 1.0 - h[i], h[j])
        else:
            amt = -min(amt, h[i], 1.0 - h[j])
        if abs(amt) < 1e-7:
            continue
        h[i] += amt
        h[j] -= amt
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] -= amt
            h[j] += amt

    return best_h, best_c5

def make_random_binary(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def rem():
        return budget_s - (time.time() - t0)

    h_init = np.array(initial_h_values, dtype=float).copy()
    n_init = len(h_init)
    h_init = project_h(h_init, n_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n_init

    # Phase 1: Subgradient at initial resolution
    t1 = min(120.0, rem() * 0.2)
    h1, c5_1 = subgradient_descent(h_init, t1, seed=seed, lr_init=0.1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()
        best_n = n_init

    # Phase 2: Local refine
    if rem() > 40:
        t2 = min(80.0, rem() * 0.15)
        h2, c5_2 = local_refine(best_h, t2, seed=seed + 100)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Try random binary restarts at n_init
    for rs in range(3):
        if rem() < 60:
            break
        h_rand = make_random_binary(n_init, seed=seed + 500 + rs * 7)
        t3 = min(30.0, rem() * 0.08)
        h_r, c5_r = subgradient_descent(h_rand, t3, seed=seed + 600 + rs, lr_init=0.15)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
            best_n = n_init

    # Phase 4: Higher resolutions
    n_list = [n_init * 2, n_init * 3]
    for n_try in n_list:
        if rem() < 80:
            break
        h_interp = interpolate_h(best_h, n_try)
        t4 = min(80.0, rem() * 0.15)
        h_hi, c5_hi = subgradient_descent(h_interp, t4, seed=seed + n_try, lr_init=0.06)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
            best_n = n_try
        if rem() > 40:
            t5 = min(40.0, rem() * 0.1)
            h_ref, c5_ref = local_refine(h_hi, t5, seed=seed + n_try + 300)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
                best_n = n_try

    # Phase 5: SA on best
    if rem() > 60:
        t6 = min(200.0, rem() * 0.35)
        h_sa, c5_sa = simulated_anneal_cf(best_h, t6, seed=seed + 1000)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 6: Final local refine
    if rem() > 20:
        t7 = min(rem() * 0.8, 120.0)
        h_f, c5_f = local_refine(best_h, t7, seed=seed + 2000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
