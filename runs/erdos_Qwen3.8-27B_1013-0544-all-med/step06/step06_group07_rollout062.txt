```python
    best_h = h.copy()
```

grad[i] = sum_k w[k] * [(1-h[i-d_k])*dx * 1{0≤i-d_k<n} - h[i+d_k]*dx * 1{0≤i+d_k<n}]

where d_k = k - (n-1) is the offset for the k-th correlation value.

This is O(n * |active|) where |active| is the number of significant weights. For n=70 and |active|≤10, that's 700 operations. Fast.

```python
import numpy as np
import time
import scipy.optimize as so
import warnings

warnings.simplefilter("ignore")

def time_left(deadline):
    return max(0.0, deadline - time.time())

def repair(h):
    x = np.array(h, dtype=float, copy=True)
    n = x.size
    if n == 0:
        return x
    target = 0.5 * float(n)
    np.clip(x, 0.0, 1.0, out=x)
    for _ in range(50):
        diff = target - x.sum()
        if abs(diff) <= 1e-14:
            break
        if diff > 0:
            slack = 1.0 - x
            mask = slack > 1e-14
            if not np.any(mask):
                break
            x[mask] += diff * slack[mask] / slack[mask].sum()
        else:
            mask = x > 1e-14
            if not np.any(mask):
                break
            x[mask] += diff * x[mask] / x[mask].sum()
        np.clip(x, 0.0, 1.0, out=x)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        k = int(np.floor(target))
        x[:] = 0.0
        x[:k] = 1.0
        rem = target - k
        if rem > 0 and k < n:
            x[k] = rem
    return x

def c_values(h):
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def exact_c(h):
    return float(np.max(c_values(h)))

def resample_to(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    if n0 == n_new:
        return repair(h0)
    dt0 = 2.0 / float(n0)
    x0 = (np.arange(n0) + 0.5) * dt0
    x1 = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x1, x0, h0)
    return repair(y)

def sa_optimize(h0, time_limit, seed):
    h = h0.copy()
    n = h.size
    dx = 2.0 / n
    c1 = c_values(h).copy()
    c2 = np.zeros_like(c1)
    c, other = c1, c2
    C = float(np.max(c))
    rng = np.random.RandomState(seed)
    best_c = C
    best_h = h.copy()
    deadline = time.time() + time_limit
    T = 0.025
    T_min = 0.0002
    cool_rate = 0.99985
    step_scale = 0.05
    accept_count = 0
    window = 0
    omf = (1.0 - h[::-1]) * dx
    hdx = h * dx
    dc = np.zeros(2 * n - 1)
    no_imp = 0
    imp_threshold = 20000
    while time_left(deadline) > 0.3:
        if window >= 2000:
            rate = accept_count / window
            if rate < 0.3:
                step_scale = max(0.01, step_scale * 0.82)
            elif rate > 0.65:
                step_scale = min(0.2, step_scale * 1.18)
            accept_count = 0
            window = 0
        if no_imp > 0 and no_imp % imp_threshold == 0:
            h = best_h.copy()
            c[:] = 0
            np.add(best_h * 0.0, best_h * 0.0, out=c)
            c = c_values(best_h).copy()
            C = float(np.max(c))
            omf[:] = (1.0 - h[::-1]) * dx
            hdx[:] = h * dx
            T = max(T, 0.015)
            no_imp = 0
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        di_max = min(1.0 - h[i], h[j])
        if di_max < 1e-14:
            continue
        di = rng.uniform(-1.0, 1.0) * min(step_scale, di_max)
        if abs(di) < 1e-14:
            continue
        dj = -di
        dc[:] = 0.0
        dc[i:i + n] += di * omf
        dc[n - 1 - i:2 * n - 1 - i] -= di * hdx
        dc[j:j + n] += dj * omf
        dc[n - 1 - j:2 * n - 1 - j] -= dj * hdx
        np.add(c, dc, out=other)
        new_C = float(other.max())
        window += 1
        if new_C <= C:
            h[i] += di
            h[j] += dj
            c, other = other, c
            C = new_C
            accept_count += 1
            omf[n - 1 - i] = (1.0 - h[i]) * dx
            omf[n - 1 - j] = (1.0 - h[j]) * dx
            hdx[i] = h[i] * dx
            hdx[j] = h[j] * dx
            if C < best_c - 1e-14:
                best_c = C
                best_h = h.copy()
                no_imp = 0
            no_imp += 1
        else:
            delta = new_C - C
            if delta < T * 5 and rng.random() < np.exp(-delta / T):
                h[i] += di
                h[j] += dj
                c, other = other, c
                C = new_C
                accept_count += 1
                omf[n - 1 - i] = (1.0 - h[i]) * dx
                omf[n - 1 - j] = (1.0 - h[j]) * dx
                hdx[i] = h[i] * dx
                hdx[j] = h[j] * dx
        T = max(T_min, T * cool_rate)
    return best_h, best_c

def gradient_step(h, c, C, tau, lr):
    n = h.size
    dx = 2.0 / n
    z = (c - C) / tau
    z -= z.max()
    w = np.exp(z)
    ws = w.sum()
    if ws < 1e-30:
        return h
    w /= ws
    grad = np.zeros(n)
    omf_full = (1.0 - h[::-1]) * dx
    hdx_full = h * dx
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-9:
            continue
        d = k - (n - 1)
        lo1 = max(0, -d)
        hi1 = min(n - 1, n - 1 - d)
        if lo1 <= hi1:
            idx = np.arange(lo1, hi1 + 1)
            grad[idx] += wk * hdx_full[idx + d]
        lo2 = max(0, d)
        hi2 = min(n - 1, n - 1 + d)
        if lo2 <= hi2:
            idx = np.arange(lo2, hi2 + 1)
            grad[idx] -= wk * hdx_full[idx - d]
    h_new = h - lr * grad
    return repair(h_new)

def slsqp_polish(h0, time_limit):
    h = h0.copy()
    n = h.size
    best_c = exact_c(h)
    best_h = h.copy()
    deadline = time.time() + time_limit
    tau_list = [0.02, 0.008, 0.003, 0.001]
    for tau in tau_list:
        if time_left(deadline) < 3.0:
            break
        def fun(x):
            cv = c_values(x)
            cm = cv.max()
            z = (cv - cm) / tau
            z -= z.max()
            return cm + tau * np.log(np.exp(z).sum())
        def jac(x):
            cv = c_values(x)
            cm = cv.max()
            z = (cv - cm) / tau
            z -= z.max()
            w = np.exp(z)
            w /= w.sum()
            n_local = x.size
            dx = 2.0 / n_local
            g = np.zeros(n_local)
            for k in range(2 * n_local - 1):
                if w[k] < 1e-8:
                    continue
                d = k - (n_local - 1)
                lo2 = max(0, d)
                hi2 = min(n_local - 1, n_local - 1 + d)
                if lo2 <= hi2:
                    g[lo2:hi2 + 1] -= w[k] * (1.0 - x[lo2 - d:hi2 + 1 - d]) * dx
                lo1 = max(0, -d)
                hi1 = min(n_local - 1, n_local - 1 - d)
                if lo1 <= hi1:
                    g[lo1:hi1 + 1] += w[k] * x[lo1 + d:hi1 + 1 + d] * dx
            return g
        bounds = [(0.0, 1.0)] * n
        cons = [{"type": "eq", "fun": lambda x: x.sum() - 0.5 * n,
                 "jac": lambda x: np.ones(n)}]
        try:
            res = so.minimize(fun, h, method="SLSQP", jac=jac,
                              bounds=bounds, constraints=cons,
                              options={"maxiter": 200, "ftol": 1e-12})
            if res is not None:
                hr = repair(res.x)
                cr = exact_c(hr)
                if cr < best_c:
                    best_c = cr
                    best_h = hr
                h = hr if cr <= best_c + 1e-6 else best_h
        except Exception:
            pass
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)
    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size >= 10:
        init = repair(np.asarray(raw, dtype=float))
    else:
        init = repair(np.full(70, 0.5))
    best_x = init.copy()
    best_c = exact_c(best_x)
    t0 = time.time()
    # Phase 1: quick SLSQP warmup
    if time_left(deadline) > 30:
        x, c = slsqp_polish(init, min(25.0, time_left(deadline) * 0.2))
        if c < best_c:
            best_c = c
            best_x = x
    # Phase 2: SA on base resolution
    if time_left(deadline) > 50:
        sa_time = min(500.0, time_left(deadline) * 0.55)
        x, c = sa_optimize(best_x, sa_time, seed_val)
        if c < best_c:
            best_c = c
            best_x = x
    # Phase 3: gradient + SLSQP polish
    if time_left(deadline) > 20:
        x, c = slsqp_polish(best_x, min(40.0, time_left(deadline) * 0.25))
        if c < best_c:
            best_c = c
            best_x = x
    # Phase 4: higher resolution attempts
    n_base = best_x.size
    for n_try in [n_base + 28, n_base + 30, n_base + 70]:
        if time_left(deadline) < 80:
            break
        if n_try > 300:
            continue
        h_try = resample_to(best_x, n_try)
        sa_t = min(120.0, time_left(deadline) * 0.4)
        x2, c2 = sa_optimize(h_try, sa_t, seed_val + n_try)
        if time_left(deadline) > 10:
            x2, c2 = slsqp_polish(x2, min(15.0, time_left(deadline) * 0.3))
        if c2 < best_c:
            best_c = c2
            best_x = x2
    # Phase 5: final SA polish at best resolution
    if time_left(deadline) > 30:
        x, c = sa_optimize(best_x, min(100.0, time_left(deadline) * 0.6), seed_val + 9999)
        if c < best_c:
            best_c = c
            best_x = x
    # Final repair and evaluation
    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
```
