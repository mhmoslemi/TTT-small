    n = len(h)

            g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def _conv_grad_single(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift >= 0:
        if shift < n:
            g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def soft_max_val_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    s = np.sum(exps)
    val = (m + t * np.log(s)) * dx
    weights = exps / s
    grad = np.zeros(n)
    thresh = 1e-8
    for idx in np.where(weights > thresh)[0]:
        shift = idx - (n - 1)
        grad += weights[idx] * _conv_grad_single(h, shift)
    return val, grad * dx
