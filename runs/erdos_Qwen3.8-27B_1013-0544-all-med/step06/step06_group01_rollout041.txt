3. Solve the LP: min ∇C(k*|h) · h s.t. 0 ≤ h ≤ 1, sum(h) = n/2
4. Update h
5. Repeat

    S = 0.5 * n
    d = S - np.sum(h)
    if abs(d) < 1e-15:

    if d > 0:

        order = np.argsort(-room, kind='stable')
        for idx in order:
            if d <= 1e-15:

            add = min(room[idx], d)
            h[idx] += add

        for idx in order:
            if -d <= 1e-15:
                break
            rem = min(room[idx], -d)
            h[idx] -= rem
            d += rem

    nc = len(C)
    m = np.max(C)
    sh = alpha * (C - m)

    vb = (b >= 0) & (b < n)

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _c5(h):
    return np.max(_corr(h))

def _project(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    S = 0.5 * n
    d = S - np.sum(h)
    if abs(d) < 1e-15:
        return h
    if d > 0:
        room = 1.0 - h
        order = np.argsort(-room, kind='stable')
        for idx in order:
            if d <= 1e-15:
                break
            add = min(room[idx], d)
            h[idx] += add
            d -= add
    else:
        room = h.copy()
        order = np.argsort(-room, kind='stable')
        for idx in order:
            if -d <= 1e-15:
                break
            rem = min(room[idx], -d)
            h[idx] -= rem
            d += rem
    return h

def _obj_grad(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _corr(h)
    nc = len(C)
    m = np.max(C)
    sh = alpha * (C - m)
    es = np.exp(sh)
    se = np.sum(es)
    f = m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    i_arr = np.arange(n)[:, None]
    k_arr = np.arange(nc)[None, :]
    a = i_arr + k_arr - (n - 1)
    b = i_arr - k_arr + (n - 1)
    va = (a >= 0) & (a < n)
    vb = (b >= 0) & (b < n)
    first = np.where(va, 1.0 - h[np.clip(a, 0, n - 1)], 0.0)
    second = np.where(vb, h[np.clip(b, 0, n - 1)], 0.0)
    grad = dx * (first - second).dot(w)
    grad += 2.0 * beta * (np.sum(h) - 0.5 * n)
    return f, grad

def _lbfgs(h, n, alpha, beta, maxiter=500):
    res = minimize(_obj_grad, h, args=(alpha, beta), jac=True,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'maxfun': maxiter * 10, 'ftol': 1e-16})
    if res.x is not None:
        return _project(res.x, n)
    return _project(h, n)

        offset = k_star - (n - 1)
        contribs = []
        for i in range(n):
            j = i + offset
            if 0 <= j < n:
                c = h[i] * (1.0 - h[j])
                if c > 1e-4:

        contribs.sort(reverse=True)
        improved = False
        for c, i, j in contribs[:25]:
            delta = min(h[i], 1.0 - h[j]) * 0.4
            if delta < 1e-6:

            h[i] -= delta
            h[j] += delta
            h = _project(h, n)
            c5n = _c5(h)
            if c5n < best_c5 - 1e-12:
                best_c5 = c5n

            else:

def _pb_local(h, n, max_rounds=30):
    """Pair-based local search: try all pairs with small transfers"""
    h = _project(h.copy(), n)
    best_c5 = _c5(h)
    C = _corr(h)
    k_star = np.argmax(C)
    offset = k_star - (n - 1)
    for _ in range(max_rounds):
        improved = False
        for i in range(n):
            if h[i] < 0.05:
                continue
            for j in range(n):
                if i == j or h[j] > 0.95:
                    continue
                for delta in (0.05, 0.1, 0.2):
                    if h[i] < delta or h[j] > 1.0 - delta:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    h = _project(h, n)
                    c5n = _c5(h)
                    if c5n < best_c5 - 1e-12:
                        best_c5 = c5n
                        improved = True
                        break
                    h[i] += delta
                    h[j] -= delta
                    h = _project(h, n)

                    break

                C = _corr(h)
                k_star = np.argmax(C)
                offset = k_star - (n - 1)
                break
        if not improved:
            break
    return _project(h, n), best_c5
