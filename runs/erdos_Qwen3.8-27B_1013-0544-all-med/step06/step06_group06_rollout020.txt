        h_try = slsqp_opt(h, n, alpha, 5000)
        c = compute_c5(h_try, n)
        if c < best_c - 1e-15:
            best_h, best_c = h_try, c
        h = best_h.copy()

    # Phase 2: Subgradient method with adaptive step and periodic SLSQP
    lr = 0.004
    no_improve = 0
    for step in range(30000):
        if budget_s - (time.time() - t0) < 30:
            break
        idx = top_lags(h, n, 8)
        g = np.zeros(n)
        for ix in idx:
            k = ix - (n - 1)
            g += grad_lag(h, n, k)
        g /= len(idx)
        h = project_feasible(h - lr * g, n)
        c = compute_c5(h, n)
        if c < best_c - 1e-15:
            best_h, best_c = h.copy(), c
            no_improve = 0
        else:
            no_improve += 1
        # Stagnation: try larger perturbation
        if no_improve == 500:
            h = project_feasible(best_h + rng.normal(0, 0.02, size=n), n)
            no_improve = 0
        # Periodic SLSQP polish
        if step % 400 == 399 and budget_s - (time.time() - t0) > 60:
            h_try = slsqp_opt(h, n, 5000, 3000)
            c_try = compute_c5(h_try, n)
            if c_try < best_c:
                best_h, best_c = h_try, c_try
            h = best_h.copy()
            no_improve = 0
        lr = 0.004 / np.sqrt(1 + step * 0.001)

    # Phase 3: Micro-perturbation + SLSQP restarts
    while budget_s - (time.time() - t0) > 40:
        scale = rng.choice([0.001, 0.003, 0.005, 0.01, 0.015, 0.03])
        h_pert = best_h + rng.normal(0, scale, size=n)
        h_pert = project_feasible(h_pert, n)
        h_try = slsqp_opt(h_pert, n, 15000, 4000)
        c = compute_c5(h_try, n)
        if c < best_c - 1e-15:
            best_h, best_c = h_try, c

    # Phase 4: 2-point swap fine-tuning
    if budget_s - (time.time() - t0) > 15:
        count = 0
        while budget_s - (time.time() - t0) > 10 and count < 80000:
            count += 1
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            for d in [0.001, -0.001, 0.002, -0.002, 0.005, -0.005]:
                hi = best_h[i] + d
                hj = best_h[j] - d
                if 0.0 <= hi <= 1.0 and 0.0 <= hj <= 1.0:
                    h_try = best_h.copy()
                    h_try[i] = hi
                    h_try[j] = hj
                    c = compute_c5(h_try, n)
                    if c < best_c - 1e-15:
                        best_h, best_c = h_try, c
                        break

    # Phase 5: Final SLSQP polish with very high alpha
    if budget_s - (time.time() - t0) > 10:
        for alpha in [30000, 100000]:
            if budget_s - (time.time() - t0) < 10:
                break
            h_try = slsqp_opt(best_h, n, alpha, 8000)
            c = compute_c5(h_try, n)
            if c < best_c:
                best_h, best_c = h_try, c

    best_h = project_feasible(best_h, n)
    best_c = compute_c5(best_h, n)
    return best_h, best_c, n
