import numpy as np

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_for_shift(h, s):
    N = len(h)
    g = np.zeros(N)
    if s >= 0:
        g[:N - s] += 1.0 - h[s:]
        g[s:] -= h[:N - s]
    else:
        s2 = -s
        g[s2:] += 1.0 - h[:N - s2]
        g[:N - s2] -= h[s2:]
    return g

def compute_grad_weighted(h, corr, weights):
    N = len(h)
    g = np.zeros(N)
    for j in range(len(weights)):
        if weights[j] < 1e-8:
            continue
        s = j - (N - 1)
        g += weights[j] * grad_for_shift(h, s)
    return g
