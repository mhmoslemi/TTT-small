def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
        else:
            idx = np.argmin(np.abs(h - 0.5))
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def obj_and_corr(h):
    N = len(h)
    dx = 2.0 / N

def grad_for_lag(h, k):
    N = len(h)
    if k == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(N)
    if k > 0:
        g[:N - k] += (1.0 - h[k:])
        g[k:] -= h[:N - k]
    else:
        m = -k
        g[m:] += (1.0 - h[:N - m])
        g[:N - m] -= h[m:]
    return g

def active_grad(h, corr, n_top=5):
    N = len(h)
    center = N - 1
    if n_top >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argpartition(corr, -n_top)[-n_top:]
    c_max = corr.max()
    w = np.exp(corr[top_idx] - c_max)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(len(top_idx))
        w[np.argmax(corr[top_idx])] = 1.0
        wsum = 1.0
    else:
        w /= wsum
    g = np.zeros(N)
    for wi, ii in zip(w, top_idx):
