            if total < 1e-15:
                break
            h += (diff / total) * room
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h += (diff / total) * h
    return np.clip(h, 0.0, 1.0)

def opt_coordinate(h, i, n):
    """Exact 1D optimization of h[i]."""
    dx = 2.0 / n
    h_save = h[i]
    h[i] = 0.0
    corr_0 = np.correlate(h, 1.0 - h, mode='full').astype(float)
    h[i] = 1.0
    corr_1 = np.correlate(h, 1.0 - h, mode='full').astype(float)
    h[i] = h_save
    
    A = corr_0 * dx
    B = (corr_1 - corr_0) * dx
    quad_coeff = dx  # coefficient of t(1-t) at k=n-1
    
    m = 2 * n - 1
    candidates = [0.0, 1.0, 0.5]
    
    # Linear-linear intersections
    for k in range(m):
        if k == n - 1:
            continue
        for l in range(k + 1, m):
            if l == n - 1:
                continue
            d = B[k] - B[l]
            if abs(d) < 1e-15:
                continue
            t = (A[l] - A[k]) / d
            if -1e-12 <= t <= 1.0 + 1e-12:
                candidates.append(0.0 if t < 0 else (1.0 if t > 1 else t))
    
    # Quadratic-linear intersections
    for k in range(m):
        if k == n - 1:
            continue
        # quad_coeff*t*(1-t) + A[n-1] = A[k] + B[k]*t
        # quad_coeff*t - quad_coeff*t^2 - B[k]*t + A[n-1] - A[k] = 0
        # -quad_coeff*t^2 + (quad_coeff - B[k])*t + (A[n-1] - A[k]) = 0
        # quad_coeff*t^2 - (quad_coeff - B[k])*t + (A[k] - A[n-1]) = 0
        a_c = quad_coeff
        b_c = -(quad_coeff - B[k])
        c_c = A[k] - A[n-1]
        disc = b_c*b_c - 4.0*a_c*c_c
        if disc >= -1e-15:
            disc = max(0.0, disc)
            sq = np.sqrt(disc)
            for t in ((-b_c + sq) / (2*a_c), (-b_c - sq) / (2*a_c)):
                if -1e-12 <= t <= 1.0 + 1e-12:
                    candidates.append(0.0 if t < 0 else (1.0 if t > 1 else t))
    
    # Evaluate
    best_t = h_save
    best_val = float('inf')
    for t in candidates:
        fvals = A + B * t
        fvals[n-1] = A[n-1] + quad_coeff * t * (1.0 - t)
        val = float(fvals.max())
        if val < best_val - 1e-16:
            best_val = val
            best_t = t
    
    return best_t, best_val

def coordinate_descent(h, n, max_sweeps=100):
    """Repeated exact 1D coordinate optimization with projection."""
    for sweep in range(max_sweeps):
        old_sum = h.sum()
        target_sum = n / 2.0
        improved = False
        for i in range(n):
            new_val, _ = opt_coordinate(h, i, n)
            if abs(new_val - h[i]) > 1e-14:
                h[i] = new_val
                improved = True
        # Project to maintain sum constraint
        h = project_feasible(h, n)
        if not improved:
            break
    return h

def smax_obj(h_vec, n_val, alpha):
    dx = 2.0 / n_val
    c = np.correlate(h_vec, 1.0 - h_vec, mode='full') * dx
    m = c.max()
    shifted = alpha * (c - m)
    mx = shifted.max()
    if mx > 500:
        return float(m + (mx + np.log(np.sum(np.exp(shifted - mx)))) / alpha)
    return float(m + np.log(np.sum(np.exp(shifted))) / alpha)
