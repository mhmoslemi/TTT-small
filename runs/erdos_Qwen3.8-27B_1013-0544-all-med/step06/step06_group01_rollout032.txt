def _obj_softmax(h, alpha, beta):
    n = len(h)
    dx = 2.0 / n
    C = _c5_array(h)
    m = np.max(C)
    es = np.exp(alpha * (C - m))
    se = np.sum(es)
    f = m + np.log(se) / alpha + beta * (np.sum(h) - 0.5 * n) ** 2
    w = es / se
    nc = len(C)
    grad = np.zeros(n)

            contrib = 0.0
            if d >= 0:
                if d <= i < n:
                    contrib += 1.0 - h[i - d]
                if 0 <= i < n - d:
                    contrib -= h[i + d]
            else:

                if 0 <= i < n - dd:
                    contrib += 1.0 - h[i + dd]
                if dd <= i < n:
                    contrib -= h[i - dd]

    C = _c5_array(h)

    n = len(h)
    dx = 2.0 / n

    return np.max(_c5_array(h))

def _project_box_sum(h, n):
    target = 0.5 * n
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    dx = 2.0 / n
    d = j - (n - 1)
    g = np.zeros(n)

            g[d:] += dx * (1.0 - h[:n - d])
            g[:n - d] -= dx * h[d:]
    else:

        hb, cb = _local_search(hb, n0, max_time=budget_s * 0.99 - _el())

    # Final short momentum polish
    if _el() < budget_s * 0.99:
        hb, cb = _momentum_opt(hb, n0, iters=50000, K=3, lr=0.03,
                              t0=t0, time_limit=budget_s * 0.995)

    hb = _project_box_sum(hb, n0)
    cb = _c5_max(hb)
    return (hb, cb, n0)
