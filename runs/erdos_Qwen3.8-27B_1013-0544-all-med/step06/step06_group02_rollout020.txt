import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_val(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_single_lag(h, lag, N):
    """Gradient of corr at a given lag with respect to h."""
    g = np.zeros(N)
    if lag >= 0:
        if lag < N:
            g[lag:] += 1.0 - h[:N-lag]
            g[:N-lag] -= h[lag:]
    else:
        L = -lag
        if L < N:
            g[:N-L] += 1.0 - h[L:]
            g[L:] -= h[:N-L]
    return g

def soft_grad(h, corr, dx, alpha):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        lag = k_star - (N - 1)
        return grad_single_lag(h, lag, N) * dx
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-5:
            continue
        lag = j - (N - 1)
        g += w[j] * grad_single_lag(h, lag, N)
    return g * dx

∇f_α = (1/α) * sum_k exp(α*corr_k*dx) / sum_j exp(α*corr_j*dx) * ∇corr_k

Which is exactly the soft gradient I've been computing! So I could use scipy.optimize.minimize with BFGS or L-BFGS-B on this smooth approximation.

One more optimization: instead of computing the full soft gradient in a loop over all lags, I can note that:

g = sum_k w_k * grad_single_lag(h, lag_k)

where lag_k = k - (N-1).

For lag ≥ 0: grad[lag:] += 1-h[:N-lag], grad[:N-lag] -= h[lag:]
For lag < 0 (L=-lag): grad[:N-L] += 1-h[L:], grad[L:] -= h[:N-L]

I can group positive and negative lags:
For positive lags l = 1, ..., N-1 with weights w_{N-1+l}:
  g[l:] += w_{N-1+l} * (1-h[:N-l])
  g[:N-l] -= w_{N-1+l} * h[l:]

For negative lags (L = 1, ..., N-1) with weights w_{N-1-L}:
  g[:N-L] += w_{N-1-L} * (1-h[L:])
  g[L:] -= w_{N-1-L} * h[:N-L]

For lag = 0 with weight w_{N-1}:
  g += w_{N-1} * (1 - 2*h)  [since grad = (1-h) - h = 1-2h]

This can be done in O(N²) but with good cache behavior. For N=120, this is fine.

import numpy as np
import time

def project_feasible(v, N):
    target = N / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.clip(v - mid, 0.0, 1.0).sum() > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        if interior.sum() > 0:
            h[interior] += diff / interior.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def c5_val(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx, corr, dx

def grad_single_lag(h, lag, N):
    g = np.zeros(N)
    if lag >= 0:
        if lag < N:
            g[lag:] += 1.0 - h[:N - lag]
            g[:N - lag] -= h[lag:]
    else:
        L = -lag
        if L < N:
            g[:N - L] += 1.0 - h[L:]
            g[L:] -= h[:N - L]
    return g

def soft_grad(h, corr, dx, alpha):
    N = len(h)
    c = corr - corr.max()
    w = np.exp(alpha * c)
    wsum = w.sum()
    if wsum < 1e-300:
        k_star = np.argmax(corr)
        lag = k_star - (N - 1)
        return grad_single_lag(h, lag, N) * dx
    w /= wsum
    g = np.zeros(N)
    for j in range(len(w)):
        if w[j] < 1e-5:
            continue
        lag = j - (N - 1)
        g += w[j] * grad_single_lag(h, lag, N)
    return g * dx

def gradient_descent(h, N, rng, t_end, try_update, max_iter, alpha_init, alpha_max):
    val, corr, dx = c5_val(h)
    try_update(h.copy(), val)
    lr = 0.03
    alpha = alpha_init
    stagnant = 0
    best_local = val
    best_local_h = h.copy()
    for it in range(max_iter):
        if time.time() > t_end:
            break
        g = soft_grad(h, corr, dx, alpha)
        h_new = project_feasible(h - lr * g, N)
        val_new, corr_new, _ = c5_val(h_new)
        if val_new < val - 1e-14:
            h = h_new
            val = val_new
            corr = corr_new
            stagnant = 0
            lr = min(lr * 1.02, 0.1)
            alpha = min(alpha * 1.003, alpha_max)
            if val < best_local:
                best_local = val
                best_local_h = h.copy()
        else:
            stagnant += 1
            lr *= 0.78
            if stagnant > 35:
                scale = 0.03 + 0.05 * min(stagnant / 100.0, 1.0)
                h = project_feasible(h + rng.randn(N) * scale, N)
                val, corr, _ = c5_val(h)
                lr = 0.03
                alpha = max(alpha_init, alpha * 0.5)
                stagnant = 0
                if val < best_local:
                    best_local = val
                    best_local_h = h.copy()
    try_update(best_local_h.copy(), best_local)
    return best_local_h.copy(), best_local

def block_perturb(h, N, rng, block_size):
    h_new = h.copy()
    start = rng.randint(0, max(1, N - block_size))
    end = min(N, start + block_size)
    perm = rng.permutation(end - start)
    h_new[start:end] = h[start:end][perm]
    return project_feasible(h_new, N)

def try_n_value(h_best, n_t, rng, t_end, try_update):
    if h_best is not None and time.time() < t_end - 10:
        x0 = np.arange(len(h_best)) / len(h_best)
        x1 = np.arange(n_t) / n_t
        h_t = np.interp(x1, x0, h_best)
        h_t = project_feasible(h_t, n_t)
    else:
        h_t = project_feasible(0.5 * np.ones(n_t) + rng.randn(n_t) * 0.05, n_t)
    h, v = gradient_descent(h_t, n_t, rng, t_end, try_update, 30000, 8.0, 40.0)
    try_update(h, v)
    return h, v

def hard_grad_polish(h, N, t_end, try_update, max_iter=30000):
    val, corr, dx = c5_val(h)
    try_update(h.copy(), val)
    lr = 0.01
    for it in range(max_iter):
        if time.time() > t_end:
            break
        k_star = np.argmax(corr)
        lag = k_star - (N - 1)
        g = grad_single_lag(h, lag, N) * dx
        h_new = project_feasible(h - lr * g, N)
        val_new, corr_new, _ = c5_val(h_new)
        if val_new < val - 1e-15:
            h = h_new
            val = val_new
            corr = corr_new
            try_update(h.copy(), val)
        else:
            lr *= 0.85
    return h, val

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t0 = time.time()
    t_end = t0 + budget_s - 5

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_init = None

    best_h = None
    best_val = float('inf')

    def try_update(h, val):
        nonlocal best_h, best_val
        if val < best_val:
            best_val = val
            best_h = h.copy()

    if h_init is not None:
        try_update(h_init.copy(), c5_val(h_init)[0])
        N0 = len(h_init)
    else:
        N0 = 100
        h_init = project_feasible(0.5 * np.ones(N0) + rng.randn(N0) * 0.1, N0)
        try_update(h_init.copy(), c5_val(h_init)[0])

    # Phase 1: Main gradient descent from initial
    if time.time() < t_end - 120:
        h, v = gradient_descent(h_init.copy(), N0, rng, t_end - 120, try_update, 80000, 8.0, 60.0)
        try_update(h, v)

    # Phase 2: Hard gradient polishing
    if best_h is not None and time.time() < t_end - 80:
        N_b = len(best_h)
        h, v = hard_grad_polish(best_h.copy(), N_b, t_end - 80, try_update, 40000)
        try_update(h, v)

    # Phase 3: Block perturbations + re-optimization
    if best_h is not None and time.time() < t_end - 60:
        N_b = len(best_h)
        for bs in [3, 5, 8, 12]:
            if time.time() > t_end - 50:
                break
            for attempt in range(15):
                if time.time() > t_end - 50:
                    break
                h_pert = block_perturb(best_h, N_b, rng, bs)
                h_r, v_r = gradient_descent(h_pert, N_b, rng, t_end - 50, try_update, 15000, 10.0, 50.0)
                try_update(h_r, v_r)

    # Phase 4: Use 1-h symmetry
    if best_h is not None and time.time() < t_end - 40:
        N_b = len(best_h)
        h_complement = project_feasible(1.0 - best_h, N_b)
        h_r, v_r = gradient_descent(h_complement, N_b, rng, t_end - 40, try_update, 30000, 8.0, 50.0)
        try_update(h_r, v_r)

    # Phase 5: Try different N values
    if time.time() < t_end - 25:
        for n_t in [60, 80, 150, 200]:
            if time.time() > t_end - 20:
                break
            try_n_value(best_h, n_t, rng, t_end - 20, try_update)

    # Phase 6: Multiple small restarts on best N
    if best_h is not None and time.time() < t_end - 15:
        N_b = len(best_h)
        for r in range(20):
            if time.time() > t_end - 15:

            scale = 0.02 + 0.02 * (r % 5)

            h_r, v_r = gradient_descent(h_pert, N_b, rng, t_end - 15, try_update, 12000, 12.0, 60.0)
            try_update(h_r, v_r)

    # Phase 7: Final fine-tuning
    if best_h is not None and time.time() < t_end:
        N_b = len(best_h)
        h, v = hard_grad_polish(best_h.copy(), N_b, t_end, try_update, 50000)
        try_update(h, v)

    # Phase 8: Ultra-fine step
    if best_h is not None and time.time() < t_end:
        N_b = len(best_h)
        h = best_h.copy()
        val, corr, dx = c5_val(h)
        lr = 0.002
        for it in range(30000):
            if time.time() > t_end:
                break
            g = soft_grad(h, corr, dx, 50.0)
            h_new = project_feasible(h - lr * g, N_b)

                val = val_new
                corr = corr_new
                try_update(h.copy(), val)
            else:
                lr *= 0.92

    if best_h is None:
        best_h = project_feasible(0.5 * np.ones(47), 47)
        best_val = c5_val(best_h)[0]

    return best_h, best_val, len(best_h)
