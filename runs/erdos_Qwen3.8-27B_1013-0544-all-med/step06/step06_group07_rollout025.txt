            if deadline - time.time() < 20:

            x, c = multi_slsqp(s0, t2 / 3.0, seed_val + 100 + start_i)
            if c < best_c:
                best_c = c
                best_x = x

    # Phase 3: Continuous SA
    if deadline - time.time() > 40:
        t3 = min(200, (deadline - time.time()) * 0.3)
        for s_i in range(2):
            if deadline - time.time() < 15:
                break
            x, c = continuous_sa(best_x, t3 / 2.0, seed_val + 200 + s_i)
            if c < best_c:
                best_c = c
                best_x = x

    # Phase 4: SLSQP refinement after SA
    if deadline - time.time() > 30:
        t4 = min(150, (deadline - time.time()) * 0.35)
        x, c = multi_slsqp(best_x, t4, seed_val + 500)
        if c < best_c:
            best_c = c
            best_x = x

    # Phase 5: Try different resolutions
    if deadline - time.time() > 40:
        for n_new in (base_n + 20, 100, 140, base_n - 20):
            if deadline - time.time() < 30:
                break
            if n_new < 20 or n_new > 200:
                continue
            hr = resample_to(best_x, n_new)
            t5 = min(80, (deadline - time.time()) * 0.25)
            x, c = multi_slsqp(hr, t5, seed_val + 1000 + n_new)
            if c < best_c:
                best_c = c
                best_x = x

    # Phase 6: Final active-set polish
    if deadline - time.time() > 5:
        n = best_x.size
        for _ in range(10):
            if deadline - time.time() < 2:
                break
            c = c_values(best_x)
            active = np.argsort(c)[::-1][:2]
            xr = repair(slsqp_active(best_x, 60, active))
            cx = exact_c(xr)
            if cx < best_c:
                best_c = cx
                best_x = xr

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
