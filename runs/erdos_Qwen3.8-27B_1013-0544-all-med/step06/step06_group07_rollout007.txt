def gradient_for_lag(x, s):
    """Gradient of c_k with respect to x, where s = k - (n-1)."""
    n = x.size
    g = np.zeros(n)
    if s >= 0:
        # x[j] * (1-x[j+s]) for j in [0, n-s)

        g[t:] += (1.0 - x[:n-t])
        g[:n-t] += -x[t:]

            y = repair(y)

        cy = float(np.max(c_values(y)))
        delta = cy - C
        if delta < -1e-14 or (rng.rand() < np.exp(-delta / max(T, 1e-12))):
            x = y
            T = max(T_min, T * 0.998)
        else:
            T = max(T_min, T * 0.9995)

    return best_x, best_c

def coord_descent(h0, time_limit, seed):
    rng = np.random.RandomState(seed)
    x = repair(h0.copy())
    best_x = x.copy()
    best_c = exact_c(x)
    n = x.size
    deadline = time.time() + time_limit
    step_sizes = [0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001]
    ss_idx = 0
    order = list(range(n))
    rng.shuffle(order)
    oi = 0

    while time.time() < deadline:
        improved = False
        for _ in range(n * 3):
            i = order[oi % n]
            oi += 1
            if time.time() > deadline:
                break
            C_cur = exact_c(x)
            for s in step_sizes[ss_idx:]:
                y = x.copy()
                y[i] = min(1.0, max(0.0, x[i] - s))
                y = repair(y)
                cy = exact_c(y)
                if cy < C_cur - 1e-13:
                    x = y
                    C_cur = cy
                    improved = True
                    break
                y = x.copy()
                y[i] = min(1.0, max(0.0, x[i] + s))
                y = repair(y)
                cy = exact_c(y)
                if cy < C_cur - 1e-13:
                    x = y
                    C_cur = cy
                    improved = True
                    break
        if not improved:
            ss_idx += 1
            if ss_idx >= len(step_sizes):
                break
            rng.shuffle(order)
        if exact_c(x) < best_c:
            best_c = exact_c(x)
            best_x = x.copy()

    return best_x, best_c

def resample_to(h0, n_new):
    h0 = np.asarray(h0, dtype=float)
    n0 = h0.size
    n_new = int(n_new)
    if n0 == n_new:
        return repair(h0)
    if n0 == 0:
        return repair(np.full(n_new, 0.5))
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x1 = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    return repair(np.interp(x1, x0, h0))

def make_starts(base, n, seed, count):
    rng = np.random.RandomState(seed)
    starts = [repair(base.copy())]
    for i in range(count - 1):
        scale = 0.03 + 0.07 * float(i)
        y = base + rng.normal(0, scale, n)
        starts.append(repair(y))
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    deadline = time.time() + max(1.0, float(budget_s))
    seed_val = int(seed)

    raw = globals().get("initial_h_values", None)
    if raw is not None and np.asarray(raw).size >= 10:
        base_h = np.asarray(raw, dtype=float)
        base_n = int(base_h.size)
    else:
        base_n = 70
        base_h = np.full(base_n, 0.5)

    # Try multiple resolutions
    resolutions = []
    if base_n <= 100:
        resolutions = [base_n, base_n + 20, base_n - 20, base_n + 40]
    else:
        resolutions = [base_n, base_n - 30, base_n - 60]
    resolutions = [r for r in resolutions if 30 <= r <= 150]
    if not resolutions:
        resolutions = [70]

    best_x = None
    best_c = 1e9

    for n in resolutions:
        if time_left(deadline) < 100.0:
            break
        base_n_here = n
        if base_n_here != base_n:
            h_base = resample_to(base_h, base_n_here)
        else:
            h_base = repair(base_h.copy())
        h_base = repair(h_base)

        c0 = exact_c(h_base)
        if c0 < best_c:
            best_c = c0
            best_x = h_base.copy()

        # Phase 1: SLSQP (30% of remaining)
        slsqp_time = min(60.0, time_left(deadline) * 0.25)
        if slsqp_time > 10.0:
            x1, c1 = slsqp_phase(h_base, slsqp_time)
            if c1 < best_c:
                best_c = c1
                best_x = x1.copy()

        # Phase 2: Hybrid SA (50% of remaining)
        sa_time = min(250.0, time_left(deadline) * 0.45)
        if sa_time > 30.0:
            x2, c2 = hybrid_sa(h_base, sa_time, seed_val + n)
            if c2 < best_c:
                best_c = c2
                best_x = x2.copy()

        # Phase 3: Coordinate descent polish
        cd_time = min(80.0, time_left(deadline) * 0.2)
        if cd_time > 10.0 and best_x is not None:
            best_here = resample_to(best_x, base_n_here) if best_x.size != base_n_here else best_x
            x3, c3 = coord_descent(best_here, cd_time, seed_val + 999 + n)
            if c3 < best_c:
                best_c = c3
                best_x = x3.copy()

        # Second SA run from SLSQP result if still have time
        if time_left(deadline) > 80.0:
            start2 = h_base if best_x.size != n else best_x
            x4, c4 = hybrid_sa(start2, min(150.0, time_left(deadline) * 0.4), seed_val + 5000 + n)
            if c4 < best_c:
                best_c = c4
                best_x = x4.copy()

    # Final polish on best solution
    if best_x is not None and time_left(deadline) > 20.0:
        final_time = min(60.0, time_left(deadline) * 0.5)
        xf, cf = coord_descent(best_x, final_time, seed_val + 77777)
        if cf < best_c:
            best_c = cf
            best_x = xf.copy()

    if best_x is None:
        best_x = repair(np.full(70, 0.5))
        best_c = exact_c(best_x)

    best_x = repair(best_x)
    best_c = exact_c(best_x)
    return best_x, best_c, int(best_x.size)
