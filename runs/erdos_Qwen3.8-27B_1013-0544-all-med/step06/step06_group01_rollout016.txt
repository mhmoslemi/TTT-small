import time

    se = np.sum(es)

    h = np.clip(h, 0.0, 1.0)
    target = 0.5 * n

        if abs(diff) < 1e-12:

        if diff > 0:

            break
        step = (f_val - L) / gn
        step = min(step, 1.0 / np.sqrt(t + 1.0))
        h_new = _project(h - step * g, n)
        cn = _c5_max(h_new)
        if cn < best_c5 - 1e-12:
            best_c5 = cn
            h = h_new
    return h, best_c5

def _focused_local_search(h, n, t_max, t_start):
    h = h.copy()
    best_c5 = _c5_max(h)
    rng = np.random.RandomState(12345)
    stall = 0
    while time.time() - t_start < t_max:
        C = _c5_values(h)
        k_star = int(np.argmax(C))
        lag = k_star - (n - 1)
        # Find contributing terms: h[i]*(1-h[i+lag]) for valid i
        valid_i = []
        for i in range(n):
            if 0 <= i + lag < n:
                valid_i.append(i)
        if not valid_i:
            break
        vi = np.array(valid_i)
        contribs = h[vi] * (1.0 - h[vi + lag])
        order = np.argsort(-contribs)
        improved = False
        for oi in order[:8]:
            i = vi[oi]
            j = i + lag
            for eps in [0.04, 0.02, 0.01, 0.005, 0.002]:
                if h[i] < eps or h[j] > 1.0 - eps:
                    continue
                hn = h.copy()
                hn[i] -= eps
                hn[j] += eps
                cn = _c5_max(hn)
                if cn < best_c5 - 1e-11:
                    h = hn
                    best_c5 = cn
                    improved = True
                    break
            if improved:
                break
        if not improved:
            # Try random moves
            for _ in range(5):
                i, j = rng.randint(0, n, 2)
                if i == j:
                    continue
                eps = 0.015
                if h[i] >= eps and h[j] <= 1.0 - eps:
                    hn = h.copy()
                    hn[i] -= eps
                    hn[j] += eps
                    cn = _c5_max(hn)
                    if cn < best_c5 - 1e-11:
                        h = hn
                        best_c5 = cn
                        improved = True
                        break
        if not improved:
            stall += 1
            if stall > 200:
                break
        else:
            stall = 0
    return h, best_c5

def _interpolate(h_old, n_old, n_new):
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return _project(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    t_left = budget_s * 0.97
    n0 = len(initial_h_values)
    hb = _project(initial_h_values.astype(float).copy(), n0)
    cb = _c5_max(hb)
    nb = n0

    def _rem():
        return t_left - (time.time() - t0)

    # Phase 1: Progressive soft-max gradient
    for alpha in [5, 10, 20, 50, 100, 200, 500, 1000, 2000]:
        if _rem() < 40:
            break
        beta = max(0.5, 0.01 * alpha)
        maxiter = min(600, int(_rem() * 3))
        res = minimize(_soft_max_obj_grad, hb, args=(alpha, beta),
                      method='L-BFGS-B', jac=True,
                      bounds=[(0.0, 1.0)] * nb,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        if res.x is not None:
            hn = _project(res.x, nb)
            cn = _c5_max(hn)
            if cn < cb - 1e-13:
                cb, hb = cn, hn

    # Phase 2: Subgradient with exact max
    if _rem() > 30:
        tp = min(_rem() * 0.25, 60.0)
        hb, cb = _subgradient_phase(hb, nb, tp, time.time())

    # Phase 3: Focused local search
    if _rem() > 20:
        hb, cb = _focused_local_search(hb, nb, min(_rem() * 0.3, 120.0), time.time())

    # Phase 4: Try different resolutions
    if _rem() > 50:
        n_cands = sorted(set([nb-10, nb-5, nb+5, nb+10, nb+15, nb+25, nb+50, 80, 100, 120, 150]))
        n_cands = [x for x in n_cands if 40 <= x <= 250 and x != nb]
        for n_try in n_cands[:6]:
            if _rem() < 40:
                break
            ht = _interpolate(hb, nb, n_try)
            for alpha in [50, 200, 500]:
                if _rem() < 30:
                    break
                beta = max(0.5, 0.01 * alpha)
                res = minimize(_soft_max_obj_grad, ht, args=(alpha, beta),
                             method='L-BFGS-B', jac=True,
                             bounds=[(0.0, 1.0)] * n_try,
                             options={'maxiter': 400, 'ftol': 1e-14})
                if res.x is not None:
                    ht = _project(res.x, n_try)
                    cn = _c5_max(ht)
                    if cn < cb - 1e-13:
                        cb, hb, nb = cn, ht, n_try
            if _rem() > 20 and nb == n_try:
                hb, cb = _focused_local_search(hb, nb, min(_rem() * 0.2, 30.0), time.time())

    # Phase 5: Random restarts
    rng = np.random.RandomState(seed)
    if _rem() > 30:
        for _ in range(20):
            if _rem() < 25:
                break
            scale = 0.02 + 0.02 * rng.rand()
            ht = _project(hb + rng.randn(nb) * scale, nb)
            for alpha in [100, 500]:
                if _rem() < 20:
                    break
                beta = max(1.0, 0.01 * alpha)
                res = minimize(_soft_max_obj_grad, ht, args=(alpha, beta),
                             method='L-BFGS-B', jac=True,
                             bounds=[(0.0, 1.0)] * nb,
                             options={'maxiter': 300, 'ftol': 1e-14})
                if res.x is not None:
                    ht = _project(res.x, nb)
                    cn = _c5_max(ht)
                    if cn < cb - 1e-13:
                        cb, hb = cn, ht

    # Phase 6: Final polish
    if _rem() > 15:
        hb, cb = _focused_local_search(hb, nb, min(_rem() * 0.5, 60.0), time.time())
    if _rem() > 10:
        for alpha in [1000, 3000]:
            if _rem() < 8:
                break
            beta = max(5.0, 0.02 * alpha)
            res = minimize(_soft_max_obj_grad, hb, args=(alpha, beta),
                         method='L-BFGS-B', jac=True,
                         bounds=[(0.0, 1.0)] * nb,
                         options={'maxiter': 500, 'ftol': 1e-15})
            if res.x is not None:
                hn = _project(res.x, nb)
                cn = _c5_max(hn)
                if cn < cb:
                    cb, hb = cn, hn

    hb = _project(hb, nb)
    cb = _c5_max(hb)
    return (hb, cb, nb)
