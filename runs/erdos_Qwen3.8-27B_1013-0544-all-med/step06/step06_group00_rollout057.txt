    n = len(h)
    grad = np.zeros(n)
    for k in range(-(n-1), n):

        if weight < 1e-15:

        # h[j-k] valid when 0 <= j-k < n, i.e., k <= j < n+k

        h_t, c = targeted_cd(best_h, n_levels=30, n_targets=10, max_sweeps=20)
        if c < best_c5:
            best_c5 = c
            best_h = h_t

    # Phase 3: Very high beta SLSQP for near-exact max minimization
    cur = best_h.copy()
    for beta in [5000.0, 20000.0]:
        if remaining() < 50:
            break
        cur = run_slsqp(cur, n, beta, maxiter=3000, ftol=1e-19)
        c = compute_c5(cur)
        if c < best_c5:
            best_c5 = c
            best_h = cur.copy()

    # Phase 4: Full coordinate descent
    if remaining() > 50:
        h_cd, c = coord_descent(best_h, n_levels=25, max_sweeps=25)
        if c < best_c5:
            best_c5 = c
            best_h = h_cd

    # Phase 5: Structured restarts to escape local optima
    n_restarts = 0
    while remaining() > 160 and n_restarts < 25:
        n_restarts += 1
        if n_restarts % 4 == 1:
            scale = 0.08 + 0.06 * np.random.rand()
            h_r = best_h + np.random.randn(n) * scale
        elif n_restarts % 4 == 2:
            h_r = best_h.copy()
            bs = np.random.choice([16, 32, 64])
            st = np.random.randint(0, max(1, n - bs))
            h_r[st:st + bs] = np.random.rand(bs)
        elif n_restarts % 4 == 3:
            alpha = np.random.uniform(0.03, 0.15)
            h_r = (1 - alpha) * best_h + alpha * np.random.rand(n)
        else:
            # Flip a random subset with binary perturbation
            h_r = best_h.copy()
            flip_mask = np.random.rand(n) < 0.1
            h_r[flip_mask] = 1.0 - h_r[flip_mask]

        h_r = project_feasible(h_r, ts)
        cur = h_r.copy()
        for beta in [20.0, 100.0, 500.0]:
            if remaining() < 60:
                break
            cur = run_slsqp(cur, n, beta, maxiter=600, ftol=1e-15)
            c = compute_c5(cur)
            if c < best_c5:
                best_c5 = c
                best_h = cur.copy()

        if remaining() > 100:
            h_cd, c = coord_descent(best_h, n_levels=15, max_sweeps=10)
            if c < best_c5:
                best_c5 = c
                best_h = h_cd

    # Phase 6: Final fine-grained coordinate descent
    if remaining() > 30:
        h_final, c = targeted_cd(best_h, n_levels=40, n_targets=12, max_sweeps=20)
        if c < best_c5:
            best_c5 = c
            best_h = h_final
        h_final, c = coord_descent(best_h, n_levels=40, max_sweeps=20)
        if c < best_c5:
            best_c5 = c
            best_h = h_final

    # Final feasibility check
    best_h = project_feasible(best_h, ts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
