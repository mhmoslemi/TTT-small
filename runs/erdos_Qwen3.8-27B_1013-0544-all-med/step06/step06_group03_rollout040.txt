def compute_c5(h):

    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def soft_max_obj(h, t, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    return t * (m / t + np.log(np.sum(exps))) * dx

def soft_max_grad(h, t, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            if shift < n:
                grad[shift:] += w * (1.0 - h[:n - shift])
                grad[:n - shift] -= w * h[shift:]
        else:
            s = -shift
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
    return grad * dx

def discrete_local_search(h_bin, max_iters=5000, improve_threshold=1e-10):

    c5 = np.max(corr) * (2.0 / n)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n, balanced to n/2 ones."""
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        # Count bits in i
        c = 0
        x = i
        while x:
            c += x & 1
            x >>= 1
        tm[i] = c % 2
    # Balance to exactly n/2 ones
    n_ones = tm.sum()
    target = n // 2
    if n_ones > target:
        # Flip some 1s to 0s
        ones = np.where(tm == 1)[0]
        to_flip = ones[:n_ones - target]
        tm[to_flip] = 0
    elif n_ones < target:
        zeros = np.where(tm == 0)[0]
        to_flip = zeros[:target - n_ones]
        tm[to_flip] = 1
    return tm.astype(float)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    return np.max(conv) * dx

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def swap_effect(corr, h, j, k):
    """Compute new corr after swapping h[j](1->0) and h[k](0->1)."""
    n = len(h)
    new_corr = corr.copy()
    # Flip j: 1->0 (delta=-1)
    new_corr[j:j+n] -= (1.0 - h[n-1::-1])
    new_corr[n-1-j:2*n-1-j] += h
    # Flip k: 0->1 (delta=+1)
    new_corr[k:k+n] += (1.0 - h[n-1::-1])
    new_corr[n-1-k:2*n-1-k] -= h
    return new_corr

    curr_max = np.max(corr)
    dx = 2.0 / n
    stagnation = 0

    for _ in range(max_passes):
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        best_improve = 0
        best_j, best_k = -1, -1
        best_new_corr = None
        # Try all zeros, limited ones candidates
        for j in zeros:
            # Only try a subset of ones for speed
            k_candidates = ones[rng.choice(len(ones), size=min(len(ones), 40), replace=False)] if len(ones) > 40 else ones
            for k in k_candidates:
                nc = swap_effect(corr, h, k, j)
                nm = np.max(nc)
                imp = curr_max - nm
                if imp > best_improve:
                    best_improve = imp
                    best_j, best_k = j, k
                    best_new_corr = nc
        if best_improve > 1e-12 and best_j >= 0:
            h[best_j] = 1.0
            h[best_k] = 0.0
            corr = best_new_corr
            curr_max -= best_improve
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 200:
                break
    return h, curr_max * dx

def binary_sa(h_bin, max_iters=5000, t_init=0.5, t_final=0.02, rng=None):
    """Simulated annealing on binary swaps."""
    n = len(h_bin)
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode="full").astype(float)
    curr_max = np.max(corr)
    best_max = curr_max
    best_h = h.copy()
    dx = 2.0 / n

    for p in range(max_iters):
        t = t_init * (t_final / t_init) ** (p / max_iters)
        zeros = np.where(h < 0.5)[0]
        ones = np.where(h > 0.5)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        j = zeros[rng.randint(len(zeros))]
        k = ones[rng.randint(len(ones))]
        nc = swap_effect(corr, h, k, j)
        nm = np.max(nc)
        delta = nm - curr_max
        if delta < 0 or rng.rand() < np.exp(-delta / max(t * dx, 1e-10)):
            h[j] = 1.0
            h[k] = 0.0
            corr = nc
            curr_max = nm
            if nm < best_max:
                best_max = nm
                best_h = h.copy()
    return best_h, best_max * dx

def thue_morse(n):
    tm = np.zeros(n, dtype=int)
    for i in range(n):
        c, x = 0, i
        while x:
            c += x & 1
            x >>= 1
        tm[i] = c % 2
    s, target = int(tm.sum()), n // 2
    if s > target:
        idx = np.where(tm == 1)[0]
        tm[idx[:s - target]] = 0
    elif s < target:
        idx = np.where(tm == 0)[0]
        tm[idx[:target - s]] = 1
    return tm.astype(float)

def balance_binary(h, n):
    h = h.copy()
    s, target = int(h.sum()), n // 2
    if s > target:
        idx = np.where(h > 0.5)[0]
        h[idx[:s - target]] = 0.0
    elif s < target:
        idx = np.where(h < 0.5)[0]
        h[idx[:target - s]] = 1.0
    return h

def soft_max_obj(h, t, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    return t * (m / t + np.log(np.sum(exps))) * dx

def soft_max_grad(h, t, n):
    dx = 2.0 / n
    conv = np.correlate(h, 1.0 - h, mode="full")
    m = np.max(conv)
    exps = np.exp((conv - m) / t)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for idx in range(len(conv)):
        w = weights[idx]
        if w < 1e-10:
            continue
        shift = idx - (n - 1)
        if shift >= 0:
            if shift < n:
                grad[shift:] += w * (1.0 - h[:n - shift])
                grad[:n - shift] -= w * h[shift:]
        else:
            s = -shift
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
    return grad * dx

def pgd_refine(h, n, max_iter=50000, lr=0.001, smooth_t=None):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for _ in range(max_iter):
        if smooth_t is not None:
            c5, grad = compute_c5(h), soft_max_grad(h, smooth_t, n)
        else:
            conv = np.correlate(h, 1.0 - h, mode="full")
            idx = np.argmax(conv)
            shift = idx - (n - 1)
            grad = np.zeros(n)
            if shift >= 0:
                grad[shift:] += (1.0 - h[:n - shift])
                grad[:n - shift] -= h[shift:]
            else:
                s = -shift
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] -= h[:n - s]
            c5 = np.max(conv) * (2.0 / n)
            grad *= (2.0 / n)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        h_new = project_feasible(h - lr * grad, n)
        c5_new = compute_c5(h_new)
        if c5_new > c5 + 1e-15:
            lr *= 0.5
            if lr < 1e-16:
                break
        else:
            h = h_new
            lr = min(lr * 1.03, 0.03)
    return best_h, best_c5

def obj_lbfgs(h, t, n):
    return soft_max_obj(h, t, n) + 200.0 * (np.sum(h) - n / 2.0) ** 2

def grad_lbfgs(h, t, n):
    return soft_max_grad(h, t, n) + 400.0 * (np.sum(h) - n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = thue_morse(64)

    n0 = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Greedy binary search from initial (thresholded)
    h_bin = balance_binary((h_init >= 0.5).astype(float), n0)
    h_g, c5_g = binary_greedy_search(h_bin, max_passes=500, rng=rng)
    if c5_g < best_c5:
        best_c5, best_h = c5_g, h_g.copy()

    # Phase 2: Multi-start with different structures
    starts = []
    starts.append(thue_morse(n0))
    alt = np.zeros(n0); alt[0::2] = 1.0
    starts.append(balance_binary(alt, n0))
    for _ in range(4):
        starts.append(balance_binary((rng.rand(n0) < 0.5).astype(float), n0))
    # Block structure
    hblk = np.zeros(n0)
    pos = 0
    while pos < n0:
        bs = max(1, n0 // 12) + rng.randint(-2, 3)
        bs = max(1, min(bs, n0 - pos))
        hblk[pos:pos+bs] = 1.0 if int(hblk[:pos].sum()) * 2 < pos else 0.0
        pos += bs
    starts.append(balance_binary(hblk, n0))

    for h_s in starts:
        if remaining() < 40:
            break
        h_g2, c5_g2 = binary_greedy_search(h_s, max_passes=300, rng=rng)
        if c5_g2 < best_c5:
            best_c5, best_h = c5_g2, h_g2.copy()
        # SA polish on best binary
        h_sa, c5_sa = binary_sa(h_g2, max_iters=3000, t_init=0.3, t_final=0.01, rng=rng)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()

    # Phase 3: Continuous refinement
    if remaining() > 60:
        hc = best_h.copy()
        for tv in [0.08, 0.02, 0.005]:
            if remaining() < 25:
                break
            try:
                res = minimize(obj_lbfgs, hc, jac=grad_lbfgs, args=(tv, n0),
                              method='L-BFGS-B', bounds=[(0.0, 1.0)] * n0,
                              options={'maxiter': 3000, 'ftol': 1e-16})
                hc = project_feasible(res.x, n0)
                c5n = compute_c5(hc)
                if c5n < best_c5:
                    best_c5, best_h = c5n, hc.copy()
            except Exception:
                pass
        if remaining() > 30:
            hp, c5p = pgd_refine(best_h, n0, max_iter=40000, lr=0.002, smooth_t=0.01)
            if c5p < best_c5:
                best_c5, best_h = c5p, hp.copy()
        if remaining() > 30:
            hp2, c5p2 = pgd_refine(best_h, n0, max_iter=60000, lr=0.0004)
            if c5p2 < best_c5:
                best_c5, best_h = c5p2, hp2.copy()

    # Phase 4: Higher resolution
    for factor in [2, 3]:
        n_hi = n0 * factor
        if n_hi > 900 or remaining() < 100:
            break
        x_lo = np.linspace(0, 2.0, n0, endpoint=False)
        x_hi = np.linspace(0, 2.0, n_hi, endpoint=False)
        h_hi = np.interp(x_hi, x_lo, best_h)
        h_hi_bin = balance_binary((h_hi >= 0.5).astype(float), n_hi)
        h_hi2, c5_hi = binary_greedy_search(h_hi_bin, max_passes=200, rng=rng)
        if c5_hi < best_c5:
            best_c5, best_h, n0 = c5_hi, h_hi2.copy(), n_hi
        if remaining() > 40:
            hp3, c5p3 = pgd_refine(best_h, n0, max_iter=30000, lr=0.001, smooth_t=0.01)
            if c5p3 < best_c5:
                best_c5, best_h = c5p3, hp3.copy()

    # Final polish
    if remaining() > 15:
        hf, c5f = pgd_refine(best_h, len(best_h), max_iter=40000, lr=0.0002)
        if c5f < best_c5:
            best_c5, best_h = c5f, hf.copy()

    return best_h, best_c5, len(best_h)
