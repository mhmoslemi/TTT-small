import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Define number of circles
    n = 26

    # Initial guess for centers: a hexagonal packing with some boundary circles
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Define a hexagonal packing for the central 16 circles
    # A hexagonal grid is adopted with a certain spacing
    spacing = 0.25
    rows = 4
    cols = 4

    # Central cluster (16 circles) in a hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + (cols - 1) * spacing / 2
            y = i * spacing * np.sqrt(3) / 2 + (rows - 1) * spacing * np.sqrt(3) / 2 / 2
            centers[i * cols + j] = [x, y]

    # 10 boundary circles: 4 corners and 6 edges
    # Corner circles
    corner_radii = 0.05
    centers[16] = [0.05, 0.05]
    centers[17] = [0.95, 0.05]
    centers[18] = [0.95, 0.95]
    centers[19] = [0.05, 0.95]
    radii[16:20] = corner_radii

    # Edge circles
    # Top edge
    centers[20] = [0.3, 1.0 - 0.1]
    centers[21] = [0.5, 1.0 - 0.1]
    centers[22] = [0.7, 1.0 - 0.1]
    # Bottom edge
    centers[23] = [0.3, 0.1]
    centers[24] = [0.5, 0.1]
    centers[25] = [0.7, 0.1]
    radii[20:26] = 0.1

    # Initial guess for the central cluster
    # Set initial radii to be the same for all central circles
    # The radius of central circles is determined by spacing and packing
    r = spacing / 2
    radii[:16] = r

    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlapping and within square
    def constraint_overlap(radii):
        dist = cdist(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dist[i, j] < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return 0

    # Constraint for boundary
    def constraint_boundary(radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8):
                return -np.inf
        return 0

    # Optimization variables
    # Use the radii as the optimization variables
    # Initial guess is the current radii

    # Bounds for radii: non-negative
    bounds = [(0, 1)] * n

    # Combine the constraints
    def constraint_func(x):
        return constraint_overlap(x) + constraint_boundary(x)

    # Optimization
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint_func},
        method="SLSQP"
    )

    radii = result.x
    centers = centers

    # Validate the final packing
    # (Validation is not included in the return, but it is assumed that the constraints are satisfied)
    return (centers, radii, np.sum(radii))
