import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Initial guess: hexagonal packing with 5 rows and 5 columns, then one extra circle
    num_circles = 26

    # Initial positions for hexagonal packing
    rows = 5
    cols = 5
    radius_initial = 0.1  # Initial radius for all circles
    positions = []
    for row in range(rows):
        for col in range(cols):
            x = col * (2 * radius_initial) + radius_initial
            y = row * (np.sqrt(3) * radius_initial) + radius_initial
            positions.append([x, y])
    # Add one extra circle
    positions.append([0.5, 1.0 - radius_initial])
    positions = np.array(positions)
    radii = np.full(num_circles, radius_initial)

    # Define the objective function to maximize the sum of radii
    def objective(radii_vec):
        return -np.sum(radii_vec)  # Negative for minimization

    # Define constraints: circles within the unit square and no overlap
    def constraint_center_in_square(radii_vec):
        # Center positions can't be too close to edges
        return np.sum((positions[:, 0] - radii_vec) >= 0) + np.sum((1 - positions[:, 1] - radii_vec) >= 0)
    
    def constraint_no_overlap(radii_vec):
        # Distance between centers should be at least sum of radii
        distances = squareform(pdist(positions))
        for i, j in combinations(range(num_circles), 2):
            if distances[i, j] < radii_vec[i] + radii_vec[j] - 1e-8:
                return False
        return True
    
    # Set up the optimization problem
    bounds = [(0.0, 1.0 - 1e-8) for _ in range(num_circles)]  # Radii must be at least 0, and not too large
    cons = [
        {'type': 'ineq', 'fun': lambda r: constraint_center_in_square(r)},
        {'type': 'ineq', 'fun': constraint_no_overlap}
    ]
    
    # Optimization
    result = minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-10)
    
    # Extract results
    centers = positions
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Validate to ensure everything is correct
    # Note: This is just to ensure that the final result is valid, but it's not required as the optimization should take care of it
    # However, in practice, we may still need to manually check for validity
    
    return centers, radii, sum_radii
