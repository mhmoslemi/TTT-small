    def objective(x):
        # x is flattened array [x1, y1, r1, x2, y2, r2, ...]
        # Reconstruct centers and radii
        n = len(centers)
        flat_centers = x[:n*2].reshape(n, 2)
        flat_radii = x[n*2:].reshape(n)

        return -sum_radii  # Minimize negative to maximize

        # Check all pairwise distances and boundary constraints
        n = len(centers)
        flat_centers = x[:n*2].reshape(n, 2)
        flat_radii = x[n*2:].reshape(n)

            x_i, y_i = flat_centers[i]
            r_i = flat_radii[i]

            for j in range(i+1, n):
                dx = flat_centers[i, 0] - flat_centers[j, 0]
