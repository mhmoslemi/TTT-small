import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing():
    # Initial placement of 25 circles using a hexagonal packing
    # 5 rows with 5, 5, 5, 5, 5 circles respectively (but since it's a square, 5 rows of 5 circles)
    # The centers are arranged in a grid, with vertical and horizontal spacing
    # We'll first create a base configuration and then optimize

    # Initial grid of centers in a 5x5 grid
    rows = 5
    cols = 5
    spacing = 1.0 / (rows - 1)  # equal spacing between centers in unit square

    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing
            centers.append([x, y])

    centers = np.array(centers)

    # Initial guess for radii: set all to 0.1 (this will be optimized)
    radii = np.full(25, 0.1)

    # Objective function: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative sum to maximize the sum

    # Constraints: non-overlap between circles and within square boundaries
    def constraint_overlap(radii):
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist - (radii[i] + radii[j]) >= -1e-12

    def constraint_boundary(radii):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        return True

    # Create bounds for radii (non-negative)
    bounds = Bounds(lb=0.0, ub=np.inf)

    # Define constraints for the optimization
    nonlinear_constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            def constraint_overlap_ij(radii):
                return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (radii[i] + radii[j]) >= -1e-12
            nonlinear_constraints.append(NonlinearConstraint(constraint_overlap_ij, -np.inf, 0))

    # Run optimization
    res = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=nonlinear_constraints,
        method='SLSQP',
        options={'ftol': 1e-12, 'maxiter': 1000}
    )

    # Extract optimized centers and radii
    optimized_centers = centers
    optimized_radii = res.x

    # Add one more circle in the corner to maximize the radius
    # Place it in the corner that has the most space
    # Try placing it at the bottom-left corner (0,0) with maximum possible radius
    def corner_circle_constraints(radii):
        # Check if the new circle is within the square
        x, y = 0, 0
        r = radii[-1]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
        # Check overlap with existing circles
        for i in range(len(optimized_centers)):
            dist = np.sqrt((x - optimized_centers[i][0])**2 + (y - optimized_centers[i][1])**2)
            if dist - (r + optimized_radii[i]) < -1e-12:
                return False
        return True

    # Add the new circle as the 26th circle with an initial guess for radius
    new_centers = np.append(optimized_centers, [[0, 0]], axis=0)
    new_radii = np.append(optimized_radii, [0.1])

    # Reformulate the problem with the new circle
    def objective_with_corner(radii):
        return -np.sum(radii)

    def constraint_corner(radii):
        x, y = 0, 0
        r = radii[-1]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            return False
        for i in range(len(optimized_centers)):
            dist = np.sqrt((x - optimized_centers[i][0])**2 + (y - optimized_centers[i][1])**2)
            if dist - (r + radii[i]) < -1e-12:
                return False
        return True

    bounds_with_corner = Bounds(lb=0.0, ub=np.inf)

    # Define constraints for the 26th circle
    constraints_with_corner = []
    constraints_with_corner.append(NonlinearConstraint(constraint_corner, -np.inf, 0))

    # Run optimization with the new circle
    res_corner = minimize(
        objective_with_corner,
        new_radii,
        bounds=bounds_with_corner,
        constraints=constraints_with_corner,
        method='SLSQP',
        options={'ftol': 1e-12, 'maxiter': 1000}
    )

    centers_final = new_centers
    radii_final = res_corner.x

    return (centers_final, radii_final, np.sum(radii_final))
