import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial guess for hexagonal packing
    n = 26
    cols = 6
    rows = 5
    spacing = 1.0 / (cols + 1)
    max_radius = 0.15

    # Initialize centers in a hexagonal grid
    centers = np.zeros((n, 2))
    radii = np.full(n, max_radius)

    # Fill the grid
    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            if idx >= n:
                break
            x = j * spacing + spacing
            y = i * (spacing * np.sqrt(3)) + spacing
            centers[idx] = [x, y]

    # Define objective function to maximize sum of radii
    def obj_func(params):
        # params: flattened array of [x1, y1, r1, x2, y2, r2, ...]
        params = params.reshape((n, 3))
        centers = params[:, :2]
        radii = params[:, 2]
        return -np.sum(radii)  # Negative for minimization

    # Define constraints: distance between circles >= sum of radii
    def cons_func(params):
        params = params.reshape((n, 3))
        centers = params[:, :2]
        radii = params[:, 2]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append(dist - (radii[i] + radii[j]) >= 0)
        return constraints

    # Bounds for parameters: x, y in [0, 1], radius >= 0
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1), (0, 1.0)])

    # Initial guess
    initial_guess = np.zeros(n * 3)
    for i in range(n):
        initial_guess[i * 3 : (i + 1) * 3] = centers[i] + [radii[i]]

    # Optimize
    result = opt.minimize(
        obj_func,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': cons_func},
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract results
    params = result.x
    params = params.reshape((n, 3))
    centers = params[:, :2]
    radii = params[:, 2]
    sum_radii = np.sum(radii)

    # Return centers, radii, sum_radii
    return centers, radii, sum_radii
