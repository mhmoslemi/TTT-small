import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    def objective(radii, centers):
        return -np.sum(radii)  # Maximize sum of radii by minimizing negative sum

    def constraint_overlap(centers, radii, i, j):
        dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
        return dist - (radii[i] + radii[j])

    def constraint_boundary(centers, radii, i):
        x, y = centers[i]
        r = radii[i]
        return x - r, y - r, x + r - 1.0, y + r - 1.0

    # Initial grid: hexagonal packing with spacing
    n = 26
    grid_size = np.sqrt(n)
    spacing = 1.0 / (grid_size + 0.5)
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = 0.2 + col * spacing
        centers[i, 1] = 0.2 + row * spacing

    radii = np.full(n, 0.01)

    # Define bounds for radii
    bounds = Bounds(np.zeros(n), np.inf)

    # Optimization with constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_overlap(centers, x, i, j)})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(centers, x, i)[0]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(centers, x, i)[1]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(centers, x, i)[2]})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(centers, x, i)[3]})

    # Create a new optimization problem
    def objective_func(x):
        return objective(x, centers)

    result = optimize.minimize(objective_func, radii, method='SLSQP', bounds=bounds, constraints=constraints)

    if not result.success:
        raise ValueError("Optimization failed")

    centers = np.array(result.x).reshape((n, 2))
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
