import numpy as np
from scipy.optimize import minimize
import time

def run_packing():
    # Initial grid-based packing with hexagonal arrangement
    num_circles = 26
    grid_size = 4
    spacing = 0.4  # Initial spacing for hexagonal grid

    # Initial hexagonal grid packing
    centers = []
    radii = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = j * spacing
            y = i * spacing + (j % 2) * spacing / 2
            if x < 1 - 1e-6 and y < 1 - 1e-6:
                centers.append([x, y])
                radii.append(1e-6)

    # Expand initial packing
    if len(centers) < num_circles:
        # Add additional circles in the remaining space
        added = 0
        for i in range(100):
            # Use random candidate positions
            x = np.random.uniform(0, 1)
            y = np.random.uniform(0, 1)
            radius = 1e-6
            overlap = False
            for j in range(len(centers)):
                dx = centers[j][0] - x
                dy = centers[j][1] - y
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[j] + radius - 1e-12:
                    overlap = True
                    break
            if not overlap:
                centers.append([x, y])
                radii.append(1e-6)
                added += 1
                if added == num_circles - len(centers):
                    break

    # Optimization: maximize sum of radii
    def objective(r):
        return -np.sum(r)

    def constraints(r):
        if np.any(r < 0):
            return np.inf
        for i in range(len(centers)):
            x, y = centers[i]
            r_i = r[i]
            if x - r_i < -1e-12 or x + r_i > 1 + 1e-12 or y - r_i < -1e-12 or y + r_i > 1 + 1e-12:
                return np.inf
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-12:
                    return np.inf
        return 0

    bounds = [(0.0, 1.0) for _ in range(num_circles)]

    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 200}
    )

    if not result.success:
        raise Exception("Optimization failed")

    final_radii = result.x
    final_centers = np.array(centers)
    sum_radii = np.sum(final_radii)

    return (final_centers, final_radii, sum_radii)
