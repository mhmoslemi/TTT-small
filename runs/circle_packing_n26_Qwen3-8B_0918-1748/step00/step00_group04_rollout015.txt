import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Initial hexagonal packing with some fixed boundary circles
    num_circles = 26
    
    # Corner circles (4 corners)
    corner_positions = np.array([
        [0.0, 0.0], [1.0, 0.0], [1.0, 1.0], [0.0, 1.0]
    ])
    
    # Remaining circles (22) placed in a hexagonal packing
    # We'll use a grid that fits within the unit square
    
    # Define initial positions for the 22 circles
    # This is a simplified 4 x 6 hexagonal grid (with spacing adjusted)
    # We'll use offset rows to simulate hexagonal packing
    positions = []
    for row in range(4):
        for col in range(6):
            x = col / 5.0 + 0.1
            y = row / 3.0 + 0.15
            positions.append([x, y])
    
    # Add corner positions
    positions = np.array(positions)
    
    # Set initial guess for radii
    # Compute minimal radius to avoid overlap in initial placement
    dist_matrix = squareform(pdist(positions))
    min_dist = np.min(dist_matrix[dist_matrix > 0])
    initial_radius = min_dist / 2.0
    
    # Initial radii
    radii = np.full(num_circles, initial_radius)
    
    # Define a function to compute constraints
    def constraints(x):
        centers = x[:num_circles * 2].reshape((num_circles, 2))
        radii = x[num_circles * 2:]
        # Check boundaries
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.array([0.0])
        # Check overlaps
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.array([0.0])
        return np.array([])
    
    # Flatten initial parameters: centers (n*2) + radii (n)
    initial_guess = np.concatenate((positions.ravel(), radii))
    
    # Optimization bounds
    bounds = Bounds(
        lb=np.concatenate((np.zeros(num_circles * 2), np.zeros(num_circles))),
        ub=np.concatenate((np.ones(num_circles * 2), np.full(num_circles, np.inf)))
    )
    
    # Define objective function: maximize sum of radii (minimize negative sum)
    def objective(x):
        centers = x[:num_circles * 2].reshape((num_circles, 2))
        radii = x[num_circles * 2:]
        return -np.sum(radii)
    
    # Optimization with constraints
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        method='SLSQP'
    )
    
    # Extract final centers and radii
    centers = result.x[:num_circles * 2].reshape((num_circles, 2))
    radii = result.x[num_circles * 2:]
    
    # Final validation before returning
    assert not np.isnan(centers).any(), "NaN values found in centers"
    assert not np.isnan(radii).any(), "NaN values found in radii"
    
    # Return centers, radii, and sum
    return centers, radii, np.sum(radii)
