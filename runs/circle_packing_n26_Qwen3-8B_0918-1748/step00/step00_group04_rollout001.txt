import numpy as np
import random

def run_packing():
    def simulate_annealing():
        num_circles = 26
        initial_centers = np.random.rand(num_circles, 2)
        initial_radii = np.random.rand(num_circles) * 0.1
        best_centers = initial_centers.copy()
        best_radii = initial_radii.copy()
        best_sum = np.sum(best_radii)

        max_iterations = 10000
        temp = 1.0
        cooling_rate = 0.995

        for _ in range(max_iterations):
            new_centers = best_centers + np.random.randn(num_circles, 2) * 0.01
            new_radii = best_radii + np.random.randn(num_circles) * 0.005
            new_radii = np.clip(new_radii, 0, 0.1)

            # Check if new configuration is valid
            valid, message = validate_packing(new_centers, new_radii)
            if valid:
                new_sum = np.sum(new_radii)
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = new_centers
                    best_radii = new_radii

            temp *= cooling_rate

        return best_centers, best_radii, best_sum

    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    centers, radii, sum_radii = simulate_annealing()
    return centers, radii, sum_radii
