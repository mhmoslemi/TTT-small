import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():
    # Initial regular grid-based seed configuration
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Seed configuration: 4 corners, 8 edges, and 14 inside
    # Corner circles
    centers[0] = [0.1, 0.1]
    centers[1] = [0.1, 0.9]
    centers[2] = [0.9, 0.9]
    centers[3] = [0.9, 0.1]
    
    # Edge circles (2 per edge)
    centers[4] = [0.1, 0.5]
    centers[5] = [0.5, 0.1]
    centers[6] = [0.5, 0.9]
    centers[7] = [0.9, 0.5]
    
    # Main body circles
    num_main = num_circles - 8
    for i in range(num_main):
        row = i // 7
        col = i % 7
        x = col * (1.0 / 6.0) + 0.1
        y = row * (1.0 / 4.0) + 0.25
        centers[8 + i] = [x, y]
    
    # Initial radii: corner circles get small radii
    radii[0:4] = 0.05
    radii[4:8] = 0.05
    radii[8:] = 0.1
    
    # Optimization problem
    def objective(radii):
        return -np.sum(radii)  # We minimize negative sum
    
    def constraints(radii):
        constraints = []
        for i in range(num_circles):
            constraints.append((radii[i], 0, None))
        return constraints
    
    bounds = Bounds(lb=0, ub=None)
    
    # Create a vector of parameters: the radii
    param_bounds = [(0, 1)] * num_circles
    param_initial = radii
    
    def optimization_function(params):
        radii = params
        # Create a list of centers with radii to check distances
        # For each pair of circles, ensure distance >= radii[i] + radii[j]
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-6:
                    return np.inf  # Penalize overlap
        return -np.sum(radii)  # Objective to minimize negative sum
    
    # Run the optimization
    result = minimize(
        optimization_function,
        param_initial,
        method='L-BFGS-B',
        bounds=param_bounds,
        options={'maxiter': 1000, 'gtol': 1e-6}
    )
    
    # Extract the result
    radii = result.x
    centers = np.array(centers)
    
    # Final validation
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError("Invalid packing: " + message)
    
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"
