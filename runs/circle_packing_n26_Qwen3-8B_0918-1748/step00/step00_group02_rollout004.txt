import numpy as np
from scipy.optimize import minimize
from itertools import combinations

def generate_initial_hexagonal_packing(n):
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i + 0.5
            if x < 0 or x >= 1 or y < 0 or y >= 1:
                continue
            centers.append([x, y])
            if len(centers) == n:
                break
        if len(centers) == n:
            break
    return np.array(centers)

def run_packing():
    n = 26
    # Generate initial configuration
    initial_centers = generate_initial_hexagonal_packing(n)
    if len(initial_centers) < n:
        initial_centers = np.random.rand(n, 2)
    initial_radii = np.full(n, 0.01)
    
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraint_center_inside(radii):
        return np.array([0.0 for _ in range(n)])  # No explicit center constraint, handled by bounds
    
    def constraint_radius_non_negative(radii):
        return np.array([-r for r in radii])
    
    def constraint_non_overlap(radii):
        constraints = []
        for i, j in combinations(range(n), 2):
            def cons(x):
                return np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2) - (radii[i] + radii[j])
            constraints.append(cons)
        return constraints
    
    # Prepare centers
    centers = initial_centers
    
    # Define constraints
    constraints = []
    constraints.append({'type': 'ineq', 'fun': constraint_radius_non_negative})
    for i, j in combinations(range(n), 2):
        def cons(x, i=i, j=j):
            return np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2) - (x[i] + x[j])
        constraints.append({'type': 'ineq', 'fun': cons})
    
    # Bounds for radii
    bounds = [(0, 1) for _ in range(n)]
    
    # Run optimization
    result = minimize(objective, initial_radii, method='SLSQP', bounds=bounds, constraints=constraints)
    
    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")
    
    return (centers, radii, sum_radii)
