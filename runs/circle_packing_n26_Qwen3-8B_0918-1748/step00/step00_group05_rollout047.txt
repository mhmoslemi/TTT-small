import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Define initial center positions based on a hexagonal grid and corner placements
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Base grid: 2 rows of 6 + 1 row of 5 = 17 circles (with spacing)
    # But we need 26, so add 9 more circles in corners and edges
    
    # Generate columns and rows for a hexagonal grid
    cols = 6
    rows = 4
    grid_size = 1.0
    spacing = 0.35  # Adjusted for better packing
    
    # Add central cluster with hexagonal grid
    cluster_centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + 0.5 * spacing
            y = i * spacing * np.sqrt(3) / 2 + 0.5 * spacing * np.sqrt(3) / 2
            cluster_centers.append((x, y))
    
    cluster_centers = np.array(cluster_centers)
    cluster_centers = cluster_centers / np.max(cluster_centers) * (1 - 0.1) + 0.05
    
    # Add corner circles
    corners = np.array([
        [0.1, 0.1],
        [0.1, 0.9],
        [0.9, 0.1],
        [0.9, 0.9]
    ])
    
    # Add edge circles (left, right, top, bottom)
    edge_centers = np.array([
        [0.1, 0.5],
        [0.9, 0.5],
        [0.5, 0.1],
        [0.5, 0.9]
    ])
    
    # Combine all centers
    centers = np.vstack((cluster_centers, corners, edge_centers))
    
    # Initial radii: use distances between clusters to define initial values
    distances = np.array([np.sqrt(np.sum((centers[i] - centers[j])**2)) for i in range(len(centers)) for j in range(i+1, len(centers))])
    initial_radius = np.min(distances) / 2
    radii = initial_radius * np.random.rand(n)
    
    # Define the constraint function
    def constraint_func(x):
        # x is a flattened vector of centers and radii
        centers_flat = x[:len(centers)*2].reshape(len(centers), 2)
        radii_flat = x[len(centers)*2:]
        
        # Check that all circles are within bounds
        for i in range(len(centers_flat)):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return 1.0  # Violation
        # Check that no two circles overlap
        for i in range(len(centers_flat)):
            for j in range(i + 1, len(centers_flat)):
                dist = np.sqrt(np.sum((centers_flat[i] - centers_flat[j])**2))
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return 1.0  # Violation
        return 0.0
    
    # Prepare variables for optimization
    x0 = np.concatenate((centers.flatten(), radii))
    
    # Define bounds for centers and radii
    bounds = Bounds(
        lb=np.concatenate((np.zeros(len(centers)*2), np.zeros(len(radii)))),
        ub=np.concatenate((np.ones(len(centers)*2), np.ones(len(radii))))
    )
    
    # Define objective function to maximize sum of radii (minimize negative sum)
    def objective(x):
        return -np.sum(x[len(centers)*2:])  # Negative for minimization
    
    # Perform optimization
    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_func}],
        method='SLSQP'
    )
    
    # Extract the optimized centers and radii
    centers_opt = result.x[:len(centers)*2].reshape(len(centers), 2)
    radii_opt = result.x[len(centers)*2:]
    
    # Ensure all radii are non-negative
    radii_opt[radii_opt < 0] = 0
    
    # Final check
    if not np.isfinite(result.fun):
        raise ValueError("Optimization failed to converge")
    
    if constraint_func(result.x) > 1e-12:
        raise ValueError("Optimization constraint violation")
    
    # Compute sum of radii
    sum_radii = np.sum(radii_opt)
    
    return (centers_opt, radii_opt, sum_radii)
