import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import itertools

def run_packing():
    # Initial guess for radii of the 24 equal circles
    r0 = 0.04

    # Define positions for the 24 equal circles in a hexagonal packing
    d = 2 * r0  # diameter
    centers = []
    # Row 0: top row
    for i in range(8):
        x = 0.5 + i * d
        y = 1.0 - r0
        centers.append([x, y])
    # Row 1: middle row
    for i in range(8):
        x = 0.5 + i * d - d / 2
        y = 0.5
        centers.append([x, y])
    # Row 2: bottom row
    for i in range(8):
        x = 0.5 + i * d
        y = r0
        centers.append([x, y])
    centers = np.array(centers)
    # Add two more circles in the corners
    r_corner = 0.05
    centers = np.vstack([centers, [1 - r_corner, r_corner], [r_corner, 1 - r_corner]])
    # Initial radii: same for 24, then corner circles
    radii = np.array([r0] * 24 + [r_corner] * 2)

    def objective(x):
        return -np.sum(x)  # Maximize sum of radii

    def constraints(x):
        # Update radii
        radii_new = x
        # Check non-overlap between all pairs
        for i, j in itertools.combinations(range(len(radii_new)), 2):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii_new[i] + radii_new[j] - 1e-8:
                return False
        # Check boundary constraints
        for i in range(len(radii_new)):
            if centers[i, 0] - radii_new[i] < -1e-8 or centers[i, 0] + radii_new[i] > 1 + 1e-8:
                return False
            if centers[i, 1] - radii_new[i] < -1e-8 or centers[i, 1] + radii_new[i] > 1 + 1e-8:
                return False
        return True

    # Optimize
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=[(0, 1)] * len(radii),
        constraints=[{'type': 'ineq', 'fun': constraints}],
        method='SLSQP'
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    radii_opt = result.x
    centers_opt = centers
    sum_radii = np.sum(radii_opt)

    return (centers_opt, radii_opt, sum_radii)
