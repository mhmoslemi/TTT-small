import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist

def run_packing():
    # Define initial seed configuration
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Define initial seed configuration with hexagonal packing
    # 3 rows of 6, 3 rows of 7, 2 rows of 9
    y_positions = [0.2, 0.5, 0.8, 0.3, 0.6, 0.9, 0.1, 0.4, 0.7]
    x_positions = [0.1, 0.25, 0.5, 0.75, 0.9]
    row_counts = [3, 3, 2]

    idx = 0
    for i, y in enumerate(y_positions):
        row = []
        for x in x_positions:
            if y in [0.2, 0.5, 0.8]:
                row.append((x, y))
            elif y in [0.3, 0.6, 0.9]:
                row.append((x, y))
            elif y in [0.1, 0.4, 0.7]:
                row.append((x, y))
        for x, y in row:
            centers[idx] = [x, y]
            idx += 1

    centers[23] = [0.5, 0.5]  # Central circle

    # Initial rough radii: peak in central rows, smaller in corner rows
    for i in range(n):
        if 23 <= i <= 25:  # Central and nearby
            radii[i] = 0.1
        elif 12 <= i <= 19:  # Mid rows
            radii[i] = 0.08
        else:  # Outer rows
            radii[i] = 0.05

    # Optimization: maximize sum of radii with non-overlap and boundary constraints
    # Objective: maximize sum of radii
    def objective(x):
        x_flat = x.reshape(-1)
        centers_flat = x_flat[:2*n].reshape(n, 2)
        radii_flat = x_flat[2*n:]
        return -np.sum(radii_flat)  # minimize negative

    # Constraints: distance between centers >= radii[i] + radii[j]
    def constraint_dist(x):
        x_flat = x.reshape(-1)
        centers_flat = x_flat[:2*n].reshape(n, 2)
        radii_flat = x_flat[2*n:]
        dist_matrix = cdist(centers_flat, centers_flat)
        for i in range(n):
            for j in range(i+1, n):
                if dist_matrix[i, j] < radii_flat[i] + radii_flat[j] - 1e-8:
                    return dist_matrix[i, j] - (radii_flat[i] + radii_flat[j] - 1e-8)
        return 0

    # Boundary constraints: x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1
    def constraint_boundary(x):
        x_flat = x.reshape(-1)
        centers_flat = x_flat[:2*n].reshape(n, 2)
        radii_flat = x_flat[2*n:]
        for i in range(n):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return (x - r + 1e-8) or (x + r - 1 - 1e-8) or (y - r + 1e-8) or (y + r - 1 - 1e-8)
        return 0

    # Initial guess
    x0 = np.concatenate((centers.flatten(), radii))

    # Bounds: x and y between 0 and 1, radii >= 0
    bounds = [(0, 1) for _ in range(2*n)] + [(0, 1) for _ in range(n)]

    # Optimization setup
    cons = [{'type': 'ineq', 'fun': constraint_dist},
            {'type': 'ineq', 'fun': constraint_boundary}]

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'ftol': 1e-8, 'maxiter': 1000})

    # Extract final centers and radii
    centers_flat = result.x[:2*n].reshape(n, 2)
    radii_flat = result.x[2*n:]
    sum_radii = np.sum(radii_flat)

    return centers_flat, radii_flat, sum_radii
