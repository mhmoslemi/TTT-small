import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Define the number of circles
    n = 26
    
    # Define the boundaries
    lb = np.array([0.0 for _ in range(2 * n)])
    ub = np.array([1.0 for _ in range(2 * n)])
    
    # Initial guess for centers (grid-like distribution)
    centers_initial = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        centers_initial[i] = [col / 4.5, row / 4.5]
    
    # Variable bounds (centers and radii)
    bounds = Bounds(lb, ub)
    x0 = centers_initial.flatten()
    
    # Objective function to maximize sum of radii (minimize negative of sum)
    def objective(x):
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]
        return -np.sum(radii)
    
    # Constraints for pairwise distances between circles
    def constraint(x):
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - (radii[i] + radii[j] - 1e-8)
        return 0
    
    # Nonlinear constraint function
    cons = {'type': 'ineq', 'fun': constraint}
    
    # Optimization
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Extract centers and radii
    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:]
    
    # Validate packing
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")
    
    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
