import numpy as np
import numpy.random as npr
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Define key parameters
    n_circles = 26
    max_iter = 100
    num_simulations = 2
    tolerance = 1e-6
    initial_radius_guess = 0.1

    # Function to compute total radius sum
    def total_radius_sum(radii):
        return np.sum(radii)

    # Function to compute valid packing
    def is_valid(centers, radii):
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False
        for i in range(n_circles):
            x, y = centers[i]
            if x - radii[i] < -tolerance or x + radii[i] > 1 + tolerance:
                return False
            if y - radii[i] < -tolerance or y + radii[i] > 1 + tolerance:
                return False
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - tolerance:
                    return False
        return True

    # Function to optimize packing with constraints
    def objective(x):
        centers = x[:2*n_circles].reshape(n_circles, 2)
        radii = x[2*n_circles:]
        if not is_valid(centers, radii):
            return -np.inf
        return -total_radius_sum(radii)

    # Function to generate initial guess
    def generate_initial_guess():
        centers = np.random.uniform(0, 1, (n_circles, 2))
        radii = np.full(n_circles, initial_radius_guess)
        return np.concatenate((centers.reshape(-1), radii))

    # Initialize best result
    best_centers = np.zeros((n_circles, 2))
    best_radii = np.zeros(n_circles)
    best_sum = 0.0

    # Try multiple simulations
    for sim in range(num_simulations):
        initial_guess = generate_initial_guess()
        bounds = Bounds([], [])
        result = minimize(objective, initial_guess, bounds=bounds, method='L-BFGS-B', options={'maxiter': max_iter})
        if not np.isfinite(result.fun):
            continue
        centers = result.x[:2*n_circles].reshape(n_circles, 2)
        radii = result.x[2*n_circles:]
        if is_valid(centers, radii):
            sum_radii = total_radius_sum(radii)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = centers
                best_radii = radii

    return best_centers, best_radii, best_sum
