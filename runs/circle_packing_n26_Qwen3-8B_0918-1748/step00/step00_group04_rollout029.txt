import numpy as np
import numpy.random as npr
import scipy.optimize as opt
from scipy.optimize import NonlinearConstraint
import matplotlib.pyplot as plt

def run_packing():
    n = 26
    # Initial guess for radii, assuming uniform size based on hexagonal packing
    r0 = 0.03
    radii = np.full(n, r0)
    centers = np.random.rand(n, 2)
    
    # Function to compute pairwise distances
    def pairwise_distances(centers):
        distances = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                distances[i, j] = np.sqrt(dx**2 + dy**2)
                distances[j, i] = distances[i, j]
        return distances
    
    # Objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)
    
    # Constraints
    def constraint_func(x):
        centers = np.reshape(x, (n, 2))
        radii = x[n:]
        dists = pairwise_distances(centers)
        overlaps = dists - (radii[:, np.newaxis] + radii[np.newaxis, :])
        return np.min(overlaps)
    
    cons = NonlinearConstraint(constraint_func, -1e-9, 0)
    
    # Optimize
    x0 = np.concatenate([centers.flatten(), radii])
    result = opt.minimize(
        objective,
        x0,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-8}
    )
    
    # Extract solution
    centers = np.reshape(result.x[:2*n], (n, 2))
    radii = result.x[2*n:]
    
    # Final validation
    assert np.all(radii >= 0), "Negative radius detected"
    assert np.all(centers >= 0), "Negative center coordinates detected"
    assert np.all(centers <= 1), "Center coordinates exceed unit square"
    
    # Check if circles are within the square (with some tolerance)
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError("Circle is outside the unit square")
    
    # Check circle overlaps
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError("Circles are overlapping")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
