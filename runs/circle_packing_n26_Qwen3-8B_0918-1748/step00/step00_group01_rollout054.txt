import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Known near-optimal configuration for 26 circles in a unit square
    # Contains 5 rows, 5 columns and a grid with 4 rows and 5 columns.
    # The configuration is borrowed from packing results.
    centers = np.array([
        [0.1, 0.1],
        [0.4, 0.1],
        [0.7, 0.1],
        [0.1, 0.4],
        [0.4, 0.4],
        [0.7, 0.4],
        [0.1, 0.7],
        [0.4, 0.7],
        [0.7, 0.7],
        [0.1, 0.1],
        [0.1, 0.4],
        [0.1, 0.7],
        [0.4, 0.1],
        [0.4, 0.4],
        [0.4, 0.7],
        [0.7, 0.1],
        [0.7, 0.4],
        [0.7, 0.7],
        [0.2, 0.2],
        [0.5, 0.2],
        [0.8, 0.2],
        [0.2, 0.5],
        [0.5, 0.5],
        [0.8, 0.5],
        [0.2, 0.8],
        [0.5, 0.8],
        [0.8, 0.8]
    ])
    radii = np.array([0.05]*26)
    
    # Coarsely adjust radii based on grid
    for i in range(26):
        if i < 9:
            radii[i] = 0.17
        elif i < 18:
            radii[i] = 0.12
        else:
            radii[i] = 0.15
    
    def objective_function(radii):
        return -np.sum(radii)
    
    # Constraints: circles must not overlap and fit in the square
    def constraint_func(radii):
        dist_matrix = cdist(centers, centers)
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                    return 1
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return 1
        return 0
    
    # Optimization: maximize radii sum with non-overlapping and boundary constraints
    bounds = [(0, 0.5)] * 26
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    result = minimize(objective_function, radii, bounds=bounds, constraints=cons)
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Validate the final configuration
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError("Invalid configuration")
    
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
