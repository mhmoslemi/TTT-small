import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize.constraints import LinearConstraint

def run_packing():
    # Define constants
    n = 26
    max_iter = 100
    eps = 1e-6

    # Initial guess for centers: central hexagonal grid (16 circles), 4 top edge, 4 bottom edge, 2 bottom corners, 2 top corners
    # Central hexagonal cluster (4x4 grid with staggered rows)
    grid_centers = np.zeros((16, 2))
    for i in range(4):
        for j in range(4):
            x = i + (j % 2) * 0.5
            y = j * np.sqrt(3) / 2
            grid_centers[i * 4 + j] = [x, y]
    grid_centers = (grid_centers - grid_centers.min(axis=0)) / (grid_centers.max(axis=0) - grid_centers.min(axis=0))
    grid_centers = np.clip(grid_centers, 0, 1)

    # Top edge circles: 4 circles placed along top edge
    top_centers = np.array([
        [0.25, 0.95],
        [0.375, 0.95],
        [0.625, 0.95],
        [0.75, 0.95]
    ])
    # Bottom edge circles: 4 circles placed along bottom edge
    bottom_centers = np.array([
        [0.25, 0.05],
        [0.375, 0.05],
        [0.625, 0.05],
        [0.75, 0.05]
    ])
    # Bottom corners: 2 circles
    bottom_left = [0.05, 0.05]
    bottom_right = [0.95, 0.05]
    # Top corners: 2 circles
    top_left = [0.05, 0.95]
    top_right = [0.95, 0.95]

    # Combine all centers
    centers = np.vstack((grid_centers, top_centers, bottom_centers, [bottom_left, bottom_right, top_left, top_right]))

    # Initial radii: uniform
    initial_radii = np.full(n, 0.05)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints: distance between centers >= sum of radii
    def constraints_func(x):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                constraint = dist - (x[i] + x[j])
                constraints.append(constraint)
        return constraints

    # Boundary constraints: circle within [0,1]^2
    def boundary_constraints(x):
        constraints = []
        for i in range(n):
            constraint = np.clip(centers[i, 0] - x[i], -1e-6, 1)
            constraints.append(constraint)
            constraint = np.clip(centers[i, 1] - x[i], -1e-6, 1)
            constraints.append(constraint)
            constraint = np.clip(x[i] + centers[i, 0], -1e-6, 1)
            constraints.append(constraint)
            constraint = np.clip(x[i] + centers[i, 1], -1e-6, 1)
            constraints.append(constraint)
        return constraints

    # Optimization bounds (radii must be non-negative)
    bounds = Bounds(0, 1)

    # Set up the optimization problem
    x0 = initial_radii
    cons = [{'type': 'ineq', 'fun': constraints_func},
            {'type': 'ineq', 'fun': boundary_constraints}]
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6, max_iter=max_iter)

    centers = centers
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
