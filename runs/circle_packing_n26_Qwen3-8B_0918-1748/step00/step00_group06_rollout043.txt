```python
import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Define initial positions for corner and side circles
    corners = np.array([
        [0.0, 0.0],
        [1.0, 0.0],
        [1.0, 1.0],
        [0.0, 1.0]
    ])
    side_centers = np.array([
        [0.0, 0.5],
        [0.5, 0.0],
        [1.0, 0.5],
        [0.5, 1.0]
    ])
    center = np.array([0.5, 0.5])

    # Initial configuration: 4 corners, 4 sides, and 18 central circles in a hexagonal pattern
    num_corner = 4
    num_side = 4
    num_central = 18
    num_total = num_corner + num_side + num_central

    # Initial positions: corner, side, and central
    centers = np.vstack([corners, side_centers, center])
    # Initialize radii as equal for all
    radii = np.ones(num_total) * 0.05  # Initial guess

    # Define an optimization problem to maximize the total radius
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum to maximize it

    # Constraints: distance between centers >= sum of radii
    def constraint_func(x):
        # x is the vector of radii
        r = x
        dists = squareform(pdist(centers))
        for i in range(num_total):
            for j in range(i + 1, num_total):
                if dists[i, j] < r[i] + r[j] - 1e-12:
                    return False
        return True

    # Prepare optimization variables
    bounds = [(0.0, 1.0) for _ in range(num_total)]
    r0 = radii.copy()

    # Optimization using SLSQP
    result = opt.minimize(
        objective,
        r0,
        bounds=bounds,
        constraints={"type": "constraint", "fun": constraint_func, "jac": None},
        method="SLSQP"
    )

    # Extract result
    final_radii = result.x
    final_centers = centers

    # Ensure all circles are within the unit square
    for i in range(num_total):
        x, y = final_centers[i]
        r = final_radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            raise ValueError("At least one circle is outside the unit square")

    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
```
