    cols = 5

    radii_hex = []
    total_radius = 0

        for j in range(cols):
            x = j * spacing + (i % 2) * spacing / 2
            y = i * spacing * np.sqrt(3) / 2
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            centers_hex.append([x, y])
            radii_hex.append(radius_base)
            total_radius += radius_base

    # Add the 26th circle in the central remaining space
    # Use a heuristic to find a position that doesn't overlap with existing circles
    # Here, we use a simple greedy approach to find the best spot

    def find_best_position(centers, radii):
        best_pos = None
        best_radius = 0
        for dx in np.linspace(0, 1, 100):
            for dy in np.linspace(0, 1, 100):
                pos = np.array([dx, dy])
                min_dist = np.inf
                for i in range(len(centers)):
                    dist = np.linalg.norm(pos - centers[i])
                    min_dist = min(min_dist, dist - radii[i])
                if min_dist < 0:
                    continue
                max_radius = min_dist
                if max_radius > best_radius:
                    best_radius = max_radius
                    best_pos = pos
        return best_pos, best_radius

    radii_hex.append(best_radius)
    total_radius += best_radius

    # We define variables for x and y of each center, and for each radius
    n = len(centers_hex)
    all_centers = np.array(centers_hex)
    all_radii = np.array(radii_hex)

    # Objective function: maximize the sum of radii
    def obj_function(variables):
        # variables are (x1, y1, r1, x2, y2, r2, ...)
        flat_vars = variables.reshape(n, 4)

        radii = flat_vars[:, 2]
        return -np.sum(radii)  # Minimize the negative sum to maximize sum

            x, y = centers[i]
            r = radii[i]

        return 0

        flat_vars = variables.reshape(n, 4)
        radii = flat_vars[:, 2]
        for r in radii:
            if r < -1e-12:
                return -1
        return 0

    def constraint_non_overlap(variables):
        flat_vars = variables.reshape(n, 4)
        centers = flat_vars[:, :2]
        radii = flat_vars[:, 2]
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    variables = np.concatenate([all_centers.reshape(-1), all_radii])

        {'type': 'ineq', 'fun': constraint_inside_unit_square},
        {'type': 'ineq', 'fun': constraint_non_negative_radius},
        {'type': 'ineq', 'fun': constraint_non_overlap},
    ]

    # Run optimization
    result = minimize(obj_function, variables, method='SLSQP', bounds=Bounds(bounds), constraints=constraints)
