import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Initial grid-based layout with hexagonal packing
    n = 26
    rows = 6
    cols = 5
    horizontal_spacing = 1.0 / (cols + 1)
    vertical_spacing = 1.0 / (rows + 1)
    
    centers = []
    for i in range(rows):
        row_centers = []
        for j in range(cols):
            x = j * horizontal_spacing + horizontal_spacing / 2
            y = i * vertical_spacing + vertical_spacing / 2
            if i % 2 == 1:
                x += horizontal_spacing / 2
            row_centers.append((x, y))
        centers.extend(row_centers)
    
    # Adjust to make it a bit denser by overlapping slightly
    for i in range(len(centers)):
        x, y = centers[i]
        centers[i] = (x, y + 0.05)
    
    # Convert to numpy array
    centers = np.array(centers)
    
    # Initial radii based on the grid spacing
    initial_radius = 0.04
    radii = np.full(n, initial_radius)
    
    # Define the objective function (negative for minimization)
    def objective(radii):
        return -np.sum(radii)
    
    # Define the constraints
    constraints = []
    
    # Boundary constraints
    for i in range(n):
        def boundary_constraint(radii, i=i):
            x, y = centers[i]
            return (x - radii[i]) + 1e-12, (y - radii[i]) + 1e-12, (1 - x - radii[i]) + 1e-12, (1 - y - radii[i]) + 1e-12
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: (r[i] + 1e-12) - (1 - centers[i, 0])})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: (r[i] + 1e-12) - (1 - centers[i, 1])})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: (r[i] + 1e-12) - (centers[i, 0] + 1e-12)})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: (r[i] + 1e-12) - (centers[i, 1] + 1e-12)})
    
    # Distance constraints between circles
    for i in range(n):
        for j in range(i + 1, n):
            def distance_constraint(radii, i=i, j=j):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                return dist - (radii[i] + radii[j] - 1e-12)
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: distance_constraint(r, i, j)})
    
    # Optimize
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=Bounds(0, 1),
        constraints=constraints
    )
    
    # Validate result
    if not result.success:
        raise ValueError("Optimization failed")
    
    # Final centers and radii
    centers_result = centers.copy()
    radii_result = result.x
    sum_radii = np.sum(radii_result)
    
    # Final validation step
    is_valid, msg = validate_packing(centers_result, radii_result)
    if not is_valid:
        raise ValueError(f"Validation failed: {msg}")
    
    return centers_result, radii_result, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
