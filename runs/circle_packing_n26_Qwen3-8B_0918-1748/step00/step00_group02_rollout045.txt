from scipy.optimize import Bounds

def run_packing():
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(centers, radii):
        # Boundary constraints: each circle's center must be at least r away from the square's sides
        for i in range(n):
            r = radii[i]
            if centers[i, 0] - r < 0 or centers[i, 0] + r > 1:
                return False
            if centers[i, 1] - r < 0 or centers[i, 1] + r > 1:
                return False
        # Non-overlap constraints: distance between centers >= sum of radii
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    n = 26
    # Initial guess: use a hexagonal grid with some random perturbations
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 4.0 + 0.1
        y = row / 4.0 + 0.1
        centers[i] = [x, y]
    # Scale to fit in unit square
    centers /= np.max(np.sqrt(np.sum(centers**2, axis=1))) * 5
    centers = (centers - np.min(centers, axis=0)) / (np.max(centers, axis=0) - np.min(centers, axis=0))
    centers = np.clip(centers, 0, 1)

    # Initial guess for radii
    radii = np.ones(n)

    bounds = Bounds([0.0 for _ in range(n)], [np.inf for _ in range(n)])

    # Define the optimization problem
    def optimize():
        result = minimize(
            objective,
            radii,
            bounds=bounds,
            constraints=[
                {
                    'type': 'ineq',
                    'fun': lambda x: constraints(centers, x)
                }
            ],
            method='SLSQP',
            jac=False
        )
        if result.success:
            return result.x, centers
        else:
            raise ValueError("Optimization failed")

    radii, centers = optimize()

    # Ensure final verification
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} is outside the unit square")
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
