import numpy as np
from scipy.optimize import minimize, Bounds
import random

def run_packing():
    # Define initial base layout: hexagonal packing split into 3 clusters
    def initial_centers():
        centers = []
        # Cluster 1: 8 circles packed hexagonally
        for r in range(3):
            for c in range(4):
                x = 0.1 + 0.2 * r + 0.1 * c
                y = 0.1 + 0.2 * r
                centers.append([x, y])
        # Cluster 2: 8 circles aligned in rows
        for r in range(3):
            for c in range(3):
                x = 0.6 + 0.2 * r + 0.1 * c
                y = 0.6 + 0.2 * r
                centers.append([x, y])
        # Cluster 3: 10 circles in a hexagonal pattern
        for r in range(3):
            for c in range(4):
                if r <= 2:
                    x = 0.85 + 0.15 * r + 0.05 * c
                    y = 0.85 + 0.15 * r
                    centers.append([x, y])
        return np.array(centers)

    centers = initial_centers()
    n = centers.shape[0]
    radii = np.ones(n) * 0.05

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        res = []
        for i in range(n):
            res.append(radii[i] >= 0)
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                res.append(dist >= radii[i] + radii[j] - 1e-6)
        return res

    def pack():
        bounds = [(0, 1) for _ in range(2*n)]
        result = minimize(objective, x0=radii, bounds=bounds, constraints=constraints)
        return result.x[:n], result.x[n:]

    centers, radii = pack()

    # Final validation
    def validate(radii, centers):
        for i in range(len(radii)):
            r = radii[i]
            if r < 0:
                return False
            x, y = centers[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        for i in range(len(radii)):
            for j in range(i+1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-6:
                    return False
        return True

    if not validate(radii, centers):
        raise Exception("Validation failed")

    return centers, radii, np.sum(radii)
