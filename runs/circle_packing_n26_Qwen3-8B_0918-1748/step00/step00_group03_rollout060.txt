import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define the hexagonal grid rows and column counts
    rows = 7
    cols = [1, 2, 3, 4, 5, 6, 10]  # First 6 rows have increasing count, then 10 in the 7th row

    # Hexagonal grid parameters
    row_offset = np.sqrt(3)/2  # Vertical spacing between rows
    col_spacing = 1           # Horizontal spacing between columns in a row

    # Create a list of circle centers in a hexagonal grid
    centers = []
    for r in range(rows):
        row_centers = []
        for c in range(cols[r]):
            # Row-wise offset (alternating for hexagonal grid)
            if r % 2 == 0:
                x = c * col_spacing + 0.5
                y = r * row_offset
            else:
                x = c * col_spacing + 0.5
                y = r * row_offset
            row_centers.append([x, y])
        centers.extend(row_centers)

    # We need to scale the coordinates to fit within [0, 1]x[1] and adjust for hexagonal grid
    # First determine the bounding box of the current grid
    x_coords = [center[0] for center in centers]
    y_coords = [center[1] for center in centers]
    x_min, x_max = min(x_coords), max(x_coords)
    y_min, y_max = min(y_coords), max(y_coords)
    width = x_max - x_min
    height = y_max - y_min

    # Define scaling factor to fit into unit square
    scale = min(1.0 / width, 1.0 / height)
    offset_x = (1 - width * scale) / 2
    offset_y = (1 - height * scale) / 2

    # Scale and shift the centers
    scaled_centers = np.array([[x * scale + offset_x, y * scale + offset_y] for x, y in centers])

    # Initial guess for radius (proportional to grid spacing)
    initial_radius = 0.08

    # Define the objective function to maximize the sum of radii
    def objective(params):
        r = params[0]
        return -r * len(centers)  # Minimize negative value to maximize

    # Define the constraints: non-overlapping and boundary conditions
    def constraints(params):
        r = params[0]
        # Check boundaries
        for i in range(len(centers)):
            x, y = scaled_centers[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        # Check overlaps
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = scaled_centers[i, 0] - scaled_centers[j, 0]
                dy = scaled_centers[i, 1] - scaled_centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < 2 * r - 1e-12:
                    return -1
        return 0

    # Optimization bounds
    bounds = [(0.001, 0.2)]

    # Use scipy's minimize with constraints
    result = minimize(objective, x0=[initial_radius], bounds=bounds, constraints={"type": "ineq", "fun": constraints})

    # Extract the result
    best_radius = result.x[0]
    centers_array = scaled_centers
    radii_array = np.full(len(centers), best_radius)

    # Validate packing manually due to scipy constraints
    # This is a simplified check; in practice, you'd use the validate_packing function
    if best_radius < 0:
        raise ValueError("All circles have negative or zero radius")

    for i in range(len(centers)):
        x, y = centers_array[i]
        r = radii_array[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")

    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dx = centers_array[i, 0] - centers_array[j, 0]
            dy = centers_array[i, 1] - centers_array[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            if dist < radii_array[i] + radii_array[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    sum_radii = np.sum(radii_array)
    return centers_array, radii_array, sum_radii
