    ])

        0.05, 0.05, 0.05, 0.05, 0.05
    ])

    n = len(radii)

        return -np.sum(radii)  # We maximize sum of radii by minimizing negative sum

    # Function to evaluate constraints
    def constraint_func(x):
        # x is a flattened array of centers and radii
        centers = x[:2*n].reshape((n, 2))
        radii = x[2*n:]

        for i in range(n):

            else:
                constraints.append(0)
        if sum(constraints) > 0:
            return 1.0

                dist = np.sqrt(dx**2 + dy**2)

                    return 1.0
        return 0.0

    x0 = np.concatenate((centers.flatten(), radii))

    res = minimize(objective, x0, method='SLSQP', bounds=Bounds([0.0, 0.0, 0.0], [1.0, 1.0, 1.0]), constraints=[{'type': 'ineq', 'fun': constraint_func}])

    # Extract the final centers and radii
