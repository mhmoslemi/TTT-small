import scipy.optimize as optimize
from scipy.optimize import Bounds

        # Cluster 6: Along top edge
        [(0.4, 0.8), 3],  # 3 circles along top edge
    ]

    centers = []

    # We assume circles are placed with spacing between them such that minimum distance is 4*radius

    def objective(radii_params):
        return -np.sum(radii_params)

    def constraint_func(centers, radii):
