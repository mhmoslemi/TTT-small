import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii, centers):
        return np.sum(radii)

    def non_overlapping(radii, centers):
        n = centers.shape[0]
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def bounds(radii):
        return [(0, 1) for _ in radii]

    def initial_centers():
        # Initial guess using a hexagonal packing with 5 rows
        rows = 5
        cols = 5
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j * (1.0 / (cols - 1)) if cols > 1 else 0.5
                y = i * (1.0 / (rows - 1)) if rows > 1 else 0.5
                centers.append([x, y])
        return np.array(centers)

    centers = initial_centers()
    n = centers.shape[0]
    dim = 2
    bounds = Bounds([0.0] * n, [1.0] * n)

    # Objective is to maximize sum of radii, so minimize negative sum
    res = minimize(
        fun=lambda x: -np.sum(x),
        x0=np.ones(n),
        bounds=bounds,
        constraints=[
            {
                'type': 'ineq',
                'fun': lambda x: non_overlapping(x, centers)
            }
        ],
        method='SLSQP'
    )

    radii = res.x
    total_sum = np.sum(radii)

    return (centers, radii, total_sum)
