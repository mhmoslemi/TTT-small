import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial hexagonal grid with approximate spacing
    num_circles = 26
    spacing = 0.365  # Fitted from previous studies for optimal packing density
    rows = int(np.sqrt(num_circles))
    cols = int(np.ceil(num_circles / rows))
    
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Generate hexagonal grid in a square
    for i in range(num_circles):
        row = i // cols
        col = i % cols
        x = col * spacing + spacing / 2
        y = row * spacing * np.sqrt(3) / 2 + spacing / 2
        centers[i] = [x, y]
        radii[i] = spacing / 2
    
    # Clip centers to fit within [0,1]^2
    centers = np.clip(centers, 0, 1)
    
    # Define objective function to maximize sum of radii
    def objective(x):
        # x is a flat array of [cx0, cy0, r0, cx1, cy1, r1, ...]
        cx = x[::3]
        cy = x[1::3]
        r = x[2::3]
        total = np.sum(r)
        return -total  # Minimize negative to maximize
    
    # Define constraints
    def constraint_overlap(x):
        cx = x[::3]
        cy = x[1::3]
        r = x[2::3]
        dist = pdist(np.column_stack((cx, cy)))
        min_dist = np.min(dist)
        required_min = np.min(r[np.newaxis, :] + r[:, np.newaxis] - 1e-8)
        return min_dist - required_min
    
    def constraint_boundary(x):
        cx = x[::3]
        cy = x[1::3]
        r = x[2::3]
        for i in range(len(r)):
            if cx[i] - r[i] < -1e-8 or cx[i] + r[i] > 1 + 1e-8 or cy[i] - r[i] < -1e-8 or cy[i] + r[i] > 1 + 1e-8:
                return -1
        return 0
    
    # Flatten centers and radii for optimization
    x0 = np.concatenate([centers.ravel(), radii])
    
    # Set bounds for all variables
    bounds = []
    for i in range(num_circles):
        bounds.append((0, 1))  # x coordinate
        bounds.append((0, 1))  # y coordinate
        bounds.append((0, 1))  # radius
    
    # Optimization
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=Bounds(bounds),
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        options={'ftol': 1e-8, 'maxiter': 200}
    )
    
    # Extract results
    x_result = result.x
    cx = x_result[::3]
    cy = x_result[1::3]
    r = x_result[2::3]
