import numpy as np
import scipy.optimize as opt

def generate_hexagonal_grid(n_rows, n_cols, spacing):
    centers = []
    for row in range(n_rows):
        y = row * spacing * np.sqrt(3) / 2
        for col in range(n_cols):
            x = col * spacing
            centers.append((x, y))
    return np.array(centers)

def create_optimization_problem(centers):
    n = len(centers)
    bounds = [(0, 1)] * (2 * n)
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - 0})
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: 1 - r[i]})
    for i in range(n):
        for j in range(i + 1, n):
            def constraint(r, i=i, j=j):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                return dist - (r[i] + r[j])
            constraints.append({'type': 'ineq', 'fun': constraint})
    return bounds, constraints

def run_packing():
    # Initial hexagonal grid with 5 rows and 5 columns, with spacing adjusted for unit square
    spacing = 0.35
    n_rows = 5
    n_cols = 5
    centers = generate_hexagonal_grid(n_rows, n_cols, spacing)
    
    # Add more circles at the top and bottom to fill gaps
    for i in range(1, n_rows - 1):
        x = centers[i, 0]
        y = centers[i, 1] + spacing * np.sqrt(3) / 2
        centers = np.vstack((centers, [x, y]))
    
    # Add more circles at the bottom
    for i in range(n_cols):
        x = centers[i, 0]
        y = centers[i, 1] - spacing * np.sqrt(3) / 2
        centers = np.vstack((centers, [x, y]))
    
    n = len(centers)
    bounds, constraints = create_optimization_problem(centers)
    
    # Initial guess for radii
    radii = np.full(n, 0.1)
    
    # Optimization problem: maximize sum of radii with constraints
    def objective(radii):
        return -np.sum(radii)
    
    result = opt.minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)
    
    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Final check to ensure no circles are outside the square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")
    
    return centers, radii, sum_radii
