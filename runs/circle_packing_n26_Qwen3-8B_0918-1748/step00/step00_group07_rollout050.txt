import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import LinearConstraint, NonlinearConstraint

def run_packing():
    # Define initial guess for circle centers and radii
    # Using a hybrid hexagonal packing with some circles near corners and edges

    # Define initial guess for centers (6 circles on bottom, 7 on top, 2 near corners)
    # Bottom row: 6 circles with spacing, offset by small amount
    # Top row: 7 circles with spacing, offset by small amount
    # Corners: 2 small circles

    # Initial positions
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Bottom row (12 circles): 6 main circles, 2 small circles on corners
    spacing = 0.2
    h_offset = 0.05
    for i in range(6):
        col = i * spacing + h_offset
        if i % 2 == 0:
            centers[i] = [col, 0.1]
            radii[i] = 0.08
        else:
            centers[i + 1] = [col + spacing, 0.1]
            radii[i + 1] = 0.07

    # Top row (remaining circles): 7 on top, 7 in between
    for i in range(7):
        col = i * spacing
        if i % 2 == 0:
            centers[i + 7] = [col, 0.9]
            radii[i + 7] = 0.08
        else:
            centers[i + 8] = [col + spacing, 0.9]
            radii[i + 8] = 0.07

    # Add 2 corner circles
    centers[15] = [0.05, 0.05]
    radii[15] = 0.05
    centers[16] = [0.95, 0.95]
    radii[16] = 0.05

    # Optimize with constraints
    def objective(x):
        return -np.sum(x[17:])

    def constraint1(x):
        # Ensure all circles are within the unit square
        for i in range(n):
            if np.any(x[i*2+17:i*2+19] < -1e-8) or np.any(x[i*2+17:i*2+19] > 1 + 1e-8):
                return -1
        return 0

    def constraint2(x):
        # Non-overlapping constraint
        for i in range(n):
            for j in range(i + 1, n):
                dx = x[i*2+17] - x[j*2+17]
                dy = x[i*2+18] - x[j*2+18]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < x[i*2+19] + x[j*2+19] - 1e-8:
                    return -1
        return 0

    # Flatten centers and radii for optimization
    flat_centers = np.array([centers[i][0] for i in range(n)])
    flat_centers = np.append(flat_centers, [centers[i][1] for i in range(n)])
    flat_radii = np.array([radii[i] for i in range(n)])
    initial_guess = np.append(flat_centers, flat_radii)

    # Bounds for centers and radii
    bounds = [(0, 1) for _ in range(2*n)] + [(0, 1) for _ in range(n)]

    # Constraints
    cons = [
        NonlinearConstraint(constraint1, -np.inf, 0),
        NonlinearConstraint(constraint2, -np.inf, 0)
    ]

    # Define optimization problem
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    )

    # Extract optimized centers and radii
    flat_centers = result.x[:2*n]
    flat_radii = result.x[2*n:]
    centers = np.zeros((n, 2))
    for i in range(n):
        centers[i][0] = flat_centers[2*i]
        centers[i][1] = flat_centers[2*i + 1]
    radii = flat_radii

    # Validate packing
    assert validate_packing(centers, radii)[0], "Packing failed validation"

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
