import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from typing import Tuple, List, Optional

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    def objective(radii):
        return -np.sum(radii)

    def constraints(centers, radii, n):
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False, "Circle out of bounds"
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, "Overlap between circles"
        return True, "Valid"

    def optimize_packing():
        n = 26
        x_centers = np.linspace(0.1, 0.9, 5)
        y_centers = np.linspace(0.1, 0.9, 5)
        centers = np.array([[x, y] for y in y_centers for x in x_centers])
        initial_radii = np.ones(n) * 0.1

        bounds = [(0, 0.5) for _ in range(n)]
        constraints = []

        def calc_distances(centers, radii):
            dists = squareform(pdist(centers))
            return dists

        def create_constraints(centers, radii):
            n = centers.shape[0]
            cons = []
            for i in range(n):
                x, y = centers[i]
                cons.append((lambda r, i=i: r[i] - (x - 0.01), -np.inf, 0.2))
                cons.append((lambda r, i=i: r[i] - (1 - x - 0.01), -np.inf, 0.2))
                cons.append((lambda r, i=i: r[i] - (y - 0.01), -np.inf, 0.2))
                cons.append((lambda r, i=i: r[i] - (1 - y - 0.01), -np.inf, 0.2))
            for i in range(n):
                for j in range(i+1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    cons.append((lambda r, i=i, j=j: (r[i] + r[j]) - np.sqrt(dx**2 + dy**2), -np.inf, 0.0))  # non-overlapping
            return cons

        cons = create_constraints(centers, initial_radii)
        constraints = [{'type': 'ineq', 'fun': c} for c in cons]

        result = minimize(
            fun=objective,
            x0=initial_radii,
            bounds=bounds,
            constraints=constraints,
            method='SLSQP'
        )

        if not result.success:
            raise ValueError("Optimization failed")

        return centers, result.x, np.sum(result.x)

    # Create initial configuration with a modified hexagonal packing
    def create_hexagonal_packing(n=26, max_radius=0.25):
        centers = []
        for row in range(5):
            if row % 2 == 0:
                for col in range(5):
                    x = col / 5
                    y = row / 5
                    centers.append([x, y])
            else:
                for col in range(4):
                    x = col / 5 + 0.1
                    y = row / 5
                    centers.append([x, y])
        centers = np.array(centers)
        return centers

    centers = create_hexagonal_packing()
    initial_radii = np.ones(centers.shape[0]) * 0.15

    # Further refine the configuration
    try:
        centers, radii, total_radii = optimize_packing()
    except Exception as e:
        print(f"Optimization failed: {e}")
        # Fallback to a simple grid packing
        centers = np.array([[i/5, j/5] for i in range(5) for j in range(5)])
        radii = np.ones(25) * 0.1
        total_radii = np.sum(radii)
        return centers, radii, total_radii

    # Ensure no circle is outside the unit square or overlaps
    validity, msg = constraints(centers, radii, centers.shape[0])
    if not validity:
        raise ValueError(msg)

    return centers, radii, total_radii
