import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt

def run_packing():
    def create_initial_grid(n_circles):
        grid = []
        for i in range(n_circles):
            r = 1 / np.sqrt(n_circles)
            x = i / np.sqrt(n_circles)
            y = 0.5
            grid.append([x, y])
        return np.array(grid)

    def energy_function(params, n_circles):
        centers = params[:2 * n_circles].reshape(n_circles, 2)
        radii = params[2 * n_circles:]
        total_energy = 0.0
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return float('inf')
            for j in range(i + 1, n_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-6:
                    return float('inf')
        return -np.sum(radii)

    def optimize_packing(n_circles):
        initial_guess = create_initial_grid(n_circles)
        initial_radii = np.ones(n_circles) * 1 / np.sqrt(n_circles)
        params = np.concatenate((initial_guess.flatten(), initial_radii))
        
        bounds = []
        for _ in range(n_circles):
            bounds.extend([(0, 1) for _ in range(2)])  # x, y for each circle
            bounds.extend([(0, 1) for _ in range(1)])  # radius for each circle
        
        result = opt.minimize(
            energy_function,
            params,
            args=(n_circles,),
            method='L-BFGS-B',
            bounds=bounds,
            options={'gtol': 1e-6, 'maxiter': 1000}
        )
        
        if result.success:
            optimized_centers = result.x[:2 * n_circles].reshape(n_circles, 2)
            optimized_radii = result.x[2 * n_circles:]
            return optimized_centers, optimized_radii, np.sum(optimized_radii)
        else:
            raise ValueError("Optimization failed")

    try:
        centers, radii, sum_radii = optimize_packing(26)
        return centers, radii, sum_radii
    except:
        raise
