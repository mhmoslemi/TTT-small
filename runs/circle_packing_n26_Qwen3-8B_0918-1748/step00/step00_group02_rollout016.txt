import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Initial layout: 4 levels with roughly 6, 6, 6, 8 circles
    # First level: 6 circles around center
    # Second level: 6 circles
    # Third level: 6 circles
    # Fourth level: 8 circles

    # Define initial centers for 26 circles
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # First level: 6 circles around center
    for i in range(6):
        angle = 2 * np.pi * i / 6
        x = 0.5 + np.cos(angle) * 0.25
        y = 0.5 + np.sin(angle) * 0.25
        centers[i] = [x, y]

    # Second level: 6 circles around first level
    for i in range(6, 12):
        angle = 2 * np.pi * i / 6
        x = 0.5 + np.cos(angle) * 0.4
        y = 0.5 + np.sin(angle) * 0
        centers[i] = [x, y]

    # Third level: 6 circles around second level
    for i in range(12, 18):
        angle = 2 * np.pi * i / 6
        x = 0.5 + np.cos(angle) * 0.5
        y = 0.5 + np.sin(angle) * 0.1
        centers[i] = [x, y]

    # Fourth level: 8 circles in a square layout
    for i in range(18, 26):
        if i % 2 == 0:
            x = 0.25 + (i - 18) * 0.15
            y = 0.25
        else:
            x = 0.25
            y = 0.25 + (i - 18) * 0.15
        centers[i] = [x, y]

    # Use optimization to improve layout
    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum

    # Constraints: non-overlap and boundary
    def constraint1(radii):
        return np.sum(radii)  # Relaxing this for optimization

    # First, apply basic constraints
    constraints = []

    # Boundary constraint: radii must be such that circles are inside square
    for i in range(26):
        def boundary_r_i(radii):
            return (0.5 - 0.5 * np.sqrt(2)) * (radii[i] ** 2 - 0.1)
        constraints.append({'type': 'ineq', 'fun': boundary_r_i})

    # Non-overlap constraint
    for i in range(26):
        for j in range(i+1, 26):
            def non_overlap(radii):
                r_i = radii[i]
                r_j = radii[j]
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                return dist - r_i - r_j - 1e-8
            constraints.append({'type': 'ineq', 'fun': non_overlap})

    # Optimization bounds: radii >= 0
    bounds = [(0, 1) for _ in range(26)]

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)

    # Extract result
    radii = result.x
    sum_radii = np.sum(radii)

    # Ensure centers are within bounds
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            raise ValueError("Circle at ({x},{y}) with radius {r} is outside the unit square")

    return centers, radii, sum_radii
