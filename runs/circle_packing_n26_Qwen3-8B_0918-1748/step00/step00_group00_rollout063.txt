import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds, LinearConstraint

def run_packing():
    # Initial hexagonal grid
    n = 26
    layer = int(np.floor((n - 1) / 6))
    remainder = n - 6 * layer - 1
    cols = 6 * layer + 1
    rows = layer + 1
    total = cols + rows - 1

    # Generate hexagonal grid
    centers = []
    radii = []
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1)
            y = i / (rows - 1)
            centers.append([x, y])
    for i in range(rows - 1):
        for j in range(1, cols):
            x = j / (cols - 1)
            y = (i + 0.5) / (rows - 1)
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(len(centers), 0.05)

    # Shift to fit in unit square
    min_x, max_x = np.min(centers[:, 0]), np.max(centers[:, 0])
    min_y, max_y = np.min(centers[:, 1]), np.max(centers[:, 1])
    scale_x = 1.0 / (max_x - min_x)
    scale_y = 1.0 / (max_y - min_y)
    centers = np.column_stack((centers[:, 0] * scale_x, centers[:, 1] * scale_y))

    # Initial sizing
    radii = np.full(len(centers), 0.05)
    for i in range(len(centers)):
        x, y = centers[i]
        r = 0.05
        if x < 0.1 or x > 0.9 or y < 0.1 or y > 0.9:
            r = min(0.1, r)
        radii[i] = r

    # Cone optimization
    def objective(r):
        return -np.sum(r)

    def constraint_overlap(x, r):
        centers = x[:2*n].reshape(n, 2)
        radii = r
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-12:
                    return -1
        return 0

    def constraint_boundary(x, r):
        centers = x[:2*n].reshape(n, 2)
        radii = r
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return -1
        return 0

    bounds = Bounds([0.0]*2*n, [1.0]*2*n)
    constraints = [LinearConstraint([], [], [], fun=constraint_overlap),
                   LinearConstraint([], [], [], fun=constraint_boundary)]

    x0 = centers.reshape(2*n)
    r0 = radii

    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-10, 'maxiter': 200})
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.fun * -1

    # Post-processing: try to increase radii of boundary circles
    for i in range(n):
        x, y = centers[i]
        if x < 0.1 or x > 0.9 or y < 0.1 or y > 0.9:
            r = radii[i]
            direction = np.array([0, 0])
            if x < 0.1:
                direction += np.array([-1, 0])
            elif x > 0.9:
                direction += np.array([1, 0])
            if y < 0.1:
                direction += np.array([0, -1])
            elif y > 0.9:
                direction += np.array([0, 1])
            direction /= np.linalg.norm(direction)
            nearest = None
            min_dist = np.inf
            for j in range(n):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    if dist < min_dist:
                        min_dist = dist
                        nearest = j
            max_possible_r = min(
                1.0 - max(centers[i, 0], 1.0 - centers[i, 0]),
                1.0 - max(centers[i, 1], 1.0 - centers[i, 1]),
                min_dist - radii[nearest]
            )
            if max_possible_r > radii[i]:
                radii[i] = max(radii[i], max_possible_r)
                # Slightly move circle away from the boundary to maintain distance
                move_radius = 1e-3
                centers[i] += direction * move_radius * 0.5

    return centers, radii, np.sum(radii)
