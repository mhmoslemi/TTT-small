import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial guess for circle positions using a hexagonal-like grid
    # We'll use a hybrid arrangement: a center cluster and edge circles
    # This is a heuristic initial placement and will be optimized

    # Initial placement: 15 in center, 10 on the edges, 1 at each corner
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Central cluster (hexagonal packing)
    for i in range(15):
        # Hexagonal grid: using a 4x4 grid with some offset
        if i < 4:
            x = 0.2 + i * 0.25
            y = 0.25
        elif i < 8:
            x = 0.25 + (i - 4) * 0.2
            y = 0.5
        elif i < 12:
            x = 0.2 + (i - 8) * 0.25
            y = 0.75
        else:
            x = 0.25 + (i - 12) * 0.2
            y = 0.9
        centers[i] = [x, y]

    # Edge circles (circles placed along the edges and corners)
    edge_indices = list(range(15, 25))
    for i, idx in enumerate(edge_indices):
        if i < 4:
            # Top edge
            x = 0.2 + i * 0.25
            y = 0.95
        elif i < 8:
            # Right edge
            x = 0.95
            y = 0.25 + (i - 4) * 0.2
        elif i < 12:
            # Bottom edge
            x = 0.9 - (i - 8) * 0.25
            y = 0.05
        else:
            # Left edge
            x = 0.05
            y = 0.25 + (i - 12) * 0.2
        centers[idx] = [x, y]

    # Corner circle
    centers[25] = [0.05, 0.05]

    # Optimization parameters
    bounds = [(0, 1) for _ in range(52)]  # 26 centers * 2 coordinates
    constraints = []

    def obj_func(params):
        # Flatten the parameters
        centers_flat = params[:52].reshape(26, 2)
        radii_flat = params[52:]

        # Objective: maximize the sum of radii
        return -np.sum(radii_flat)

    def constraint_func(params):
        centers_flat = params[:52].reshape(26, 2)
        radii_flat = params[52:]

        # Check if any circle is outside the square
        for i in range(26):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return -1
        # Check for overlaps
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -1
        return 0

    # Optimization with bounds and constraints
    result = opt.minimize(
        obj_func,
        x0=np.concatenate((centers.flatten(), np.ones(26))),
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_func}]
    )

    # Extract optimized centers and radii
    centers_opt = result.x[:52].reshape(26, 2)
    radii_opt = result.x[52:]

    # Validate the packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
