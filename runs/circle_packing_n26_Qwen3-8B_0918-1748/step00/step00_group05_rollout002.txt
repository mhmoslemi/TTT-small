import numpy as np
import scipy.optimize as optimize
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial grid positions for 26 circles (approximate hexagonal packing)
    def initial_positions(n):
        rows = int(np.ceil(n**0.5))
        cols = int(np.ceil(n / rows))
        positions = []
        for i in range(rows):
            for j in range(cols):
                x = 0.1 + j * 0.25
                y = 0.1 + i * 0.25
                if i % 2 == 1:
                    x += 0.125
                positions.append([x, y])
        # Add more positions if needed
        while len(positions) < n:
            positions.append([0.5, 0.5])
        return np.array(positions)
    
    initial_centers = initial_positions(26)
    n = 26

    # Initial radii: max possible without overlapping
    def initial_radii(centers):
        radii = np.zeros(n)
        for i in range(n):
            min_dist_to_wall = min(centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1])
            min_dist_to_other = np.inf
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < min_dist_to_other:
                    min_dist_to_other = dist
            radii[i] = min(min_dist_to_wall, min_dist_to_other / 2)
        return radii

    initial_radii = initial_radii(initial_centers)

    # Optimize radii and positions
    def objective(x, centers, radii):
        # x is a flattened array of positions and radii
        # Extract positions and radii
        flat_pos = x[:2 * n]
        flat_rad = x[2 * n:]
        centers_opt = flat_pos.reshape(n, 2)
        radii_opt = flat_rad
        # Compute total radius
        total = np.sum(radii_opt)
        return -total  # Minimize negative total (maximize total)

    def constraint_eq(x, centers, radii):
        flat_pos = x[:2 * n]
        flat_rad = x[2 * n:]
        centers_opt = flat_pos.reshape(n, 2)
        radii_opt = flat_rad
        # Check non-overlap and boundary constraints
        for i in range(n):
            xi, yi = centers_opt[i]
            ri = radii_opt[i]
            # Ensure circle is inside the square
            if xi - ri < -1e-12 or xi + ri > 1 + 1e-12 or yi - ri < -1e-12 or yi + ri > 1 + 1e-12:
                return -1
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers_opt[i] - centers_opt[j])
                if dist < radii_opt[i] + radii_opt[j] - 1e-12:
                    return -1
        return 0

    # Initial guess: positions + radii
    flat_pos = initial_centers.reshape(-1)
    flat_rad = initial_radii
    x0 = np.concatenate([flat_pos, flat_rad])

    bounds = []
    for _ in range(n * 2):
        bounds.append((0, 1))
    for _ in range(n):
        bounds.append((0, 1))

    # Optimization
    result = optimize.minimize(
        objective,
        x0,
        args=(initial_centers, initial_radii),
        bounds=bounds,
        constraints=[{'type': 'equality', 'fun': constraint_eq, 'args': (initial_centers, initial_radii)}],
        method='SLSQP'
    )

    # Extract optimized positions and radii
    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]

    # Final validation
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
