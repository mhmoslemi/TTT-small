import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint
from typing import Tuple, Any

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial guess: hybrid hexagonal + corner placement
    # Generate 26 points in a grid-like and hexagonal arrangement
    # This will be used as initial guess for optimization
    centers = []
    radii = []
    num_circles = 26

    # Initial placement: first layer (row 0) with 4 circles (corners)
    # Second layer (row 1) with 5 circles
    # Third layer (row 2) with 5 circles
    # Fourth layer (row 3) with 3 circles
    # Fifth layer (row 4) with 3 circles
    # Sixth layer (row 5) with 4 circles
    # Total: 4+5+5+3+3+4=24, need 26, so add 2 more circles
    # To get a creative shape, we add 2 more near the bottom right and middle bottom

    # First, populate the hexagonal-like columns
    # Define layer positions (x, y)
    positions = []
    for i in range(5):
        if i % 2 == 0:
            for j in range(5):
                positions.append((j / 4.5, i / 5))
            positions.append((5 / 4.5, i / 5))
        else:
            for j in range(5):
                positions.append((j / 4.5 - 0.5, i / 5))
            positions.append((5 / 4.5 - 1, i / 5))

    # Add 2 more circles at bottom right and middle bottom
    positions.append((0.9, 0.8))
    positions.append((0.4, 0.9))

    # Normalize to fit into [0, 1]×[0, 1]
    # Using simplex transformation
    def simplex_transform(point, size=1, max_distance=1.0):
        # Normalize the coordinates to [0, 1]×[0, 1] by scaling
        # For example, in a hexagonal lattice, we can transform the grid
        # into a square.
        # For simplicity, assume all positions are in a grid.
        # We will use the sum of x and y to determine the scaling factor.
        x, y = point[0], point[1]
        max_x, max_y = 1.0, 1.0
        r = np.sqrt(x**2 + y**2)
        if r == 0:
            return (x, y)
        scale = max_distance / r
        return (x * scale, y * scale)

    normalized_centers = [simplex_transform(p) for p in positions]
    centers = np.array(normalized_centers)
    # Initial guess for radii: all equal, based on average spacing
    spacing = np.min(np.sqrt(np.sum((centers[i] - centers[j])**2) for i in range(len(centers)) for j in range(i+1, len(centers))))
    initial_radius = np.min([spacing / 2, 0.1])  # Safety factor
    radii = np.full(len(centers), initial_radius)

    # Optimization
    def objective(radii):
        return -np.sum(radii)

    def constraint1(x):
        # x is (x1, y1, r1, x2, y2, r2, ..., xn, yn, rn)
        n = len(centers)
        positions = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        return np.sum(radii)

    def constraint2(x):
        n = len(centers)
        positions = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        return np.sum(radii)

    bounds = Bounds([0.0] * (2 * n + n), [1.0] * (2 * n + n))
    # Nonlinear constraint: distance between centers >= sum of radii
    def nonlin_eq(x):
        n = len(centers)
        positions = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        dist = np.sqrt(np.sum((positions[i] - positions[j])**2) for i in range(n) for j in range(i+1, n))
        return dist - (radii[i] + radii[j] for i in range(n) for j in range(i+1, n))

    # Optimization problem
    x0 = np.concatenate([centers.flatten(), radii])
    cons1 = LinearConstraint(np.ones(2 * n + n), 0, np.inf)
    cons2 = LinearConstraint(np.repeat(np.arange(2 * n + n), 2), 0, np.inf)
    nonlin_cons = NonlinearConstraint(nonlin_eq, 0, 0)

    result = minimize(objective, x0, bounds=bounds, constraints=[cons1, cons2, nonlin_cons])

    # Extract the result
    x_result = result.x
    centers = x_result[:2*n].reshape(n, 2)
    radii = x_result[2*n:]

    # Validate before returning
    success, message = validate_packing(centers, radii)
    if not success:
        raise RuntimeError(message)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
