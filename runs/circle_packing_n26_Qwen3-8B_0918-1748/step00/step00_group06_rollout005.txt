import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Define starting grid positions
    num_circles = 26
    grid_rows = 5
    grid_cols = 5
    spacing = 1.0 / (grid_cols - 1)
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # First 25 circles in a 5x5 grid with hexagonal offset
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j * spacing
            y = i * spacing
            if (i + j) % 2 == 0:
                y += spacing / 2
            centers[i * grid_cols + j] = [x, y]

    # Add the 26th circle in the center
    centers[-1] = [0.5, 0.5]

    # Initial radii based on spacing
    initial_radius = spacing / 2
    radii[:] = initial_radius

    # Optimization setup
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_overlap(centers, radii):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    def constraint_boundary(centers, radii):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        return 0

    # Optimization variables: radii
    bounds = [(0, 1) for _ in range(num_circles)]
    constraints = []
    for _ in range(num_circles * (num_circles - 1) // 2):
        constraints.append({'type': 'ineq', 'fun': lambda r, i, j: constraint_overlap(centers, r) if i == j else 0})
    for _ in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': lambda r, i: constraint_boundary(centers, r) if i == 0 else 0})

    # Run optimization
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 1000}
    )

    # Extract results
    centers = result.x.reshape(-1, 2)
    radii = result.x
    sum_radii = np.sum(radii)

    # Final check for validity
    if np.isnan(centers).any() or np.isnan(radii).any():
        raise ValueError("NaN values present in packing")

    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    return centers, radii, sum_radii
