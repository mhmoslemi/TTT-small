import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Base configuration: structured packing with 26 circles, starting with a grid
    # We'll start with a 4x6 grid (24 circles) and add 2 more circles in the remaining space
    base_centers = np.zeros((26, 2))
    base_radii = np.zeros(26)

    # Step 1: Initialize in a grid arrangement
    grid_size = 0.2
    for i in range(24):
        row = i // 6
        col = i % 6
        x = col * grid_size
        y = row * grid_size
        base_centers[i] = [x, y]
        base_radii[i] = grid_size / 2

    # Step 2: Add two more circles near the boundaries
    base_centers[24] = [0.0, 0.8]
    base_radii[24] = 0.1
    base_centers[25] = [0.8, 0.0]
    base_radii[25] = 0.1

    # Define objective function: maximize total radius sum
    def objective(radii, centers):
        return -np.sum(radii)

    # Define constraints: circle boundaries and non-overlapping
    def constraint_boundary(radii, centers):
        return np.array([np.min(centers[i, 0] - radii[i]) >= -1e-12 and np.max(centers[i, 0] + radii[i]) <= 1 + 1e-12 and
                        np.min(centers[i, 1] - radii[i]) >= -1e-12 and np.max(centers[i, 1] + radii[i]) <= 1 + 1e-12 for i in range(26)])

    def constraint_overlap(radii, centers):
        dists = cdist(centers, centers)
        return np.array([dists[i, j] - (radii[i] + radii[j]) >= -1e-12 for i in range(26) for j in range(i+1, 26)]).flatten()

    # Optimization setup
    x0 = np.concatenate((base_centers.flatten(), base_radii))

    # Separate variables into centers and radii
    def objective_func(x):
        centers = x[:52].reshape(26, 2)
        radii = x[52:]
        return -np.sum(radii)

    def constraint_func(x):
        centers = x[:52].reshape(26, 2)
        radii = x[52:]
        boundary_constraints = []
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.array([False, False])
            boundary_constraints.append(1)
        overlap_constraints = []
        for i in range(26):
            for j in range(i+1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.array([False, False])
                overlap_constraints.append(1)
        return np.array([True, True])

    # We will use a simpler optimization approach here to avoid divergence
    # Define bounds for centers and radii
    bounds = []
    for i in range(26):
        bounds.extend([(0, 1) for _ in range(2)])  # for x and y of each center
        bounds.extend([(0, 1) for _ in range(1)])  # for radius of each circle

    res = minimize(objective_func, x0, bounds=bounds, constraints={"type": "ineq", "fun": constraint_func})

    centers = res.x[:52].reshape(26, 2)
    radii = res.x[52:]
    total = np.sum(radii)
    return centers, radii, total
