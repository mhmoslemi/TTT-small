import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Define the number of circles
    n = 26

    # Initial guess for centers
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # First ring: 6 circles
    for i in range(6):
        angle = 2 * np.pi * (i / 6)
        center = np.array([0.5 + 0.5 * np.cos(angle), 0.5 + 0.5 * np.sin(angle)])
        # Check if the center is within the square
        if (center[0] < 1e-12 or center[0] > 1 - 1e-12 or
            center[1] < 1e-12 or center[1] > 1 - 1e-12):
            continue
        centers[i] = center
        radii[i] = 0.2

    # Second ring: 12 circles
    for i in range(6, 18):
        angle = 2 * np.pi * (i - 6) / 12
        center = np.array([0.5 + 0.5 * np.cos(angle), 0.5 + 0.5 * np.sin(angle)])
        # Check if the center is within the square
        if (center[0] < 1e-12 or center[0] > 1 - 1e-12 or
            center[1] < 1e-12 or center[1] > 1 - 1e-12):
            continue
        centers[i] = center
        radii[i] = 0.15

    # Third ring: 8 circles to make total of 26
    for i in range(18, 26):
        angle = 2 * np.pi * (i - 18) / 8
        center = np.array([0.5 + 0.3 * np.cos(angle), 0.5 + 0.3 * np.sin(angle)])
        # Check if the center is within the square
        if (center[0] < 1e-12 or center[0] > 1 - 1e-12 or
            center[1] < 1e-12 or center[1] > 1 - 1e-12):
            continue
        centers[i] = center
        radii[i] = 0.1

    # Define optimization problem
    def objective(x):
        return -np.sum(radii)

    def constraint(x, i, j):
        return np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - radii[i] - radii[j]

    # Optimization parameters
    bounds = Bounds([], [])
    cons = []
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint(x, i, j)})

    x0 = np.concatenate((centers.flatten(), radii))
    result = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons)

    # Extract centers and radii
    x_result = result.x
    centers_opt = x_result[:n * 2].reshape((n, 2))
    radii_opt = x_result[n * 2:]

    # Validate the packing
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(message)

    # Return centers, radii, and sum of radii
    return centers_opt, radii_opt, np.sum(radii_opt)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
