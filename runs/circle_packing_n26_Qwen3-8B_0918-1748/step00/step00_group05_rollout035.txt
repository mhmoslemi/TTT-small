import numpy as np
import random
import math

def run_packing():
    # Initial configuration: 5x5 grid with some variation
    centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])
    radii = np.full(26, 0.05)
    
    # Optimization parameters
    temp = 1.0
    cooling_rate = 0.999
    max_iterations = 10000
    perturbation_scale = 0.02
    radius_perturbation_scale = 0.01
    
    # Simulated Annealing
    for _ in range(max_iterations):
        # Randomly select a circle to perturb
        idx = random.randint(0, 25)
        x, y = centers[idx]
        r = radii[idx]
        
        # Perturb the center
        dx = random.uniform(-perturbation_scale, perturbation_scale)
        dy = random.uniform(-perturbation_scale, perturbation_scale)
        new_x = x + dx
        new_y = y + dy
        
        # Check if new center is within [0, 1]^2
        if not (0 <= new_x <= 1 and 0 <= new_y <= 1):
            continue
        
        centers[idx] = [new_x, new_y]
        
        # Perturb the radius
        dr = random.uniform(-radius_perturbation_scale, radius_perturbation_scale)
        new_r = r + dr
        
        # Check if new radius is non-negative
        if new_r < 0:
            continue
        
        # Check if new radius doesn't cause circle to go out of bounds
        if new_x - new_r < -1e-12 or new_x + new_r > 1 + 1e-12 or new_y - new_r < -1e-12 or new_y + new_r > 1 + 1e-12:
            continue
        
        radii[idx] = new_r
        
        # Calculate energy before and after
        old_energy = 0.0
        new_energy = 0.0
        
        # Compute old_energy
        for i in range(26):
            for j in range(i + 1, 26):
                dist = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                old_energy += math.log(dist - radii[i] - radii[j])
        
        # Compute new_energy
        for i in range(26):
            for j in range(i + 1, 26):
                dist = math.hypot(centers[i, 0] - centers[j, 0], centers[i, 1] - centers[j, 1])
                new_energy += math.log(dist - radii[i] - radii[j])
        
        # Acquire delta_energy
        delta_energy = new_energy - old_energy
        
        # Accept or reject based on temperature
        if random.random() < math.exp(delta_energy / temp):
            continue
        
        # If not accepted, revert the changes
        centers[idx] = [x, y]
        radii[idx] = r
        
        # Cool down
        temp *= cooling_rate
    
    # Compute sum_radii
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
