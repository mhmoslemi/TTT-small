import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.spatial import ConvexHull
from scipy.optimize import Bounds

def run_packing():
    # Initial grid parameters
    num_inner_circles = 24
    num_outer_circles = 2
    total_circles = num_inner_circles + num_outer_circles
    
    # Hexagonal packing configuration for inner circles
    rows = 4
    cols = 6
    radius_ratio = 0.8
    inner_radius = 0.15
    inner_centers = np.zeros((num_inner_circles, 2))
    index = 0
    for row in range(rows):
        for col in range(cols):
            x = 0.5 + 0.5 * (col - (cols - 1) / 2) * (2 * inner_radius) / np.sqrt(3)
            y = 0.5 + row * (2 * inner_radius) * 2 / np.sqrt(3)
            if row % 2 == 0:
                x += 0.5 * (2 * inner_radius) / np.sqrt(3)
            inner_centers[index, 0] = x
            inner_centers[index, 1] = y
            index += 1
    
    # Define the outer grid points
    outer_points = np.array([
        [0.1, 0.1],
        [0.9, 0.1],
        [0.1, 0.9],
        [0.9, 0.9]
    ])
    
    # Create a convex hull for the inner circles to determine outer regions
    inner_hull = ConvexHull(inner_centers)
    outer_centers = np.empty((num_outer_circles, 2))
    index = 0
    for point in outer_points:
        outer_centers[index, 0] = point[0]
        outer_centers[index, 1] = point[1]
        index += 1
    
    # Combine centers
    centers = np.vstack((inner_centers, outer_centers))
    radii = np.zeros(total_circles)
    
    # Function to compute max radius for a point depending on neighbors and boundaries
    def compute_initial_radii(centers, radii, num_circles):
        for i in range(num_circles):
            x, y = centers[i]
            min_dist = np.inf
            for j in range(num_circles):
                if i != j:
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    if dist < min_dist:
                        min_dist = dist
            # Compute distance to nearest boundary
            dist_left = x
            dist_right = 1 - x
            dist_bottom = y
            dist_top = 1 - y
            min_boundary = np.min([dist_left, dist_right, dist_bottom, dist_top])
            radii[i] = min(min_dist / 2, min_boundary)
    
    compute_initial_radii(centers, radii, total_circles)
    
    # Optimization to maximize sum of radii
    def obj_func(radii):
        return -np.sum(radii)
    
    def constraint(x, centers, radii):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1
        return 0
    
    bounds = [(0, 1) for _ in range(total_circles)]
    res = minimize(obj_func, radii, bounds=bounds, constraints=({"type": "ineq", "fun": constraint, "args": (centers, radii)}))
    
    radii = res.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
