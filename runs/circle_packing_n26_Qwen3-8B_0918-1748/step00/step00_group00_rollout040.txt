import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Define the number of circles
    n = 26

    # Initial guess for the positions of the circles (hexagonal lattice)
    # Start with a hexagonal packing with 5 rows (since 5*5=25, slightly adjusted for 26)
    # This is a rough initial guess to be refined by the optimization
    positions = []
    radii = []

    # First 5 rows of hexagonal packing (rows 0 to 4)
    for i in range(5):
        for j in range(5):
            if i % 2 == 0:
                x = j * (1 / 5)
            else:
                x = j * (1 / 5) + (1 / 10)
            y = i * (1 / 5)
            positions.append([x, y])
    # Add an extra circle at the end for the 26th circle
    # To keep the packing symmetric, place it at the center of the square
    positions.append([0.5, 0.5])
    
    # Convert initial positions to a numpy array
    initial_positions = np.array(positions)
    
    # Initial radii: all start with the same value
    initial_radii = np.ones(n) * 0.1  # Initial guess

    # Define variables: positions (x, y) and radii
    # We will use a combined vector for optimization
    # The variables are arranged as [x0, y0, x1, y1, ..., xn-1, yn-1, r0, r1, ..., rn-1]

    # Flatten the positions and radii into a single vector for optimization
    def vector_to_params(x_vec):
        n = len(x_vec) // (2 + 1)
        positions = x_vec[:2 * n].reshape((n, 2))
        radii = x_vec[2 * n:].reshape((n,))
        return positions, radii

    def params_to_vector(positions, radii):
        return np.hstack((positions.reshape(-1), radii))

    # Objective function: maximize the sum of the radii
    def objective(x_vec):
        positions, radii = vector_to_params(x_vec)
        return -np.sum(radii)  # Negative for maximization

    # Constraint functions for the optimization
    # 1. Each circle must be inside the unit square
    def constraint_inside(x_vec):
        positions, radii = vector_to_params(x_vec)
        for i in range(n):
            x, y = positions[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return 1.0  # Violation
        return 0.0

    # 2. No overlapping between circles
    def constraint_no_overlapping(x_vec):
        positions, radii = vector_to_params(x_vec)
        for i in range(n):
            for j in range(i + 1, n):
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return 1.0  # Violation
        return 0.0

    # 3. Radii must be non-negative
    def constraint_radii_non_negative(x_vec):
        positions, radii = vector_to_params(x_vec)
        for r in radii:
            if r < -1e-12:
                return 1.0  # Violation
        return 0.0

    # Combine all constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_inside},
        {'type': 'ineq', 'fun': constraint_no_overlapping},
        {'type': 'ineq', 'fun': constraint_radii_non_negative}
    ]

    # Initial bounds on variables
    # For positions: each x, y in [0, 1]
    # For radii: each radius >= 0
    # We will use the 'Bounds' object from scipy
    bounds = Bounds(0, 1, lb=np.repeat(0, 2 * n), ub=np.repeat(1, 2 * n))
    bounds = Bounds(np.repeat(0, 2 * n + n), np.repeat(1, 2 * n + n))

    # Initial guess is the vectorized version of initial_positions and initial_radii
    x0_vec = params_to_vector(initial_positions, initial_radii)

    # Run the optimization with Sequential Least Squares Programming (SLSQP)
    result = opt.minimize(
        objective,
        x0_vec,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        tol=1e-8,
        options={
            'maxiter': 200,
            'ftol': 1e-8,
            'eps': 1e-8,
            'disp': False
        }
    )

    # Extract the final positions and radii
    positions, radii = vector_to_params(result.x)
    sum_radii = np.sum(radii)

    # Validate the packing and return result
    # Note: If the optimization fails, this will raise an error
    if not validate_packing(positions, radii)[0]:
        raise ValueError("Optimization failed, packing is invalid")

    return positions, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
