import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial configuration based on a hexagonal grid with some adjustment for square shape
    num_circles = 26
    grid_rows = 5
    grid_cols = 6
    spacing = 1.0 / (grid_rows + 1)
    grid_centers = []
    for i in range(grid_rows):
        row = []
        for j in range(grid_cols):
            x = j * (1.0 / (grid_cols - 1)) + 0.5 * (j % 2) * spacing
            y = i * spacing
            row.append((x, y))
        grid_centers.extend(row)
    # Adjust for square constraints
    centers = np.array(grid_centers)
    # Calculate initial radii as uniform
    radii = np.full(num_circles, 0.1)
    
    # Define objective function: maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)
    
    # Define constraints: non-overlapping and boundary constraints
    def constraint_func(radii, centers):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12
                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    # Optimization
    result = optimize.minimize(
        fun=lambda x: -np.sum(x),
        x0=radii,
        bounds=[(0.0, 0.5)] * num_circles,
        constraints=[{
            'type': 'ineq',
            'fun': lambda x: constraint_func(x, centers)
        }],
        method='SLSQP'
    )
    
    # Final centers and radii
    final_centers = centers
    final_radii = result.x
    sum_radii = np.sum(final_radii)
    
    return (final_centers, final_radii, sum_radii)
