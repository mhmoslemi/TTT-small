import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint
import time

def run_packing():
    # Initial configuration: hexagonal packing with some adjustment for boundary conditions
    # We will start with a hexagonal grid with 2 rows of 13 circles and 1 row of 7 circles to make 26 circles

    # Initial placement with hexagonal grid
    centers = []
    radii = []
    row_count = 2
    col_count = 13
    spacing = 1.0 / np.sqrt(3)  # Hexagonal grid spacing

    # First layer (top row)
    for i in range(col_count):
        x = i * spacing
        y = 0.5
        # Adjust to fit in the square
        if x > 1 - 1e-6:
            x = 1.0 - 1e-6
        if y > 1 - 1e-6:
            y = 1.0 - 1e-6
        centers.append([x, y])
        radii.append(0.1)

    # Second layer (bottom row)
    for i in range(col_count):
        x = i * spacing
        y = 0.5 + spacing
        # Adjust to fit in the square
        if x > 1 - 1e-6:
            x = 1.0 - 1e-6
        if y > 1 - 1e-6:
            y = 1.0 - 1e-6
        centers.append([x, y])
        radii.append(0.1)

    # Third layer (middle row)
    for i in range(7):
        x = i * spacing
        y = spacing * 2
        # Adjust to fit in the square
        if x > 1 - 1e-6:
            x = 1.0 - 1e-6
        if y > 1 - 1e-6:
            y = 1.0 - 1e-6
        centers.append([x, y])
        radii.append(0.1)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Define the optimization problem
    def objective(x):
        # x has shape (26*2, ), flatten to get centers and radii
        flat_centers = x[:26*2].reshape(26, 2)
        flat_radii = x[26*2:]
        return -np.sum(flat_radii)

    def constraints(x):
        flat_centers = x[:26*2].reshape(26, 2)
        flat_radii = x[26*2:]
        constraints = []
        # Boundary constraint: distance from boundary must be >= radius
        for i in range(26):
            x_i, y_i = flat_centers[i]
            r_i = flat_radii[i]
            # Ensure x - r >= 0 and x + r <= 1
            constraints.append(x_i - r_i >= -1e-8)
            constraints.append(x_i + r_i <= 1 + 1e-8)
            constraints.append(y_i - r_i >= -1e-8)
            constraints.append(y_i + r_i <= 1 + 1e-8)
        # Non-overlapping constraint: distance between centers >= r_i + r_j
        for i in range(26):
            for j in range(i+1, 26):
                dist = np.sqrt(np.sum((flat_centers[i] - flat_centers[j])**2))
                constraints.append(dist >= flat_radii[i] + flat_radii[j] - 1e-8)
        return constraints

    # Bounds: centers in [0,1]^2, radii >= 0
    bounds = [(0, 1) for _ in range(26*2)] + [(0, 1) for _ in range(26)]

    # Initial guess: flatten centers and radii
    initial_guess = np.concatenate([centers.flatten(), radii])

    # Set up the optimization problem
    problem = {
        'fun': objective,
        'bounds': bounds,
        'constraints': {
            'type': 'ineq',
            'fun': constraints
        }
    }

    # Run the optimization
    result = minimize(**problem)

    # Extract the solution
    optimal_centers = result.x[:26*2].reshape(26, 2)
    optimal_radii = result.x[26*2:]

    # Check if the solution is valid
    valid, message = validate_packing(optimal_centers, optimal_radii)
    if not valid:
        raise ValueError("Optimal configuration is invalid:", message)

    # Calculate total sum of radii
    total_sum = np.sum(optimal_radii)

    return (optimal_centers, optimal_radii, total_sum)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
