import numpy as np
import cvxpy as cp

def run_packing():
    # Define number of circles
    n = 26

    # Initial configuration: layers of circles with increasing spacing
    # Layer 1: 13 circles along the bottom
    # Layer 2: 12 circles stacked above
    # Layer 3: 1 circle at top
    # This forms a central cluster with dense layers

    # Initial centers (tentative placement)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Layer 1: 13 circles along the bottom
    spacing = 0.1
    for i in range(13):
        centers[i, 0] = i * spacing + 0.05
        centers[i, 1] = 0.9

    # Layer 2: 12 circles stacked above
    for i in range(13, 25):
        centers[i, 0] = (i - 13) * spacing + 0.05
        centers[i, 1] = 0.85

    # Layer 3: 1 circle at the top
    centers[25, 0] = 0.5
    centers[25, 1] = 0.95

    # Initial radii: uniform (will be updated)
    radii = np.full(n, 0.05)

    # Variables for optimization
    centers_vars = cp.Variable((n, 2))
    radii_vars = cp.Variable(n, non_negative=True)

    # Constraints: circles inside the square
    constraints = []
    for i in range(n):
        x = centers_vars[i, 0]
        y = centers_vars[i, 1]
        r = radii_vars[i]
        constraints.append(x - r >= -1e-12)
        constraints.append(x + r <= 1 + 1e-12)
        constraints.append(y - r >= -1e-12)
        constraints.append(y + r <= 1 + 1e-12)

    # Constraints: non-overlapping circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt(cp.sum_squares(centers_vars[i] - centers_vars[j]))
            constraints.append(dist >= radii_vars[i] + radii_vars[j] - 1e-12)

    # Objective: maximize the total sum of radii
    obj = cp.Maximize(cp.sum(radii_vars))

    # Solve the optimization problem
    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.SCS, eps=1e-6)

    # Return the result
    centers_result = centers_vars.value
    radii_result = radii_vars.value
    sum_radii = np.sum(radii_result)

    return (centers_result, radii_result, sum_radii)
