import numpy as np
import cvxpy as cp
import math

def run_packing():
    # Hexagonal grid parameters
    num_hex = 24
    grid_rows = int(math.ceil(math.sqrt(num_hex)))
    grid_cols = int(math.ceil(num_hex / grid_rows))
    
    # Define the positions of the hexagonal grid (origin at top-left corner)
    centers_hex = []
    for y in range(grid_rows):
        for x in range(grid_cols):
            # Hexagonal grid coordinates
            x_center = x + 0.5 * y
            y_center = y + 0.5 * x
            centers_hex.append([x_center, y_center])
    
    # Scale to fit unit square with some margin
    scale = 1.0 / (np.max(centers_hex, axis=0) - np.min(centers_hex, axis=0))
    scaled_centers = np.array(centers_hex) * scale
    # Adjust to fit into [0, 1]^2
    scaled_centers -= np.min(scaled_centers, axis=0)
    scaled_centers /= np.max(scaled_centers, axis=0) - np.min(scaled_centers, axis=0)

    # Initial radii: assuming uniform size for 24 circles
    r_hex = 0.08
    radii_hex = [r_hex] * num_hex

    # Introduce two more circles with optimization
    extra_circles = 2
    num_circles = num_hex + extra_circles
    all_centers = np.vstack([scaled_centers, np.zeros((extra_circles, 2))])
    all_radii = radii_hex + [0.0] * extra_circles

    # Optimization variables
    r = cp.Variable(num_circles)
    x = cp.Variable((num_circles, 2))

    # Constraints
    constraints = []
    for i in range(num_circles):
        # Bound the center within [0, 1]^2
        constraints.append(x[i, 0] >= 0)
        constraints.append(x[i, 0] <= 1)
        constraints.append(x[i, 1] >= 0)
        constraints.append(x[i, 1] <= 1)
        # Ensure radius is non-negative
        constraints.append(r[i] >= 0)

    # Distance constraints between circles
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            # Distance between centers squared
            dist_sq = (x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2
            # Constraint: distance >= r_i + r_j
            constraints.append(dist_sq >= (r[i] + r[j])**2 - 1e-12)

    # Objective: maximize total radius sum
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    prob.solve(solver=cp.SCS, eps=1e-5)

    if not prob.success:
        raise ValueError("Optimization failed")

    centers = x.value
    radii = r.value
    sum_radii = np.sum(radii)

    # Final validation: ensure all circles within bounds, non-overlapping
    for i in range(num_circles):
        if (centers[i, 0] - radii[i] < -1e-12 or
            centers[i, 0] + radii[i] > 1 + 1e-12 or
            centers[i, 1] - radii[i] < -1e-12 or
            centers[i, 1] + radii[i] > 1 + 1e-12):
            raise ValueError("Circle out of bounds")

        for j in range(i + 1, num_circles):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError("Overlap between circles")

    return (centers, radii, sum_radii)
