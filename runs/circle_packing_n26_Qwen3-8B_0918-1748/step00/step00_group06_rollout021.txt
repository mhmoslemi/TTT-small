import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    # Initial guess for centers and radii
    centers = np.random.rand(26, 2)
    radii = np.random.rand(26) * 0.1  # Initial guess for radii

    # Define the objective function to maximize sum of radii
    def objective(x):
        # x contains all parameters: first 52 values are centers (26 * 2), next 26 are radii
        centers_flat = x[:52].reshape(26, 2)
        radii_flat = x[52:]
        return -np.sum(radii_flat)  # Minimize negative to maximize sum

    # Define constraints for optimization
    def constraint(x):
        centers_flat = x[:52].reshape(26, 2)
        radii_flat = x[52:]
        for i in range(26):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -1
        return 0

    # Initial parameter vector
    x0 = np.concatenate([centers.flatten(), radii])

    # Set bounds for centers (0 to 1) and radii (0 to 0.5)
    bounds = []
    for _ in range(52):
        bounds.append((0, 1))
    for _ in range(26):
        bounds.append((0, 0.5))

    # Run optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraint})

    # Extract centers and radii from result
    centers = result.x[:52].reshape(26, 2)
    radii = result.x[52:]

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing is invalid: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
