import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Initial guess: 6x4 grid with some adjustment to fit within unit square
    # Start with a grid of 6 rows and 4 columns
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    # Generate initial guess
    for i in range(4):
        for j in range(6):
            row = i
            col = j
            x = col / 4.0 + 0.05
            y = row / 5.0 + 0.05
            centers[n - 1 - (i * 4 + j)] = [x, y]
    # Define function to compute total area
    def objective(radii):
        return -np.sum(np.pi * radii**2)  # Negative for minimization

    # Define constraints: circle stays within square, circles don't overlap
    def constraint1(radii, centers):
        return radii - 1e-10  # Radii must be non-negative

    def constraint2(radii, centers):
        return np.zeros(n)

    def constraint3(radii, centers):
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return 0

    # Set up optimization
    bounds = Bounds([0.0]*n, [np.inf]*n)
    result = minimize(objective, radii, bounds=bounds, constraints=[{'type': 'ineq', 'fun': constraint1, 'args': (centers,)}, 
                                                                    {'type': 'eq', 'fun': constraint2, 'args': (centers,)}, 
                                                                    {'type': 'ineq', 'fun': constraint3, 'args': (centers,)}])

    # Validate result
    if not result.success:
        raise ValueError("Optimization failed")
    radii = result.x
    # Ensure radii are non-negative
    radii[radii < 0] = 0
    # Final validation
    if np.isnan(centers).any() or np.isnan(radii).any():
        raise ValueError("NaN values present")
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
