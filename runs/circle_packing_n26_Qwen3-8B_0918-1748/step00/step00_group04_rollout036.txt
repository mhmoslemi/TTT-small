import numpy as np

import scipy.spatial.distance as distance
from scipy.optimize import minimize

def generate_hexagonal_grid(n, side_length, square_size=(1, 1)):
    centers = []
    for i in range(n):
        row = i // (n // 3 + 1)
        col = i % (n // 3 + 1)
        x = col * (side_length * np.sqrt(3))
        y = row * (side_length * 1.5)
        if x > square_size[0] or y > square_size[1]:
            continue
        centers.append([x, y])
    return np.array(centers)

def run_packing():

    base_centers = generate_hexagonal_grid(25, 0.2, square_size=(1, 1))

    max_r = np.zeros_like(base_radii)
    for i in range(base_centers.shape[0]):

        dx = min(x, 1 - x)
        dy = min(y, 1 - y)
        max_r[i] = min(dx, dy)

    base_radii = np.minimum(base_radii, max_r)
