import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26

    # Initial guess for circle positions using hexagonal packing
    # We'll use a staggered grid with some circles in the corners
    # This is a heuristic approach to create an initial configuration

    # Define centers with some circles at the corners
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Place 4 large circles in the corners
    # Each corner circle has to touch two sides of the square
    # They will have radius r = 1 - 2 * d, where d is the distance to the nearest adjacent circle
    # Start with some initial guess for these corner circles

    # Define a function to compute the max radius for a corner circle
    def max_corner_radius(centers, radii, i):
        # Compute the distance to the nearest adjacent circle
        distances = np.array([np.sqrt(np.sum((centers[i] - centers[j])**2)) for j in range(n)])
        min_dist = np.min(distances)
        return min_dist - 1e-6  # small buffer to prevent overlap

    # Place 4 corner circles
    # Bottom-left corner
    centers[0] = [0.1, 0.1]
    # Top-left corner
    centers[1] = [0.1, 0.9]
    # Top-right corner
    centers[2] = [0.9, 0.9]
    # Bottom-right corner
    centers[3] = [0.9, 0.1]

    # Use initial radii as 0.35 for corner circles
    radii[0] = 0.35
    radii[1] = 0.35
    radii[2] = 0.35
    radii[3] = 0.35

    # Fill the remaining 22 circles in a staggered hexagonal grid
    # We'll use a 5x5 grid with 5 rows and 5 columns, but with 22 circles (so leaving 4 empty positions)
    # Start filling from bottom to top, with staggered positions

    # Generate grid positions with hexagonal spacing
    def generate_hex_grid(n, seed=42):
        np.random.seed(seed)
        pos = []
        for i in range(n):
            # Randomize the position to avoid perfect symmetry
            x = np.random.uniform(0.2, 0.8)
            y = np.random.uniform(0.2, 0.8)
            pos.append([x, y])
        return np.array(pos)

    # Fill remaining circles with a staggered grid
    for i in range(4, n):
        centers[i] = np.random.uniform(0.1, 0.9, 2)
        radii[i] = 0.1  # Initial small radius

    # Optimize using convex optimization with CVXPY
    # Variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Objective: Maximize total radius
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []

    # Boundary constraints: each circle must lie within [0, 1]^2
    for i in range(n):
        constraints.append(x[i, 0] - r[i] >= 0)
        constraints.append(x[i, 0] + r[i] <= 1)
        constraints.append(x[i, 1] - r[i] >= 0)
        constraints.append(x[i, 1] + r[i] <= 1)

    # Non-overlapping constraints: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dist_expr = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist_expr >= r[i] + r[j] - 1e-6)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, max_iter=1000)

    # Extract the solution
    centers_opt = x.value
    radii_opt = r.value

    # Apply a small threshold to ensure non-negative radii
    radii_opt[radii_opt < 0] = 0

    # Validate that all radii are non-negative and centers are within bounds
    # This is important to prevent invalid solutions due to numerical precision

    # Check for invalid values
    for i in range(n):
        if np.isnan(centers_opt[i, 0]) or np.isnan(centers_opt[i, 1]) or np.isnan(radii_opt[i]):
            raise ValueError(f"NaN values found in the solution: circle {i}")

        if radii_opt[i] < 0:
            raise ValueError(f"Negative radius found for circle {i}")

    # Check that all circles are within the unit square
    for i in range(n):
        x, y = centers_opt[i]
        r_i = radii_opt[i]
        if (x - r_i < -1e-6 or x + r_i > 1 + 1e-6 or
            y - r_i < -1e-6 or y + r_i > 1 + 1e-6):
            raise ValueError(f"Circle {i} is outside the unit square")

    # Check that no circles overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers_opt[i, 0] - centers_opt[j, 0])**2 +
                          (centers_opt[i, 1] - centers_opt[j, 1])**2)
            if dist < radii_opt[i] + radii_opt[j] - 1e-6:
                raise ValueError(f"Circles {i} and {j} overlap")

    return centers_opt, radii_opt, np.sum(radii_opt)
