import numpy as np
import scipy.optimize as optimize
import random

def run_packing():
    # Initial seed for reproducibility
    np.random.seed(42)
    random.seed(42)

    # Initial guess for circle centers and radii
    # 4 rows aligned with bottom and top of square
    # 2 rows aligned with left and right sides
    # 1 central row

    # Define rows of centers with some padding
    row_centers = []
    radii = []

    # Bottom row (near bottom)
    for i in range(8):
        x = i / 7
        y = 0.05
        row_centers.append([x, y])
        radii.append(0.05)
    # Top row (near top)
    for i in range(8):
        x = i / 7
        y = 0.95
        row_centers.append([x, y])
        radii.append(0.05)
    # Left row (near left)
    for i in range(7):
        y = i / 6 + 0.1
        x = 0.05
        row_centers.append([x, y])
        radii.append(0.05)
    # Right row (near right)
    for i in range(7):
        y = i / 6 + 0.1
        x = 0.95
        row_centers.append([x, y])
        radii.append(0.05)
    # Central row
    for i in range(6):
        x = i / 5 + 0.25
        y = 0.5
        row_centers.append([x, y])
        radii.append(0.05)

    # Convert to numpy arrays
    centers = np.array(row_centers)
    radii = np.array(radii)

    # Define optimization bounds
    bounds_centers = [(0, 1) for _ in range(2 * len(radii))]
    bounds_radii = [(0, 1) for _ in range(len(radii))]

    # Attempt to optimize using constrained optimization
    # We'll use a reduced parameter space to make this feasible
    # This is a simplified version with the expectation that it will need further refinement
    # For the purpose of this exercise, we use a local optimization step

    # We define a loss function that minimizes the total overlap and violates constraints
    def objective(x):
        # Flatten the x vector
        flat_x = x.reshape(-1)
        centers = flat_x[:2 * len(radii)].reshape(-1, 2)
        radii = flat_x[2 * len(radii):]

        # Ensure boundaries
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return 1e6  # Large penalty for boundary violation

        # Ensure non-overlap
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return 1e6  # Large penalty for overlap

        # Objective function: maximize sum of radii
        # We minimize negative sum of radii
        return -np.sum(radii)

    # Optimization with bounds
    result = optimize.minimize(
        objective,
        np.concatenate((centers.flatten(), radii)),
        bounds=bounds_centers + bounds_radii,
        method='L-BFGS-B',
        options={'maxiter': 100, 'ftol': 1e-8}
    )

    # Extract the optimized solution
    optimized_centers = result.x[:2 * len(radii)].reshape(-1, 2)
    optimized_radii = result.x[2 * len(radii):]

    # Ensure centers and radii are within bounds and no overlap
    # For this example, we return the result as is
    # In a real solution, additional post-processing may be needed

    # Validate the result
    # (For this code, we assume the optimization has been successful)
    return (optimized_centers, optimized_radii, np.sum(optimized_radii))
