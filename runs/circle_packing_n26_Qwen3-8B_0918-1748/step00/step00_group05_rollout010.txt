import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds

def run_packing():
    # Initial configuration: 5x5 grid of circles with some spacing to allow for 26th circle
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Generate 25 circles in a 5x5 grid
    grid_spacing = 0.2
    for i in range(5):
        for j in range(5):
            x = i * grid_spacing + 0.05
            y = j * grid_spacing + 0.05
            centers[5*i + j] = [x, y]
            radii[5*i + j] = 0.05
    
    # Add the 26th circle to a corner to maximize space
    centers[25] = [0.05, 0.05]
    radii[25] = 0.05
    
    # Optimization parameters
    bounds = Bounds([0, 0], [1, 1])
    x0 = centers.copy()
    r0 = radii.copy()
    
    # Objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)
    
    # Constraints
    def constraint1(x):
        return x
    
    def constraint2(x):
        return np.sqrt(np.sum((centers - centers[25]) ** 2)) - radii[25] - radii[25]
    
    cons = [{'type': 'ineq', 'fun': constraint2}]
    
    # Optimization
    result = minimize(objective, x0, bounds=bounds, constraints=cons)
    
    # Extract optimized centers and radii
    optimized_centers = result.x
    optimized_radii = radii
    
    # Ensure all circles are within the square and no overlaps
    for i in range(n):
        x, y = optimized_centers[i]
        if x - optimized_radii[i] < -1e-12 or x + optimized_radii[i] > 1 + 1e-12:
            optimized_radii[i] = 1 - max(x, 1 - x)
    
    for i in range(n):
        x, y = optimized_centers[i]
        r = optimized_radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            optimized_radii[i] = 1 - max(x, 1 - x)
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((optimized_centers[i] - optimized_centers[j]) ** 2))
            if dist < optimized_radii[i] + optimized_radii[j] - 1e-12:
                optimized_radii[i] = (dist - optimized_radii[j]) / 2
                optimized_radii[j] = (dist - optimized_radii[i]) / 2
    
    return (optimized_centers, optimized_radii, np.sum(optimized_radii))
