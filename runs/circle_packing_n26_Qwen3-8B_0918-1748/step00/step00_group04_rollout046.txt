import numpy as np
import scipy.optimize as optimize
from scipy.optimize import LinearConstraint, Bounds

def run_packing():
    # Initial guess for centers and radii
    # We will use a hybrid of hexagonal and corner-based packing
    # Start with a hexagonal grid for the center and add corner circles

    # First, define the centers of the 26 circles
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Hexagonal grid for central region (16 circles)
    for i in range(4):
        for j in range(4):
            if i % 2 == 0:
                centers[4 * i + j] = [0.1 + j * 0.15, 0.1 + i * 0.15]
            else:
                centers[4 * i + j] = [0.1 + j * 0.15 + 0.075, 0.1 + i * 0.15]
    
    # Add 9 more circles in an extended hexagonal pattern
    for i in range(3):
        for j in range(3):
            if i % 2 == 0:
                centers[16 + 4 * i + j] = [0.1 + j * 0.15 + 0.15, 0.1 + i * 0.15 + 0.15]
            else:
                centers[16 + 4 * i + j] = [0.1 + j * 0.15 + 0.15 + 0.075, 0.1 + i * 0.15 + 0.15]
    
    # Add 2 corner circles
    centers[24] = [0.05, 0.05]
    centers[25] = [0.95, 0.95]
    
    # Initial radii guess based on spacing and boundary constraints
    # Target radius for central and main grid circles
    r = 0.04
    for i in range(24):
        radii[i] = r
    radii[24] = 0.02  # corner circle
    radii[25] = 0.02  # corner circle
    
    # Optimization: adjust radii and positions to maximize total radius sum
    # Define the objective function to maximize sum of radii
    def objective(x):
        # x contains all parameters: first half is centers, next half is radii
        x_centers = x[:26*2].reshape(26, 2)
        x_radii = x[26*2:]
        return -np.sum(x_radii)  # We minimize -sum(radii) to maximize sum(radii)
    
    # Constraints: circles must be within the square, non-overlapping
    def constraint1(x):
        x_centers = x[:26*2].reshape(26, 2)
        x_radii = x[26*2:]
        for i in range(26):
            xi, yi = x_centers[i]
            ri = x_radii[i]
            if xi - ri < -1e-12 or xi + ri > 1 + 1e-12 or yi - ri < -1e-12 or yi + ri > 1 + 1e-12:
                return -1
        return 0
    
    def constraint2(x):
        x_centers = x[:26*2].reshape(26, 2)
        x_radii = x[26*2:]
        for i in range(26):
            for j in range(i + 1, 26):
                xi, yi = x_centers[i]
                xj, yj = x_centers[j]
                ri = x_radii[i]
                rj = x_radii[j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                if dist < ri + rj - 1e-12:
                    return -1
        return 0
    
    # Bounds: centers must be in [0,1]^2, radii non-negative
    bounds = [(0, 1) for _ in range(26*2)] + [(0, 1) for _ in range(26)]
    
    # Initial guess for x: centers from above, radii from initial guess
    initial_guess = np.concatenate((centers.flatten(), radii))
    
    # Optimization with constraints
    result = optimize.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}
        ],
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    # Extract results
    final_centers = result.x[:26*2].reshape(26, 2)
    final_radii = result.x[26*2:]
    total_sum = np.sum(final_radii)
    
    return final_centers, final_radii, total_sum
