import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt
from matplotlib.patches import Circle
import time

def run_packing():
    # Define initial grid configuration (4x4 grid with spacing)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial configuration: 16 circles in a 4x4 grid
    grid_size = 4
    grid_spacing = 0.25  # initial spacing
    for i in range(4):
        for j in range(4):
            idx = i * 4 + j
            centers[idx, 0] = i * grid_spacing
            centers[idx, 1] = j * grid_spacing
            radii[idx] = 0.05  # initial radius

    # Add 10 additional circles in the remaining space
    # Try to fit circles in gaps around the grid
    add_circle_params = [
        [0.1, 0.1, 0.08],  # near top-left
        [0.8, 0.1, 0.08],  # near top-right
        [0.1, 0.8, 0.08],  # near bottom-left
        [0.8, 0.8, 0.08],  # near bottom-right
        [0.5, 0.2, 0.08],  # center-left
        [0.5, 0.8, 0.08],  # center-right
        [0.2, 0.5, 0.08],  # left-center
        [0.8, 0.5, 0.08],  # right-center
        [0.3, 0.3, 0.08],  # top-center
        [0.7, 0.7, 0.08],  # bottom-center
    ]
    for i in range(10):
        idx = 16 + i
        x, y, r = add_circle_params[i]
        centers[idx, 0] = x
        centers[idx, 1] = y
        radii[idx] = r

    # Initial sum of radii
    sum_radii = np.sum(radii)

    # Optimization parameters
    bounds = [(0.0, 1.0) for _ in range(2 * n)]
    constraints = []
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - x[2*i + 1] + x[2*i + 1] - 0.001})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1.0 - x[2*i] - x[2*i + 1] + x[2*i + 1] - 0.001})
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: np.sqrt((x[2*j] - x[2*i])**2 + (x[2*j + 1] - x[2*i + 1])**2) - x[2*n + i] - x[2*n + j] + 0.001})

    # Objective function to maximize sum of radii
    def objective(x):
        return np.sum(x[2*n:])

    # Optimization
    x0 = np.concatenate([centers.flatten(), radii])
    result = minimize(objective, x0, bounds=bounds, constraints=constraints, method='SLSQP', options={'ftol': 1e-9, 'maxiter': 1000})

    # Extract results
    centers = result.x[0:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    sum_radii = np.sum(radii)

    # Final validation check
    # Ensure no circles are out of bounds
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            raise ValueError(f"Circle {i} is out of bounds")

    # Ensure no circles are overlapping
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} are overlapping")

    return centers, radii, sum_radii
