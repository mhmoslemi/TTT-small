```python
import numpy as np
import torch
import torch.optim as optim
import torch.nn as nn
import time

def run_packing():
    # Define initial grid-based centers for 26 circles
    # Using a 5x5 grid with 1 extra circle at the end
    grid_size = 5
    centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i / (grid_size - 1)
            y = j / (grid_size - 1)
            centers.append([x, y])
    # Add one more circle to make 26
    centers.append([0.5, 0.5])
    centers = np.array(centers)

    # Convert to PyTorch tensors for optimization
    centers = torch.tensor(centers, requires_grad=True, dtype=torch.float64)
    radii = torch.nn.Parameter(torch.full((26,), 0.05, dtype=torch.float64), requires_grad=True)

    # Define the loss function to maximize total radius
    # We will use a custom neural network to represent the loss
    class CustomLoss(nn.Module):
        def __init__(self, centers, radii, boundary_tolerance=1e-3):
            super(CustomLoss, self).__init__()
            self.centers = centers
            self.radii = radii
            self.boundary_tolerance = boundary_tolerance

        def forward(self):
            # Compute total radius
            total_radius = torch.sum(self.radii)

            # Penalize boundary violations
            boundary_penalty = 0
            for i in range(26):
                x, y = self.centers[i]
                r = self.radii[i]
                if x - r < -self.boundary_tolerance:
                    boundary_penalty += (x - r + self.boundary_tolerance) ** 2
                if x + r > 1 + self.boundary_tolerance:
                    boundary_penalty += (x + r - 1 - self.boundary_tolerance) ** 2
                if y - r < -self.boundary_tolerance:
                    boundary_penalty += (y - r + self.boundary_tolerance) ** 2
                if y + r > 1 + self.boundary_tolerance:
                    boundary_penalty += (y + r - 1 - self.boundary_tolerance) ** 2

            # Penalize overlaps between circles
            overlap_penalty = 0
            for i in range(26):
                for j in range(i + 1, 26):
                    dx = self.centers[i, 0] - self.centers[j, 0]
                    dy = self.centers[i, 1] - self.centers[j, 1]
                    dist = torch.sqrt(dx ** 2 + dy ** 2)
                    if dist < self.radii[i] + self.radii[j] - 1e-3:
                        overlap_penalty += (dist - (self.radii[i] + self.radii[j] - 1e-3)) ** 2

            # Our goal is to maximize total_radius, so we minimize -total_radius + boundary_penalty + overlap_penalty
            return -total_radius + boundary_penalty + overlap_penalty

    # Instantiate the loss function
    loss_fn = CustomLoss(centers, radii)

    # Set up optimizer
    optimizer = optim.Adam([centers, radii], lr=0.01)

    # Training loop
    num_iterations = 10000
    for iteration in range(num_iterations):
        optimizer.zero_grad()
        loss = loss_fn()
        loss.backward()
        optimizer.step()

        if iteration % 100 == 0:
            print(f"Iteration {iteration}, Current loss: {loss.item()}")

    # Convert to numpy arrays for final output
    centers_np = centers.detach().numpy()
    radii_np = radii.detach().numpy()
    sum_radii = np.sum(radii_np)

    # Validate packing (for demonstration purposes, in real use case, this would be done before submission)
    # However, the validate_packing function is read-only, so we won't call it here

    return centers_np, radii_np, sum_radii
```
