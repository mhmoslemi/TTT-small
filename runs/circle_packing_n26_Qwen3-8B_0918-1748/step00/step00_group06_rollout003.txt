import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Initial configuration with hexagonal packing
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Place 24 circles in a hexagonal grid
    for i in range(24):
        row = i // 4
        col = i % 4
        x = col + 0.5 * (i % 2)
        y = row + 0.5 * (i // 2)
        centers[i] = [x / 5, y / 5]

    centers[24] = [0.1, 0.1]
    centers[25] = [0.9, 0.9]

    def objective(x):
        # x is a flat array of [x0, y0, r0, x1, y1, r1, ...]
        # Reshape to centers and radii
        centers_flat = x[:2 * num_circles].reshape(num_circles, 2)
        radii_flat = x[2 * num_circles:].reshape(num_circles)
        
        # Objective is to maximize sum of radii
        return -np.sum(radii_flat)  # Minimize negative sum
    
    # Constraints: non-overlapping and boundary
    def constraint_non_overlap(x):
        centers_flat = x[:2 * num_circles].reshape(num_circles, 2)
        radii_flat = x[2 * num_circles:].reshape(num_circles)
        
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return dist - (radii_flat[i] + radii_flat[j] - 1e-8)
        return 0
    
    def constraint_boundary(x):
        centers_flat = x[:2 * num_circles].reshape(num_circles, 2)
        radii_flat = x[2 * num_circles:].reshape(num_circles)
        
        for i in range(num_circles):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]

                return x_i - r_i + 1e-8

                return y_i - r_i + 1e-8
        return 0
    
    # Bounds: x and y in [0, 1], radii in [0, 1]

        bounds.extend([(0, 1), (0, 1)])  # x, y
        bounds.append((0, 1))  # radius
    
    # Initial guess: flatten centers and radii
    x0 = []
    for i in range(num_circles):
        x0.extend(centers[i])
        x0.append(radii[i])
    x0 = np.array(x0)
    
    # Optimization with constraints
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=Bounds(bounds),
        constraints=[
            {'type': 'ineq', 'fun': constraint_non_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    # Extract results
    centers_flat = result.x[:2 * num_circles].reshape(num_circles, 2)
