import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26

    # Initial guess for circle positions: hexagonal packing in a square
    # We will create a grid of circles with some spacing and adjust via optimization
    # Start with a grid layout to estimate initial positions and radii

    # Create a grid of circle centers with some spacing
    # We will use a nested approach with rows and columns
    # Let's use a square grid with some space to allow for optimization
    # Initial guess for positions
    positions = np.zeros((n, 2))
    radii = np.zeros(n)

    # Start with a grid layout
    rows = 5
    cols = 5
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col / (cols - 1)
        y = row / (rows - 1)
        positions[i] = [x, y]

    # Initial guess for radii (equal size for all)
    r_initial = 0.1
    radii[:] = r_initial

    # Define optimization variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n, non_negative=True)

    # Objective: maximize total radius sum
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    # 1. Each circle must be within the unit square
    constraints = []
    for i in range(n):
        constraints.append(x[i, 0] - r[i] >= 0)
        constraints.append(x[i, 0] + r[i] <= 1)
        constraints.append(x[i, 1] - r[i] >= 0)
        constraints.append(x[i, 1] + r[i] <= 1)

    # 2. No overlap between circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-8)

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-5)

    if not prob.success:
        raise ValueError("Optimization failed")

    # Extract the final centers and radii
    centers = np.round(x.value, 6)
    radii_values = np.round(r.value, 6)

    # Validate the packing
    is_valid, message = validate_packing(centers, radii_values)
    if not is_valid:
        raise ValueError("Invalid packing: " + message)

    total_sum = np.sum(radii_values)
    return centers, radii_values, total_sum

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"
