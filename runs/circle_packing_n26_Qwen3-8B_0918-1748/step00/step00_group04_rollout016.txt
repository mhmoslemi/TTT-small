import numpy as np
from scipy.optimize import minimize
from itertools import combinations

def run_packing():
    # Initial layout based on hexagonal packing
    num_circles = 26
    rows = 5
    cols = 6
    spacing = 1.0  # initial spacing for horizontal alignment
    centers = []
    radii = []
    
    # Calculate the initial radius based on spacing between circles
    # We'll use a heuristic to fit as many circles as possible into the square
    # and then adjust based on boundary constraints
    
    # Start by placing circles in a hexagonal grid
    for row in range(rows):
        for col in range(cols):
            x = col * spacing / (cols - 1) if cols > 1 else 0.5
            y = row * (spacing * np.sqrt(3)) / (rows - 1) if rows > 1 else 0.5
            centers.append([x, y])
    
    # Trim to 26 circles
    centers = centers[:26]
    
    # Now we'll optimize the radii to maximize the total sum, under constraints
    # Define an objective function to maximize the sum of radii
    
    def obj_func(radii):
        return -np.sum(radii)  # Negative for minimization
    
    # Define constraints: circles must fit in the square, and be non-overlapping
    
    def constraint_in_square(radii_vec):
        return np.sum(np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2) - (radii_vec[i] + radii_vec[j]) for i, j in combinations(range(26), 2)) >= 0
    
    # Initial guess for radii
    initial_radii = np.array([0.05] * 26)
    
    # Optimization with constraints
    constraints = [
        {'type': 'ineq', 'fun': lambda r: np.array([centers[i][0] - r[i] for i in range(26)]) + 1e-12},
        {'type': 'ineq', 'fun': lambda r: np.array([1.0 - centers[i][0] - r[i] for i in range(26)]) + 1e-12},
        {'type': 'ineq', 'fun': lambda r: np.array([centers[i][1] - r[i] for i in range(26)]) + 1e-12},
        {'type': 'ineq', 'fun': lambda r: np.array([1.0 - centers[i][1] - r[i] for i in range(26)]) + 1e-12},
    ]
    
    result = minimize(obj_func, initial_radii, constraints=constraints)
    
    # Extract the optimized centers and radii
    radii = result.x
    centers = np.array(centers)
    
    # Ensure all circles are within the square and non-overlapping
    # This is manually enforced via constraints to avoid NaNs
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
