import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Initial guess for circle centers: grid-based hexagonal packing with 18 circles
    # and 4 corner clusters with 2 circles each, total 26 circles

    # Hexagonal grid parameters
    n_hex = 18
    r_hex = 0.05  # Initial guess for radius

    # Generate hexagonal grid points in the center of the square
    centers_hex = []
    x = np.linspace(0.1, 0.9, 6)
    y = np.linspace(0.1, 0.9, 4)
    for i, xi in enumerate(x):
        for j, yi in enumerate(y):
            if i % 2 == 0:
                centers_hex.append([xi, yi])
            else:
                centers_hex.append([xi, yi + 0.1])
    centers_hex = np.array(centers_hex)

    # Corner clusters: 4 corners, each with 2 circles
    centers_corner = []
    for dx, dy in [(0, 0), (1, 1), (1, 0), (0, 1)]:
        if dx == 1 and dy == 1:
            x = 0.9
            y = 0.9
        elif dx == 1 and dy == 0:
            x = 0.9
            y = 0.1
        elif dx == 0 and dy == 1:
            x = 0.1
            y = 0.9
        else:
            x = 0.1
            y = 0.1
        centers_corner.append([x, y])
        centers_corner.append([x + 0.1, y + 0.1])
    centers_corner = np.array(centers_corner)

    # Initial centers
    centers = np.vstack((centers_hex, centers_corner))
    radii = np.full(centers.shape[0], r_hex)

    # Optimization parameters
    max_iter = 100
    tol = 1e-6
    bounds = [(0.0, 1.0) for _ in range(2 * centers.shape[0])]
    bounds += [(0.0, 1.0) for _ in range(centers.shape[0])]

    def objective(radii):
        return -np.sum(radii)

    def constraints_func(x):
        center_x = x[:2 * centers.shape[0]].reshape(centers.shape[0], 2)
        radius_x = x[2 * centers.shape[0]:]
        # Check if circles are inside the square
        for i in range(centers.shape[0]):
            if center_x[i, 0] - radius_x[i] < -1e-6 or center_x[i, 0] + radius_x[i] > 1.0 + 1e-6:
                return np.inf
            if center_x[i, 1] - radius_x[i] < -1e-6 or center_x[i, 1] + radius_x[i] > 1.0 + 1e-6:
                return np.inf
        # Check if circles are non-overlapping
        distances = squareform(pdist(center_x))
        for i in range(centers.shape[0]):
            for j in range(i + 1, centers.shape[0]):
                if distances[i, j] < radius_x[i] + radius_x[j] - 1e-6:
                    return np.inf
        return 0

    # Optimize
    result = minimize(objective, np.concatenate((centers.reshape(-1), radii)), bounds=bounds, method='SLSQP', constraints={'type': 'ineq', 'fun': constraints_func}, options={'maxiter': max_iter, 'ftol': tol, 'eps': 1e-8})

    if not result.success:
        raise ValueError("Optimization failed.")

    centers_opt = result.x[:2 * centers.shape[0]].reshape(centers.shape[0], 2)
    radii_opt = result.x[2 * centers.shape[0]:]
    sum_radii = np.sum(radii_opt)

    # Final validation
    for i in range(centers_opt.shape[0]):
        x, y = centers_opt[i]
        r = radii_opt[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")

    for i in range(centers_opt.shape[0]):
        for j in range(i + 1, centers_opt.shape[0]):
            dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
            if dist < radii_opt[i] + radii_opt[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    return centers_opt, radii_opt, sum_radii
