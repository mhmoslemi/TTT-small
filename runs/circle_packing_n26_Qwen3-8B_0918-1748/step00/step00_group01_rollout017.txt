import numpy as np
import matplotlib.pyplot as plt

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    spacing = 0.75  # spacing between centers of adjacent circles
    col_offset = spacing / 2
    row_offset = (np.sqrt(3) / 2) * spacing  # vertical offset for hexagonal packing

    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + (col_offset if i % 2 == 0 else 0)
            y = i * row_offset
            if i < rows - 1 and j < cols - 1:
                centers.append([x, y])
    centers = np.array(centers)
    centers = centers[:n]

    # Place corner circles
    corners = np.array([
        [0.0, 0.0],
        [1.0, 0.0],
        [1.0, 1.0],
        [0.0, 1.0]
    ])
    corner_indices = [0, 1, 2, 3]
    centers[corner_indices] = corners

    # Calculate initial radius for all circles
    def distance_squared(a, b):
        return np.sum((a - b) ** 2)

    max_initial_radius = 0.25
    radii = np.full(n, max_initial_radius)

    # Optimization function to maximize total radius
    def objective(x):
        return -np.sum(x)

    # Constraints: distance between centers must be at least sum of radii
    def constraint_func(x):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(distance_squared(centers[i], centers[j]))
                if dist < x[i] + x[j] - 1e-12:
                    return -1
        return 0

    # Adjust centers slightly to make space for larger radii
    def adjust_centers(centers):
        return centers + np.random.normal(0, 0.05, centers.shape)

    # Run optimization with adjusted centers
    def optimize_packing():
        # First, compute initial radius
        radii = np.zeros(n)
        for i in range(n):
            r = min(0.5, 1 - centers[i, 0] - centers[i, 1])
            radii[i] = r

        # We will perform some sort of gradient descent with constraints
        # For simplicity, we will incrementally increase the radii
        # while ensuring that all constraints are satisfied

        # Iteratively refine radii
        for _ in range(10):
            # Adjust centers slightly to make space
            centers = adjust_centers(centers)

            # Compute current max radius
            current_max = 0
            for i in range(n):
                max_r = 0.5  # max possible radius to stay within square
                for j in range(i + 1, n):
                    dist = np.sqrt(distance_squared(centers[i], centers[j]))
                    max_r = min(max_r, (dist - 1e-12) / 2)
                max_r = min(max_r, 1 - centers[i, 0] - centers[i, 1])
                max_r = min(max_r, centers[i, 0] + centers[i, 1])
                radii[i] = min(radii[i] + 0.02, max_r)

        return centers, radii

    centers, radii = optimize_packing()
    return centers, radii, np.sum(radii)
