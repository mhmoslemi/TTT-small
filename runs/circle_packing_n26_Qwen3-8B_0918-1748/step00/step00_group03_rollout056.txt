import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run_packing():
    # Define initial centers based on a grid pattern
    grid_size = 5
    grid = np.linspace(0.1, 0.9, grid_size)
    centers = []
    for x in grid:
        for y in grid:
            centers.append([x, y])
    # Add some additional circles for the remaining 26 - grid_size**2
    extra = 26 - len(centers)
    for i in range(extra):
        # Add circles near the corners
        if i % 3 == 0:
            centers.append([0.1, 0.1])
        elif i % 3 == 1:
            centers.append([0.9, 0.1])
        else:
            centers.append([0.1, 0.9])
    centers = np.array(centers)
    
    # Initial guess for radii
    radii = np.full(len(centers), 0.1)

    def objective(radii_guess):
        return -np.sum(radii_guess)

    def constraint_feasibility(radii_guess):
        # Ensure all radii are non-negative
        return radii_guess >= 0

    def boundary_constraint(radii_guess):
        # Ensure the circles do not go out of bounds
        # This is indirectly enforced by adjusting the centers
        return np.array([])

    def circle_to_circle_constraint(radii_guess):
        # Check that circles do not overlap
        n = len(centers)
        dist_matrix = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :])**2, axis=2))
        for i in range(n):
            for j in range(i+1, n):
                if dist_matrix[i, j] < radii_guess[i] + radii_guess[j] - 1e-8:
                    return np.array([0.0])
        return np.array([0.0])

    # Optimization bounds
    bounds = Bounds([0.0]*len(radii), [1.0]*len(radii))

    # Initial guess for radii
    x0 = radii

    # Optimization setup
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_feasibility},
            {'type': 'ineq', 'fun': circle_to_circle_constraint},
        ],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 200}
    )

    # Extract the result
    radii_opt = result.x
    centers_opt = centers.copy()

    # Validate the packing
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    is_valid, msg = validate_packing(centers_opt, radii_opt)
    if not is_valid:
        raise ValueError(f"Validation failed: {msg}")

    return centers_opt, radii_opt, np.sum(radii_opt)
