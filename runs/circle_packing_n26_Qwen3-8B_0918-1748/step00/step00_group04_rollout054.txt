import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Initial grid of 5x5 circles
    grid_size = 5
    grid_spacing = 1.0 / (grid_size - 1)
    centers = np.zeros((grid_size * grid_size, 2))
    radii = np.zeros(grid_size * grid_size)
    
    for i in range(grid_size):
        for j in range(grid_size):
            idx = i * grid_size + j
            x = i * grid_spacing
            y = j * grid_spacing
            centers[idx] = [x, y]
            radii[idx] = grid_spacing / 2

    # Add the 26th circle near the top-left corner
    # Start with a guess based on corner placement
    corner_circle = np.array([0.1, 0.1])
    corner_radius = 0.15
    centers = np.vstack((centers, corner_circle))
    radii = np.append(radii, corner_radius)

    # Optimization function
    def objective(radii):
        return -np.sum(radii)  # We want to maximize the sum, so minimize negative sum

    def constraints(radii):
        # Create constraints for non-overlapping and boundary
        n = len(radii)
        con = []
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            # Constraint: circle stays within square
            con.append((x_i - r_i >= -1e-12))
            con.append((x_i + r_i <= 1 + 1e-12))
            con.append((y_i - r_i >= -1e-12))
            con.append((y_i + r_i <= 1 + 1e-12))
        
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                con.append((dist >= radii[i] + radii[j] - 1e-12))
        
        return con

    # Setup optimization
    lb = np.zeros_like(radii)
    ub = np.full_like(radii, 1.0)
    
    result = minimize(objective, radii, bounds=list(zip(lb, ub)), constraints=constraints, method='SLSQP')

    # Extract results
    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
