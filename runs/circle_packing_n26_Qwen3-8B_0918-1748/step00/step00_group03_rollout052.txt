import numpy as np
from scipy.optimize import minimize, Bounds
import math

def run_packing():
    # Base grid dimensions
    rows = 5
    cols = 6

    # Hexagonal grid coordinates
    grid = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * i
            y = i
            grid.append((x, y))
    
    # Create a numpy array of grid points
    centers = np.array(grid)
    radii = np.zeros(len(centers))
    sum_radii = 0.0

    # First pass: initial placement based on grid and boundary constraints
    for i, (x, y) in enumerate(centers):
        # Determine distance to nearest boundary
        top = 1.0 - y
        bottom = y
        left = x
        right = 1.0 - x
        min_bound = min(top, bottom, left, right)
        radii[i] = min_bound * 0.85  # 85% of minimum boundary distance
        sum_radii += radii[i]

    # Optimization to maximize sum of radii with non-overlapping and boundary constraints
    def objective(radii_vec):
        return -np.sum(radii_vec)  # minimize negative sum to maximize sum

    def constraints(radii_vec):
        constraints = []
        for i in range(len(centers)):
            constraints.append(radii_vec[i] >= 0)
        return constraints

    # Prepare optimization variables
    bounds = [(0, 1) for _ in range(len(centers))]
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    radii = result.x
    sum_radii = np.sum(radii)

    # Final validation: adjust centers to bring circles closer to boundaries if needed
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            centers[i, 0] = r
        elif x + r > 1:
            centers[i, 0] = 1 - r
        if y - r < 0:
            centers[i, 1] = r
        elif y + r > 1:
            centers[i, 1] = 1 - r

    # Final validation: adjust radii to ensure no overlap
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            if dist < radii[i] + radii[j]:
                # Adjust radii based on distance
                max_r = (dist - 1e-12) / 2
                radii[i] = max(radii[i], max_r)
                radii[j] = max(radii[j], max_r)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
