import numpy as np
from scipy.optimize import minimize
from itertools import combinations

def run_packing():
    # Initial configuration: place 26 circles with varying radii using a hybrid approach
    # Group some circles near corners, some near edges, and use a hexagonal approximation for the rest
    
    # Define initial centers using a staggered pattern with some circles at corners
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Grouping strategy: 3 clusters in each corner, 5 in center, 10 along edges
    
    # Top-left corner cluster
    for i in range(3):
        theta = 2 * np.pi / 3 * i
        r = 0.1 + 0.05 * i
        centers[i] = (0.1 + r * np.cos(theta), 0.1 + r * np.sin(theta))
        radii[i] = r
    
    # Top-right corner cluster
    for i in range(3):
        theta = 2 * np.pi / 3 * i
        r = 0.1 + 0.05 * i
        centers[3 + i] = (0.9 - r * np.cos(theta), 0.1 + r * np.sin(theta))
        radii[3 + i] = r
    
    # Bottom-left corner cluster
    for i in range(3):
        theta = 2 * np.pi / 3 * i
        r = 0.1 + 0.05 * i
        centers[6 + i] = (0.1 + r * np.cos(theta), 0.9 - r * np.sin(theta))
        radii[6 + i] = r
    
    # Bottom-right corner cluster
    for i in range(3):
        theta = 2 * np.pi / 3 * i
        r = 0.1 + 0.05 * i
        centers[9 + i] = (0.9 - r * np.cos(theta), 0.9 - r * np.sin(theta))
        radii[9 + i] = r
    
    # Central cluster
    for i in range(5):
        r = 0.15 + 0.05 * i
        centers[12 + i] = (0.5, 0.5)
        radii[12 + i] = r
    
    # Edge cluster (top, right, bottom, left)
    for i in range(10):
        if i < 4:
            # Top edge: near (x, 1)
            x = 0.25 + (i * 0.2)
            radii[17 + i] = 0.08
            centers[17 + i] = (x, 1 - radii[17 + i])
        elif i < 8:
            # Right edge: near (1, y)
            y = 0.25 + ((i - 4) * 0.2)
            radii[17 + i] = 0.08
            centers[17 + i] = (1 - radii[17 + i], y)
        else:
            # Bottom edge: near (x, 0)
            x = 0.25 + ((i - 8) * 0.2)
            radii[17 + i] = 0.08
            centers[17 + i] = (x, radii[17 + i])
    
    # Define the objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize
    
    # Define constraints for boundary and non-overlap
    def constraint_boundary(radii):
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        return 0
    
    def constraint_overlap(radii):
        for i, j in combinations(range(26), 2):
            x1, y1 = centers[i]
            x2, y2 = centers[j]
            r1, r2 = radii[i], radii[j]
            dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            if dist < r1 + r2 - 1e-12:
                return -1
        return 0
    
    # Combine constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    # Optimize using bounds and constraints
    result = minimize(
        objective,
        radii,
        bounds=[(0, 1) for _ in range(26)],
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-6, 'maxiter': 1000}
    )
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    return centers, result.x, np.sum(result.x)
