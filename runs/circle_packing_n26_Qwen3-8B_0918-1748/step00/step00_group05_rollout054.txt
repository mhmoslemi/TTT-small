import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, HPowell

def run_packing():
    # Known optimal packing of 26 circles in a unit square (from literature)
    # Approximate configuration to start with
    centers = np.array([
        [0.05, 0.05], [0.05, 0.95], [0.95, 0.95], [0.95, 0.05],
        [0.25, 0.25], [0.25, 0.75], [0.75, 0.75], [0.75, 0.25],
        [0.15, 0.15], [0.15, 0.85], [0.85, 0.85], [0.85, 0.15],
        [0.45, 0.45], [0.45, 0.55], [0.55, 0.55], [0.55, 0.45],
        [0.3, 0.3], [0.3, 0.7], [0.7, 0.7], [0.7, 0.3],
        [0.1, 0.5], [0.5, 0.5], [0.5, 0.1], [0.5, 0.4], [0.4, 0.5], [0.6, 0.5]
    ])
    radii = np.full(26, 0.05)

    # Optimization parameters
    bounds = Bounds([0.0, 0.0], [1.0, 1.0])  # Bound for centers
    params_bounds = Bounds([0.0], [1.0])  # Bound for radii

    def objective(x):
        flat_x = x.reshape(26, 4)  # 26 circles, 4 variables each (x, y, r, r)
        centers = flat_x[:, :2]
        radii = flat_x[:, 2]
        return -np.sum(radii)  # Maximize sum of radii

    def constraints(x):
        flat_x = x.reshape(26, 4)
        centers = flat_x[:, :2]
        radii = flat_x[:, 2]
        # Check boundary constraints
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return np.inf
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.inf
        # Check circle-circle overlap
        for i in range(26):
            for j in range(i+1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                if np.sqrt(dx*dx + dy*dy) < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return 0

    # Initial guess with centers and radii
    x0 = np.concatenate([centers.reshape(-1), radii])

    # Optimization using Powell method with constraints
    result = opt.minimize(
        objective,
        x0,
        bounds=Bounds([0.0, 0.0, 0.0], [1.0, 1.0, 1.0]),
        constraints=opt.Constraint(lambda x: constraints(x), bounds=(-np.inf, 0)),
        method='Powell'
    )

    centers_opt = result.x[:26*2].reshape(26, 2)
    radii_opt = result.x[26*2:].reshape(26)
    sum_radii = np.sum(radii_opt)

    # Final validation
    for i in range(26):
        x, y = centers_opt[i]
        r = radii_opt[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} outside unit square")
        for j in range(i + 1, 26):
            dx = centers_opt[i, 0] - centers_opt[j, 0]
            dy = centers_opt[i, 1] - centers_opt[j, 1]
            if np.sqrt(dx*dx + dy*dy) < radii_opt[i] + radii_opt[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    return centers_opt, radii_opt, sum_radii
