import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Initial configuration: 5 rows of 5 circles each, with spacing
    init_centers = np.array([
        [0.1, 0.1], [0.3, 0.1], [0.5, 0.1], [0.7, 0.1], [0.9, 0.1],
        [0.1, 0.3], [0.3, 0.3], [0.5, 0.3], [0.7, 0.3], [0.9, 0.3],
        [0.1, 0.5], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5], [0.9, 0.5],
        [0.1, 0.7], [0.3, 0.7], [0.5, 0.7], [0.7, 0.7], [0.9, 0.7],
        [0.1, 0.9], [0.3, 0.9], [0.5, 0.9], [0.7, 0.9], [0.9, 0.9]
    ])
    init_radii = np.full(26, 0.05)
    
    def objective(x):
        return -np.sum(x[1:])  # Minimize negative sum to maximize sum
    
    def constraints(x):
        centers = x[:26*2].reshape(26, 2)
        radii = x[26*2:]
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return False
        for (i, j) in combinations(range(26), 2):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - 1e-12:
                return False
        return True
    
    bounds = [(0, 1) for _ in range(26*2)] + [(0, 1) for _ in range(26)]
    x0 = np.concatenate([init_centers.reshape(-1), init_radii])
    
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraints}],
        method='SLSQP',
        options={'disp': False, 'ftol': 1e-12, 'maxiter': 1000}
    )
    
    centers = result.x[:26*2].reshape(26, 2)
    radii = result.x[26*2:]
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
