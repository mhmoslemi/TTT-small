import numpy as np
from scipy.optimize import minimize
import random

def run_packing():
    # Initial grid layout with 5 rows and 5 columns (25 circles) and 1 extra circle
    num_circles = 26
    initial_centers = []
    for i in range(5):
        for j in range(5):
            x = i / 4.0 + 0.05
            y = j / 4.0 + 0.05
            initial_centers.append([x, y])
    # Add the 26th circle near the top-right corner
    initial_centers.append([0.9, 0.9])
    initial_centers = np.array(initial_centers)

    # Initial radii as a rough estimate based on distances between adjacent circles
    def initial_radii_func(centers):
        radii = np.zeros(len(centers))
        for i in range(len(centers)):
            min_dist = float('inf')
            for j in range(len(centers)):
                if i != j:
                    dist = np.linalg.norm(centers[i] - centers[j])
                    if dist < min_dist:
                        min_dist = dist
            radii[i] = min_dist / 2.0
        return radii

    initial_radii = initial_radii_func(initial_centers)

    # Flatten positions and radii for optimization
    x0 = np.concatenate([initial_centers.reshape(-1), initial_radii])

    # Objective function to maximize sum of radii (minimize negative sum)
    def objective(x):
        flat_centers = x[:2 * num_circles].reshape(num_circles, 2)
        flat_radii = x[2 * num_circles:]
        return -np.sum(flat_radii)

    # Constraint: circle i is within the unit square with radius r_i
    def constraint_in_square(x):
        flat_centers = x[:2 * num_circles].reshape(num_circles, 2)
        flat_radii = x[2 * num_circles:]
        for i in range(num_circles):
            x_i, y_i = flat_centers[i]
            r_i = flat_radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        return 0

    # Constraint: circle i and j are not overlapping
    def constraint_no_overlap(x):
        flat_centers = x[:2 * num_circles].reshape(num_circles, 2)
        flat_radii = x[2 * num_circles:]
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = flat_centers[i, 0] - flat_centers[j, 0]
                dy = flat_centers[i, 1] - flat_centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < flat_radii[i] + flat_radii[j] - 1e-12:
                    return -1
        return 0

    # Bounds: x and y in [0, 1], radii in [0, 1]
    bounds = []
    for _ in range(num_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
    for _ in range(num_circles):
        bounds.extend([(0.0, 1.0)])

    # Optimization
    res = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=[
        {'type': 'ineq', 'fun': constraint_in_square},
        {'type': 'ineq', 'fun': constraint_no_overlap}
    ])

    # Extract results
    flat_centers = res.x[:2 * num_circles].reshape(num_circles, 2)
    flat_radii = res.x[2 * num_circles:]
    sum_radii = np.sum(flat_radii)

    return flat_centers, flat_radii, sum_radii
