import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize.optimize import LinearConstraint, NonlinearConstraint

def run_packing():
    # Initial hexagonal packing of 25 circles with a central circle
    # Generate grid points for hexagonal packing
    n = 25
    centers = np.zeros((n, 2))
    r = 0.05  # Initial uniform radius

    # Hexagonal grid with 5 rows: 5, 5, 5, 5, 5 circles
    for i in range(5):
        for j in range(5):
            x = j + 0.5 * (i % 2)
            y = i + 0.5 * i
            centers[i * 5 + j] = [x, y]

    # Add center circle
    centers = np.vstack((centers, [0.5, 0.5]))
    # Initial radii
    radii = np.full(n + 1, r)

    def objective(radii):
        return -np.sum(radii)

    def constraint_func(radii):
        # Ensure all circles fit within the square
        for i in range(n + 1):
            if radii[i] < 0:
                return False
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return False
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
        # Ensure no overlaps
        for i in range(n + 1):
            for j in range(i + 1, n + 1):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Define bounds for each radius
    bounds = [(0, 1) for _ in range(n + 1)]

    # Initial guess for radii
    initial_radii = np.full(n + 1, r)

    # Create a constraint function for minimization
    def constraint(x):
        return constraint_func(x)

    # Use SLSQP for constrained optimization
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints={"type": "constraint", "fun": constraint},
        method="SLSQP"
    )

    # Extract the result
    radii = result.x
    centers = np.array(centers)

    # Validate packing
    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(msg)

    return (centers, radii, np.sum(radii))

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
