import numpy as np
from scipy.optimize import minimize
import random

def run_packing():
    # Initial guess: 4 corner circles, 8 edge circles, and 14 central circles
    np.random.seed(42)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Corner circles
    corners = [(0, 0), (1, 0), (1, 1), (0, 1)]
    for i, (x, y) in enumerate(corners):
        centers[i] = [x + np.random.uniform(-0.05, 0.05), y + np.random.uniform(-0.05, 0.05)]
        radii[i] = np.random.uniform(0.05, 0.1)
    
    # Edge circles (top, right, bottom, left)
    edge_positions = [
        [(0.25, 0.5), (0.5, 0.25), (0.75, 0.5)],  # Left, bottom, right
        [(0.5, 0.75), (0.25, 0.25), (0.75, 0.75)],  # Top, left, right
        [(0.25, 0.75), (0.5, 0.5), (0.75, 0.25)],  # Bottom, top, left
        [(0.5, 0.25), (0.75, 0.5), (0.25, 0.5)],   # Right, bottom, left
    ]
    for i, positions in enumerate(edge_positions):
        for j, (x, y) in enumerate(positions):
            centers[4 + i * 3 + j] = [x + np.random.uniform(-0.05, 0.05), y + np.random.uniform(-0.05, 0.05)]
            radii[4 + i * 3 + j] = np.random.uniform(0.05, 0.1)
    
    # Central circles arranged in a hexagonal grid
    for i in range(12):
        angle = 2 * np.pi * i / 12
        r = 0.2 + 0.05 * i
        x = 0.5 + r * np.cos(angle)
        y = 0.5 + r * np.sin(angle)
        centers[12 + i] = [x + np.random.uniform(-0.05, 0.05), y + np.random.uniform(-0.05, 0.05)]
        radii[12 + i] = np.random.uniform(0.02, 0.06)
    
    # Optimization objective function
    def objective(x):
        return -np.sum(x[12:])

    # Constraints
    def constraint(x):
        centers_flat = x[:26*2].reshape(-1, 2)
        radii_flat = x[26*2:]
        for i in range(26):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt((centers_flat[i][0] - centers_flat[j][0])**2 + (centers_flat[i][1] - centers_flat[j][1])**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -1
        return 0

    # Initial guess
    initial_guess = np.concatenate((centers.flatten(), radii))
    
    # Optimization parameters
    result = minimize(objective, initial_guess, method='SLSQP', bounds=[(0, 1) for _ in range(26*2)] + [(0, 1) for _ in range(26)], constraints={'type': 'ineq', 'fun': constraint})
    
    # Extract final centers and radii
    centers_final = result.x[:26*2].reshape(-1, 2)
    radii_final = result.x[26*2:]
    
    # Post-check validity
    if not constraint(result.x):
        raise ValueError("Optimization result failed to satisfy constraints.")
    
    # Return centers, radii, and sum of radii
    return centers_final, radii_final, np.sum(radii_final)
