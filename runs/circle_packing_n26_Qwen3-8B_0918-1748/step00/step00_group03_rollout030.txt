import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

def run_packing():
    # Initial configuration based on hexagonal packing and grid
    # Step 1: Place 6 central circles (hexagonal)
    center_x = np.array([0.3, 0.5, 0.7, 0.3, 0.5, 0.7])
    center_y = np.array([0.5, 0.5, 0.5, 0.6, 0.7, 0.8])
    radii = np.full(6, 0.1)
    centers = np.column_stack((center_x, center_y))
    
    # Step 2: Place 12 mid-sized circles (hexagonal layers)
    mid_centers = []
    mid_radii = []
    for i in range(3):
        for j in range(4):
            mid_centers.append([0.2 + i*0.2, 0.2 + j*0.2])
            mid_radii.append(0.08)
    for i in range(3):
        for j in range(4):
            mid_centers.append([0.8 - i*0.2, 0.2 + j*0.2])
            mid_radii.append(0.08)
    mid_centers = np.array(mid_centers)
    mid_radii = np.array(mid_radii)
    centers = np.vstack((centers, mid_centers))
    radii = np.concatenate((radii, mid_radii))
    
    # Step 3: Place 10 small corner circles
    corner_centers = np.array([
        [0.1, 0.1], [0.1, 0.9], [0.9, 0.9], [0.9, 0.1],
        [0.2, 0.3], [0.8, 0.3], [0.2, 0.7], [0.8, 0.7],
        [0.4, 0.4], [0.6, 0.6]
    ])
    corner_radii = np.full(10, 0.05)
    centers = np.vstack((centers, corner_centers))
    radii = np.concatenate((radii, corner_radii))
    
    def objective(x, radii):
        # x is a flat array of (center_x, center_y, radius) for each circle
        # Reshape x into (n, 3)
        n = len(radii)
        flat_centers = x[:n*2].reshape(n, 2)
        flat_radii = x[n*2:].reshape(n)
        return -np.sum(flat_radii)
    
    def constraints(x, radii):
        n = len(radii)
        flat_centers = x[:n*2].reshape(n, 2)
        flat_radii = x[n*2:].reshape(n)
        cons = []
        for i in range(n):
            for j in range(i+1, n):
                dist = np.linalg.norm(flat_centers[i] - flat_centers[j])
                cons.append(dist - (flat_radii[i] + flat_radii[j]))
        for i in range(n):
            x_i, y_i = flat_centers[i]
            r_i = flat_radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                cons.append(-1)
        return cons
    
    x0 = np.concatenate((centers.flatten(), radii))
    
    # Optimization parameters
    bounds = []
    for i in range(len(radii)):
        bounds.append((0, 1))
    bounds.extend([(0, 1) for _ in range(len(radii))])
    
    result = minimize(
        fun=objective,
        x0=x0,
        method='SLSQP',
        constraints=[{'type': 'ineq', 'fun': constraints, 'args': (radii,)}],
        bounds=bounds,
        options={'maxiter': 1000, 'ftol': 1e-8, 'disp': False}
    )
    
    x = result.x
    centers = x[:len(radii)*2].reshape(len(radii), 2)
    radii = x[len(radii)*2:]
    return centers, radii, np.sum(radii)
