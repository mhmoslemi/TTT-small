import numpy as np
import random
import math
from scipy.optimize import minimize

def generate_initial_grid(n=26, offset=0.1, max_radius=0.15):
    positions = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col + offset
        y = row + offset
        positions.append([x, y])
    return np.array(positions)

def run_packing():
    n = 26
    centers = generate_initial_grid(n)
    radii = np.full(n, 0.1)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                if dist_sq < (radii[i] + radii[j])**2 - 1e-12:
                    raise ValueError(f"Circles {i} and {j} overlap")
        return True
    
    radii = np.full(n, 0.1)
    try:
        res = minimize(
            objective,
            radii,
            method='SLSQP',
            constraints=[
                {'type': 'ineq', 'fun': lambda r: np.array([r[i] for i in range(n)]) - 0.1},
                {'type': 'ineq', 'fun': lambda r: np.array([sum(1 for i in range(n)) if i == 0 else 0 for i in range(n)])}
            ],
            bounds=[(0, 0.2) for _ in range(n)]
        )
        radii = res.x
        return (centers, radii, np.sum(radii))
    
    except Exception as e:
        raise RuntimeError(f"Optimization failed: {e}")
