import numpy as np
import numpy.random as npr
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial hexagonal grid packing
    num_circles = 26
    radius_estimate = 0.1

    # Generate hexagonal grid points
    # Use a hexagonal packing where centers are spaced by 2r for non-overlapping
    # We will adjust radii later to maximize the sum
    r = radius_estimate
    grid_points = []
    for i in range(num_circles):
        x = i / (num_circles - 1)
        y = 0.5 * (i % 2)
        grid_points.append([x, y])
    centers = np.array(grid_points)

    # Initial radii: all same
    radii = np.full(num_circles, r)

    # Define the minimize function to maximize sum of radii
    # Cost function: -sum(radii) to minimize
    def cost_function(x):
        # x has shape (2*num_circles + num_circles) = (3*num_circles)
        # First num_circles * 2: centers
        # Last num_circles: radii
        centers_flat = x[:2*num_circles].reshape((num_circles, 2))
        radii_flat = x[2*num_circles:].reshape((num_circles,))

        # Compute pairwise distances between centers
        dist_matrix = np.zeros((num_circles, num_circles))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist_matrix[i, j] = np.sqrt(dx**2 + dy**2)
                dist_matrix[j, i] = dist_matrix[i, j]

        # Check non-overlapping constraints
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dist_matrix[i, j] < radii_flat[i] + radii_flat[j] - 1e-8:
                    return np.inf

        # Check boundaries
        for i in range(num_circles):
            x, y = centers_flat[i]
            if x - radii_flat[i] < -1e-8 or x + radii_flat[i] > 1 + 1e-8:
                return np.inf
            if y - radii_flat[i] < -1e-8 or y + radii_flat[i] > 1 + 1e-8:
                return np.inf

        # Return negative of sum of radii (to minimize)
        return -np.sum(radii_flat)

    # Bounds: x and y in [0,1], radii >= 0
    bounds = []
    for _ in range(num_circles):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
    # Add bounds for radii
    for _ in range(num_circles):
        bounds.append((0.0, 1.0))

    # Initial guess: centers as hexagonal grid, radii as initial estimate
    initial_guess = []
    for i in range(num_circles):
        initial_guess.extend([centers[i, 0], centers[i, 1], radii[i]])
    initial_guess = np.array(initial_guess)

    # Options for optimization
    options = {
        'method': 'L-BFGS-B',
        'ftol': 1e-6,
        'disp': False
    }

    # Run optimization
    result = opt.minimize(
        fun=cost_function,
        x0=initial_guess,
        bounds=Bounds(bounds),
        method='L-BFGS-B',
        options=options
    )

    # Extract final centers and radii
    x_opt = result.x
    centers = x_opt[:2*num_circles].reshape((num_circles, 2))
    radii = x_opt[2*num_circles:].reshape((num_circles,))

    # Final check
    success, message = validate_packing(centers, radii)
    if not success:
        raise ValueError(message)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
