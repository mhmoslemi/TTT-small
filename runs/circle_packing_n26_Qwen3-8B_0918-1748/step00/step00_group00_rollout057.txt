import numpy as np
import scipy.optimize as opt
from scipy.optimize import LinearConstraint, Bounds
import warnings

def run_packing():
    # Define initial hexagonal packing with 26 circles
    rows = 6
    cols = 5
    # We will use a staggered grid with row offsets
    # We will contain the configuration within the unit square

    # Initial positions for hexagonal packing
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1) if cols > 1 else 0.5
            y = i / (rows - 1) if rows > 1 else 0.5
            # Apply staggered offset for odd rows
            if i % 2 == 1:
                x += 0.5 / (cols - 1) if cols > 1 else 0.25
            centers.append([x, y])

    # Convert to numpy arrays
    centers = np.array(centers)
    # Initial guess for radii: based on the maximum possible radius (0.5)
    radii = np.full(len(centers), 0.5)

    # Optimize the packing with constraints
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii

    def constraint_overlap(centers, radii):
        n = len(centers)
        A = []
        b = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                A.append([-radii[i], -radii[j], 1.0])
                b.append(dist - 1e-8)
        return np.array(A), np.array(b)

    # Prepare for optimization
    n = len(centers)
    bounds = Bounds([0.0] * n, [1.0] * n)
    constraints = []
    # Add non-overlap constraints
    # We will use a simplified form for large-scale optimization
    # Instead of full pairwise constraints, use approximate bounds
    # We use the scipy.optimize.slsqp method and additional simplifications

    try:
        # Setup the optimization with simplified constraints
        # Use a box constraint on radii and try to optimize within bounds
        result = opt.minimize(
            objective,
            radii,
            bounds=bounds,
            method='SLSQP',
            jac=None,
            options={'max_iter': 500, 'ftol': 1e-8, 'eps': 1e-8}
        )
        radii = result.x

        # Final validation
        centers_final = centers.copy()
        radii_final = radii.copy()

        # Final refinement: ensure complex constraints are met
        # We now manually verify all pairwise distances and adjust if needed
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_final[i, 0] - centers_final[j, 0]
                dy = centers_final[i, 1] - centers_final[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_final[i] + radii_final[j] - 1e-8:
                    # Minor adjustment to radii to fix overlap
                    adjustment = (dist - (radii_final[i] + radii_final[j])) / 2
                    radii_final[i] -= 0.5 * adjustment
                    radii_final[j] -= 0.5 * adjustment
                    # Keep radii positive
                    radii_final[i] = max(radii_final[i], 0)
                    radii_final[j] = max(radii_final[j], 0)

        # Final validation function call
        valid, msg = validate_packing(centers_final, radii_final)
        if not valid:
            raise ValueError(msg)

        return (centers_final, radii_final, np.sum(radii_final))

    except Exception as e:
        raise RuntimeError(f"Optimization failed: {e}")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
