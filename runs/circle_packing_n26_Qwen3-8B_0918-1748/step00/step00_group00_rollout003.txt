import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():
    # Define initial guess for radii
    r0 = np.full(26, 0.05)
    centers = np.random.rand(26, 2)
    
    # Optimize radii and positions
    def objective(x):
        return -np.sum(x[::2])  # Maximize sum of radii
    
    def constraint_overlap(x):
        centers = x[1:].reshape(-1, 2)
        radii = x[::2]
        dists = squareform(pdist(centers))
        for i in range(25):
            for j in range(i + 1, 26):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0
    
    def constraint_boundary(x):
        centers = x[1:].reshape(-1, 2)
        radii = x[::2]
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return 1
        return 0
    
    bounds = [(0, 0.2) for _ in range(52)]
    constraints = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]
    
    result = minimize(
        objective,
        np.concatenate((r0, centers.flatten())),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-8, 'disp': False}
    )
    
    x = result.x
    radii = x[::2]
    centers = x[1:].reshape(-1, 2)
    
    # Validate packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")
    
    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    
    return True, "ok"
