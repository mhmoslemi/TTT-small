    def objective(x):
        # x is a flat array: first n_circles*2 are centers, then n_circles are radii

        radii_flat = x[n_circles*2:]
        return -np.sum(radii_flat)

    def non_overlap_constraints(x):
        centers_flat = x[:n_circles*2].reshape(n_circles, 2)
        radii_flat = x[n_circles*2:]
        constraints = []
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                dist = np.sqrt(np.sum((centers_flat[i] - centers_flat[j])**2))
                constraints.append(dist >= radii_flat[i] + radii_flat[j] - 1e-8)
        return constraints

    def boundary_constraints(x):
        centers_flat = x[:n_circles*2].reshape(n_circles, 2)
        radii_flat = x[n_circles*2:]
        constraints = []

            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            constraints.append(x_i - r_i >= -1e-8)
            constraints.append(x_i + r_i <= 1 + 1e-8)
            constraints.append(y_i - r_i >= -1e-8)
            constraints.append(y_i + r_i <= 1 + 1e-8)

    x0 = np.concatenate([centers.flatten(), radii])

    cons = non_overlap_constraints(x0) + boundary_constraints(x0)

        bounds=bounds,
        constraints=cons,

    )
