import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define the positions and radii for the 26 circles
    # The layout is a hybrid of hexagonal packing and corner-based packing
    # Divided into 12 corner circles and 14 gap-fill circles for optimal use

    # Initial positions for the 12 corner circles (placed at corners and edges)
    corner_circles = [
        (0.1, 0.1), (0.9, 0.1), (0.9, 0.9), (0.1, 0.9),
        (0.1, 0.5), (0.5, 0.1), (0.5, 0.9), (0.9, 0.5),
        (0.25, 0.25), (0.75, 0.25), (0.25, 0.75), (0.75, 0.75)
    ]
    corner_centers = np.array(corner_circles)

    # Define the initial positions for the 14 gap-fill circles
    gap_fill_centers = [
        (0.05, 0.05), (0.95, 0.05), (0.95, 0.95), (0.05, 0.95),
        (0.15, 0.1), (0.85, 0.1), (0.1, 0.85), (0.9, 0.85),
        (0.25, 0.45), (0.75, 0.45), (0.45, 0.25), (0.45, 0.75),
        (0.55, 0.55), (0.5, 0.5)
    ]
    gap_fill_centers = np.array(gap_fill_centers)

    # Combine all centers
    centers = np.vstack((corner_centers, gap_fill_centers))

    # Initial guess for radii: all set to 0.05
    radii = np.full(centers.shape[0], 0.05)

    # Define the optimization problem
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii by minimizing negative sum

    def constraint_overlap(i, j, radii):
        x1, y1 = centers[i]
        x2, y2 = centers[j]
        dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        return dist - (radii[i] + radii[j]) >= 0

    def constraint_boundary(i, radii):
        x, y = centers[i]
        return x - radii[i] >= 0 and x + radii[i] <= 1 and y - radii[i] >= 0 and y + radii[i] <= 1

    # Prepare constraints
    constraints = []
    for i in range(centers.shape[0]):
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: constraint_boundary(i, r)})
    for i in range(centers.shape[0]):
        for j in range(i + 1, centers.shape[0]):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_overlap(i, j, r)})

    # Run the optimization
    result = minimize(objective, radii, method='SLSQP', constraints=constraints)

    # Extract the result
    final_radii = result.x
    final_centers = centers

    # Validate the packing
    # This is just to confirm; it's not part of the return
    # validate_packing(final_centers, final_radii)

    return final_centers, final_radii, np.sum(final_radii)
