import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():
    # Initial corner and edge circle placements
    corner_radii = [0.24, 0.24, 0.24, 0.24]  # Approximate max radius for corner circles
    edge_radii = [0.12] * 4  # Edge circles between corners

    # Place corner circles at (r, r), (1 - r, r), (1 - r, 1 - r), (r, 1 - r)
    corner_centers = np.array([
        [corner_radii[0], corner_radii[0]],
        [1 - corner_radii[1], corner_radii[2]],
        [1 - corner_radii[3], 1 - corner_radii[3]],
        [corner_radii[4], 1 - corner_radii[4]]
    ])

    # Place edge circles along the sides
    edge_centers = np.array([
        [corner_radii[0] + edge_radii[0], 0.5],
        [0.5, corner_radii[0] + edge_radii[1]],
        [1 - corner_radii[1] - edge_radii[2], 0.5],
        [0.5, 1 - corner_radii[2] - edge_radii[3]]
    ])

    # Initialize inner circles with hexagonal packing
    num_inner = 26 - len(corner_radii) - len(edge_radii)
    inner_radius = 0.05
    inner_positions = np.zeros((num_inner, 2))
    for i in range(num_inner):
        inner_positions[i] = [0.5 + (i % 5 - 2.5) * inner_radius, 0.5 + int(i / 5) * inner_radius]

    # Combine all circles
    centers = np.vstack([corner_centers, edge_centers, inner_positions])
    radii = np.concatenate([corner_radii, edge_radii, [inner_radius] * num_inner])

    # Define objective to maximize total radii
    def objective(x):
        return -np.sum(x)  # Minimize negative total for maximization

    # Define constraints for non-overlapping and boundary
    def constraint(x):
        dists = squareform(pdist(centers))
        overlaps = np.sum((dists < x[:len(x)//2] + x[len(x)//2:] - 1e-12))
        return overlaps

    # Get initial radii estimate
    initial_radii = np.copy(radii)

    # Setup bounds for radii
    bounds = [(0, 1)] * len(radii)

    # Optimization
    result = minimize(objective, initial_radii, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraint})
    radii = result.x

    # Recheck centers to ensure they are within bounds
    centers = np.vstack([corner_centers, edge_centers, inner_positions])
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        if x < -1e-12 or x > 1 + 1e-12 or y < -1e-12 or y > 1 + 1e-12:
            centers[i] = [x, y]

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
