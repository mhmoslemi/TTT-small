import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import math

def run_packing():
    # Initial guess for number of circles per row and column
    rows = 6
    cols = 4
    num_circles = rows * cols

    # Initial spacing between centers
    d = 0.25

    # Generate initial grid points
    x = np.linspace(0, 1, cols + 1)
    y = np.linspace(0, 1, rows + 1)
    grid = np.array([[xi, yi] for xi in x for yi in y])

    # Remove the last point to stay within the unit square
    grid = grid[:-1]

    # Initial radii
    radii = np.full(len(grid), d / 2)

    # Objective function to maximize sum of radii
    def objective(radii_params):
        return -np.sum(radii_params)

    # Constraint functions
    def constraint1(radii_params):
        return np.sum(radii_params)

    def constraint2(radii_params):
        return np.sum(radii_params) - 2.636

    # Non-overlapping and boundary constraints
    def non_overlapping(radii_params):
        n = len(grid)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((grid[i] - grid[j]) ** 2))
                if dist < radii_params[i] + radii_params[j] - 1e-8:
                    return -np.inf
        return np.sum(radii_params)

    def boundary(radii_params):
        for i in range(n):
            x, y = grid[i]
            r = radii_params[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return -np.inf
        return np.sum(radii_params)

    # Initial guess
    initial_guess = radii.copy()

    # Optimization settings
    bounds = Bounds(0, 1.0)
    constraints = [
        {'type': 'ineq', 'fun': non_overlapping},
        {'type': 'ineq', 'fun': boundary}
    ]

    # Run optimization
    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints)

    centers = grid
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
