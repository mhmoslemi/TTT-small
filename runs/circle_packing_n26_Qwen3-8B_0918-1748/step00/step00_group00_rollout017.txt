import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from itertools import product

def run_packing():
    # Initial configuration for 25 circles in a 5x5 grid
    n = 25
    grid_size = 5
    spacing = 2 / np.sqrt(5)
    centers = np.array([[i * spacing, j * spacing] for i, j in product(range(grid_size), range(grid_size))])
    radii = np.full(n, spacing / 2)
    
    # Add the 26th circle in the largest gap
    def objective_function(position_and_radius):
        x, y, r = position_and_radius
        return -r  # Maximize r by minimizing -r
    
    def constraint_center_inside(position_and_radius):
        x, y, r = position_and_radius
        return [x - r >= -1e-12, x + r <= 1 + 1e-12, y - r >= -1e-12, y + r <= 1 + 1e-12]
    
    def constraint_no_overlap(position_and_radius, other_centers, other_radii):
        x, y, r = position_and_radius
        constraints = []
        for i in range(len(other_centers)):
            dx = x - other_centers[i][0]
            dy = y - other_centers[i][1]
            dist = np.sqrt(dx**2 + dy**2)
            constraints.append(dist >= other_radii[i] + r - 1e-12)
        return constraints
    
    best_sum = 0
    best_centers = centers
    best_radii = radii
    
    # Try placing the 26th circle in different positions
    for i in range(n):
        # Compute gaps between circles to find potential positions
        # This is a simplified heuristic
        x, y = centers[i]
        r = radii[i]
        # Try placing the new circle in the middle of the gap
        new_center = np.array([x, y])
        # Use Simulated Annealing to find optimal new radius and position
        def objective(position_and_radius):
            x_new, y_new, r_new = position_and_radius
            return -r_new
        
        def constraints(position_and_radius):
            x_new, y_new, r_new = position_and_radius
            constraints = []
            constraints.append([x_new - r_new >= -1e-12, x_new + r_new <= 1 + 1e-12, y_new - r_new >= -1e-12, y_new + r_new <= 1 + 1e-12])
            for j in range(n):
                dx = x_new - centers[j][0]
                dy = y_new - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist >= radii[j] + r_new - 1e-12)
            return constraints
        
        bounds = [(0, 1), (0, 1), (0, 1)]
        
        result = minimize(objective, x0=[x, y, r], bounds=bounds, constraints=[{'type': 'ineq', 'fun': lambda x: x[0] - x[2], 'jac': lambda x: (1, 0, -1)}, 
                                                                             {'type': 'ineq', 'fun': lambda x: x[0] + x[2] - 1, 'jac': lambda x: (1, 0, 1)},
                                                                             {'type': 'ineq', 'fun': lambda x: x[1] - x[2], 'jac': lambda x: (0, 1, -1)},
                                                                             {'type': 'ineq', 'fun': lambda x: x[1] + x[2] - 1, 'jac': lambda x: (0, 1, 1)}])
        
        if result.success:
            new_center = np.array([result.x[0], result.x[1]])
            new_radius = result.x[2]
            temp_centers = np.vstack((centers, new_center))
            temp_radii = np.concatenate((radii, [new_radius]))
            sum_radii = np.sum(temp_radii)
            if sum_radii > best_sum:
                best_sum = sum_radii
                best_centers = temp_centers
                best_radii = temp_radii
    
    return best_centers, best_radii, best_sum
