import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds

def run_packing():
    # Define the number of circles and initial positions
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Initial guess: hybrid layout (hexagonal, edge, and corner circles)
    # place hexagonal lattice in the middle
    n_hex = 16
    r_initial = 0.1
    radii[:n_hex] = r_initial
    r_hex = r_initial
    
    # hexagonal packing: positions
    for i in range(n_hex):
        row = i // 4
        col = i % 4
        x = 0.2 + col * (1 - 0.4) / 3
        y = 0.2 + row * (1 - 0.4) / 3 + (i % 2) * r_hex
        centers[i] = [x, y]
    
    # Edge and corner circles: small ones placing along the edges
    # 8 circles along the perimeter, at 0.25, 0.5, 0.75
    # 4 circles on the top, left, bottom, right
    for i in range(4):
        if i == 0:
            # top edge: (0.25, 0.75) to (0.75, 0.75)
            x = 0.25 + i * 0.5
            y = 0.95
        elif i == 1:
            # right edge: (0.95, 0.25) to (0.95, 0.75)
            x = 0.95
            y = 0.25 + i * 0.5
        elif i == 2:
            # bottom edge: (0.25, 0.05) to (0.75, 0.05)
            x = 0.25 + i * 0.5
            y = 0.05
        elif i == 3:
            # left edge: (0.05, 0.25) to (0.05, 0.75)
            x = 0.05
            y = 0.25 + i * 0.5
        # Also add 4 corner circles
        if i == 0:
            x, y = 0.05, 0.95
        elif i == 1:
            x, y = 0.95, 0.95
        elif i == 2:
            x, y = 0.95, 0.05
        elif i == 3:
            x, y = 0.05, 0.05
        centers[n_hex + i] = [x, y]
        radii[n_hex + i] = 0.05
    # Center of the square
    centers[n_hex + 4] = [0.5, 0.5]
    radii[n_hex + 4] = 0.05
    
    # Define bounds for centers and radii
    bounds_centers = [(0, 1) for _ in range(2 * num_circles)]
    bounds_radii = [(0, 1) for _ in range(num_circles)]
    
    # Define the objective function to maximize sum of radii
    def objective(x):
        # x is a flat list: first 2n are centers, then n are radii
        centers_flat = x[:2 * num_circles]
        radii_flat = x[2 * num_circles:]
        centers = centers_flat.reshape((num_circles, 2))
        radii = radii_flat
        return -np.sum(radii)
    
    # Define constraints
    def constraints(x):
        centers_flat = x[:2 * num_circles]
        radii_flat = x[2 * num_circles:]
        centers = centers_flat.reshape((num_circles, 2))
        radii = radii_flat
        
        # Ensure circles are within the square
        for i in range(num_circles):
            x_center, y_center = centers[i]
            r = radii[i]
            if x_center - r < -1e-12 or x_center + r > 1 + 1e-12 or y_center - r < -1e-12 or y_center + r > 1 + 1e-12:
                return np.inf
        
        # Ensure circles are not overlapping
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        
        return 0
    
    # Combine bounds and constraints
    bounds = Bounds(bounds_centers + bounds_radii)
    
    # Optimization: minimize -sum_radii with constraints
    result = minimize(
        objective,
        np.concatenate([centers.flatten(), radii]),
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        method='SLSQP'
    )
    
    centers_opt = result.x[:2 * num_circles].reshape((num_circles, 2))
    radii_opt = result.x[2 * num_circles:]
    
    # Validate the packing to check for errors or overlaps
    success, message = validate_packing(centers_opt, radii_opt)
    
    if not success:
        raise RuntimeError(f"Packing validation failed: {message}")
    
    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    
    return True, "ok"
