import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
import math

def run_packing():
    # Define initial guesses for centers and radii
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Base layer: 4x4 grid of circles
    base_n = 16
    for i in range(base_n):
        row = i // 4
        col = i % 4
        x = col * 0.25 + 0.125
        y = row * 0.25 + 0.125
        centers[i] = [x, y]
    
    # Top layer: 10 circles placed near corners and edges
    for i in range(base_n, 26):
        if i - base_n == 0:
            # Top-left corner
            centers[i] = [0.1, 0.9]
        elif i - base_n == 1:
            # Top-right corner
            centers[i] = [0.9, 0.9]
        elif i - base_n == 2:
            # Bottom-left corner
            centers[i] = [0.1, 0.1]
        elif i - base_n == 3:
            # Bottom-right corner
            centers[i] = [0.9, 0.1]
        elif i - base_n == 4:
            # Left edge
            centers[i] = [0.1, 0.5]
        elif i - base_n == 5:
            # Right edge
            centers[i] = [0.9, 0.5]
        elif i - base_n == 6:
            # Top edge
            centers[i] = [0.5, 0.9]
        elif i - base_n == 7:
            # Bottom edge
            centers[i] = [0.5, 0.1]
        elif i - base_n == 8:
            # Mid-left
            centers[i] = [0.2, 0.5]
        elif i - base_n == 9:
            # Mid-right
            centers[i] = [0.8, 0.5]

    # Initial radii guess based on spacing from grid
    for i in range(base_n):
        radii[i] = 0.12
    for i in range(base_n, 26):
        radii[i] = 0.1

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints
    constraints = []

    # Boundary constraint: each circle must be within the unit square
    def boundary_constraint(radii):
        return np.array([
            centers[i, 0] - radii[i] >= 0,
            centers[i, 0] + radii[i] <= 1,
            centers[i, 1] - radii[i] >= 0,
            centers[i, 1] + radii[i] <= 1
        for i in range(26)
        ]).flatten()

    constraints.append({'type': 'ineq', 'fun': boundary_constraint})

    # Non-overlap constraint: distance between centers >= sum of radii
    def non_overlap_constraint(radii):
        sum_distances = 0
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                sum_distances += dist - (radii[i] + radii[j])
        return sum_distances

    constraints.append({'type': 'ineq', 'fun': non_overlap_constraint})

    # Optimization bounds
    bounds = Bounds(np.zeros(26), np.inf)

    # Run optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'disp': False, 'maxiter': 1000}
    )

    # Extract final centers and radii
    centers = np.array(centers)
    radii = result.x

    # Compute and return sum of radii
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
