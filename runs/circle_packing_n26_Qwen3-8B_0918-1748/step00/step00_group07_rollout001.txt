import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from collections import namedtuple

Circle = namedtuple('Circle', ['x', 'y', 'r'])

def run_packing():
    # Initial hexagonal packing of 26 circles in a unit square
    n = 26
    centers = []
    radii = []
    
    # Generate initial hexagonal grid with varying rows
    row_height = 1.0 / np.sqrt(3)
    cols = 6
    rows = 5
    
    for r in range(rows):
        for c in range(cols):
            x = c / (cols - 1)
            y = r / (rows - 1)
            centers.append([x, y])
    
    # Add more circles if needed for a total of 26
    if len(centers) < n:
        for i in range(len(centers), n):
            centers.append([0.5, 0.5])
    
    # Initial uniform radius prediction
    total_area = n * np.pi * (0.5)**2
    available_area = 1.0
    initial_radius = np.sqrt(total_area / (np.pi * n)) * 0.85
    
    # Adjust the radii based on initial spacing
    radii = [initial_radius] * n
    
    # Define the objective function for optimization
    def objective(x):
        # x is a flat array of parameters: [x0, y0, r0, x1, y1, r1, ...]
        centers_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        sum_radii = sum(radii_flat)
        
        # Check boundary constraints
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1:
                return -np.inf
            if y_i - r_i < 0 or y_i + r_i > 1:
                return -np.inf
        
        # Check overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -np.inf
        
        # Return sum of radii as the objective to maximize
        return sum_radii
    
    # Initial guess: centers in hexagonal pattern, radii as initial_radius
    initial_guess = []
    for i in range(n):
        initial_guess.extend(centers[i])
        initial_guess.append(initial_radius)
    
    # Perform optimization
    result = minimize(lambda x: -objective(x), initial_guess, method='L-BFGS-B', options={'maxiter': 1000, 'ftol': 1e-12})
    
    # Extract results
    x_opt = result.x
    centers_opt = x_opt[:2*n].reshape(n, 2)
    radii_opt = x_opt[2*n:]
    sum_radii = objective(x_opt)
    
    # Validate the result
    if np.isnan(sum_radii):
        raise ValueError("Invalid packing configuration")
    
    return (centers_opt, radii_opt, sum_radii)
