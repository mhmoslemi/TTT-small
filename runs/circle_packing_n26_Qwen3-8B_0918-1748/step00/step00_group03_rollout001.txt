import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26
    
    # Initial grid spacing
    grid_spacing = 0.15
    
    # Generate initial positions based on hexagonal grid
    centers = []
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * grid_spacing
        y = row * grid_spacing + (col % 2) * grid_spacing / 2
        centers.append([x, y])
    centers = np.array(centers)
    
    # Initial guess for radii as uniform
    radii = np.full(n, 0.05)
    
    # Optimization variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)
    
    # Constraints
    constraints = []
    for i in range(n):
        # Boundary constraints
        constraints += [
            x[i, 0] - r[i] >= -1e-12,
            x[i, 0] + r[i] <= 1 + 1e-12,
            x[i, 1] - r[i] >= -1e-12,
            x[i, 1] + r[i] <= 1 + 1e-12
        ]
    
    for i in range(n):
        for j in range(i + 1, n):
            # Non-overlap constraint
            constraints += [
                cp.norm(x[i] - x[j]) >= r[i] + r[j] - 1e-12
            ]
    
    # Objective: Maximize the sum of radii
    objective = cp.Maximize(cp.sum(r))
    
    # Solve the problem
    problem = cp.Problem(objective, constraints)
    problem.solve()
    
    # Extract the solution
    centers = x.value
    radii = r.value
    
    # Validate the solution to ensure it meets all constraints
    validation_result, message = validate_packing(centers, radii)
    if not validation_result:
        raise ValueError(f"Packing validation failed: {message}")
    
    # Return the result
    return (centers, radii, np.sum(radii))

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
