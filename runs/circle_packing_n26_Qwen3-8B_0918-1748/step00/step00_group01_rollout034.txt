import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint

def run_packing():
    # Initial configuration: hexagonal packing with 5 rows
    num_circles = 26
    rows = 5
    cols = 6  # 5 rows of 5 circles, plus 1 extra in the last row
    radius_initial = 0.1
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Generate initial grid-like positions
    for i in range(rows):
        for j in range(cols):
            x = j * (1 + 0.1)
            y = i * (1 + 0.1)
            if x > 1:
                x = 1 - (x - 1) * 0.5
            if y > 1:
                y = 1 - (y - 1) * 0.5
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = radius_initial

    def objective(radii):
        return -sum(radii)

    def constraint_overlap(x, y, r):
        dist = np.sqrt((x - y)**2)
        return dist - (r[0] + r[1])

    bounds = Bounds(lb=np.zeros(num_circles), ub=np.full(num_circles, 1.0))

    cons = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            cons.append(NonlinearConstraint(lambda x: constraint_overlap(x[i, 0], x[j, 0], x[i, 1], x[j, 1]), -np.inf, 0))

    x0 = centers.copy()
    res = optimize.minimize(
        objective,
        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-6, 'maxiter': 1000}
    )

    centers = res.x[:, :2]
    radii = res.fun * np.ones(num_circles)

    # Post-optimization refinement: adjust radii to prevent boundary overlaps
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        dx = min(x - r, 1.0 - r)
        dy = min(y - r, 1.0 - r)
        if dx < 1e-6 or dy < 1e-6:
            radii[i] = min(r, 1.0 - max(x, y))

    # Final validation
    if not validate_packing(centers, radii)[0]:
        raise ValueError("Optimization result is invalid - check constraints")

    return centers, radii, sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
