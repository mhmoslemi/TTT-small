import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist
import time

def run_packing():
    np.random.seed(42)
    n = 26
    centers = np.random.rand(n, 2) * 0.9 + 0.05
    radii = np.random.rand(n) * 0.1 + 0.01

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(n):
            constraints.append(radii[i] >= 0.0)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append(dist >= radii[i] + radii[j] - 1e-8)
        for i in range(n):
            x, y = centers[i]
            constraints.append(radii[i] >= (x - 1e-8) if x - 1e-8 > 0 else 0.0)
            constraints.append(radii[i] >= (1.0 - 1e-8 - x) if 1.0 - 1e-8 - x > 0 else 0.0)
            constraints.append(radii[i] >= (y - 1e-8) if y - 1e-8 > 0 else 0.0)
            constraints.append(radii[i] >= (1.0 - 1e-8 - y) if 1.0 - 1e-8 - y > 0 else 0.0)
        return constraints

    def optimize():
        bounds = Bounds([0.0] * n, [1.0] * n)
        x0 = radii.copy()
        res = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints=(lambda x: np.array(constraints(x))),
            method='SLSQP',
            options={'ftol': 1e-8, 'maxiter': 1000, 'disp': False}
        )
        return res.x

    radii = optimize()
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
