import numpy as np
from scipy.optimize import minimize, Bounds
from itertools import combinations

def run_packing():
    # Define a helper function to calculate distance between two points
    def dist(p1, p2):
        return np.linalg.norm(p1 - p2)

    # Define a helper function to calculate the constraint for non-overlapping circles
    def non_overlap_constraint(p, r, i, j):
        d = dist(p[i], p[j])
        return d - (r[i] + r[j])

    # Define a helper function to calculate the constraint for boundary
    def boundary_constraint(p, r, i):
        x, y = p[i]
        return (x - r[i]) - 0.0  # Ensure circle stays within [0,1]^2

    # Initialize core hexagonal grid positions
    def hexagonal_grid(n, dx, dy, offset_x, offset_y):
        grid = []
        for row in range(1, n + 1):
            for col in range(n - row + 1):
                x = offset_x + col * dx
                y = offset_y + (row - 1) * dy if row % 2 == 1 else offset_y + row * dy
                grid.append([x, y])
        return np.array(grid)

    # Initial positions of 16 circles in hexagonal grid
    n_rows = 4
    n_cols = 4
    dx = 0.3
    dy = np.sqrt(3) / 2 * dx
    offset_x = 0.5
    offset_y = 0.5
    core_centers = hexagonal_grid(16, dx, dy, offset_x, offset_y)
    core_radii = np.ones(16) * 0.05  # Initial small radius

    # Function to optimize core radii
    def objective_core(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints_core(radii):
        constraints = []
        for i in range(len(core_centers)):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: boundary_constraint(core_centers[i], r, i)})
        for i, j in combinations(range(len(core_centers)), 2):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: non_overlap_constraint(core_centers, r, i, j)})
        return constraints

    bounds_core = Bounds([0.0] * len(core_radii), [np.inf] * len(core_radii))

    result_core = minimize(objective_core, core_radii, method='SLSQP', bounds=bounds_core, constraints=constraints_core)
    core_radii = result_core.x
    core_centers = core_centers

    # Add 10 edge/corner circles in unused gaps
    # Generate random candidate positions
    edge_centers = []
    for _ in range(10):
        x = np.random.uniform(0.1, 0.9)
        y = np.random.uniform(0.1, 0.9)
        edge_centers.append([x, y])

    edge_radii = np.ones(10) * 0.02  # Smaller radii to fit in gaps

    # Function to optimize edge radii
    def objective_edge(radii):
        return -np.sum(radii)

    def constraints_edge(radii):
        constraints = []
        for i in range(len(edge_centers)):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i: boundary_constraint(edge_centers[i], r, i)})
        for i, j in combinations(range(len(edge_centers)), 2):
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: non_overlap_constraint(edge_centers, r, i, j)})
        for i in range(len(edge_centers)):
            for j in range(len(core_centers)):
                constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: non_overlap_constraint(np.append(edge_centers, core_centers), np.append(r, core_radii), i, j)})
        return constraints

    bounds_edge = Bounds([0.0] * len(edge_radii), [np.inf] * len(edge_radii))

    result_edge = minimize(objective_edge, edge_radii, method='SLSQP', bounds=bounds_edge, constraints=constraints_edge)
    edge_radii = result_edge.x
    edge_centers = np.array(edge_centers)

    # Combine the centers and radii
    all_centers = np.vstack((core_centers, edge_centers))
    all_radii = np.concatenate((core_radii, edge_radii))
    sum_radii = np.sum(all_radii)

    return all_centers, all_radii, sum_radii
