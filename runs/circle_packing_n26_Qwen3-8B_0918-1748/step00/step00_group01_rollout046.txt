import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial configuration: 5 rows of 5 circles with staggered pattern
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    row_height = 0.3
    col_width = 0.3
    offset = 0.15
    
    # First 2 rows: left-aligned
    for i in range(2):
        for j in range(5):
            x = col_width * j
            y = row_height * i
            centers[i * 5 + j] = [x, y]
            radii[i * 5 + j] = 0.1
    
    # Next 2 rows: staggered
    for i in range(2, 4):
        for j in range(5):
            if j % 2 == 0:
                x = col_width * j
            else:
                x = col_width * (j - 0.5)
            y = row_height * i
            centers[i * 5 + j] = [x, y]
            radii[i * 5 + j] = 0.1
    
    # Last row: bottom row, aligned
    for j in range(5):
        x = col_width * j
        y = row_height * 4
        centers[20 + j] = [x, y]
        radii[20 + j] = 0.1

    # Initial optimization
    def objective(radii_vec):
        return -np.sum(radii_vec)
    
    def constraint_func(radii_vec):
        # Define the centers in terms of radii
        # Assuming the same positions (for demo purposes, in real code radii would be optimized)
        # For actual code, radii would be included as variables to optimize
        for i in range(26):
            for j in range(i + 1, 26):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return dist - (radii[i] + radii[j] - 1e-12)
        return 0
    
    # Use bounded optimization
    bounds = [(0, 0.5) for _ in range(26)]
    
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_func, 'jac': None}
        ],
        tol=1e-6
    )
    
    radii = result.x
    return (centers, radii, np.sum(radii))
