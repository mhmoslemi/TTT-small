import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Define initial packing - divide the square into two regions and use hexagonal packing
    # Region 1: left half, Region 2: right half
    num_circles = 26
    num_per_region = 13

    # Helper to generate hexagonal packing in a given region
    def hexagonal_packing(region_start, region_end, num_circles_in_region):
        x_start, y_start = region_start
        x_end, y_end = region_end
        width = x_end - x_start
        height = y_end - y_start
        base_radius = 0.1
        positions = []
        for i in range(num_circles_in_region):
            row = i // 2
            col = i % 2
            x = x_start + col * (2 * base_radius) + row * (base_radius * np.sqrt(3))
            y = y_start + row * (2 * base_radius)
            if x > x_end - base_radius or y > y_end - base_radius:
                x = x_end - base_radius
                y = y_end - base_radius
            positions.append([x, y])
        return np.array(positions)

    # Generate initial positions
    region1 = [0, 0.5]
    region2 = [0.5, 1]
    region1_positions = hexagonal_packing(region1, region2, num_per_region)
    region2_positions = hexagonal_packing(region2, region2, num_per_region)
    initial_centers = np.vstack((region1_positions, region2_positions))

    # Optimize radii and positions
    def objective(x, initial_centers, num_circles):
        centers = x[:2 * num_circles].reshape((num_circles, 2))
        radii = x[2 * num_circles:]
        return -np.sum(radii)

    def constraints(x, initial_centers, num_circles):
        centers = x[:2 * num_circles].reshape((num_circles, 2))
        radii = x[2 * num_circles:]
        constraints = []
        for i in range(num_circles):
            constraints.append((centers[i, 0] - radii[i]) >= -1e-12)
            constraints.append((centers[i, 0] + radii[i]) <= 1 + 1e-12)
            constraints.append((centers[i, 1] - radii[i]) >= -1e-12)
            constraints.append((centers[i, 1] + radii[i]) <= 1 + 1e-12)
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                constraints.append((dist - radii[i] - radii[j]) >= -1e-12)
        return constraints

    bounds = []
    for _ in range(num_circles):
        bounds.append((0, 1))
        bounds.append((0, 1))
    for _ in range(num_circles):
        bounds.append((0, 1))

    x0 = np.concatenate((initial_centers.flatten(), np.full(num_circles, 0.1)))

    res = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': lambda x: 0}]
    )

    centers = res.x[:2 * num_circles].reshape((num_circles, 2))
    radii = res.x[2 * num_circles:]

    # Final validation
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} out of bounds")

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
