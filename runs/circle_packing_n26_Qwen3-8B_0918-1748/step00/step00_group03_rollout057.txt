import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Define initial positions: 4 corners, 8 edge, 14 central in hexagonal pattern
    # Corner circles
    corners = np.array([[0, 0], [1, 0], [1, 1], [0, 1]])
    # Edge circles (distance from corner)
    edges = np.array([
        [0.25, 0], [0.75, 0], [1, 0.25], [1, 0.75], [0.75, 1], [0.25, 1], [0, 0.75], [0, 0.25]
    ])
    # Central hexagonal packing: 14 circles arranged in a hexagonal pattern
    central = []
    for row in range(4):
        for col in range(4):
            if row % 2 == 0:
                central.append([col * 0.3 + 0.3, row * 0.3 + 0.3])
            else:
                central.append([col * 0.3 + 0.35, row * 0.3 + 0.3])
    central = np.array(central)
    # Combine all positions
    centers = np.vstack((corners, edges, central))
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_sum(radii):
        return np.sum(radii)
    
    # Initial guess: uniform radii based on distance to corners and center
    def initial_radii(centers):
        dist = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
        min_dist = np.min(dist, axis=1)
        return 0.5 / (1 + min_dist)
    
    radii_guess = initial_radii(centers)
    
    # Optimization problem: maximize sum of radii subject to constraints
    def constraint_overlap(radii):
        dist = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
        return dist - (radii[:, np.newaxis] + radii[np.newaxis, :]) + 1e-6
    
    def constraint_boundary(radii):
        dist_to_edge = np.abs(centers - np.array([[0, 0], [1, 0], [1, 1], [0, 1]]))
        return np.min(dist_to_edge, axis=1) - radii + 1e-6
    
    bounds = Bounds([0] * len(radii_guess), [np.inf] * len(radii_guess))
    
    n = len(centers)
    A = np.zeros((n, n))
    A[np.triu_indices(n, 1)] = 1
    
    # Use scipy.optimize to solve the problem
    res = minimize(
        objective,
        radii_guess,
        bounds=bounds,
        constraints=[{
            'type': 'ineq',
            'fun': lambda x: np.min(np.concatenate((constraint_overlap(x), constraint_boundary(x)), axis=0))
        }],
        method='SLSQP'
    )
    
    # Extract final solution
    centers_final = centers
    radii_final = res.x
    sum_radii = np.sum(radii_final)
    
    # Validate solution
    validated, msg = validate_packing(centers_final, radii_final)
    if not validated:
        raise RuntimeError(msg)
    
    return (centers_final, radii_final, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
