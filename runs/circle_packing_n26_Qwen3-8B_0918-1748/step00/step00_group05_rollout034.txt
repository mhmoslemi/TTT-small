import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize.minimize_scalar import Bounds as ScalarBounds
import time

def run_packing():
    # Initial grid-based arrangement with 4x4 grid and 10 edge circles
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # 4x4 grid in the center, spaced with 0.25 units between adjacent centers
    grid_size = 0.25
    grid_x = np.linspace(0.25, 0.75, 4)
    grid_y = np.linspace(0.25, 0.75, 4)
    
    idx = 0
    for x in grid_x:
        for y in grid_y:
            centers[idx] = [x, y]
            radii[idx] = 0.1
            idx += 1
    
    # Add 10 edge circles: 4 on the top, 4 on the bottom, 2 on the left and 2 on the right
    # Top edge (offset from corners)
    centers[idx] = [0.25, 0.9]
    radii[idx] = 0.1
    idx += 1
    centers[idx] = [0.5, 0.9]
    radii[idx] = 0.1
    idx += 1
    centers[idx] = [0.75, 0.9]
    radii[idx] = 0.1
    idx += 1
    # Bottom edge (offset from corners)
    centers[idx] = [0.25, 0.1]
    radii[idx] = 0.1
    idx += 1
    centers[idx] = [0.5, 0.1]
    radii[idx] = 0.1
    idx += 1
    centers[idx] = [0.75, 0.1]
    radii[idx] = 0.1
    idx += 1
    # Left edge (offset from corners)
    centers[idx] = [0.1, 0.25]
    radii[idx] = 0.1
    idx += 1
    centers[idx] = [0.1, 0.5]
    radii[idx] = 0.1
    idx += 1
    # Right edge (offset from corners)
    centers[idx] = [0.9, 0.25]
    radii[idx] = 0.1
    idx += 1
    centers[idx] = [0.9, 0.5]
    radii[idx] = 0.1
    idx += 1
    
    # Objective function to maximize the sum of radii
    def objective(x):
        return -x  # Negative for minimization
    
    # Constraint: distance between circles >= sum of radii
    def constraint_func(x):
        total = 0
        for i in range(26):
            for j in range(i + 1, 26):
                xi = x[2*i]
                yi = x[2*i + 1]
                xj = x[2*j]
                yj = x[2*j + 1]
                r_i = radii[i]
                r_j = radii[j]
                dist = np.sqrt((xi - xj)**2 + (yi - yj)**2)
                total += max(0, (r_i + r_j) - dist)
        return total
    
    # Bound for each circle: x and y in [0,1], radius >= 0
    bounds = []
    for i in range(26):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((0.0, 1.0))  # radius
    
    # Initial guess: positions fixed, radii fixed
    x_initial = []
    for i in range(26):
        x_initial.append(centers[i][0])
        x_initial.append(centers[i][1])
        x_initial.append(radii[i])
    
    # Optimization steps
    res = minimize(objective, x_initial, bounds=bounds, constraints={"type": "ineq", "fun": constraint_func, "jac": None})
    
    # Update centers and radii based on result
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    for i in range(26):
        x = res.x[2*i]
        y = res.x[2*i + 1]
        r = res.x[2*i + 2]
        centers[i] = [x, y]
        radii[i] = r
    
    # Validate the packing
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing invalid: {message}")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
