import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for centers: 3 rows of 8, 8, 10 circles offset in hexagonal fashion
    n = 26
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05  # Initial small radii to start optimization

    # Hexagonal packing: top row, then each row offset
    row_height = 1.1  # slightly more than diameter of small circles for spacing
    col_width = 1.1  # same as row_height
    row_offset = 0.5 * row_height

    # Initial placement of centers
    row = 0
    col = 0
    for i in range(n):
        if i % 8 == 0:
            row += 1
            col = 0
        else:
            col += 1

        if row % 2 == 0:
            x = col * col_width
            y = row * row_height
        else:
            x = (col - 0.5) * col_width
            y = row * row_height

        centers[i] = [x, y]

    # Define optimization variables: center coordinates and radii
    n_vars = n * 2 + n  # 2 per center, 1 per radius
    x0 = np.zeros(n_vars)
    x0[:2*n] = centers.flatten()
    x0[2*n:] = radii

    def objective(x):
        return -np.sum(x[2*n:])  # Maximize sum of radii by minimizing negative

    def bounds(x):
        return Bounds(lb=0.0, ub=1.0, indices=range(2*n)) + Bounds(lb=-np.inf, ub=1.0, indices=range(2*n, n_vars))

    def constraints(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                cons = dist - (radii[i] + radii[j])
                constraints.append(cons)
        return constraints

    cons = [{'type': 'ineq', 'fun': constraints}]

    # Optimization
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 300})

    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    sum_radii = np.sum(radii)

    # Validate and return
    if np.isnan(centers).any() or np.isnan(radii).any():
        raise ValueError("NaN values in optimization result")
    if np.any(radii < 0):
        raise ValueError("Negative radius found in optimization result")
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")
    return (centers, radii, sum_radii)
