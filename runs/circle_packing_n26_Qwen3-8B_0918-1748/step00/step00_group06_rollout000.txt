import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Hexagonal packing of 26 circles
    # We use a 6-row arrangement: [5, 6, 6, 6, 6, 4]
    # Each row is spaced with 2*pi/3 radians angle (hexagonal)
    # Initial offset: x = 0.1, y = 0.1
    initial_centers = []
    for row in range(6):
        if row == 0 or row == 5:
            num_circles = 5
        else:
            num_circles = 6
        row_centers = []
        for col in range(num_circles):
            x = 0.1 + col * (1 - 2 * 0.1) / (num_circles - 1)
            y = 0.1 + row * (1 - 2 * 0.1) / (5)
            row_centers.append([x, y])
        initial_centers.extend(row_centers)
    centers = np.array(initial_centers)
    
    # Optimization: maximize sum of radii subject to constraints
    def objective(radii):
        return -np.sum(radii)
    
    def constraint1(radii):
        return np.sum(radii)
    
    def constraint2(radii):
        for i in range(len(radii)):
            if radii[i] < 0:
                return -1
        return 0
    
    def constraint3(radii):
        for i in range(len(radii)):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return -1
        return 0
    
    def constraint4(radii):
        dist_matrix = cdist(centers, centers)
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0
    
    bounds = [(0, None) for _ in range(len(radii))]
    cons = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2},
        {'type': 'ineq', 'fun': constraint3},
        {'type': 'ineq', 'fun': constraint4}
    ]
    result = minimize(objective, np.ones(len(radii)), bounds=bounds, constraints=cons)
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
