        dist_right = 1 - x
        dist_bottom = y
        dist_top = 1 - y
        dist_boundary = min(dist_left, dist_right, dist_bottom, dist_top)
        return min(layer_radius, dist_boundary)

        radii[i] = estimate_radius(i // 10, x, y)  # Arbitrary layer assignment for initial guess

        return -np.sum(radii)  # Negative because we minimize

    def constraint_non_overlapping(centers, radii):
        n = len(centers)
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                    return False
        return True

        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return False
        return True

    def optimize_packing(centers, radii):
        # Flatten the parameters for optimization
        params = np.concatenate((centers.flatten(), radii))

            centers_flat = params[:2 * n].reshape(n, 2)

            return -np.sum(radii_flat)
