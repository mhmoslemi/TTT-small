import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Define number of circles
    n = 26

    # Initial guess for centers (grid layout)
    centers = np.zeros((n, 2))
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 4 + 0.125
        y = row / 4 + 0.125
        centers[i, :] = [x, y]

    # Initial guess for radii (based on grid spacing)
    radii = np.full(n, 0.1)

    # Define constraints
    def constraints_func(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        constraints = []

        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return False
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return False

        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return False

        return True

    # Objective function to maximize the sum of radii
    def objective(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        return -np.sum(radii_flat)  # Minimize negative sum to maximize sum

    # Bounds for centers (x and y must be in [0, 1])
    bounds = []
    for _ in range(n):
        bounds.extend([(0, 1), (0, 1)])  # x and y for each center
    bounds.extend([(0, 1)] * n)  # radii for each center

    # Optimization
    x0 = np.concatenate([centers.flatten(), radii])

    # Set options for optimization
    options = {'maxiter': 1000, 'disp': False}

    result = minimize(objective, x0, bounds=bounds, constraints={'type': 'soft', 'fun': constraints_func}, options=options)

    # Extract result
    x_result = result.x
    centers_result = x_result[:n*2].reshape(n, 2)
    radii_result = x_result[n*2:]

    # Validate result
    validation_result, message = validate_packing(centers_result, radii_result)
    if not validation_result:
        raise ValueError(f"Validation failed: {message}")

    # Return centers, radii, and sum of radii
    return centers_result, radii_result, np.sum(radii_result)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
