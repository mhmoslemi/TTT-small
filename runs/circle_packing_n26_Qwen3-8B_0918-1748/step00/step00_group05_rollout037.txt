import numpy as np
from scipy.optimize import Bounds, minimize
from scipy.spatial.distance import cdist
import matplotlib.pyplot as plt

def run_packing():
    # Define the number of circles
    n = 26

    # Initial estimate: use hexagonal packing with appropriate spacing
    # These are initial positions for a hexagonal grid
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # First 6 circles form a hexagon at the bottom
    for i in range(6):
        angle = 2 * np.pi / 6 * i
        centers[i] = np.array([0.5 + 0.15 * np.cos(angle), 0.1 + 0.15 * np.sin(angle)])
    # Then next 3 circles on top of the hexagon
    for i in range(6, 9):
        angle = 2 * np.pi / 6 * (i - 6)
        centers[i] = np.array([0.5 + 0.13 * np.cos(angle), 0.35 + 0.13 * np.sin(angle)])
    # Then next 3 circles on the next layer
    for i in range(9, 12):
        angle = 2 * np.pi / 6 * (i - 9)
        centers[i] = np.array([0.5 + 0.11 * np.cos(angle), 0.45 + 0.11 * np.sin(angle)])
    # Then next 3 circles on the next layer
    for i in range(12, 15):
        angle = 2 * np.pi / 6 * (i - 12)
        centers[i] = np.array([0.5 + 0.09 * np.cos(angle), 0.55 + 0.09 * np.sin(angle)])
    # Next 3 circles
    for i in range(15, 18):
        angle = 2 * np.pi / 6 * (i - 15)
        centers[i] = np.array([0.5 + 0.07 * np.cos(angle), 0.65 + 0.07 * np.sin(angle)])
    # Next 2 circles
    for i in range(18, 20):
        angle = 2 * np.pi / 6 * (i - 18)
        centers[i] = np.array([0.5 + 0.05 * np.cos(angle), 0.75 + 0.05 * np.sin(angle)])
    # Last 2 circles
    for i in range(20, 22):
        angle = 2 * np.pi / 6 * (i - 20)
        centers[i] = np.array([0.5 + 0.03 * np.cos(angle), 0.85 + 0.03 * np.sin(angle)])
    # Extra circles
    centers[22] = np.array([0.1, 0.1])
    centers[23] = np.array([0.9, 0.1])
    centers[24] = np.array([0.1, 0.9])
    centers[25] = np.array([0.9, 0.9])

    # Radii are approximately consistent based on the spacing
    for i in range(n):
        # Initial radius for all circles
        radii[i] = 0.05

    # Optimization bounds for centers
    bounds_centers = [(0.0, 1.0) for _ in range(n * 2)]
    # Optimization bounds for radii
    bounds_radii = [(0.0, 1.0) for _ in range(n)]

    # Define the optimization objective function
    def objective(x):
        # x[:2*n] are the centers, x[2*n:] are the radii
        centers_opt = x[:2*n].reshape(n, 2)
        radii_opt = x[2*n:].reshape(n)
        return -np.sum(radii_opt)  # Minimize negative sum to maximize sum

    # Define the constraints
    def constraint(x):
        centers_opt = x[:2*n].reshape(n, 2)
        radii_opt = x[2*n:].reshape(n)
        # Boundary constraints
        for i in range(n):
            x_i, y_i = centers_opt[i]
            r_i = radii_opt[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1  # Violation of boundary constraint
        # Non-overlapping constraints
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j]) ** 2))
                if dist < radii_opt[i] + radii_opt[j]:
                    return -1  # Violation of non-overlapping constraint
        return 0  # All constraints satisfied

    # Initial guess
    x0 = np.array([centers[i][0] for i in range(n)] + [centers[i][1] for i in range(n)] + [radii[i] for i in range(n)])

    # Set up optimization problem
    cons = [{'type': 'ineq', 'fun': constraint}]
    result = minimize(objective, x0, method='SLSQP', bounds=Bounds(bounds_centers + bounds_radii), constraints=cons, tol=1e-8)

    # Extract the result
    result_centers = result.x[:2*n].reshape(n, 2)
    result_radii = result.x[2*n:].reshape(n)
    total_radii = np.sum(result_radii)

    return (result_centers, result_radii, total_radii)
