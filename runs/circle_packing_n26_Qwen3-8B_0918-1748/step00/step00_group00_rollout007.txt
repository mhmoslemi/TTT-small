import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Number of circles
    n = 26

    # Initial grid positions using hexagonal packing
    max_rows = 5
    row_height = 1 - 1e-6
    col_width = 1 - 1e-6

    # Define initial positions in hexagonal grid
    centers = []
    for row in range(max_rows):
        for col in range(2 * (max_rows - row - 1) + 1):
            x = col * (col_width / (2 * max_rows))
            y = row * (row_height / (max_rows - 1)) if row < max_rows - 1 else 0.5
            if row % 2 == 0:
                x = col * (col_width / (2 * max_rows))
            else:
                x = (col + 0.5) * (col_width / (2 * max_rows))
            centers.append([x, y])
    centers = np.array(centers)

    # Add some circles near the corners
    corners = np.array([
        [0.1, 0.1],
        [0.9, 0.1],
        [0.1, 0.9],
        [0.9, 0.9]
    ])
    centers = np.vstack((centers, corners))

    # Initial radius guess (based on spacing between centers)
    dists = pdist(centers)
    min_dist = np.min(dists)
    radii = np.full(n, min_dist / 2.0)

    def objective(radii):
        return -np.sum(radii)

    def constraint_dist(radii):
        dists = pdist(centers)
        return dists - (radii[:, np.newaxis] + radii[np.newaxis, :]) + 1e-6

    def constraint_boundary(radii):
        return np.array([np.clip(centers[i, 0] - radii[i], 0, 1) for i in range(n)]) + \
               np.array([np.clip(centers[i, 1] - radii[i], 0, 1) for i in range(n)]) + \
               np.array([np.clip(centers[i, 0] + radii[i], 0, 1) for i in range(n)]) + \
               np.array([np.clip(centers[i, 1] + radii[i], 0, 1) for i in range(n)])

    bounds = Bounds(0, np.inf)

    result = minimize(
        fun=objective,
        x0=radii,
        method='SLSQP',
        constraints=[
            {'type': 'ineq', 'fun': constraint_dist},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        bounds=bounds
    )

    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
