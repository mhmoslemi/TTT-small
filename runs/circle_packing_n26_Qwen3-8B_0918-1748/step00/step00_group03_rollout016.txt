import numpy as np
import scipy.optimize as optimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial guess for circle positions
    # Two rows of 7 circles, one row of 6 circles, and 12 small corner/additional circles
    def initial_guess():
        centers = []
        radii = []
        
        # First row of 7 circles (left to right)
        for i in range(7):
            x = i / 6.0 + 0.05
            y = 0.2
            centers.append([x, y])
            radii.append(0.1)
        
        # Second row of 7 circles (offset to the right by 0.5*r)
        for i in range(7):
            x = i / 6.0 + 0.05 + 0.15
            y = 0.7
            centers.append([x, y])
            radii.append(0.1)
        
        # Row of 6 circles in the middle
        for i in range(6):
            x = i / 5.0 + 0.2
            y = 0.5
            centers.append([x, y])
            radii.append(0.05)
        
        # Small corner circles
        corner_positions = [
            [0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]
        ]
        for pos in corner_positions:
            centers.append(pos)
            radii.append(0.05)
        
        # Additional small circles in remaining gaps
        for i in range(4):
            x = 0.3 + i * 0.15
            y = 0.4
            centers.append([x, y])
            radii.append(0.05)
        
        # Add more small circles in corners
        for i in range(4):
            x = 0.2 + i * 0
            y = 0.8
            centers.append([x, y])
            radii.append(0.05)
        
        # Convert to numpy arrays
        centers = np.array(centers)
        radii = np.array(radii)
        return centers, radii
    
    # Initial guess
    centers, radii = initial_guess()
    n = len(centers)
    
    # Objective function: maximize the sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum
    
    # Constraints: no overlaps and within the square
    def constraint(x):
        # Update radii and centers
        nonlocal centers, radii
        centers = np.array([list(centers[i]) for i in range(n)])
        radii = x
        
        # Check boundary constraints
        for i in range(n):
            x_pos, y_pos = centers[i]
            r = radii[i]
            if x_pos - r < -1e-12 or x_pos + r > 1 + 1e-12:
                return -1
            if y_pos - r < -1e-12 or y_pos + r > 1 + 1e-12:
                return -1
        
        # Check pairwise distance between centers
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        
        return 0
    
    # Optimization bounds
    bounds = [(0, 0.5) for _ in range(n)]
    
    # Optimization problem
    problem = optimize.OptimizeResult()
    problem.fun = objective
    problem.x = radii
    problem.success = True
    problem.message = "Optimization completed"
    
    # Perform optimization
    result = optimize.minimize(
        objective,
        x0=radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint}
        ],
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    # Extract optimized result
    optimal_radii = result.x
    optimal_centers = centers.copy()
    
    # Validate packing
    success, message = validate_packing(optimal_centers, optimal_radii)
    assert success, f"Packing validation failed: {message}"
    
    sum_radii = np.sum(optimal_radii)
    return (optimal_centers, optimal_radii, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
