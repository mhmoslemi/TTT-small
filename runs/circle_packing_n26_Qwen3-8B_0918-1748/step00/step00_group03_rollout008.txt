import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():

    # We start with a hexagonal packing of circles in the square, then refine

    # Initial grid-based layout with some overlap for optimization
    # This is a heuristic layout that maximizes the number of circles while being flexible for optimization
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    rows = 6
    cols = 5
    spacing = 1.0 / (cols + 1)
    half_spacing = spacing / 2

    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing
            if i % 2 == 0:
                x += spacing / 2
            centers[i * cols + j] = [x, y]

    # Initial radii estimate based on the nearest neighbor distance
    distances = squareform(pdist(centers))
    min_distance = np.min(distances)
    initial_radius = min_distance / (2 * np.sqrt(3))  # for hexagonal packing
    radii = np.full(26, initial_radius)

    # Define the objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Maximize sum by minimizing negative sum

    # Define constraints
    def constraint_center_bounds(x):
        return x

    def constraint_boundary(x):
        # x is a flattened array of [x1, y1, r1, x2, y2, r2, ...]
        # Check that all circles are within the unit square
        for i in range(26):
            idx = i * 3
            x = x[idx]
            y = x[idx + 1]
            r = x[idx + 2]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return -np.inf
        return 0

    def constraint_overlap(x):
        # x is a flattened array of [x1, y1, r1, x2, y2, r2, ...]
        # Check that all circles are non-overlapping
        for i, j in combinations(range(26), 2):
            idx_i = i * 3
            idx_j = j * 3
            x1, y1, r1 = x[idx_i:idx_i + 3]
            x2, y2, r2 = x[idx_j:idx_j + 3]
            dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
            if dist < r1 + r2 - 1e-8:
                return -np.inf
        return 0

    # Flatten initial centers and radii
    x0 = np.concatenate([centers.flatten(), radii])

    # Define bounds: x and y must be within [0, 1], radii must be >= 0
    bounds = []

        bounds.extend([(0, 1), (0, 1), (0, 1)])

    # Set up the optimization problem
    cons = [
        {'type': 'ineq', 'fun': constraint_center_bounds},
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8, options={'maxiter': 1000})

    # Extract centers and radii
    centers_new = result.x[::3].reshape((26, 2))
