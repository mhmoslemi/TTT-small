import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Initial configuration: hexagonal packing with staggered rows
    num_rows = 7
    max_cols = 7
    centers = []
    radii = []

    # Calculate vertical and horizontal spacing for hexagonal packing
    vertical_spacing = 1.0 / (num_rows + 1)
    horizontal_spacing = 1.0 / (max_cols + 1)

    # Place circles in staggered rows
    for row in range(num_rows):
        if row % 2 == 0:
            start_col = 0
        else:
            start_col = 0.5 * horizontal_spacing

        for col in range(max_cols):
            x = start_col + col * horizontal_spacing
            y = row * vertical_spacing + 0.5 * vertical_spacing
            centers.append([x, y])

    # Add additional circles for total 26
    # If we need more than 7 rows, add extra rows
    # To simplify, we'll place extra rows at the top
    for row in range(8, num_rows + 2):
        start_col = 0
        for col in range(2):
            x = start_col + col * horizontal_spacing
            y = row * vertical_spacing + 0.5 * vertical_spacing
            centers.append([x, y])

    # We now have 26 circles arranged in a hexagonal grid
    # Now, we'll perform an optimization to maximize the sum of radii
    # We'll assume all circles have the same radius first
    # Then, we'll optimize radii to maximize their sum

    # Initial estimate of radius based on spacing
    initial_radius = 0.1

    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Define constraints: non-overlapping and within the unit square
    def constraint_overlap(radii):
        distances = squareform(pdist(centers))
        for i, j in combinations(range(len(centers)), 2):
            if distances[i, j] < radii[i] + radii[j] - 1e-8:
                return distances[i, j] - radii[i] - radii[j] + 1e-8
        return 0

    def constraint_boundary(radii):
        for i in range(len(centers)):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return (x - radii[i]) + (x + radii[i] - 1 - 1e-8) + (y - radii[i]) + (y + radii[i] - 1 - 1e-8)
        return 0

    # Initial guess for radii
    initial_guess = [initial_radius] * len(centers)

    # Bounds for radii: non-negative
    bounds = [(0, 1)] * len(centers)

    # Optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP'
    )

    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    # Extract the final centers and radii
    centers = np.array(centers)
    radii = np.array(result.x)

    # Validate the solution
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")

    total_sum = np.sum(radii)
    print(f"Total sum of radii: {total_sum}")

    return (centers, radii, total_sum)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
