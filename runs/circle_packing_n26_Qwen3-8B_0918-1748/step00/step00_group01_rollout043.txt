import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def generate_initial_grid(n_rows, n_cols, square_size=1.0):
    spacing = square_size / (n_cols + 1)
    horizontal_offsets = [i * spacing for i in range(n_cols + 1)]
    vertical_offsets = [i * spacing for i in range(n_rows + 1)]
    centers = []
    for i in range(n_rows):
        for j in range(n_cols):
            x = horizontal_offsets[j] + spacing / 2
            y = vertical_offsets[i] + spacing / 2
            centers.append([x, y])
    return np.array(centers)

def generate_boundary_circles(square_size=1.0):
    # Top left corner
    r1 = 0.2
    center1 = [r1, r1]
    # Bottom right corner
    r2 = 0.2
    center2 = [1 - r2, 1 - r2]
    return np.array([center1, center2]), np.array([r1, r2])

def calculate_total_radius_sum(centers, radii):
    return np.sum(radii)

def optimize_packing(centers_guess, radii_guess, square_size=1.0, max_iter=100, tol=1e-8):
    def objective(x):
        idx = 0
        new_centers = np.zeros((len(centers_guess), 2))
        new_radii = np.zeros(len(radii_guess))
        for i in range(len(centers_guess)):
            new_centers[i] = x[idx:idx+2]
            idx += 2
        new_radii = x[idx:]
        return -calculate_total_radius_sum(new_centers, new_radii)

    def constraints(x):
        idx = 0
        new_centers = np.zeros((len(centers_guess), 2))
        new_radii = np.zeros(len(radii_guess))
        for i in range(len(centers_guess)):
            new_centers[i] = x[idx:idx+2]
            idx += 2
        new_radii = x[idx:]
        for i in range(len(new_radii)):
            x_i, y_i = new_centers[i]
            r_i = new_radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.inf
        for i in range(len(new_radii)):
            for j in range(i + 1, len(new_radii)):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < new_radii[i] + new_radii[j] - 1e-12:
                    return np.inf
        return 0

    bounds = []
    for i in range(len(centers_guess)):
        bounds.extend([(0, 1) for _ in range(2)])
    for i in range(len(radii_guess)):
        bounds.extend([(0, 1) for _ in range(1)])
    
    x0 = np.concatenate([centers_guess.flatten(), radii_guess])
    result = minimize(objective, x0, method='L-BFGS-B', bounds=Bounds(bounds), constraints=constraints, tol=tol, max_iter=max_iter)
    if not result.success:
        raise Exception("Optimization failed")
    x_opt = result.x
    idx = 0
    centers_opt = np.zeros((len(centers_guess), 2))
    for i in range(len(centers_guess)):
        centers_opt[i] = x_opt[idx:idx+2]
        idx += 2
    radii_opt = x_opt[idx:]
    return centers_opt, radii_opt

def run_packing():
    # Step 1: Generate initial grid for 24 circles
    n_rows = 4
    n_cols = 6
    initial_centers = generate_initial_grid(n_rows, n_cols)
    initial_radii = np.ones(len(initial_centers)) * 0.05

    # Step 2: Add boundary circles
    boundary_centers, boundary_radii = generate_boundary_circles()

    # Step 3: Combine initial grid and boundary circles
    combined_centers = np.vstack((initial_centers, boundary_centers))
    combined_radii = np.concatenate((initial_radii, boundary_radii))

    # Step 4: Optimize the combined configuration
    centers_opt, radii_opt = optimize_packing(combined_centers, combined_radii)

    # Step 5: Validate the packing
    is_valid, message = validate_packing(centers_opt, radii_opt)
    if not is_valid:
        raise ValueError(f"Invalid packing: {message}")

    return centers_opt, radii_opt, calculate_total_radius_sum(centers_opt, radii_opt)
