import numpy as np
from scipy.optimize import minimize

def run_packing():
    def objective(s):
        s = np.clip(s, 1e-8, 1 - 1e-8)
        n = 5
        x = np.linspace(0, 1, n + 1)[:-1]
        y = np.linspace(0, 1, n + 1)[:-1]
        centers = np.array([(xi, yi) for xi in x for yi in y])
        r = (s) / 2
        radii = np.full(centers.shape[0], r)
        return -np.sum(radii)

    result = minimize(lambda s: objective(s), x0=0.5, bounds=[(1e-8, 1 - 1e-8)], method='L-BFGS-B')
    s_opt = result.x[0]
    n = 5
    x = np.linspace(0, 1, n + 1)[:-1]
    y = np.linspace(0, 1, n + 1)[:-1]
    centers = np.array([(xi, yi) for xi in x for yi in y])
    r = s_opt / 2
    radii = np.full(centers.shape[0], r)

    # Add 26th circle in top-right corner
    corner = np.array([1 - r, 1 - r])
    dist_to_corner = np.linalg.norm(centers[-1] - corner)
    max_radius_corner = dist_to_corner / 2
    extra_radius = max_radius_corner - r
    radii[-1] += extra_radius
    centers[-1] = corner

    # Ensure all circles are within bounds
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            raise ValueError("Circle out of bounds")

    # Validate non-overlapping
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError("Circles overlap")

    return centers, radii, np.sum(radii)
