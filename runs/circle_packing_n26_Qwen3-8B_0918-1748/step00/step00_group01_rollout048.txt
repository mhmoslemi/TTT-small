import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from itertools import combinations

def run_packing():
    # Base configuration with hexagonal packing
    n = 26
    centers = []
    radii = []
    for i in range(n):
        x = i / np.sqrt(3)
        y = (i // 2) / 2
        if x > 1 or y > 1:
            x = 1 - (i % 2) / np.sqrt(3)
            y = 1 - (i // 2) / 2
        centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(n, 0.05)

    # Function to compute total sum of radii
    def total_sum(radii):
        return np.sum(radii)

    # Objective function to maximize sum of radii
    # with constraints: circles within square and non-overlapping
    def objective(radii):
        return -total_sum(radii)  # Minimize negative sum

    # Constraint 1: circles must be within square
    def constraint1(radii):
        for i in range(n):
            x, y = centers[i]
            if not (0 <= x - radii[i] and x + radii[i] <= 1 and 0 <= y - radii[i] and y + radii[i] <= 1):
                return -1
        return 0

    # Constraint 2: circles must not overlap
    def constraint2(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Optimization
    bounds = [(0, 1) for _ in range(n)]
    res = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}
        ],
        method='SLSQP'
    )

    # Update radii with optimized values
    radii = res.x

    # Final validation (not shown here but should be done in practice)
    # print(validate_packing(centers, radii))

    return (centers, radii, total_sum(radii))
