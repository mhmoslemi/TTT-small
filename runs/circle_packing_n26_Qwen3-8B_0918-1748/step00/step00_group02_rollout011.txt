import numpy as np
import math
from scipy.optimize import minimize

def run_packing():
    # Define the number of circles
    n = 26
    # Initial guess for centers in a hexagonal grid around a central circle
    # We'll create a nested pattern with center circle and layers around it
    
    def hex_grid(n_layers, center, radius, spacing):
        centers = []
        for layer in range(n_layers + 1):
            layer_centers = []
            for i in range(6 * layer + 1):
                angle = 2 * math.pi / 6 * i
                x = center[0] + (radius + spacing) * layer * math.cos(angle)
                y = center[1] + (radius + spacing) * layer * math.sin(angle)
                layer_centers.append((x, y))
            centers.extend(layer_centers)
        return np.array(centers)
    
    # Initial parameters for hexagonal packing
    center = [0.5, 0.5]
    initial_radius = 0.1
    spacing = 0.1
    num_layers = 3  # Number of layers around the central circle
    
    centers = hex_grid(num_layers, center, initial_radius, spacing)
    
    # Objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    # Constraint: circles must be within the unit square
    def constraint1(radii):
        return np.array([r for r in radii])
    
    # Constraint: circles must not overlap
    def constraint2(radii):
        dists = np.sqrt(np.sum((centers[i] - centers[j])**2) for i in range(n) for j in range(i+1, n))
        min_allowed = np.array([radii[i] + radii[j] for i in range(n) for j in range(i+1, n)])
        return dists - min_allowed
    
    # Constraints: boundary condition for each circle
    def constraint3(radii):
        dists = np.array([np.min([centers[i][0] - radii[i], centers[i][1] - radii[i], 
                                 1 - centers[i][0] - radii[i], 1 - centers[i][1] - radii[i]]) 
                          for i in range(n)])
        return dists
    
    # Set up optimization problem
    initial_guess = np.full(n, initial_radius)
    
    # Constraints are all >= 0
    cons = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2},
        {'type': 'ineq', 'fun': constraint3}
    ]
    
    result = minimize(objective, initial_guess, constraints=cons)
    
    radii = result.x
    centers = np.array(centers)
    
    # Function to check if the current configuration is valid
    def is_valid(centers, radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < 0 or x + r > 1 or
                y - r < 0 or y + r > 1):
                return False
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = math.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    if is_valid(centers, radii):
        sum_radii = np.sum(radii)
        return (centers, radii, sum_radii)
    else:
        raise ValueError("Optimization result is invalid")
