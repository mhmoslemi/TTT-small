from scipy.optimize import minimize

def run_packing():

    # Initial guess: place circles in a hexagonal grid pattern
    # Start with a grid of columns and rows
    cols = 6
    rows = 5
    # Initial placement: place circles in a grid with spacing to avoid overlap
    # We will later optimize this
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        col = i % cols
        row = i // cols
        x = col / (cols - 1)
        y = row / (rows - 1)
        centers[i] = [x, y]
        radii[i] = 0.04

    def constraint_func(x):
        # x is a flat array of 26*3 variables: [x0, y0, r0, x1, y1, r1, ...]
        x_vec = x.reshape(n, 3)
        centers_opt = x_vec[:, :2]
        radii_opt = x_vec[:, 2]
        # Check boundaries
        for i in range(n):
            x, y = centers_opt[i]
            r = radii_opt[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return 1.0
        # Check circle overlaps
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers_opt[i]
                x2, y2 = centers_opt[j]
                r1 = radii_opt[i]
                r2 = radii_opt[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2 - 1e-8:
                    return 1.0
        return 0.0

    def objective(x):
        x_vec = x.reshape(n, 3)

        x0[i * 3 + 2] = radii[i]

    # Optimization

        objective,

        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_func}
