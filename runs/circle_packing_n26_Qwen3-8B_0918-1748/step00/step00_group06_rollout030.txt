import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define initial grid positions for 5x5 grid with an extra circle
    grid = np.linspace(0.1, 0.9, 5)
    x_grid, y_grid = np.meshgrid(grid, grid)
    centers = np.column_stack((x_grid.flatten(), y_grid.flatten()))
    # Add an extra circle in the top right corner
    centers = np.vstack((centers, [0.95, 0.95]))
    radii = np.zeros(26)
    
    # Initial guess for radii based on grid spacing
    spacing = 0.2
    initial_radius = spacing / 2
    radii[:25] = initial_radius
    # Adjust the extra circle radius
    radii[25] = 0.05
    
    # Optimization function to maximize total radius
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraints(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return False
        return True
    
    # Optimization with constraints
    def optimize_packing():
        bounds = [(0, 0.5) for _ in range(26)]
        result = optimize.minimize(
            objective,
            radii,
            args=(centers,),
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': lambda r: constraints(r, centers)}
            ]
        )
        return result
    
    result = optimize_packing()
    radii = result.x
    # Ensure centers are within [0,1]^2 and adjust radii to fit bound
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
            r = np.min([x, 1 - x, y, 1 - y])
            radii[i] = r
    # Final validation
    if not constraints(radii, centers):
        raise ValueError("Optimization failed to satisfy constraints")
    return (centers, radii, np.sum(radii))
