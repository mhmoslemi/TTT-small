    def objective(radii):

    def constraint_func(radii):
        constraints = []
        for i, j in combinations(range(len(radii)), 2):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))

    bounds = [(0, 1e-3) for _ in range(len(radii))]

    x0 = radii.copy()

    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint_func},
        method="SLSQP"
    )

    if not result.success:

    # Final validation
    centers = np.array(centers)
