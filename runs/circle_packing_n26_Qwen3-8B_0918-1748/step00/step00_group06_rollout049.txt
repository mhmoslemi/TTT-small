import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from scipy.optimize import Bounds
from collections import defaultdict

def run_packing():
    # Initial guess for circle centers using a grid pattern
    num_circles = 26
    grid_rows, grid_cols = 5, 5  # 5x5 grid with one extra circle
    spacing = 1.0 / (grid_rows - 1)
    centers = np.zeros((num_circles, 2))

    # Fill the grid with 5x5 circles
    for i in range(grid_rows):
        for j in range(grid_cols):
            if i * grid_cols + j < num_circles:
                x = j * spacing
                y = i * spacing
                centers[i * grid_cols + j] = [x, y]

    # Add the 26th circle near the top right
    centers[-1] = [0.8, 0.8]

    # Apply a small perturbation to avoid symmetry
    centers += np.random.normal(0, 0.02, centers.shape)

    # Function to compute the total sum of radii
    def objective(radii, centers):
        return -np.sum(radii)

    # Constraints
    def constraint_dist(radii, centers, i, j):
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist - (radii[i] + radii[j])

    # Optimization parameters
    bounds = [(0, 1) for _ in range(num_circles)]
    constraints = []

    # Define radius variables
    n = num_circles
    radii = np.ones(n) * 0.1  # Initial guess for radius

    # Optimization
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=Bounds([0] * n, [1] * n),
        constraints=[
            {'type': 'ineq', 'fun': lambda r: 1 - r[i] for i in range(n)},
            {'type': 'ineq', 'fun': lambda r: r[i] for i in range(n)},
        ],
        # Add distance constraints
        # For each pair of circles:
        *[
            {'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_dist(r, centers, i, j)}
            for i in range(n)
            for j in range(i + 1, n)
        ],
    )

    # Extract results
    radii = result.x
    centers = centers

    # Final validation
    success, message = validate_packing(centers, radii)
    if not success:
        raise ValueError(message)

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
