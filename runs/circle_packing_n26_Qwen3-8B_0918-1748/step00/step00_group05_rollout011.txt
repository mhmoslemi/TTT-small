import numpy as np
import matplotlib.pyplot as plt
import itertools
import random
from scipy.optimize import minimize

def run_packing():
    # Initial configuration: hexagonal packing with 26 circles
    # This is a hybrid approach with some circles near the corners and edges
    # We will start with a hexagonal packing in the center and then adjust for boundary constraints
    # Define starting positions and radii using a hexagonal packing pattern
    centers = []
    radii = []

    # Hexagonal packing in the center
    num_layers = 5
    for layer in range(num_layers):
        num_circles = 6 * layer + 1
        if num_circles > 26:
            break
        for i in range(num_circles):
            angle = 2 * np.pi / 6 * i
            r = 1.0 / (num_layers + 1) * (layer + 1)
            x = 0.5 + r * np.cos(angle)
            y = 0.5 + r * np.sin(angle)
            centers.append([x, y])
            radii.append(r)

    # This will be adjusted to fit 26 circles, some near the corners and edges
    # Now fill the remaining positions with small circles at the corners and edges
    remaining = 26 - len(centers)
    if remaining > 0:
        # Add small circles near the corners
        corner_positions = np.array([
            [0.05, 0.05], [0.95, 0.05], [0.95, 0.95], [0.05, 0.95]
        ])
        for pos in corner_positions:
            centers.append(pos)
            radii.append(0.04)  # small radius to avoid overlap with main circles

        # Add small circles near the edges (non-corner)
        edge_positions = [
            [0.1, 0.5], [0.5, 0.1], [0.9, 0.5], [0.5, 0.9]
        ]
        for pos in edge_positions:
            centers.append(pos)
            radii.append(0.04)

    # Trim to 26 circles and make sure we have exactly 26
    centers = np.array(centers[:26])
    radii = np.array(radii[:26])

    # Now optimize the configuration using a metaheuristic and local optimization
    # Initial guess for positions and radii
    def objective(x):
        # x has shape (26*2 + 26) -> centers flattened and radii appended
        centers_flat = x[:26*2].reshape((26, 2))
        radii = x[26*2:]
        # Check all constraints and return negative sum of radii to maximize
        if not all(0 <= center[0] - r <= 1 + 1e-12 and 0 <= center[1] - r <= 1 + 1e-12 for center, r in zip(centers_flat, radii)):
            return np.inf
        if not all(center[0] + r <= 1 + 1e-12 and center[1] + r <= 1 + 1e-12 for center, r in zip(centers_flat, radii)):
            return np.inf
        for i in range(len(centers_flat)):
            for j in range(i+1, len(centers_flat)):
                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        return -np.sum(radii)

    # Initial guess
    initial_guess = np.array([0.5 for _ in range(26*2 + 26)])
    initial_centers = initial_guess[:26*2].reshape(26, 2)
    initial_radii = initial_guess[26*2:]
    # Use local optimization to refine the configuration
    res = minimize(objective, initial_guess, method='L-BFGS-B', bounds=[(0, 1) for _ in range(26*2)] + [(0, 1) for _ in range(26)])
    final_centers = res.x[:26*2].reshape(26, 2)
    final_radii = res.x[26*2:]

    # Final Check
    # Ensure centers are within [0,1]^2 and radii are non-negative
    for i in range(26):
        x, y = final_centers[i]
        r = final_radii[i]
        if x < 0 or x > 1 or y < 0 or y > 1:
            print("Center out of bounds")
            return np.array([]), np.array([]), 0.0
        if r < 0:
            print("Negative radius")
            return np.array([]), np.array([]), 0.0

    # Final validation to ensure no overlaps
    for i in range(26):
        for j in range(i+1, 26):
            dx = final_centers[i][0] - final_centers[j][0]
            dy = final_centers[i][1] - final_centers[j][1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < final_radii[i] + final_radii[j] - 1e-12:
                print("Overlap detected")
                return np.array([]), np.array([]), 0.0

    return final_centers, final_radii, np.sum(final_radii)
