import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial guess for circle centers using a hexagonal grid
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Generate a hexagonal grid pattern in the unit square
    # This is a simplified hexagonal grid with spacing adjusted to fit the square
    # We'll create a grid with ~6 rows and ~4 columns for 26 circles
    rows, cols = 6, 5
    spacing = 0.25

    # Create hexagonal grid
    for i in range(rows):
        for j in range(cols):
            col = j - (cols - 1) / 2
            row = i - (rows - 1) / 2
            x = col * spacing * np.sqrt(3)
            y = row * spacing * 3 / 2
            x = np.clip(x, 0, 1)
            y = np.clip(y, 0, 1)
            centers[i * cols + j] = [x, y]

    # Initial guess for radii: same for all
    initial_radius = 0.1
    radii.fill(initial_radius)

    # Objective function to maximize the total radius sum
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Constraints: no overlapping circles
    def constraint_overlap(x):
        centers_flat = x[:num_circles, :]
        radii_flat = x[num_circles:]

        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dist = np.sqrt(np.sum((centers_flat[i] - centers_flat[j]) ** 2))
                if dist < radii_flat[i] + radii_flat[j] - 1e-6:
                    return dist - (radii_flat[i] + radii_flat[j] - 1e-6)

        return 0

    # Boundary constraints: circles must not exceed the square
    def constraint_boundary(x):
        centers_flat = x[:num_circles, :]
        radii_flat = x[num_circles:]

        for i in range(num_circles):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < -1e-6 or x + r > 1 + 1e-6 or y - r < -1e-6 or y + r > 1 + 1e-6:
                return -1
        return 0

    # Initial guess: centers + radii
    x0 = np.concatenate((centers.flatten(), radii))

    # Set up optimization problem
    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=[(-np.inf, np.inf)] * (num_circles * 3),
        constraints=cons,
        tol=1e-6,
        options={'maxiter': 500, 'ftol': 1e-9, 'eps': 1e-8}
    )

    centers = result.x[:num_circles * 2].reshape((num_circles, 2))
    radii = result.x[num_circles * 2:]

    # Final check
    if np.isnan(centers).any() or np.isnan(radii).any():
        raise ValueError("NaN values present")

    # Final validation pass
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-6 or x + r > 1 + 1e-6 or y - r < -1e-6 or y + r > 1 + 1e-6:
            raise ValueError(f"Circle {i} is outside the unit square")

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-6:
                raise ValueError(f"Circles {i} and {j} overlap")

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
