import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    # Initial grid layout: 5 rows, 5 columns (25 circles) with one extra at the end
    centers = []
    radii = []
    for i in range(5):
        for j in range(5):
            x = j / 4.0
            y = i / 4.0
            centers.append([x, y])
    # Add an extra circle to make 26
    centers.append([0.1, 0.9])
    centers = np.array(centers)
    radii = [1.0 / 8.0] * 25 + [0.0]
    
    # Define the objective function to maximize the sum of radii
    def objective(x):
        # x is a flat array of [x0, y0, r0, x1, y1, r1, ...]
        # Reconstruct centers and radii
        flat = np.array(x)
        n = len(flat) // 2
        centers = flat[:n*2].reshape((n, 2))
        radii = flat[n*2:]
        total = np.sum(radii)
        return -total  # Minimize negative total to maximize total
        
    # Define constraints
    def constraint(x):
        flat = np.array(x)
        n = len(flat) // 2
        centers = flat[:n*2].reshape((n, 2))
        radii = flat[n*2:]
        # Check if any radii is negative
        if np.any(radii < 0):
            return -1.0
        # Check if any circle is out of bounds
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return -1.0
        # Check for overlap between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-10:
                    return -1.0
        return 0.0
    
    # Convert centers and radii to flat array for optimization
    flat_centers = centers.reshape(-1)
    flat_radii = np.array(radii)
    flat_initial = np.concatenate([flat_centers, flat_radii])
    
    # Set bounds and constraints for optimization
    bounds = []
    for _ in range(26):
        bounds.append((0.0, 1.0))  # For x and y coordinates
    for _ in range(26):
        bounds.append((0.0, 1.0))  # For radii
    
    # Use scipy.optimize.minimize with constraints
    result = minimize(
        objective,
        flat_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint}]
    )
    
    # Extract final centers and radii
    flat_centers_final = result.x[:26*2]
    flat_radii_final = result.x[26*2:]
    centers_final = flat_centers_final.reshape((26, 2))
    radii_final = flat_radii_final
    
    # Return the result
    return (centers_final, radii_final, np.sum(radii_final))
