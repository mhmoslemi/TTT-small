import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Initial hexagonal grid configuration
    n = 26
    grid = np.zeros((n, 2))
    radii = np.zeros(n)
    count = 0
    for i in range(2):
        for j in range(2):
            grid[count] = [i + 0.5, j + 0.5]
            count += 1
    for i in range(13):
        grid[count] = [i + 0.5, 0.5]
        count += 1
    for i in range(13):
        grid[count] = [i + 0.5, 0.5]
        count += 1
    for i in range(13):
        grid[count] = [i + 0.5, 0.5]
        count += 1
    radii = np.full(n, 0.04)

    def objective(radii_vector):
        return -np.sum(radii_vector)

    def constraints(radii_vector):
        constraints = []
        for i in range(n):
            constraints.append((radii_vector[i], 0, np.inf))
        return constraints

    def bounds(radii_vector):
        return [(0, np.inf) for _ in range(n)]

    def distance_matrix(centers):
        n = centers.shape[0]
        dist = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist[i, j] = np.sqrt(dx ** 2 + dy ** 2)
                dist[j, i] = dist[i, j]
        return dist

    def compute_overlap(centers, radii):
        dist = distance_matrix(centers)
        for i in range(n):
            for j in range(i + 1, n):
                if dist[i, j] < radii[i] + radii[j] - 1e-12:
                    return True
        return False

    def compute_boundary(centers, radii):
        for i in range(n):
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12:
                return True
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12:
                return True
        return False

    def constraint_func(x):
        radii = x
        if compute_overlap(grid, radii) or compute_boundary(grid, radii):
            return 10000
        return 0

    bounds = Bounds([0] * n, [np.inf] * n)

    # Initial guess for radii
    initial_guess = np.full(n, 0.04)

    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint_func},
        method="SLSQP",
        options={"ftol": 1e-10, "max_iter": 1000}
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    centers = grid
    radii = result.x
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
