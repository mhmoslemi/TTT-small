import numpy as np
import cvxpy as cp

def run_packing():
    n = 26

    # Initial hexagonal grid positions (scaled to fit unit square)
    # Rows: 4 rows of circles with increasing number of circles per row (6, 6, 6, 8)
    rows = [6, 6, 6, 8]
    total_circles = sum(rows)
    assert total_circles == n - 2, "Main grid circle count mismatch"

    # Scale for unit square
    scale = 1.0 / (np.sqrt(3) + 1.0)

    # Generate main grid positions
    main_centers = []
    main_radii = []
    for i, row_size in enumerate(rows):
        row_y = i * (1.0 / (np.sqrt(3) * (row_size - 1) + 1))
        for j in range(row_size):
            if i % 2 == 0:  # even rows: left to right
                x = j / (row_size - 1) * (1.0 - 2 * row_y)
            else:  # odd rows: offset for hexagonal packing
                x = (j + 0.5) / (row_size - 1) * (1.0 - 2 * row_y)
            main_centers.append([x, row_y])
        if i % 2 == 1:
            # Adjust odd rows: shift to make hexagonal grid
            for j in range(row_size):
                main_centers[-1 + j][0] += 0.5 * (1.0 - 2 * row_y)

    # Convert main_centers to numpy array
    main_centers = np.array(main_centers)

    # Determine optimal radii
    # We'll optimize radii to be as large as possible without overlap
    # Positions are fixed, but radii will be variables to be optimized
    # We'll use a convex optimization approach with CVXPY

    # Define variables
    r = cp.Variable(n)
    x = cp.Variable((n, 2))

    # Constraints
    constraints = []
    for i in range(n):
        # Ensure circles are inside the square
        constraints.append(x[i, 0] - r[i] >= -1e-8)
        constraints.append(x[i, 0] + r[i] <= 1 + 1e-8)
        constraints.append(x[i, 1] - r[i] >= -1e-8)
        constraints.append(x[i, 1] + r[i] <= 1 + 1e-8)
        # Non-negative radii
        constraints.append(r[i] >= 0)

    # For overlapping circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-8)

    # Objective: maximize sum of radii
    obj = cp.Maximize(cp.sum(r))

    # Solve
    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.SCS, eps=1e-5)

    # Extract centers and radii
    centers = np.array([x.value[i] for i in range(n)])
    radii = r.value

    # We can refine the center positions if needed for better packing, but for simplicity,
    # we'll use the initial grid as the center positions
    # (Note: In a real scenario, we would propagate radii back to centers using an iterative method)
    # For now, we'll use the main grid as a base
    centers = main_centers

    # We need to determine the radii values that maximize the sum of radii
    # Since we fixed the centers, we can compute radii as the minimal distance to nearest neighbors
    # or find consistent radii such that all circles are as large as possible without overlapping

    # Since the above optimization might not have converged, we'll compute approximate radii manually
    # We'll take the minimum distance to neighbors and compute radii accordingly
    dists = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dists[i, j] = np.sqrt(dx**2 + dy**2)

    # Compute radii as half the minimal distance between circles (approximate method)
    # We assume circles are as large as possible without overlapping, so radii[i] = min(dists[i, j] / 2 for j in range(n) if j != i)
    radii = np.zeros(n)
    for i in range(n):
        min_dist = np.inf
        for j in range(n):
            if i == j:
                continue
            if dists[i, j] < min_dist:
                min_dist = dists[i, j]
        radii[i] = min_dist / 2

    # We need to ensure all circles are within the square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            # Adjust radius to fit in square
            radii[i] = min(r, 1 - x, x, 1 - y)

    # Compute sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
