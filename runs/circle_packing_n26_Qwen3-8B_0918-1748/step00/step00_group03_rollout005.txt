import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt
from scipy.spatial.distance import pdist, squareform

def generate_hexagonal_grid(n, scale=1.0, offset=0.0):
    rows = int(np.ceil((np.sqrt(2 * n) - 1) / np.sqrt(3)))
    cols = int(np.ceil(n / rows))
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = (j + 0.5 * (i % 2)) * scale
            y = i * np.sqrt(3)/2 * scale
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append((x, y))
    return np.array(centers)

def get_initial_centers(n):
    centers = generate_hexagonal_grid(n)
    if len(centers) < n:
        extra = n - len(centers)
        for i in range(extra):
            x = np.random.uniform(0, 1)
            y = np.random.uniform(0, 1)
            centers = np.vstack((centers, [x, y]))
    return centers

def objective_function(radii, centers):
    return -np.sum(radii)

def constraint_radius(radii, centers):
    return np.sum(radii)

def constraint_center_bounds(radii, centers):
    return np.array([0.0, 0.0])

def constraint_boundary(radii, centers):
    return np.sum(np.maximum(centers[:, 0] - radii, 0) + np.maximum(centers[:, 1] - radii, 0) - 1 + 1e-12)

def constraint_inter_circle(radii, centers):
    dist = squareform(pdist(centers))
    return np.sum(np.sum(np.maximum(dist - (radii[:, np.newaxis] + radii[np.newaxis, :]) + 1e-12, 0), axis=1))

def run_packing():
    n = 26
    initial_centers = get_initial_centers(n)
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))
    
    if len(initial_centers) != n:
        raise ValueError("Initial center generation failed to return n circles")
    
    radius = np.full(n, 0.05)
    
    def objective(x):
        radii = x[:n]
        return -np.sum(radii)
    
    def constraint(x):
        radii = x[:n]
        centers = initial_centers.copy()
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -np.inf
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.sum(radii)
    
    x0 = np.concatenate((radius, np.zeros(n)))
    
    result = opt.minimize(
        objective,
        x0,
        bounds=[(0.0, 1.0)] * (2 * n),
        constraints=[opt.constraints.DELAYED penalty]
    )
    
    if result.success:
        radii = result.x[:n]
        centers = initial_centers.copy()
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                raise ValueError("Boundary constraint violation")
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    raise ValueError("Inter-circle overlap")
        return (centers, radii, np.sum(radii))
    else:
        raise ValueError("Optimization failed")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"


    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"

if __name__ == "__main__":
    centers, radii, sum_radii = run_packing()
    print(f"Total sum of radii: {sum_radii}")
