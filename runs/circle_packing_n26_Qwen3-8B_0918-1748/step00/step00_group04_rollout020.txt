import numpy as np
import matplotlib.pyplot as plt
import math
from scipy.optimize import minimize

def run_packing():
    def objective(radii, centers):
        return -np.sum(radii)

    def constraints(radii, centers):
        n = len(radii)
        cons = []
        for i in range(n):
            r_i = radii[i]
            x_i, y_i = centers[i]
            if x_i - r_i < 0 or x_i + r_i > 1:
                cons.append((r_i, 'ineq', 0, 0))
            if y_i - r_i < 0 or y_i + r_i > 1:
                cons.append((r_i, 'ineq', 0, 0))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                cons.append((dist - (radii[i] + radii[j]), 'ineq', 0, 0))
        return cons

    # Define base configuration: 5 rows of 5 circles, then hexagonal stacking
    def generate_base_config():
        centers = []
        radii = []
        for i in range(5):
            for j in range(5):
                x = j / 4.0
                y = i / 4.0
                if i % 2 == 0:
                    centers.append([x, y])
                    radii.append(0.08)
                else:
                    centers.append([x + 0.25, y])
                    radii.append(0.08)
        # Add corner circles
        corners = [[0, 0], [1, 0], [1, 1], [0, 1]]
        for x, y in corners:
            centers.append([x, y])
            radii.append(0.06)
        # Add top center circle
        centers.append([0.5, 1])
        radii.append(0.06)
        # Add bottom center circle
        centers.append([0.5, 0])
        radii.append(0.06)
        # Add extra circles in hexagonal arrangement
        for i in range(2):
            for j in range(2):
                x = j / 3.0
                y = i / 3.0 + 0.2
                centers.append([x, y])
                radii.append(0.05)
        centers = np.array(centers)
        radii = np.array(radii)
        return centers, radii

    # Generate initial configuration
    centers, radii = generate_base_config()
    n = len(centers)
    radii = radii.astype(float)
    centers = centers.astype(float)

    # Define bounds for radii
    bounds = [(0, 0.2) for _ in range(n)]

    # Define initial guess for radii
    initial_guess = radii.copy()

    # Optimization
    def optimize_config(centers):
        def objective_func(radii):
            return -np.sum(radii)

        def constraint_func(radii):
            n = len(radii)
            cons = []
            for i in range(n):
                r_i = radii[i]
                x_i, y_i = centers[i]
                if x_i - r_i < 0 or x_i + r_i > 1:
                    cons.append((r_i, 'ineq', 0, 0))
                if y_i - r_i < 0 or y_i + r_i > 1:
                    cons.append((r_i, 'ineq', 0, 0))
            for i in range(n):
                for j in range(i + 1, n):
                    dx = centers[i, 0] - centers[j, 0]
                    dy = centers[i, 1] - centers[j, 1]
                    dist = np.sqrt(dx*dx + dy*dy)
                    cons.append((dist - (radii[i] + radii[j]), 'ineq', 0, 0))
            return cons

        res = minimize(objective_func, initial_guess, bounds=bounds, constraints=constraint_func)
        return res.x

    optimized_radii = optimize_config(centers)

    # Validate packing
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, "Circle outside unit square"
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, "Circles overlap"
        return True, "ok"

    success, message = validate_packing(centers, optimized_radii)
    assert success, message

    sum_radii = np.sum(optimized_radii)
    return centers, optimized_radii, sum_radii
