import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    # Initial grid-based seed with 26 points
    n = 26
    grid_size = 3
    grid_points = np.zeros((n, 2))
    for i in range(n):
        row = i // grid_size
        col = i % grid_size
        grid_points[i, 0] = col / (grid_size - 1)
        grid_points[i, 1] = row / (grid_size - 1)

    # Perturb grid to avoid symmetry
    perturbation = 0.1
    grid_points += np.random.uniform(-perturbation, perturbation, size=grid_points.shape)

    # Initial radii: small values to allow expansion
    radii = np.full(n, 0.02)

    # Optimization parameters
    max_iterations = 100
    learning_rate = 0.01
    tolerance = 1e-6

    # Define objective function to maximize sum of radii
    def objective(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        return -np.sum(radii)

    # Define constraints for optimization
    def constraint(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Initial guess with grid points and small radii
    x_initial = np.concatenate((grid_points.flatten(), radii))

    # Optimization with constraints
    result = minimize(
        objective,
        x_initial,
        method='SLSQP',
        bounds=[(-1, 1) for _ in range(2*n)] + [(0, 1) for _ in range(n)],
        constraints={'type': 'ineq', 'fun': constraint},
        options={'maxiter': max_iterations, 'ftol': tolerance}
    )

    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    sum_radii = np.sum(radii)

    # Final validation
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing validation failed: {message}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
