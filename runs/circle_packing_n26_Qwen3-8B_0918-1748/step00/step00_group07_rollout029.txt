import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt

def run_packing():
    # Define the number of circles
    n = 26
    # Initial guess for centers: Hexagonal grid
    # We'll create a grid with 5 rows and 5 or 6 columns, interleaved
    centers = []
    for row in range(5):
        for col in range(5):
            if (row % 2 == 0):
                x = col + 0.5
            else:
                x = col + 0.4
            y = row + 0.1
            if y < 1.0 and x < 1.0 and y > 0.0 and x > 0.0:
                centers.append([x, y])
    
    # Manually add 6 more circles to reach 26
    # These will be placed in the remaining positions
    # These are added with a bit of offset to avoid overlapping
    # Total of 26 circles
    centers += [[0.1, 0.9], [0.9, 0.9], [0.1, 0.8], [0.9, 0.8], [0.1, 0.7], [0.9, 0.7]]
    
    # Convert to numpy array
    centers = np.array(centers)
    
    # Now, we will use scipy.optimize to maximize the total radius sum
    # We will define a function that, given radii, tries to adjust them + centers
    # to maximize the total, under constraints
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint1(radii):
        return np.sum(radii)
    
    x0 = np.full(n, 0.03)
    
    cons = ({'type': 'ineq', 'fun': lambda r: np.sum(r)})
    
    result = opt.minimize(
        objective,
        x0,
        method='SLSQP',
        constraints=cons,
        bounds=[(0, 1) for _ in range(n)]
    )
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Now, we need to adjust the positions to prevent overlaps
    # We'll use a gradient descent idea to move the centers
    def overlap_penalty(centers, radii):
        penalty = 0.0
        n = len(centers)
        for i in range(n):
            x1, y1 = centers[i]
            r1 = radii[i]
            for j in range(i + 1, n):
                x2, y2 = centers[j]
                r2 = radii[j]
                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r1 + r2 - 1e-12:
                    penalty += (r1 + r2 - dist)**2
        return penalty
    
    def boundary_penalty(centers, radii):
        penalty = 0.0
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12:
                penalty += (x - r + 1e-12)**2
            if x + r > 1 + 1e-12:
                penalty += (x + r - 1 - 1e-12)**2
            if y - r < -1e-12:
                penalty += (y - r + 1e-12)**2
            if y + r > 1 + 1e-12:
                penalty += (y + r - 1 - 1e-12)**2
        return penalty
    
    def total_penalty(centers, radii):
        return overlap_penalty(centers, radii) + boundary_penalty(centers, radii)
    
    # Optimization step to adjust centers
    def total_energy(centers):
        return total_penalty(centers, radii)
    
    # Define bounds for centers
    bounds = [(0, 1) for _ in range(n * 2)]
    
    result = opt.minimize(
        total_energy,
        centers.reshape(-1),
        method='L-BFGS-B',
        bounds=bounds
    )
    
    centers = result.x.reshape(-1, 2)
    
    # Final check
    # Now, we need to check if the configuration is valid
    # We use a simple way to avoid overlaps by reducing radii slightly
    # We'll run a second pass to minimize overlap
    def calc_radii(centers):
        n = len(centers)
        radii = np.full(n, 0.05)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    # Reduce radii to prevent overlap
                    radii[i] = (dist - 1e-12) / 2
                    radii[j] = (dist - 1e-12) / 2
        return radii
    
    radii = calc_radii(centers)
    sum_radii = np.sum(radii)
    
    # Final validation and optimization
    # We'll use a more accurate optimization here to raise radii as much as possible
    def energy(radii):
        penalty = 0.0
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12:
                penalty += (x - r + 1e-12)**2
            if x + r > 1 + 1e-12:
                penalty += (x + r - 1 - 1e-12)**2
            if y - r < -1e-12:
                penalty += (y - r + 1e-12)**2
            if y + r > 1 + 1e-12:
                penalty += (y + r - 1 - 1e-12)**2
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    penalty += (radii[i] + radii[j] - dist)**2
        return penalty
    
    result = opt.minimize(
        energy,
        radii,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n)]
    )
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
