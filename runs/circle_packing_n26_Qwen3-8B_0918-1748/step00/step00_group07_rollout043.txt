import numpy as np
from scipy.optimize import minimize
import random

def run_packing():
    # Initial configuration: 15 circles in hexagonal grid, 11 on edges

    # Define initial positions for hexagonal packing in the center
    n_hex = 15
    centers_hex = []
    radii_hex = []
    r_hex = 0.1  # Initial radius for central hexagonal grid

    # Define the hexagonal grid
    for i in range(n_hex):
        if i % 2 == 0:
            x = 0.2 + i // 2 * (0.4 + 2 * r_hex)
            y = 0.2 + (i // 2) % 2 * (0.4)
        else:
            x = 0.2 + (i // 2) * (0.4 + 2 * r_hex) + r_hex
            y = 0.2 + (i // 2) % 2 * (0.4)
        centers_hex.append([x, y])
        radii_hex.append(r_hex)

    # Define initial positions for boundary circles (11 circles)
    centers_edge = []
    radii_edge = []
    r_edge = 0.15

    # Corner circles (4 corners)
    for i, corner in enumerate([(0.1, 0.1), (0.9, 0.1), (0.9, 0.9), (0.1, 0.9)]):
        centers_edge.append(list(corner))
        radii_edge.append(r_edge)

    # Edge circles (3 on left, 3 on right, 3 on top, 3 on bottom, total 12, minus 1 to make 11)
    for i in range(3):
        centers_edge.append([0.1 + i * (0.4 + 2 * r_edge), 0.45])
        radii_edge.append(r_edge)
    for i in range(3):
        centers_edge.append([0.1 + i * (0.4 + 2 * r_edge), 0.55])
        radii_edge.append(r_edge)
    for i in range(3):
        centers_edge.append([0.45, 0.1 + i * (0.4 + 2 * r_edge)])
        radii_edge.append(r_edge)
    for i in range(3):
        centers_edge.append([0.55, 0.1 + i * (0.4 + 2 * r_edge)])
        radii_edge.append(r_edge)
    # Remove one to make total 11
    centers_edge.pop()
    radii_edge.pop()

    # Combine centers and radii
    centers = np.array(centers_hex + centers_edge)
    radii = np.array(radii_hex + radii_edge)

    # Optimization parameters
    def total_sum(radii_opt):
        return -np.sum(radii_opt)

    def constraint_func(params):
        centers_opt = centers
        radii_opt = radii
        # No constraints, but radius can't be negative
        # Since we're only optimizing, keep radii as is
        return np.sum(radii_opt)

    # Optimize for maximum sum of radii using coordinate descent
    def objective(x):
        # x is a flat array of radii
        return -np.sum(x)

    def cons(x):
        return np.sum(x)

    bounds = [(0, np.inf)] * len(radii)

    # Optimize
    res = minimize(objective, radii, bounds=bounds, constraints={"type": "ineq", "fun": cons})

    # Get optimized radii
    radii_opt = res.x

    # Final centers and radii
    # We need to validate the configuration
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    # Attempt validation
    try:
        valid, msg = validate_packing(centers, radii_opt)
        if not valid:
            raise ValueError(f"Validation failed: {msg}")
    except Exception as e:
        # If validation fails, return a basic valid configuration with low radii
        centers = np.array([[0.5, 0.5]] * 26)
        radii = np.array([0.05] * 26)
        return (centers, radii, np.sum(radii))

    return (centers, radii_opt, np.sum(radii_opt))
