import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial hexagonal grid parameters
    rows = 5
    cols = 6
    radius_base = 0.15
    hex_offset = np.sqrt(3) / 2 * radius_base
    horizontal_step = 2 * radius_base
    vertical_step = hex_offset

    # Generate initial hexagonal grid
    centers = []
    radii = []

    # Place primary circles in hexagonal grid
    for row in range(rows):
        for col in range(cols):
            x = col * horizontal_step
            y = row * vertical_step
            if row % 2 == 0:
                x += horizontal_step / 2
            centers.append([x, y])
            radii.append(radius_base)

    # Adjust for unit square
    scale = 1.0
    for i in range(len(centers)):
        centers[i][0] *= scale
        centers[i][1] *= scale

    # Initial guess for optimization
    centers = np.array(centers)
    radii = np.array(radii)

    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize

    # Define constraints: boundaries and non-overlap
    def constraint_boundary(radii, centers):
        return np.array([np.min(centers[:, 0] - radii) > -1e-8,
                         np.min(centers[:, 1] - radii) > -1e-8,
                         np.min(1 - centers[:, 0] - radii) > -1e-8,
                         np.min(1 - centers[:, 1] - radii) > -1e-8])

    def constraint_non_overlap(radii, centers):
        n = len(centers)
        dists = np.zeros(n * (n - 1) // 2)
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dists[idx] = np.sqrt(dx**2 + dy**2) - (radii[i] + radii[j])
                idx += 1
        return np.min(dists) > -1e-8

    # Optimization
    bounds = [(0, 1.0) for _ in range(len(radii))]

    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda x: constraint_boundary(x, centers)},
            {'type': 'ineq', 'fun': lambda x: constraint_non_overlap(x, centers)}
        ]
    )

    # Final centers and radii
    optimized_centers = centers
    optimized_radii = result.x
    sum_radii = np.sum(optimized_radii)

    return (optimized_centers, optimized_radii, sum_radii)
