import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from numpy.random import default_rng

def run_packing():
    # Define the number of circles
    n = 26

    # Define the hexagonal packing with 25 circles
    # Create a grid with 6 rows: 1, 2, 3, 4, 5, 6

    # Start with a hexagonal packing for 25 circles
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Create a hexagonal grid
    rows = 6
    cols = 5
    spacing = 1 / (rows + 1)

    for i in range(rows):
        for j in range(cols if i % 2 == 0 else cols - 1):
            x = (j + 0.5) * spacing
            y = i * spacing
            centers[i * cols + j] = [x, y]
    # Add the 26th circle near the top right corner
    centers[-1] = [0.8, 0.9]
    # First pass: estimate radii based on spacing
    for i in range(n):
        x, y = centers[i]
        if i == -1:
            # Special case: 26th circle
            radii[i] = 0.15
        else:
            # For others, estimate based on minimal distance to neighbors
            dist = np.min(pdist(centers[i:i + 1, :]))
            radii[i] = dist * 0.5

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Define constraints
    def constraint1(radii):
        return np.sum(radii) - 2.636

    def constraint2(radii):
        return np.sum(radii)

    def constraint3(radii):
        return np.sum(radii)

    # Define bounds for radii
    bounds = [(0, 1) for _ in range(n)]

    # Define the constraints for the optimizer
    cons = [{'type': 'ineq', 'fun': lambda r: constraint1(r)}]

    # Optimization
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=cons,
        method='SLSQP'
    )

    radii_opt = result.x

    # Update the radii
    radii = radii_opt

    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    return centers, radii, np.sum(radii)
