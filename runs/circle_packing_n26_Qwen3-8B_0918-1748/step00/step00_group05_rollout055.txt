import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26

    # Initial guess for circle centers (grid-like with some variance)
    centers = np.random.rand(n, 2)
    # Initial guess for radii (small to start with)
    radii = np.full(n, 0.05)

    # Optimization variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []

    # Boundary constraints: each circle must be within [0,1]^2
    for i in range(n):
        constraints.append(x[i, 0] - r[i] >= 0)
        constraints.append(x[i, 0] + r[i] <= 1)
        constraints.append(x[i, 1] - r[i] >= 0)
        constraints.append(x[i, 1] + r[i] <= 1)

    # Non-overlapping constraints: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-6)

    # Non-negativity of radii
    constraints.append(cp.Minimize(cp.sum(r)))

    # Solve using CVXPY
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.OSQP)

    if not prob.success:
        raise ValueError("Optimization failed.")

    # Extract the centers and radii
    centers = x.value
    radii = r.value

    # Compute sum of radii
    sum_radii = np.sum(radii)

    # Validate the result
    success, message = validate_packing(centers, radii)
    if not success:
        raise ValueError(f"Packing validation failed: {message}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
