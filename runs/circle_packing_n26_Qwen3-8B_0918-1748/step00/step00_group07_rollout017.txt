import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist

def run_packing():
    # Define initial configuration with clusters and hexagonal packing
    # Cluster 1: Bottom-left corner with hexagonal packing
    n1 = 6
    x_vals1 = np.linspace(0.1, 0.3, n1)
    y_vals1 = np.linspace(0.1, 0.3, n1)
    centers1 = np.column_stack((x_vals1, y_vals1))
    radii1 = np.full(n1, 0.08)

    # Cluster 2: Bottom-right corner with hexagonal packing
    n2 = 6
    x_vals2 = np.linspace(0.7, 0.9, n2)
    y_vals2 = np.linspace(0.1, 0.3, n2)
    centers2 = np.column_stack((x_vals2, y_vals2))
    radii2 = np.full(n2, 0.08)

    # Cluster 3: Top-left corner with hexagonal packing
    n3 = 6
    x_vals3 = np.linspace(0.1, 0.3, n3)
    y_vals3 = np.linspace(0.7, 0.9, n3)
    centers3 = np.column_stack((x_vals3, y_vals3))
    radii3 = np.full(n3, 0.08)

    # Cluster 4: Top-right corner with hexagonal packing
    n4 = 6
    x_vals4 = np.linspace(0.7, 0.9, n4)
    y_vals4 = np.linspace(0.7, 0.9, n4)
    centers4 = np.column_stack((x_vals4, y_vals4))
    radii4 = np.full(n4, 0.08)

    # Central cluster with hexagonal packing
    n5 = 2
    x_vals5 = np.linspace(0.4, 0.6, n5)
    y_vals5 = np.linspace(0.4, 0.6, n5)
    centers5 = np.column_stack((x_vals5, y_vals5))
    radii5 = np.full(n5, 0.06)

    # Additional cluster near the center
    n6 = 2
    x_vals6 = np.array([0.5, 0.5])
    y_vals6 = np.array([0.5, 0.5])
    centers6 = np.column_stack((x_vals6, y_vals6))
    radii6 = np.full(n6, 0.05)

    # Combine all centers and radii
    centers = np.vstack((centers1, centers2, centers3, centers4, centers5, centers6))
    radii = np.concatenate((radii1, radii2, radii3, radii4, radii5, radii6))

    # Flatten the centers and radii to form the initial guess
    initial_centers = centers.reshape(-1)
    initial_radii = radii

    # Define total number of circles
    n = centers.shape[0]

    # Define the variables to optimize (centers and radii)
    # We will treat centers as variables for x and y, and radii as variables
    # Convert to vector for optimization
    x0 = np.concatenate((initial_centers, initial_radii))

    # Define the optimization function to maximize the total sum of radii
    def objective(x):
        flat_centers = x[:2 * n].reshape(n, 2)
        flat_radii = x[2 * n:]
        return -np.sum(flat_radii)  # We minimize the negative to maximize

    # Constraints: non-overlapping and boundary constraints
    def constraint(x):
        flat_centers = x[:2 * n].reshape(n, 2)
        flat_radii = x[2 * n:]
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(flat_centers[i] - flat_centers[j])
                if dist < flat_radii[i] + flat_radii[j] - 1e-9:
                    return dist - (flat_radii[i] + flat_radii[j]) + 1e-9
        # Check boundary constraints
        for i in range(n):
            x_i, y_i = flat_centers[i]
            r_i = flat_radii[i]
            if x_i - r_i < -1e-9 or x_i + r_i > 1 + 1e-9:
                return x_i + r_i - 1 - 1e-9
            if y_i - r_i < -1e-9 or y_i + r_i > 1 + 1e-9:
                return y_i + r_i - 1 - 1e-9
        return 0  # If all constraints are satisfied

    # Set bounds for the optimization: x and y between 0 and 1, radii >= 0
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1), (0, 1)])
        bounds.append((0, 1))  # radius for each circle

    # Perform optimization
    result = minimize(objective, x0=x0, bounds=bounds, constraints={"type": "ineq", "fun": constraint})

    # Extract results
    optimized_x = result.x
    flat_centers = optimized_x[:2 * n].reshape(n, 2)
    flat_radii = optimized_x[2 * n:]

    # Return the centers, radii, and sum of radii
    return flat_centers, flat_radii, np.sum(flat_radii)
