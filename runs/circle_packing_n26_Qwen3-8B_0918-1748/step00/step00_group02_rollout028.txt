import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    np.random.seed(42)
    
    # Initial hexagonal packing in a square
    def initial_positions(n):
        cols = int(np.sqrt(n))
        rows = (n + cols - 1) // cols
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * i
                y = i + 0.5 * j
                if x <= 1 and y <= 1:
                    centers.append([x, y])
        return np.array(centers)
    
    # Initial guess for centers and radii
    centers = initial_positions(26)
    radii = np.full(centers.shape[0], 0.05)
    
    # Define the objective function
    def objective(x):
        centers_flat = x[:2 * centers.shape[0]].reshape(centers.shape)
        radii_flat = x[2 * centers.shape[0]:]
        return -np.sum(radii_flat)
    
    # Define constraints
    def constraint_center_boundary(x):
        centers_flat = x[:2 * centers.shape[0]].reshape(centers.shape)
        radii_flat = x[2 * centers.shape[0]:]
        for i in range(centers.shape[0]):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1
        return 0
    
    def constraint_circle_overlap(x):
        centers_flat = x[:2 * centers.shape[0]].reshape(centers.shape)
        radii_flat = x[2 * centers.shape[0]:]
        for i, j in combinations(range(centers.shape[0]), 2):
            x_i, y_i = centers_flat[i]
            x_j, y_j = centers_flat[j]
            r_i = radii_flat[i]
            r_j = radii_flat[j]
            dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            if dist < r_i + r_j - 1e-6:
                return -1
        return 0
    
    # Combine all constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_center_boundary},
        {'type': 'ineq', 'fun': constraint_circle_overlap}
    ]
    
    # Bounds for centers (in [0,1]^2) and radii (non-negative)
    bounds = []
    for _ in range(centers.shape[0]):
        bounds.extend([(0, 1), (0, 1)])
    for _ in range(centers.shape[0]):
        bounds.append((0, 1))
    
    # Initial guess
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Optimization
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 200}
    )
    
    # Extract centers and radii
    centers_opt = result.x[:2 * centers.shape[0]].reshape(centers.shape)
    radii_opt = result.x[2 * centers.shape[0]:]
    
    # Validate and return
    if not validate_packing(centers_opt, radii_opt)[0]:
        raise ValueError("Validation failed: Packing is invalid")
    
    total_sum = np.sum(radii_opt)
    return centers_opt, radii_opt, total_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
