def run_packing():

    centers[16] = [0.5, 0.15]
    centers[17] = [0.75, 0.15]
    centers[18] = [0.15, 0.35]
    centers[19] = [0.5, 0.4]
    centers[20] = [0.85, 0.35]
    centers[21] = [0.1, 0.55]
    centers[22] = [0.6, 0.55]
    centers[23] = [0.3, 0.75]
    centers[24] = [0.7, 0.75]
    centers[25] = [0.4, 0.85]
    radii[15:26] = 0.04

    # Now we will optimize radii and positions while ensuring constraints

    # Objective function: maximize sum of radii
    def objective(x):
        # x is a flat array of [x1, y1, r1, x2, y2, r2, ...]
        # Reshape for easier handling
        flat = np.reshape(x, (-1, 3))
        centers = flat[:, :2]
        radii = flat[:, 2]

        # Compute sum of radii
        total = np.sum(radii)

        # Penalize for circles going out of bounds
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1.0 + 1e-8 or y - r < -1e-8 or y + r > 1.0 + 1e-8:
                total -= 1000 * r

        # Penalize for overlaps
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-8:
                    total -= 1000 * (radii[i] + radii[j])

        return -total  # minimize negative to maximize total

    # Constraints: 0 <= radii
    bounds = []
    for _ in range(26):
        bounds.extend([(0.0, 1.0)] * 3)  # x, y, r for each circle

    # Initial guess
    initial_guess = []
    for i in range(26):
        initial_guess.extend(centers[i])
        initial_guess.extend(radii[i])

    # Perform optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-10, 'maxiter': 1000}
    )

    # Extract final centers and radii
    flat = np.reshape(result.x, (-1, 3))
    centers = flat[:, :2]
    radii = flat[:, 2]

    # Return centers, radii, and sum of radii
    return centers, radii, np.sum(radii)
