import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from itertools import combinations

def run_packing():
    # Define the number of circles
    n = 26
    
    # Initial grid configuration for base packing
    grid_rows = [2, 3, 4, 4, 3, 2, 1]
    grid_cols = [2, 3, 4, 4, 3, 2]
    
    # Total grid circles
    total_grid_circles = sum(grid_rows)
    
    # Start with a base radius based on the grid size
    base_radius = 0.12
    
    # Define initial centers as grid points
    centers = []
    radii = []
    
    # Define spacing and offset for hexagonal packing
    row_start = 0.5
    col_start = 0.5
    row_spacing = 1.5 * base_radius  # For hexagonal packing between rows
    col_spacing = 1.5 * base_radius  # Between columns
    
    # Build the grid with hexagonal pattern
    row_index = 0
    for row in grid_rows:
        for col in range(row):
            x = col_start + col * col_spacing
            y = row_start + row_index * row_spacing
            centers.append([x, y])
            radii.append(base_radius)
        row_index += 1
    
    # Adjust grid to fit in unit square and handle hexagonal offset
    # Adjust the row index for hexagonal offset (alternating row offsets)
    row_index = 0
    for row in grid_rows:
        for col in range(row):
            x = col_start + col * col_spacing
            y = row_start + row_index * row_spacing
            if row_index % 2 == 1:
                y += (row_spacing / 2)
            centers.append([x, y])
            radii.append(base_radius)
        row_index += 1
    
    # Ensure we only have 26 circles
    centers = centers[:26]
    radii = radii[:26]
    
    # Define the objective function to maximize sum of radii
    def objective(radii):
        return -sum(radii)  # Negative because we'll minimize
    
    # Define constraints for non-overlapping and boundary conditions
    def constraints(radii, centers):
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers[i]
                r1 = radii[i]
                x2, y2 = centers[j]
                r2 = radii[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                constraints.append(dist - (r1 + r2) >= -1e-12)
        return constraints
    
    # Initial guess is the base radius for all circles
    initial_guess = np.full(n, base_radius)
    
    # Set bounds for radii (must be non-negative)
    bounds = [(0, 1) for _ in range(n)]
    
    # Optimization
    res = minimize(objective, initial_guess, bounds=bounds, constraints=constraints)

    # Extract the result
    optimal_radii = res.x
    optimal_centers = np.array(centers)

    # Validate the packing
    is_valid, msg = validate_packing(optimal_centers, optimal_radii)
    if not is_valid:
        print(f"Validation failed: {msg}")
        return np.array([]), np.array([]), 0.0

    # Compute the sum of radii
    sum_radii = sum(optimal_radii)

    return optimal_centers, optimal_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
