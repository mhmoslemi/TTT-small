import numpy as np
import cvxpy as cp

def run_packing():
    # Define number of circles
    n = 26

    # Define a hexagonal grid layout with rows and columns
    rows = 6
    cols = 5
    row_spacing = 0.4
    col_spacing = 0.4
    offset = 0.5 * row_spacing

    # Step 1: Generate initial positions using a hexagonal grid
    centers = np.empty((n, 2))
    radii = np.zeros(n)

    # Generate positions with row-wise offsetting
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col * col_spacing
        y = row * row_spacing + (col % 2) * offset
        centers[i] = [x, y]

    # Compute initial radii based on minimum distance to nearest neighbors
    # and to boundaries
    def compute_radii(centers):
        radii = np.zeros_like(centers[:, 0])
        for i in range(n):
            min_dist = np.inf
            for j in range(n):
                if i == j:
                    continue
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < min_dist:
                    min_dist = dist
            # Distance to nearest boundary
            dist_to_left = centers[i, 0]
            dist_to_right = 1 - centers[i, 0]
            dist_to_bottom = centers[i, 1]
            dist_to_top = 1 - centers[i, 1]
            min_boundary_dist = min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top)
            radii[i] = min(0.5 * min_dist, 0.5 * min_boundary_dist)
        return radii

    radii = compute_radii(centers)

    # Optimize radii to maximize sum, with non-overlapping constraints
    # Define variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n, nonneg=True)

    # Define boundaries for the square [0, 1]x[0, 1]
    # For each circle, constraint that it's entirely within the square
    constraints = []
    for i in range(n):
        x_i, y_i = x[i]
        r_i = r[i]
        constraints.append(x_i - r_i >= 0)
        constraints.append(x_i + r_i <= 1)
        constraints.append(y_i - r_i >= 0)
        constraints.append(y_i + r_i <= 1)

    # For each pair of circles, constraint that they don't overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j])

    # Objective: maximize sum of radii
    obj = cp.Maximize(cp.sum(r))

    # Solve the problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, eps=1e-4)

    # Extract the optimal solution
    centers = x.value
    radii = r.value
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
