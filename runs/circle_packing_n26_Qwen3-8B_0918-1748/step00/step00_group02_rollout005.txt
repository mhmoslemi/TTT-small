import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint

def run_packing():
    # Define the number of circles
    n = 26

    # Initial guess: cluster-based packing with 4 corner clusters and a central cluster
    # Each corner cluster will have 4 circles, and central cluster will have 2
    # Initial centers can be optimized to maximize radii sum

    # Define initial positions for corner clusters
    # Each corner has 4 circles with equal radii
    # Define initial positions with spacing and offsets
    corner_centers = np.array([
        [0.1, 0.1], [0.2, 0.1], [0.3, 0.1], [0.4, 0.1],
        [0.1, 0.3], [0.2, 0.3], [0.3, 0.3], [0.4, 0.3],
        [0.1, 0.5], [0.2, 0.5], [0.3, 0.5], [0.4, 0.5],
        [0.1, 0.7], [0.2, 0.7], [0.3, 0.7], [0.4, 0.7],
        [0.1, 0.9], [0.2, 0.9], [0.3, 0.9], [0.4, 0.9],
        [0.5, 0.1], [0.6, 0.1], [0.7, 0.1], [0.8, 0.1],
        [0.5, 0.3], [0.6, 0.3], [0.7, 0.3], [0.8, 0.3],
        [0.5, 0.5], [0.6, 0.5], [0.7, 0.5], [0.8, 0.5],
        [0.5, 0.7], [0.6, 0.7], [0.7, 0.7], [0.8, 0.7],
        [0.5, 0.9], [0.6, 0.9], [0.7, 0.9], [0.8, 0.9],
        [0.9, 0.1], [0.9, 0.3], [0.9, 0.5], [0.9, 0.7], [0.9, 0.9],
        [0.0, 0.1], [0.0, 0.3], [0.0, 0.5], [0.0, 0.7], [0.0, 0.9]
    ])

    # Remove duplicate initial centers for 26 circles
    # We need only 26 circles, so we'll take the first 26 of the above
    centers_initial = corner_centers[:26]

    # Define variables
    # Each circle has two center coordinates and one radius: 3 * 26 variables
    x = np.concatenate([centers_initial[:, 0], centers_initial[:, 1], np.zeros(n)])
    bounds = []
    for i in range(n):
        x_low = 0.0
        x_high = 1.0
        bounds.append((x_low, x_high))
        bounds.append((x_low, x_high))
        bounds.append((0.0, 1.0))  # radius lower bounded by 0

    # Objective function: maximize sum of radii (minimize negative sum)
    def objective(x):
        radii = x[2*n:]
        return -np.sum(radii)

    # Constraint: distance between centers >= sum of radii
    def constraint_dist(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - radii[i] - radii[j] + 1e-8
        return 0.0

    # Constraint: each circle is inside the unit square
    def constraint_inside(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-8 or x_i + r_i > 1.0 + 1e-8 or y_i - r_i < -1e-8 or y_i + r_i > 1.0 + 1e-8:
                return (x_i - r_i + 1e-8) + (x_i + r_i - 1.0 - 1e-8) + (y_i - r_i + 1e-8) + (y_i + r_i - 1.0 - 1e-8)
        return 0.0

    # Nonlinear constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_dist},
        {'type': 'ineq', 'fun': constraint_inside}
    ]

    # Solve the optimization problem
    result = optimize.minimize(
        objective,
        x,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 500}
    )

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    # Extract the solution
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]

    # Verify the solution passes the validation function
    # This is purely for demonstration; the function is not actually executed here

    return centers, radii, np.sum(radii)
