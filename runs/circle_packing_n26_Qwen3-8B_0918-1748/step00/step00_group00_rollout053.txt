import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist

def run_packing():
    def place_hexagonal_layer(num_layers, initial_radius):
        centers = []
        radii = []
        layer_radii = [initial_radius * (1 + 0.1 * i) for i in range(num_layers)]
        center = np.array([0.5, 0.5])
        centers.append(center)
        radii.append(layer_radii[0])
        for layer_idx in range(1, num_layers):
            layer_radius = layer_radii[layer_idx]
            angle = 0
            for _ in range(6):
                angle += 2 * np.pi / 6
                dx = layer_radius * np.cos(angle)
                dy = layer_radius * np.sin(angle)
                pos = center + np.array([dx, dy])
                if 0 <= pos[0] <= 1 and 0 <= pos[1] <= 1:
                    centers.append(pos)
                    radii.append(layer_radius)
        return np.array(centers), np.array(radii)

    def compute_max_circle_gap(centers, radii, square_size=1.0):
        def objective(x, radii, centers, square_size):
            x, y = x
            r = 0
            for i in range(len(centers)):
                dx = x - centers[i][0]
                dy = y - centers[i][1]
                dist = np.sqrt(dx**2 + dy**2)
                r = min(r, dist - radii[i])
            return -r

        result = minimize(
            lambda x: objective(x, radii, centers, square_size),
            x0=np.array([0.5, 0.5]),
            bounds=[(0, 1), (0, 1)],
            method="L-BFGS-B"
        )
        if result.success:
            x, y = result.x
            r = -result.fun
            return np.array([x, y]), r
        else:
            raise ValueError("Failed to compute max circle gap")

    def optimize_radii(centers, num_circles, square_size=1.0):
        def objective(radii):
            return -np.sum(radii)

        def constraints(radii):
            constraints = []
            for i in range(num_circles):
                constraints.append((radii[i], 0, np.inf))
            for i in range(num_circles):
                for j in range(i + 1, num_circles):
                    dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    constraints.append((radii[i] + radii[j], 0, dist))
            return constraints

        bounds = Bounds(0, np.inf)
        result = minimize(
            objective,
            x0=np.ones(num_circles) * 0.01,
            constraints=constraints,
            bounds=bounds,
            method="SLSQP"
        )
        if result.success:
            return result.x
        else:
            raise ValueError("Optimization failed")

    def check_boundary_constraints(centers, radii):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        return True

    def compute_magnitude(vec):
        return np.sqrt(np.sum(vec**2))

    # Step 1: Create base hexagonal layer (25 circles)
    centers, radii = place_hexagonal_layer(5, 0.03)
    if len(centers) > 25:
        centers = centers[:25]
        radii = radii[:25]

    # Step 2: Optimize radii for the base layer
    try:
        opt_radii = optimize_radii(centers, len(centers))
        radii = opt_radii
    except:
        raise ValueError("Optimization failed")

    # Step 3: Find optimal gap for 26th circle
    new_center, new_radius = compute_max_circle_gap(centers, radii)
    if not check_boundary_constraints([new_center], [new_radius]):
        raise ValueError("26th circle would exceed boundary")

    # Step 4: Validate and finalize
    final_centers = np.vstack((centers, new_center))
    final_radii = np.concatenate((radii, [new_radius]))
    if not check_boundary_constraints(final_centers, final_radii):
        raise ValueError("Final configuration exceeds boundary")

    return final_centers, final_radii, np.sum(final_radii)
