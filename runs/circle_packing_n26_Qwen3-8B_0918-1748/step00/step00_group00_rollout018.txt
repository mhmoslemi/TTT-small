import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Initial configuration: 2x2 grid of large circles at corners, and hexagonal grid in center
    # First 4 circles at the corners with small radii
    # Remaining 22 circles in a hexagonal grid
    centers = []
    radii = []
    for i in range(4):
        # Place 4 circles at corners with small radii
        if i == 0:
            center = np.array([0.1, 0.1])
            radius = 0.05
        elif i == 1:
            center = np.array([0.9, 0.1])
            radius = 0.05
        elif i == 2:
            center = np.array([0.9, 0.9])
            radius = 0.05
        elif i == 3:
            center = np.array([0.1, 0.9])
            radius = 0.05
        centers.append(center)
        radii.append(radius)
    
    # Hexagonal grid in the middle
    # Create 5 rows of 5 circles
    for row in range(5):
        for col in range(5):
            x = 0.2 + col * (0.25)
            y = 0.2 + row * (0.25)
            if row % 2 == 0:
                x += 0.125
            center = np.array([x, y])
            radius = 0.1
            centers.append(center)
            radii.append(radius)
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Optimization parameters
    bounds = Bounds([0.0, 0.0], [1.0, 1.0])
    constraints = []
    
    # Constraint to ensure circles are inside the square
    def in_square(x):
        return np.all(np.abs(x) <= 1.0)
    
    # Constraint to ensure circles are non-overlapping
    def non_overlapping(x, y, r, s):
        distance = np.sqrt((x - y)**2 + (r - s)**2)
        return distance >= r + s - 1e-8
    
    # Create a function that converts the parameters to the required format
    def objective(params):
        # Unpack the parameters
        flat_centers = params[:2 * len(centers)]
        flat_radii = params[2 * len(centers):]
        centers_opt = flat_centers.reshape((len(centers), 2))
        radii_opt = flat_radii
        # Compute total radius sum
        total_radius = np.sum(radii_opt)
        return -total_radius  # Negative for minimization
    
    # Create constraints based on non_overlapping function
    def constraint_func(params):
        flat_centers = params[:2 * len(centers)]
        flat_radii = params[2 * len(centers):]
        centers_opt = flat_centers.reshape((len(centers), 2))
        radii_opt = flat_radii
        # Check non-overlapping for all pairs
        for i in range(len(centers_opt)):
            for j in range(i + 1, len(centers_opt)):
                x1, y1 = centers_opt[i]
                r1 = radii_opt[i]
                x2, y2 = centers_opt[j]
                r2 = radii_opt[j]
                if np.sqrt((x1 - x2)**2 + (y1 - y2)**2) < r1 + r2 - 1e-8:
                    return -np.inf
        # Check boundary constraints
        for i in range(len(centers_opt)):
            x, y = centers_opt[i]
            r = radii_opt[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return -np.inf
        return 0
    
    # Initial guess: centers as placed and radii as initial values
    initial_guess = np.concatenate([centers.flatten(), radii])
    bounds = Bounds([0.0, 0.0], [1.0, 1.0])  # For centers
    bounds_radii = Bounds([0.0], [1.0])  # For radii
    
    # Create a helper function to pack
    def pack(params):
        flat_centers = params[:2 * len(centers)]
        flat_radii = params[2 * len(centers):]
        centers_opt = flat_centers.reshape((len(centers), 2))
        radii_opt = flat_radii
        return centers_opt, radii_opt
    
    # Optimization: use simulated annealing to maximize total radius
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_func}]
    )
    
    centers_opt, radii_opt = pack(result.x)
    
    total_radius = np.sum(radii_opt)
    
    return centers_opt, radii_opt, total_radius
