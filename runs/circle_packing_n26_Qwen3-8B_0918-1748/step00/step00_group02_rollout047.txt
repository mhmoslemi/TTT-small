```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds
from itertools import combinations

def run_packing():
    # Initial grid-based packing with 5 rows and 5 columns (total 25 circles) with some adjustments for 26
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Start with a grid-based layout
    # First 25 circles in a grid layout (5x5)
    for i in range(5):
        for j in range(5):
            x = j / 4
            y = i / 4
            centers[i * 5 + j] = [x, y]
    
    # Add one more circle at the bottom right corner
    centers[25] = [0.95, 0.95]
    
    # Initial radii - start with equal radii adjusted to fit first 25 circles
    # The max radius for 25 circles in a 5x5 grid would be about 0.15
    initial_radius = 0.15
    radii[:25] = initial_radius
    radii[25] = initial_radius  # The 26th circle starts with the same radius
    
    # Function to calculate distance between two points
    def dist(p1, p2):
        return np.hypot(p1[0] - p2[0], p1[1] - p2[1])
    
    # Objective function to maximize the sum of radii
    def objective(x):
        # x is a flattened array [r0, r1, ..., r25]
        radii[:] = x
        return -np.sum(radii)  # Negative for minimization
    
    # Constraints: for each pair of circles, distance >= r_i + r_j
    bounds = Bounds([0.0] * num_circles, [1.0] * num_circles)
    
    # Create constraint for each circle
    cons = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def constraint_func(x):
                r_i = x[i]
                r_j = x[j]
                return dist(centers[i], centers[j]) - r_i - r_j
            cons.append({'type': 'ineq', 'fun': constraint_func})
    
    # Optimization
    result = opt.minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8)
    
    # Extract the optimized centers and radii
    centers_opt = centers.copy()
    radii_opt = result.x
    
    # Final validation step (re-check for safety)
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"
    
    valid, msg = validate_packing(centers_opt, radii_opt)
    
    if not valid:
        raise ValueError(f"Packing is invalid: {msg}")
    
    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
```
