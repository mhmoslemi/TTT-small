import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from functools import lru_cache

def run_packing():
    # Initial grid dimensions (rows, cols)
    rows = 6
    cols = 5
    # Initial radii for grid circles
    initial_radii = np.full(rows * cols, 0.1)
    # Initial grid spacing
    grid_spacings = [0.25, 0.25]
    # Initial grid positions
    centers = np.zeros((rows * cols, 2))
    for i in range(rows):
        for j in range(cols):
            x = j * grid_spacings[0] + (grid_spacings[0] / 2)
            y = i * grid_spacings[1] + (grid_spacings[1] / 2)
            centers[i * cols + j] = [x, y]
    # Add corner circles
    corner_radii = [0.05, 0.05, 0.05, 0.05]
    corner_centers = np.array([
        [0.1, 0.1],
        [0.1, 0.9],
        [0.9, 0.9],
        [0.9, 0.1]
    ])
    centers = np.vstack((centers, corner_centers))
    radii = np.concatenate((initial_radii, corner_radii))
    
    def objective(radii_flat):
        return -np.sum(radii_flat)
    
    def constraints(radii_flat):
        constraints = []
        for i in range(len(radii_flat)):
            for j in range(i + 1, len(radii_flat)):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return False
        for i in range(len(radii_flat)):
            x, y = centers[i]
            r = radii_flat[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e8 or y + r > 1 + 1e-8:
                return False
        return True
    
    bounds = []
    for _ in range(len(radii)):
        bounds.append((0.0, 1.0))
    
    initial_guess = radii.copy()
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'callback', 'fun': constraints}
    )
    
    if not result.success:
        raise RuntimeError("Optimization failed")
    
    radii = result.x
    centers = np.vstack((centers[:len(radii)], centers[len(radii):]))
    
    if not validate_packing(centers, radii)[0]:
        raise RuntimeError("Validation failed")
    
    total_sum = np.sum(radii)
    return centers, radii, total_sum

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"
