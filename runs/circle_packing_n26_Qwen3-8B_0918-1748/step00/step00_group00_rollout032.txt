from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():

    grid_size = 5

    grid_points = np.array(np.meshgrid(grid, grid)).T.reshape(-1, 2)

    dist_matrix = squareform(pdist(centers))
    radii = np.array([np.min(dist_matrix[i]) * 0.4 for i in range(len(centers))])

    bounds = Bounds([0, 0], [1, 1])

        centers[:] = x[:2*len(centers)].reshape((len(centers), 2))
        radii[:] = x[2*len(centers):]
        return -np.sum(radii)  # Minimize negative sum to maximize
    
    def constraint(x):
        centers[:] = x[:2*len(centers)].reshape((len(centers), 2))
        radii[:] = x[2*len(centers):]
        
        # Check if all circles are inside the square
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return -np.inf

        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):

                dist = np.sqrt(dx**2 + dy**2)

                    return -np.inf
        
        return np.sum(radii)
    
    result = minimize(objective, x0=np.concatenate((centers.ravel(), radii)), bounds=Bounds([0, 0], [1, 1]), method='SLSQP', constraints=[{'type': 'ineq', 'fun': constraint}])
