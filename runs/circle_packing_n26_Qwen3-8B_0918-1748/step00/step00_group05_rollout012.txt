import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def hexagonal_grid(n):
    """
    Generate a hexagonal grid of n circles. Assumes a pattern of rows with offsets.
    """
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    grid = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * (i % 2)
            y = i
            grid.append((x, y))
    return np.array(grid)

def run_packing():
    # Initial placement using hexagonal grid
    centers = hexagonal_grid(26)
    # Normalization to unit square
    centers = (centers - np.min(centers)) / (np.max(centers) - np.min(centers))

    # Objective function for optimization: minimize negative of sum of radii
    def objective(x):
        r = x[26:]
        return -np.sum(r)

    # Constraints: distance between centers >= radii[i] + radii[j]
    def constraint_func(x):
        r = x[26:]
        centers = x[:26].reshape((26, 2))
        distances = np.sqrt(np.sum((centers[:, np.newaxis, :] - centers[np.newaxis, :, :]) ** 2, axis=2))
        return distances - (r[np.newaxis, :] + r[:, np.newaxis]) + 1e-8

    # Create bounds for centers and radii
    bounds = []
    for i in range(26):
        bounds.append((0.0, 1.0))  # for x-coordinate
        bounds.append((0.0, 1.0))  # for y-coordinate
    for i in range(26):
        bounds.append((0.0, 1.0))  # for radius

    # Initial guess: assign radius to be 0.05 for all circles
    x0 = np.array([0.5] * 52)
    x0[26:] = 0.05

    # Create constraints of the form: f(x) >= 0
    # Use a quadratic constraint for distance >= sum of radii
    # For the purpose of this simple implementation, we'll use a direct approach

    # Optimization
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        method="SLSQP",
        constraints=[
            {
                'type': 'ineq',
                'fun': lambda x: np.min(np.sqrt(np.sum((x[:26].reshape((26, 2))[:, np.newaxis, :] - x[:26].reshape((26, 2))[np.newaxis, :, :]) ** 2, axis=2)) - (x[26:] + x[26:][:, np.newaxis]).min() + 1e-8)
            }
        ]
    )

    # Extract centers and radii
    centers = result.x[:26].reshape((26, 2))
    radii = result.x[26:]
    sum_radii = np.sum(radii)

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
