import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Define an initial hexagonal arrangement with 26 circles
    # We'll approximate a hexagonal packing with 7 rows (4, 4, 4, 3, 3, 3, 3)
    # We'll try to fit these circles in a grid, adjust for the unit square, and then optimize radii

    # Initial guess for circle positions
    # We'll use a hexagonal pattern with row offsets
    num_circles = 26
    rows = [4, 4, 4, 3, 3, 3, 3]
    cols = [4, 4, 4, 3, 3, 3, 3]

    # Generate initial coordinates
    centers = []
    for i, (row_count, col_count) in enumerate(zip(rows, cols)):
        row_offset = i * np.sqrt(3)
        for j in range(col_count):
            col_offset = j - (col_count - 1) / 2
            x = col_offset * 2 * 0.5 + 0.5
            y = row_offset + (i % 2) * (np.sqrt(3) * 0.5)
            centers.append([x, y])
    centers = np.array(centers)

    # Initial guess for radii
    radii_initial = np.ones(num_circles) * 0.1  # start with small radii

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # minimize negative sum to maximize sum

    # Constraints:
    # 1. Each circle's center and radius must be within [0,1] (we'll handle this with bounds)
    # 2. Distance between centers >= sum of radii

    # Define bounds for x, y, and radii
    bounds = []
    for i in range(num_circles):
        bounds.append((0.0, 1.0))  # for x
        bounds.append((0.0, 1.0))  # for y
        bounds.append((0.0, 1.0))  # for radius

    # Define constraints for non-overlapping circles
    constraints = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            def constraint_func(x):
                idx_i = i * 3
                idx_j = j * 3
                x_i, y_i, r_i = x[idx_i], x[idx_i + 1], x[idx_i + 2]
                x_j, y_j, r_j = x[idx_j], x[idx_j + 1], x[idx_j + 2]
                distance = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                return distance - (r_i + r_j)
            constraints.append({
                'type': 'ineq',
                'fun': constraint_func
            })

    # Prepare variables for optimization: x, y, r for each circle
    variables = np.concatenate([centers.flatten(), radii_initial])

    # Define the optimization problem
    result = opt.minimize(
        objective,
        variables,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-8, 'disp': False}
    )

    # Extract centers and radii from result
    variables = result.x
    centers_opt = variables[:num_circles * 2].reshape((num_circles, 2))
    radii_opt = variables[num_circles * 2:]

    # Final validation
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii


def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
