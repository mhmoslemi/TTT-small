import numpy as np
from scipy.optimize import minimize, Bounds

def run_packing():

    # Total number of circles

    # We'll use a staggered grid
    # We estimate radius based on spacing
    # Let's define a grid with m rows and columns along the square
    # Heuristically, we'll start with a grid of 6 rows with 4, 5, 5, 5, 5, 4 circles

            if (row % 2) == 1:
                col -= 1
                row += 1
            else:
                row += 1
        return points

    # Try to fit the 26 circles in a way that they can be sized up

    # We'll minimize the negative of the sum of radii
    # Subject to constraints:
    # 1. All circles are within the square
    # 2. No circle extends beyond the square
    # 3. Distance between centers is >= r_i + r_j
    # We'll use a constrained optimization with scipy.optimize.minimize

    # Define the objective function
    def objective(x):
        # x is a vector of (centers, radii)

        centers = x[:n * 2].reshape(n, 2)
        radii = x[n * 2:]
        total = np.sum(radii)

    def constraint1(x):
        # Constraint: All circles are within the square

        for i in range(n):
            x_i = centers[i, 0]
            y_i = centers[i, 1]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:

        return 0

    def constraint2(x):
        # Constraint: Distance between centers is >= r_i + r_j
        centers = x[:n * 2].reshape(n, 2)
        radii = x[n * 2:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]

                if dist < radii[i] + radii[j] - 1e-12:

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
