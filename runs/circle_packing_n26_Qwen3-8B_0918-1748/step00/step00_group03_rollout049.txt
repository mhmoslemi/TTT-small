import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Initial hexagonal grid with 4 rows
    rows = 4
    cols_per_row = 6
    centers = []
    radii = []
    
    # Set up the initial hexagonal grid
    for row in range(rows):
        for col in range(cols_per_row):
            x = col / (cols_per_row - 1) if col < cols_per_row - 1 else 1.0
            y = row / (rows - 1) if row < rows - 1 else 1.0
            # Apply hexagonal offset
            if row % 2 == 1:
                x = (col + 0.5) / (cols_per_row - 1)
            centers.append([x, y])
            # Initial radius based on grid spacing
            if row == 0 and col == 0:
                radii.append(0.1)
            else:
                radii.append(0.05)
    
    # Add remaining circles in corners and edges
    corner_circles = [(0.1, 0.1, 0.05), (0.9, 0.1, 0.05), (0.1, 0.9, 0.05), (0.9, 0.9, 0.05)]
    edge_circles = [(0.5, 0.1, 0.05), (0.5, 0.9, 0.05)]
    
    for x, y, r in corner_circles + edge_circles:
        centers.append([x, y])
        radii.append(r)
    
    # Convert lists to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Objective function to maximize sum of radii (to minimize negative)
    def objective(x):
        # x has shape (2*26) = 52 elements: centers[0][0], centers[0][1], ..., centers[25][1]
        # Reconstruct centers and radii
        flat_centers = x[:2*len(centers)].reshape((len(centers), 2))
        flat_radii = x[2*len(centers):]
        return -np.sum(flat_radii)  # Maximize sum by minimizing negative
    
    # Constraint: non-overlapping circles
    def constraint(x):
        flat_centers = x[:2*len(centers)].reshape((len(centers), 2))
        flat_radii = x[2*len(centers):]
        dists = squareform(pdist(flat_centers))
        for i in range(len(centers)):
            for j in range(i+1, len(centers)):
                if dists[i,j] < flat_radii[i] + flat_radii[j] - 1e-8:
                    return dists[i,j] - (flat_radii[i] + flat_radii[j] - 1e-8)
        return 0.0
    
    # Constraint: circles within unit square
    def square_constraint(x):
        flat_centers = x[:2*len(centers)].reshape((len(centers), 2))
        flat_radii = x[2*len(centers):]
        for i in range(len(centers)):
            x_i, y_i = flat_centers[i]
            r_i = flat_radii[i]
            if x_i - r_i < -1e-8 or x_i + r_i > 1 + 1e-8 or y_i - r_i < -1e-8 or y_i + r_i > 1 + 1e-8:
                return 1.0
        return 0.0
    
    # Bounds: centers in [0,1]^2, radii >= 0
    bounds = []
    for _ in range(len(centers)):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
    for _ in range(len(radii)):
        bounds.extend([(0.0, 0.5)])
    
    # Initial guess
    x0 = np.concatenate([centers.flatten(), radii])
    
    # Optimization
    cons = [
        {'type': 'ineq', 'fun': constraint},
        {'type': 'ineq', 'fun': square_constraint}
    ]
    
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 1000})
    
    # Extract centers and radii
    flat_centers = result.x[:2*len(centers)].reshape((len(centers), 2))
    flat_radii = result.x[2*len(centers):]
    
    return flat_centers, flat_radii, np.sum(flat_radii)
