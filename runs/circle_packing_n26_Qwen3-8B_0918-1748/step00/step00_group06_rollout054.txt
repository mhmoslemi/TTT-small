import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for positions: hexagonal packing with 5 rows
    n = 26
    positions = np.zeros((n, 2))
    radii = np.zeros(n)
    # Generate a hexagonal packing in a square

    # First row with 6 circles
    for i in range(6):
        positions[i, 0] = i / 5
        positions[i, 1] = 0.1

    # Second row (offset)
    for i in range(6):
        positions[6 + i, 0] = (i + 0.5) / 5
        positions[6 + i, 1] = 0.35

    # Third row
    for i in range(5):
        positions[12 + i, 0] = i / 4.5
        positions[12 + i, 1] = 0.6

    # Fourth row (offset)
    for i in range(5):
        positions[17 + i, 0] = (i + 0.5) / 4.5
        positions[17 + i, 1] = 0.85

    # Fifth row
    for i in range(4):
        positions[22 + i, 0] = i / 3.5
        positions[22 + i, 1] = 0.95

    # Add extra circles in corners or tight spaces
    positions[23, 0], positions[23, 1] = 0.05, 0.05
    positions[24, 0], positions[24, 1] = 0.95, 0.05
    positions[25, 0], positions[25, 1] = 0.95, 0.95

    # Initial guess for radii
    for i in range(n):
        if i < 6:
            radii[i] = 0.1
        elif 6 <= i < 12:
            radii[i] = 0.1
        elif 12 <= i < 17:
            radii[i] = 0.1
        elif 17 <= i < 22:
            radii[i] = 0.1
        else:
            radii[i] = 0.05

    # Define constraints
    def constraint1(x):
        # x = [pos[0], pos[1], radii]
        positions_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        for i in range(n):
            x_i, y_i = positions_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return 0
        return 1

    def constraint2(x):
        positions_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        for i in range(n):
            for j in range(i+1, n):
                dx = positions_flat[i, 0] - positions_flat[j, 0]
                dy = positions_flat[i, 1] - positions_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return 0
        return 1

    # Define objective function to maximize sum of radii
    def objective(x):
        positions_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        return -np.sum(radii_flat)

    # Bounds: positions in [0,1], radii >= 0
    bounds = []
    for _ in range(n*2):
        bounds.append((0.0, 1.0))
    for _ in range(n):
        bounds.append((0.0, 0.5))  # max radius can't exceed 0.5 in square

    # Initial guess
    x0 = np.zeros(n*2 + n)
    x0[:n*2] = positions.flatten()
    x0[n*2:] = radii

    # Optimization with constraints
    cons = [{'type': 'ineq', 'fun': constraint1},
           {'type': 'ineq', 'fun': constraint2}]

    result = opt.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)

    centers = result.x[:n*2].reshape(n, 2)
    radii = result.x[n*2:]

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
