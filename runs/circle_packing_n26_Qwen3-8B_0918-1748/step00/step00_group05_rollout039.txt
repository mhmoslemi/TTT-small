import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Number of circles
    n = 26

    # Define initial cluster layout: 6 clusters with 4 circles each and 2 clusters with 1 circle
    # Total: 6*4 + 2 = 26
    cluster_config = [
        (0.2, 0.2, 0.15),
        (0.8, 0.2, 0.15),
        (0.2, 0.8, 0.15),
        (0.8, 0.8, 0.15),
        (0.5, 0.5, 0.15),
        (0.5, 0.3, 0.15),
        (0.5, 0.7, 0.15),
        (0.3, 0.5, 0.15),
        (0.7, 0.5, 0.15),
        (0.5, 0.1, 0.15),
        (0.5, 0.9, 0.15),
        (0.1, 0.5, 0.15),
        (0.9, 0.5, 0.15),
        (0.4, 0.4, 0.15),
        (0.6, 0.6, 0.15),
        (0.4, 0.6, 0.15),
        (0.6, 0.4, 0.15),
        (0.4, 0.3, 0.15),
        (0.6, 0.3, 0.15),
        (0.4, 0.7, 0.15),
        (0.6, 0.7, 0.15),
        (0.3, 0.4, 0.15),
        (0.7, 0.6, 0.15),
        (0.3, 0.6, 0.15),
        (0.7, 0.4, 0.15),
        (0.6, 0.6, 0.15),
        (0.4, 0.4, 0.15)
    ]

    centers = np.array([cluster_config[i][:-1] for i in range(n)])
    radii = np.array([cluster_config[i][-1] for i in range(n)])

    # Optimize radii to maximize sum while satisfying constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_overlap(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return dist_matrix[i, j] - radii[i] - radii[j] + 1e-8
        return 0.0

    def constraint_boundary(radii):
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return (x - radii[i]) + (x + radii[i] - 1) + (y - radii[i]) + (y + radii[i] - 1)
        return 0.0

    bounds = Bounds(lb=0, ub=np.inf)
    overlap_constraint = NonlinearConstraint(constraint_overlap, -np.inf, 0)
    boundary_constraint = NonlinearConstraint(constraint_boundary, -np.inf, 0)

    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[overlap_constraint, boundary_constraint],
        method='SLSQP'
    )

    centers = np.array([cluster_config[i][:-1] for i in range(n)])
    radii = result.x
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
