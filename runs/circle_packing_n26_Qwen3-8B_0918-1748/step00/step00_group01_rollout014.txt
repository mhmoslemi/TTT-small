import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Define initial configuration as a hexagonal grid with adjustments for 26 circles
    # We will use a hybrid approach with 5 rows of 5, 5, 5, 5, and 6 circles
    # Initial centers are generated with hexagonal packing

    # Generate initial centers
    def generate_hex_grid(n, max_x, max_y, hex_rows, hex_cols):
        centers = []
        for row in range(hex_rows):
            for col in range(hex_cols):
                x = col + (row % 2) * 0.5
                y = row
                centers.append([x, y])
        return np.array(centers)

    # Initial layout: 5 rows with 5, 5, 5, 5, and 6 circles
    initial_centers = []
    for r in range(5):
        if r < 4:
            initial_centers.extend([[col, r] for col in range(5)])
        else:
            initial_centers.extend([[col, 4] for col in range(6)])
    initial_centers = np.array(initial_centers)

    # Scale the initial centers to fit in the unit square
    initial_centers[:, 0] /= 10
    initial_centers[:, 1] /= 10
    initial_centers[:, 1] += 0.5

    n = 26

    # Define radii as all the same initially for easy starting point
    initial_radii = np.ones(n) * 0.05

    # Optimization function to maximize the sum of radii
    def objective(x):
        return -np.sum(x[n:])  # Negative because we use minimize

    # Define bounds for each circle's position and radius
    bounds = []
    for i in range(n):
        # Center coordinates must be within [0, 1]
        bounds.append((0, 1))
        bounds.append((0, 1))
        # Radius must be non-negative
        bounds.append((0, 1))  # We use 1 as an upper bound for radius

    # Define constraints: distance between centers >= sum of radii
    def constraint_func(x):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - (radii[i] + radii[j] - 1e-8)
        return 0

    # Initial guess: centers and radii
    x0 = np.concatenate([initial_centers.reshape(-1), initial_radii])

    # Define constraints and bounds
    cons = [{'type': 'ineq', 'fun': constraint_func}]
    bounds = Bounds(x0, x0)

    # Run the optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-10)

    # Extract the optimized centers and radii
    centers = result.x[:n*2].reshape(n, 2)
    radii = result.x[n*2:]

    # Check if the solution is valid
    if not validate_packing(centers, radii)[0]:
        raise ValueError("Optimization resulted in invalid packing.")

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
