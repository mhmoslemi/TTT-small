import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Define the number of circles
    n = 26

    # Define initial positions for corner circles (offset from square boundaries)
    corner_radii = [0.15, 0.15, 0.15, 0.15]
    corner_centers = np.array([
        [0.15, 0.15],  # bottom-left
        [0.85, 0.15],  # bottom-right
        [0.15, 0.85],  # top-left
        [0.85, 0.85]   # top-right
    ])

    # Define initial positions for interior circles in a hexagonal grid
    # We'll create a grid of 10 rows with 5 circles in each row alternately offset
    interior_centers = np.empty((22, 2))
    r = 0.1  # initial radius for interior circles
    x_start = 0.2
    y_start = 0.2
    rows = 10
    cols = 5

    for i in range(rows):
        for j in range(cols):
            x = x_start + j * (1.5 * r)
            y = y_start + i * (np.sqrt(3) * r)
            if i % 2 == 0:
                x = x_start + j * (1.5 * r)
            else:
                x = x_start + (j + 0.5) * (1.5 * r)
            interior_centers[i * cols + j] = [x, y]

    # Combine corner and interior centers
    centers = np.vstack((corner_centers, interior_centers))
    radii = np.zeros(n)
    radii[:4] = corner_radii
    radii[4:] = np.full(22, r)

    # Optimization parameters
    bounds = Bounds([], [])

    # Define the optimization objective: maximize sum of radii
    def objective(x):
        # x contains all parameters: first 4*2 for corner centers, then 22*2 for interior centers, then 26 radii
        # Reshape to get centers and radii
        # First 4*2 = 8 elements: corners
        # Next 22*2 = 44 elements: interior centers
        # Last 26 elements: radii
        num_corners = 4
        num_interior = 22
        corner_centers = x[:num_corners * 2].reshape(num_corners, 2)
        interior_centers = x[num_corners * 2 : (num_corners + num_interior) * 2].reshape(num_interior, 2)
        radii = x[(num_corners + num_interior) * 2 :]

        # Centers and radii
        centers = np.vstack((corner_centers, interior_centers))
        radii = radii

        # Compute total sum of radii
        total_sum = np.sum(radii)
        return -total_sum  # Minimize negative to maximize

    # Define constraints: non-overlapping and boundary constraints
    def constraint_func(x):
        # x contains all parameters: first 4*2 for corner centers, then 22*2 for interior centers, then 26 radii
        # Reshape to get centers and radii
        num_corners = 4
        num_interior = 22
        corner_centers = x[:num_corners * 2].reshape(num_corners, 2)
        interior_centers = x[num_corners * 2 : (num_corners + num_interior) * 2].reshape(num_interior, 2)
        radii = x[(num_corners + num_interior) * 2 :]

        # Centers and radii
        centers = np.vstack((corner_centers, interior_centers))
        radii = radii

        # Check boundary constraints
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12
                or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return -1  # Violation

        # Check non-overlapping constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1  # Violation

        return 0

    # Combine constraints into a single function
    def constraint(x):
        return constraint_func(x)

    cons = {'type': 'ineq', 'fun': constraint_func}

    # Initial bounds for all variables
    # Centers: use initial positions as bounds
    x_low = np.full(n * 2, 0.0)
    x_high = np.full(n * 2, 1.0)
    bounds = Bounds(x_low, x_high)

    # Initial guess for x: centers and radii
    x0 = np.zeros(n * 2 + n)
    x0[:n * 2] = np.ravel(centers)
    x0[n * 2:] = radii

    # Run optimization
    result = optimize.minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 100})

    # Extract the optimized centers and radii
    optimized_x = result.x
    centers = optimized_x[:n * 2].reshape(n, 2)
    radii = optimized_x[n * 2:]

    # Ensure non-negative radii
    radii = np.maximum(radii, 0)

    # Final validation (optional: remove for performance)
    # validity, message = validate_packing(centers, radii)
    # if not validity:
    #     raise ValueError(message)

    return (centers, radii, np.sum(radii))
