import numpy as np
from scipy.optimize import minimize
from itertools import combinations

def run_packing():
    # Define initial guess: dense hexagonal packing
    def initial_guess():
        n = 26
        centers = []
        radii = []
        for i in range(n):
            row = i // 4
            col = i % 4
            x = col * 0.25 + 0.125
            y = row * 0.25 + 0.125
            centers.append([x, y])
            radii.append(0.05)
        return np.array(centers), np.array(radii)

    centers, radii = initial_guess()

    # Parameters for optimization
    bounds = [(0, 1)] * 2 * 26
    bounds += [(0, 1)] * 26

    # Objective function to maximize sum of radii
    def objective(x):
        new_centers = x[:2*26].reshape(26, 2)
        new_radii = x[2*26:]
        return -np.sum(new_radii)

    # Constraint: circle is inside the unit square
    def constraint_inside(x):
        new_centers = x[:2*26].reshape(26, 2)
        new_radii = x[2*26:]
        for i in range(26):
            x_i, y_i = new_centers[i]
            r_i = new_radii[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or
                y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return -1
        return 0

    # Constraint: circles do not overlap
    def constraint_overlap(x):
        new_centers = x[:2*26].reshape(26, 2)
        new_radii = x[2*26:]
        for i, j in combinations(range(26), r=2):
            dx = new_centers[i, 0] - new_centers[j, 0]
            dy = new_centers[i, 1] - new_centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < new_radii[i] + new_radii[j] - 1e-12:
                return -1
        return 0

    # Combine constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_inside},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Initial guess with small perturbations and radii
    x0 = np.concatenate([centers.flatten(), radii])

    # Optimization
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
    )

    # Extract result
    centers = result.x[:2*26].reshape(26, 2)
    radii = result.x[2*26:]
    sum_radii = np.sum(radii)

    # Final validation
    # (Assuming validate_packing is an external utility, we assume it's used elsewhere and skip it here)
    return centers, radii, sum_radii
