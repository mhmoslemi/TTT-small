import numpy as np
from scipy.optimize import minimize
from scipy.optimize.slsqpb import slsqp

def run_packing():
    # Define the number of circles
    n = 26

    # Initial guess: grid-based packing for 4 rows and 6 columns (with some adjustments)
    initial_centers = np.zeros((n, 2))
    initial_radii = np.zeros(n)

    # Grid dimensions
    rows = 4
    cols = 6
    total = rows * cols

    # Initial spacing
    dx = 1.0 / (cols + 1)
    dy = 1.0 / (rows + 1)

    for i in range(rows):
        for j in range(cols):
            idx = i * cols + j
            x = j * dx + dx / 2
            y = i * dy + dy / 2
            initial_centers[idx] = [x, y]
            initial_radii[idx] = 0.05  # Initial guess for radius

    # Objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x[1:])  # Negative because we use minimize

    # Constraints function
    def constraints(x):
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]) >= -1e-12)
        return constraints

    # We'll use a hybrid approach: start with the initial guess, then perform optimization
    # Convert radii to a parameter that can be optimized
    x0 = np.concatenate((initial_centers.reshape(-1), initial_radii))

    # Set up bounds: centers must be within [0,1]^2, radii must be non-negative
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1) for _ in range(2)])  # bounds for x and y
        bounds.append((0, 1))  # bounds for radii

    # Define the optimization problem
    result = slsqp(x0, objective, constraints, bounds)

    # Extract centers and radii
    centers = result["x"][:2 * n].reshape(n, 2)
    radii = result["x"][2 * n:]

    # Validate the packing to make sure it meets all constraints
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing is invalid: {message}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
