import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial configuration: 4 boundary rows, 24 circles total
    # 4 circles at corners, 4x6 grid of circles along the boundary
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    sum_radii = 0.0
    
    # Place 4 corner circles
    for i in range(4):
        if i == 0:
            centers[i] = [0, 1]
            radii[i] = 0.1
        elif i == 1:
            centers[i] = [1, 1]
            radii[i] = 0.1
        elif i == 2:
            centers[i] = [1, 0]
            radii[i] = 0.1
        else:
            centers[i] = [0, 0]
            radii[i] = 0.1
    sum_radii = sum(radii)
    
    # Place 6 circles along top and bottom edges
    # Top edge: 3 circles at x = 0.25, 0.5, 0.75
    for i in range(3):
        centers[4 + i] = [i * 0.25 + 0.125, 1]
        radii[4 + i] = 0.15
    sum_radii += sum(radii[4:7])
    
    # Bottom edge: 3 circles at x = 0.25, 0.5, 0.75
    for i in range(3):
        centers[7 + i] = [i * 0.25 + 0.125, 0]
        radii[7 + i] = 0.15
    sum_radii += sum(radii[7:10])
    
    # Left edge: 3 circles at y = 0.25, 0.5, 0.75
    for i in range(3):
        centers[10 + i] = [0, i * 0.25 + 0.125]
        radii[10 + i] = 0.15
    sum_radii += sum(radii[10:13])
    
    # Right edge: 3 circles at y = 0.25, 0.5, 0.75
    for i in range(3):
        centers[13 + i] = [1, i * 0.25 + 0.125]
        radii[13 + i] = 0.15
    sum_radii += sum(radii[13:16])
    
    # Place inner circles in a hexagonal grid, arranged in concentric rings
    # First ring: 6 circles
    for i in range(6):
        angle = 2 * np.pi / 6 * i
        x = 0.5 + 0.3 * np.cos(angle)
        y = 0.5 + 0.3 * np.sin(angle)
        centers[16 + i] = [x, y]
        radii[16 + i] = 0.18
    sum_radii += sum(radii[16:22])
    
    # Second ring: 12 circles
    for i in range(12):
        angle = 2 * np.pi / 12 * i
        x = 0.5 + 0.52 * np.cos(angle)
        y = 0.5 + 0.52 * np.sin(angle)
        centers[22 + i] = [x, y]
        radii[22 + i] = 0.15
    sum_radii += sum(radii[22:34])
    
    # Optimization to refine the packing
    def objective(radii_params):
        return -sum(radii_params)
    
    def constraints(radii_params):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii_params[i] + radii_params[j] - 1e-12:
                    return np.inf
        for i in range(n):
            x, y = centers[i]
            r = radii_params[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return np.inf
        return 0
    
    bounds = [(0.0, 0.2) for _ in range(n)]
    
    # Initial guess for radii
    radii_guess = radii.copy()
    
    # Optimization
    result = optimize.minimize(
        objective, radii_guess, bounds=bounds, constraints={'type': 'ineq', 'fun': constraints}
    )
    
    radii = result.x
    sum_radii = sum(radii)
    
    return centers, radii, sum_radii
