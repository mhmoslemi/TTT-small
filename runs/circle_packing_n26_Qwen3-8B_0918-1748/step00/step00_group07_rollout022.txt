import numpy as np
from scipy.optimize import minimize
import random

def run_packing():
    # Initial configuration: hexagonal grid with multiple layers
    def initial_config(n_circles):
        centers = []
        radii = []
        for i in range(n_circles):
            x = i % 5 / 5.0 + 0.1
            y = i // 5 / 5.0 + 0.1
            centers.append([x, y])
            radii.append(0.05)
        return np.array(centers), np.array(radii)

    centers, radii = initial_config(26)
    
    def objective(x):
        return -np.sum(x[1:])  # maximize sum of radii

    def constraint1(x):
        return np.sum(x[1:])  # radii must be non-negative

    def constraint2(x):
        for i in range(26):
            x_i, y_i = x[0 + 2*i], x[1 + 2*i]
            r_i = x[26 + i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -1
        return 1

    def constraint3(x):
        for i in range(26):
            for j in range(i + 1, 26):
                x_i, y_i = x[0 + 2*i], x[1 + 2*i]
                x_j, y_j = x[0 + 2*j], x[1 + 2*j]
                r_i = x[26 + i]
                r_j = x[26 + j]
                dx = x_i - x_j
                dy = y_i - y_j
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r_i + r_j - 1e-8:
                    return -1
        return 1

    # Flatten centers and radii into a single vector for optimization
    x0 = np.concatenate([centers.flatten(), radii])

    # Set bounds for centers (0 to 1) and radii (non-negative)
    bounds = []
    for _ in range(26 * 2):
        bounds.append((0.0, 1.0))
    for _ in range(26):
        bounds.append((0.0, 1.0))

    # Optimization setup
    cons = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2},
        {'type': 'ineq', 'fun': constraint3}
    ]

    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8}
    )

    # Extract the optimal centers and radii
    x_opt = result.x
    centers_opt = x_opt[:26 * 2].reshape(26, 2)
    radii_opt = x_opt[26 * 2:]

    # Return the result
    return (centers_opt, radii_opt, np.sum(radii_opt))
