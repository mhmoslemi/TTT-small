```python
import numpy as np
from scipy.optimize import minimize
import math

def generate_hex_grid(n):
    rows = int(np.ceil(n / 2))
    cols = int(np.ceil(n / rows))
    grid = []
    for i in range(rows):
        for j in range(cols):
            x = j + 0.5 * i
            y = i
            grid.append([x, y])
    return np.array(grid)

def run_packing():
    initial_grid = generate_hex_grid(26)
    centers = initial_grid
    radii = np.ones(26) * 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        constraints = []
        for i in range(26):
            constraints.append(('bounds', 0, 1))
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append(('inequality', dist - radii[i] - radii[j], 0.0))
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            constraints.append(('inequality', x - r, -1e-12))
            constraints.append(('inequality', 1.0 + 1e-12 - x - r, 0.0))
            constraints.append(('inequality', y - r, -1e-12))
            constraints.append(('inequality', 1.0 + 1e-12 - y - r, 0.0))
        return constraints

    def optimize():
        bounds = [(0, 1) for _ in range(26)]
        res = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)
        return res.x

    radii = optimize()

    def check_packing(centers, radii):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False, f"Circle {i} at ({x}, {y}) r={r} outside unit square"
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    valid, message = check_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    return centers, radii, np.sum(radii)
```
