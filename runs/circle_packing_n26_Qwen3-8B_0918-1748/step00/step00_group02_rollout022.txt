import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.spatial.distance import pdist, squareform
import itertools

def run_packing():
    n = 26
    # Initial guess: hexagonal packing
    rows = 6
    cols = 5
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col + 0.5 * (1 - 2 * (i % 2))
        y = row * (np.sqrt(3) / 2)
        centers[i] = [x, y]
    
    # Initial radius based on distance to nearest boundary
    max_dist = np.min([centers[i, 0] for i in range(n)] + [1 - centers[i, 0] for i in range(n)] +
                     [centers[i, 1] for i in range(n)] + [1 - centers[i, 1] for i in range(n)])
    radii = np.full(n, max_dist / 2)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_func(radii):
        # Distance between any two circles must be at least sum of radii
        dist_matrix = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True
    
    # Setup optimization problem
    bounds = Bounds(lb=0, ub=np.inf)
    cons = {'type': 'ineq', 'fun': constraint_func}
    
    result = minimize(objective, radii, bounds=bounds, constraints=cons)
    
    # Post-process to ensure all circles are within the square
    for i in range(n):
        x, y = centers[i]
        r = result.x[i]
        if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
            # Adjust radius to fit within square
            max_dist = min(x, 1 - x, y, 1 - y)
            result.x[i] = max_dist
    
    return centers, result.x, np.sum(result.x)
