import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Initial configuration: 5 rows with 5, 5, 5, 5, 6 circles
    # This is a modified hexagonal packing with 5 rows, where the last row has 6 circles
    centers = []
    radii = []
    total_sum = 0

    # First row (top)
    for i in range(5):
        x = 0.1 + i * 0.18
        y = 0.95
        centers.append([x, y])
        radii.append(0.05)
        total_sum += 0.05

    # Second row
    for i in range(5):
        x = 0.1 + i * 0.18
        y = 0.65
        centers.append([x, y])
        radii.append(0.05)
        total_sum += 0.05

    # Third row
    for i in range(5):
        x = 0.1 + i * 0.18
        y = 0.35
        centers.append([x, y])
        radii.append(0.05)
        total_sum += 0.05

    # Fourth row
    for i in range(5):
        x = 0.1 + i * 0.18
        y = 0.05
        centers.append([x, y])
        radii.append(0.05)
        total_sum += 0.05

    # Fifth row (with 6 circles)
    for i in range(6):
        x = 0.1 + i * (0.18 / 6)
        y = 0.05 + 0.3
        centers.append([x, y])
        radii.append(0.05)
        total_sum += 0.05

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Define the constraints
    def constraint1(radii):
        for i in range(len(radii)):
            if radii[i] < 0:
                return False
        return True

    def constraint2(radii):
        for i in range(len(radii)):
            x, y = centers[i]
            if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12
                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
                return False
        return True

    def constraint3(radii):
        distances = squareform(pdist(centers))
        for i, j in combinations(range(len(radii)), 2):
            if distances[i, j] < radii[i] + radii[j] - 1e-12:
                return False
        return True

    # Set up the optimization problem
    bounds = [(0, 1) for _ in range(len(radii))]
    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2},
            {'type': 'ineq', 'fun': constraint3}]

    # Perform the optimization
    result = minimize(objective, radii, bounds=bounds, constraints=cons)

    # Extract results
    centers_optimized = centers
    radii_optimized = result.x
    sum_radii = np.sum(radii_optimized)

    return (centers_optimized, radii_optimized, sum_radii)
