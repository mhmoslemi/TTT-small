from scipy.optimize import minimize

def run_packing():
    # Initial configuration: 5 rows of hexagonal packing with some spacing
    num_circles = 26

                centers.append([x, y])
    # Add corner clusters

    ]
    for x, y in corners:
        centers.append([x, y])
    # Initialize radii
    radii = [hex_rad] * 24 + [0.1] * 2

    # Define function to compute total sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Define constraints: distance between centers >= r_i + r_j
    def constraint_func(x):
        constraints = []
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                r_i = x[i]
                r_j = x[j]
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r_i + r_j - 1e-9:
                    constraints.append(-dist + r_i + r_j - 1e-9)
        return np.array(constraints)

    # Define box constraints: radii >= 0
    bounds = [(0, 1)] * len(radii)

    # Initial guess
    x0 = radii

    # Prepare optimization
    import numpy as np

    # Define combined constraints
    def constraint(x):
        constraints = []
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                r_i = x[i]
                r_j = x[j]
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r_i + r_j - 1e-9:
                    constraints.append(-dist + r_i + r_j - 1e-9)
        return np.array(constraints)

    # Optimize
    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=constraint,
        method='SLSQP'
    )

    # Extract centers and radii
    centers = np.array(centers)
    radii = result.x

    # Validate packing
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    is_valid, msg = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
