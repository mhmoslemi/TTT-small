import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.spatial.distance import pdist, squareform

def run_packing():
    n = 26
    # Initial grid-based packing with some space for optimization
    grid_size = int(np.ceil(np.sqrt(n)))
    positions = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // grid_size
        col = i % grid_size
        x = col / (grid_size - 1)
        y = row / (grid_size - 1)
        positions[i] = [x, y]
    
    # Initial radius based on spacing between adjacent circles
    initial_radius = 0.1
    radii = np.full(n, initial_radius)
    
    # Objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)
    
    # Bounds for positions and radii
    bounds = []
    for i in range(n):
        # x and y between 0 and 1
        bounds.append((0, 1))
        bounds.append((0, 1))
        # radii must be non-negative
        bounds.append((0, 1))
    
    # Constraints: distance between circles >= sum of radii
    def constraint_func(x):
        positions_flat = x[:2*n].reshape(n, 2)
        radii_flat = x[2*n:]
        constraints = []
        for i in range(n):
            for j in range(i+1, n):
                dx = positions_flat[i, 0] - positions_flat[j, 0]
                dy = positions_flat[i, 1] - positions_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                required_dist = radii_flat[i] + radii_flat[j]
                if dist < required_dist - 1e-8:
                    return dist - required_dist + 1e-8
        return 0
    
    # Optimization
    x0 = np.concatenate([positions.flatten(), radii])
    cons = {'type': 'ineq', 'fun': constraint_func}
    result = minimize(
        objective,
        x0,
        bounds=Bounds(bounds),
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    sum_radii = np.sum(radii)
    
    # Final validation
    if np.isnan(centers).any() or np.isnan(radii).any():
        raise ValueError("NaN values present")
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")
    
    return (centers, radii, sum_radii)
