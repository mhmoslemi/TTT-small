import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Define initial centers and radii for 26 circles
    # Using a hybrid hexagonal grid and boundary circles strategy
    
    # Core circle
    centers = np.array([
        [0.5, 0.5],  # Core
    ])
    
    radii = np.array([0.2])  # Initial guess for radii
    
    # Generate hexagonal layers around the core
    # Layer 1: 6 circles around the core
    layer1_centers = []
    for angle in np.linspace(0, 2*np.pi, 6, endpoint=False):
        x = 0.5 + np.cos(angle) * 0.4
        y = 0.5 + np.sin(angle) * 0.4
        layer1_centers.append([x, y])
    centers = np.vstack([centers, layer1_centers])
    radii = np.append(radii, np.full(len(layer1_centers), 0.1))
    
    # Layer 2: 12 circles forming a hexagonal ring (2 layers around the core)
    layer2_centers = []
    for angle in np.linspace(0, 2*np.pi, 12, endpoint=False):
        x = 0.5 + np.cos(angle) * 0.6
        y = 0.5 + np.sin(angle) * 0.6
        layer2_centers.append([x, y])
    centers = np.vstack([centers, layer2_centers])
    radii = np.append(radii, np.full(len(layer2_centers), 0.08))
    
    # Layer 3: 7 edge/corner circles
    # 4 corner circles and 3 mid-edge circles
    corner_centers = [
        [0.1, 0.1], [0.9, 0.1], [0.9, 0.9], [0.1, 0.9]
    ]
    mid_edge_centers = [
        [0.5, 0.1], [0.5, 0.9], [0.1, 0.5]
    ]
    layer3_centers = corner_centers + mid_edge_centers
    centers = np.vstack([centers, layer3_centers])
    radii = np.append(radii, np.full(len(layer3_centers), 0.05))
    
    # Define optimization objective
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraint_func(radii):
        # Compute pairwise distances between centers
        dists = squareform(pdist(centers))
        # Ensure circles do not overlap and stay within the unit square
        for i in range(len(centers)):

                return -1
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                if dists[i, j] < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Optimization
    bounds = [(0.001, 0.5) for _ in range(len(radii))]  # Small lower bound to prevent zero radii
    res = minimize(objective, radii, bounds=bounds, constraints={"type": "ineq", "fun": constraint_func, "jac": None})
