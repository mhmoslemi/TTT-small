import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import random

def run_packing():
    def fitness(radii, centers):
        return -np.sum(radii)

    def constraint_overlap(i, j, centers, radii):
        return np.sqrt(np.sum((centers[i] - centers[j])**2)) - (radii[i] + radii[j])

    def constraint_boundary(i, center, radius):
        return np.all((center[0] - radius >= 0) & (center[0] + radius <= 1) &
                     (center[1] - radius >= 0) & (center[1] + radius <= 1))

    def validate(radius, center, constraints):
        if radius < 0:
            return False
        if not constraints:
            return True
        for c in constraints:
            if not c(radius, center):
                return False
        return True

    def get_constraints(n):
        constraints = []
        for i in range(n):
            def c(radius, center):
                return constraint_boundary(i, center, radius)
            constraints.append(c)
        for i in range(n):
            for j in range(i+1, n):
                def c(radius, center):
                    return constraint_overlap(i, j, center, radius)
                constraints.append(c)
        return constraints

    n = 26
    bounds = [(0, 1)] * 2 * n
    initial_centers = np.random.rand(n, 2)
    initial_radii = np.random.rand(n)
    initial_radii = initial_radii / np.max(initial_radii)

    result = minimize(
        lambda x: fitness(x[:n], x[n:]),
        x0=np.concatenate((initial_radii, initial_centers.flatten())),
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': lambda x, i: constraint_overlap(i, x[n:][2*i:2*i+2], x[n:], x[:n])}]
    )

    if not result.success:
        best = np.random.rand(n, 2)
        best_radii = np.random.rand(n) / 10
        return best, best_radii, np.sum(best_radii)

    best_radii = result.x[:n]
    best_centers = result.x[n:].reshape(n, 2)
    return best_centers, best_radii, np.sum(best_radii)
