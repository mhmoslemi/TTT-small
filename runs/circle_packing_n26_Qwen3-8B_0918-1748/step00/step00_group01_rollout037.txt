import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial guess: square grid with a perturbation for the 26th circle
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial grid configuration (5x5, with 1 extra circle)
    # First 25 circles: grid from (0.1, 0.1) to (0.9, 0.9) spaced by 0.2
    for i in range(5):
        for j in range(5):
            idx = i * 5 + j

            radii[idx] = 0.1

    # 26th circle: try to place in a corner with maximum possible radius
    # Start with a guess slightly outside of the square to allow optimization
    centers[25] = [0.05, 0.05]
    radii[25] = 0.1

    def objective(radii):
        return -np.sum(radii)

    def constraints(centers, radii):
        # Boundary constraints: each circle must be within [0,1]^2
        bounds = Bounds([0, 0], [1, 1])
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0:
                return False, "Circle %d goes outside left boundary" % i
            if x + r > 1:
                return False, "Circle %d goes outside right boundary" % i
            if y - r < 0:
                return False, "Circle %d goes outside bottom boundary" % i
            if y + r > 1:
                return False, "Circle %d goes outside top boundary" % i
        # Non-overlapping constraints: distance between centers >= sum of radii
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False, "Circles %d and %d overlap" % (i, j)
        return True, "Constraints satisfied"

    # Optimization
    bounds = Bounds([0, 0], [1, 1])
    result = opt.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': lambda x: np.sum(x) - 1e-8}],
        method='SLSQP'
    )

    # Refine the centers based on optimized radii
    # Use the same initial grid but with adjusted radii
    for i in range(n):

            # Adjust for grid spacing
            x = 0.1 + (i % 5) * 0.2

            # Adjust for the 26th circle
            x = 0.05
            y = 0.05
        centers[i] = [x, y]
        radii[i] = r

    # Final check for boundary and non-overlapping
    for i in range(n):
        x, y = centers[i]

            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - 1e-8:
                raise ValueError("Final check failed: circles %d and %d overlap" % (i, j))

    return centers, radii, np.sum(radii)
