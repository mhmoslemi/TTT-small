from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():

    # First, define a grid of coordinates for a hexagonal packing
    n_rows = 5
    n_cols = 5
    hex_grid = []
    for i in range(n_rows):
        for j in range(n_cols):
            x = j + 0.5 * i
            y = i * np.sqrt(3) / 2
            if x <= 1 and y <= 1:
                hex_grid.append((x, y))

    boundary_circles = [
        (0.5, 0.0),       # bottom
        (0.5, 1.0),       # top
        (0.0, 0.5),       # left
        (1.0, 0.5),       # right
        (0.25, 0.75),     # bottom-right
        (0.75, 0.75),     # bottom-left
        (0.25, 0.25),     # top-right
        (0.75, 0.25),     # top-left
    ]
    
    # Combine all proposed circles
    all_centers = hex_grid + boundary_circles
    n = len(all_centers)
    
    # Define radius vector with equal initial values
    radii = np.ones(n) * 0.05
    
    # Initial bounds for all circles
    bounds = [(0.0, 1.0) for _ in range(n * 2)]
    
    # Optimization problem: maximize sum(radii) with constraints
    def objective(radii_vec):
        return -np.sum(radii_vec)  # minimize negative sum to maximize

        def cons(x):
            # i-th circle
            x_i, y_i = all_centers[i]
            r_i = x[n * i : n * i + 2]
            # j-th circle
            x_j, y_j = all_centers[j]
            r_j = x[n * j : n * j + 2]
            # Distance between centers
            dx, dy = x_i - x_j, y_i - y_j
            dist_sq = dx**2 + dy**2
            if dist_sq < (r_i[0] + r_j[0])**2 - 1e-12:
                return (r_i[0] + r_j[0])**2 - dist_sq
            else:
                return 0.0
        return cons

        def cons(x):

                return y_i - r_i[0] + 1e-12
            return 0.0
        return cons

    cons = []

        cons.append((constraint_boundary(i), 0.0))
    
    for i in range(n):
        for j in range(i + 1, n):
            cons.append((constraint_overlap(i, j), 0.0))

    x0 = np.zeros(n * 2)
    x0[:n] = 0.05
    x0[n:] = 0.05
    
    # Solve the optimization problem
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=Bounds(0.0, 1.0),
