```python
import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    def objective_function(x):
        r = x[:26]
        centers = x[26:].reshape(26, 2)
        return -np.sum(r)  # We want to maximize sum of radii, so minimize negative sum

    def constraint_boundary(x):
        r = x[:26]
        centers = x[26:].reshape(26, 2)
        for i in range(26):
            x_i, y_i = centers[i]
            if (x_i - r[i] < -1e-12 or x_i + r[i] > 1 + 1e-12 or
                y_i - r[i] < -1e-12 or y_i + r[i] > 1 + 1e-12):
                return -1.0  # Violates boundary constraint
        return 0.0

    def constraint_overlap(x):
        r = x[:26]
        centers = x[26:].reshape(26, 2)
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                if dist_sq < (r[i] + r[j])**2 - 1e-12:
                    return -1.0  # Violates overlap constraint
        return 0.0

    def setup_initial_positions():
        rows = 5
        cols = 5
        spacing = 0.2
        centers = np.zeros((26, 2))
        for i in range(rows):
            for j in range(cols):
                row = i
                col = j
                if i % 2 == 1:
                    col += 0.5
                x = col * spacing
                y = row * spacing
                if row % 2 == 0:
                    x += spacing / 2.0
                centers[i * cols + j] = [x, y]
        # Add additional circles near the corners
        for i in range(26):
            if i >= rows * cols:
                if i == 26 - 1:
                    centers[i] = [0.95, 0.95]
                elif i == 25 - 1:
                    centers[i] = [0.05, 0.95]
                elif i == 24 - 1:
                    centers[i] = [0.95, 0.05]
                elif i == 23 - 1:
                    centers[i] = [0.05, 0.05]
        return centers

    initial_centers = setup_initial_positions()
    initial_r = np.full(26, 0.05)
    initial_x = np.concatenate((initial_r, initial_centers.flatten()))

    # We'll use a modified version of the scipy.optimize.minimize
    # with bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(26)] + \
            [(0.0, 1.0) for _ in range(52)]

    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    result = minimize(
        objective_function,
        initial_x,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    x_result = result.x
    radii = x_result[:26]
    centers = x_result[26:].reshape(26, 2)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
```
