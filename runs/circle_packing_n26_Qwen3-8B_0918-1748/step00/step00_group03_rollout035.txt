import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial hexagonal grid packing with 20 circles
    def generate_hex_grid(num_rows, num_cols, spacing, offset):
        grid = []
        for i in range(num_rows):
            for j in range(num_cols):
                x = j * spacing
                y = i * spacing * np.sqrt(3)/2
                grid.append((x, y))
        return np.array(grid)

    # 18 columns, 12 rows (adjusted to fit 20 circles)
    grid = generate_hex_grid(12, 18, 0.08, 0.05)
    centers = grid.copy()
    radii = np.full(grid.shape[0], 0.05)

    # Convert to numpy arrays
    centers = np.array(centers, dtype=np.float64)
    radii = np.array(radii, dtype=np.float64)

    # Nearest neighbor search for extra circles
    def find_gaps(centers, radii, n_extra=6):
        gap_indices = []
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] + 0.001:
                    gap_indices.append((i, j))
        return np.unique(gap_indices)

    # Initial extra circles positions
    extra_centers = []
    extra_radii = []
    used = np.zeros(len(centers), dtype=bool)

    # Place extra circles in the most promising gaps
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] + 0.001:
                x = (centers[i][0] + centers[j][0]) / 2
                y = (centers[i][1] + centers[j][1]) / 2
                extra_centers.append([x, y])
                extra_radii.append(0.03)
                used[i] = True
                used[j] = True
                if len(extra_centers) == 6:
                    break

    # Remove duplicates
    used = np.zeros(len(centers), dtype=bool)
    for i in range(len(centers)):
        if not used[i] and np.sum(centers[i] == centers[i]) > 0:
            used[i] = True

    # Combine base grid and extra circles
    centers = np.vstack([centers, np.array(extra_centers)])
    radii = np.concatenate([radii, np.array(extra_radii)])
    n = len(centers)

    # Optimization parameters
    lb = np.array([[0.0] * 2 for _ in range(n)])
    ub = np.array([[1.0] * 2 for _ in range(n)])
    bounds = Bounds(lb, ub)

    def objective(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:]
        return -np.sum(radii_flat)

    def constraints(x):
        centers_flat = x[:2 * n].reshape(n, 2)
        radii_flat = x[2 * n:]
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return np.inf
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return np.inf
        return 0

    x0 = np.concatenate([centers.reshape(n * 2), radii])
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-6, "maxiter": 300}
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    centers_opt = result.x[:2 * n].reshape(n, 2)
    radii_opt = result.x[2 * n:]

    # Final validation
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii_opt)
    return (centers_opt, radii_opt, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
