import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for circle centers in a hexagonal close-packed arrangement
    # We'll start with a horizontal row of 13 circles and then stack rows on top
    # with a slight offset for hexagonal packing
    
    num_circles = 26
    radius_base = 0.15
    spacing = 1.5 * radius_base
    
    centers = []
    radii = []
    
    # First row (bottom)
    for i in range(13):
        x = i * spacing
        y = radius_base
        centers.append([x, y])
        radii.append(radius_base)
    
    # Second row (slightly offset)
    for i in range(12):
        x = i * spacing + spacing / 2
        y = 2 * radius_base + spacing / 2
        centers.append([x, y])
        radii.append(radius_base)
    
    # Third row (offset again)
    for i in range(11):
        x = i * spacing + spacing / 2
        y = 3 * radius_base + spacing / 2
        centers.append([x, y])
        radii.append(radius_base)
    
    # Adjust the last few centers to fit in the unit square and avoid overlap
    # This is a rough estimate to get a starting point for optimization
    
    # Convert to numpy arrays for easier manipulation
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Define the optimization problem
    def objective(x):
        # x is a flattened array of all variables: [x0, y0, r0, x1, y1, r1, ...]
        x_flat = x.reshape(-1, 3)
        centers_opt = x_flat[:, :2]
        radii_opt = x_flat[:, 2]
        
        # Calculate the total sum of radii to maximize
        return -np.sum(radii_opt)  # Negative because we are minimizing
    
    def constraint_boundary(x):
        x_flat = x.reshape(-1, 3)
        centers_opt = x_flat[:, :2]
        radii_opt = x_flat[:, 2]
        
        # Ensure circles are within the unit square
        for i in range(num_circles):
            x, y = centers_opt[i]
            r = radii_opt[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return 1.0
        return 0.0
    
    def constraint_overlap(x):
        x_flat = x.reshape(-1, 3)
        centers_opt = x_flat[:, :2]
        radii_opt = x_flat[:, 2]
        
        # Check for overlaps between circles
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_opt[i] + radii_opt[j] - 1e-8:
                    return 1.0
        return 0.0
    
    # Flatten the initial guess
    x0 = []
    for i in range(num_circles):
        x0.append(centers[i, 0])
        x0.append(centers[i, 1])
        x0.append(radii[i])
    x0 = np.array(x0)
    
    # Define bounds: each circle must have its center within [0,1]x[0,1], and radius non-negative
    bounds = []
    for _ in range(num_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0), (0.0, np.inf)])
    
    # Define constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    # Solve the optimization problem
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP'
    )
    
    # Extract the results
    x_result = result.x
    centers_result = x_result.reshape(-1, 3)[:, :2]
    radii_result = x_result.reshape(-1, 3)[:, 2]
    
    # Return the centers, radii, and sum of radii
    return centers_result, radii_result, np.sum(radii_result)
