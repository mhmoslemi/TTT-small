import numpy as np
import random
from scipy.optimize import minimize
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    def fitness_function(x: np.ndarray) -> float:
        centers = x[:26 * 2].reshape(26, 2)
        radii = x[26 * 2:].reshape(26)
        valid, message = validate_packing(centers, radii)
        if not valid:
            return -np.inf
        return np.sum(radii)

    def create_initial_guess() -> np.ndarray:
        grid = np.linspace(0, 1, 5)
        x_centers = []
        y_centers = []
        for i in range(5):
            for j in range(5):
                x_centers.append(grid[i])
                y_centers.append(grid[j])
        x_centers = np.array(x_centers)
        y_centers = np.array(y_centers)
        centers = np.column_stack((x_centers, y_centers))
        radii = np.full(len(centers), 0.05)
        return np.concatenate((centers.ravel(), radii))

    def gradient_function(x: np.ndarray) -> np.ndarray:
        return np.zeros_like(x)

    initial_guess = create_initial_guess()
    result = minimize(
        fitness_function,
        initial_guess,
        method='L-BFGS-B',
        jac=gradient_function,
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    optimal_x = result.x
    centers = optimal_x[:26 * 2].reshape(26, 2)
    radii = optimal_x[26 * 2:].reshape(26)
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
