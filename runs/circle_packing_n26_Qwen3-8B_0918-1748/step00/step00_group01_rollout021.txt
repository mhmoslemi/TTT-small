import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint

def run_packing():
    # Define the number of circles
    n_circles = 26

    # Initial seed: 4 rows with increasing number of circles
    # First row: 6 circles
    # Second row: 6 circles
    # Third row: 6 circles
    # Fourth row: 8 circles
    # Try to adjust radii to make the packing as tight as possible
    
    # Initial centers (assumed to be in hexagonal grid format)
    centers = np.zeros((n_circles, 2))
    radii = np.zeros(n_circles)
    
    # Create a hexagonal lattice
    radius = 0.1
    row_height = np.sqrt(3) * radius
    col_width = 2 * radius
    
    # Arrange the centers in a hexagonal grid
    row = 0
    col = 0
    for i in range(n_circles):
        if i % 2 == 0:
            centers[i, 0] = col * col_width
            centers[i, 1] = row * row_height
        else:
            centers[i, 0] = (col + 0.5) * col_width
            centers[i, 1] = (row + 0.5) * row_height
        col += 1
        if col % 2 == 0:
            row += 1
    
    # Define the optimization problem
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_radius(radii):
        return np.sum(radii)
    
    def constraint_center(x, y, radius):
        return np.array([
            x - radius >= 0,
            x + radius <= 1,
            y - radius >= 0,
            y + radius <= 1
        ])
    
    def constraint_overlap(x1, y1, r1, x2, y2, r2):
        dx = x1 - x2
        dy = y1 - y2
        dist_sq = dx ** 2 + dy ** 2
        return np.sqrt(dist_sq) >= r1 + r2
    
    # Define bounds and constraints
    bounds = Bounds(0, 1)  # Radii bounds for each circle
    constraints = []
    
    for i in range(n_circles):
        constraints.append(LinearConstraint(
            np.array([1, 1]),  # Coefficients for x and y
            0, 1))  # x + y <= 1
        for j in range(i + 1, n_circles):
            constraints.append(NonlinearConstraint(lambda r: (
                np.sqrt((centers[i, 0] - centers[j, 0]) ** 2 + (centers[i, 1] - centers[j, 1]) ** 2) - (r[i] + r[j])
            ), -np.inf, 0))
    
    # Solve the optimization problem with bounds and constraints
    result = opt.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )
    
    if not result.success:
        raise RuntimeError("Optimization failed.")
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Final check to ensure circles are fully within the square
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            raise RuntimeError("Circle is partially outside the unit square.")
    
    return (centers, radii, sum_radii)
