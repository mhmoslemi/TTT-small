import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial grid-based packing with a slight offset for hexagonal arrangement
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create a hexagonal grid of 5 rows with varying number of columns
    # First 4 rows will have 6 circles, the fifth will have 5 circles
    # This uses a hexagonal packing to increase density
    
    # Define initial positions with hexagonal packing
    for i in range(n):
        if i < 6:  # First row
            x = i / 5.0
            y = 0.2
        elif i < 12:  # Second row
            x = (i - 6) / 5.0
            y = 0.3
        elif i < 18:  # Third row
            x = (i - 12) / 5.0
            y = 0.4
        elif i < 24:  # Fourth row
            x = (i - 18) / 5.0
            y = 0.5
        else:  # Fifth row
            x = (i - 24) / 4.0
            y = 0.6
        centers[i] = [x, y]
    
    # Add a circle in the center to optimize the packing
    centers[-1] = [0.5, 0.5]  # This is our 26th circle
    
    # Optimization parameters
    bounds = Bounds([0.0, 0.0], [1.0, 1.0])
    initial_radii = np.ones(n) * 0.05  # Initial guess for radii
    
    def objective(radii: np.ndarray) -> float:
        return -np.sum(radii)  # Minimize negative sum to maximize radii
    
    def constraints(radii: np.ndarray) -> np.ndarray:
        res = []
        for i in range(n):
            # Boundary constraint
            res.append(radii[i] - (1.0 - centers[i, 0]) - 1e-8)
            res.append(radii[i] - (centers[i, 0] - 1e-8))
            res.append(radii[i] - (1.0 - centers[i, 1]) - 1e-8)
            res.append(radii[i] - (centers[i, 1] - 1e-8))
        
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                res.append(radii[i] + radii[j] - dist - 1e-8)
        
        return np.array(res)
    
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 1000}
    )
    
    radii = result.x
    centers = np.array(centers)
    
    # Ensure non-negative radii and adjust if needed
    radii[radii < 0] = 0
    # Check and adjust overlapping circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-8:
                # Adjust radii to prevent overlapping
                overlap = radii[i] + radii[j] - dist
                radii[i] -= overlap / 2
                radii[j] -= overlap / 2
                radii[radii < 0] = 0
    
    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} is outside the unit square")
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
