import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():
    # Initial guess for circle centers using a hexagonal grid
    def hexagonal_grid(n, side_length):
        rows = int(np.ceil(n / 2))
        cols = int(np.ceil(n / rows))
        centers = np.zeros((n, 2))
        idx = 0
        for r in range(rows):
            for c in range(cols):
                if idx < n:
                    x = c * (2 * side_length * np.sqrt(3)/2) + side_length * np.sqrt(3)/2
                    y = r * (2 * side_length) + side_length
                    if x > 1 or y > 1 or x < 0 or y < 0:
                        continue
                    centers[idx] = [x, y]
                    idx += 1
        return centers

    # Initial guess for radii using the spacing between circles in hexagonal grid
    def initial_radii(centers, side_length):
        radii = np.full(len(centers), side_length / 2)
        # Adjust radii to fit within the unit square
        for i in range(len(centers)):
            x, y = centers[i]
            radii[i] = min(radii[i], 1 - max(x, y))
            radii[i] = min(radii[i], max(x, y))
        return radii

    # Optimize configuration to maximize total radius sum
    def optimize_config(centers_guess, radii_guess):
        def objective(x, centers, radii):
            return -np.sum(radii)

        def constraint_overlap(x, centers, radii):
            n = len(radii)
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.linalg.norm(x[2*i:2*i+2] - x[2*j:2*j+2])
                    if dist < radii[i] + radii[j] - 1e-6:
                        return dist - radii[i] - radii[j] + 1e-6
            return 0

        def constraint_boundary(x, centers, radii):
            n = len(radii)
            for i in range(n):
                x_i = x[2*i]
                y_i = x[2*i + 1]
                r_i = radii[i]
                if x_i - r_i < -1e-6 or x_i + r_i > 1 + 1e-6 or y_i - r_i < -1e-6 or y_i + r_i > 1 + 1e-6:
                    return x_i - r_i + 1e-6
            return 0

        n = len(radii_guess)
        bounds = [(0, 1) for _ in range(2 * n)]
        x0 = np.zeros(2 * n)
        for i in range(n):
            x0[2*i] = centers_guess[i, 0]
            x0[2*i + 1] = centers_guess[i, 1]

        # Initial radii guess
        radii = radii_guess.copy()
        for i in range(n):
            x, y = centers_guess[i]
            radii[i] = min(radii[i], 1 - max(x, y))
            radii[i] = min(radii[i], max(x, y))

        # Optimization with constraints
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                cons.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: np.linalg.norm(x[2*i:2*i+2] - x[2*j:2*j+2]) - radii[i] - radii[j] + 1e-6})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i] - radii[i] + 1e-6})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2*i + 1] - radii[i] + 1e-6})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i] - radii[i] + 1e-6})
            cons.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[2*i + 1] - radii[i] + 1e-6})

        result = minimize(objective, x0, args=(centers_guess, radii), bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 200})
        if not result.success:
            raise RuntimeError("Optimization failed")
        centers_opt = np.zeros((n, 2))
        for i in range(n):
            centers_opt[i, 0] = result.x[2*i]
            centers_opt[i, 1] = result.x[2*i + 1]
        radii_opt = np.full(n, 0)
        for i in range(n):
            x, y = centers_opt[i]
            r = min(1 - max(x, y), max(x, y))
            radii_opt[i] = r
        return centers_opt, radii_opt

    # Try initial hexagonal packing and optimize
    try:
        centers_guess = hexagonal_grid(26, 0.2)
        radii_guess = initial_radii(centers_guess, 0.2)
        centers_opt, radii_opt = optimize_config(centers_guess, radii_guess)
        radii_opt = np.maximum(radii_opt, 0)
        # Final validation
        def validate_config(centers, radii):
            n = len(radii)
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                if x - r < -1e-6 or x + r > 1 + 1e-6 or y - r < -1e-6 or y + r > 1 + 1e-6:
                    return False
            for i in range(n):
                for j in range(i + 1, n):
                    dist = np.linalg.norm(centers[i] - centers[j])
                    if dist < radii[i] + radii[j] - 1e-6:
                        return False
            return True

        if validate_config(centers_opt, radii_opt):
            total_sum = np.sum(radii_opt)
            return (centers_opt, radii_opt, total_sum)
        else:
            raise RuntimeError("Invalid packing detected")
    except Exception as e:
        raise RuntimeError(f"Optimization failed: {str(e)}")
