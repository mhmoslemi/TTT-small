import numpy as np
from scipy.optimize import minimize
import time

def run_packing():
    def objective(x):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_overlap(i, j):
        def cons(x):
            idx_i = i * 2
            idx_j = j * 2
            x_i, y_i = x[idx_i], x[idx_i + 1]
            x_j, y_j = x[idx_j], x[idx_j + 1]
            dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
            return dist - (radii[i] + radii[j])
        return cons

    def constraint_boundary(i):
        def cons(x):
            idx = i * 2
            x_i, y_i = x[idx], x[idx + 1]
            r = radii[i]
            if x_i - r < -1e-8 or x_i + r > 1 + 1e-8:
                return x_i - r + 1e-8
            if y_i - r < -1e-8 or y_i + r > 1 + 1e-8:
                return y_i - r + 1e-8
            return 0
        return cons

    def initial_guess_hexagonal(n, spacing):
        centers = []
        for i in range(n):
            row = i // spacing
            col = i % spacing
            x = col / spacing + 0.5 / spacing
            y = row / spacing + 0.5 / spacing
            if (row + col) % 2 == 1:
                y += np.sqrt(3) / (2 * spacing)
            centers.append([x, y])
        return np.array(centers)

    n = 26
    spacing = 5
    centers = initial_guess_hexagonal(n, spacing)
    radii = np.full(n, 1 / (spacing**2 * np.pi))

    # Local refinement
    for i in range(n):
        for j in range(n):
            if i != j:
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                radii[i] = min(radii[i], dist / 2)
                radii[j] = min(radii[j], dist / 2)

    # Define bounds
    bounds = []
    for i in range(n):
        bounds.append((0, 1))
        bounds.append((0, 1))
    bounds = np.array(bounds).reshape(n, 2)

    # Optimization setup
    x0 = centers.flatten()
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})
    result = minimize(objective, x0, bounds=bounds, constraints=constraints, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    centers_opt = result.x.reshape((n, 2))
    radii_opt = np.full(n, 1 / (spacing**2 * np.pi))
    for i in range(n):
        for j in range(n):
            if i != j:
                dist = np.sqrt((centers_opt[i, 0] - centers_opt[j, 0])**2 + (centers_opt[i, 1] - centers_opt[j, 1])**2)
                radii_opt[i] = min(radii_opt[i], dist / 2)
                radii_opt[j] = min(radii_opt[j], dist / 2)

    # Secondary optimization to densify
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers_opt[i, 0] - centers_opt[j, 0])**2 + (centers_opt[i, 1] - centers_opt[j, 1])**2)
            radii_opt[i] = min(radii_opt[i], dist / 2)
            radii_opt[j] = min(radii_opt[j], dist / 2)

    # Ensuring boundary constraints
    for i in range(n):
        x, y = centers_opt[i]
        r = radii_opt[i]
        if x - r < -1e-8:
            radii_opt[i] = x
        elif x + r > 1 + 1e-8:
            radii_opt[i] = 1 - x
        if y - r < -1e-8:
            radii_opt[i] = y
        elif y + r > 1 + 1e-8:
            radii_opt[i] = 1 - y

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
