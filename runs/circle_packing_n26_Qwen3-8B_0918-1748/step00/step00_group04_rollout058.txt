import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.spatial.distance import pdist, squareform
from numpy import linalg as LA
import time

def run_packing():
    np.random.seed(42)
    
    # Define number of circles
    n = 26
    
    # Initial guess: coarse hexagonal packing
    def hexagonal_grid(n, side_length):
        cols = int(np.sqrt(n))
        rows = (n + cols - 1) // cols
        x = np.linspace(0, 1, rows)
        y = np.linspace(0, 1, cols)
        grid = np.array(np.meshgrid(x, y)).T
        grid = np.vstack([grid.reshape(-1, 2), np.random.rand(n, 2) * 1e-5])
        return grid
    
    centers_initial = hexagonal_grid(n, 0.1)
    radii_initial = np.full(n, 0.05)
    
    # Optimization function to maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative sum
    
    # Constraints: center-to-center distance >= r_i + r_j
    def constraint_center_to_center(centers, radii):
        dist = squareform(pdist(centers))
        return dist - (radii[:, np.newaxis] + radii[np.newaxis, :])
    
    # Constraint: distance to boundary >= r_i
    def constraint_boundary(centers, radii):
        dist_to_boundary = np.min(np.array([
            centers[:, 0] - radii,
            1 - centers[:, 0] - radii,
            centers[:, 1] - radii,
            1 - centers[:, 1] - radii
        ][:, np.newaxis], axis=0), axis=0)
        return dist_to_boundary
    
    # Bounds for circles
    bounds = Bounds(np.zeros(n), np.inf)
    
    result = minimize(
        fun=objective,
        x0=radii_initial,
        method='SLSQP',
        jac=None,
        bounds=bounds,
        constraints=[
            {
                'type': 'ineq',
                'fun': lambda radii: np.min(constraint_center_to_center(centers_initial, radii)),
                'jac': None
            },
            {
                'type': 'ineq',
                'fun': lambda radii: np.min(constraint_boundary(centers_initial, radii)),
                'jac': None
            }
        ],
        options={
            'ftol': 1e-12,
            'maxiter': 1000,
            'disp': False
        }
    )
    
    radii = result.x
    centers = centers_initial
    
    # Final validation
    def validate_packing(centers, radii):
        n = centers.shape[0]
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"
        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"
    
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError("Invalid packing: " + message)
    
    sum_radii = np.sum(radii)
    if sum_radii < 2.636:
        raise RuntimeError("Packing does not meet the target sum of radii")
    
    return (centers, radii, sum_radii)
