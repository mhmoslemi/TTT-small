import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for centers and radii
    centers = np.random.rand(26, 2)
    radii = np.random.rand(26)
    
    # Bounds for centers and radii
    center_bounds = [(0, 1) for _ in range(26 * 2)]
    radius_bounds = [(0, 1) for _ in range(26)]
    
    # Objective function to maximize total radius sum
    def objective(x):
        centers_flat = x[:26*2].reshape(26, 2)
        radii_flat = x[26*2:]
        return -np.sum(radii_flat)  # Negative for minimization
    
    # Constraints: non-overlapping between circles and in the square
    def constraints(x):
        centers_flat = x[:26*2].reshape(26, 2)
        radii_flat = x[26*2:]
        constraints = []
        for i in range(26):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            # Boundary constraints
            if x_i - r_i < -1e-12:
                return False
            if x_i + r_i > 1 + 1e-12:
                return False
            if y_i - r_i < -1e-12:
                return False
            if y_i + r_i > 1 + 1e-12:
                return False
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                r_i = radii_flat[i]
                r_j = radii_flat[j]
                if dist < r_i + r_j - 1e-12:
                    return False
        return True
    
    # Bounds for x (flattened)
    bounds = Bounds(center_bounds + radius_bounds)
    
    # Optimization
    result = opt.minimize(
        objective,
        np.concatenate([centers.flatten(), radii]),
        bounds=bounds,
        constraints={"type": "custom", "fun": constraints, "args": []},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 500}
    )
    
    centers_opt = result.x[:26*2].reshape(26, 2)
    radii_opt = result.x[26*2:]
    
    # Revalidation and adjustment to ensure constraints are met
    # (If needed, further refinement or manual correction could be applied)
    
    return centers_opt, radii_opt, np.sum(radii_opt)
