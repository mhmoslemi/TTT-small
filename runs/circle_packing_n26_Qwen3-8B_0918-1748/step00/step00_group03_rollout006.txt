import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist

def run_packing():
    np.random.seed(42)
    
    # Initial grid-based layout
    num_circles = 26
    grid_rows = 5
    grid_cols = 5
    initial_centers = np.zeros((num_circles, 2))
    for i in range(grid_rows):
        for j in range(grid_cols):
            idx = i * grid_cols + j
            if idx < num_circles:
                x = j / (grid_cols - 1)
                y = i / (grid_rows - 1)
                initial_centers[idx] = [x, y]
    
    # Initial guess for radii
    initial_radii = np.ones(num_circles) * 0.05
    
    def objective(r):
        return -np.sum(r)  # Minimize negative sum to maximize sum
    
    def constraints_func(r, centers):
        constraints = []
        for i in range(num_circles):
            x, y = centers[i]
            if (x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12 or
                y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12):
                return False
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-12:
                    return False
        return True
    
    result = minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=Bounds(lb=0, ub=1),
        constraints=[{'type': 'ineq', 'fun': lambda r: constraints_func(r, initial_centers)}]
    )
    
    final_radii = result.x
    final_centers = np.copy(initial_centers)
    
    # Local refinement to fill gaps
    for _ in range(10):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = final_centers[i, 0] - final_centers[j, 0]
                dy = final_centers[i, 1] - final_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < final_radii[i] + final_radii[j] - 1e-12:
                    direction = [dx / dist, dy / dist]
                    overlap = dist - (final_radii[i] + final_radii[j])
                    push_amount = overlap * 0.5
                    final_centers[i] += direction * push_amount
                    final_centers[j] -= direction * push_amount
    
    # Ensure all circles are within the unit square
    for i in range(num_circles):
        x, y = final_centers[i]
        r = final_radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            final_centers[i] = [np.clip(x, 0, 1), np.clip(y, 0, 1)]
    
    return final_centers, final_radii, np.sum(final_radii)
