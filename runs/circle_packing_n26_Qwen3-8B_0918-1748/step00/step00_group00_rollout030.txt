import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define 6 central circles in a hexagonal cluster
    # Distance between centers is twice the radius of the inner circles
    # Initial radius for inner circles is 0.15
    # Central circle has a larger radius to compress the cluster
    central_centers = np.array([
        [0.5, 0.5],
        [0.5 + 0.3, 0.5],
        [0.5 - 0.3, 0.5],
        [0.5, 0.5 + 0.3],
        [0.5, 0.5 - 0.3],
        [0.5 + 0.3 * np.cos(30 * np.pi / 180), 0.5 + 0.3 * np.sin(30 * np.pi / 180)]
    ])
    central_radii = np.array([0.2, 0.15, 0.15, 0.15, 0.15, 0.15])
    
    # Define 20 surrounding circles in a hexagonal grid pattern
    # Start from the top edge and work downwards
    surrounding_centers = []
    surrounding_radii = []
    for i in range(5):
        for j in range(5):
            x = 0.2 + 0.2 * i
            y = 0.2 + 0.2 * j
            surrounding_centers.append([x, y])
            surrounding_radii.append(0.08)
    
    # Add more circles in the remaining space
    for i in range(3):
        x = 0.7 + 0.1 * i
        y = 0.2
        surrounding_centers.append([x, y])
        surrounding_radii.append(0.08)
    
    for i in range(3):
        x = 0.2
        y = 0.7 + 0.1 * i
        surrounding_centers.append([x, y])
        surrounding_radii.append(0.08)
    
    for i in range(4):
        x = 0.2 + 0.2 * i
        y = 0.8
        surrounding_centers.append([x, y])
        surrounding_radii.append(0.08)
    
    # Combine all centers and radii
    centers = np.vstack([central_centers, surrounding_centers])
    radii = np.concatenate([central_radii, surrounding_radii])
    
    # Optimization parameters
    bounds = [(0, 1) for _ in range(26 * 2)]
    constraints = []
    
    def objective(x):
        return -np.sum(x[1::2])
    
    def constraint_func(x):
        centers_flat = x[::2]
        radii_flat = x[1::2]
        for i in range(26):
            x_i, y_i = centers_flat[2 * i], centers_flat[2 * i + 1]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        for i in range(26):
            for j in range(i + 1, 26):
                x_i, y_i = centers_flat[2 * i], centers_flat[2 * i + 1]
                x_j, y_j = centers_flat[2 * j], centers_flat[2 * j + 1]
                r_i = radii_flat[i]
                r_j = radii_flat[j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                if dist < r_i + r_j - 1e-12:
                    return -1
        return 0
    
    constraints.append({'type': 'ineq', 'fun': constraint_func})
    
    result = optimize.minimize(
        objective,
        x0=np.concatenate([centers.flatten(), radii]),
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
