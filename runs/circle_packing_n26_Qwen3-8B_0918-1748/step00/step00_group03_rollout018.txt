import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint
from scipy.optimize import minimize_scalar

def run_packing():
    # Define initial hexagonal grid
    # 7 rows with 4, 5, 5, 5, 5, 4, 4 circles
    num_rows = 7
    num_cols_per_row = [4, 5, 5, 5, 5, 4, 4]
    num_circles = sum(num_cols_per_row)
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Define initial radius for all circles
    radius_base = 0.15
    radius = radius_base

    # Build initial hexagonal packing
    row_index = 0
    for i, col_count in enumerate(num_cols_per_row):
        col_index = 0
        for j in range(col_count):
            x = j + 0.5 * (col_count - (j % 2)) / 2
            y = i + 0.5
            centers[row_index] = np.array([x, y])
            radii[row_index] = radius
            row_index += 1
        # Adjust row index to account for offset in hexagonal packing
        row_index += (col_count - (col_count % 2)) // 2

    # Define function to compute sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints for non-overlapping circles and boundary conditions
    def constraints_func(centers, radii, max_radius):
        constraints = []
        for i in range(len(radii)):
            # Ensure circle stays within square
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return False
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
            # Ensure circle is non-empty
            if radii[i] < 0:
                return False
            # Ensure circle does not overlap with others
            for j in range(i + 1, len(radii)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Initial guess for radii
    initial_guess = np.full(len(radii), radius_base)

    # Optimization to maximize total radii
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=Bounds([0]*len(radii), [1.0]*len(radii)),
        constraints=[
            NonlinearConstraint(
                lambda r: constraints_func(centers, r, 1.0),
                None,
                1.0
            )
        ]
    )

    # Extract optimized radii and centers
    radii = result.x
    # Ensure that the centers are adjusted within the square
    for i in range(len(radii)):
        x, y = centers[i]
        r = radii[i]
        # Clip to keep within [0,1]
        x = np.clip(x, 0.0, 1.0)
        y = np.clip(y, 0.0, 1.0)
        # Ensure circle lies within bounds
        if x - r < 0:
            delta = x - r
            x = 0
            r = x + delta  # make sure at least one bound is satisfied
        elif x + r > 1:
            delta = (x + r) - 1
            x = 1 - delta
            r = x - (x - delta)
        if y - r < 0:
            delta = y - r
            y = 0
            r = y + delta  # make sure at least one bound is satisfied
        elif y + r > 1:
            delta = (y + r) - 1
            y = 1 - delta
            r = y - (y - delta)
        centers[i] = [x, y]
        radii[i] = r

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
