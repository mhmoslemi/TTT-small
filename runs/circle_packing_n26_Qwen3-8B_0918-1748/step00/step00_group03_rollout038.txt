import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess for centers: grid-based with some randomness
    num_circles = 26
    centers = np.random.rand(num_circles, 2)
    radii = np.full(num_circles, 0.05)

    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(i, j):
        def cons(radii):
            dist = np.linalg.norm(centers[i] - centers[j])
            return dist - radii[i] - radii[j] + 1e-8
        return cons

    def constraint_boundary(i):
        def cons(radii):
            x, y = centers[i]
            r = radii[i]
            return (x - r) - 0.0 + 1e-8
        return cons

    def constraint_boundary_right(i):
        def cons(radii):
            x, y = centers[i]
            r = radii[i]
            return (1.0 - x - r) + 1e-8
        return cons

    def constraint_boundary_top(i):
        def cons(radii):
            x, y = centers[i]
            r = radii[i]
            return (1.0 - y - r) + 1e-8
        return cons

    def constraint_boundary_bottom(i):
        def cons(radii):
            x, y = centers[i]
            r = radii[i]
            return (y - r) + 1e-8
        return cons

    # Create constraints
    constraints = []

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})

    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_right(i)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_top(i)})
        constraints.append({'type': 'ineq', 'fun': constraint_boundary_bottom(i)})

    # Optimization
    result = optimize.minimize(
        objective,
        radii,
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1.0) for _ in range(num_circles)],
        tol=1e-8
    )

    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
