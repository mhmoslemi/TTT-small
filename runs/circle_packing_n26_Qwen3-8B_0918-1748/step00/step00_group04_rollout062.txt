import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial import distance_matrix
from scipy.optimize import Bounds
from functools import partial

def run_packing():
    # Define initial positions and radii for the 26 circles
    # Corner circles (4)
    corners = np.array([
        [0.0, 0.0],
        [1.0, 0.0],
        [1.0, 1.0],
        [0.0, 1.0]
    ])
    radii_corners = np.array([0.15, 0.15, 0.15, 0.15])

    # Edge circles (12): 3 on each side (excluding corners)
    edge_centers = []
    edge_radii = []

    # Bottom edge (excluding corner at (0,0))
    for i in range(1, 4):
        x = i / 3
        y = 0.0
        edge_centers.append([x, y])
        edge_radii.append(0.1)

    # Right edge (excluding corner at (1,0))
    for i in range(1, 4):
        x = 1.0
        y = i / 3
        edge_centers.append([x, y])
        edge_radii.append(0.1)

    # Top edge (excluding corner at (1,1))
    for i in range(2, -1, -1):
        x = i / 3
        y = 1.0
        edge_centers.append([x, y])
        edge_radii.append(0.1)

    # Left edge (excluding corner at (0,1))
    for i in range(2, -1, -1):
        x = 0.0
        y = i / 3
        edge_centers.append([x, y])
        edge_radii.append(0.1)

    # Inner grid (10): 2 rows of 5 circles in the center
    inner_centers = []
    inner_radii = []
    for i in range(2):
        for j in range(5):
            x = 0.2 + i * 0.25
            y = 0.2 + j * 0.25
            inner_centers.append([x, y])
            inner_radii.append(0.1)

    centers = np.vstack([corners, edge_centers, inner_centers])
    radii = np.concatenate([radii_corners, edge_radii, inner_radii])

    def objective(x):
        return -np.sum(radii)

    def constraint_overlap(x, i, j):
        dx = x[2*i] - x[2*j]
        dy = x[2*i + 1] - x[2*j + 1]
        dist = np.sqrt(dx*dx + dy*dy)
        return dist - (radii[i] + radii[j]) >= 0

    def constraint_boundary(x, i):
        x_center = x[2*i]
        y_center = x[2*i + 1]
        return x_center - radii[i] >= 0 and x_center + radii[i] <= 1 and y_center - radii[i] >= 0 and y_center + radii[i] <= 1

    bounds = []
    for i in range(len(centers)):
        x = centers[i, 0]
        y = centers[i, 1]
        bounds.append((x - 1, x + 1))
        bounds.append((y - 1, y + 1))

    x0 = np.concatenate([centers.reshape(-1), radii])

    cons = []
    for i in range(len(centers)):
        cons.append({'type': 'ineq', 'fun': partial(constraint_boundary, i=i), 'args': ()})
        for j in range(i + 1, len(centers)):
            cons.append({'type': 'ineq', 'fun': partial(constraint_overlap, i=i, j=j), 'args': ()})

    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-8, options={'maxiter': 1000, 'ftol': 1e-8})

    x_opt = result.x
    centers_opt = x_opt[:2*len(centers)].reshape(-1, 2)
    radii_opt = x_opt[2*len(centers):]

    # Post-processing: refine radii for boundary circles
    for i in range(len(centers_opt)):
        x, y = centers_opt[i]
        r = radii_opt[i]
        if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
            if x - r < -1e-8:
                radii_opt[i] = x - 1e-8
            elif x + r > 1 + 1e-8:
                radii_opt[i] = 1 - x - 1e-8
            elif y - r < -1e-8:
                radii_opt[i] = y - 1e-8
            elif y + r > 1 + 1e-8:
                radii_opt[i] = 1 - y - 1e-8

    # Validate packing
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    sum_radii = np.sum(radii_opt)
    return (centers_opt, radii_opt, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
