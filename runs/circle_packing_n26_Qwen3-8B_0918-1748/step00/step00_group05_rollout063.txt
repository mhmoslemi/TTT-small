import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial import distance_matrix

def run_packing():
    def objective_function(radii, centers):
        return -np.sum(radii)

    def constraint_overlap(radii, centers):
        n = len(radii)
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.sum(radii)

    def constraint_boundary(radii, centers):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            if x - radii[i] < -1e-8 or x + radii[i] > 1 + 1e-8 or y - radii[i] < -1e-8 or y + radii[i] > 1 + 1e-8:
                return -np.inf
        return np.sum(radii)

    # Initial guess: 5 rows of 5 circles + 1 in the center (total 26)
    initial_centers = np.array([

    ])

        def objective(params):
            return -np.sum(radii)
        
        def constraints(params):
            for i in range(len(centers)):
                x, y = params[2*i], params[2*i + 1]
                r = radii[i]
                if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                    return -np.inf
            for i in range(len(centers)):
                for j in range(i + 1, len(centers)):
                    x1, y1 = params[2*i], params[2*i + 1]

                    dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                    if dist < radii[i] + radii[j] - 1e-8:
                        return -np.inf
            return np.sum(radii)
        
        bounds = [(0, 1) for _ in range(2 * len(centers))]
        result = minimize(objective, x0=params, bounds=bounds, constraints=constraints)
        return result["x"].reshape(-1, 2)
    
    # Refine centers and radii
    params = np.concatenate([initial_centers.reshape(-1), initial_radii])
    result = minimize(objective_function, x0=params, bounds=Bounds([0, 1] * 26), constraints=constraint_overlap)
    params_result = result["x"]
    refined_centers = params_result[:2*26].reshape(-1, 2)
    refined_radii = params_result[2*26:]
    
    centers = refined_centers
    radii = refined_radii
    sum_radii = np.sum(radii)
    
    # Further refinement using Voronoi diagram and distances
    def refine_voronoi(centers, radii):
        n = len(centers)
        D = cdist(centers, centers)
        for i in range(n):
            for j in range(i + 1, n):
                d = D[i, j]
                if d < radii[i] + radii[j] - 1e-8:
                    delta = (radii[i] + radii[j] - d) / 2
                    if radii[i] > radii[j]:
                        centers[i] += delta * (centers[i] - centers[j]) / d
                        radii[i] -= delta
                    else:
                        centers[j] += delta * (centers[j] - centers[i]) / d
                        radii[j] -= delta
        return centers, radii
    
    centers, radii = refine_voronoi(centers, radii)
    
    # Final check and return
    return (centers, radii, np.sum(radii))
