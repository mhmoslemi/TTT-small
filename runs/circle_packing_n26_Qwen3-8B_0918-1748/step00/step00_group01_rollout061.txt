        centers = np.array([[

            x, y = centers[i]

                dx = centers[i, 0] - centers[j, 0]

                dist = np.sqrt(dx**2 + dy**2)

                    violations.append(1)
        return np.sum(violations)
    
    def constraint(x):
        return constraints_func(x)
    
    # Set up bounds for radii
    bounds = [(0, 1) for _ in range(26)]
    
    # Run optimization
    result = minimize(
        objective,
        initial_radii,
        bounds=bounds,
