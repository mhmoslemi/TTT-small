import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds, LinearConstraint
from scipy.spatial.distance import cdist
from scipy.constants import pi
import time

def run_packing():
    # Define the number of circles
    N = 26

    # Step 1: Generate an initial guess using a hexagonal grid
    # We'll use a modified hexagonal grid with varying rows to fit 26 circles
    # We'll initialize them using a hexagonal lattice adjusted to fit the square

    # Hexagonal grid parameters
    rows = 5
    cols = 5
    remaining = N - rows * cols

    # Generate initial positions using hexagonal grid
    spacing = 1.0 / (rows + 1)
    angles = np.linspace(0, 2 * np.pi, rows + 1)
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * spacing * np.sqrt(3) / 2
            centers.append((x, y))
    # Add remaining circles
    for _ in range(remaining):
        # Add a circle near the bottom right corner
        idx = len(centers) - 1
        x, y = centers[idx]
        if idx % cols == cols - 1:
            # Right edge, go down
            x_new = x
            y_new = y + spacing * np.sqrt(3) / 2
        else:
            # Bottom edge, go right
            x_new = x + spacing
            y_new = y
        centers.append((x_new, y_new))

    # Move centers to be within [0,1]x[0,1]
    centers = np.array(centers)
    centers += np.random.uniform(-0.05, 0.05, size=centers.shape)
    centers = np.clip(centers, 0, 1)

    # Step 2: Optimize the radii and positions to maximize sum of radii
    # We'll use a constrained optimization approach

    # Define the objective function: maximize the sum of radii
    def objective_function(vars):
        radii = vars[:N]
        X = vars[N:]
        X = X.reshape((N, 2))
        return -np.sum(radii)

    # Define constraints
    constraints = []
    # Constraints: each circle must be within the unit square
    for i in range(N):
        def constraint_i(vars):
            radii = vars[:N]
            X = vars[N:]
            x, y = X[i]
            r = radii[i]
            return -r + (1 - x) * (1 - y)
        constraints.append({'type': 'ineq', 'fun': lambda vars, i=i: constraint_i(vars)})

    # Constraints: distance between circles must be at least sum of radii
    for i in range(N):
        for j in range(i + 1, N):
            def constraint_ij(vars):
                radii = vars[:N]
                X = vars[N:]
                xi, yi = X[i]
                xj, yj = X[j]
                ri = radii[i]
                rj = radii[j]
                return -np.sqrt((xi - xj)**2 + (yi - yj)**2) + ri + rj
            constraints.append({'type': 'ineq', 'fun': lambda vars, i=i, j=j: constraint_ij(vars)})

    # Bounds: radii must be non-negative, positions must be within [0,1]^2
    bounds = []
    for _ in range(N):
        bounds.append((0, 1))
    for _ in range(N * 2):
        bounds.append((0, 1))

    # Initial guess: uniform radii and positions
    initial_vars = np.zeros(N * 2 + N)
    initial_vars[:N] = 0.1
    initial_vars[N:] = centers.flatten()

    # Run the optimization
    result = minimize(
        objective_function,
        initial_vars,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'eps': 1e-6, 'maxiter': 1000}
    )

    # Extract results
    radii = result.x[:N]
    centers_optimized = result.x[N:].reshape((N, 2))

    # Final validation step
    def validate_and_dump():
        # Check if optimization was successful
        if not result.success:
            raise ValueError("Optimization failed")

        # Check if any radius is negative
        if np.any(radii < 0 - 1e-10):
            raise ValueError("Negative radii found")

        # Check boundary constraints
        for i in range(N):
            x, y = centers_optimized[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12:
                raise ValueError(f"Circle {i} is outside on x-axis")
            if y - r < -1e-12 or y + r > 1 + 1e-12:
                raise ValueError(f"Circle {i} is outside on y-axis")

        # Check pairwise distances
        for i in range(N):
            for j in range(i + 1, N):
                dist = np.sqrt((centers_optimized[i, 0] - centers_optimized[j, 0])**2 + (centers_optimized[i, 1] - centers_optimized[j, 1])**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    raise ValueError(f"Circles {i} and {j} overlap")

        return centers_optimized, radii, np.sum(radii)

    # Return the valid packing
    centers, radii, sum_radii = validate_and_dump()
    return centers, radii, sum_radii
