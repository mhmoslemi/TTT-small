import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Define a pattern with 26 circles
    # Initial estimate: 16 circles in a hexagonal grid, 10 corner circles
    # Approximate positions and radii based on known dense packings
    
    # Hexagonal grid for central cluster (16 circles)
    def hexagonal_grid(n, radius_base):
        centers = []
        for i in range(n):
            row = i // 4
            col = i % 4
            x = col + 0.5
            y = row + 0.5
            centers.append([x, y])
        return np.array(centers), np.full(n, radius_base)
    
    # Corner circles
    def corner_circles(n, radius_base):
        centers = []
        for i in range(n):
            if i == 0:
                centers.append([0.1, 0.1])  # corner near (0,0)
            elif i == 1:
                centers.append([0.9, 0.1])  # corner near (1,0)
            elif i == 2:
                centers.append([0.1, 0.9])  # corner near (0,1)
            elif i == 3:
                centers.append([0.9, 0.9])  # corner near (1,1)
        return np.array(centers), np.full(n, radius_base)
    
    # Create initial configuration with 16 hexagonal circles and 10 corner circles
    central_centers, central_radii = hexagonal_grid(16, 0.12)
    corner_centers, corner_radii = corner_circles(10, 0.1)
    
    centers = np.vstack([central_centers, corner_centers])
    radii = np.concatenate([central_radii, corner_radii])
    
    # Objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize
    
    # Constraints
    def constraint(x):
        return x - 0.0
    
    # Initial guess for radii
    initial_guess = radii
    
    # Optimization
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        jac=None,
        bounds=[(0.0, 1.0) for _ in range(26)],
        constraints=[{'type': 'ineq', 'fun': lambda x: np.min(
            (1 - (centers - x[None, :][:, 0])**2 - (centers - x[None, :][:, 1])**2 - (radii - x[None, :])**2 - (radii + x[None, :])**2).flatten())}]
    )
    
    if not result.success:
        raise RuntimeError("Optimization failed")
    
    # Final centers and radii
    final_centers = centers
    final_radii = result.x
    
    # Final sum of radii
    final_sum = np.sum(final_radii)
    
    return final_centers, final_radii, final_sum
