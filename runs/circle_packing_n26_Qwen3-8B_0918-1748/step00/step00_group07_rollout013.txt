import numpy as np
from scipy.optimize import minimize, constraint, Bounds
from scipy.optimize import LinearConstraint, NonlinearConstraint

def run_packing():
    # Initial grid-based configuration with hexagonal packing
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    # Start with a grid layout for a hexagonal packing
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col + 0.5 * (col % 2) / 5
        y = row + 0.5 * (row % 2) / 5
        centers[i] = [x, y]
    # Initial radii based on spacing
    initial_radius = 0.1
    radii.fill(initial_radius)
    
    # Define objective function
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints
    def constraint_func(radii):
        dists = np.zeros((26, 26))
        for i in range(26):
            for j in range(i + 1, 26):
                dists[i, j] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dists - (radii[:, np.newaxis] + radii[np.newaxis, :])
    
    nonlin_cons = NonlinearConstraint(
        constraint_func, 
        -1e-12, 0, 
        jac=lambda radii: np.zeros((26, 26)), 
        name="non_overlap"
    )
    
    # Bounds for radii
    bounds = Bounds(0, 1)
    
    # Optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=nonlin_cons,
        options={'ftol': 1e-6, 'maxiter': 1000}
    )
    
    centers = np.array(centers)
    radii = np.array(result.x)
