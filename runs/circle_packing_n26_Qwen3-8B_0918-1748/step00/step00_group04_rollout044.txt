import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run_packing():
    # Define initial grid structure with hexagonal packing
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Hexagonal packing with 5 rows
    rows = 5
    cols = 5
    r = 0.1
    radius = r
    offset = radius * np.sqrt(3)
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * 2 * radius
                y = i * (2 * radius)
            else:
                x = j * 2 * radius + radius
                y = i * (2 * radius) + radius
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = radius
    
    # Optimize the radii to maximize total sum
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        n = len(radii)
        constraints = []
        for i in range(n):
            constraints.append((radii[i] - 1e-12) >= 0)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append((dist - (radii[i] + radii[j] - 1e-12)) >= 0)
        return constraints
    
    bounds = Bounds(0, 1)
    constraints = LinearConstraint(np.eye(n), -np.inf, np.inf)
    
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=constraints)
    radii_opt = result.x
    sum_radii = np.sum(radii_opt)
    
    # Adjust positions for better packing
    # For simplicity, we will not adjust positions in this version but ensure constraints are met
    
    # Revalidation to ensure everything is okay
    if not validate_packing(centers, radii_opt)[0]:
        raise ValueError("Validation failed after optimization")
    
    return centers, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
