import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define initial positions with a hexagonal grid pattern
    num_circles = 26
    num_rows = 6
    num_cols = 5
    spacing = 1.0 / num_rows  # vertical spacing
    horizontal_spacing = 1.0 / num_cols  # horizontal spacing

    # Create initial grid positions
    positions = []
    for row in range(num_rows):
        y = row * spacing
        for col in range(num_cols):
            x = col * horizontal_spacing
            positions.append((x, y))
    positions = np.array(positions)

    # Add one more circle in the center
    positions = np.vstack((positions, [0.5, 0.5]))

    # Initial radii for the grid
    radii = np.ones(num_circles)

    # Define the initial configuration
    centers = positions
    radii = radii

    # Run optimization to maximize the sum of radii
    def objective(params):
        # Unpack parameters into centers and radii
        new_centers = params[:num_circles * 2].reshape((num_circles, 2))
        new_radii = params[num_circles * 2:].astype(float)

        # Check for valid configuration
        valid, _ = validate_packing(new_centers, new_radii)
        if not valid:
            return float('-inf')

        # Compute sum of radii
        return np.sum(new_radii)

    # Initial guess
    initial_guess = np.concatenate((centers.flatten(), radii))

    # Optimization settings
    bounds = []
    for _ in range(num_circles * 2):
        bounds.append((0.0, 1.0))  # For centers x and y
    for _ in range(num_circles):
        bounds.append((0.0, 1.0))  # For radii

    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8}
    )

    # Extract the optimized centers and radii
    centers_opt = result.x[:num_circles * 2].reshape((num_circles, 2))
    radii_opt = result.x[num_circles * 2:]

    # Validate the final configuration
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Validation failed: {message}")

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
