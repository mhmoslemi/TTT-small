import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    n = 26
    # Initial grid-based placement with hexagonal-inspired adjustments
    # Top 3 rows with 2 circles each (for total of 6)
    # Middle 4 rows with 3 circles each (for total of 12)
    # Bottom 2 rows with 2 circles each (for total of 4)
    # Total so far: 6 + 12 + 4 = 22
    # Add 4 more circles to fill in the gaps
    centers = []
    radii = []

    # Horizontal spacing and vertical spacing
    h_spacing = 0.5
    v_spacing = 0.5

    # Top rows
    for i in range(3):
        x = i * h_spacing + 0.2
        y = 1 - (i + 0.5) * v_spacing
        if i == 0:
            # First two circles in top row
            centers.append([x, y])
            radii.append(0.1)
            centers.append([x + h_spacing - 0.2, y])
            radii.append(0.1)
        else:
            # Middle rows
            x = 0.5
            y = 1 - (i + 0.5) * v_spacing
            centers.append([x, y])
            radii.append(0.1)

    # Middle rows
    for i in range(4):
        x = i * h_spacing + 0.2
        y = 0.5 - (i + 0.5) * v_spacing
        if i == 0 or i == 3:
            centers.append([x, y])
            radii.append(0.1)
        else:
            centers.append([x, y])
            radii.append(0.1)

    # Bottom rows
    for i in range(2):
        x = i * h_spacing + 0.2
        y = 0.5 - (i + 0.5) * v_spacing
        centers.append([x, y])
        radii.append(0.1)

    # Add 4 extra circles
    # First extra circle - bottom right
    centers.append([0.8, 0.3])
    radii.append(0.1)
    # Second extra circle - top right
    centers.append([0.8, 0.8])
    radii.append(0.1)
    # Third extra circle - top left
    centers.append([0.2, 0.8])
    radii.append(0.1)
    # Fourth extra circle - bottom left
    centers.append([0.2, 0.3])
    radii.append(0.1)

    # Optimize the configuration
    def objective(params):
        flat_params = params.reshape(-1)
        centers = flat_params[:n*2].reshape(n, 2)
        radii = flat_params[n*2:].reshape(n)
        sum_radii = np.sum(radii)
        return -sum_radii  # Minimize negative to maximize

    def constraints(params):
        flat_params = params.reshape(-1)
        centers = flat_params[:n*2].reshape(n, 2)
        radii = flat_params[n*2:].reshape(n)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Boundary constraints
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    bounds = Bounds(
        lb=np.array([0.0 for _ in range(n*2)] + [0.0 for _ in range(n)]),
        ub=np.array([1.0 for _ in range(n*2)] + [1.0 for _ in range(n)])
    )

    # Flatten initial centers and radii
    initial_centers = np.array(centers)
    initial_radii = np.array(radii)
    initial_params = np.concatenate([initial_centers.flatten(), initial_radii])

    result = minimize(
        objective,
        initial_params,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        options={'maxiter': 500, 'ftol': 1e-10}
    )

    centers = result.x[:n*2].reshape(n, 2)
    radii = result.x[n*2:].reshape(n)
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
