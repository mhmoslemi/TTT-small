import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    # Initial hexagonal packing with 26 circles
    def hexagonal_packing(n):
        rows = int(np.ceil(n / 2))
        col = int(np.ceil(n / rows))
        centers = np.zeros((n, 2))
        x, y = 0.5, 0.5
        for i in range(n):
            row = i // col
            col_idx = i % col
            if row % 2 == 0:
                x = 0.5 - (col_idx + 0.5) / col
            else:
                x = 0.5 - (col_idx + 0.5) / col + 1 / (2 * col)
            y = 0.5 - (row + 0.5) / rows
            centers[i] = (x, y)
        return centers

    # Initial centers and radii
    centers = hexagonal_packing(26)
    radii = np.full(26, 0.1)

    # Define objective function to maximize sum of radii
    def objective(r):
        return -np.sum(r)

    # Define constraints
    def constraint_func(r):
        dists = distance.cdist(centers, centers)
        for i in range(26):
            for j in range(i + 1, 26):
                if dists[i, j] < r[i] + r[j] - 1e-8:
                    return dists[i, j] - r[i] - r[j] + 1e-8
        return 0

    # Optimize
    bounds = [(0, 1) for _ in range(26)]
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={"type": "ineq", "fun": constraint_func},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 1000}
    )

    radii = result.x
    # Adjust radii to ensure no circle exceeds the boundary
    for i in range(26):
        r = radii[i]
        x, y = centers[i]
        if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
            radii[i] = np.min([x, 1 - x, y, 1 - y])

    # Final validation
    if validate_packing(centers, radii)[0]:
        return (centers, radii, np.sum(radii))
    else:
        raise ValueError("Packing validation failed")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
