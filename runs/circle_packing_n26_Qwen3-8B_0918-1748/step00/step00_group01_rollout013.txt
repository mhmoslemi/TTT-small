import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing():
    # Initial guess: 5x5 grid of circles, slightly offset from the square boundaries
    centers = np.zeros((26, 2))
    for i in range(5):
        for j in range(5):
            centers[5*i + j] = [0.1 + i * 0.2, 0.1 + j * 0.2]
    # Add an extra circle at the bottom right
    centers[25] = [0.9, 0.9]
    radii = np.zeros(26)
    # Initial guess for radii: small value
    for i in range(25):
        radii[i] = 0.05
    radii[25] = 0.05

    # Objective function: maximize sum of radii (minimize negative sum)
    def objective(x):
        # x has shape (26*2 + 26) = 78
        # Unpack centers and radii
        flat_centers = x[:26*2].reshape(26, 2)
        radii = x[26*2:]
        return -np.sum(radii)

    # Bounds: for each center (x, y), 0 <= x <= 1, 0 <= y <= 1
    # for each radius, 0 <= r <= 1
    bounds = Bounds(np.zeros(26*2 + 26), np.ones(26*2 + 26))

    # Nonlinear constraint: for each pair of circles, distance >= r_i + r_j - 1e-12
    def constraint_func(x):
        flat_centers = x[:26*2].reshape(26, 2)
        radii = x[26*2:]
        constraints = []
        for i in range(26):
            for j in range(i + 1, 26):
                dx = flat_centers[i, 0] - flat_centers[j, 0]
                dy = flat_centers[i, 1] - flat_centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_i = radii[i]
                r_j = radii[j]
                constraints.append(dist - r_i - r_j + 1e-12)
        return np.array(constraints)

    # Ensure that each circle is within the square (r <= distance to nearest edge)
    def boundary_constraint(x):
        flat_centers = x[:26*2].reshape(26, 2)
        radii = x[26*2:]
        constraints = []
        for i in range(26):
            x_i, y_i = flat_centers[i]
            r_i = radii[i]
            # Distance to left edge
            d_left = x_i - r_i
            if d_left < -1e-12:
                constraints.append(d_left + 1e-12)
            # Distance to right edge
            d_right = 1 - x_i - r_i
            if d_right < -1e-12:
                constraints.append(d_right + 1e-12)
            # Distance to bottom edge
            d_bottom = y_i - r_i
            if d_bottom < -1e-12:
                constraints.append(d_bottom + 1e-12)
            # Distance to top edge
            d_top = 1 - y_i - r_i
            if d_top < -1e-12:
                constraints.append(d_top + 1e-12)
        return np.array(constraints)

    # Create constraints
    cons = [
        NonlinearConstraint(constraint_func, lb=-np.inf, ub=0),
        NonlinearConstraint(boundary_constraint, lb=-np.inf, ub=0)
    ]

    # Initial guess
    initial_guess = np.zeros(26*2 + 26)
    initial_guess[:26*2] = np.array([0.1 + i * 0.2 for i in range(26)] + [0.1 + j * 0.2 for j in range(26)])
    initial_guess[26*2:] = np.full(26, 0.05)

    # Run the optimization
    result = minimize(
        objective,
        initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    # Check if the optimization was successful
    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    # Extract centers and radii
    flat_centers = result.x[:26*2].reshape(26, 2)
    radii = result.x[26*2:]

    # Return centers, radii, and their sum
    return flat_centers, radii, np.sum(radii)
