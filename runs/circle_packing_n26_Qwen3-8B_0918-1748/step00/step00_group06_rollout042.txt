import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint
import time

def run_packing():
    n = 26
    # Initial random packing with equal radii
    centers = np.random.rand(n, 2)
    # Compute initial radii such that all circles are within the square
    min_dist = np.min(np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :])**2, axis=2)))
    initial_radius = min_dist / (n + 1)
    radii = np.full(n, initial_radius)
    
    # Define the optimization problem
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_dist(radii):
        dist_matrix = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :])**2, axis=2))
        return dist_matrix - (radii[:, np.newaxis] + radii[np.newaxis, :])
    
    def constraint_boundary(radii):
        x, y = centers.T
        return np.minimum(x - radii, 1 - x - radii, y - radii, 1 - y - radii)
    
    # Set up bounds for radii
    bounds = Bounds(0, 1)
    
    # Nonlinear constraints
    nonlin_cons = NonlinearConstraint(constraint_dist, -1e-12, 0)
    nonlin_cons2 = NonlinearConstraint(lambda r: constraint_boundary(r), -1e-12, 0)
    
    # Initial guess
    initial_guess = initial_radius * np.ones(n)
    
    # Run optimization
    result = minimize(
        objective, 
        initial_guess, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=[nonlin_cons, nonlin_cons2],
        options={'ftol': 1e-10, 'maxiter': 500}
    )
    
    centers = centers
    radii = result.x
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
