import numpy as np
import scipy.optimize as optimize

def run_packing():
    np.random.seed(42)
    
    # Initial grid-based seed configuration
    def initial_centers():
        grid = np.zeros((26, 2))
        grid[0] = [0.5, 0.5]
        grid[1:17] = np.array([(i / 10 + 0.05, 0.5) for i in range(1, 17)])
        grid[17:33] = np.array([(i / 10 + 0.05, 0.5 + 0.25) for i in range(1, 17)])
        grid = grid[:26]
        return grid
    
    centers = initial_centers()
    radii = np.full(26, 0.04)
    
    def objective(radii):
        return -np.sum(radii)  # Maximize sum by minimizing negative sum
    
    def constraint_overlap(i, j):
        def cons(x):
            dx = x[i][0] - x[j][0]
            dy = x[i][1] - x[j][1]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - (radii[i] + radii[j])
        return cons
    
    def constraint_boundary(i):
        def cons(x):
            x_i, y_i = x[i]
            return np.array([x_i - radii[i], x_i + radii[i], y_i - radii[i], y_i + radii[i]])
        return cons
    
    constraints = []
    for i in range(26):
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})
    
    for i in range(26):
        for j in range(i + 1, 26):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})
    
    def optimize_packing():
        bounds = []
        for i in range(26):
            bounds.append((0, 1))
        
        result = optimize.minimize(
            objective,
            centers.flatten(),
            bounds=bounds,
            constraints=constraints,
            method='SLSQP',
            options={'ftol': 1e-12, 'maxiter': 1000}
        )
        
        if not result.success:
            raise RuntimeError(f"Optimization failed: {result.message}")
        
        return result.x.reshape(26, 2), radii
    
    centers, radii = optimize_packing()
    
    # Final validation and return
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
