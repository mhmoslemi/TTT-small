```python
import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing():
    n = 26
    # Initial guess for positions: hexagonal packing
    positions = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col / 4.0
        y = row / 4.0
        if row % 2 == 1:
            y += 0.5 / 4.0
        positions[i] = [x, y]
        radii[i] = 0.05

    # Optimization variables: x, y, r
    x0 = positions.flatten()
    x0 = np.concatenate((x0, radii))

    # Objective function: maximize sum of radii (minimize negative sum)
    def objective(x):
        pos = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        return -np.sum(radii)

    # Constraints: circles stay within unit square
    def bounds(x):
        pos = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        return np.concatenate((pos.flatten(), radii))

    # Bounds on positions and radii
    pos_bounds = [(0, 1) for _ in range(2*n)]
    radii_bounds = [(0, 1) for _ in range(n)]
    bounds = Bounds(pos_bounds + radii_bounds)

    # Nonlinear constraints: distance between circles >= sum of radii
    def constraint_func(x):
        pos = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        dists = np.sqrt(np.sum((pos[:, np.newaxis] - pos[np.newaxis, :])**2, axis=2))
        return dists - (radii[:, np.newaxis] + radii[np.newaxis, :])

    # Bounds are already handled by the position and radius constraints
    # We need to ensure that circles are at least 0 distance from each other, so:
    # dists - (radii[:, np.newaxis] + radii[np.newaxis, :]) >= 0
    # So the inequality is: dists >= radii[:, np.newaxis] + radii[np.newaxis, :]
    # Thus, constraint_func(x) >= 0
    # We use the NonlinearConstraint with inequality > 0
    nonlin_cons = NonlinearConstraint(constraint_func, -np.inf, 0)

    # Minimize the negative sum (maximize the sum)
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[nonlin_cons],
        method='SLSQP',
        options={'ftol': 1e-6, 'maxiter': 1000}
    )

    # Extract centers and radii
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    sum_radii = np.sum(radii)

    # Final check for valid packing
    assert np.all(centers >= 0) and np.all(centers <= 1)
    assert np.all(radii >= 0)

    return centers, radii, sum_radii
```
