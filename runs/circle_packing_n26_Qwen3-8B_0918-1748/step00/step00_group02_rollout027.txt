import numpy as np
import random
from scipy.optimize import minimize

def run_packing():
    # Initial packing: hexagonal grid with rows and columns
    num_circles = 26
    radius_guess = np.ones(num_circles) * 0.03
    centers = np.random.rand(num_circles, 2)
    
    # Perturb initial centers to avoid exact overlap
    for i in range(num_circles):
        centers[i] = [np.random.uniform(0.05, 0.95), np.random.uniform(0.05, 0.95)]
    
    # Define objective function to maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative total to maximize
    
    # Define constraints for boundaries
    def boundary_constraint(radii, centers, i):
        x, y = centers[i]
        return x - radii[i] >= -1e-12 and x + radii[i] <= 1 + 1e-12 and y - radii[i] >= -1e-12 and y + radii[i] <= 1 + 1e-12
    
    # Define constraints for circle-circle distances
    def circle_circle_constraint(radii, centers, i, j):
        dx = centers[i][0] - centers[j][0]
        dy = centers[i][1] - centers[j][1]
        dist = np.sqrt(dx*dx + dy*dy)
        return dist >= radii[i] + radii[j] - 1e-12
    
    # Prepare constraints for optimization
    bounds = [(0.0, 1.0) for _ in range(num_circles)]
    constraints = []
    for i in range(num_circles):
        constraints.append(('ineq', lambda radii: boundary_constraint(radii, centers, i)))
    
    for i in range(num_circles):
        for j in range(i+1, num_circles):
            constraints.append(('ineq', lambda radii, i=i, j=j: circle_circle_constraint(radii, centers, i, j)))
    
    # Initial guess for radii
    initial_guess = np.random.uniform(0.01, 0.1, num_circles)
    
    # Optimize
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'disp': False, 'maxiter': 1000}
    )
    
    radii = result.x
    centers = np.array([[np.random.uniform(0.05, 0.95), np.random.uniform(0.05, 0.95)] for _ in range(num_circles)])
    
    # Final validation pass for center placement
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            r = max(0, r - 1e-12)
            radii[i] = r
    
    for i in range(num_circles):
        for j in range(i+1, num_circles):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - 1e-12:
                overlap = (radii[i] + radii[j]) - dist
                radii[i] = max(0, radii[i] - overlap / 2)
                radii[j] = max(0, radii[j] - overlap / 2)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
