import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    # Initial configuration: 5x5 grid with spacing between centers
    # Let's start with 25 circles in a grid and one extra circle near a corner
    centers = []
    radii = []
    total_sum = 0
    
    # Create a grid of 5x5 circles
    n = 5
    grid_spacing = 1.0 / (n - 1)
    
    for i in range(n):
        for j in range(n):
            x = i * grid_spacing
            y = j * grid_spacing
            centers.append([x, y])
            # We'll use radius as (1/(2*sqrt(2))) for a tight grid
            r = 1 / (2 * np.sqrt(2)) 
            radii.append(r)
    
    # Add one more circle near the corner to utilize space
    # Place it near (0.2, 0.2) with a radius that doesn't overlap with the grid
    x = 0.2
    y = 0.2
    # Compute the radius that doesn't overlap with the nearest grid points
    # nearest grid point is (0,0), so distance is sqrt((0.2)^2 + (0.2)^2)
    dist = np.sqrt(0.2**2 + 0.2**2)
    r = (dist - 1e-12) / 2
    centers.append([x, y])
    radii.append(r)
    
    # Now define an optimization problem to maximize the sum of radii
    # We define a function to compute the total sum of radii given some parameters
    # and constraints to prevent overlaps and stay within square
    
    def objective(params):
        # params: list of radii for existing circles
        # The first 25 are grid points, the last one is the corner
        # We need to adjust the radii to maximize total_sum
        # The corner circle's radius is already determined based on the grid
        # But let's try to refine all radii
        # For simplicity, we'll scale all radii by a factor
        scale = np.max(params) if params else 1.0
        return -sum(params) * scale  # Minimize the negative to maximize the sum
    
    def constraints(params):
        # Check if any circle is outside the square
        for idx, r in enumerate(params):
            x, y = centers[idx]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        # Check if any two circles overlap
        for i in range(len(params)):
            for j in range(i + 1, len(params)):
                x1, y1 = centers[i]
                r1 = params[i]
                x2, y2 = centers[j]
                r2 = params[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2 - 1e-12:
                    return False
        return True
    
    # Initial guess for the radii
    initial_guess = [1 / (2 * np.sqrt(2))] * 25 + [0.1]
    
    # Set bounds: radii must be non-negative
    bounds = [(0, None) for _ in range(26)]
    
    # Use a constrained optimization algorithm
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints={"type": "constraint", "fun": constraints},
        method="SLSQP"
    )
    
    # Extract the optimized radii
    final_radii = result.x
    final_centers = centers
    
    # Calculate the total sum of radii
    total_sum = np.sum(final_radii)
    
    return np.array(final_centers), np.array(final_radii), total_sum
