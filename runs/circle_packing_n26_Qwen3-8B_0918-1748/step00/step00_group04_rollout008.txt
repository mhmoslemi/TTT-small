            r2 = radii[j]
            return np.sqrt((x1 - x2)**2 + (y1 - y2)**2) - (r1 + r2)

            r1 = radii[i]
            return (x1 - r1) - 0.0  # x1 - r1 >= 0
        return cons

            centers = x[:26*2].reshape(26, 2)

            x1, y1 = centers[i]
            r1 = radii[i]
            return (1.0 + 1e-12) - (x1 + r1)  # x1 + r1 <= 1
