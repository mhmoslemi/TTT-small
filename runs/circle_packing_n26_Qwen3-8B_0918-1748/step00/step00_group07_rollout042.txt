```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, minimize

def run_packing():
    n = 26

    # Initial guess for circle centers in a hexagonal-like grid
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))

    angles = np.linspace(0, 2 * np.pi, cols, endpoint=False)
    radii_initial = np.full(n, 0.05)
    centers_initial = np.zeros((n, 2))

    for i in range(n):
        row = i // cols
        col = i % cols
        angle = angles[col]
        x = 0.5 * (1 - 1 / (rows + 1)) + (row % 2) * (1 / (rows + 1)) + (col + 0.5) / (cols + 1)
        y = 0.5 * (1 - 1 / (rows + 1)) + (row + 0.5) / (rows + 1)
        centers_initial[i] = [x, y]

    # Optimized parameters
    bounds = Bounds([0, 0], [1, 1])
    cons = []

    def objective(x):
        return -np.sum(x[n*2:])

    def constraint_overlap(x):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        for i in range(n):
            for j in range(i+1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return dist - radii[i] - radii[j] + 1e-8
        return 0

    def constraint_boundary(x):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        for i in range(n):
            x_center, y_center = centers[i]
            r = radii[i]
            if x_center - r < -1e-8 or x_center + r > 1 + 1e-8:
                return x_center - r + 1e-8
            if y_center - r < -1e-8 or y_center + r > 1 + 1e-8:
                return y_center - r + 1e-8
        return 0

    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[n*2 + i]})

    cons.append({'type': 'ineq', 'fun': constraint_overlap})
    cons.append({'type': 'ineq', 'fun': constraint_boundary})

    x0 = np.concatenate([centers_initial.reshape(n*2), radii_initial])

    result = minimize(objective, x0, bounds=Bounds([0, 0, 0], [1, 1, 1]), constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 500})

    centers = result.x[:n*2].reshape(n, 2)
    radii = result.x[n*2:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
```
