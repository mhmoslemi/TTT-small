import cvxpy as cp

    n = 26

            centers[5*i + j] = [x, y]
    
    # 26th circle in the center
    centers[25] = [0.5, 0.5]
    
    # Optimization variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)
    
    # Constraints: circles must be within the unit square
    constraints = []
    for i in range(n):
        constraints += [x[i, 0] - r[i] >= 0]
        constraints += [x[i, 0] + r[i] <= 1]
        constraints += [x[i, 1] - r[i] >= 0]
        constraints += [x[i, 1] + r[i] <= 1]
    
    # Constraints: circles must not overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints += [dist >= r[i] + r[j] - 1e-8]
    
    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
    
    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False)
