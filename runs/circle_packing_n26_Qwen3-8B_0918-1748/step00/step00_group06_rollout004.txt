import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial import distance
from itertools import combinations

def run_packing():
    # Initial configuration: 26 circles in a compact cluster with some at the corners
    # Start with a central cluster and 7 circles at corners and edges
    
    def objective(radii, centers):
        return -np.sum(radii)  # Maximize sum of radii
    
    def constraint_overlap(i, j, centers, radii):
        def cons(x):
            return np.sqrt(np.sum((x[i] - x[j])**2)) - (radii[i] + radii[j])
        return cons
    
    def constraint_boundary(i, centers, radii):
        def cons(x):
            x_i, y_i = centers[i]
            r_i = radii[i]
            return x_i - r_i - 1e-12  # Left boundary
        return cons
    
    def constraint_boundary_right(i, centers, radii):
        def cons(x):
            x_i, y_i = centers[i]
            r_i = radii[i]
            return 1 + 1e-12 - x_i - r_i  # Right boundary
        return cons
    
    def constraint_boundary_top(i, centers, radii):
        def cons(x):
            x_i, y_i = centers[i]
            r_i = radii[i]
            return 1 + 1e-12 - y_i - r_i  # Top boundary
        return cons
    
    def constraint_boundary_bottom(i, centers, radii):
        def cons(x):
            x_i, y_i = centers[i]
            r_i = radii[i]
            return y_i - r_i - 1e-12  # Bottom boundary
        return cons
    
    # Initial guesses: start with a central cluster and some at the corners
    n = 26
    centers = np.random.rand(n, 2)
    centers[0] = [0.5, 0.5]
    centers[1] = [0.5, 0.55]
    centers[2] = [0.45, 0.5]
    centers[3] = [0.55, 0.5]
    centers[4] = [0.4, 0.5]
    centers[5] = [0.6, 0.5]
    centers[6] = [0.45, 0.45]
    centers[7] = [0.55, 0.45]
    centers[8] = [0.4, 0.4]
    centers[9] = [0.6, 0.4]
    centers[10] = [0.45, 0.35]
    centers[11] = [0.55, 0.35]
    centers[12] = [0.4, 0.3]
    centers[13] = [0.6, 0.3]
    centers[14] = [0.45, 0.25]
    centers[15] = [0.55, 0.25]
    centers[16] = [0.4, 0.2]
    centers[17] = [0.6, 0.2]
    centers[18] = [0.45, 0.15]
    centers[19] = [0.55, 0.15]
    centers[20] = [0.4, 0.1]
    centers[21] = [0.6, 0.1]
    centers[22] = [0.45, 0.05]
    centers[23] = [0.55, 0.05]
    centers[24] = [0.45, 0.0]
    centers[25] = [0.55, 0.0]
    
    # Initial radii guess
    radii = np.random.rand(n)
    
    # Optimization variables
    variables = np.concatenate((centers.flatten(), radii))
    
    # Set up constraints
    cons = []
    
    # Boundary constraints for each circle
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': constraint_boundary(i, centers, radii)})
        cons.append({'type': 'ineq', 'fun': constraint_boundary_right(i, centers, radii)})
        cons.append({'type': 'ineq', 'fun': constraint_boundary_top(i, centers, radii)})
        cons.append({'type': 'ineq', 'fun': constraint_boundary_bottom(i, centers, radii)})
    
    # Overlap constraints for all pairs
    for i, j in combinations(range(n), 2):
        cons.append({'type': 'ineq', 'fun': constraint_overlap(i, j, centers, radii)})
    
    # Optimization problem
    bounds = [(0, 1) for _ in range(2 * n)] + [(0, 1) for _ in range(n)]
    
    result = minimize(
        fun=lambda x: objective(x[2 * n:], x[:2 * n].reshape((n, 2))),
        x0=variables,
        method='SLSQP',
        bounds=bounds,
        constraints=cons
    )
    
    centers = result.x[:2 * n].reshape((n, 2))
    radii = result.x[2 * n:]
    
    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise Exception(f"Packing is invalid: {msg}")
    
    # Calculate sum of radii
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
