import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    def objective(x):
        return -np.sum(x[::2] + x[1::2])  # Negative for minimization

    def constraint1(x):
        return x[::2] - 1e-6  # Ensure radii are non-negative

    def constraint2(x):
        for i in range(26):
            x_i = x[2*i]
            y_i = x[2*i + 1]
            r_i = x[2*i + 2]
            if x_i - r_i < -1e-6 or x_i + r_i > 1.0 + 1e-6 or y_i - r_i < -1e-6 or y_i + r_i > 1.0 + 1e-6:
                return -1.0
        return 0.0

    def constraint3(x):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = x[2*i] - x[2*j]
                dy = x[2*i + 1] - x[2*j + 1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_i = x[2*i + 2]
                r_j = x[2*j + 2]
                if dist < r_i + r_j - 1e-6:
                    return -1.0
        return 0.0

    def initial_guess():
        x = np.zeros(26 * 3)
        for i in range(26):
            col = i // 5
            row = i % 5
            x[2*i] = row / 4.0 + 0.05
            x[2*i + 1] = col / 4.0 + 0.05
            x[2*i + 2] = 0.04
        return x

    x0 = initial_guess()

    cons = [
        {'type': 'ineq', 'fun': constraint1},
        {'type': 'ineq', 'fun': constraint2},
        {'type': 'ineq', 'fun': constraint3}
    ]

    result = minimize(objective, x0, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 500})

    if not result.success:
        raise RuntimeError("Optimization failed.")

    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    for i in range(26):
        centers[i, 0] = result.x[2*i]
        centers[i, 1] = result.x[2*i + 1]
        radii[i] = result.x[2*i + 2]

    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
