import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Define number of circles
    n = 26
    # Initial grid positions
    num_rows = 5
    num_cols = 5
    spacing = 0.25  # initial spacing for grid
    
    # Generate grid-based initial centers
    x_centers = np.linspace(0.25, 0.75, num_cols)
    y_centers = np.linspace(0.25, 0.75, num_rows)
    centers = []
    for y in y_centers:
        for x in x_centers:
            centers.append([x, y])
    # Add an extra circle near a corner
    centers.append([0.1, 0.1])
    
    # Convert to numpy arrays
    centers = np.array(centers)
    # Initial guess for radii
    radii = np.ones(n) * 0.05
    
    # Objective function to maximize the sum of radii
    def objective(params):
        new_centers = params[:2*n].reshape(n, 2)
        new_radii = params[2*n:]
        return -np.sum(new_radii)  # Minimize negative sum to maximize actual sum
    
    # Constraints: 
    # 1. No overlaps between circles
    # 2. All circles within the square
    # 3. Radii non-negative
    
    def constraint_overlap(params):
        new_centers = params[:2*n].reshape(n, 2)
        new_radii = params[2*n:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < new_radii[i] + new_radii[j] - 1e-8:
                    return dist - (new_radii[i] + new_radii[j] - 1e-8)
        return 0
    
    def constraint_bounds(params):
        new_centers = params[:2*n].reshape(n, 2)
        new_radii = params[2*n:]
        for i in range(n):
            x, y = new_centers[i]
            r = new_radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return (x - r + 1e-8) + (x + r - 1 - 1e-8) + (y - r + 1e-8) + (y + r - 1 - 1e-8)
        return 0
    
    def constraint_radii(params):
        new_radii = params[2*n:]
        for r in new_radii:
            if r < 0:
                return r + 1e-8
        return 0
    
    # Initial guess
    initial_params = np.concatenate((centers.reshape(-1), np.full(n, 0.05)))
    
    # Bounds: all positions must be in [0, 1] and radii must be >= 0
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
        bounds.extend([(0.0, 1.0)])
    
    # Set up constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_bounds},
        {'type': 'ineq', 'fun': constraint_radii}
    ]
    
    # Run the optimization
    result = minimize(
        objective,
        initial_params,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'disp': False, 'maxiter': 200}
    )
    
    # Extract centers and radii
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    
    # Validate packing (optional, but if failed, the error would be raised)
    # validate_packing(centers, radii)
    
    return centers, radii, np.sum(radii)
