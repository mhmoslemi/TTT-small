import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from itertools import combinations
from scipy.spatial.distance import pdist, squareform

def run_packing():
    # Initial approximate centers (moebius-like, hexagonal clusters, corners)
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Define a set of hexagonal-like integer grid centers with small offsets (low density)
    # First, core cluster
    core_centers = np.array([
        [0.3, 0.3], [0.7, 0.3], [1.0, 0.3], [0.7, 0.6], [0.3, 0.6], [0.5, 0.5]
    ])
    
    # Edge clusters: top, bottom, left, right
    edge_centers = np.array([
        [0.2, 0.9], [0.4, 0.9], [0.6, 0.9], [0.8, 0.9],
        [0.9, 0.2], [0.9, 0.4], [0.9, 0.6], [0.9, 0.8],
        [0.2, 0.2], [0.4, 0.2], [0.6, 0.2], [0.8, 0.2],
        [0.2, 0.4], [0.4, 0.4], [0.6, 0.4], [0.8, 0.4],
        [0.1, 0.5], [0.3, 0.5], [0.7, 0.5], [0.9, 0.5]
    ])
    
    # Add corner circles
    corner_centers = np.array([
        [0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]
    ])
    
    # Combine all centers in a suitable order
    centers = np.vstack((core_centers, edge_centers, corner_centers))
    centers = centers[:26]
    
    # Initial guess for radii: 0.04 for larger circles, 0.02 for small ones
    radii = np.zeros(26)
    for i in range(len(centers)):
        if i < 6:  # Core cluster
            radii[i] = 0.08
        elif 6 <= i < 22:  # Edge clusters
            radii[i] = 0.04
        else:  # Corner circles
            radii[i] = 0.02
    
    # Define the objective: sum of radii
    def objective(radii_param, centers):
        return -np.sum(radii_param)  # Minimize the negative to maximize sum
    
    # Define constraints: make sure circles are within [0,1]^2 and don't overlap
    def constraint1(radii_param, centers):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii_param[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return False
        return True
    
    def constraint2(radii_param, centers):
        for i, j in combinations(range(len(centers)), 2):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii_param[i] + radii_param[j] - 1e-8:
                return False
        return True
    
    # Optimization parameters
    bounds = Bounds(0, 0.5)  # All radii must be non-negative
    
    # Run optimization
    result = minimize(
        fun=objective,
        x0=radii,
        args=(centers,),
        method='SLSQP',
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda x: constraint1(x, centers)},
            {'type': 'ineq', 'fun': lambda x: constraint2(x, centers)}
        ]
    )
    
    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Optional: visualize
    plt.figure(figsize=(6, 6))
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        plt.gca().add_patch(plt.Circle((x, y), r, color='blue', alpha=0.5))
    plt.xlim(0, 1)
    plt.ylim(0, 1)
    plt.axis('equal')
    plt.title(f"Circle Packing with Sum Radii = {sum_radii:.4f}")
    plt.show()
    
    return centers, radii, sum_radii
