import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    # Hexagonal packing initial guess for 26 circles
    def hexagonal_grid(n):
        layers = int(np.ceil((n - 1) / 6))
        centers = []
        for layer in range(layers + 1):
            for angle in np.linspace(0, 2 * np.pi, 7, endpoint=False):
                x = 0.5 + (layer + 0.5) * np.cos(angle)
                y = 0.5 + (layer + 0.5) * np.sin(angle)
                centers.append((x, y))
        return np.array(centers)

    # Generate initial guess for 26 circles using hexagonal packing
    initial_centers = hexagonal_grid(26)
    initial_radii = np.full(26, 0.01)

    # Define the objective function to maximize total radius sum
    def objective(radii, centers):
        return -np.sum(radii)

    # Define constraints
    constraints = []

    # Constraint 1: Distance to nearest boundary
    def boundary_constraint(r, x, y):
        return r - np.min([x - r, y - r, 1 - x - r, 1 - y - r])

    # Constraint 2: Distance between circles
    def circle_dist_constraint(r1, r2, x1, y1, x2, y2):
        dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
        return dist - (r1 + r2)

    # Define all constraints for optimization
    def add_constraints(centers, radii):
        for i in range(len(radii)):
            x, y = centers[i]
            constraints.append({
                'type': 'ineq',
                'fun': lambda r, x=x, y=y: r - np.min([x - r, y - r, 1 - x - r, 1 - y - r])
            })
            for j in range(i + 1, len(radii)):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                constraints.append({
                    'type': 'ineq',
                    'fun': lambda r1, r2, x1=x1, y1=y1, x2=x2, y2=y2: np.sqrt((x1 - x2)**2 + (y1 - y2)**2) - (r1 + r2)
                })

    add_constraints(initial_centers, initial_radii)

    # Optimization
    result = minimize(
        objective,
        x0=initial_radii,
        args=(initial_centers,),
        method='SLSQP',
        constraints=constraints,
        bounds=[(0, 1)] * 26,
        options={
            'ftol': 1e-8,
            'gtol': 1e-8,
            'maxiter': 500,
            'disp': False
        }
    )

    # Extract results
    centers = initial_centers
    radii = result.x
    sum_radii = np.sum(radii)

    # Validate packing
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing validation failed: {msg}")

    return centers, radii, sum_radii
