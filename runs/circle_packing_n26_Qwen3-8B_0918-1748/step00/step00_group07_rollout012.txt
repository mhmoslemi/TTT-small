import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define initial grid parameters for hexagonal packing
    N_circles = 26
    num_rows = 5  # Number of rows for hexagonal grid
    num_cols = 6  # Number of columns in the first row

    # Generate initial positions for hexagonal packing
    angle = np.pi / 3  # 60 degrees
    dx = np.cos(angle)
    dy = np.sin(angle)
    G = np.zeros((N_circles, 2))
    G[0, 0] = 0.5
    G[0, 1] = 0.5
    for i in range(1, N_circles):
        row = i // num_cols
        col = i % num_cols
        if row % 2 == 0:
            x = col * (1 - dx) + 0.5 * dx
            y = row * dy
        else:
            x = col * (1 - dx) + dx
            y = row * dy
        G[i, 0] = x
        G[i, 1] = y

    # Generate grid
    grid = np.zeros((N_circles, 2))
    for i in range(N_circles):
        grid[i, 0] = G[i, 0]
        grid[i, 1] = G[i, 1]

    # Initial guess for radii
    base_radius = 0.1
    radii = np.full(N_circles, base_radius)

    # Define function to minimize (negative of total radius sum)
    def obj_func(radii):
        return -np.sum(radii)

    # Constraints: non-overlapping and in unit square
    def constraint_func(radii, centers):
        for i in range(len(radii)):
            x1, y1 = centers[i]
            r1 = radii[i]
            for j in range(i + 1, len(radii)):
                x2, y2 = centers[j]
                r2 = radii[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2 - 1e-12:
                    return False
            if x1 - r1 < -1e-12 or x1 + r1 > 1 + 1e-12 or y1 - r1 < -1e-12 or y1 + r1 > 1 + 1e-12:
                return False
        return True

    # Initial constraints to ensure all circles are in the square
    for i in range(N_circles):
        x, y = grid[i]
        if x - base_radius < -1e-12 or x + base_radius > 1 + 1e-12 or y - base_radius < -1e-12 or y + base_radius > 1 + 1e-12:
            base_radius *= 0.9

    # Finalize centers and radii
    centers = grid
    radii = np.full(N_circles, base_radius)

    # Step 1: Refine radii with optimization
    def constraint(radii):
        for i in range(len(radii)):
            x1, y1 = centers[i]
            r1 = radii[i]
            for j in range(i + 1, len(radii)):
                x2, y2 = centers[j]
                r2 = radii[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                if dist < r1 + r2 - 1e-12:
                    return False
            if x1 - r1 < -1e-12 or x1 + r1 > 1 + 1e-12 or y1 - r1 < -1e-12 or y1 + r1 > 1 + 1e-12:
                return False
        return True

    # Initial guess for radii
    base_radius = 0.15
    initial_guess = np.full(N_circles, base_radius)

    # Optimize
    result = minimize(obj_func, initial_guess, method='SLSQP', constraints={'type': 'ineq', 'fun': constraint, 'args': (centers,)})

    # Assign final centers and radii
    centers = result.x.reshape(-1, 2)
    radii = np.full(26, 0.13)
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            radii[i] = min(radii[i], dist - 1e-12)
            radii[j] = min(radii[j], dist - 1e-12)

    # Final adjustment to ensure all are within bounds
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12:
            radii[i] = x - 1e-12
        elif x + r > 1 + 1e-12:
            radii[i] = 1 - x - 1e-12
        if y - r < -1e-12:
            radii[i] = y - 1e-12
        elif y + r > 1 + 1e-12:
            radii[i] = 1 - y - 1e-12

    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
