import numpy as np
from scipy.optimize import minimize
import itertools

def run_packing():
    # Initial guess for center positions based on a hexagonal grid
    # Each circle will be placed with hexagonal spacing, starting from the center
    # We will have 5 rows, each with 5 circles, and 1 additional circle in the middle
    # That's 5*5 = 25 circles, plus 1 more to make 26 total

    # Initial positions with hexagonal spacing
    def hexagonal_positions(num_rows, row_spacing, col_spacing, center):
        positions = []
        for row in range(num_rows):
            for col in range(num_rows - row):
                x = center[0] + (col - (num_rows - 1) / 2) * row_spacing
                y = center[1] + row * row_spacing * np.sqrt(3) / 2
                positions.append((x, y))
        return np.array(positions)

    # First, generate an initial hexagonal grid with 5 rows
    hex_grid = hexagonal_positions(5, 0.2, 0.2, (0.5, 0.5))
    # This gives 15 positions (5+4+3+2+1 = 15)
    # Add additional circles in the corners and edges

    # Create a list of all centers
    centers = []

    # Lower left corner
    centers.append((0.1, 0.1))
    # Lower right corner
    centers.append((0.9, 0.1))
    # Upper left corner
    centers.append((0.1, 0.9))
    # Upper right corner
    centers.append((0.9, 0.9))
    # Middle bottom
    centers.append((0.5, 0.1))
    # Middle top
    centers.append((0.5, 0.9))

    # Extend to 26 circles by adding additional circles into the hex grid
    # We first generate the full hex grid
    full_hex_grid = hexagonal_positions(5, 0.2, 0.2, (0.5, 0.5))
    centers.extend(full_hex_grid)

    # Trim to 26 circles by taking the first 26
    centers = centers[:26]
    centers = np.array(centers)

    # Initial guess for radii
    # We set all radii to the same initial value based on the hex grid spacing
    # Then we will optimize with constraints
    initial_guess_radii = np.array([0.1] * 26)

    def objective(radii):
        return -np.sum(radii)

    def constraint_func(x):
        # x has shape (2*26,)
        # First 2*26 elements are centers, then the radii
        centers = x[:2*26].reshape(26, 2)
        radii = x[2*26:]
        # Check boundaries
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1:
                return -np.inf
            if y_i - r_i < 0 or y_i + r_i > 1:
                return -np.inf
        # Check overlaps
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return -np.sum(radii)

    # Prepare bounds and constraints
    x_bounds = []
    for i in range(26):
        x_bounds.extend([(0, 1)] * 2)
    x_bounds.extend([(0, 1)] * 26)

    # Use a numerical optimization to find the best configuration
    result = minimize(
        fun=lambda x: -constraint_func(x),
        x0=np.concatenate((centers.flatten(), initial_guess_radii)),
        bounds=x_bounds,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-10, 'disp': False}
    )

    if not result.success:
        raise ValueError("Optimization failed")

    best_centers = result.x[:2*26].reshape(26, 2)
    best_radii = result.x[2*26:]

    sum_radii = np.sum(best_radii)

    return best_centers, best_radii, sum_radii
