import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    np.random.seed(42)

    n = 26

    # Initial grid-based configuration with 5 rows and 5 columns (25 circles)
    grid_size = 5
    grid_spacing = 1.0 / (grid_size - 1)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initialize centers in a grid pattern
    for i in range(grid_size):
        for j in range(grid_size):
            idx = i * grid_size + j
            if idx < n:
                x = i * grid_spacing
                y = j * grid_spacing
                centers[idx] = np.array([x, y])

    # Add the 26th circle in a corner
    centers[n - 1] = np.array([0.0, 0.0])
    radii[n - 1] = 0.05

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize radii

    def square_boundary(radii, centers, idx):
        x, y = centers[idx]
        r = radii[idx]
        return (x - r >= 0) and (x + r <= 1) and (y - r >= 0) and (y + r <= 1)

    # Constraints: non-overlapping between circles
    def non_overlapping(radii, centers, i, j):
        x1, y1 = centers[i]
        x2, y2 = centers[j]
        r1, r2 = radii[i], radii[j]
        return np.sqrt((x1 - x2)**2 + (y1 - y2)**2) >= r1 + r2

    bounds = [(0.0, 0.5) for _ in range(n)]
