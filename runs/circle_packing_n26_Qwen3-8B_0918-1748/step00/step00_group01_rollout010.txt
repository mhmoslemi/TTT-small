import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize.slsqp import inequality

def run_packing():
    # Parameters
    num_circles = 26
    boundary = 1e-6
    max_radius = 1.0 - boundary

    # Base grid: a 5x5 grid in the center with some spacing
    grid_x = np.linspace(0.15, 0.85, 5)
    grid_y = np.linspace(0.15, 0.85, 5)
    base_grid = np.array([(x, y) for x in grid_x for y in grid_y])
    base_radii = np.full(25, 0.15)  # Initial estimated radius

    # Add corner circles
    corner_centers = np.array([
        [0.1, 0.1],
        [0.9, 0.1],
        [0.1, 0.9],
        [0.9, 0.9]
    ])
    corner_radii = np.full(4, 0.05)
    centers = np.vstack([base_grid, corner_centers])
    radii = np.concatenate([base_radii, corner_radii])
    sum_radii = np.sum(radii)

    def objective(x):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints(x):
        # Each circle lies within the unit square
        for i in range(num_circles):
            x_i, y_i = x[2*i], x[2*i + 1]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return False
        # All circles are non-overlapping
        dists = squareform(pdist(x.reshape(-1, 2)))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dists[i, j] < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Flatten centers and radii into a single vector for optimization
    x0 = centers.flatten()
    bounds = [(0, 1) for _ in range(2 * num_circles)]

    # Add constraints for radii bounds
    cons = []
    for i in range(num_circles):
        cons.append(inequality(lambda x, idx=i: x[2*idx + 1] - x[2*idx], None, None, None))
        cons.append(inequality(lambda x, idx=i: x[2*idx] - x[2*idx + 1], None, None, None))

    res = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-10, 'maxiter': 1000})

    # Extract the optimized centers and radii
    centers_opt = res.x.reshape(-1, 2)
    # Reconstruct radii with fixed values (as initial guess is good)
    radii_opt = radii.copy()

    # Final validation
    assert constraints(res.x), "Optimization constraints violated"

    return centers_opt, radii_opt, np.sum(radii_opt)
