import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    def validate_packing(centers, radii):
        n = centers.shape[0]

        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    def compute_sum_radii(radii):
        return np.sum(radii)

    def objective_function(params, centers, radii):
        for i in range(len(radii)):
            radii[i] = params[i]
        return -compute_sum_radii(radii)

    def constraints_function(params, centers, radii):
        for i in range(len(radii)):
            radii[i] = params[i]
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Step 1: Define initial guess (grid and cluster arrangement)
    initial_centers = []
    initial_radii = []

    # 4x4 grid in the center
    for i in range(4):
        for j in range(4):
            x = 0.25 + i * 0.25
            y = 0.25 + j * 0.25
            initial_centers.append([x, y])
            initial_radii.append(0.1)

    # 10 circles arranged in a hexagonal-like pattern around the grid
    hex_centers = []
    for i in range(10):
        if i % 2 == 0:
            offset = i // 2
            x = 0.2 + offset * 0.15
            y = 0.8
            hex_centers.append([x, y])
        else:
            offset = (i - 1) // 2
            x = 0.75 - offset * 0.15
            y = 0.7
            hex_centers.append([x, y])

    for center in hex_centers:
        initial_centers.append(center)
        initial_radii.append(0.1)

    # Refine the initial configuration to improve packing
    centers = np.array(initial_centers)
    radii = np.array(initial_radii)

    # Create a flattened parameter vector for optimization
    n = len(radii)
    initial_params = radii.copy()

    # Optimization objective function
    def obj(params):
        return objective_function(params, centers, radii)

    # Constraints function
    def cons(params):
        return constraints_function(params, centers, radii)

    # Set up the optimization problem
    bounds = [(0.0, 1.0) for _ in range(n)]
    result = minimize(obj, initial_params, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': cons})

    final_radii = result.x
    final_centers = centers.copy()

    # Check validity of the result
    valid, message = validate_packing(final_centers, final_radii)
    if not valid:
        print(f"Optimization failed: {message}")
        return np.array([]), np.array([]), 0.0

    # Return the final configuration
    return final_centers, final_radii, compute_sum_radii(final_radii)
