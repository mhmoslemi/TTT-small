    # Create a central circle
    centers[0] = [0.5, 0.5]
    radii[0] = 0.2

    for i in range(1, 26):

        radii_flat = x[52:]

        # Compute total sum of radii
        total_radii = np.sum(radii_flat)

        # Compute constraint violations
        constraint_violations = 0.0
        for i in range(26):
            r = radii_flat[i]
            x_i, y_i = centers_flat[i]
            # Check boundary constraints
            if x_i - r < -1e-12 or x_i + r > 1 + 1e-12:
                constraint_violations += 1e6
            if y_i - r < -1e-12 or y_i + r > 1 + 1e-12:
                constraint_violations += 1e6

    x0 = np.concatenate((centers.flatten(), radii))

    result = minimize(
