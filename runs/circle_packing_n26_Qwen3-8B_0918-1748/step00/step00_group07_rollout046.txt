import numpy as np
import scipy.optimize as optimize
from scipy.optimize.linesearch import line_search_wolfe_powell
import time

def run_packing():
    # Initial grid-based hexagonal packing
    n = 26
    rows = int(np.sqrt(n))
    cols = (n + rows - 1) // rows
    spacing = 1.0 / (cols - 1)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    for i in range(n):
        col = i % cols
        row = i // cols
        x = col * spacing
        y = row * spacing * np.sqrt(3) / 2
        if row % 2 == 1:
            x += spacing / 2
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = 0.05  # Initial radius
    
    # Optimize radii using gradient descent with constraints
    def objective(radii):
        return -np.sum(radii)  # Maximize sum of radii
    
    def constraint1(radii):
        return np.sum(radii)  # Non-negativity constraint
    
    def constraint2(radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        return True
    
    def constraint3(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    bounds = [(0, np.inf) for _ in range(n)]
    
    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2},
            {'type': 'ineq', 'fun': constraint3}]
    
    result = optimize.minimize(objective, radii, bounds=bounds, constraints=cons, method='SLSQP')
    radii = result.x
    
    # Final validation
    centers_final = centers.copy()
    radii_final = radii.copy()
    return centers_final, radii_final, np.sum(radii_final)
