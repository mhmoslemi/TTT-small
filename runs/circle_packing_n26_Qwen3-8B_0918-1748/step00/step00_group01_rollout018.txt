import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist

def run_packing():
    # Initial configuration with hexagonal packing in 2D
    n = 26
    # Two rows: 13 circles in first row (horizontal), 13 in second row (shifted)
    # Generate initial approximate positions (centered in unit square)
    rows = 2
    cols = 13
    spacing = 1.0 / (cols - 1)  # horizontal spacing
    diag_spacing = np.sqrt(3) / 2 * spacing  # vertical spacing in hexagonal packing
    base_x = np.linspace(0.5 / cols, 1 - 0.5 / cols, cols)
    base_y = [0.5 / rows, 1 - 0.5 / rows]
    
    # Generate grid of positions
    positions = []
    for y in base_y:
        for x in base_x:
            positions.append((x, y))
    
    # Convert to numpy array and add some circles to corners
    positions = np.array(positions)
    # Check if the number of initial positions is less than n
    if len(positions) < n:
        # If not enough, generate some more with a hexagonal pattern
        base_x = np.linspace(0.2, 0.8, 10)
        base_y = np.linspace(0.2, 0.8, 10)
        additional = np.column_stack((base_x, base_y))
        positions = np.vstack((positions, additional))
    
    # Ensure we have exactly n circles
    positions = positions[:n]
    centers = positions.copy()
    
    # Initial radius: based on minimum spacing between circles
    # We'll optimize for this
    radii = np.ones(n) * 0.03  # initial guess
    
    # Optimization variables: centers (2D) and radii
    # Flatten centers to a 1D vector of size 2n
    n_vars = 2 * n
    x0 = np.concatenate((centers.ravel(), radii))
    
    # Objective function: maximize sum of radii
    def objective(x):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    # Constraints: no overlap, within unit square
    constraints = []
    
    # Constraint: circle i is within unit square
    def square_constraint(x, i):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        x_i, y_i = centers[i]
        r_i = radii[i]
        return (x_i - r_i >= 0) * (x_i + r_i <= 1) * (y_i - r_i >= 0) * (y_i + r_i <= 1)
    
    for i in range(n):
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: square_constraint(x, i)})
    
    # Constraint: no overlap between any pair of circles
    def overlap_constraint(x, i, j):
        centers = x[:2*n].reshape(n, 2)
        radii = x[2*n:]
        x_i, y_i = centers[i]
        x_j, y_j = centers[j]
        r_i = radii[i]
        r_j = radii[j]
        dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
        return dist - (r_i + r_j)
    
    for i in range(n):
        for j in range(i+1, n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: overlap_constraint(x, i, j)})
    
    # Optimization bounds: centers in [0,1]^2, radii >= 0
    bounds = Bounds(np.zeros(n_vars), np.inf)
    
    # Perform optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-8)
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    # Extract results
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    
    # Final check
    success, message = validate_packing(centers, radii)
    if not success:
        raise ValueError(f"Validation failed: {message}")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
