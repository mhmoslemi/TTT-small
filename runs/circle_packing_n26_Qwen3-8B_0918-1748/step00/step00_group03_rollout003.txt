import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    def energy_function(params, n_circles):
        centers = params[:n_circles*2].reshape(n_circles, 2)
        radii = params[n_circles*2:].reshape(n_circles)
        
        # Boundary constraints
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return np.inf
        
        # Circle-circle constraints
        dist_matrix = squareform(pdist(centers))
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                dist = dist_matrix[i, j]
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.inf
        
        # Energy function based on repulsive forces
        energy = 0
        for i in range(n_circles):
            for j in range(n_circles):
                if i != j:
                    dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                    if dist < 1e-12:
                        continue
                    energy += 1 / dist
        
        return energy

    def optimize_packing(n_circles):
        initial_centers = np.random.rand(n_circles, 2)
        initial_radii = np.random.rand(n_circles)
        params = np.concatenate((initial_centers.reshape(-1), initial_radii))
        
        bounds = []
        for i in range(n_circles):
            bounds.extend([(0, 1) for _ in range(2)])  # x, y of center
            bounds.extend([(0, 1) for _ in range(1)])  # radius
        
        result = minimize(energy_function, params, args=(n_circles,), bounds=bounds, method='L-BFGS-B', options={'gtol': 1e-5, 'ftol': 1e-5})
        
        if not result.success:
            raise ValueError("Optimization failed.")
        
        centers = result.x[:n_circles*2].reshape(n_circles, 2)
        radii = result.x[n_circles*2:]
        
        return centers, radii

    n_circles = 26
    centers, radii = optimize_packing(n_circles)
    
    # Apply additional optimization to increase sum of radii
    def objective_function(radii):

    def constraint1(radii):
        return sum(radii)
    
    # Re-optimize radii given centers
    # This is a placeholder due to the complexity of reoptimizing radii with fixed centers
    # In practice, this would need to be integrated with center optimization
    
    # Final validation
    for i in range(n_circles):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} is outside the unit square")
    
    for i in range(n_circles):
        for j in range(i+1, n_circles):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
