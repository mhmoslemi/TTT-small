import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    # Define initial grid with 4 rows and 7 columns, adjusted for hexagonal packing
    rows = 4
    cols = 7
    spacing = 1.0 / (cols - 1)
    row_spacing = math.sqrt(3) / 2

    centers = np.zeros((rows * cols, 2))
    radii = np.zeros(rows * cols)

    # Fill grid with hexagonal packing
    for i in range(rows):
        for j in range(cols):
            x = j * spacing
            y = i * row_spacing
            centers[i * cols + j] = [x, y]

    # Initial radii, assuming regular hexagonal packing
    initial_r = 1.0 / (2 * (cols + rows))
    radii[:] = initial_r

    # Optimization variables: midpoints of jumps in each row for boundary adjustments
    # Each row has a control variable to adjust vertical shift
    num_rows = rows
    bounds = []
    for i in range(num_rows):
        bounds.append((0, 1))
    x0 = np.array([0.5] * num_rows)

    def objective(x):
        # Adjust vertical positions of rows
        for i in range(num_rows):
            row_idx = i
            centers[row_idx * cols : (row_idx + 1) * cols, 1] = y_offsets[i]

        # Calculate total sum of radii
        total = np.sum(radii)
        return -total  # Minimize negative to maximize total

    def constraint(x):
        # Ensure circles do not overlap
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Set up problem
    constraints = [{'type': 'ineq', 'fun': constraint}]

    # Run optimization
    result = minimize(objective, x0, bounds=bounds, constraints=constraints, method='SLSQP')

    # Apply changes to centers
    y_offsets = result.x
    for i in range(num_rows):
        row_start = i * cols
        row_end = (i + 1) * cols
        centers[row_start:row_end, 1] = y_offsets[i]

    # Adjust radii based on boundary constraints
    for i in range(len(centers)):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
            # Reduce radius to fit within square
            max_r = min(1 - y, 1 - (1 - y), x, 1 - x)
            radii[i] = max_r

    # Final validation
    # Ensure circles do not overlap
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            if dist < radii[i] + radii[j] - 1e-8:
                raise ValueError("Overlap detected in final configuration")

    # Calculate total sum of radii
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
