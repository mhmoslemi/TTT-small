import numpy as np
import random
import math
from scipy.optimize import minimize

def run_packing():
    # Initial configuration: 2 rows with 12 circles in the first row and 13 in the second, with hexagonal packing
    # Adjust for boundary constraints

    def compute_hexagonal_grid(n_circles, side_length):
        row_count = int(math.ceil(n_circles / 2))
        col_count = (n_circles + 1) // 2
        positions = []
        for i in range(row_count):
            for j in range(col_count):
                x = j * (side_length * math.sqrt(3))
                y = i * (side_length * 1.5)
                if i % 2 == 1:
                    x += side_length * math.sqrt(3) / 2
                positions.append([x, y])
        return np.array(positions)
    
    def adjust_for_boundary(positions, boundary_distance):
        adjusted = []
        for x, y in positions:
            new_x = x + boundary_distance
            new_y = y + boundary_distance
            adjusted.append([new_x, new_y])
        return np.array(adjusted)
    
    def is_valid_config(positions, radii):
        for i in range(len(positions)):
            x, y = positions[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
        for i in range(len(positions)):
            for j in range(i + 1, len(positions)):
                dist = math.hypot(positions[i][0] - positions[j][0], positions[i][1] - positions[j][1])
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    def optimize_radii(positions, n_circles):
        def objective(radii):
            return -np.sum(radii)
        
        def constraints(radii):
            constraints = []
            for i in range(n_circles):
                constraints.append((radii[i], 0, None))
            for i in range(n_circles):
                for j in range(i + 1, n_circles):
                    dist = math.hypot(positions[i][0] - positions[j][0], positions[i][1] - positions[j][1])
                    constraints.append((radii[i] + radii[j], 0, dist + 1e-12))
            return constraints
        
        bounds = [(0, None) for _ in range(n_circles)]
        result = minimize(objective, x0=np.ones(n_circles), bounds=bounds, constraints=constraints)
        return result.x
    
    # Create initial hexagonal grid in a 1x1 square
    n_circles = 26
    initial_positions = compute_hexagonal_grid(n_circles, 0.1)
    # Adjust positions to be within unit square
    boundary_distance = 0.1
    adjusted_positions = adjust_for_boundary(initial_positions, boundary_distance)
    # Optimize radii
    radii = optimize_radii(adjusted_positions, n_circles)
    # Refine positions based on radii
    refined_positions = []
    for i in range(len(radii)):
        x, y = adjusted_positions[i]
        refined_positions.append([x, y])
    refined_positions = np.array(refined_positions)
    # Final check and return
    return (refined_positions, radii, np.sum(radii))
