import numpy as np
from scipy.optimize import minimize
from scipy.spatial import cKDTree

def run_packing():
    # Base configuration: 2 rows of 13 circles in a structured layout
    # First row: 6 circles with centers along x from 0.2 to 0.8
    # Second row: 7 circles with centers along x from 0.15 to 0.9
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # First row: 6 circles
    for i in range(6):
        x = 0.2 + i * 0.12
        y = 0.5
        centers[i] = [x, y]
        radii[i] = 0.2
    
    # Second row: 7 circles with slight offset
    for i in range(6, 13):
        x = 0.15 + (i - 6) * 0.12
        y = 0.5
        centers[i] = [x, y]
        radii[i] = 0.18
    
    # Third row: 7 circles
    for i in range(13, 20):
        x = 0.1 + (i - 13) * 0.15
        y = 0.85
        centers[i] = [x, y]
        radii[i] = 0.15
    
    # Fourth row: 6 circles
    for i in range(20, 26):
        x = 0.1 + (i - 20) * 0.15
        y = 0.25
        centers[i] = [x, y]
        radii[i] = 0.12
    
    # Define optimization bounds
    bounds = []
    for _ in range(26):
        bounds.append((0.0, 1.0))  # x-coordinate
        bounds.append((0.0, 1.0))  # y-coordinate
    
    # Objective function to maximize sum of radii
    def objective(params):
        params = params.reshape(26, 2)
        new_centers = params
        new_radii = radii
        total = np.sum(new_radii)
        return -total
    
    # Constraints: circles must not overlap and stay within square
    def constraint(params):
        params = params.reshape(26, 2)
        new_centers = params
        for i in range(26):
            x1, y1 = new_centers[i]
            r1 = radii[i]
            if x1 - r1 < -1e-12 or x1 + r1 > 1 + 1e-12 or y1 - r1 < -1e-12 or y1 + r1 > 1 + 1e-12:
                return -1
        for i in range(26):
            for j in range(i + 1, 26):
                dx = new_centers[i, 0] - new_centers[j, 0]
                dy = new_centers[i, 1] - new_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0
    
    constraints = [{'type': 'ineq', 'fun': constraint}]
    
    # Initial guess: centers and radii as defined
    initial_centers = centers.reshape(52)
    initial_radii = radii
    
    # Optimization
    result = minimize(
        objective,
        initial_centers,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-10, 'eps': 1e-8}
    )
    
    # Extract final centers and radii
    final_centers = result.x.reshape(26, 2)
    final_radii = radii
    
    # Compute sum of radii
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
