import numpy as np
import cvxpy as cp
import time

def run_packing():
    # Initial configuration: Hexagonal grid with 5x5 = 25 circles, then add one more
    num_circles = 26
    grid_size = 5
    spacing = 0.285  # Initial spacing for hexagonal grid
    radius_initial = spacing / 2  # Initial radius for each circle

    # Generate hexagonal grid
    centers = []
    radii = []

    # Generate 5 rows, with alternating columns
    for row in range(grid_size):
        for col in range(grid_size):
            x = col * spacing
            y = row * spacing if row % 2 == 0 else row * spacing + spacing / 2
            centers.append([x, y])
            radii.append(radius_initial)

    # Add one extra circle in the center to make 26
    if num_circles > len(centers):
        extra_circle = [grid_size / 2 * spacing, grid_size / 2 * spacing]
        centers.append(extra_circle)
        radii.append(radius_initial)

    # Optimization variables
    x = cp.Variable((num_circles, 2))
    r = cp.Variable(num_circles, non_negative=True)

    # Objective: Maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Constraints
    constraints = []

    # Boundary constraints
    for i in range(num_circles):
        constraints.append(x[i, 0] - r[i] >= 0)
        constraints.append(x[i, 0] + r[i] <= 1)
        constraints.append(x[i, 1] - r[i] >= 0)
        constraints.append(x[i, 1] + r[i] <= 1)

    # Non-overlap constraints
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-6)

    # Solve with CVXPY
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, eps=1e-5, max_iter=1000, time_limit=60)

    if not problem.status == 'optimal':
        raise RuntimeError(f"Optimization failed: {problem.status}")
