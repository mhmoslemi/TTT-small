import numpy as np
import cvxpy as cp

def run_packing():
    n = 26
    # Initial seed positions using a hexagonal grid inside the unit square
    # We place 5 rows of circles with 5, 5, 5, 5, and 6 circles respectively
    # This ensures a roughly hexagonal packing and uses the unit square efficiently
    centers = np.empty((n, 2))
    r = 0.1  # Initial spacing between circles
    col = 0
    row = 0
    for i in range(n):
        if i % 5 == 0:
            row += 1
            col = 0
        x = col * (1 + r)
        y = row * (np.sqrt(3)/2 * r)
        # Scale to fit within unit square
        scale = 1.0 / (np.sqrt(3)/2 * r + 0.1)  # Adjust scale to fit in unit square
        x = x * scale
        y = y * scale
        centers[i] = [x, y]
        col += 1
    
    # Optimization variables
    c = cp.Variable((n, 2))
    r_var = cp.Variable(n)

    # Objective: maximize the sum of radii
    objective = cp.Maximize(cp.sum(r_var))

    # Constraints
    constraints = []

    # 1. Each circle fits within the unit square
    for i in range(n):
        x, y = c[i]
        r = r_var[i]
        constraints.append(x - r >= -1e-8)
        constraints.append(x + r <= 1 + 1e-8)
        constraints.append(y - r >= -1e-8)
        constraints.append(y + r <= 1 + 1e-8)

    # 2. Non-overlap between circles
    for i in range(n):
        for j in range(i + 1, n):
            dx = c[i, 0] - c[j, 0]
            dy = c[i, 1] - c[j, 1]
            dist = cp.sqrt(dx**2 + dy**2)
            constraints.append(dist >= r_var[i] + r_var[j] - 1e-8)

    # 3. Initial positions as a seed
    for i in range(n):
        constraints.append(c[i] == centers[i])

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.MOSEK, verbose=False)

    # Extracting results
    centers = c.value
    radii = r_var.value
    sum_radii = np.sum(radii)

    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
