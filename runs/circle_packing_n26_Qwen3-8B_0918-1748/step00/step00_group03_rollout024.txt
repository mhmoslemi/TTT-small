import numpy as np
import scipy.optimize as optimize
from scipy.spatial.distance import cdist

def run_packing():
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize radii

    def constraints(radii, centers):
        return np.sum(radii)  # Just for demonstration, actual constraint handling is in validate_packing

    # Initial guess for radii - all equal
    initial_radii = np.full(26, 0.1)

    # Initial centers based on hexagonal packing
    def generate_hexagonal_centers():
        n = 26
        rows = int(np.ceil(np.sqrt(n))) + 1
        cols = int(np.ceil(np.sqrt(n))) + 1
        centers = []
        for i in range(rows):
            for j in range(cols):
                if i * cols + j < n:
                    x = j * 1.0 / (cols - 1)
                    y = i * 1.0 / (rows - 1)
                    centers.append([x, y])
        return np.array(centers)

    centers = generate_hexagonal_centers()
    # Normalize centers to fit in the unit square
    centers /= np.max(centers)

    # Optimize using differential evolution
    result = optimize.differential_evolution(
        objective,
        bounds=[(0, 1.0) for _ in range(26)],
        args=(centers,),
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-10,
        polish=True,
        mutation=(0.5, 1),
        recombination=0.9,
        seed=42,
        disp=False
    )

    radii = result.x
    # Final validation to ensure positions are adjusted and valid
    # This is a simplification as actual validation is done in a separate step
    # For the purpose of this function, we assume the optimization produces a valid configuration
    # and adjust the centers to be within the unit square and non-overlapping

    # Final check to ensure centers are in [0,1]^2
    centers = np.clip(centers, 0, 1)

    # Ensure no overlaps by adjusting radii
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                # Adjust radii to prevent overlap
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2

    # Final adjustment to ensure radii are non-negative
    radii = np.clip(radii, 0, 1)

    # Final check to ensure all circles are within the unit square
    for i in range(26):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            # Adjust radius to fit within the square
            radii[i] = np.min([x, 1 - x, y, 1 - y])

    return centers, radii, np.sum(radii)
