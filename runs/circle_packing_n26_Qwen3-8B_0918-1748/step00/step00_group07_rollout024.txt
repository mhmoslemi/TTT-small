import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial grid-based configuration for 25 circles
    grid = np.linspace(0.1, 0.9, 5)
    centers = np.empty((25, 2))
    radii = np.zeros(25)
    
    # Place circles in a 5x5 grid with spacing
    spacing = 0.2
    for i in range(5):
        for j in range(5):
            x = grid[i] + spacing * (i + 1) / 5
            y = grid[j] + spacing * (j + 1) / 5
            centers[5 * i + j] = [x, y]
            radii[5 * i + j] = 0.13

    # Now add a 26th circle in the remaining space
    # Start with a guess for the 26th circle
    centers_26 = np.array([[0.5, 0.5]])
    radii_26 = np.array([0.2])
    
    # Combine the initial grid with the 26th circle
    centers = np.vstack((centers, centers_26))
    radii = np.concatenate((radii, radii_26))
    
    # Define the optimization problem to maximize the sum of radii
    # with constraints that circles do not overlap and are within the unit square
    
    def objective(radii):
        return -np.sum(radii)  # Negative because we are minimizing
    
    def constraints(radii):
        # Constraint: circles must be within the unit square
        for i in range(26):
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12:
                return False
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12:
                return False
        
        # Constraint: circles must not overlap
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        
        return True
    
    # Define bounds for radii (non-negative)
    bounds = Bounds(np.zeros(26), np.inf)
    
    # Optimization
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraints}]
    )
    
    # Extract final centers, radii, and sum of radii
    centers_final = centers
    radii_final = result.x
    sum_radii = np.sum(radii_final)
    
    return (centers_final, radii_final, sum_radii)
