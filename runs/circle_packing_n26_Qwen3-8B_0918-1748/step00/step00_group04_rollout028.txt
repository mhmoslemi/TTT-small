import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import random

def run_packing():
    n = 26
    # Initial grid-based placement with some variations to explore different configurations
    centers = np.zeros((n, 2))
    radii = np.ones(n) * 0.05
    
    # Create a hexagonal grid based on a grid of points
    grid_centers = []
    grid_radii = []
    grid_density = 0.1
    for i in range(n):
        row = i // 6
        col = i % 6
        x = col * (1 - 2*grid_density) / 5
        y = row * (1 - 2*grid_density) / 5
        grid_centers.append([x, y])
        grid_radii.append(grid_density)
    
    # Fixing to a subset of the hexagonal grid
    centers = np.array(grid_centers)
    radii = np.array(grid_radii)
    
    # Optimization to maximize sum of radii
    # Use a modified version of the scipy.optimize.minimize function
    def obj_func(radii):
        return -np.sum(radii)
    
    constraints = []
    
    for i in range(n):
        # Boundary constraints
        def boundary_constr_i(radii):
            if not np.isclose(radii[i], radii[i]):
                return np.inf
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return np.inf
            return 0
        constraints.append({'type': 'ineq', 'fun': lambda r, i=i: boundary_constr_i(r)})
    
    for i in range(n):
        for j in range(i + 1, n):
            def dist_constr_i_j(radii):
                ri = radii[i]
                rj = radii[j]
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                dist = np.sqrt((x1 - x2)**2 + (y1 - y2)**2)
                return dist - ri - rj
            constraints.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: dist_constr_i_j(r)})
    
    # Set initial radii (not optimized yet)
    initial_guess = radii.copy()
    
    result = minimize(obj_func, initial_guess, method='SLSQP', constraints=constraints)
    optimized_radii = result.x
    
    # Ensure no negative radii
    optimized_radii = np.clip(optimized_radii, 0, 1)
    
    # Calculate sum of radii
    sum_radii = np.sum(optimized_radii)
    
    # Return the centers and radii
    return (centers, optimized_radii, sum_radii)
