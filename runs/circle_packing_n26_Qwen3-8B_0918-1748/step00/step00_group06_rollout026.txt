import numpy as np
import scipy.optimize as optimize
from scipy.optimize import NonlinearConstraint, Bounds

def run_packing():
    # Initial grid-based layout with a hexagonal packing strategy
    rows = 7
    cols = [3, 4, 3, 4, 3, 4, 3]
    centers = []
    radii = []
    for i, row in enumerate(cols):
        if row == 0:
            continue
        row_centers = []
        for j in range(row):
            if i % 2 == 0:
                x = j * (1.0 / (row - 1)) + 0.5 / row
            else:
                x = (j + 0.5) / row
            y = (i + 0.5) / rows
            row_centers.append([x, y])
        centers.extend(row_centers)
    
    # Initial radius approximation
    total_area = 26 * np.pi
    initial_radius = np.sqrt(total_area / 26)
    radii = [initial_radius] * len(centers)
    
    # Optimization variables: centers and radii
    n = len(centers)
    x0 = np.array([centers[i][0] for i in range(n)] + [radii[i] for i in range(n)])
    
    # Bounds: x and y between 0 and 1, radii non-negative
    bounds = []
    for i in range(n):
        bounds.append((0.0, 1.0))  # x
        bounds.append((0.0, 1.0))  # y
        bounds.append((0.0, 1.0))  # radius
    
    # Objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x[n*2:])

    # Nonlinear constraints: distance between centers >= r_i + r_j
    def constraint_func(x):
        centers = x[:n*2].reshape(n, 2)
        radii = x[n*2:]
        constraints = []
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            constraints.append([x_i - r_i >= -1e-8, x_i + r_i <= 1 + 1e-8,
                               y_i - r_i >= -1e-8, y_i + r_i <= 1 + 1e-8])
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append([dist >= radii[i] + radii[j] - 1e-8])
        return constraints
    
    # Create constraints
    constraints = []
    for i in range(n):
        x_i, y_i = centers[i]
        r_i = radii[i]
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*2] - r_i + 1e-8})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i*2] - r_i + 1e-8})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i*2 + 1] - r_i + 1e-8})
        constraints.append({'type': 'ineq', 'fun': lambda x, i=i: 1 - x[i*2 + 1] - r_i + 1e-8})
    for i in range(n):
        for j in range(i+1, n):
            def dist_ineq(x, i=i, j=j):
                centers_i = x[i*2 : i*2 + 2]
                centers_j = x[j*2 : j*2 + 2]
                r_i = x[n*2 + i]
                r_j = x[n*2 + j]
                dist = np.sqrt((centers_i[0] - centers_j[0])**2 + (centers_i[1] - centers_j[1])**2)
                return dist - r_i - r_j + 1e-8
            constraints.append({'type': 'ineq', 'fun': dist_ineq})
    
    # Optimization options
    options = {'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8}
    
    # Run the optimization
    result = optimize.minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options=options
    )
    
    # Extract results
    final_centers = result.x[:n*2].reshape(n, 2)
    final_radii = result.x[n*2:]
    
    # Return the centers, radii, and total sum
    return final_centers, final_radii, np.sum(final_radii)
