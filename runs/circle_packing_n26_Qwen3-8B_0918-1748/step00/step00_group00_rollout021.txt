import numpy as np
import cvxpy as cp

def run_packing():
    n = 26
    # Initial grid-based configuration
    rows = 5
    cols = 6
    x = np.linspace(0.1, 0.9, cols)
    y = np.linspace(0.1, 0.9, rows)
    centers = np.array([[xi, yi] for yi in y for xi in x])
    
    # Optimization variables
    radii = cp.Variable(n)
    # Objective: maximize sum of radii
    obj = cp.Maximize(cp.sum(radii))
    
    # Constraints
    constraints = []
    # Each circle must be within the square
    for i in range(n):
        x_i, y_i = centers[i]
        constraints.append(x_i - radii[i] >= 0)
        constraints.append(x_i + radii[i] <= 1)
        constraints.append(y_i - radii[i] >= 0)
        constraints.append(y_i + radii[i] <= 1)
    
    # Non-overlapping constraints
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j])
    
    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.SCS, eps=1e-5)
    
    if not prob.status in ['OPTIMAL', 'OPTIMAL_INFEASIBLE']:
        raise RuntimeError(f"Optimization failed with status: {prob.status}")
    
    centers = centers.astype(np.float64)
    radii = radii.value.astype(np.float64)
    sum_radii = np.sum(radii)
    
    # Validate packing
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Packing validation failed: {message}")
    
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
