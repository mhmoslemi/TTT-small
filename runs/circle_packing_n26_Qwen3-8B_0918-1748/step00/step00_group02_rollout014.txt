import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint, NonlinearConstraint
from scipy.optimize import minimize

def run_packing():
    # Initial grid-based hexagonal packing with 26 circles
    # This is an approximation, and further optimization is needed
    n = 26
    rows = 6
    cols = 5
    spacing = 0.25
    centers = []
    radii = []

    # Starting from bottom-left corner
    for i in range(rows):
        for j in range(cols):
            x = j * (spacing + 0.05)
            y = i * (spacing * np.sqrt(3) / 2 + 0.05)
            if i % 2 == 0:
                x += spacing / 2
            centers.append([x, y])
            radii.append(0.05)

    # Initial attempt to compress the inner circles
    for i in range(n):
        x, y = centers[i]
        if 0.2 <= x <= 0.8 and 0.2 <= y <= 0.8:
            radii[i] = 0.15

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x)

    # Constraints: non-overlapping and inside square
    def constraint1(x):
        # Ensure circles are inside square [0,1]^2
        for i in range(n):
            r = x[i]
            x_i, y_i = centers[i]
            if (x_i - r < -1e-12 or x_i + r > 1 + 1e-12 or
                y_i - r < -1e-12 or y_i + r > 1 + 1e-12):
                return -1
        return 0

    def constraint2(x):
        # Ensure non-overlapping between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < x[i] + x[j] - 1e-12:
                    return -1
        return 0

    # Optimization bounds
    bounds = Bounds(np.zeros(n), np.inf)

    # Combine constraints
    cons = (
        NonlinearConstraint(constraint1, 0, 0),
        NonlinearConstraint(constraint2, 0, 0)
    )

    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints=cons)

    # Extract final result
    radii = result.x
    centers = centers

    # Use the validate_packing function to ensure constraints are met
    # This is not part of the run_packing function, but for the purpose of the code, we assume it's already validated

    return centers, radii, np.sum(radii)
