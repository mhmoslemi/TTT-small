import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds, minimize

def run_packing():
    # Define initial guess for circle centers (grid-based)
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Initial grid layout with some offset for hexagonal pattern
    rows = 5
    cols = 6
    
    # Calculate the vertical and horizontal spacing
    vertical_spacing = 1.0 / (rows + 1)
    horizontal_spacing = 1.0 / (cols + 1)
    
    # Place circles in a hexagonal grid
    for i in range(rows):
        for j in range(cols):
            x = j * horizontal_spacing
            y = i * vertical_spacing
            centers[i * cols + j] = [x, y]
    
    # Refine positions using optimization
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraints(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True
    
    # Set up bounds for radii
    bounds = [(0, 1) for _ in range(num_circles)]
    
    # Optimization
    result = minimize(objective, radii, method='SLSQP', bounds=bounds, constraints={'type': 'ineq', 'fun': constraints})
    
    radii = result.x
    # Adjust positions to ensure they are within the unit square
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        if x - r < 0:
            centers[i, 0] = r
        elif x + r > 1:
            centers[i, 0] = 1 - r
        if y - r < 0:
            centers[i, 1] = r
        elif y + r > 1:
            centers[i, 1] = 1 - r
    
    # Recalculate distances with adjusted positions
    dist_matrix = squareform(pdist(centers))
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                # Adjust radii to ensure no overlap
                radii[i] = dist_matrix[i, j] - radii[j]
                radii[j] = dist_matrix[i, j] - radii[i]
    
    # Final validation and sum calculation
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
