import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run_packing():
    # Initial guess for circle centers using a hexagonal packing
    # Adjust spacing and bounds to fit within the unit square
    num_circles = 26
    # Initial grid parameters
    rows = 6
    cols = 5
    spacing = 0.2  # Spacing between centers to avoid overlap
    initial_centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + (spacing / 2)
            y = i * spacing + (spacing / 2)
            if i % 2 == 1:
                x += spacing / 2
            initial_centers.append([x, y])
    initial_centers = np.array(initial_centers)
    # Trim circles that go over the square
    valid_indices = []
    for idx, center in enumerate(initial_centers):
        x, y = center
        if 0 <= x <= 1 and 0 <= y <= 1:
            valid_indices.append(idx)
    initial_centers = initial_centers[valid_indices]
    initial_centers = initial_centers[:26]
    initial_radii = np.full(len(initial_centers), spacing / 2)
    
    # Define the optimization problem
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    def constraint_func(x):
        centers = x[:26*2].reshape((26, 2))
        radii = x[26*2:]
        # Check boundaries
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -np.inf
        # Check overlaps
        for i in range(26):
            for j in range(i+1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        # All constraints satisfied
        return np.sum(radii)

    # Prepare initial guess
    x0 = np.concatenate([initial_centers.flatten(), initial_radii])
    bounds = Bounds([0, 0], [1, 1])  # for centers
    bounds = Bounds([0, 0], [1, 1])  # for centers
    # Define bounds for radii
    radii_bounds = np.full(26, 1)
    radii_bounds = np.tile(radii_bounds, 2)  # type: ignore
    bounds = Bounds([0, 0], [1, 1])  # for centers
    radii_bounds = np.full(26, 1)
    radii_bounds = np.tile(radii_bounds, 2)  # type: ignore
    bounds = Bounds([0, 0], [1, 1])  # for centers
    # Combine bounds for centers and radii
    full_bounds = []
    for i in range(26):
        full_bounds.append((0, 1))
        full_bounds.append((0, 1))
    full_bounds = np.array(full_bounds)
    # Run optimization
    res = minimize(objective, x0, bounds=full_bounds, method='SLSQP', 
                  constraints=[{'type': 'ineq', 'fun': constraint_func}])
    # Extract results
    centers = res.x[:26*2].reshape((26, 2))
    radii = res.x[26*2:]
    total_sum_radii = np.sum(radii)
    return (centers, radii, total_sum_radii)
