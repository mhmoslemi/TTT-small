import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Initial hexagonal grid with 26 circles
    n = 26
    # Create a hexagonal grid layout
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row = 0
    col = 0
    for i in range(n):
        if i % 2 == 0:
            centers[i, 0] = (i / 2) / (np.sqrt(3) * 0.5)
            centers[i, 1] = (row) / 2
            row += 1
        else:
            centers[i, 0] = (i // 2) / (np.sqrt(3) * 0.5) + 0.5 / np.sqrt(3)
            centers[i, 1] = (row) / 2
            row += 1
    # Scale centers to [0,1]^2
    max_x = np.max(centers[:, 0])
    max_y = np.max(centers[:, 1])
    centers /= max_x if max_x > 0 else 1
    centers[:, 1] /= max_y if max_y > 0 else 1
    # Initial radii based on distance to nearest neighbor
    dists = np.zeros(n)
    for i in range(n):
        dists[i] = np.min(np.sqrt(np.sum((centers[i] - centers[j])**2) for j in range(n) if j != i))
    radii = np.clip(dists / 2, 0, 1)

    def objective(radii_vec):
        return -np.sum(radii_vec)

    def constraint_positions(x):
        for i in range(n):
            x_i, y_i = x[i*2], x[i*2+1]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1e6
        return 0

    def constraint_overlap(x):
        for i in range(n):
            for j in range(i+1, n):
                dx = x[i*2] - x[j*2]
                dy = x[i*2+1] - x[j*2+1]
                dist = np.sqrt(dx**2 + dy**2)
                r_i = radii[i]
                r_j = radii[j]
                if dist < r_i + r_j - 1e-12:
                    return -1e6
        return 0

    bounds = Bounds([0]*2*n, [1]*2*n)

    x0 = centers.flatten()

    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_positions},
            {'type': 'ineq', 'fun': constraint_overlap}
        ]
    )

    centers_opt = result.x.reshape((n, 2))
    radii_opt = np.zeros(n)
    for i in range(n):
        x, y = centers_opt[i]
        r = 0.5 * np.min(np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2) for j in range(n) if j != i))
        radii_opt[i] = r

    # Final validation
    if not validate_packing(centers_opt, radii_opt)[0]:
        raise Exception("Validation failed")

    return centers_opt, radii_opt, np.sum(radii_opt)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
