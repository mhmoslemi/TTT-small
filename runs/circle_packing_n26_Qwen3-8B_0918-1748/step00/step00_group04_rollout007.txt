import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, Constraint, NonlinearConstraint

def run_packing():
    # Initial grid setup
    grid_size = 5
    num_circles = 26
    centers = []
    radii = []
    initial_radii = []

    # Create initial grid for 5x5 = 25 circles
    for i in range(grid_size):
        for j in range(grid_size):
            x = i + 0.5
            y = j + 0.5
            if (i + j) % 2 == 0:
                # Even positions for hexagonal packing
                offset_x = 0.25 * np.sin(np.pi / 3)
                offset_y = 0.25 * np.cos(np.pi / 3)
                x += offset_x
                y += offset_y
            centers.append([x, y])

    # Add the 26th circle in the center
    centers.append([0.5, 0.5])

    # Calculate initial radii based on grid spacing
    spacing = 1.0 / grid_size
    for i in range(len(centers)):
        x, y = centers[i]
        # Find minimum distance to boundaries
        r_boundary = min(x, 1 - x, y, 1 - y)
        # Find minimum distance to neighbors
        r_neighbors = float('inf')
        for j in range(len(centers)):
            if i != j:
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r_neighbors:
                    r_neighbors = dist
        r = min(r_boundary, r_neighbors / 2)
        initial_radii.append(r)
    radii = initial_radii

    # Define the optimization problem
    n = len(centers)
    center_vars = np.array(centers).flatten()
    radius_vars = np.array(radii)

    # Objective function: maximize sum of radii
    def objective(x):
        x_centers = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        return -np.sum(r)  # Negative for minimization

    # Constraints: circle within the square
    def boundary_constraint(x):
        x_centers = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        for i in range(n):
            x_i, y_i = x_centers[i]
            r_i = r[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.array([0.0])  # Return a scalar if any violation
        return np.array([1.0])

    # Constraints: circles do not overlap
    def non_overlap_constraint(x):
        x_centers = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = x_centers[i, 0] - x_centers[j, 0]
                dy = x_centers[i, 1] - x_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-12:
                    return np.array([0.0])  # Return a scalar if any violation
        return np.array([1.0])

    # Set bounds for center coordinates (0,1) and radii (0,1)
    center_bounds = Bounds([0.0]*2*n, [1.0]*2*n)
    radius_bounds = Bounds([0.0]*n, [1.0]*n)

    # Set up constraints
    cons = []
    cons.append(Constraint(lambda x: boundary_constraint(x), 'ineq', 1.0))
    cons.append(Constraint(lambda x: non_overlap_constraint(x), 'ineq', 1.0))

    # Initial guess
    x0 = np.concatenate([center_vars, radius_vars])

    # Solve the optimization problem
    result = opt.minimize(objective, x0, bounds=[center_bounds, radius_bounds], constraints=cons, method='SLSQP')

    # Extract results
    x_opt = result.x
    centers_opt = x_opt[:2*n].reshape(n, 2)
    radii_opt = x_opt[2*n:]

    # Validate the packing
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    return centers_opt, radii_opt, np.sum(radii_opt)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
