```python
import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds, NonlinearConstraint

def run_packing():
    # Define the initial grid of centers
    rows = 3
    cols = 9
    grid_centers = []
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1)
            y = i / (rows - 1)
            grid_centers.append([x, y])
    grid_centers = np.array(grid_centers)

    # Remove one circle to have 26 circles (e.g., remove the last one)
    grid_centers = grid_centers[:-1]

    # Initial guess for radii: based on initial spacing
    initial_radii = np.full(grid_centers.shape[0], 0.1)

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative sum to maximize the sum

    # Constraints: for each pair of circles, distance between centers >= r_i + r_j
    def constraint_dist(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Nonlinear constraint: no overlaps
    cons = ({'type': 'ineq', 'fun': lambda radii: constraint_dist(grid_centers, radii)})

    # Bounds: radii >= 0
    bounds = Bounds(np.zeros_like(initial_radii), np.inf * np.ones_like(initial_radii))

    # Optimization
    result = opt.minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8,
        options={'maxiter': 1000, 'ftol': 1e-10, 'eps': 1e-8}
    )

    # Check for feasible solution
    if not result.success:
        raise ValueError("Optimization failed: " + result.message)

    centers = grid_centers
    radii = result.x

    # Ensure all circles are within the unit square
    for i in range(len(radii)):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            raise ValueError(f"Circle {i} is outside the unit square")

    # Final sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
```
