import numpy as np
import math
import random
from scipy.optimize import minimize, Bounds

def run_packing():
    # Define initial hexagonal grid with 25 circles
    n = 25
    rows = int(math.sqrt(n))
    cols = (n + rows - 1) // rows
    radius_initial = 0.05
    centers = []
    for i in range(rows):
        for j in range(cols):
            x = j * (1.0 + radius_initial) / (cols - 1)
            y = i * (1.0 + radius_initial) / (rows - 1)
            centers.append((x, y))
    centers = np.array(centers)
    radii = np.full(n, radius_initial)
    
    # Add the 26th circle in a tentative position
    # We attempt to place it near the corner (0.1, 0.1) with small radius
    centers = np.vstack((centers, [0.1, 0.1]))
    radii = np.append(radii, 0.03)
    
    # Define the optimization problem with constraints
    def objective(x):
        # x is flattened vector of centers and radii
        x_flat = x.reshape(-1)
        centers_opt = x_flat[:2 * n].reshape(n, 2)
        radii_opt = x_flat[2 * n:]
        total_sum = np.sum(radii_opt)
        return -total_sum  # Minimize negative to maximize
    
    def constraint_boundary(x):
        x_flat = x.reshape(-1)
        centers_opt = x_flat[:2 * n].reshape(n, 2)
        radii_opt = x_flat[2 * n:]
        for i in range(n):
            x, y = centers_opt[i]
            r = radii_opt[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        return 0
    
    def constraint_overlap(x):
        x_flat = x.reshape(-1)
        centers_opt = x_flat[:2 * n].reshape(n, 2)
        radii_opt = x_flat[2 * n:]
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers_opt[i] - centers_opt[j])**2))
                if dist < radii_opt[i] + radii_opt[j] - 1e-12:
                    return -1
        return 0
    
    # Initial guess
    x_initial = np.concatenate((centers.flatten(), radii))
    
    # Bounds: centers in [0,1], radii >= 0
    bounds = Bounds(0, 1, len(x_initial) - n)
    bounds = Bounds([0.0 if i < 2 * n else 0.0 for i in range(2 * n + n)], [1.0 if i < 2 * n else np.inf for i in range(2 * n + n)])
    
    # Use scipy.optimize.minimize with constraints
    result = minimize(objective, x_initial, method='SLSQP', bounds=bounds, constraints=[
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ])
    
    final_centers = result.x[:2 * n].reshape(n, 2)
    final_radii = result.x[2 * n:]
    total_sum = np.sum(final_radii)
    
    # Validate the solution
    if not validate_packing(final_centers, final_radii)[0]:
        raise ValueError("Validation failed for the computed packing")
    
    return final_centers, final_radii, total_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
