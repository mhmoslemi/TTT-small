import numpy as np
import math
from scipy.optimize import minimize

def run_packing():
    # Define initial guess for centers and radii
    # Start with a hexagonal packing with some circles on the edges

    # Initial hexagonal packing with center at (0.5, 0.5)
    # We will use 24 circles in a hexagonal grid and 2 on the corners

    def generate_hexagonal_centers():
        centers = []
        radii = []
        # Center circle
        centers.append([0.5, 0.5])
        radii.append(0.01)
        # First layer (6 circles)
        for i in range(6):
            angle = 2 * math.pi * i / 6
            x = 0.5 + 0.3 * math.cos(angle)
            y = 0.5 + 0.3 * math.sin(angle)
            centers.append([x, y])
            radii.append(0.01)
        # Second layer (12 circles)
        for i in range(12):
            angle = 2 * math.pi * i / 12
            x = 0.5 + 0.6 * math.cos(angle)
            y = 0.5 + 0.6 * math.sin(angle)
            centers.append([x, y])
            radii.append(0.01)
        # Third layer (6 circles)
        for i in range(6):
            angle = 2 * math.pi * i / 6
            x = 0.5 + 0.9 * math.cos(angle)
            y = 0.5 + 0.9 * math.sin(angle)
            centers.append([x, y])
            radii.append(0.01)
        # Two corner circles
        centers.append([0.1, 0.1])
        radii.append(0.01)
        centers.append([0.9, 0.9])
        radii.append(0.01)
        return centers, radii

    # Generate initial positions and radii
    centers, radii = generate_hexagonal_centers()
    num_circles = len(centers)

    # Objective function: maximize the sum of radii
    def objective(x, r, centers, num_circles):
        # x contains all variables: first 2*num_circles are positions, then radii
        # Extract positions and radii
        pos = x[:2 * num_circles].reshape((num_circles, 2))
        new_radii = x[2 * num_circles:]
        # Objective: maximize sum of radii (minimize negative sum)
        return -np.sum(new_radii)

    # Constraint functions:
    # 1. Positions within [0,1]^2
    # 2. Radii nonnegative
    # 3. No overlap between circles
    # 4. No circle outside square

    def constraint_1(x):
        # Positions within [0,1]^2
        pos = x[:2 * num_circles].reshape((num_circles, 2))
        return np.all(pos >= 0) and np.all(pos <= 1)

    def constraint_2(x):
        # Radii must be nonnegative
        r = x[2 * num_circles:]
        return np.all(r >= 0)

    def constraint_3(x):
        # No overlap between circles
        pos = x[:2 * num_circles].reshape((num_circles, 2))
        r = x[2 * num_circles:]
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = pos[i, 0] - pos[j, 0]
                dy = pos[i, 1] - pos[j, 1]
                dist = math.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return False
        return True

    def constraint_4(x):
        # No circle outside square
        pos = x[:2 * num_circles].reshape((num_circles, 2))
        r = x[2 * num_circles:]
        for i in range(num_circles):
            if pos[i, 0] - r[i] < -1e-8 or pos[i, 0] + r[i] > 1 + 1e-8:
                return False
            if pos[i, 1] - r[i] < -1e-8 or pos[i, 1] + r[i] > 1 + 1e-8:
                return False
        return True

    # Initial guess: positions and radii as initial guess
    x0 = np.concatenate([np.array(centers).flatten(), np.array(radii)])
    bounds = []
    for i in range(num_circles):
        bounds.extend([(0, 1) for _ in range(2)])
        bounds.extend([(0, 1)])
    constraints = [
        {'type': 'ineq', 'fun': constraint_1},
        {'type': 'ineq', 'fun': constraint_2},
        {'type': 'ineq', 'fun': constraint_3},
        {'type': 'ineq', 'fun': constraint_4}
    ]

    # Run optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, options={'ftol': 1e-8, 'maxiter': 1000})

    if not result.success:
        raise RuntimeError(f"Optimization failed: {result.message}")

    x = result.x
    pos = x[:2 * num_circles].reshape((num_circles, 2))
    radii = x[2 * num_circles:]
    sum_radii = np.sum(radii)

    return pos, radii, sum_radii
