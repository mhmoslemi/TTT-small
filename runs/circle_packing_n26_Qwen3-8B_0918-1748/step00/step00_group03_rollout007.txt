import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Initial approximation: hexagonal packing with some adjustment for square boundaries
    num_circles = 26
    # Start with a hexagonal grid with 5 rows and 6 columns (1 row with 5 circles, then alternating)
    # Adjust to fit in [0,1]x[0,1]
    # Initial radii approximated as 0.1 to start with

    # Initial hexagonal grid coordinates
    rows = 5
    cols = 5

    # Calculate the side length of the hexagonal grid
    # We'll fit it into [0,1]x[0,1]
    # Hexagonal grid height = (rows - 1) * sqrt(3) / 2 * side_length
    # Hexagonal grid width = (cols) * side_length
    # We'll fit it to the square

    # We'll fit the grid such that the full height is 1
    side_length = 1 / (rows - 1) * np.sqrt(3) / 2
    # Now adjust to fit into [0,1]x[0,1]
    # Also, allow for some adjustment to fit the centering

    centers = []
    for i in range(rows):
        for j in range(cols):
            if i % 2 == 0:
                x = j * side_length + 0.5 * side_length
            else:
                x = j * side_length + 0.25 * side_length
            y = i * (side_length * np.sqrt(3)) / 2
            centers.append([x, y])
    # Adjust initial centers to fit inside the square
    centers = np.array(centers)
    # For simplicity, adjust by scaling and centering
    scale_factor = 1 / (np.max(centers, axis=0) - np.min(centers, axis=0))
    centers = (centers - np.min(centers, axis=0)) * scale_factor
    centers = np.clip(centers, 0, 1)

    # Introduce some adjustment for corner circles
    # Place 4 circles at the corners (with small radii to prevent overlap)
    # Let’s use some of the circles to be the corner circles
    corners = np.array([
        [0.1, 0.1],
        [0.9, 0.1],
        [0.9, 0.9],
        [0.1, 0.9]
    ])
    # Split centers into core and corners
    # Select first 4 circles as corners
    core_centers = np.array([centers[i] for i in range(4, num_circles)])
    corner_radii = np.full(4, 0.05)
    core_radii = np.full(num_circles - 4, 0.1)

    # Define the initial guess for positions and radii
    initial_centers = np.vstack([corners, core_centers])
    initial_radii = np.concatenate([corner_radii, core_radii])

    # Define the optimization problem
    def objective(x):
        return -np.sum(x[1:])  # Minimize negative sum to maximize sum

    def constraints(x):
        # x[0:2] = center of first circle
        # x[2] = radius of first circle
        # x[3:5] = center of second circle
        # x[5] = radius of second circle
        # etc.
        # All circles must be non-overlapping and within the square
        # Reshape x into centers and radii
        n = num_circles
        centers = x[::n+1].reshape(n, 2)
        radii = x[1::n+1]
        # Check boundaries
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return np.inf
        # Check overlaps
        for i, j in combinations(range(n), 2):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                return np.inf
        return 0

    # Flatten the initial guess
    x0 = []
    for i in range(num_circles):
        x0.append(initial_centers[i][0])
        x0.append(initial_centers[i][1])
        x0.append(initial_radii[i])
    x0 = np.array(x0)

    # Set up the optimization problem
    result = minimize(objective, x0, constraints=lambda x: constraints(x), method='SLSQPN', tol=1e-4)

    # Extract the final centers and radii
    centers = result.x[::num_circles+1].reshape(num_circles, 2)
    radii = result.x[1::num_circles+1]

    # Return the result
    return centers, radii, np.sum(radii)
