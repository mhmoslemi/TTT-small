def generate_hex_grid(n, side_length, spacing, square_size=1.0):
    """
    Generate a hexagonal grid of circles with given spacing and side length.
    """
    grid = []
    for i in range(n):
        for j in range(n):
            x = i * spacing
            y = j * spacing * np.sqrt(3) / 2
            grid.append([x, y])
    grid = np.array(grid)
    # Clip to square
    grid = np.clip(grid, 0, square_size)
    return grid
