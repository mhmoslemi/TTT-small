import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers and radii
    def initial_packing():
        # Hexagonal packing for 4 rows with 6 circles each (total 24)
        r_initial = 0.15
        centers = []
        for row in range(4):
            for col in range(6):
                x = col * (2 * r_initial) + r_initial
                y = row * (np.sqrt(3) * r_initial) + r_initial
                centers.append([x, y])
        # Bring the centers within [0,1]
        scaling = 1.0 / (np.max(centers, axis=0) + 1e-6)
        centers = np.array(centers) * scaling
        # Add two corner circles
        r_corner = 0.15
        centers.append([r_corner, 1 - r_corner])
        centers.append([1 - r_corner, r_corner])
        radii = np.full(len(centers), r_initial)
        radii[-2:] = r_corner
        return np.array(centers), radii

    centers_initial, radii_initial = initial_packing()
    
    def objective(params):
        flat_params = params.reshape(-1)
        centers = flat_params[:2*26].reshape(26, 2)
        radii = flat_params[2*26:]
        if np.any(radii < 0):
            return -np.inf
        # Check boundary conditions
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -np.inf
        # Check circle-circle overlaps
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        # Objective is to maximize sum of radii
        return -np.sum(radii)
    
    bounds = []
    for _ in range(26):
        bounds.extend([(0, 1) for _ in range(2)])  # for centers
    for _ in range(26):
        bounds.append((0, 1))  # for radii
    
    # Initial guess
    flat_params = np.concatenate([centers_initial.flatten(), radii_initial])
    
    # Optimization
    result = minimize(
        objective,
        flat_params,
        bounds=bounds,
        method='L-BFGS-B',
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8},
        jac=None
    )
    
    if not result.success:
        raise Exception("Optimization failed")
    
    centers = result.x[:2*26].reshape(26, 2)
    radii = result.x[2*26:]
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
