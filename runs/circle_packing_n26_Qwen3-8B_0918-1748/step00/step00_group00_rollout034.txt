import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize.lbfgs import minimize_l_bfgs_b

def run_packing():
    # Initial grid layout with 5x5 = 25 circles
    num_circles = 26
    grid_size = 5
    grid_spacing = 1.0 / grid_size
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    for i in range(grid_size):
        for j in range(grid_size):
            x = i * grid_spacing
            y = j * grid_spacing
            centers[i * grid_size + j] = [x, y]
    
    # Add one more circle at the bottom right corner
    centers[-1] = [grid_spacing, grid_spacing]
    
    # Initial radii: based on grid spacing
    radii = np.full(num_circles, grid_spacing / 2)
    
    # Constraint function for optimization
    def constraint_func(radii):
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True
    
    # Bounds for radii
    bounds = [(0, 1) for _ in range(num_circles)]
    
    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Optimization
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_func}],
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    
    centers = np.array(centers)
    radii = result.x
    
    # Final validation
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"
    
    valid, _ = validate_packing(centers, radii)
    if not valid:
        raise ValueError("Optimization failed to satisfy constraints")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
