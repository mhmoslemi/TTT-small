import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run_packing():
    # Define initial grid layout
    n = 26
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial grid layout
    for i in range(n):
        row = i // cols
        col = i % cols
        x = col / (cols - 1)
        y = row / (rows - 1)
        centers[i] = [x, y]
        radii[i] = 0.05  # initial small radius
    
    # Optimization bounds
    bounds = Bounds(np.zeros(n), np.inf)
    
    # Constraints
    def constraint_func(x):
        # x is a 2n-dimensional array: first n for centers, then n for radii
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        
        # Check if all centers are within [0,1]^2 and radii are non-negative
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i < 0 - r_i or x_i > 1 + r_i or y_i < 0 - r_i or y_i > 1 + r_i:
                return -1.0
            if r_i < 0:
                return -1.0
        
        # Check if all circles are non-overlapping
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -1.0
        
        return 0.0
    
    # Objective function: maximize sum of radii
    def objective_func(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        return -np.sum(radii_flat)  # minimize negative sum to maximize sum
    
    # Initial guess
    x0 = np.zeros(n * 2 + n)
    x0[:n*2] = centers.reshape(n * 2)
    x0[n*2:] = radii
    
    # Optimization
    result = minimize(
        fun=objective_func,
        x0=x0,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint_func}],
        method='SLSQP',
        options={'maxiter': 200, 'ftol': 1e-8}
    )
    
    # Extract solution
    centers_flat = result.x[:n*2].reshape(n, 2)
    radii_flat = result.x[n*2:]
    
    # Post-processing: prune very small radii
    pruning_threshold = 1e-5
    radii = radii_flat[radii_flat > pruning_threshold]
    centers = centers_flat[np.where(radii_flat > pruning_threshold)[0]]
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
