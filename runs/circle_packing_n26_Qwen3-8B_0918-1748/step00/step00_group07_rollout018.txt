import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial import distance_matrix
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run_packing():
    # Initial attempt: Create a hexagonal-like packing with uneven radii
    # Based on strategic placement of circles near corners and edges
    
    # Define corner and edge positions for boundary circles
    corner_circles = [
        (0.1, 0.1),
        (0.1, 0.9),
        (0.9, 0.9),
        (0.9, 0.1)
    ]
    
    # Define edge circles (along the midpoints of edges)
    edge_circles = [
        (0.5, 0.1),
        (0.5, 0.9),
        (0.1, 0.5),
        (0.9, 0.5)
    ]
    
    # Define core circles in a hexagonal packing in the center
    # Generating 22 circles: 12 rows, with decreasing number of circles per row
    # Starting from 6 circles in the middle, and decreasing alternating rows
    core_circles = []
    for i in range(7):
        if i == 3:
            # Skip the middle row for hexagonal packing
            continue
        for j in range(6 - abs(i - 3)):
            x = 0.5 + (j - 2.5) * 0.2
            y = 0.5 + (i - 3) * 0.2
            core_circles.append((x, y))
    
    # Total circles: corners (4) + edges (4) + core (22) = 26
    centers = np.array(core_circles + corner_circles + edge_circles)
    radii = np.zeros(len(centers))
    # Initialize radii by estimating maximum radius for circles in core positions
    # Using hexagonal packing in a unit square (not exact, just an estimate)
    max_core_radius = 0.16
    max_edge_radius = 0.14
    max_corner_radius = 0.15
    for i, (x, y) in enumerate(centers):
        if i < len(core_circles):
            radii[i] = max_core_radius
        elif len(core_circles) <= i < len(core_circles) + len(edge_circles):
            radii[i] = max_edge_radius
        else:
            radii[i] = max_corner_radius
    
    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints: ensure circles stay within square and do not overlap
    # First, define bounds for each radius: non-negative, and radius <= 1 - max distance to nearest boundary
    def constraint_boundary(radii):
        for i in range(len(centers)):
            x, y = centers[i]
            # Maximum possible radius for circle i is min(1 - x, x, 1 - y, y)
            max_radius = min(1 - x, x, 1 - y, y)
            if radii[i] > max_radius:
                return False
        return True
    
    # Define constraints for non-overlapping: distance between centers >= radii[i] + radii[j]
    def constraint_overlap(radii):
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-6:
                    return False
        return True
    
    # Optimization parameters
    bounds = [(0, 1 - 1e-6) for _ in range(len(centers))]
    constraints = [
        {'type': 'ineq', 'fun': lambda r: constraint_boundary(r)},
        {'type': 'ineq', 'fun': lambda r: constraint_overlap(r)}
    ]
    
    # Perform optimization using a gradient descent approach
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    centers_result = centers
    radii_result = result.x
    sum_radii = np.sum(radii_result)
    
    # Final validation
    valid, message = validate_packing(centers_result, radii_result)
    if not valid:
        raise ValueError(f"Optimization failed with message: {message}")
    
    return centers_result, radii_result, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
