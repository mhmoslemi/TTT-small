import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint
import time

def run_packing():
    # Initial guess: grid of circles with spacing such that there's room for 26 circles
    # We use a 5x5 grid with 2 circles in the center rows/columns for a total of 26 circles
    # The grid is [x, y], and each point has a radius which we will optimize
    
    # Initial center positions
    centers = np.zeros((26, 2))
    idx = 0
    for row in range(5):
        for col in range(5):
            x = col / 4.0
            y = row / 4.0
            centers[idx] = [x, y]
            idx += 1
    # Add 6 more circles in the middle for a total of 26
    # These are placed in the center 3x3 grid shifted slightly
    for row in range(3):
        for col in range(3):
            x = col / 4.0 + 0.25
            y = row / 4.0 + 0.25
            centers[25 - 9 + idx] = [x, y]
            idx += 1
    
    # Initial radius guesses
    radii = np.ones(26) * 0.05
    
    # Define objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraint_center_bounds(radii):
        for i in range(26):
            if radii[i] < 0:
                return False, f"Negative radius for circle {i}"
        return True, "ok"

    def constraint_boundary(radii):
        for i in range(26):
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12:
                return False, f"Circle {i} outside left or right boundary"
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12:
                return False, f"Circle {i} outside bottom or top boundary"
        return True, "ok"

    def constraint_overlap(radii):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.hypot(dx, dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    # Optimize
    bounds = Bounds(0, 1.0)
    nl_constraints = []
    nl_constraints.append(NonlinearConstraint(constraint_overlap, None, 0, jac=None))
    nl_constraints.append(NonlinearConstraint(constraint_boundary, None, 0, jac=None))
    res = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=nl_constraints,
        options={'ftol': 1e-10, 'maxiter': 200}
    )
    
    # Extract results
    centers_opt = centers
    radii_opt = res.x
    
    # Verify the packing (needed for external validation, though not used in this context)
    # This can be commented out as the function is only called once and the result is not validated here
    # success, message = validate_packing(centers_opt, radii_opt)
    # if not success:
    #     raise ValueError(message)
    
    total_sum = np.sum(radii_opt)
    return centers_opt, radii_opt, total_sum
