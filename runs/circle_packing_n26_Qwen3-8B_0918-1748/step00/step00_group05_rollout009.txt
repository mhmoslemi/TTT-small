import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def compute_max_radius(centers, index, radius, other_radii, other_centers):
    r = radius
    x, y = centers[index]
    max_r = 1.0 - max(x, 1 - x, y, 1 - y)
    for i in range(len(other_centers)):
        dx = x - other_centers[i][0]
        dy = y - other_centers[i][1]
        dist = np.sqrt(dx**2 + dy**2)
        if dist < r + other_radii[i] - 1e-8:
            max_r = min(max_r, dist - other_radii[i] - 1e-8)
    return max_r

def run_packing():
    # Initial grid-based configuration with 5 rows and 5 columns
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Create a grid-like base configuration (e.g., 5x5 with one extra circle)
    base_grid = np.indices((5, 5)).reshape(25, 2)
    centers[:25] = base_grid / 4.0
    centers[25] = [0.5, 0.5]
    
    # Initial estimate of radii based on minimal overlap
    for i in range(num_circles):
        max_r = 1.0 - np.max([centers[i][0], 1 - centers[i][0], centers[i][1], 1 - centers[i][1]])
        for j in range(i + 1, num_circles):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[j] + radii[i] + 1e-8:
                max_r = min(max_r, dist - radii[j] - 1e-8)
        radii[i] = max_r
    
    bounds = Bounds([0, 0], [1, 1])  # Bounds for x and y
    constraints = []

    def objective(radii):
        return -np.sum(radii)
    
    def constraint_overlap(i, j):
        def cons(x):
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx**2 + dy**2)
            return dist - (radii[i] + radii[j]) + 1e-8
        return cons

    def constraint_boundary(i):
        def cons(x):
            x = centers[i][0]
            return 1.0 - max(x, 1 - x)
        return cons
    
    for i in range(num_circles):
        constraints.append({'type': 'ineq', 'fun': constraint_boundary(i)})
        for j in range(i + 1, num_circles):
            constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j)})
    
    result = optimize.minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP'
    )
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    final_radii = result.x
    final_centers = centers.copy()
    final_sum = np.sum(final_radii)
    
    return final_centers, final_radii, final_sum
