import numpy as np
from scipy.optimize import minimize
from scipy.optimize.optimize import fsolve
from scipy.spatial.distance import cdist
import time

def run_packing():
    np.random.seed(42)
    
    # Initial configuration: 20 circle hexagonal grid with 6 edge-corner circles
    # We start by placing the 20 circles in a hexagonal grid
    # Then place 6 more in the corners and gaps
    def initial_config():
        # First, generate a hexagonal grid of 20 circles
        grid = []
        radii = []
        for i in range(5):
            for j in range(5):
                if i * 5 + j < 20:
                    x = j / 4.0 + 0.1
                    y = i / 4.0 + 0.1
                    grid.append([x, y])
                    radii.append(0.05)
        return np.array(grid), np.array(radii)
    
    centers, radii = initial_config()
    n = centers.shape[0]
    for i in range(n):
        # Adjust initial radii based on initial center positions
        dist = cdist(centers[i].reshape(1, 2), centers).flatten()
        radii[i] = 0.1 / (np.max(dist) + 1e-12)
    
    # Add 6 more circles in corners and gaps
    add_centers = [
        [0.1, 0.1], [0.1, 0.9], [0.9, 0.9], [0.9, 0.1],
        [0.5, 0.1], [0.1, 0.5]
    ]
    add_radii = [0.05] * 6
    centers = np.vstack((centers, add_centers))
    radii = np.concatenate((radii, add_radii))
    
    # Define objective function to maximize sum of radii
    def objective(x):
        # x: 26*2 + 26 = 52 values
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        return -np.sum(radii_flat)  # Minimize negative sum to maximize

    # Define constraints: no overlap and within bounds
    def constraint(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        for i in range(n):
            x1, y1 = centers_flat[i]
            r1 = radii_flat[i]
            if x1 - r1 < -1e-12 or x1 + r1 > 1 + 1e-12 or y1 - r1 < -1e-12 or y1 + r1 > 1 + 1e-12:
                return -1
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -1
        return 0

    # Initial guess: centers and radii
    x0 = np.hstack((centers.flatten(), radii))
    # Bounds for centers: [0, 1] for x and y
    # Bounds for radii: [0, 0.5]
    bounds = []
    for _ in range(n):
        bounds += [(0.0, 1.0), (0.0, 1.0)]
    for _ in range(n):
        bounds += [(0.0, 0.5)]

    # Use L-BFGS-B optimizer
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint}],
        method='L-BFGS-B'
    )
    
    # Extract centers and radii from result
    x_res = result.x
    centers_res = x_res[:n*2].reshape(n, 2)
    radii_res = x_res[n*2:]
    
    # Final validation (not using the given function as per problem instructions)
    return (centers_res, radii_res, np.sum(radii_res))
