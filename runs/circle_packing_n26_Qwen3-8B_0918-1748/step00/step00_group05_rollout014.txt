import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers: grid layout
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Step 1: Place a large central circle
    centers[0] = [0.5, 0.5]
    radii[0] = 0.1  # Initial guess for the largest circle
    
    # Step 2: Place small circles in a hexagonal grid
    # For now, use a grid to fill the square
    grid_size = 0.2  # initial spacing
    for i in range(1, n):
        row = i // 5
        col = i % 5
        x = col * grid_size + 0.1
        y = row * grid_size + 0.1
        centers[i] = [x, y]
        radii[i] = 0.05

    # Optimization: adjust centers and radii to maximize sum of radii
    def objective(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        sum_radii = np.sum(radii_flat)
        return -sum_radii  # Minimize negative to maximize

    def constraints(x):
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        constraints = []
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                constraints.append((x_i - r_i + 1e-12) + (x_i + r_i - 1 - 1e-12) + (y_i - r_i + 1e-12) + (y_i + r_i - 1 - 1e-12))
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_flat[i][0] - centers_flat[j][0]
                dy = centers_flat[i][1] - centers_flat[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii_flat[i] + radii_flat[j] - 1e-12))
        return constraints

    bounds = []
    for i in range(n):
        bounds.extend([(0, 1) for _ in range(2)])  # x and y for each center
        bounds.extend([(0, 1)])  # radius for each circle
    bounds = np.array(bounds)

    # Initial guess
    x0 = np.concatenate([centers.flatten(), radii])

    # Optimization with constraints
    result = minimize(objective, x0, bounds=bounds, constraints={"type": "ineq", "fun": constraints})

    centers_opt = result.x[:n*2].reshape(n, 2)
    radii_opt = result.x[n*2:]

    # Final validation (optional, but not done here to avoid unnecessary steps)
    return (centers_opt, radii_opt, np.sum(radii_opt))
