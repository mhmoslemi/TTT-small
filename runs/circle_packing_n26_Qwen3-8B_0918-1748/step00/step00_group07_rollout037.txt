import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    # Define the number of circles
    num_circles = 26

    # Define a hexagonal packing with varying layers
    # First, define a base hexagonal grid with spacing
    spacing = 0.25  # initial spacing value to be adjusted
    radius_scale = 0.1  # initial scale for radii

    # Function to compute the centers of a hexagonal grid based on offsets
    def hex_grid(offsets, spacing):
        centers = []
        for x, y in offsets:
            # Scale to unit square
            x_scaled = x * spacing
            y_scaled = y * spacing
            centers.append((x_scaled, y_scaled))
        return np.array(centers)

    # Define desired offsets for 26 circles in hexagonal grid pattern
    # First layer (bottom): 5 circles
    offset_1 = [(i, 0) for i in range(5)]
    # Second layer (middle): 10 circles around the bottom layer
    offset_2 = []
    for j in range(1, 6):
        offset_2.extend([(j + 0.5, -0.5 + j * spacing), (j + 0.5, -0.5 - j * spacing)])
    # Third layer (middle): 10 circles above the second layer
    offset_3 = []
    for j in range(2, 6):
        offset_3.extend([(j + 0.5, 0.5 + j * spacing), (j + 0.5, 0.5 - j * spacing)])
    # Fourth layer (top): 1 circle
    offset_4 = [(5, 1.0)]

    # Combining all the offsets
    all_offsets = offset_1 + offset_2 + offset_3 + offset_4

    # Check if total circles == 26
    assert len(all_offsets) == num_circles, f"Expected {num_circles} circles, got {len(all_offsets)}"

    # Initial guess for spacing and radius_scale
    def objective(params):
        spacing, radius_scale = params
        centers = hex_grid(all_offsets, spacing)
        # Ensure circles are within unit square
        x_min, x_max = np.min(centers[:, 0]), np.max(centers[:, 0])
        y_min, y_max = np.min(centers[:, 1]), np.max(centers[:, 1])
        # Scale centers to fit within the unit square
        scale_x = (1 - 1e-6) / (x_max - x_min)
        scale_y = (1 - 1e-6) / (y_max - y_min)
        scaled_centers = np.vstack((scale_x * (centers[:, 0] - x_min), scale_y * (centers[:, 1] - y_min))).T
        radii = np.full(num_circles, radius_scale)
        # Check for overlaps
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = scaled_centers[i, 0] - scaled_centers[j, 0]
                dy = scaled_centers[i, 1] - scaled_centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-6:
                    return float('inf')  # Penalize overlap
        # Compute total radii
        return -np.sum(radii)

    # Initial spacing and radius scale
    initial_params = (0.5, 0.15)

    # Optimize
    result = minimize(objective, initial_params, method='L-BFGS-B', bounds=[(0.1, 1.0), (0.05, 0.3)])

    # Extract optimized parameters
    spacing_opt, radius_scale_opt = result.x

    # Generate centers and radii
    centers = hex_grid(all_offsets, spacing_opt)
    x_min, x_max = np.min(centers[:, 0]), np.max(centers[:, 0])
    y_min, y_max = np.min(centers[:, 1]), np.max(centers[:, 1])
    scale_x = (1 - 1e-6) / (x_max - x_min)
    scale_y = (1 - 1e-6) / (y_max - y_min)
    scaled_centers = np.vstack((scale_x * (centers[:, 0] - x_min), scale_y * (centers[:, 1] - y_min))).T
    radii = np.full(num_circles, radius_scale_opt)

    # Final check for boundary and overlaps
    for i in range(num_circles):
        x, y = scaled_centers[i]
        r = radii[i]
        if (x - r < -1e-6 or x + r > 1 + 1e-6 or y - r < -1e-6 or y + r > 1 + 1e-6):
            raise ValueError(f"Circle {i} at ({x}, {y}) r={r} outside unit square")
        for j in range(i + 1, num_circles):
            dx = scaled_centers[i, 0] - scaled_centers[j, 0]
            dy = scaled_centers[i, 1] - scaled_centers[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii[i] + radii[j] - 1e-6:
                raise ValueError(f"Circles {i} and {j} overlap")

    sum_radii = np.sum(radii)
    return (scaled_centers, radii, sum_radii)
