import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():
    # Initialize centers in a hexagonal packing with some spacing
    # Creation of a grid of points in a hexagonal pattern
    # Adjusting the number of rows and columns to fit 26 circles
    rows = 5
    cols = 5
    centers = []
    radii = []
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1)
            y = i / (rows - 1)
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(len(centers), 0.05)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_func(radii, centers):
        constraints = []
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
            if np.any(centers[i] - radii[i] < -1e-12) or np.any(centers[i] + radii[i] > 1 + 1e-12):
                return False
        return True
    
    # Optimization parameters
    max_iterations = 100
    tolerance = 1e-6
    bounds = Bounds(0, 0.5)
    
    for _ in range(3):
        result = minimize(
            fun=objective,
            x0=radii,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'ineq', 'fun': lambda r: constraint_func(r, centers)}],
            options={'maxiter': max_iterations, 'ftol': tolerance, 'gtol': tolerance}
        )
        
        if result.success:
            radii = result.x
            # Refine by checks and adjustments
            for i in range(len(radii)):
                x, y = centers[i]
                if (x - radii[i] < -1e-12) or (x + radii[i] > 1 + 1e-12) or (y - radii[i] < -1e-12) or (y + radii[i] > 1 + 1e-12):
                    radii[i] = min(0.5, 1 - max(x, 1 - x, y, 1 - y))
            
            # Final validation
            for i in range(len(radii)):
                for j in range(i + 1, len(radii)):
                    dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                    if dist < radii[i] + radii[j] - 1e-12:
                        radii[i] = max(0, radii[i] - (dist - (radii[i] + radii[j])) * 0.5)
                        radii[j] = max(0, radii[j] - (dist - (radii[i] + radii[j])) * 0.5)
            
            break
    
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
