import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Define initial grid of positions for 25 circles in a 5x5 grid
    # with some spacing to allow for optimization
    num_circles = 26
    grid_size = 5
    grid_positions = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i / (grid_size - 1)
            y = j / (grid_size - 1)
            grid_positions.append([x, y])
    # Add one more circle at a corner, e.g., (0.05, 0.05)
    grid_positions.append([0.05, 0.05])
    centers = np.array(grid_positions)
    
    # Initial guess for radii: uniform based on square packing
    # Approximate radius from area: total area filled = n * πr² ≈ 1
    # r ≈ sqrt(1/(n * π))
    initial_radius = np.sqrt(1 / (num_circles * np.pi))
    radii = initial_radius * np.ones(num_circles)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        # Ensure all circles are within the unit square
        for i in range(num_circles):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return False
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:
                return False
        # Ensure all pairs of circles are non-overlapping
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True
    
    # Set up bounds
    bounds = Bounds(0, 1)
    
    # Run optimization
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints={'type': 'callback', 'fun': constraints},
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    # Adjust centers to avoid overlap during optimization
    # Reoptimize with adjusted positions
    # For simplicity, use the initial positions, but with optimized radii
    # In a real scenario, we would refine the positions as well
    
    # Return centers, radii, and sum of radii
    return (centers, result.x, np.sum(result.x))
