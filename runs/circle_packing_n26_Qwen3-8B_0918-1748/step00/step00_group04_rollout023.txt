import random

    # Fourth row: 8 circles shifted right by half the spacing

    # Seventh row: 9 circles along the bottom
    for i in range(9):
        x = i / 8

        centers.append([x, y])

    # Eighth row: 8 circles shifted right by half the spacing
    for i in range(8):
        x = i / 7 + 0.5 / 7
        y = 0.05 + 0.7
        centers.append([x, y])
        radii.append(0.05)

    # Ninth row: 9 circles along the bottom
    for i in range(9):
        x = i / 8
        y = 0.05 + 0.8
        centers.append([x, y])
        radii.append(0.05)

    # Adjust radii to avoid overlap and scale
    def distance(c1, c2):
        return np.sqrt((c1[0] - c2[0])**2 + (c1[1] - c2[1])**2)

    # Start with a uniform scaling of the radii based on the initial spacing between centers
    initial_spacing = 0.1
    initial_radius = initial_spacing / (2 * np.sqrt(3))  # for hexagonal packing

    # Estimate the sum of radii based on the initial configuration
    sum_radii = 26 * initial_radius
    print(f"Initial estimate of sum of radii: {sum_radii}")

    # Create an optimization problem to maximize the sum of radii
    # Objective function: maximize sum(radii)
    # Constraints:
    # 1. Each circle must be inside the unit square
    # 2. Each pair of circles must have center-to-center distance >= sum of radii

    n = 26
    centers = np.array(centers)
    radii = np.array(radii)

    # Objective function
    def objective(x):
        return -np.sum(x)  # maximize sum of radii

    # Bounds for radii: non-negative
    bounds = [(0, 1) for _ in range(n)]

    # Constraint function for center positions
    def constraint_center(x, centers, radii):
        for i in range(n):
            if not (0 <= centers[i, 0] - radii[i] and centers[i, 0] + radii[i] <= 1):
                return -1
            if not (0 <= centers[i, 1] - radii[i] and centers[i, 1] + radii[i] <= 1):
                return -1
        return 0

    # Constraint function for circle overlap
    def constraint_overlap(x, centers):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                if dist < x[i] + x[j] - 1e-12:
                    return -1
        return 0

    # Optimize
    result = optimize.minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': lambda x: constraint_center(x, centers, x)},
            {'type': 'ineq', 'fun': lambda x: constraint_overlap(x, centers)}
        ],
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )

    if not result.success:
        raise ValueError(f"Optimization failed: {result.message}")

    # Extract the result
    radii = result.x
    centers = np.array(centers)
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
