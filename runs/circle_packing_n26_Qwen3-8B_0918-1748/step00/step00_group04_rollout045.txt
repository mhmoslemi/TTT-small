import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist

def run_packing():
    # Initial regular hexagonal packing with 16 circles in the interior
    # First, define 4 rows with 4, 5, 5, 4 circles
    # Distances between centers will be slightly adjusted for square packing

    # Define center positions in hexagonal pattern
    row_height = 0.7  # adjusted to fit in square
    col_width = 0.5  # adjusted to fit in square
    centers = []
    radii = []

    # Row 0: 4 circles
    for i in range(4):
        x = i * col_width
        y = 0.25
        centers.append([x, y])
    
    # Row 1: 5 circles, offset by col_width / 2
    for i in range(5):
        x = i * col_width - col_width / 2
        y = 0.25 + row_height
        centers.append([x, y])
    
    # Row 2: 5 circles, offset by col_width / 2
    for i in range(5):
        x = i * col_width - col_width / 2
        y = 0.25 + 2 * row_height
        centers.append([x, y])
    
    # Row 3: 4 circles
    for i in range(4):
        x = i * col_width
        y = 0.25 + 3 * row_height
        centers.append([x, y])

    # Initialize radii
    # For hexagonal packing, radius is min distance to adjacent circles and to boundary
    # Calculate initial approximate radii
    initial_radii = []
    for i, (x, y) in enumerate(centers):
        min_dist_to_boundary = min(x, 1 - x, y, 1 - y)
        min_dist_to_neighbors = float('inf')
        for j in range(len(centers)):
            if i == j:
                continue
            dx = centers[i][0] - centers[j][0]
            dy = centers[i][1] - centers[j][1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < min_dist_to_neighbors:
                min_dist_to_neighbors = dist
        initial_radius = min(min_dist_to_boundary, min_dist_to_neighbors / 2)
        initial_radii.append(initial_radius)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(initial_radii)

    # Optimization to improve packing
    # Define the objective function to maximize the sum of radii
    def objective(x):
        # x contains both centers and radii, we need to reorganize
        # First extract the centers
        idx = 0
        new_centers = []
        for i in range(len(centers)):
            new_centers.append(x[idx:idx+2])
            idx += 2
        new_centers = np.array(new_centers)
        new_radii = x[idx:]
        sum_radii = np.sum(new_radii)
        return -sum_radii  # minimize negative sum to maximize sum

    # Define constraints:
    # 1. Radii must be non-negative
    # 2. Centers must be within the unit square
    # 3. Distance between centers must be at least sum of radii

    # Flatten centers and radii to create a vector for the optimization
    n = len(centers)
    flat_centers = centers.flatten()
    flat_radii = radii
    x0 = np.concatenate([flat_centers, flat_radii])

    # Bounds: centers in [0,1]^2 and radii >= 0
    bounds = []
    for i in range(2 * n):
        bounds.append((0, 1))  # for x and y coordinates
    for i in range(n):
        bounds.append((0, np.inf))  # for radii

    # Define constraints for non-overlapping
    def non_overlap_constraints(x):
        # x contains centers and radii
        idx = 0
        centers = []
        radii = []
        for i in range(n):
            centers.append(x[idx:idx+2])
            radii.append(x[idx+2])
            idx += 2 + 1
        centers = np.array(centers)
        radii = np.array(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                constraints.append(dist - (radii[i] + radii[j]) >= -1e-8)
        return constraints

    # Optimize the configuration
    result = minimize(
        objective,
        x0,
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': non_overlap_constraints},
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract the optimal centers and radii
    optimal_centers = np.reshape(result.x[:2 * n], (n, 2))
    optimal_radii = result.x[2 * n:]

    # Check if the result is valid
    if np.any(np.isnan(optimal_centers)) or np.any(np.isnan(optimal_radii)):
        raise ValueError("NaN values in optimal configuration")

    # Apply clamping for radii to avoid overstepping
    optimal_radii = np.clip(optimal_radii, 0, 1)

    # Ensure all circles are within unit square
    for i in range(n):
        x, y = optimal_centers[i]
        r = optimal_radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12:
            optimal_radii[i] = np.clip(r, 0, 1 - np.max([x - 1e-12, 1 - x - 1e-12]))
        if y - r < -1e-12 or y + r > 1 + 1e-12:
            optimal_radii[i] = np.clip(r, 0, 1 - np.max([y - 1e-12, 1 - y - 1e-12]))

    # Final check for overlap
    for i in range(n):
        for j in range(i + 1, n):
            dx = optimal_centers[i, 0] - optimal_centers[j, 0]
            dy = optimal_centers[i, 1] - optimal_centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < optimal_radii[i] + optimal_radii[j] - 1e-8:
                # Adjust radii to prevent overlap
                overlap = optimal_radii[i] + optimal_radii[j] - dist
                optimal_radii[i] -= overlap / 2
                optimal_radii[j] -= overlap / 2
                optimal_radii[i] = np.clip(optimal_radii[i], 0, 1)
                optimal_radii[j] = np.clip(optimal_radii[j], 0, 1)

    # Final validation of the configuration
    # Re-apply all constraints and ensure all are satisfied
    for i in range(n):
        x, y = optimal_centers[i]
        r = optimal_radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12:
            optimal_radii[i] = np.clip(r, 0, 1 - max(x - 1e-12, 1 - x - 1e-12))
        if y - r < -1e-12 or y + r > 1 + 1e-12:
            optimal_radii[i] = np.clip(r, 0, 1 - max(y - 1e-12, 1 - y - 1e-12))

    # Check for possible overlaps again
    for i in range(n):
        for j in range(i + 1, n):
            dx = optimal_centers[i, 0] - optimal_centers[j, 0]
            dy = optimal_centers[i, 1] - optimal_centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < optimal_radii[i] + optimal_radii[j] - 1e-8:
                # Adjust for overlap
                overlap = optimal_radii[i] + optimal_radii[j] - dist
                optimal_radii[i] -= overlap / 2
                optimal_radii[j] -= overlap / 2
                optimal_radii[i] = np.clip(optimal_radii[i], 0, 1)
                optimal_radii[j] = np.clip(optimal_radii[j], 0, 1)

    return optimal_centers, optimal_radii, np.sum(optimal_radii)
