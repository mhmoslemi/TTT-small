import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    # Initial configuration: hexagonal grid-based packing
    def get_hexagonal_positions(n):
        rows = int(math.ceil(n ** 0.5))
        cols = int(math.ceil(n / rows))
        positions = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * i
                y = i * (math.sqrt(3) / 2)
                positions.append((x, y))
        # Trim to n
        positions = positions[:n]
        return np.array(positions)

    # Scale positions to unit square
    def scale_to_unit(positions):
        max_x = positions[:, 0].max()
        max_y = positions[:, 1].max()
        scaled = positions / np.array([max_x, max_y])
        return scaled

    # Generate initial positions
    positions = get_hexagonal_positions(26)
    positions = scale_to_unit(positions)

    # Define variables: radii for each circle
    # Optimization will adjust positions and radii

    # Objective: maximize sum of radii (minimize negative sum)
    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlap between circles, and boundary constraints
    def constraints(radii):
        def non_overlap(radii, positions):
            n = len(radii)
            for i in range(n):
                for j in range(i + 1, n):
                    dx = positions[i, 0] - positions[j, 0]
                    dy = positions[i, 1] - positions[j, 1]
                    dist = math.hypot(dx, dy)
                    if dist < radii[i] + radii[j] - 1e-6:
                        return False
            return True

        def boundary(radii, positions):
            n = len(radii)
            for i in range(n):
                x, y = positions[i]
                if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                    return False
            return True

        return non_overlap(radii, positions) and boundary(radii, positions)

    # Initial guess: equal radii
    initial_radii = np.ones(26) * 0.05

    # Optimization with bounds
    bounds = [(0, 1) for _ in range(26)]

    # Optimize
    result = minimize(
        fun=objective,
        x0=initial_radii,
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': lambda r: constraints(r)}]
    )

    # Extract results
    radii = result.x
    centers = positions

    # Final validation
    if not constraints(radii, centers):
        raise ValueError("Optimization failed to satisfy constraints")

    # Calculate sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
