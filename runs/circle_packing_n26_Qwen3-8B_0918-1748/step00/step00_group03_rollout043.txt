import numpy as np
from scipy.optimize import minimize
import matplotlib.pyplot as plt

def run_packing():
    # Initial hexagonal packing with 25 circles in a square grid
    # Start with a 5x5 grid and adjust to hexagonal arrangement
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Central layer: 25 circles in a near-hexagonal grid
    for i in range(5):
        for j in range(5):
            x = 0.2 + i * 0.25
            y = 0.2 + j * 0.25
            centers[5*i + j] = [x, y]
    # Add the 26th circle at a corner
    centers[25] = [0.9, 0.9]
    
    # Initial radii
    for i in range(25):
        radii[i] = 0.1
    radii[25] = 0.1
    
    # Define objective function
    def objective(x):
        nonlocal centers, radii
        # x is a flat array of parameters: [r0, r1, ..., r25]
        radii = x
        # Calculate sum of radii
        total = np.sum(radii)
        # Apply constraints to detect overlaps and boundary issues
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return -np.inf
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return total
    
    # Optimization with constraints using Nelder-Mead
    bounds = [(0, 0.5) for _ in range(n)]
    initial_guess = np.full(n, 0.1)
    
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        method='L-BFGS-B',
        options={'ftol': 1e-10, 'gtol': 1e-10}
    )
    
    # Extract result
    radii_result = result.x
    centers_result = centers
    
    # Return centers, radii, and total sum
    return (centers_result, radii_result, np.sum(radii_result))
