    ])

    centers = np.vstack((base_centers, [0.5, 0.5]))
    radii = np.concatenate((base_radii, [0.0]))

    # Optimization: maximize the sum of radii with non-overlapping and boundary constraints
    def objective(x):
        # x is a flat array of [x1, y1, r1, x2, y2, r2, ...]
        # Split into centers and radii
        n = len(centers)
        x_centers = x[:n * 2].reshape(n, 2)
        x_radii = x[n * 2:].reshape(n)
        # Objective: maximize the sum of radii
        return -np.sum(x_radii)  # Minimize negative of sum

    def constraint1(x):
        # Ensure all circles are within the unit square
        n = len(centers)
        x_centers = x[:n * 2].reshape(n, 2)
        x_radii = x[n * 2:].reshape(n)
        for i in range(n):
            x_i, y_i = x_centers[i]
            r_i = x_radii[i]
            if not (0 <= x_i - r_i and x_i + r_i <= 1 and 0 <= y_i - r_i and y_i + r_i <= 1):
                return -1
        return 0

    def constraint2(x):
        # Ensure no two circles overlap
        n = len(centers)
        x_centers = x[:n * 2].reshape(n, 2)
        x_radii = x[n * 2:].reshape(n)
        for i in range(n):
            for j in range(i + 1, n):
                dx = x_centers[i, 0] - x_centers[j, 0]
                dy = x_centers[i, 1] - x_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < x_radii[i] + x_radii[j] - 1e-8:
                    return -1
        return 0

    # Initial guess: centers as base_centers, radii as base_radii + some adjustment
    initial_guess = np.empty(26 * 2 + 26)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
