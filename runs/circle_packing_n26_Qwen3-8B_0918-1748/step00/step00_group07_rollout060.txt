def run_packing():

    grid_points = []

        for j in range(5):
            grid_points.append([i / 4, j / 4])

    grid_points.append([0.5, 0.5])

        # Objective: maximize sum of radii -> minimize negative sum
        return -np.sum(z[n_circles*2:])

        # Enforce that each circle stays within the square [0,1]^2

            idx = i * 2

            y = z[idx + 1]
            r = z[n_circles*2 + i]

                return 1.0
        return 0.0

        for i in range(n_circles):
            for j in range(i + 1, n_circles):

                idx_j = j * 2
                x1 = z[idx_i]
                y1 = z[idx_i + 1]
                r1 = z[n_circles*2 + i]

                y2 = z[idx_j + 1]

        {'type': 'ineq', 'fun': constraint_boundary},
