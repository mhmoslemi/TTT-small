import numpy as np
from scipy.optimize import minimize, Bounds
import time

def run_packing():
    # Initial hexagonal grid configuration for 26 circles
    def initial_hex_grid(n):
        rows = int(np.ceil(np.sqrt(n)))
        cols = int(np.ceil(n / rows))
        grid = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * i
                y = i
                grid.append([x, y])
        return np.array(grid)

    # Optimize centers and radii
    def objective(params, n):
        params = params.reshape(n, 4)
        centers = params[:, :2]
        radii = params[:, 2]
        sum_radii = np.sum(radii)
        return -sum_radii  # Minimize negative to maximize sum_radii

    def constraints(params, n):
        params = params.reshape(n, 4)
        centers = params[:, :2]
        radii = params[:, 2]
        constraints = []
        for i in range(n):
            constraints.append((centers[i, 0] - radii[i]) >= 0)
            constraints.append((1 - centers[i, 0] - radii[i]) >= 0)
            constraints.append((centers[i, 1] - radii[i]) >= 0)
            constraints.append((1 - centers[i, 1] - radii[i]) >= 0)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append((dist - (radii[i] + radii[j])) >= 0)
        return constraints

    # Generate initial configuration
    initial_centers = initial_hex_grid(26)
    initial_radii = np.minimum(0.1, np.sqrt(1 - (initial_centers[:, 0] ** 2 + initial_centers[:, 1] ** 2)))
    initial_params = np.concatenate([initial_centers, initial_radii.reshape(-1, 1)], axis=1)

    # Optimization
    n = 26
    bounds = Bounds([0, 0, 0, 0], [1, 1, 1, 1])

    result = minimize(
        objective,
        initial_params.flatten(),
        args=(n,),
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': lambda x: 0}],
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )

    if not result.success:
        raise RuntimeError("Optimization failed")

    params = result.x.reshape(n, 4)
    centers = params[:, :2]
    radii = params[:, 2]

    # Validate packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(f"Validation failed: {msg}")

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
