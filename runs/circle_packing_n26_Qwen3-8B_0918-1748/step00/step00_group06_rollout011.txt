import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from itertools import combinations

def run_packing():
    # Initial configuration: hexagonal packing with 6 rows
    # Row 1: 5 circles, Row 2: 5, Row 3: 5, Row 4: 5, Row 5: 4, Row 6: 2
    # Hexagonal grid with 2r horizontal spacing and sqrt(3)r vertical spacing

    # Initial guess for radii (equal for all)
    r_initial = 0.08

    # Create initial centers
    rows = [5, 5, 5, 5, 4, 2]
    centers = []
    for i, row_size in enumerate(rows):
        if i % 2 == 0:
            # Even rows (starting from 0) shift right by r
            offset = r_initial
        else:
            # Odd rows align left
            offset = 0
        for j in range(row_size):
            x = j * 2 * r_initial + offset
            y = i * np.sqrt(3) * r_initial
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(len(centers), r_initial)

    # Optimization variables: positions and radii
    n = len(centers)
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # radius

    def objective(params):
        pos = params[:2*n].reshape(n, 2)
        radii = params[2*n:]
        sum_r = np.sum(radii)
        return -sum_r  # Minimize negative sum to maximize sum

    def constraint_dist(params):
        pos = params[:2*n].reshape(n, 2)
        radii = params[2*n:]
        dists = cdist(pos, pos)
        min_dist = np.minimum(radii[:, np.newaxis], radii[np.newaxis, :])
        dists -= min_dist
        return np.min(dists)

    def constraint_boundary(params):
        pos = params[:2*n].reshape(n, 2)
        radii = params[2*n:]
        for i in range(n):
            x, y = pos[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return x - r + 1e-8  # positive if outside
        return 1.0

    constraints = [
        {'type': 'ineq', 'fun': constraint_dist},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]

    # Initial guess: positions and radii
    initial_guess = np.concatenate([centers.flatten(), radii])

    result = minimize(objective, initial_guess, bounds=bounds, constraints=constraints, method='SLSQP', tol=1e-8)

    # Extract final centers and radii
    final_centers = result.x[:2*n].reshape(n, 2)
    final_radii = result.x[2*n:]

    # Validate the packing
    valid, msg = validate_packing(final_centers, final_radii)
    if not valid:
        raise ValueError(msg)

    return final_centers, final_radii, np.sum(final_radii)
