import numpy as np
import scipy.optimize as opt
import itertools

def run_packing():
    # Initial configuration: a structured grid of circles
    # Start with placing 2 circles on each side (corners and edges), and fill the interior
    # This uses a hybrid arrangement with some circles near the edges and some in the center
    
    # Generate initial placement with some randomness to avoid symmetry issues
    # First, place 4 circles at the corners
    centers = np.array([
        [0.1, 0.1],
        [0.9, 0.1],
        [0.1, 0.9],
        [0.9, 0.9]
    ])
    
    # Then, place additional circles along the edges, 4 on each side (excluding corners)
    for i in range(4):
        # Distribute 6 circles along each edge (excluding corners), creating a total of 4 edges × 6 = 24
        # (already placed 4)
        # This will give us 20 more circles to place, alongside the 4 corners, total 24
        # No need for the 26th circle in the initial placement, since we will optimize from here
        pass
    
    # We'll use a more systematic approach to generate the initial configuration
    # Create a grid of points inside the square, and then add some edge-based and corner-based points
    
    # Generate initial points in a grid: 6 rows and 6 columns of points
    grid_points = []
    for i in range(6):
        for j in range(6):
            x = i / 5.0
            y = j / 5.0
            grid_points.append([x, y])
    grid_points = np.array(grid_points)
    centers = grid_points[::2]  # Take every other point to reduce density

    # Add edge-based points: small circles at the midpoints of the square's edges
    edge_points = [
        [0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5]
    ]
    centers = np.vstack((centers, edge_points))

    # Add corner-based small circles
    corners = [
        [0.1, 0.1], [0.1, 0.9], [0.9, 0.1], [0.9, 0.9]
    ]
    centers = np.vstack((centers, corners))

    # Define radii, initially all 0.05
    radii = np.full(centers.shape[0], 0.05)

    # Optimize the configuration using a local optimization algorithm
    def objective(radii):
        return -np.sum(radii)

    def constraint_func(x):
        # x contains all variables: first, the coordinates of the centers, then the radii
        # Split x into centers and radii
        n = len(centers)
        x_centers = x[:2 * n].reshape(n, 2)
        x_radii = x[2 * n:]
        # Check boundaries
        for i in range(n):
            x_i, y_i = x_centers[i]
            r_i = x_radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return -np.inf
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -np.inf
        # Check overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = x_centers[i, 0] - x_centers[j, 0]
                dy = x_centers[i, 1] - x_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < x_radii[i] + x_radii[j] - 1e-12:
                    return -np.inf
        return -np.sum(x_radii)

    bounds = []
    for i in range(len(centers)):
        bounds.extend([(0, 1) for _ in range(2)])  # bounds for x and y
        bounds.extend([(0, 0.5) for _ in range(len(centers))])  # bounds for radii

    # Prepare initial guess
    x0 = np.concatenate([centers.reshape(-1), radii])

    # Optimize
    result = opt.minimize(
        fun=constraint_func,
        x0=x0,
        bounds=bounds,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )

    # Extract final centers and radii
    final_centers = result.x[:2 * len(centers)].reshape((len(centers), 2))
    final_radii = result.x[2 * len(centers):]

    # Apply small heuristic adjustments
    for i in range(len(final_radii)):
        x, y = final_centers[i]
        r = final_radii[i]
        if x - r < -1e-12:
            final_radii[i] = x - 1e-12
        if x + r > 1 + 1e-12:
            final_radii[i] = 1 - x - 1e-12
        if y - r < -1e-12:
            final_radii[i] = y - 1e-12
        if y + r > 1 + 1e-12:
            final_radii[i] = 1 - y - 1e-12

    # Return result
    sum_radii = np.sum(final_radii)
    return final_centers, final_radii, sum_radii
