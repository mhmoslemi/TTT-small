import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Initial grid-based configuration with some spatial variation
    # We'll use a hybrid approach combining multiple grids and clusters
    # The idea is to start with a central cluster and then place the rest
    # in a hexagonal or semi-hexagonal formation
    
    # Initial guess for centers
    centers = []
    radii = []
    
    # Central cluster - 4 circles
    for dx in [-0.2, 0.2]:
        for dy in [-0.2, 0.2]:
            centers.append([0.5 + dx, 0.5 + dy])
            radii.append(0.1)
    
    # Top cluster - 5 circles
    for i in range(5):
        angle = 2 * np.pi * i / 5
        x = 0.5 + 0.3 * np.cos(angle)
        y = 0.95 - 0.3 * np.sin(angle)
        centers.append([x, y])
        radii.append(0.1)
    
    # Bottom cluster - 5 circles
    for i in range(5):
        angle = 2 * np.pi * i / 5
        x = 0.5 + 0.3 * np.cos(angle)
        y = 0.05 + 0.3 * np.sin(angle)
        centers.append([x, y])
        radii.append(0.1)
    
    # Left cluster - 5 circles
    for i in range(5):
        angle = 2 * np.pi * i / 5
        x = 0.05 + 0.3 * np.sin(angle)
        y = 0.5 + 0.3 * np.cos(angle)
        centers.append([x, y])
        radii.append(0.1)
    
    # Right cluster - 5 circles
    for i in range(5):
        angle = 2 * np.pi * i / 5
        x = 0.95 - 0.3 * np.sin(angle)
        y = 0.5 + 0.3 * np.cos(angle)
        centers.append([x, y])
        radii.append(0.1)
    
    # Additional clusters - 2 circles
    centers.append([0.15, 0.15])
    radii.append(0.1)
    centers.append([0.85, 0.15])
    radii.append(0.1)
    centers.append([0.15, 0.85])
    radii.append(0.1)
    centers.append([0.85, 0.85])
    radii.append(0.1)
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Now perform optimization
    # We will use a constrained optimization approach to maximize the sum of radii
    # The constraints are:
    # 1. Each circle must be fully within the unit square (0 <= x - r, x + r <= 1, same for y)
    # 2. Distances between centers >= sum of radii for overlapping circles
    
    # Objective function: maximize sum of radii
    def objective(x):
        # x is flattened array of centers and radii
        # First decode x into centers and radii
        n = len(centers)
        flat_centers = x[:n*2].reshape(n, 2)
        flat_radii = x[n*2:].reshape(n)
        return -np.sum(flat_radii)  # Minimize negative to maximize
    
    # Constraints for boundaries
    def boundary_constraints(x):
        n = len(centers)
        flat_centers = x[:n*2].reshape(n, 2)
        flat_radii = x[n*2:].reshape(n)
        # Each circle must be within the square
        # x_i - r_i >= 0
        # x_i + r_i <= 1
        # y_i - r_i >= 0
        # y_i + r_i <= 1
        cons = []
        for i in range(n):
            x_i, y_i = flat_centers[i]
            r_i = flat_radii[i]
            cons.append(x_i - r_i >= -1e-12)
            cons.append(x_i + r_i <= 1 + 1e-12)
            cons.append(y_i - r_i >= -1e-12)
            cons.append(y_i + r_i <= 1 + 1e-12)
        return cons
    
    # Constraints for non-overlapping circles
    def non_overlap_constraints(x):
        n = len(centers)
        flat_centers = x[:n*2].reshape(n, 2)
        flat_radii = x[n*2:].reshape(n)
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = flat_centers[i, 0] - flat_centers[j, 0]
                dy = flat_centers[i, 1] - flat_centers[j, 1]
                dist_sq = dx**2 + dy**2
                cons.append(np.sqrt(dist_sq) >= flat_radii[i] + flat_radii[j] - 1e-12)
        return cons
    
    # Combine all constraints
    all_constraints = boundary_constraints + non_overlap_constraints
    
    # Initial guess for optimization
    # Start with initial centers and radii
    x0 = np.concatenate([centers.reshape(-1), radii])
    
    # Bounds for all variables: centers must be between 0 and 1, radii must be >= 0
    bounds = []
    for _ in range(centers.shape[0] * 2):
        bounds.append((0.0, 1.0))
    for _ in range(centers.shape[0]):
        bounds.append((0.0, 1.0))
    
    # Set up optimization problem
    res = minimize(objective, x0, bounds=bounds, constraints=all_constraints)
    
    # Extract final solution
    final_centers = res.x[:len(centers)*2].reshape(len(centers), 2)
    final_radii = res.x[len(centers)*2:].reshape(len(centers))
    
    # Compute sum of radii
    sum_radii = np.sum(final_radii)
    
    return final_centers, final_radii, sum_radii
