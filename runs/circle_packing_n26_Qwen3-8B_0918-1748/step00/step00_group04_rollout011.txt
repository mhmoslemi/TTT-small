import numpy as np
import scipy.optimize as opt

def run_packing():
    # Define initial guess for centers and radii
    # Use a hybrid packing: 1 + 2 + 3 + 4 + 4 + 12 = 26 circles
    # In a hexagonal-like grid with some circles at the edges
    
    # Step 1: Generate initial centers in a hexagonal pattern
    def generate_hex_centers(n):
        centers = []
        for row in range(n):
            for col in range(n - row):
                x = col + row / 2
                y = row
                centers.append((x, y))
        return np.array(centers)
    
    # Generate initial configuration
    def generate_initial_centers():
        centers = []
        # First 4 rows of hexagonal grid
        for row in range(4):
            for col in range(5 - row):
                x = col + row / 2
                y = row
                centers.append((x, y))
        # Add more circles at the bottom to reach 26
        for col in range(8):
            x = col / 8
            y = 0.95
            centers.append((x, y))
        # Adjust positions to fit in [0, 1]x[0, 1]
        centers = np.array(centers)
        max_x = np.max(centers[:, 0])
        max_y = np.max(centers[:, 1])
        scaling = 1.0 / max(max_x, max_y)
        centers *= scaling
        return centers

    centers = generate_initial_centers()
    
    # Step 2: Optimize radii and positions
    def objective(x, n_circles=26):
        # x contains: x1, y1, r1, x2, y2, r2, ..., xn, yn, rn
        positions = x[:2*n_circles].reshape(n_circles, 2)
        radii = x[2*n_circles:]
        total_sum = np.sum(radii)
        return -total_sum  # Minimize negative to maximize

    def constraint_overlap(x, n_circles=26):
        positions = x[:2*n_circles].reshape(n_circles, 2)
        radii = x[2*n_circles:]
        for i in range(n_circles):
            for j in range(i+1, n_circles):
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return {'type': 'ineq', 'fun': lambda x: (radii[i] + radii[j]) - dist - 1e-8}
        return {'type': 'ineq', 'fun': lambda x: 0}

    def constraint_boundary(x, n_circles=26):
        positions = x[:2*n_circles].reshape(n_circles, 2)
        radii = x[2*n_circles:]
        for i in range(n_circles):
            x, y = positions[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return {'type': 'ineq', 'fun': lambda x: 1e-8 - (x - r if x - r < -1e-8 else 0) - (x + r - 1 if x + r > 1 + 1e-8 else 0) - (y - r if y - r < -1e-8 else 0) - (y + r - 1 if y + r > 1 + 1e-8 else 0)}
        return {'type': 'ineq', 'fun': lambda x: 0}

    def constraint_non_negative(x, n_circles=26):
        radii = x[2*n_circles:]
        for i in range(n_circles):
            if radii[i] < -1e-8:
                return {'type': 'ineq', 'fun': lambda x: radii[i] + 1e-8}
        return {'type': 'ineq', 'fun': lambda x: 0}

    bounds = []
    for _ in range(26):
        bounds.extend([(0, 1)] * 2)
        bounds.extend([(0, 1)])
    
    # Initial guess with equal radii
    x0 = np.concatenate((centers.reshape(-1), np.array([1.0]*26)))
    
    # Setup optimization problem
    constraints = [constraint_overlap, constraint_boundary, constraint_non_negative]
    res = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints)
    
    # Extract final centers and radii
    positions = res.x[:2*26].reshape(26, 2)
    radii = res.x[2*26:]
    
    # Validate packing
    from validate_packing import validate_packing
    validation_result, message = validate_packing(positions, radii)
    if not validation_result:
        raise ValueError(f"Packing validation failed: {message}")
    
    sum_radii = np.sum(radii)
    return positions, radii, sum_radii
