import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial import distance_matrix

def run_packing():
    # Initial guess for centers: 4 corners, 24 along edges
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Corners
    centers[0] = [0.25, 0.25]
    centers[1] = [0.75, 0.25]
    centers[2] = [0.75, 0.75]
    centers[3] = [0.25, 0.75]
    radii[0] = 0.25
    radii[1] = 0.25
    radii[2] = 0.25
    radii[3] = 0.25
    
    # Horizontal edges
    for i in range(4, 16):
        x = (i - 4) / 12 * 1.0
        y = 0.25 if i % 2 == 0 else 0.75
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Vertical edges
    for i in range(16, 24):
        x = 0.25 if i % 2 == 0 else 0.75
        y = (i - 16) / 12 * 1.0
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Center cluster
    for i in range(24, 26):
        if i == 24:
            centers[i] = [0.5, 0.5]
        else:
            centers[i] = [0.6, 0.6]
        radii[i] = 0.05
    
    # Create bounds for centers (each center must be in the unit square)
    bounds_centers = [(0, 1) for _ in range(26 * 2)]
    bounds_radii = [(0, 1) for _ in range(26)]
    
    # Define objective function: maximize sum of radii
    def objective(x):
        # x contains 26*2 centers and 26 radii
        centers = x[:26 * 2].reshape((26, 2))
        radii = x[26 * 2:]
        return -np.sum(radii)  # Negative for minimization
    
    # Define constraints: for each circle, center + radius <= 1, center - radius >= 0
    def constraint_center_radius(x):
        centers = x[:26 * 2].reshape((26, 2))
        radii = x[26 * 2:]
        for i in range(26):
            for j in range(2):
                if centers[i][j] + radii[i] > 1 + 1e-8:
                    return -1
                if centers[i][j] - radii[i] < -1e-8:
                    return -1
        return 0
    
    # Define constraints: no overlap between circles
    def constraint_overlap(x):
        centers = x[:26 * 2].reshape((26, 2))
        radii = x[26 * 2:]
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    # Initial guess for x
    x0 = np.zeros(26 * 2 + 26)
    x0[:52] = np.array([0.25, 0.25, 0.75, 0.25, 0.75, 0.75, 0.25, 0.75])
    x0[52:] = radii
    
    # Constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_center_radius},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    # Bounds
    bnds = Bounds(bounds_centers + bounds_radii)
    
    # Optimize
    result = minimize(objective, x0, method='SLSQP', bounds=bnds, constraints=cons)
    
    # Extract centers and radii
    centers = result.x[:52].reshape((26, 2))
    radii = result.x[52:]
    
    # Validate
    assert validate_packing(centers, radii)[0], "Packing is invalid"
    
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
