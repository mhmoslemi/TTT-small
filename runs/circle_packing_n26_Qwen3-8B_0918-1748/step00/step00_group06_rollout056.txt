import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Initial guess: Hexagonal packing with some rows
    # Let's try 5 rows of 5 circles (with some adjusted for boundary constraints)
    # We will start with standard hexagonal packing and then optimize

    # Initial configuration with 5 rows of 5 circles
    n_circles = 26
    row_cols = [5, 5, 5, 5, 6]  # The first 4 rows have 5 circles, 5th has 6
    centers = np.zeros((n_circles, 2))
    x = 0.5
    y = 0.5
    row_offset = 0.1
    col_offset = 0.25

    for i in range(5):
        for j in range(row_cols[i]):
            centers[i * row_cols[i] + j] = [x + j * (1 - row_offset), y + i * row_offset]

    # Refine the initial guess using optimization
    def objective(radii):
        return -np.sum(radii)  # Negative for minimization

    # Define bounds for radii (non-negative)
    bounds = [(0, 1.0) for _ in range(n_circles)]

    # Define constraints: distance between centers >= sum of radii
    def constraint_func(x):
        x = np.reshape(x, (n_circles, 2))
        radii = x[:, 2:]
        centers = x[:, :2]
        dist_matrix = squareform(pdist(centers))
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return dist_matrix[i, j] - (radii[i] + radii[j] - 1e-8) + 1e-6
        return 0.0

    # Flatten centers and radii for optimization
    x0 = np.concatenate((centers.reshape(-1), np.ones(n_circles)))

    # Add constraints to the optimization
    cons = ({'type': 'ineq', 'fun': constraint_func, 'jac': None})

    # Optimize
    result = minimize(objective, x0, bounds=bounds, constraints=cons)

    # Extract the optimized centers and radii
    centers_opt = result.x[:2 * n_circles].reshape(n_circles, 2)
    radii_opt = result.x[2 * n_circles:]

    # Validate the packing
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Packing validation failed: {message}")

    # Improve by adjusting boundary circles (for better packing efficiency)
    # Let's make sure the circles are touching the boundaries where possible
    # First, separate boundary and internal circles
    boundary_indices = []
    for i in range(n_circles):
        x, y = centers_opt[i]
        if (x - radii_opt[i] < 0 or x + radii_opt[i] > 1 or y - radii_opt[i] < 0 or y + radii_opt[i] > 1):
            boundary_indices.append(i)

    # For boundary circles, adjust their radii to touch the boundary
    for i in boundary_indices:
        x, y = centers_opt[i]
        r = radii_opt[i]
        # Find closest boundary
        dist_to_left = x - r
        dist_to_right = 1 - x - r
        dist_to_bottom = y - r
        dist_to_top = 1 - y - r

        min_dist = min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top)
        if min_dist < 1e-8:
            # Adjust the radius to be as large as possible without going out of bounds
            # For example, adjust to make it exactly touch one side
            if dist_to_left < dist_to_right and dist_to_left < dist_to_bottom and dist_to_left < dist_to_top:
                radii_opt[i] = x
            elif dist_to_right < dist_to_left and dist_to_right < dist_to_bottom and dist_to_right < dist_to_top:
                radii_opt[i] = 1 - x
            elif dist_to_bottom < dist_to_left and dist_to_bottom < dist_to_right and dist_to_bottom < dist_to_top:
                radii_opt[i] = y
            elif dist_to_top < dist_to_left and dist_to_top < dist_to_right and dist_to_top < dist_to_bottom:
                radii_opt[i] = 1 - y
            else:
                # If multiple directions are equal, choose one at random
                # For simplicity, adjust to make it touch the left boundary
                radii_opt[i] = x

    # Final validation
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Packing validation failed: {message}")

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
