import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
from typing import Tuple

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    # Initial hexagonal packing configuration
    def hexagonal_packing(n: int) -> Tuple[np.ndarray, np.ndarray]:
        num_rows = int(np.ceil(n / 2))
        centers = []
        radii = []
        for row in range(num_rows):
            for col in range(2 * row + 1, 2 * row + 1 + n - 2 * row):
                if col >= n:
                    break
                x = (col + 0.5 * row) / (2 * row + 1)
                y = (row + 0.5) / num_rows
                centers.append([x, y])
        centers = np.array(centers)
        radii = np.full(len(centers), 1 / (2 * num_rows))
        return centers, radii

    # Initial packing
    centers, radii = hexagonal_packing(26)
    
    # Constraint function for optimization
    def constraint_func(x: np.ndarray) -> np.ndarray:
        # x is flattened [x1, y1, r1, x2, y2, r2, ...]
        n = len(radii)
        centers_opt = x[:2*n].reshape(n, 2)
        radii_opt = x[2*n:]
        constraints = []
        for i in range(n):
            x_i, y_i = centers_opt[i]
            r_i = radii_opt[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                constraints.append(-1)
            else:
                constraints.append(0)
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_opt[i] + radii_opt[j] - 1e-12:
                    constraints.append(-1)
                else:
                    constraints.append(0)
        return np.array(constraints)

    # Objective function to maximize sum of radii
    def objective(x: np.ndarray) -> float:
        n = len(radii)
        centers_opt = x[:2*n].reshape(n, 2)
        radii_opt = x[2*n:]
        return np.sum(radii_opt)

    # Initial guess with centers and radii
    initial_guess = np.concatenate([centers.flatten(), radii])
    
    # Perform optimization
    result = minimize(
        fun=objective,
        x0=initial_guess,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(2*26)] + [(0, 1) for _ in range(26)],
        constraints={
            'type': 'ineq',
            'fun': constraint_func,
            'jac': lambda x: np.concatenate([np.zeros(2*26), np.ones(26)])
        },
        options={'maxiter': 1000, 'ftol': 1e-8}
    )
    
    # Extract final solution
    centers_opt = result.x[:2*26].reshape(26, 2)
    radii_opt = result.x[2*26:]
    total_sum = np.sum(radii_opt)
    
    return centers_opt, radii_opt, total_sum
