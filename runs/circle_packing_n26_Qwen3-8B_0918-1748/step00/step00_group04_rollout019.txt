import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define initial guess for centers
    centers = np.zeros((26, 2))
    # Hexagonal packing with 5 rows
    for i in range(5):
        for j in range(5 + (i % 2)):
            centers[i * 6 + j] = [j / 5.0, i / 5.0]
    
    # Initial guess for radii
    radii = np.full(26, 0.1)
    
    # Define optimization variables: positions (x, y) and radii
    n = 26
    bounds = []
    for i in range(n):
        # Positions must be within [0, 1]
        bounds.append((0, 1))
        bounds.append((0, 1))
        # Radii must be non-negative
        bounds.append((0, 1))  # Radius is a fraction of the square size
    
    # Optimization variables: x, y, r for each circle
    x0 = np.concatenate([centers.reshape(-1), radii])
    
    def objective(x):
        # Objective function: maximize sum of radii (minimize negative sum)
        pos = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        return -np.sum(r)
    
    def constraint_positions(x):
        pos = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        for i in range(n):
            x_i, y_i = pos[i]
            r_i = r[i]
            if x_i - r_i < -1e-8 or x_i + r_i > 1 + 1e-8:
                return -1
            if y_i - r_i < -1e-8 or y_i + r_i > 1 + 1e-8:
                return -1
        return 0
    
    def constraint_overlap(x):
        pos = x[:2*n].reshape(n, 2)
        r = x[2*n:]
        for i in range(n):
            for j in range(i + 1, n):
                dx = pos[i, 0] - pos[j, 0]
                dy = pos[i, 1] - pos[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < r[i] + r[j] - 1e-8:
                    return -1
        return 0
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_positions},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'maxiter': 500, 'ftol': 1e-8, 'disp': False}
    )
    
    pos = result.x[:2*n].reshape(n, 2)
    r = result.x[2*n:]
    
    # Ensure non-negative radii
    r = np.clip(r, 0, 1)
    
    # Validate packing
    valid, msg = validate_packing(pos, r)
    if not valid:
        raise ValueError(msg)
    
    sum_radii = np.sum(r)
    return pos, r, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
