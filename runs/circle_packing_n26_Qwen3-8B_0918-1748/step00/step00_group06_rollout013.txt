import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import LinearConstraint

def run_packing():
    np.random.seed(42)
    
    # Initial base packing: hexagonal arrangement of 20 circles with small radii
    # Start from the center, then build outward in a hexagonal grid
    centers = []
    radii = []
    
    # First circle at the center
    centers.append([0.5, 0.5])
    radii.append(0.1)
    
    # Build hexagonal lattice around the center
    for layer in range(1, 5):
        for angle in np.linspace(0, 2 * np.pi, 6, endpoint=False):
            x = 0.5 + (layer + 1) * np.cos(angle)
            y = 0.5 + (layer + 1) * np.sin(angle)
            if 0 <= x <= 1 and 0 <= y <= 1:
                centers.append([x, y])
                radii.append(0.1)
    
    # Refine the base packing using local optimization
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Define optimization variables: positions and radii
    n = len(centers)
    variables = np.concatenate([centers.ravel(), radii])
    
    # Objective function: maximize total sum of radii
    def objective(x):
        return -np.sum(x[n*2: ])  # negate because we minimize
    
    # Constraints: circle centers within [0, 1]^2 and radii non-negative
    bounds = Bounds(np.zeros(n*2 + n), np.inf)
    
    # Constraint: distance between centers >= sum of radii
    def constraint(x):
        pos = x[:n*2].reshape(n, 2)
        rad = x[n*2:]
        dists = squareform(pdist(pos))
        return dists - (rad[:, None] + rad[None, :])
    
    # Create a constraint that this is >= 0
    def constraint_func(x):
        return constraint(x).flatten()
    
    # Initialize an optimization problem
    opt_problem = minimize(
        objective,
        variables,
        bounds=bounds,
        constraints=LinearConstraint(
            constraint,
            lb=np.zeros(len(constraint)),
            ub=np.inf
        )
    )
    
    # Extract optimized centers and radii
    centers_opt = opt_problem.x[:n*2].reshape(n, 2)
    radii_opt = opt_problem.x[n*2:]
    
    # Final check to ensure the sqrt(2) spacing for largest circles
    # Consider placing additional circles in corners
    num_corners = 4
    corner_positions = [
        [0.1, 0.1], [0.9, 0.1], [0.9, 0.9], [0.1, 0.9]
    ]
    corner_radii = [0.05] * num_corners
    
    # Check for overlaps with existing circles
    def check_overlap(corners, corner_radii, centers, radii):
        for i in range(len(corners)):
            for j in range(len(centers)):
                dist = np.sqrt((corners[i][0] - centers[j][0])**2 + (corners[i][1] - centers[j][1])**2)
                if dist < corner_radii[i] + radii[j] - 1e-8:
                    return True
        return False
    
    # If there is no overlap, add the corner circles
    if not check_overlap(corner_positions, corner_radii, centers_opt, radii_opt):
        centers_opt = np.vstack([centers_opt, corner_positions])
        radii_opt = np.concatenate([radii_opt, corner_radii])
    
    # Final validation
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"
    
    valid, msg = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")
    
    sum_radii = np.sum(radii_opt)
    
    return centers_opt, radii_opt, sum_radii
