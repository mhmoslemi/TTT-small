import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    # Initial guess for centers: hexagonal packing with 25 circles
    def hex_grid(n):
        rows = int(np.ceil(n**0.5))
        cols = int(np.ceil(n / rows))
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * (i % 2)
                y = i + np.sqrt(3)/2 * (i % 2)
                if i % 2 == 1:
                    x += 0.5
                centers.append([x, y])
        centers = np.array(centers)
        centers = np.clip(centers, 0, 1)
        return centers

    # Adjust the grid to fit inside the unit square
    def scale_grid(centers):
        x_min, x_max = np.min(centers[:, 0]), np.max(centers[:, 0])
        y_min, y_max = np.min(centers[:, 1]), np.max(centers[:, 1])
        scale = 1.0 / max(x_max - x_min, y_max - y_min)
        centers = centers * scale
        return centers

    # Initial configuration with 25 circles in a hexagonal grid
    centers = hex_grid(25)
    centers = scale_grid(centers)

    # Add the 26th circle at a strategic position
    centers = np.vstack((centers, [0.5, 0.5]))
    radii = np.full(26, 1.0 / 26)

    # Optimization: maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    def constraint_overlap(i, j, centers, radii):
        dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
        return dist - radii[i] - radii[j]

    def constraint_boundary(i, centers, radii):
        x, y = centers[i]
        return (x - radii[i]) >= 0 and (x + radii[i]) <= 1 and (y - radii[i]) >= 0 and (y + radii[i]) <= 1

    # Define all constraints
    constraints = []
    for i in range(26):
        for j in range(i + 1, 26):
            constraints.append((lambda x, i=i, j=j: constraint_overlap(i, j, centers, x), 0))
        constraints.append((lambda x, i=i: constraint_boundary(i, centers, x), 0))

    # Optimize
    result = minimize(
        objective,
        radii,
        constraints=[{'type': 'ineq', 'fun': c} for c in constraints],
        bounds=[(0, 1) for _ in range(26)],
        method='SLSQP'
    )

    # Validate the result
    radii = result.x
    centers = np.array(centers)
    return (centers, radii, np.sum(radii))
