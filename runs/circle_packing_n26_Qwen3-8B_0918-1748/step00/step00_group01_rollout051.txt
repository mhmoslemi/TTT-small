import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.spatial import distance
from numpy.linalg import norm

def run_packing():
    # Define the number of circles
    n = 26
    # Initial guesses: spread the circles in a grid shape
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    for i in range(n):
        row = i // 5
        col = i % 5
        centers[i, 0] = col / 4.5 + 0.05
        centers[i, 1] = row / 4.5 + 0.05
        radii[i] = 0.05

    # Objective function: maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)

    # Constraints: non-overlapping and boundary
    def constraint_overlap(radii, centers, i, j):
        dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
        return dist - (radii[i] + radii[j])

    def constraint_boundary(radii, centers, i):
        x, y = centers[i]
        return x - radii[i] >= 0 and y - radii[i] >= 0 and x + radii[i] <= 1 and y + radii[i] <= 1

    # Optimization problem
    bounds = [(0, 0.2) for _ in range(n)]

    # Initial guess
    x0 = np.concatenate([centers.flatten(), radii])

    def objective_with_params(x):
        r = x[n*2:]
        c = x[:n*2].reshape((n, 2))
        return -np.sum(r)

    def constraint_overlap_with_params(x):
        r = x[n*2:]
        c = x[:n*2].reshape((n, 2))
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((c[i] - c[j])**2))
                if dist < r[i] + r[j] - 1e-12:
                    return -dist + r[i] + r[j] - 1e-2
        return 0

    def constraint_boundary_with_params(x):
        r = x[n*2:]
        c = x[:n*2].reshape((n, 2))
        for i in range(n):
            x_i, y_i = c[i]
            if x_i - r[i] < -1e-12 or x_i + r[i] > 1 + 1e-12 or y_i - r[i] < -1e-12 or y_i + r[i] > 1 + 1e-12:
                return -1
        return 0

    # Optimization with bounds
    res = minimize(objective_with_params, x0, bounds=bounds, constraints=[
        {'type': 'ineq', 'fun': constraint_overlap_with_params},
        {'type': 'ineq', 'fun': constraint_boundary_with_params}
    ])

    # Extract results
    centers_opt = res.x[:n*2].reshape((n, 2))
    radii_opt = res.x[n*2:]
    sum_radii = np.sum(radii_opt)

    return centers_opt, radii_opt, sum_radii
