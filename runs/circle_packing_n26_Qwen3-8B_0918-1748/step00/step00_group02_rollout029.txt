import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds, InequalityConstraint
import matplotlib.pyplot as plt

def run_packing():
    # Initial configuration: 6 rows of circles with 4, 5, 5, 5, 4, 3 circles in each row
    # The rows are placed in a hexagonal pattern, alternating spacing
    # Initial radii are chosen to fit within the square with minimal overlap

    initial_centers = []
    initial_radii = []

    # Define the initial offset and spacing
    row_height = 0.8
    col_width = 0.6
    offset = 0.3

    # Row 0: 4 circles
    for i in range(4):
        x = i * col_width
        y = 0.5
        initial_centers.append([x, y])
        initial_radii.append(0.1)

    # Row 1: 5 circles
    for i in range(5):
        x = i * col_width - 0.2
        y = 0.5 + row_height / 2
        initial_centers.append([x, y])
        initial_radii.append(0.1)

    # Row 2: 5 circles
    for i in range(5):
        x = i * col_width - 0.2
        y = 0.5 + row_height
        initial_centers.append([x, y])
        initial_radii.append(0.1)

    # Row 3: 5 circles
    for i in range(5):
        x = i * col_width - 0.2
        y = 0.5 + 1.5 * row_height / 2
        initial_centers.append([x, y])
        initial_radii.append(0.1)

    # Row 4: 4 circles
    for i in range(4):
        x = i * col_width
        y = 0.5 + 2 * row_height
        initial_centers.append([x, y])
        initial_radii.append(0.1)

    # Row 5: 3 circles
    for i in range(3):
        x = i * col_width + 0.2
        y = 0.5 + 2.5 * row_height / 2
        initial_centers.append([x, y])
        initial_radii.append(0.1)

    initial_centers = np.array(initial_centers)
    initial_radii = np.array(initial_radii)

    # Define the objective function to maximize sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative of sum to maximize sum

    # Define constraints
    def constraint_boundary(radii, centers):
        return np.sum(centers[:, 0] - radii + centers[:, 1] - radii)

    def constraint_overlap(radii, centers):
        n = centers.shape[0]
        constraints = []
        for i in range(n):
            x, y = centers[i]
            dist = np.sqrt((x - centers[i])**2 + (y - centers[i])**2)
            constraints.append(dist >= radii[i] + radii[i] - 1e-6)
        return np.array(constraints)

    # Bounds for radii (non-negative)
    bounds = Bounds(0, 1)

    # Initial guess for radii
    initial_guess = initial_radii

    # Optimization
    result = minimize(
        objective,
        initial_guess,
        args=(initial_centers,),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_boundary, 'args': (initial_centers,)},
            {'type': 'ineq', 'fun': constraint_overlap, 'args': (initial_centers,)},
        ]
    )

    # Extract the final centers and radii
    final_centers = initial_centers
    final_radii = result.x

    # Ensure valid final_radii
    valid = True
    for r in final_radii:
        if r < 0:
            valid = False
            break
    if not valid:
        raise ValueError("Optimization resulted in negative radii.")

    # Final sum of radii
    sum_radii = np.sum(final_radii)

    return (final_centers, final_radii, sum_radii)
