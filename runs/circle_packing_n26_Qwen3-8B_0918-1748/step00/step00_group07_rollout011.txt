import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial configuration: hexagonal packing with 26 circles
    # Start with 5 rows of circles, with 5, 5, 5, 5, and 6 circles respectively
    # Hexagonal packing will help with spacing

    # Define initial guess for centers and radii
    # Each circle placed in hexagonal grid
    # We'll use a compact hexagonal arrangement that fits into the unit square
    # Start with compact grid and then optimize

    # Number of circles
    n = 26

    # Generate initial positions
    # Initialize with a compact hexagonal grid
    # Rows: 5 rows with varying numbers of circles
    # First row: 6 circles (left-right)
    # Second row: 5 circles (offset)
    # Third row: 5 circles (left-right)
    # Fourth row: 5 circles (offset)
    # Fifth row: 5 circles (left-right)
    # Total of 6 + 5 + 5 + 5 + 5 = 26

    # Define initial grid positions
    grid = []
    for i in range(5):
        row = []
        if i % 2 == 0:
            for j in range(6 if i == 0 else 5):
                x = j / (6 - 1) if i == 0 else (j + 0.5) / (5 - 1)
                y = i / (5 - 1)
                row.append((x, y))
        else:
            for j in range(5):
                x = (j + 0.5) / (5 - 1)
                y = i / (5 - 1)
                row.append((x, y))
        grid.append(row)

    # Flatten the grid into centers
    centers = np.array([np.array([x, y]) for row in grid for x, y in row])

    # Initial radii as uniform value based on grid spacing
    # Base spacing: approx 1/6 or 1/5 of square length
    spacing = 0.2
    initial_r = spacing * 0.7  # Start with smaller radii to prevent initial overlap
    radii = initial_r * np.ones(n)

    # Optimization parameters
    bounds = []
    for i in range(n):
        # Bound x: [0, 1]
        # Bound y: [0, 1]
        # Bound r: [0, 1]
        bounds.append((0, 1))
        bounds.append((0, 1))
        bounds.append((0, 1))

    def objective(x):
        # x contains: x1, y1, r1, x2, y2, r2, ..., xn, yn, rn
        # Reshape into centers and radii
        centers_flat = x[:n * 2].reshape(n, 2)
        radii_flat = x[n * 2:].reshape(n)
        # Compute the sum of radii
        return -np.sum(radii_flat)

    def constraints_func(x):
        # x contains: x1, y1, r1, x2, y2, r2, ..., xn, yn, rn
        centers_flat = x[:n * 2].reshape(n, 2)
        radii_flat = x[n * 2:].reshape(n)
        # Constraints: distance between centers >= sum of radii
        # Also, check if circles are within the unit square
        # List to store constraint violations
        violations = []
        # Check if circles are within square
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if not (0 - r_i <= x_i <= 1 + r_i and 0 - r_i <= y_i <= 1 + r_i):
                violations.append(1)
        # Check if circles are overlapping
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers_flat[i] - centers_flat[j])**2))
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    violations.append(1)
        return violations

    # Optimize with L-BFGS-B
    result = opt.minimize(
        objective,
        np.r_[centers.flatten(), radii.flatten()],
        bounds=bounds,
        constraints={"type": "ineq", "fun": lambda x: -constraints_func(x)},
        method='L-BFGS-B',
        options={'maxiter': 500, 'ftol': 1e-8, 'gtol': 1e-8}
    )

    # Extract final centers and radii
