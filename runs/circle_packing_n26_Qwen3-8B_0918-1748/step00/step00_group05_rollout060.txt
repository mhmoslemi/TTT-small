import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run_packing():
    # Initial guess: grid-based layout with 5 rows and 5 columns
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # 5x5 grid in the center
    spacing = 1.0 / (5 + 1)  # Spacing between centers
    for i in range(5):
        for j in range(5):
            centers[5 * i + j] = [spacing + i * spacing * 2, spacing + j * spacing * 2]
            radii[5 * i + j] = spacing * 0.5

    # 26th circle in the bottom-left corner
    centers[25] = [spacing, spacing]
    radii[25] = spacing * 0.5

    # Objective function: maximize the sum of radii
    def objective(x):
        return -np.sum(x[::2] + x[1::2])  # Minimize the negative of the sum

    # Constraints: non-overlapping and boundary constraints
    def non_overlapping_and_boundary(x):
        # Extract centers and radii
        centers = np.reshape(x[:num_circles * 2], (num_circles, 2))
        radii = x[num_circles * 2:]

        # Constraint function
        constraints = []
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-10:
                    return -1
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-10 or x_i + r_i > 1 + 1e-10 or y_i - r_i < -1e-10 or y_i + r_i > 1 + 1e-10:
                return -1
        return 0

    # Bounds: all positions must be between 0 and 1, radii must be non-negative
    bounds = []
    for i in range(num_circles):
        bounds.append((0, 1))  # x-coordinate
        bounds.append((0, 1))  # y-coordinate
        bounds.append((0, 1))  # radius

    # Initial guess
    x0 = np.concatenate([centers.ravel(), radii])

    # Set up the optimization problem
    def constraint_func(x):
        return non_overlapping_and_boundary(x)

    cons = {'type': 'ineq', 'fun': constraint_func}

    # Run the optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-8, options={'maxiter': 500, 'ftol': 1e-8})

    # Check if the result is successful
    if not result.success:
        raise ValueError(f"Optimization failed with message: {result.message}")

    # Extract centers and radii
    centers = np.reshape(result.x[:num_circles * 2], (num_circles, 2))
    radii = result.x[num_circles * 2:]

    # Check for any issues
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} is outside the unit square")

    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
