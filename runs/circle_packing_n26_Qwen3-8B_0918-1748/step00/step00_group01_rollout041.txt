import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Initial guess for circle positions and radii based on hexagonal packing
    n = 26
    rows = 6
    cols = [4, 5, 5, 5, 4, 1]
    cols = [4, 5, 5, 5, 4, 1]
    total_cols = sum(cols)

    # Calculate initial radius based on unit square constraints
    # Estimate a radius based on the height of the hexagonal arrangement
    radius_initial = 0.1  # Initial guess, will be optimized

    # Generate initial grid positions
    def generate_initial_positions(rows, cols):
        positions = []
        for i in range(rows):
            col_count = cols[i]
            for j in range(col_count):
                x = j * (2 * radius_initial) + radius_initial
                y = i * (np.sqrt(3) * radius_initial) + radius_initial
                positions.append([x, y])
        return np.array(positions)

    # Adjust positions to fit within [0, 1] x [0, 1]
    def scale_positions(positions):
        x_min, x_max = positions[:, 0].min(), positions[:, 0].max()
        y_min, y_max = positions[:, 1].min(), positions[:, 1].max()
        scale_x = (1 - 1e-6) / (x_max - x_min)
        scale_y = (1 - 1e-6) / (y_max - y_min)
        scaled_positions = np.array([
            [x * scale_x + 1e-6, y * scale_y + 1e-6]
            for x, y in positions
        ])
        return scaled_positions

    # Initial positions
    initial_positions = generate_initial_positions(rows, cols)
    initial_positions = scale_positions(initial_positions)

    # Optimization parameters
    radius_bounds = (0, 0.5)
    radius_initial = 0.12

    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraints: non-overlapping circles and within unit square
    def constraints(x):
        positions = x[:2 * n].reshape((n, 2))
        radii = x[2 * n:]
        # Check if all circles are inside unit square
        for i in range(n):
            x_pos, y_pos = positions[i]
            r = radii[i]
            if not (0 <= x_pos - r and x_pos + r <= 1 and 0 <= y_pos - r and y_pos + r <= 1):
                return False
        # Check if all circles are non-overlapping
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.hypot(positions[i, 0] - positions[j, 0], positions[i, 1] - positions[j, 1])
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    # Initial guess: positions and radii
    initial_guess = np.r_[initial_positions.ravel(), np.full(n, radius_initial)]

    # Bounds for positions and radii
    bounds = []
    for _ in range(2 * n):
        bounds.append((0, 1))
    for _ in range(n):
        bounds.append(radius_bounds)

    # Optimization with constraints
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints={"type": "success", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-6, "maxiter": 100}
    )

    # Extract centers and radii
    centers = result.x[:2 * n].reshape((n, 2))
    radii = result.x[2 * n:]

    # Validate the packing
    validation_result, message = validate_packing(centers, radii)
    if not validation_result:
        raise ValueError(f"Packing validation failed: {message}")

    # Compute sum of radii
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
