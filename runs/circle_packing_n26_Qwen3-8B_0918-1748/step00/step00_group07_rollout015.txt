import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial import distance
import random

def run_packing():
    # Initial configuration with a hybrid layout: 
    # - 3 rows of 4 circles in the center
    # - 3 rows of 3 circles near the top and bottom
    # - 2 circles in the corners
    initial_centers = np.array([
        [0.2, 0.2], [0.4, 0.2], [0.6, 0.2], [0.8, 0.2],
        [0.2, 0.4], [0.4, 0.4], [0.6, 0.4], [0.8, 0.4],
        [0.2, 0.6], [0.4, 0.6], [0.6, 0.6], [0.8, 0.6],
        [0.1, 0.9], [0.5, 0.9], [0.9, 0.9],
        [0.9, 0.1], [0.5, 0.1], [0.1, 0.1]
    ])
    
    # Initial radii based on spacing
    initial_radii = np.full(26, 0.05)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraints(radii):
        # Constraints: circles cannot go out of bounds or overlap
        constraints = []
        for i in range(26):
            if radii[i] < 0:
                constraints.append((radii[i] + 1e-12, 0))
        
        # Check if all circles are within the square [0,1]^2
        constraints.append((np.sum(radii) + 1e-12, 2.636))
        
        # Check for circle overlaps
        for i in range(26):
            for j in range(i + 1, 26):
                # Compute distance between centers
                centers_i = initial_centers[i]
                centers_j = initial_centers[j]
                dist = np.sqrt(np.sum((centers_i - centers_j)**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    constraints.append((radii[i] + radii[j] - dist + 1e-12, 0))
        
        return constraints
    
    # Initial optimization with bounds
    bounds = Bounds(np.zeros(26), np.full(26, 1))
    result = minimize(objective, initial_radii, bounds=bounds, constraints=constraints)
    
    # Post-processing: refine the configuration by fine-tuning positions and radii
    # Introduce small random perturbations to centers
    perturbation = 0.02 * np.random.rand(26, 2)
    centers = initial_centers + perturbation
    centers[centers < 0] = 0
    centers[centers > 1] = 1
    
    # Calculate radii based on nearest neighbors and boundary constraints
    radii = np.zeros(26)
    for i in range(26):
        min_dist_to_boundary = np.min([centers[i, 0], 1 - centers[i, 0], centers[i, 1], 1 - centers[i, 1]])
        radii[i] = min_dist_to_boundary * 0.8  # 80% of boundary distance
    
    # Add additional distance constraints between circles
    for i in range(26):
        for j in range(i + 1, 26):
            dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
            if dist < radii[i] + radii[j] - 1e-12:
                radii[i] = (dist - 1e-12) / 2
                radii[j] = (dist - 1e-12) / 2
                
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
