import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint

def run_packing():
    # Initial estimate for radii (average radius based on target sum)
    target_sum = 2.636
    initial_radius = target_sum / 26
    initial_radii = np.full(26, initial_radius)
    
    # Define bounds for radii
    radius_bounds = [(0, 1) for _ in range(26)]
    
    # Define positions for a hexagonal grid layout
    def hexagonal_positions(n):
        rows = int(np.sqrt(n))
        cols = n // rows + (1 if n % rows else 0)
        positions = []
        for i in range(rows):
            for j in range(cols):
                x = j + 0.5 * i
                y = i + 0.5
                positions.append((x, y))
        return np.array(positions)
    
    # Initial positions in a hexagonal grid
    initial_centers = hexagonal_positions(26)
    
    # Adjust initial positions to fit in the unit square
    initial_centers[:, 0] /= np.max(initial_centers[:, 0])
    initial_centers[:, 1] /= np.max(initial_centers[:, 1])
    
    # Define objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative of sum for maximization
    
    # Define constraints for non-overlapping and boundary conditions
    def constraint_func(radii):
        centers = np.zeros((26, 2))
        for i in range(26):
            centers[i, 0] = initial_centers[i, 0] * (1 - radii[i] / 0.5)
            centers[i, 1] = initial_centers[i, 1] * (1 - radii[i] / 0.5)
        
        # Boundary constraint (each circle must fit in [0,1]^2)
        for i in range(26):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12:
                return np.array([1.0])
            if y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return np.array([1.0])
        
        # Non-overlap constraint (distance between any two circles must be >= sum of radii)
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.array([1.0])
        
        return np.array([0.0])
    
    # Define bounds and constraints for the optimization
    bounds = Bounds([0.0, 0.0], [1.0, 1.0])
    con = NonlinearConstraint(constraint_func, 0, 0)
    
    # Use SLSQP to solve the optimization problem
    result = optimize.minimize(
        objective,
        initial_radii,
        bounds=bounds,
        constraints=con,
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 1000}
    )
    
    # Extract optimized centers and radii
    optimized_radii = result.x
    optimized_centers = np.zeros((26, 2))
    for i in range(26):
        x = initial_centers[i, 0] * (1 - optimized_radii[i] / 0.5)
        y = initial_centers[i, 1] * (1 - optimized_radii[i] / 0.5)
        optimized_centers[i, 0] = x
        optimized_centers[i, 1] = y
    
    total_sum_radii = np.sum(optimized_radii)
    return (optimized_centers, optimized_radii, total_sum_radii)
