import numpy as np
import cvxpy as cp

def run_packing():
    # Define the positions of 26 circles in a truncated hexagonal pattern
    centers = []
    radii = []

    # Central circle
    centers.append([0.5, 0.5])

    # Ring 1: 2 circles
    for angle in [0, np.pi]:
        x = 0.5 + np.cos(angle) * 0.3
        y = 0.5 + np.sin(angle) * 0.3
        centers.append([x, y])

    # Ring 2: 7 circles
    for i in range(7):
        angle = 2 * np.pi * i / 7
        x = 0.5 + np.cos(angle) * 0.45
        y = 0.5 + np.sin(angle) * 0.45
        centers.append([x, y])

    # Ring 3: 16 circles
    for i in range(16):
        angle = 2 * np.pi * i / 16
        x = 0.5 + np.cos(angle) * 0.55
        y = 0.5 + np.sin(angle) * 0.55
        centers.append([x, y])

    centers = np.array(centers)
    n = 26

    # Initial radii estimate
    max_radius_guess = 0.1
    radii = np.full(n, max_radius_guess)

    # Optimization variables
    r = cp.Variable(n)
    x = cp.Variable((n, 2))

    # Constraints
    constraints = []

    # Boundary constraints: radius cannot exceed distance to nearest boundary
    for i in range(n):
        x_i, y_i = centers[i]
        dist_to_left = x_i - 1e-12
        dist_to_right = 1 - x_i + 1e-12
        dist_to_bottom = y_i - 1e-12
        dist_to_top = 1 - y_i + 1e-12
        min_dist = np.minimum(np.minimum(dist_to_left, dist_to_right), np.minimum(dist_to_bottom, dist_to_top))
        constraints.append(r[i] <= min_dist)

    # Non-overlapping constraint: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0]) ** 2 + (x[i, 1] - x[j, 1]) ** 2)
            constraints.append(dist >= r[i] + r[j] - 1e-12)

    # Center and radius constraints
    for i in range(n):
        constraints.append(x[i, 0] >= 1e-12)
        constraints.append(x[i, 0] <= 1 - 1e-12)
        constraints.append(x[i, 1] >= 1e-12)
        constraints.append(x[i, 1] <= 1 - 1e-12)
        constraints.append(r[i] >= 0)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))

    # Solve the problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.OSQP, eps=1e-5)

    # Extract results
    centers = x.value
    radii = r.value
    sum_radii = np.sum(radii)

    # Validate packing
    valid, message = validate_packing(centers, radii)
    if not valid:
        raise ValueError("Packing validation failed: " + message)

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
