import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, NonlinearConstraint

def run_packing():
    # Define initial grid with large circles
    large_circles = []
    large_radii = []
    large_positions = []
    
    # Create a 3x4 grid of large circles (total of 12 large circles)
    for i in range(3):
        for j in range(4):
            x = j / 3.0 + 0.1
            y = i / 2.0 + 0.1
            large_positions.append([x, y])
            large_radii.append(0.1)
            large_circles.append(0.1)
    
    # Create 6 small circles around each large circle (for total of 72 small circles)
    small_positions = []
    small_radii = []
    
    for x, y in large_positions:
        # Add small circles around the large circle
        for dx, dy in [(-0.5, -0.5), (-0.5, 0.5), (0.5, -0.5), (0.5, 0.5),
                      (-0.7, 0), (0, -0.7), (0, 0.7), (0.7, 0)]:
            nx = x + dx * 0.45
            ny = y + dy * 0.45
            small_positions.append([nx, ny])
            small_radii.append(0.05)
    
    # Check if small circles are within bounds and not overlapping
    def check_bounds(positions):
        for x, y in positions:
            if x < -1e-12 or x > 1 + 1e-12 or y < -1e-12 or y > 1 + 1e-12:
                return False
        return True
    
    # Cull small circles that are out of bounds
    bounded_small_positions = []
    for pos in small_positions:
        if check_bounds([pos]):
            bounded_small_positions.append(pos)
    
    # Construct full positions and radii
    all_positions = large_positions + bounded_small_positions
    all_radii = large_radii + small_radii
    
    # Define optimization variables
    n = len(all_positions)
    x0 = np.array([pos[0] for pos in all_positions] + [rad for rad in all_radii])
    
    # Define bounds for positions and radii
    bounds = []
    for i in range(n):
        if i < len(all_positions):
            # Position bounds: [0,1]^2
            bounds.append((0.0, 1.0))
            bounds.append((0.0, 1.0))
        else:
            # Radius bounds: non-negative
            bounds.append((0.0, 1.0))
    
    # Define constraints: non-overlapping (distance >= r1 + r2 - 1e-12)
    def constraint_func(x):
        pos = x[:2*n].reshape(n, 2)
        rad = x[2*n:]
        dist = np.sqrt(np.sum((pos - pos[:, np.newaxis, :] ** 2), axis=2))
        non_overlapping = dist - (rad[:, np.newaxis] + rad[np.newaxis, :]) + 1e-12
        return np.min(non_overlapping)
    
    # Define constraints: should be >= 0
    def constraint_func_non_negative(x):
        return constraint_func(x) >= 0
    
    # Define bounds for variables
    optimal_bounds = Bounds([0.0]*2*n + [0.0]*n, [1.0]*2*n + [1.0]*n)
    
    # Use SLSQP optimization method
    result = optimize.minimize(
        lambda x: -np.sum(x[2*n:]),
        x0,
        bounds=optimal_bounds,
        constraints=[NonlinearConstraint(lambda x: constraint_func(x), -np.inf, 0.0)]
    )
    
    centers = result.x[:2*n].reshape(n, 2)
    radii = result.x[2*n:]
    
    # Final check for non-overlapping, bounds, and non-negativity
    for i in range(n):
        if radii[i] < 0:
            raise ValueError(f"Circle {i} has negative radius {radii[i]}")
        x, y = centers[i]
        if (x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12
                or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={radii[i]} outside unit square")
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")
    
    return centers, radii, np.sum(radii)
