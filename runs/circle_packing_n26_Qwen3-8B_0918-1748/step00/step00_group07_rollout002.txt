import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import LinearConstraint, NonlinearConstraint
import time

def run_packing():
    # Define the number of circles
    n = 26

    # Define initial positions and radii
    # First, place 16 circles in a hexagonal close packing arrangement
    # Then, add 10 more circles in gaps and at the boundaries

    # Initial centers for hexagonal close packing
    # The hexagonal packing will be 4 rows with 4, 5, 5, 4 circles respectively
    # We'll adjust the spacing to fit within the unit square

    # Initial y positions for rows in hexagonal packing
    y_positions = np.array([0.2, 0.35, 0.5, 0.65, 0.8])
    y_positions = (y_positions - y_positions.min()) / (y_positions.max() - y_positions.min()) * 1.0

    centers = []
    radii = []

    # First 16 circles in hexagonal packing
    for y in y_positions:
        if y < 0.2:
            continue
        row_y = y
        for x in np.linspace(0, 1, 5):
            if np.abs(x) < 0.05:
                continue
            centers.append([x, row_y])
            # Initial radius based on distance to nearby circles
            r = 0.03
            radii.append(r)

    # Second layer
    row_y = y_positions[1]
    x_positions = np.linspace(0.1, 0.9, 5)
    for x in x_positions:
        if np.abs(x - 0.5) < 0.05:
            continue
        centers.append([x, row_y])
        radii.append(0.03)

    # Third layer
    for y in y_positions[2:]:
        x_positions = np.linspace(0.1, 0.9, 5)
        for x in x_positions:
            if np.abs(x - 0.5) < 0.05:
                continue
            centers.append([x, y])
            radii.append(0.03)

    # Add more circles in the remaining space
    # Add 10 more circles in the gaps
    # Try to place them in the middle and spread them
    remaining = 26 - len(centers)
    for _ in range(remaining):
        # Try to place in a grid
        x = np.random.uniform(0.1, 0.9)
        y = np.random.uniform(0.1, 0.9)
        # Check if the proposed position is valid with current circles
        valid = True
        for i in range(len(centers)):
            dist = np.sqrt((x - centers[i][0])**2 + (y - centers[i][1])**2)
            if dist < radii[i] + 0.01:
                valid = False
                break
        if valid:
            centers.append([x, y])
            radii.append(0.02)

    centers = np.array(centers)
    radii = np.array(radii)

    # Now, perform a local optimization to refine the positions and radii
    def objective(x):
        return -np.sum(radii)

    def constraint1(x):
        # Ensure circles are within the square
        for i in range(len(centers)):
            x_i = x[i*2]
            y_i = x[i*2 + 1]
            r_i = radii[i]
            if (x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1):
                return -1
        return 0

    def constraint2(x):
        # Ensure no overlapping between circles
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                x_i = x[i*2]
                y_i = x[i*2 + 1]
                r_i = radii[i]
                x_j = x[j*2]
                y_j = x[j*2 + 1]
                r_j = radii[j]
                dist = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                if dist < r_i + r_j - 1e-8:
                    return -1
        return 0

    x0 = np.zeros(2 * len(centers))
    for i in range(len(centers)):
        x0[2*i] = centers[i, 0]
        x0[2*i + 1] = centers[i, 1]

    bounds = Bounds([0, 0], [1, 1])

    # Set up constraints
    cons = [
        NonlinearConstraint(constraint1, 0, 0),
        NonlinearConstraint(constraint2, 0, 0)
    ]

    # Optimize
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6, options={'maxiter': 200})

    # Extract final centers and radii
    centers_opt = np.zeros((len(centers), 2))
    for i in range(len(centers)):
        centers_opt[i, 0] = result.x[2*i]
        centers_opt[i, 1] = result.x[2*i + 1]
    radii_opt = radii.copy()

    # Final check to ensure all centers and radii are valid
    for i in range(len(centers_opt)):
        x = centers_opt[i, 0]
        y = centers_opt[i, 1]
        r = radii_opt[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            raise ValueError("Circle outside the unit square")

    # Now check for overlaps
    for i in range(len(centers_opt)):
        for j in range(i + 1, len(centers_opt)):
            dx = centers_opt[i, 0] - centers_opt[j, 0]
            dy = centers_opt[i, 1] - centers_opt[j, 1]
            dist = np.hypot(dx, dy)
            if dist < radii_opt[i] + radii_opt[j] - 1e-12:
                raise ValueError("Overlap detected between circles")

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
