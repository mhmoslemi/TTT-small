import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define the layer structure: [1, 2, 5, 9, 7, 2]
    layers = [1, 2, 5, 9, 7, 2]
    total_circles = sum(layers)
    assert total_circles == 26, "Total circles must be 26"
    
    # Define the number of rows for each layer
    rows = [1, 2, 3, 4, 5, 2]
    
    # Define the starting point for each layer
    top_left = np.array([0.1, 0.9])
    bottom_right = np.array([0.9, 0.1])
    
    # Create a grid of points
    centers = []
    for layer_idx, (num_rows, num_circles) in enumerate(zip(rows, layers)):
        if num_circles == 0:
            continue
        row_increment = 1.0 / (num_rows + 1)
        col_increment = 1.0 / (num_rows + 1)
        for row in range(num_rows + 1):
            for col in range(num_rows + 1):
                x = col * col_increment
                y = row * row_increment
                centers.append([x, y])
    
    # Convert to numpy array
    centers = np.array(centers)
    
    # Initial radius guess (based on layer count)
    radii = np.full(len(centers), 0.05)
    
    # Sequential Quadratic Programming (SLSQP) with bounds
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraint1(radii):
        for i in range(len(radii)):
            r = radii[i]
            if r < 0:
                return -1
            if (centers[i, 0] - r < 0 or centers[i, 0] + r > 1 or
                centers[i, 1] - r < 0 or centers[i, 1] + r > 1):
                return -1
        return 0
    
    constraints = []
    for i in range(len(centers)):
        for j in range(i + 1, len(centers)):
            def constraint2(radii):
                r_i = radii[i]
                r_j = radii[j]
                if r_i < 0 or r_j < 0:
                    return -1
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < r_i + r_j - 1e-8:
                    return -1
                return 0
            constraints.append(('in', constraint2))
    
    bounds = [(0, 1) for _ in range(len(radii))]
    
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    # Enforce non-negative radii
    radii = np.maximum(result.x, 0)
    
    # Validate packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
