import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers: grid-based with spacing 0.1 and random perturbation
    num_circles = 26
    centers = np.random.rand(num_circles, 2) * 0.9 + 0.05  # small perturbation to avoid edge issues
    radii = np.full(num_circles, 0.05)

    def objective(radii_vec):
        return -np.sum(radii_vec)  # maximize sum of radii

    def constraint_boundary(radii_vec):
        for i in range(num_circles):
            x, y = centers[i]
            r = radii_vec[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1.0
        return 0.0

    def constraint_overlap(radii_vec):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                r1 = radii_vec[i]
                r2 = radii_vec[j]
                if dist < r1 + r2 - 1e-12:
                    return -1.0
        return 0.0

    # Initial guess for radii
    initial_radii = np.full(num_circles, 0.05)

    # Optimization bounds
    bounds = [(0.0, 1.0) for _ in range(num_circles)]

    # For scipy, the objective needs to be in terms of a vector of variables
    # We can parameterize the problem with radii only, and keep positions fixed

    # Reformulate: optimize radii given the centers
    def objective_func(radii_vec):
        return -np.sum(radii_vec)

    def constraint_boundary_func(radii_vec):
        for i in range(num_circles):
            r = radii_vec[i]
            x, y = centers[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1.0
        return 0.0

    def constraint_overlap_func(radii_vec):
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                r1 = radii_vec[i]
                r2 = radii_vec[j]
                if dist < r1 + r2 - 1e-12:
                    return -1.0
        return 0.0

    # We'll run optimization multiple times with different initial guesses
    # to avoid local minima

    # Try multiple initial configurations
    best_sum = 0.0
    best_centers = centers.copy()
    best_radii = initial_radii.copy()

    for _ in range(5):  # multiple trials to escape local minima
        # Randomize centers slightly to avoid same initial configuration
        centers = np.random.rand(num_circles, 2) * 0.9 + 0.05

        # Initial guess for radii
        initial_radii = np.full(num_circles, 0.05)

        # Optimization problem
        result = minimize(
            objective_func,
            initial_radii,
            bounds=bounds,
            constraints=[
                {'type': 'ineq', 'fun': constraint_boundary_func},
                {'type': 'ineq', 'fun': constraint_overlap_func}
            ],
            method='SLSQP'
        )

        if result.success:
            current_sum = np.sum(result.x)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centers.copy()
                best_radii = result.x.copy()

    # Final packing
    return best_centers, best_radii, best_sum
