import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Initial guess: place circles in a grid-like pattern with some spacing
    # We'll start with 5 rows of 5 circles
    # Start with all radii equal to a base value
    base_radius = 0.05
    
    # Create initial centers
    centers = np.zeros((26, 2))
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col * 0.3 + 0.15
        y = row * 0.3 + 0.15
        centers[i] = [x, y]
    
    radii = np.full(26, base_radius)
    
    # Define objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Negative because we use minimization
    
    # Define constraints
    def constraint_func(x):
        # x is a flattened array of (centers, radii)
        centers = x[:26*2].reshape((26, 2))
        radii = x[26*2:]
        
        # Create a list of all pairwise distances
        dist = pdist(centers)
        dist = squareform(dist)
        # Add boundary constraints
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return np.inf
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.inf
        
        # Add circle-circle constraints
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                distance = np.sqrt(dx**2 + dy**2)
                if distance < radii[i] + radii[j] - 1e-12:
                    return np.inf
        
        return 0
    
    # Flattened initial guess
    x0 = np.concatenate((centers.flatten(), radii))
    
    # Use a constrained optimization method
    cons = {'type': 'ineq', 'fun': constraint_func}
    result = opt.minimize(objective, x0, constraints=cons, method='SLSQP')
    
    # Extract results
    centers = result.x[:26*2].reshape((26, 2))
    radii = result.x[26*2:]
    
    # Ensure all radii are non-negative
    radii[radii < 0] = 0
    
    # Validate the packing
    validation_result, message = validate_packing(centers, radii)
    if not validation_result:
        raise ValueError(message)
    
    total_radius = np.sum(radii)
    return centers, radii, total_radius

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
