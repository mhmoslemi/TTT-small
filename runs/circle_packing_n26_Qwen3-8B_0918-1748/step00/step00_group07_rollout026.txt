import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Define initial guess for centers and radii
    # Using a hybrid cluster approach: 5 clusters with 5 circles each, with some small circles along the edges
    
    # Initial clusters (circles in each cluster are hexagonally packed)
    # A square packing can be used for simplicity
    num_clusters = 5
    circles_per_cluster = 5
    centers = []
    radii = []
    
    # First cluster (bottom-left)
    x0, y0 = 0.15, 0.15
    r0 = 0.07
    centers.append([x0, y0])
    radii.append(r0)
    centers.append([x0 + r0 + r0, y0])
    radii.append(r0)
    centers.append([x0, y0 + r0 + r0])
    radii.append(r0)
    centers.append([x0 + r0 + r0, y0 + r0 + r0])
    radii.append(r0)
    centers.append([x0 + r0, y0 + r0])
    radii.append(r0)
    
    # Second cluster (top-right)
    x0, y0 = 0.65, 0.65
    r0 = 0.07
    centers.append([x0, y0])
    radii.append(r0)
    centers.append([x0 + r0 + r0, y0])
    radii.append(r0)
    centers.append([x0, y0 + r0 + r0])
    radii.append(r0)
    centers.append([x0 + r0 + r0, y0 + r0 + r0])
    radii.append(r0)
    centers.append([x0 + r0, y0 + r0])
    radii.append(r0)
    
    # Third cluster (bottom-right)
    x0, y0 = 0.65, 0.15
    r0 = 0.07
    centers.append([x0, y0])
    radii.append(r0)
    centers.append([x0 - r0 - r0, y0])
    radii.append(r0)
    centers.append([x0, y0 + r0 + r0])
    radii.append(r0)
    centers.append([x0 - r0 - r0, y0 + r0 + r0])
    radii.append(r0)
    centers.append([x0 - r0, y0 + r0])
    radii.append(r0)
    
    # Fourth cluster (top-left)
    x0, y0 = 0.15, 0.65
    r0 = 0.07
    centers.append([x0, y0])
    radii.append(r0)
    centers.append([x0 + r0 + r0, y0])
    radii.append(r0)
    centers.append([x0, y0 - r0 - r0])
    radii.append(r0)
    centers.append([x0 + r0 + r0, y0 - r0 - r0])
    radii.append(r0)
    centers.append([x0 + r0, y0 - r0])
    radii.append(r0)
    
    # Fifth cluster (center cluster)
    x0, y0 = 0.5, 0.5
    r0 = 0.07
    centers.append([x0, y0])
    radii.append(r0)
    centers.append([x0 + r0 + r0, y0])
    radii.append(r0)
    centers.append([x0, y0 + r0 + r0])
    radii.append(r0)
    centers.append([x0 + r0 + r0, y0 + r0 + r0])
    radii.append(r0)
    centers.append([x0 + r0, y0 + r0])
    radii.append(r0)
    
    # Additional small circles along the edges
    # Bottom edge
    centers.append([0.2, 0.05])
    radii.append(0.03)
    centers.append([0.8, 0.05])
    radii.append(0.03)
    # Top edge
    centers.append([0.2, 0.95])
    radii.append(0.03)
    centers.append([0.8, 0.95])
    radii.append(0.03)
    # Left edge
    centers.append([0.05, 0.2])
    radii.append(0.03)
    centers.append([0.05, 0.8])
    radii.append(0.03)
    # Right edge
    centers.append([0.95, 0.2])
    radii.append(0.03)
    centers.append([0.95, 0.8])
    radii.append(0.03)
    
    # Flatten the list
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Optimization parameters
    bounds = []
    for i in range(26):
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))
    
    # Objective function to maximize sum of radii
    def objective(x):
        centers_flat = x[:52].reshape((26, 2))
        radii_flat = x[52:]
        return -np.sum(radii_flat)
    
    # Constraints: spacing between circles
    def constraint_func(x):
        centers_flat = x[:52].reshape((26, 2))
        radii_flat = x[52:]
        violated = False
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    violated = True
        return violated
    
    # Optimization
    result = optimize.minimize(
        objective,
        x0=np.concatenate((centers.flatten(), radii)),
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': lambda x: 0 if not constraint_func(x) else -1}]
    )
    
    # Extract the result
    if not result.success:
        raise Exception("Optimization failed")
    
    centers_opt = result.x[:52].reshape((26, 2))
    radii_opt = result.x[52:]
    
    # Final validation
    def validate_packing(centers, radii):
        n = centers.shape[0]
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"
        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"
    
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise Exception(message)
    
    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
