import numpy as np
import random
from scipy.optimize import minimize
from scipy.spatial import distance
import matplotlib.pyplot as plt

def run_packing():
    # Initial estimate for the maximum sum of radii
    target_sum = 2.636

    # Define the number of circles
    n_circles = 26

    # Start with a grid-based initial guess
    # Place 10 circles in a grid on the bottom half
    # Place 16 circles in a hexagonal pattern on the top half
    # This is a starting point and will be refined by the optimization

    # Grid-based arrangement on the bottom half
    grid_rows = 4
    grid_cols = 5
    grid_spacing = 0.25

    centers = []
    radii = []

    # Bottom half grid
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j * grid_spacing
            y = i * grid_spacing
            if i < grid_rows // 2:
                centers.append([x, y])
                radii.append(0.05)

    # Top half hexagonal packing
    for i in range(grid_rows):
        for j in range(grid_cols):
            x = j * grid_spacing + (i % 2) * grid_spacing / 2
            y = 1 - i * grid_spacing
            if i < grid_rows // 2:
                centers.append([x, y])
                radii.append(0.05)

    # Initial refinement
    centers = np.array(centers)
    radii = np.array(radii)

    # Define the objective function to maximize the total radius
    def objective(radii, centers):
        return -np.sum(radii)

    # Constraints: circles must not overlap and stay within the square
    def constraints(radii, centers):
        constraints = []
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                d = np.linalg.norm(centers[i] - centers[j])
                constraints.append(d >= radii[i] + radii[j] - 1e-12)
            # Ensure circles are within the unit square
            x, y = centers[i]
            constraints.append(x - radii[i] >= -1e-12)
            constraints.append(x + radii[i] <= 1 + 1e-12)
            constraints.append(y - radii[i] >= -1e-12)
            constraints.append(y + radii[i] <= 1 + 1e-12)
        return constraints

    # Optimize the radii
    result = minimize(
        objective,
        radii,
        args=(centers,),
        method='SLSQP',
        constraints=[{'type': 'ineq', 'fun': lambda r: 1e-12 - r[i] for i in range(n_circles)}]
    )

    optimized_radii = result.x
    optimized_centers = centers.copy()

    # Final validation
    valid, msg = validate_packing(optimized_centers, optimized_radii)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    total_sum = np.sum(optimized_radii)
    print(f"Total sum of radii: {total_sum}")

    return (optimized_centers, optimized_radii, total_sum)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
