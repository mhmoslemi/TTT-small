import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26
    
    # Define the unit square
    square = np.array([[0.0, 0.0], [1.0, 1.0]], dtype=np.float64)
    
    # Define initial positions based on a hybrid hexagonal and quadrant-based packing
    # Divide square into 4 quadrants with edges at x = 0.25, 0.75 and y = 0.25, 0.75
    quadrants = [
        [[0.0, 0.0], [0.75, 0.25]],
        [[0.75, 0.0], [1.0, 0.25]],
        [[0.0, 0.75], [0.25, 1.0]],
        [[0.75, 0.75], [1.0, 1.0]],
    ]
    
    # Initialize centers
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Distribute circles into quadrants in a hexagonal pattern
    def place_hexagonal_circles(quadrant, num_circles):
        # Get quadrant boundaries
        x_min, x_max = quadrant[0][0], quadrant[1][0]
        y_min, y_max = quadrant[0][1], quadrant[1][1]
        width = x_max - x_min
        height = y_max - y_min
        
        # Determine spacing and radius based on quadrant size
        spacing = min(width, height) * 0.85
        radius = spacing / 2
        
        # Hexagonal packing: 2 rows with 3, 2, 1 circles
        if num_circles == 6:
            # Row 1: 3 circles
            for i in range(3):
                centers[0 + i] = [x_min + i * spacing, y_min + radius]
                radii[0 + i] = radius
            # Row 2: 2 circles
            for i in range(2):
                centers[3 + i] = [x_min + (i + 1.5) * spacing, y_min + 1.5 * radius]
                radii[3 + i] = radius
            # Center circle
            centers[5] = [x_min + 1.5 * spacing, y_min + 1.5 * radius]
            radii[5] = radius
        elif num_circles == 5:
            # Center circle
            centers[0] = [x_min + width / 2, y_min + height / 2]
            radii[0] = min(width, height) / 3
            # 4 circles around it
            for i in range(4):
                angle = 2 * np.pi * i / 4
                x = x_min + width / 2 + radii[0] * np.cos(angle)
                y = y_min + height / 2 + radii[0] * np.sin(angle)
                centers[1 + i] = [x, y]
                radii[1 + i] = radii[0]
        else:
            raise ValueError("Unsupported number of circles for quadrant placement")
    
    # Place circles in each quadrant
    quadrant_index = 0
    for quad in quadrants:
        quadrant_index = 0
        if quadrant_index < n:
            if quadrant_index + 6 <= n:
                place_hexagonal_circles(quad, 6)
                quadrant_index += 6
            else:
                place_hexagonal_circles(quad, n - quadrant_index)
                quadrant_index = n
    
    # Define optimization variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n, non_negative=True)
    obj = cp.Maximize(cp.sum(r))
    
    # Constraints
    constraints = []
    for i in range(n):
        # Center must be inside the unit square
        if i < 4:
            constraints.append(x[i, 0] >= 0)
            constraints.append(x[i, 1] >= 0)
            constraints.append(x[i, 0] <= 1)
            constraints.append(x[i, 1] <= 1)
        else:
            # Other circles follow the same constraint
            constraints.append(x[i, 0] >= 0)
            constraints.append(x[i, 1] >= 0)
            constraints.append(x[i, 0] <= 1)
            constraints.append(x[i, 1] <= 1)
    
    # Add constraints for distance between each pair of circles
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.norm(x[i] - x[j])
            constraints.append(dist >= r[i] + r[j])
    
    # Set up and solve the optimization problem
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, eps=1e-5)
    except:
        pass  # Fallback to initial placement if optimization fails
    
    # Extract the solution
    centers = x.value
    radii = r.value
    
    # Validate the packing to ensure no errors and correct behavior
    success, message = validate_packing(centers, radii)
    if not success:
        raise ValueError(f"Invalid packing: {message}")
    
    # Return the packing
    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
