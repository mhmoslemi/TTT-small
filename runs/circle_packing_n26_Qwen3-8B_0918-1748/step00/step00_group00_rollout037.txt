import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Define a starting configuration for 26 circles
    # We use a combination of grid-based and corner allocations
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    total = 0

    # Place 16 circles in a grid near the center
    grid_x = np.linspace(0.1, 0.9, 4)
    grid_y = np.linspace(0.1, 0.9, 4)
    for i in range(4):
        for j in range(4):
            idx = i * 4 + j
            centers[idx] = [grid_x[i], grid_y[j]]
            radii[idx] = 0.05  # Initial guess

    # Place 10 circles at the corners
    corners = [
        (0.1, 0.1),
        (0.9, 0.1),
        (0.1, 0.9),
        (0.9, 0.9),
        (0.2, 0.2),
        (0.8, 0.2),
        (0.2, 0.8),
        (0.8, 0.8),
        (0.15, 0.15),
        (0.85, 0.15)
    ]
    for i, (x, y) in enumerate(corners):
        centers[16 + i] = [x, y]
        radii[16 + i] = 0.05

    # Objective function to maximize total radius sum
    def objective(radii):
        return -np.sum(radii)  # Minimize the negative sum

    # Constraints:
    # 1. Each circle must be within the square
    # 2. Distance between centers must be >= sum of radii

    def constraint_within_bounds(radii):
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1
        return 0

    def constraint_between_centers(radii):
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    # Set up optimization problem
    bounds = Bounds(np.zeros(26), np.inf)
    res = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_within_bounds},
            {'type': 'ineq', 'fun': constraint_between_centers}
        ],
        method='SLSQP'
    )

    return (centers, res.x, -res.fun)
