import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import itertools

def run_packing():
    # Initial grid-based layout for 25 circles
    num_circles = 26
    grid_size = 5
    x = np.linspace(0.1, 0.9, grid_size)
    y = np.linspace(0.1, 0.9, grid_size)
    centers = np.array([np.array([xi, yi]) for yi in y for xi in x])
    centers = np.vstack((centers, np.array([0.5, 0.5])))  # Add one more circle in the center
    
    # Initial guess for radii: maximum possible without overlapping with boundaries
    max_radius_initial = 0.4  # A safe initial estimate
    radii = np.full(num_circles, max_radius_initial)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_overlap(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-12:
                    return -1
        return 0

    def constraint_boundary(radii):
        for i in range(num_circles):
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return -1
        return 0

    bounds = Bounds(0, 10)  # Radii must be non-negative and not exceed a large value

    # Initial optimization with constraints
    result = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP',
        options={'ftol': 1e-10, 'disp': False}
    )

    # Extract result
    radii_result = result.x
    sum_radii = np.sum(radii_result)
    return (centers, radii_result, sum_radii)
