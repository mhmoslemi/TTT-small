import numpy as np
import itertools
import scipy.optimize as optimize
import cvxpy as cp

def run_packing():
    # Initial grid-based packing with spacing
    grid_size = 5
    spacing = 0.2
    centers = []
    radii = []
    
    # First layer: 5x5 grid
    for i in range(grid_size):
        for j in range(grid_size):
            x = i * spacing
            y = j * spacing
            r = spacing / 2 - 1e-5
            centers.append([x, y])
            radii.append(r)
            
    # Second layer: additional circles around the grid
    for i in range(grid_size):
        for j in range(grid_size):
            # Add circles around the grid
            x = i * spacing + spacing / 2
            y = j * spacing + spacing / 2
            r = spacing / 2 - 1e-5
            centers.append([x, y])
            radii.append(r)
            
    # Final layer: fill remaining spots
    for i in range(grid_size):
        for j in range(grid_size):
            x = i * spacing + spacing / 2
            y = j * spacing + spacing / 2
            r = spacing / 2 - 1e-5
            centers.append([x, y])
            radii.append(r)
    
    # We need to reduce the number of circles to 26
    # We'll keep the first 26
    centers = centers[:26]
    radii = radii[:26]
    centers = np.array(centers)
    radii = np.array(radii)
    sum_radii = np.sum(radii)
    
    # Optimization to increase sum_radii by redistributing space
    # We will use a simple gradient-based optimization with constraints
    
    n = len(centers)
    x = cp.Variable((n, 2))
    r = cp.Variable(n)
    
    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
    
    # Constraints: circles within bounds
    for i in range(n):
        x_i, y_i = x[i, 0], x[i, 1]
        constraints = [
            x_i >= r[i],
            x_i <= 1 - r[i],
            y_i >= r[i],
            y_i <= 1 - r[i]
        ]
    
    # Non-overlapping constraint: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0]) ** 2 + (x[i, 1] - x[j, 1]) ** 2)
            constraints.append(dist >= r[i] + r[j] - 1e-8)
    
    # Initial guess based on current centers and radii
    initial_centers = np.vstack([centers, np.zeros((n, 2))])
    initial_radii = np.zeros(n)
    
    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, max_iters=1000)
    
    # Extract the optimized positions and radii
    optimized_centers = x.value
    optimized_radii = r.value
    
    # Filter out any radii that are below 0
    valid_indices = np.where(optimized_radii >= 1e-8)[0]
    optimized_centers = optimized_centers[valid_indices]
    optimized_radii = optimized_radii[valid_indices]
    
    # Ensure we have exactly 26 circles
    if len(optimized_radii) < 26:
        # If not enough, add placeholder circles at the end
        needed = 26 - len(optimized_radii)
        for _ in range(needed):
            optimized_centers = np.vstack([optimized_centers, np.array([0.5, 0.5])])
            optimized_radii = np.append(optimized_radii, 1e-8)
    
    if len(optimized_radii) > 26:
        optimized_radii = optimized_radii[:26]
        optimized_centers = optimized_centers[:26]
    
    return optimized_centers, optimized_radii, np.sum(optimized_radii)
