import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds, LinearConstraint, NonlinearConstraint

def run_packing():
    # Initial grid-based configuration with 5x5 grid and one extra circle
    grid_size = 5
    num_circles = 26
    grid_centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i / (grid_size - 1)
            y = j / (grid_size - 1)
            grid_centers.append([x, y])
    grid_centers = np.array(grid_centers)
    grid_centers = np.vstack((grid_centers, [0.5, 0.5]))  # Add central circle
    num_centers = len(grid_centers)
    
    # Initial radii estimates based on spacing and boundary constraints
    def initial_radii(centers):
        radii = np.zeros_like(centers[:, 0])
        for i in range(len(centers)):
            x, y = centers[i]
            min_dist = np.inf
            for j in range(len(centers)):
                if i == j:
                    continue
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < min_dist:
                    min_dist = dist
            # Ensure circle is within the square
            x_dist = min(x, 1 - x)
            y_dist = min(y, 1 - y)
            min_boundary = min(x_dist, y_dist)
            radii[i] = min(min_dist / 2, min_boundary - 1e-8)
        return radii
    
    initial_radii_values = initial_radii(grid_centers)
    initial_radii_values[25] = 0.1  # Adjust the last one to be a placeholder
    
    # Define variables: positions and radii
    positions = grid_centers.copy()
    radii = initial_radii_values.copy()
    
    # Optimization setup
    bounds = []
    for i in range(num_centers):
        x, y = positions[i]
        # Bounds for x and y positions: [0, 1]
        bounds.append((0, 1))
        bounds.append((0, 1))
        # Bounds for radius: [0, max_radius]
        max_radius = min(min(1 - x, x, 1 - y, y), min(1 - x, x, 1 - y, y))
        bounds.append((0, max_radius))
    
    # Objective function: maximize the sum of radii
    def objective(x):
        flat_x = x.reshape(-1)
        positions = flat_x[:num_centers * 2].reshape((num_centers, 2))
        radii = flat_x[num_centers * 2:]
        return -np.sum(radii)  # Minimize the negative sum
    
    # Nonlinear constraints: distance >= r_i + r_j
    def constraint_func(x):
        flat_x = x.reshape(-1)
        positions = flat_x[:num_centers * 2].reshape((num_centers, 2))
        radii = flat_x[num_centers * 2:]
        constraints = []
        for i in range(num_centers):
            for j in range(i + 1, num_centers):
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                constraints.append(dist - (radii[i] + radii[j]))
        return np.array(constraints)
    
    # Additional constraints: circles within the square
    def boundary_constraints(x):
        flat_x = x.reshape(-1)
        positions = flat_x[:num_centers * 2].reshape((num_centers, 2))
        radii = flat_x[num_centers * 2:]
        for i in range(num_centers):
            x, y = positions[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12:
                return np.inf
            if y - r < -1e-12 or y + r > 1 + 1e-12:
                return np.inf
        return 0
    
    # Initial guess for positions and radii
    x0 = np.zeros(len(bounds))
    for i in range(num_centers):
        x0[2 * i] = positions[i, 0]
        x0[2 * i + 1] = positions[i, 1]
        x0[2 * num_centers + i] = radii[i]
    
    # Optimization options
    options = {
        'maxiter': 500,
        'ftol': 1e-8,
        'eps': 1e-8
    }
    
    # Solve the optimization problem
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=[
            NonlinearConstraint(constraint_func, np.zeros(len(constraint_func)), np.inf),
            NonlinearConstraint(boundary_constraints, 0, 0)
        ],
        method='SLSQP',
        options=options
    )
    
    # Extract final positions and radii
    final_x = result.x.reshape(-1)
    positions = final_x[:num_centers * 2].reshape((num_centers, 2))
    radii = final_x[num_centers * 2:]
    
    # Validate the packing
    _, message = validate_packing(positions, radii)
    if message != "ok":
        raise ValueError(f"Validation failed: {message}")
    
    return positions, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
