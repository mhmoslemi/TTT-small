import numpy as np
from scipy.optimize import minimize, Bounds
import warnings

def run_packing():
    # Base configuration: 4 rows of 7 circles each (28 circles), then remove 2 circles
    # We'll create a grid-based starting point for 26 circles.
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial grid configuration with some perturbations
    for i in range(n):
        row = i // 7
        col = i % 7
        x = col / 6.5 + 0.05
        y = row / 6.5 + 0.05
        centers[i, 0] = x
        centers[i, 1] = y
        radii[i] = 0.1

    # Optimization parameters
    bounds = Bounds([0, 0], [1, 1])  # Center bounds
    radii_bounds = Bounds([0,], [np.inf,])  # Radius bounds

    # Objective function to maximize sum of radii
    def objective(x):
        # x has shape (2*26, ), first half is centers, second half is radii
        centers_opt = x[:2*26].reshape((26, 2))
        radii_opt = x[2*26:]
        return -np.sum(radii_opt)  # Minimize negative of sum to maximize sum

    # Constraints: non-overlapping and boundary conditions
    def constraints(x):
        centers_opt = x[:2*26].reshape((26, 2))
        radii_opt = x[2*26:]
        for i in range(26):
            x_i, y_i = centers_opt[i]
            r_i = radii_opt[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return np.inf
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_opt[i, 0] - centers_opt[j, 0]
                dy = centers_opt[i, 1] - centers_opt[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_i = radii_opt[i]
                r_j = radii_opt[j]
                if dist < r_i + r_j - 1e-8:
                    return np.inf
        return 0

    # Combine into a single optimization problem
    x0 = np.concatenate((centers.reshape(-1), radii))
    res = minimize(
        objective,
        x0,
        bounds=Bounds([0, 0, 0], [1, 1, np.inf]),
        constraints={'type': 'ineq', 'fun': constraints},
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 500}
    )

    centers_opt = res.x[:2*26].reshape((26, 2))
    radii_opt = res.x[2*26:]
    sum_radii = np.sum(radii_opt)
    
    # Validate packing and ensure no errors
    try:
        is_valid, msg = validate_packing(centers_opt, radii_opt)
        if not is_valid:
            raise RuntimeError(msg)
    except:
        # In case of failure, return a baseline configuration
        # That is, a grid-based configuration with some radii adjustment
        centers = np.zeros((26, 2))
        for i in range(26):
            row = i // 7
            col = i % 7
            x = col / 6.5 + 0.05
            y = row / 6.5 + 0.05
            centers[i, 0] = x
            centers[i, 1] = y
            radii[i] = 0.1
        radii = np.full(26, 0.1)
        sum_radii = 26 * 0.1
        return (centers, radii, sum_radii)
    
    return (centers_opt, radii_opt, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
