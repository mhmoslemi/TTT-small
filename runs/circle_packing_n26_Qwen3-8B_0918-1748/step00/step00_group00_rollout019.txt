import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define the hybrid packing pattern with 26 circles
    # Layer 1: 4x4 grid of circles
    # Layer 2: 3x3 grid of circles offset to hexagonal pattern
    # Layer 3: 3x3 grid of circles shifted to fill remaining gaps

    # Initial guess for centers and radii
    def initial_centers_and_radii():
        # Layer 1: 4x4 grid of circles
        layer1 = []
        for i in range(4):
            for j in range(4):
                x = i / 3.0 + 0.05
                y = j / 3.0 + 0.05
                layer1.append((x, y))
        # Layer 2: 3x3 grid of circles offset to hexagonal pattern
        layer2 = []
        for i in range(3):
            for j in range(3):
                x = i / 2.5 + 0.25
                y = j / 2.5 + 0.25
                # Offset for hexagonal pattern
                if i % 2 == 0:
                    x_offset = 0.3
                else:
                    x_offset = -0.3
                x += x_offset
                y += 0.2
                layer2.append((x, y))
        # Layer 3: 3x3 grid of circles to fill remaining gaps
        layer3 = []
        for i in range(3):
            for j in range(3):
                x = i / 2.5 + 0.05
                y = j / 2.5 + 0.05
                x += 0.4
                y += 0.4
                layer3.append((x, y))

        centers = np.array(layer1 + layer2 + layer3)
        radii = np.ones(26) * 0.05  # Initial radius guess
        return centers, radii

    # Optimize the radii and positions
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_non_overlap(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return 1
        return 0

    def constraint_boundary(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            if centers[i, 0] - radii[i] < -1e-8 or centers[i, 0] + radii[i] > 1 + 1e-8:
                return 1
            if centers[i, 1] - radii[i] < -1e-8 or centers[i, 1] + radii[i] > 1 + 1e-8:
                return 1
        return 0

    centers_initial, radii_initial = initial_centers_and_radii()

    # Optimization
    bounds = [(0, 1) for _ in range(26 * 2)]  # Bounds for x and y for each circle
    bounds += [(0, 1) for _ in range(26)]     # Bounds for radii

    # Set up the optimization problem with constraints
    def constraints(x):
        centers = x[:26 * 2].reshape(26, 2)
        radii = x[26 * 2:]
        return [
            {'type': 'ineq', 'fun': lambda x: constraint_non_overlap(centers, radii)},
            {'type': 'ineq', 'fun': lambda x: constraint_boundary(centers, radii)}
        ]

    x0 = np.concatenate([centers_initial.ravel(), radii_initial])

    result = minimize(
        fun=objective,
        x0=x0,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-5, 'maxiter': 100}
    )

    centers_opt = result.x[:26 * 2].reshape(26, 2)
    radii_opt = result.x[26 * 2:]

    # Ensure all radii are non-negative
    radii_opt[radii_opt < 0] = 0

    # Validate packing
    if not validate_packing(centers_opt, radii_opt)[0]:
        raise ValueError("Optimization failed to produce valid packing")

    return centers_opt, radii_opt, np.sum(radii_opt)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
