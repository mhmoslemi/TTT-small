import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial guess for centers and radii
    # Start with a hexagonal packing with some adjustments for the unit square
    # 5 rows of circles with varying numbers of circles
    
    # Define the initial centers based on a hexagonal packing
    # Each circle has an initial radius based on the spacing between rows
    
    # Adjusted hexagonal packing with 5 rows for 26 circles
    # We will use a hybrid approach with 5 rows: 5, 5, 5, 7, 4
    
    # First, compute a grid of points with hexagonal spacing
    # We'll adjust the spacing to ensure the total area fits in the unit square
    # Let's choose a base spacing of 0.2 units between centers
    
    # Define the initial centers
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # First row (top row)
    row_offset = 0.5
    for i in range(5):
        x = i * 0.2
        y = row_offset
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Second row (offset)
    row_offset -= np.sqrt(3) * 0.1
    for i in range(5, 10):
        x = i - 5
        x = x * 0.2
        y = row_offset
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Third row
    row_offset += np.sqrt(3) * 0.1
    for i in range(10, 15):
        x = i - 10
        x = x * 0.2
        y = row_offset
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Fourth row (offset)
    row_offset -= np.sqrt(3) * 0.1
    for i in range(15, 21):
        x = i - 15
        x = x * 0.2
        y = row_offset
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # Fifth row
    row_offset += np.sqrt(3) * 0.1
    for i in range(21, 26):
        x = i - 21
        x = x * 0.2
        y = row_offset
        centers[i] = [x, y]
        radii[i] = 0.1
    
    # These initial centers are in a hexagonal grid but need to be adjusted
    # to fit within [0,1]^2 and optimize the radii for maximum total sum
    
    # Define the objective function
    def objective(radii):
        return -np.sum(radii)  # We maximize by minimizing the negative
    
    # Constraints: non-overlap and boundary conditions
    def constraint_overlap(radii):
        # Non-overlap constraints
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1.0
        return 0.0
    
    def constraint_boundary(radii):
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if not (0 <= x - r and x + r <= 1 and 0 <= y - r and y + r <= 1):
                return -1.0
        return 0.0
    
    # Define bounds for radii
    bounds = [(0.0, 1.0) for _ in range(26)]
    
    # Create constraints for the optimization problem
    constraints = [
        {'type': 'ineq', 'fun': constraint_overlap},
        {'type': 'ineq', 'fun': constraint_boundary}
    ]
    
    # Initial guess for radii
    initial_guess = radii
    
    # Run the optimization
    result = minimize(objective, initial_guess, method='SLSQP', bounds=bounds, constraints=constraints)
    
    # Update centers and radii based on the result
    radii = result.x
    centers = centers
    
    # Return the centers, radii, and sum of radii
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
