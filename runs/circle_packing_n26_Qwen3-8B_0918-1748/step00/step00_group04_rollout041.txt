import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Initial guess: place circles in a grid-like pattern with varying radii
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Initial grid layout (7 rows, 4 columns) with some side circles
    for i in range(7):
        for j in range(4):
            idx = i * 4 + j
            if idx < num_circles:
                x = j / 3.0 + 0.05
                y = i / 6.0 + 0.05
                centers[idx] = [x, y]
    
    # Add corner circles
    corners = [(0.05, 0.05), (0.95, 0.05), (0.05, 0.95), (0.95, 0.95)]
    for k in range(4):
        idx = 28 + k
        if idx < num_circles:
            centers[idx] = corners[k]
    
    # Initial radii using a density approach
    avg_radius = 0.06
    radii = np.full(num_circles, avg_radius)
    
    # Bounds for optimization
    bounds = Bounds([0, 0], [1, 1])
    
    # Objective function: maximize sum of radii
    def objective(x):
        x_flat = x.reshape(-1)
        centers_new = x_flat[:2*num_circles].reshape(num_circles, 2)
        radii_new = x_flat[2*num_circles:]
        return -np.sum(radii_new)  # Minimize negative sum to maximize
    
    # Constraints: boundary conditions
    def constraint_boundary(x):
        x_flat = x.reshape(-1)
        centers_new = x_flat[:2*num_circles].reshape(num_circles, 2)
        radii_new = x_flat[2*num_circles:]
        for i in range(num_circles):
            x, y = centers_new[i]
            r = radii_new[i]
            if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
                return 1
        return 0
    
    # Non-overlapping constraints
    def constraint_overlap(x):
        x_flat = x.reshape(-1)
        centers_new = x_flat[:2*num_circles].reshape(num_circles, 2)
        radii_new = x_flat[2*num_circles:]
        for i in range(num_circles):
            for j in range(i+1, num_circles):
                dx = centers_new[i, 0] - centers_new[j, 0]
                dy = centers_new[i, 1] - centers_new[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_new[i] + radii_new[j] - 1e-8:
                    return 1
        return 0
    
    # Optimization
    x0 = np.concatenate((centers.flatten(), radii))
    cons = [
        {'type': 'ineq', 'fun': lambda x: -constraint_boundary(x)},
        {'type': 'ineq', 'fun': lambda x: -constraint_overlap(x)}
    ]
    
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 1000, 'ftol': 1e-8, 'eps': 1e-8}
    )
    
    centers_opt = result.x[:2*num_circles].reshape(num_circles, 2)
    radii_opt = result.x[2*num_circles:]
    
    # Final validation and return
    centers_final = centers_opt
    radii_final = radii_opt
    sum_radii = np.sum(radii_final)
    
    return centers_final, radii_final, sum_radii
