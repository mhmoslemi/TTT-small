import numpy as np
import numpy.random as npr
from scipy.optimize import minimize

def run_packing():
    np.random.seed(42)
    n = 26
    
    # Initial guess: regular grid with some variation
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # First, place 3 circles in a triangle near the bottom left
    radii[0] = 0.1
    centers[0] = [0.1, 0.1]
    radii[1] = 0.1
    centers[1] = [0.1, 0.8]
    radii[2] = 0.1
    centers[2] = [0.8, 0.1]
    
    # Now create 23 more circles in regular grid
    remaining = n - 3
    for i in range(remaining):
        r = 0.1 + np.random.uniform(-0.05, 0.05)
        x = 0.1 + i / 2
        y = 0.1 + (i // 2) / 2
        if i % 3 == 0:
            x += np.random.uniform(-0.05, 0.05)
        if i % 2 == 0:
            y += np.random.uniform(-0.05, 0.05)
        centers[3 + i] = [x, y]
        radii[3 + i] = r
    
    # Create a small optimization function
    def objective(radii_flat):
        return -np.sum(radii_flat)
    
    def constraint_overlap(radii_flat):
        centers_flat = np.reshape(centers, (-1))
        r = np.reshape(radii_flat, (-1))
        for i in range(n):
            for j in range(i+1, n):
                dist = np.sqrt( (centers_flat[2*i] - centers_flat[2*j])**2 + 
                              (centers_flat[2*i+1] - centers_flat[2*j+1])**2 )
                if dist < r[i] + r[j] - 1e-12:
                    return -1
        return 0
    
    def constraint_bounds(radii_flat):
        r = np.reshape(radii_flat, (-1))
        for i in range(n):
            if r[i] < 0:
                return -1
        return 0
    
    def constraint_square(radii_flat):
        r = np.reshape(radii_flat, (-1))
        for i in range(n):
            x, y = centers[i]
            if x - r[i] < -1e-12 or x + r[i] > 1 + 1e-12:
                return -1
            if y - r[i] < -1e-12 or y + r[i] > 1 + 1e-12:
                return -1
        return 0
    
    # Optimization parameters
    x0 = radii.copy()
    
    # Set up bounds
    bounds = [(0, 1) for _ in range(n)]
    
    # Constraints
    constraints = [
        {'type': 'ineq', 'fun': constraint_overlap, 'args': (radii,)},
        {'type': 'ineq', 'fun': constraint_bounds, 'args': (radii,)},
        {'type': 'ineq', 'fun': constraint_square, 'args': (radii,)}
    ]
    
    # Run optimization
    result = minimize(objective, x0, bounds=bounds, constraints=constraints, method='SLSQP')
    
    # Extract solution
    radii_opt = result.x
    centers_opt = centers.copy()
    
    # Post-processing: adjust slightly if needed
    for i in range(n):
        x, y = centers_opt[i]
        r = radii_opt[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12:
            radii_opt[i] = min(r, x - 1e-12, 1 - x - 1e-12)
        if y - r < -1e-12 or y + r > 1 + 1e-12:
            radii_opt[i] = min(r, y - 1e-12, 1 - y - 1e-12)
    
    return centers_opt, radii_opt, np.sum(radii_opt)
