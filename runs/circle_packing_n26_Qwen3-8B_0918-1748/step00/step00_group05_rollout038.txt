import numpy as np
import scipy.optimize as optimize

def run_packing():
    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        return np.sum(radii)

    def bounds(radii):
        return [(0, 1)] * len(radii)

    def solve_packing():
        N = 26
        centers = np.random.rand(N, 2)
        radii = np.random.rand(N) * 0.1 + 0.05

        def objective_function(x):
            return -np.sum(x)

        def inequality_constraints(x):
            dists = np.sqrt(np.sum((centers[i] - centers[j])**2) for i in range(N) for j in range(i+1, N))
            return [dists[i] - x[i] - x[j] for i in range(N) for j in range(i+1, N)]

        def bounds_constraints(x):
            return [x[i] <= 1 - np.min(centers[i, 0] - x[i], centers[i, 1] - x[i], 1 - centers[i, 0] - x[i], 1 - centers[i, 1] - x[i]) for i in range(N)]

        res = optimize.minimize(
            objective_function,
            radii,
            bounds=[(0, 1)] * N,
            constraints=[
                {'type': 'ineq', 'fun': lambda x: np.array([1 - np.max(centers[i, 0] + x[i]) for i in range(N)])},
                {'type': 'ineq', 'fun': lambda x: np.array([1 - np.max(centers[i, 1] + x[i]) for i in range(N)])},
                {'type': 'ineq', 'fun': lambda x: np.array([np.min(centers[i, 0] - x[i]) for i in range(N)])},
                {'type': 'ineq', 'fun': lambda x: np.array([np.min(centers[i, 1] - x[i]) for i in range(N)])},
                {'type': 'ineq', 'fun': lambda x: np.array([np.array(inequality_constraints(x))])},
                {'type': 'ineq', 'fun': lambda x: np.array([np.array(bounds_constraints(x))])}
            ],
            method='SLSQP'
        )
        return res.x, centers

    radii, centers = solve_packing()
    return centers, radii, np.sum(radii)
