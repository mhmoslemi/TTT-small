import numpy as np
import math
from scipy.optimize import minimize

def run_packing():
    # Base hexagonal packing with boundary adjustments
    def hexagonal_packing(n):
        centers = []
        radii = []
        for i in range(n):
            row = i // 4
            col = i % 4
            x = 0.1 + col * 0.25
            y = 0.1 + row * 0.25
            if row % 2 == 1:
                x += 0.125
            r = 0.1
            # Adjust to fit in square
            x = np.clip(x, 0.0, 1.0)
            y = np.clip(y, 0.0, 1.0)
            dist_to_left = x - r
            dist_to_right = 1.0 - x - r
            dist_to_bottom = y - r
            dist_to_top = 1.0 - y - r
            min_dist = min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top)
            r = max(0.0, r - 0.1 * min_dist)
            centers.append([x, y])
            radii.append(r)
        return np.array(centers), np.array(radii)

    # Initial packing
    centers, radii = hexagonal_packing(26)
    total_radii = np.sum(radii)

    # Define a function to evaluate and optimize
    def objective(x):
        # x is a flattened array: [x1, y1, r1, x2, y2, r2, ...]
        # Reconstruct centers and radii
        n = 26
        centers = np.reshape(x[:2 * n], (n, 2))
        radii = x[2 * n:]
        # Check if all radii are non-negative
        if np.any(radii < 0):
            return -np.inf
        # Check if all circles are inside the square
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if not (0 - r <= x <= 1 + r and 0 - r <= y <= 1 + r):
                return -np.inf
        # Check for overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        # Compute total radii
        return np.sum(radii)

    # Initial guess
    x0 = np.concatenate((centers.flatten(), radii))
    # Bounds for centers (x and y between 0 and 1), radii >= 0
    bounds = []
    for _ in range(26):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])
    for _ in range(26):
        bounds.append((0.0, 1.0))
    # Optimization
    result = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', options={'maxiter': 500})
    if not result.success:
        raise RuntimeError("Optimization failed")
    x = result.x
    centers = np.reshape(x[:2 * 26], (26, 2))
    radii = x[2 * 26:]
    total_radii = np.sum(radii)
    return (centers, radii, total_radii)
