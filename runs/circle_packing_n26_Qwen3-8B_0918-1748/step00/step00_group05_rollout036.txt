import numpy as np
import random
from scipy.optimize import minimize

def run_packing():
    # Initial layout: 5x5 grid in the middle with 10 small circles at corners/edges
    centers = []
    radii = []

    # Create 5x5 grid in the center
    for i in range(5):
        for j in range(5):
            x = 0.2 + i * 0.25
            y = 0.2 + j * 0.25
            centers.append([x, y])
            radii.append(0.1)

    # Add 10 small circles at strategic positions near edges and corners
    corner_offsets = [(0.1, 0.1), (0.1, 0.9), (0.9, 0.9), (0.9, 0.1)]
    edge_offsets = [(0.1, 0.5), (0.5, 0.1), (0.5, 0.9), (0.9, 0.5)]

    for offset in corner_offsets:
        x, y = offset
        centers.append([x, y])
        radii.append(0.05)

    for offset in edge_offsets:
        x, y = offset
        centers.append([x, y])
        radii.append(0.05)

    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)

    # Define the objective function to maximize sum of radii
    def objective(x):
        return -np.sum(x)  # Negative for minimization

    # Define constraints: circles must not overlap and stay within unit square
    def constraint_overlap(i, j):
        def cons(x):
            # Extract positions and radii
            pos_i = x[2*i:2*i+2]
            pos_j = x[2*j:2*j+2]
            r_i = radii[i]
            r_j = radii[j]
            dist = np.sqrt((pos_i[0] - pos_j[0])**2 + (pos_i[1] - pos_j[1])**2)
            return dist - r_i - r_j
        return {'type': 'ineq', 'fun': cons}

    # Define constraints for boundaries: circle must be fully inside square
    def constraint_boundary(i):
        def cons(x):
            pos = x[2*i:2*i+2]
            r = radii[i]
            return (pos[0] - r) - 1e-12  # Left boundary
        return {'type': 'ineq', 'fun': cons}

    def constraint_boundary_right(i):
        def cons(x):
            pos = x[2*i:2*i+2]
            r = radii[i]
            return (1.0 - pos[0] - r) - 1e-12  # Right boundary
        return {'type': 'ineq', 'fun': cons}

    def constraint_boundary_top(i):
        def cons(x):
            pos = x[2*i:2*i+2]
            r = radii[i]
            return (1.0 - pos[1] - r) - 1e-12  # Top boundary
        return {'type': 'ineq', 'fun': cons}

    def constraint_boundary_bottom(i):
        def cons(x):
            pos = x[2*i:2*i+2]
            r = radii[i]
            return (pos[1] - r) - 1e-12  # Bottom boundary
        return {'type': 'ineq', 'fun': cons}

    # Set up the optimization problem
    n = len(centers)
    bounds = [(0, 1) for _ in range(2 * n)]

    # Create constraints
    cons = []
    for i in range(n):
        cons.append(constraint_boundary(i))
        cons.append(constraint_boundary_right(i))
        cons.append(constraint_boundary_top(i))
        cons.append(constraint_boundary_bottom(i))

    for i in range(n):
        for j in range(i+1, n):
            cons.append(constraint_overlap(i, j))

    # Initial guess: positions of circles
    x0 = np.concatenate([centers[i] for i in range(n)])

    # Optimization
    result = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'maxiter': 1000})

    # Update centers and radii with optimized positions
    optimized_centers = result.x.reshape((n, 2))
    optimized_radii = np.array(radii)

    # Recheck constraints
    valid, msg = validate_packing(optimized_centers, optimized_radii)
    if not valid:
        # Fallback to original configuration if validation fails
        return (centers, radii, np.sum(radii))

    # Return optimized result
    return (optimized_centers, optimized_radii, np.sum(optimized_radii))

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
