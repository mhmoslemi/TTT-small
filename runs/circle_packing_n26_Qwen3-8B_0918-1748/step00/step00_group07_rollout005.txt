from scipy.optimize import minimize
from scipy.optimize import Bounds

            x_i, y_i = centers[i]
            r_i = radii[i]

                constraints.append(y_i + r_i - 1 + 1e-12)

                constraints.append(0)

    def non_overlap_constraints(x):
        centers_flat = x[:num_circles * 2]

                dx = centers[i, 0] - centers[j, 0]

                dist = np.sqrt(dx * dx + dy * dy)
                overlap = dist - (radii[i] + radii[j])
                if overlap < -1e-12:
                    constraints.append(overlap + 1e-12)
                else:
                    constraints.append(0)
        return np.array(constraints)

    bounds = []
    for _ in range(num_circles):
        bounds.extend([(0, 1), (0, 1)])  # x and y coordinates

    x0 = np.concatenate([centers.reshape(-1), radii])

    result = minimize(

    return best_centers, best_radii, best_sum
