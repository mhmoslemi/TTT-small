import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance

def run_packing():
    # Define grid for 25 small circles in a hexagonal pattern
    layers = [6, 12, 6]  # layer sizes
    positions = []
    for layer in range(3):
        if layer == 0:
            offset = 0.1
        elif layer == 1:
            offset = 0.25
        else:
            offset = 0.4
        for i in range(layers[layer]):
            angle = 2 * np.pi * i / layers[layer]
            x = 0.5 + offset * np.cos(angle)
            y = 0.5 + offset * np.sin(angle)
            positions.append([x, y])
    positions = np.array(positions)
    
    # Add large central circle at (0.5, 0.5)
    positions = np.vstack([positions, [0.5, 0.5]])
    
    # Define the optimization problem
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraints(radii):
        # Constraints: circle-to-circle distance >= sum of radii
        # Also check if circles are within the square [0,1]^2
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((positions[i] - positions[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
            # Check if circles are within the square
            if (positions[i, 0] - radii[i] < -1e-8 or positions[i, 0] + radii[i] > 1 + 1e-8 or
                positions[i, 1] - radii[i] < -1e-8 or positions[i, 1] + radii[i] > 1 + 1e-8):
                return False
        return True
    
    # Initial guess for radii (same for all except the last one which is smaller)
    initial_radii = np.full(26, 0.05)
    initial_radii[-1] = 0.1  # Last one is the large central circle
    
    # Optimization
    result = minimize(
        objective,
        initial_radii,
        bounds=[(0, 1)] * 26,
        constraints=[{'type': 'ineq', 'fun': constraints}],
        method='SLSQP',
        options={'ftol': 1e-8, 'disp': False}
    )
    
    centers = positions
    radii = result.x
    sum_radii = np.sum(radii)
    
    return (centers, radii, sum_radii)
