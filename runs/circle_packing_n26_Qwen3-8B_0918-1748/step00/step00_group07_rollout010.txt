import numpy as np
from scipy.optimize import minimize
from itertools import combinations

def run_packing():
    # Hexagonal packing with 26 circles as initial guess
    # Start with a 4x4 grid with spacing and adjust radii for hexagonal packing
    # Center of square is (0.5, 0.5)
    # Create initial estimate of centers and radii using hexagonal grid
    # The radii will be adjusted during optimization

    # Define the initial centers in a hexagonal grid (as a 5x5 grid with 3 layers)
    # Layer 0: 1 circle (center)
    # Layer 1: 6 circles
    # Layer 2: 12 circles
    # Layer 3: 12 circles
    # However, for 26 circles, we'll have to choose appropriate layers

    # Initial configuration: the first 41 positions (from a 7x7 grid) with a few layers
    # For the purpose of this notebook, we'll manually create an initial configuration
    # of 26 points in a hexagonal-like cluster

    # Create an initial guess of 26 points with symmetric positions
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Hexagonal grid layout with some center points
    # This is a manually designed initial configuration
    # Arrange circles in a cluster to form a hexagonal grid

    # 1st layer - center circle
    centers[0] = [0.5, 0.5]
    radii[0] = 0.1

    # 2nd layer - 6 circles around the center
    for i in range(1, 7):
        angle = 2 * np.pi * i / 6
        r = 0.3
        centers[i] = [0.5 + r * np.cos(angle), 0.5 + r * np.sin(angle)]
        radii[i] = 0.1

    # 3rd layer - 12 circles
    for i in range(7, 19):
        layer = 2
        angle = 2 * np.pi * (i - 7) / 12
        r = 0.4
        centers[i] = [0.5 + r * np.cos(angle), 0.5 + r * np.sin(angle)]
        radii[i] = 0.1

    # 4th layer - 12 circles
    for i in range(19, 26):
        layer = 3
        angle = 2 * np.pi * (i - 19) / 12
        r = 0.5
        centers[i] = [0.5 + r * np.cos(angle), 0.5 + r * np.sin(angle)]
        radii[i] = 0.1

    # Define objective function
    def objective(x):
        # x contains all variables: first 52 values are centers, then 26 radii
        centers_flat = x[:52].reshape(26, 2)
        radii_flat = x[52:]
        return -np.sum(radii_flat)

    # Define constraints
    def constraint1(x):
        centers_flat = x[:52].reshape(26, 2)
        radii_flat = x[52:]
        # Check if circles are within the square
        for i in range(26):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1.0
        return 0.0

    def constraint2(x):
        centers_flat = x[:52].reshape(26, 2)
        radii_flat = x[52:]
        # Check circular overlaps
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers_flat[i] - centers_flat[j])**2))
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -1.0
        return 0.0

    # Initial guess
    x0 = np.concatenate((centers.flatten(), radii))

    # Wrap the constraints
    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]

    # Run optimization
    result = minimize(objective, x0, method='SLSQP', constraints=cons, tol=1e-8, options={'maxiter': 500})

    centers_opt = result.x[:52].reshape(26, 2)
    radii_opt = result.x[52:]

    if not np.isfinite(result.fun):
        # Fallback in case optimization fails
        return np.zeros((26, 2)), np.zeros(26), 0.0

    return centers_opt, radii_opt, -result.fun
