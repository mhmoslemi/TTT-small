import numpy as np
import random
import matplotlib.pyplot as plt
from scipy.optimize import minimize

def run_packing():
    # Initial centers and radii
    centers = np.random.rand(26, 2)
    radii = np.full(26, 0.05)

    # Initial perturbation to avoid symmetry
    for i in range(26):
        centers[i] += np.random.uniform(-0.05, 0.05, 2)

    # Optimization parameters
    max_iter = 1000
    learning_rate = 0.1
    tolerance = 1e-6
    best_sum = 0
    best_centers = centers.copy()
    best_radii = radii.copy()

    for _ in range(max_iter):
        # Compute pairwise distances
        dist_matrix = np.zeros((26, 26))
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_matrix[i, j] = np.sqrt(dx * dx + dy * dy)
                dist_matrix[j, i] = dist_matrix[i, j]

        # Compute force magnitudes
        force = np.zeros((26, 2))
        for i in range(26):
            for j in range(i + 1, 26):
                diff = centers[i] - centers[j]
                dist = dist_matrix[i, j]
                if dist < 1e-12:
                    continue
                force[i] += (radii[j] / dist) * diff
                force[j] -= (radii[i] / dist) * (-diff)

        # Update centers
        for i in range(26):
            # Boundary constraints
            for j in range(2):
                if centers[i, j] < 0:
                    centers[i, j] = 0
                if centers[i, j] > 1:
                    centers[i, j] = 1
            # Update center
            centers[i] -= learning_rate * force[i]

        # Update radii
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - tolerance:
                    # Adjust radii to prevent overlap
                    if dist < radii[i] + radii[j] - tolerance:
                        if dist < radii[i] - tolerance:
                            radii[i] = max(0, radii[i] - (radii[i] - dist))
                        else:
                            radii[i] = max(0, radii[i] - (radii[i] + radii[j] - dist) / 2)
                            radii[j] = max(0, radii[j] - (radii[i] + radii[j] - dist) / 2)

        # Compute sum of radii
        sum_radii = np.sum(radii)
        if sum_radii > best_sum:
            best_sum = sum_radii
            best_centers = centers.copy()
            best_radii = radii.copy()

    # Final validation
    if validate_packing(best_centers, best_radii)[0]:
        return best_centers, best_radii, best_sum
    else:
        raise ValueError("Validation failed for the final configuration.")

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
