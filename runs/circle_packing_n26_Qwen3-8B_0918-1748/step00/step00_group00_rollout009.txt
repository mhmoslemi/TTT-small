import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for centers: 1 row near top, 1 row near bottom, and 4 rows in the middle
    # These are approximate and can be refined with optimization

    # Initial centers
    centers = np.zeros((26, 2))
    # Corner circles (4 corners)
    centers[0] = [0.1, 0.1]
    centers[1] = [0.1, 0.9]
    centers[2] = [0.9, 0.9]
    centers[3] = [0.9, 0.1]
    # Edge circles (4 edges)
    centers[4] = [0.1, 0.5]
    centers[5] = [0.5, 0.9]
    centers[6] = [0.9, 0.5]
    centers[7] = [0.5, 0.1]
    # Central circles in grid (ideal for hexagonal packing, but adjusted for unit square)
    for i in range(4):
        for j in range(4):
            if i == 0 and j == 0:
                continue  # Skip the first one to avoid overlapping with edge circles
            row = i * 0.25
            col = j * 0.25
            centers[8 + i * 4 + j] = [col + 0.2, row + 0.2]

    # Make sure the number of centers is 26
    assert len(centers) == 26, "Incorrect number of centers!"

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_func(radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(len(radii)):
            for j in range(i + 1, len(radii)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-6:
                    return 1.0
        return 0.0

    # Bounds for radii: all radii must be non-negative
    bounds = [(0, 1) for _ in range(26)]

    # Optimization
    result = opt.minimize(
        objective,
        x0=np.ones(26),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_func}
        ],
        method='SLSQP'
    )

    radii = result.x
    # Validate packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
