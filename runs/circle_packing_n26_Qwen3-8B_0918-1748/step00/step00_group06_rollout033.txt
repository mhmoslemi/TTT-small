import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Define the number of circles
    n = 26

    # Initial packed configuration: 5 rows of 5 circles, with one extra circle at the bottom
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Initial layout: 5 rows of 5 circles (first 25 circles), and one extra at bottom
    row_height = 1 / (np.sqrt(3))  # Hexagonal packing height
    col_width = 1 / 5  # Equal spacing

    for i in range(5):
        for j in range(5):
            x = j * col_width
            y = i * row_height
            centers[i * 5 + j] = [x, y]
            radii[i * 5 + j] = 0.1

    # Add one extra circle at the bottom center
    centers[25] = [0.5, 1 - 1e-12]
    radii[25] = 0.15

    # Define bounds for positions and radii
    bounds = [(0, 1) for _ in range(2 * n)] + [(0, 0.25) for _ in range(n)]

    # Initial guess for positions and radii
    x0 = np.hstack((centers.flatten(), radii))

    # Define the cost function to minimize (negative of total radii)
    def cost(x):
        positions = x[:2 * n].reshape((n, 2))
        radii = x[2 * n:]

        # Compute total radii to be maximized (we minimize negative of it)
        total_radii = -np.sum(radii)

        # Check if any circle is outside the square
        for i in range(n):
            x, y = positions[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return float('inf')

        # Check for overlaps between circles
        for i in range(n):
            for j in range(i + 1, n):
                dx = positions[i, 0] - positions[j, 0]
                dy = positions[i, 1] - positions[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return float('inf')

        return total_radii

    # Set up optimization problem
    cons = []
    for i in range(n):
        cons.append({'type': 'ineq', 'fun': lambda x, i=i: x[2 * n + i]})

    result = minimize(
        cost,
        x0,
        bounds=Bounds(np.zeros(3 * n), np.array([1] * (2 * n) + [0.25] * n)),
        constraints=cons,
        method='L-BFGS-B',
        options={'maxiter': 500, 'ftol': 1e-9}
    )

    if not result.success:
        raise ValueError("Optimization failed")

    # Extract final positions and radii
