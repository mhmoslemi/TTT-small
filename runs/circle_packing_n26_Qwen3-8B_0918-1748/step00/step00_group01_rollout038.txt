import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

def run_packing():
    # Initial guess: 20 circles in a hexagonal grid, 6 along the boundary
    initial_centers = np.array([
        [0.3, 0.3], [0.7, 0.3], [1.0, 0.3], [0.7, 0.7], [0.3, 0.7], [0.3, 0.5], [0.5, 0.5], [0.7, 0.5],
        [0.2, 0.4], [0.8, 0.4], [0.2, 0.6], [0.8, 0.6], [0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75],
        [0.15, 0.15], [0.85, 0.15], [0.15, 0.85], [0.85, 0.85], [0.5, 0.1], [0.5, 0.9], [0.1, 0.5], [0.9, 0.5]
    ])
    initial_radii = np.array([0.1] * 26)

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraints_func(radii):
        constraints = []
        for i in range(26):
            constraints.append((radii[i] >= 0))
        return constraints

    def bounds_func(radii):
        return [(0, 1) for _ in range(26)]

    def distance_matrix(centers, radii):
        dists = cdist(centers, centers)
        for i in range(26):
            for j in range(i + 1, 26):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        return True

    def optimize():
        res = minimize(
            fun=objective,
            x0=initial_radii,
            bounds=bounds_func(initial_radii),
            method='SLSQP',
            constraints=[
                {'type': 'ineq', 'fun': lambda x: np.array([distance_matrix(initial_centers, x)])}
            ]
        )
        return res.x

    # Optimization loop
    start_time = time.time()
    radii = optimize()
    centers = initial_centers.copy()
    return (centers, radii, np.sum(radii))
