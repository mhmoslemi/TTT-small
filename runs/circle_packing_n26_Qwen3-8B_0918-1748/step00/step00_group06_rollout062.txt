import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Initial hexagonal packing with 17 circles
    centers = []
    radii = []
    
    # Central circle
    centers.append([0.5, 0.5])
    radii.append(0.0)
    
    # First ring (distance from center = 1.0)
    for i in range(6):
        angle = 2 * np.pi / 6 * i
        x = 0.5 + np.cos(angle)
        y = 0.5 + np.sin(angle)
        centers.append([x, y])
        radii.append(0.0)
    
    # Second ring (distance from center = 1.5)
    for i in range(6):
        angle = 2 * np.pi / 6 * i
        x = 0.5 + 1.5 * np.cos(angle)
        y = 0.5 + 1.5 * np.sin(angle)
        centers.append([x, y])
        radii.append(0.0)
    
    # Third ring (distance from center = 2.0)
    for i in range(6):
        angle = 2 * np.pi / 6 * i
        x = 0.5 + 2.0 * np.cos(angle)
        y = 0.5 + 2.0 * np.sin(angle)
        centers.append([x, y])
        radii.append(0.0)
    
    # Add 9 circles manually in corners and edges
    # Bottom-left corner
    centers.append([0.1, 0.1])
    radii.append(0.0)
    # Bottom-right corner
    centers.append([0.9, 0.1])
    radii.append(0.0)
    # Top-right corner
    centers.append([0.9, 0.9])
    radii.append(0.0)
    # Top-left corner
    centers.append([0.1, 0.9])
    radii.append(0.0)
    # Bottom edge
    centers.append([0.4, 0.1])
    radii.append(0.0)
    # Top edge
    centers.append([0.6, 0.9])
    radii.append(0.0)
    # Left edge
    centers.append([0.1, 0.5])
    radii.append(0.0)
    # Right edge
    centers.append([0.9, 0.5])
    radii.append(0.0)
    # Center-top
    centers.append([0.5, 0.75])
    radii.append(0.0)
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraint_overlap(radii):
        n = len(radii)
        distances = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if distances[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0
    
    def constraint_boundary(radii):
        for i in range(len(radii)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return -1
        return 0
    
    bounds = Bounds(0, 1)
    
    res = minimize(
        objective,
        radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP'
    )
    
    return centers, res.x, np.sum(res.x)
