import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Initial guess for circle centers in a grid-like configuration
    # This configuration is a hybrid of hexagonal and rectangular packing
    # with some circles near the corners and edges for better utilization of space

    # Define initial positions (flat grid of 5 rows and 5 columns + 1 extra)
    num_circles = 26
    grid_rows = 5
    grid_cols = 5
    extra_circles = 1
    initial_centers = np.array([
        [i / grid_cols, j / grid_rows] for i in range(grid_cols)
        for j in range(grid_rows)
    ])[:num_circles - extra_circles]
    # Add a few extra circles at the corners and edges
    corners = [
        [0.05, 0.05], [0.95, 0.05], [0.95, 0.95], [0.05, 0.95]
    ]
    edges = [
        [0.5, 0.05], [0.5, 0.95], [0.05, 0.5], [0.95, 0.5]
    ]
    initial_centers = np.vstack((initial_centers, corners[:2], edges[:2], [0.5, 0.5]))
    # Initial radii are set based on the grid spacing
    spacing = 1 / grid_rows
    initial_radii = np.full(num_circles, spacing * 0.4)
    initial_radii[-1] *= 2  # One circle at center

    # Define the objective function: maximize sum of radii
    def objective(x):
        # x has shape (2 * num_circles)
        centers = x[:2 * num_circles].reshape((num_circles, 2))
        radii = x[2 * num_circles:]
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints
    def constraints(x):
        centers = x[:2 * num_circles].reshape((num_circles, 2))
        radii = x[2 * num_circles:]
        # Boundary constraints: x_i - r_i >= 0 and x_i + r_i <= 1
        bounds = []
        for i in range(num_circles):
            x_i, y_i = centers[i]
            r_i = radii[i]
            bounds.append((x_i - r_i, 1e-12))
            bounds.append((x_i + r_i, 1 - 1e-12))
            bounds.append((y_i - r_i, 1e-12))
            bounds.append((y_i + r_i, 1 - 1e-12))
        # Non-overlapping constraints: distance between centers[i] and centers[j] >= radii[i] + radii[j]
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                bounds.append((dist - (radii[i] + radii[j]), 1e-12))
        return bounds

    # Set up optimization problem
    x0 = np.concatenate((initial_centers.reshape(-1), initial_radii))
    bounds = Bounds([0.0] * (2 * num_circles) + [0.0] * num_circles, [1.0] * (2 * num_circles) + [1.0] * num_circles)

    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        method='SLSQP',
        constraints=constraints
    )

    # Extract results
    centers = result.x[:2 * num_circles].reshape((num_circles, 2))
    radii = result.x[2 * num_circles:]
    total_radii = np.sum(radii)

    return (centers, radii, total_radii)
