import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds

def run_packing():
    # Define the number of circles
    n = 26
    
    # Initial guess: grid-based placement with varying radii
    # Place 2 rows of 13 circles, with spacing and centering within the square
    # First row: 13 circles in the lower half
    # Second row: 13 circles in the upper half
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # First row of circles (lower half)
    for i in range(13):
        x = i / 12.0
        y = 0.25
        centers[i] = [x, y]
        radii[i] = 0.15
    
    # Second row of circles (upper half)
    for i in range(13):
        x = i / 12.0
        y = 0.75
        centers[13 + i] = [x, y]
        radii[13 + i] = 0.15
    
    # Define the objective function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    # Define constraints for non-overlapping and boundary conditions
    def constraints(x):
        # x is a flat array of all parameters: first n*2 parameters are centers, then radii
        centers_flat = x[:n*2].reshape(n, 2)
        radii_flat = x[n*2:]
        
        # Check boundary constraints
        for i in range(n):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return -np.inf
        
        # Check non-overlapping constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -np.inf
        
        return np.sum(radii_flat)
    
    # Prepare initial guess
    initial_guess = np.concatenate([centers.flatten(), radii])
    
    # Set bounds for centers (within [0,1]^2) and radii (non-negative)
    bounds = []
    for _ in range(n*2):
        bounds.append((0, 1))  # x and y coordinates
    for _ in range(n):
        bounds.append((0, 1))  # radii
    
    # Use SLSQP for optimization with bounds and constraints
    result = minimize(
        fun=constraints,
        x0=initial_guess,
        bounds=bounds,
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    # Extract optimized centers and radii
    optimized_centers = result.x[:n*2].reshape(n, 2)
    optimized_radii = result.x[n*2:]
    
    # Return the result
    return (optimized_centers, optimized_radii, np.sum(optimized_radii))
