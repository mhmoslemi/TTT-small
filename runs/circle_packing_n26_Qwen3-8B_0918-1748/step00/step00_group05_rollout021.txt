import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint
from scipy.optimize import NonlinearConstraint

def run_packing():
    # Define initial placement in a hexagonal grid with some extra circles at the corners
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Initial placement in hexagonal grid
    rows = 6
    cols = 6
    spacing = 1.0 / (rows + 1)
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing / 2
            y = i * spacing + spacing / 2
            if i % 2 == 0:
                x = j * spacing + spacing / 2
            else:
                x = j * spacing + spacing / 2 + spacing / 2
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = spacing / 2
    
    # Add extra circles at the corners
    corner_radii = spacing / 2
    for i in range(n - rows * cols):
        idx = rows * cols + i
        if i % 4 == 0:
            centers[idx] = [spacing, spacing]
        elif i % 4 == 1:
            centers[idx] = [1 - spacing, spacing]
        elif i % 4 == 2:
            centers[idx] = [1 - spacing, 1 - spacing]
        elif i % 4 == 3:
            centers[idx] = [spacing, 1 - spacing]
        radii[idx] = spacing / 2
    
    # Define optimization variables
    bounds = Bounds([0, 0], [1, 1])
    constraint = LinearConstraint(np.eye(2 * n), np.zeros(2 * n), np.full(2 * n, 1.0))
    
    # Define objective function
    def obj(x):
        radii = x[n:]
        return -np.sum(radii ** 2)
    
    # Define constraints
    def cons(x):
        centers = x[:n * 2].reshape(n, 2)
        radii = x[n:]
        dists = np.zeros(n * (n - 1) // 2)
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                dists[idx] = np.linalg.norm(centers[i] - centers[j])
                idx += 1
        return dists - (radii[i] + radii[j] for i, j in zip(range(n), range(n)))
    
    result = minimize(obj, np.concatenate((centers.flatten(), radii)), bounds=bounds, constraints=constraint)
    centers = result.x[:n * 2].reshape(n, 2)
    radii = result.x[n:]
    
    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} at ({x},{y}) r={r} outside unit square")
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.linalg.norm(centers[i] - centers[j])
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
