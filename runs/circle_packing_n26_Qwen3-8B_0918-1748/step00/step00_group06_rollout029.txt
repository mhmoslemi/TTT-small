import numpy as np
import cvxpy as cp

def run_packing():
    # Initial grid-based configuration
    n = 26
    grid_x = np.linspace(0.1, 0.9, 5)
    grid_y = np.linspace(0.1, 0.9, 5)
    centers = []
    radii = []
    
    # 25 circles in grid
    for i in range(5):
        for j in range(5):
            centers.append([grid_x[i], grid_y[j]])
            radii.append(0.125)  # Initial radius
    
    # 1 extra circle in the upper right corner
    centers.append([0.92, 0.95])
    radii.append(0.06)  # Initial radius
    
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Objective: maximize sum of radii
    # Constraints:
    # 1. All circles within the unit square
    # 2. No two circles overlap
    # 3. Radii non-negative
    
    # Optimization variables
    x = cp.Variable((n, 2))  # centers
    r = cp.Variable(n)       # radii
    
    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(r))
    
    # Constraints
    constraints = []
    
    # All circles within the unit square
    for i in range(n):
        constraints.append(x[i, 0] - r[i] >= -1e-6)
        constraints.append(x[i, 0] + r[i] <= 1 + 1e-6)
        constraints.append(x[i, 1] - r[i] >= -1e-6)
        constraints.append(x[i, 1] + r[i] <= 1 + 1e-6)
    
    # Non-overlapping constraint
    for i in range(n):
        for j in range(i+1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints.append(dist >= r[i] + r[j] - 1e-6)
    
    # Non-negative radii
    constraints.append(r >= 1e-6)
    
    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, eps=1e-8)
    
    # Extract solution
    centers_opt = x.value
    radii_opt = r.value
    
    # Verification
    assert np.isfinite(centers_opt).all(), "Invalid center coordinates"
    assert np.isfinite(radii_opt).all(), "Invalid radius values"
    assert (radii_opt >= 0).all(), "Negative radii found"
    
    # Return result
    return (centers_opt, radii_opt, np.sum(radii_opt))
