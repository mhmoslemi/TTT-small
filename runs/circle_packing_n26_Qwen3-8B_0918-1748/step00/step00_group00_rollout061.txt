import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial guess for centers and radii
    # Using a hexagonal close packing (HCP) arrangement
    # Two rows of 13 circles each
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)

    # Width and height of the square
    W, H = 1.0, 1.0

    # Initial guess for radius
    r0 = 0.05  # Adjusted to allow a better starting point

    # Initial row positions
    row_1 = np.linspace(0.5 * r0, W - 0.5 * r0, 13)
    row_2 = np.linspace(0.5 * r0, W - 0.5 * r0, 13)

    # Set centers for first row
    for i in range(13):
        centers[i, 0] = row_1[i]
        centers[i, 1] = 0.5 * r0
        radii[i] = r0

    # Set centers for second row (with offset)
    for i in range(13):
        centers[13 + i, 0] = row_2[i]
        centers[13 + i, 1] = H - 0.5 * r0
        radii[13 + i] = r0

    # Objective function to maximize the sum of radii
    def objective(x):
        nonlocal centers, radii
        # x is a vector: [r1, r2, ..., r26]
        # Update radii
        radii = x
        # Calculate the sum of radii
        total = np.sum(radii)
        return -total  # Minimize negative to maximize

    # Constraint for non-overlapping between circles
    def non_overlap(x):
        nonlocal centers, radii
        radii = x
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-8:

        return 0

        nonlocal centers, radii
        radii = x

    )

    # Optimization with bounds
    bounds = [(0, 1.0) for _ in range(n)]

    # Run the optimization
    result = optimize.minimize(

        x0=x0,
        bounds=bounds,
        constraints=cons,
        method='SLSQP'
    )
