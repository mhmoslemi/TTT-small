import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run_packing():
    # Initial guess: Hexagonal packing with 2 layers
    # First layer: 6 circles in a hexagon
    # Second layer: 12 circles in a larger hexagon
    # Third layer: 10 circles in a larger hexagon
    # Total circles: 6 + 12 + 10 = 28, but we take 26

    # Anchor circle at (0.5, 0.5)
    # Radii start as uniform, but we will optimize them

    # Generate initial positions for a hexagonal packing
    def generate_hexagonal_positions(n):
        positions = []
        for layer in range(2):
            for i in range(6 * (layer + 1)):
                angle = 2 * np.pi / 6 * i
                r = (layer + 1) * 1
                x = 0.5 + r * np.cos(angle)
                y = 0.5 + r * np.sin(angle)
                positions.append((x, y))
            if layer == 0:
                # Add the center circle
                positions.append((0.5, 0.5))
        # Remove duplicates and take first 26
        positions = np.array(positions)
        unique_positions = np.unique(positions, axis=0)
        return unique_positions[:26]

    # Initial positions
    centers = generate_hexagonal_positions(2)
    radii = np.ones(26) * 0.05  # Initial small radii

    # Optimization parameters
    bounds_centers = [(0, 1) for _ in range(2 * 26)]
    bounds_radii = [(0, 1) for _ in range(26)]

    # Objective function: Maximize sum of radii
    def objective(x):
        centers_flat = x[:2 * 26].reshape(26, 2)
        radii_flat = x[2 * 26:]
        return -np.sum(radii_flat)

    # Constraints: distance between centers >= radii[i] + radii[j]
    def constraint_func(x):
        centers_flat = x[:2 * 26].reshape(26, 2)
        radii_flat = x[2 * 26:]
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return dist - radii_flat[i] - radii_flat[j] + 1e-8
        return 0

    # Initial guess with some randomness
    np.random.seed(0)
    centers_initial = centers + np.random.uniform(-0.05, 0.05, centers.shape)
    centers_initial = np.clip(centers_initial, 0, 1)
    x0 = np.concatenate((centers_initial.flatten(), radii))

    # Optimization with bounds
    cons = {'type': 'ineq', 'fun': constraint_func}
    result = minimize(
        objective,
        x0,
        method='SLSQP',
        bounds=Bounds(bounds_centers + bounds_radii),
        constraints=cons,
        options={'maxiter': 500, 'ftol': 1e-8, 'eps': 1e-8}
    )

    # Extract results
    centers_opt = result.x[:2 * 26].reshape(26, 2)
    radii_opt = result.x[2 * 26:]
    sum_radii = np.sum(radii_opt)

    # Validate the packing
    def validate_packing(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                    or y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Validation failed: {message}")

    return centers_opt, radii_opt, sum_radii
