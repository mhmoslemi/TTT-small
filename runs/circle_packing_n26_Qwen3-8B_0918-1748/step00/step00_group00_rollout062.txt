import numpy as np
import scipy.optimize as opt
import warnings

def run_packing():
    """
    Packs 26 circles in a unit square to maximize the sum of radii.

    Returns:
        (centers, radii, sum_radii) where centers has shape (26, 2) and radii has shape (26,).
    """
    # Initial grid of 5x5 positions for 25 circles, plus 1 center circle
    grid = np.linspace(0, 1, 6)  # 6 points from 0 to 1 inclusive
    x_coords = grid[1:-1]  # Remove the ends to get inner positions
    y_coords = grid[1:-1]
    grid_centers = np.array([[x, y] for x in x_coords for y in y_coords])
    centers = np.vstack((grid_centers, [0.5, 0.5]))  # Add center circle

    # Initial radius guess based on grid spacing
    initial_radii = np.full(26, 0.1)
    for i in range(25):
        x, y = grid_centers[i]
        # Compute initial radius as 1/3 of the minimum distance to boundaries
        r = min(x, 1 - x, y, 1 - y) / 3
        initial_radii[i] = r
    initial_radii[25] = 0.1  # Center circle

    # Define constraints for optimization
    def constraints(x):
        # x has shape (26 * 3) -> (x1, y1, r1, x2, y2, r2, ..., x26, y26, r26)
        x_centers = x[:26*2].reshape((26, 2))
        radii = x[26*2:]
        # Ensure each circle is inside the unit square
        for i in range(26):
            x_i, y_i = x_centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return np.array([0.0])
        # Ensure circles are non-overlapping
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt(np.sum((x_centers[i] - x_centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.array([0.0])
        return np.array([])

    # Objective function to maximize total radius sum
    def obj(x):
        x_centers = x[:26*2].reshape((26, 2))
        radii = x[26*2:]
        return -np.sum(radii)  # Negative for minimization

    # Initial guess: centers are fixed, radii are initial_radii
    x0 = np.concatenate((centers.reshape(-1), initial_radii))

    # Optimization settings
    opt_options = {
        'ftol': 1e-8,
        'maxiter': 200,
        'disp': False,
        'constraints': {'type': 'ineq', 'fun': constraints}
    }

    # Suppress warnings for optimization issues
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        result = opt.minimize(obj, x0, method='SLSQP', options=opt_options)

    # Extract results
    x_result = result.x
    centers_result = x_result[:26*2].reshape((26, 2))
    radii_result = x_result[26*2:]
    sum_radii = np.sum(radii_result)

    # Validate packing
    valid, msg = validate_packing(centers_result, radii_result)
    if not valid:
        raise ValueError(f"Validation failed: {msg}")

    return centers_result, radii_result, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
