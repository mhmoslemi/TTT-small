def run_packing():
    # Initial hexagonal packing with slight modification for boundary
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # First 5 rows of hexagonal packing
    for i in range(5):
        for j in range(5):
            x = j + 0.5 * (i % 2)
            y = i + 0.4 * (i % 2)
            centers[i * 5 + j] = [x, y]
    
    # Add 5 remaining circles at the end (positions 20 to 25)
    for i in range(5):
        x = 0.5 + 0.1 * i
        y = 0.75
        centers[20 + i] = [x, y]
    
    # Initial guess for radii
    for i in range(n):
        if i < 20:
            if i % 5 == 0:
                radii[i] = 0.15
            else:
                radii[i] = 0.13
        else:
            radii[i] = 0.12
    
    # Define the objective function to maximize total radius sum
    def objective(x):
        # x is a flat array: radii followed by centers
        new_radii = x[:n]
        new_centers = x[n:].reshape(n, 2)
        
        # Enforce constraints to ensure self-consistency
        obj = -np.sum(new_radii)  # We minimize the negative to maximize the sum
        return obj
    
    # Define the constraints
    def constraints(x):
        new_radii = x[:n]
        new_centers = x[n:].reshape(n, 2)
        cons = []
        for i in range(n):
            x_i, y_i = new_centers[i]
            r_i = new_radii[i]
            # Ensure circle is within the square
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return False
        for i in range(n):
            for j in range(i+1, n):
                x_i, y_i = new_centers[i]
                x_j, y_j = new_centers[j]
                r_i = new_radii[i]
                r_j = new_radii[j]
                distance = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                if distance < r_i + r_j - 1e-12:
                    return False
        return True
    
    # Combine all the constraints
    def constraint_func(x):
        new_radii = x[:n]
        new_centers = x[n:].reshape(n, 2)
        for i in range(n):
            x_i, y_i = new_centers[i]
            r_i = new_radii[i]

                x_j, y_j = new_centers[j]

                r_j = new_radii[j]
                distance = np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2)
                if distance < r_i + r_j - 1e-12:

    # Define the optimization problem with bounds and constraints

    initial_guess = np.zeros(3 * n)

    result = minimize(

        return (optimized_centers, optimized_radii, total_radius_sum)

        raise RuntimeError("Optimization failed to converge")
