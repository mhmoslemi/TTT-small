import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import minimize, Bounds
from scipy.optimize import LinearConstraint
import time

def run_packing():
    # Define the number of circles
    n_circles = 26

    # Initial grid-based placement for 25 circles in hexagonal close packing
    def hexagonal_grid(num_circles):
        row_count = int(np.ceil(np.sqrt(num_circles)))
        col_count = int(np.ceil(num_circles / row_count))
        centers = np.zeros((num_circles, 2))
        for i in range(num_circles):
            row = i // col_count
            col = i % col_count
            x = col + 0.5 * (row % 2)
            y = row
            centers[i] = [x, y]
        return centers

    # Initial placement for 25 circles
    initial_centers = hexagonal_grid(25)
    initial_centers = initial_centers / np.max(initial_centers, axis=0)  # scale to unit square

    # Optimization parameters
    max_iter = 100
    tolerance = 1e-2

    # Define the optimization variables
    # We will optimize the positions (x, y) and radii (r) of the circles
    # For 26 circles, we have 26 * 2 + 26 = 78 variables

    # Initial guess: positions from initial placement, radii all equal to some initial value
    initial_guess = np.concatenate([initial_centers.flatten(), np.full(n_circles, 0.01)])

    # Constraints: distance between each pair of circles >= sum of their radii
    # Also, each circle must fit within the unit square

    # Define the optimization problem
    def objective(x):
        return -np.sum(x[n_circles * 2:])  # Negative because we are minimizing

    def bounds(x):
        # Bounds for positions (x, y) between 0 and 1
        # Bounds for radii >= 0
        return Bounds(0, 1) + Bounds(0, np.inf)

    def constraints(x):
        # Extract positions and radii
        positions = x[:n_circles * 2].reshape(n_circles, 2)
        radii = x[n_circles * 2:]

        # Distance between each pair of circles
        distances = squareform(pdist(positions))
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                if distances[i, j] < radii[i] + radii[j] - 1e-12:
                    return False
        # Check if each circle is within the unit square
        for i in range(n_circles):
            x_i, y_i = positions[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return False
        return True

    # Use a local optimizer to refine the configuration
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': constraints},
        options={'maxiter': max_iter, 'ftol': tolerance}
    )

    centers = result.x[:n_circles * 2].reshape(n_circles, 2)
    radii = result.x[n_circles * 2:]

    # Final validation to ensure we meet the constraints
    validation_result, message = validate_packing(centers, radii)
    if not validation_result:
        raise RuntimeError(f"Validation failed: {message}")

    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
