import numpy as np
from scipy.optimize import minimize, Bounds, LinearConstraint
import time
import random

def run_packing():
    # Define the number of circles
    n = 26

    # Initial grid-based placement of centers
    # 4 rows x 6 columns = 24 circles, with 2 extra circles at the end
    cols = 6
    rows = 4
    dx = 1.0 / cols
    dy = 1.0 / rows
    centers = np.zeros((n, 2))

    # Fill the grid with 24 circles
    for i in range(rows):
        for j in range(cols):
            centers[i * cols + j] = [j * dx + dx / 2, i * dy + dy / 2]

    # Add two more circles at the end of the last row
    centers[-2] = [cols * dx - dx / 2, rows * dy - dy / 2]
    centers[-1] = [cols * dx - dx / 2, rows * dy - dy / 2]

    # Initial radii estimate
    initial_radius = 0.098  # Approximately half the bin size
    radii = np.full(n, initial_radius)

    # Objective: Maximize sum of radii (minimize negative radii)
    def neg_sum_radii(radii):
        return -np.sum(radii)

    # Constraints:
    # 1. Radii are non-negative
    bounds = Bounds(np.zeros(n), np.inf)

    # 2. Constraint: Distance between centers >= radii[i] + radii[j]
    def constraint_func(radii):
        dist = np.zeros(n * (n - 1) // 2)
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                dist[idx] = np.sqrt(np.sum((centers[i] - centers[j]) ** 2)) - radii[i] - radii[j]
                idx += 1
        return dist

    # Define the nonlinear constraint: dist >= 0
    # This is converted to a linear constraint for optimization
    # Instead of direct constraint, we use a penalty approach

    # Define a penalty function to be minimized
    def penalty_function(radii):
        penalty = 0.0
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                overlap = dist - radii[i] - radii[j]
                if overlap < 0:
                    penalty += -overlap * 1000  # Penalty for overlap
        # Also ensure that circles are within unit square
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                penalty += 10000  # Penalty for going outside
        return penalty

    # Optimization: Use Nelder-Mead to maximize sum of radii
    result = minimize(
        fun=penalty_function,
        x0=radii,
        bounds=bounds,
        method='Nelder-Mead',
        options={'xatol': 1e-8, 'maxiter': 500}
    )

    radii = result.x

    # Validate the packing
    success, msg = validate_packing(centers, radii)
    if not success:
        raise ValueError(f"Validation failed: {msg}")

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
