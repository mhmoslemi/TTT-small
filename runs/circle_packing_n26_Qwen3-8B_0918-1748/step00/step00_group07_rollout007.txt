import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from scipy.spatial import Delaunay, Voronoi
from scipy.optimize import LinearConstraint, NonlinearConstraint
from itertools import combinations

def run_packing():
    # Initial random placement
    np.random.seed(42)
    centers = np.random.rand(26, 2)
    radii = np.full(26, 0.02)

    # Step 1: Delaunay triangulation to spread points
    tri = Delaunay(centers)
    edges = np.unique(np.concatenate((tri.simplices[:, [0, 1]], tri.simplices[:, [1, 2]], tri.simplices[:, [2, 0]])), axis=0)
    distances = np.zeros(26)
    for i in range(26):
        distances[i] = np.min(np.sqrt(np.sum((centers[i] - centers[edges[np.where(edges == i)[0]]]) ** 2, axis=1)))
    radii = np.maximum(radii, distances / 2 - 1e-5)

    # Step 2: Local optimization using Nelder-Mead
    def objective(radii_vector):
        return -np.sum(radii_vector)

    def constraint_func(x):
        centers = x[:2*26].reshape(26, 2)
        radii = x[2*26:]
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or
                y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return -np.inf
        for i in range(26):
            for j in range(i+1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        return np.sum(radii)

    bounds = Bounds(np.zeros(26*2 + 26), np.inf)
    result = minimize(objective, np.concatenate((centers.flatten(), radii)), bounds=bounds,
                     constraints=NonlinearConstraint(lambda x: constraint_func(x), -np.inf, 0),
                     method='Nelder-Mead')
    best_centers = result.x[:2*26].reshape(26, 2)
    best_radii = result.x[2*26:]
    sum_radii = np.sum(best_radii)

    # Step 3: Fix centers near boundaries
    for i in range(26):
        x, y = best_centers[i]
        r = best_radii[i]
        if x - r < -1e-12:
            dx = max(0, r - (x - r))
            best_centers[i, 0] = x + dx
        elif x + r > 1 + 1e-12:
            dx = max(0, (x + r) - (1 + 1e-12))
            best_centers[i, 0] = x - dx
        if y - r < -1e-12:
            dy = max(0, r - (y - r))
            best_centers[i, 1] = y + dy
        elif y + r > 1 + 1e-12:
            dy = max(0, (y + r) - (1 + 1e-12))
            best_centers[i, 1] = y - dy

    # Step 4: Local refinement using gradient descent
    def objective2(radii_vec):
        return -np.sum(radii_vec)

    def constraint_func2(x):
        centers = x[:2*26].reshape(26, 2)
        radii = x[2*26:]
        for i in range(26):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if (x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or
                y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12):
                return -np.inf
        for i in range(26):
            for j in range(i+1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        return np.sum(radii)

    bounds2 = Bounds(np.zeros(26*2 + 26), np.inf)
    result2 = minimize(objective2, np.concatenate((best_centers.flatten(), best_radii)),
                      bounds=bounds2,
                      constraints=NonlinearConstraint(lambda x: constraint_func2(x), -np.inf, 0),
                      method='L-BFGS-B')
    centers = result2.x[:2*26].reshape(26, 2)
    radii = result2.x[2*26:]
    sum_radii = np.sum(radii)

    # Final validation
    if not validate_packing(centers, radii)[0]:
        raise ValueError("Validation failed")

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
