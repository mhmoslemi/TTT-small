import numpy as np
import scipy.optimize as optimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial guess of centers and radii
    centers = np.random.rand(26, 2)
    radii = np.random.rand(26) * 0.1

    # Optimization variables: centers (26, 2) and radii (26,)
    # Flatten for optimization
    variables = np.concatenate([centers.flatten(), radii])

    def objective(variables):
        # Unflatten variables
        centers_flat = variables[:26 * 2]
        radii_flat = variables[26 * 2:]
        centers = centers_flat.reshape(26, 2)
        radii = radii_flat

        # Objective: maximize sum of radii
        return -np.sum(radii)  # Minimize negative sum

    def bounds():
        # Bounds for centers (0 to 1) and radii (0 to 1)
        bounds = []
        for _ in range(26 * 2):
            bounds.append((0, 1))
        for _ in range(26):
            bounds.append((0, 1))
        return bounds

    def constraints(variables):
        # Unflatten variables
        centers_flat = variables[:26 * 2]
        radii_flat = variables[26 * 2:]
        centers = centers_flat.reshape(26, 2)
        radii = radii_flat

        # Boundary constraints: circles must be inside the square
        constraints = []
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            # Check if circle is entirely within the square
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        # Non-overlap constraints: distance between centers >= sum of radii
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Optimization settings
    result = optimize.minimize(
        objective,
        variables,
        bounds=bounds(),
        constraints={"type": "ineq", "fun": constraints},
        method="SLSQP",
        options={"max_iter": 1000, "ftol": 1e-12, "eps": 1e-12}
    )

    # Extract result
    best_centers = result.x[:26 * 2].reshape(26, 2)
    best_radii = result.x[26 * 2:]
    sum_radii = np.sum(best_radii)

    # Validate the result
    valid, msg = validate_packing(best_centers, best_radii)
    if not valid:
        raise ValueError(f"Packing invalid: {msg}")

    return best_centers, best_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
