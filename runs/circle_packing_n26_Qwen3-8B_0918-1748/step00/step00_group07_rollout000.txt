import numpy as np
import numpy.random as npr
import scipy.optimize as optimize
import itertools

def run_packing():
    def place_corner_circles():
        corners = np.array([[0, 0], [1, 0], [1, 1], [0, 1]])
        radii = np.array([0.1, 0.1, 0.1, 0.1])
        return corners, radii

    def place_row_circles(num_rows, spacing, offset):
        centers = []
        for row in range(num_rows):
            y = row * spacing + offset
            x_positions = np.linspace(0, 1, num_rows, endpoint=False)
            x_positions = np.sort(x_positions)
            for x in x_positions:
                centers.append([x, y])
        return np.array(centers)

    def place_hexagonal_rows(num_rows, spacing, offset):
        centers = []
        for row in range(num_rows):
            y = row * spacing + offset
            if row % 2 == 0:
                x_positions = np.linspace(0, 1, num_rows, endpoint=False)
                x_positions = np.sort(x_positions)
                for x in x_positions:
                    centers.append([x, y])
            else:
                x_positions = np.linspace(0.5 / (num_rows - 1), 1 - 0.5 / (num_rows - 1), num_rows - 1)
                x_positions = np.sort(x_positions)
                for x in x_positions:
                    centers.append([x, y])
        return np.array(centers)

    def initial_configuration():
        num_corner_circles = 4
        num_hex_rows = 5
        num_hex_cols = 5

        corner_circles, _ = place_corner_circles()
        hex_centers = place_hexagonal_rows(num_hex_rows, 0.2, 0.25)
        centers = np.vstack((corner_circles, hex_centers))

        num_total = len(centers)
        if num_total > 26:
            raise ValueError("Exceeded the number of circles allowed in the configuration.")

        return centers

    def objective(radii, centers):
        if np.any(radii < 0):
            return -np.inf
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -np.inf
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -np.inf
        return np.sum(radii)

    def perturb_centers(centers, perturbation):
        return np.clip(centers + npr.uniform(-perturbation, perturbation, size=centers.shape), 0, 1)

    def optimize_packing(centers):
        np.random.seed(42)
        N = len(centers)
        initial_radii = np.full(N, 0.1)
        bounds = [(0, 1) for _ in range(N)]
        result = optimize.minimize(
            lambda r: -objective(r, centers),
            x0=initial_radii,
            bounds=bounds,
            method='L-BFGS-B',
            options={'ftol': 1e-6, 'maxiter': 1000}
        )
        if not result.success:
            raise RuntimeError("Optimization failed with message: " + result.message)
        return result.x

    clusters = initial_configuration()
    radii = optimize_packing(clusters)
    centers = clusters

    for i in range(len(centers)):
        if radii[i] < 0.001:
            radii[i] = 0.001

    max_radius = 0
    for i in range(len(centers)):
        if radii[i] > max_radius:
            max_radius = radii[i]

    for i in range(len(centers)):
        if radii[i] > max_radius:
            radii[i] = max_radius

    return centers, radii, np.sum(radii)
