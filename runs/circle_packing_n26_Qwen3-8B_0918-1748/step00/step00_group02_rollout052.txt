import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist
import time

def run_packing():
    # Initial hexagonal grid layout with 4 rows: 6, 7, 6, 7 circles
    num_rows = 4
    row_widths = [6, 7, 6, 7]
    total_circles = sum(row_widths)
    num_cols = max(row_widths)
    
    # Initial grid positions
    centers = np.zeros((total_circles, 2))
    col_idx = 0
    for row in range(num_rows):
        if row % 2 == 0:
            row_centers = np.linspace(0, 1, row_widths[row])
        else:
            row_centers = np.linspace(0.5 / (row_widths[row] - 1), 0.5 + 0.5 / (row_widths[row] - 1), row_widths[row])
        for i in range(row_widths[row]):
            centers[col_idx] = [row_centers[i], row + 0.5]
            col_idx += 1
    
    # Initial radii: estimate from hexagonal packing
    # Calculate max r for 4 rows in hexagonal packing with equal radii
    # Height constraint: sqrt(3)*r*(number of rows - 1) + 2r <= 1
    r_initial = 1 / (3 * np.sqrt(3) + 2)
    radii = np.full(total_circles, r_initial)

    # Constraints: non-overlapping, within square
    # Objective: maximize sum of radii

    def obj_func(x):
        # x contains radii
        return -np.sum(x)  # minimize -sum(R) to maximize sum(R)

    def constraint_func(x):
        # x contains radii and centers
        centers = x[:2 * total_circles].reshape((-1, 2))
        radii = x[2 * total_circles:]
        
        # Check if centers are within [0,1]
        if np.any(centers < -1e-12) or np.any(centers > 1 + 1e-12):
            return -np.inf
        
        # Check if radii are non-negative
        if np.any(radii < -1e-12):
            return -np.inf
        
        # Check if circles are inside unit square
        for i in range(total_circles):
            x_center, y_center = centers[i]
            r = radii[i]
            if (x_center - r < -1e-12 or x_center + r > 1 + 1e-12 or
                y_center - r < -1e-12 or y_center + r > 1 + 1e-12):
                return -np.inf
        
        # Check for overlaps between circles
        for i in range(total_circles):
            for j in range(i + 1, total_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist_sq = dx*dx + dy*dy
                if dist_sq < (radii[i] + radii[j])**2 - 1e-12:
                    return -np.inf
        
        # If all checks passed
        return np.sum(radii)

    # Flatten centers and radii into single vector for optimization
    x0 = np.concatenate([centers.reshape(-1), radii])
    bounds = [(0, 1) for _ in range(2 * total_circles + total_circles)]

    # Set up optimization
    result = minimize(
        fun=obj_func,
        x0=x0,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 1000},
        jac=None,
        callback=lambda x: None
    )

    # Extract optimized centers and radii
    optimized_x = result.x
    centers_opt = optimized_x[:2 * total_circles].reshape(-1, 2)
    radii_opt = optimized_x[2 * total_circles:]
    
    # Validate packing to ensure it's correct
    # This is to catch cases where the optimization failed
    def validate_packing(centers, radii):
        n = centers.shape[0]
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"

        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"

        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"

        return True, "ok"

    valid, message = validate_packing(centers_opt, radii_opt)
    
    if not valid:
        raise RuntimeError(f"Invalid packing: {message}")
    
    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
