import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint, NonlinearConstraint

def run_packing():
    class PackingProblem:
        def __init__(self, num_circles):
            self.num_circles = num_circles
            self.dimensions = 2
            self.n_vars = num_circles * (self.dimensions + 1)
            self.x_low = np.zeros(self.n_vars)
            self.x_high = np.zeros(self.n_vars)
            self.x_low[:self.dimensions * num_circles] = 0.0
            self.x_high[:self.dimensions * num_circles] = 1.0
            self.x_low[self.dimensions * num_circles:] = 0.0
            self.x_high[self.dimensions * num_circles:] = 1.0

            self.c = np.zeros(self.n_vars)
            self.c[self.dimensions * num_circles:] = 1.0

            self.angles = np.linspace(0, 2 * np.pi, num_circles, endpoint=False)
            self.radius_base = 0.1

        def objective(self, x):
            return -np.sum(x[self.dimensions * self.num_circles:])

        def inequality(self, x):
            n = self.num_circles
            centers = x[:n * self.dimensions].reshape(n, self.dimensions)
            radii = x[n * self.dimensions:]
            dists = np.sqrt(np.sum((centers[:, np.newaxis] - centers[np.newaxis, :])**2, axis=-1))
            for i in range(n):
                for j in range(i + 1, n):
                    if dists[i, j] < radii[i] + radii[j] - 1e-8:
                        return 1.0
            return 0.0

        def boundary(self, x):
            n = self.num_circles
            centers = x[:n * self.dimensions].reshape(n, self.dimensions)
            radii = x[n * self.dimensions:]
            for i in range(n):
                x, y = centers[i]
                r = radii[i]
                if not (0 - r < x < 1 + r) or not (0 - r < y < 1 + r):
                    return 1.0
            return 0.0

    def initial_guess(num_circles):
        n_rows = 5
        row_radii = []
        centers = []
        col_radii = []
        for i in range(n_rows):
            row_idx = i
            row_count = 5 + i if i < 2 else 6 - i
            if i == 0:
                row_radii.append(1.0)
                col_radii.append(1.0)
                for j in range(row_count):
                    centers.append((j / (row_count - 1), i / (n_rows - 1)))
            else:
                row_radii.append(0.8)
                col_radii.append(0.8)
                for j in range(row_count):
                    centers.append(
                        (j / (row_count - 1), i / (n_rows - 1))
                    )
        return centers, row_radii + col_radii

    centers, radii = initial_guess(26)
    x0 = np.array([x for row in centers for x in row] + radii)
    bounds = Bounds(x0, x0)
    problem = PackingProblem(26)
    constraints = [
        NonlinearConstraint(problem.inequality, 0, 1e-6, 1e3),
        NonlinearConstraint(problem.boundary, 0, 1e-6, 1e3)
    ]
    result = minimize(
        problem.objective,
        x0,
        bounds=bounds,
        constraints=constraints,
        method='SLSQP',
        options={'ftol': 1e-6, 'maxiter': 500}
    )
    x_opt = result.x
    centers_opt = x_opt[:26 * 2].reshape(26, 2)
    radii_opt = x_opt[26 * 2:]
    total_radius = np.sum(radii_opt)
    return centers_opt, radii_opt, total_radius
