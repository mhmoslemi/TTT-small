import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    # Define initial configuration: 4 rows of 6 circles, 2 rows of 5 circles, and 1 row of 2 circles in a staggered hexagonal arrangement
    # First layer: 6 circles in a row, aligned with the square
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # First row of 6 circles
    dx = 1 / 6
    dy = np.sqrt(3) / 6
    for i in range(6):
        centers[i, 0] = i * dx
        centers[i, 1] = 0.5
        radii[i] = 0.15

    # Second row, 5 circles, shifted right by dx / 2
    for i in range(5):
        centers[6 + i, 0] = i * dx + dx / 2
        centers[6 + i, 1] = 0.5 + dy
        radii[6 + i] = 0.15

    # Third row, 5 circles, shifted right by dx / 2
    for i in range(5):
        centers[11 + i, 0] = i * dx + dx / 2
        centers[11 + i, 1] = 0.5 + 2 * dy
        radii[11 + i] = 0.15

    # Fourth row, 6 circles, shifted right by dx / 2
    for i in range(6):
        centers[16 + i, 0] = i * dx + dx / 2
        centers[16 + i, 1] = 0.5 + 3 * dy
        radii[16 + i] = 0.15

    # Fifth row, 2 circles
    centers[22, 0] = 0.4
    centers[22, 1] = 0.8
    radii[22] = 0.15
    centers[23, 0] = 0.6
    centers[23, 1] = 0.8
    radii[23] = 0.15

    def objective(radii):
        return -np.sum(radii)

    def constraints(radii):
        # Ensure all circles are within the square
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return False
        # Ensure no two circles overlap
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Use the initial radii as starting point for optimization
    initial_radii = radii.copy()
    result = minimize(objective, initial_radii, bounds=[(0, 1)]*26, constraints={"type": "ineq", "fun": constraints})
    
    if not result.success:
        raise RuntimeError("Optimization failed")
    
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
