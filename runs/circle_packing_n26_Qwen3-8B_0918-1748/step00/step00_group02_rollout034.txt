import numpy as np
import cvxpy as cp
import math

def run_packing():
    # Initial hexagonal packing with 5 rows, 5 columns
    n = 25
    r = 0.25  # Initial guess for radius
    centers = np.zeros((n, 2))
    for i in range(5):
        for j in range(5):
            x = j * (2 * r) + (i % 2) * r
            y = i * (2 * r) * math.sqrt(3) / 2
            centers[i * 5 + j] = [x, y]
    
    # Introduce a 26th circle in the best gap, initial guess
    # Find the gap with the most available space
    center = [0.5, 0.5]  # Initial guess, can be adjusted
    radii = np.full(n + 1, r)
    radii[-1] = 0.2  # Initial guess for the 26th circle

    # Create optimization variables
    centers_var = cp.Variable((n + 1, 2))
    radii_var = cp.Variable(n + 1, non_negative=True)

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii_var))

    # Constraints: within unit square
    constraints = []
    for i in range(n + 1):
        x, y = centers_var[i]
        r = radii_var[i]
        constraints.append(x - r >= 0)
        constraints.append(x + r <= 1)
        constraints.append(y - r >= 0)
        constraints.append(y + r <= 1)

    # Constraint: non-overlapping
    for i in range(n + 1):
        for j in range(i + 1, n + 1):
            dist = cp.sqrt((centers_var[i, 0] - centers_var[j, 0]) ** 2 + 
                           (centers_var[i, 1] - centers_var[j, 1]) ** 2)
            constraints.append(dist >= radii_var[i] + radii_var[j] - 1e-8)

    # Initial guess for centers
    centers_guess = np.vstack((centers, [0.5, 0.5]))
    centers_var.value = centers_guess

    # Initial guess for radii
    radii_var.value = radii

    # Solve the optimization problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.OSQP, eps_abs=1e-8, eps_rel=1e-8, max_iter=1000)
    except:
        # If solver fails, fall back to a simpler configuration
        centers_simple = np.vstack((np.array([[0.2, 0.2]] * 25), [0.5, 0.5]))
        radii_simple = np.full(26, 0.2)
        return (centers_simple, radii_simple, np.sum(radii_simple))

    # Extract results
    centers_result = centers_var.value
    radii_result = radii_var.value
    sum_radii = np.sum(radii_result)

    # Return centers, radii, and sum of radii
    return (centers_result, radii_result, sum_radii)
