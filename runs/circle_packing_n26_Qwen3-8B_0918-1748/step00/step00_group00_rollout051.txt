import numpy as np
from scipy.optimize import Bounds, minimize

def run_packing():
    n = 26
    # Initial guess for centers: hexagonal packing within square
    centers = np.zeros((n, 2))
    row_height = 1.0 / np.sqrt(3)
    col_width = 1.0 / (np.sqrt(n))
    
    # Create a hexagonal grid pattern
    for i in range(n):
        row = i // 5
        col = i % 5
        x = col * col_width
        y = row * row_height
        centers[i, 0] = x
        centers[i, 1] = y

    # Initial radius guess: equal radii
    radii = np.full(n, 0.05)

    # Define the objective function: maximize the sum of radii
    def obj(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Define constraints: non-overlapping and boundary
    def cons(x):
        centers_flat = x[:2*n].reshape((n, 2))
        radii_flat = x[2*n:]
        for i in range(n):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < 0 or x_i + r_i > 1 or y_i - r_i < 0 or y_i + r_i > 1:
                return 1
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_i = radii_flat[i]
                r_j = radii_flat[j]
                if dist < r_i + r_j - 1e-8:
                    return 1
        return 0

    # Combine centers and radii into a single vector for optimization
    x0 = np.concatenate([centers.flatten(), radii])

    # Define bounds for centers (within [0,1]^2) and radii (non-negative)
    bounds = Bounds(np.concatenate([np.zeros(2*n), np.zeros(n)]), np.concatenate([np.ones(2*n), np.inf * np.ones(n)]))

    # Run optimization
    result = minimize(obj, x0, bounds=bounds, constraints={"type": "ineq", "fun": cons})

    # Extract final centers and radii
    centers = result.x[:2*n].reshape((n, 2))
    radii = result.x[2*n:]
    sum_radii = np.sum(radii)

    return centers, radii, sum_radii
