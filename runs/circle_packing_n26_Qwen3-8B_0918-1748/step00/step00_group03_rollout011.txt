import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial configuration: 4 rows of circles in hexagonal pattern
    # First row: 7 circles
    # Second row: 8 circles, offset vertically
    # Third row: 7 circles
    # Fourth row: 4 circles, centered

    # Define initial positions and radii based on a rough hexagonal packing
    # We'll optimize these later

    # 1st row
    row1 = np.array([(i / 6.0, 0.2) for i in range(7)])
    # 2nd row: offset vertically by r*sqrt(3)/2 and shifted left/right
    row2 = np.array([(i / 6.0 + 0.5, 0.4) for i in range(8)])
    # 3rd row: same as first
    row3 = np.array([(i / 6.0, 0.6) for i in range(7)])
    # 4th row: centered
    row4 = np.array([(0.5, 0.8) for _ in range(4)])
    centers = np.vstack([row1, row2, row3, row4])
    radii = np.full(26, 0.06)

    # Define optimization variables: positions and radii
    n = 26
    bounds = [(0, 1) for _ in range(2 * n)]  # 2D positions for all circles

    # Define the objective: maximize the sum of radii (minimize negative sum)
    def objective(x):
        pos = x.reshape((n, 2))
        r = x[n * 2:]  # radii after positions
        return -np.sum(r)

    # Define constraints: circle i and j must be non-overlapping
    def constraint_overlap(x, i, j):
        pos = x.reshape((n, 2))
        r = x[n * 2:]
        dx = pos[i, 0] - pos[j, 0]
        dy = pos[i, 1] - pos[j, 1]
        dist_sq = dx**2 + dy**2
        if dist_sq < 1e-12:
            return 0
        return 1 - (r[i] + r[j])**2 / dist_sq

    # Generate all pairwise constraints
    constraints = []
    for i in range(n):
        for j in range(i + 1, n):
            cons = {'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_overlap(x, i, j)}
            constraints.append(cons)

    # Boundary constraints: each circle must be within [0,1]x[0,1]
    def constraint_boundary(x, i):
        pos = x.reshape((n, 2))
        r = x[n * 2:]
        x_i, y_i = pos[i]

        cons = {'type': 'ineq', 'fun': lambda x, i=i: constraint_boundary(x, i)}
        constraints.append(cons)

    # Initial guess: positions and radii
    x0 = np.concatenate([centers.reshape(-1), radii])

    # Run optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints)
