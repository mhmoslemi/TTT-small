import numpy as np
from scipy.optimize import minimize
import itertools

def run_packing():
    # Initial grid-based configuration
    grid_size = 5
    grid_spacing = 1 / (grid_size - 1)
    centers = []
    for i in range(grid_size):
        for j in range(grid_size):
            x = i * grid_spacing
            y = j * grid_spacing
            centers.append([x, y])
    centers = np.array(centers)
    radii = np.full(grid_size ** 2, 1 / (2 * grid_size))
    radii[-1] = 0.05  # Adjust the last circle to fit near the corner

    # Add one extra circle near the bottom right corner
    centers = np.vstack((centers, [0.9, 0.9]))
    radii = np.append(radii, 0.1)

    # Define objective function to maximize total radius
    def objective(x):
        return -np.sum(x)  # Minimize negative sum to maximize sum

    # Define constraints for optimization
    def constraint_overlap(x, idx1, idx2):
        dist = np.sqrt(np.sum((centers[idx1] - centers[idx2]) ** 2))
        return dist - (radii[idx1] + radii[idx2])

    # Optimization parameters
    bounds = [(0, 1) for _ in range(26 * 2)]
    constraints = []
    for i in range(26):
        for j in range(i + 1, 26):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i, j=j: constraint_overlap(x, i, j)})

    # Optimization
    initial_guess = np.concatenate((centers.flatten(), radii))
    result = minimize(objective, initial_guess, bounds=bounds, constraints=constraints, method='SLSQP')

    # Extract centers and radii from result
    optimized_centers = result.x[:26 * 2].reshape(26, 2)
    optimized_radii = result.x[26 * 2:]

    # Validate the packing
    valid, message = validate_packing(optimized_centers, optimized_radii)
    if not valid:
        raise ValueError(f"Packaging validation failed: {message}")

    return optimized_centers, optimized_radii, np.sum(optimized_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"
