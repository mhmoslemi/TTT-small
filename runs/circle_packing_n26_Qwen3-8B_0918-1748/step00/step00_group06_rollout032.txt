import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial configuration: hexagonal packing with 25 circles and 1 in the center
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)

    # Generate a hexagonal grid of 25 circles in the corners of the square
    rows = 5
    cols = 5
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1)
            y = i / (rows - 1)
            centers[i * cols + j] = [x, y]
            radii[i * cols + j] = 0.1

    # Add a central circle
    centers[-1] = [0.5, 0.5]
    radii[-1] = 0.1

    # Initial radius trial
    max_radius = 0.1
    # Objective: maximize sum of radii
    # Constraints: distance between centers >= sum of radii
    # Objective is to maximize sum(radii), we will turn it into a minimization problem

    # Define the objective function (negative of sum of radii)
    def objective(radii):
        return -np.sum(radii)

    # Define constraints
    def constraint_func(radii):
        dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
        return dist - (radii[i] + radii[j])

    # Create constraints for all pairs
    cons = []
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            cons.append({'type': 'ineq', 'fun': lambda r, i=i, j=j: constraint_func(r)})

    # Bounds: radii >= 0
    bounds = [(0, 1) for _ in range(num_circles)]

    # Optimization
    result = opt.minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-6, 'maxiter': 1000}
    )

    if not result.success:
        raise ValueError("Optimization failed with message: " + result.message)

    radii = result.x
    total_radius = -result.fun

    # Final validation
    if not validate_packing(centers, radii)[0]:
        raise ValueError("Final packing is invalid")

    return centers, radii, total_radius

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
