import scipy.optimize as optimize

    # and use optimization to refine the positions and radii for maximum sum of radii

    cols = 5
    total_circles = rows * cols
    centers = np.zeros((total_circles, 2))
    for i in range(rows):
        for j in range(cols):
            x = j * (1 / (cols - 1)) * 2 / (1 + np.sqrt(3))
            y = i * (1 / (rows - 1)) * 2 / (np.sqrt(3) + 1)
            centers[i * cols + j] = [x, y]

    # We'll use optimization to find the optimal radii while respecting constraints

    # Optimization variables: radii
    radii = np.ones(total_circles) * 0.1

    # Objective: maximize sum of radii
    # Constraints:
    # 1. Radii must be non-negative
    # 2. Each circle must be within the unit square
    # 3. Distance between centers must be at least sum of radii

    # Define bounds for radii
    bounds = [(0, 1) for _ in range(total_circles)]

    # Define the objective function
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_func(radii):
        n = len(radii)
