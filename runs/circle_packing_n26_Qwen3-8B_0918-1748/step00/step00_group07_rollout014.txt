import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist
import time

def run_packing():
    # Initial positions and radii for a hybrid grid and hexagonal packing
    centers = np.zeros((26, 2))
    radii = np.zeros(26)
    
    # Grid-based packing with 5 rows and 5 columns
    for i in range(5):
        for j in range(5):
            idx = i * 5 + j
            centers[idx, 0] = j / 4.0 + 0.05
            centers[idx, 1] = i / 4.0 + 0.05
            radii[idx] = 0.1
    
    # Add one more circle in a corner
    centers[25, 0] = 0.05
    centers[25, 1] = 0.05
    radii[25] = 0.1
    
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    def constraints(radii, centers):
        n = len(radii)
        cons = []
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                cons.append((dist - (radii[i] + radii[j]), 0.0))
        return cons
    
    # Bounds for radii to be non-negative
    bounds = [(0, 1) for _ in range(26)]
    
    # Optimization
    res = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        args=(centers,)
    )
    
    # Extract the result
    radii_opt = res.x
    centers_opt = centers.copy()
    
    # Final validation step to ensure no overlap and within square
    for i in range(26):
        x, y = centers_opt[i]
        r = radii_opt[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            raise ValueError(f"Circle {i} is outside the unit square")
    
    for i in range(26):
        for j in range(i + 1, 26):
            dx = centers_opt[i, 0] - centers_opt[j, 0]
            dy = centers_opt[i, 1] - centers_opt[j, 1]
            dist = np.sqrt(dx * dx + dy * dy)
            if dist < radii_opt[i] + radii_opt[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")
    
    return centers_opt, radii_opt, np.sum(radii_opt)
