import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds
from functools import lru_cache

def run_packing():
    # Initial grid for centers
    num_circles = 26
    grid = np.linspace(0, 1, 13)
    grid_points = np.array([[x, y] for x in grid for y in grid])
    centers = grid_points[:26]
    
    # Initial radii: equal spacing with some reduction
    initial_radii = np.full(num_circles, 0.05)
    for i in range(num_circles):
        if i % 3 == 0:
            initial_radii[i] *= 1.5
        if i % 5 == 0:
            initial_radii[i] *= 1.2
    
    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)
    
    # Constraints: non-overlapping and boundary conditions
    def constraint_overlap(radii, centers):
        n = len(centers)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers[i]
                x2, y2 = centers[j]
                dist_sq = (x1 - x2)**2 + (y1 - y2)**2
                r1, r2 = radii[i], radii[j]
                constraints.append(dist_sq - (r1 + r2)**2)
        return constraints
    
    def constraint_boundary(radii, centers):
        n = len(centers)
        constraints = []
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            constraints.append(x - r)
            constraints.append(x + r - 1)
            constraints.append(y - r)
            constraints.append(y + r - 1)
        return constraints
    
    # Optimization parameters
    bounds = Bounds([0.0] * num_circles, [np.inf] * num_circles)
    
    # Initial guess
    initial_guess = initial_radii
    
    # Run optimization
    result = minimize(
        fun=objective,
        x0=initial_guess,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap, 'args': (centers,)},
            {'type': 'ineq', 'fun': constraint_boundary, 'args': (centers,)}
        ],
        method='SLSQP'
    )
    
    # Extract results
    radii = result.x
    centers = grid_points[:26]
    
    # Validate the packing
    def validate(radii, centers):
        for i in range(len(centers)):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return False
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
                if dist < radii[i] + radii[j]:
                    return False
        return True
    
    if not validate(radii, centers):
        raise ValueError("Invalid packing found")
    
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
