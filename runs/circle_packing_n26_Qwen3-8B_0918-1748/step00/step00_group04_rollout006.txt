import numpy as np
from scipy.optimize import minimize
from scipy.spatial import Voronoi, voronoi_plot_2d
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Initial trial configuration with 16 circles in a hexagonal grid and 10 in a grid
    def initial_config():
        # Hexagonal grid of 16 circles
        hex_grid = []
        def get_hex_coords(n_rows, n_cols, row, col):
            y = (np.sqrt(3)/2) * (row - n_rows / 2)
            x = (0.5 + col - n_cols / 2)
            return x, y
        hex_grid = []
        for i in range(5):
            for j in range(4):
                x = (0.5 + j - 1.5) * 0.5
                y = (np.sqrt(3)/2) * (i - 2)
                hex_grid.append((x, y))
        hex_grid = np.array(hex_grid)
        # Add 10 circles in a grid along edges
        edge_grid = []
        for i in range(5):
            edge_grid.append((0.1, i * 0.2))
            edge_grid.append((0.9, i * 0.2))
        edge_grid = np.array(edge_grid)
        # Combine and normalize
        combined = np.vstack((hex_grid, edge_grid))
        combined[:, 0] *= 1.0 / (np.max(combined[:, 0]) - np.min(combined[:, 0]))
        combined[:, 1] *= 1.0 / (np.max(combined[:, 1]) - np.min(combined[:, 1]))
        return combined

    centers = initial_config()
    n = centers.shape[0]

    def objective(radii):
        return -np.sum(radii)

    def constraint1(radii):
        return np.sum(radii)

    def constraint2(radii):
        for i in range(n):
            if radii[i] < 0:
                return -np.inf
        return np.inf

    def constraint3(radii):
        for i in range(n):
            if centers[i, 0] - radii[i] < -1e-12 or centers[i, 0] + radii[i] > 1 + 1e-12:
                return -np.inf
            if centers[i, 1] - radii[i] < -1e-12 or centers[i, 1] + radii[i] > 1 + 1e-12:
                return -np.inf
        return np.inf

    def constraint4(radii):
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        return np.inf

    bounds = [(0, 1.0) for _ in range(n)]

    result = minimize(
        objective,
        x0=np.ones(n),
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint2},
            {'type': 'ineq', 'fun': constraint3},
            {'type': 'ineq', 'fun': constraint4}
        ],
        method='SLSQP'
    )

    radii = result.x
    sum_radii = np.sum(radii)
    centers = centers
    return (centers, radii, sum_radii)
