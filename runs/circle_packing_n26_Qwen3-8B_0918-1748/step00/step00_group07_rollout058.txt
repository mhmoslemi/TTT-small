import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial import distance
import random

def run_packing():
    # Define parameters
    n = 26
    max_iter = 100
    tol = 1e-6
    boundary_factor = 0.95
    edge_radius = 0.05

    # Initial guess: uniform grid with some perturbation
    centers = np.random.rand(n, 2)
    radii = np.full(n, 0.05)

    # Function to calculate total radius sum
    def total_radius_sum(radii):
        return np.sum(radii)

    # Function to check boundary constraints
    def check_boundary(centers, radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return False
        return True

    # Function to check circle-circle constraints
    def check_overlap(centers, radii):
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Function to update radii based on boundary and circle constraints
    def update_radii(centers, radii):
        new_radii = np.copy(radii)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Adjust radius based on distance to boundaries
            dist_left = x - 1e-12
            dist_right = 1 - x - 1e-12
            dist_bottom = y - 1e-12
            dist_top = 1 - y - 1e-12
            new_r = r * min(dist_left, dist_right, dist_bottom, dist_top) / 0.95
            new_radii[i] = new_r
        return new_radii

    # Initialize with a low radius and some boundary-aligned circles
    for i in range(n):
        if i % 5 == 0:
            # Place circles near edges
            if i < 5:
                centers[i] = [0.1, 0.1]
                radii[i] = 0.08
            elif i < 10:
                centers[i] = [0.9, 0.1]
                radii[i] = 0.08
            elif i < 15:
                centers[i] = [0.1, 0.9]
                radii[i] = 0.08
            else:
                centers[i] = [0.9, 0.9]
                radii[i] = 0.08

    # Optimization loop
    for iteration in range(max_iter):
        # Check current state
        if not check_boundary(centers, radii) or not check_overlap(centers, radii):
            break

        # Try to increase radii by moving circles
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Try to expand radius
            new_r = r * 1.1
            # Check if it fits within the square
            if x - new_r >= -1e-12 and x + new_r <= 1 + 1e-12 and y - new_r >= -1e-12 and y + new_r <= 1 + 1e-12:
                # Check if it overlaps with other circles
                overlap = False
                for j in range(n):
                    if i == j:
                        continue
                    dx = x - centers[j, 0]
                    dy = y - centers[j, 1]
                    dist = np.sqrt(dx**2 + dy**2)
                    if dist < new_r + radii[j] - 1e-12:
                        overlap = True
                        break
                if not overlap:
                    radii[i] = new_r

        # Try to move circles to better positions
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            # Try to move towards boundaries
            dist_left = x - 1e-12
            dist_right = 1 - x - 1e-12
            dist_bottom = y - 1e-12
            dist_top = 1 - y - 1e-12
            direction = np.array([dist_left, dist_right, dist_bottom, dist_top])
            direction = direction / np.max(np.abs(direction)) if np.max(np.abs(direction)) != 0 else np.array([0, 0, 0, 0])
            move = r * direction * 0.5
            new_x = x + move[0] - move[1]
            new_y = y + move[2] - move[3]
            # Ensure within bounds
            new_x = np.clip(new_x, 1e-12, 1 - 1e-12)
            new_y = np.clip(new_y, 1e-12, 1 - 1e-12)
            centers[i] = [new_x, new_y]

        # Update radii based on boundary proximity
        radii = update_radii(centers, radii)

    # Final validation
    if not check_boundary(centers, radii) or not check_overlap(centers, radii):
        raise ValueError("Invalid packing configuration")

    return centers, radii, np.sum(radii)
