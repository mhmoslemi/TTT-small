import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing():
    # Initial guess for centers using a grid with slight perturbations
    centers = np.zeros((26, 2))
    for i in range(26):
        row = i // 5
        col = i % 5
        x = col / 5 + np.random.uniform(-0.1, 0.1)
        y = row / 5 + np.random.uniform(-0.1, 0.1)
        centers[i] = [x, y]
    
    # Initial guess for radii
    radii = np.full(26, 0.05)
    
    # Function to compute the loss (we will minimize the negative of the sum of radii)
    def loss_function(radii):
        nonlocal centers
        sum_radii = np.sum(radii)
        # Expand the centers with radii to check collisions
        expanded_centers = np.column_stack((centers, radii))
        # Sort centers by x-coordinate (for efficiency)
        sorted_indices = np.argsort(centers[:, 0])
        sorted_centers = centers[sorted_indices]
        sorted_radii = radii[sorted_indices]
        # Check for overlaps
        for i in range(len(sorted_centers)):
            for j in range(i + 1, len(sorted_centers)):
                dx = sorted_centers[i, 0] - sorted_centers[j, 0]
                dy = sorted_centers[i, 1] - sorted_centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < sorted_radii[i] + sorted_radii[j] - 1e-6:
                    return -sum_radii - 1e6  # Penalize overlaps heavily
        # Check for boundary violations
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-6 or x + r > 1 + 1e-6 or y - r < -1e-6 or y + r > 1 + 1e-6:
                return -sum_radii - 1e6
        return -sum_radii
    
    # Define bounds for radii
    bounds = [(0.0, 0.3) for _ in range(26)]
    
    # Optimization settings
    result = optimize.minimize(
        loss_function,
        radii,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 200, 'gtol': 1e-4}
    )
    
    # Extract final centers and radii
    best_centers = centers
    best_radii = result.x
    
    # Final validation
    valid, msg = validate_packing(best_centers, best_radii)
    if not valid:
        raise RuntimeError("Invalid packing after optimization: " + msg)
    
    sum_radii = np.sum(best_radii)
    return best_centers, best_radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
