import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Define initial positions for 26 circles in a hybrid hexagonal + nested arrangement
    # First 4 circles at corners
    corners = np.array([
        [0.1, 0.1],
        [0.9, 0.1],
        [0.1, 0.9],
        [0.9, 0.9]
    ])
    # Then 4 circles on edges (midpoints)
    edges = np.array([
        [0.5, 0.1],
        [0.5, 0.9],
        [0.1, 0.5],
        [0.9, 0.5]
    ])
    # Then 8 circles in 2 rows of 4 (hexagonal)
    row1 = np.array([
        [0.2, 0.3],
        [0.4, 0.3],
        [0.6, 0.3],
        [0.8, 0.3]
    ])
    row2 = np.array([
        [0.2, 0.6],
        [0.4, 0.6],
        [0.6, 0.6],
        [0.8, 0.6]
    ])
    # Then 6 circles in a hexagonal arrangement
    hex1 = np.array([
        [0.3, 0.4],
        [0.5, 0.4],
        [0.7, 0.4]
    ])
    hex2 = np.array([
        [0.3, 0.5],
        [0.5, 0.5],
        [0.7, 0.5]
    ])
    # Finally, 2 circles in a staggered hexagonal arrangement
    hex3 = np.array([
        [0.3, 0.6],
        [0.7, 0.6]
    ])

    # Combine all positions into a single array
    centers = np.vstack([corners, edges, row1, row2, hex1, hex2, hex3])
    n = centers.shape[0]

    # Initial radii as 1 / (sqrt(2)*n) to be conservative
    radii = np.ones(n) * 1 / (np.sqrt(2) * n)

    # Define objective to maximize sum of radii
    def objective(params):
        # params should be a 1D array of length 2n: first n for centers, next n for radii
        # Extract centers and radii
        centers_flat = params[:2*n].reshape(n, 2)
        radii_flat = params[2*n:]
        # Objective is to maximize -sum(radii_flat) (minimize negative)
        return -np.sum(radii_flat)

    # Define constraints for boundaries and non-overlapping
    def constraint_boundary(params):
        centers_flat = params[:2*n].reshape(n, 2)
        radii_flat = params[2*n:]
        # Check if any circle goes out of bounds
        for i in range(n):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return -1
        return 0

    def constraint_overlap(params):
        centers_flat = params[:2*n].reshape(n, 2)
        radii_flat = params[2*n:]
        for i in range(n):
            for j in range(i+1, n):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < radii_flat[i] + radii_flat[j] - 1e-8:
                    return -1
        return 0

    # Set up bounds for centers and radii
    bounds = []
    for i in range(n):
        bounds.extend([(0, 1) for _ in range(2)])  # for centers
        bounds.extend([(0, 1) for _ in range(1)])  # for radii

    # Initial guess
    initial_guess = np.concatenate([centers.flatten(), radii])

    # Constraints
    cons = (
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    )

    # Minimize the negative sum of radii
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-8
    )

    # Extract the final centers and radii
    centers_result = result.x[:2*n].reshape(n, 2)
    radii_result = result.x[2*n:]

    # Validate the packing
    is_valid, message = validate_packing(centers_result, radii_result)
    if not is_valid:
        raise ValueError(f"Packing is invalid: {message}")

    sum_radii = np.sum(radii_result)

    return centers_result, radii_result, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
