import numpy as np
import math
from scipy.optimize import minimize

def run_packing():
    # Initial configuration: 26 circles in a hexagonal pattern with adjustments
    # First, create a hexagonal grid in the center and place additional circles in corners/edges
    centers = []
    radii = []
    total_sum = 0

    # Define a helper function to generate grid points in hexagon
    def generate_hexagonal_grid(side, num_rows, origin, max_radius):
        points = []
        for i in range(num_rows):
            for j in range(i + 1):
                x = origin[0] + (j - (i + 1) / 2) * side
                y = origin[1] + i * (side * math.sqrt(3) / 2)
                points.append((x, y))
        return points

    # First layer: 5 circles in a hexagonal pattern
    layer1 = generate_hexagonal_grid(0.18, 5, (0.5, 0.5), 0.1)
    # Second layer: 5 more circles around it
    layer2 = generate_hexagonal_grid(0.14, 5, (0.5, 0.5), 0.1)
    # Third layer: 5 more circles filling remaining spaces
    layer3 = generate_hexagonal_grid(0.1, 5, (0.5, 0.5), 0.05)
    # Add 6 circles in the corners
    corners = [
        (0.1, 0.1),
        (0.1, 0.9),
        (0.9, 0.9),
        (0.9, 0.1),
        (0.05, 0.5),
        (0.95, 0.5),
    ]
    corners = [np.array(p) for p in corners]

    # Combine all points
    all_points = layer1 + layer2 + layer3 + corners

    # Adjust radii for each point
    radii = []
    for p in all_points:
        r = 0.1
        if p[0] < 0.1 or p[1] < 0.1 or p[0] > 0.9 or p[1] > 0.9:
            r = 0.08
        radii.append(r)

    # Convert to numpy arrays
    centers = np.array(all_points)
    radii = np.array(radii)

    # Validate initial configuration
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)

    # Define optimization objective
    def objective(x):
        r = x[0]
        centers = np.vstack((centers, [x[1], x[2]]))
        radii = np.append(radii, r)
        valid, msg = validate_packing(centers, radii)
        if not valid:
            return -np.inf
        return np.sum(radii)

    # Run optimization (simplified for this example)
    bounds = [(0.001, 0.1), (0.05, 1), (0.05, 1)]
    result = minimize(lambda x: -objective(x), x0=[0.05, 0.5, 0.5], bounds=bounds, method='L-BFGS-B')
    optimal_r = result.x[0]
    new_center = [result.x[1], result.x[2]]

    # Update centers and radii
    centers = np.vstack((centers, new_center))
    radii = np.append(radii, optimal_r)

    # Final validation
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(msg)

    total_sum = np.sum(radii)
    return centers, radii, total_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
