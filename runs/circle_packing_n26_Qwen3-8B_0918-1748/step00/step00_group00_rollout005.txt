import numpy as np
import scipy.optimize as opt
from scipy.spatial import distance
from scipy.optimize import LinearConstraint, Bounds

def run_packing():
    n = 26
    # Initial grid-based configuration
    rows, cols = 5, 5  # 5x5 grid gives 25 circles
    spacing = 1.0 / (cols - 1)
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Place 25 circles in a grid
    for i in range(5):
        for j in range(5):
            x = i * spacing
            y = j * spacing
            centers[i * 5 + j] = [x, y]
    
    # Add one more circle to make 26
    # Place in the center of the square
    centers[25] = [0.5, 0.5]
    
    # Initial radii for grid circles
    for i in range(25):
        max_dist = min(1.0 / (cols + 1), 1.0 / (rows + 1))
        radii[i] = max_dist
    
    # Initial radius for the central circle
    radii[25] = 0.1
    
    # Function to compute total sum of radii
    def total_sum(radii):
        return np.sum(radii)
    
    # Objective: maximize sum of radii
    def objective(radii):
        return -total_sum(radii)  # Minimize negative sum
    
    # Constraint: non-overlapping
    def non_overlap(radii):
        return np.array([distance.pdist(centers) >= radii[i] + radii[j] for i in range(n) for j in range(i+1)])
    
    # Constraints for boundary
    def boundary(radii):
        return np.array([centers[i][0] - radii[i] >= 0, centers[i][0] + radii[i] <= 1 for i in range(n)])
    
    # Use a constrained optimization
    bounds = Bounds(np.zeros(n), np.inf)
    
    # Create constraints
    non_overlap_constraints = LinearConstraint(np.ones(n), -np.inf, np.inf, name='non_overlap')
    
    # Initial guess
    initial_guess = radii
    
    # Run optimization
    result = opt.minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[non_overlap_constraints],  # This is not a linear constraint but a custom one
        method='SLSQP'
    )
    
    # Update radii with optimized values
    radii = result.x
    
    # Final validation
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Final check for non-overlap and boundaries
    for i in range(n):
        if centers[i][0] - radii[i] < -1e-12 or centers[i][0] + radii[i] > 1 + 1e-12:
            raise ValueError("Circle out of bounds")
        if centers[i][1] - radii[i] < -1e-12 or centers[i][1] + radii[i] > 1 + 1e-12:
            raise ValueError("Circle out of bounds")
    
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt((centers[i][0] - centers[j][0])**2 + (centers[i][1] - centers[j][1])**2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError("Circles overlap")
    
    return (centers, radii, total_sum(radii))
