import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Initial guess for centers (hexagonal grid with 6 rows)
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    
    # Create initial hexagonal packing
    rows = 6
    cols = [5, 6, 6, 6, 6, 5]
    total_cols = sum(cols)
    
    for i in range(rows):
        row_index = i
        col_index = i
        if i % 2 == 0:
            col_index = i
        else:
            col_index = i - 1
        
        for j in range(cols[i]):
            col = j
            row = i
            x = col * (1 + (1 / (total_cols - 1))) * 0.5
            y = row * (np.sqrt(3) / 2) * 0.5
            centers[i * cols[i] + j] = [x, y]
    
    # Initial radius guess
    initial_radius = 0.2
    radii[:] = initial_radius
    
    # Objective function to maximize sum of radii
    def objective(x):
        # x contains [x1, y1, r1, x2, y2, r2, ...]
        # Extract centers and radii
        centers_flat = x[:n*2].reshape((n, 2))
        radii_flat = x[n*2:]
        
        # Calculate total sum of radii
        total = np.sum(radii_flat)
        return -total  # Minimize negative to maximize
    
    # Constraints: non-overlapping circles and within unit square
    def constraint(x):
        centers_flat = x[:n*2].reshape((n, 2))
        radii_flat = x[n*2:]
        
        # Check if all circles are within the unit square
        for i in range(n):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -1
        
        # Check if all circles are non-overlapping
        distances = squareform(pdist(centers_flat))
        for i, j in combinations(range(n), 2):
            if distances[i, j] < radii_flat[i] + radii_flat[j] - 1e-12:
                return -1
        
        return 0
    
    # Optimization setup
    bounds = []
    for i in range(n):
        bounds.append((0, 1))  # x
        bounds.append((0, 1))  # y
        bounds.append((0, 1))  # r
    
    # Initial guess: centers from hexagonal grid plus radii
    initial_guess = []
    for i in range(n):
        initial_guess.extend(centers[i].tolist())
        initial_guess.append(radii[i])
    
    # Optimization
    result = minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=[{'type': 'ineq', 'fun': constraint}]
    )
    
    # Extract final centers and radii
    centers_final = result.x[:n*2].reshape((n, 2))
    radii_final = result.x[n*2:]
    
    # Calculate total sum of radii
    sum_radii = np.sum(radii_final)
    
    # Validate packing for final check (this is just for validation, not for the actual output)
    valid, _ = validate_packing(centers_final, radii_final)
    
    return centers_final, radii_final, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
