import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import cdist

def run_packing():
    """
    Packs 26 circles in a unit square [0,1]x[0,1] to maximize the total sum of radii.
    The packing uses a hybrid approach with boundary circles, edge circles, and interior circles in a hexagonal-like arrangement.
    We optimize the positions and radii to maximize the total sum of radii under non-overlap and boundary constraints.
    """

    # Define initial positions and radii estimates
    num_circles = 26

    # Corner circles (4 total)
    corners = np.array([
        [0, 0],
        [1, 0],
        [1, 1],
        [0, 1]
    ])

    # Edge circles (6 per side minus corners, total 24 edges but we will use 6 per side)
    edge_spacings = np.linspace(0, 1, 7)[1:-1]  # 6 points per side, excluding corners

    # Bottom edge
    bottom_edge = np.array([[x, 0] for x in edge_spacings])
    # Top edge
    top_edge = np.array([[x, 1] for x in edge_spacings])
    # Left edge
    left_edge = np.array([[0, y] for y in edge_spacings])
    # Right edge
    right_edge = np.array([[1, y] for y in edge_spacings])

    # Combine all edge circles
    edge_circles = np.vstack([bottom_edge, top_edge, left_edge, right_edge])

    # Interior circles (26 - 4 - 24 = -2, wait this is not correct - let's adjust. The total is 26)
    # Let's use hexagonal packing for interior circles, with 12 circles

    num_interior = 26 - len(corners) - len(edge_circles)

    # Interior circles - hexagonal grid with 3 rows of 4, 4 rows of 3
    interior = []
    for i in range(3):
        for j in range(4):
            interior.append([0.1 + j * 0.25, 0.2 + i * 0.4])
        for j in range(3):
            interior.append([0.25 + j * 0.5, 0.4 + i * 0.4])
    interior = np.array(interior)[:num_interior]

    # Combine all circles
    centers = np.vstack([corners, edge_circles, interior])

    # Initial radii guess: corners get large radii, edges get moderate, interior get smaller
    radii = np.zeros_like(centers)
    radii[:4] = np.random.uniform(0.1, 0.2, size=4)
    radii[4:4 + 24] = np.random.uniform(0.05, 0.15, size=24)
    radii[4 + 24:] = np.random.uniform(0.02, 0.08, size=num_interior)

    # Define objective function
    def objective(x):
        # x contains both centers and radii: first n*2 values are centers, then radii
        n = len(centers)
        flat = x.reshape((n * 2 + n, ))
        centers_flat = flat[:n * 2].reshape((n, 2))
        radii_flat = flat[n * 2:].reshape((n, ))
        return -np.sum(radii_flat)

    # Define constraints
    def constraint_boundary(x):
        n = len(centers)
        flat = x.reshape((n * 2 + n, ))
        centers_flat = flat[:n * 2].reshape((n, 2))
        radii_flat = flat[n * 2:].reshape((n, ))
        for i in range(n):
            x, y = centers_flat[i]
            r = radii_flat[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return -1
        return 0

    def constraint_overlap(x):
        n = len(centers)
        flat = x.reshape((n * 2 + n, ))
        centers_flat = flat[:n * 2].reshape((n, 2))
        radii_flat = flat[n * 2:].reshape((n, ))
        for i in range(n):
            for j in range(i + 1, n):
                x1, y1 = centers_flat[i]
                x2, y2 = centers_flat[j]
                r1 = radii_flat[i]
                r2 = radii_flat[j]
                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < r1 + r2 - 1e-12:
                    return -1
        return 0

    # Combine constraints
    cons = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]

    # Initial guess: flatten centers and radii
    initial_guess = np.concatenate([centers.reshape(-1), radii])

    # Optimize
    result = opt.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        constraints=cons,
        options={'ftol': 1e-8, 'disp': False}
    )

    # Reshape result to centers and radii
    final_guess = result.x
    centers_final = final_guess[:len(centers)*2].reshape((len(centers), 2))
    radii_final = final_guess[len(centers)*2:].reshape((len(centers), ))

    # Check if optimization succeeded
    if not result.success:
        raise RuntimeError("Optimization failed: " + result.message)

    # Ensure non-negative radii
    radii_final[radii_final < 0] = 0

    # Final validation
    valid, msg = validate_packing(centers_final, radii_final)
    if not valid:
        raise RuntimeError("Validation failed: " + msg)

    sum_radii = np.sum(radii_final)

    return centers_final, radii_final, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
