import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt
import random

def run_packing():
    # Initial guess: a hexagonal packing with some adjustments
    def initial_guess(n):
        centers = []
        radii = []
        for i in range(n):
            angle = 2 * np.pi * i / n
            r = 0.1 + 0.05 * (i % 4)
            x = 0.5 + np.cos(angle) * (0.5 - r)
            y = 0.5 + np.sin(angle) * (0.5 - r)
            centers.append([x, y])
            radii.append(r)
        return np.array(centers), np.array(radii)

    # Objective function: maximize sum of radii, so minimize negative sum
    def objective(radii, centers):
        return -np.sum(radii)

    # Constraint function: ensure circles do not overlap and stay within square
    def constraint(radii, centers):
        n = len(radii)
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                return -1000000  # Penalize for going out of bounds
            for j in range(i + 1, n):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    return -1000000  # Penalize for overlap
        return 0

    # Initial guess for centers and radii
    centers_initial, radii_initial = initial_guess(26)

    # Optimization setup
    n = 26
    bounds = [(0, 1)] * 2 * n  # Bounds for centers and radii

    # Define optimization problem
    def optimize():
        # initial guess for centers and radii
        # Flatten the initial guess
        flat_initial = np.concatenate([centers_initial.flatten(), radii_initial])

        def obj_func(x):
            centers = x[:2 * n].reshape(n, 2)
            radii = x[2 * n:]
            return objective(radii, centers)

        def cons_func(x):
            centers = x[:2 * n].reshape(n, 2)
            radii = x[2 * n:]
            return constraint(radii, centers)

        # Setup the optimization
        result = minimize(obj_func, flat_initial, method='L-BFGS-B', bounds=bounds, constraints=[{'type': 'ineq', 'fun': cons_func}])
        if not result.success:
            raise RuntimeError("Optimization failed")

        centers_opt = result.x[:2 * n].reshape(n, 2)
        radii_opt = result.x[2 * n:]
        return centers_opt, radii_opt, np.sum(radii_opt)

    # Run the optimization
    centers, radii, sum_radii = optimize()
    return centers, radii, sum_radii
