import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds
import time

def run_packing():
    # Initial grid layout: 25 circles in 5x5 grid, 26th circle at top-right corner
    grid_size = 5
    grid_spacing = 1.0 / grid_size
    centers = []
    radii = []

    # Place 25 circles in a 5x5 grid
    for i in range(grid_size):
        for j in range(grid_size):
            x = j * grid_spacing + grid_spacing / 2
            y = i * grid_spacing + grid_spacing / 2
            centers.append([x, y])
            radii.append(0.0)

    # Place the 26th circle at the top-right corner
    x, y = 1 - grid_spacing / 2, 1 - grid_spacing / 2
    centers.append([x, y])
    radii.append(0.0)

    centers = np.array(centers)
    radii = np.array(radii)

    # Optimization parameters
    bounds = [(0, 1) for _ in range(2 * len(centers))]
    constraints = []

    # Objective function: maximize sum of radii
    def objective(x):
        return -np.sum(x[1::2])  # Minimize negative sum to maximize positive sum

    # Bounds for centers and radii
    center_bounds = [(0, 1) for _ in range(2 * len(centers))]
    radius_bounds = [(0, 1) for _ in range(len(centers))]

    # Constraints for circles to be inside the square
    def center_bounds_func(x):
        for i in range(len(centers)):
            x_i = x[2 * i]
            y_i = x[2 * i + 1]
            if x_i < 0 or x_i > 1 or y_i < 0 or y_i > 1:
                return -1
        return 0

    # Constraint for circles to be fully inside the square
    constraints.append({
        'type': 'ineq',
        'fun': center_bounds_func
    })

    # Constraints for circle radii
    def radius_bounds_func(x):
        for i in range(len(radii)):
            if x[2 * len(centers) + i] < 0:
                return -1
        return 0

    constraints.append({
        'type': 'ineq',
        'fun': radius_bounds_func
    })

    # Constraint for no overlaps between circles
    def no_overlap_func(x):
        centers_flat = x[:2 * len(centers)]
        radii_flat = x[2 * len(centers):]
        for i in range(len(centers_flat) // 2):
            x1, y1 = centers_flat[2 * i], centers_flat[2 * i + 1]
            r1 = radii_flat[i]
            for j in range(i + 1, len(centers_flat) // 2):
                x2, y2 = centers_flat[2 * j], centers_flat[2 * j + 1]
                r2 = radii_flat[j]
                dx = x1 - x2
                dy = y1 - y2
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < r1 + r2 - 1e-12:
                    return -1
        return 0

    constraints.append({
        'type': 'ineq',
        'fun': no_overlap_func
    })

    # Optimization
    try:
        result = optimize.minimize(
            objective,
            x0=np.concatenate([centers.flatten(), np.zeros(len(radii))]),
            bounds=Bounds(center_bounds + radius_bounds),
            constraints=constraints,
            method='SLSQP',
            options={'ftol': 1e-10, 'maxiter': 200}
        )
    except:
        raise RuntimeError("Optimization failed")

    # Extract centers and radii
    opt_centers = result.x[:2 * len(centers)].reshape(-1, 2)
    opt_radii = result.x[2 * len(centers):]

    # Ensure radii are non-negative
    opt_radii = np.maximum(opt_radii, 0)

    # Ensure circles are fully within the unit square
    for i in range(len(opt_centers)):
        x, y = opt_centers[i]
        r = opt_radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            opt_radii[i] = np.minimum(r, np.minimum(1 - x, 1 - y))

    return opt_centers, opt_radii, np.sum(opt_radii)
