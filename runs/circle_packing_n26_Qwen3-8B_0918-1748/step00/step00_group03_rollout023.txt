import numpy as np
import random
import time
from scipy.optimize import minimize

def run_packing():
    # Initial configuration: place circles in corners and along edges
    num_circles = 26
    centers = np.zeros((num_circles, 2))
    radii = np.zeros(num_circles)
    
    # Place 4 circles in corners (smaller radius)
    corner_radii = 0.05
    for i in range(4):
        if i == 0:
            centers[i] = [corner_radii, corner_radii]
        elif i == 1:
            centers[i] = [1 - corner_radii, corner_radii]
        elif i == 2:
            centers[i] = [corner_radii, 1 - corner_radii]
        else:
            centers[i] = [1 - corner_radii, 1 - corner_radii]
        radii[i] = corner_radii
    
    # Place 18 circles along edges in hexagonal fashion
    edge_spacing = 0.15
    for i in range(4, num_circles):
        if i % 2 == 0:
            pos = i // 2
            if pos < 9:
                centers[i] = [pos * edge_spacing, 0.5]
            else:
                centers[i] = [1 - (pos - 9) * edge_spacing, 0.5]
        else:
            pos = (i - 1) // 2
            if pos < 9:
                centers[i] = [0.5, pos * edge_spacing]
            else:
                centers[i] = [0.5, 1 - (pos - 9) * edge_spacing]
        radii[i] = 0.075
    
    # Optimization with gradient descent and simulated annealing
    def objective(radii):
        nonlocal centers
        current_radii = radii
        for i in range(num_circles):
            if current_radii[i] < 0:
                return -np.inf
        for i in range(num_circles):
            x, y = centers[i]
            r = current_radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
                return -np.inf
        
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist < current_radii[i] + current_radii[j] - 1e-12:
                    return -np.inf
        
        return np.sum(current_radii)
    
    def gradient(radii):
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(radii)
        for i in range(num_circles):
            radii_plus = radii.copy()
            radii_plus[i] += eps
            obj_plus = objective(radii_plus)
            radii_minus = radii.copy()
            radii_minus[i] -= eps
            obj_minus = objective(radii_minus)
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        return grad
    
    # Initial optimization with gradient descent
    result = minimize(
        fun=objective,
        x0=radii,
        method='L-BFGS-B',
        jac=gradient,
        bounds=[(0, 1) for _ in range(num_circles)],
        options={'maxiter': 100, 'ftol': 1e-10, 'gtol': 1e-8}
    )
    
    radii = result.x
    centers = np.array(centers)
    
    # Final validation
    for i in range(num_circles):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            raise Exception("Circle outside the unit square")
    
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx*dx + dy*dy)
            if dist < radii[i] + radii[j] - 1e-12:
                raise Exception("Circles overlap")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
