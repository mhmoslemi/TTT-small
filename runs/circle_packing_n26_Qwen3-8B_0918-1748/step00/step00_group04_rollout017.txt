import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial import distance
from scipy.optimize import Bounds

def run_packing():
    def energy_function(params, n_circles):
        centers = params[:n_circles*2].reshape(n_circles, 2)
        radii = params[n_circles*2:].reshape(n_circles)
        
        # Boundary energy
        boundary_energy = 0.0
        for i in range(n_circles):
            x, y = centers[i]
            r = radii[i]
            if x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1:
                boundary_energy += 1000.0
        
        # Overlap energy
        overlap_energy = 0.0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                r_sum = radii[i] + radii[j]
                if dist < r_sum - 1e-8:
                    overlap_energy += (r_sum - dist) ** 2
        
        # Repulsion energy between centers
        repulsion_energy = 0.0
        for i in range(n_circles):
            for j in range(i + 1, n_circles):
                dx = centers[i][0] - centers[j][0]
                dy = centers[i][1] - centers[j][1]
                dist = np.sqrt(dx*dx + dy*dy)
                if dist > 1e-8:
                    repulsion_energy += 1.0 / dist
        
        return boundary_energy + overlap_energy + repulsion_energy

    n_circles = 26
    initial_centers = np.random.rand(n_circles, 2)
    initial_radii = np.random.rand(n_circles) * 0.1
    initial_params = np.concatenate((initial_centers.flatten(), initial_radii))

    # Bounds: centers must be in [0, 1]^2 and radii must be non-negative
    bounds = []
    for _ in range(n_circles):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])  # x and y coordinates for each circle
        bounds.extend([(0.0, 1.0)])             # radius for each circle

    # Minimize energy to find optimal configuration
    result = minimize(energy_function, initial_params, args=(n_circles,), bounds=bounds, method='L-BFGS-B', tol=1e-8, options={'maxiter': 1000})

    centers = result.x[:n_circles*2].reshape(n_circles, 2)
    radii = result.x[n_circles*2:].reshape(n_circles)
    sum_radii = np.sum(radii)

    # Validate packing
    def validate_packing(centers, radii):
        n = centers.shape[0]
        if np.isnan(centers).any() or np.isnan(radii).any():
            return False, "NaN values present"
        for i in range(n):
            if radii[i] < 0:
                return False, f"Circle {i} has negative radius {radii[i]}"
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if (x - r < -1e-12 or x + r > 1 + 1e-12 or
                y - r < -1e-12 or y + r > 1 + 1e-12):
                return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False, f"Circles {i} and {j} overlap"
        return True, "ok"

    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing validation failed: {msg}")

    return (centers, radii, sum_radii)
