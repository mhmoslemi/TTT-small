import numpy as np
import cvxpy as cp

def run_packing():
    # Define variables for optimization
    n = 26
    radii = cp.Variable(n)
    centers = cp.Variable((n, 2))

    # Objective: maximize sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Constraints
    constraints = []

    # Boundary constraints: centers within [0,1]^2
    for i in range(n):
        constraints.append(centers[i, 0] >= 0)
        constraints.append(centers[i, 0] <= 1)
        constraints.append(centers[i, 1] >= 0)
        constraints.append(centers[i, 1] <= 1)

    # Non-overlap constraints: distance between centers >= sum of radii
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
            constraints.append(dist >= radii[i] + radii[j] - 1e-8)

    # Non-negativity of radii
    constraints.append(radii >= 0)

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(max_iter=1000, eps=1e-6)

    # Extract the optimal centers and radii
    centers_opt = centers.value
    radii_opt = radii.value

    # Compute the total sum of radii
    sum_radii = np.sum(radii_opt)

    return (centers_opt, radii_opt, sum_radii)
