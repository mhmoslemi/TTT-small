import numpy as np
from scipy.optimize import minimize
import warnings

def run_packing():
    # Define the initial positions using staggered hexagonal grid
    # First row: 13 circles spaced horizontally
    # Second row: 13 circles offset horizontally and vertically
    spacing = 0.25
    offset_x = spacing / 2
    offset_y = spacing * np.sqrt(3) / 2

    centers = []
    for i in range(13):
        x = i * spacing
        y = 0.25
        centers.append([x, y])
    for i in range(13):
        x = i * spacing + offset_x
        y = 0.75
        centers.append([x, y])
    centers = np.array(centers)

    # Initial uniform radius based on minimum distance to boundary
    def compute_initial_radius(x, y):
        margin = 0.25  # Approximately
        return min(margin, x, 1 - x, y, 1 - y)

    radii = np.array([compute_initial_radius(x, y) for x, y in centers])

    # Define constraints
    def constraint_circle_boundary(x, y, r):
        return x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1

    # Objective function to maximize sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Constraints function: distance >= sum of radii
    def constraint_distance(centers, radii):
        n = len(centers)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j])**2))
                constraints.append(dist - (radii[i] + radii[j]) >= 1e-8)
        return constraints

    # Prepare constraints and bounds
    bounds = [(0, 1) for _ in range(26)]
    cons = []
    for i in range(len(centers)):
        x, y = centers[i]
        cons.append({'type': 'ineq', 'fun': lambda r, i=i: r[i] - (np.min([x, 1 - x, y, 1 - y]))})
    for c in constraint_distance(centers, radii):
        cons.append(c)

    # Optimization
    with warnings.catch_warnings():
        warnings.simplefilter("ignore")
        result = minimize(objective, radii, bounds=bounds, constraints=cons)

    # Extract results
    if not result.success:
        raise RuntimeError("Optimization failed")
    radii = result.x
    sum_radii = np.sum(radii)

    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise RuntimeError(msg)

    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
