import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Initial grid-based placement with some corner adjustments
    num_circles = 26
    grid_size = 5
    grid = np.linspace(0, 1, grid_size)
    centers = []
    radii = []

    # Use a hybrid grid with hexagonal packing and corner adjustments
    for i in range(grid_size):
        for j in range(grid_size):
            x = grid[i]
            y = grid[j]
            if i % 2 == 0:
                x += 0.1
            if j % 2 == 0:
                y += 0.1
            # Avoid placing near edges
            if (x < 0.1 and y < 0.1) or (x > 0.9 and y < 0.1) or (x < 0.1 and y > 0.9) or (x > 0.9 and y > 0.9):
                continue
            centers.append([x, y])
            radii.append(0.05)

    # Add corner circles
    for i in [0, 1, 2]:
        for j in [0, 1, 2]:
            if i == 1 and j == 1:
                continue
            x = 0.1 + 0.1 * i
            y = 0.1 + 0.1 * j
            centers.append([x, y])
            radii.append(0.05)

    # Ensure the total number of circles is 26
    if len(centers) < num_circles:
        # Add more circles in the middle
        for _ in range(num_circles - len(centers)):
            x = np.random.uniform(0.3, 0.7)
            y = np.random.uniform(0.3, 0.7)
            centers.append([x, y])
            radii.append(0.05)

    # Initial guess
    centers = np.array(centers)
    radii = np.array(radii)

    # Optimization parameters
    bounds = [(0, 1) for _ in range(2 * num_circles)]
    def objective(x):
        centers_flat = x[:2 * num_circles].reshape((num_circles, 2))
        radii_flat = x[2 * num_circles:]
        return -np.sum(radii_flat)

    def constraint1(x):
        centers_flat = x[:2 * num_circles].reshape((num_circles, 2))
        radii_flat = x[2 * num_circles:]
        for i in range(num_circles):
            x_i, y_i = centers_flat[i]
            r_i = radii_flat[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12:
                return -1
            if y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -1
        return 0

    def constraint2(x):
        centers_flat = x[:2 * num_circles].reshape((num_circles, 2))
        radii_flat = x[2 * num_circles:]
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                dx = centers_flat[i, 0] - centers_flat[j, 0]
                dy = centers_flat[i, 1] - centers_flat[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii_flat[i] + radii_flat[j] - 1e-12:
                    return -1
        return 0

    # Initial guess for optimization
    initial_guess = np.concatenate([centers.flatten(), radii])

    # Set up optimization problem
    cons = [{'type': 'ineq', 'fun': constraint1}, {'type': 'ineq', 'fun': constraint2}]
    result = minimize(objective, initial_guess, bounds=bounds, constraints=cons, method='SLSQP')

    # Extract final centers and radii
    final_centers = result.x[:2 * num_circles].reshape((num_circles, 2))
    final_radii = result.x[2 * num_circles:]

    # Check if the result is valid
    if not validate_packing(final_centers, final_radii)[0]:
        raise ValueError("Optimization failed to produce a valid packing.")

    # Return the result
    return final_centers, final_radii, np.sum(final_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
