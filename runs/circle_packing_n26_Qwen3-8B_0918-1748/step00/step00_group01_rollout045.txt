import numpy as np
from scipy.optimize import minimize
import math

def run_packing():
    # Define the number of circles
    n = 26
    
    # Initial guess: layer-based hexagonal packing with some adjustments
    # First, place 25 circles in a hexagonal grid with radii r
    # Then, place the 26th circle in a gap
    # We'll use a nested approach with some symmetry
    
    # Start configuration: 15 circles in the bottom layer, 10 in the middle, 1 on top (just an initial guess)
    # All radii start as same value for initial guess
    # Optimize radii and positions
    
    # Define initial positions and radii
    # Layer 1: 15 circles in the bottom layer
    # Layer 2: 10 circles in the middle layer
    # Layer 3: 1 circle on top
    # Use hexagonal grid with spacing of 2r
    
    # Use SLSQP optimization with constraints
    
    # Define variables: positions (x, y) and radii
    
    # Create initial guess for positions
    # For each layer, generate a hexagonal grid
    
    # Define initial positions for layer 1 (bottom) with 15 circles
    # Radii will be optimized
    # Assume a square of side length approx 0.8 for layer 1
    layer1_pos = []
    for i in range(3):
        for j in range(5):
            x = 0.1 + i * (1.0 / 3)
            y = 0.1 + j * (0.2)
            layer1_pos.append([x, y])
    
    # Define initial positions for layer 2 (middle) with 10 circles
    # Assume a square of side length approx 0.5
    layer2_pos = []
    for i in range(2):
        for j in range(5):
            x = 0.3 + i * (0.4)
            y = 0.5 + j * (0.1)
            layer2_pos.append([x, y])
    
    # Define initial positions for layer 3 (top) with 1 circle
    layer3_pos = []
    layer3_pos.append([0.5, 0.9])
    
    # Combine all positions
    initial_pos = np.array(layer1_pos + layer2_pos + layer3_pos)
    
    # Define initial guess for radii - same for all circles
    initial_r = np.full(n, 0.04)
    
    # Define variables: positions and radii
    # reshape into flat list for optimization
    bounds = []
    for _ in range(n):
        bounds.extend([(0.0, 1.0)] * 2)
        bounds.extend([(0.0, 0.5)])  # radius cannot exceed 0.5
    
    # Objective function: maximize sum of radii (minimize -sum)
    def objective(x):
        # x contains positions (flatten) and radii
        # reshape to get positions and radii
        pos = x[:2 * n].reshape((n, 2))
        radii = x[2 * n:]
        return -np.sum(radii)
    
    # Constraints: distance between centers must be >= sum of radii
    # Also, centers must be at least radius away from the boundary
    # Define constraints
    def constraint_func(x):
        pos = x[:2 * n].reshape((n, 2))
        radii = x[2 * n:]
        # Check boundary constraints
        for i in range(n):
            x_i, y_i = pos[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -np.inf
        # Check non-overlap constraints
        for i in range(n):
            for j in range(i + 1, n):
                dx = pos[i, 0] - pos[j, 0]
                dy = pos[i, 1] - pos[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        return 0
    
    # Optimization
    x0 = np.concatenate([initial_pos.flatten(), initial_r])
    
    res = minimize(objective, x0, bounds=bounds, constraints={'type': 'ineq', 'fun': constraint_func})
    
    # Extract the optimal positions and radii
    centers = res.x[:2 * n].reshape((n, 2))
    radii = res.x[2 * n:]
    
    # Compute total sum of radii
    total_sum = np.sum(radii)
    
    # Ensure the solution is valid
    # If the optimization is not successful, return degenerate values
    if not np.isfinite(total_sum):
        return np.array([]), np.array([]), 0.0
    
    # Return results
    return centers, radii, total_sum
