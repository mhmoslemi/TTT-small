import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize
from itertools import combinations

def run_packing():
    def compute_min_distance_to_boundary(center):
        x, y = center
        dist_to_left = x
        dist_to_right = 1 - x
        dist_to_bottom = y
        dist_to_top = 1 - y
        return min(dist_to_left, dist_to_right, dist_to_bottom, dist_to_top)

    def compute_min_distance_to_other_circles(center, centers, radii):
        min_dist = np.inf
        for i in range(len(centers)):
            dx = centers[i, 0] - center[0]
            dy = centers[i, 1] - center[1]
            dist = np.sqrt(dx**2 + dy**2)
            min_dist = min(min_dist, dist - radii[i])
        return min_dist

    def fitness_function(params):
        centers = params[:26*2].reshape(26, 2)
        radii = params[26*2:]
        sum_radii = sum(radii)
        for i in range(26):
            x, y = centers[i]
            r = radii[i]
            if r < 0:
                return -np.inf
            if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
                return -np.inf
        for i in range(26):
            for j in range(i+1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        return sum_radii

    initial_centers = np.random.rand(26, 2)
    initial_radii = np.random.rand(26) * 0.1
    initial_params = initial_centers.reshape(-1).tolist() + initial_radii.tolist()

    result = minimize(fitness_function, initial_params, method='L-BFGS-B', options={'maxiter': 1000})
    
    centers = result.x[:26*2].reshape(26, 2)
    radii = result.x[26*2:]
    sum_radii = sum(radii)
    
    return (centers, radii, sum_radii)
