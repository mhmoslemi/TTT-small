import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
import cvxpy as cp

def run_packing():
    # Initial hexagonal packing with 26 circles
    n = 26
    max_radius = 0.0
    centers = []
    radii = []
    # Hexagonal grid with row-wise arrangement
    rows = int(np.ceil(np.sqrt(n)))
    cols = int(np.ceil(n / rows))
    spacing = 0.5
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + 0.5 * (i % 2)
            y = i * spacing * np.sqrt(3)
            if i % 2 == 1:
                x += 0.5 * spacing
            if 0 <= x < 1 and 0 <= y < 1:
                centers.append([x, y])
                radii.append(0.1)
                max_radius = max(max_radius, 0.1)
    centers = np.array(centers)
    radii = np.array(radii)
    while len(centers) < n:
        # Add more circles in the remaining space
        # A simple heuristic to add circles in the center if there is space
        if len(centers) < n:
            # Find the largest empty space
            # In this case, we'll add in the center if possible
            cx, cy = 0.5, 0.5
            r = 0.1
            valid = True
            for i in range(len(centers)):
                dx = cx - centers[i, 0]
                dy = cy - centers[i, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radi[i] + r - 1e-8:
                    valid = False
                    break
            if valid:
                centers = np.vstack([centers, [cx, cy]])
                radii = np.append(radii, r)
    
    # Optimization to maximize sum of radii
    # We mix gradient descent and constraints
    def objective(radii):
        return -np.sum(radii)  # Minimize negative to maximize

    def constraints(centers, radii):
        return [
            (centers[i, 0] - radii[i] >= 0, centers[i, 0] + radii[i] <= 1, 
             centers[i, 1] - radii[i] >= 0, centers[i, 1] + radii[i] <= 1)
            for i in range(len(radii))
        ]
    
    # Define CVXPY variables
    x = cp.Variable((n, 2))
    r = cp.Variable(n)
    obj = cp.Maximize(cp.sum(r))
    constraints = []
    # Each circle is within the unit square
    for i in range(n):
        constraints += [
            x[i, 0] - r[i] >= 0,
            x[i, 0] + r[i] <= 1,
            x[i, 1] - r[i] >= 0,
            x[i, 1] + r[i] <= 1
        ]
    # Pairwise circle non-overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2)
            constraints += [dist >= r[i] + r[j]]
    problem = cp.Problem(obj, constraints)
    problem.solve(solver=cp.SCS, eps=1e-5)
    
    centers = x.value
    radii = r.value
    if centers is None or radii is None:
        return centers, radii, 0.0
    
    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        return np.array([[0.5, 0.5]]), np.array([0.0]), 0.0
    
    return centers, radii, np.sum(radii)

def validate_packing(centers, radii):
    n = centers.shape[0]
    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"
    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12 or
            y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"
    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"
    return True, "ok"
