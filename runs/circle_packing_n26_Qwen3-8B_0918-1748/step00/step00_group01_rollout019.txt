from scipy.optimize import minimize, Bounds, NonlinearConstraint
from scipy.spatial.distance import pdist, squareform

def run_packing():

    # We'll adjust this later with optimization
    num_circles = 26
    grid = np.linspace(0, 1, 13)

        row = i // 13
        col = i % 13

            centers[i, 1] = row * (1 / (num_circles ** 0.5)) + (1 / (num_circles ** 0.5)) / 2

    max_radius = 0.1

        centers = x[:2*num_circles].reshape(num_circles, 2)
        radii = x[2*num_circles:]
        distances = squareform(pdist(centers))

    result = minimize(

        x0,
        bounds=bounds,
        constraints=[constraint1, constraint2],
