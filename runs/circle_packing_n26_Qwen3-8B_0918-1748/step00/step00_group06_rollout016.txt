import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import matplotlib.pyplot as plt

def run_packing():
    # Define initial grid parameters for hexagonal packing
    num_circles = 26
    rows = 13
    cols = 2

    # Initial positions for hexagonal packing
    centers = np.zeros((num_circles, 2))
    row_height = np.sqrt(3) / 2
    row_offset = 0.5

    for i in range(rows):
        for j in range(cols):
            row = i
            col = j
            if i % 2 == 0:
                x = col + 0.5
            else:
                x = col + 0.5 - row_offset
            y = i * row_height
            centers[i * cols + j] = [x, y]

    # Define the function to maximize the sum of radii
    def objective(radii):
        return -np.sum(radii)

    # Define constraints
    def constraint1(radii):
        return np.sum(radii)

    def constraint2(radii):
        return np.sum(radii) - 2.636

    def constraint3(radii):
        return np.sum(radii) - 2.636

    # Define bounds for radii
    bounds = [(0, 1)] * num_circles

    # Constraint to ensure the total sum of radii is at most 2.636
    cons = [{'type': 'ineq', 'fun': lambda x: 2.636 - np.sum(x)}]

    # Optimization
    result = minimize(objective, x0=np.ones(num_circles), bounds=bounds, constraints=cons)

    # Reconstruct the centers and radii from the optimization result
    radii = result.x

    # Final validation before returning
    centers_final = centers
    radii_final = radii

    # Ensure all circles are within the square
    for i in range(num_circles):
        x, y = centers_final[i]
        r = radii_final[i]
        if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
            raise ValueError(f"Circle {i} is outside the unit square")

    # Ensure no overlap between circles
    for i in range(num_circles):
        for j in range(i + 1, num_circles):
            dist = np.sqrt(np.sum((centers_final[i] - centers_final[j]) ** 2))
            if dist < radii_final[i] + radii_final[j] - 1e-8:
                raise ValueError(f"Circles {i} and {j} overlap")

    return centers_final, radii_final, np.sum(radii_final)
