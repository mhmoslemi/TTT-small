import numpy as np
import scipy.optimize as opt
from scipy.spatial.distance import cdist
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    def hexagonal_grid(n):
        rows = int(np.ceil(n / 6))
        cols = int(np.ceil(n / rows))
        centers = np.zeros((n, 2))
        row = 0
        col = 0
        for i in range(n):
            if i % 6 == 0:
                row += 1
                col = 0
            centers[i] = [col / (cols - 1), row / (rows - 1)]
            col += 1
        return centers

    def adjust_for_boundary(centers, radii):
        new_centers = np.copy(centers)
        new_radii = np.copy(radii)
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            max_r = min(1 - x, x, 1 - y, y)
            if max_r < r:
                new_radii[i] = max_r
                new_centers[i] = [x, y]
        return new_centers, new_radii

    def optimize_radii(centers, max_radii, max_iter=100, tol=1e-5):
        n = len(centers)
        radii = np.full(n, max_radii)
        for _ in range(max_iter):
            for i in range(n):
                dx = np.sqrt((centers[i, 0] - centers[:, 0])**2 + (centers[i, 1] - centers[:, 1])**2)
                dx = dx[dx != 0]
                sum_overlap = np.sum(radii[dx < radii[i] + 1e-12])
                if sum_overlap + radii[i] > 1:
                    radii[i] = 1 - sum_overlap
            if np.abs(np.sum(radii) - np.sum(radii)) < tol:
                break
        return radii

    centers = hexagonal_grid(26)
    max_radii = 1 / (np.sqrt(3) + 2)
    centers, radii = adjust_for_boundary(centers, max_radii)
    radii = optimize_radii(centers, max_radii)
    max_sum = np.sum(radii)
    return (centers, radii, max_sum)
