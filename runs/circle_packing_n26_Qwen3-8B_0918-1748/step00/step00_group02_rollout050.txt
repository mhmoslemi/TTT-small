import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run_packing():
    # Define a function that computes the total sum of radii to be maximized
    def total_sum(radii):
        return -np.sum(radii)  # Negative for minimization

    # Define the initial positions and radii as a guess
    # We start with a hexagonal cluster of 19 circles in the left half of the square
    # and 7 circles in the top-right corner

    # Generate initial guess for centers and radii
    # Set the initial radii to 1/10 for all to ensure they don't overlap initially
    radii_guess = np.full(26, 0.1)
    centers_guess = np.zeros((26, 2))

    # Place 19 circles in a hexagonal cluster
    cluster_centers = []
    for i in range(5):
        for j in range(5 - i):
            x = (i + 0.5) * 0.25 + 0.1
            y = (j + 0.5) * 0.25 + 0.1
            cluster_centers.append([x, y])
    for i in range(4):
        for j in range(4 - i):
            x = (i + 0.5) * 0.25 + 0.1
            y = (j + 0.5) * 0.25 + 0.1
            cluster_centers.append([x, y])
    cluster_centers = np.array(cluster_centers)
    for i in range(19):
        centers_guess[i] = cluster_centers[i]

    # Place 7 circles in the top-right corner
    for i in range(19, 26):
        centers_guess[i] = [0.8, 0.8 + (i - 19) * 0.1]

    # Define the optimization problem
    def objective(x):
        return total_sum(x[19:])

    def constraint(x):
        return np.sum(x[19:])  # We're maximizing, so we'll minimize -sum

    cons = [{'type': 'ineq', 'fun': lambda x: x[19:]}]

    # Prepare the bounds for the variables
    bounds = [(0, 1) for _ in range(26)]  # Radii must be non-negative

    # Perform the optimization
    result = minimize(
        objective,
        x0=radii_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'maxiter': 1000, 'ftol': 1e-7, 'eps': 1e-8}
    )

    # Extract the optimized radii
    optimized_radii = result.x

    # Validate the distribution to ensure the centers are placed in the correct region
    # and the radii values are valid

    # Now, reposition centers to ensure better packing
    optimized_centers = np.zeros((26, 2))
    for i in range(19):
        # Adjust the cluster to be more tightly packed
        optimized_centers[i] = [(0.1 + i * 0.165), (0.2 + i * 0.165)]

    for i in range(19, 26):
        # Place the remaining circles in a slight offset in the top-right
        optimized_centers[i] = [0.8 + (i - 19) * 0.13, 0.8 + (i - 19) * 0.13]

    # Ensure the radii are adjusted to fit within the total space
    adjusted_radii = np.clip(optimized_radii * 0.8, 0, 1)

    # Now validate the packing to make sure no circles go out of bounds
    def validate():
        for i in range(26):
            x, y = optimized_centers[i]
            r = adjusted_radii[i]
            if x - r < 0 or x + r > 1:
                return False
            if y - r < 0 or y + r > 1:
                return False
        for i in range(26):
            for j in range(i + 1, 26):
                dist = np.sqrt((optimized_centers[i, 0] - optimized_centers[j, 0]) ** 2 +
                               (optimized_centers[i, 1] - optimized_centers[j, 1]) ** 2)
                if dist < adjusted_radii[i] + adjusted_radii[j] - 1e-12:
                    return False
        return True

    if not validate():
        # If validation fails, try a different placement strategy
        optimized_centers = np.random.rand(26, 2)
        adjusted_radii = np.random.rand(26) * 0.1
        return optimized_centers, adjusted_radii, np.sum(adjusted_radii)

    return optimized_centers, adjusted_radii, np.sum(adjusted_radii)
