import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    # Initial positions: 10 in the center, 16 near the corners
    # Center circles are arranged in a hexagonal grid
    # Corner circles are placed at the square's corners with some spacing

    # Initial center positions
    center_centers = []
    for x in np.linspace(0.1, 0.9, 4):  # X positions
        for y in np.linspace(0.1, 0.9, 4):  # Y positions
            center_centers.append([x, y])

    # Convert to numpy array
    center_centers = np.array(center_centers)

    # Initial corner positions
    corner_centers = []
    for x in [0.1, 0.9]:
        for y in [0.1, 0.9]:
            corner_centers.append([x, y])

    # Convert to numpy array
    corner_centers = np.array(corner_centers)

    # Combine initial centers
    initial_centers = np.vstack([center_centers, corner_centers])
    initial_centers = initial_centers[:26]  # Only 26 circles

    # Initial guess for radii: same for all
    initial_radii = np.full(26, 0.05)

    # Define the optimization variables: centers and radii
    n = 26
    lb = np.concatenate([np.zeros(2 * n), np.zeros(n)])
    ub = np.concatenate([np.ones(2 * n), np.ones(n)])
    bounds = Bounds(lb, ub)

    # Objective function: maximize the sum of radii
    def objective(x):
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]
        return -np.sum(radii)

    # Constraints:
    # 1. Circles are within the unit square
    # 2. Circles do not overlap
    # 3. Radii are non-negative

    # Define constraints
    def constraint_func(x):
        centers = x[:2 * n].reshape(n, 2)
        radii = x[2 * n:]
        # Check if all circles are within the unit square
        for i in range(n):
            x_i, y_i = centers[i]
            r_i = radii[i]
            if x_i - r_i < -1e-12 or x_i + r_i > 1 + 1e-12 or y_i - r_i < -1e-12 or y_i + r_i > 1 + 1e-12:
                return -np.inf
        # Check if all circles do not overlap
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx * dx + dy * dy)
                if dist < radii[i] + radii[j] - 1e-12:
                    return -np.inf
        # Check if all radii are non-negative
        for r in radii:
            if r < -1e-12:
                return -np.inf
        return 0.0

    cons = {'type': 'ineq', 'fun': constraint_func}

    # Run optimization
    result = optimize.minimize(
        objective,
        x0=np.concatenate([initial_centers.reshape(2 * n), initial_radii]),
        bounds=bounds,
        constraints=cons,
        method='L-BFGS-B',
        options={'maxiter': 500, 'ftol': 1e-10, 'gtol': 1e-10}
    )

    centers = result.x[:2 * n].reshape(n, 2)
    radii = result.x[2 * n:]

    # Check for any NaNs or invalid values
    if np.isnan(centers).any() or np.isnan(radii).any():
        raise ValueError("NaN values present in the result")

    # Final validation
    is_valid, message = validate_packing(centers, radii)
    if not is_valid:
        raise ValueError(message)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
