import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from itertools import combinations

def run_packing():
    # Define a grid of candidate positions (hexagonal and boundary)
    grid = []
    for y in np.linspace(0, 1, 10):
        for x in np.linspace(0, 1, 10):
            grid.append([x, y])
    grid = np.array(grid)
    # Add some offset points for flexibility
    for y in np.linspace(0.2, 0.8, 5):
        for x in np.linspace(0.2, 0.8, 5):
            grid.append([x, y])
    grid = np.unique(grid, axis=0)
    # Select 26 positions
    selected = grid[np.random.permutation(len(grid))[:26]]
    
    # Initial radius assignment based on hexagonal packing
    def initial_radii(positions):
        radii = np.zeros(len(positions))
        for i in range(len(positions)):
            x, y = positions[i]
            row = int(y * 10)  # approx row index in grid
            col = int(x * 10)
            if row % 2 == 0:
                neighbor_dist = min(1 - y, y, 1 - x, x)
            else:
                neighbor_dist = min(1 - y, y, 1 - x, x)
            radii[i] = min(neighbor_dist, 0.1)
        return radii
    
    # Optimization objective function
    def objective(radii, positions):
        return -np.sum(radii)
    
    # Constraint functions for non-overlapping
    def non_overlap_constraints(radii, positions):
        n = len(positions)
        constraints = []
        for i, j in combinations(range(n), 2):
            def constraint(radii):
                dist = np.linalg.norm(positions[i] - positions[j])
                return dist - (radii[i] + radii[j])
            constraints.append({'type': 'ineq', 'fun': constraint})
        return constraints
    
    # Constraint for boundary
    def boundary_constraints(radii, positions):
        n = len(positions)
        constraints = []
        for i in range(n):
            def constraint(radii):
                x, y = positions[i]
                return x - radii[i]
            constraints.append({'type': 'ineq', 'fun': constraint})
            def constraint2(radii):
                x, y = positions[i]
                return 1 - x - radii[i]
            constraints.append({'type': 'ineq', 'fun': constraint2})
            def constraint3(radii):
                x, y = positions[i]
                return y - radii[i]
            constraints.append({'type': 'ineq', 'fun': constraint3})
            def constraint4(radii):
                x, y = positions[i]
                return 1 - y - radii[i]
            constraints.append({'type': 'ineq', 'fun': constraint4})
        return constraints
    
    # Initial radii
    radii = initial_radii(selected)
    # Optimize
    result = minimize(
        lambda r: -np.sum(r),
        x0=radii,
        constraints=non_overlap_constraints(radii, selected) + boundary_constraints(radii, selected),
        bounds=[(0, 0.15) for _ in range(len(selected))],
        method='SLSQP'
    )
    
    # Extract final solution
    centers = selected
    radii = result.x
    sum_radii = np.sum(radii)
    return (centers, radii, sum_radii)
