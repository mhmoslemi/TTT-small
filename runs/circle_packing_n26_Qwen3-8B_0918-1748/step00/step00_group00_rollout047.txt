import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import itertools

def run_packing():
    # Initial guess for centers: 16 in hexagonal grid, 6 on edges, 4 at corners
    # First, define the centers in a hexagonal grid in the center of the square
    n_center = 16
    n_edge = 6
    n_corner = 4
    n_total = n_center + n_edge + n_corner

    # Step 1: Generate a hexagonal grid of 16 circles in the center
    # Generate a 4x4 grid with hexagonal spacing
    # Distances between centers in the grid are 2r, where r is the radius of each circle
    # We'll need to adjust the radius later based on constraints

    # Generate grid centers
    centers_hex = []
    for i in range(4):
        for j in range(4):
            if i % 2 == 0:
                x = 0.25 + j * 0.25
                y = 0.25 + i * 0.25 + (i + 1) * 0.1
            else:
                x = 0.25 + j * 0.25 + 0.125
                y = 0.25 + i * 0.25 + (i + 1) * 0.1
            centers_hex.append([x, y])

    # Step 2: Generate 6 edge circles (midpoints of edges)
    edge_centers = []
    edge_centers.append([0.5, 0.1])  # bottom edge
    edge_centers.append([0.1, 0.5])  # left edge
    edge_centers.append([0.9, 0.5])  # right edge
    edge_centers.append([0.5, 0.9])  # top edge
    edge_centers.append([0.25, 0.5])  # left mid
    edge_centers.append([0.75, 0.5])  # right mid
    centers_hex.extend(edge_centers)

    # Step 3: Add 4 corner circles
    corner_centers = [[0.1, 0.1], [0.9, 0.1], [0.9, 0.9], [0.1, 0.9]]
    centers_hex.extend(corner_centers)

    initial_centers = np.array(centers_hex)

    # Initial radius guess: all circles same size
    radius_guess = np.full(n_total, 0.1)

    def objective(radii):
        return -np.sum(radii)

    def constraint_dist(centers, radii):
        dist_matrix = squareform(pdist(centers))
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return 1.0
        return 0.0

    def constraint_boundary(centers, radii):
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return 1.0
        return 0.0

    # Optimization bounds: radius must be >=0
    bounds = [(0, 1) for _ in range(n_total)]

    # Optimization problem:
    # Objective: maximize sum(radii)
    # Constraints: distance between circles >= sum of their radii, and within square

    # Solve the problem using a local optimizer
    result = minimize(
        fun=objective,
        x0=radius_guess,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_dist, 'args': (initial_centers,)},
            {'type': 'ineq', 'fun': constraint_boundary, 'args': (initial_centers,)},
        ],
        method='SLSQP',
    )

    radii = result.x
    centers = initial_centers.copy()

    # Now adjust the centers based on the radii to maximize the sum, ensuring non-overlap
    # We'll perform a local optimization to refine the positions of the circles

    # Define the function to compute the sum of radii
    def objective_pos(centers_flat):
        centers = centers_flat.reshape((n_total, 2))
        dist_matrix = squareform(pdist(centers))
        for i in range(n_total):
            for j in range(i + 1, n_total):
                if dist_matrix[i, j] < radii[i] + radii[j] - 1e-8:
                    return -100000  # Penalize overlap
        for i in range(n_total):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return -100000  # Penalize boundary violation
        return -np.sum(radii)

    # Initial guess for centers: use the initial_centers
    centers_flat = initial_centers.flatten()

    # Bounds for each center coordinate
    center_bounds = []
    for i in range(n_total):
        center_bounds.extend([(0, 1), (0, 1)])

    result = minimize(
        fun=objective_pos,
        x0=centers_flat,
        bounds=center_bounds,
        method='L-BFGS-B',
    )

    centers = result.x.reshape((n_total, 2))

    # Redefine radii based on the distances and positions
    dist_matrix = squareform(pdist(centers))
    radii = []
    for i in range(n_total):
        for j in range(i + 1, n_total):
            if dist_matrix[i, j] < 1e-8:
                continue
            radii_i = dist_matrix[i, j] / 2
            radii_j = dist_matrix[i, j] / 2
            radii.append(radii_i)
            radii.append(radii_j)
        # Add the last radius (for non-overlap with boundaries)
        x, y = centers[i]
        r = min(x, 1 - x, y, 1 - y) * 0.9
        radii.append(r)

    radii = np.array(radii)

    # Check that the number of radii matches the number of circles
    if len(radii) != n_total:
        raise ValueError("Mismatch in number of radii")

    # Final validation and return
    centers = centers.astype(np.float64)
    radii = radii.astype(np.float64)
    return centers, radii, np.sum(radii)
