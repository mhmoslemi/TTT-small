import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run_packing():
    # Initial hexagonal grid with 5 rows and varying columns to total 26 circles
    num_rows = 5
    columns = [5, 6, 5, 5, 5]
    total_circles = sum(columns)
    
    # Define grid positions
    def generate_hex_grid(columns, num_rows, spacing):
        centers = []
        for row in range(num_rows):
            for col in range(columns[row]):
                x = col * spacing
                y = row * spacing * np.sqrt(3) / 2
                centers.append([x, y])
        return np.array(centers)
    
    # Generate base grid with spacing of 1.0
    grid = generate_hex_grid(columns, num_rows, spacing=1.0)
    
    # Translate grid to fit within [0,1]^2
    trans_x = (1 - np.max(grid[:, 0])) / 2
    trans_y = (1 - np.max(grid[:, 1])) / 2
    grid += [trans_x, trans_y]
    
    # Initial radii based on grid spacing
    initial_radii = np.full(len(grid), 1.0 / (2 * np.sqrt(3)), dtype=np.float64)
    
    # Optimizer setup
    def objective(x):
        return -np.sum(x)  # Negative for maximization
    
    def constraint_func(x):
        dists = np.zeros((len(grid), len(grid)))
        for i in range(len(grid)):
            for j in range(i + 1, len(grid)):
                dists[i, j] = np.sqrt(np.sum((grid[i] - grid[j]) ** 2))
                if dists[i, j] < x[i] + x[j] - 1e-8:
                    return -np.inf
        return np.inf
    
    constraints = [{'type': 'ineq', 'fun': constraint_func}]
    
    bounds = Bounds([0.0] * len(grid), [1.0] * len(grid))
    
    # Initial guess
    x0 = np.copy(initial_radii)
    
    # Run optimization
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints, tol=1e-12)
    
    # Extract results
    centers = grid
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Final validation (optional but included for completeness)
    # if not validate_packing(centers, radii)[0]:
    #     raise ValueError("Invalid packing configuration")
    
    return (centers, radii, sum_radii)
