```python
import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

def run_packing():
    np.random.seed(42)
    
    def objective(radii):
        return -np.sum(radii)
    
    def constraint_overlap(i, j, centers, radii):
        def cons(x):
            centers[i] = x[:2]
            centers[j] = x[2:]
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            return dist - (radii[i] + radii[j])
        return cons
    
    def constraint_boundary(i, centers, radii):
        def cons(x):
            centers[i] = x[:2]
            r = radii[i]
            return np.array([
                centers[i][0] - r,
                centers[i][1] - r,
                1 - centers[i][0] - r,
                1 - centers[i][1] - r
            ])
        return cons
    
    def constraint_radius(i, x):
        return x[i] >= 0
    
    def run_optimization():
        n = 26
        centers = np.random.rand(n, 2) * 1.0
        radii = np.random.rand(n) * 0.1
        
        constraints = []
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: x[i] >= 0})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: (x[i*2] - x[i*2 + 1]) >= -1e-12})
            constraints.append({'type': 'ineq', 'fun': lambda x, i=i: (x[i*2] - x[i*2 + 1]) <= 1 + 1e-12})
        
        for i in range(n):
            constraints.append({'type': 'ineq', 'fun': constraint_boundary(i, centers, radii)})
        
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append({'type': 'ineq', 'fun': constraint_overlap(i, j, centers, radii)})
        
        bounds = Bounds([0.0 for _ in range(2*n)], [1.0 for _ in range(2*n)])
        
        result = optimize.minimize(
            objective,
            x0=np.concatenate((centers.flatten(), radii)),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'eps': 1e-10, 'maxiter': 1000}
        )
        
        centers = result.x[:2*n].reshape(n, 2)
        radii = result.x[2*n:]
        
        return centers, radii
    
    centers, radii = run_optimization()
    
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
```
