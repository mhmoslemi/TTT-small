import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():
    # Initial configuration: hybrid hexagonal + ring
    # 16 circles in central hexagonal grid, 10 circles in outer ring
    # This hybrid approach balances density and spacing

    # Function to compute bounding box for a circle
    def circle_bbox(x, y, r):
        return np.array([[x - r, y - r], [x + r, y + r]])

    # Initialize centers
    def initial_centers():
        # Central hexagonal grid
        n_hex = 16
        hex_centers = []
        for i in range(n_hex):
            if i < 6:
                row = i
            else:
                row = i - 6
            col = i % 6
            x = 0.4 + col * 0.2
            y = 0.4 + row * 0.2
            hex_centers.append([x, y])

        # Outer ring
        ring_centers = []
        for i in range(10):
            angle = 2 * np.pi * i / 10
            x = 0.5 + 0.3 * np.cos(angle)
            y = 0.5 + 0.3 * np.sin(angle)
            ring_centers.append([x, y])

        return np.array(hex_centers + ring_centers)

    centers = initial_centers()
    n = centers.shape[0]
    radii = np.full(n, 0.1)

    # Cost function: negative of sum of radii to maximize
    def objective(radii):
        return -np.sum(radii)

    # Constraints: non-overlap and boundary
    def constraint_func(radii):
        if np.isnan(radii).any():
            raise ValueError("NaN in radii")
        if np.any(radii < 0):
            raise ValueError("Negative radius")
        dists = squareform(pdist(centers))
        for i in range(n):
            for j in range(i + 1, n):
                if dists[i, j] < radii[i] + radii[j] - 1e-8:
                    return False
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
                return False
        return True

    # Optimization
    result = minimize(
        objective,
        radii,
        method='SLSQP',
        bounds=[(0, 0.5) for _ in range(n)],
        constraints={'type': 'inequality', 'fun': constraint_func},
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    if not result.success:
        raise ValueError("Optimization failed")

    # Final centers and radii
    final_centers = centers
    final_radii = result.x
    sum_radii = np.sum(final_radii)

    return (final_centers, final_radii, sum_radii)
