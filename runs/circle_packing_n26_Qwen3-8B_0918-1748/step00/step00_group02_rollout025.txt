import numpy as np
import scipy.optimize as opt

def run_packing():
    # Initial placement: 4 large in corners, 8 medium on edges, 10 small in gaps, 4 small in interior
    # We'll fix positions for small circles for better control and let radii be optimized
    
    # Define initial positions for large circles in corners (radius = r)
    large_centers = np.array([
        [0.1, 0.1],
        [0.1, 0.9],
        [0.9, 0.1],
        [0.9, 0.9]
    ])
    
    # Define initial positions for medium circles along the edges (radius = r)
    edge_centers = np.array([
        [0.05, 0.5],
        [0.95, 0.5],
        [0.5, 0.05],
        [0.5, 0.95],
        [0.25, 0.25],
        [0.75, 0.25],
        [0.25, 0.75],
        [0.75, 0.75]
    ])
    
    # Define initial positions for small circles in the gaps (radius = r)
    gap_centers = np.array([
        [0.3, 0.3],
        [0.3, 0.7],
        [0.7, 0.3],
        [0.7, 0.7],
        [0.2, 0.6],
        [0.8, 0.6],
        [0.6, 0.2],
        [0.6, 0.8],
        [0.4, 0.4],
        [0.6, 0.6]
    ])
    
    # Define initial positions for small interior circles (radius = r)
    interior_centers = np.array([
        [0.4, 0.6],
        [0.6, 0.4],
        [0.5, 0.45],
        [0.55, 0.5]
    ])
    
    # Combine all initial centers
    centers = np.vstack([large_centers, edge_centers, gap_centers, interior_centers])
    
    # Optimization: minimize negative sum of radii, with constraints
    # We define a function that computes the sum of radii and returns negative for minimization
    def objective(radii):
        return -np.sum(radii)
    
    # Define constraints for each circle: it must stay within [0,1]^2
    def constraint_center(x, i):
        x_i, y_i = x[2*i], x[2*i + 1]
        r_i = x[-26 + i]
        return x_i - r_i >= 0, x_i + r_i <= 1, y_i - r_i >= 0, y_i + r_i <= 1
    
    # Define constraints for pair of circles: distance between centers >= sum of radii
    def constraint_overlap(x, i, j):
        x_i, y_i = x[2*i], x[2*i + 1]
        r_i = x[-26 + i]
        x_j, y_j = x[2*j], x[2*j + 1]
        r_j = x[-26 + j]
        return np.sqrt((x_i - x_j)**2 + (y_i - y_j)**2) >= r_i + r_j
    
    # Initial guess: same radius for all circles for a rough starting point
    n = 26
    initial_guess = np.concatenate([centers.flatten(), np.ones(n) * 0.1])
    
    # List of constraints
    cons = []
    for i in range(n):
        # Constraint for center of circle i
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[2*i] - x[-26 + i],
            'jac': lambda x, i=i: np.concatenate([np.zeros(2*i + 1), np.array([1.0] + [0.0]*25), np.zeros(26 - i - 1)]))
        })
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[2*i + 1] - x[-26 + i],
            'jac': lambda x, i=i: np.concatenate([np.zeros(2*i + 2), np.array([0.0] + [0.0]*25), np.zeros(26 - i - 1)]))
        })
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[2*i + 1] - x[-26 + i],
            'jac': lambda x, i=i: np.concatenate([np.zeros(2*i + 2), np.array([0.0] + [0.0]*25), np.zeros(26 - i - 1)]))
        })
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[2*i + 1] - x[-26 + i],
            'jac': lambda x, i=i: np.concatenate([np.zeros(2*i + 2), np.array([0.0] + [0.0]*25), np.zeros(26 - i - 1)]))
        })
    
    for i in range(n):
        for j in range(i + 1, n):
            cons.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: np.sqrt((x[2*i] - x[2*j])**2 + (x[2*i + 1] - x[2*j + 1])**2) - x[-26 + i] - x[-26 + j],
                'jac': lambda x, i=i, j=j: np.concatenate([
                    np.zeros(2*i),
                    np.zeros(2*j - 1),
                    np.array([1.0, 0.0] + [0.0]*2*j),
                    np.zeros(26 - j - 1),
                    np.array([-1.0, 0.0] + [0.0]*2*i),
                    np.zeros(26 - i - 1)
                ])
            })
    
    # Optimization bounds: x must be within [0,1], radii must be >= 0
    bounds = [(0.0, 1.0) for _ in range(2*n)] + [(0.0, 1.0) for _ in range(n)]
    
    # Perform the optimization
    result = opt.minimize(
        objective,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        tol=1e-10,
        options={'maxiter': 1000, 'ftol': 1e-10, 'disp': False}
    )
    
    # Extract centers and radii
    centers_opt = result.x[:2*n].reshape(26, 2)
    radii_opt = result.x[2*n:]
    
    # Calculate total sum
    total_sum = np.sum(radii_opt)
    
    return centers_opt, radii_opt, total_sum
