import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize import differential_evolution
from itertools import combinations

def run_packing():
    # Define the number of circles
    n = 26

    # Initial guess: hexagonal packing with 3 rows
    # Row 1: 10 circles
    # Row 2: 10 circles (offset by 0.5*r)
    # Row 3: 6 circles (offset by 0.5*r)
    # Total of 26 circles

    # Initial guess for radii (assume a starting value: 0.05)
    r0 = 0.05  # initial guess for radii
    r = np.full(n, r0)

    # Hexagonal grid: coordinates for 3 rows
    # First row: 10 circles
    # Second row: 10 circles, shifted by 0.5*r
    # Third row: 6 circles, shifted by 0.5*r
    # We'll define centers as follows:

    centers = []

    r_row_1 = r0
    r_row_2 = r0
    r_row_3 = r0

    # First row
    for i in range(10):
        x = i * (r_row_1 + r_row_1) * 0.5
        y = r_row_1
        centers.append([x, y])

    # Second row
    for i in range(10):
        x = i * (r_row_2 + r_row_2) * 0.5 + r_row_2 * 0.5
        y = r_row_2 + np.sqrt(3) * r_row_2
        centers.append([x, y])

    # Third row
    for i in range(6):
        x = i * (r_row_3 + r_row_3) * 0.5 + r_row_3 * 0.5
        y = r_row_3 + 2 * np.sqrt(3) * r_row_3
        centers.append([x, y])

    centers = np.array(centers)

    # Optimization parameters
    bounds = []
    for i in range(n):
        # x and y between 0 and 1
        bounds.append((0.0, 1.0))
        bounds.append((0.0, 1.0))

    # Objective: maximize sum of radii
    def objective(x):
        # x is flattened: [x0, y0, x1, y1, ..., x25, y25]
        centers_flat = x.reshape((n, 2))
        radii = r.copy()
        # Check if centers are within bounds
        for i in range(n):
            x, y = centers_flat[i]
            r = radii[i]
            if not (0.0 - 1e-8 <= x - r <= 1.0 + 1e-8 and 0.0 - 1e-8 <= y - r <= 1.0 + 1e-8):
                return -float('inf')
        # Check distances between circles
        for i, j in combinations(range(n), 2):
            dx = centers_flat[i, 0] - centers_flat[j, 0]
            dy = centers_flat[i, 1] - centers_flat[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-8:
                return -float('inf')
        return np.sum(radii)

    # Initial guess x0: centers flattened
    x0 = centers.flatten()

    # Setup bounds
    bounds = []
    for i in range(n):
        bounds.extend([(0.0, 1.0), (0.0, 1.0)])

    # Optimization
    result = differential_evolution(objective, bounds, strategy='best1bin', popsize=15, maxiter=150, tol=1e-6, mutation=0.5, recombination=0.9)

    # Extract solution
    x_opt = result.x
    centers = x_opt.reshape((n, 2))

    # Ensure no overlap or boundary violations
    def check_valid(centers, radii):
        for i in range(n):
            x, y = centers[i]
            r = radii[i]
            if not (0.0 - 1e-8 <= x - r <= 1.0 + 1e-8 and 0.0 - 1e-8 <= y - r <= 1.0 + 1e-8):
                return False
        for i, j in combinations(range(n), 2):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-8:
                return False
        return True

    # Now perform a final Nelder-Mead optimization to refine
    def refine_objective(x):
        centers = x.reshape((n, 2))
        # Ensure circles are within the square and non-overlapping
        if not check_valid(centers, r):
            return -float('inf')
        return np.sum(r)

    # Refinement bounds
    refine_bounds = []
    for i in range(n):
        refine_bounds.extend([(0.0, 1.0), (0.0, 1.0)])

    result_refine = minimize(refine_objective, x0, method='Nelder-Mead', bounds=refine_bounds, options={'xatol': 1e-6, 'fatol': 1e-6})

    # Extract final solution
    centers = result_refine.x.reshape((n, 2))

    # Construct radii vector
    radii = np.full(n, r0)

    # Ensure all circles are within the square
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-8 or x + r > 1 + 1e-8 or y - r < -1e-8 or y + r > 1 + 1e-8:
            radii[i] = np.min([np.max([0.0, x - 1e-8]), np.max([0.0, 1.0 - 1e-8])])

    # Final validation
    for i, j in combinations(range(n), 2):
        dx = centers[i, 0] - centers[j, 0]
        dy = centers[i, 1] - centers[j, 1]
        dist = np.sqrt(dx**2 + dy**2)
        if dist < radii[i] + radii[j] - 1e-8:
            radii[np.argmin([radii[i], radii[j]])] *= 0.9

    # Compute sum of radii
    sum_radii = np.sum(radii)

    # Return centers, radii, sum_radii
    return (centers, radii, sum_radii)
