import numpy as np
import scipy.optimize as opt
import matplotlib.pyplot as plt

def run_packing():
    # Initial configuration: 25 circles placed in a grid (5x5) with spacing
    # We'll optimize the 26th circle to fit into the remaining space

    # Initialize grid
    num_rows = 5
    num_cols = 5
    grid_spacing = 1.0 / (num_rows - 1)
    grid_x = np.linspace(0, 1, num_cols)
    grid_y = np.linspace(0, 1, num_rows)
    grid_centers = np.array([ [x, y] for y in grid_y for x in grid_x ])

    # Initial guess for radii: based on the distance to neighbors
    initial_radii = np.zeros(len(grid_centers))
    for i in range(len(grid_centers)):
        x, y = grid_centers[i]
        dist_to_left = x - grid_centers[i - 1][0] if i % num_cols != 0 else np.inf
        dist_to_right = grid_centers[i + 1][0] - x if i % num_cols != num_cols - 1 else np.inf
        dist_to_top = grid_centers[i - num_cols][1] - y if i >= num_cols else np.inf
        dist_to_bottom = y - grid_centers[i + num_cols][1] if i < num_cols * (num_rows - 1) else np.inf
        min_dist = min(dist_to_left, dist_to_right, dist_to_top, dist_to_bottom)
        initial_radii[i] = min_dist / 2

    # Flatten and combine grid and the 26th circle
    centers = np.vstack((grid_centers, [0.5, 0.5]))  # 26th circle at center
    radii = initial_radii.tolist() + [0.0]  # Initial radius for 26th circle

    # Define optimization variables: positions and radii
    # Unpacked variables: [x0, y0, r0, x1, y1, r1, ..., x25, y25, r25]
    n = len(centers)
    bounds = []

    for i in range(n):
        if i == n - 1:
            # 26th circle has no bounds (we will optimize it)
            bounds.extend([(-1, 1), (-1, 1), (0, 1)])
        else:
            # For grid circles, enforce center within [0,1]^2 and radius >=0
            bounds.extend([(0, 1), (0, 1), (0, 1)])

    # Define objective function (maximize sum of radii)
    def objective(x):
        return -np.sum(x[2::3])  # Negative for minimization

    # Define constraints:
    # 1. Center stays within [0,1]^2
    # 2. Radius >= 0
    # 3. Distance between centers >= sum of radii

    constraints = []

    # Constraints for center within square
    for i in range(n):
        if i < n - 1:
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[2 * i + 0] - x[2 * i + 2] + 1e-12
            })
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[2 * i + 1] - x[2 * i + 2] + 1e-12
            })
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[2 * i + 0] + x[2 * i + 2] - 1 - 1e-12
            })
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[2 * i + 1] + x[2 * i + 2] - 1 - 1e-12
            })

    # Constraints for radii >= 0
    for i in range(n):
        if i < n - 1:
            constraints.append({
                'type': 'ineq',
                'fun': lambda x, i=i: x[2 * i + 2]
            })

    # Distance constraints between circles
    for i in range(n):
        for j in range(i + 1, n):
            def dist_constr(x):
                x1, y1, r1 = x[2 * i + 0], x[2 * i + 1], x[2 * i + 2]
                x2, y2, r2 = x[2 * j + 0], x[2 * j + 1], x[2 * j + 2]
                return np.sqrt((x1 - x2)**2 + (y1 - y2)**2) - r1 - r2 - 1e-12
            constraints.append({
                'type': 'ineq',
                'fun': dist_constr
            })

    # Optimize
    x0 = np.concatenate([centers.flatten(), radii])
    result = opt.minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraints)

    # Extract the result
    optimized_vars = result.x
    centers_opt = np.array(optimized_vars[0::3]).reshape(-1, 2)
    radii_opt = optimized_vars[2::3]

    # Ensure radii are nonnegative
    radii_opt = np.maximum(radii_opt, 0.0)

    # Validate packing
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Packing is invalid: {message}")

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
