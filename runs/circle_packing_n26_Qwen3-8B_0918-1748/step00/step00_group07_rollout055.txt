import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Define initial grid positions using a hexagonal packing
    def generate_hex_grid(n):
        rows = int(np.sqrt(n))
        cols = (n + rows - 1) // rows
        grid = []
        x = np.linspace(0, 1, cols)
        y = np.linspace(0, 1, rows)
        for i in range(len(y)):
            for j in range(len(x)):
                if i % 2 == 0:
                    grid.append((x[j], y[i]))
                else:
                    grid.append((x[j] + 0.5, y[i]))
        return np.array(grid)
    
    # Initial configuration using hexagonal grid
    centers = generate_hex_grid(26)
    radii = np.full(26, 0.05)
    
    # Objective function: maximize sum of radii
    def objective(radii):
        return -np.sum(radii)  # Minimize negative to maximize
    
    # Constraint: no two circles overlap
    def constraint_overlap(x):
        r = x[26:]
        c = x[:26]
        for i in range(26):
            for j in range(i + 1, 26):
                dx = c[i, 0] - c[j, 0]
                dy = c[i, 1] - c[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < r[i] + r[j] - 1e-8:
                    return 1
        return 0
    
    # Constraint: circles within square [0,1]x[0,1]
    def constraint_boundary(x):
        r = x[26:]
        c = x[:26]
        for i in range(26):
            if c[i, 0] - r[i] < -1e-8 or c[i, 0] + r[i] > 1 + 1e-8:
                return 1
            if c[i, 1] - r[i] < -1e-8 or c[i, 1] + r[i] > 1 + 1e-8:
                return 1
        return 0
    
    # Combine constraints
    def constraints(x):
        if constraint_overlap(x) > 0:
            return np.array([1])
        if constraint_boundary(x) > 0:
            return np.array([1])
        return np.array([])
    
    # Initial guess
    x0 = np.concatenate((centers.flatten(), radii))
    
    # Optimization settings
    res = minimize(
        objective,
        x0,
        method='SLSQP',
        constraints={'type': 'ineq', 'fun': constraints},
        bounds=[(0, 1) for _ in range(52)]  # 26 centers (x, y), 26 radii
    )
    
    if not res.success:
        raise ValueError("Optimization failed")
    
    centers_opt = res.x[:52].reshape(26, 2)
    radii_opt = res.x[52:]
    
    # Final validation

    if not valid:
        raise ValueError(msg)
    
    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii
