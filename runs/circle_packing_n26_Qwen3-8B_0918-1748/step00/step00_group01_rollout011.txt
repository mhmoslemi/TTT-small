import numpy as np
import matplotlib.pyplot as plt
from scipy.optimize import minimize, Bounds

def run_packing():
    # Define the hexagonal packing with staggered rows
    def initial_guess_hexagonal(n_rows, n_cols, spacing):
        centers = []
        for i in range(n_rows):
            y = i * spacing
            for j in range(n_cols):
                x = j * spacing
                centers.append((x, y))
            if i % 2 == 1:
                for j in range(n_cols - 1):
                    x = (j + 0.5) * spacing
                    y = i * spacing + spacing / 2
                    centers.append((x, y))
        centers = np.array(centers)
        return centers

    # Define objective function to maximize sum of radii
    def objective(params):
        x = params[::2]
        y = params[1::2]
        r = params[26:]
        return -np.sum(r)

    # Define constraints for boundary and non-overlap
    def constraint_boundary(params):
        x = params[::2]
        y = params[1::2]
        r = params[26:]
        for i in range(26):
            if x[i] - r[i] < -1e-12 or x[i] + r[i] > 1 + 1e-12:
                return -1
            if y[i] - r[i] < -1e-12 or y[i] + r[i] > 1 + 1e-12:
                return -1
        return 0

    def constraint_overlap(params):
        x = params[::2]
        y = params[1::2]
        r = params[26:]
        for i in range(26):
            for j in range(i + 1, 26):
                dx = x[i] - x[j]
                dy = y[i] - y[j]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < r[i] + r[j] - 1e-12:
                    return -1
        return 0

    # Create initial centers using a staggered hexagonal layout
    n_rows = 6
    n_cols = 5
    spacing = 0.5
    centers = initial_guess_hexagonal(n_rows, n_cols, spacing)
    # Initial guess for radii
    radii = np.full(26, 0.1)

    # Flatten centers and radii into a single parameter vector
    params = np.concatenate((centers.flatten(), radii))

    # Bounds for x, y coordinates and radii
    bounds = []
    for _ in range(26 * 2):
        bounds.append((0.0, 1.0))
    for _ in range(26):
        bounds.append((0.0, 1.0))  # Radii bounds

    # Set up the optimization problem
    constraints = [
        {'type': 'ineq', 'fun': constraint_boundary},
        {'type': 'ineq', 'fun': constraint_overlap}
    ]
    result = minimize(
        objective,
        params,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 200, 'ftol': 1e-8}
    )

    # Extract the optimized parameters
    optimized_params = result.x
    centers_opt = optimized_params[:26 * 2].reshape((26, 2))
    radii_opt = optimized_params[26 * 2:]

    # Validate the packing
    valid, message = validate_packing(centers_opt, radii_opt)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")

    sum_radii = np.sum(radii_opt)
    return centers_opt, radii_opt, sum_radii

def validate_packing(centers, radii):
    import numpy as np

    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
