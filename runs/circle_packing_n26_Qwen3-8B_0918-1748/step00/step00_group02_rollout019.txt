import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import cdist

def run_packing():
    # Define the number of circles
    n = 26
    
    # Initial layout: 16 circles in central hexagonal grid and 10 on boundaries
    # Hexagonal packing for central 16 circles
    hex_rows = 4
    hex_cols = 4
    hex_radius = 0.15
    hex_spacing = 2 * hex_radius  # Distance between centers
    
    # Generate central hexagonal grid
    centers = []
    radii = []
    for i in range(hex_rows):
        for j in range(hex_cols):
            x = 0.5 - (j - hex_cols // 2) * hex_spacing
            y = 0.5 - (i - hex_rows // 2) * hex_spacing
            centers.append([x, y])
            radii.append(hex_radius)
    
    # Generate boundary circles (10 circles)
    boundary_centers = [
        [0.1, 0.5], [0.9, 0.5], [0.5, 0.1], [0.5, 0.9],
        [0.2, 0.2], [0.8, 0.8], [0.2, 0.8], [0.8, 0.2],
        [0.3, 0.25], [0.7, 0.75]
    ]
    boundary_radii = [0.1] * 10
    centers += boundary_centers
    radii += boundary_radii
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.array(radii)
    
    # Define the objective function: maximize the sum of radii
    def objective(x):
        return -np.sum(radii)
    
    # Define constraints: distance between centers >= sum of radii
    def constraint_func(x):
        # x is a single flattened vector of all centers and radii
        # Extract centers and radii
        x_centers = x[:2 * n].reshape(n, 2)
        x_radii = x[2 * n:]
        
        # Check boundaries
        for i in range(n):
            x_pos, y_pos = x_centers[i]
            r = x_radii[i]
            if (x_pos - r < -1e-12 or x_pos + r > 1 + 1e-12
                or y_pos - r < -1e-12 or y_pos + r > 1 + 1e-12):
                return -np.inf  # Penalize if outside
    
        # Check circle overlaps
        for i in range(n):
            for j in range(i + 1, n):
                dx = x_centers[i, 0] - x_centers[j, 0]
                dy = x_centers[i, 1] - x_centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < x_radii[i] + x_radii[j] - 1e-12:
                    return -np.inf  # Penalize if overlapping
        
        return np.sum(x_radii)
    
    # Create bounds for the optimization variables
    # For each center: [0, 1], [0, 1]
    # For each radius: [0, 1]
    bounds = []
    for _ in range(n):
        bounds.extend([(0., 1.), (0., 1.)])
    for _ in range(n):
        bounds.append((0., 1.))
    
    # Initial guess
    initial_guess = []
    for i in range(n):
        initial_guess.extend(centers[i])
    for i in range(n):
        initial_guess.append(radii[i])
    
    # Optimization using Sequential Least Squares Programming (SLSQP)
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[{'fun': constraint_func, 'type': 'eq'}],
        method='SLSQP',
        tol=1e-6,
        options={'disp': False}
    )
    
    # Extract the optimal centers and radii
    optimal_centers = result.x[:2 * n].reshape(n, 2)
    optimal_radii = result.x[2 * n:]
    
    # Validate the packing to ensure correctness
    valid, msg = validate_packing(optimal_centers, optimal_radii)
    if not valid:
        raise ValueError(f"Invalid packing: {msg}")
    
    total_sum = np.sum(optimal_radii)
    return optimal_centers, optimal_radii, total_sum

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
