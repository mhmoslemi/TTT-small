```python
import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import time

def run_packing():
    # Initial placement: hexagonal grid with some variations
    num_circles = 26
    centers = []
    for i in range(num_circles):
        row = i // 6
        col = i % 6
        x = col + 0.5 * (1 - (1 - np.sqrt(3)) / 2)
        y = row + np.sqrt(3) / 2
        x = np.clip(x, 0, 1)
        y = np.clip(y, 0, 1)
        centers.append([x, y])
    centers = np.array(centers)

    # Initial guess for radii (uniform)
    radii = np.full(num_circles, 0.1)

    # Constraints for the optimization
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_overlap(radii):
        distances = squareform(pdist(centers))
        for i in range(num_circles):
            for j in range(i + 1, num_circles):
                if distances[i, j] < radii[i] + radii[j] - 1e-8:
                    return -1
        return 0

    def constraint_boundary(radii):
        for i in range(num_circles):
            x, y = centers[i]
            if x - radii[i] < 0 or x + radii[i] > 1 or y - radii[i] < 0 or y + radii[i] > 1:
                return -1
        return 0

    # Optimization setup
    bounds = [(0, 1) for _ in range(num_circles)]
    result = minimize(
        fun=objective,
        x0=radii,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_overlap},
            {'type': 'ineq', 'fun': constraint_boundary}
        ],
        method='SLSQP',
        options={'ftol': 1e-8, 'maxiter': 1000}
    )

    # Extract results
    radii = result.x
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
```
