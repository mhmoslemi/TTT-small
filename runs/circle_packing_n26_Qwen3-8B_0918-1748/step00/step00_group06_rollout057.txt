import numpy as np
import scipy.optimize as optimize
from scipy.optimize import LinearConstraint, Bounds
from itertools import combinations

def run_packing():
    # Initial configuration: a grid of circles with some near the boundaries
    # This is a hybrid approach combining square and hexagonal packing
    centers = np.empty((26, 2))
    radii = np.zeros(26)
    
    # Define the grid-based configuration with variations
    # We'll use a hybrid arrangement combining multiple small clusters
    # Start with a known grid-like configuration, then add an extra circle near the corner
    
    # First 25 circles: grid with 5 rows and 5 columns, slightly offset
    rows = 5
    cols = 5
    spacing = 1 / (rows + 1)
    for i in range(rows):
        for j in range(cols):
            x = j * spacing + spacing / 2
            y = i * spacing + spacing / 2
            centers[i * cols + j] = [x, y]
    
    # 26th circle: place near top-right corner
    x = 0.92
    y = 0.92
    centers[25] = [x, y]
    
    # Compute an initial radius estimate: size of circles in grid assuming non-overlapping
    # Assuming each circle touches its neighbors
    r = spacing / 2
    radii[:] = r
    
    # Optimization step: adjust radii and positions to maximize sum of radii
    # We formulate this as a constrained optimization problem
    # We'll use a combination of constraints: no overlaps, within square
    
    def objective(radii):
        return -np.sum(radii)  # Maximize sum by minimizing negative sum
    
    def non_overlap_constraints(radii, centers):
        n = len(radii)
        constraints = []
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.linalg.norm(centers[i] - centers[j])
                if dist < radii[i] + radii[j] - 1e-8:
                    return False
            constraints.append((radii[i], None, None))
        return True
    
    # Initial guess for radii
    initial_radii = radii.copy()
    
    # Optimization
    result = optimize.minimize(
        objective,
        initial_radii,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(26)],
        constraints=[{
            'type': 'ineq',
            'fun': lambda r: np.array([np.linalg.norm(centers[i] - centers[j]) - (r[i] + r[j]) for i in range(26) for j in range(i + 1, 26)]),
            'jac': lambda r: np.array([[(np.linalg.norm(centers[i] - centers[j]) - (r[i] + r[j])) for j in range(i + 1, 26)] for i in range(26)]).flatten()
        }],
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    if not result.success:
        raise ValueError("Optimization failed")
    
    # Extract optimized centers and radii
    optimized_centers = centers
    optimized_radii = result.x
    
    # Final validation
    valid, message = validate_packing(optimized_centers, optimized_radii)
    if not valid:
        raise ValueError(f"Invalid packing: {message}")
    
    total_sum = np.sum(optimized_radii)
    return (optimized_centers, optimized_radii, total_sum)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
