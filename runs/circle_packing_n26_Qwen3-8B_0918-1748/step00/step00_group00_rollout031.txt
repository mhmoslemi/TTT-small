import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.optimize.constraints import Constraint
import itertools

def run_packing():
    # Define initial hexagonal grid layout for 26 circles
    def hexagonal_layout(n_circles):
        rows = int(np.sqrt(2 * n_circles / 3))
        cols = int(np.sqrt(2 * n_circles / 3))
        while rows * cols < n_circles:
            rows += 1
            cols += 1
        centers = []
        for i in range(rows):
            for j in range(cols):
                x = j / cols
                y = i / rows
                centers.append((x, y))
        # Adjust rows and cols to fit exactly n_circles
        while len(centers) > n_circles:
            centers.pop()
        return np.array(centers)

    # Initial layout with hexagonal grid
    initial_centers = hexagonal_layout(26)
    n = len(initial_centers)

    # Define optimization variables: flattened centers and radii
    # Each circle has (x, y, r), flattened to 3 * n variables
    initial_guess = np.concatenate([initial_centers.flatten(), np.ones(n) / 2])

    # Objective function to maximize sum of radii (minimize negative sum)
    def objective(x):
        flat_centers = x[:3 * n].reshape(n, 3)
        radii = flat_centers[:, 2]
        return -np.sum(radii)

    # Bounds: x and y in [0, 1], radii >= 0
    bounds = Bounds([0, 0, 0] * n, [1, 1, np.inf] * n)

    # Constraints: boundary and non-overlapping
    def constraint_func(x):
        flat_centers = x[:3 * n].reshape(n, 3)
        radii = flat_centers[:, 2]
        centers = flat_centers[:, :2]
        violations = []

        # Check boundaries (x - r >= 0, x + r <= 1, y - r >= 0, y + r <= 1)
        for i in range(n):
            x_i, y_i, r_i = flat_centers[i]
            if x_i - r_i < -1e-8:
                violations.append(-1)
            if x_i + r_i > 1 + 1e-8:
                violations.append(-1)
            if y_i - r_i < -1e-8:
                violations.append(-1)
            if y_i + r_i > 1 + 1e-8:
                violations.append(-1)
        # Check non-overlapping (distance between circles >= r_i + r_j)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-8:
                    violations.append(-1)
        return np.array(violations)

    # Constraints object
    cons = Constraint(fun=constraint_func, type='ineq')

    # Run optimization
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'ftol': 1e-10, 'maxiter': 1000}
    )

    # Extract centers and radii from result
    x_opt = result.x
    centers = x_opt[:3 * n].reshape(n, 2)
    radii = x_opt[2 * n : 3 * n]

    # Ensure radii are non-negative
    radii = np.clip(radii, 0, np.inf)

    # Final validation
    # (this step is not implemented as the problem says to let it fail loudly if invalid)

    return (centers, radii, -result.fun)
