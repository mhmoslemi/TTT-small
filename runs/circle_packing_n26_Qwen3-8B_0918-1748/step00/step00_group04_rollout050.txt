import numpy as np
import scipy.optimize as optimize

def run_packing():
    # Initial estimate of centers in a hexagonal-like grid
    # This is a rough estimate and will be refined during optimization
    
    # First, define a grid of positions that could be used as a starting point
    # We will use a hexagonal grid with multiple rows
    # This is a simplified starting grid without optimization
    
    # Create a starting grid with hexagonal pattern
    centers = []
    radii = []
    
    # First layer
    for i in range(5):
        for j in range(5):
            x = (i + 0.5) / 5
            y = (j + 0.5) / 5
            centers.append([x, y])
    
    # Add a second layer around the first
    for i in range(5):
        for j in range(5):
            x = (i + 0.5) / 5 + 0.25
            y = (j + 0.5) / 5 + 0.25
            centers.append([x, y])
    
    # Add a third layer
    for i in range(5):
        for j in range(5):
            x = (i + 0.5) / 5 + 0.5
            y = (j + 0.5) / 5 + 0.5
            centers.append([x, y])
    
    # Convert to numpy arrays
    centers = np.array(centers)
    radii = np.full(len(centers), 0.05)  # Initial guess for radii
    
    # Define the optimization problem
    def objective(radii):
        return -np.sum(radii)  # We want to maximize the sum
    
    def constraint_distances(radii):
        dists = np.zeros((len(centers), len(centers)))
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dists[i, j] = np.sqrt(dx**2 + dy**2)
        return dists - (radii[:, np.newaxis] + radii[np.newaxis, :]) + 1e-12
    
    # Optimize
    bounds = [(0, 1) for _ in range(len(centers))]
    result = optimize.minimize(
        objective, 
        radii,
        bounds=bounds,
        constraints={'type': 'ineq', 'fun': lambda r: constraint_distances(r).flatten()}
    )
    
    # Extract the results
    centers = np.array(centers)
    radii = result.x
    sum_radii = np.sum(radii)
    
    return centers, radii, sum_radii
