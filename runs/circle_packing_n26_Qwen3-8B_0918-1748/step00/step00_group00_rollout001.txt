import numpy as np
import cvxpy as cp

def run_packing():
    # Define the number of circles
    n = 26

    # Define variables for the centers and radii
    centers = cp.Variable((n, 2))
    radii = cp.Variable(n)

    # Define the constraints
    constraints = []

    # All circles must be within the unit square
    for i in range(n):
        constraints.append(centers[i, 0] - radii[i] >= 0)
        constraints.append(centers[i, 0] + radii[i] <= 1)
        constraints.append(centers[i, 1] - radii[i] >= 0)
        constraints.append(centers[i, 1] + radii[i] <= 1)

    # No two circles can overlap
    for i in range(n):
        for j in range(i + 1, n):
            dist = cp.norm(centers[i] - centers[j])
            constraints.append(dist >= radii[i] + radii[j] - 1e-12)

    # Objective function: maximize the sum of radii
    objective = cp.Maximize(cp.sum(radii))

    # Solve the optimization problem
    problem = cp.Problem(objective, constraints)
    problem.solve(solver=cp.SCS, eps=1e-7)

    # Extract the results
    centers_np = centers.value
    radii_np = radii.value

    # Check if the solution is feasible
    if not problem.status in ['OPTIMAL', 'OPTIMAL_INFEASIBLE']:
        raise RuntimeError(f"Optimization failed with status: {problem.status}")

    # Convert to numpy arrays
    centers = np.array(centers_np)
    radii = np.array(radii_np)

    # Compute the sum of radii
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
