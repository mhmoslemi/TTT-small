    radii = []

    radii.append(side_length / 2)

    centers = np.array(centers)

        # x is a flat array of [x1, y1, r1, x2, y2, r2, ...]

        flat_centers = flat_params[:2 * len(centers)].reshape((len(centers), 2))

        radii = flat_radii

        return -np.sum(radii)  # Minimize negative sum to maximize sum

        bounds.extend([(0, 1), (0, 1)])

        # x is a flat array of [x1, y1, r1, x2, y2, r2, ...]

        flat_centers = flat_params[:2 * len(centers)].reshape((len(centers), 2))
        flat_radii = flat_params[2 * len(centers):]
        radii = flat_radii

            x, y = flat_centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12:
                return -1

        # x is a flat array of [x1, y1, r1, x2, y2, r2, ...]
