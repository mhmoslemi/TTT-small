import numpy as np
import scipy.optimize as optimize
import matplotlib.pyplot as plt

def run_packing():
    # Define initial positions for a hexagonal packing of 26 circles
    # We will use a 5x5 grid with 2 rows offset (like a hex grid)
    # This is a preliminary layout, to be optimized later

    # Initial configuration
    centers = []
    # Create a hexagonal grid with 5 rows (alternating offsets)
    for i in range(5):
        for j in range(5):
            x = j + 0.5 * (i % 2)
            y = i / 2
            centers.append([x, y])
    for i in range(5):
        for j in range(5):
            x = j + 0.5 * (i % 2)
            y = i / 1.5
            centers.append([x, y])
    # Remove duplicates to make it 26
    centers = np.array(centers[:26])
    
    # Initial estimate for radii based on area
    area_total = 1.0
    radii = np.full(26, np.sqrt(area_total / np.pi / 26))
    
    # Optimization: maximize sum of radii with constraints
    def objective(r):
        return -np.sum(r)
    
    def constraint1(r):
        return np.sum(r**2) - 1.0 / np.pi
    
    def constraint2(r):
        return np.sum(r) - 2.636  # Target
    
    # Remove the last radius to allow for optimization
    r = radii.copy()
    r[-1] = 0.0
    cons = [{'type': 'ineq', 'fun': constraint1},
            {'type': 'ineq', 'fun': constraint2}]
    bounds = [(0, None) for _ in range(26)]
    
    # Optimize
    result = optimize.minimize(objective, r, bounds=bounds, constraints=cons, method='SLSQP')
    
    # Extract radii from result
    radii_opt = result.x
    # Ensure the sum of radii is at least 2.636
    sum_radii = np.sum(radii_opt)
    if sum_radii < 2.636:
        radii_opt = radii_opt * (2.636 / sum_radii) if sum_radii > 0 else np.full(26, 0)
    
    # Final check: scale radii to fit within the unit square
    max_dist = np.sqrt(2)
    radii_opt = radii_opt * (1.0 / max_dist)
    # Final validation
    centers = np.array(centers)
    radii_opt = np.array(radii_opt)
    # Scale radii if needed based on packing
    scale = 1.0
    for i in range(26):
        dist = np.sqrt((centers[i, 0] - centers[0, 0])**2 + (centers[i, 1] - centers[0, 1])**2)
        if dist < 1.0:
            scale = min(scale, 1.0 / dist)
    radii_opt = radii_opt * scale
    
    # Final validation for boundary and overlap
    for i in range(26):
        x, y = centers[i]
        r = radii_opt[i]
        if (x - r < 0 or x + r > 1 or y - r < 0 or y + r > 1):
            radii_opt[i] = 0
    for i in range(26):
        for j in range(i + 1, 26):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii_opt[i] + radii_opt[j]:
                radii_opt[i] = (dist - radii_opt[j]) * 0.9
                radii_opt[j] = (dist - radii_opt[i]) * 0.9
    
    return (centers, radii_opt, np.sum(radii_opt))
