import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial grid configuration (hexagonal packing with offset rows)
    n = 26
    grid_rows = 5
    grid_cols = 6
    x_coords = np.linspace(0, 1, grid_cols, endpoint=False)
    y_coords = np.linspace(0, 1, grid_rows, endpoint=False)
    centers = np.array([[
        x + (i % 2) * (x_coords[1] - x_coords[0])/2,
        y + (i // 2) * (y_coords[1] - y_coords[0])
    ] for i in range(grid_rows) for x in x_coords for y in y_coords])

    # Function to calculate total radius sum
    def total_radius(radii):
        return np.sum(radii)

    # Constraint: centers are within unit square
    def in_unit_square(x):
        return np.all((x >= 0) & (x <= 1))

    # Constraint: radius can't be negative
    def non_negative_radius(x):
        return np.all(x >= 0)

    # Constraint: circles don't overlap
    def no_overlap(x):
        centers = x[:, :2]
        radii = x[:, 2]
        for i in range(n):
            for j in range(i + 1, n):
                dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    # Initial guess for radii: grid-based
    initial_radii = np.ones(n) * 0.1

    # Optimization bounds and constraints
    bounds = [(0, 1) for _ in range(2 * n + n)]  # 2n for centers, n for radii
    constraints = [
        {'type': 'ineq', 'fun': lambda x: in_unit_square(x[:2*n].reshape(n, 2))},
        {'type': 'ineq', 'fun': lambda x: non_negative_radius(x[2*n:])},
        {'type': 'ineq', 'fun': lambda x: no_overlap(x)}
    ]

    # Optimization: minimize negative of total radius
    result = minimize(lambda x: -total_radius(x[2*n:]), x0=np.concatenate([centers.flatten(), initial_radii]),

    if not result.success:
