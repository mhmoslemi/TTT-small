import numpy as np
from scipy.optimize import minimize, Bounds
from scipy.spatial.distance import pdist, squareform
import math

def run_packing():
    # Initial configuration: Hexagonal packing
    # We'll start with a hexagonal grid in the unit square
    # We'll use a grid of points and adjust radii to fit within the square and avoid overlaps

    # Define grid parameters
    n_rows = 6
    n_cols = 5
    grid_spacing = 1.0 / (n_cols + 1)  # spacing between points
    grid_offset = grid_spacing / 2  # offset for hexagonal grid
    centers = []
    for row in range(n_rows):
        for col in range(n_cols):
            x = col * grid_spacing + grid_offset
            y = row * (grid_spacing * math.sqrt(3) / 2) + grid_offset
            centers.append([x, y])
    centers = np.array(centers)
    centers = np.vstack((centers, [0.5, 0.5], [0.5, 0.25], [0.5, 0.75], [0.25, 0.5], [0.75, 0.5]))  # add some extra points
    
    # Initial radii: all set to the minimum of distances to boundaries
    radii = []
    for x, y in centers:
        r = min(x, 1 - x, y, 1 - y)
        radii.append(r)
    radii = np.array(radii)
    
    # Optimization: maximize sum of radii, avoiding overlaps and boundaries
    # Define objective
    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum
    
    # Define constraints: distance between centers >= sum of radii
    def constraint_func(radii):
        dists = squareform(pdist(centers))
        return dists - (radii[:, np.newaxis] + radii[np.newaxis, :])
    
    # Optimization bounds
    bounds = [(0, 1) for _ in range(len(centers))]
    
    # Initial guess
    initial_guess = radii.copy()
    
    # Solve with Sequential Least Squares Programming (SLSQP)
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[
            {
                'type': 'ineq',
                'fun': lambda r: np.min(constraint_func(r)),
                'name': 'non_overlap'
            }
        ],
        method='SLSQP'
    )
    
    # Get result
    radii = result.x
    sum_radii = np.sum(radii)
    
    # Validate the packing
    valid, msg = validate_packing(centers, radii)
    if not valid:
        raise ValueError(f"Packing validation failed: {msg}")
    
    return (centers, radii, sum_radii)

def validate_packing(centers, radii):
    n = centers.shape[0]

    if np.isnan(centers).any() or np.isnan(radii).any():
        return False, "NaN values present"

    for i in range(n):
        if radii[i] < 0:
            return False, f"Circle {i} has negative radius {radii[i]}"

    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if (x - r < -1e-12 or x + r > 1 + 1e-12
                or y - r < -1e-12 or y + r > 1 + 1e-12):
            return False, f"Circle {i} at ({x},{y}) r={r} outside unit square"

    for i in range(n):
        for j in range(i + 1, n):
            dist = np.sqrt(np.sum((centers[i] - centers[j]) ** 2))
            if dist < radii[i] + radii[j] - 1e-12:
                return False, f"Circles {i} and {j} overlap"

    return True, "ok"
