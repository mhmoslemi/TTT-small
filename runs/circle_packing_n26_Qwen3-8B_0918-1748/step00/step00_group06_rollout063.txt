import numpy as np
from scipy.optimize import minimize

def run_packing():
    # Initial packing: hybrid grid with boundary clusters
    # Rows and columns based on a grid density, with some circles near edges
    rows = 5
    cols = 5
    base_radii = np.full(rows * cols, 1 / (2 * np.sqrt(2)))  # Initial estimate based on square packing
    centers = np.empty((rows * cols, 2))
    for i in range(rows):
        for j in range(cols):
            x = j / (cols - 1)
            y = i / (rows - 1)
            centers[i * cols + j] = [x, y]
    # Adjust radii based on position to include edge circles
    radii = np.copy(base_radii)
    radii[0] = 0.3  # Left bottom corner
    radii[1] = 0.3  # Left top corner
    radii[-2] = 0.3  # Right bottom corner
    radii[-1] = 0.3  # Right top corner
    radii[cols//2] = 0.25  # Center column

    def objective(radii):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_overlap(i, j):
        def cons(x):
            idx = np.arange(len(radii))
            radii[idx] = x
            # Check if circles i and j overlap
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx ** 2 + dy ** 2)
            if dist < radii[i] + radii[j] - 1e-8:
                return dist - radii[i] - radii[j] + 1e-8
            return 0
        return cons

    cons = []
    for i in range(len(radii)):
        for j in range(i + 1, len(radii)):
            cons.append({"type": "ineq", "fun": lambda x, i=i, j=j: constraint_overlap(i, j)(x)})

    result = minimize(objective, radii, constraints=cons, method='SLSQP', options={'ftol': 1e-8, 'eps': 1e-8})
    optimized_radii = result.x
    centers = np.array(centers)
    return centers, optimized_radii, np.sum(optimized_radii)
