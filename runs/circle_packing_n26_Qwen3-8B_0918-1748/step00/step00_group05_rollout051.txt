import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize import Bounds

def run_packing():
    def objective(radii_and_centers):
        nonlocal centers, radii
        centers = radii_and_centers[:26*2].reshape(26, 2)
        radii = radii_and_centers[26*2:].reshape(26)
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    def constraint_circle_in_square(x):
        nonlocal centers, radii
        centers = x[:26*2].reshape(26, 2)
        radii = x[26*2:].reshape(26)
        return np.array([centers[i, 0] - radii[i] >= -1e-12 and centers[i, 0] + radii[i] <= 1 + 1e-12 and
                         centers[i, 1] - radii[i] >= -1e-12 and centers[i, 1] + radii[i] <= 1 + 1e-12
                        for i in range(26)])

    def constraint_no_overlap(x):
        nonlocal centers, radii
        centers = x[:26*2].reshape(26, 2)
        radii = x[26*2:].reshape(26)
        distances = squareform(pdist(centers))
        for i in range(26):
            for j in range(i+1, 26):
                if distances[i, j] < radii[i] + radii[j] - 1e-12:
                    return np.array([False])
        return np.array([True])

    # Initialize with a hexagonal packing inside the square
    rows = 5
    cols = 5
    centers = np.zeros((26, 2))
    radii = np.zeros(26)

    # Create a hexagonal grid with varying row lengths
    offset = 0.5
    for row in range(5):
        if row % 2 == 0:
            start_col = 0
        else:
            start_col = 0.5
        for col in range(5):
            x = col + start_col
            y = row * np.sqrt(3) / 2
            if x < 1 and y < 1 and x + 1 < 1 and y + 1 < 1:
                centers[row * 5 + col, 0] = x
                centers[row * 5 + col, 1] = y
                radii[row * 5 + col] = 0.1

    # Add more circles for total of 26
    for i in range(26):
        if i >= 5 * 5:
            x = np.random.uniform(0, 1)
            y = np.random.uniform(0, 1)
            radii[i] = 0.05
            centers[i, 0] = x
            centers[i, 1] = y

    # Flatten centers and radii for optimization
    initial_guess = np.concatenate((centers.flatten(), radii))

    # Define bounds for positions and radii
    bounds = []
    for _ in range(26):
        bounds.append((0, 1))  # x-coordinate
        bounds.append((0, 1))  # y-coordinate
        bounds.append((0, 0.5))  # radius

    # Optimize using Sequential Least Squares Programming
    result = minimize(
        objective,
        initial_guess,
        bounds=bounds,
        constraints=[
            {'type': 'ineq', 'fun': constraint_circle_in_square},
            {'type': 'ineq', 'fun': constraint_no_overlap}
        ],
        method='SLSQP'
    )

    centers = result.x[:26*2].reshape(26, 2)
    radii = result.x[26*2:].reshape(26)
    sum_radii = np.sum(radii)

    return (centers, radii, sum_radii)
