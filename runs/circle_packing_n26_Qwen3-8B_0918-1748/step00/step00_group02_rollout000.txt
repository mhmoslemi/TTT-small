import numpy as np
from scipy.optimize import minimize
from scipy.spatial.distance import pdist, squareform
from scipy.optimize.linesearch import line_search_wolfe

def run_packing():
    # Initial configuration: 3 rows of 9 circles in a hexagonal pattern (total of 27)
    # Remove one circle to make it 26
    n = 26
    centers = np.zeros((n, 2))
    radii = np.zeros(n)
    row_radii = []
    
    # Calculate initial spacing for hexagonal arrangement
    total_width = 1.0
    total_height = 1.0
    row_count = 3
    col_count = 9
    spacing = 0.13
    row_spacing = spacing * np.sqrt(3)
    
    # Calculate column positions (left to right)
    col_positions = np.linspace(0 + spacing, 1 - spacing, col_count)
    
    # Calculate row positions (top to bottom)
    row_positions = np.linspace(0 + spacing, 1 - spacing, row_count)
    
    # Initial radii estimation
    max_radius = (total_width - 2 * spacing) / (col_count - 1) / 2
    max_radius = min(max_radius, (total_height - 2 * spacing) / (row_count - 1) / 2)
    
    # Create and populate the centers and radii
    index = 0
    for row in range(row_count):
        for col in range(col_count):
            if row % 2 == 0:
                x = col_positions[col]
            else:
                x = col_positions[col] + spacing / 2
            y = row_positions[row]
            centers[index, 0] = x
            centers[index, 1] = y
            radii[index] = max_radius
            index += 1
    
    # Remove one circle to get exactly 26 circles
    # Remove the last one
    centers = np.delete(centers, -1, axis=0)
    radii = np.delete(radii, -1)
    
    # Optimization to maximize sum of radii with constraints
    def objective(radii):
        return -np.sum(radii)  # minimize the negative
    
    def constraint_center(radii):
        return np.sum(radii) - 2.636
    
    def constraint_min_radius(radii):
        return np.min(radii)
    
    constraints = [
        {'type': 'ineq', 'fun': constraint_center, 'args': []},
        {'type': 'ineq', 'fun': constraint_min_radius, 'args': []}
    ]
    
    # Initial guess: same as current radii
    radii_initial = radii.copy()
    
    # Optimization with bounds
    bounds = [(0, 10) for _ in range(n)]
    
    result = minimize(objective, radii_initial, method='SLSQP', bounds=bounds, constraints=constraints)
    
    # Apply the optimization result
    radii = result.x
    
    # Final validation
    for i in range(n):
        x, y = centers[i]
        r = radii[i]
        if x - r < -1e-12 or x + r > 1 + 1e-12 or y - r < -1e-12 or y + r > 1 + 1e-12:
            raise ValueError(f"Circle {i} at ({x},{y}) with radius {r} is outside the unit square")
    
    for i in range(n):
        for j in range(i + 1, n):
            dx = centers[i, 0] - centers[j, 0]
            dy = centers[i, 1] - centers[j, 1]
            dist = np.sqrt(dx**2 + dy**2)
            if dist < radii[i] + radii[j] - 1e-12:
                raise ValueError(f"Circles {i} and {j} overlap")
    
    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
