import numpy as np
from scipy.optimize import minimize
from scipy.spatial import distance
from scipy.optimize import Bounds

def run_packing():
    def is_valid(centers, radii):
        n = centers.shape[0]
        for i in range(n):
            if radii[i] < 0:
                return False
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return False
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    def objective(x):
        centers = x[:26*2].reshape(26, 2)
        radii = x[26*2:]
        sum_radii = np.sum(radii)
        return -sum_radii  # Minimize negative sum to maximize

    def constraints(x):
        centers = x[:26*2].reshape(26, 2)
        radii = x[26*2:]
        for i in range(26):
            if radii[i] < 0:
                return np.array([np.inf])
            x, y = centers[i]
            if x - radii[i] < -1e-12 or x + radii[i] > 1 + 1e-12 or y - radii[i] < -1e-12 or y + radii[i] > 1 + 1e-12:
                return np.array([np.inf])
        for i in range(26):
            for j in range(i + 1, 26):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx**2 + dy**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return np.array([np.inf])
        return np.array([0.0])

    x0 = np.random.rand(26*2 + 26)
    bounds = Bounds(0, 1)
    result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=[{'type': 'ineq', 'fun': constraints}])
    x_opt = result.x
    centers = x_opt[:26*2].reshape(26, 2)
    radii = x_opt[26*2:]
    if is_valid(centers, radii):
        return (centers, radii, np.sum(radii))
    else:
        raise ValueError("Invalid packing found")

run_packing()
