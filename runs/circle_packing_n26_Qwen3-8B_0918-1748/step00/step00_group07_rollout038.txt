import numpy as np
import scipy.optimize as opt
from scipy.optimize import Bounds

def run_packing():
    # Initial base symmetric grid with 5x5 layout
    grid = np.mgrid[0:1:5j, 0:1:5j]
    centers = grid.T.reshape(-1, 2)
    radii = np.ones(25) * 0.1  # Initial guess for radii

    # Objective function to maximize the sum of radii
    def objective(radii, centers):
        return -np.sum(radii)  # Minimize negative sum to maximize sum

    # Constraint 1: All circles within the unit square
    def constraint1(radii, centers):
        return np.array([np.sqrt((centers[i, 0] - radii[i]) ** 2 + (centers[i, 1] - radii[i]) ** 2) <= 1 + 1e-12 for i in range(len(centers))])

    # Constraint 2: No overlapping between circles
    def constraint2(radii, centers):
        n = len(centers)
        dist = np.zeros((n, n))
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist[i, j] = np.sqrt(dx ** 2 + dy ** 2)
        return np.array([dist[i, j] >= radii[i] + radii[j] - 1e-12 for i in range(n) for j in range(i + 1, n)])

    # Optimize radii with constraints
    bounds = Bounds(np.zeros(len(radii)), np.inf)
    result = opt.minimize(
        fun=objective,
        x0=radii,
        args=(centers,),
        method='SLSQP',
        constraints=[
            {'type': 'ineq', 'fun': constraint1, 'args': (centers,)},
            {'type': 'ineq', 'fun': constraint2, 'args': (centers,)}
        ],
        bounds=bounds
    )

    # Add 26th circle at the center
    centers = np.vstack((centers, np.array([0.5, 0.5])))
    radii = np.append(result.x, 0.0)

    # Final optimization step to adjust all circles
    def optimize_all(radii, centers):
        n = len(centers)
        for i in range(n):
            for j in range(i + 1, n):
                dx = centers[i, 0] - centers[j, 0]
                dy = centers[i, 1] - centers[j, 1]
                dist = np.sqrt(dx ** 2 + dy ** 2)
                if dist < radii[i] + radii[j] - 1e-12:
                    # Adjust radii to maintain non-overlapping
                    radii[i] = (dist - 1e-12) / 2
                    radii[j] = (dist - 1e-12) / 2
        return radii

    radii = optimize_all(radii, centers)

    sum_radii = np.sum(radii)
    return centers, radii, sum_radii
