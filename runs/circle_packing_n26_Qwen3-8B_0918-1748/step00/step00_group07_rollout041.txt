```python
import numpy as np
import matplotlib.pyplot as plt
import scipy.optimize as opt
from scipy.spatial.distance import pdist, cdist
from scipy.optimize import Bounds
from typing import Tuple, List

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    def maximize_sum_radii(centers: np.ndarray, radii: np.ndarray) -> float:
        return np.sum(radii)

    def generate_hexagonal_grid(n_rows: int, spacing: float) -> np.ndarray:
        x = np.arange(0, n_rows * spacing, spacing)
        y = np.arange(0, np.sqrt(3) * spacing * (n_rows - 1) / 2, np.sqrt(3) * spacing)
        grid = np.empty((0, 2), dtype=np.float64)
        for i in range(n_rows):
            row_len = i + 1 if i < n_rows - 1 else 1
            for j in range(row_len):
                x_pos = x[j] if i % 2 == 0 else x[j] + spacing / 2
                y_pos = y[i]
                grid = np.vstack((grid, [x_pos, y_pos]))
        return grid

    def adjust_for_square(grid: np.ndarray, size: float = 1.0) -> np.ndarray:
        margin = size / 2
        grid[:, 0] = np.clip(grid[:, 0], margin, size - margin)
        grid[:, 1] = np.clip(grid[:, 1], margin, size - margin)
        return grid

    def compute_max_radius(grid: np.ndarray, size: float = 1.0) -> float:
        max_x = np.max(grid[:, 0])
        min_x = np.min(grid[:, 0])
        max_y = np.max(grid[:, 1])
        min_y = np.min(grid[:, 1])
        width = size - max_x - min_x
        height = size - max_y - min_y
        return np.min([width, height]) / 2

    def add_corner_circles(grid: np.ndarray, size: float = 1.0, radius: float = 0.05) -> np.ndarray:
        corner_radii = [0.05, 0.05, 0.05, 0.05]
        corner_centers = [
            [radius, radius],
            [size - radius, radius],
            [size - radius, size - radius],
            [radius, size - radius]
        ]
        return np.vstack((grid, corner_centers))

    def add_edge_circles(grid: np.ndarray, size: float = 1.0, radius: float = 0.05) -> np.ndarray:
        edge_radii = [0.05, 0.05, 0.05, 0.05]
        edge_centers = [
            [size / 2, radius],
            [size / 2, size - radius],
            [radius, size / 2],
            [size - radius, size / 2]
        ]
        return np.vstack((grid, edge_centers))

    def refine_packing(centers: np.ndarray, radii: np.ndarray, size: float = 1.0) -> Tuple[np.ndarray, np.ndarray]:
        n = len(centers)
        constraints = []
        for i in range(n):
            constraints.append(('ineq', lambda x: x[i, 0] - radii[i]))
            constraints.append(('ineq', lambda x: x[i, 1] - radii[i]))
            constraints.append(('ineq', lambda x: (size - x[i, 0]) - radii[i]))
            constraints.append(('ineq', lambda x: (size - x[i, 1]) - radii[i]))
        for i in range(n):
            for j in range(i + 1, n):
                constraints.append(('ineq', lambda x, i=i, j=j: np.sqrt((x[i, 0] - x[j, 0])**2 + (x[i, 1] - x[j, 1])**2) - (radii[i] + radii[j])))

        bounds = Bounds(0, 1)

        initial_centers = centers.copy()
        initial_radii = radii.copy()

        result = opt.minimize(
            fun=lambda x: -np.sum(x[n:]),
            x0=np.concatenate((initial_centers.flatten(), initial_radii)),
            constraints=constraints,
            bounds=bounds,
            method='SLSQP'
        )
        refined_centers = result.x[:n * 2].reshape(n, 2)
        refined_radii = result.x[n * 2:]
        return refined_centers, refined_radii

    # Step 1: Generate hexagonal grid
    hex_grid = generate_hexagonal_grid(7, 0.2)
    hex_grid = adjust_for_square(hex_grid)
    max_radius = compute_max_radius(hex_grid)
    hex_radii = np.full(len(hex_grid), max_radius)

    # Step 2: Add corner circles
    hex_grid = add_corner_circles(hex_grid)
    corner_radii = np.full(len(hex_grid) - len(hex_grid), 0.05)
    hex_radii = np.concatenate((hex_radii, corner_radii))

    # Step 3: Add edge circles
    hex_grid = add_edge_circles(hex_grid)
    edge_radii = np.full(len(hex_grid) - len(hex_grid), 0.05)
    hex_radii = np.concatenate((hex_radii, edge_radii))

    # Step 4: Refine packing
    refined_centers, refined_radii = refine_packing(hex_grid, hex_radii)

    # Final validation
    def validate_boundaries(centers: np.ndarray, radii: np.ndarray, size: float = 1.0) -> bool:
        for i in range(len(centers)):
            x, y = centers[i]
            r = radii[i]
            if x - r < -1e-12 or x + r > 1 + 1e-12:
                return False
            if y - r < -1e-12 or y + r > 1 + 1e-12:
                return False
        return True

    def validate_spacing(centers: np.ndarray, radii: np.ndarray) -> bool:
        for i in range(len(centers)):
            for j in range(i + 1, len(centers)):
                dist = np.sqrt((centers[i, 0] - centers[j, 0])**2 + (centers[i, 1] - centers[j, 1])**2)
                if dist < radii[i] + radii[j] - 1e-12:
                    return False
        return True

    if not validate_boundaries(refined_centers, refined_radii) or not validate_spacing(refined_centers, refined_radii):
        raise ValueError("Packing constraints not satisfied")

    sum_radii = np.sum(refined_radii)
    return refined_centers, refined_radii, sum_radii
```
