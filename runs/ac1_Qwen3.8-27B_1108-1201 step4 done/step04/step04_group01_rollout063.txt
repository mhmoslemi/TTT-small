def project_simplex(v):
    n=len(v)
    if n==1: return np.array([max(v[0],0.0)])
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    cond=(u*np.arange(1,n+1) > (cssv-1.0))
    if not cond.any(): return np.ones(n)/n
    rho=np.nonzero(cond)[0][-1]
    theta=(cssv[rho]-1.0)/(rho+1.0)
    return np.maximum(v-theta,0.0)

def polish_with_lse(x0, time_limit, seed=0, beta_start=8.0, beta_end=50000.0, maxiter=None):
    x=normalize_nonneg(x0)
    if x[0] < 1e-12 * x.sum() or (x.size>0 and x[0] < 0.001*np.max(x)):
        x=x[::-1].copy()
    c=fast_convolve(x)
    best_score=2.0*len(x)*c.max()
    best_x=x.copy()
    t0=time.monotonic(); deadline=t0+max(1.0,float(time_limit))
    n=len(x)
    if maxiter is None:
       maxiter = 50 if n <= 512 else 30
    bounds=[(-80.0,80.0)]*(n-1)
    beta=float(beta_start)
    stall=0
    attempt=0
    while time.monotonic() < deadline - 3.0:
        attempt += 1
        # if beta too high, final maxiter larger
        cur_maxiter = maxiter
        if beta > 5000:
            cur_maxiter = int(maxiter*1.5)
            if beta > 20000:
                cur_maxiter = int(maxiter*2)
        # perturb if stall
        if stall >= 2:
            scale = 0.003 + 0.004 * min(1.0, float(stall) / 10.0)
            rng = np.random.default_rng(seed + attempt * 1000003)
            x_pert = project_simplex(best_x + rng.standard_normal(n) * scale)
            # maybe reverse if base small
            if x_pert[0] < 0.001 * np.max(x_pert):
                x_pert = x_pert[::-1].copy()
            x = x_pert
        d = init_d(x)
        try:
            res = minimize(_lse_fun, d, args=(beta,), method='L-BFGS-B', jac=True, bounds=bounds,
                           options={'maxiter': cur_maxiter, 'ftol': 1e-12, 'gtol': 1e-7, 'maxls': 20})
            cand = None
            if res.x is not None and np.all(np.isfinite(res.x)):
                cand = d_to_x(res.x)
            if cand is not None:
                x = cand
            else:
                x = best_x.copy()
        except Exception:
            x = best_x.copy()
        c=fast_convolve(x)
        cur_score=2.0*n*c.max()
        if np.isfinite(cur_score) and cur_score < best_score - 1e-12:
            best_score=cur_score; best_x=x.copy(); stall=0
        else:
            stall += 1
        # If current much worse, revert
        if not np.isfinite(cur_score) or cur_score > best_score * 1.05 + 1e-12:
            x = best_x.copy()
        # beta update
        if stall == 0:
            beta *= 1.55
        elif stall == 1:
            beta *= 1.35
        elif stall == 2:
            beta *= 1.25
        else:

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for nonnegative a.

Algorithm:
- Work on the simplex (sum a = 1).
- Use several structured starts: generalized binomial prefixes
  (coefficients of (1-z)^(-alpha)), skewed mixes, interpolated parent,
  and random smooth starts.
- Polish each start by minimizing a log-sum-exp smoothing of the peak
  coefficients.  The sequence is parameterized as a softmax with index 0
  fixed, and L-BFGS-B is used on the resulting unconstrained problem.
- Keep the best actual sequence and periodically do a high-temperature
  final polish.
"""

import time
import numpy as np
from scipy.optimize import minimize


def next_power_of_two(m: int) -> int:
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_convolve(a: np.ndarray) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    out_len = 2 * n - 1
    fft_size = next_power_of_two(out_len)
    A = np.fft.rfft(a, fft_size)
    c = np.fft.irfft(A * A, fft_size)[:out_len]
    return np.maximum(c, 0.0)


def compute_grad_weights(w: np.ndarray, x: np.ndarray) -> np.ndarray:
    """
    For c_k = sum_i x_i x_{k-i}, return grad of sum_k w_k c_k wrt x:
    grad_j = 2 * sum_k w_k x_{k-j} over valid k.
    """
    x = np.asarray(x, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = x.size
    total_len = len(w) + n - 1
    fft_size = next_power_of_two(total_len)
    W = np.fft.rfft(w, fft_size)
    X = np.fft.rfft(x[::-1], fft_size)
    conv = np.fft.irfft(W * X, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def normalize_nonneg(a) -> np.ndarray:
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def project_simplex(v: np.ndarray) -> np.ndarray:
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([max(0.0, float(v[0]))], dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def ensure_base_positive(x: np.ndarray) -> np.ndarray:
    x = normalize_nonneg(x)
    if x.size <= 1:
        return x
    mx = float(np.max(x))
    if mx > 0.0 and x[0] < 5e-3 * mx:
        x = x[::-1].copy()
    return x


def d_to_x(d: np.ndarray) -> np.ndarray:
    d = np.clip(np.asarray(d, dtype=np.float64), -80.0, 80.0)
    n = d.size + 1
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    z = np.zeros(n, dtype=np.float64)
    z[1:] = d
    z -= z.max()
    p = np.exp(z)
    p /= p.sum()
    return p


def init_d(x: np.ndarray) -> np.ndarray:
    x = np.asarray(x, dtype=np.float64)
    n = x.size
    if n == 1:
        return np.array([], dtype=np.float64)
    x = np.maximum(x, 1e-30)
    s = float(x.sum())
    if s <= 0.0:
        return np.zeros(n - 1, dtype=np.float64)
    x = x / s
    if x[0] <= 1e-30:
        return np.full(n - 1, -80.0, dtype=np.float64)
    d = np.log(x[1:] / x[0])
    return np.clip(d, -80.0, 80.0)


def _lse_fun(d: np.ndarray, beta: float):
    """
    L-BFGS-B objective and gradient.
    Objective: log sum_k exp(beta * (a*a)_k)
    Gradient wrt softmax parameters d.
    """
    x = d_to_x(d)
    n = x.size
    c = fast_convolve(x)
    cm = float(np.max(c))

    shifted = beta * (c - cm)
    e = np.exp(shifted)
    s = float(e.sum())
    val = float(beta * cm + np.log(max(s, 1e-300)))

    w = e / max(s, 1e-300)
    gx = beta * compute_grad_weights(w, x)
    dot = float(np.dot(gx, x))

    if n == 1:
        grad = np.array([], dtype=np.float64)
    else:
        grad = gx[1:] * x[1:] - x[1:] * dot

    return val, grad


def generalized_binomial(n: int, alpha: float) -> np.ndarray:
    """
    Coefficients of (1-z)^(-alpha), normalized.
    For alpha=0.5: 1, 1/2, 3/8, ...
    """
    n = int(max(1, n))
    c = np.ones(n, dtype=np.float64)
    for i in range(1, n):
        c[i] = c[i - 1] * (alpha + i - 1.0) / float(i)
    c = np.maximum(c, 0.0)
    s = float(c.sum())
    if s <= 0.0:
        return np.ones(n, dtype=np.float64) / n
    return c / s


def mixed_alpha_start(n: int, alpha: float, p: float) -> np.ndarray:
    n = int(max(1, n))
    base = generalized_binomial(n, alpha)
    p = float(np.clip(p, 0.0, 1.0))
    if p >= 1.0:
        x = base.copy()
    elif p <= 0.0:
        x = base[::-1].copy()
    else:
        x = p * base + (1.0 - p) * base[::-1]
    return ensure_base_positive(x)


def random_smooth_start(n: int, seed: int) -> np.ndarray:
    n = int(max(1, n))
    rng = np.random.default_rng(seed)
    if n < 8:
        y = np.maximum(rng.standard_normal(n), 0.0) + 1e-6
        return ensure_base_positive(y)
    y = rng.standard_normal(n)
    klen = max(3, n // 6)
    t = np.linspace(-3.0, 3.0, klen)
    kernel = np.exp(-0.5 * t * t)
    kernel /= kernel.sum()
    y = np.convolve(y, kernel, mode="same")
    y = np.maximum(y, 1e-6)
    return ensure_base_positive(y)


def interpolate_sequence(x: np.ndarray, n: int) -> np.ndarray:
    x = normalize_nonneg(x)
    n = int(max(1, n))
    if x.size == n:
        return ensure_base_positive(x)
    if x.size == 1:
        return ensure_base_positive(np.full(n, x[0]))
    old_t = np.linspace(0.0, 1.0, x.size)
    new_t = np.linspace(0.0, 1.0, n)
    y = np.interp(new_t, old_t, x)
    return ensure_base_positive(y)


def polish_with_lse(
    x0: np.ndarray,
    time_limit: float,
    seed: int = 0,
    beta_start: float = 8.0,
    beta_end: float = 50000.0,
    maxiter=None,
):
    """
    Local polish by continuation in beta for log-sum-exp smoothing.
    Returns (best_x, best_internal_score), where best_internal_score = 2*n*max conv.
    """
    x = ensure_base_positive(x0)
    n = x.size
    if n <= 1:
        return x, float(2.0 * n * np.max(fast_convolve(x))) if n > 0 else float("inf")

    c = fast_convolve(x)
    best_score = float(2.0 * n * np.max(c))
    best_x = x.copy()

    t0 = time.monotonic()
    deadline = t0 + max(1.0, float(time_limit))

    if maxiter is None:
        if n <= 384:
            maxiter = 150
        elif n <= 448:
            maxiter = 120
        elif n <= 600:
            maxiter = 80
        else:
            maxiter = 50

    bounds = [(-80.0, 80.0)] * (n - 1)
    beta = float(beta_start)
    if beta > beta_end:
        beta = float(beta_end)

    stall = 0
    final_count = 0
    attempt = 0

    while time.monotonic() < deadline - 2.0:
        attempt += 1

        cur_maxiter = int(maxiter)
        if beta > 5000.0:
            cur_maxiter = int(cur_maxiter * 1.5)
        if beta > 20000.0:
            cur_maxiter = int(cur_maxiter * 2.0)

        # Occasional small random restarts from the best known point.
        if stall >= 2:
            scale = 0.004 + 0.005 * min(1.0, float(stall) / 8.0)
            rng = np.random.default_rng(seed + attempt * 1000003)
            x_pert = project_simplex(best_x + rng.standard_normal(n) * scale)
            if x_pert[0] < 5e-3 * float(np.max(x_pert)):
                x_pert = x_pert[::-1].copy()
            x = x_pert

        d = init_d(x)

        try:
            res = minimize(
                _lse_fun,
                d,
                args=(beta,),
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                options={
                    "maxiter": max(1, cur_maxiter),
                    "ftol": 1e-13,
                    "gtol": 1e-8,
                    "maxls": 15,
                },
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                x = d_to_x(res.x)
            else:
                x = best_x.copy()
        except Exception:
            x = best_x.copy()

        c = fast_convolve(x)
        cur_score = float(2.0 * n * np.max(c))

        if np.isfinite(cur_score) and cur_score < best_score - 1e-12:
            best_score = cur_score
            best_x = x.copy()
            stall = 0
        else:
            stall += 1

        # Re-center the softmax base if index 0 becomes very small.
        mx = float(np.max(x))
        if mx > 0.0 and x[0] < 5e-4 * mx:
            x = x[::-1].copy()
            best_x = best_x[::-1].copy()

        # If the current point has moved too far from the best in exact score, revert.
        if (not np.isfinite(cur_score)) or cur_score > best_score * 1.05 + 1e-10:
            x = best_x.copy()

        if attempt % 5 == 0:
            elapsed = time.monotonic() - t0
            print(
                f"      LSE beta={beta:.0f} stall={stall} "
                f"score={best_score:.10f} elapsed={elapsed:.1f}s"
            )

        if beta >= beta_end:
            final_count += 1
            if final_count >= 4:
                break
            beta = float(beta_end)
        else:
            final_count = 0
            if stall == 0:
                beta *= 1.55
            elif stall == 1:
                beta *= 1.35
            elif stall == 2:
                beta *= 1.25
            else:
                beta *= 2.0
            if beta > beta_end:
                beta = float(beta_end)

        if beta >= beta_end and stall >= 5:
            break

    return best_x, best_score


def evaluate_with_fallback(x: np.ndarray, fallback_score: float) -> float:
    try:
        val = evaluate_sequence(np.asarray(x, dtype=np.float64).tolist())

    except Exception:

        return float("inf")


def attempt_polish(
    x0: np.ndarray,
    time_limit: float,
    seed: int,
    best_score: float,
    best_x: np.ndarray,
    beta_start: float = 8.0,

        return best_score, best_x

    if np.isfinite(actual) and actual < best_score:

        best_x = x_opt.copy()

    return best_score, best_x

        final_score = evaluate_with_fallback(
            seq, float(2.0 * n_parent * np.max(fast_convolve(seq)))
        )

    print(f"FINAL returned length={len(seq)} score={final_score:.10f}")
    return [float(v) for v in seq]
