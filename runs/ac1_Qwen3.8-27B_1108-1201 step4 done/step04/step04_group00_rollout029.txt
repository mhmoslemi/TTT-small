def coordinate_descent_gsf(f0, time_limit):
    """Coordinate descent with golden section search per coordinate."""
    f = normalize(f0)
    n = len(f)
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    golden_ratio = (1 + sqrt(5)) / 2
    inv_phi2 = 1 - 1/golden_ratio  # ≈ 0.382
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[i]
            if old_val < 1e-12:
                continue
            
            # Golden section search for optimal value of coordinate i
            # t ranges from 0 to 1 (but practically from 0 to 1 - epsilon)
            lo, hi = 0.0, 1.0 - 1e-10
            
            def obj_1d(t):
                # Compute max convolution peak when coordinate i is set to t
                new_f = f.copy()
                new_f[i] = t
                scale = (1.0 - t) / (1.0 - old_val) if (1.0 - old_val) > 1e-12 else 0
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                new_f[mask] *= scale
                new_f = new_f / new_f.sum()  # ensure exact normalization
                return fast_convolve(new_f).max()
            
            # Golden section
            c = hi - inv_phi2 * (hi - lo)
            d = lo + inv_phi2 * (hi - lo)
            fc = obj_1d(c)
            fd = obj_1d(d)
            
            for _ in range(30):  # ~30 iterations for good precision
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - inv_phi2 * (hi - lo)
                    fc = obj_1d(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + inv_phi2 * (hi - lo)
                    fd = obj_1d(d)
            
            best_t = (lo + hi) / 2
            best_obj_new = obj_1d(best_t)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. LP-Frank-Wolfe for initial convergence
2. Fast coordinate descent with golden section search (O(n) per 1D eval)
3. Subgradient polishing with adaptive peak weighting
4. Multi-length exploration with interpolation and diverse restarts

import numpy as np
from scipy.optimize import linprog
import time
import math


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64)
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normalize(a):
    """Normalize to sum=1, clip negatives."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive M relaxation."""
    f = normalize(f0.copy())
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                m_exp = 1.0
            continue
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 100:
                break
            m_exp = min(m_exp * 1.05, 5.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > 80:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.003 * min(no_improve // 50, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def cd_golden_search(f0, time_limit, verbose=False):
    """
    Coordinate descent with golden section search.
    Uses O(n) per 1D evaluation by exploiting the structure of single-coordinate changes.
    """
    f = normalize(f0.copy())
    n = len(f)
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    inv_phi2 = 1.0 - 1.0 / phi  # ≈ 0.381966
    
    pass_count = 0
    no_improve_passes = 0
    
    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        pass_improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[i]
            if old_val < 1e-14:
                continue
            
            # Compute full convolution once
            conv_full = fast_convolve(f)
            
            # Precompute S_k and off_diag for coordinate i
            # S_k = conv_full[k] - 2*a_i*a_{k-i} (if k-i valid and != i)
            #              = conv_full[k] - a_i^2 (if k == 2i)
            #              = conv_full[k] (otherwise)
            # off_diag_k = a_{k-i} (if valid and != i), else 0
            # delta_k = 1 if k == 2i, else 0
            
            S = conv_full.copy()
            off_diag = np.zeros(2 * n - 1)
            delta = np.zeros(2 * n - 1)
            
            # For each k, the term a_i * a_{k-i} exists if 0 <= k-i < n
            k_start = max(0, i)  # k-i >= 0 => k >= i... wait, k ranges 0 to 2n-2
            # Actually k-i must be in [0, n-1], so k must be in [i, i+n-1]
            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)
            
            for k in range(k_lo, k_hi + 1):
                ki = k - i  # = k - i, which is in [0, n-1]
                if ki == i:
                    # 2i = k
                    S[k] -= old_val * old_val
                    delta[k] = 1.0
                else:
                    S[k] -= 2.0 * old_val * f[ki]
                    off_diag[k] = f[ki]
            
            # Now do golden section search for t in [0, 1)
            # s(t) = (1-t) / (1-old_val)
            # q_k(t) = s(t)^2 * S[k] + 2*t*s(t)*off_diag[k] + t^2 * delta[k]
            # obj(t) = max_k q_k(t)
            
            denom = 1.0 - old_val
            if denom < 1e-12:
                continue
            
            def obj_1d(t):
                s = (1.0 - t) / denom
                # q = s^2 * S + 2*t*s*off_diag + t^2 * delta
                q = s * s * S + 2.0 * t * s * off_diag + t * t * delta
                return q.max()
            
            # Golden section in [0, 1 - eps]
            lo = 0.0
            hi = 1.0 - 1e-10
            
            c = hi - inv_phi2 * (hi - lo)
            d = lo + inv_phi2 * (hi - lo)
            fc = obj_1d(c)
            fd = obj_1d(d)
            
            for _ in range(25):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - inv_phi2 * (hi - lo)
                    fc = obj_1d(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + inv_phi2 * (hi - lo)
                    fd = obj_1d(d)
            
            best_t = (lo + hi) / 2.0
            best_obj_new = obj_1d(best_t)
            
            if best_obj_new < best_obj - 1e-13:
                # Update
                f[i] = best_t
                scale = (1.0 - best_t) / denom
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                f[mask] *= scale
                f = f / f.sum()
                best_obj = best_obj_new
                best_f = f.copy()
                pass_improved = True
        
        if verbose and pass_count % 5 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}")
        
        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 3:
                break
        else:
            no_improve_passes = 0
    
    return best_f, best_obj

            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def cd_fast(f0, time_limit, verbose=False, seed=0):
    """
    Fast coordinate descent using O(n) per 1D evaluation.
    When coordinate i changes to t and others are scaled by (1-t)/(1-a_i),
    each convolution value is a quadratic in t, computable in O(1) after precomputation.
    """
    f = normalize(f0.copy())
    n = len(f)
    best_f = f.copy()
    conv = fast_convolve(f)
    best_obj = conv.max()
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)

    phi = (1.0 + math.sqrt(5.0)) / 2.0
    inv_phi2 = 1.0 - 1.0 / phi

    pass_count = 0
    no_improve_passes = 0
    order = list(range(n))

    while time.monotonic() - t0 < time_limit:
        pass_count += 1
        pass_improved = False
        rng.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[i]
            if old_val < 1e-14:
                continue

            denom = 1.0 - old_val
            if denom < 1e-12:
                continue

            # Precompute S_k, off_diag, delta for this coordinate
            conv_full = fast_convolve(f)
            S = conv_full.copy()
            off_diag = np.zeros(2 * n - 1)

            k_lo = max(0, i)
            k_hi = min(2 * n - 2, i + n - 1)

            for k in range(k_lo, k_hi + 1):
                ki = k - i
                if ki == i:
                    S[k] -= old_val * old_val
                else:
                    S[k] -= 2.0 * old_val * f[ki]
                    off_diag[k] = f[ki]

            # delta is 1 at k=2i
            delta_pos = 2 * i
            if delta_pos < 2 * n - 1:
                # The t^2 coefficient at delta_pos gets +1
                pass
            else:
                delta_pos = -1

            # 1D objective: q_k(t) = s^2*S_k + 2*t*s*off_k + t^2*delta_k
            # where s = (1-t)/denom
            # Expand: s = (1-t)/denom
            # q_k = (1-t)^2/denom^2 * S_k + 2*t*(1-t)/denom * off_k + t^2 * delta_k
            #     = (1 - 2t + t^2)*S_k/denom^2 + (2t - 2t^2)*off_k/denom + t^2*delta_k
            # Coefficients:
            # constant: S_k/denom^2
            # linear: -2*S_k/denom^2 + 2*off_k/denom
            # quadratic: S_k/denom^2 - 2*off_k/denom + delta_k

            # Precompute coefficients for all k
            denom2 = denom * denom
            # For each k: A_k*t^2 + B_k*t + C_k
            # C_k = S[k] / denom^2
            # B_k = -2*S[k]/denom^2 + 2*off_diag[k]/denom
            # A_k = S[k]/denom^2 - 2*off_diag[k]/denom + (1 if k==2i else 0)

            C_vec = S / denom2
            B_vec = -2.0 * S / denom2 + 2.0 * off_diag / denom
            A_vec = S / denom2 - 2.0 * off_diag / denom
            if delta_pos >= 0:
                A_vec[delta_pos] += 1.0

            # obj(t) = max_k (A_k*t^2 + B_k*t + C_k)
            # Use sampling + local search

            # First: try a set of candidate t values
            candidates = [0.0, old_val * m for m in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0]]
            candidates = [max(0.0, min(1.0 - 1e-10, c)) for c in candidates]
            candidates = [1.0 - 1e-10] + candidates

            best_t = old_val
            best_val = best_obj

            for t_val in candidates:
                q = A_vec * t_val * t_val + B_vec * t_val + C_vec
                val = q.max()
                if val < best_val - 1e-15:
                    best_val = val
                    best_t = t_val

            # Local golden section refinement around best_t
            span = 0.05 * max(old_val, 0.01)
            lo_g = max(0.0, best_t - span)
            hi_g = min(1.0 - 1e-10, best_t + span)

            if hi_g - lo_g > 1e-12:
                c_g = hi_g - inv_phi2 * (hi_g - lo_g)
                d_g = lo_g + inv_phi2 * (hi_g - lo_g)

                q_c = A_vec * c_g * c_g + B_vec * c_g + C_vec
                q_d = A_vec * d_g * d_g + B_vec * d_g + C_vec
                fc = q_c.max()
                fd = q_d.max()

                for _ in range(20):
                    if fc < fd:
                        hi_g = d_g
                        d_g = c_g
                        fd = fc
                        c_g = hi_g - inv_phi2 * (hi_g - lo_g)
                        q_c = A_vec * c_g * c_g + B_vec * c_g + C_vec
                        fc = q_c.max()
                    else:
                        lo_g = c_g
                        c_g = d_g
                        fc = fd
                        d_g = lo_g + inv_phi2 * (hi_g - lo_g)
                        q_d = A_vec * d_g * d_g + B_vec * d_g + C_vec
                        fd = q_d.max()

                local_best_t = (lo_g + hi_g) / 2.0
                q_lb = A_vec * local_best_t * local_best_t + B_vec * local_best_t + C_vec
                local_val = q_lb.max()
                if local_val < best_val - 1e-15:
                    best_val = local_val
                    best_t = local_best_t

            # Accept if improved
            if best_val < best_obj - 1e-13:
                f[i] = best_t
                scale = (1.0 - best_t) / denom
                mask = np.ones(n, dtype=bool)
                mask[i] = False
                f[mask] *= scale
                f = f / f.sum()
                best_obj = best_val
                best_f = f.copy()
                pass_improved = True

        if verbose and pass_count % 3 == 0:
            print(f"    CD pass {pass_count}: obj={best_obj:.10f}")

        if not pass_improved:
            no_improve_passes += 1
            if no_improve_passes >= 2:
                break
        else:
            no_improve_passes = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.93, top_k=30, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = normalize(f0.copy())
    n = len(f)
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-13:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idxs = np.arange(js, je + 1)
            sg[idxs] += 2.0 * w[idx] * f[k - idxs]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9995
        cur_temp *= 1.004

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def load_parent():
    """Load parent sequence from global variable if available."""
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            return p / p.sum()
    except Exception:
        pass
    return None


def propose_candidate(budget_s=1800, **kwargs):
    """
    Multi-phase optimization combining LP-FW, fast coordinate descent with
    golden section, subgradient polishing, and multi-length search.
    """
    rng = np.random.default_rng(12345)
    deadline = time.monotonic() + max(1.0, float(budget_s)) - 25.0

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            return
        a_list = [float(x) for x in a]
        score = float(evaluate_sequence(a_list))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = load_parent()
    if parent is not None:
        print(f"Parent loaded: n={len(parent)}")
        p_score = try_seq(parent, "parent")
        print(f"  Parent score: {p_score:.6f}")
    else:
        parent = np.ones(384) / 384
        print("No parent, using uniform n=384")

    # === Phase 1: LP-FW on parent ===
    print("\n=== Phase 1: LP-FW on parent ===")
    t1 = min(120, deadline - time.monotonic() - 600)
    if t1 > 30:
        a_opt, obj = lp_fw_optimize(parent, time_limit=t1, verbose=True)
        try_seq(a_opt, "lpfw_parent")
    else:
        a_opt = parent.copy()

    base = best_seq if best_seq is not None else a_opt
    n_base = len(base)

    # === Phase 2: Fast CD on parent length ===
    print(f"\n=== Phase 2: Fast CD (n={n_base}) ===")
    rem = deadline - time.monotonic() - 400
    if rem > 20:
        t_cd = min(rem * 0.5, 200)
        a_opt, obj = cd_fast(base, time_limit=t_cd, verbose=True, seed=1)
        try_seq(a_opt, "cd_parent")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else base
    target_lengths = []
    n_base = len(base)
    # Try lengths around and above the current best
    for tlen in [512, 768, 1024, 1536, 2048]:
        if tlen > n_base or (tlen == n_base and tlen != 768):
            target_lengths.append(tlen)
    # Also include n_base if not already there
    if n_base not in target_lengths and n_base >= 384:
        target_lengths.insert(0, n_base)
    target_lengths = sorted(target_lengths)

    for tlen in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 30:
            break
        print(f"  Trying n={tlen}, remaining={rem:.0f}s")
        f_interp = interpolate_sequence(base, tlen)

        # Quick LP-FW
        t_lp = min(rem * 0.3, 60)
        if t_lp > 15:
            a_lp, _ = lp_fw_optimize(f_interp, time_limit=t_lp)
            try_seq(a_lp, f"lpfw_n{tlen}")
            f_interp = best_seq if best_seq is not None and len(best_seq) == tlen else a_lp

        # Fast CD
        rem = deadline - time.monotonic() - 150
        if rem > 15:
            t_cd2 = min(rem * 0.5, 120)
            a_cd, _ = cd_fast(f_interp, time_limit=t_cd2, verbose=False, seed=tlen)
            try_seq(a_cd, f"cd_n{tlen}")

    # === Phase 4: Deep optimization on best ===
    print(f"\n=== Phase 4: Deep CD (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        # Multiple CD runs with different seeds and perturbations
        for p in range(6):
            rem = deadline - time.monotonic() - 80
            if rem < 15:
                break
            t_alloc = min(rem * 0.4, 100)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.5 * best_seq + 0.5 * best_seq[::-1]
                f_init = project_simplex(f_init)
            elif p == 3:
                f_init = 0.8 * best_seq + 0.2 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)
            elif p == 4:
                # Sparse perturbation
                f_init = best_seq.copy()
                idx = rng.choice(n_best, size=max(1, n_best // 10), replace=False)
                f_init[idx] *= rng.uniform(0.5, 1.5, size=len(idx))
                f_init = project_simplex(f_init)
            else:
                # Shifted perturbation
                shift = rng.integers(1, max(2, n_best // 50))
                f_init = np.roll(best_seq, shift)
                f_init = project_simplex(f_init)

            a_opt, _ = cd_fast(f_init, time_limit=t_alloc, verbose=(p == 0), seed=100 + p)
            try_seq(a_opt, f"deep_cd_{p}")

    # === Phase 5: Subgradient polish ===
    print(f"\n=== Phase 5: SG Polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 50
        if rem > 15:
            a_opt, _ = sg_polish(best_seq, time_limit=min(rem * 0.6, 60),
                                 lr=0.001, mom=0.94,
                                 top_k=max(15, n_best // 10), temp=0.00003)
            try_seq(a_opt, "sg_polish")
            if best_seq is not None:
                # CD after SG
                rem = deadline - time.monotonic() - 30
                if rem > 10:
                    a_opt, _ = cd_fast(best_seq, time_limit=min(rem, 40), seed=999)
                    try_seq(a_opt, "cd_after_sg")

    # === Phase 6: Final micro-optimization ===
    print(f"\n=== Phase 6: Final (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        rem = deadline - time.monotonic() - 15
        if rem > 5:
            # Tiny perturbations + quick CD
            for attempt in range(3):
                rem = deadline - time.monotonic() - 10
                if rem < 5:
                    break
                noise = rng.standard_normal(n_best) * 0.001
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = cd_fast(f_pert, time_limit=min(rem, 15), seed=200 + attempt)
                try_seq(a_opt, f"micro_{attempt}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence([float(x) for x in best_seq]))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
