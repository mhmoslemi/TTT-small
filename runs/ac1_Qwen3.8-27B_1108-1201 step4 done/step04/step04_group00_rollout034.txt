import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)]) * s
    # First project onto {x >= 0, sum(x) = 1}, then scale by s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) * s / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    result_sum = result.sum()
    if result_sum > 0:
        result = result * s / result_sum
    else:
        result = np.ones(n) * s / n
    return result

def subgradient_of_peak(a, peaks, weights):
    """Compute weighted subgradient of max_k (a*a)[k] at point a."""
    n = len(a)
    sg = np.zeros(n)
    for k, w in zip(peaks, weights):
        if w < 1e-15:
            continue
        k = int(k)
        # (g_k)_j = 2*a_{k-j} if 0 <= k-j < n
        js = max(0, k - (n - 1))
        je = min(n - 1, k)
        # sg[j] += w * 2 * a[k-j]
        # j ranges from js to je, and k-j ranges from k-je to k-js
        idx = np.arange(js, je + 1)
        partners = k - idx
        sg[idx] += w * 2.0 * a[partners]
    return sg

    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v_mom = np.zeros(n)
    lr = lr_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    prev_a = a.copy()
    prev_sgd = None
    
    while time.monotonic() - t0 < time_limit:
        it += 1

        k_actual = min(top_k, 2 * n - 1)
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        temp = max(peak_vals.max() * 0.01, 1e-12)

def pair_swaps(a0, time_limit, n_swaps_per_iter=100, verbose=False):

def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

def diverse_initializations(n, seed=42):

    e = np.exp(-0.5 * ((x - n/2) / (n/4))**2)

    tri = np.maximum(tri, 0.1)
    inits.append(tri / tri.sum())
    
    # Random with various distributions
    r1 = rng.dirichlet(np.ones(n))
    inits.append(r1)

    alt = np.ones(n)

    # Flat center
    flat = np.ones(n)

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v, s=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)]) * s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) * s / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    rs = result.sum()
    if rs > 1e-15:
        result = result * s / rs
    else:
        result = np.ones(n) * s / n
    return result

    # j + m ranges from j to j+n-1, which is in [0, 2n-2] since j in [0,n-1]
    for j in range(n):
        # m from 0 to n-1, index j+m from j to j+n-1
        sg[j] = 2.0 * np.dot(weights[j:j+n], a)
    return sg


def compute_softmax_subgradient_fft(a, conv, temperature):
    """Compute smoothed subgradient using FFT for O(n log n) cost."""
    n = len(a)
    scaled = conv / temperature
    scaled -= scaled.max()
    weights = np.exp(scaled)
    ws = weights.sum()
    if ws < 1e-30:
        weights = np.zeros(len(conv))
        weights[np.argmax(conv)] = 1.0
        ws = 1.0
    weights /= ws
    
    # sg[j] = 2 * sum_{m=0}^{n-1} weights[j+m] * a[m]
    # This is a cross-correlation. Define:
    # corr[j] = sum_m weights[j+m] * a[m]
    # = sum_m a[m] * weights[j+m]
    # If we reverse weights: w_rev[k] = weights[2n-2-k]
    # Then corr[j] = sum_m a[m] * w_rev[2n-2-j-m] = (a * w_rev)[2n-2-j] reversed...
    # 
    # Alternative: corr[j] = inner product of a with weights[j:j+n]
    # This is a sliding dot product, which is a cross-correlation.
    # corr = correlate(weights, a) [circular], but we need linear.
    
    # Use: corr[j] = sum_{m=0}^{n-1} a[m] * weights[j+m]
    # Let's define b = a, c = weights. We want (c ⊛ b)[j] = sum_m c[j+m] * b[m]
    # = sum_m b[m] * c[j+m]
    # = (b * c_flipped)[len(b)-1+j] ... hmm.
    
    # Simplest FFT approach:
    # corr[j] = sum_{m=0}^{n-1} a[m] * weights[j+m]
    # Pad both to size >= n + (2n-1) - 1 = 3n - 2
    fft_size = 1 << (3 * n + 2).bit_length()
    A_fft = np.fft.rfft(a, fft_size)
    # weights[j+m] for m=0..n-1: this is weights shifted
    # corr[j] = sum_m a[m] * weights[j+m]
    # = <a, weights[j:j+n]>
    # = (a_flipped * weights)[j] where a_flipped[i] = a[n-1-i]
    # = ifft(fft(a_flipped) * fft(weights))[j]... not quite.
    
    # Let me just do: corr = (a_conj * weights) [linear convolution] evaluated at specific indices
    # Actually, the cleanest way:
    # corr[j] = sum_m a[m] * w[j+m] 
    # Let u[m] = a[m], v[k] = w[k]
    # corr[j] = sum_m u[m] * v[j+m]
    # Define v2[k] = v[-k] (reversed). Then v[j+m] = v2[-j-m].
    # corr[j] = sum_m u[m] * v2[-j-m] = (u * v2)[-j] (linear convolution)
    # So corr[j] = conv(u, v2)[-j] where conv is linear convolution.
    # For j >= 0: conv(u, v2)[N-1-j] where appropriate.
    
    # Actually, let me just use a different trick:
    # corr[j] = sum_m a[m] * weights[j+m]
    # = sum_{m+j = k} a[m] for k ranging... no.
    
    # I'll use the direct convolution:
    # Extend weights to length n + (2n-1) = 3n-1 (pad with zeros at end)
    # Then corr[j] = dot(a, weights[j:j+n]) for j = 0, 1, ..., n-1
    # This is the "cross-correlation" of a and weights.
    # Using numpy: np.correlate(weights, a, mode='valid') gives exactly this!
    
    # np.correlate(weights, a, mode='valid')[j] = sum_m weights[j+m] * a[m]
    # Wait, numpy correlate: (a * b)[n] = sum_k a[k] * conj(b[n-k])
    # For real: np.correlate(weights, a, 'valid')[j] = sum_k weights[j+k] * a[k]
    # Yes! That's exactly what we want.
    
    corr = np.correlate(weights, a, mode='valid')
    # corr has length (2n-1) - n + 1 = n
    # corr[j] = sum_{k=0}^{n-1} weights[j+k] * a[k]
    sg = 2.0 * corr[:n]
    return sg


def smoothed_subgradient_optimize(a0, time_limit, lr_init=0.05, momentum=0.9,
                                    temp_init=0.001, temp_decay=0.999,
                                    verbose=False):
    """Optimize using smoothed subgradient with multiplicative weights and momentum."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    peak = conv.max()
    best_obj = peak
    best_a = a.copy()
    
    v_mom = np.zeros(n)
    lr = lr_init
    temp = temp_init
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if it > 500000:
            break
        
        # Compute smoothed subgradient
        sg = compute_softmax_subgradient_fft(a, conv, temp)
        
        # BB-like step size adjustment
        if it > 1 and hasattr(smoothed_subgradient_optimize, '_prev'):
            pass  # Skip BB for simplicity
        
        # Momentum update
        v_mom = momentum * v_mom + lr * sg
        a_candidate = a - v_mom
        
        # Project to simplex
        a_candidate = project_simplex(a_candidate, 1.0)
        
        # Evaluate
        conv_new = fast_convolve(a_candidate)
        peak_new = conv_new.max()
        
        if peak_new < best_obj - 1e-15:
            best_obj = peak_new
            best_a = a_candidate.copy()
            no_improve = 0
            a = a_candidate
            conv = conv_new
        else:
            a = a_candidate
            conv = conv_new
            no_improve += 1
        
        # Decay temperature (focus more on the main peak)
        temp *= temp_decay
        # Slightly decay learning rate
        lr *= 0.9999
        
        if verbose and it % 500 == 0:
            print(f"      iter {it}: peak={best_obj:.10f}, temp={temp:.6f}, lr={lr:.6f}")
        
        if no_improve > 300:
            # Plateau: perturb and reset
            rng = np.random.default_rng(it * 7 + 3)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003, 1.0)
            v_mom = np.zeros(n)
            conv = fast_convolve(a)
            no_improve = 0
            lr = max(lr_init * 0.5, 0.005)
            temp = temp * 2.0  # Spread weights more
    
    return best_a, best_obj


def multiplicative_weights_optimize(a0, time_limit, lr=0.01, verbose=False):
    """Mirror descent (multiplicative weights) on the simplex."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute subgradient
        sg = compute_softmax_subgradient_fft(a, conv, temp)
        
        # Multiplicative weights update (mirror descent with entropy)
        # a_new = a * exp(-lr * sg) / sum(a * exp(-lr * sg))
        exponent = -lr * sg
        exponent -= exponent.max()  # Numerical stability
        scales = np.exp(exponent)
        a_new = a * scales
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-30:
            a_new = a.copy()
        else:
            a_new /= a_new_sum
        
        conv_new = fast_convolve(a_new)
        peak_new = conv_new.max()
        
        if peak_new < best_obj - 1e-15:
            best_obj = peak_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        conv = conv_new
        temp *= 0.9999
        lr *= 0.99995
        
        if no_improve > 200:
            rng = np.random.default_rng(it * 11 + 5)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.004, 1.0)
            conv = fast_convolve(a)
            no_improve = 0
            lr = max(lr * 1.5, 0.005)
    
    return best_a, best_obj


def coordinate_descent_fast(a0, time_limit, verbose=False):
    """Coordinate descent with local optimization per coordinate."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        cur_peak = fast_convolve(a).max()
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_v = a[j]
            if old_v < 1e-14:
                continue
            
            best_j_val = old_v
            best_j_obj = cur_peak
            
            # Try geometric multipliers around current value
            multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 0.3, 3.0]
            # Also try the uniform value
            multipliers.append((1.0 / n) / old_v if old_v > 1e-15 else 1.0)
            
            for mult in multipliers:
                nv = old_v * mult
                if nv < 0 or nv > 0.99:
                    continue
                rest = 1.0 - nv
                old_rest = 1.0 - old_v
                if old_rest < 1e-12 or rest < 0:
                    continue
                scale = rest / old_rest
                a_trial = a.copy()
                a_trial[j] = nv
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = nv
            
            if best_j_val != old_v:
                rest = 1.0 - best_j_val
                old_rest = 1.0 - old_v
                if old_rest > 1e-12 and rest > 0:
                    scale = rest / old_rest
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    cur_peak = fast_convolve(a).max()
                    if cur_peak < best_obj - 1e-15:
                        best_obj = cur_peak
                        best_a = a.copy()
                        improved = True
        
        if not improved and passes > 3:
            break
        if verbose and passes % 5 == 0:
            print(f"      CD pass {passes}: obj={best_obj:.10f}")
    
    return best_a, best_obj


def local_pair_move(a0, time_limit, n_moves=500, verbose=False):
    """Local search: try moving mass between pairs of coordinates."""
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False
        
        for _ in range(n_moves):
            if time.monotonic() - t0 > time_limit:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            
            # Try moving mass from i to j
            frac = rng.uniform(0.05, 0.5)
            amount = a[i] * frac
            if amount < 1e-12:
                continue
            
            a_trial = a.copy()
            a_trial[i] -= amount
            a_trial[j] += amount
            
            val = fast_convolve(a_trial).max()
            if val < best_obj - 1e-15:
                a = a_trial
                conv = fast_convolve(a)
                best_obj = conv.max()
                best_a = a.copy()
                improved = True
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 30:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def diverse_initializations(n, seed=42):
    """Generate diverse initial sequences."""
    rng = np.random.default_rng(seed)
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Gaussian centered
    g = np.exp(-0.5 * ((x - mid) / (n / 6.0)) ** 2)
    inits.append(g / g.sum())
    
    # Inverted Gaussian (edges heavy)
    ig = np.exp(-0.5 * ((x - mid) / (n / 4.0)) ** 2)
    ig = np.maximum(1.0 - ig, 0.01)
    inits.append(ig / ig.sum())
    
    # Raised cosine
    rc = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / n))
    inits.append(rc / rc.sum())
    
    # Triangular
    tri = np.minimum(x, n - 1 - x).astype(np.float64)
    tri = np.maximum(tri, 0.1)
    inits.append(tri / tri.sum())
    
    # Linear ramp
    lr = np.linspace(0.1, 1.0, n)
    inits.append(lr / lr.sum())
    
    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    
    # Random Dirichlet (spread)
    inits.append(rng.dirichlet(np.ones(n) * 3.0))
    
    # Two bumps
    bump = np.exp(-0.5 * ((x - n * 0.25) / (n / 10.0)) ** 2) + \
           np.exp(-0.5 * ((x - n * 0.75) / (n / 10.0)) ** 2)
    bump = np.maximum(bump, 0.01)
    inits.append(bump / bump.sum())
    
    # Flat center
    flat = np.ones(n)
    flat[n // 4: 3 * n // 4] = 1.5
    inits.append(flat / flat.sum())
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.7
    inits.append(alt / alt.sum())
    
    return inits


def lp_fw_step(f, M, n):
    """Single LP-FW step: solve LP for direction, line search."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception:
        return None
    
    if not res.success:
        return None
    
    g = res.x
    gs = g.sum()
    if gs < 1e-15:
        return None
    g = g / gs
    
    # Line search
    best_alpha = 0.5
    best_val = fast_convolve(f).max()
    for alpha in [0.1, 0.3, 0.5, 0.7, 0.9]:
        ft = f + alpha * (g - f)
        ft = np.maximum(ft, 0.0)
        ft = ft / ft.sum()
        val = fast_convolve(ft).max()
        if val < best_val:
            best_val = val
            best_alpha = alpha
    
    ft = f + best_alpha * (g - f)
    ft = np.maximum(ft, 0.0)
    ft = ft / ft.sum()
    return ft


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with restarts."""
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64)
    f = f / f.sum()
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        ft = lp_fw_step(f, M * 1.01, n)
        if ft is None:
            ft = lp_fw_step(f, M * 1.1, n)
            if ft is None:
                no_improve += 1
                continue
        
        conv = fast_convolve(ft)
        M_new = conv.max()
        
        if M_new < best_obj - 1e-14:
            best_obj = M_new
            best_f = ft.copy()
            f = ft
            M = M_new
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 80:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004, 1.0)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj

        if rem > 15:
            a_opt, _ = coordinate_descent_fast(best_seq, time_limit=min(rem, 40))
            check_and_update(a_opt, "final_cd")
        
        # Pair moves
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = local_pair_move(best_seq, time_limit=min(rem, 25), n_moves=500)
            check_and_update(a_opt, "final_pairs")
    
    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))
    
    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs
    
    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
