           sum(d) = 0, d >= -a

"""
Multi-phase optimizer using active-set LP Frank-Wolfe with diverse initializations,
multi-length exploration, adaptive active-set refinement, coordinate descent,
and subgradient polishing. Key innovation: active-set LP dramatically reduces
constraint count per iteration, enabling 5-10x more LP-FW steps.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_active_set(c, M, n, frac=0.005):
    """Find indices where convolution is within frac of the peak."""
    threshold = M - frac * M
    active = np.where(c >= threshold)[0]
    # Also always include the argmax
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    return active


def solve_active_set_lp(a, active_indices, c_vals, n):
    """
    Solve active-set LP:
    min t s.t. c[k] + 2*(a*d)[k] <= t for k in active_indices,
              sum(d) = 0, d >= -a
    Returns (d, t) or (None, None).
    """
    m = len(active_indices)
    # Objective: minimize t (variable index n)
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Inequality constraints: 2*(a*d)[k] - t <= -c[k]
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)

    for idx, k in enumerate(active_indices):
        # (a*d)[k] = sum_j a[j] * d[k-j] where 0 <= k-j < n
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        for j in range(lo, hi + 1):
            A_ub[idx, j] = 2.0 * a[k - j]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -c_vals[k]

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    # Bounds
    bounds = [(-a[j] - 1e-15, None) for j in range(n)]
    bounds.append((0.0, None))

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x[:n], res.x[n]
    return None, None


def solve_full_lp(a, M, n):
    """Solve full LP: max sum(g) s.t. (a*g)[k] <= M for all k, g >= 0."""
    c_obj = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = a
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None

def _perturb_restart(f, best_f, n):

    rng = np.random.default_rng()

def coordinate_descent(f0, time_limit):
    """Coordinate descent with local search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = f[j]
            if old_val < 1e-12:
                continue

            cur_peak = fast_convolve(f).max()
            best_j_val = old_val
            best_j_obj = cur_peak

            for mult in [0.5, 0.7, 0.9, 1.1, 1.3, 1.6]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale_f = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale_f
                f_trial = f_trial / f_trial.sum()

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale_f = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale_f
                    f = f / f.sum()
                    if best_j_obj < best_obj - 1e-14:
                        best_obj = best_j_obj
                        best_f = f.copy()
                        improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.002, mom=0.94, top_k=40, temp=5e-5):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.002

        if no_improve > 400:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0004)
            no_improve = 0

    return best_f, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    target_lengths = [nl for nl in target_lengths if abs(nl - n_parent) > 32]

    per_len = 70
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 850
        if rem < 30:
            break
        t_alloc = min(per_len, rem)
        print(f"  Exploring n={target_n}, t={t_alloc:.0f}s")

        # Try multiple starts
        starts = make_diverse_starts(target_n, base)
        phase_best = None
        phase_best_obj = float('inf')

        for si, st in enumerate(starts[:3]):  # Limit to 3 starts per length
            t_per = t_alloc / 3.0
            if t_per < 10:
                break
            a_opt, obj = lpfw_active_set(st, time_limit=t_per, verbose=False)
            if obj < phase_best_obj:
                phase_best_obj = obj
                phase_best = a_opt

        if phase_best is not None:
            try_seq(phase_best, f"ml_n{target_n}")

    # === Phase 3: Deep optimization (900s) ===
    print(f"\n=== Phase 3: Deep optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple deep passes with different init perturbations
        n_passes = 5
        for p in range(n_passes):
            rem = deadline - time.monotonic() - 250
            if rem < 30:
                break
            t_alloc = min(rem / (n_passes - p), 200)

            if p == 0:
                f_init = best_seq.copy()
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.005
                f_init = project_simplex(best_seq + noise)
            elif p == 2:
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init)
            elif p == 3:
                # Shifted version
                shift = n_best // 16
                f_init = np.roll(best_seq, shift)
                f_init = np.maximum(f_init, 0)
                f_init = f_init / f_init.sum()
            else:
                f_init = 0.9 * best_seq + 0.1 * (np.ones(n_best) / n_best)
                f_init = project_simplex(f_init)

            print(f"  Deep pass {p+1}/{n_passes}, n={n_best}, t={t_alloc:.0f}s")
            a_opt, obj = lpfw_active_set(f_init, time_limit=t_alloc, verbose=(p == 0))
            try_seq(a_opt, f"deep_{p}")

    # === Phase 4: Coordinate descent + SG (250s) ===
    print(f"\n=== Phase 4: Polishing (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 30:
            t_cd = min(rem * 0.5, 120)
            print(f"  CD t={t_cd:.0f}s")
            a_opt, _ = coordinate_descent(best_seq, time_limit=t_cd)
            try_seq(a_opt, "cd")

        rem = deadline - time.monotonic() - 60
        if rem > 20:
            t_sg = min(rem, 100)
            print(f"  SG t={t_sg:.0f}s")
            n_best = len(best_seq)
            a_opt, _ = sg_polish(best_seq, time_limit=t_sg,
                                 lr=0.0015, mom=0.95,
                                 top_k=max(20, n_best // 8),
                                 temp=3e-5)
            try_seq(a_opt, "sg")

    # === Phase 5: Final micro-optimization (100s) ===
    print(f"\n=== Phase 5: Final micro-opt (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 20
        if rem > 20:
            # Short active-set LP-FW
            t_final = min(rem, 60)
            a_opt, _ = lpfw_active_set(best_seq, time_limit=t_final, verbose=False)
            try_seq(a_opt, "final_lpfw")

        rem = deadline - time.monotonic() - 15
        if rem > 15:
            # Tiny perturbations + SG
            n_best = len(best_seq)
            t_pert = min(rem, 50)
            a_init = project_simplex(best_seq + rng.standard_normal(n_best) * 0.002)
            a_opt, _ = sg_polish(a_init, time_limit=t_pert,
                                 lr=0.0008, mom=0.96,
                                 top_k=max(10, n_best // 4),
                                 temp=1e-5)
            try_seq(a_opt, "perturb_sg")

        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = lpfw_active_set(best_seq, time_limit=min(rem, 20), verbose=False)
            try_seq(a_opt, "micro_lpfw")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = min(best_score, fs)

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
