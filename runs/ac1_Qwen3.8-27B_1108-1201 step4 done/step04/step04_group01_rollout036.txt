import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_grad(weights, f):
    """Compute gradient of LSE: 2 * cross_correlation(weights, f).
    weights has length 2n-1, f has length n.
    grad[i] = 2 * sum_{j} weights[i+j] * f[j]
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase adaptive LSE gradient
descent with peak-targeted refinement and local search heuristics.
Key improvements: Nesterov momentum, adaptive beta with annealing,
peak-focused subgradient steps, coordinate swap local search, and
reduced reliance on slow LP solves.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(weights, f):
    """Compute gradient of LSE(conv(a)): grad[i] = 2 * sum_j weights[i+j] * f[j]."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def compute_subgrad_peak(k_star, f):
    """Subgradient of conv[k_star] w.r.t. f: 2*f[k_star - j] for valid j."""
    n = len(f)
    grad = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(n - 1, k_star)
    idx = np.arange(js, je + 1)
    grad[idx] = 2.0 * f[k_star - idx]
    return grad


def lse_gradient_descent(f0, time_limit, beta_start=10.0, beta_end=3000.0,
                         lr_init=None, momentum_init=0.90, verbose=True):
    """LSE-smoothed gradient descent with momentum and adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 4.0 / n

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    momentum = momentum_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    anneal_counter = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Softmax weights (numerically stable)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        # Gradient via FFT
        grad = compute_lse_gradient(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 200
            break
        grad = grad / grad_norm

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
