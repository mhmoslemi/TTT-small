    # = cross_correlation(p, a)[i]
    # Using the relation: cross_correlation(p,a)[i] = conv(p, a_rev)[i + n - 1]
    # conv(p, a_rev)[m] = sum_k p[k] * a_rev[m-k] = sum_k p[k] * a[n-1-m+k]
    # We want: sum_k p[k] * a[k-i]
    # Set n-1-m+k = k-i => n-1-m = -i => m = n-1+i
    # So: cross_correlation(p,a)[i] = conv(p, a_rev)[n-1+i]
    
    grad = np.zeros(n)
    for i in range(n):
        idx = n - 1 + i
        if idx < full_size:
            grad[i] = 2.0 * conv_pa[idx]
    
    return grad


def compute_gradient_direct(r, beta, a):
    """Direct gradient computation (slower but simpler, for verification)."""
    n = len(a)
    conv_len = len(r)
    scaled = beta * r
    scaled -= scaled.max()
    p = np.exp(scaled)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros(conv_len)
        p[np.argmax(r)] = 1.0
    else:
        p /= p_sum

    grad = np.zeros(n)
    for i in range(n):
        # grad[i] = 2 * sum_k p[k] * a[k-i] for valid k-i
        js = max(0, i)
        je = min(n, conv_len - i + 1)
        if je > js:
            # k ranges from i to i+n-1, a[k-i] ranges from a[0] to a[n-1]
            # p[i:i+n] * a[0:n]
            k_start = i
            k_end = min(conv_len, i + n)
            if k_end > k_start:
                grad[i] = 2.0 * np.dot(p[k_start:k_end], a[:k_end - k_start])
    return grad


def smoothed_minmax_optimize(a0, time_limit, beta_start=10.0, beta_end=5000.0,
                             lr_start=0.01, mom=0.9, max_iter=None):
    """
    Smoothed min-max optimization using softmax-approximated max.
    Minimizes (1/beta)*log(sum_k exp(beta * r[k])) which approximates max_k r[k].
    Uses adaptive beta (increasing over time) and Nesterov momentum.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    r = fast_convolve(a)
    best_obj = r.max()
    best_a = a.copy()

    t0 = time.monotonic()
    v = np.zeros(n)  # momentum
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)

    total_iters_target = 50000 if max_iter is None else max_iter

    while time.monotonic() - t0 < time_limit:
        it += 1
        # Adaptive beta
        progress = min(1.0, it / total_iters_target)
        # Exponential schedule
        beta = beta_start * (beta_end / beta_start) ** progress
        # Learning rate decay
        lr = lr_start * (0.5 + 0.5 * (1 - progress))
        # Momentum increases slightly
        cur_mom = mom + 0.05 * progress

        # Compute gradient
        grad = compute_gradient_smoothed(r, beta, a)
        
        # Nesterov momentum update
        v = cur_mom * v - lr * grad
        a_new = project_simplex(a + v)

        # Evaluate
        r_new = fast_convolve(a_new)
        obj_new = r_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        r = r_new

        # Periodic restart if stuck
        if no_improve > 200:
            noise_scale = 0.002 + 0.003 * (no_improve // 200)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            v = np.zeros(n)
            r = fast_convolve(a)
            no_improve = 0

        # Check time
        if time.monotonic() - t0 > time_limit - 5:
            break

    return best_a, best_obj


def coordinate_descent(a0, time_limit, n_sweeps_max=5000, verbose=False):
    """
    Coordinate descent for min max_k conv(a,a)[k] subject to a>=0, sum(a)=1.
    For each coordinate, tries multiple candidates and picks the best.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    r = fast_convolve(a)
    best_obj = r.max()
    best_a = a.copy()

    t0 = time.monotonic()
    sweep = 0
    no_improve = 0
    rng = np.random.default_rng(42)

    while time.monotonic() - t0 < time_limit and sweep < n_sweeps_max:
        sweep += 1
        improved = False

        # Randomize coordinate order for diversification
        order = rng.permutation(n) if sweep % 10 == 0 else np.arange(n)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            if old_val < 1e-15:
                # Try setting to small value
                candidates = [0.0, 0.001, 0.005, 0.01]
            else:
                # Try relative changes
                candidates = [0.0, old_val * 0.3, old_val * 0.5, old_val,
                              old_val * 1.5, old_val * 2.0,
                              0.5 / n, 2.0 / n]
                # Also try value that would equalize with current peak
                # Peak k* has value M; if a[i] contributes 2*val*a[k*-i] to r[k*],
                # set val so that 2*val*a[k*-i] = M/2 (rough heuristic)
                k_star = int(np.argmax(r))
                k_minus_i = k_star - i
                if 0 <= k_minus_i < n and a[k_minus_i] > 1e-10:
                    target_val = r.max() / (4.0 * a[k_minus_i])
                    candidates.append(max(0.0, min(target_val, 0.5)))

            best_cand_val = old_val
            best_cand_obj = best_obj

            for x in candidates:
                if x < -1e-10 or x > 0.1:
                    continue
                # Create new sequence
                a_trial = a.copy()
                a_trial[i] = x
                # Rescale to sum=1
                s_trial = a_trial.sum()
                if s_trial < 1e-10:
                    continue
                a_trial = a_trial / s_trial
                # Evaluate
                r_trial = fast_convolve(a_trial)
                obj_trial = r_trial.max()
                if obj_trial < best_cand_obj - 1e-14:
                    best_cand_obj = obj_trial
                    best_cand_val = x

            if best_cand_val != old_val:
                a[i] = best_cand_val
                s_new = a.sum()
                if s_new > 1e-10:
                    a = a / s_new
                    r = fast_convolve(a)
                    if r.max() < best_obj - 1e-14:
                        best_obj = r.max()
                        best_a = a.copy()
                        improved = True

        if improved:
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 50:
            # Restart with perturbation
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            r = fast_convolve(a)
            no_improve = 0
            if time.monotonic() - t0 > time_limit - 10:
                break

        if verbose and sweep % 100 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}")

    return best_a, best_obj


def spectral_shaping(a0, time_limit, n_iters=1000):
    """
    Frequency-domain spectral shaping: iteratively flatten the power spectrum
    while maintaining non-negativity and length-n support.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    r = fast_convolve(a)
    best_obj = r.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    alpha = 0.1

    while time.monotonic() - t0 < time_limit and it < n_iters:
        it += 1

        # A~~~~: Compute power spectrum
        fft_size = 1 << (2 * n - 1 + 1).bit_length()
        A = np.fft.rfft(a, fft_size)
        power = np.abs(A) ** 2
        phase = np.angle(A)

        # Compute autocorrelation
        r = np.fft.irfft(power, fft_size)[:2 * n - 1]

        # Find peak
        peak_val = r.max()
        peak_loc = int(np.argmax(r))

        # Modify power spectrum to reduce peak
        # The peak at lag k corresponds to certain frequency components
        # We reduce power at frequencies that contribute to the peak
        # Simple approach: reduce overall power at high-magnitude frequencies
        # that correlate with the peak location
        
        # Alternative: iteratively reduce the "effective bandwidth"
        # Multiply power by a window that suppresses concentration
        power_flat = power.copy()
        # Reduce the maximum power bins
        p_max = power_flat.max()
        if p_max > 0:
            # Soft thresholding
            threshold = 0.7 * p_max
            excess = np.maximum(power_flat - threshold, 0)
            power_flat = power_flat - alpha * excess

        # Ensure non-negative power
        power_flat = np.maximum(power_flat, 0)
        amp = np.sqrt(power_flat)

        # Reconstruct time-domain signal
        A_new = amp * np.exp(1j * phase)
        a_new_full = np.fft.irfft(A_new, fft_size)
        a_new = np.maximum(a_new_full[:n], 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new = a_new / s_new

        # Evaluate
        r_new = fast_convolve(a_new)
        obj_new = r_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()

        a = a_new
        # Decay alpha
        alpha = max(0.01, alpha * 0.995)

    return best_a, best_obj


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, top_k=40):
    """Solve reduced linearized LP for Frank-Wolfe direction."""
    n = len(f)
    B = build_conv_matrix(f)

    # Use top_k peak constraints
    if top_k < len(conv_f):
        peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    else:
        peaks = np.arange(len(conv_f))

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((len(peaks), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv_f[peaks]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def line_search(f, target, current_obj, max_alpha=1.0):
    """Find best alpha for f_new = (1-alpha)*f + alpha*target."""
    best_alpha = 0.0
    best_obj = current_obj

    alphas = np.linspace(0.05, max_alpha, 10)
    for alpha in alphas:
        f_new = (1.0 - alpha) * f + alpha * target
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha

    if best_alpha > 0:
        lo = max(0.01, best_alpha * 0.7)
        hi = min(max_alpha, best_alpha * 1.3)
        for _ in range(6):
            mid = (lo + hi) / 2.0
            f_new = (1.0 - mid) * f + mid * target
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-15:
                break
            f_new = f_new / s
            obj = fast_convolve(f_new).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = mid
                lo = mid
            else:
                hi = mid

    return best_alpha, best_obj


def linearized_fw_optimize(f0, time_limit, reduced_k=30):
    """Frank-Wolfe with linearized LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(99)

    while time.monotonic() - t0 < time_limit:
        it += 1
        g = solve_linearized_lp(f, conv, top_k=reduced_k)

        if g is None:
            no_improve += 5
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                no_improve = 0
            continue

        target = np.maximum(f + g, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        alpha, obj_val = line_search(f, target, conv.max())

        if alpha > 1e-8:
            f = np.maximum((1.0 - alpha) * f + alpha * target, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            if conv.max() < best_obj - 1e-14:
                best_obj = conv.max()
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            f = project_simplex(best_f + rng.standard_normal(n) * (0.002 + 0.001 * (no_improve // 60)))
            no_improve = 0

    return best_f, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []

    # Uniform
    inits.append(np.ones(n) / n)

    # Exponential decay (left-heavy)
    x = np.arange(n, dtype=np.float64)
    for decay in [0.5, 1.0, 2.0, 3.0]:
        e = np.exp(-decay * x / n)
        e = np.maximum(e, 1e-10)
        inits.append(e / e.sum())

    # Exponential decay (right-heavy)
    e2 = np.exp(-decay * (n - 1 - x) / n)
    e2 = np.maximum(e2, 1e-10)
    inits.append(e2 / e2.sum())

    # Gaussian center
    sigma = 0.3 * n
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
    inits.append(g / g.sum())

    # Sine wave
    s1 = np.abs(np.sin(np.pi * x / n)) + 0.1
    inits.append(s1 / s1.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / (2 * n))) + 0.05
    inits.append(c1 / c1.sum())

    # Two peaks
    for frac in [0.25, 0.33]:
        p = np.exp(-((x - frac * (n - 1)) / (0.1 * n)) ** 2) + \
            np.exp(-((x - (1 - frac) * (n - 1)) / (0.1 * n)) ** 2)
        inits.append(p / p.sum())

    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-strategy search combining:
    - Diverse initializations
    - Smoothed min-max optimization (primary)
    - Coordinate descent (polish)
    - Spectral shaping (global exploration)
    - Linearized LP Frank-Wolfe (backup)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent: n={len(parent)}, score={parent_score:.8f}")
    except Exception:
        pass

    # === Phase 1: Diverse initializations and quick evaluation ===
    print("=== Phase 1: Initialization ===")
    t_start = time.monotonic()

    candidate_list = []

    if parent is not None:
        n_default = len(parent)
        candidate_list.append((parent, "parent"))
    else:
        n_default = 384
        candidate_list.append((np.ones(n_default) / n_default, "uniform"))

    # Try multiple lengths
    for n_try in [n_default, 256, 320, 448, 512, 640]:
        inits = generate_diverse_inits(n_try, parent, rng)
        for i, ini in enumerate(inits):
            candidate_list.append((ini, f"n{n_try}_i{i}"))

    # Quick evaluate all
    for cand, label in candidate_list:
        if time.monotonic() > deadline - 300:
            break
        try_seq(cand, label)

    if best_seq is None:
        best_seq = np.ones(384) / 384
        try_seq(best_seq, "fallback")

    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    n_best = len(best_seq)

    # === Phase 2: Smoothed min-max optimization (primary) ===
    print("\n=== Phase 2: Smoothed Min-Max Optimization ===")
    remaining = deadline - time.monotonic() - 200
    t_p2 = min(remaining * 0.40, 700)

    if t_p2 > 30:
        # Run on best with different beta ranges
        t_each = t_p2 / 3
        for run_id, (b_start, b_end, lr) in enumerate(
                [(5.0, 2000.0, 0.01), (20.0, 8000.0, 0.005), (50.0, 20000.0, 0.003)]):
            rem = deadline - time.monotonic() - 150
            if rem < 15:
                break
            t_alloc = min(t_each, rem * 0.6)
            a_opt, _ = smoothed_minmax_optimize(best_seq, time_limit=t_alloc,
                                                 beta_start=b_start, beta_end=b_end, lr_start=lr)
            try_seq(a_opt, f"smm_run{run_id}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Coordinate descent ===
    print("\n=== Phase 3: Coordinate Descent ===")
    remaining = deadline - time.monotonic() - 150
    t_p3 = min(remaining * 0.30, 400)

    if t_p3 > 20:
        a_opt, _ = coordinate_descent(best_seq, time_limit=t_p3 * 0.7)
        try_seq(a_opt, "cd_main")

        # CD after perturbation
        rem = deadline - time.monotonic() - 100
        if rem > 15:
            a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.005)
            a_opt, _ = coordinate_descent(a_pert, time_limit=min(rem * 0.5, 150))
            try_seq(a_opt, "cd_pert")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Spectral shaping ===
    print("\n=== Phase 4: Spectral Shaping ===")
    remaining = deadline - time.monotonic() - 100
    t_p4 = min(remaining * 0.20, 200)

    if t_p4 > 15:
        a_opt, _ = spectral_shaping(best_seq, time_limit=t_p4 * 0.6, n_iters=500)
        try_seq(a_opt, "spectral")

        # Spectral from perturbed
        rem = deadline - time.monotonic() - 80
        if rem > 10:
            a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)
            a_opt, _ = spectral_shaping(a_pert, time_limit=min(rem * 0.5, 80), n_iters=300)
            try_seq(a_opt, "spectral_pert")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Linearized LP Frank-Wolfe ===
    print("\n=== Phase 5: Linearized LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic() - 80
    t_p5 = min(remaining * 0.25, 250)

    if t_p5 > 20:
        a_opt, _ = linearized_fw_optimize(best_seq, time_limit=t_p5 * 0.7, reduced_k=35)
        try_seq(a_opt, "linfw")

        rem = deadline - time.monotonic() - 60
        if rem > 15:
            a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)
            a_opt, _ = linearized_fw_optimize(a_pert, time_limit=min(rem * 0.6, 100), reduced_k=25)
            try_seq(a_opt, "linfw_pert")

    print(f"  Phase 5 best: {best_score:.8f}")

    # === Phase 6: Alternating polish (smoothed + CD alternate
