def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def fast_cross_convolve(a, b):
    """Linear cross-convolution via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_pnorm_grad(weights, f):
    """Gradient of sum_k weights_k * conv(f,f)_k w.r.t. f.
    = 2 * cross_correlation(weights, f) of length n.

    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_rev, fft_size)
    conv = np.fft.irfft(W * F, fft_size)

                if st < 1e-15:

                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

            conv = fast_convolve(f)

def adam_pnorm_optimize(f0, time_limit, p_start=10.0, p_end=200.0,
                        lr_init=0.02, beta1=0.8, beta2=0.999, eps=1e-8):
    """Adam gradient descent on p-norm smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    p = p_start
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        t_step += 1

        conv = fast_convolve(f)
        c_max = conv.max()
        c_min = conv.min()

        # p-norm weights (numerically stable)
        # w_k = |c_k|^p / sum |c_i|^p  (use shifted for stability)
        c_shifted = conv - c_min + 1e-12
        log_c = np.log(c_shifted) * p
        log_c -= log_c.max()
        exp_c = np.exp(log_c)
        w = exp_c / exp_c.sum()

        # Gradient
        grad = compute_pnorm_grad(w, f)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 100
            break

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.95
            if no_improve > 200:
                f = best_f.copy()
                m = np.zeros(n)
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.001)
                p = min(p * 1.5, p_end)
                no_improve = 0

            p *= 1.005

            print(f"    Adam it={t_step}, obj={best_obj:.12f}, p={p:.1f}, lr={lr:.6f}, t={elapsed:.0f}s")

def peak_chase_optimize(f0, time_limit, lr_init=0.005, momentum=0.93,
                        top_k=16, temp_init=0.00002):
    """Peak-chasing subgradient with momentum and top-k weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    v = np.zeros(n)
    cur_mom = momentum
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)

        # Top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        # Subgradient: 2 * sum_k w_k * f_{k-i}
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = cur_mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                cur_lr *= 0.96
            if no_improve > 400:
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.5, 0.0005)
                no_improve = 0

        cur_temp *= 1.003

    return best_f, best_obj

        m = 0.8 * m + 0.2 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1 - 0.8 ** t_step)
        v_hat = v / (1 - 0.999 ** t_step)

        f_new = best_f - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # Enforce symmetry
        f_new = 0.5 * (f_new + f_new[::-1])
        f_new = project_simplex(f_new)
        # Re-enforce symmetry after projection
        f_new = 0.5 * (f_new + f_new[::-1])
        f_new = np.maximum(f_new, 0)
        f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 20 == 0:
                lr *= 0.95
            if no_improve > 200:
                m = np.zeros(n)
                v = np.zeros(n)
                lr = max(lr_init * 0.5, 0.001)
                p = min(p * 1.5, p_end)
                no_improve = 0

        if p < p_end:
            p *= 1.005

    return best_f, best_obj


def fourier_init(n, rng):
    """Generate a Fourier-structured initial distribution."""
    x = np.arange(n) / n
    best_p = None
    best_val = np.inf

    for trial in range(50):
        a1 = rng.uniform(-0.5, 0.5)
        a2 = rng.uniform(-0.3, 0.3)
        a3 = rng.uniform(-0.1, 0.1)
        a4 = rng.uniform(-0.05, 0.05)
        phase1 = rng.uniform(0, 2 * np.pi)
        phase2 = rng.uniform(0, 2 * np.pi)

        p = 1.0 + a1 * np.cos(2 * np.pi * x + phase1) \
                    + a2 * np.cos(4 * np.pi * x + phase2) \
                    + a3 * np.cos(6 * np.pi * x) \
                    + a4 * np.cos(8 * np.pi * x)
        p = np.maximum(p, 0.0)
        s = p.sum()
        if s < 1e-10:
            continue
        p = p / s
        val = fast_convolve(p).max()
        if val < best_val:
            best_val = val
            best_p = p.copy()

    if best_p is None:
        best_p = np.ones(n) / n
    return best_p


def try_sequence(a, best_tracker):
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization combining:
    - Fourier-structured initialization
    - Adam p-norm gradient descent
    - FW-LP for global moves
    - Symmetric-constrained optimization
    - Peak-chasing subgradient
    - Multi-length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_score = float('inf')
    best_seq = None
    best_tracker = [float('inf'), None]

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.8f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is None:
        parent = fourier_init(448, rng)
        try_sequence(parent, best_tracker)

    n_parent = len(parent)

    # === Phase 1: Quick FW-LP warm start ===
    print("\n=== Phase 1: FW-LP warm start ===")
    t1 = min(120, deadline - time.monotonic() - 1200)
    if t1 > 30:
        print(f"  Allocating {t1:.0f}s")
        a_opt, _ = fw_lp_optimize(parent, time_limit=t1)
        try_sequence(a_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 2: Adam p-norm (main optimization) ===
    print(f"\n=== Phase 2: Adam p-norm (n={len(current)}) ===")
    t2 = min(500, deadline - time.monotonic() - 500)
    if t2 > 60:
        print(f"  Allocating {t2:.0f}s")
        a_opt, _ = adam_pnorm_optimize(current, time_limit=t2,
                                       p_start=8.0, p_end=150.0, lr_init=0.025)
        try_sequence(a_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 3: FW-LP refinement ===
    print(f"\n=== Phase 3: FW-LP refinement ===")
    t3 = min(200, deadline - time.monotonic() - 350)
    if t3 > 40:
        print(f"  Allocating {t3:.0f}s")
        a_opt, _ = fw_lp_optimize(current, time_limit=t3)
        try_sequence(a_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 4: Symmetric Adam ===
    print(f"\n=== Phase 4: Symmetric Adam ===")
    t4 = min(150, deadline - time.monotonic() - 250)
    if t4 > 40:
        print(f"  Allocating {t4:.0f}s")
        a_opt, _ = symmetric_adam_optimize(current, time_limit=t4,
                                           p_start=15.0, p_end=250.0, lr_init=0.02)
        try_sequence(a_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 5: Peak-chasing polish ===
    print(f"\n=== Phase 5: Peak-chasing polish ===")
    t5 = min(100, deadline - time.monotonic() - 150)
    if t5 > 30:
        print(f"  Allocating {t5:.0f}s")
        n_best = len(current)
        a_opt, _ = peak_chase_optimize(current, time_limit=t5,
                                       lr_init=0.003, momentum=0.94,
                                       top_k=max(8, n_best // 10),
                                       temp_init=0.00001)
        try_sequence(a_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 6: Second Adam pass with higher p ===
    print(f"\n=== Phase 6: High-p Adam polish ===")
    t6 = min(150, deadline - time.monotonic() - 100)
    if t6 > 30:
        print(f"  Allocating {t6:.0f}s")
        a_opt, _ = adam_pnorm_optimize(current, time_limit=t6,
                                       p_start=50.0, p_end=500.0, lr_init=0.01)
        try_sequence(a_opt, best_tracker)

    current = best_tracker[1] if best_tracker[1] is not None else parent

    # === Phase 7: Alternative lengths ===
    print(f"\n=== Phase 7: Alternative lengths ===")
    n_best = len(current) if current is not None else n_parent

    alt_lengths = [320, 384, 480, 512, 576]
    if n_best in alt_lengths:
        alt_lengths.remove(n_best)

    for n_alt in alt_lengths[:3]:
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 60:
            break
        t_alloc = min(100, t_remaining / 3.0)
        if t_alloc < 30:

        # Interpolate
        ref = best_tracker[1] if best_tracker[1] is not None else parent
        x_old = np.linspace(0, 1, len(ref))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, ref)
        f_interp = np.maximum(f_interp, 0)
        if f_interp.sum() > 1e-10:
            f_interp = f_interp / f_interp.sum()

        # Quick Adam
        t_adam = t_alloc * 0.7
        t_fw = t_alloc * 0.3
        a_opt, _ = adam_pnorm_optimize(f_interp, time_limit=t_adam,
                                       p_start=10.0, p_end=100.0, lr_init=0.02)
        try_sequence(a_opt, best_tracker)

        current_alt = best_tracker[1] if best_tracker[1] is not None else f_interp
        if len(current_alt) == n_alt and t_fw > 20:
            a_opt, _ = fw_lp_optimize(current_alt, time_limit=t_fw)
            try_sequence(a_opt, best_tracker)

    # === Final: One more FW-LP on the best ===
    if best_tracker[1] is not None:
        t_final = deadline - time.monotonic() - 10
        if t_final > 40:
            print(f"\n=== Final FW-LP: {t_final:.0f}s ===")
            a_opt, _ = fw_lp_optimize(best_tracker[1], time_limit=t_final)
            try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(n_parent) / n_parent
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
