def fast_convolve(a):

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    it = 0
    no_improve = 0

            break
        
        it += 1
        conv = fast_convolve(f)
        obj = conv.max()

            best_f = f.copy()
            no_improve = 0

        # (Q_k a)[j] = sum_m Q_k[j,m] a[m] = a[k-j] if 0 <= k-j < n
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        idx = np.arange(js, je + 1)
        sg[idx] = 2.0 * f[k_star - idx]

        gap = obj - f_lb

        if sg_norm_sq < 1e-30:

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FW-LP with reduced active constraints and exact line search
2. Coordinate descent polishing with renormalization
3. Multi-restart with perturbation

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_value(a):
    """Compute the objective value 2*n*max(conv(a,a))/(sum(a))^2."""
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    a_norm = a / s
    conv = fast_convolve(a_norm)
    return 2.0 * len(a_norm) * conv.max()


def build_active_lp(f, active_indices, M):
    """Build sparse LP matrix for active constraints only."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in active_indices:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        rows.extend([k] * len(js))
        cols.extend(js.tolist())
        vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))


def solve_reduced_lp(f, active_indices, M):
    """Solve the reduced LP for FW direction."""
    n = len(f)
    A = build_active_lp(f, active_indices, M)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. FW-LP with reduced active constraints and exact line search for large jumps
2. Fast cyclic coordinate descent with O(n) incremental convolution updates
3. Multi-restart with perturbation to escape flat regions
4. Alternative length exploration
The CD phase is the main workhorse, exploiting the fact that changing one
coordinate only affects O(n) convolution entries linearly/quadratically.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_active_lp_matrix(f, active_indices):
    """Build sparse LP matrix for given active constraint indices."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in active_indices:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1)
        rows.extend([int(k)] * len(js))
        cols.extend(js.tolist())
        vals.extend((f[k - js]).tolist())
    return csr_matrix((vals, (rows, cols)), shape=(len(active_indices), n))


def solve_lp_direction(f, active_indices, M, timeout=5):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    A = build_active_lp_matrix(f, active_indices)
    c = -np.ones(n)
    b_ub = np.full(len(active_indices), M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': timeout})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def fw_reduced_lp(f0, time_limit, eps_init=0.1, eps_min=0.02):
    """Frank-Wolfe with reduced LP oracle and active-set line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
        s = 1.0
    f = f / s
    n = len(f)

    conv_f = fast_convolve(f)
    M = conv_f.max()
    best_f = f.copy()
    best_obj = 2.0 * n * M

    t0 = time.monotonic()
    eps = eps_init
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break
        it += 1

        # Select active constraints
        threshold = M - eps * M
        active_mask = conv_f >= threshold
        active_indices = np.where(active_mask)[0]
        if len(active_indices) < 3:
            top_k = min(10, 2 * n - 1)
            active_indices = np.argpartition(conv_f, -top_k)[-top_k:]

        # Solve LP
        g = solve_lp_direction(f, active_indices, M, timeout=3)
        if g is None or g.sum() < 1e-15:
            # Fallback: try all constraints
            all_idx = np.arange(2 * n - 1)
            g = solve_lp_direction(f, all_idx, M, timeout=5)
            if g is None or g.sum() < 1e-15:
                no_improve += 1
                if no_improve > 15:
                    break
                eps = max(eps * 0.9, eps_min)
                continue

        g = np.maximum(g, 0.0)
        g_sum = g.sum()
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 25:
                break
            eps = max(eps * 0.95, eps_min)
            continue

        # Exact line search
        conv_g = fast_convolve(g)
        fft_size = 1 << (2 * n - 2).bit_length()
        F = np.fft.rfft(f, fft_size)
        G = np.fft.rfft(g, fft_size)
        conv_fg = np.fft.irfft(F * G, fft_size)[:2 * n - 1]

        C = conv_f
        P = conv_fg
        Q = conv_g

        t = 0.5
        for _ in range(40):
            vals = (1.0 - t) ** 2 * C + 2.0 * t * (1.0 - t) * P + t ** 2 * Q
            k_star = int(np.argmax(vals))
            a_k = Q[k_star] - 2.0 * P[k_star] + C[k_star]
            b_k = 2.0 * P[k_star] - 2.0 * C[k_star]
            if abs(a_k) < 1e-15:
                t_new = 0.0 if b_k >= 0 else 1.0
            else:
                t_new = -b_k / (2.0 * a_k)
                t_new = max(0.0, min(1.0, t_new))
            if abs(t_new - t) < 1e-13:
                t = t_new
                break
            t = t_new

        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            no_improve += 1
            continue
        f_new = f_new / s_new

        conv_new = fast_convolve(f_new)
        M_new = conv_new.max()
        obj_new = 2.0 * n * M_new

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            eps = max(eps * 0.97, eps_min)
        else:
            no_improve += 1

        f = f_new
        conv_f = conv_new
        M = M_new

        if no_improve > 40:
            rng = np.random.default_rng(it * 13 + 5)
            scale = 0.002 + 0.003 * min(1.0, (no_improve - 40) / 40.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv_f = fast_convolve(f)
            M = conv_f.max()
            eps = eps_init
            no_improve = 0

    return best_f, best_obj


def cd_polish(f0, time_limit, verbose=True):
    """
    Fast cyclic coordinate descent with O(n) incremental convolution updates.
    Works in unnormalized space; objective = 2n * max(conv) / sum^2.
    """
    a = np.maximum(f0.astype(np.float64), 0.0)
    a_sum = a.sum()
    if a_sum < 1e-15:
        a = np.ones(len(f0))
        a_sum = len(f0)
    n = len(a)

    conv = fast_convolve(a).copy()
    obj = 2.0 * n * conv.max() / (a_sum * a_sum)
    best_obj = obj
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve_sweeps = 0
    sweep = 0
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    rng = np.random.default_rng(777)

    # Precompute affected ranges for each coordinate
    k_ranges = []
    for i in range(n):
        k_min = max(0, i)
        k_max = min(2 * n - 2, i + n - 1)
        k_ranges.append((k_min, k_max))

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        any_improved = False

        # Random coordinate order
        order = np.arange(n)
        rng.shuffle(order)

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old = a[i]
            cur_sum = a_sum
            k_min, k_max = k_ranges[i]
            affected_len = k_max - k_min + 1 if k_max >= k_min else 0
            two_i = 2 * i

            # Precompute the "slope" vector for this coordinate
            # conv_as_delta[k] = conv[k] + 2*delta*a[k-i] + delta^2*(k==2i)
            # For k in [k_min, k_max]: a[k-i] where j = k-i is in [0, n-1]
            # a[k-i] for k in [k_min, k_max] = a[0 : affected_len]
            slope_vec = a[0:affected_len].copy() if affected_len > 0 else None

            def eval_delta(delta):
                """Evaluate objective for a[i] changed by delta (unnormalized)."""
                new_sum = cur_sum + delta
                if new_sum < 1e-12:
                    return float('inf')
                # Compute max of updated conv
                max_val = conv.max()
                if affected_len > 0:
                    updated = conv[k_min:k_max + 1] + 2.0 * delta * slope_vec
                    if two_i >= k_min and two_i <= k_max:
                        updated = updated.copy()
                        updated[two_i - k_min] += delta * delta
                    local_max = updated.max()
                    if local_max > max_val:
                        max_val = local_max
                return 2.0 * n * max_val / (new_sum * new_sum)

            # Golden section search
            lo = -old  # a[i] = 0
            hi = max(old, cur_sum * 0.01)
            hi = min(hi, cur_sum * 0.5)
            if hi <= lo:
                hi = lo + 0.01 * cur_sum

            c = hi - gr * (hi - lo)
            d = lo + gr * (hi - lo)
            fc = eval_delta(c)
            fd = eval_delta(d)

            for _ in range(35):
                if fc < fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - gr * (hi - lo)
                    fc = eval_delta(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + gr * (hi - lo)
                    fd = eval_delta(d)

            delta_opt = (lo + hi) / 2.0
            obj_opt = eval_delta(delta_opt)

            if obj_opt < best_obj - 1e-14:
                # Accept: update conv incrementally
                delta = delta_opt
                if affected_len > 0:
                    conv[k_min:k_max + 1] += 2.0 * delta * slope_vec
                if two_i < 2 * n - 1:
                    conv[two_i] += delta * delta
                a[i] += delta
                a_sum = cur_sum + delta
                best_obj = obj_opt
                best_a = a.copy()
                any_improved = True

        if not any_improved:
            no_improve_sweeps += 1
            if no_improve_sweeps > 3:
                # Perturb to escape
                scale = 0.0005 + 0.002 * min(1.0, no_improve_sweeps / 5.0)
                perturbation = rng.standard_normal(n) * scale
                a = np.maximum(a + perturbation, 0.0)
                if a_sum <= 0:
                    a = best_a.copy()
                    a_sum = best_a.sum()
                else:
                    a_sum = a.sum()
                    conv = fast_convolve(a)
                best_obj = 2.0 * n * conv.max() / (a_sum * a_sum)
                best_a = a.copy()
                no_improve_sweeps = 0
        else:
            no_improve_sweeps = 0

        if verbose and sweep % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD sweep {sweep}, obj={best_obj:.10f}, elapsed={elapsed:.0f}s")

    return best_a, best_obj

        a_opt, _ = fw_reduced_lp(parent, time_limit=t_alloc, eps_init=0.15, eps_min=0.03)
        try_and_track(a_opt, best_tracker)

    # === Phase 2: CD Polishing (main) ===
    print(f"\n=== Phase 2: CD Polishing ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(700, deadline - time.monotonic() - 600)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, _ = cd_polish(current, time_limit=t_alloc)
        try_and_track(a_opt, best_tracker)

    # === Phase 3: FW-LP refinement ===
    print(f"\n=== Phase 3: FW-LP Refinement ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(150, deadline - time.monotonic() - 400)
    if t_alloc > 20:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, _ = fw_reduced_lp(current, time_limit=t_alloc, eps_init=0.08, eps_min=0.02)
        try_and_track(a_opt, best_tracker)

    # === Phase 4: Final CD ===
    print(f"\n=== Phase 4: Final CD Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(400, deadline - time.monotonic() - 100)
    if t_alloc > 30:
        print(f"  Allocating {t_alloc:.0f}s")
        a_opt, _ = cd_polish(current, time_limit=t_alloc)
        try_and_track(a_opt, best_tracker)

    # === Phase 5: Alternative lengths ===
    print(f"\n=== Phase 5: Alternative Lengths ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current)

    alt_lengths = [384, 512, 576, 640]
    for L in alt_lengths:
        if L == n_best:
            continue
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 60:
            break
        t_alloc = min(100, t_remaining - 30)
        if t_alloc < 30:
            break

        # Interpolate parent to new length
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, L)
        f_interp = np.interp(x_new, x_old, current)
        f_interp = np.maximum(f_interp, 0.0)
        if f_interp.sum() < 1e-10:
            continue
        f_interp = f_interp / f_interp.sum()
        try_and_track(f_interp, best_tracker)

        # Quick CD on interpolated
        a_opt, _ = cd_polish(f_interp, time_limit=min(t_alloc, 80))
        try_and_track(a_opt, best_tracker)

    # === Final: one more CD on absolute best ===
    print(f"\n=== Final Polish ===")
    current = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic() - 10
    if t_remaining > 20:
        a_opt, _ = cd_polish(current, time_limit=min(t_remaining, 60))
        try_and_track(a_opt, best_tracker)

    # Final verification
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(448) / 448.0
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
