def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    """Compute gradient of 2n * LSE(conv(a), beta) with respect to a."""
    # p = softmax(beta * conv(a)) -- precomputed

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smoothed gradient descent
with FFT-based gradient computation, Adam optimizer, beta annealing,

Key innovation over LP-based Frank-Wolfe: O(n log n) gradient steps via FFT

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(a, p, n):
    """
    Compute gradient of 2n * LSE(conv(a), beta) w.r.t. a.
    p = softmax(beta * conv(a)), precomputed.
    grad[j] = 2n * (sum_m a[m]*p[j+m] + p[2j]*a[j])
    Uses FFT for the cross-correlation term.
    """
    fft_size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    h = np.fft.irfft(A * np.conj(P), fft_size)[:n]
    term2 = p[2 * np.arange(n)] * a
    return 2.0 * n * (h + term2)


def softmax_stable(x, beta):
    """Compute softmax(beta * x) numerically stably."""
    scaled = beta * x
    scaled -= scaled.max()
    e = np.exp(scaled)
    s = e.sum()
    if s < 1e-300:
        return np.full(len(x), 1.0 / len(x))
    return e / s


def smoothed_gradient_descent(a_init, time_limit, beta_start=30.0, beta_end=20000.0,
                               lr_init=0.002, momentum=0.9, verbose=False):
    """
    Smoothed gradient descent with Adam, beta annealing, and simplex projection.
    Returns best (sequence, objective) found.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    step_count = 0
    beta_power = 0.999
    beta = beta_start

    no_improve = 0
    lr = lr_init
    eval_interval = max(1, n // 10)

    while time.monotonic() - t0 < time_limit:
        step_count += 1

        # Compute convolution and softmax weights
        conv = fast_convolve(a)
        p = softmax_stable(conv, beta)

        # Compute gradient
        grad = compute_smoothed_gradient(a, p, n)

        # Adam update
        step_count_f = float(step_count)
        bc1 = 1.0 - 0.9**step_count_f
        bc2 = 1.0 - 0.999**step_count_f
        m_state = 0.9 * m_state + 0.1 * grad
        v_state = 0.999 * v_state + 0.001 * grad**2
        m_hat = m_state / bc1
        v_hat = v_state / bc2
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # Project to simplex
        a = project_simplex(a_new)

        # Evaluate true objective periodically
        if step_count % eval_interval == 0:
            conv_check = fast_convolve(a)
            obj = conv_check.max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                no_improve = 0
                if verbose and step_count % (eval_interval * 10) == 0:
                    elapsed = time.monotonic() - t0
                    print(f"      SGD step {step_count}: obj={best_obj:.10f}, beta={beta:.1f}, t={elapsed:.1f}s")
            else:
                no_improve += 1

        # Anneal beta
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        beta = beta_start * (beta_end / beta_start) ** frac

        # Decay learning rate slowly
        lr = lr_init * 0.9999 ** step_count

        # If stuck, try a kick
        if no_improve > 50 * eval_interval:
            rng = np.random.default_rng(step_count % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            m_state = np.zeros(n)
            v_state = np.zeros(n)
            step_count = 0
            lr = lr_init * 0.8
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_polish(a_init, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate, try to find optimal value
    by testing candidate multipliers based on current gradient direction.
    """
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    # Precompute order: central coordinates first (they affect the peak most)
    center = (n - 1) / 2.0
    order = np.argsort(np.abs(np.arange(n) - center))

    passes = 0
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for idx in order:
            if time.monotonic() - t0 > time_limit:
                break
            j = int(idx)
            old_val = a[j]
            if old_val < 1e-10:
                # Try small positive values
                for new_val in [0.001, 0.005, 0.01]:
                    trial = a.copy()
                    trial[j] = new_val
                    denom = 1.0 - old_val
                    if denom > 1e-10:
                        mask = np.ones(n, dtype=bool)
                        mask[j] = False
                        trial[mask] *= (1.0 - new_val) / denom
                        trial = trial / trial.sum()
                        obj = fast_convolve(trial).max()
                        if obj < best_obj - 1e-13:
                            best_obj = obj
                            best_a = trial.copy()
                            a = trial.copy()
                            improved = True
                continue

            # Try multipliers centered around 1.0
            multipliers = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0]
            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                trial = a.copy()
                trial[j] = new_val
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                trial[mask] *= (1.0 - new_val) / denom
                trial_sum = trial.sum()
                if trial_sum < 1e-10:
                    continue
                trial = trial / trial_sum
                obj = fast_convolve(trial).max()
                if obj < best_j_obj - 1e-14:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[j] = best_j_val
                    a[mask] *= (1.0 - best_j_val) / denom
                    a = a / a.sum()
                if best_j_val > 0:
                    a[j] = best_j_val
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-13:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved and passes > 2:
            # Try random perturbations
            if passes > 5:
                break
            a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
            best_a = a.copy()
            best_obj = fast_convolve(a).max()

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using cubic-like interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def generate_initialization(n, kind, seed=0):
    """Generate various initialization patterns."""
    rng = np.random.default_rng(seed)
    x = np.arange(n)
    mid = (n - 1) / 2.0

    if kind == 'uniform':
        a = np.ones(n)
    elif kind == 'cosine_peak':
        # Peak at center: a[i] = 1 - cos(2*pi*(i-mid)/(n-1)) = 2*sin^2(...)
        if n > 1:
            a = 1.0 - np.cos(2 * np.pi * (x - mid) / (n - 1))
        else:
            a = np.ones(n)
    elif kind == 'cosine_edges':
        # Peak at edges: a[i] = 1 + cos(2*pi*(i-mid)/(n-1)) = 2*cos^2(...)
        if n > 1:
            a = 1.0 + np.cos(2 * np.pi * (x - mid) / (n - 1))
        else:
            a = np.ones(n)
    elif kind == 'gaussian':
        sigma = n / 4.0
        a = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
    elif kind == 'random_smooth':
        a = rng.random(n)
        # Smooth with a simple moving average
        kernel_size = max(3, n // 20)
        kernel = np.ones(kernel_size) / kernel_size
        a = np.convolve(a, kernel, mode='same')
    elif kind == 'two_humps':
        sigma = n / 8.0
        a = np.exp(-0.5 * ((x - n * 0.25) / sigma) ** 2) + np.exp(-0.5 * ((x - n * 0.75) / sigma) ** 2)
    elif kind == 'flat_center':
        # High at edges, flat low in center
        a = 1.0 + 0.5 * np.cos(4 * np.pi * x / (n - 1)) if n > 1 else np.ones(n)
        a = np.maximum(a, 0.01)
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a

        t_alloc = min(rem * 0.4, 100)
        a_opt, _ = smoothed_gradient_descent(f_interp, time_limit=t_alloc,
                                               beta_start=30.0, beta_end=5000.0,
                                               lr_init=0.002, verbose=False)
        rem2 = deadline - time.monotonic() - 150
        if rem2 > 10:
            a_opt, _ = coordinate_descent_polish(a_opt, time_limit=min(rem2, 40))
        try_seq(a_opt, f"len_{target_n}")

    # === Phase 4: Deep optimization on best ===
    print(f"\n=== Phase 4: Deep optimization (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)

        # Multiple refinement passes with different parameters
        for p in range(3):
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break

            if p == 0:
                f_init = best_seq.copy()
                beta_s, beta_e = 50.0, 10000.0
                lr = 0.001
            elif p == 1:
                noise = rng.standard_normal(n_best) * 0.002
                f_init = project_simplex(best_seq + noise)
                beta_s, beta_e = 100.0, 20000.0
                lr = 0.0005
            else:
                # Slight structural perturbation: mix with reversed
                f_init = 0.7 * best_seq + 0.3 * best_seq[::-1]
                f_init = project_simplex(f_init)
                beta_s, beta_e = 30.0, 15000.0
                lr = 0.0015

            a_opt, _ = smoothed_gradient_descent(f_init, time_limit=min(rem, 80),
                                                   beta_start=beta_s, beta_end=beta_e,
                                                   lr_init=lr, verbose=(p == 0))
            rem2 = deadline - time.monotonic() - 80
            if rem2 > 8:
                a_opt, _ = coordinate_descent_polish(a_opt, time_limit=min(rem2, 30))
            try_seq(a_opt, f"deep_{p}")

    # === Phase 5: Final intensive polish ===
    print(f"\n=== Phase 5: Final polish (best={best_score:.6f}) ===", flush=True)
    if best_seq is not None:
        n_best = len(best_seq)

        # Long coordinate descent
        rem = deadline - time.monotonic() - 30
        if rem > 10:
            a_opt, _ = coordinate_descent_polish(best_seq, time_limit=min(rem, 120), verbose=True)
            try_seq(a_opt, "final_cd")

        # Very fine gradient
        rem = deadline - time.monotonic() - 15
        if rem > 8:
            a_opt, _ = smoothed_gradient_descent(best_seq, time_limit=min(rem, 40),
                                                   beta_start=500.0, beta_end=100000.0,
                                                   lr_init=0.0001, verbose=False)
            try_seq(a_opt, "final_fine_grad")

        # Tiny perturbations + quick optimization
        for _ in range(3):
            rem = deadline - time.monotonic() - 10
            if rem < 8:
                break
            noise = rng.standard_normal(n_best) * 0.0005
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = coordinate_descent_polish(f_pert, time_limit=min(rem, 15))
            try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
