    n = len(a)

    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)

    """Smoothed max of conv(a,a) where a = exp(au) / sum(exp(au))"""
    # au is log-scale parameters
    amax = au.max()
    a = np.exp(au - amax)
    a = a / a.sum()
    conv = fast_convolve(a)
    return tau * logsumexp(conv / tau)

def soft_max_grad(au, tau):
    amax = au.max()
    a = np.exp(au - amax)
    a = a / a.sum()
    n = len(a)
    conv = fast_convolve(a)
    
    # Softmax weights over peaks
    cv = conv / tau
    w = np.exp(cv - cv.max())
    w /= w.sum()
    
    # Gradient of conv(a,a)[k] w.r.t. a[m] = 2*a[k-m] if valid

    """Project v onto {x >= 0, sum(x) = 1}"""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
1. L-BFGS on softmax-parameterized soft-max smoothed objective
2. Nesterov-accelerated mirror descent with adaptive temperature
3. Exact peak subgradient with momentum and restarts
4. Simulated annealing for local minimum escape
5. Coordinate descent final polish
Uses FFT-based gradient computation and adaptive time allocation.

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_cross_corr(w, a):
    """Compute cross-correlation R[m] = sum_j w[m+j] * a[j] for m=0..n-1."""
    n = len(a)
    size = 3 * n
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    return corr[:n]


def compute_softgrad(a, w):
    """Gradient of soft-max(conv(a,a)) w.r.t. a, given softmax weights w."""
    return 2.0 * compute_cross_corr(w, a)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_obj_and_grad(z, tau):
    """Objective and gradient for L-BFGS with softmax parameterization."""
    zs = z - z.max()
    a = np.exp(zs)
    a_sum = a.sum()
    a = a / a_sum
    
    conv = fast_convolve(a)
    cv = conv / tau
    cv -= cv.max()
    exp_cv = np.exp(cv)
    obj_val = tau * (np.log(exp_cv.sum()) + cv.max())
    
    w = exp_cv / exp_cv.sum()
    grad_a = compute_softgrad(a, w)
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)
    
    return obj_val, grad_z


def lbgfs_phase(a0, time_limit, tau_list, maxiter_per_tau=150):

    a = a0.copy()

            soft_max_obj_and_grad, z0, method='L-BFGS-B',

            options={'maxiter': maxiter_per_tau, 'ftol': 1e-14, 'gtol': 1e-10}

        z_sol = result.x

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using multi-phase optimization:
1. L-BFGS on softmax-parameterized soft-max smoothed objective (fast convergence)
2. Nesterov-accelerated multiplicative weight descent with adaptive temperature
3. Exact peak subgradient with momentum and periodic restarts
4. Simulated annealing for local minimum escape
5. Coordinate descent and micro-perturbation final polish
Uses FFT-based gradient computation, softmax parameterization for L-BFGS,
and adaptive time allocation across phases.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_cross_corr(w, a):
    """Compute R[m] = sum_j w[m+j] * a[j] for m=0..n-1 using FFT."""
    n = len(a)
    size = 3 * n
    fft_size = 1 << (size - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    A = np.fft.rfft(a, fft_size)
    corr = np.fft.irfft(W * np.conj(A), fft_size)
    return corr[:n]


def compute_softgrad(a, w):
    """Gradient of soft-max(conv(a,a)) w.r.t. a, given normalized peak weights w."""
    return 2.0 * compute_cross_corr(w, a)


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_max_obj_and_grad(z, tau):
    """
    Objective and gradient for L-BFGS with softmax parameterization.
    a = softmax(z), minimizes soft_max(conv(a,a), tau).
    Returns (f, g) tuple.
    """
    zs = z - z.max()
    a = np.exp(zs)
    a_sum = a.sum()
    a = a / a_sum

    conv = fast_convolve(a)
    cv = conv / tau
    cv_max = cv.max()
    exp_cv = np.exp(cv - cv_max)
    obj_val = tau * (np.log(exp_cv.sum()) + cv_max)

    w = exp_cv / exp_cv.sum()
    grad_a = compute_softgrad(a, w)
    mean_grad = np.dot(a, grad_a)
    grad_z = a * (grad_a - mean_grad)

    return obj_val, grad_z


def lbgfs_phase(a0, time_limit, tau_list, maxiter_per_tau=200):
    """L-BFGS optimization with soft-max smoothing and decreasing temperature."""
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    for tau in tau_list:
        if time.monotonic() - t0 >= time_limit:
            break

        z0 = np.log(np.maximum(a, 1e-30))

        result = minimize(
            soft_max_obj_and_grad, z0, method='L-BFGS-B',
            args=(tau,), jac=True,
            options={'maxiter': maxiter_per_tau, 'ftol': 1e-15, 'gtol': 1e-12}
        )

        z_sol = result.x
        zs = z_sol - z_sol.max()
        a_new = np.exp(zs)
        a_new = a_new / a_new.sum()

        obj_val = fast_convolve(a_new).max()
        elapsed = time.monotonic() - t0
        if obj_val < best_obj:
            best_obj = obj_val
            best_a = a_new.copy()
            a = a_new.copy()
            print(f"    L-BFGS tau={tau:.5f}: score={2*n*obj_val:.6f} (t={elapsed:.0f}s)")
        else:
            print(f"    L-BFGS tau={tau:.5f}: no improvement (score={2*n*obj_val:.6f})")

def nesterov_mirror_phase(a0, time_limit, tau_init=0.004, tau_final=0.0002,
                          lr_init=0.005, momentum=0.90):
    """Nesterov-accelerated multiplicative weight descent on soft-max objective."""
    a = a0.copy()
    n = len(a)
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        tau = tau_init * (tau_final / tau_init) ** progress

        # Nesterov lookahead
        y = a - momentum * v
        y = np.maximum(y, 1e-15)
        y = y / y.sum()

        # Soft-max gradient at lookahead point
        conv_y = fast_convolve(y)
        cv = conv_y / tau
        cv_max = cv.max()
        exp_cv = np.exp(cv - cv_max)
        w = exp_cv / exp_cv.sum()
        grad_y = compute_softgrad(y, w)

        # Update momentum and apply multiplicative step
        v = momentum * v + lr * grad_y
        log_step = -np.clip(lr * v, -40, 40)
        a = a * np.exp(log_step)
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0

            a = a / a.sum()
            v = np.zeros(n)
            lr = max(lr_init * 0.5, 0.001)
            no_improve = 0

    return best_a, best_obj


def exact_peak_sg_phase(a0, time_limit, lr_init=0.0004, momentum=0.93, top_k=5):
    """Exact peak subgradient method with Nesterov momentum on top-k peaks."""
    a = a0.copy()
    n = len(a)
    v = np.zeros(n)
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)

        # Select top-k peaks and weight by height
        k_sel = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        pv = conv[peaks]
        weights = pv / pv.sum()

        # Weighted subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            indices = np.arange(js, je + 1)
            sg[indices] += weights[i] * 2.0 * a[k - indices]

        # Nesterov update with simplex projection
        v = momentum * v + lr * sg
        a_new = a - lr * v
        a_new = project_simplex(a_new)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 300:
                rng = np.random.default_rng((it * 13 + 7) % (2**31))
                scale = 0.001 + 0.003 * min(1.0, no_improve / 800.0)
                a = project_simplex(best_a + rng.standard_normal(n) * scale)
                v = np.zeros(n)
                lr = max(lr_init * 0.6, 0.0001)
                no_improve = 0

        lr *= 0.99985

    return best_a, best_obj


def simulated_anneal_phase(a0, time_limit, temp_init=0.001, temp_final=0.00005,
                           rng_seed=777):
    """Simulated annealing: random perturbations with Metropolis accept/reject."""
    a = a0.copy()
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    rng = np.random.default_rng(rng_seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        temp = temp_init * (temp_final / temp_init) ** progress
        scale = 0.08 * temp

        a_pert = a + rng.standard_normal(n) * scale
        a_pert = project_simplex(a_pert)

        obj = fast_convolve(a_pert).max()

        if obj < best_obj:
            best_obj = obj
            best_a = a_pert.copy()
            a = a_pert.copy()
            accept_count += 1
        elif obj - best_obj < temp * 5 and rng.random() < np.exp(-(obj - best_obj) / max(temp, 1e-15)):
            a = a_pert.copy()
            accept_count += 1

        if it % 5000 == 0:
            print(f"    SA it={it}, best_score={2*n*best_obj:.6f}, temp={temp:.6f}, accepts={accept_count}")

    return best_a, best_obj

        a_pert = project_simplex(parent + rng.standard_normal(n) * 0.01)
        score = float(evaluate_sequence(a_pert.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_pert.copy()
            print(f"  Phase 0 perturbed: {best_score:.8f}")

    # === Phase 1: L-BFGS with soft-max (200s) ===
    rem = deadline - time.monotonic()
    t1 = min(200, rem * 0.12)
    if t1 > 20:
        print(f"\n--- Phase 1: L-BFGS Soft-Max (t={t1:.0f}s) ---")
        tau_list = [0.05, 0.03, 0.02, 0.015, 0.01, 0.007, 0.005, 0.003]
        a_opt, _ = lbgfs_phase(best_seq, t1, tau_list)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Nesterov Mirror Descent (500s) ===
    rem = deadline - time.monotonic()
    t2 = min(500, rem * 0.28)
    if t2 > 30:
        print(f"\n--- Phase 2: Nesterov Mirror (t={t2:.0f}s) ---")
        a_opt, _ = nesterov_mirror_phase(best_seq, t2,
                                         tau_init=0.003, tau_final=0.0002,
                                         lr_init=0.004, momentum=0.90)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  Phase 2 best: {best_score:.8f}")

    # Second Nesterov pass with different hyperparams
    rem = deadline - time.monotonic()
    t2b = min(250, rem * 0.14)
    if t2b > 20:
        print(f"\n--- Phase 2b: Nesterov round 2 (t={t2b:.0f}s) ---")
        a_opt, _ = nesterov_mirror_phase(best_seq, t2b,
                                         tau_init=0.002, tau_final=0.0001,
                                         lr_init=0.002, momentum=0.93)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  Phase 2b best: {best_score:.8f}")

    # === Phase 3: Exact Peak Subgradient (400s total, two passes) ===
    rem = deadline - time.monotonic()
    t3 = min(350, rem * 0.2)
    if t3 > 30:
        print(f"\n--- Phase 3: Exact Peak SG (t={t3:.0f}s) ---")
        a_opt, _ = exact_peak_sg_phase(best_seq, t3,
                                       lr_init=0.0003, momentum=0.94, top_k=5)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  Phase 3 best: {best_score:.8f}")

    rem = deadline - time.monotonic()
    t3b = min(150, rem * 0.08)
    if t3b > 20:
        print(f"\n--- Phase 3b: Peak SG fine (t={t3b:.0f}s) ---")
        a_opt, _ = exact_peak_sg_phase(best_seq, t3b,
                                       lr_init=0.00015, momentum=0.96, top_k=3)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  Phase 3b best: {best_score:.8f}")

    # === Phase 4: Simulated Annealing (300s) ===
    rem = deadline - time.monotonic()
    t4 = min(300, rem * 0.17)
    if t4 > 20:
        print(f"\n--- Phase 4: Simulated Annealing (t={t4:.0f}s) ---")
        a_opt, _ = simulated_anneal_phase(best_seq, t4,
                                          temp_init=0.0008, temp_final=0.00003,
                                          rng_seed=42+n)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Coordinate Descent + Final Polish (150s) ===
    rem = deadline - time.monotonic()
    t5 = min(150, rem * 0.08)
    if t5 > 10:
        print(f"\n--- Phase 5: Coord Descent (t={t5:.0f}s) ---")
        a_opt, _ = coord_descent_phase(best_seq, t5)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_opt.copy()
            print(f"  Phase 5 best: {best_score:.8f}")

    # Final micro-polish with tiny perturbations
    rem = deadline - time.monotonic()
    if rem > 15:
        print(f"\n--- Final Micro-Polish (t={rem:.0f}s) ---")
        # Very fine Nesterov
        t_fp = min(rem * 0.4, 60)
        if t_fp > 8:
            a_opt, _ = nesterov_mirror_phase(best_seq, t_fp,
                                             tau_init=0.0005, tau_final=0.0001,
                                             lr_init=0.0008, momentum=0.95)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_opt.copy()
                print(f"  Final Nesterov: {best_score:.8f}")

        # Tiny random perturbations
        rng = np.random.default_rng(31337)
        rem2 = deadline - time.monotonic()
        max_perts = max(10, int(rem2 / 0.5))
        for p_i in range(min(max_perts, 200)):
            if time.monotonic() >= deadline - 5:
                break
            for scale in [0.0002, 0.0005, 0.001]:
                a_pert = best_seq + rng.standard_normal(n) * scale
                a_pert = project_simplex(a_pert)
                score = float(evaluate_sequence(a_pert.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a_pert.copy()
                    print(f"  Micro-pert ({scale}): {best_score:.8f}")
                    break

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = min(best_score, final_score)

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    print(f"{'='*50}")
    return [float(x) for x in best_seq]
