where grad[i] is the subgradient of `max_k (a*a)[k]` w.r.t. a[i].

The subgradient of `max_k (a*a)[k]` at a point where the max is attained at k* is:

        if best_alpha > 0.01:

            hi = min(1.0, best_alpha + 0.05)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)

        if no_improve > 30:
            rng = np.random.default_rng((it * 7919 + n * 104729) % (2**31))
            scale = 0.002 + 0.001 * min(no_improve // 30, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def mirror_descent_optimize(f0, time_limit, T_init=0.003, T_min=0.00005, lr_init=0.01, mom_beta=0.85):
    """Mirror descent with momentum and adaptive temperature annealing."""
    a = np.maximum(f0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    T = T_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    momentum = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        grad = softmax_gradient(conv, a, T)
        momentum = mom_beta * momentum + (1.0 - mom_beta) * grad
        eff_grad = momentum - np.mean(momentum)
        a_new = mirror_descent_step(a, eff_grad, lr)
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                lr = max(lr * 0.92, 0.0001)

        T = max(T * 0.9985, T_min)

        if no_improve > 250:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            lr = max(lr_init * 0.5, 0.002)
            T = max(T_init * 0.3, T_min)
            momentum = np.zeros(n)
            no_improve = 0

    return best_a, best_obj


def coordinate_descent_sweep(a, n_sweeps=1, n_trials=12):
    """Coordinate descent: for each coordinate, try several rescaled values."""
    a = a.copy()
    n = len(a)
    current_obj = fast_convolve(a).max()

    for sweep in range(n_sweeps):
        for i in range(n):
            best_t = a[i]
            best_local = current_obj
            orig_val = a[i]
            if orig_val < 1e-15:
                continue
            for frac in np.linspace(0.0, 2.5, n_trials):
                t = orig_val * frac
                if abs(t - orig_val) < 1e-15:
                    continue
                a_test = a.copy()
                a_test[i] = t
                if a_test.sum() < 1e-10:
                    continue
                a_test = a_test / a_test.sum()
                obj = fast_convolve(a_test).max()
                if obj < best_local:
                    best_local = obj
                    best_t = t

            if best_t != orig_val:
                a[i] = best_t
                a = a / a.sum()
                current_obj = best_local

    return a, current_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for a given length."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # U-shaped (edge-concentrated)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine window
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.02
    inits.append(c1 / c1.sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    two_hump = h1 + h2
    inits.append(two_hump / two_hump.sum())

    # Flat-top window
    ft = np.ones(n)
    rolloff = max(3, n // 10)
    for i in range(rolloff):
        ft[i] = 0.5 * (1 + np.cos(np.pi * i / rolltoff))
        ft[n - 1 - i] = ft[i]
    inits.append(ft / ft.sum())

    # Symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def evaluate_and_update(a, best_seq, best_score, label=""):
    """Evaluate sequence and return (best_seq, best_score, improved)."""
    if a is None:
        return best_seq, best_score, False
    a = np.maximum(a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        return best_seq, best_score, False
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_seq = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return best_seq, best_score, True
    return best_seq, best_score, False


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Explore multiple lengths and parametric initializations.
    Phase 2: Deep optimization alternating mirror descent, LP-FW, and coordinate descent.
    Phase 3: Fine polish with reduced temperature and step size.
    Phase 4: Final tricks - nearby lengths, perturbations, mirror averaging.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Broad Exploration ===
    print("=== Phase 1: Broad Exploration ===")
    t_p1 = time.monotonic()
    p1_budget = 400.0

    base_n = len(parent) if parent is not None else 368
    lengths_to_try = []
    for delta in [0, -4, 4, -8, 8, -12, 12, -16, 16, -20, 20, -24, 24, -32, 32, -40, 40, -48, 48, -64, 64, 80, -80]:
        cand = base_n + delta
        if 128 <= cand <= 800:
            lengths_to_try.append(cand)
    lengths_to_try = list(dict.fromkeys(lengths_to_try))  # dedupe preserving order

    # Prioritize: parent length first, then nearby
    lengths_to_try.sort(key=lambda x: abs(x - base_n))

    for n in lengths_to_try[:10]:
        if time.monotonic() > deadline - (budget_s - p1_budget) - 60:
            break

        inits = generate_inits(n, parent, rng)
        # Rank inits by quick objective
        ranked = []
        for f_init in inits:
            conv = fast_convolve(f_init)
            obj = conv.max()
            ranked.append((obj, f_init))
        ranked.sort(key=lambda x: x[0])

        # Try top 3 initializations with quick mirror descent
        for rank, (init_obj, f_init) in enumerate(ranked[:3]):
            if time.monotonic() > deadline - (budget_s - p1_budget) - 30:
                break
            t_alloc = min(35.0, (deadline - (budget_s - p1_budget) - 30) - time.monotonic())
            if t_alloc < 10:
                break
            a_opt, _ = mirror_descent_optimize(f_init, time_limit=t_alloc,
                                               T_init=0.005, T_min=0.0001,
                                               lr_init=0.012, mom_beta=0.8)
            best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, f"p1_n{n}_r{rank}")

    print(f"  Phase 1 done. Best: {best_score:.8f}")

    # === Phase 2: Deep Optimization ===
    print("=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()
    p2_budget = 800.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_best = len(best_seq)

        n_cycles = 5
        for cycle in range(n_cycles):
            rem = min(p2_budget, budget_s) - (time.monotonic() - t_p2) - 250
            if rem < 20:
                break

            # Sub-phase A: Mirror descent
            t_md = min(rem * 0.35, 120.0)
            if t_md > 15:
                T_start = max(0.001, 0.004 * fast_convolve(f_work).max())
                a_opt, _ = mirror_descent_optimize(f_work, time_limit=t_md,
                                                   T_init=T_start, T_min=0.00003,
                                                   lr_init=0.008, mom_beta=0.85)
                best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, f"p2_md{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            rem = min(p2_budget, budget_s) - (time.monotonic() - t_p2) - 250
            if rem < 15:
                break

            # Sub-phase B: LP-based Frank-Wolfe
            t_fw = min(rem * 0.3, 100.0)
            if t_fw > 15:
                a_opt, _ = fw_lp_optimize(f_work, time_limit=t_fw)
                best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, f"p2_fw{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            rem = min(p2_budget, budget_s) - (time.monotonic() - t_p2) - 250
            if rem < 10:
                break

            # Sub-phase C: Coordinate descent
            t_cd = min(rem * 0.25, 60.0)
            if t_cd > 5:
                n_sweeps = max(1, int(t_cd / 8))
                a_cd, _ = coordinate_descent_sweep(f_work, n_sweeps=n_sweeps, n_trials=10)
                best_seq, best_score, _ = evaluate_and_update(a_cd, best_seq, best_score, f"p2_cd{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            rem = min(p2_budget, budget_s) - (time.monotonic() - t_p2) - 250
            if rem < 10:
                break

            # Sub-phase D: Additional mirror descent with lower T
            t_md2 = min(rem * 0.3, 60.0)
            if t_md2 > 10:
                a_opt, _ = mirror_descent_optimize(f_work, time_limit=t_md2,
                                                   T_init=0.0008, T_min=0.00002,
                                                   lr_init=0.005, mom_beta=0.9)
                best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, f"p2_md2_{cycle}")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

    print(f"  Phase 2 done. Best: {best_score:.8f}")

    # === Phase 3: Fine Polish ===
    print("=== Phase 3: Fine Polish ===")
    t_p3 = time.monotonic()
    p3_budget = 250.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        # Multiple rounds of increasingly fine optimization
        configs = [
            {"T_init": 0.0005, "T_min": 0.00001, "lr": 0.003, "beta": 0.9},
            {"T_init": 0.0003, "T_min": 0.000005, "lr": 0.002, "beta": 0.92},
            {"T_init": 0.0002, "T_min": 0.000003, "lr": 0.001, "beta": 0.93},
        ]

        for i, cfg in enumerate(configs):
            rem = p3_budget - (time.monotonic() - t_p3)
            if rem < 20:
                break
            t_alloc = min(60.0, rem - 10)
            if t_alloc < 10:
                break
            a_opt, _ = mirror_descent_optimize(f_work, time_limit=t_alloc,
                                               T_init=cfg["T_init"], T_min=cfg["T_min"],
                                               lr_init=cfg["lr"], mom_beta=cfg["beta"])
            best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, f"p3_md{i}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Coordinate descent fine
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 10:
            a_cd, _ = coordinate_descent_sweep(f_work, n_sweeps=3, n_trials=15)
            best_seq, best_score, _ = evaluate_and_update(a_cd, best_seq, best_score, "p3_cd")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # One more LP-FW
        rem = p3_budget - (time.monotonic() - t_p3)
        if rem > 20:
            a_opt, _ = fw_lp_optimize(f_work, time_limit=min(40.0, rem - 10))
            best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, "p3_fw")

    print(f"  Phase 3 done. Best: {best_score:.8f}")

    # === Phase 4: Final Tricks ===
    print("=== Phase 4: Final Tricks ===")

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_best = len(best_seq)

        # Try very nearby lengths
        for delta in [1, -1, 2, -2, 3, -3]:
            if time.monotonic() > deadline - 15:
                break
            new_n = n_best + delta
            if new_n < 100 or new_n > 1000:
                continue
            px = np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, n_best), f_work)
            px = project_simplex(px)
            a_opt, _ = mirror_descent_optimize(px, time_limit=min(12.0, max(5, (deadline - time.monotonic()) - 15)),
                                               T_init=0.0005, T_min=0.00001, lr_init=0.003)
            best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, f"p4_n{new_n}")

        # Tiny perturbations
        for _ in range(4):
            if time.monotonic() > deadline - 10:
                break
            n_cur = len(best_seq)
            noise = rng.standard_normal(n_cur) * 0.0003
            f_pert = project_simplex(f_work + noise)
            a_opt, _ = mirror_descent_optimize(f_pert, time_limit=min(8.0, max(3, (deadline - time.monotonic()) - 10)),
                                               T_init=0.0003, T_min=0.000005, lr_init=0.002)
            best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, "p4_pert")

        # Mirror symmetric version
        if time.monotonic() < deadline - 20:
            n_cur = len(best_seq)
            mirrored = 0.5 * f_work + 0.5 * f_work[::-1]
            mirrored = mirrored / mirrored.sum()
            a_opt, _ = mirror_descent_optimize(mirrored, time_limit=min(20.0, (deadline - time.monotonic()) - 15),
                                               T_init=0.0005, T_min=0.00001, lr_init=0.003)
            best_seq, best_score, _ = evaluate_and_update(a_opt, best_seq, best_score, "p4_mirror")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
