    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_load(a, active_w):
    """Compute load[i] = sum_j a[j]*active_w[j+i] via FFT cross-correlation."""
    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(active_w, fft_size)
    cross = np.fft.irfft(np.conj(A) * W, fft_size)
    return cross[:n]


def multiplicative_opt(a_init, time_limit, alpha=0.35, power_init=2.0, power_final=8.0, sym=True):
    """Sink-knitting multiplicative updates with adaptive power scheduling."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    if sym:
        a = (a + a[::-1]) / 2.0
        a = a / a.sum()

    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    power = power_init

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        if M < 1e-15:
            break

        active_w = (conv / M) ** power
        sw = active_w.sum()
        if sw < 1e-30:
            active_w = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            active_w = active_w / sw

        load = compute_load(a, active_w)
        load = np.maximum(load, 1e-30)
        nu = 1.0 / n
        ratio = nu / load
        ratio = np.clip(ratio, 0.01, 100.0)
        a = a * (ratio ** alpha)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            break
        a = a / s

        if sym and it % 3 == 0:
            a = (a + a[::-1]) / 2.0
            a = a / a.sum()

        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive power: increase focus on peaks over time
        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        power = power_init + (power_final - power_init) * frac

        if no_improve > 3000:
            alpha = max(alpha * 0.8, 0.05)
            no_improve = 0

    return best_a, best_obj


def smoothed_gd_opt(a_init, time_limit, T_init=None, T_min=0.0005, lr_init=0.008):
    """Smoothed gradient descent on simplex with log-sum-exp smoothing."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    if T_init is None:
        conv0 = fast_convolve(a)
        T_init = max(0.0005, 0.3 * conv0.max())

    T = T_init
    lr = lr_init
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        scaled = conv / T
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-300:
            p = np.zeros(len(conv))
            p[np.argmax(conv)] = 1.0
        else:
            p = w / ws

        fft_size = 1 << (3 * n).bit_length()
        A = np.fft.rfft(a, fft_size)
        P = np.fft.rfft(p, fft_size)
        grad = 2.0 * np.fft.irfft(np.conj(A) * P, fft_size)[:n]
        grad = grad - np.mean(grad)

        a = project_simplex(a - lr * grad)
        obj = fast_convolve(a).max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 150 == 0:
                lr = max(lr * 0.88, 0.0001)

        T = max(T * 0.998, T_min)

        if no_improve > 400:
            rng = np.random.default_rng((it * 7919 + n) % (2 ** 31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.0015)
            lr = max(lr * 1.5, 0.003)
            T = max(T_init * 0.4, T_min)
            no_improve = 0

    return best_a, best_obj


def solve_lp_direction(f, M):
    """LP: max 1^T g s.t. conv(f,g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def fw_lp_opt(a_init, time_limit):
    """Frank-Wolfe with LP directions and line search."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        g = solve_lp_direction(a, M * m_exp)

        if g is not None and g.sum() > 1e-10:
            g = g / g.sum()
            if np.max(np.abs(g - a)) < 1e-10:
                no_improve += 1
                m_exp = min(m_exp * 1.05, 3.0)
            else:
                best_alpha = 0.0
                best_val = M
                for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_t = (1 - alpha) * a + alpha * g
                    a_t = np.maximum(a_t, 0.0)
                    st = a_t.sum()
                    if st < 1e-15:
                        continue
                    a_t = a_t / st
                    val = fast_convolve(a_t).max()
                    if val < best_val:
                        best_val = val
                        best_alpha = alpha
                if best_alpha > 1e-10:
                    a = (1 - best_alpha) * a + best_alpha * g
                    a = np.maximum(a, 0.0)
                    a = a / max(a.sum(), 1e-15)
                    obj = fast_convolve(a).max()
                    if obj < best_obj - 1e-14:
                        best_obj = obj
                        best_a = a.copy()
                        no_improve = 0
                        m_exp = 1.0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
                    m_exp = min(m_exp * 1.05, 3.0)
        else:
            m_exp = min(m_exp * 1.1, 5.0)
            no_improve += 1

        if no_improve > 40:
            rng = np.random.default_rng((it * 733 + n * 31) % (2 ** 31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.003)
            no_improve = 0
            m_exp = 1.0

    return best_a, best_obj


def coordinate_polish(a_init, time_limit, top_k_frac=0.1, temp=0.00004):
    """Subgradient polish focusing on top-k peaks."""
    n = len(a_init)
    a = np.maximum(a_init.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = 0.0012
    top_k = max(5, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

        a = project_simplex(a - lr * sg)
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 100 == 0:
            lr *= 0.997
            temp *= 1.01

        if no_improve > 300:
            rng = np.random.default_rng((it * 449 + 13) % (2 ** 31))
            a = project_simplex(best_a + rng.standard_normal(n) * 0.0008)
            lr = 0.0008
            no_improve = 0

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for a given length."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c1 / c1.sum())

    h1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())

    hamming = 0.54 - 0.46 * np.cos(2 * np.pi * x / (n - 1)) if n > 1 else np.array([1.0])
    inits.append(hamming / hamming.sum())

    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.8)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Main entry point. Multi-phase hybrid search:
    Phase 1: Quick exploration (multiplicative updates on parent + other lengths).
    Phase 2: Deep optimization (alternating multiplicative, smoothed GD, LP-FW).
    Phase 3: Fine polishing (coordinate polish, low-T smoothed GD).
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Quick Exploration (250s) ===
    print("=== Phase 1: Quick Exploration ===")
    t_p1 = time.monotonic()
    p1_budget = 250.0

    if parent is not None:
        try_seq(parent, "parent")
        t_alloc = min(80.0, p1_budget - 60.0)
        if t_alloc > 20:
            print(f"  MU on parent n={len(parent)}, t={t_alloc:.0f}s")
            a_opt, _ = multiplicative_opt(parent, t_alloc, alpha=0.35, power_init=2.0, power_final=10.0, sym=True)
            try_seq(a_opt, "p1_mu_parent")

    base_n = len(parent) if parent is not None else 384
    other_lengths = []
    for delta in [-16, -8, 8, 16, 32, -32, 64, -64]:
        cn = base_n + delta
        if 128 <= cn <= 1024 and cn != base_n:
            other_lengths.append(cn)
    rng.shuffle(other_lengths)

    for n in other_lengths[:3]:
        rem = p1_budget - (time.monotonic() - t_p1)
        if rem < 30:
            break
        inits = generate_inits(n, parent, rng)
        best_init = None
        bi_obj = float('inf')
        for f_init in inits[:4]:
            c = fast_convolve(f_init)
            if c.max() < bi_obj:
                bi_obj = c.max()
                best_init = f_init
        if best_init is not None:
            t_alloc = min(40.0, rem - 20.0)
            if t_alloc > 10:
                a_opt, _ = multiplicative_opt(best_init, t_alloc, alpha=0.3, sym=True)
                try_seq(a_opt, f"p1_n{n}")

    print(f"  Phase 1 done: best={best_score:.6f}")

    # === Phase 2: Deep Optimization (1150s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()
    p2_budget = 1150.0

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()

        # Alternating optimization cycles
        while time.monotonic() < deadline - 350:
            rem = deadline - 350 - time.monotonic()
            if rem < 30:
                break

            # Cycle: MU (300s) -> SGD (200s) -> LP-FW (300s)
            t_mu = min(300.0, rem / 3.0)
            t_sgd = min(200.0, rem / 3.0)
            t_fw = min(300.0, rem / 3.0)

            if t_mu > 20:
                alpha = 0.25 if best_score < 1.51 else 0.4
                pw_f = 12.0 if best_score < 1.51 else 6.0
                print(f"  MU (t={t_mu:.0f}s, alpha={alpha}, pw_final={pw_f})")
                a_opt, _ = multiplicative_opt(f_work, t_mu, alpha=alpha,
                                              power_init=2.0, power_final=pw_f, sym=True)
                try_seq(a_opt, "p2_mu")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            rem = deadline - 350 - time.monotonic()
            if rem < 30:
                break
            if t_sgd > 20:
                T_init = max(0.0005, 0.2 * fast_convolve(f_work).max())
                print(f"  SGD (t={t_sgd:.0f}s, T={T_init:.6f})")
                a_opt, _ = smoothed_gd_opt(f_work, t_sgd, T_init=T_init, T_min=0.0003, lr_init=0.006)
                try_seq(a_opt, "p2_sgd")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            rem = deadline - 350 - time.monotonic()
            if rem < 30:
                break
            if t_fw > 20:
                print(f"  LP-FW (t={t_fw:.0f}s)")
                a_opt, _ = fw_lp_opt(f_work, t_fw)
                try_seq(a_opt, "p2_fw")
                if best_seq is not None:
                    f_work = best_seq / best_seq.sum()

            # Try symmetrization
            rem = deadline - 350 - time.monotonic()
            if rem > 30:
                symmed = (f_work + f_work[::-1]) / 2.0
                symmed = symmed / symmed.sum()
                if np.max(np.abs(symmed - f_work)) > 1e-5:
                    a_opt, _ = multiplicative_opt(symmed, min(30.0, rem / 3), alpha=0.2, sym=True)
                    try_seq(a_opt, "p2_sym")
                    if best_seq is not None:
                        f_work = best_seq / best_seq.sum()

    print(f"  Phase 2 done: best={best_score:.6f}")

    # === Phase 3: Fine Polish (300s) ===
    print("\n=== Phase 3: Fine Polish ===")
    t_p3 = time.monotonic()

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
        n_best = len(best_seq)

        # Round 1: Coordinate polish
        t_alloc = min(80.0, 300.0 - (time.monotonic() - t_p3))
        if t_alloc > 20:
            print(f"  Coord polish 1 (t={t_alloc:.0f}s)")
            a_opt, _ = coordinate_polish(f_work, t_alloc, top_k_frac=0.08, temp=0.00003)
            try_seq(a_opt, "p3_cp1")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Round 2: Fine smoothed GD
        rem = 300.0 - (time.monotonic() - t_p3)
        if rem > 30:
            T_init = max(0.0003, 0.1 * fast_convolve(f_work).max())
            t_alloc = min(80.0, rem - 20.0)
            print(f"  Fine SGD (t={t_alloc:.0f}s, T={T_init:.7f})")
            a_opt, _ = smoothed_gd_opt(f_work, t_alloc, T_init=T_init, T_min=0.0002, lr_init=0.003)
            try_seq(a_opt, "p3_fine_sgd")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Round 3: Fine multiplicative
        rem = 300.0 - (time.monotonic() - t_p3)
        if rem > 30:
            t_alloc = min(60.0, rem - 20.0)
            print(f"  Fine MU (t={t_alloc:.0f}s)")
            a_opt, _ = multiplicative_opt(f_work, t_alloc, alpha=0.15, power_init=5.0, power_final=15.0, sym=True)
            try_seq(a_opt, "p3_fine_mu")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Round 4: Coord polish again
        rem = 300.0 - (time.monotonic() - t_p3)
        if rem > 20:
            t_alloc = min(40.0, rem - 10.0)
            a_opt, _ = coordinate_polish(f_work, t_alloc, top_k_frac=0.05, temp=0.00001)
            try_seq(a_opt, "p3_cp2")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Final tiny LP-FW
        rem = 300.0 - (time.monotonic() - t_p3)
        if rem > 20:
            t_alloc = min(30.0, rem - 5.0)
            a_opt, _ = fw_lp_opt(f_work, t_alloc)
            try_seq(a_opt, "p3_final_fw")

    print(f"  Phase 3 done: best={best_score:.6f}")

    # === Phase 4: Last-ditch attempts ===
    print("\n=== Phase 4: Final Attempts ===")
    if best_seq is not None and time.monotonic() < deadline - 10:
        f_work = best_seq / best_seq.sum()
        n_best = len(f_work)

        # Tiny perturbations
        for attempt in range(5):
            if time.monotonic() > deadline - 5:
                break
            noise = rng.standard_normal(n_best) * 0.0004
            f_pert = project_simplex(f_work + noise)
            a_opt, _ = multiplicative_opt(f_pert, 5.0, alpha=0.2, power_init=5.0, power_final=15.0, sym=True)
            try_seq(a_opt, f"p4_pert_{attempt}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # Try one more smoothed GD at very low T
        if time.monotonic() < deadline - 15:
            T_init = max(0.0002, 0.05 * fast_convolve(f_work).max())
            a_opt, _ = smoothed_gd_opt(f_work, 10.0, T_init=T_init, T_min=0.0001, lr_init=0.002)
            try_seq(a_opt, "p4_final_sgd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
