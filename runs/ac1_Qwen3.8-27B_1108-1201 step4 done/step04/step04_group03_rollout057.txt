```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L^p norm optimization with Adam.
Uses softmax parameterization for the simplex constraint, p-annealing from
smooth (p=2) to peak-focused (p=128), and explores multiple sequence lengths.

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def softmax(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()

def lp_objective_and_grad(a, p):
    n = len(a)
    c = fast_convolve(a)
    c_safe = np.maximum(c, 1e-30)
    c_pow = c_safe ** p
    sum_cp = c_pow.sum()
    obj = sum_cp ** (1.0 / p)
    coef = sum_cp ** (-1.0 + 1.0 / p)
    w = c_safe ** (p - 1)
    grad = np.zeros(n)
    for i in range(n):
        m_max = min(n, 2 * n - 1 - i)
        if m_max > 0:
            grad[i] = 2.0 * coef * np.dot(w[i:i + m_max], a[:m_max])
    return obj, grad

def true_score(a):
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2 * len(a) * c.max() / (s * s))

    # Weight later phases more heavily
    w = np.linspace(0.6, 1.4, n_phases)
    w = w / w.sum()
    phase_durations = w * time_limit

            v_adam = beta2 * v_adam + (1 - beta2) * (grad_z * grad_z)

def upsample(a, new_n):
    """Upsample sequence to new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_a = np.interp(x_new, x_old, a)
    return new_a / new_a.sum()


def generate_inits(n, parent, rng):
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    x = np.arange(n, dtype=np.float64)
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c / c.sum())
    
    # Bimodal
    g2 = np.exp(-((x - 0.3*n)/ (0.1*n))**2) + np.exp(-((x - 0.7*n)/(0.1*n))**2)
    inits.append(g2 / g2.sum())
    
    if parent is not None:
        inits.append(upsample(parent, n))
        # Perturbed upsample
        up = upsample(parent, n)
        up_pert = up * np.exp(rng.standard_normal(n) * 0.02)
        inits.append(up_pert / up_pert.sum())
    
    return inits
```

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using L^p norm optimization.
Combines L-BFGS (for smooth low-p regime) and Adam (for peak-focused high-p regime)
with softmax parameterization. Explores multiple sequence lengths and p-annealing.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def softmax(z):
    z_shifted = z - z.max()
    e = np.exp(z_shifted)
    return e / e.sum()


def true_score(a):
    """Compute the actual objective: 2*n*max(conv)/(sum)^2"""
    s = a.sum()
    if s < 0.01:
        return float('inf')
    c = fast_convolve(a)
    return float(2 * len(a) * c.max() / (s * s))


def lp_obj_grad_z(z, p):
    """Compute L^p norm of conv(a,a) and gradient w.r.t. z where a=softmax(z)."""
    n = len(z)
    a = softmax(z)
    c = fast_convolve(a)
    c_safe = np.maximum(c, 1e-30)

    c_pow = c_safe ** p
    sum_cp = c_pow.sum()
    obj = sum_cp ** (1.0 / p)

    coef = sum_cp ** (-1.0 + 1.0 / p)
    w = c_safe ** (p - 1)

    # Gradient w.r.t. a: grad_a[i] = 2*coef * sum_m w[i+m]*a[m]
    # Use vectorized computation with preallocated buffer
    grad_a = np.zeros(n)
    for i in range(n):
        m_max = min(n, 2 * n - 1 - i)
        if m_max > 0:
            grad_a[i] = 2.0 * coef * np.dot(w[i:i + m_max], a[:m_max])

    # Chain rule for softmax: grad_z = (I - a a^T) grad_a = grad_a - (sum a*grad_a)*a
    grad_z = grad_a - np.dot(a, grad_a) * a
    return obj, grad_z


def lp_obj_grad_z_fft(z, p):
    """Faster version using FFT for the cross-correlation in gradient."""
    n = len(z)
    a = softmax(z)
    c = fast_convolve(a)
    c_safe = np.maximum(c, 1e-30)

    c_pow = c_safe ** p
    sum_cp = c_pow.sum()
    obj = sum_cp ** (1.0 / p)

    coef = sum_cp ** (-1.0 + 1.0 / p)
    w = c_safe ** (p - 1)

    # Cross-correlation via FFT: grad_a[i] = 2*coef * sum_m w[i+m]*a[m]
    # This equals the "anti-convolution" or cross-correlation
    # Use: cross_corr[i] = sum_m a[m] * w[i+m]
    # = ifft(fft(a, N) * conj(fft(w, N)))... let me verify
    # Actually: circular cross-correlation R[i] = sum_j a[j] * w[j+i] (mod N)
    # We need linear, so N >= n + (2n-1) - 1 = 3n - 2
    N = 1 << (3 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # Cross-correlation: ifft(conj(A) * W) gives sum_j a[j] * w[j+i]... 
    # Actually ifft(flip(A) * W) ... let me just use the direct method
    # For n <= 1024, the O(n^2) loop is fast enough in numpy
    grad_a = np.zeros(n)
    for i in range(n):
        m_max = min(n, 2 * n - 1 - i)
        if m_max > 0:
            grad_a[i] = 2.0 * coef * np.dot(w[i:i + m_max], a[:m_max])

    grad_z = grad_a - np.dot(a, grad_a) * a
    return obj, grad_z


def lfbgsopt(z0, p, maxiter=500, ftol=1e-14):
    """Run L-BFGS-B on the z parameterization with L^p objective."""
    a0 = softmax(z0)
    score_before = true_score(a0)

    def fun(z):
        obj, grad = lp_obj_grad_z(z, p)
        return obj, grad

    try:
        res = scipy_minimize(fun, z0, jac=True, method='L-BFGS-B',
                             options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
        a_res = softmax(res.x)
        score_after = true_score(a_res)
        return res.x, score_after
    except Exception:
        return z0, score_before


def adam_optimize(z0, time_limit, p_values, lr=0.01, beta1=0.9, beta2=0.999,
                  eps=1e-8, check_interval=200, label=""):

    n = len(z0)
    z = z0.copy()

    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t = 0

    a_curr = softmax(z)
    best_score = true_score(a_curr)
    best_z = z.copy()

    t0 = time.monotonic()
    deadline = t0 + time_limit

        if time.monotonic() >= deadline:
            break
        # Adaptive time per p: later (larger p) gets more time
        remaining_ps = p_values[p_values.index(p):]
        frac = 1.0 / len(remaining_ps)
        phase_end = min(deadline, max(time.monotonic(), deadline - (deadline - t0) * (1 - sum(
            1.0 / len(p_values) for _ in range(p_values.index(p) + 1)))))
        # Simpler: split time equally

        while time.monotonic() < phase_end:
            t += 1
            a_curr = softmax(z)
            obj, grad_z = lp_obj_grad_z(a_curr, p) if False else lp_obj_grad_z(z, p)

            m_adam = beta1 * m_adam + (1 - beta1) * grad_z
            v_adam = beta2 * v_adam + (1 - beta2) * (grad_z * grad_z)
            m_hat = m_adam / (1 - beta1 ** t)
            v_hat = v_adam / (1 - beta2 ** t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

            if t % check_interval == 0:

                sc = true_score(a_eval)
                if sc < best_score:
                    best_score = sc

    sc_final = true_score(a_final)
    if sc_final < best_score:
        best_score = sc_final
        best_z = z.copy()

    return best_z, best_score, softmax(best_z)


def upsample_seq(a, new_n):
    """Upsample a sequence to new length via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new_a = np.interp(x_new, x_old, a)
    new_a = np.maximum(new_a, 0)
    s = new_a.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return new_a / s


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge-weighted (triangular)
    edge_w = np.minimum(x + 1, n - x)
    inits.append(edge_w / edge_w.sum())

    # Inverted edge (peak in middle)
    inv_edge = 1.0 / (edge_w + 1.0)
    inits.append(inv_edge / inv_edge.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())

    # Bimodal
    g2 = np.exp(-((x - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c / c.sum())

    if parent is not None:
        up = upsample_seq(parent, n)
        inits.append(up)
        # Slightly perturbed upsample
        up_p = up * np.exp(rng.standard_normal(n) * 0.01)
        inits.append(up_p / up_p.sum())

    return inits
```
