Multi-strategy optimizer for minimum peak auto-correlation sequences.
Uses mirror descent (exponential weights), peak-clipping, population soups,
and frequency-domain initialization to find low-autocorrelation non-negative
sequences.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def mirror_descent_step(b, conv, eta, top_k, temperature):
    """One mirror descent step targeting top_k peaks."""
    n = len(b)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * w[i] * b[k - idx]
    
    # Multiplicative update
    b_new = b * np.exp(-eta * grad)
    b_new = np.maximum(b_new, 1e-30)
    b_new = b_new / b_new.sum()
    return b_new

def optimize_mirror(b0, time_limit, eta0=1.0, top_k=30, temp=0.001,
                    eta_decay=0.9995, patience=200, seed=0):
    """Mirror descent optimization with adaptive step size."""
    b = b0.copy()
    n = len(b)
    rng = np.random.default_rng(seed)
    
    conv = fast_convolve(b)
    best_obj = conv.max()
    best_b = b.copy()
    
    eta = eta0
    temp = temp
    no_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = mirror_descent_step(b, conv, eta, top_k, temp)
        conv = fast_convolve(b)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_b = b.copy()
            no_improve = 0
            # Slightly increase eta if improving well
            eta = min(eta * 1.001, eta0 * 2)
        else:
            no_improve += 1
            eta *= eta_decay
        
        # Anneal temperature (focus more on top peaks)
        temp *= 1.002
        
        # Restart if stuck
        if no_improve > patience:
            # Try a random restart near best
            b = project_simplex(best_b + rng.standard_normal(n) * 0.005)
            conv = fast_convolve(b)
            eta = eta0 * 0.5
            no_improve = 0
    
    return best_b, best_obj

def peak_clipping(b0, time_limit, top_k=50, alpha=0.1, seed=0):
    """Peak clipping: reduce a values that contribute to peaks."""
    b = b0.copy()
    n = len(b)
    rng = np.random.default_rng(seed)
    
    conv = fast_convolve(b)
    best_obj = conv.max()
    best_b = b.copy()
    
    t0 = time.monotonic()
    lr = 0.01
    
    while time.monotonic() - t0 < time_limit:
        # Find peaks
        peak_threshold = conv.max() * 0.95
        peaks = np.where(conv > peak_threshold)[0]
        
        if len(peaks) == 0:
            peaks = np.array([int(np.argmax(conv))])
        
        # Compute pressure
        pressure = np.zeros(n)
        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            pressure[idx] += b[k - idx]
        
        # Apply pressure (reduce values that contribute to peaks)
        b = b - alpha * lr * pressure
        b = np.maximum(b, 0)
        b_s = b.sum()
        if b_s < 1e-15:
            b = best_b.copy()
        else:
            b = b / b_s
        
        conv = fast_convolve(b)
        obj = conv.max()
        
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_b = b.copy()
            lr = min(lr * 1.01, 0.1)
        else:
            lr *= 0.99
        
        # Periodic restart
        if (int((time.monotonic() - t0) * 100) % 5000) == 0:
            if obj > best_obj * 1.01:
                b = np.maximum(best_b + rng.standard_normal(n) * 0.003, 0)
                b = b / b.sum()
    
    return best_b, best_obj

def soup_combine(b_list, weights=None):
    """Combine multiple solutions via geometric mean (on simplex)."""
    n = len(b_list[0])
    if weights is None:
        weights = np.ones(len(b_list)) / len(b_list)
    
    log_b = np.zeros(n)
    for i, b in enumerate(b_list):
        log_b += weights[i] * np.log(np.maximum(b, 1e-30))
    
    b_soup = np.exp(log_b - log_b.max())
    b_soup /= b_soup.sum()
    return b_soup

def generate_structured_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Parent (interpolated)
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        inits.append(px / px.sum())
    
    # Gaussian (various widths)
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Inverse Gaussian (U-shape)
    for sigma_frac in [0.2, 0.3, 0.4]:
        sigma = sigma_frac * n
        g = np.exp(0.5 * ((x - mid) / sigma) ** 2)
        g = g / g.sum()
        inits.append(g)
    
    # Bessel-like
    for order in [1, 2, 3]:
        t = np.abs(x - mid) / n * np.pi
        # Approximate Bessel J_0 with cos for simplicity
        b = np.cos(t * order) ** 2 + 0.1
        inits.append(b / b.sum())
    
    # Random Dirichlet
    for alpha_d in [0.5, 1.0, 2.0, 5.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha_d))
    
    # Smoothed random
    for _ in range(3):
        r = rng.standard_normal(n)
        # Smooth with a kernel
        kernel_size = max(3, n // 32)
        kernel = np.ones(kernel_size) / kernel_size
        r_smooth = np.convolve(r, kernel, mode='same')
        r_smooth = np.maximum(r_smooth, 0)
        rs = r_smooth.sum()
        if rs > 1e-10:
            inits.append(r_smooth / rs)
    
    return inits

    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():

def compute_gradient(b, peaks, weights):

    n = len(b)

        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * weights[i] * b[k - idx]

"""
Multi-phase optimizer for minimum peak auto-correlation non-negative sequences.
Combines fast mirror descent (exponential weights) with stochastic peak targeting,
population-based souping, and adaptive restarts. Uses FFT-based convolution for
speed and multi-start strategy for robustness.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_multi_gradient(b, peaks, weights):
    """Compute weighted sum of gradients of convolution at given peaks."""
    n = len(b)
    grad = np.zeros(n)
    for i in range(len(peaks)):
        wi = weights[i]
        if wi < 1e-15:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        grad[idx] += 2.0 * wi * b[k - idx]
    return grad


def mirror_descent_phase(b0, time_limit, eta0=0.8, top_k=40, temp=0.0005,
                        stochastic_frac=0.6, patience=400, seed=0,
                        mom_decay=0.999, restart_noise=0.004):
    """Mirror descent with adaptive step size and stochastic peaks."""
    b = b0.astype(np.float64).copy()
    n = len(b)
    rng = np.random.default_rng(seed)
    
    conv = fast_convolve(b)
    best_obj = conv.max()
    best_b = b.copy()
    
    eta = eta0
    cur_temp = temp
    no_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        # Select peaks
        n_sel = max(8, int(top_k * stochastic_frac))
        if n_sel >= 2 * n - 1:
            peak_idx = np.arange(2 * n - 1)
        else:
            peak_idx = np.argpartition(conv, -n_sel)[-n_sel:]
        
        pv = conv[peak_idx]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros_like(scaled)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        grad = compute_multi_gradient(b, peak_idx, w)
        
        # Mirror descent (multiplicative weights)
        b_new = b * np.exp(-eta * grad)
        b_new = np.maximum(b_new, 1e-15)
        b_sum = b_new.sum()
        if b_sum < 1e-15:
            b_new = best_b.copy()
        else:
            b_new /= b_sum
        
        conv_new = fast_convolve(b_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_b = b_new.copy()
            no_improve = 0
            b = b_new
            conv = conv_new
            eta = min(eta * 1.0005, eta0 * 3.0)
        else:
            no_improve += 1
            eta *= mom_decay
            if eta < eta0 * 0.01:
                eta = eta0 * 0.5
            
            if no_improve > patience:
                b = project_simplex(best_b + rng.standard_normal(n) * restart_noise)
                conv = fast_convolve(b)
                eta = eta0
                no_improve = 0
                restart_noise *= 0.9
        
        cur_temp *= 1.005
    
    return best_b, best_obj


def gradient_descent_phase(b0, time_limit, lr0=0.01, top_k=40, temp=0.001,
                          mom=0.9, patience=400, seed=0):
    """Projected gradient descent with momentum on the simplex."""
    b = b0.astype(np.float64).copy()
    n = len(b)
    rng = np.random.default_rng(seed)
    
    conv = fast_convolve(b)
    best_obj = conv.max()
    best_b = b.copy()
    
    lr = lr0
    cur_temp = temp
    v = np.zeros(n)
    no_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        n_sel = max(8, int(top_k * 0.7))
        if n_sel >= 2 * n - 1:
            peak_idx = np.arange(2 * n - 1)
        else:
            peak_idx = np.argpartition(conv, -n_sel)[-n_sel:]
        
        pv = conv[peak_idx]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros_like(scaled)
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        grad = compute_multi_gradient(b, peak_idx, w)
        
        # Momentum update
        v = mom * v - lr * grad
        b_new = project_simplex(b + v)
        
        conv_new = fast_convolve(b_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_b = b_new.copy()
            no_improve = 0
            b = b_new
            conv = conv_new
            lr = min(lr * 1.001, lr0 * 5.0)
        else:
            no_improve += 1
            lr *= 0.998
            if no_improve > patience:
                b = project_simplex(best_b + rng.standard_normal(n) * 0.003)
                v = np.zeros(n)
                lr = lr0 * 0.5
                conv = fast_convolve(b)
                no_improve = 0
        
        cur_temp *= 1.004
    
    return best_b, best_obj


def soup_combine(b_list, weights=None):
    """Geometric mean combination on the simplex (Fréchet mean under KL)."""
    n = len(b_list[0])
    k = len(b_list)
    if weights is None:
        weights = np.ones(k) / k
    weights = np.asarray(weights, dtype=np.float64)
    weights = weights / weights.sum()
    
    log_sum = np.zeros(n)
    for i in range(k):
        log_sum += weights[i] * np.log(np.maximum(b_list[i], 1e-30))
    
    b_soup = np.exp(log_sum - log_sum.max())
    b_soup = b_soup / b_soup.sum()
    return b_soup


def arith_combine(b_list, weights=None):
    """Arithmetic mean combination (simpler, often works well)."""
    k = len(b_list)
    if weights is None:
        weights = np.ones(k) / k
    b = np.zeros_like(b_list[0])
    for i in range(k):
        b += weights[i] * b_list[i]
    return b / b.sum()


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    mid = 0.5 * (n - 1)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Parent interpolated
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    # Gaussians
    for sf in [0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
        sigma = sf * n
        g = np.exp(-0.5 * ((x - mid) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Asymmetric (shifted Gaussian)
    for shift_frac in [-0.2, -0.1, 0.1, 0.2]:
        center = mid + shift_frac * n
        sigma = 0.25 * n
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Beta distributions
    for a_b, b_b in [(1.5, 2.5), (2.5, 1.5), (2, 3), (3, 2), (1.5, 1.5)]:
        t = x / (n - 1 + 1e-10)
        bv = t**(a_b-1) * (1-t)**(b_b-1) + 1e-10
        inits.append(bv / bv.sum())
    
    # Cosine windows
    for alpha_c in [0.5, 1.0, 1.5, 2.0]:
        c = 0.5 * (1 + np.cos(np.pi * alpha_c * (2 * x / (n-1) - 1)))
        c = np.maximum(c, 1e-10)
        inits.append(c / c.sum())
    
    # Random
    for _ in range(5):
        inits.append(rng.dirichlet(np.ones(n) * 1.0))
    
    # Smoothed random
    for _ in range(3):
        r = rng.standard_normal(n)
        ksize = max(5, n // 20)
        kern = np.exp(-0.5 * (np.arange(ksize) - ksize//2)**2 / (ksize/4)**2)
        kern /= kern.sum()
        r = np.convolve(r, kern, mode='same')
        r = np.maximum(r, 0)
        rs = r.sum()
        if rs > 1e-10:
            inits.append(r / rs)
    
    return inits
