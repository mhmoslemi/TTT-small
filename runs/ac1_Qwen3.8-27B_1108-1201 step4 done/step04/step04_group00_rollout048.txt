"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using analytical subgradient methods,
exact coordinate descent with 1D quadratic maximization, and multi-length
exploration with adaptive time allocation.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 1:
        return np.array([a[0] * a[0]])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, T=0.001, top_k=50):
    """Compute softmax-weighted subgradient of max_k conv[k]."""
    n = len(a)
    size = 2 * n - 1
    k_use = min(top_k, size)
    peak_idx = np.argpartition(conv, -k_use)[-k_use:]
    vals = conv[peak_idx]
    scaled = vals / T
    scaled -= scaled.max()
    weights = np.exp(scaled)
    W = weights.sum()
    if W < 1e-30:
        return np.zeros(n)
    
    sg = np.zeros(n)
    for i in range(k_use):
        k = int(peak_idx[i])
        w_i = 2.0 * weights[i] / W
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            sg[j_start:j_end + 1] += w_i * a[idx]
    
    return sg


def subgradient_optimize(a0, time_limit, verbose=False):
    """Subgradient method with momentum and adaptive step size."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    beta = 0.9  # momentum coefficient
    lr = 0.01  # learning rate
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Compute subgradient
        T = max(0.0001, 0.001 * (1 + 0.01 * it))  # slowly increasing temperature
        sg = compute_subgradient(a, conv, T=T, top_k=min(80, n))
        
        # Momentum update
        v = beta * v - lr * sg
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            conv = conv_new
        else:
            no_improve += 1
            a = a_new  # still update current position
            conv = conv_new
        
        # Adaptive learning rate
        if no_improve > 100:
            lr *= 0.95
            if no_improve > 500:
                # Restart
                rng = np.random.default_rng(it)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                conv = fast_convolve(a)
                v = np.zeros(n)
                lr = 0.01
                no_improve = 0
                best_obj = conv.max()
                best_a = a.copy()
        
        # Periodically inject noise
        if it % 200 == 0:
            noise_scale = 0.001 * (1 + no_improve / 100.0)
            a = project_simplex(a + np.random.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
    
    return best_a, best_obj


def coordinate_descent_exact(a0, time_limit, verbose=False):
    """Coordinate descent with exact 1D minimization via golden section search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    passes = 0
    
    def obj_1d(x, j, a_base):
        """Compute max conv when a[j]=x, others scaled to maintain sum=1."""
        a_trial = a_base.copy()
        old_val = a_base[j]
        if old_val >= 1.0 - 1e-15:
            # All mass on j
            a_trial = np.zeros(n)
            a_trial[j] = 1.0
        else:
            scale = (1.0 - x) / (1.0 - old_val)
            a_trial[j] = x
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a_trial[mask] = a_base[mask] * scale
        if a_trial.sum() < 1e-15:
            return 1e10
        conv = fast_convolve(a_trial)
        return conv.max()
    
    golden_ratio = (np.sqrt(5) - 1) / 2  # ≈ 0.618
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-10:
                # Try to give it some mass
                x_lo, x_hi = 0.0, 0.01
            else:
                x_lo, x_hi = 0.0, 1.0 - 1e-10
            
            # Golden section search
            a_b = x_lo + (1 - golden_ratio) * (x_hi - x_lo)
            b_b = x_lo + golden_ratio * (x_hi - x_lo)
            fa = obj_1d(a_b, j, a)
            fb = obj_1d(b_b, j, a)
            
            for _ in range(40):
                if fa < fb:
                    x_hi = b_b
                    b_b = a_b
                    fb = fa
                    a_b = x_lo + (1 - golden_ratio) * (x_hi - x_lo)
                    fa = obj_1d(a_b, j, a)
                else:
                    x_lo = a_b
                    a_b = b_b
                    fa = fb
                    b_b = x_lo + golden_ratio * (x_hi - x_lo)
                    fb = obj_1d(b_b, j, a)
            
            x_opt = (x_lo + x_hi) / 2.0
            f_opt = obj_1d(x_opt, j, a)
            
            if f_opt < best_obj - 1e-10:
                # Apply the update
                if old_val < 1.0 - 1e-15:
                    scale = (1.0 - x_opt) / (1.0 - old_val)
                    a[j] = x_opt
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                    best_obj = f_opt
                    best_a = a.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a normalized sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
