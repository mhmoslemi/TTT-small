    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    tmp = u * np.arange(1, n + 1, dtype=np.float64) - (cssv - 1.0)
    cond = tmp > 0
    if not cond.any():
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def softmax_feat_grad(a, weights):
    """
    Compute gradient of sum_k weights[k] * conv(a,a)[k] w.r.t. a.
    Result: grad[i] = 2 * sum_k weights[k] * a[k-i] (valid indices only).
    """
    n = len(a)
    conv_len = len(weights)
    total_len = conv_len + n - 1
    fs = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fs)
    Ar = np.fft.rfft(a[::-1], fs)
    corr = np.fft.irfft(W * Ar, fs)[:total_len]
    return 2.0 * corr[n - 1: n - 1 + n]


def run_softmax_gd(a0, time_limit, T0, T_min, lr0, mom, seed):
    """
    Smoothed gradient descent using softmax approximation of max convolution.
    Returns (best_a, best_obj).
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n, dtype=np.float64)
    T = T0
    lr = lr0
    conv = fast_convolve(a)
    best_obj = float(conv.max())
    best_a = a.copy()

    t0 = time.monotonic()
    no_imp = 0
    it = 0
    rng = np.random.default_rng(seed)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)

        # Softmax weights over conv lags
        sc = conv / T
        sc -= sc.max()
        w = np.exp(sc)
        sw = w.sum()
        if sw < 1e-30:
            w = np.ones(len(conv)) / len(conv)
        else:
            w = w / sw

        grad = softmax_feat_grad(a, w)
        v = mom * v - lr * grad
        a_new = project_simplex(a + v)

        c_new = fast_convolve(a_new)
        obj_new = float(c_new.max())

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_a = a_new.copy()
            no_imp = 0
        else:
            no_imp += 1

        a = a_new

        # Anneal temperature
        if it % 4 == 0:
            T = max(T_min, T * 0.997)

        # Adaptive learning rate / restart
        if no_imp > 150:
            if no_imp > 500:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
                v = np.zeros(n)
                T = max(T0 * 0.4, T * 2.0)
                lr = lr0
            else:
                lr = max(lr * 0.8, 0.0005)
            no_imp = 0

    return best_a, best_obj


def solve_lp_dir(a, conv, top_k):
    """
    Solve the linearized Frank-Wolfe LP:
      min t
      s.t. t >= conv[k] + 2 * sum_i a[k-i] * g[i]  for k in top_k peaks
           sum(g) = 0
           g[i] >= -a[i]
    Returns direction g or None.
    """
    n = len(a)
    if top_k >= len(conv):
        peaks = np.arange(len(conv), dtype=np.intp)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peaks = np.sort(peaks)

    m = len(peaks)
    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)

    for idx in range(m):
        k = int(peaks[idx])
        A_ub[idx, 0] = -1.0
        b_ub[idx] = -conv[k]
        # A_ub[idx, 1+i] = 2 * a[k-i] if 0 <= k-i < n
        lo_i = max(0, k - (n - 1))
        hi_i = min(n - 1, k)
        if lo_i <= hi_i:
            indices = np.arange(lo_i, hi_i + 1)
            A_ub[idx, 1 + lo_i: 1 + hi_i + 1] = 2.0 * a[k - indices]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.array([0.0])

    bounds = [(None, None)]
    for i in range(n):
        bounds.append((-float(a[i]), None))

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[1:]
    return None


def run_lp_fw(a0, time_limit, top_k_init, top_k_max, seed):
    """
    LP-based Frank-Wolfe with adaptive constraint count.
    Starts with top_k_init constraints, increases to top_k_max as we converge.
    """
    a = np.maximum(a0, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = float(conv.max())
    best_a = a.copy()

    t0 = time.monotonic()
    no_imp = 0
    rng = np.random.default_rng(seed)
    it = 0
    cur_top_k = top_k_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Adaptive top_k: increase as we converge
        if no_imp > 20 and cur_top_k < top_k_max:
            cur_top_k = min(top_k_max, cur_top_k + 3)

        conv = fast_convolve(a)
        g = solve_lp_dir(a, conv, cur_top_k)

        if g is None:
            no_imp += 3
            if no_imp > 40:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                no_imp = 0
                cur_top_k = top_k_init
            continue

        target = a + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_imp += 1
            continue
        target = target / ts

        # Fixed-point check
        if np.max(np.abs(target - a)) < 1e-11:
            no_imp += 1
            if no_imp > 30:
                a = project_simplex(best_a + rng.standard_normal(n) * 0.004)
                no_imp = 0
                cur_top_k = top_k_init
            continue

        # Line search (coarse + fine)
        best_alpha = 0.0
        best_val = float(conv.max())
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * a + alpha * target
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = float(fast_convolve(ft).max())
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha * 0.7)
            hi = min(1.0, best_alpha * 1.3)
            for _ in range(6):
                mid = (lo + hi) * 0.5
                ft = (1.0 - mid) * a + mid * target
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = float(fast_convolve(ft).max())
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-9:
            a = (1.0 - best_alpha) * a + best_alpha * target
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
        else:
            no_imp += 1

        if no_imp > 50:
            scale = 0.003 + 0.001 * min(no_imp // 50, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            no_imp = 0
            cur_top_k = top_k_init

    return best_a, best_obj
