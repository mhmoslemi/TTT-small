import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(f, conv, top_k=10, temp=0.001):
    """Compute weighted subgradient of max(conv) using top-k peaks."""
    n = len(f)
    # Get top-k peaks
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        w = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        # Softmax weights
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
    
    # Subgradient: for each peak k, the subgradient is f reversed and shifted
    # conv[k] = sum_i f[i] * f[k-i]
    # d(conv[k])/d(f[j]) = f[k-j] (for valid j)
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-10:
            continue
        k = int(peaks[idx])
        wv = w[idx]
        # f[j] * f[k-j] for j where both are valid
        # j ranges from max(0, k-n+1) to min(n-1, k)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # The subgradient contribution to f[j] is f[k-j]
        for j in range(js, je + 1):
            sg[j] += wv * f[k - j]
    
    return sg

def egd_optimize(f0, time_limit, top_k=20, temp_init=0.0005, temp_end=0.00005,
                 lr_init=0.1, lr_end=0.001, diversity=0.001):
    """Exponential Gradient Descent for minimizing max(auto-conv)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    total_iter = 0
    
    while time.monotonic() - t0 < time_limit:
        total_iter += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / time_limit)
        
        # Anneal temperature and learning rate
        temp = temp_init * (temp_end / temp_init) ** progress
        lr = lr_init * (lr_end / lr_init) ** progress
        
        # Compute subgradient
        sg = compute_subgradient(f, conv, top_k=top_k, temp=temp)
        
        # EGD update: f_new[i] = f[i] * exp(-lr * sg[i])
        # To avoid underflow, work in log-space
        log_f = np.log(np.maximum(f, 1e-15))
        log_f -= lr * sg
        log_f -= log_f.max()  # numerical stability
        f_new = np.exp(log_f)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / f_new_sum
        
        # Add diversity to prevent collapse
        if diversity > 0:
            f_new = (1 - diversity) * f_new + diversity / n
        
        conv = fast_convolve(f_new)
        M_new = conv.max()
        
        if M_new < best_obj - 1e-12:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Try smaller effective lr
            if no_improve > 20:
                f = best_f.copy()
                f_new = f * (1 - lr * 0.1 * sg)
                f_new = np.maximum(f_new, 0)
                f_new_sum = f_new.sum()
                if f_new_sum > 1e-15:
                    f_new = f_new / f_new_sum
                conv = fast_convolve(f_new)
                M_new = conv.max()
                if M_new < best_obj - 1e-12:
                    best_obj = M_new
                    best_f = f_new.copy()
                    f = f_new
                    no_improve = 0
                else:
                    f = f_new  # still move, even if not improving
                    if no_improve > 100:
                        # Restart with perturbation
                        rng = np.random.default_rng(total_iter)
                        f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
                        conv = fast_convolve(f)
                        no_improve = 0
        
        f = f_new
    
    return best_f, best_obj

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient_fast(f, conv, top_k=20, temp=0.001):
    """Vectorized subgradient computation."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        w = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
    
    # For each peak k, subgradient contribution to f[j] is f[k-j] for valid j
    # Vectorize: for peak k, the valid j range is [max(0,k-n+1), min(n-1,k)]
    # f[k-j] for j in that range is f[min(n-1,k) : max(0,k-n+1)-1 : -1] (reversed slice)
    
    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        wv = float(w[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        # f[k-j] for j=js,js+1,...,je
        # = f[k-js], f[k-js-1], ..., f[k-je]

"""
Multi-strategy optimizer for minimizing 2*n*max(auto-conv)/(sum)^2.
Combines: (1) momentum-accelerated exponential gradient descent with annealed
softmax peak weighting, (2) LP-based Frank-Wolfe, (3) symmetric sequence search,
(4) coordinate descent polishing, and (5) multi-length exploration.
Key innovations over previous: vectorized subgradient, momentum EGD, adaptive
annealing schedules, edge/bimodal initializations, and structured restarts.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    if size <= 0:
        return np.array([0.0])
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient_fast(f, conv, top_k=20, temp=0.001):
    """Vectorized subgradient of weighted sum of top-k convolution peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
        w = np.ones(len(peaks)) / len(peaks)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()

    sg = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        wv = float(w[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        # Subgradient at position k: sg[j] += w * f[k-j] for j in [js, je]
        # f[k-j] for j=js..je is f[k-js], f[k-js-1], ..., f[k-je] = f[je], f[je-1], ..., f[js]
        sg[js:je + 1] += wv * f[js:je + 1][::-1]

    return sg


def egd_optimize(f0, time_limit, top_k=20, temp_init=0.001, temp_end=0.00005,
                 lr_init=0.05, lr_end=0.001, diversity=0.001, momentum=0.9,
                 force_symmetric=False):
    """Exponential Gradient Descent with momentum for minimizing max(auto-conv)."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    total_iter = 0
    v_mom = np.zeros(n)
    rng = np.random.default_rng(54321 + n)

    while time.monotonic() - t0 < time_limit:
        total_iter += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / max(time_limit, 1e-6))

        # Cosine annealing for smoother transitions
        cos_factor = 0.5 * (1.0 + np.cos(np.pi * progress))
        temp = temp_end + (temp_init - temp_end) * cos_factor
        lr = lr_end + (lr_init - lr_end) * cos_factor

        # Compute subgradient
        sg = compute_subgradient_fast(f, conv, top_k=top_k, temp=temp)

        # Nesterov-style momentum
        v_mom = momentum * v_mom - lr * sg

        # EGD update in log-space
        log_f = np.log(np.maximum(f, 1e-15))
        log_f += v_mom
        # Numerical stability
        log_f_max = log_f.max()
        if log_f_max > 50:
            log_f -= log_f_max
        f_new = np.exp(log_f)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15 or not np.isfinite(f_new_sum):
            f_new = np.ones(n) / n
        else:
            f_new = f_new / f_new_sum

        # Enforce symmetry if requested
        if force_symmetric:
            f_new = 0.5 * (f_new + f_new[::-1])
            f_new = f_new / f_new.sum()

        # Diversity regularization to prevent collapse
        if diversity > 0:
            f_new = (1.0 - diversity) * f_new + diversity / n

        conv = fast_convolve(f_new)
        M_new = conv.max()

        if M_new < best_obj - 1e-12:
            best_obj = M_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        f = f_new

        # Adaptive restart if stuck
        if no_improve > 150:
            if M_new > best_obj * 1.005:
                f = best_f.copy()
                v_mom = np.zeros(n)
            else:
                scale = 0.003 + 0.002 * (no_improve // 200)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                if force_symmetric:
                    f = 0.5 * (f + f[::-1])
                    f = f / f.sum()
                v_mom = np.zeros(n)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0

        # Early termination if converged
        if total_iter > 50 and M < best_obj * (1 + 1e-10) and no_improve > 500:
            break

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle for precise optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng_local = np.random.default_rng(777 + n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 3:
            break

        # Build LP matrix (sparse-friendly construction)
        c = -np.ones(n)
        A_ub = np.zeros((2 * n - 1, n))
        for j in range(n):
            A_ub[j:min(j + n, 2 * n - 1), j] = f[:min(n, 2 * n - 1 - j)]
        b_ub = np.full(2 * n - 1, M * m_exp)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
            g = res.x if res.success else None
        except Exception:
            g = None

        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        # Fixed-point check
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 40:
                break
            continue

        # Line search: coarse then fine
        best_alpha = 0.0
        best_val = M
        alphas_coarse = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        for alpha in alphas_coarse:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection around best coarse alpha
        if 0.1 < best_alpha < 0.9:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        # Structured restart
        if no_improve > 50:
            strat = no_improve % 4
            if strat == 0:
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.004)
            elif strat == 1:
                blend = 0.3 + 0.05 * (no_improve // 60)
                f = project_simplex((1 - blend) * best_f + blend * best_f[::-1])
            elif strat == 2:
                # Boost edges
                edge_boost = np.ones(n)
                bw = max(1, n // 20)
                edge_boost[:bw] = 1.5
                edge_boost[-bw:] = 1.5
                f = project_simplex(best_f * edge_boost)
            else:
                f = project_simplex(rng_local.dirichlet(np.ones(n) * 0.7))
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit):
    """Efficient coordinate descent: try moving mass between pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(9999)

    # Precompute: for each pair (i, j), the change in max conv when moving mass
    # is expensive, so we sample pairs

    while time.monotonic() - t0 < time_limit:
        improved = False
        n_papers = 50  # number of random pairs to try per sweep

        for _ in range(n_papers):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue

            # Try moving a small amount of mass from j to i
            for frac in [0.02, 0.05, 0.1]:
                delta = f[j] * frac
                if delta < 1e-15:
                    continue
                f_new = f.copy()
                f_new[i] += delta
                f_new[j] -= delta
                if f_new[j] < 0:
                    f_new[i] += f_new[j]
                    f_new[j] = 0.0
                f_new = np.maximum(f_new, 0.0)
                sn = f_new.sum()
                if sn < 1e-15:
                    continue
                f_new = f_new / sn
                val = fast_convolve(f_new).max()
                if val < best_obj - 1e-15:
                    f = f_new
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-concentrated (U-shaped)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.7
    inits.append(edge_w / edge_w.sum())

    # Strong edge (bimodal at ends)
    bimodal = (np.exp(-((x - 0) ** 2) / (2 * (n / 10.0) ** 2)) +
              np.exp(-((x - (n - 1)) ** 2) / (2 * (n / 10.0) ** 2)))
    inits.append(bimodal / bimodal.sum())

    # Three-peak
    three = (np.exp(-((x - 0) ** 2) / (2 * (n / 8.0) ** 2)) +
            0.5 * np.exp(-((x - n / 2.0) ** 2) / (2 * (n / 10.0) ** 2)) +
            np.exp(-((x - (n - 1)) ** 2) / (2 * (n / 8.0) ** 2)))
    inits.append(three / three.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Random Dirichlet (concentrated)
    inits.append(rng.dirichlet(np.ones(n) * 0.4))

    # Cosine
    cos_w = np.cos(np.pi * x / (2.0 * (n - 1))) ** 2
    inits.append(cos_w / cos_w.sum())

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Power law decay from edges
    pl = 1.0 / (np.minimum(x + 1.0, n - x) ** 1.2)
    inits.append(pl / pl.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 1:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def try_sequence(a, label, state):

    return score
