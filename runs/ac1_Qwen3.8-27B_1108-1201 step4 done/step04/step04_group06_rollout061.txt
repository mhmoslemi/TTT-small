def lp_direction(a, M):

    This finds direction d to increase sum while keeping peaks bounded."""

    C = build_conv_matrix(a)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid approach:
1. Multi-start with diverse window initializations and gradient warmup
2. LP-based direction finding (minimize peak of a*g subject to sum(g)=1, g>=0)
3. Adaptive switching between LP and gradient methods
4. Final polishing with coordinate descent and micro-steps
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.special import i0


def fast_convolve(a):
    """FFT-based full linear convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_soft_grad(a, conv, top_k=50, temp=0.005):
    """Compute softmax-weighted gradient of peak autocorrelation (vectorized)."""
    n = len(a)
    if top_k >= len(conv):
        sorted_idx = np.argsort(conv)[::-1]
        peaks = sorted_idx
        peak_vals = conv[peaks]
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peak_vals = conv[peaks]

    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    # grad[i] = sum_k w[k] * a[k-i] where k-i in [0, n-1]
    # Use FFT cross-correlation
    L = 1 << (2 * n - 1).bit_length()
    A_fft = np.fft.rfft(a, L)
    # Build full w array of length 2n-1 (padded to L)
    w_full = np.zeros(L)
    for idx in range(len(peaks)):
        if w[idx] > 1e-14:
            w_full[peaks[idx]] = w[idx]
    W_fft = np.fft.rfft(w_full, L)
    # Cross-correlation: grad[i] = sum_j a[j] * w[j+i] = cross at negative lag i
    cross = np.fft.irfft(A_fft * np.conj(W_fft), L)
    # cross[(L-i) % L] gives sum_j a[j]*w[j-i] circular = sum_j a[j]*w[j+i] for our purposes
    grad = np.zeros(n)
    for i in range(n):
        grad[i] = cross[(L - i) % L]
    return grad


def gradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=50, temp_init=0.003):
    """Nesterov-accelerated gradient descent for the minimax objective."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov lookahead
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y
        else:
            y = a.copy()

        conv_y = fast_convolve(y)
        grad = compute_soft_grad(y, conv_y, top_k=top_k, temp=temp)

        # Project gradient onto sum=0 hyperplane
        grad_mean = grad.sum() / n
        grad = grad - grad_mean

        # Gradient step
        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()
            v = np.zeros(n)
            y = a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        it += 1
        # Adaptive parameters
        lr *= 0.9992
        temp *= 1.004
        mom = min(0.97, mom + 0.0003)

        if no_improve >= 400:
            # Restart velocity
            v = np.zeros(n)
            no_improve = 200

        if no_improve >= 800:
            break

    return best_a, best_obj

def build_conv_mat(a):
    """Build matrix C such that C @ g = linear_convolve(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def lp_find_direction(a, verbose=False):
    """
    Solve LP: min t s.t. C@g <= t*e, sum(g)=1, g>=0
    Returns g (the direction) and t (the predicted peak of a*g).
    """
    n = len(a)
    C = build_conv_mat(a)
    m = 2 * n - 1

    # Variables: g[0..n-1], t (index n)
    # Objective: minimize t -> c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[n] = 1.0

    # Inequalities: C@g - t <= 0 for each row
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = C
    A_ub[:, n] = -1.0
    b_ub = np.zeros(m)

    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = res.x[:n]
            t_val = res.x[n]
            g = np.maximum(g, 0)
            gs = g.sum()
            if gs > 1e-15:
                g = g / gs
            return g, t_val, True
    except Exception:
        pass
    return a.copy(), np.inf, False


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with perturbations."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    n = len(a)
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()

    t0 = time.monotonic()
    iter_count = 0
    no_improve = 0
    perturb_count = 0
    rng = np.random.default_rng(12345 + n)
    max_no_improve = 40

    while time.monotonic() - t0 < time_limit:
        # Find LP direction
        g, t_pred, success = lp_find_direction(a)

        if success:
            # Try several step sizes
            best_step_a = a.copy()
            best_step_obj = 2.0 * n * fast_convolve(a).max()

            candidates = [0.8, 0.5, 0.3, 0.2, 0.1, 0.05]
            for alpha in candidates:
                a_try = (1 - alpha) * a + alpha * g
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_step_obj - 1e-14:
                    best_step_obj = obj_try
                    best_step_a = a_try.copy()
                    break

            if best_step_obj < 2.0 * n * conv.max():
                a = best_step_a
                conv = fast_convolve(a)
                M = conv.max()
                cur_obj = 2.0 * n * M
                if cur_obj < best_obj - 1e-14:
                    best_obj = cur_obj
                    best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        iter_count += 1
        elapsed = time.monotonic() - t0

        if verbose and iter_count % 20 == 0:
            print(f"    LP iter {iter_count}, obj={best_obj:.8f}, M={M:.6f}, t={elapsed:.1f}s")

        # Handle stagnation
        if no_improve >= max_no_improve:
            scale = 0.002 + 0.003 * min(perturb_count, 5)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0)
            s = a.sum()
            if s > 1e-15:
                a = a / s
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(150, 40 + 20 * perturb_count)

            if perturb_count > 50:
                # Try gradient escape
                a_tmp, obj_tmp = gradient_optimize(best_a, time_limit=min(5.0, time_limit - elapsed - 5),
                                                   lr_init=0.008, top_k=50, temp_init=0.002)
                if obj_tmp < best_obj:
                    best_obj = obj_tmp
                    best_a = a_tmp.copy()
                a = best_a.copy()
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0
                perturb_count = 0
                max_no_improve = 40

    return best_a, best_obj


def window_init(n, wtype):
    """Generate diverse window function initializations."""
    x = np.arange(n, dtype=np.float64)
    if n <= 2:
        return np.ones(n) / n

    if wtype == 'hann':
        a = 0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1)))
    elif wtype == 'hamming':
        a = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / (n - 1))
    elif wtype == 'blackman':
        a = 0.42 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1)) + 0.08 * np.cos(4.0 * np.pi * x / (n - 1))
    elif wtype == 'blackman_harris':
        a = (0.35875 - 0.48829 * np.cos(2.0 * np.pi * x / (n - 1))
             + 0.14128 * np.cos(4.0 * np.pi * x / (n - 1))
             - 0.01168 * np.cos(6.0 * np.pi * x / (n - 1)))
    elif wtype == 'kaiser5':
        beta = 5.0
        r = 2.0 * x / (n - 1) - 1.0
        a = i0(beta * np.sqrt(np.maximum(1.0 - r * r, 0))) / i0(beta)
    elif wtype == 'kaiser10':
        beta = 10.0
        r = 2.0 * x / (n - 1) - 1.0
        a = i0(beta * np.sqrt(np.maximum(1.0 - r * r, 0))) / i0(beta)
    elif wtype == 'root_hann':
        a = np.sqrt(np.maximum(0.5 * (1.0 + np.cos(2.0 * np.pi * x / (n - 1))), 0))
    elif wtype == 'sine':
        a = np.sin(np.pi * (x + 0.5) / n) ** 2
    elif wtype == 'power03':
        a = (x + 1.0) ** 0.3
    elif wtype == 'power_minus':
        a = (n - x) ** 0.3
    elif wtype == 'bimodal':
        edge = max(1, n // 4)
        a = np.zeros(n)
        a[:edge] = 1.0
        a[-edge:] = 1.0
        # Smooth edges
        for i in range(min(edge, 20)):
            frac = 0.5 * (1.0 + np.cos(np.pi * i / edge))
            a[i] *= frac
            a[-1 - i] *= frac
    elif wtype == 'flat_top':
        a = (0.21557895 - 0.41663158 * np.cos(2.0 * np.pi * x / (n - 1))
             + 0.277263158 * np.cos(4.0 * np.pi * x / (n - 1))
             - 0.083578947 * np.cos(6.0 * np.pi * x / (n - 1))
             + 0.006947368 * np.cos(8.0 * np.pi * x / (n - 1)))
    elif wtype == 'triangle':
        mid = (n - 1) / 2.0
        a = 1.0 - np.abs(x - mid) / mid
    else:
        a = np.ones(n)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def coordinate_descent(a0, time_limit, rng=None):
    """Random coordinate descent: swap small mass between pairs."""
    if rng is None:
        rng = np.random.default_rng(999)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    n = len(a)
    s = a.sum()
    a = a / s

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    # Precompute which lags each pair affects
    while time.monotonic() - t0 < time_limit:
        # Try many random pairs
        n_trials = max(100, min(5000, 200000 // n))
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < 1e-10:
                continue

            # Try moving delta from i to j
            for delta_frac in [0.01, 0.005, 0.02]:
                delta = a[i] * delta_frac
                if delta < 1e-12:
                    continue
                a[i] -= delta
                a[j] += delta
                conv_try = fast_convolve(a)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-14:
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
                else:
                    a[i] += delta
                    a[j] -= delta
            else:
                no_improve += 1

        if no_improve > 20000:
            break

    return best_a, best_obj

    n_cands = len(top_candidates)

    for i, (n, score, name, a0) in enumerate(top_candidates):
        if time.monotonic() > deadline - 250:
            break
        t_remaining_for_cands = deadline - time.monotonic() - 250
        t_alloc = min(t_remaining_for_cands / max(1, n_cands - i) * 0.85, 400.0)
        if t_alloc < 10:
            break

        print(f"  LP cand {i}: n={n}, {name}, score={score:.6f}, alloc={t_alloc:.0f}s")

        # Gradient warmup (short)
        t_warm = min(8.0, t_alloc * 0.1)
        if t_warm > 3:
            a_warm, _ = gradient_optimize(a0, time_limit=t_warm,
                                          lr_init=0.01, top_k=40, temp_init=0.003)
        else:
            a_warm = a0.copy()

        # LP optimization
        t_lp = t_alloc - t_warm - 5
        if t_lp > 5:
            a_opt, obj = lp_optimize(a_warm, time_limit=t_lp, verbose=(n <= 640))
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"    LP done: {score:.6f} *NEW BEST*")
                else:
                    print(f"    LP done: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic() - t2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Polish the best candidate (350s)
    # =========================================================
    if best_sequence is not None and best_score < float('inf'):
        print("=== Phase 3: Polish ===")
        t3 = time.monotonic()
        t_remain = deadline - time.monotonic() - 30

        # Multiple LP runs with different perturbations
        npolish = 4
        for pi in range(npolish):
            t_left = deadline - time.monotonic() - 20
            if t_left < 15:
                break
            t_alloc = min(t_left / (npolish - pi) * 0.7, 120.0)
            if t_alloc < 10:
                break

            if pi == 0:
                a0 = best_sequence.copy()
            else:
                scale = 0.0015 * (pi + 1)
                noise = rng.standard_normal(len(best_sequence)) * scale
                a0 = np.maximum(best_sequence + noise, 0)
                s = a0.sum()
                a0 = a0 / s if s > 1e-15 else best_sequence.copy()

            # Short gradient then LP
            t_g = min(5.0, t_alloc * 0.25)
            t_l = t_alloc - t_g
            if t_g > 2:
                a0, _ = gradient_optimize(a0, time_limit=t_g, lr_init=0.005,
                                          top_k=80, temp_init=0.001)
            if t_l > 5:
                a_opt, obj = lp_optimize(a0, time_limit=t_l)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Polish {pi}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Polish {pi}: {score:.6f}")

        # Coordinate descent fine-tuning
        t_left = deadline - time.monotonic() - 10
        t_cd = min(30.0, t_left * 0.5)
        if t_cd > 5:
            a_cd, obj_cd = coordinate_descent(best_sequence, time_limit=t_cd, rng=rng)
            score = float(evaluate_sequence(a_cd.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_cd.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

        # Final micro-gradient
        t_left = deadline - time.monotonic() - 5
        if t_left > 5:
            a_opt, _ = gradient_optimize(best_sequence, time_limit=t_left,
                                         lr_init=0.002, momentum=0.96,
                                         top_k=100, temp_init=0.0005)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  FinalGrad: {score:.6f} *NEW BEST*")

    print(f"Phase 3 done ({time.monotonic() - t3:.0f}s). Best: {best_score:.6f}")

    # Fallback
    if best_sequence is None or not np.isfinite(best_score):
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
