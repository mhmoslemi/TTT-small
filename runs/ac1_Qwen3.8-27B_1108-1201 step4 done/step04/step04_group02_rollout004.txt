            conv = fast_convolve(f)
            v = np.zeros(n)
            cur_lr = lr
            no_improve = 0

    return best_f, best_obj


def resample_to_length(f, new_n):
    x_old = np.linspace(0, 1, len(f))
    x_new = np.linspace(0, 1, new_n)
    f_new = np.interp(x_new, x_old, f)
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return np.ones(new_n) / new_n
    return f_new / s


def generate_structured_inits(n, rng):
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Decreasing ramp
    r = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(r / r.sum())

    # Increasing ramp
    r2 = np.arange(1, n + 1, dtype=np.float64)
    inits.append(r2 / r2.sum())

    # Cosine window
    c = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 4 + 0.02
    inits.append(c / c.sum())

    # Sine
    s = np.sin(np.pi * (x + 0.5) / n) + 0.05
    inits.append(s / s.sum())

    # Exponential (from left)
    e = np.exp(-3.0 * x / n)
    inits.append(e / e.sum())

    # Two-peaked
    t = x / n
    two = np.exp(-20 * (t - 0.25) ** 2) + np.exp(-20 * (t - 0.75) ** 2) + 0.01
    inits.append(two / two.sum())

    # Middle-heavy
    mid = np.exp(-5.0 * (t - 0.5) ** 2) + 0.05
    inits.append(mid / mid.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function. Uses accelerated smoothed gradient, FW-LP,
    and exponentiated gradient in alternating fashion with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})", flush=True)
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}", flush=True)
    except Exception:
        pass

    # === Phase 0: Quick parent evaluation ===
    print("=== Phase 0: Quick Evaluation ===", flush=True)
    if parent is not None:
        try_seq(parent, "parent_raw")

    # === Phase 1: Short diversity search (30s) ===
    print("\n=== Phase 1: Diversity Search ===", flush=True)
    t1_start = time.monotonic()
    t1_limit = min(30.0, deadline - time.monotonic() - 1400.0)

    candidate_lengths = [384, 256, 512]
    if parent is not None:
        parent_n = len(parent)
        if parent_n not in candidate_lengths:
            candidate_lengths.insert(0, parent_n)

    for n in candidate_lengths:
        rem = deadline - time.monotonic() - 1350
        if rem < 10:
            break
        t_alloc = min(rem, t1_limit / len(candidate_lengths) + 5)
        if t_alloc < 5:
            break

        if parent is not None and n == len(parent):
            inits = [parent]
        else:
            if parent is not None:
                inits = [resample_to_length(parent, n)]
            else:
                inits = [np.ones(n) / n]

        for f_init in inits:
            if time.monotonic() > deadline - 1350:
                break
            a_opt, obj = fw_lp_refine(f_init, time_limit=min(t_alloc, 15),
                                      m_factor=1.0, lazy_k=n)
            try_seq(a_opt, f"div_n{n}")

    print(f"  After Phase 1: best={best_score:.8f}", flush=True)

    # === Phase 2: Main optimization (bulk of time) ===
    print("\n=== Phase 2: Main Optimization ===", flush=True)

    if best_seq is None:
        n_main = 384
        best_f = np.ones(n_main) / n_main
    else:
        n_main = len(best_seq)
        best_f = best_seq / best_seq.sum()

    # Run accelerated smoothed method on the best
    rem = deadline - time.monotonic() - 200
    if rem > 60:
        a_opt, obj = accelerated_smoothed_method(
            best_f, time_limit=min(rem, 600),
            tau_init=0.003, tau_min=0.00005,
            lr_init=0.002, top_k=max(32, n_main // 8),
            lp_every=30, lazy_k=min(150, 2 * n_main - 1),
            eg_every=50)
        try_seq(a_opt, "accel_main")

    # Continue with another round from the updated best
    if best_seq is not None:
        best_f = best_seq / best_seq.sum()
        rem = deadline - time.monotonic() - 150
        if rem > 30:
            a_opt, obj = accelerated_smoothed_method(
                best_f, time_limit=min(rem, 400),
                tau_init=0.001, tau_min=0.00002,
                lr_init=0.001, top_k=max(32, n_main // 6),
                lp_every=20, lazy_k=min(100, 2 * n_main - 1),
                eg_every=40)
            try_seq(a_opt, "accel_main2")

    # FW-LP deep runs
    if best_seq is not None:
        for p in range(3):
            rem = deadline - time.monotonic() - 100
            if rem < 20:
                break

            if p == 0:
                f_init = best_seq / best_seq.sum()
            elif p == 1:
                f_init = project_simplex(best_f + rng.standard_normal(n_main) * 0.003)
            else:
                f_init = project_simplex(0.9 * best_f + 0.1 * (best_f[::-1]))

            t_alloc = min(rem // (3 - p), 120)
            a_opt, obj = fw_lp_refine(f_init, time_limit=t_alloc,
                                      m_factor=1.0, lazy_k=min(120, 2 * n_main - 1))
            try_seq(a_opt, f"fw_deep_{p}")

    print(f"  After Phase 2: best={best_score:.8f}", flush=True)

    # === Phase 3: Final polishing (remaining time) ===
    print("\n=== Phase 3: Final Polish ===", flush=True)

    if best_seq is not None:
        n_best = len(best_seq)
        best_f = best_seq / best_seq.sum()

        # Micro subgradient polish
        rem = deadline - time.monotonic() - 20
        if rem > 10:
            a_opt, obj = micro_polish(best_f, time_limit=min(rem, 60),
                                      lr=0.0003, top_k=max(40, n_best // 4))
            try_seq(a_opt, "micro1")

        # FW polish with very tight M
        if best_seq is not None:
            rem = deadline - time.monotonic() - 10
            if rem > 8:
                a_opt, obj = fw_lp_refine(best_seq / best_seq.sum(),
                                          time_limit=min(rem, 40),
                                          m_factor=1.0, lazy_k=min(80, 2 * n_best - 1))
                try_seq(a_opt, "fw_polish")

        # Final micro perturbations
        if best_seq is not None:
            rem = deadline - time.monotonic() - 5
            if rem > 5:
                for _ in range(5):
                    rem = deadline - time.monotonic() - 3
                    if rem < 3:
                        break
                    f_pert = project_simplex(best_f + rng.standard_normal(n_best) * 0.001)
                    a_opt, obj = micro_polish(f_pert, time_limit=min(rem, 8),
                                              lr=0.0002, top_k=max(30, n_best // 5))
                    try_seq(a_opt, "micro_pert")

    # Fallback
    if best_seq is None:
        print("  WARNING: No valid sequence found, using fallback", flush=True)
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})", flush=True)
    return [float(x) for x in best_seq]
