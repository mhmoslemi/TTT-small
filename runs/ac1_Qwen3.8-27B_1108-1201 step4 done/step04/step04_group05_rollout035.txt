            if m_mult > 5.0:
                m_mult = 1.0
                g = solve_lp_topk(f, M * 1.5, K_lp)
                if g is None:
                    continue
            else:
                continue
        
        gs = g.sum()
        if gs < 1e-15:
            g = np.ones(n) / n
        else:
            g = g / gs
        
        # Fixed point check
        if np.max(np.abs(g - f)) < 5e-12:
            no_improve += 1
            if no_improve > 30:
                break
            m_mult = min(m_mult * 1.03, 2.5)
            continue
        
        # Line search
        best_t, best_val = line_search_parabolic(f, g, M, max_evals=25)
        
        if best_t > 1e-8:
            f = (1 - best_t) * f + best_t * g
            f = np.maximum(f, 0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
                conv = fast_convolve(f)
                M = conv.max()
            else:
                f = f / fs
                conv = fast_convolve(f)
                M = conv.max()
            
            if M < best_obj - 1e-13:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                m_mult = max(0.95, m_mult * 0.97)
                if verbose and it % 20 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}, M={M:.8f}")
            else:
                no_improve += 1
                m_mult = min(m_mult * 1.02, 2.0)
        else:
            no_improve += 1
            m_mult = min(m_mult * 1.05, 3.0)
        
        # Restart if stuck
        if no_improve > 40:
            strategies = [
                "noise_small", "noise_med", "mirror", "gaussian"
            ]
            strat = strategies[no_improve % 4]
            if strat == "noise_small":
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            elif strat == "noise_med":
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.01)
            elif strat == "mirror":
                f = project_simplex(0.8 * best_f + 0.2 * best_f[::-1])
            else:
                g_center = n / 2.0
                scale = n * 0.15
                gauss = np.exp(-0.5 * ((np.arange(n) - g_center) / scale) ** 2)
                f = project_simplex(0.7 * best_f + 0.3 * gauss / gauss.sum())
            
            conv = fast_convolve(f)
            M = conv.max()
            m_mult = 1.0
            no_improve = 0
    
    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr_init=0.004, mom=0.93, top_k=15, temp=0.0001):
    """Subgradient descent on simplex with softmax peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp
    rng_local = np.random.default_rng(9999)
    
    while time.monotonic() - t0 < time_limit:
        # Get peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
            pv = conv
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
            pv = conv[peaks]
        
        # Softmax weights
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            sg[js:je+1] += w[i] * f[k-je:k-js+1][::-1]
        
        # Update
        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.9995
        cur_temp *= 1.003
        
        if no_improve > 200:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.3, 0.0008)
            no_improve = 0
    
    return best_f, best_obj


def targeted_peak_reduction(f0, time_limit, n_iter_max=None):
    """Directly target the highest peak: reduce mass at contributing positions."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng_local = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]
        
        # Find which positions contribute most to this peak
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        contribs = f[peak_k - je:peak_k - js + 1][::-1] * f[js:je+1]
        
        # Try moving mass from high-contributing to low-contributing positions
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break
        
        # Target: reduce the top contributor
        local_order = np.argsort(contribs)[::-1]
        move_from = js + local_order[0]
        
        # Find a position to move mass to (one that doesn't contribute to the peak)
        peak_set = set(range(js, je+1))
        candidates = [j for j in range(n) if j not in peak_set and f[j] > 0]
        if not candidates:
            candidates = [j for j in range(n) if j != move_from and f[j] > 1e-10]
        if not candidates:
            break
        
        # Pick candidate that minimizes contribution to peak
        cand_arr = np.array(candidates)
        best_cand = cand_arr[0]
        best_gain = -1
        
        for c in cand_arr[:min(20, len(cand_arr))]:
            # Estimate effect: moving epsilon from move_from to c
            # Reduces peak by ~f[peak_k-move_from]*eps (if move_from contributes)
            # But might increase other peaks
            old_contrib = f[peak_k - move_from] * f[move_from] if (0 <= peak_k - move_from < n) else 0
            gain = old_contrib
            if gain > best_gain:
                best_gain = gain
                best_cand = c
        
        # Try several step sizes
        improved = False
        for frac in [0.02, 0.05, 0.1, 0.2]:
            delta = f[move_from] * frac
            if delta < 1e-12:
                continue
            f_new = f.copy()
            f_new[move_from] -= delta
            f_new[int(best_cand)] += delta
            f_new = np.maximum(f_new, 0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            val = fast_convolve(f_new).max()
            if val < best_obj - 1e-14:
                f = f_new
                best_obj = val
                best_f = f.copy()
                improved = True
                break
        
        if not improved:
            # Try direct noise perturbation
            noise = rng_local.standard_normal(n) * 0.005
            f_new = project_simplex(f + noise)
            val = fast_convolve(f_new).max()
            if val < best_obj - 1e-14:
                f = f_new
                best_obj = val
                best_f = f.copy()
            else:
                break
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit):
    """Cycle through coordinates trying pair-swap moves."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng_local = np.random.default_rng(424242)
    
    while time.monotonic() - t0 < time_limit:
        improved_this_round = False
        order = rng_local.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 2:
                break
            
            # Try moving mass from j to a few candidates
            for source in [(j + 1) % n, (j - 1) % n, j ^ (n // 2), int(rng_local.integers(n))]:
                source = int(source) % n
                if source == j or f[source] < 1e-12:
                    continue
                
                for frac in [0.03, 0.08, 0.15]:
                    delta = f[source] * frac
                    if delta < 1e-14:
                        continue
                    f_new = f.copy()
                    f_new[j] += delta
                    f_new[source] -= delta
                    if f_new[source] < 0:
                        f_new[j] += f_new[source]
                        f_new[source] = 0.0
                    f_new = np.maximum(f_new, 0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-15:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
                        improved_this_round = True
                        break
                if improved_this_round:
                    break
        
        if not improved_this_round:
            break
    
    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.4))
    
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Asymmetric ramp
    ramp = (x + 1) ** 1.5
    inits.append(ramp / ramp.sum())
    
    # Two-peak
    peak = (np.exp(-((x - n * 0.3) ** 2) / (2 * (n * 0.1) ** 2)) +
            np.exp(-((x - n * 0.7) ** 2) / (2 * (n * 0.1) ** 2)))
    inits.append(peak / peak.sum())
    
    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    # Power law
    pl = (x + 1.0) ** (-0.3)
    inits.append(pl / pl.sum())
    
    # Random smooth (low-pass filtered noise)
    noise = rng.standard_normal(n)
    kernel_size = max(3, n // 20)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(noise, kernel, mode='same')
    smooth = np.maximum(smooth, 0.01)
    inits.append(smooth / smooth.sum())
    
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def try_seq(a, label, state):
    """Evaluate and update state."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry: multi-phase search combining LP-FW, subgradient, 
    targeted peak reduction, and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    
    state = {'best_seq': None, 'best_score': float('inf')}
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.8f}")
                state['best_score'] = min(state['best_score'], s)
                state['best_seq'] = parent.copy()
    except Exception:
        pass
    
    if parent is None:
        parent = np.ones(384) / 384
        print("No parent found, using uniform 384")
    
    # === Phase 1: Exploration ===
    print("\n=== Phase 1: Exploration ===")
    phase1_end = time.monotonic() + float(budget_s) * 0.22
    
    # Quick run on parent
    n_p = len(parent)
    t1 = min(120, phase1_end - time.monotonic() - 60)
    if t1 > 20:
        print(f"  FW on parent n={n_p}, t={t1:.0f}s")
        a_opt, obj = fw_optimize(parent, time_limit=t1, K_lp=25)
        try_seq(a_opt, "parent_fw", state)
    
    # Try nearby lengths
    base_n = len(state['best_seq']) if state['best_seq'] is not None else n_p
    lengths_to_try = sorted(set([
        base_n - 16, base_n - 8, base_n, base_n + 8, base_n + 16,
        base_n + 32, 512, 448
    ]), key=lambda x: abs(x - base_n))
    lengths_to_try = [l for l in lengths_to_try if 256 <= l <= 640]
    
    for n_test in lengths_to_try[:5]:
        if time.monotonic() > phase1_end - 45:
            break
        
        best_seq = state['best_seq']
        inits = generate_inits(n_test, best_seq, rng)
        
        # Try 2-3 initializations
        n_trials = min(3, len(inits))
        t_per = max(15, (phase1_end - time.monotonic() - 30) / (n_trials * max(1, 4)))
        t_per = min(t_per, 45)
        
        for idx in range(n_trials):
            if time.monotonic() > phase1_end - 25:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 15)
            if t_alloc < 12:
                break
            f_init = inits[idx]
            a_opt, obj = fw_optimize(f_init, time_limit=t_alloc, K_lp=20)
            try_seq(a_opt, f"exp_n{n_test}_i{idx}", state)
    
    print(f"  Phase 1 best: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    
    # === Phase 2: Deep Optimization ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + float(budget_s) * 0.45
    
    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Optimizing n={n_best}")
        
        # Main deep run
        t_main = min(300, (phase2_end - time.monotonic()) * 0.4)
        if t_main > 30:
            print(f"  Deep FW: t={t_main:.0f}s")
            a_opt, obj = fw_optimize(state['best_seq'], time_limit=t_main, K_lp=35, verbose=True)
            try_seq(a_opt, "deep_fw", state)
        
        # Perturbation runs
        for ps in [0.003, 0.008, 0.02, 0.05]:
            rem = phase2_end - time.monotonic() - 25
            if rem < 20:
                break
            t_alloc = min(rem * 0.25, 100)
            if t_alloc < 15:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            a_opt, obj = fw_optimize(f_init, time_limit=t_alloc, K_lp=30)
            try_seq(a_opt, f"pert_{ps}", state)
        
        # Mirror
        rem = phase2_end - time.monotonic() - 20
        if rem > 25:
            f_mirror = project_simplex(0.8 * state['best_seq'] + 0.2 * state['best_seq'][::-1])
            a_opt, obj = fw_optimize(f_mirror, time_limit=min(rem, 80), K_lp=30)
            try_seq(a_opt, "mirror", state)
        
        # Full LP run (slower but more accurate)
        rem = phase2_end - time.monotonic() - 15
        if rem > 30:
            print(f"  Full LP run: t={min(rem, 120):.0f}s")
            a_opt, obj = fw_optimize(state['best_seq'], time_limit=min(rem, 120),
                                     use_full_lp=True, K_lp=50)
            try_seq(a_opt, "full_lp", state)
    
    print(f"  Phase 2 best: {state['best_score']:.8f}")
    
    # === Phase 3: Polishing ===
    print(f"\n=== Phase 3: Polishing ===")
    phase3_end = deadline - 15
    
    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        
        # Targeted peak reduction
        rem = phase3_end - time.monotonic() - 60
        if rem > 20:
            a_opt, obj = targeted_peak_reduction(state['best_seq'], time_limit=min(rem, 100))
            try_seq(a_opt, "target_peak", state)
        
        # Subgradient (medium)
        rem = phase3_end - time.monotonic() - 40
        if rem > 15:
            a_opt, obj = subgradient_polish(state['best_seq'], time_limit=min(rem, 100),
                                            lr_init=0.002, mom=0.94,
                                            top_k=max(8, n_best // 8), temp=0.00005)
            try_seq(a_opt, "sg_med", state)
        
        # Coordinate descent
        rem = phase3_end - time.monotonic() - 25
        if rem > 12:
            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=min(rem, 80))
            try_seq(a_opt, "cd", state)
        
        # Fine subgradient
        rem = phase3_end - time.monotonic() - 15
        if rem > 12:
            a_opt, obj = subgradient_polish(state['best_seq'], time_limit=min(rem, 60),
                                            lr_init=0.0005, mom=0.96,
                                            top_k=max(5, n_best // 10), temp=0.00002)
            try_seq(a_opt, "sg_fine", state)
        
        # Final FW squeeze
        rem = phase3_end - time.monotonic()
        if rem > 10:
            a_opt, obj = fw_optimize(state['best_seq'], time_limit=min(rem, 50), K_lp=40)
            try_seq(a_opt, "final_fw", state)
        
        # One more CD pass
        rem = phase3_end - time.monotonic()
        if rem > 8:
            a_opt, obj = coordinate_descent(state['best_seq'], time_limit=min(rem, 30))
            try_seq(a_opt, "cd_final", state)
    
    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback: {state['best_score']:.6f}")
    
    # Final verification
    final_score = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(final_score):
        state['best_score'] = final_score
    
    print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
