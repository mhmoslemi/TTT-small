a >= 0, sum(a) = 1

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    """Symmetrize a sequence."""
    return (a + a[::-1]) / 2.0

    n = len(a)

    js = np.arange(n)
    kj = k - js
    valid = (kj >= 0) & (kj < n)
    g[valid] = 2.0 * a[kj[valid]]
    return g

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Symmetric Frank-Wolfe with LP oracle (exploits optimal symmetry)
2. Nesterov accelerated projected subgradient
3. Coordinate descent with 1D line search
4. Multi-length exploration (n=384, 512, 768, 1024)
Key insight: optimal solution is WLOG symmetric (a[j]=a[n-1-j]), halving the search space.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def symmetrize(a):
    return (a + a[::-1]) / 2.0


def compute_subgrad(a, k):
    n = len(a)
    g = np.zeros(n)
    js = np.arange(n)
    kj = k - js
    valid = (kj >= 0) & (kj < n)
    g[valid] = 2.0 * a[kj[valid]]
    return g


def build_lp_matrix_symmetric(f, n):
    m = n // 2
    ncon = n
    A = np.zeros((ncon, m))
    for k in range(ncon):
        for t in range(m):
            coeff = 0.0
            if t <= k:
                coeff += f[k - t]
            idx2 = k - n + 1 + t
            if 0 <= idx2 < n and t >= n - 1 - k:
                coeff += f[idx2]
            A[k, t] = coeff
    return A, m


def solve_lp_oracle_symmetric(f, M):
    n = len(f)
    if n % 2 == 1:
        n_solve = n + 1
        f_ext = np.concatenate([f, [0.0]])
        A, m = build_lp_matrix_symmetric(f_ext, n_solve)
        c = -np.ones(m)
        b_ub = np.full(n_solve, M)
        bounds = [(0, None)] * m
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None
        x = np.maximum(res.x, 0.0)
        g = np.zeros(n)
        for j in range(n // 2):
            g[j] = x[j]
            g[n - 1 - j] = x[j]
        if n % 2 == 1:
            g[n // 2] = x[n // 2]
        return g
    else:
        m = n // 2
        A, m2 = build_lp_matrix_symmetric(f, n)
        c = -np.ones(m2)
        b_ub = np.full(n, M)
        bounds = [(0, None)] * m2
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            return None
        x = np.maximum(res.x, 0.0)
        g = np.zeros(n)
        for j in range(m2):
            g[j] = x[j]
            g[n - 1 - j] = x[j]
        return g


def solve_lp_oracle_general(f, M):
    n = len(f)
    m = n
    ncon = 2 * n - 1
    A = np.zeros((ncon, n))
    for j in range(n):
        for k in range(n):
            A[j + k, j] = f[k]
    c = -np.ones(n)
    b_ub = np.full(ncon, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return np.maximum(res.x, 0.0)

def nesterov_sg(f0, time_limit, top_k=None, verbose=False):
    f = symmetrize(np.maximum(f0.astype(np.float64), 0.0))
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if top_k is None:
        top_k = max(5, n // 8)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    y = f.copy()
    prev_f = f.copy()
    t_nest = 1.0
    lr_scale = 1.0

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 0.5:
            break

        conv_y = fast_convolve(y)
        idx = np.argpartition(conv_y, -top_k)[-top_k:]
        pv = conv_y[idx]
        order = np.argsort(pv)[::-1]
        weights = np.zeros(len(idx))
        for rank, oi in enumerate(order):
            weights[oi] = 1.0 / (rank + 1) ** 1.2
        weights /= weights.sum()

        sg = np.zeros(n)
        for ii in range(len(idx)):
            if weights[ii] < 1e-14:
                continue
            k = int(idx[ii])
            g_k = compute_subgrad(y, k)
            sg += weights[ii] * g_k

        t_nest += 1.0
        momentum = (t_nest - 1.0) / (t_nest + 1.0)
        lr = lr_scale / (t_nest ** 0.5)

        y_new = y - momentum * (f - y)
        y_new = y_new - lr * sg
        f = project_simplex(y_new)
        f = symmetrize(f)
        f = f / f.sum()
        y = f.copy()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            seed_val = (int(time.monotonic() * 10000) + it) % (2**31)
            rng = np.random.default_rng(seed_val)
            f = symmetrize(project_simplex(best_f + rng.standard_normal(n) * 0.003))
            f = f / f.sum()
            y = f.copy()
            t_nest = 1.0
            lr_scale = max(lr_scale * 0.7, 0.1)
            no_improve = 0

        if verbose and it % 5000 == 0:
            print(f"    Nest iter {it}, obj={obj:.8f}, best={best_obj:.8f}")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_trials=15, verbose=False):
    f = symmetrize(np.maximum(f0.astype(np.float64), 0.0))
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    total_passes = 0
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        obj_before = best_obj
        total_passes += 1

        m = n // 2
        order = np.random.permutation(m) if total_passes > 1 else np.arange(m)

        for j_idx in order:
            j = int(j_idx)
            fj = f[j]
            if fj < 1e-15:
                candidates = [f.sum() / n * 0.5, f.sum() / n, f.sum() / n * 1.5]
            else:
                base = fj
                candidates = [base * (1.0 + d) for d in
                              [-0.5, -0.3, -0.2, -0.1, -0.05, 0.0, 0.05, 0.1, 0.2, 0.3, 0.5]]
                candidates = [max(0.0, c) for c in candidates]

            best_x = fj
            best_local = conv.max()

            for x in candidates:
                f[j] = x
                f[n - 1 - j] = x
                fs = f.sum()
                if fs < 1e-15:
                    f[j] = fj
                    f[n - 1 - j] = fj
                    continue
                f_norm = f / fs
                val = fast_convolve(f_norm).max()
                if val < best_local:
                    best_local = val
                    best_x = x

            f[j] = best_x
            f[n - 1 - j] = best_x
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            f = symmetrize(f)

        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_f = f.copy()
            no_improve_passes = 0
        else:
            no_improve_passes += 1

        if no_improve_passes > 15:
            seed_val = (int(time.monotonic() * 10000) + total_passes * 31) % (2**31)
            rng = np.random.default_rng(seed_val)
            scale = 0.001 + 0.002 * min(1.0, no_improve_passes / 30.0)
            f = symmetrize(project_simplex(best_f + rng.standard_normal(n) * scale))
            f = f / f.sum()
            no_improve_passes = 0

        if verbose and total_passes % 5 == 0:
            print(f"    CD pass {total_passes}, obj={best_obj:.8f}")

    return best_f, best_obj
