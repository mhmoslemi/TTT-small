def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])

- `x = 0`
- For each lag `k` in `[i, i+n-1]` with `k != 2i` and `a[k-i] > 0`:

- `total_max[j] = max(base_max_outside, max_inside[j])`
- `obj[j] = 2*n*total_max[j] / (S + X[j])^2`

This is `O(m * l)` = `O(n^2)` per coordinate, vectorizable with numpy.

combined with Nesterov-accelerated subgradient and basin hopping.

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def obj_val(a):
    """Objective assuming sum(a)=1: 2*n*max(a*a)."""
    return 2.0 * len(a) * fast_convolve(a).max()


def coordinate_descent(a, time_limit=60.0, verbose=False):
    """Exact coordinate descent. a should be normalized (sum=1)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n) / n

    current_obj = obj_val(a)
    best_a = a.copy()
    best_obj = current_obj
    t0 = time.monotonic()
    pass_num = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        if no_improve > 5:
            break
        pass_num += 1
        improved = False
        
        for i in range(n):
            # Remove coordinate i
            old_val = a[i]
            a[i] = 0.0
            base = fast_convolve(a)
            S = a.sum()
            
            # Relevant lags: k in [i, min(i+n-1, 2n-2)]
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)

            other_idx = lags - i  # = 0, 1, ..., len(lags)-1

            # Find which lag is the "quadratic" one (k = 2i)
            quad_mask = (lags == 2 * i)
            
            # Precompute max of base outside relevant lags
            if k_min > 0 or k_max < 2 * n - 2:
                outside = np.ones(2 * n - 1, dtype=bool)
                outside[k_min:k_max+1] = False
                base_max_out = np.max(base[outside]) if outside.any() else 0.0
            else:
                base_max_out = 0.0
            
            # Candidate x values
            candidates = [0.0]

            valid_linear = (~quad_mask) & (coeffs > 1e-15)

    n = len(a)
    f = np.maximum(a.astype(np.float64), 0.0)
    s = f.sum()

        f = np.ones(n) / n

        f = f / s
    
    y = f.copy()
    v = np.zeros(n)
    best_f = f.copy()
    best_obj = obj_val(f)
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(y)
        
        # Get top-k peaks
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Subgradient
        sg = np.zeros(n)
        for idx in range(len(peaks)):
            if w[idx] < 1e-13:
                continue
            k = int(peaks[idx])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx_range = np.arange(js, je + 1)
            sg[idx_range] += 2.0 * w[idx] * y[k - idx_range]
        
        # Nesterov update
        v = momentum * v - cur_lr * sg
        f_new = y + v
        f_new = project_simplex(f_new)
        
        # Nesterov: update y as convex combination
        t_it = 2.0 / (it + 1 + momentum)
        y = (1 - t_it) * y + t_it * f_new
        y = project_simplex(y)
        
        obj = obj_val(y)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        cur_lr *= 0.999
        cur_temp *= 1.005
        
        if no_improve > 200:
            rng = np.random.default_rng((it * 7 + 3) % 100000)
            f_new = project_simplex(best_f + rng.standard_normal(n) * 0.005)
            y = f_new.copy()
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)

    if s < 1e-15:
        best_f = np.ones(n) / n

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
