"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using improved gradient-based methods
with backtracking line search, Nesterov momentum, targeted peak reduction,
and enhanced coordinate descent with swap operations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_gradient(f, conv, top_k=5, temp=0.001):
    """Compute smoothed gradient of max(conv) using softmax weighting over peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    
    return sg


def backtracking_gradient_descent(f0, time_limit, top_k=10, temp=0.001, 
                                    max_iter=10000, use_momentum=True):
    """Backtracking line search gradient descent on the simplex."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    lr = 0.05
    min_lr = 1e-6
    max_backtrack = 20
    c1 = 1e-4  # Armijo condition
    no_improve = 0
    rng_local = np.random.default_rng(12345)
    
    v = np.zeros(n)  # momentum
    momentum = 0.9
    
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 200:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = max(0.01, min_lr * 10)
            no_improve = 0
            continue
        
        # Compute gradient
        grad = compute_gradient(f, conv, top_k=top_k, temp=temp)
        
        # Nesterov momentum
        if use_momentum:
            v = momentum * v - lr * grad
            step = v
        else:
            step = -lr * grad
        
        # Backtracking line search
        accepted = False
        for bt in range(max_backtrack):
            alpha = 1.0 / (2.0 ** bt)
            f_new = project_simplex(f + alpha * step)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()
            
            if obj_new <= obj + c1 * alpha * np.dot(step, grad):
                f = f_new
                accepted = True
                break
            elif obj_new < obj:
                # Accept even if Armijo not satisfied (practical)
                f = f_new
                accepted = True
                break
        
        if not accepted:
            lr = max(lr * 0.5, min_lr)
            if lr <= min_lr:
                # Try random restart
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.001)
                v = np.zeros(n)
                lr = 0.01
        
        # Adapt top_k and temp
        if it % 100 == 0 and it > 0:
            top_k = min(20, top_k + 1)
            temp = max(0.0001, temp * 0.95)
    
    return best_f, best_obj


def peak_reduction(f0, time_limit, n_iters=5000):
    """Targeted peak reduction: directly reduce the maximum of the convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng_local = np.random.default_rng(54321)
    no_improve = 0
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.002)
            no_improve = 0
            continue
        
        # Find the peak position
        k_star = int(np.argmax(conv))
        
        # Find positions that contribute to this peak
        js = max(0, k_star - n + 1)
        je = min(n - 1, k_star)
        contributors = np.arange(js, je + 1)
        
        # The contribution of position i to the peak is f[i] * f[k_star - i]
        contributions = f[contributors] * f[k_star - contributors]
        total_contrib = contributions.sum()
        
        if total_contrib < 1e-15:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.001)
            continue
        
        # Find positions NOT contributing to the peak (have zero contribution)
        # These are positions i where k_star - i is out of range for the window
        # or where f[k_star - i] = 0
        all_positions = np.arange(n)
        non_contributors = np.setdiff1d(all_positions, contributors)
        
        # Also consider positions where f is very small
        non_contrib = non_contributors[f[non_contributors] < 0.01 * f.max()]
        if len(non_contrib) == 0:
            non_contrib = non_contributors
        
        if len(non_contrib) == 0:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.001)
            continue
        
        # Move mass from high-contributing positions to non-contributing positions
        # Pick the top contributors
        top_contrib_idx = np.argsort(contributions)[-min(5, len(contributors)):]
        top_contrib_positions = contributors[top_contrib_idx]
        
        # Pick target positions (non-contributors with least mass)
        n_targets = min(3, len(non_contrib))
        targets = non_contrib[np.argsort(f[non_contrib])[:n_targets]]
        
        # Move a small fraction of mass
        move_frac = 0.02 + 0.01 * rng_local.random()
        
        total_move = 0.0
        f_new = f.copy()
        
        for pos in top_contrib_positions:
            move = f_new[pos] * move_frac
            f_new[pos] -= move
            total_move += move
        
        # Distribute to targets
        if total_move > 0 and len(targets) > 0:
            per_target = total_move / len(targets)
            for t in targets:
                f_new[t] += per_target
        
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            continue
        f_new = f_new / sn
        
        # Check if this helps (consider all peaks, not just k_star)
        conv_new = fast_convolve(f_new)
        if conv_new.max() < obj - 1e-15:
            f = f_new
        else:
            # Try smaller move
            f_try = f.copy()
            for pos in top_contrib_positions[:2]:
                move = f_try[pos] * move_frac * 0.3
                f_try[pos] -= move
                total_move = move
            if total_move > 0 and len(targets) > 0:
                f_try[int(targets[0])] += total_move
            f_try = np.maximum(f_try, 0.0)
            f_try = f_try / f_try.sum()
            conv_try = fast_convolve(f_try)
            if conv_try.max() < obj - 1e-15:
                f = f_try
    
    return best_f, best_obj

    fractions = [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5]
    
    for round_idx in range(n_rounds):
        if time.monotonic() - t0 > time_limit - 10:
            break
        
        improved = False
        order = rng_local.permutation(n)
        
        for j in order:
            if time.monotonic() - t0 > time_limit - 5:
                    break

            candidates = set()

            candidates.add((j - 1) % n)
            candidates.add(j ^ (n // 3))
            candidates.add(j ^ (n // 2))
            candidates.add(rng_local.integers(n))
            candidates.add(rng_local.integers(n))
            
            # Also try symmetric position
            candidates.add((n - 1 - j) % n)
            
            for source in candidates:
                source = int(source) % n

                    continue
                
                for frac in fractions:

                        continue

                    f_new[i1] += delta
                    f_new[i2] -= delta
                    if f_new[i2] < 0:
                        f_new[i1] += f_new[i2]

                    f_new = np.maximum(f_new, 0.0)

                    val = fast_convolve(f_new).max()

    best_f = f.copy()

    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break
        
        g = solve_lp(f, M * m_exp)

            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        diff = np.max(np.abs(g - f))
        if diff < 1e-10:

            m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine
        if 0.05 < best_alpha < 0.95:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(6):

                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)
        
        if no_improve > 40:
            strategies = ["gaussian", "mirror", "random"]
            strat = strategies[no_improve % len(strategies)]
            if strat == "gaussian":
                f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)

                f = project_simplex(rng_local.dirichlet(np.ones(n) * 0.5))

            M = conv.max()
            m_exp = 1.0

    return best_f, best_obj


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.8))
    
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian center

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smoothed_gradient(f, conv, top_k, temp):
    """Compute smoothed gradient using softmax weighting over top peaks."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-13:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]

    return sg

    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    rng_local = np.random.default_rng(999 + n)

    while time.monotonic() - t0 < time_limit - 2:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 2:
            break

        g = solve_lp(f, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-10:
            no_improve += 1
            if no_improve > 25:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if 0.1 < best_alpha < 0.9:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(5):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                ft = ft / ft.sum()
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            M = fast_convolve(f).max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)

        if no_improve > 35:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.003)
            M = fast_convolve(f).max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.8))

    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    ham = 0.54 - 0.46 * np.cos(2 * np.pi * x / max(1, n - 1))
    inits.append(ham / ham.sum())

    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())

    peak3 = (np.exp(-((x - n / 6.0) ** 2) / (2.0 * (n / 12.0) ** 2)) +
             np.exp(-((x - n / 2.0) ** 2) / (2.0 * (n / 12.0) ** 2)) +
             np.exp(-((x - 5.0 * n / 6.0) ** 2) / (2.0 * (n / 12.0) ** 2)))
    inits.append(peak3 / peak3.sum())

    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.3
    inits.append(alt / alt.sum())

    # Block pattern
    block = np.zeros(n)
    block[::3] = 1.0
    block[1::3] = 0.4
    inits.append(block / block.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def try_sequence(a, label, state):
    """Evaluate and update best if improved."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def palindromic_fold(a):
    """Fold a sequence to its palindromic part (first half)."""
    n = len(a)
    half = (n + 1) // 2
    return a[:half]


def palindromic_unfold(half, n):
    """Unfold a half-sequence to full palindromic sequence."""
    a = np.zeros(n)
    a[:len(half)] = half[:n]
    a[n - 1::-1] = np.minimum(a[:n], a[n - 1::-1]) if n > 1 else a
    # Proper unfold
    a = np.zeros(n)
    for i in range(len(half)):
        a[i] = half[i]
        if n - 1 - i != i:
            a[n - 1 - i] = half[i]
    return a


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-stage optimization combining adaptive-temperature gradient descent,
    targeted peak reduction, swap descent, and FW-LP. Uses structured
    exploration, palindromic symmetrization, and deep polishing.
    """
    rng = np.random.default_rng(seed)
    total_budget = max(180.0, float(budget_s))
    deadline = time.monotonic() + total_budget - 45.0

    state = {'best_seq': None, 'best_score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.6f}")
                state['best_score'] = s
                state['best_seq'] = parent.copy()
    except Exception:
        pass

    # === Phase 1: Exploration (20% of time) ===
    print("\n=== Phase 1: Exploration ===")
    phase1_time = total_budget * 0.20
    phase1_end = time.monotonic() + phase1_time

    if parent is not None:
        t_alloc = min(50, phase1_end - time.monotonic() - 50)
        if t_alloc > 20:
            print(f"  Adaptive GD on parent (n={len(parent)}), t={t_alloc:.0f}s")
            a_opt, obj = adaptive_gradient_descent(parent, time_limit=t_alloc,
                                                    top_k_init=8, temp_init=0.005)
            try_sequence(a_opt, "parent_adapt", state)

    # Explore different lengths
    lengths = [320, 352, 384, 400, 416]
    if parent is not None:
        if len(parent) in lengths:
            lengths.remove(len(parent))
        lengths.insert(0, len(parent))

    for n in lengths:
        if time.monotonic() > phase1_end - 45:
            break

        inits = generate_inits(n, parent, rng)
        n_try = min(2, len(inits))
        remaining_lengths = len(lengths) - lengths.index(n)
        t_per = max(12, (phase1_end - time.monotonic() - 30) / (n_try * max(1, remaining_lengths)))
        t_per = min(t_per, 35)

        for i in range(n_try):
            if time.monotonic() > phase1_end - 20:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 15)
            if t_alloc < 10:
                break
            f_init = inits[i]
            a_opt, obj = adaptive_gradient_descent(f_init, time_limit=t_alloc,
                                                    top_k_init=5, temp_init=0.01)
            try_sequence(a_opt, f"expl_n{n}_i{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Deep Optimization (45% of time) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_time = total_budget * 0.42
    phase2_end = time.monotonic() + phase2_time

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Optimizing n={n_best}")

        # Long adaptive gradient descent
        rem = phase2_end - time.monotonic() - 80
        if rem > 30:
            t_main = min(rem * 0.35, 150)
            print(f"  Adaptive GD (long), t={t_main:.0f}s")
            a_opt, obj = adaptive_gradient_descent(state['best_seq'], time_limit=t_main,
                                                    top_k_init=10, temp_init=0.003)
            try_sequence(a_opt, "deep_adapt", state)

        # Peak reduction
        rem = phase2_end - time.monotonic() - 50
        if rem > 25:
            t_pr = min(rem * 0.3, 80)
            print(f"  Peak reduction, t={t_pr:.0f}s")
            a_opt, obj = peak_reduction(state['best_seq'], time_limit=t_pr)
            try_sequence(a_opt, "peak_red", state)

        # FW-LP
        rem = phase2_end - time.monotonic() - 40
        if rem > 25:
            t_fw = min(rem * 0.35, 80)
            print(f"  FW-LP, t={t_fw:.0f}s")
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_fw)
            try_sequence(a_opt, "deep_fw", state)

        # Palindromic projection + re-optimize
        rem = phase2_end - time.monotonic() - 30
        if rem > 20:
            pal = palindromic_unfold(palindromic_fold(state['best_seq']), n_best)
            pal = pal / pal.sum()
            t_pal = min(rem * 0.4, 50)
            a_opt, obj = adaptive_gradient_descent(pal, time_limit=t_pal,
                                                    top_k_init=8, temp_init=0.002)
            try_sequence(a_opt, "palindromic", state)

        # Perturbation restarts
        perturb_scales = [0.003, 0.008, 0.015, 0.04]
        for ps in perturb_scales:
            rem = phase2_end - time.monotonic() - 20
            if rem < 15:
                break
            t_alloc = min(rem / 2.5, 45)
            if t_alloc < 12:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            a_opt, obj = adaptive_gradient_descent(f_init, time_limit=t_alloc,
                                                    top_k_init=8, temp_init=0.003)
            try_sequence(a_opt, f"pert_{ps}", state)

    print(f"  Phase 2 best: {state['best_score']:.6f}")

    # === Phase 3: Polishing (35% of time) ===
    print(f"\n=== Phase 3: Polishing ===")
    phase3_end = deadline - 5

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Current best: {state['best_score']:.6f}")

        # Multiple polish passes
        for pass_idx in range(4):
            if time.monotonic() > phase3_end - 20:
                break

            # Swap descent
            rem = phase3_end - time.monotonic() - 15
            if rem < 10:
                break
            t_swap = min(rem * 0.3, 50)
            if t_swap > 8:
                print(f"  Polish {pass_idx+1}: swap, t={t_swap:.0f}s")
                a_opt, obj = swap_descent(state['best_seq'], time_limit=t_swap, n_rounds=150)
                try_sequence(a_opt, f"pol_{pass_idx}_swap", state)

            # Fine gradient
            rem = phase3_end - time.monotonic() - 15
            if rem < 10:
                break
            t_grad = min(rem * 0.3, 40)
            if t_grad > 8:
                print(f"  Polish {pass_idx+1}: fine grad, t={t_grad:.0f}s")
                a_opt, obj = adaptive_gradient_descent(state['best_seq'], time_limit=t_grad,
                                                        top_k_init=5, temp_init=0.0005,
                                                        temp_min=0.00005)
                try_sequence(a_opt, f"pol_{pass_idx}_grad", state)

            # Peak reduction (fine)
            rem = phase3_end - time.monotonic() - 15
            if rem < 10:
                break
            t_peak = min(rem * 0.3, 35)
            if t_peak > 8:
                print(f"  Polish {pass_idx+1}: peak, t={t_peak:.0f}s")
                a_opt, obj = peak_reduction(state['best_seq'], time_limit=t_peak)
                try_sequence(a_opt, f"pol_{pass_idx}_peak", state)

        # Final squeeze
        rem = phase3_end - time.monotonic() - 5
        if rem > 10:
            a_opt, obj = swap_descent(state['best_seq'], time_limit=min(rem, 30), n_rounds=200)
            try_sequence(a_opt, "final_swap", state)

        # One more adaptive GD with very low temperature
        rem = phase3_end - time.monotonic() - 5
        if rem > 10:
            a_opt, obj = adaptive_gradient_descent(state['best_seq'], time_limit=min(rem, 30),
                                                    top_k_init=3, temp_init=0.0003,
                                                    temp_min=0.00001)
            try_sequence(a_opt, "final_grad", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(256) / 256
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
