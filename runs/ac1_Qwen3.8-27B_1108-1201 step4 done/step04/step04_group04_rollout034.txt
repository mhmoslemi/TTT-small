def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)

    rho = np.nonzero(cond)[0][-1]

    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]


def lbgfbs_optimize(f0, time_limit, T_init=0.002, T_min=0.00005, 
                    T_decay=0.99, verbose=False):
    """Optimize using L-BFGS-B with annealed log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    T = T_init
    stage = 0
    
    while time.monotonic() - t0 < time_limit:
        # Define objective and gradient for current T
        def obj_func(a):
            conv = fast_convolve(a)
            scaled = conv / T
            scaled -= scaled.max()
            ws = np.exp(scaled).sum()
            if ws < 1e-300:
                return T * (scaled.max() + 0)  # fallback
            return T * (scaled.max() + np.log(ws))
        
        def obj_grad(a):
            conv = fast_convolve(a)
            scaled = conv / T
            scaled -= scaled.max()
            w = np.exp(scaled)
            ws = w.sum()
            if ws < 1e-300:
                p = np.zeros(len(conv))
                p[np.argmax(conv)] = 1.0
            else:
                p = w / ws
            return compute_grad(a, p)
        
        # Run L-BFGS-B
        bounds = [(0.0, None)] * n
        res = minimize(
            obj_func, f.copy(),
            jac=obj_grad,
            method='L-BFGS-B',
            bounds=bounds,
            options={
                'maxiter': 200,
                'maxfun': 500,
                'gtol': 1e-8,
                'ftol': 1e-14,
                'disp': False
            }
        )
        
        if res.success or res.fun < obj_func(f):
            f_new = np.maximum(res.x, 0.0)
            s_new = f_new.sum()
            if s_new > 1e-10:
                f_new = f_new / s_new
                obj_new = fast_convolve(f_new).max()
                if obj_new < best_obj:
                    best_obj = obj_new
                    best_f = f_new.copy()
                    f = f_new
                    stage = 0
                else:
                    stage += 1
            else:
                stage += 1
        else:
            stage += 1
        
        # Anneal T
        T = max(T * T_decay, T_min)
        
        # If stuck, try a restart
        if stage > 3:
            if time.monotonic() - t0 > time_limit * 0.5:
                # In second half: smaller perturbations
                scale = 0.001
            else:
                scale = 0.005
            rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            T = max(T * 2, T_init * 0.3)
            stage = 0
        
        if verbose:
            elapsed = time.monotonic() - t0
            print(f"    L-BFGS-B: t={elapsed:.0f}s, obj={best_obj:.10f}, T={T:.6f}")
        
        if time.monotonic() - t0 > time_limit - 5:
            break
    
    return best_f, best_obj

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def grad_smoothed_max(a, T):
    """Gradient of T*log(sum(exp(conv(a)/T))) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    p = w / ws
    # grad_i = 2 * sum_k p_k * a_{k-i} (valid indices)
    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]

def lbgfbs_round(a_init, T, maxiter=300):
    """One round of L-BFGS-B optimization at fixed T."""
    n = len(a_init)
    a = a_init.copy()
    
    def obj(a):
        conv = fast_convolve(a)
        s = conv / T
        s -= s.max()
        return T * np.log(np.exp(s).sum())
    
    def obj_grad(a):
        return grad_smoothed_max(a, T)
    
    bounds = [(0.0, None)] * n
    res = minimize(obj, a, jac=obj_grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15})
    return res.x

"""
L-BFGS-B + Adam hybrid optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses smoothed max (log-sum-exp) with adaptive temperature annealing,
L-BFGS-B for fast quasi-Newton convergence, Adam for robustness,
multi-length exploration, and population-based diversity.

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_from_p(a, p):
    """Compute gradient of sum_k p_k * conv_k w.r.t. a = 2 * cross_corr(a, p)."""
    n = len(a)
    fft_size = 1 << (2 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]


def _smooth_conv_obj(a, T):
    """Smoothed max of convolution: T*log(sum(exp(conv/T)))."""
    conv = fast_convolve(a)
    m = conv.max()
    s = (conv - m) / T
    return m + T * np.log(np.exp(s).sum())


def _smooth_conv_grad(a, T):
    """Gradient of smoothed max convolution w.r.t. a."""
    conv = fast_convolve(a)
    m = conv.max()
    s = (conv - m) / T
    w = np.exp(s)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws
    return _grad_from_p(a, p)


class LbfgsObj:

    def __init__(self, T):

    def fun(self, a):

    def jac(self, a):

def lbfgs_single_round(a_init, T, maxiter=500):

    n = len(a_init)
    a = np.maximum(a_init, 0.0)

    bounds = [(0.0, None)] * n
    res = minimize(
        obj_wrapper.fun, a,
        jac=obj_wrapper.jac,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'gtol': 1e-10, 'ftol': 1e-15, 'disp': False}
    )
    return np.maximum(res.x, 0.0)


def lbfgs_anneal(f0, time_limit, T_init=0.003, T_min=0.00003, T_decay=0.985,

def adam_optimize(f0, time_limit, lr=0.005, T_init=0.002, T_min=0.0001,
                  beta1=0.9, beta2=0.999, eps=1e-8, verbose=False):
    """Adam optimizer on the simplex with log-sum-exp smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    T = T_init
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t_step = 0
    stuck = 0

    while time.monotonic() - t0 < time_limit - 5:
        t_step += 1
        conv = fast_convolve(f)
        grad = _smooth_conv_grad(f, T)

        # Project gradient onto tangent space of simplex
        grad = grad - np.mean(grad)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            stuck = 0
        else:
            stuck += 1

        # Anneal T
        T = max(T * 0.997, T_min)

        # Adaptive learning rate
        if stuck > 50 and stuck % 50 == 0:
            lr = max(lr * 0.8, 0.0001)

        # Restart
        if stuck > 200:
            rng = np.random.default_rng(t_step * 31 + n)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            lr = max(lr * 2.0, 0.002)
            T = max(T_init * 0.5, T_min)
            stuck = 0
            m = np.zeros(n)
            v = np.zeros(n)

        if verbose and t_step % 200 == 0:
            print(f"    Adam: step={t_step}, obj={best_obj:.10f}, T={T:.6f}")

def generate_inits(n, rng, parent=None):
    """Generate diverse initializations of length n on the simplex."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-concentrated (U-shape)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())

    # Asymmetric (left-heavy)
    asym = np.exp(-x / (0.3 * n))
    inits.append(asym / asym.sum())

    # Asymmetric (right-heavy)
    asym2 = np.exp(-(n - 1 - x) / (0.3 * n))
    inits.append(asym2 / asym2.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Random Dirichlet (sparse)
    inits.append(rng.dirichlet(np.ones(n) * 0.1))

    # Mirrored random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    n_remain = n - 2 * half_n
    if n_remain > 0:
        sym[half_n:half_n + n_remain] = 0.01
    inits.append(sym / sym.sum())

    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n),
                       np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def ensemble_solutions(solutions, weights=None):
    """Average multiple solutions (all must be same length)."""
    if not solutions:
        return None
    n = len(solutions[0])
    if weights is None:
        weights = np.ones(len(solutions)) / len(solutions)
    avg = np.zeros(n)
    for sol, w in zip(solutions, weights):
        avg += w * sol
    return project_simplex(avg)


def sg_polish_quick(f0, time_limit, lr=0.001, temp=0.00003):
    """Quick subgradient polish on top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()

    t0 = time.monotonic()
    top_k = max(5, int(n * 0.05))

    while time.monotonic() - t0 < time_limit - 2:
        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        f = project_simplex(f - lr * sg)
        co = fast_convolve(f).max()

        if co < best_obj - 1e-12:
            best_obj = co
            best_f = f.copy()

    return best_f, best_obj

        if np.isfinite(score) and score < best_score:

            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except NameError:
        pass
    except Exception:
        pass

    if parent is not None:
        base_n = len(parent)
    else:
        base_n = 384

    print(f"Base length: {base_n}, budget: {budget_s}s")

    # === Phase 1: Quick exploration (250s) ===
    print("\n=== Phase 1: Quick Exploration ===")
    t1_start = time.monotonic()
    phase1_budget = 250.0
    all_good = []

    # Optimize parent quickly
    if parent is not None:
        rem = phase1_budget - (time.monotonic() - t1_start)
        if rem > 30:
            t_alloc = min(80.0, rem - 20)
            print(f"  L-BFGS-B parent (n={base_n}, t={t_alloc:.0f}s)")
            a_opt, _ = lbfgs_anneal(parent, time_limit=t_alloc,
                                     T_init=0.003, T_min=0.0001, T_decay=0.98)
            try_seq(a_opt, "p1_parent")
            all_good.append(a_opt)

    # Try diverse inits at base length
    inits = generate_inits(base_n, rng, parent)
    # Quick evaluate and pick top 3
    init_objs = []
    for idx, f_init in enumerate(inits):
        obj = fast_convolve(f_init).max()
        init_objs.append((obj, idx))
    init_objs.sort()

    rem = phase1_budget - (time.monotonic() - t1_start)
    for rank, (obj_val, idx) in enumerate(init_objs[:4]):
        if rem < 30:
            break
        t_alloc = min(50.0, rem - 15)
        print(f"  L-BFGS-B init{idx} (obj={obj_val:.6f}, t={t_alloc:.0f}s)")
        a_opt, _ = lbfgs_anneal(inits[idx], time_limit=t_alloc,
                                 T_init=0.004, T_min=0.0001, T_decay=0.98)
        try_seq(a_opt, f"p1_init{idx}")
        all_good.append(a_opt)
        rem = phase1_budget - (time.monotonic() - t1_start)

    # Try a few other lengths
    for delta in [32, -32, 64, -64]:
        new_n = base_n + delta
        if new_n < 128 or new_n > 768:
            continue
        rem = phase1_budget - (time.monotonic() - t1_start)
        if rem < 30:
            break
        t_alloc = min(40.0, rem - 15)
        f_init = generate_inits(new_n, rng, parent)[0]  # uniform
        a_opt, _ = lbfgs_anneal(f_init, time_limit=t_alloc,
                                 T_init=0.005, T_min=0.0002, T_decay=0.98)
        try_seq(a_opt, f"p1_n{new_n}")
        all_good.append(a_opt)
        rem = phase1_budget - (time.monotonic() - t1_start)

    print(f"  Phase 1 done: best={best_score:.6f}")

    # === Phase 2: Deep optimization (850s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t2_start = time.monotonic()
    phase2_budget = 850.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()
    else:
        f_work = parent if parent is not None else np.ones(base_n) / base_n

    # Multiple rounds of L-BFGS-B with decreasing T
    n_rounds = 5
    for rnd in range(n_rounds):
        rem = phase2_budget - (time.monotonic() - t2_start)
        if rem < 30:
            break
        t_alloc = min(150.0, rem / (n_rounds - rnd + 1))
        T_start = max(0.001, 0.004 / (rnd + 1))
        T_end = max(0.00002, T_start * 0.02)
        print(f"  Round {rnd}: L-BFGS-B anneal (T: {T_start:.5f}->{T_end:.5f}, t={t_alloc:.0f}s)")
        a_opt, _ = lbfgs_anneal(f_work, time_limit=t_alloc,
                                 T_init=T_start, T_min=T_end, T_decay=0.985)
        try_seq(a_opt, f"p2_r{rnd}")
        if best_seq is not None:
            f_work = best_seq / best_seq.sum()

        # Adam interlude
        rem = phase2_budget - (time.monotonic() - t2_start)
        if rem > 40 and (rnd % 2 == 0):
            t_adam = min(80.0, rem / 2)
            print(f"    Adam (t={t_adam:.0f}s)")
            a_opt, _ = adam_optimize(f_work, time_limit=t_adam,
                                      lr=0.003, T_init=T_start,
                                      T_min=max(0.00005, T_end))
            try_seq(a_opt, f"p2_adam_r{rnd}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

    # Try ensemble of good solutions
    if len(all_good) >= 2:
        rem = phase2_budget - (time.monotonic() - t2_start)
        if rem > 30:
            # Ensemble all same-length solutions
            same_len = [s for s in all_good if len(s) == base_n]
            if len(same_len) >= 2:
                ens = ensemble_solutions(same_len)
                if ens is not None:
                    t_ens = min(60.0, rem / 2)
                    a_opt, _ = lbfgs_anneal(ens, time_limit=t_ens,
                                             T_init=0.001, T_min=0.00005,
                                             T_decay=0.99)
                    try_seq(a_opt, "p2_ensemble")

    print(f"  Phase 2 done: best={best_score:.6f}")

    # === Phase 3: Multi-length + Adam (500s) ===
    print(f"\n=== Phase 3: Multi-length & Adam ===")
    t3_start = time.monotonic()
    phase3_budget = 500.0

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()
    else:
        n_best = base_n
        f_work = parent if parent is not None else np.ones(n_best) / n_best

    # Try lengths around best
    for delta in [16, -16, 48, -48, 8, -8]:
        new_n = n_best + delta
        if new_n < 100 or new_n > 1024:
            continue
        rem = phase3_budget - (time.monotonic() - t3_start)
        if rem < 40:
            break
        # Interpolate best solution to new length
        px = np.interp(np.linspace(0, 1, new_n),
                       np.linspace(0, 1, n_best), f_work)
        px = project_simplex(px)
        t_alloc = min(80.0, rem / 2)
        print(f"  Length {new_n}: L-BFGS-B (t={t_alloc:.0f}s)")
        a_opt, _ = lbfgs_anneal(px, time_limit=t_alloc,
                                 T_init=0.002, T_min=0.00005, T_decay=0.988)
        try_seq(a_opt, f"p3_n{new_n}")
        if best_seq is not None:
            f_work = best_seq / best_seq.sum()
            n_best = len(best_seq)
        rem = phase3_budget - (time.monotonic() - t3_start)

    # Adam with long run
    rem = phase3_budget - (time.monotonic() - t3_start)
    if rem > 60 and best_seq is not None:
        f_work = best_seq / best_seq.sum()
        t_adam = min(200.0, rem - 20)
        print(f"  Long Adam (t={t_adam:.0f}s)")
        a_opt, _ = adam_optimize(f_work, time_limit=t_adam,
                                  lr=0.002, T_init=0.001, T_min=0.00003)
        try_seq(a_opt, "p3_adam_long")

    # Another L-BFGS-B round from best
    rem = phase3_budget - (time.monotonic() - t3_start)
    if rem > 40 and best_seq is not None:
        f_work = best_seq / best_seq.sum()
        t_alloc = min(100.0, rem - 20)
        print(f"  Final L-BFGS-B (t={t_alloc:.0f}s)")
        a_opt, _ = lbfgs_anneal(f_work, time_limit=t_alloc,
                                 T_init=0.0005, T_min=0.00002, T_decay=0.992)
        try_seq(a_opt, "p3_final_lbfgs")

    print(f"  Phase 3 done: best={best_score:.6f}")

    # === Phase 4: Final polish (200s) ===
    print(f"\n=== Phase 4: Final Polish ===")
    t4_start = time.monotonic()
    phase4_budget = 200.0

    if best_seq is not None:
        f_work = best_seq / best_seq.sum()

        # SG polish
        rem = phase4_budget - (time.monotonic() - t4_start)
        if rem > 30:
            t_sg = min(60.0, rem - 10)
            print(f"  SG polish (t={t_sg:.0f}s)")
            a_opt, _ = sg_polish_quick(f_work, time_limit=t_sg,
                                       lr=0.0008, temp=0.00002)
            try_seq(a_opt, "p4_sg")

        # Very fine L-BFGS-B
        rem = phase4_budget - (time.monotonic() - t4_start)
        if rem > 30 and best_seq is not None:
            f_work = best_seq / best_seq.sum()
            t_fine = min(80.0, rem - 10)
            print(f"  Fine L-BFGS-B (t={t_fine:.0f}s)")
            a_opt, _ = lbfgs_anneal(f_work, time_limit=t_fine,
                                     T_init=0.0003, T_min=0.00001,
                                     T_decay=0.995)
            try_seq(a_opt, "p4_fine")

        # Tiny perturbations
        rem = phase4_budget - (time.monotonic() - t4_start)
        if rem > 20 and best_seq is not None:
            f_work = best_seq / best_seq.sum()
            n_best = len(f_work)
            for attempt in range(3):
                if time.monotonic() > deadline - 15:
                    break
                noise = rng.standard_normal(n_best) * 0.0003
                f_pert = project_simplex(f_work + noise)
                t_pert = min(25.0, (deadline - time.monotonic()) / 2)
                if t_pert > 10:
                    a_opt, _ = lbfgs_anneal(f_pert, time_limit=t_pert,
                                             T_init=0.0002, T_min=0.00002,
                                             T_decay=0.995)
                    try_seq(a_opt, f"p4_pert{attempt}")

        # Mirror average
        if best_seq is not None:
            f_work = best_seq / best_seq.sum()
            mirrored = 0.5 * f_work + 0.5 * f_work[::-1]
            mirrored = project_simplex(mirrored)
            if np.max(np.abs(mirrored - f_work)) > 1e-6:
                if time.monotonic() < deadline - 20:
                    a_opt, _ = lbfgs_anneal(mirrored, time_limit=20,
                                             T_init=0.0002, T_min=0.00005,
                                             T_decay=0.99)
                    try_seq(a_opt, "p4_mirror")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(base_n) / base_n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
