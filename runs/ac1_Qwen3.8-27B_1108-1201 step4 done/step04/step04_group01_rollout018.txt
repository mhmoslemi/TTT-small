"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Diverse structural initializations (symmetric, frequency-shaped)
2. Piecewise quadratic smoothed objective with active-set tracking
3. Projected gradient with backtracking line search and momentum
4. Symmetric optimization (optimize half, mirror)
5. Multi-restart with annealing
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_grad_smoothed(f, conv, t, p=2):
    """Compute gradient of sum(max(0, conv - t)^p) w.r.t. f.
    Active lags: those where conv[k] > t.
    grad[i] = 2*p * sum_{k: conv[k]>t} (conv[k]-t)^(p-1) * f[k-i] (with bounds)
    """
    n = len(f)
    active = conv > t
    if not active.any():
        return np.zeros(n)
    
    # Weight for each active lag
    weights = (conv[active] - t) ** (p - 1) * active
    # grad[i] = 2*p * sum_k weights[k] * f[k-i] where 0 <= k-i < n
    # This is a cross-correlation of weights with f
    # Use FFT for efficiency
    total_len = len(weights) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F_flip = np.fft.rfft(f[::-1], fft_size)
    corr = np.fft.irfft(W * F_flip, fft_size)
    return 2.0 * p * corr[n-1:n-1+n]

def smoothed_obj(f, t, p=2):
    """Compute sum(max(0, conv(f) - t)^p)."""
    conv = fast_convolve(f)
    excess = np.maximum(conv - t, 0)
    return np.sum(excess ** p)

def obj_gradient_descent(f0, time_limit, t_init=None, p=2, lr_init=0.01, 
                         momentum=0.9, max_iter=None):
    """Optimize using gradient descent on the smoothed objective with adaptive t."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    # Initialize t slightly above the current max
    if t_init is None:
        t = conv.max() * 0.95
    else:
        t = t_init
    
    t0 = time.monotonic()
    lr = lr_init
    v = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if max_iter and it >= max_iter:
            break
        it += 1
        
        conv = fast_convolve(f)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.9
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = lr_init * 0.5
                no_improve = 0
        
        # Compute gradient
        grad = compute_grad_smoothed(f, conv, t, p)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Gradient step with momentum
        v = momentum * v - lr * grad / (grad_norm + 1e-15)
        f_new = project_simplex(f + v)
        
        # Check if step improved things
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < obj:
            f = f_new
            # Decrease t slowly to tighten
            t = max(conv_new.max() * 0.9, t * 0.999)
        else:
            # Revert
            v = momentum * v
            lr *= 0.95
            t = min(t * 1.01, conv.max())
        
        # Ramp t down over time
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        t_target = best_obj * (1.0 - 0.1 * progress)
        t = max(t, t_target * 0.8)
        
        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PGD it={it}, obj={best_obj:.10f}, t={t:.6f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-track approach:
1. Soft-threshold gradient descent (different smoothing from LSE)
2. Active-set refinement with local quadratic approximation
3. Symmetric optimization track (exploit reversibility)

5. Multi-restart with structured perturbations

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_conv_grad(f, conv, active_mask, weights):
    """Gradient of sum(weights[k] * conv[k]) w.r.t. f.
    grad[i] = 2 * sum_k weights[k] * f[k-i] (with appropriate bounds)
    Uses FFT cross-correlation.
    """
    n = len(f)
    w = np.zeros(len(conv))
    w[active_mask] = weights[active_mask]
    if w.max() < 1e-18:
        return np.zeros(n)
    total_len = len(w) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, fft_size)
    F_flip = np.fft.rfft(f[::-1].copy(), fft_size)
    corr = np.fft.irfft(W * F_flip, fft_size)
    return 2.0 * corr[n - 1:n - 1 + n]


def softplus_objective_grad(f, beta, t_target):
    """Compute gradient of sum(softplus(beta*(conv - t_target))) w.r.t. f.
    softplus(x) = log(1 + exp(x)), stable version.
    Returns (gradient, objective_value).
    """
    n = len(f)
    conv = fast_convolve(f)
    x = beta * (conv - t_target)
    # Stable softplus
    sp = np.where(x > 30, x, np.log1p(np.exp(x)))
    # Stable sigmoid (derivative of softplus)
    sig = np.where(x > 30, 1.0, 1.0 / (1.0 + np.exp(-x)))
    obj_val = np.sum(sp)
    # Gradient: sum_k sig[k] * d(conv[k])/df
    # d(conv[k])/df[i] = f[k-i] + f[i-k] (for valid indices)
    # Actually for conv[k] = sum_j f[j]*f[k-j], d/df[i] = f[k-i] + f[i-k] if k != 0
    # For k = 0: d/df[0] = 2*f[0], others 0
    # In general: d(conv[k])/df[i] = f[k-i] (if 0<=k-i<n) + f[i-k] (if 0<=i-k<n and i-k != k-i)
    # Simplification: the gradient of conv[k] w.r.t. f is the "cross-correlation" of f with the delta at k plus delta at -k
    # Which is: (f convolved with delta_k) + (f convolved with delta_{-k}) 
    # = f[k-i] (shifted) ... 
    # Actually the simplest way: grad[i] = 2 * sum_k sig[k] * f[k-i] (with bounds checking)
    # This works because conv is quadratic and symmetric in f, so the gradient is 2 * (f cross-corr with weights)
    active = sig > 1e-15
    if not active.any():
        return np.zeros(n), obj_val
    grad = compute_conv_grad(f, conv, active, sig)
    return grad, obj_val


def topk_smoothed_grad(f, k, temperature):
    """Gradient using top-k selection with softmax weights.
    Select top-k lags, weight them with softmax, compute gradient.
    """
    n = len(f)
    conv = fast_convolve(f)
    # Select top-k
    if k >= len(conv):
        indices = np.arange(len(conv))
        vals = conv
    else:
        indices = np.argpartition(conv, -k)[-k:]
        vals = conv[indices]
    # Softmax weights (focused on the largest)
    scaled = (vals - vals.max()) / temperature
    w = np.exp(scaled)
    w = w / w.sum()
    full_w = np.zeros(len(conv))
    full_w[indices] = w
    active = full_w > 1e-15
    grad = compute_conv_grad(f, conv, active, full_w)
    return grad, conv.max()


def optimize_softplus(f0, time_limit, beta_start=50.0, beta_end=500.0, 
                      lr_init=0.05, momentum=0.92, t_decay=0.9995):

    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    beta = beta_start
    t_target = conv.max()
    lr = lr_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1

        if grad_norm < 1e-16:

        grad = grad / grad_norm
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Tighten target
            t_target = min(t_target * t_decay, obj_new * 0.95)
            # Slight lr increase
            lr = min(lr * 1.001, 0.15)
        else:
            no_improve += 1
            f = best_f.copy()
            v = momentum * v
            if no_improve >= 5:
                lr *= 0.93
            if no_improve > 60:
                f = best_f.copy()
                v = np.zeros(n)
                lr = lr_init
                beta = min(beta * 1.3, beta_end)
                t_target = best_obj
                no_improve = 0

        t_target = min(t_target, best_obj * 1.001)

            f = best_f.copy()

            if no_improve >= 8:
                lr *= 0.92
            if no_improve > 80:
                f = best_f.copy()
                v = np.zeros(n)
                lr = lr_init * 0.7
                temp = temp_init * 0.5
                no_improve = 0
        
        temp *= temp_growth
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    TopK it={it}, obj={best_obj:.10f}, k={cur_k}, temp={temp:.5f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_obj


def optimize_symmetric(f0, time_limit, lr_init=0.03, momentum=0.91, 
                        beta_start=30.0, beta_end=400.0):
    """Optimize assuming symmetric structure: a[i] = a[n-1-i].
    Only optimize the first half.
    Returns the full symmetric sequence.
    """
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s
    
    # Extract half
    half_len = (n + 1) // 2
    f_half = f[:half_len].copy()
    
    def expand(half):
        """Expand half to full symmetric sequence."""
        full = np.zeros(n)
        full[:half_len] = half
        if n % 2 == 0:
            full[half_len:] = half[::-1]
        else:
            full[half_len:] = half[-2::-1]
        return full
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_half = f_half.copy()
    
    beta = beta_start
    t_target = conv.max()
    lr = lr_init
    v = np.zeros(half_len)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1
        
        full = expand(f_half)
        conv = fast_convolve(full)
        x = beta * (conv - t_target)
        sp = np.where(x > 30, x, np.log1p(np.exp(x)))
        sig = np.where(x > 30, 1.0, 1.0 / (1.0 + np.exp(-x)))
        
        # Compute gradient w.r.t. full, then map to half
        full_grad = compute_conv_grad(full, conv, sig > 1e-15, sig)
        
        # Map gradient to half: grad_half[i] = full_grad[i] + full_grad[n-1-i]
        if n % 2 == 0:
            grad_half = full_grad[:half_len] + full_grad[half_len-1::-1]
        else:
            grad_half = np.zeros(half_len)
            grad_half[:n//2] = full_grad[:n//2] + full_grad[n-1:n//2-1:-1]
            grad_half[n//2] = full_grad[n//2]  # center element
            grad_half[-1] += full_grad[0] if n//2 == half_len-1 else 0
            # Simpler: just take first half_len and add mirrored
            grad_half = full_grad[:half_len].copy()
            mirror = full_grad[n-1:n-1-half_len:-1]
            grad_half += mirror
        
        grad_norm = np.linalg.norm(grad_half)
        if grad_norm < 1e-16:
            break
        grad_half = grad_half / grad_norm
        
        v = momentum * v - lr * grad_half
        f_half_new = np.maximum(f_half + v, 0.0)
        f_half_new = f_half_new / (f_half_new.sum() + 1e-30)
        
        full_new = expand(f_half_new)
        conv_new = fast_convolve(full_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_half = f_half_new.copy()
            f_half = f_half_new
            no_improve = 0
            t_target = min(t_target * 0.9995, obj_new * 0.96)
        else:
            no_improve += 1
            f_half = best_half.copy()
            v *= 0.5
            if no_improve >= 8:
                lr *= 0.92
            if no_improve > 80:
                f_half = best_half.copy()
                v = np.zeros(half_len)
                lr = lr_init
                beta = min(beta * 1.3, beta_end)
                no_improve = 0
        
        beta = min(beta * 1.003, beta_end)
        t_target = min(t_target, best_obj * 1.001)
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Sym it={it}, obj={best_obj:.10f}, beta={beta:.1f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")
    
    best_full = expand(best_half)
    best_full = np.maximum(best_full, 0.0)
    s = best_full.sum()
    if s > 1e-15:
        best_full = best_full / s
    return best_full, best_obj


def penalty_method_optimize(f0, time_limit, lr_init=0.01, momentum=0.90,
                             lambda_start=100.0, lambda_growth=1.005):
    """Penalty method: minimize t + lambda * sum(max(0, c-t)^2).
    Start with large lambda, gradually increase to enforce constraints harder.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    lam = lambda_start
    t = conv.max() * 0.9
    lr = lr_init
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break
        it += 1
        
        conv = fast_convolve(f)
        excess = np.maximum(conv - t, 0.0)
        # Gradient of sum(excess^2) = 2 * sum_k excess[k] * d(conv[k])/df
        active = excess > 1e-15
        if active.any():
            grad = 2.0 * lam * compute_conv_grad(f, conv, active, excess[active] if False else excess)
            # Fix: weights should be the excess values at active positions
            w = np.zeros(len(conv))
            w[active] = excess[active]
            grad = 2.0 * lam * compute_conv_grad(f, conv, active, w[active])
        else:
            grad = np.zeros(n)
        
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-16:
            break
        grad = grad / grad_norm
        
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Decrease t
            t = max(t * 0.999, obj_new * 0.92)
        else:
            no_improve += 1
            f = best_f.copy()
            v *= 0.5
            if no_improve >= 10:
                lr *= 0.93
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                lr = lr_init * 0.5
                t = best_obj * 0.95
                no_improve = 0
        
        lam *= lambda_growth
        t = min(t, best_obj * 1.002)
        
        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Penalty it={it}, obj={best_obj:.10f}, lam={lam:.1f}, t={t:.6f}, lr={lr:.6f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_obj


def freq_domain_init(n, mode='trapezoid', rng=None):
    """Generate diverse initial sequences using frequency domain shaping."""
    if rng is None:
        rng = np.random.default_rng()
    
    fft_size = 1 << (2 * n - 1).bit_length()
    
    if mode == 'flat':
        # Flat spectrum -> white noise time domain
        A = np.ones(n / fft_size)
        A[n:] = 0
    elif mode == 'trapezoid':
        # Trapezoidal spectrum
        A = np.trapezoid if hasattr(np, 'trapezoid') else None
        A = np.zeros(n)
        center = n // 2
        width = n // 3
        for i in range(width, n - width):
            A[i] = 1.0
        for i in range(width):
            A[i] = i / width
            A[n - 1 - i] = i / width
    elif mode == 'gaussian':
        x = np.arange(n) - n / 2
        A = np.exp(-x**2 / (2 * (n / 4)**2))
    elif mode == 'bump':
        # Concentrated in frequency -> spread in time
        center = n // 2
        A = np.zeros(n)
        bw = max(2, n // 8)
        A[center - bw:center + bw] = 1.0
    elif mode == 'two_peaks':
        A = np.zeros(n)
        c1 = n // 4
        c2 = 3 * n // 4
        bw = max(2, n // 16)
        A[c1 - bw:c1 + bw] = 1.0
        A[c2 - bw:c2 + bw] = 1.0
    else:
        A = np.ones(n) / n
    
    # Add small random perturbation
    A = A + rng.standard_normal(n) * 0.01
    A = np.maximum(A, 0)
    
    # Inverse FFT to get time domain
    a = np.fft.irfft(A * fft_size)
    a = a[:n]
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    else:
        a = np.ones(n) / n
    
    return a


def diverse_perturb(f, rng, scale=0.05):
    """Generate a perturbed version of f."""
    n = len(f)
    mode = rng.integers(4)
    
    if mode == 0:
        # Gaussian perturbation
        perturbed = f + rng.standard_normal(n) * scale
    elif mode == 1:
        # Block perturbation
        perturbed = f.copy()
        block_size = max(1, n // 10)
        start = rng.integers(0, n - block_size)
        perturbed[start:start + block_size] += rng.standard_normal(block_size) * scale * 2
    elif mode == 2:
        # Frequency perturbation
        fft_size = 1 << (2 * n - 1).bit_length()
        A = np.fft.rfft(f, fft_size)
        A = A * (1 + rng.standard_normal(A.size) * 0.05)
        perturbed = np.fft.irfft(A, fft_size)[:n]
    else:
        # End emphasis
        perturbed = f.copy()
        frac = max(1, n // 8)
        perturbed[:frac] *= (1 + rng.standard_normal() * 0.1)
        perturbed[-frac:] *= (1 + rng.standard_normal() * 0.1)
    
    perturbed = np.maximum(perturbed, 0)
    s = perturbed.sum()
    if s > 1e-10:
        perturbed = perturbed / s
    else:
        perturbed = f.copy()
    
    return perturbed


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score
