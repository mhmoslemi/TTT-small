def fast_convolve(a):

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def coord_line_search(f, i, n):
    """For coordinate i, find optimal value via line search."""
    old_val = f[i]
    if old_val < 1e-14:
        return old_val, None
    
    others_sum = 1.0 - old_val
    if others_sum < 1e-14:
        return old_val, None
    
    best_t = old_val
    best_obj = fast_convolve(f).max()
    
    # Try a range of values

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def solve_lp_direction(f, M, n, method='highs'):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[j] * g[k-j] for valid j
    # For each k, (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[j] * g[k-j]
    # Actually, (f*g)[k] = sum_j f[j]*g[k-j] where 0<=j<n and 0<=k-j<n
    # So j ranges from max(0, k-n+1) to min(n-1, k)
    # The coefficient of g[j] in (f*g)[k] is f[k-j] (if 0<=k-j<n)
    # So (f*g)[k] = sum_j g[j] * f[k-j] where both j and k-j are in [0,n-1]
    
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # g[j] appears in (f*g)[k] for k = j, j+1, ..., j+n-1
        # with coefficient f[k-j] = f[0], f[1], ..., f[n-1]
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method=method)
        if res.success:
            return res.x
    except Exception:
        pass
    return None

    it = 0
    m_exp = 1.0

        it += 1

                m_exp = 2.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:

            if no_improve > 100:

            continue

        # Line search with golden section + fixed points
        best_alpha = 0.0
        best_val = M

        # Try fixed alphas
        for alpha in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()

                best_val = val
                best_alpha = alpha

                if st1 > 1e-15:
                    ft1 = ft1 / st1

                else:
                    val1 = best_val
                ft2 = f + t2 * (g - f)

                if st2 > 1e-15:

                else:
                    val2 = best_val
                if val1 < val2:
                    hi = t2
                else:
                    lo = t1
                if hi - lo < 1e-6:
                    break
            mid = (lo + hi) / 2
            ft = f + mid * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st > 1e-15:
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)

            conv = fast_convolve(f_new)
            M = conv.max()
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()

        else:

def sg_polish(f0, time_limit, lr=0.003, momentum=0.94, top_k=50, temp=0.0001):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v_mom = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v_mom = momentum * v_mom - cur_lr * sg
        f = project_simplex(f + v_mom)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.005

        if no_improve > 300:
            rng = np.random.default_rng(it * 41 + 3)
            scale = 0.001 + 0.002 * min(no_improve // 500, 3)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v_mom = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def diverse_initializer(n, rng):
    """Generate diverse initializations."""
    types = ['uniform', 'bell', 'bell2', 'edges', 'sine', 'random']
    t = rng.integers(len(types))
    
    if t == 0:  # uniform
        a = np.ones(n)
    elif t == 1:  # bell (centered)
        x = np.arange(n) / n
        a = np.exp(-4.0 * (x - 0.5)**2)
    elif t == 2:  # wider bell
        x = np.arange(n) / n
        a = np.exp(-2.0 * (x - 0.5)**2)
    elif t == 3:  # edges
        x = np.arange(n) / n
        a = 0.3 + 0.7 * np.cos(np.pi * x)**2
    elif t == 4:  # sine
        x = np.arange(n) / n
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    else:  # random
        a = rng.random(n)
    
    a = np.maximum(a, 0)
    ss = a.sum()
    if ss > 1e-15:
        a /= ss
    return a

Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines LP-based Frank-Wolfe, coordinate descent with golden section search,
subgradient polishing, and multi-length exploration with diverse initializations.

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_sequence(f, new_n):
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:

        pass
    return None
