def project_simplex(v):

    lr = 2.0 / n
    momentum = 0.90
    v = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        bc = beta * conv
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()
        grad = compute_lse_grad(w, f)

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.002, 4.0 / n)
        else:
            no_improve += 1
            if no_improve >= 10:
                lr *= 0.92
                momentum = max(0.85, momentum - 0.005)
            if no_improve > 80:
                f = best_f.copy()
                v = np.zeros(n)
                lr = 2.0 / n
                beta = min(beta * 1.5, beta_end)
                momentum = 0.88
                no_improve = 0

        if beta < beta_end:
            beta *= 1.003

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    StdLSE it={it}, obj={best_obj:.12f}, beta={beta:.0f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit, lp_timeout=15):
    """Frank-Wolfe with LP oracle, adaptive relaxation, and restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        g = solve_lp_direction(f, M * m_exp, n, lp_timeout)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 20:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.03, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.06, 5.0)

        if no_improve > 30:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.004 * min(1.0, (no_improve - 30) / 40.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def topk_subgrad(f0, time_limit, top_k_frac=0.08, lr_init=0.004, temp_init=0.00004):
    """Subgradient descent using top-k peak weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    top_k = max(8, int(n * top_k_frac))
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    v = np.zeros(n)
    momentum = 0.93

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w = w / w_sum

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = momentum * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                cur_lr *= 0.94
            if no_improve > 250:
                rng = np.random.default_rng(int(time.monotonic() * 1000000) % 999983)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                v = np.zeros(n)
                cur_lr = max(lr_init * 0.6, 0.0008)
                cur_temp *= 0.5
                no_improve = 0

        cur_temp *= 1.003

    return best_f, best_obj


def local_search(f0, time_limit, num_trials=200):
    """Perturbation-based local search."""
    f = np.maximum(f0.astype(np.float64), 0.0).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    trial = 0

    while time.monotonic() - t0 < time_limit and trial < num_trials:
        trial += 1
        rng = np.random.default_rng(trial * 104729 + 13)

        # Random perturbation: change a random subset of coordinates
        n_perturb = max(3, n // 20)
        indices = rng.choice(n, size=n_perturb, replace=False)
        scale = 0.005 + 0.01 * rng.random()
        perturbation = rng.standard_normal(n_perturb) * scale
        f_trial = f.copy()
        f_trial[indices] += perturbation
        f_trial = np.maximum(f_trial, 0.0)
        s_t = f_trial.sum()
        if s_t < 1e-15:
            continue
        f_trial = f_trial / s_t

        conv_trial = fast_convolve(f_trial)
        obj_trial = conv_trial.max()

        if obj_trial < best_obj - 1e-14:
            best_obj = obj_trial
            best_f = f_trial.copy()
            f = f_trial

    return best_f, best_obj


def make_diverse_init(n, seed):
    """Create diverse initial patterns."""
    rng = np.random.default_rng(seed)
    choice = seed % 6

    if choice == 0:
        # Cosine bump
        x = np.linspace(0, 1, n)
        f = np.cos(np.pi * (x - 0.5)) ** 2
    elif choice == 1:
        # Triangular
        x = np.linspace(0, 1, n)
        f = np.maximum(0, 1 - 2 * np.abs(x - 0.5))
    elif choice == 2:
        # Two peaks
        x = np.linspace(0, 1, n)
        f = np.exp(-50 * (x - 0.3) ** 2) + np.exp(-50 * (x - 0.7) ** 2)
    elif choice == 3:
        # Random smooth
        f = rng.standard_normal(n) ** 2
    elif choice == 4:
        # Flat with noise
        f = np.ones(n) + 0.3 * rng.standard_normal(n)
    else:
        # Multi-frequency
        f = 1.0 + 0.2 * np.sin(2 * np.pi * 3 * np.linspace(0, 1, n)) + \
            0.1 * np.sin(2 * np.pi * 7 * np.linspace(0, 1, n))

    f = np.maximum(f, 0.0)
    s = f.sum()
    if s > 1e-15:
        f = f / s
    else:
        f = np.ones(n) / n
    return f


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score

            parent = p / p.sum()

            try_sequence(parent, best_tracker)

        print(f"No parent: {e}")

    print("\n=== Phase 1: Diverse Init + Quick Optim ===")
