def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Asymmetric Gaussian (left-shifted)
    x = np.arange(n, dtype=np.float64)
    g_left = np.exp(-0.5 * ((x - 0.3 * n) / (0.15 * n)) ** 2)
    inits.append(g_left / g_left.sum())
    
    # Asymmetric Gaussian (right-shifted)
    g_right = np.exp(-0.5 * ((x - 0.7 * n) / (0.15 * n)) ** 2)
    inits.append(g_right / g_right.sum())
    
    # Edge-concentrated (U-shape, asymmetric)
    edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    edge_asym = 1.0 / (x + 1.0) + 0.5 / (n - x + 1.0)
    inits.append(edge_asym / edge_asym.sum())
    
    # Cosine (asymmetric)
    c1 = np.abs(np.cos(np.pi * x / (n + 1) + 0.3)) + 0.1
    inits.append(c1 / c1.sum())
    
    # Two humps (asymmetric positions)
    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.65 * n) / (0.12 * n)) ** 2)
    two = 0.6 * h1 + 0.4 * h2
    inits.append(two / two.sum())
    
    # From parent (if available)
    if parent is not None:
        # Resize parent to n
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Asymmetric perturbation of parent
            pert = px * (1 + 0.1 * rng.standard_normal(n))
            pert = np.maximum(pert, 0)
            ps2 = pert.sum()
            if ps2 > 1e-10:
                inits.append(pert / ps2)
            # Mirror parent
            mir = (px + px[::-1]) / 2
            mir = np.maximum(mir, 0)
            ms = mir.sum()
            if ms > 1e-10:
                inits.append(mir / ms)
    
    return inits
