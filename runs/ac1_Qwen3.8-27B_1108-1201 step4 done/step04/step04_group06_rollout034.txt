        if cu < best - 1e-12:
            best = cu
            best_a = a.copy()
            ni = 0
        else:
            ni += 1
        # Adaptive parameters
        lr *= 0.9995
        temp *= 1.003
        mom = min(0.96, mom + 0.0001)
        if ni >= 500:
            break
    return best_a, best


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def optimize_lp(a0, time_limit, use_symmetry=True):
    """LP-based iterative optimization with adaptive step sizing."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    best_a = a.copy()
    best_M = M
    t0 = time.monotonic()
    ni = 0
    max_no_improve = 60
    perturb_count = 0
    C = build_conv_matrix(a)
    while time.monotonic() - t0 < time_limit:
        eps = 0.0003
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * (1.0 - eps))
        bounds = [(0, None)] * n
        try:
            res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                   'dual_feasibility_tolerance': 1e-9})
        except Exception:
            ni += 1
            if ni > 10:
                break
            continue
        if not res.success:
            ni += 1
        else:
            g = res.x
            if use_symmetry:
                g = 0.5 * (g + g[::-1])
            gs = g.sum()
            if gs < 1e-10:
                ni += 1
            else:
                g = g / gs
                found = False
                for t in [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]:
                    a_try = np.maximum((1 - t) * a + t * g, 0.0)
                    if use_symmetry:
                        a_try = 0.5 * (a_try + a_try[::-1])
                    st = a_try.sum()
                    if st < 1e-15:
                        continue
                    a_try = a_try / st
                    ct = fast_convolve(a_try)
                    Mt = ct.max()
                    if Mt < M - 1e-14:
                        a = a_try
                        M = Mt
                        if M < best_M:
                            best_M = M
                            best_a = a.copy()
                        C = build_conv_matrix(a)
                        ni = 0
                        found = True
                        break
                if not found:
                    ni += 1
        if ni >= max_no_improve:
            rng = np.random.default_rng(perturb_count * 1000 + 42)
            scale = 0.002 + 0.004 * min(perturb_count, 6)
            a = np.maximum(best_a + rng.standard_normal(n) * scale, 0.0)
            if use_symmetry:
                a = 0.5 * (a + a[::-1])
            s2 = a.sum()
            if s2 > 1e-15:
                a = a / s2
            else:
                a = best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            C = build_conv_matrix(a)
            ni = 0
            perturb_count += 1
            max_no_improve = min(400, 60 + 30 * perturb_count)
            if perturb_count > 30:
                break
    return best_a, 2.0 * n * best_M


def coordinate_polish(a0, time_limit, use_symmetry=True):
    """Coordinate descent: try adjusting pairs to reduce the peak."""
    a = np.maximum(np.asarray(a0, dtype=np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, np.inf
    a = a / s
    n = len(a)
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(99999)
    ni = 0
    while time.monotonic() - t0 < time_limit:
        batch = max(1, min(200000 // n, 500))
        for _ in range(batch):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            improved = False
            for delta in [0.005, 0.002, 0.01, 0.001, -0.005, -0.002, -0.01, -0.001]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                if use_symmetry:
                    a_try = 0.5 * (a_try + a_try[::-1])
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try = a_try / st
                ct = fast_convolve(a_try)
                ot = 2.0 * n * ct.max()
                if ot < best_obj - 1e-11:
                    a = a_try
                    best_obj = ot
                    best_a = a.copy()
                    ni = 0
                    improved = True
                    break
            if not improved:
                ni += 1
        if ni > 300:
            break
    return best_a, best_obj


def init_central_binomial(n):
    """Initialize with central binomial coefficients C(2k,k)/4^k."""
    x = np.ones(n)
    for k in range(1, n):
        x[k] = x[k - 1] * (2 * k) * (2 * k - 1) / (4 * k * k)
    s = x.sum()
    if s > 1e-15:
        x = x / s
    return x


def init_sine(n, freq=1.0):
    """Sine-based initialization."""
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * freq * x / n) ** 2 + 0.01
    return a / a.sum()


def init_bimodal(n):
    """Bimodal: Gaussian at edges."""
    x = np.arange(n, dtype=np.float64)
    edge_w = max(1, n // 8)
    a_left = np.exp(-0.5 * ((x - 0) / edge_w) ** 2)
    a_right = np.exp(-0.5 * ((x - (n - 1)) / edge_w) ** 2)
    a = a_left + a_right + 0.01
    return a / a.sum()


def init_diverse(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    inits['uniform'] = np.ones(n) / n
    inits['binom'] = init_central_binomial(n)
    inits['sine1'] = init_sine(n, 1.0)
    inits['sine2'] = init_sine(n, 2.0)
    inits['bimodal'] = init_bimodal(n)
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    # Smooth random
    r = rng.standard_normal(n)
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
    r_smooth = np.convolve(r, kernel, mode='same')
    r_smooth = np.maximum(r_smooth, 0)
    rs = r_smooth.sum()
    if rs > 1e-15:
        inits['smooth_rand'] = r_smooth / rs
    # From parent
    if parent is not None and np.isfinite(parent).all() and parent.sum() > 1e-10:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and len(p) == n:
            inits['parent'] = p / ps
        elif ps > 1e-10:
            pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
            pp = np.maximum(pp, 0)
            pps = pp.sum()
            if pps > 1e-10:
                inits['parent_interp'] = pp / pps
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Hybrid optimization with multiple strategies.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass

    # Determine target lengths to try
    if parent is not None:
        n_parent = len(parent)
        target_lengths = [n_parent]
        # Try a slightly different length too
        if n_parent >= 256:
            target_lengths.append(n_parent // 2)
    else:
        target_lengths = [512, 768]

    # =========================================================
    # Phase 1: Quick gradient scan (200s)
    # =========================================================
    print("=== Phase 1: Gradient Scan ===")
    t_start = time.monotonic()
    phase1_limit = 180.0
    scan_results = []

    for n in target_lengths:
        if time.monotonic() > deadline - 600:
            break
        inits = init_diverse(n, parent, rng)
        priority = ['parent', 'parent_interp', 'binom', 'sine1', 'smooth_rand', 'uniform', 'random']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:4]

        t_per = max(15.0, (phase1_limit - (target_lengths.index(n) * 60)) / max(1, len(init_names)))

        for name in init_names:
            if time.monotonic() > deadline - 500:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 200)
            if t_alloc < 5:
                break
            a0 = inits[name]
            a_opt, obj_val = optimize_gradient(a0, time_limit=t_alloc, lr=0.01,
                                               mom=0.92, top_k=50, temp=0.003,
                                               use_symmetry=True)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad n={n} {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    if best_sequence is not None:
        print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: LP deep optimization (400s)
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start = time.monotonic()
    phase2_limit = 400.0

    # Select top candidates
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results:
        if n not in seen:
            top_configs.append((n, a_opt))
            seen.add(n)
        if len(top_configs) >= 2:
            break
    if best_sequence is not None and len(best_sequence) not in seen:
        top_configs.append((len(best_sequence), best_sequence.copy()))

    t_per_lp = max(60.0, phase2_limit / max(1, len(top_configs)))

    for i, (n, a0) in enumerate(top_configs):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 150)
        if t_alloc < 15:
            break
        print(f"  LP cand {i}: n={n}, t={t_alloc:.0f}s")
        a_opt, obj_val = optimize_lp(a0, time_limit=t_alloc, use_symmetry=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Extended LP with perturbations (800s)
    # =========================================================
    print("=== Phase 3: Extended LP ===")
    t_start = time.monotonic()
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = min(800.0, deadline - time.monotonic() - 200.0)
        if remaining > 60:
            # Run LP in chunks with increasing perturbation
            for chunk in range(4):
                if time.monotonic() > deadline - 150:
                    break
                t_chunk = min(remaining / (4 - chunk) * 0.85, deadline - time.monotonic() - 100)
                if t_chunk < 15:
                    break
                if chunk == 0:
                    a0 = best_sequence.copy()
                else:
                    rng_c = np.random.default_rng(chunk * 777 + 13)
                    scale = 0.003 * (chunk + 1)
                    noise = rng_c.standard_normal(n_best) * scale
                    a0 = np.maximum(best_sequence + noise, 0.0)
                    a0 = 0.5 * (a0 + a0[::-1])
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s
                    else:
                        a0 = best_sequence.copy()
                a_opt, obj_val = optimize_lp(a0, time_limit=t_chunk, use_symmetry=True)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  ExtLP {chunk} (t={t_chunk:.0f}s): {score:.6f} *NEW BEST*")
                else:
                    print(f"  ExtLP {chunk} (t={t_chunk:.0f}s): {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Gradient + LP alternating (200s)
    # =========================================================
    print("=== Phase 4: Alternating Opt ===")
    t_start = time.monotonic()
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = min(200.0, deadline - time.monotonic() - 50.0)
        if remaining > 30:
            # Alternate between gradient and LP
            a_cur = best_sequence.copy()
            for alt in range(6):
                if time.monotonic() > deadline - 40:
                    break
                t_sub = min(remaining / (6 - alt) * 0.8, deadline - time.monotonic() - 30)
                if t_sub < 5:
                    break
                if alt % 2 == 0:
                    a_opt, obj_val = optimize_gradient(a_cur, time_limit=t_sub, lr=0.005,
                                                       mom=0.93, top_k=60, temp=0.002,
                                                       use_symmetry=True)
                else:
                    a_opt, obj_val = optimize_lp(a_cur, time_limit=t_sub, use_symmetry=True)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score):
                    if score < best_score:
                        best_score = score
                        best_sequence = a_opt.copy()
                    a_cur = a_opt if score < best_score else a_cur
                    print(f"  Alt {alt} ({'grad' if alt%2==0 else 'LP'}): {score:.6f}")

    print(f"Phase 4 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Final polishing (remaining time)
    # =========================================================
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None:
        n_best = len(best_sequence)
        remaining = deadline - time.monotonic() - 10.0
        if remaining > 20:
            # Coordinate descent
            t_cd = min(remaining * 0.4, 40.0)
            if t_cd > 5:
                a_opt, obj_val = coordinate_polish(best_sequence, time_limit=t_cd, use_symmetry=True)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  CoordDesc: {score:.6f} *NEW BEST*")

            remaining = deadline - time.monotonic() - 10.0
            if remaining > 15:
                # Short gradient with very fine params
                t_sg = min(remaining * 0.5, 30.0)
                if t_sg > 5:
                    opts_list = [
                        {'lr': 0.003, 'mom': 0.95, 'top_k': 80, 'temp': 0.001},
                        {'lr': 0.008, 'mom': 0.89, 'top_k': 30, 'temp': 0.004},
                        {'lr': 0.005, 'mom': 0.94, 'top_k': 100, 'temp': 0.0015},
                    ]
                    for gi in range(3):
                        if time.monotonic() > deadline - 10:
                            break
                        t_g = min((deadline - time.monotonic() - 10) / (3 - gi) * 0.8, 20.0)
                        if t_g < 3:
                            break
                        cfg = opts_list[gi]
                        a_opt, obj_val = optimize_gradient(best_sequence, time_limit=t_g,
                                                          use_symmetry=True, **cfg)
                        score = float(evaluate_sequence(a_opt.tolist()))
                        if np.isfinite(score) and score < best_score:
                            best_score = score
                            best_sequence = a_opt.copy()
                            print(f"  Final_grad {gi}: {score:.6f} *NEW BEST*")

    # Final verification
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
