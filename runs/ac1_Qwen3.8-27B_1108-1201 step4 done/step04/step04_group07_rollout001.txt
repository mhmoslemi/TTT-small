Minimizes 2*n*max(autoconv(a,a))/(sum(a))^2 using a multi-phase approach:
LP Frank-Wolfe, coordinate descent, spectral shaping, and basin hopping.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def get_objective(f):
    """Get the peak of auto-convolution for a sum-1 sequence."""
    conv = fast_convolve(f)
    return float(conv.max())

def build_conv_matrix_sparse(f):
    """Build convolution matrix B where (B@g)[k] = (f*g)[k]."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B

def solve_peak_lp(f, conv, top_k=40):
    """LP oracle: find direction g that reduces the peak of conv(f, f+eps*g)."""
    n = len(f)
    B = build_conv_matrix_sparse(f)
    
    if top_k < len(conv):
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    else:
        peaks = np.arange(len(conv))
    
    c = np.zeros(n + 1)
    c[0] = 1.0
    
    A_ub = np.zeros((len(peaks), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B[peaks, :]
    b_ub = -conv[peaks]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x[1:]
    return None

def solve_maxsum_lp(f, M, top_k=60):
    """LP oracle: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    B = build_conv_matrix_sparse(f)
    
    c = -np.ones(n)
    
    if top_k < (2 * n - 1):
        # Use all constraints for correctness, but could subset
        A_ub = B
        b_ub = np.full(2 * n - 1, M)
    else:
        A_ub = B
        b_ub = np.full(2 * n - 1, M)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and res.x.sum() > 1e-15:
        return res.x / res.x.sum()
    return None

    rng = np.random.default_rng(999)
    t0 = time.monotonic()
    
    for restart in range(n_restarts):
        if time.monotonic() - t0 > time_limit:
            break
        
        # Make a jump
        jump_scale = 0.01 * (restart + 1) * (0.5 + rng.random())
        f_jump = project_simplex(f + rng.standard_normal(n) * jump_scale)
        
        # Re-optimize
        remaining = time_limit - (time.monotonic() - t0)
        opt_time = max(1.0, remaining * opt_time_frac / max(1, n_restarts - restart))
        
        if opt_fn is not None:
            f_opt, obj = opt_fn(f_jump, opt_time)
        else:
            f_opt = f_jump
            obj = get_objective(f_opt)
        
        if obj < best_obj:
            best_obj = obj
            best_f = f_opt.copy()
            f = f_opt.copy()
        
        # Occam: also try the jumped point directly
        obj_jump = get_objective(f_jump)
        if obj_jump < best_obj:
            best_obj = obj_jump
            best_f = f_jump.copy()
    
    return best_f, best_obj

def generate_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(('uniform', np.ones(n) / n))
    
    # Random Dirichlet
    inits.append(('dirichlet', rng.dirichlet(np.ones(n) * 0.3)))
    
    # Gaussian centered
    x = np.arange(n, dtype=np.float64)
    for sigma in [0.1, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (sigma * n)) ** 2)
        inits.append((f'gauss_{sigma}', g / g.sum()))
    
    # Cosine window
    for freq in [1, 2, 3]:
        c = 0.5 + 0.5 * np.cos(2 * np.pi * freq * x / n)
        c = np.maximum(c, 0.01)
        inits.append((f'cos_{freq}', c / c.sum()))
    
    # Triangle
    tri = 1.0 - 2.0 * np.abs(x / n - 0.5)
    tri = np.maximum(tri, 0.01)
    inits.append(('tri', tri / tri.sum()))
    
    # Two bumps
    for pos in [0.25, 0.33]:
        b = (np.exp(-0.5 * ((x - pos * n) / (0.1 * n)) ** 2) +
             np.exp(-0.5 * ((x - (1-pos) * n) / (0.1 * n)) ** 2))
        inits.append((f'twobump_{pos}', b / b.sum()))
    
    # Sine squared
    s2 = np.sin(np.pi * x / n) ** 2
    s2 = np.maximum(s2, 0.01)
    inits.append(('sinesq', s2 / s2.sum()))
    
    # Ramp
    ramp = np.linspace(0.1, 1.0, n)
    inits.append(('ramp', ramp / ramp.sum()))
    
    # Parent interpolation
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(('parent', px / ps))
    
    return inits
