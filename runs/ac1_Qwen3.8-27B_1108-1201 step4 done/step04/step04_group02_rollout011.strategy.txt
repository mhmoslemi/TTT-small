import numpy as np
from scipy.optimize import minimize
import time

def evaluate_fitness(x):
    n = len(x)
    A = np.fft.rfft(x, n)
    conv_sq = np.fft.irfft(A * A, n)[:2*n-1]
    if np.isnan(conv_sq).any(): return np.inf
    
    # Validate 
    if n < 2: return np.inf
    if not np.all(x >= 0.01): return np.inf
    s = np.sum(x)
    if s < 0.01 or s > 1000: return np.inf
    
    max_val = np.max(conv_sq)
    return 2*n*max_val/(s**2)

def generate_localized_init(length):
    # Only three possible step positions: 0.1, 0.3, 0.5 in unit-space
    # Normalize to uniform first
    base = np.ones(length) / length
    
    # Three variations of localized steps
    def localize_factor(factor, position):
        return np.minimum(0.5, factor) * (position > 0.7) * np.exp(-(position - 0.6)**2 / (2* (factor-0.5)**2))

    # Variation 1: steps at index 10-20% with factor 0.5
    for pos, fact in zip([0.25, 0.35], [1.0, 1.5]):
        for j in range(length):
            if abs(j/length - pos) < 0.01:
                base[j] += localize_factor(fact, np.logspace(-1, 0))

    # Variation 2: steps at 0.9 and 0.1 with amplitude 0.8
    if hasattr(base, "size") and base.size>4:
        base[0] = base[0] * (1 - np.argpartition(base, -1)[-1]*0.05)
        base[3] = base[3] * (1 + np.diff(base[:4]).max())

    # Variation 3: local perturbations
    for i in range(min(length//4, 8)): base[i] *= 0.9

    return base

def main_optimization(f):
    # Goldstein-Zangwill PC algorithm setup
    n = len(f)
    current = f
    best_fitness = evaluate_fitness(current)
    
    # Project to convex hull
    simplex_proj = minimize(lambda x: (-np.sum(x))**2, current, bounds=[(0, None)]*n, method='SLSQP')

    return simplex_proj.x, best_fitness
