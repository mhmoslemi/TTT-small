        sweep += 1
        sweep_improved = False

        for i in range(n):
            if time.monotonic() - t0 > time_limit - 10:
                break

            t_val = a[i]
            di = 2 * i
            b = np.zeros(2 * n - 1)
            js = max(0, di - (n - 1))
            je = min(n - 1, di)
            for j in range(js, je + 1):
                k = di - j
                if 0 <= k < 2 * n - 1:
                    b[k] = t_val

            def f_of_delta(d):
                if d < -t_val + 1e-15:
                    return 1e10
                new_conv = conv - b
                if di < 2 * n - 1:
                    new_conv[di] += d * d
                val = np.max(new_conv)
                S_new = s + d
                if S_new < 0.01:
                    return 1e10
                return val / (S_new * S_new)

            lo = 0.0
            hi = max(5.0 * t_val, 0.02)
            f_lo = f_of_delta(lo)
            f_hi = f_of_delta(hi)
            if f_hi < f_lo:
                for _ in range(8):
                    mid = (lo + hi) / 2.0
                    if f_of_delta(mid) < f_lo:
                        lo = mid
                        f_lo = f_of_delta(lo)
                    else:
                        hi = mid
                hi = min(hi * 1.5, 1.0)

            golden = 0.618033988749895
            c1 = 1.0 - golden
            c2 = golden
            x1 = lo + c1 * (hi - lo)
            x2 = lo + c2 * (hi - lo)
            f1 = f_of_delta(x1)
            f2 = f_of_delta(x2)

            for _ in range(60):
                if f1 < f2:
                    hi = x2
                    x2 = x1
                    f2 = f1
                    x1 = lo + c1 * (hi - lo)
                    f1 = f_of_delta(x1)
                else:
                    lo = x1
                    x1 = x2
                    f1 = f2
                    x2 = lo + c2 * (hi - lo)
                    f2 = f_of_delta(x2)
                if abs(hi - lo) < 1e-12 * max(1.0, abs(lo)):
                    break

            best_d = (lo + hi) / 2.0
            f_best = f_of_delta(best_d)
            f_zero = f_of_delta(0.0)

            if f_best < f_zero - 1e-15:
                a[i] = t_val + best_d
                s += best_d
                conv = conv + 2.0 * best_d * a - b
                if di < 2 * n - 1:
                    conv[di] += best_d * best_d
                current_obj = conv.max() / (s * s)
                if current_obj < best_obj:
                    best_obj = current_obj
                    best_a = a.copy()
                    sweep_improved = True

        if not sweep_improved:
            no_improve_sweeps += 1
            if no_improve_sweeps >= 5:
                break
        else:
            no_improve_sweeps = 0

        if sweep % 50 == 0:
            print(f"    CD sweep {sweep}, obj={best_obj:.8f}, time={time.monotonic()-t0:.0f}s")

    return best_a, best_obj


def lp_escape(f, M, n):
    """Try LP-based Frank-Wolfe step: find g that reduces max convolution."""
    from scipy.optimize import linprog
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            gs = g.sum()
            if gs > 1e-10:
                g = g / gs
                return g
    except Exception:
        pass
    return None


def generate_inits(n, parent=None, rng=None):
    """Generate diverse structured initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())

    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())

    q = (x - n / 2.0) ** 2 + 1.0
    inits.append(q / q.sum())

    e1 = np.exp(-x / (n / 3.0))
    inits.append(e1 / e1.sum())

    c = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())

    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(sn / sn.sum())

    # Gaussian-like
    g = np.exp(-0.5 * ((x - n / 2.0) / (n / 6.0)) ** 2)
    inits.append(g / g.sum())

    # Two-peak
    two_peak = np.exp(-0.5 * ((x - n * 0.25) / (n / 10.0)) ** 2) + \
               np.exp(-0.5 * ((x - n * 0.75) / (n / 10.0)) ** 2) + 0.01
    inits.append(two_peak / two_peak.sum())

    # Random
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Blend
            for blend in [0.3, 0.5, 0.7]:
                bl = blend * (px / ps) + (1 - blend) * (np.ones(n) / n)
                inits.append(bl / bl.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search combining L-BFGS-B smoothed optimization, coordinate descent,
    and multi-start exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Intensive optimization on parent ===
    print("=== Phase 1: Parent Optimization (L-BFGS-B + CD) ===")
    if parent is not None:
        n_p = len(parent)

        # L-BFGS-B with annealing
        rem = deadline - time.monotonic() - 600
        if rem > 60:
            print(f"  L-BFGS-B on parent (n={n_p}), time={min(rem, 400):.0f}s")
            a_opt, _ = lbfgs_optimize(parent, time_limit=min(rem, 400),
                                      T_start=300.0, T_min=30.0)
            try_seq(a_opt, "parent_lbfgs")

            # Coordinate descent polishing
            rem2 = deadline - time.monotonic() - 400
            if rem2 > 30:
                a_opt2, _ = coord_descent(a_opt, time_limit=min(rem2, 200))
                try_seq(a_opt2, "parent_cd")

        # LP escape from best
        rem3 = deadline - time.monotonic() - 300
        if rem3 > 30 and best_seq is not None:
            b_norm = best_seq / best_seq.sum()
            nn = len(b_norm)
            conv_b = fast_convolve(b_norm)
            M_b = conv_b.max()
            g = lp_escape(b_norm, M_b * 1.01, nn)
            if g is not None:
                for alpha in [0.1, 0.2, 0.3, 0.5, 0.7]:
                    ft = project_simplex((1 - alpha) * b_norm + alpha * g)
                    try_seq(ft, f"parent_lp_{alpha}")

    # === Phase 2: Multi-length exploration ===
    print(f"\n=== Phase 2: Multi-Length Exploration ===")
    print(f"  Current best: {best_score:.8f}")

    lengths_to_try = [384, 512, 256]
    if parent is not None:
        if len(parent) in lengths_to_try:
            lengths_to_try.remove(len(parent))
        lengths_to_try.insert(0, len(parent))

    # Remove duplicates and Already-done
    seen_lengths = set()
    final_lengths = []
    for L in lengths_to_try:
        if L not in seen_lengths:
            seen_lengths.add(L)
            final_lengths.append(L)

    for n in final_lengths:
        if time.monotonic() > deadline - 200:
            break

        inits = generate_inits(n, parent, rng)
        rem = deadline - time.monotonic() - 100
        n_init = len(inits)

        # Keep only top initializations based on initial objective
        scored_inits = []
        for f_init in inits:
            fc = fast_convolve(f_init).max()
            scored_inits.append((fc, f_init))
        scored_inits.sort(key=lambda x: x[0])
        scored_inits = scored_inits[:min(5, len(scored_inits))]

        for i, (init_obj, f_init) in enumerate(scored_inits):
            if time.monotonic() > deadline - 80:
                break
            t_alloc = max(20, (deadline - time.monotonic() - 60) / max(1, len(scored_inits) - i))
            t_alloc = min(t_alloc, 120)
            if t_alloc < 15:
                break

            # Split time between L-BFGS-B and CD
            t_lbfgs = int(t_alloc * 0.6)
            t_cd = int(t_alloc * 0.4)

            a_opt, _ = lbfgs_optimize(f_init, time_limit=t_lbfgs,
                                      T_start=200.0, T_min=30.0)
            if t_cd > 10:
                a_opt2, _ = coord_descent(a_opt, time_limit=t_cd)
                try_seq(a_opt2, f"n{n}_i{i}_cd")
            else:
                try_seq(a_opt, f"n{n}_i{i}")

    # === Phase 3: Deep optimization on best ===
    print(f"\n=== Phase 3: Deep Optimization ===")
    print(f"  Current best: {best_score:.8f}")

    if best_seq is not None:
        n_best = len(best_seq)
        b_norm = best_seq / best_seq.sum()

        # Multiple L-BFGS-B restarts with different T schedules
        for p in range(3):
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            t_alloc = min(int(rem / (3 - p)), 150)

            if p == 0:
                f_init = b_norm.copy()
                T_s, T_m = 100.0, 20.0
            elif p == 1:
                f_init = project_simplex(b_norm + rng.standard_normal(n_best) * 0.003)
                T_s, T_m = 50.0, 15.0
            else:
                f_init = project_simplex(0.9 * b_norm + 0.1 * (best_seq[::-1] / best_seq.sum()))
                T_s, T_m = 80.0, 20.0

            a_opt, _ = lbfgs_optimize(f_init, time_limit=t_alloc,
                                      T_start=T_s, T_min=T_m)
            # Follow with CD
            rem2 = deadline - time.monotonic() - 40
            if rem2 > 20:
                a_opt2, _ = coord_descent(a_opt, time_limit=min(rem2, 60))
                try_seq(a_opt2, f"deep_{p}_cd")
            else:
                try_seq(a_opt, f"deep_{p}")

        # LP escape attempts
        for p in range(3):
            rem = deadline - time.monotonic() - 30
            if rem < 15:
                break
            b_norm = best_seq / best_seq.sum()
            nn = len(b_norm)
            conv_b = fast_convolve(b_norm)
            M_b = conv_b.max()
            g = lp_escape(b_norm, M_b * (1.005 + 0.005 * p), nn)
            if g is not None:
                for alpha in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    ft = project_simplex((1 - alpha) * b_norm + alpha * g)
                    sc = try_seq(ft, f"lp_deep_{p}_{alpha}")
                    if sc < best_score:
                        break

    # === Phase 4: Final CD polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 10
        if rem > 15:
            b_norm = best_seq / best_seq.sum()
            a_final, _ = coord_descent(b_norm, time_limit=min(rem, 120))
            try_seq(a_final, "final_cd")

        # Micro perturbations + CD
        rem = deadline - time.monotonic() - 5
        if rem > 10:
            n_best = len(best_seq)
            b_norm = best_seq / best_seq.sum()
            for _ in range(3):
                rem = deadline - time.monotonic() - 5
                if rem < 5:
                    break
                f_pert = project_simplex(b_norm + rng.standard_normal(n_best) * 0.001)
                a_micro, _ = coord_descent(f_pert, time_limit=min(rem - 2, 20))
                try_seq(a_micro, "micro_cd")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
