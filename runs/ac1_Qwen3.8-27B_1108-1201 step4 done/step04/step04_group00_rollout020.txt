import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def objective(a):
    """Compute the objective value (without calling evaluate_sequence)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def compute_subgradient(a, conv, top_k, temp):
    """Compute weighted subgradient of max(conv)."""
    n = len(a)
    M = conv.max()
    # Select top-k peaks
    k_actual = min(top_k, len(conv))
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    pv = conv[peaks]
    # Softmax weights
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    w /= ws
    
    # Compute subgradient: g[j] = 2 * sum_{k in peaks} w[k_idx] * a[k-j]
    g = np.zeros(n)
    for idx in range(k_actual):
        k = int(peaks[idx])
        wi = w[idx]
        if wi < 1e-15:
            continue
        # g[j] += 2 * wi * a[k-j] for valid j
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            # a[k-j] for j in [js, je]
            # When j = js: a[k-js] = a[min(n-1, k)] (largest index)
            # When j = je: a[k-je] = a[max(0, k-n+1)] (smallest index)
            # So a[k-j] for j in [js,je] is a[je..js] (reversed)
            indices = np.arange(k - je, k - js + 1)
            g[js:je+1] += 2.0 * wi * a[indices]
    
    return g

def lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[max(0,k-n+1)..min(n-1,k)] shifted
    # (f*g)[k] = sum_j f[j] * g[k-j] for valid j,k-j
    # = sum_j g[j] * f[k-j] for valid j,k-j
    # So A_ub[k, j] = f[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A_ub[k, j] = f[k - j]
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        g = lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                break
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            f = f + best_alpha * (g - f)
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"      LP-FW {it}: {best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 17 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit):
    """Coordinate descent polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    
    coord_order = list(range(n))
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(int(time.monotonic() * 1000) % (2**31))
        rng.shuffle(coord_order)
        
        for j in coord_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            if old_val < 1e-14:
                continue
            
            best_j_val = old_val
            best_j_obj = fast_convolve(f).max()
            
            for mult in [0.5, 0.7, 0.85, 1.0, 1.2, 1.4, 1.8]:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-12:
                    continue
                scale = (1.0 - new_val) / denom
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = f_trial / f_trial.sum()
                
                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-12:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = f / f.sum()
                
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_f, best_obj

def make_ramp(n, power=1.0):
    """Create a ramp sequence: a[i] ∝ (i+1)^power, normalized."""
    a = np.arange(1, n + 1, dtype=np.float64) ** power
    a = a / a.sum()
    return a

def make_bimodal(n, peak_frac=0.3):
    """Create a bimodal sequence with peak in the middle."""
    a = np.zeros(n)
    center = n // 2
    width = int(n * peak_frac)
    a[center - width//2:center + width//2 + 1] = 1.0
    a = a / a.sum()
    return a

def make_exponential(n, rate=0.5):
    """Create an exponential ramp: a[i] ∝ exp(rate * i / n)."""
    a = np.exp(rate * np.arange(n) / n)
    a = a / a.sum()
    return a

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
