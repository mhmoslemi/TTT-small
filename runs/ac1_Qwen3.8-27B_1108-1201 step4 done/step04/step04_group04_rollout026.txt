Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Improvements over previous: CVXPY LP, momentum-augmented gradient,
peak-directed QP, better annealing, diverse multi-n exploration.

import numpy as np
from scipy.optimize import linprog
import time

try:
    import cvxpy as cp
    HAS_CVXPY = True
except ImportError:
    HAS_CVXPY = False


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def fast_cross_convolve(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        A[k_start:k_end + 1, j] = f
    return A

    A = build_conv_matrix(f)
    g = cp.Variable(n)
    constraints = [A @ g <= M, g >= 0]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.GLOP)
        if g.value is not None:
            return np.maximum(g.value, 0)
    except Exception:
        pass
    try:
        prob.solve(solver=cp.HIGHS)
        if g.value is not None:
            return np.maximum(g.value, 0)
    except Exception:
        pass
    return None


def solve_lp_scipy(f, M):
    """Solve LP via scipy linprog."""
    n = len(f)
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return np.maximum(res.x, 0)
    return None


def solve_lp_dir(f, M):
    """Solve LP direction with best available solver."""
    if HAS_CVXPY:
        result = solve_lp_cvxpy(f, M)
        if result is not None and result.sum() > 1e-15:
            return result
    return solve_lp_scipy(f, M)


def smoothed_gradient_fft(conv, a, T):
    """Gradient of T*log(sum(exp(conv/T))) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws
    # grad[i] = 2 * sum_k p[k] * (contribution of a[i] to conv[k])
    # = 2 * cross_correlation(p, a)[i]  (with appropriate definition)
    # conv[k] = sum_j a[j]*a[k-j], d/da[i] = a[k-i] (if valid) + a[k-i] (if k-i valid and != i)
    # Simplified: grad[i] ≈ 2 * sum_k p[k] * a[k-i] (with boundary handling via cross-corr)
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    # cross[i] = sum_k p[k] * a[k-i] (this is the correlation)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]


def fw_lp_optimize(f0, time_limit, verbose=False, use_cvxpy=True):
    """LP-based Frank-Wolfe with momentum and adaptive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    momentum = np.zeros(n)
    eta = 0.4  # momentum coefficient for direction smoothing

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        g = solve_lp_dir(f, M)
        if g is None or g.sum() < 1e-15:
            no_improve += 1
            if no_improve > 30:
                break
            continue

        g = g / g.sum()

        # Blend with momentum
        g_smooth = (1 - eta) * g + eta * momentum
        g_smooth = np.maximum(g_smooth, 0)
        gs = g_smooth.sum()
        if gs < 1e-15:
            g_smooth = g
            gs = 1.0
        g_smooth = g_smooth / gs
        momentum = g_smooth

        if np.max(np.abs(g_smooth - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g_smooth
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine bisection
        if best_alpha > 0.02:
            lo = max(0.0, best_alpha - 0.06)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g_smooth
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g_smooth
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.12f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 30:
            rng = np.random.default_rng((it * 7919 + n * 31) % (2**31))
            scale = 0.001 + 0.002 * min(no_improve // 30, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
            momentum = np.zeros(n)

    return best_f, best_obj


def smoothed_grad_descent(f0, time_limit, T_init=0.002, T_min=0.00005,
                          lr_init=0.005, verbose=False):
    """Gradient descent on simplex with momentum and adaptive smoothing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    T = T_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = lr_init
    velocity = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros(n)
    v = np.zeros(n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        grad = smoothed_gradient_fft(conv, f, T)
        grad = grad - np.mean(grad)

        # Adam-like update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        update = m_hat / (np.sqrt(v_hat) + 1e-8)

        f_new = project_simplex(f - lr * update)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    SGD iter {it}: obj={best_obj:.12f}, T={T:.7f}, lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve % 50 == 0:
                lr *= 0.92

        T = max(T * 0.9985, T_min)

        if no_improve > 400:
            rng = np.random.default_rng((it * 331 + 17 + n) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0008)
            lr = max(lr * 2.0, lr_init * 0.5)
            T = max(T_init * 0.2, T_min)
            m = np.zeros(n)
            v = np.zeros(n)
            no_improve = 0

    return best_f, best_obj


def peak_attack(f0, time_limit, verbose=False):
    """Focused attack on the worst peak using LP directions targeting that peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        M = conv.max()
        # Find the top peaks
        top_k = max(3, n // 20)
        peak_indices = np.argsort(conv)[-top_k:]
        peak_vals = conv[peak_indices]
        max_peak_idx = int(np.argmax(conv))

        # Solve LP with tighter constraint on the worst peak
        # We want to reduce conv(f,g)[max_peak_idx] specifically
        # Use a modified M that's tighter at the peak
        M_tight = M * 0.995  # slightly tighter overall
        g = solve_lp_dir(f, M_tight)

        if g is None:
            # Try even tighter
            M_tight = M * 0.99
            g = solve_lp_dir(f, M_tight)
        if g is None or g.sum() < 1e-15:
            no_improve += 1
            if no_improve > 20:
                break
            continue

        g = g / g.sum()

        # Check if this direction reduces the worst peak
        cross = fast_cross_convolve(f, g)
        peak_change = cross[max_peak_idx] - M

        if peak_change >= -1e-10:
            # Direction doesn't help the peak, try a different approach
            # Directly reduce mass at positions contributing to the peak
            k = max_peak_idx
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            contrib = np.zeros(n)
            for j in range(js, je + 1):
                contrib[j] = f[k - j]
            total_contrib = contrib.sum()
            if total_contrib > 1e-15:
                # Move mass from high-contributing positions to low ones
                weights = contrib / total_contrib
                low_mask = np.argsort(contrib)[:n // 4]
                target = np.zeros(n)
                target[low_mask] = 1.0 / len(low_mask)
                alpha = 0.02
                f_new = f + alpha * (target - weights)
                f_new = np.maximum(f_new, 0)
                f_new = f_new / f_new.sum()
            else:
                f_new = f.copy()
        else:
            alpha = 0.05
            f_new = (1 - alpha) * f + alpha * g
            f_new = np.maximum(f_new, 0)
            f_new = f_new / f_new.sum()

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 100 == 0:
                print(f"    Peak attack iter {it}: obj={best_obj:.12f}")
        else:
            no_improve += 1

        if no_improve > 25:
            rng = np.random.default_rng((it * 101 + 7) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0005)
            no_improve = 0

    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr=0.001, top_k_frac=0.05, temp=0.00002):
    """Subgradient polish focusing on top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    top_k = max(3, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        f = project_simplex(f - lr * sg)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 100 == 0:
            lr *= 0.995
            temp *= 1.005

        if no_improve > 200:
            rng = np.random.default_rng((it * 733 + 41 + n) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.0004)
            lr = 0.0005
            no_improve = 0

    return best_f, best_obj


def mirror_symmetrize(f):
    """Average f with its reverse and re-normalize."""
    mirrored = 0.5 * (f + f[::-1])
    s = mirrored.sum()
    if s < 1e-15:
        return f
    return mirrored / s


def spectral_iteration(f0, time_limit, verbose=False):
    """Gerchberg-Saxton style spectral iteration to flatten the autocorrelation."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    fft_size = 1 << (2 * n).bit_length()
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Forward: compute spectrum
        F = np.fft.rfft(f, fft_size)
        power = np.abs(F) ** 2

        # Target: flatten the power spectrum (keep DC, flatten the rest)
        dc = power[0]
        # Blend towards flat
        target = np.ones_like(power) * (power.sum() - dc) / max(len(power) - 1, 1)
        target[0] = dc
        # Geometric mean (smooth flattening)
        blend = 0.3
        new_power = (1 - blend) * power + blend * target
        new_power = np.maximum(new_power, 0)

        # Inverse: reconstruct with new amplitude, keep phase
        phase = np.angle(F)
        F_new = np.sqrt(new_power) * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, fft_size)[:n]

        # Project to simplex
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            if verbose and it % 200 == 0:
                print(f"    Spectral iter {it}: obj={best_obj:.12f}")
        else:
            no_improve += 1
            # Try with less blending
            if no_improve % 100 == 0:
                blend *= 0.9

        if no_improve > 300:
            break

    return best_f, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.03
    inits.append(c1 / c1.sum())

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.08 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.08 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())

    # Triangular
    tri = 1.0 - np.abs(x - 0.5 * (n - 1)) / (0.5 * n)
    tri = np.maximum(tri, 0.01)
    inits.append(tri / tri.sum())

    # W-shape (4 peaks)
    centers = [0.125, 0.375, 0.625, 0.875]
    w_shape = np.zeros(n)
    for c in centers:
        w_shape += np.exp(-0.5 * ((x - c * n) / (0.05 * n)) ** 2)
    inits.append(w_shape / w_shape.sum())

    # Mirror-symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.4)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[::-1]
    inits.append(sym / sym.sum())

    # Mirror-antisymmetric-ish (high in second half)
    half = rng.dirichlet(np.ones(half_n) * 0.5)
    asy = np.zeros(n)
    asy[half_n:] = half
    inits.append(asy / asy.sum())

    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            # Noisy version of parent
            px_noisy = np.maximum(px + rng.standard_normal(n) * 0.01, 0)
            pxs = px_noisy.sum()
            if pxs > 1e-10:
                inits.append(px_noisy / pxs)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search combining LP-FW, smoothed gradient,
    spectral iterations, peak attack, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(p.tolist()):.8f}")
    except Exception:
        pass

    # === Phase 1: Exploration (250s) ===
    print("=== Phase 1: Exploration ===")
    t_p1 = time.monotonic()
    p1_budget = 250.0

    base_n = len(parent) if parent is not None else 384
    all_results = []

    # Optimize parent
    if parent is not None:
        t_alloc = min(80.0, p1_budget - 50.0)
        if t_alloc > 20:
            print(f"  FW on parent n={base_n}, t={t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(parent, time_limit=t_alloc, verbose=True)
            all_results.append((a_opt, obj))
            try_seq(a_opt, "parent_fw")

        # Spectral on parent
        rem = p1_budget - (time.monotonic() - t_p1)
        if rem > 40:
            t_alloc = min(30.0, rem - 20.0)
            print(f"  Spectral on parent, t={t_alloc:.0f}s")
            a_opt, obj = spectral_iteration(parent, time_limit=t_alloc, verbose=True)
            all_results.append((a_opt, obj))
            try_seq(a_opt, "parent_spectral")

    # Try other lengths
    lengths_to_try = []
    for delta in [-16, -8, -4, 4, 8, 16, 32, 64, -32]:
        cn = base_n + delta
        if 128 <= cn <= 1024 and cn not in lengths_to_try:
            lengths_to_try.append(cn)

    rng.shuffle(lengths_to_try)
    for n in lengths_to_try[:4]:
        rem = p1_budget - (time.monotonic() - t_p1)
        if rem < 30:
            break
        t_alloc = min(35.0, rem - 15.0)
        if t_alloc < 15:
            break

        inits = generate_diverse_inits(n, parent, rng)
        # Quick evaluation to pick best init
        best_init = None
        best_init_obj = float('inf')
        for fi in inits[:5]:
            c = fast_convolve(fi)
            o = c.max()
            if o < best_init_obj:
                best_init_obj = o
                best_init = fi

        if best_init is not None:
            print(f"  FW n={n}, t={t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(best_init, time_time if False else time_limit=t_alloc)
            all_results.append((a_opt, obj))
            try_seq(a_opt, f"n{n}_fw")

    print(f"  Phase 1 done. Best score: {best_score:.8f}")

    # === Phase 2: Deep Optimization (1100s) ===
    print("\n=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()

        n_best = len(f_work)
