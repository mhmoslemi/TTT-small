            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 15 * perturb_count)
            relax_factor = min(1.5, relax_factor + 0.1)
            T = max(T, 0.001)

            if perturb_count > 50:
                break

    return best_a, best_obj


def coordinate_descent(a0, time_limit):
    """Coordinate descent: try moving mass between pairs of elements."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        for _ in range(200):
            if time.monotonic() - t0 > time_limit - 0.5:
                break
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            found = False
            for delta in [0.002, 0.005, 0.01, 0.02, -0.002, -0.005, -0.01, -0.02]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    found = True
                    break
            if not found:
                no_improve += 1

        if no_improve > 500:
            break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random1'] = rng.dirichlet(np.ones(n) * 0.5)
    inits['random2'] = rng.dirichlet(np.ones(n) * 0.1)

    s1 = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine1'] = s1 / s1.sum()
    s2 = np.sin(2 * np.pi * x / n) ** 2 + 0.01
    inits['sine2'] = s2 / s2.sum()
    s3 = np.sin(0.5 * np.pi * x / n) ** 2 + 0.01
    inits['sine_half'] = s3 / s3.sum()

    # Bimodal
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 2.0
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Gaussian centered
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.15 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Triangular
    tri = np.abs(x - n / 2) / (n / 2)
    tri = 1.0 - tri + 0.01
    inits['triangle'] = tri / tri.sum()

    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point: hybrid multi-peak LP + SA + gradient + coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_sequence = None
    best_score = float('inf')

    # Load parent
    parent = None
    n_target = 768
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            n_target = len(p)
            print(f"Parent loaded: n={n_target}, sum={p.sum():.4f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(n_target) / n_target

    # =========================================================
    # Phase 1: Quick gradient scan (150s)
    # =========================================================
    print("=== Phase 1: Quick Gradient Scan ===")
    t_start = time.monotonic()
    phase1_limit = 150.0

    inits = generate_inits(n_target, parent, rng)
    priority = ['parent', 'parent_interp', 'sine1', 'uniform', 'alternating', 'random1']
    init_names = [nm for nm in priority if nm in inits]
    init_names += [nm for nm in inits if nm not in priority]
    init_names = init_names[:5]

    scan_results = []
    t_per = max(15.0, (phase1_limit - 10) / len(init_names))

    for name in init_names:
        if time.monotonic() > deadline - 1400:
            break
        t_alloc = min(t_per, deadline - time.monotonic() - 1300)
        if t_alloc < 5:
            break
        a0 = inits[name]
        a_opt, obj = nesterov_opt(a0, time_limit=t_alloc, lr_init=0.015,
                                   momentum=0.92, top_k=40, temp_init=0.002)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            scan_results.append((score, name, a_opt.copy()))
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Grad {name}: {score:.6f}")

    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 2: Deep LP+SA optimization on best (1300s)
    # =========================================================
    print("=== Phase 2: Deep LP+SA Optimization ===")
    t_start2 = time.monotonic()

    # Take top 2-3 candidates from scan + parent
    candidates = []
    if best_sequence is not None:
        candidates.append(best_sequence.copy())
    if parent is not None:
        pc = np.maximum(parent, 0).astype(np.float64)
        pcs = pc.sum()
        if pcs > 1e-10:
            candidates.append(pc / pcs)
    for score, name, a_opt in scan_results[:2]:
        if name not in ['parent']:
            candidates.append(a_opt.copy())

    # Deduplicate
    unique_cands = [candidates[0]]
    for c in candidates[1:]:
        if np.linalg.norm(c - unique_cands[0]) > 0.01 * len(c):
            unique_cands.append(c)
        if len(unique_cands) >= 3:
            break

    remaining = deadline - time.monotonic() - 200.0
    n_cands = len(unique_cands)
    t_per_lp = max(60.0, remaining * 0.7 / n_cands)

    for i, a0 in enumerate(unique_cands):
        if time.monotonic() > deadline - 180:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 150)
        if t_alloc < 20:
            break
        print(f"  LP+SA candidate {i}: n={len(a0)}, t_alloc={t_alloc:.0f}s")
        a_opt, obj = lp_sa_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP+SA {i}: {score:.6f} *NEW BEST*")
            else:
                print(f"  LP+SA {i}: {score:.6f}")

    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 3: Second round of LP+SA on best (200s)
    # =========================================================
    print("=== Phase 3: LP+SA Second Round ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 100.0
        if remaining > 50:
            # Try with slight perturbation
            for pi in range(2):
                if time.monotonic() > deadline - 60:
                    break
                t_alloc = min(remaining / (2 - pi) * 0.8, deadline - time.monotonic() - 50)
                if t_alloc < 15:
                    break
                if pi == 0:
                    a0 = best_sequence.copy()
                else:
                    noise = rng.standard_normal(len(best_sequence)) * 0.002
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s
                    else:
                        a0 = best_sequence.copy()

                a_opt, obj = lp_sa_optimize(a0, time_limit=t_alloc)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Round2 LP+SA {pi}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Round2 LP+SA {pi}: {score:.6f}")

    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 4: Gradient + coordinate descent polish (80s)
    # =========================================================
    print("=== Phase 4: Polish ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0

        # Gradient polish
        t_grad = min(remaining * 0.4, 40.0)
        if t_grad > 5:
            cfgs = [
                {'lr_init': 0.005, 'momentum': 0.93, 'top_k': 60, 'temp_init': 0.001},
                {'lr_init': 0.01, 'momentum': 0.88, 'top_k': 30, 'temp_init': 0.003},
                {'lr_init': 0.003, 'momentum': 0.95, 'top_k': 100, 'temp_init': 0.0005},
            ]
            for gi in range(3):
                if time.monotonic() > deadline - 15 or t_grad < 5:
                    break
                cfg = cfgs[gi % len(cfgs)]
                t_alloc = min(t_grad / (3 - gi) * 0.9, t_grad)
                a_opt, obj = nesterov_opt(best_sequence, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Grad polish {gi}: {score:.6f} *NEW BEST*")
                t_grad -= t_alloc

        # Coordinate descent
        remaining = deadline - time.monotonic() - 15.0
        if remaining > 5:
            t_cd = min(remaining * 0.6, 30.0)
            a_opt, obj = coordinate_descent(best_sequence, time_limit=t_cd)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  CoordDesc: {score:.6f} *NEW BEST*")

    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Final micro LP (10s)
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 5:
        t_final = min(15.0, deadline - time.monotonic() - 3)
        if t_final > 3:
            a_opt, obj = lp_sa_optimize(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final: {score:.6f} *NEW BEST*")

    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(n_target) / n_target

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
