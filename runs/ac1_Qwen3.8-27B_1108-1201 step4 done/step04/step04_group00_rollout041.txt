import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_flatten(a, time_limit, lr_init=0.5, cooling=0.9999):

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()

    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        A = np.fft.fft(a, fft_size)
        C = A * A  # This is FFT of the zero-padded auto-convolution
        c = np.fft.ifft(C, fft_size)[:size]
        
        target = c[size//2]  # Use center value as target (or mean)
        # Actually, use mean as target
        target = c.mean()
        
        # Compute excess
        excess = np.maximum(c - target, 0)
        
        if excess.max() < 1e-15:
            break
        
        # Modify C in frequency domain to reduce excess
        # E = FFT(excess_padded) where excess is zero-padded to fft_size
        excess_padded = np.zeros(fft_size)
        excess_padded[:size] = excess
        E = np.fft.fft(excess_padded)

        C_new = np.maximum(C_new.real, 0) + 1j * C_new.imag

        C_new = 0.5 * (C_new + np.conj(C_new[np.mod(fft_size - np.arange(fft_size), fft_size)]))

        C_new = np.maximum(C_new, 1e-20 + 0j)  # avoid zero
        # For square root of complex numbers with conjugate symmetry
        A_new = np.sqrt(C_new)

        A_new[0] = np.abs(A_new[0])  # DC is real

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_shaping(a, time_limit, lr_init=0.3):
    """
    Spectral shaping: modify the amplitude of the FFT to flatten the auto-convolution.
    Key: we work with A^2 (not |A|^2) since we're doing auto-convolution.
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        A = np.fft.fft(a, fft_size)
        C = A * A
        c = np.fft.ifft(C, fft_size)[:size].real
        
        # We want to reduce the peaks in c
        target = c.mean()
        excess = np.maximum(c - target, 0)
        
        if excess.max() < 1e-14:
            break
        
        # Compute the "force" in frequency domain
        # We want to reduce c where it's above target
        # d(c)/dC = IFFT(1) ... actually c = IFFT(C), so dc/dC[m] = exp(2πi*m*k/fft_size)
        # To reduce c at peak positions, we should reduce C at all frequencies
        # weighted by the conjugate of the DFT basis at those positions.
        
        # Simpler: compute E = FFT of the excess (zero-padded)
        exc = np.zeros(fft_size)
        exc[:size] = excess
        E = np.fft.fft(exc)
        
        # C_new = C - lr * E
        C_new = C - lr * E
        
        # Enforce conjugate symmetry (C should be Hermitian since c is real)
        C_new[0] = C_new[0].real
        if fft_size % 2 == 0:
            C_new[fft_size // 2] = C_new[fft_size // 2].real
        for m in range(1, fft_size // 2 + 1):
            C_new[fft_size - m] = np.conj(C_new[m])
        
        # Take square root to get new A
        # Use polar form to handle complex square roots
        magnitude = np.abs(C_new)
        phase = np.angle(C_new)
        
        A_new_mag = np.sqrt(np.maximum(magnitude, 1e-30))
        A_new_phase = phase / 2.0
        
        A_new = A_new_mag * np.exp(1j * A_new_phase)
        
        # Enforce conjugate symmetry on A_new
        A_new[0] = np.abs(A_new[0])
        if fft_size % 2 == 0:
            A_new[fft_size // 2] = np.abs(A_new[fft_size // 2])
        for m in range(1, fft_size // 2 + 1):
            A_new[fft_size - m] = np.conj(A_new[m])
        
        a_new = np.fft.ifft(A_new, fft_size)[:n].real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-15:
            lr *= 0.5
            no_improve += 10
            continue
        a_new = a_new / s
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        lr *= 0.999
        if no_improve > 50:
            lr = max(lr * 1.5, lr_init * 0.3)
            no_improve = 0
    
    return best_a, best_obj

def spectral_amplitude_method(a, time_limit, lr_init=0.5, cooling_rate=0.9998):
    """
    Novel spectral method: iteratively modify the amplitude of FFT(a)
    to flatten the auto-convolution. Works by:
    1. Computing C = FFT(a)^2 (which is FFT of the auto-convolution)
    2. Identifying peaks in the time-domain convolution
    3. Reducing C at frequencies that contribute to those peaks
    4. Taking sqrt of modified C to get new FFT
    5. Converting back to time domain, clipping negatives, normalizing
    """
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    
    a = normalize_nonneg(a)
    best_a = a.copy()
    conv = fast_convolve(a)
    best_obj = conv.max()
    
    t0 = time.monotonic()
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute current convolution and identify excess
        conv = fast_convolve(a)
        target = conv.mean()
        excess = np.maximum(conv - target, 0)
        
        if excess.max() < 1e-15:
            break
        
        # Compute the frequency-domain "force" to reduce excess
        exc_fft = np.fft.fft(excess, fft_size)
        
        # Get current spectrum
        A = np.fft.fft(a, fft_size)
        C = A * A
        
        # Modify C
        C_new = C - lr * exc_fft
        
        # Enforce Hermitian symmetry (C must be FFT of real signal)
        C_new[0] = C_new[0].real
        if fft_size % 2 == 0:
            C_new[fft_size // 2] = C_new[fft_size // 2].real
        for m in range(1, fft_size // 2 + 1):
            C_new[fft_size - m] = np.conj(C_new[m])
        
        # Take square root (polar decomposition)
        mag = np.abs(C_new)
        phase = np.angle(C_new)
        
        A_new_mag = np.sqrt(np.maximum(mag, 1e-30))
        A_new_phase = 0.5 * phase
        
        A_new = A_new_mag * np.exp(1j * A_new_phase)
        
        # Enforce conjugate symmetry on A_new
        A_new[0] = np.abs(A_new[0])
        if fft_size % 2 == 0:
            A_new[fft_size // 2] = np.abs(A_new[fft_size // 2])
        for m in range(1, fft_size // 2 + 1):
            A_new[fft_size - m] = np.conj(A_new[m])
        
        # Convert back to time domain
        a_new = np.fft.ifft(A_new, fft_size)[:n].real
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:

            no_improve += 5

            continue
        a_new = a_new / s

        if obj < best_obj - 1e-14:

            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        lr *= cooling_rate
        
        if no_improve > 100:

            lr = min(lr * 2.0, lr_init)
            no_improve = 0
    
    return best_a, best_obj

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:

                    best_obj = obj
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break

        if not improved:
            no_improve += 1
            if no_improve > 30:
                break
        else:
            no_improve = 0

    return best_a, best_obj


def single_coord_descent(a, time_limit, multipliers=None):
    """Standard single-coordinate descent with multiplier search."""
    if multipliers is None:
        multipliers = [0.4, 0.6, 0.8, 0.9, 1.1, 1.3, 1.6]

    n = len(a)
    a = normalize_nonneg(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    t0 = time.monotonic()
    no_improve_passes = 0

    while time.monotonic() - t0 < time_limit:
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[j]
            if old_val < 1e-14:
                continue

            best_j_val = old_val
            best_j_obj = best_obj

            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val > 0.99:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = np.maximum(a_trial, 0)
                ts = a_trial.sum()
                if ts < 1e-10:
                    continue
                a_trial /= ts

                obj = fast_convolve(a_trial).max()
                if obj < best_j_obj - 1e-15:
                    best_j_obj = obj
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = np.maximum(a, 0)
                    a /= a.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True

        if not improved:
            no_improve_passes += 1
            if no_improve_passes > 3:
                break
        else:
            no_improve_passes = 0

    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Linearly interpolate a sequence to a new length and normalize."""
    old_n = len(f)
    if old_n == new_n:
        return normalize_nonneg(f)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    return normalize_nonneg(f_interp)


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-strategy optimization combining spectral methods, subgradient descent,
    pairwise coordinate descent, and single-coordinate descent with multi-start.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    # Get parent sequence
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(384) / 384
        print(f"Using default parent: n={len(parent)}")

    parent_n = len(parent)
    best_seq, best_score = try_and_update(parent, best_seq, best_score, "parent")

    # Phase 1: Quick spectral + subgradient on parent
    print("\n=== Phase 1: Quick optimization on parent ===")
    t1 = min(150, (deadline - time.monotonic()) * 0.12)
    if t1 > 30:
        a_spec, _ = spectral_amplitude_method(parent, time_limit=t1 * 0.5, lr_init=0.5)
        best_seq, best_score = try_and_update(a_spec, best_seq, best_score, "spec_p1")

        base = best_seq if best_seq is not None and len(best_seq) == parent_n else parent
        a_sg, _ = subgradient_optimize(base, time_limit=t1 * 0.5, n_peaks=10,
                                        lr_init=0.004, momentum=0.92,
                                        temp_init=0.005, temp_growth=1.002)
        best_seq, best_score = try_and_update(a_sg, best_seq, best_score, "sg_p1")

    # Phase 2: Explore different lengths
    print("\n=== Phase 2: Length exploration ===")
    candidate_lengths = []
    for dn in [-128, -64, 0, 64, 128, 256, 384]:
        n_try = parent_n + dn
        if 192 <= n_try <= 2048 and n_try not in candidate_lengths:
            candidate_lengths.append(n_try)
    candidate_lengths.sort()
    candidate_lengths = [n for n in candidate_lengths if n != parent_n]
    print(f"  Testing lengths: {candidate_lengths}")

    for n_try in candidate_lengths:
        rem = deadline - time.monotonic() - 80
        if rem < 40:
            break

        a_init = interpolate_sequence(parent, n_try)
        t_alloc = min(rem * 0.25, 120)

        a_spec, _ = spectral_amplitude_method(a_init, time_limit=t_alloc * 0.5, lr_init=0.4)
        best_seq, best_score = try_and_update(a_spec, best_seq, best_score, f"spec_n{n_try}")

        a_sg, _ = subgradient_optimize(a_spec, time_limit=t_alloc * 0.5, n_peaks=10,
                                        lr_init=0.003, momentum=0.92,
                                        temp_init=0.004, temp_growth=1.002)
        best_seq, best_score = try_and_update(a_sg, best_seq, best_score, f"sg_n{n_try}")

    # Phase 3: Deep optimization on best sequence
    print(f"\n=== Phase 3: Deep optimization (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        strategies = []
        # Strategy 0: direct
        strategies.append(("direct", best_seq.copy()))
        # Strategy 1: small perturbation
        strategies.append(("perturb_small", project_simplex(best_seq + rng.standard_normal(n_best) * 0.003)))
        # Strategy 2: reverse mix
        strategies.append(("reverse_mix", project_simplex(0.7 * best_seq + 0.3 * best_seq[::-1])))
        # Strategy 3: uniform mix
        strategies.append(("uniform_mix", project_simplex(0.8 * best_seq + 0.2 * np.ones(n_best) / n_best)))
        # Strategy 4: larger perturbation
        strategies.append(("perturb_large", project_simplex(best_seq + rng.standard_normal(n_best) * 0.01)))

        for strat_name, a_init in strategies:
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            t_alloc = min(rem * 0.3, 150)

            # Spectral
            a_spec, _ = spectral_amplitude_method(a_init, time_limit=t_alloc * 0.4, lr_init=0.3)
            best_seq, best_score = try_and_update(a_spec, best_seq, best_score, f"deep_spec_{strat_name}")

            # Subgradient
            base = a_spec
            if best_seq is not None and len(best_seq) == n_best:
                base = normalize_nonneg(0.5 * best_seq + 0.5 * a_spec)
            a_sg, _ = subgradient_optimize(base, time_limit=t_alloc * 0.4, n_peaks=15,
                                            lr_init=0.003, momentum=0.93,
                                            temp_init=0.003, temp_growth=1.002)
            best_seq, best_score = try_and_update(a_sg, best_seq, best_score, f"deep_sg_{strat_name}")

            # Pairwise
            if best_seq is not None and len(best_seq) == n_best:
                a_pw, _ = pairwise_coord_descent(best_seq, time_limit=t_alloc * 0.2,
                                                  n_swaps_per_pass=50)
                best_seq, best_score = try_and_update(a_pw, best_seq, best_score, f"deep_pw_{strat_name}")

    # Phase 4: Fine polishing
    print(f"\n=== Phase 4: Fine polishing (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 20
        if rem > 30:
            t_cd = min(rem * 0.4, 120)
            print(f"  Single-coordinate descent: {t_cd:.0f}s")
            a_cd, _ = single_coord_descent(best_seq, time_limit=t_cd,
                                            multipliers=[0.8, 0.9, 0.95, 1.05, 1.1, 1.25])
            best_seq, best_score = try_and_update(a_cd, best_seq, best_score, "polish_cd")

        rem = deadline - time.monotonic() - 15
        if rem > 20:
            t_pw = min(rem * 0.4, 80)
            print(f"  Pairwise descent: {t_pw:.0f}s")
            a_pw, _ = pairwise_coord_descent(best_seq, time_limit=t_pw, n_swaps_per_pass=80)
            best_seq, best_score = try_and_update(a_pw, best_seq, best_score, "polish_pw")

        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_sg = min(rem * 0.5, 60)
            print(f"  Fine subgradient: {t_sg:.0f}s")
            a_sg, _ = subgradient_optimize(best_seq, time_limit=t_sg, n_peaks=20,
                                            lr_init=0.001, momentum=0.95,
                                            temp_init=0.002, temp_growth=1.003)
            best_seq, best_score = try_and_update(a_sg, best_seq, best_score, "polish_sg")

        rem = deadline - time.monotonic() - 10
        if rem > 15:
            t_spec = min(rem * 0.5, 40)
            print(f"  Fine spectral: {t_spec:.0f}s")
            a_spec, _ = spectral_amplitude_method(best_seq, time_limit=t_spec, lr_init=0.15)
            best_seq, best_score = try_and_update(a_spec, best_seq, best_score, "polish_spec")

    # Phase 5: Final random restarts
    print(f"\n=== Phase 5: Final restarts (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        for i in range(5):
            rem = deadline - time.monotonic() - 10
            if rem < 20:
                break
            scale = 0.002 + 0.002 * i
            a_pert = project_simplex(best_seq + rng.standard_normal(n_best) * scale)
            t_alloc = min(rem * 0.3, 50)
            a_sg, _ = subgradient_optimize(a_pert, time_limit=t_alloc, n_peaks=10,
                                            lr_init=0.003, momentum=0.92,
                                            temp_init=0.004, temp_growth=1.002)
            best_seq, best_score = try_and_update(a_sg, best_seq, best_score, f"restart_{i}")

    # Fallback
    if best_seq is None:
        best_seq = parent
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
