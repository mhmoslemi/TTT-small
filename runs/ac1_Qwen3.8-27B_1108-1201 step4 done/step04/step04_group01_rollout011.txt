    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient of LSE objective. weights has length 2n-1, f has length n.
    grad[i] = 2 * sum_m weights[i+m] * f[m]
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def nesterov_lse(f0, time_limit, beta_start=50.0, beta_end=50000.0,
                 lr_scale=0.8, momentum_base=0.9):
    """Nesterov-accelerated LSE gradient descent with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    t = 1.0  # Nesterov momentum parameter
    lr = lr_scale * 2.0 / n
    y = f.copy()
    prev_f = f.copy()
    v = np.zeros(n)

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Compute gradient at y (Nesterov extrapolation point)
        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_grad(w, y)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new

        # Update with momentum (blending Nesterov and heavy-ball)
        momentum = min(0.99, momentum_base + 0.09 * t / (t + 10.0))
        v = momentum * v - lr * grad
        f_new = y + beta_mom * (y - prev_f) + v
        f_new = project_simplex(f_new)

        prev_f = f.copy()
        f = f_new
        y = f.copy()
        t = t_new

        # Evaluate
        conv_new = fast_convolve(f)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
            # Slightly increase lr
            lr = min(lr * 1.003, 3.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
            if no_improve > 100:
                # Smart restart: jump to best + small structured perturbation
                rng = np.random.default_rng(it)
                freq = rng.integers(1, max(2, n // 4))
                phase = rng.uniform(0, 2 * np.pi)
                x = np.linspace(0, 2 * np.pi * freq, n)
                perturbation = 0.003 * np.cos(x + phase)
                f = project_simplex(best_f + perturbation)
                y = f.copy()
                prev_f = f.copy()
                v = np.zeros(n)
                t = 1.0
                lr = lr_scale * 2.0 / n
                beta = min(beta * 1.3, beta_end)
                no_improve = 0

        # Ramp up beta (accelerating late in the run)
        progress = elapsed / time_limit
        beta = beta_start * (beta_end / beta_start) ** progress
        beta = min(beta, beta_end)

        if it % 1000 == 0:
            print(f"    NLSE it={it}, obj={best_obj:.12f}, beta={beta:.1f}, lr={lr:.6f}, t={t:.1f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def nesterov_subgrad(f0, time_limit, top_k_start=4, top_k_end=None,
                     lr_init=0.01, use_averaging=True):
    """Nesterov-accelerated subgradient method with peak focus and averaging."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if top_k_end is None:
        top_k_end = max(4, n // 4)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t = 1.0
    lr = lr_init
    y = f.copy()
    prev_f = f.copy()
    v = np.zeros(n)

    # For Nesterov averaging
    avg_f = f.copy()
    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        if elapsed > time_limit:
            break

        # Determine current top_k (ramp up over time)
        progress = elapsed / time_limit
        cur_top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        cur_top_k = max(1, min(cur_top_k, 2 * n - 1))

        # Compute subgradient at y
        conv_y = fast_convolve(y)
        if cur_top_k >= len(conv_y):
            peaks = np.arange(len(conv_y))
            w = np.ones(len(peaks)) / len(peaks)
        else:
            peaks = np.argpartition(conv_y, -cur_top_k)[-cur_top_k:]
            pv = conv_y[peaks]
            # Use softmax weighting over peaks
            scaled = (pv - pv.max()) * 50.0
            w = np.exp(scaled)
            w = w / w.sum()

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * (k - idx)
            sg[idx] += 2.0 * w[i] * (k - idx)  # Wait, this is wrong

        # Recompute properly: grad[i] = 2 * sum_k w_k * a[k-i] for valid k-i
        sg = np.zeros(n)
        for i_prog in range(len(peaks)):
            if w[i_prog] < 1e-14:
                continue
            k = int(peaks[i_prog])
            # For position i in a, the contribution is 2*w_k*a[k-i] if 0<=k-i<n
            # So sg[i] += 2 * w_k * a[k-i]
            for m in range(n):
                ki = k - m  # this is k - m, we need 0 <= k-m < n
                # Actually: sg[i] = 2 * sum_k w_k * a[k-i]
                # Let's compute differently: for each peak k, add 2*w_k*a shifted
                pass

        # Better: compute via the correlation formula
        # sg[i] = 2 * sum_{k} w_k * a[k-i] where 0<=k-i<n, i.e., i <= k < i+n
        # = 2 * (w * a_reversed) cross-correlation
        # Build sparse weight vector
        w_full = np.zeros(2 * n - 1)
        for i_prog in range(len(peaks)):
            if w[i_prog] > 1e-14:
                w_full[int(peaks[i_prog])] = w[i_prog]

        sg = compute_lse_grad(w_full, y)

        grad_norm = np.linalg.norm(sg)
        if grad_norm < 1e-18:
            no_improve += 100
            break
        sg = sg / grad_norm

        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta_mom = (t - 1.0) / t_new

        momentum = min(0.99, 0.85 + 0.14 * t / (t + 20.0))
        v = momentum * v - lr / np.sqrt(t) * sg
        f_new = y + beta_mom * (y - prev_f) + v
        f_new = project_simplex(f_new)

        prev_f = f.copy()
        f = f_new
        y = f.copy()
        t = t_new

        # Nesterov averaging
        if use_averaging:
            avg_f = ((t - 1.0) / t_new * avg_f + f_new)

        # Evaluate
        conv_new = fast_convolve(f)
        obj_new = conv_new.max()

        # Also check averaged iterate periodically
        if use_averaging and it % 50 == 0:
            avg_norm = avg_f.sum()
            if avg_norm > 1e-10:
                avg_f_norm = avg_f / avg_norm
                obj_avg = fast_convolve(avg_f_norm).max()
                if obj_avg < obj_new:
                    obj_new = obj_avg
                    f_new = avg_f_norm.copy()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 200:
                rng = np.random.default_rng(it * 7 + 3)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                y = f.copy()
                prev_f = f.copy()
                v = np.zeros(n)
                t = 1.0
                avg_f = f.copy()
                no_improve = 0

        if it % 2000 == 0:
            print(f"    NSG it={it}, obj={best_obj:.12f}, top_k={cur_top_k}, t={t:.1f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def peak_chase(f0, time_limit, n_iter_max=50000):
    """Aggressive peak-chasing: iteratively reduce the single largest peak."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    lr = 0.05
    no_improve = 0

    while time.monotonic() - t0 < time_limit and it < n_iter_max:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(f)
        peak_k = int(np.argmax(conv))
        peak_val = conv[peak_k]

        # Subgradient of the peak
        sg = np.zeros(n)
        js = max(0, peak_k - n + 1)
        je = min(n - 1, peak_k)
        idx = np.arange(js, je + 1)
        sg[idx] = 2.0 * (peak_k - idx)
        sg[idx] = 2.0 * f[peak_k - idx]  # This is the derivative contribution

        # Wait: db_k/da_i = 2*a[k-i] for valid k-i
        # So sg[i] = 2 * f[peak_k - i] for i in [max(0,k-n+1), min(n-1,k)]
        sg = np.zeros(n)
        for i in range(js, je + 1):
            sg[i] = 2.0 * f[peak_k - i]

        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-18:
            break
        sg = sg / sg_norm

        f_new = project_simplex(f - lr * sg)
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            lr = min(lr * 1.01, 0.5)
            no_improve = 0
        else:
            no_improve += 1
            lr *= 0.95
            if no_improve > 100:
                # Try to also reduce second peak
                conv = fast_convolve(f)
                peak_k = int(np.argmax(conv))
                # Find second peak (excluding neighbors of first)
                conv_masked = conv.copy()
                lo = max(0, peak_k - 3)
                hi = min(len(conv), peak_k + 4)
                conv_masked[lo:hi] = -np.inf
                peak_k2 = int(np.argmax(conv_masked))

                # Combined subgradient
                sg2 = np.zeros(n)
                for pk in [peak_k, peak_k2]:
                    js2 = max(0, pk - n + 1)
                    je2 = min(n - 1, pk)
                    for i in range(js2, je2 + 1):
                        sg2[i] += f[pk - i]

                sg2_norm = np.linalg.norm(sg2)
                if sg2_norm > 1e-18:
                    f_new2 = project_simplex(f - lr * 2.0 * sg2 / sg2_norm)
                    conv_new2 = fast_convolve(f_new2)
                    if conv_new2.max() < obj_new:
                        f = f_new2
                        obj_new = conv_new2.max()
                        if obj_new < best_obj:
                            best_obj = obj_new
                            best_f = f.copy()
                        no_improve = 0

            if no_improve > 300:
                rng = np.random.default_rng(it)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
                lr = 0.05
                no_improve = 0

    return best_f, best_obj


def fw_lp_quick(f0, time_limit, lp_timeout=10):
    """Quick FW-LP for initial structure finding."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    it = 0
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Build constraint matrix
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A[k, j] += f[k - j]
                if j != k - j and 0 <= k - j < n and k - j != j:
                    pass  # Already handled by the sum

        # Actually, A[k,j] should be f[k-j] if 0<=k-j<n
        A = np.zeros((2 * n - 1, n))
        for k in range(2 * n - 1):
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            for j in range(js, je + 1):
                A[k, j] = f[k - j]

        # Solve LP: max sum(g) s.t. A@g <= M*1.05, g >= 0
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M * 1.05)
        bounds = [(0.0, None)] * n
        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={'presolve': True, 'time_limit': lp_timeout})
            if res.success:
                g = np.maximum(res.x, 0)
                g_sum = g.sum()
                if g_sum > 1e-15:
                    g = g / g_sum
                    # Line search
                    best_alpha = 0.5
                    best_val = M
                    for alpha in [0.3, 0.5, 0.7, 0.9, 1.0]:
                        ft = (1 - alpha) * f + alpha * g
                        ft = project_simplex(ft)
                        val = fast_convolve(ft).max()
                        if val < best_val:
                            best_val = val
                            best_alpha = alpha
                    if best_alpha > 0:
                        f = project_simplex((1 - best_alpha) * f + best_alpha * g)
                        conv = fast_convolve(f)
                        M = conv.max()
                        if M < best_obj:
                            best_obj = M
                            best_f = f.copy()
                            no_improve = 0
                        else:
                            no_improve += 1
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        except Exception:
            no_improve += 1

        if no_improve > 20:
            break

    return best_f, best_obj


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def structured_init(n, seed):
    """Generate diverse structured initializations."""
    rng = np.random.default_rng(seed)
    inits = []

    # 1. Uniform
    inits.append(np.ones(n) / n)

    # 2. Triangular
    x = np.arange(n)
    tri = np.abs(x - n / 2.0)
    tri = tri / tri.sum()
    inits.append(tri)

    # 3. Cosine
    cos = np.cos(np.pi * x / n) ** 2 + 0.1
    cos = np.maximum(cos, 0)
    cos = cos / cos.sum()
    inits.append(cos)

    # 4. Random smooth
    rough = rng.standard_normal(n)
    # Smooth with a window
    kernel = np.exp(-0.5 * (np.arange(-5, 6) / 2.0) ** 2)
    smooth = np.convolve(rough, kernel, mode='same')
    smooth = np.maximum(smooth, 0)
    if smooth.sum() > 1e-10:
        smooth = smooth / smooth.sum()
        inits.append(smooth)

    # 5. Two-peak
    c1 = rng.uniform(0.2, 0.4)
    c2 = rng.uniform(0.6, 0.8)
    sigma = rng.uniform(0.05, 0.15)
    g1 = np.exp(-0.5 * ((x / n - c1) / sigma) ** 2)
    g2 = np.exp(-0.5 * ((x / n - c2) / sigma) ** 2)
    two = g1 + g2
    two = two / two.sum()
    inits.append(two)

    # 6. Random on simplex
    r = rng.standard_normal(n)
    r = np.maximum(r, 0)
    if r.sum() > 1e-10:
        r = r / r.sum()
        inits.append(r)

    # 7. Sinusoidal modulation
    freq = rng.integers(2, max(3, n // 10))
    phase = rng.uniform(0, 2 * np.pi)
    sin_mod = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x / n + phase)
    sin_mod = np.maximum(sin_mod, 0.01)
    sin_mod = sin_mod / sin_mod.sum()
    inits.append(sin_mod)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Multi-start with quick local search
    Phase 2: Nesterov-accelerated LSE (high beta)
    Phase 3: Nesterov subgradient with averaging
    Phase 4: Peak chasing
    Phase 5: Length exploration
    Phase 6: Final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_score = float('inf')
    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = evaluate_sequence(parent.tolist())
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.10f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    n_best = n_parent

    # === Phase 1: Multi-start with quick local search (120s) ===
    print("\n=== Phase 1: Multi-start ===")
    t0 = time.monotonic()
    t_phase1 = min(120, deadline - t0 - 600)
    if t_phase1 > 30:
        per_start = t_phase1 / 8
        # Try parent with quick FW-LP
        t1 = min(per_start, 40)
        a_opt, _ = fw_lp_quick(parent, time_limit=t1)
        try_sequence(a_opt, best_tracker)

        # Try structured inits
        inits = structured_init(n_parent, seed=seed)
        for i, init in enumerate(inits):
            if time.monotonic() + per_start > deadline - 550:
                break
            t1 = min(per_start * 0.4, 20)
            t2 = min(per_start * 0.6, 30)
            if t1 < 5:
                continue
            a_opt, _ = nesterov_lse(init, time_limit=t1,
                                     beta_start=50.0, beta_end=5000.0)
            try_sequence(a_opt, best_tracker)
            if t2 > 5:
                cur = best_tracker[1] if best_tracker[1] is not None and len(best_tracker[1]) == n_parent else a_opt
                a_opt2, _ = peak_chase(cur, time_limit=t2)
                try_sequence(a_opt2, best_tracker)

    print(f"  Phase 1 done, best={best_tracker[0]:.10f}, elapsed={time.monotonic()-t0:.0f}s")

    # === Phase 2: Nesterov LSE high beta (500s) ===
    print("\n=== Phase 2: Nesterov LSE (high beta) ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)
    t0 = time.monotonic()
    t_phase2 = min(500, deadline - t0 - 400)
    if t_phase2 > 30:
        a_opt, obj = nesterov_lse(current_best, time_limit=t_phase2,
                                   beta_start=30.0, beta_end=80000.0,
                                   lr_scale=1.0)
        try_sequence(a_opt, best_tracker)
    print(f"  Phase 2 done, best={best_tracker[0]:.10f}, elapsed={time.monotonic()-t0:.0f}s")

    # === Phase 3: Nesterov Subgradient with averaging (300s) ===
    print("\n=== Phase 3: Nesterov Subgradient ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)
    t0 = time.monotonic()
    t_phase3 = min(300, deadline - t0 - 300)
    if t_phase3 > 30:
        a_opt, obj = nesterov_subgrad(current_best, time_limit=t_phase3,
                                       top_k_start=2, top_k_end=max(8, n_best // 6),
                                       lr_init=0.02, use_averaging=True)
        try_sequence(a_opt, best_tracker)
    print(f"  Phase 3 done, best={best_tracker[0]:.10f}, elapsed={time.monotonic()-t0:.0f}s")

    # === Phase 4: Peak chasing (200s) ===
    print("\n=== Phase 4: Peak Chase ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)
    t0 = time.monotonic()
    t_phase4 = min(200, deadline - t0 - 200)
    if t_phase4 > 20:
        a_opt, obj = peak_chase(current_best, time_limit=t_phase4)
        try_sequence(a_opt, best_tracker)
    print(f"  Phase 4 done, best={best_tracker[0]:.10f}, elapsed={time.monotonic()-t0:.0f}s")

    # === Phase 5: Second round of Nesterov LSE on best (200s) ===
    print("\n=== Phase 5: Second LSE polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)
    t0 = time.monotonic()
    t_phase5 = min(200, deadline - t0 - 100)
    if t_phase5 > 20:
        a_opt, obj = nesterov_lse(current_best, time_limit=t_phase5,
                                   beta_start=200.0, beta_end=200000.0,
                                   lr_scale=0.5)
        try_sequence(a_opt, best_tracker)
    print(f"  Phase 5 done, best={best_tracker[0]:.10f}, elapsed={time.monotonic()-t0:.0f}s")

    # === Phase 6: Length exploration ===
    print("\n=== Phase 6: Length Exploration ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t0 = time.monotonic()
    t_phase6 = min(200, deadline - t0 - 50)

    if t_phase6 > 40:
        # Try nearby lengths
        alt_lengths = [n_best - 16, n_best - 8, n_best + 8, n_best + 16,
                       n_best - 32, n_best + 32]
        alt_lengths = [x for x in alt_lengths if 50 <= x <= 2000]

        per_alt = t_phase6 / max(1, len(alt_lengths))
        for n_alt in alt_lengths[:4]:
            if time.monotonic() > deadline - 50:
                break
            if n_alt in [len(best_tracker[1])] if best_tracker[1] is not None else False:
                continue

            # Interpolate
            x_old = np.linspace(0, 1, len(current_best))
            x_new = np.linspace(0, 1, n_alt)
            f_interp = np.interp(x_new, x_old, current_best)
            f_interp = np.maximum(f_interp, 0)
            if f_interp.sum() > 1e-10:
                f_interp = f_interp / f_interp.sum()

            t1 = min(per_alt * 0.6, 50)
            t2 = min(per_alt * 0.4, 30)
            if t1 > 5:
                a_opt, _ = nesterov_lse(f_interp, time_limit=t1,
                                         beta_start=50.0, beta_end=10000.0)
                try_sequence(a_opt, best_tracker)
            if t2 > 5:
                cur = best_tracker[1] if best_tracker[1] is not None and len(best_tracker[1]) == n_alt else a_opt
                a_opt2, _ = nesterov_subgrad(cur, time_limit=t2,
                                              top_k_start=2, top_k_end=max(4, n_alt // 10))
                try_sequence(a_opt2, best_tracker)

    print(f"  Phase 6 done, best={best_tracker[0]:.10f}, elapsed={time.monotonic()-t0:.0f}s")

    # === Phase 7: Final micro-polish ===
    print("\n=== Phase 7: Final Micro-Polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t0 = time.monotonic()
    t_phase7 = max(10, deadline - t0 - 5)
    if t_phase7 > 10:
        # Very high beta LSE with tiny learning rate
        a_opt, _ = nesterov_lse(current_best, time_limit=t_phase7,
                                 beta_start=1000.0, beta_end=500000.0,
                                 lr_scale=0.2)
        try_sequence(a_opt, best_tracker)
    print(f"  Phase 7 done, best={best_tracker[0]:.10f}, elapsed={time.monotonic()-t0:.0f}s")

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
        else:
            best_score = best_tracker[0]
    else:
        final_seq = np.ones(448) / 448
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.10f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
