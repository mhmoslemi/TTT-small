Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a combination of:
1. Smoothed Nesterov accelerated gradient descent with soft-max approximation
2. LP-based Frank-Wolfe iterations with improved line search
3. Spectral-based initialization and multi-scale refinement
4. Adaptive restarts with structured perturbations

import numpy as np
from scipy.optimize import linprog
from scipy.linalg import null_space
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_grad(a, T, conv=None):
    """Compute gradient of smooth max of auto-convolution.
    
    Returns (smooth_obj, grad) where grad is the gradient of (1/T)*log(sum exp(T*conv_k))
    with respect to a.
    """
    n = len(a)
    if conv is None:
        conv = fast_convolve(a)
    
    # Softmax weights
    scaled = T * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()
    
    smooth_obj = (scaled.max() + np.log(w.sum())) / T  # This is log-sum-exp / T

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

                a = best_a_try
                if best_new_obj < best_obj - 1e-15:
                    best_obj = best_new_obj
                    best_a = a.copy()
                    no_improve = 0
                    improved = True
                # Update conv for next iteration
                conv = fast_convolve(a)
                peak_indices = np.argpartition(conv, -top_k)[-top_k:]
            else:
                no_improve += 1
        
        if not improved and no_improve > 50:
            # Perturb and restart
            scale = min(0.01 + 0.02 * no_improve / 100, 0.1)
            a = project_simplex(best_a + rng_local.standard_normal(n) * scale)
            conv = fast_convolve(a)
            no_improve = 0
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

def lp_frank_wolfe(a0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with improved line search."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng_local = np.random.default_rng(999 + n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 3:
            break
        
        # Solve LP
        g = _solve_lp_direction(a, M * m_exp)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum
        
        # Fixed point check
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 3.0)
            if no_improve > 30:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        # Coarse
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
            at = np.maximum((1 - alpha) * a + alpha * g, 0)
            st = at.sum()
            if st < 1e-15:
                continue
            at = at / st
            val = fast_convolve(at).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Fine
        if 0.1 < best_alpha < 0.9:
            lo, hi = max(0, best_alpha - 0.1), min(1, best_alpha + 0.1)
            for _ in range(6):
                mid = (lo + hi) / 2
                at = np.maximum((1 - mid) * a + mid * g, 0)
                st = at.sum()
                if st < 1e-15:
                    break
                at = at / st
                val = fast_convolve(at).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            a = np.maximum((1 - best_alpha) * a + best_alpha * g, 0)
            fs = a.sum()
            if fs < 1e-15:
                a = best_a.copy()
            else:
                a = a / fs
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
                if verbose and it % 20 == 0:
                    print(f"    LP-FW {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 4.0)
        
        # Restart
        if no_improve > 40:
            strategies = [
                lambda: project_simplex(best_a + rng_local.standard_normal(n) * 0.003),
                lambda: project_simplex(0.8 * best_a + 0.2 * best_a[::-1]),
                lambda: project_simplex(best_a + 0.05 * (np.random.default_rng().random(n) - 0.5)),
            ]
            a = strategies[no_improve % len(strategies)]()
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0
    
    return best_a, best_obj

def _solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    # Build constraint matrix: (f*g)[k] = sum_j f[k-j] * g[j] for valid k-j
    # A_ub has 2n-1 rows, n columns
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        # (f*g)[k] includes f[k-j]*g[j] for k-j in [0, n-1]
        # So A_ub[k, j] = f[k-j] for k in [j, j+n-1]
        A_ub[j:j+n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None
