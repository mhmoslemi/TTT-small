def project_simplex(x):

    n = len(x)
    s = np.sum(x)
    y = x.copy()
    changed = True
    while changed:

        for i in range(len(x)):
            for j in range(len(x)):
                if i == j: continue
                if y[i] < 1e-10: continue
                a = y[i] + y[j]
