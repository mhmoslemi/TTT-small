def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smoothed_max(conv, temp, top_k=64):
    """Compute smoothed max using softmax over top-k peaks."""
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        return pv.max(), float(np.argmax(pv))
    return float((w * pv).sum() / wsum), int(peaks[np.argmax(pv)])


def compute_grad_smoothed(f, conv, temp, top_k=64):
    """Compute gradient of softmax-smoothed max of conv w.r.t. f."""
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            grad[j] += 2.0 * w[idx] * f[k - j]
    return grad


def solve_lp_direction(f, M, time_limit=15):
    """LP oracle: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    n = len(f)
    if n > 1024:
        return None
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None:
            gx = np.maximum(res.x, 0)
            if gx.sum() > 1e-15:
                return gx / gx.sum()
    except Exception:
        pass
    return None


def sg_descent(f0, time_limit, initial_lr=0.005, momentum=0.9, top_k=32,
               temp_init=0.001, temp_final=0.00001, lp_escape=True,
               lp_every=400, verbose=False):
    """Subgradient descent with momentum and adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    lr = initial_lr
    temp = temp_init
    total_time = time_limit

    while time.monotonic() - t0 < total_time:
        it += 1
        rem = total_time - (time.monotonic() - t0)
        if rem < 1.0:
            break

        grad = compute_grad_smoothed(f, conv, temp, top_k)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break

        # Backtracking line search
        step = lr
        best_alpha = 0.0
        best_val = best_obj
        for trial in range(8):
            alpha = step / (grad_norm + 1e-30)
            f_try = f - alpha * grad
            f_try = project_simplex(f_try)
            conv_try = fast_convolve(f_try)
            val = conv_try.max()
            if val < best_val - 1e-14:
                best_val = val
                best_alpha = alpha
            if trial < 3:
                step *= 0.6
            else:
                step *= 1.3

        if best_alpha > 1e-12:
            f_new = f - best_alpha * grad
            f_new = project_simplex(f_new)
            conv_new = fast_convolve(f_new)
            obj_new = conv_new.max()

            # Momentum
            v = momentum * v - best_alpha * grad
            f_mom = project_simplex(f_new + 0.25 * v)
            conv_mom = fast_convolve(f_mom)
            if conv_mom.max() < obj_new:
                f = f_mom
                conv = conv_mom
                obj = conv.max()
            else:
                f = f_new
                conv = conv_new
                obj = obj_new

            if obj < best_obj - 1e-13:
                best_obj = obj
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        # Adaptive parameters
        progress = (time.monotonic() - t0) / total_time
        temp = temp_init * (temp_final / temp_init) ** progress
        lr = initial_lr * (0.5 ** progress)

        # LP escape
        if lp_escape and no_improve > lp_every and rem > 30:
            g = solve_lp_direction(f, best_obj * 1.01, time_limit=10)
            if g is not None:
                for alpha in [0.1, 0.3, 0.5, 0.7, 1.0]:
                    ft = project_simplex((1 - alpha) * f + alpha * g)
                    ct = fast_convolve(ft).max()
                    if ct < best_obj:
                        best_obj = ct
                        best_f = ft.copy()
                        f = ft
                        conv = fast_convolve(f)
                        break
            no_improve = 0
            v = np.zeros(n)

        # Random restart
        if no_improve > 2000:
            rng = np.random.default_rng(it * 7 + 3)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            conv = fast_convolve(f)
            v = np.zeros(n)
            no_improve = 0
            lr = initial_lr * 0.3

        if verbose and it % 500 == 0:
            print(f"    SG it={it}, obj={best_obj:.10f}, rem={rem:.0f}s")

    return best_f, best_obj

def pair_exchange_polish(f0, time_limit, verbose=False):

            break

        top_peaks = np.argpartition(conv, -8)[-8:]

        improved_this_round = False
        for k in top_peaks[:4]:
            if time.monotonic() - t0 > time_limit:
                break
            # Contributors to conv[k]: pairs (j, k-j) where both in [0, n-1]

            je = min(n - 1, k)
            if js > je:
                continue

            # Find the largest contributor
            contribs = []
            for j in range(js, je + 1):
                kj = k - j
                if 0 <= kj < n:
                    contribs.append((j, kj, f[j] * f[kj]))

            if not contribs:
                continue

            # Sort by contribution
            contribs.sort(key=lambda x: -x[2])
            j_big, kj_big, _ = contribs[0]

            # Find a non-contributor to receive mass
            non_contribs = []
            for j in range(n):
                if j == j_big or j == kj_big:
                    continue
                # Check if j contributes to any top peak
                is_contrib = False
                for kp in top_peaks:
                    if max(0, kp - n + 1) <= j <= min(n - 1, kp):
                        is_contrib = True
                        break

            lo_dt = -f[j_big]

def multiplicative_update(f0, time_limit, target=None, verbose=False):
    """Multiplicative update to flatten the convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:

        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    if target is None:
        target = conv.max() * 0.95

    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit:
            break

        if rem < 0.5:
            break

        # Compute update ratio for each element
        # For element i, it contributes to conv[k] via a[i]*a[k-i]
        # We want to reduce elements that contribute to high peaks
        ratios = np.zeros(n)
        for i in range(n):

                ki = k - i

                    den += f[ki] ** 2

            else:
                ratios[i] = 1.0

        # Apply multiplicative update with damping
        lr = 0.1
        f = f * (1.0 + lr * (1.0 - ratios / (np.max(ratios) + 1e-30)))
        f = np.maximum(f, 1e-10)
        f = f / f.sum()
        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-13:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
            # Decrease target
            target = conv.max() * 0.98
        else:
            no_improve += 1

        if no_improve > 200:
            break

        if verbose:
            pass

    return best_f, best_obj

    e1 = np.exp(-x / (n / 3.0))
    inits.append(e1 / e1.sum())

    e2 = np.exp(-x[::-1] / (n / 3.0))
    inits.append(e2 / e2.sum())

    c = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())

    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(sn / sn.sum())

    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    # Bimodal
    bm = np.exp(-((x - n * 0.25) / (n * 0.1)) ** 2) + np.exp(-((x - n * 0.75) / (n * 0.1)) ** 2) + 0.01
    inits.append(bm / bm.sum())

    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            blended = 0.5 * (px / ps) + 0.5 * (ramp_d / ramp_d.sum())
            inits.append(blended / blended.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search function combining multiple optimization strategies.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    t_start = time.monotonic()

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)}) (was {old:.8f}) [t={time.monotonic()-t_start:.0f}s]")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # Phase 1: Quick evaluation of initializations
    print("=== Phase 1: Initialization ===")
    candidate_inits = []

    if parent is not None:
        n_p = len(parent)
        candidate_inits.append(('parent', parent.copy(), n_p))
        # Try parent at different n
        for n_try in [256, 384, 512]:
            if n_try != n_p:
                px = np.interp(np.arange(n_try), np.linspace(0, 1, n_p), parent)
                px = np.maximum(px, 0)
                ps = px.sum()
                if ps > 1e-10:
                    candidate_inits.append((f'parent_n{n_try}', px / ps, n_try))
    else:
        n_p = 384

    # Add structured inits at parent length
    if parent is not None:
        inits = generate_inits(n_p, parent, rng)
        for i, init in enumerate(inits):
            candidate_inits.append((f'init_{i}', init, n_p))

    # Evaluate all inits
    for label, init, n in candidate_inits:
        if time.monotonic() > deadline - 300:
            break
        score = try_seq(init, f"init_{label}")

    if best_seq is not None:
        n_best_init = len(best_seq)
        print(f"  Best init: score={best_score:.8f}, n={n_best_init}")

    # Phase 2: Subgradient descent on best
    print(f"\n=== Phase 2: SG Descent ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        f_work = best_seq / best_seq.sum()
        t_alloc = min(200, (deadline - time.monotonic()) * 0.15)
        print(f"  SG descent, n={n_cur}, t={t_alloc:.0f}s")
        a_opt, obj = sg_descent(f_work, time_limit=t_alloc,
                                initial_lr=0.008, momentum=0.9,
                                top_k=max(16, n_cur // 16),
                                temp_init=0.002, temp_final=0.00005,
                                lp_escape=True, lp_every=300)
        try_seq(a_opt, "sg_phase2")

    # Phase 3: FW-LP optimization
    print(f"\n=== Phase 3: FW-LP ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        f_work = best_seq / best_seq.sum()
        t_alloc = min(300, (deadline - time.monotonic()) * 0.2)
        print(f"  FW-LP, n={n_cur}, t={t_alloc:.0f}s")
        a_opt, obj = fw_optimize(f_work, time_limit=t_alloc)
        try_seq(a_opt, "fw_phase3")

    # Phase 4: Second SG round (finer)
    print(f"\n=== Phase 4: Fine SG ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        f_work = best_seq / best_seq.sum()
        t_alloc = min(300, (deadline - time.monotonic()) * 0.2)
        print(f"  Fine SG, n={n_cur}, t={t_alloc:.0f}s")
        a_opt, obj = sg_descent(f_work, time_limit=t_alloc,
                                initial_lr=0.003, momentum=0.92,
                                top_k=max(32, n_cur // 8),
                                temp_init=0.0005, temp_final=0.00001,
                                lp_escape=True, lp_every=200)
        try_seq(a_opt, "sg_fine")

    # Phase 5: Pair-exchange polishing
    print(f"\n=== Phase 5: Pair Exchange Polish ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        f_work = best_seq / best_seq.sum()
        t_alloc = min(400, (deadline - time.monotonic()) * 0.25)
        print(f"  Pair exchange, n={n_cur}, t={t_alloc:.0f}s")
        a_opt, obj = pair_exchange_polish(f_work, time_limit=t_alloc)
        try_seq(a_opt, "pair_exch")

    # Phase 6: Final polish rounds
    print(f"\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        n_cur = len(best_seq)
        f_work = best_seq / best_seq.sum()

        # Quick SG with very small steps
        rem = (deadline - time.monotonic()) * 0.5
        if rem > 10:
            a_opt, _ = sg_descent(f_work, time_limit=min(rem, 150),
                                  initial_lr=0.001, momentum=0.95,
                                  top_k=max(64, n_cur // 4),
                                  temp_init=0.0001, temp_final=0.000003,
                                  lp_escape=False)
            try_seq(a_opt, "final_sg")

        # Final pair exchange
        rem = (deadline - time.monotonic()) * 0.5
        if rem > 10:
            f_work = best_seq / best_seq.sum()
            a_opt, _ = pair_exchange_polish(f_work, time_limit=min(rem, 150))
            try_seq(a_opt, "final_pe")

    # Fallback
    if best_seq is None:
        print("  WARNING: No valid sequence found, using fallback")
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    print(f"\n=== Final Verification ===")
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"FINAL SCORE: {best_score:.8f} (n={len(best_seq)}), time={time.monotonic()-t_start:.0f}s")
    return [float(x) for x in best_seq]
