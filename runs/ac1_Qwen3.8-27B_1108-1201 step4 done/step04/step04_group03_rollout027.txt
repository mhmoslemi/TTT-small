def lse_mirror_descent(a0, time_limit, t_init=0.005, t_decay=0.9998, t_min=0.0001,

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t = t_init
    cur_lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        max_c = conv.max()
        
        # Compute LSE gradient
        shifted = (conv - max_c) / t
        shifted = np.maximum(shifted, -50)  # clip for numerical stability
        unnormalized = np.exp(shifted)
        weights = unnormalized / unnormalized.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            w = weights[k]
            if w < 1e-14:
                continue

            je = min(n - 1, k)
            grad[js:je+1] += 2.0 * w * a[k-je:k-js+1][::-1]
        
        # Mirror descent update
        a_new = a * np.exp(-cur_lr * grad)
        a_new = np.maximum(a_new, 1e-30)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a_new = a.copy()
        else:

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
        else:
            no_improve += 1
            a = a_new  # Still take the step (might help later)
        
        # Adaptive temperature
        t = max(t * t_decay, t_min)
        cur_lr = max(cur_lr * lr_decay, 0.0005)

    return best_a, best_obj


def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp(f, M):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def non_smooth_polish(a0, time_limit, lr=0.001, mom=0.93):

    a = np.maximum(a0.astype(np.float64), 0.0)

        a = a / s

    v = np.zeros(n)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    
    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(a)
        M = conv.max()
        
        # Find active peaks (within 1% of max)
        threshold = M * 0.99
        active = np.where(conv >= threshold)[0]
        
        if len(active) == 0:
            active = np.array([np.argmax(conv)])
        
        # Weight peaks by their value (sharper for higher peaks)
        pv = conv[active]
        weights = pv / pv.sum()

        sg = np.zeros(n)
        for k in active:
            w = weights[k]

            sg[js:je+1] += 2.0 * w * a[k-je:k-js+1][::-1]
        
        # Momentum update
        v = mom * v - cur_lr * sg
        a = project_simplex(a + v)
        
        conv = fast_convolve(a)
        obj = conv.max()

            best_obj = obj
            best_a = a.copy()
            no_improve = 0

        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 1e6) % 1000000)
            a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
            v = np.zeros(n)
            cur_lr = max(lr * 0.6, 0.0001)

    return best_a, best_obj
