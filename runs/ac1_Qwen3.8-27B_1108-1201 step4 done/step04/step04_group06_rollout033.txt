def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a, active_indices=None):
    """Build matrix C such that C @ g = a * g. If active_indices given, only those rows."""
    n = len(a)
    if active_indices is None:
        active_indices = np.arange(2 * n - 1)
    C = np.zeros((len(active_indices), n))
    for ii, k in enumerate(active_indices):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            C[ii, j_start:j_end + 1] = a[k - np.arange(j_start, j_end + 1)]
    return C


def compute_lse_gradient(a, conv, T, top_k_ratio=0.1):
    """Compute gradient of log-sum-exp of convolution values."""
    n = len(a)
    # Use top-K values for efficiency
    n_conv = len(conv)
    k_keep = max(10, int(n_conv * top_k_ratio))
    top_idx = np.argpartition(conv, -k_keep)[-k_keep:]
    top_vals = conv[top_idx]
    
    scaled = T * top_vals
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(w)
        w[np.argmax(top_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        if w[ii] < 1e-10:
            continue
        k = int(top_idx[ii])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[ii] * a[k - idxs]
    return grad


def lp_solve_active(a, M, active_indices, C=None):
    """Solve LP with active constraints only. Returns (g, success)."""
    from scipy.optimize import linprog
    n = len(a)
    if C is None:
        C = build_conv_matrix(a, active_indices)
    c_vec = -np.ones(n)
    b_vec = np.full(len(active_indices), M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c_vec, A_ub=C, b_ub=b_vec, bounds=bounds,
                         method='highs', options={'presolve': True})
        if result.success:
            g = np.maximum(result.x, 0)
            if g.sum() > 1e-12:
                return g, True
    except Exception:
        pass
    return None, False

def lse_adam_optimize(a0, time_limit, T_init=500.0, T_end=3000.0, lr_init=0.02, top_k_ratio=0.15):
    """Optimize log-sum-exp of convolution using Adam."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    
    t0 = time.monotonic()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    while time.monotonic() - t0 < time_limit:
        progress = min(1.0, (time.monotonic() - t0) / time_limit)
        T = T_init * (T_end / T_init) ** progress
        lr = lr_init * (1.0 - 0.7 * progress)
        
        conv = fast_convolve(a)
        grad = compute_lse_gradient(a, conv, T, top_k_ratio)
        
        # Normalize gradient to have similar scale
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-15:
            grad = grad / grad_norm * 0.1
        
        t_step += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t_step)
        v_hat = v / (1 - beta2 ** t_step)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            a = best_a.copy()
            continue
        a = a_new / s_new
        
        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-14:
            best_obj = cur_obj
            best_a = a.copy()
    
    return best_a, best_obj


def peak_shaving(a0, time_limit):
    """Peak-shaving coordinate descent: directly reduce the top peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(999)
    no_improve = 0
    max_no_improve = 500
    
    while time.monotonic() - t0 < time_limit:
        # Find the peak lag
        peak_k = int(np.argmax(conv))
        
        # Identify which a[j] contribute to this peak
        j_start = max(0, peak_k - n + 1)
        j_end = min(n - 1, peak_k)
        contributing = np.arange(j_start, j_end + 1)
        partners = peak_k - contributing
        
        # Try reducing the largest contributors
        contrib_vals = a[contributing] * a[partners]
        order = np.argsort(contrib_vals)[::-1]
        
        improved_this_round = False
        for oi in order[:5]:
            j = contributing[oi]
            # Try reducing a[j] and redistributing to random positions
            for target in rng.integers(0, n, size=3):
                if target == j:
                    continue
                # Move small amount from j to target
                epsilon = min(a[j] * 0.1, 0.005)
                if epsilon < 1e-8:
                    continue
                a_try = a.copy()
                a_try[j] -= epsilon
                a_try[target] += epsilon
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                a_try = a_try / s_try
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-12:
                    a = a_try
                    conv = conv_try
                    best_obj = obj_try
                    best_a = a.copy()
                    improved_this_round = True
                    no_improve = 0
                    break
            if improved_this_round:
                break
        
        if not improved_this_round:
            no_improve += 1
            if no_improve >= max_no_improve:
                break
        else:
            conv = fast_convolve(a)
    
    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n)
    inits['gauss_center'] = np.exp(-0.5 * ((x - n/2) / (n/4)) ** 2)
    inits['gauss_left'] = np.exp(-0.5 * ((x - n/4) / (n/6)) ** 2)
    inits['gauss_right'] = np.exp(-0.5 * ((x - 3*n/4) / (n/6)) ** 2)
    inits['bimodal'] = np.exp(-0.5 * ((x - n/4) / (n/8)) ** 2) + np.exp(-0.5 * ((x - 3*n/4) / (n/8)) ** 2)
    inits['sine1'] = 1 + np.sin(2 * np.pi * x / n)
    inits['sine2'] = 1 + np.sin(4 * np.pi * x / n)
    inits['sine5'] = 1 + np.sin(10 * np.pi * x / n)
    inits['ramp'] = x + 1
    inits['inv_ramp'] = n - x
    inits['triangle'] = 1 - np.abs(2 * x / n - 1)
    inits['random'] = np.abs(rng.standard_normal(n)) + 0.1
    inits['random_dirichlet'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # From parent
    if parent is not None:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits['parent'] = p.copy()
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
                # Also try with noise
                pnoisy = np.maximum(pp + rng.standard_normal(n) * 0.02, 0)
                pns = pnoisy.sum()
                if pns > 1e-10:
                    inits['parent_noisy'] = pnoisy / pns
    
    # Normalize all
    for k in inits:
        inits[k] = np.maximum(inits[k], 0).astype(np.float64)
        s = inits[k].sum()
        if s > 1e-10:
            inits[k] = inits[k] / s
    
    return inits

    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass
    
    # =========================================================
    # Phase 1: Quick gradient scan (150s)
    # =========================================================
    print("=== Phase 1: Quick Gradient Scan ===")
    t_start = time.monotonic()
    p1_budget = 150.0
    
    lengths = [384, 512, 768]
    scan_results = []
    
    for n in lengths:
        if time.monotonic() > deadline - 700:
            break
        inits = generate_diverse_inits(n, parent, rng)
        
        # Priority order
        priority = ['parent', 'parent_interp', 'parent_noisy', 'uniform', 'bimodal', 
                    'gauss_left', 'sine1', 'random', 'triangle']
        init_names = [nm for nm in priority if nm in inits][:5]
        
        t_per = max(8.0, (p1_budget - 10) / (len(lengths) * 5))
        
        for name in init_names:
            if time.monotonic() > deadline - 600:
                break
            t_alloc = min(t_per, deadline - time.monotonic() - 500)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            # Quick LSE-Adam
            a_opt, obj = lse_adam_optimize(a0, time_limit=t_alloc, T_init=300, T_end=1500, lr_init=0.015)
            
            # Quick LP step
            a_opt2, obj2 = lp_optimize(a_opt, time_limit=min(10, deadline - time.monotonic() - 400), verbose=False)
            if obj2 < obj:
                a_opt, obj = a_opt2, obj2
            
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score):
                scan_results.append((score, n, name, a_opt.copy()))
                if score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Phase1 n={n}, {name}: {score:.6f}")
    
    scan_results.sort(key=lambda x: x[0])
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 2: LP deep optimization (700s)
    # =========================================================
    print("=== Phase 2: LP Deep Optimization ===")
    t_start2 = time.monotonic()
    
    # Collect top candidates
    top_configs = []
    seen = set()
    for score, n, name, a_opt in scan_results[:6]:
        if n not in seen:
            top_configs.append((n, a_opt))
            seen.add(n)
    if best_sequence is not None and len(best_sequence) not in seen:
        top_configs.append((len(best_sequence), best_sequence.copy()))
    
    remaining = deadline - time.monotonic() - 500.0
    t_per_lp = max(30.0, remaining * 0.7 / max(1, len(top_configs)))
    
    for i, (n, a0) in enumerate(top_configs[:3]):
        if time.monotonic() > deadline - 400:
            break
        t_alloc = min(t_per_lp, deadline - time.monotonic() - 350)
        if t_alloc < 15:
            break
        print(f"  LP {i}: n={n}, t={t_alloc:.0f}s")
        a_opt, obj = lp_optimize(a0, time_limit=t_alloc, verbose=True)
        score = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(score):
            if score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  LP {i} (n={n}): {score:.6f} *NEW BEST*")
            else:
                print(f"  LP {i} (n={n}): {score:.6f}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 3: LSE-Adam polish with annealing (400s)
    # =========================================================
    print("=== Phase 3: LSE-Adam Polish ===")
    t_start3 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 300.0
        if remaining > 30:
            configs = [
                {'T_init': 200, 'T_end': 2000, 'lr_init': 0.01, 'top_k_ratio': 0.1},
                {'T_init': 500, 'T_end': 5000, 'lr_init': 0.008, 'top_k_ratio': 0.05},
                {'T_init': 100, 'T_end': 1000, 'lr_init': 0.015, 'top_k_ratio': 0.2},
            ]
            for ci, cfg in enumerate(configs):
                if time.monotonic() > deadline - 200:
                    break
                t_alloc = min(remaining / (len(configs) - ci) * 0.8, deadline - time.monotonic() - 150)
                if t_alloc < 10:
                    break
                
                # Add small perturbation for diversity
                if ci > 0:
                    noise = rng.standard_normal(len(best_sequence)) * 0.003 * ci
                    a0 = np.maximum(best_sequence + noise, 0)
                    s = a0.sum()
                    if s > 1e-15:
                        a0 = a0 / s
                    else:
                        a0 = best_sequence.copy()
                else:
                    a0 = best_sequence.copy()
                
                a_opt, obj = lse_adam_optimize(a0, time_limit=t_alloc, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  LSE-Adam {ci}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  LSE-Adam {ci}: {score:.6f}")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 4: Peak shaving (200s)
    # =========================================================
    print("=== Phase 4: Peak Shaving ===")
    t_start4 = time.monotonic()
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 150.0
        if remaining > 20:
            t_ps = min(remaining * 0.6, 120.0)
            a_opt, obj = peak_shaving(best_sequence, time_limit=t_ps)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  PeakShave: {score:.6f} *NEW BEST*")
            else:
                print(f"  PeakShave: {score:.6f}")
    
    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 5: Final combined polish (remaining time)
    # =========================================================
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        if remaining > 30:
            # Alternate short LP and LSE runs
            half = remaining / 2
            a_opt, obj = lp_optimize(best_sequence, time_limit=half * 0.7, verbose=False)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *NEW BEST*")
            
            remaining = deadline - time.monotonic() - 15.0
            if remaining > 10:
                a_opt, obj = lse_adam_optimize(best_sequence, time_limit=remaining,
                                                T_init=1000, T_end=5000, lr_init=0.005, top_k_ratio=0.05)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Final LSE: {score:.6f} *NEW BEST*")
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(512)
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
