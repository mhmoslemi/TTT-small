    b = np.maximum(b,0)

def project_simplex(v):
    n=len(v)
    if n==0: return v
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    cond = u * np.arange(1,n+1) > (cssv - 1.0)
    if not cond.any(): return np.ones(n)/n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0)/(rho+1.0)
    return np.maximum(v - theta, 0.0)
```

def softmax_from_logits(y):
    y = np.asarray(y, dtype=np.float64)

    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
       return np.ones(y.size)/y.size
    return e/s

def build_cheb_basis(n, D):
    B = np.zeros((n, D), dtype=np.float64)
    if n <= 1 or D <=0: return B
    t = 2.0 * np.arange(n, dtype=np.float64) / (n -1) - 1.0
    T0 = np.ones(n)
    T1 = t.copy()
    if D >=1:
       B[:,0] = T1
    for r in range(2, D+1):
       T2 = 2.0*t*T1 - T0
       B[:, r-1] = T2
       T0, T1 = T1, T2
    for j in range(D):
       col = B[:,j]
       col -= col.mean()
       norm = np.linalg.norm(col)
       if norm > 1e-14:
           col /= norm
       else:
           col[:] = 0.0
    return B
```

def cheb_fun(x, beta, n, B):
    y = B.dot(x)
    y = y - y.mean()
    a = softmax_from_logits(y)
    c = fast_convolve(a)
    c = np.maximum(c,0)
    m = float(np.max(c))
    scaled = beta*(c-m)
    smax = float(np.max(scaled))
    ex = np.exp(scaled - smax)
    sulf = float(ex.sum())
    lse = smax + np.log(sulf)
    val = 2.0*n*(m + lse/beta)
    w = ex/sulf
    d = cross_d_from_weights(w,a)
    mc = float(np.dot(w,c))
    gy = 2.0*a*(d - mc)
    gx = B.T.dot(gy)
    return val, gx

```python
"""
Search for a non-negative sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))^2.

The program uses a multi-scale, multi-start strategy:
1. Resample the parent construction to several larger lengths.
2. Generate smooth low-dimensional Chebyshev-log initializations and refine
   them with limited-memory BFGS on a smoothed-maximum objective.
3. Generate diverse one-parameter power-law / beta starts and refine the best
   ones with full-dimensional softmax L-BFGS.
4. Deeply refine the best candidates with increasingly sharp smoothing.
5. Apply zero-pruning and a short multiplicative peak-balancing polish.

All internally used sequences are normalized to sum 1; the objective is
scale-invariant.
"""

import numpy as np
import time
import warnings
from scipy.optimize import minimize

warnings.filterwarnings("ignore")


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    size = 2 * n - 1
    fsize = _next_pow2(size)
    A = np.fft.rfft(a, fsize)
    out = np.fft.irfft(A * A, fsize)
    return out[:size]


def cross_d_from_weights(w, a):
    """
    Returns d_j = sum_k w_k a_{k-j}, valid over j=0..n-1.
    Used for gradients of convolution coefficients under logit parameters.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    ar = a[::-1]
    size = len(w) + n - 1
    fsize = _next_pow2(size)
    W = np.fft.rfft(w, fsize)
    AR = np.fft.rfft(ar, fsize)
    conv = np.fft.irfft(W * AR, fsize)
    return conv[n - 1:n + n]


def softmax_from_logits(y):
    y = np.asarray(y, dtype=np.float64)
    if y.size == 0:
        return np.array([], dtype=np.float64)
    z = y - np.max(y)
    e = np.exp(z)
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.ones(y.size, dtype=np.float64) / y.size
    return e / s


def normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64)
    if a.ndim != 1:
        a = a.ravel()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-15:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return float("inf")
    a = normalize_nonneg(a)
    c = fast_convolve(a)
    m = float(np.max(c))
    return float(2.0 * a.size * m)


def project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not cond.any():
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def resample_sequence(a, n):
    a = normalize_nonneg(np.asarray(a, dtype=np.float64))
    if n <= 0:
        return np.array([], dtype=np.float64)
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / n
    if a.size == 1:
        return np.ones(n, dtype=np.float64) / n
    if n == 1:
        return np.array([1.0])
    x0 = np.linspace(0.0, 1.0, a.size, endpoint=True)
    x1 = np.linspace(0.0, 1.0, n, endpoint=True)
    b = np.interp(x1, x0, a)
    b = np.where(np.isfinite(b), b, 0.0)
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s < 1e-15:
        return np.ones(n, dtype=np.float64) / n
    return b / s


def make_gauge(n, pivot):
    idx = np.arange(n)
    mask = idx != pivot
    fill = idx[mask]
    return fill, mask


def gauge_fun(x, beta, n, fill_idx, mask):
    """
    Smoothed objective and gradient with respect to logit parameters x,
    where the pivot coordinate is fixed to 0.
    """
    if n == 1:
        return 2.0, np.array([], dtype=np.float64)

    y = np.zeros(n, dtype=np.float64)
    y[fill_idx] = x
    a = softmax_from_logits(y)

    c = fast_convolve(a)
    c = np.maximum(c, 0.0)
    m = float(np.max(c))

    beta = float(beta)
    scaled = beta * (c - m)
    smax = float(np.max(scaled))
    ex = np.exp(scaled - smax)
    sulf = float(ex.sum())
    lse = smax + np.log(sulf)
    val = 2.0 * n * (m + lse / beta)
    w = ex / sulf

    d = cross_d_from_weights(w, a)
    mc = float(np.dot(w, c))
    gy = 2.0 * a * (d - mc)

    gx = np.empty(n - 1, dtype=np.float64)
    gx[mask] = gy[mask]
    return float(val), gx


def build_cheb_basis(n, D):
    B = np.zeros((n, D), dtype=np.float64)
    if n <= 1 or D <= 0:
        return B

    t = 2.0 * np.arange(n, dtype=np.float64) / float(n - 1) - 1.0
    T0 = np.ones(n, dtype=np.float64)
    T1 = t.copy()

    if D >= 1:
        B[:, 0] = T1

    for r in range(2, D + 1):
        T2 = 2.0 * t * T1 - T0
        B[:, r - 1] = T2
        T0, T1 = T1, T2

    for j in range(D):
        col = B[:, j]
        col -= np.mean(col)
        norm = np.linalg.norm(col)
        if norm > 1e-14:
            col /= norm
        else:
            col[:] = 0.0
    return B


def cheb_fun(x, beta, n, B):
    """
    Smoothed objective in a low-dimensional zero-mean Chebyshev-log subspace.
    """
    if n == 1:
        return 2.0, np.zeros(B.shape[1], dtype=np.float64)

    y = B.dot(x)
    y = y - np.mean(y)
    a = softmax_from_logits(y)

    c = fast_convolve(a)
    c = np.maximum(c, 0.0)
    m = float(np.max(c))

    beta = float(beta)
    scaled = beta * (c - m)
    smax = float(np.max(scaled))
    ex = np.exp(scaled - smax)
    sulf = float(ex.sum())
    lse = smax + np.log(sulf)
    val = 2.0 * n * (m + lse / beta)
    w = ex / sulf

    d = cross_d_from_weights(w, a)
    mc = float(np.dot(w, c))
    gy = 2.0 * a * (d - mc)
    gx = B.T.dot(gy)
    return float(val), gx


def optimize_gauge_lbfgs(a0, betas, maxiters, deadline=None):
    """
    Full-dimensional refinement on softmax logits.
    The pivot coordinate is fixed to 0 to fix the softmax gauge.
    """
    a = normalize_nonneg(a0)
    n = a.size
    if n <= 1:
        best_a = a.copy()
        best_s = float(internal_score(a))
        return best_a, best_s

    best_a = a.copy()
    best_s = float(internal_score(a))

    pivot = int(np.argmax(a))
    fill_idx, mask = make_gauge(n, pivot)
    eps = np.finfo(np.float64).tiny
    y = np.log(a + eps) - np.log(a[pivot] + eps)
    x = y[fill_idx].astype(np.float64)
    x = np.nan_to_num(x, nan=0.0, posinf=100.0, neginf=-100.0)
    x = np.clip(x, -200.0, 200.0)

    if not isinstance(maxiters, (list, tuple)):
        maxiters = [int(maxiters)] * len(betas)

    for bi in range(len(betas)):
        if deadline is not None and time.monotonic() >= deadline - 5.0:
            break

        beta = float(betas[bi])
        maxit = int(maxiters[bi])
        if maxit < 1:
            maxit = 1

        try:
            options = {
                "maxiter": maxit,
                "maxfun": max(50, 5 * maxit),
                "ftol": 1e-12,
                "gtol": 1e-7,
                "maxls": 20,
                "disp": False,
            }
            res = minimize(
                gauge_fun,
                x,
                args=(beta, n, fill_idx, mask),
                method="L-BFGS-B",
                jac=True,
                options=options,
            )
            rx = getattr(res, "x", None)
            if rx is not None:
                rx = np.asarray(rx, dtype=np.float64)
                if rx.size == n - 1:
                    x = np.nan_to_num(rx, nan=0.0, posinf=100.0, neginf=-100.0)
                    x = np.clip(x, -200.0, 200.0)
        except Exception:
            pass

        y = np.zeros(n, dtype=np.float64)
        y[fill_idx] = x
        a_new = softmax_from_logits(y)
        s_new = float(internal_score(a_new))

        if np.isfinite(s_new) and s_new < best_s:
            best_s = s_new
            best_a = a_new.copy()

        if not np.isfinite(s_new):
            pivot = int(np.argmax(best_a))
            fill_idx, mask = make_gauge(n, pivot)
            y2 = np.log(best_a + eps) - np.log(best_a[pivot] + eps)
            x = y2[fill_idx].astype(np.float64)
            x = np.nan_to_num(x, nan=0.0, posinf=100.0, neginf=-100.0)
            x = np.clip(x, -200.0, 200.0)

    return best_a, best_s


def refine_sequence(a0, deadline, betas, maxiters):
    a0 = normalize_nonneg(a0)
    s0 = float(internal_score(a0))
    try:
        a1, s1 = optimize_gauge_lbfgs(a0, betas, maxiters, deadline)
    except Exception:
        return a0.copy(), s0
    if not np.isfinite(s1) or s1 > s0:
        return a0.copy(), s0
    return a1, s1


def low_dim_search(n, D, restarts, beta, maxiter, deadline, rng):
    if n <= 0 or D <= 0:
        return None, float("inf")

    B = build_cheb_basis(n, D)
    best_a = None
    best_s = float("inf")

    for _ in range(int(max(0, restarts))):
        if time.monotonic() >= deadline - 5.0:
            break

        amp = float(rng.uniform(0.5, 3.0))
        x = amp * rng.standard_normal(D)
        if D > 3:
            x *= np.linspace(1.0, 0.5, D, dtype=np.float64)

        try:
            options = {
                "maxiter": int(maxiter),
                "maxfun": max(50, 5 * int(maxiter)),
                "ftol": 1e-9,
                "gtol": 1e-5,
                "maxls": 20,
                "disp": False,
            }
            res = minimize(
                cheb_fun,
                x,
                args=(float(beta), n, B),
                method="L-BFGS-B",
                jac=True,
                options=options,
            )
            rx = getattr(res, "x", None)
            if rx is not None:
                rx = np.asarray(rx, dtype=np.float64)
                if rx.size == D:
                    rx = np.nan_to_num(rx, nan=0.0, posinf=100.0, neginf=-100.0)
                    rx = np.clip(rx, -100.0, 100.0)
                    y = B.dot(rx)
                    y = y - np.mean(y)
                    a = softmax_from_logits(y)
                    s = float(internal_score(a))
                    if np.isfinite(s) and s < best_s:
                        best_s = s
                        best_a = a.copy()
        except Exception:
            pass

    return best_a, best_s

def select_top_candidates(cand_list, max_total, max_per_len):
    items = []
    for i, a in enumerate(cand_list):
        if a is None:
            continue
        a = np.asarray(a, dtype=np.float64)
        if a.size == 0:
            continue
        s = float(internal_score(a))
        if np.isfinite(s):
            items.append((s, i, a))
    items.sort()

    selected = []
    counts = {}
    for s, i, a in items:
        n = a.size
        cnt = counts.get(n, 0)
        if cnt >= max_per_len:
            continue
        selected.append((s, a))
        counts[n] = cnt + 1
        if len(selected) >= max_total:
            break
    return selected


def zero_prune_search(a, fracs=(0.001, 0.005, 0.01, 0.02, 0.05)):
    a = normalize_nonneg(a)
    n = a.size
    if n <= 1:
        return a.copy(), float(internal_score(a))

    best_a = a.copy()
    best_s = float(internal_score(a))

    for q in fracs:
        k = int(float(q) * n)
        if k <= 0 or k >= n:
            continue
        idx = np.argsort(a)[:k]
        a2 = a.copy()
        a2[idx] = 0.0
        a2 = normalize_nonneg(a2)
        s2 = float(internal_score(a2))
        if np.isfinite(s2) and s2 < best_s:
            best_s = s2
            best_a = a2.copy()

    return best_a, best_s


def polish_sequence(a0, deadline, rng):
    """Short exact peak-balancing multiplicative polish."""
    a = normalize_nonneg(a0)
    n = a.size
    if n <= 1:
        return a.copy(), float(internal_score(a))

    best_a = a.copy()
    best_s = float(internal_score(a))
    tiny = np.finfo(np.float64).tiny
    alpha = 0.05
    heavy = 0

    for _ in range(50000):
        if time.monotonic() >= deadline - 10.0:
            break

        c = fast_convolve(a)
        M = float(np.max(c))
        if M <= 0.0:
            break

        temp = max(1e-12, 0.03 * M)
        sel = c > M - temp
        if not np.any(sel):
            sel[int(np.argmax(c))] = True
        cv = c[sel]
        scaled = (cv - M) / temp
        scaled -= np.max(scaled)
        wsel = np.exp(scaled)
        wsel /= wsel.sum()
        w = np.zeros_like(c, dtype=np.float64)
        w[sel] = wsel

        p = cross_d_from_weights(w, a)
        logs = np.log(a + tiny)

        best_local = best_s
        best_local_a = best_a
        for al in (alpha * 0.2, alpha * 0.5, alpha, alpha * 1.5, alpha * 2.0, -alpha * 0.5):
            a_new = softmax_from_logits(logs - al * p)
            s_new = float(internal_score(a_new))
            if s_new < best_local:
                best_local = s_new
                best_local_a = a_new

        if best_local < best_s - 1e-12:
            a = best_local_a.copy()
            best_s = best_local
            best_a = a.copy()
            alpha = max(0.005, alpha * 0.97)
            heavy = 0
        else:
            heavy += 1
            alpha *= 0.9
            if alpha < 0.003 or heavy > 25:
                logs = np.log(best_a + tiny)
                scale = 0.01 + 0.015 * float(heavy % 5)
                logs = logs + scale * rng.standard_normal(n)
                a = softmax_from_logits(logs)
                alpha = 0.03
                heavy = 0

    return best_a, best_s


def consider_candidate(candidates, new_a, current_seq, current_best):
    if new_a is None:
        return current_seq, current_best
    a = normalize_nonneg(new_a)
    if a.size == 0:
        return current_seq, current_best
    s = float(internal_score(a))
    if not np.isfinite(s):
        return current_seq, current_best
    candidates.append(a.copy())
    if s < current_best:
        return a.copy(), s
    return current_seq, current_best

            continue
        alloc = float(min(70.0, rem))
        if n <= 640:
            D = 32
            restarts = max(4, int(alloc / 12.0))
            a_low2, s_low2 = low_dim_search(n, D, restarts, 8000, 60, time.monotonic() + alloc, rng)
            if a_low2 is not None and np.isfinite(s_low2) and s_low2 < best_score:
                print(f"Low-dim D=32 new best: n={n}, score={s_low2:.10f}")
                best_seq, best_score = consider_candidate(candidates, a_low2, best_seq, best_score)
                low_best[n] = a_low2.copy()

    # Phase 2: diverse starts + quick full-dimensional refinement.
    phase2_deadline = min(deadline - 420.0, time.monotonic() + 500.0)
    quick_betas = (2000.0, 6000.0, 15000.0)
    for n in lengths:
        now = time.monotonic()
        if now >= phase2_deadline - 60.0:
            break

        starts = generate_deterministic_starts(n, parent, rng, num_beta=16)
        if n in low_best:
            starts.append(low_best[n])

        items = []
        for idx, a in enumerate(starts):
            a = normalize_nonneg(a)
            s = float(internal_score(a))
            if np.isfinite(s):
                items.append((s, idx, a))
        items.sort()

        if n <= 768:
            max_quick = 5
        elif n <= 1536:
            max_quick = 3
        else:
            max_quick = 2

        for rank in range(min(max_quick, len(items))):
            now = time.monotonic()
            if now >= phase2_deadline - 35.0:
                break
            _, _, a = items[rank]
            per = float(min(80.0, (phase2_deadline - now - 25.0) / max(1, max_quick - rank)))
            cand_deadline = min(phase2_deadline - 25.0, now + max(15.0, per))

            if n <= 768:
                maxiters = [55, 70, 90]
            elif n <= 1536:
                maxiters = [35, 45, 60]
            else:
                maxiters = [25, 35, 45]

            a2, s2 = refine_sequence(a, cand_deadline, quick_betas, maxiters)
            before = best_score
            best_seq, best_score = consider_candidate(candidates, a2, best_seq, best_score)
            if best_score < before:
                print(f"Quick refine improved: n={a2.size}, score={best_score:.10f}")

    # Phase 3: deep refinement of the best known candidates.
    phase3_deadline = min(deadline - 120.0, time.monotonic() + 500.0)
    top = select_top_candidates(candidates, max_total=8, max_per_len=2)
    deep_betas = (5000.0, 15000.0, 40000.0, 100000.0)

    for idx, (s0, a) in enumerate(top):
        now = time.monotonic()
        if now >= phase3_deadline - 60.0:
            break
        per = float(min(160.0, (phase3_deadline - now - 45.0) / max(1, len(top) - idx)))
        cand_deadline = min(phase3_deadline - 45.0, now + max(25.0, per))

        n = a.size
        if n <= 768:
            maxiters = [90, 110, 130, 160]
        elif n <= 1536:
            maxiters = [60, 75, 90, 120]
        else:
            maxiters = [40, 55, 70, 90]

        a2, s2 = refine_sequence(a, cand_deadline, deep_betas, maxiters)
        before = best_score
        best_seq, best_score = consider_candidate(candidates, a2, best_seq, best_score)
        if best_score < before:
            print(f"Deep refine improved: n={a2.size}, score={best_score:.10f}")

    # Phase 4: zero-pruning and short multiplicative polish on remaining best.
    final_top = select_top_candidates(candidates, max_total=5, max_per_len=2)
    for _, a in final_top:
        if time.monotonic() >= deadline - 35.0:
            break

        a_pruned, s_pruned = zero_prune_search(a)
        before = best_score
        best_seq, best_score = consider_candidate(candidates, a_pruned, best_seq, best_score)
        if best_score < before:
            print(f"Zero-prune improved: n={a_pruned.size}, score={best_score:.10f}")

        if time.monotonic() < deadline - 30.0:
            n = a_pruned.size
            polish_time = min(deadline - 25.0, time.monotonic() + (45.0 if n <= 1024 else 25.0))
            if polish_time > time.monotonic() + 10.0:
                a_pol, s_pol = polish_sequence(a_pruned, polish_time, rng)
                before = best_score
                best_seq, best_score = consider_candidate(candidates, a_pol, best_seq, best_score)
                if best_score < before:
                    print(f"Polish improved: n={a_pol.size}, score={best_score:.10f}")

    # Final verification against the actual evaluator.
    if best_seq is None:
        best_seq = np.ones(384, dtype=np.float64) / 384.0
        best_score = float(internal_score(best_seq))
        candidates.append(best_seq.copy())

    final_score = float("inf")
    final_seq = best_seq
    final_items = select_top_candidates(candidates, max_total=8, max_per_len=4)
    final_list = [final_seq]
    for _, a in final_items:
        final_list.append(a)

    for a in final_list:
        try:
            a = normalize_nonneg(a)
            score_eval = float(evaluate_sequence([float(x) for x in a]))
        except Exception:
            score_eval = float("inf")
        if np.isfinite(score_eval) and score_eval < final_score:
            final_score = score_eval
            final_seq = a.copy()

    if not np.isfinite(final_score):
        final_seq = best_seq
        final_score = float(internal_score(final_seq))

    print(f"FINAL internal-like score: {final_score:.10f}, length: {final_seq.size}")
    return [float(x) for x in final_seq]
