Multi-phase optimizer for minimizing 2*n*max(auto_conv(a))/sum(a)^2.
New approaches vs previous: Adam optimizer on smoothed objective,

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_dir(f, M):
    """Solve LP: max 1^T g s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def lse_gradient(conv, a, T):
    """Gradient of T*log(sum(exp(conv/T))) w.r.t. a using FFT cross-correlation."""
    n = len(a)
    scaled = conv / T
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-300:
        p = np.zeros(len(conv))
        p[np.argmax(conv)] = 1.0
    else:
        p = w / ws
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]


def power_mean_gradient(conv, a, p):
    """Gradient of (sum conv^p)^(1/p) w.r.t. a."""
    n = len(a)
    cp = np.maximum(conv, 1e-30) ** p
    total = cp.sum()
    if total < 1e-300:
        return np.zeros(n)
    # d/d a_i of (sum conv^p)^(1/p) = (1/p) * (sum conv^p)^(1/p-1) * p * sum_k conv_k^(p-1) * d conv_k / d a_i
    # = (sum conv^p)^(1/p-1) * sum_k conv_k^(p-1) * 2*a_{k-i}
    weights = cp * (conv / total)  # proportional to conv^p * conv / sum(conv^p) = conv^(p+1)/total
    # Actually: weights_k = conv_k^(p-1) / (sum conv_j^p)^(1/p) * (1/p) * p = conv_k^(p-1) / total^(1/p)
    # Let's just use: grad_i = 2 * sum_k w_k * a_{k-i} where w_k proportional to conv_k^(p-1)
    w = cp / (total * total)  # normalized weights proportional to conv^p
    # Better: w_k = conv_k^(p-1) / sum(conv_j^(p-1))
    pred = np.maximum(conv, 1e-30) ** (p - 1)
    ps = pred.sum()
    if ps < 1e-300:
        return np.zeros(n)
    p_norm = pred / ps
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    P = np.fft.rfft(p_norm, fft_size)
    cross = np.fft.irfft(np.conj(A) * P, fft_size)
    return 2.0 * cross[:n]

    it = 0

            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()

            # Slowly tighten M target
            m_exp = max(m_exp * m_tighten_rate, m_min)

            if M < best_obj - 1e-15:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}, m_exp={m_exp:.4f}")
            else:
                no_improve += 1
                if no_improve % 15 == 0:
                    m_exp = max(m_exp * 0.97, m_min * 0.9)
        else:
            no_improve += 1
            m_exp = min(m_exp * m_relax_rate, 4.0)

        if no_improve > 50:
            rng = np.random.default_rng((it * 7919 + n * 104729) % (2**31))
            scale = 0.001 + 0.002 * min(no_improve // 50, 4)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = m_start
            no_improve = 0

    return best_f, best_obj


def adam_optimize(f0, time_limit, T_init=0.004, T_min=0.00005, beta1=0.8, beta2=0.999,
                  lr_init=0.005, lr_min=0.0001, verbose=False):
    """Adam optimizer on simplex with log-sum-exp smoothed max objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    T = T_init
    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    eps = 1e-8

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_step += 1

        conv = fast_convolve(f)
        grad = lse_gradient(conv, f, T)

        # Project gradient to tangent space of simplex
        grad = grad - np.mean(grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f_new = project_simplex(f_new)

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    Adam iter {it}: obj={best_obj:.10f}, T={T:.6f}, lr={lr:.6f}")
        else:
            no_improve += 1
            if no_improve % 30 == 0:
                lr = max(lr * 0.9, lr_min)

        # Anneal T
        T = max(T * 0.996, T_min)

        # Periodic restart
        if no_improve > 400:
            rng = np.random.default_rng((it * 331 + 17) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            m = np.zeros(n)
            v = np.zeros(n)
            t_step = 0
            lr = max(lr_init * 0.3, lr_min)
            T = max(T_init * 0.2, T_min)
            no_improve = 0

    return best_f, best_obj


def nesterov_subgrad(f0, time_limit, lr=0.002, temp=0.00005, top_k_frac=0.1, verbose=False):
    """Nesterov-accelerated subgradient method on the max of auto-convolution."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    y = f.copy()  # Nesterov's y
    mu = 1.0  # momentum parameter

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    t_global = np.zeros(n)  # for adaptive lr
    top_k = max(5, int(n * top_k_frac))

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_global += 1

        # Nesterov lookahead
        y = project_simplex(f + mu * (f - project_simplex(y - mu * 0)))
        # Simpler: y = f + beta * (f - f_prev), but let's use standard formula
        # Actually, let's just do: f_new = y - lr * subgrad(y), then y = f_new + beta*(f_new - f)
        # Standard Nesterov: y_k = x_k + beta_k (x_k - x_{k-1}), then x_{k+1} = y_k - step * g(y_k)

        conv_y = fast_convolve(y)
        peaks = np.argpartition(conv_y, -top_k)[-top_k:]
        pv = conv_y[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[k - idx]

        # Project gradient to tangent
        sg = sg - np.mean(sg)

        # Adaptive step size
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            step = lr
        else:
            step = lr * 0.01 / (np.linalg.norm(f - y) + 0.01)
            step = min(step, lr)

        f_new = project_simplex(y - step * sg)

        # Update momentum (increasing)
        mu = 2.0 / (it ** 0.5 + 2.0)
        y = f_new + mu * (f_new - f)
        y = np.maximum(y, 0.0)
        ys = y.sum()
        if ys > 1e-15:
            y = y / ys
        else:
            y = f_new.copy()

        f = f_new
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-15:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
            if verbose and it % 500 == 0:
                print(f"    Nesterov iter {it}: obj={best_obj:.10f}")
        else:
            no_improve += 1

        if no_improve > 500:
            rng = np.random.default_rng((it * 733 + 41) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            y = f.copy()
            lr = lr * 0.7
            no_improve = 0

    return best_f, best_obj


def power_mean_optimize(f0, time_limit, p=8, lr_init=0.003, verbose=False):
    """Optimize using power-mean objective with Adam-style updates."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    lr = lr_init
    m = np.zeros(n)
    v = np.zeros(n)
    t_step = 0
    beta1, beta2 = 0.85, 0.999
    eps = 1e-8
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        t_step += 1

        conv = fast_convolve(f)
        grad = power_mean_gradient(conv, f, p)
        grad = grad - np.mean(grad)

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)

        f_new = project_simplex(f - lr * m_hat / (np.sqrt(v_hat) + eps))

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            no_improve = 0
            f = f_new
            if verbose and it % 200 == 0:
                print(f"    PM iter {it}: obj={best_obj:.10f}, p={p}")
        else:
            no_improve += 1
            if no_improve % 40 == 0:
                lr = max(lr * 0.92, 0.0001)

        if no_improve > 500:
            rng = np.random.default_rng((it * 557 + 23) % (2**31))
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            m = np.zeros(n)
            v = np.zeros(n)
            t_step = 0
            lr = lr_init * 0.3
            no_improve = 0

    return best_f, best_obj


def u_shape_init(n, sharpness=2.0, rng=None):
    """Generate a U-shaped initialization (high at edges, low in middle)."""
    x = np.arange(n, dtype=np.float64)
    dist_from_center = np.abs(x - 0.5 * (n - 1))
    # U-shape: higher at edges
    w = 1.0 / (1.0 + (dist_from_center / (0.5 * n)) ** sharpness)
    # Add small noise for diversity
    if rng is not None:
        w += rng.standard_normal(n) * 0.01
    w = np.maximum(w, 0.0)
    w = w / w.sum()
    return w


def edge_concentrated_init(n, alpha=1.5, rng=None):
    """Edge-concentrated initialization."""
    x = np.arange(n, dtype=np.float64)
    d = np.minimum(x, n - 1 - x) + 1.0
    w = d ** (-alpha)
    if rng is not None:
        w += rng.standard_normal(n) * 0.02 * w.max()
    w = np.maximum(w, 0.0)
    w = w / w.sum()
    return w


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search with Adam, Nesterov, LP-FW,
    and power-mean methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    # === Phase 1: Length & Shape Exploration (250s) ===
    print("=== Phase 1: Exploration ===")
    t_p1 = time.monotonic()
    p1_budget = 250.0

    # Base length
    base_n = len(parent) if parent is not None else 384

    # Try several lengths
    lengths = sorted(set([
        base_n,
        base_n - 16, base_n - 32, base_n + 16, base_n + 32, base_n + 64,
        256, 320, 448, 512
    ]))
    lengths = [l for l in lengths if 192 <= l <= 640]
    # Shuffle non-base
    other = [l for l in lengths if l != base_n]
    rng.shuffle(other)
    eval_lengths = [base_n] + other[:5]

    for n in eval_lengths:
        rem = p1_budget - (time.monotonic() - t_p1)
        if rem < 20:
            break
        t_alloc = min(45.0, rem - 10.0)
        if t_alloc < 10:
            break

        # Multiple inits
        inits = [
            u_shape_init(n, 2.0, rng),
            u_shape_init(n, 3.0, rng),
            edge_concentrated_init(n, 1.5, rng),
            np.ones(n) / n,
        ]
        if parent is not None:
            px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                inits.append(px / ps)

        # Pick best init by quick eval
        best_init = None
        best_init_obj = float('inf')
        for fi in inits:
            obj = fast_convolve(fi).max()
            if obj < best_init_obj:
                best_init_obj = obj
                best_init = fi

        if best_init is None:
            best_init = np.ones(n) / n

        print(f"  Exploring n={n}, t={t_alloc:.0f}s")
        a_opt, _ = fw_lp_optimize(best_init, time_limit=t_alloc)
        try_seq(a_opt, f"exp_n{n}")

    print(f"  Phase 1 done: best={best_score:.6f}")

    # === Phase 2: Deep Optimization (900s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t_p2 = time.monotonic()
    p2_budget = 900.0

    if best_seq is None:
        best_seq = np.ones(base_n) / base_n
        best_score = float(evaluate_sequence(best_seq.tolist()))

    n_best = len(best_seq)
    f_work = best_seq / best_seq.sum()

    # Alternate methods
    n_cycles = 5
    t_per_cycle = p2_budget / n_cycles

    for cycle in range(n_cycles):
        rem = p2_budget - (time.monotonic() - t_p2)
        if rem < 20:
            break

        # A: LP-FW
        t_a = min(t_per_cycle * 0.35, rem * 0.35)
        if t_a > 10:
            print(f"  Cycle {cycle}: LP-FW ({t_a:.0f}s)")
            a_opt, _ = fw_lp_optimize(f_work, time_limit=t_a, m_start=1.0, m_min=0.93)
            try_seq(a_opt, f"p2_fw_{cycle}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        rem = p2_budget - (time.monotonic() - t_p2)
        # B: Adam
        t_b = min(t_per_cycle * 0.25, rem * 0.3)
        if t_b > 10:
            T_init = max(0.001, 0.003 * fast_convolve(f_work).max())
            print(f"  Cycle {cycle}: Adam ({t_b:.0f}s, T={T_init:.5f})")
            a_opt, _ = adam_optimize(f_work, time_limit=t_b, T_init=T_init, T_min=0.00003)
            try_seq(a_opt, f"p2_adam_{cycle}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        rem = p2_budget - (time.monotonic() - t_p2)
        # C: Nesterov subgradient
        t_c = min(t_per_cycle * 0.2, rem * 0.3)
        if t_c > 10:
            temp = max(0.00002, 0.0005 * fast_convolve(f_work).max())
            print(f"  Cycle {cycle}: Nesterov ({t_c:.0f}s)")
            a_opt, _ = nesterov_subgrad(f_work, time_limit=t_c, lr=0.002, temp=temp, top_k_frac=0.08)
            try_seq(a_opt, f"p2_nes_{cycle}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        rem = p2_budget - (time.monotonic() - t_p2)
        # D: Power-mean
        t_d = min(t_per_cycle * 0.2, rem * 0.3)
        if t_d > 10:
            print(f"  Cycle {cycle}: PowerMean ({t_d:.0f}s)")
            a_opt, _ = power_mean_optimize(f_work, time_limit=t_d, p=8, lr_init=0.002)
            try_seq(a_opt, f"p2_pm_{cycle}")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

    print(f"  Phase 2 done: best={best_score:.6f}")

    # === Phase 3: Fine Polish (500s) ===
    print(f"\n=== Phase 3: Fine Polish ===")
    t_p3 = time.monotonic()
    p3_budget = 500.0

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()

        # 3a: Very tight LP-FW
        rem = p3_budget - (time.monotonic() - t_p3)
        t_alloc = min(150.0, rem - 20)
        if t_alloc > 10:
            print(f"  Fine LP-FW ({t_alloc:.0f}s)")
            a_opt, _ = fw_lp_optimize(f_work, time_limit=t_alloc, m_start=0.98, m_min=0.93)
            try_seq(a_opt, "p3_fw_fine")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # 3b: Fine Adam
        rem = p3_budget - (time.monotonic() - t_p3)
        t_alloc = min(120.0, rem - 20)
        if t_alloc > 10:
            T_init = max(0.0003, 0.0008 * fast_convolve(f_work).max())
            print(f"  Fine Adam ({t_alloc:.0f}s)")
            a_opt, _ = adam_optimize(f_work, time_limit=t_alloc, T_init=T_init, T_min=0.00001,
                                     lr_init=0.001, beta1=0.9, beta2=0.999)
            try_seq(a_opt, "p3_adam_fine")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # 3c: Nesterov fine
        rem = p3_budget - (time.monotonic() - t_p3)
        t_alloc = min(100.0, rem - 20)
        if t_alloc > 10:
            temp = max(0.00001, 0.0002 * fast_convolve(f_work).max())
            print(f"  Fine Nesterov ({t_alloc:.0f}s)")
            a_opt, _ = nesterov_subgrad(f_work, time_limit=t_alloc, lr=0.001, temp=temp, top_k_frac=0.05)
            try_seq(a_opt, "p3_nes_fine")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

        # 3d: Power-mean p=16
        rem = p3_budget - (time.monotonic() - t_p3)
        t_alloc = min(100.0, rem - 10)
        if t_alloc > 10:
            print(f"  Fine PowerMean p=16 ({t_alloc:.0f}s)")
            a_opt, _ = power_mean_optimize(f_work, time_limit=t_alloc, p=16, lr_init=0.001)
            try_seq(a_opt, "p3_pm16")
            if best_seq is not None:
                f_work = best_seq / best_seq.sum()

    print(f"  Phase 3 done: best={best_score:.6f}")

    # === Phase 4: Final tricks (100s) ===
    print(f"\n=== Phase 4: Final ===")

    if best_seq is not None:
        n_best = len(best_seq)
        f_work = best_seq / best_seq.sum()

        # Try small length changes
        for delta in [8, -8, 16, -16]:
            if time.monotonic() > deadline - 15:
                break
            new_n = n_best + delta
            if new_n < 128 or new_n > 1024:
                continue
            px = np.interp(np.linspace(0, 1, new_n), np.linspace(0, 1, n_best), f_work)
            px = project_simplex(px)
            t_alloc = min(25.0, deadline - time.monotonic() - 10)
            if t_alloc > 10:
                a_opt, _ = fw_lp_optimize(px, time_limit=t_alloc, m_start=0.98, m_min=0.94)
                try_seq(a_opt, f"p4_len{new_n}")

        # Mirror + polish
        if time.monotonic() < deadline - 20:
            mirrored = 0.5 * f_work + 0.5 * f_work[::-1]
            mirrored = project_simplex(mirrored)
            if np.max(np.abs(mirrored - f_work)) > 1e-8:
                t_alloc = min(20.0, deadline - time.monotonic() - 10)
                if t_alloc > 5:
                    a_opt, _ = fw_lp_optimize(mirrored, time_limit=t_alloc)
                    try_seq(a_opt, "p4_mirror")

        # Tiny perturbation restarts
        for _ in range(3):
            if time.monotonic() > deadline - 10:
                break
            noise = rng.standard_normal(n_best) * 0.0003
            f_pert = project_simplex(f_work + noise)
            t_alloc = min(15.0, deadline - time.monotonic() - 5)
            if t_alloc > 5:
                a_opt, _ = adam_optimize(f_pert, time_limit=t_alloc, T_init=0.0005,
                                         T_min=0.00001, lr_init=0.001)
                try_seq(a_opt, "p4_perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
