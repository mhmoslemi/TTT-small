"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using continuation L-BFGS on softmax
parameterization with log-sum-exp smoothing, followed by LP-based active-set
polish and structured peak-targeting perturbations.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smooth_max_objective(a, tau):
    """
    Compute smooth approximation of max(conv(a,a)) and gradient w.r.t. a.
    Returns (obj, grad_a, S, p) where obj = (1/tau)*log(sum exp(tau*S_k)),
    grad_a = d(obj)/da, S = conv(a,a), p = softmax(tau*S).
    """
    n = len(a)
    S = fast_convolve(a)
    S_max = S.max()
    scaled = tau * (S - S_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    p_sum = exp_s.sum()
    p = exp_s / p_sum
    obj = S_max + np.log(p_sum) / tau
    
    # Gradient w.r.t. a: grad[j] = 2 * sum_k p[k] * a[k-j]
    # Compute via convolution
    a_flip = a[::-1]
    h = np.convolve(p, a_flip)
    g = h[n-1 : 2*n-1]
    grad = 2.0 * g
    
    return obj, grad, S, p

def smooth_objective_and_grad_theta(theta, tau):
    """
    Objective and gradient w.r.t. theta for softmax parameterization.
    a = softmax(theta), minimize smooth_max(a, tau).
    """
    # Compute softmax
    theta_shift = theta - theta.max()
    exp_t = np.exp(theta_shift)
    a = exp_t / exp_t.sum()
    
    obj, grad_a, S, p = smooth_max_objective(a, tau)
    
    # Convert gradient from a-space to theta-space
    # dobj/dtheta_j = a_j * (grad_a_j - dot(grad_a, a))
    ga_dot_a = np.dot(grad_a, a)
    grad_theta = a * (grad_a - ga_dot_a)
    
    return obj, grad_theta

def lbfgs_continue(a_init, tau_schedule, time_limit, verbose=False):
    """
    Run L-BFGS optimization with continuation on tau.
    a_init: initial sequence (will be normalized).
    tau_schedule: list of tau values to try (increasing).
    Returns best sequence found.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_true_obj = fast_convolve(a).max()
    
    t0 = time.monotonic()
    theta = np.log(a)  # initial theta
    
    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        
        def fun(X):
            obj, grad = smooth_objective_and_grad_theta(X, tau)
            return obj, grad
        
        try:
            res = minimize(fun, theta, method='L-BFGS-B',
                          jac=True, options={'maxiter': 200, 'ftol': 1e-12,
                                             'gtol': 1e-8, 'maxls': 50})
        except Exception as e:
            if verbose:
                print(f"    L-BFGS error at tau={tau}: {e}")
            continue
        
        # Convert back to a
        theta_final = res.x
        theta_shift = theta_final - theta_final.max()
        exp_t = np.exp(theta_shift)
        a_new = exp_t / exp_t.sum()
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        true_obj = fast_convolve(a_new).max()
        if true_obj < best_true_obj:
            best_true_obj = true_obj
            best_a = a_new.copy()
            if verbose:
                print(f"    tau={tau}: true_obj={true_obj:.10f}, res_obj={res.fun:.10f}, iters={res.nit}")
        else:
            if verbose:
                print(f"    tau={tau}: no improvement (true_obj={true_obj:.10f} >= best={best_true_obj:.10f})")
        
        theta = theta_final
    
    return best_a, best_true_obj

def solve_peak_reducing_lp(a, S, top_k, M_target):
    """
    Solve LP to find direction that reduces peak convolution.
    min t s.t. (a+g)*a conv <= M_target at top_k peaks, sum(g)=0, g >= -a
    """
    n = len(a)
    
    # Get top_k peak indices
    if top_k >= len(S):
        peaks = np.arange(len(S))
    else:
        peaks = np.argpartition(S, -top_k)[-top_k:]
    
    # Build constraint matrix for the selected peaks
    # (a+g)*a at lag k = S[k] + 2*(a*g)[k] - ... 
    # Actually: ((a+g)* (a+g))[k] ≈ S[k] + 2*(a*g)[k] (linearized)
    # We want: S[k] + 2*(a*g)[k] <= M_target
    # => 2*(a*g)[k] <= M_target - S[k]
    # (a*g)[k] = sum_j a[j]*g[k-j]
    
    # Variables: g_0, ..., g_{n-1}
    # Constraint for peak k: sum_j a[j]*g[k-j] <= (M_target - S[k])/2
    
    A_ub = np.zeros((len(peaks), n))
    for idx, k in enumerate(peaks):
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                A_ub[idx, j] = a[m]
    
    b_ub = (M_target - S[peaks]) / 2.0
    
    # Equality: sum(g) = 0
    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    
    # Bounds: g >= -a
    bounds = [(-a[i], None) for i in range(n)]
    
    # Objective: minimize sum of positive part of g (to prefer spreading)
    # Or just minimize 0 (feasibility)
    c = np.zeros(n)
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs', options={'presolve': True})
    if res.success and res.x is not None:
        return res.x
    return None

def lp_polish(a, time_limit, verbose=False):
    """
    LP-based active set polish. Iteratively reduce the peak using LP directions.
    """
    n = len(a)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    
    S = fast_convolve(a)
    M = S.max()
    best_a = a.copy()
    best_M = M
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    rng = np.random.default_rng(777)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        S = fast_convolve(a)
        M = S.max()
        
        # Try to reduce M by a small amount
        for frac in [0.001, 0.005, 0.01, 0.02, 0.05]:
            M_target = M * (1 - frac)
            g = solve_peak_reducing_lp(a, S, top_k=min(20, 2*n-1), M_target=M_target)
            if g is None:
                continue
            
            # Line search on (1-alpha)*a + alpha*(a+g)
            best_alpha = 0.0
            best_val = M
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_trial = a + alpha * g
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                val = fast_convolve(a_trial).max()
                if val < best_val:
                    best_val = val
                    best_alpha = alpha
            
            if best_alpha > 0:
                a = a + best_alpha * g
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                val = fast_convolve(a).max()
                if val < best_M - 1e-12:
                    best_M = val
                    best_a = a.copy()
                    no_improve = 0
                    break
                else:
                    # Revert
                    a = best_a.copy()
            else:
                continue
        
        # If we didn't break out of the frac loop, we made progress
        # Check no_improve
        S = fast_convolve(a)
        if S.max() < best_M - 1e-12:
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 30:
            # Perturb and restart
            pert = rng.standard_normal(n) * 0.003
            a = project_simplex(best_a + pert)
            no_improve = 0
            if verbose and it % 50 == 0:
                print(f"    LP polish iter {it}: best_M={best_M:.10f}")
    
    return best_a, best_M

def momentum_smoothed(a, time_limit, tau=50, lr=0.01, momentum=0.9, verbose=False):
    """
    Momentum-based optimization using smooth gradient.
    Works in a-space with simplex projection.
    """
    n = len(a)
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    
    v = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    noise_scale = 0.002
    rng = np.random.default_rng(42)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        _, grad_a, S, _ = smooth_max_objective(a, tau)
        
        # Project gradient to tangent space of simplex
        grad_proj = grad_a - np.mean(grad_a)
        
        v = momentum * v - lr * grad_proj
        a_new = project_simplex(a + v)
        
        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
        
        a = a_new
        
        # Adaptive learning rate
        lr = max(lr * 0.9998, 0.0001)
        
        if it % 200 == 0:
            # Small perturbation to escape local minima
            a = project_simplex(a + rng.standard_normal(n) * noise_scale)
            noise_scale *= 0.9
    
    return best_a, best_obj

def build_conv_matrix_rows(a, peaks):
    """Build rows of the convolution matrix for given peak indices."""
    n = len(a)
    m = len(peaks)
    B = np.zeros((m, n))
    for idx, k in enumerate(peaks):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            B[idx, j] = a[k - j]
    return B

def structured_perturb(a, S, rng, scale=0.01):
    """
    Perturb the sequence to specifically target peak-reduction.
    Reduce mass at positions that contribute to peaks, increase at others.
    """
    n = len(a)
    S_max = S.max()
    # Find active peaks (within 1% of max)
    active = np.where(S > 0.995 * S_max)[0]
    
    # For each active peak k, the pairs (i, k-i) contribute.
    # Create a "repulsion" signal: positions that are in many active peak pairs get reduced.
    repulsion = np.zeros(n)
    for k in active:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                repulsion[i] += a[i] * a[j]
                repulsion[j] += a[i] * a[j]
    
    # Perturb: reduce mass where repulsion is high, increase where low
    perturbation = -repulsion * scale + rng.standard_normal(n) * scale * 0.3
    a_new = project_simplex(a + perturbation)
    return a_new

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)  # uniform
    
    x = np.arange(n, dtype=np.float64)
    cx = 0.5 * (n - 1)
    
    # Hann window
    hann = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    hann = hann / hann.sum()
    inits.append(hann)
    
    # Raised cosine with different widths
    for sigma in [0.2, 0.35, 0.5]:
        gauss = np.exp(-0.5 * ((x - cx) / (sigma * n / 2))**2)
        inits.append(gauss / gauss.sum())
    
    # Two-lobe
    two_lobe = np.exp(-0.5 * ((x - 0.3*(n-1)) / (0.15*n))**2) + \
                np.exp(-0.5 * ((x - 0.7*(n-1)) / (0.15*n))**2)
    inits.append(two_lobe / two_lobe.sum())
    
    # Three-lobe
    three_lobe = np.exp(-0.5 * ((x - 0.2*(n-1)) / (0.1*n))**2) + \
                 np.exp(-0.5 * ((x - 0.5*(n-1)) / (0.1*n))**2) + \
                 np.exp(-0.5 * ((x - 0.8*(n-1)) / (0.1*n))**2)
    inits.append(three_lobe / three_lobe.sum())
    
    # U-shape (edge-heavy)
    u = 1.0 / (0.5 * np.abs(x - cx) / (n/2) + 0.1)
    inits.append(u / u.sum())
    
    # Cosine modulated
    cosm = (1 + 0.5 * np.cos(2 * np.pi * 2 * x / n)) / (1 + 0.5*np.cos(2*np.pi*2*x/n)).sum()
    inits.append(cosm)
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using continuation L-BFGS on softmax
parameterization with log-sum-exp smoothing, momentum-based smooth gradient
descent, LP-based active-set polish, and structured peak-targeting perturbations.
Key innovation: the log-sum-exp smoothing provides a differentiable proxy for
the max, enabling fast Newton-like steps via L-BFGS that the previous LP-based
Frank-Wolfe could not match.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_max_and_grad_a(a, tau):
    """
    Compute smooth approximation of max(conv(a,a)) and gradient w.r.t. a.
    obj = (1/tau)*log(sum_k exp(tau*S_k))
    grad_a[j] = 2 * sum_k p[k]*a[k-j]
    Returns (obj, grad_a, S, p).
    """
    n = len(a)
    S = fast_convolve(a)
    S_max = S.max()
    scaled = tau * (S - S_max)
    scaled = np.clip(scaled, -500, 500)
    exp_s = np.exp(scaled)
    p_sum = exp_s.sum()
    p = exp_s / p_sum
    obj = S_max + np.log(p_sum) / tau

    a_flip = a[::-1]
    h = np.convolve(p, a_flip)
    g = h[n - 1: 2 * n - 1]
    grad_a = 2.0 * g

    return obj, grad_a, S, p


def smooth_obj_grad_theta(theta, tau):
    """
    Smooth objective and gradient w.r.t. theta (softmax params).
    a = softmax(theta), minimize smooth_max(a, tau).
    dobj/dtheta_j = a_j * (grad_a_j - dot(grad_a, a))
    """
    theta_shift = theta - theta.max()
    exp_t = np.exp(theta_shift)
    a = exp_t / exp_t.sum()

    obj, grad_a, S, p = smooth_max_and_grad_a(a, tau)
    ga_dot_a = np.dot(grad_a, a)
    grad_theta = a * (grad_a - ga_dot_a)

    return obj, grad_theta


def lbfgs_continuation(a_init, tau_schedule, time_limit, verbose=False):
    """
    Continuation L-BFGS: start with small tau (smooth), increase for sharpness.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_true = fast_convolve(a).max()

    t0 = time.monotonic()
    theta = np.log(a + 1e-30)

    for tau in tau_schedule:
        if time.monotonic() - t0 > time_limit:
            break

        def fun(X):
            obj, grad = smooth_obj_grad_theta(X, tau)
            return obj, grad

        try:
            res = minimize(fun, theta, method='L-BFGS-B', jac=True,
                           options={'maxiter': 300, 'ftol': 1e-14,
                                    'gtol': 1e-10, 'maxls': 80})
        except Exception:
            continue

        theta_final = res.x
        theta_shift = theta_final - theta_final.max()
        exp_t = np.exp(theta_shift)
        a_new = exp_t / exp_t.sum()
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        true_obj = fast_convolve(a_new).max()
        if true_obj < best_true:
            best_true = true_obj
            best_a = a_new.copy()
            if verbose:
                print(f"      tau={tau:.0f}: true={true_obj:.10f} (iters={res.nit})")
        else:
            if verbose:
                print(f"      tau={tau:.0f}: no improvement ({true_obj:.10f})")

        theta = theta_final

    return best_a, best_true


def momentum_smooth_descent(a_init, time_limit, tau=50.0, lr=0.006,
                            momentum=0.90, verbose=False):
    """
    Momentum-based descent using smooth gradient in a-space with simplex projection.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-15)
    a = a / a.sum()

    best_a = a.copy()

        grad_proj = grad_a - np.mean(grad_a)

        v = momentum * v - lr * grad_proj
        a_new = project_simplex(a + v)

        obj = fast_convolve(a_new).max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:

def build_lp_constraints(a, S, peaks):
    """Build A_ub, b_ub for linearized peak-reduction LP at given peaks."""
    n = len(a)
    m = len(peaks)
    A = np.zeros((m, n))
    for idx, k in enumerate(peaks):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        for j in range(js, je + 1):
            A[idx, j] = a[k - j]
    return A


def solve_peak_lp(a, S, M_target, top_k=15):
    """
    Solve: find g s.t. linearized (a+g)*a <= M_target at top peaks.
    min 0 s.t. 2*(a*g)[k] <= M_target - S[k] for k in top_k, sum(g)=0, g >= -a.
    Returns g or None.
    """
    n = len(a)
    if top_k >= len(S):
        peaks = np.arange(len(S))
    else:
        peaks = np.argpartition(S, -top_k)[-top_k:]
    peaks = np.sort(peaks)

    A = build_lp_constraints(a, S, peaks)
    b_ub = (M_target - S[peaks]) / 2.0

    A_eq = np.ones((1, n))
    b_eq = np.zeros(1)
    bounds = [(-a[i], None) for i in range(n)]
    c = np.zeros(n)

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
    except Exception:
        return None

    if res.success and res.x is not None:
        return res.x
    return None


def lp_active_polish(a_init, time_limit, verbose=False):

    Iterative LP-based peak reduction with adaptive target and line search.
    """

        S = fast_convolve(a)
        M = S.max()

        progress = False
        for frac in [0.0005, 0.001, 0.003, 0.008, 0.02]:
            M_target = M * (1.0 - frac)
            if M_target <= 0:
                continue
            g = solve_peak_lp(a, S, M_target, top_k=12)
            if g is None:
                continue

            best_alpha = 0.0
            best_val = M
            for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                a_t = a + alpha * g
                a_t = np.maximum(a_t, 0.0)
                s = a_t.sum()

                if val < best_val:

                    best_alpha = alpha

        if verbose and it % 100 == 0:

def structured_perturb(a, S, rng, scale=0.008):
    """
    Perturb to reduce peak-contributing pairs: reduce mass at positions
    that participate in many high-convolution pairs.
    """
    n = len(a)
    S_max = S.max()
    active = np.where(S > 0.99 * S_max)[0]

    repulsion = np.zeros(n)
    for k in active:
        w = S[k] / S_max
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                repulsion[i] += w * a[j]
                repulsion[j] += w * a[i]

    pert = -repulsion * scale + rng.standard_normal(n) * scale * 0.2
    return project_simplex(a + pert)


def gen_inits(n, parent, rng):
    """Generate diverse initializations of length n."""
    inits = []
    inits.append(np.ones(n) / n)

    x = np.arange(n, dtype=np.float64)
    cx = 0.5 * (n - 1)

    # Hann
    h = 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1 + 1e-10)))
    inits.append(h / h.sum())

    # Gaussians
    for sig in [0.15, 0.25, 0.4]:
        g = np.exp(-0.5 * ((x - cx) / (sig * n)) ** 2)
        inits.append(g / g.sum())

    # Two-lobe
    t = np.exp(-0.5 * ((x - 0.3 * (n - 1)) / (0.12 * n)) ** 2) + \
        np.exp(-0.5 * ((x - 0.7 * (n - 1)) / (0.12 * n)) ** 2)
    inits.append(t / t.sum())

    # Three-lobe
    t3 = np.exp(-0.5 * ((x - 0.2 * (n - 1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.08 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.8 * (n - 1)) / (0.08 * n)) ** 2)
    inits.append(t3 / t3.sum())

    # U-shape
    u = 1.0 / (np.abs(x - cx) / (n / 2.0) + 0.08)
    inits.append(u / u.sum())

    # Cosine ripple
    cr = 1.0 + 0.4 * np.cos(4.0 * np.pi * x / n)
    cr = np.maximum(cr, 0.01)
    inits.append(cr / cr.sum())

    # Sine-modulated Gaussian
    smg = np.exp(-0.5 * ((x - cx) / (0.3 * n)) ** 2) * (1 + 0.3 * np.sin(6 * np.pi * x / n))
    smg = np.maximum(smg, 0.0)
    if smg.sum() > 1e-10:
        inits.append(smg / smg.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0.0)
        if px.sum() > 1e-10:
            inits.append(px / px.sum())

    return inits

    t_p3 = min(remaining * 0.25, 300)

    if t_p3 > 20:
        for tau_val, lr_val in [(30, 0.008), (100, 0.004), (300, 0.002)]:
            rem = deadline - time.monotonic() - 250
            if rem < 15:
                break
            t_each = min(rem * 0.35, 120)
            a_opt, _ = momentum_smooth_descent(best_seq, time_limit=t_each,
                                               tau=tau_val, lr=lr_val)
            s = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_seq = a_opt.copy()
                print(f"  MomSG tau={tau_val}: {s:.8f}")

    print(f"  Phase 3 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 4: LP Active Set Polish ===
    print("\n=== Phase 4: LP Active Set Polish ===")
    n_w = len(best_seq)
    remaining = deadline - time.monotonic() - 120
    t_p4 = min(remaining * 0.55, 350)

    if t_p4 > 20:
        a_opt, _ = lp_active_polish(best_seq, time_limit=t_p4 * 0.6, verbose=True)
        s = float(evaluate_sequence(a_opt.tolist()))
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_seq = a_opt.copy()
            print(f"  LP polish: {s:.8f}")

        # With perturbation
        rem = deadline - time.monotonic() - 60
        if rem > 20:
            pert = rng.standard_normal(n_w) * 0.004
            a_pert = project_simplex(best_seq + pert)
            a_opt, _ = lp_active_polish(a_pert, time_limit=min(rem * 0.6, 100))
            s = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_seq = a_opt.copy()
                print(f"  LP polish pert: {s:.8f}")

    print(f"  Phase 4 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 5: Structured Perturbation + Re-optimize ===
    print("\n=== Phase 5: Structured Perturbation ===")
    n_w = len(best_seq)
    remaining = deadline - time.monotonic() - 25

    if remaining > 20:
        S = fast_convolve(best_seq)
        for attempt in range(6):
            if remaining < 8:
                break
            S = fast_convolve(best_seq)
            a_pert = structured_perturb(best_seq, S, rng, scale=0.004 * (attempt + 1))

            # Quick L-BFGS
            t_each = min(remaining * 0.3, 35)
            if t_each > 5:
                a_opt, _ = lbfgs_continuation(a_pert, [80, 250], time_limit=t_each)
                s = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_seq = a_opt.copy()
                    print(f"  struct_{attempt} lbfgs: {s:.8f}")

            # Quick momentum
            rem2 = deadline - time.monotonic() - 15
            if rem2 > 8:
                a_opt, _ = momentum_smooth_descent(a_pert, time_limit=min(rem2 * 0.4, 25),
                                                   tau=100, lr=0.003)
                s = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_seq = a_opt.copy()
                    print(f"  struct_{attempt} mom: {s:.8f}")

            remaining = deadline - time.monotonic() - 20

    # === Final: try nearby lengths ===
    rem = deadline - time.monotonic() - 12
    if rem > 15 and best_seq is not None:
        n_best = len(best_seq)
        for n_try in [n_best - 16, n_best + 16]:
            if n_try < 32 or rem < 8:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0.0)
            px = px / px.sum()
            a_opt, _ = lbfgs_continuation(px, [50, 200], time_limit=min(rem * 0.5, 20))
            s = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_seq = a_opt.copy()
                print(f"  final n={n_try}: {s:.8f}")
            rem = deadline - time.monotonic() - 10

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK: {best_score:.10f}")

    return [float(x) for x in best_seq]
