import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_oracle(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return np.maximum(res.x, 0)
    return None


def compute_active_set(conv, M, tol=1e-8):
    """Find peaks within tol of the maximum."""
    threshold = M - tol * M
    return np.where(conv >= threshold)[0]


def afw_optimize(f0, time_limit, verbose=False):

    Accelerated Frank-Wolfe with adaptive M relaxation and active set tracking.
    Uses the AFW step size rule: t_k = 2/(k+2) for the main iterate,
    with backtracking for practical stability.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    y = f.copy()  # AFW auxiliary iterate

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    k = 0  # AFW counter

    while time.monotonic() - t0 < time_limit:
        it += 1
        k += 1

        # AFW step size: tau = 2/(k+2)
        tau_aw = 2.0 / (k + 2.0)

        # Solve LP oracle with current M
        g = solve_lp_oracle(y, M * m_exp)
        if g is None:
            m_exp *= 1.1
            if m_exp > 3.0:
                m_exp = 1.0
                # Fallback: use previous f
                g = f.copy()
            else:
                continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            g = f.copy()
        else:
            g = g / g_sum

        # Check if g ≈ y (convergence)
        if np.max(np.abs(g - y)) < 1e-10:
            no_improve += 1

                    break
                continue

        candidates = list(set(candidates))

            y_new = np.maximum(y_new, 0.0)
            sy = y_new.sum()
            if sy < 1e-15:

            val = fast_convolve(y_new).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Fine line search around best

            lo = max(0.0, best_alpha - 0.03)
            hi = min(1.0, best_alpha + 0.03)

                mid = (lo + hi) / 2.0
                y_new = (1.0 - mid) * y + mid * g
                y_new = np.maximum(y_new, 0.0)

                if sy < 1e-15:
                    break
                y_new = y_new / sy
                val = fast_convolve(y_new).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            y = (1.0 - best_alpha) * y + best_alpha * g
            y = np.maximum(y, 0.0)
            y = y / y.sum()
            conv = fast_convolve(y)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = y.copy()

def newton_polish(f0, time_limit, verbose=False):
    """
    Second-order polish using Newton's method on the active peak system.
    Identifies active peaks and solves the KKT-like system iteratively.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        # Identify active peaks (top few)
        top_k = max(3, min(15, n // 40))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        peaks = peaks[np.argsort(conv[peaks])[::-1]]  # Sort by value descending

        # Build the "active Hessian" approximation
        # For the sum of active peaks: sum_r w_r * (a*a)(k_r)
        # The Hessian is 2 * sum_r w_r * Q_{k_r}
        # We approximate by just using the gradient and a Levenberg-Marquardt style step

        # Weight peaks by their excess over the second-highest
        pv = conv[peaks]
        pv_max = pv[0]
        # Sharp weights: emphasize the top peak most
        weights = np.exp(-(pv_max - pv) / (0.01 * pv_max + 1e-10))
        weights /= weights.sum()

        # Compute gradient of weighted sum of peaks
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if weights[i] < 1e-12:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * weights[i] * f[k - idx]

        # Center gradient (for sum constraint)
        grad_centered = grad - grad.mean()

        # Levenberg-Marquardt step: f = project_simplex(f - lr * grad_centered / (1 + lr * ||grad||))
        grad_norm = np.linalg.norm(grad_centered)
        if grad_norm < 1e-15:
            break

        # Adaptive learning rate
        base_lr = 0.01
        lr = base_lr / (1.0 + 0.1 * grad_norm)

        # Try step
        for trial_lr in [lr, lr * 0.5, lr * 2.0, lr * 0.1]:
            f_new = f - trial_lr * grad_centered
            f_new = np.maximum(f_new, 0.0)
            sn = f_new.sum()
            if sn < 1e-15:
                continue
            f_new = f_new / sn
            val = fast_convolve(f_new).max()
            if val < M - 1e-15:
                f = f_new
                break

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 999983)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            no_improve = 0

    return best_f, best_obj


def mw_fine_polish(f0, time_limit, lr_init=0.003, decay_lr=0.9997, verbose=False):

    it = 0

        for rank, idx in enumerate(order):

        log_update = np.zeros(n)

            log_update[idx] -= cur_lr * weights[i] * f[k - idx]

        f = np.maximum(f, 1e-30)

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-16:

            cur_lr = max(lr_init * 0.5, 0.0005)
            no_improve = 0

def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))

    x = np.arange(n, dtype=np.float64)
    # Cauchy-like
    c1 = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(c1 / c1.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c2 = np.abs(np.cos(np.pi * x / n)) + 0.02
    inits.append(c2 / c2.sum())

    # Power law
    c3 = (x + 1) ** (-0.5)
    inits.append(c3 / c3.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

            print(f"  AFW on parent (n={n_use}), t={t_alloc:.0f}s")
            a_opt, _ = afw_optimize(parent, time_limit=t_alloc)
            try_seq(a_opt, "afw_parent")

    # Quick AFW on alternative inits
    rem = deadline - time.monotonic() - 1000
    if rem > 30:
        inits = generate_inits(n_use, parent, rng)
        for fi in range(1, min(4, len(inits))):
            rem = deadline - time.monotonic() - 900
            if rem < 20:
                break
            t_alloc = min(rem, 60)
            a_opt, _ = afw_optimize(inits[fi], time_limit=t_alloc)
            try_seq(a_opt, f"afw_init{fi}")

    print(f"  Phase 1 best: {best_score:.8f}")

    # === Phase 2: Deeper AFW + perturbations (~400s) ===
    print("\n=== Phase 2: Deep AFW ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 600
        if rem > 30:
            t_alloc = min(rem, 250)
            print(f"  Deep AFW, t={t_alloc:.0f}s")
            a_opt, _ = afw_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "deep_afw")

        # Perturbed starts
        rem = deadline - time.monotonic() - 400
        if rem > 30:
            for pi in range(3):
                rem = deadline - time.monotonic() - 350
                if rem < 20:
                    break
                noise = rng.standard_normal(len(best_seq)) * (0.003 + 0.002 * pi)
                f_pert = project_simplex(best_seq + noise)
                t_alloc = min(rem, 80)
                a_opt, _ = afw_optimize(f_pert, time_limit=t_alloc)
                try_seq(a_opt, f"afw_pert{pi}")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Newton Polish (~200s) ===
    print("\n=== Phase 3: Newton Polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 300
        if rem > 20:
            t_alloc = min(rem, 200)
            print(f"  Newton polish, t={t_alloc:.0f}s")
            a_opt, _ = newton_polish(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "newton")

        # Active set refinement
        rem = deadline - time.monotonic() - 200
        if rem > 20:
            t_alloc = min(rem, 150)
            print(f"  Active-set refine, t={t_alloc:.0f}s")
            a_opt, _ = active_set_refine(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "active_set")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Fine MW Polish (~200s) ===
    print("\n=== Phase 4: Fine MW Polish ===")
    if best_seq is not None:
        mw_configs = [
            {'lr_init': 0.003, 'decay_lr': 0.9997},
            {'lr_init': 0.002, 'decay_lr': 0.9998},
            {'lr_init': 0.005, 'decay_lr': 0.9995},
        ]
        rem = deadline - time.monotonic() - 150
        t_per = max(20, rem / len(mw_configs))
        for ci, cfg in enumerate(mw_configs):
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            t_alloc = min(t_per, rem)
            print(f"  MW polish config {ci}, t={t_alloc:.0f}s")
            a_opt, _ = mw_fine_polish(
                best_seq, time_limit=t_alloc,
                lr_init=cfg['lr_init'], decay_lr=cfg['decay_lr']
            )
            try_seq(a_opt, f"mw_{ci}")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Multi-length exploration (~150s) ===
    print("\n=== Phase 5: Multi-length Exploration ===")
    alt_lengths = [256, 512, 768]
    if best_seq is not None:
        for n_alt in alt_lengths:
            rem = deadline - time.monotonic() - 60
            if rem < 30:
                break
            # Interpolate parent to new length
            px = np.interp(np.linspace(0, 1, n_alt),
                           np.linspace(0, 1, len(best_seq)), best_seq)
            px = np.maximum(px, 0)
            px = px / px.sum()
            t_alloc = min(rem, 60)
            a_opt, _ = afw_optimize(px, time_limit=t_alloc)
            try_seq(a_opt, f"len_{n_alt}")

    # === Phase 6: Final ultra-fine polish (~100s) ===
    print("\n=== Phase 6: Final Polish ===")
    if best_seq is not None:
        # Very fine MW
        rem = deadline - time.monotonic() - 30
        if rem > 15:
            t_alloc = min(rem, 50)
            print(f"  Ultra-fine MW, t={t_alloc:.0f}s")
            a_opt, _ = mw_fine_polish(
                best_seq, time_limit=t_alloc,
                lr_init=0.001, decay_lr=0.9999
            )
            try_seq(a_opt, "ultra_mw")

        # Tiny active-set steps
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            a_opt, _ = active_set_refine(best_seq, time_limit=min(rem, 30))
            try_seq(a_opt, "final_active")

        # Micro perturbations
        rem = deadline - time.monotonic() - 5
        if rem > 5:
            for _ in range(3):
                rem = deadline - time.monotonic() - 3
                if rem < 3:
                    break
                noise = rng.standard_normal(len(best_seq)) * 0.0005
                f_pert = project_simplex(best_seq + noise)
                a_opt, _ = mw_fine_polish(
                    f_pert, time_limit=min(rem, 10),
                    lr_init=0.001, decay_lr=0.9999
                )
                try_seq(a_opt, "micro")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
