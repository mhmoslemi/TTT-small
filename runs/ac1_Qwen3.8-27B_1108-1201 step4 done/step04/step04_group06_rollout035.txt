import numpy as np
import time
from scipy.optimize import linprog

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

def compute_objective(a):
    """Compute the objective value."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)

def lp_direction(a, M, C=None):
    """Solve LP to find descent direction."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs')
    if result.success:
        return result.x, result.x.sum()
    return None, None

def lp_optimize(a0, time_limit, verbose=True):
    """LP-based optimization with adaptive step sizing."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = 2.0 * n * M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    C = build_conv_matrix(a)
    
    while time.monotonic() - t0 < time_limit:
        # Recompute C if needed (it changes with a)
        C = build_conv_matrix(a)
        
        # Try slightly tighter bound
        for factor in [1.0, 0.999, 0.998, 0.997, 0.995, 0.99]:
            g, g_sum = lp_direction(a, M * factor, C)
            if g is not None and g_sum > 1.0 + 1e-6:
                break
        
        if g is None:
            no_improve += 1
            if no_improve > 50:
                # Perturb
                rng = np.random.default_rng(int(time.monotonic() * 1000))
                a = best_a + rng.standard_normal(n) * 0.005
                a = np.maximum(a, 0)
                a = a / a.sum()
                conv = fast_convolve(a)
                M = conv.max()
                no_improve = 0
            continue
        
        d = g - a
        norm_d = np.linalg.norm(d)
        if norm_d < 1e-10:
            no_improve += 1
            continue
        
        # Line search
        best_t = 0
        for t in [1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.01]:
            a_try = a + t * d
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-15:
                continue
            a_try = a_try / s_try
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < best_obj - 1e-10:
                best_obj = obj_try
                best_a = a_try.copy()
                best_t = t
                break
        
        if best_t > 0:
            a = best_a.copy()
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and no_improve % 20 == 0:
            print(f"  obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s, no_improve={no_improve}")
    
    return best_a, best_obj

3. Project onto simplex: `a_proj = proj_simplex(a_new)`

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build C (2n-1 x n) such that C@g = convolve(a, g)."""
    n = len(a)
    m = 2 * n - 1
    C = np.zeros((m, n))
    for k in range(m):
        l_start = max(0, k - n + 1)
        l_end = min(n - 1, k)
        seg_start = k - l_end
        seg_end = k - l_start
        C[k, l_start:l_end + 1] = a[seg_start:seg_end + 1][::-1]
    return C


def simplex_project(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(0.0, v[0])])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def compute_subgradient(a, conv, top_k=40, temp=0.005):
    """Compute softmax-weighted subgradient of max conv(a,a)_k."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp - peak_vals.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        indices = np.arange(i_start, i_end + 1)
        grad[indices] += 2.0 * w[idx] * a[k - indices]
    return grad


def lp_solve_direction(a, M, C):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0. Returns g or None."""
    n = len(a)
    try:
        result = linprog(-np.ones(n), A_ub=C, b_ub=np.full(2 * n - 1, M),
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = result.x
            if g.sum() > 1.0 + 1e-6:
                return g
    except Exception:
        pass
    return None

    """
    LP-based iterative optimization with adaptive tightness.
    The LP finds a direction g where cross-conv conv(a,g) <= M at all lags.

def nesterov_subgrad(a0, time_limit, lr_init=0.008, momentum=0.91, top_k=50, temp_init=0.003):
    """Nesterov-accelerated subgradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a))
    a = a / s
    n = len(a)

    v = np.zeros(n)
    lr = lr_init
    temp = temp_init
    best_obj = 2.0 * n * fast_convolve(a).max()
    best_a = a.copy()
    t0 = time.monotonic()
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + momentum * v
        y = simplex_project(y)
        conv_y = fast_convolve(y)
        grad = compute_subgradient(y, conv_y, top_k=top_k, temp=temp)

        v = momentum * v - lr * grad
        a_new = simplex_project(a + v)
        a = a_new

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()
        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp *= 1.004
        if no_improve > 300:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations for optimization."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sine-based
    s1 = np.sin(np.pi * x / n) ** 2 + 0.01
    inits['sine1'] = s1 / s1.sum()

    s2 = np.sin(2 * np.pi * x / n) ** 2 + 0.01
    inits['sine2'] = s2 / s2.sum()

    # Bimodal (mass at edges)
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.25 * n)) ** 2) + 0.001
    inits['gaussian'] = g / g.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Triangular
    tri = 1.0 - np.abs(x - 0.5 * (n - 1)) / (0.5 * (n - 1) + 1)
    tri = np.maximum(tri, 0.01)
    inits['triangular'] = tri / tri.sum()

    # From parent (interpolated)
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits

                t_sg = min(remaining / (3 - gi) * 0.7, 40.0)
                if t_sg < 5:
                    break
                a_opt, obj = nesterov_subgrad(best_sequence, time_limit=t_sg, **cfg)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Subgrad {gi}: {score:.6f} *** NEW BEST ***")

    print(f"Phase 4 done ({time.monotonic()-t4:.0f}s). Best: {best_score:.6f}")

    # =========================================================
    # Phase 5: Final LP micro-polish
    # =========================================================
    print("=== Phase 5: Final Polish ===")
    if best_sequence is not None and time.monotonic() < deadline - 10:
        t_final = min(30.0, deadline - time.monotonic() - 5)
        if t_final > 5:
            a_opt, obj = lp_optimize(best_sequence, time_limit=t_final)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Final LP: {score:.6f} *** NEW BEST ***")

    # Fallback
    if best_sequence is None:
        n_fb = 768
        if parent is not None:
            best_sequence = parent.copy()
        else:
            best_sequence = np.ones(n_fb) / n_fb
        best_score = float(evaluate_sequence(best_sequence.tolist()))

    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score

    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
