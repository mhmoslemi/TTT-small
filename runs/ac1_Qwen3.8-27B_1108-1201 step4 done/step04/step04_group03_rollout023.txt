import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def soft_max(conv, temperature):
    """Soft approximation of max."""
    shifted = conv / temperature
    shifted -= shifted.max()
    exp_val = np.exp(shifted)
    return temperature * (np.log(exp_val.sum()) + shifted.max())

def top_k_sum(conv, k):
    """Sum of top k values."""
    if k >= len(conv):
        return conv.sum()
    return np.sum(np.partition(conv, -k)[-k:])

def smoothed_objective(conv, temperature, top_k=50):
    """A smooth surrogate for max(conv)."""
    # Combine soft-max with top-k sum
    sm = soft_max(conv, temperature)
    tk = top_k_sum(conv, top_k) / top_k
    return 0.5 * sm + 0.5 * tk
