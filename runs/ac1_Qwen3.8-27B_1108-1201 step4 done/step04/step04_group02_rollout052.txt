def project_simplex(v):

    if n == 1:

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with exact
1D per-coordinate optimization, combined with subgradient pre-optimization
and multi-length exploration.

import numpy as np
from scipy.optimize import minimize_scalar, linprog
import time


def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def sg_descent(f0, time_limit, initial_lr=0.01, mom=0.9, top_k=64, temp=0.001):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        sg = compute_subgradient(f, conv, top_k=top_k, temp=temp)
        v = mom * v - initial_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = f_new  # still move, for exploration
            no_improve += 1
        if no_improve > 200:
            f = project_simplex(best_f + np.random.default_rng(it).standard_normal(n) * 0.01)
            conv = fast_convolve(f)
            v = np.zeros(n)
            no_improve = 0

    return best_f, best_obj


def coord_descent(f0, time_limit=60.0, max_sweeps=5000, tol=1e-14):
    """Coordinate descent with exact 1D optimization per coordinate."""
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    S = f.sum()

        S = 1.0
    else:
        f = f / S
        S = 1.0
    
    t0 = time.monotonic()
    prev_obj = 2 * n * fast_convolve(f).max() / (S * S)
    
    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break
        
        for i in range(n):
            old = f[i]
            S_base = S - old
            
            if S_base < 1e-15:
                # Only one non-zero element; can't improve much
                continue
            
            # Compute conv with f[i] = 0
            f[i] = 0.0
            conv_base = fast_convolve(f)
            f[i] = old

            # k in [i, i+n-1] intersect [0, 2n-2]
            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            
            # Base values at relevant indices (with f[i]=0 already subtracted)
            # conv_base already has f[i]=0, so conv_base[k] is the value without f[i]'s contribution

            # h_k(x) = conv_base[k] + 2*x*f[k-i] for k in [k_min, k_max], k != 2i
            # h_other = max of conv_base[k] for k not in [k_min, k_max]

                if k_min == 0 and k_max == 2 * n - 2:
                    h_other = 0.0
                else:

                    elif k_max == 2 * n - 2:

                    else:
                        h_other = max(conv_base[:k_min].max(), conv_base[k_max+1:].max())
            else:
                h_other = 0.0

    S = f.sum()
    if S > 1e-15:

    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(f, conv, top_k=64, temp=0.001):
    n = len(f)
    if top_k >= len(conv):
        peaks = np.arange(len(conv))
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    wsum = w.sum()
    if wsum < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= wsum
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        idx = np.arange(js, je + 1)
        sg[idx] += 2.0 * w[i] * f[k - idx]
    return sg


def sg_descent(f0, time_limit, initial_lr=0.01, mom=0.9, top_k=64, temp=0.001, seed=0):
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        sg = compute_subgradient(f, conv, top_k=top_k, temp=temp)
        v = mom * v - initial_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        obj = conv.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            f = f_new
            no_improve += 1
        if no_improve > 300:
            rng = np.random.default_rng(seed + it)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.01)
            conv = fast_convolve(f)
            v = np.zeros(n)
            no_improve = 0

    return best_f, best_obj


def coord_descent(f0, time_limit=30.0, max_sweeps=10000, tol=1e-14, shuffle=False, seed=0):
    n = len(f0)
    f = np.maximum(f0.astype(np.float64), 0.0)
    S = f.sum()
    if S < 1e-15:
        f = np.ones(n) / n
        S = 1.0
    else:
        f = f / S
        S = 1.0

    t0 = time.monotonic()
    prev_obj = 2.0 * n * fast_convolve(f).max() / (S * S)
    rng = np.random.default_rng(seed)

    for sweep in range(max_sweeps):
        if time.monotonic() - t0 > time_limit:
            break

        if shuffle:
            order = rng.permutation(n)
        else:
            order = range(n)

        for i in order:
            old = f[i]
            S_base = S - old

            if S_base < 1e-15:
                continue

            # Compute conv with f[i] = 0
            f[i] = 0.0
            conv_base = fast_convolve(f)
            f[i] = old

            k_min = i
            k_max = min(2 * n - 2, i + n - 1)
            idx_2i = 2 * i - k_min

            # h_other: max of conv_base outside [k_min, k_max]
            if k_min == 0 and k_max == 2 * n - 2:
                h_other = 0.0
            elif k_min == 0:
                h_other = conv_base[k_max + 1:].max()
            elif k_max == 2 * n - 2:
                h_other = conv_base[:k_min].max()
            else:
                h_other = max(conv_base[:k_min].max(), conv_base[k_max + 1:].max())

            offsets = np.arange(k_min, k_max + 1)
            coeff = 2.0 * f[offsets - i]
            base_rel = conv_base[k_min:k_max + 1]
            len_rel = k_max - k_min + 1

            if len_rel > 1:
                mask_2i = np.ones(len_rel, dtype=bool)
                mask_2i[idx_2i] = False
                coeff_other = coeff[mask_2i]
                base_other = base_rel[mask_2i]
            else:
                coeff_other = None
                base_other = None

            base_2i = base_rel[idx_2i]
            h_other_val = h_other

            # 1D objective
            def obj_fn(x):
                S_new = S_base + x
                if S_new < 1e-15:
                    return 1e18
                h0 = base_2i + x * x
                if coeff_other is not None:
                    hks = (base_other + x * coeff_other).max()
                    max_rel = h0 if h0 > hks else hks
                else:
                    max_rel = h0
                mv = max_rel if max_rel > h_other_val else h_other_val
                if mv < 0.0:
                    mv = 0.0
                return 2.0 * n * mv / (S_new * S_new)

            ub = max(old * 3.0 + 0.05, 0.1)
            res = minimize_scalar(obj_fn, bounds=(0.0, ub), method='bounded',
                                  options={'xatol': 1e-13})
            new_val = res.x if res.x > 0.0 else 0.0

            S += (new_val - old)
            f[i] = new_val

        conv = fast_convolve(f)
        cur_obj = 2.0 * n * conv.max() / (S * S)
        if abs(prev_obj - cur_obj) < tol:
            break
        prev_obj = cur_obj

    S = f.sum()
    if S > 1e-15:
        f = f / S
    return f


def solve_lp(f, M, time_limit=10):
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_step(f, M, time_limit=10):
    g = solve_lp(f, M, time_limit)
    if g is None:
        return None
    gs = g.sum()
    if gs < 1e-15:
        return None
    g = g / gs
    return g


def generate_inits(n, parent=None, rng=None):
    if rng is None:
        rng = np.random.default_rng(42)
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)

    ramp_d = np.arange(n, 0, -1, dtype=np.float64)
    inits.append(ramp_d / ramp_d.sum())

    ramp_i = np.arange(1, n + 1, dtype=np.float64)
    inits.append(ramp_i / ramp_i.sum())

    q = (x - n / 2.0) ** 2 + 1.0
    inits.append(q / q.sum())

    e1 = np.exp(-x / (n / 3.0))
    inits.append(e1 / e1.sum())

    c = np.cos(np.pi * (x / max(n - 1, 1) - 0.5)) ** 4 + 0.01
    inits.append(c / c.sum())

    sn = np.sin(np.pi * x / n) + 0.05
    inits.append(sn / sn.sum())

    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)

    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
            blended = 0.5 * (px / ps) + 0.5 * (np.ones(n) / n)
            inits.append(blended / blended.sum())

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 40.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        if a.sum() < 0.01:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            old = best_score
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.10f} (n={len(a)}) (was {old:.10f})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, obj={2.0*len(parent)*fast_convolve(parent).max():.8f}")
    except Exception:
        pass

    # === Phase 1: Optimize parent ===
    print("=== Phase 1: Parent Optimization ===")
    if parent is not None:
        n_p = len(parent)
        t_alloc = min(60, deadline - time.monotonic() - 600)
        if t_alloc > 10:
            print(f"  SG descent on parent (n={n_p}), t={t_alloc:.0f}s")
            a_opt, _ = sg_descent(parent, time_limit=t_alloc, initial_lr=0.008,
                                  top_k=max(32, n_p // 8), temp=0.0005, seed=seed)
            try_seq(a_opt, "parent_sg")
            if best_seq is not None:
                parent = best_seq / best_seq.sum()

        # Coordinate descent on parent
        t_alloc = min(180, deadline - time.monotonic() - 500)
        if t_alloc > 10 and parent is not None:
            print(f"  Coord descent on parent (n={n_p}), t={t_alloc:.0f}s")
            a_opt = coord_descent(parent, time_limit=t_alloc, shuffle=False, seed=seed)
            try_seq(a_opt, "parent_cd")
            if best_seq is not None:
                parent = best_seq / best_seq.sum()

    # === Phase 2: Multi-length exploration ===
    print("\n=== Phase 2: Multi-length Exploration ===")

    lengths = [256, 320, 384, 448, 512]
    if parent is not None:
        n_p = len(parent)
        if n_p in lengths:
            lengths.remove(n_p)
        lengths.insert(0, n_p)

    for n in lengths:
        if time.monotonic() > deadline - 200:
            break

        inits = generate_inits(n, parent, rng)
        n_inits = len(inits)

        # Distribute time among inits for this length
        time_for_length = min(250, deadline - time.monotonic() - 150)
        if time_for_length < 30:
            continue
        t_per_sg = max(5, time_for_length / (n_inits * 3.5))
        t_per_cd = max(15, time_for_length / (n_inits * 1.5))

        for i, f_init in enumerate(inits):
            if time.monotonic() > deadline - 100:
                break

            # Quick SG
            rem = deadline - time.monotonic() - 100
            t_sg = min(t_per_sg, rem * 0.2)
            if t_sg > 3:
                a_sg, _ = sg_descent(f_init, time_limit=t_sg, initial_lr=0.01,
                                     top_k=max(32, n // 10), temp=0.001, seed=seed + i)
                try_seq(a_sg, f"n{n}_i{i}_sg")
                f_cur = a_sg if (best_seq is not None and len(a_sg) == n) else f_init
            else:
                f_cur = f_init

            # Coordinate descent
            rem = deadline - time.monotonic() - 80
            t_cd = min(t_per_cd, rem * 0.7)
            if t_cd > 5:
                a_cd = coord_descent(f_cur, time_limit=t_cd, shuffle=True, seed=seed + n + i)
                try_seq(a_cd, f"n{n}_i{i}_cd")

    print(f"  Phase 2 best: {best_score:.10f}")

    # === Phase 3: Deep optimization of best ===
    print(f"\n=== Phase 3: Deep Optimization ===")

    if best_seq is not None:
        n_best = len(best_seq)
        best_norm = best_seq / best_seq.sum()

        # Multiple CD passes with different shuffles
        for p in range(5):
            rem = deadline - time.monotonic() - 60
            if rem < 10:
                break
            t_alloc = min(rem // (6 - p), 120)
            if p < 2:
                f_init = best_norm
            elif p == 2:
                f_init = project_simplex(best_norm + rng.standard_normal(n_best) * 0.003)
            elif p == 3:
                f_init = project_simplex(0.9 * best_norm + 0.1 * (np.ones(n_best) / n_best))
            else:
                f_init = project_simplex(best_norm + rng.standard_normal(n_best) * 0.01)

            a_opt = coord_descent(f_init, time_limit=t_alloc, shuffle=(p >= 2), seed=seed + 100 + p)
            try_seq(a_opt, f"deep_cd_{p}")
            if best_seq is not None:
                best_norm = best_seq / best_seq.sum()

        # LP-based Frank-Wolfe escape
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            conv = fast_convolve(best_norm)
            M = conv.max()
            g = solve_lp(best_norm, M * 1.001, time_limit=min(rem // 3, 30))
            if g is not None:
                gs = g.sum()
                if gs > 1e-10:
                    g = g / gs
                    for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                        ft = (1 - alpha) * best_norm + alpha * g
                        ft = np.maximum(ft, 0)
                        sts = ft.sum()
                        if sts > 1e-10:
                            ft = ft / sts
                            a_try = ft.copy()
                            s_try = try_seq(a_try, f"fw_escape_{alpha}")
                            if np.isfinite(s_try) and s_try < best_score:
                                best_norm = a_try / a_try.sum()
                                break

    # === Phase 4: Final polish ===
    print(f"\n=== Phase 4: Final Polish ===")
    print(f"  Current best: {best_score:.10f}")

    if best_seq is not None:
        n_best = len(best_seq)
        best_norm = best_seq / best_seq.sum()

        # Very tight CD
        rem = deadline - time.monotonic() - 20
        if rem > 5:
            a_opt = coord_descent(best_norm, time_limit=min(rem, 60),
                                  tol=1e-15, shuffle=False, seed=seed + 200)
            try_seq(a_opt, "polish_cd")

        # Micro perturbations + CD
        if best_seq is not None:
            best_norm = best_seq / best_seq.sum()
            for p in range(3):
                rem = deadline - time.monotonic() - 10
                if rem < 4:
                    break
                f_pert = project_simplex(best_norm + rng.standard_normal(n_best) * 0.001)
                a_opt = coord_descent(f_pert, time_limit=min(rem, 20),
                                      tol=1e-15, shuffle=True, seed=seed + 300 + p)
                try_seq(a_opt, f"micro_{p}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
