from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def fast_convolve(a):

    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_lse_gradient(weights, f):
    """Gradient of log-sum-exp(beta * conv(f,f)) w.r.t. f.
    grad[i] = 2 * sum_{m=0}^{n-1} weights[i+m] * f[m] ... 
    Actually, let me use the formula from the previous code.
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]

def build_lp_matrix(f):
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp_direction(f, M):
    n = len(f)
    A = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        f = f / s
    n = len(f)

    momentum = momentum_init
    y = f.copy()  # Nesterov lookahead
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        y = f + momentum * v

        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 100
            break
        grad = grad / grad_norm
        
        # Update
        v = momentum * v - lr * grad
        f_new = project_simplex(y + v)  # project from lookahead

def fw_lp_optimize(f0, time_limit, max_iters=5000):
    """Frank-Wolfe with LP oracle."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit and it < max_iters:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        g = solve_lp_direction(f, M)
        if g is None:
            if no_improve > 30:
                break
            no_improve += 1
            continue
        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = np.maximum(g, 0.0) / g_sum
        
        if np.max(np.abs(g - f)) < 1e-12:
            no_improve += 1
            if no_improve > 50:
                break
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        # Refine with bisection
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid
        
        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.001)
            conv = fast_convolve(f)
            M = conv.max()
            no_improve = 0
    
    return best_f, best_obj

def coordinate_descent(f0, time_limit, max_passes=200, search_range=0.02):
    """Coordinate descent with golden section line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    inv_phi = 1.0 / phi
    inv_phi2 = inv_phi * inv_phi
    
    t0 = time.monotonic()
    pass_count = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit and pass_count < max_passes:
        pass_count += 1
        if time.monotonic() - t0 > time_limit:
            break
        
        rng = np.random.default_rng(pass_count * 997 + 42)
        order = rng.permutation(n)
        improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[i]
            # Search range for f[i]
            lo = max(0.0, old_val - search_range)
            hi = old_val + search_range
            
            # Golden section search
            x1 = hi - inv_phi * (hi - lo)
            x2 = lo + inv_phi * (hi - lo)
            
            f_test = f.copy()
            f_test[i] = x1
            f_test = f_test / f_test.sum()
            obj1 = fast_convolve(f_test).max()
            
            f_test[i] = x2
            f_test = f_test / f_test.sum()
            obj2 = fast_convolve(f_test).max()
            
            for _ in range(20):
                if obj1 < obj2:
                    hi = x2
                    x2 = x1
                    obj2 = obj1
                    x1 = hi - inv_phi * (hi - lo)
                    f_test[i] = x1
                    f_test = f_test / f_test.sum()
                    obj1 = fast_convolve(f_test).max()
                else:
                    lo = x1
                    x1 = x2
                    obj1 = obj2
                    x2 = lo + inv_phi * (hi - lo)
                    f_test[i] = x2
                    f_test = f_test / f_test.sum()
                    obj2 = fast_convolve(f_test).max()
            
            best_x = (lo + hi) / 2.0
            f[i] = best_x
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            
            if abs(f[i] - old_val) > 1e-8:
                improved = True
        
        # Check improvement at end of pass
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if not improved:
            if no_improve > 3:
                break
        if no_improve > 10:
            break
        
        if pass_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD pass={pass_count}, obj={best_obj:.12f}, elapsed={elapsed:.0f}s")
    
    return best_f, best_obj

def simulated_annealing(f0, time_limit, t_start=0.01, t_end=0.0001):
    """Simulated annealing for the normalized sequence."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    conv = fast_convolve(f)
    current_obj = conv.max()
    best_obj = current_obj
    best_f = f.copy()
    
    t0 = time.monotonic()
    total_time = time_limit
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = elapsed / total_time
        temperature = t_start * (t_end / t_start) ** progress
        
        rng = np.random.default_rng(it)
        # Perturb 5-20 random coordinates
        n_perturb = rng.integers(3, min(20, n // 4) + 1)
        indices = rng.choice(n, size=n_perturb, replace=False)
        
        f_new = f.copy()
        for idx in indices:
            f_new[idx] = max(0.0, f_new[idx] + rng.gaussian(0, temperature))
        
        s_new = f_new.sum()
        if s_new < 1e-10:
            continue
        f_new = f_new / s_new
        
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < current_obj or rng.random() < np.exp(-(obj_new - current_obj) / temperature):
            f = f_new
            current_obj = obj_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()
        
        if it % 500 == 0:
            print(f"      SA it={it}, best={best_obj:.12f}, temp={temperature:.6f}, "
                  f"elapsed={elapsed:.0f}s")
    
    return best_f, best_obj

def try_sequence(a, best_tracker):
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.8f} (n={len(a)})")
    return score

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Multi-start initialization with quick LSE screening
2. Nesterov-accelerated LSE gradient descent with adaptive beta (up to 300k)
3. FW-LP refinement with small steps
4. Coordinate descent with golden section search (adaptive range)
5. Simulated annealing for global exploration
6. Length exploration around parent length

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(weights, f):
    """Compute gradient of LSE objective via FFT cross-correlation.
    grad[i] = 2 * sum_{m=0}^{n-1} weights[i+m] * f[m]
    """
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP oracle.
    A[k, j] = f[k-j] if 0 <= k-j < n, else 0.
    """
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(f[k - js].tolist())
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_direction(f, M, A_cache=None):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    if A_cache is None:
        A_cache = build_lp_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def lse_nesterov(f0, time_limit, beta_start=30.0, beta_end=50000.0,
                 lr_init=None, momentum_init=0.85):
    """Nesterov-accelerated LSE gradient descent with adaptive parameters."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    if lr_init is None:
        lr_init = 0.8 / np.sqrt(n)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = lr_init
    momentum = momentum_init
    y = f.copy()
    v = np.zeros(n)
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    eval_count = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        # Nesterov lookahead
        y = f + momentum * v

        # Ensure y is valid
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy < 1e-15:
            y = f.copy()
        else:
            y = y / sy

        # Compute gradient at y
        conv_y = fast_convolve(y)
        bc = beta * conv_y
        bc_max = bc.max()
        exp_bc = np.exp(bc - bc_max)
        w = exp_bc / exp_bc.sum()

        grad = compute_lse_gradient(w, y)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-18:
            no_improve += 100
            break
        grad = grad / grad_norm

        # Update
        v = momentum * v - lr * grad
        f_new = project_simplex(y + v)

        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        eval_count += 1

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slightly increase lr when making progress
            lr = min(lr * 1.001, 2.0 / np.sqrt(n))
        else:
            f = f_new
            no_improve += 1
            if no_improve >= 30:
                lr *= 0.95
                momentum = max(0.75, momentum - 0.003)
            if no_improve > 150:
                f = best_f.copy()
                v = np.zeros(n)
                y = f.copy()
                lr = max(0.3 / np.sqrt(n), lr * 0.5)
                momentum = 0.82
                beta = min(beta * 1.3, beta_end)
                no_improve = 0

        # Ramp up beta
        if beta < beta_end:
            beta *= 1.004
            if beta > beta_end:
                beta = beta_end

        if it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"      LSE-N it={it}, obj={best_obj:.12f}, beta={beta:.0f}, "
                  f"lr={lr:.6f}, mom={momentum:.3f}, no_imp={no_improve}, "
                  f"elapsed={elapsed:.0f}s")

    return best_f, best_obj


def fw_lp_optimize(f0, time_limit, min_alpha=0.01):
    """Frank-Wolfe with LP oracle and small step sizes."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    A_cache = build_lp_matrix(f)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit:
            break

        g = solve_lp_direction(f, M, A_cache)
        if g is None:
            no_improve += 1
            if no_improve > 20:
                # Perturb and rebuild
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                conv = fast_convolve(f)
                M = conv.max()
                A_cache = build_lp_matrix(f)
                no_improve = 0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        diff = np.max(np.abs(g - f))
        if diff < 1e-12:
            no_improve += 1
            if no_improve > 40:
                break
            continue

        # Line search with small step emphasis
        best_alpha = 0.0
        best_val = M
        alphas = [min_alpha, 0.03, 0.05, 0.08, 0.12, 0.2, 0.35, 0.5, 0.8, 1.0]
        for alpha in alphas:
            ft = (1.0 - alpha) * f + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        # Bisection refinement
        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.04)
            hi = min(1.0, best_alpha + 0.04)
            for _ in range(6):
                mid = (lo + hi) / 2.0
                ft = (1.0 - mid) * f + mid * g
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = (1.0 - best_alpha) * f + best_alpha * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            A_cache = build_lp_matrix(f)
            conv = fast_convolve(f)
            M = conv.max()

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.001 + 0.002 * min(1.0, (no_improve - 60) / 40.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            A_cache = build_lp_matrix(f)
            no_improve = 0

        if it % 200 == 0:
            elapsed = time.monotonic() - t0
            print(f"      FW-LP it={it}, obj={best_obj:.12f}, M={M:.12f}, "
                  f"elapsed={elapsed:.0f}s")

    return best_f, best_obj


def coordinate_descent(f0, time_limit, max_passes=1000, initial_range=0.015):
    """Coordinate descent with golden section search and adaptive range."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    inv_phi = 2.0 / (1.0 + np.sqrt(5.0))

    t0 = time.monotonic()
    pass_count = 0
    no_improve = 0
    search_range = initial_range
    total_evals = 0

    while time.monotonic() - t0 < time_limit and pass_count < max_passes:
        pass_count += 1
        if time.monotonic() - t0 > time_limit:
            break

        rng = np.random.default_rng(pass_count * 997 + 42)
        order = rng.permutation(n)
        pass_improved = False

        for i in order:
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[i]
            # Adaptive search range: larger for larger values
            span = max(search_range, 0.1 * old_val)
            lo = max(0.0, old_val - span)
            hi = old_val + span

            # Golden section search
            a = lo
            b = hi
            c = b - inv_phi * (b - a)
            d = a + inv_phi * (b - a)

            f_work = f.copy()
            f_work[i] = c
            f_work = np.maximum(f_work, 0.0)
            f_work = f_work / f_work.sum()
            fc = fast_convolve(f_work).max()
            total_evals += 1

            f_work[i] = d
            f_work = np.maximum(f_work, 0.0)
            f_work = f_work / f_work.sum()
            fd = fast_convolve(f_work).max()
            total_evals += 1

            for _ in range(18):
                if fc < fd:
                    b = d
                    d = c
                    fd = fc
                    c = b - inv_phi * (b - a)
                    f_work[i] = c
                    f_work = np.maximum(f_work, 0.0)
                    f_work = f_work / f_work.sum()
                    fc = fast_convolve(f_work).max()
                    total_evals += 1
                else:
                    a = c
                    c = d
                    fc = fd
                    d = a + inv_phi * (b - a)
                    f_work[i] = d
                    f_work = np.maximum(f_work, 0.0)
                    f_work = f_work / f_work.sum()
                    fd = fast_convolve(f_work).max()
                    total_evals += 1

            best_x = (a + b) / 2.0
            f[i] = best_x
            f = np.maximum(f, 0.0)
            f = f / f.sum()

            if abs(f[i] - old_val) > 1e-8:
                pass_improved = True

        # End of pass: check improvement
        conv = fast_convolve(f)
        obj = conv.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 0 and no_improve % 5 == 0:
                search_range *= 0.85

        if not pass_improved:
            if no_improve > 5:
                break

        if no_improve > 20:
            break

        if pass_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"      CD pass={pass_count}, obj={best_obj:.12f}, "
                  f"range={search_range:.6f}, evals={total_evals}, "
                  f"elapsed={elapsed:.0f}s")

    return best_f, best_obj


def simulated_annealing(f0, time_limit, t_start=0.008, t_end=0.00005):
    """Simulated annealing for global exploration."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    current_obj = conv.max()
    best_obj = current_obj
    best_f = f.copy()

    t0 = time.monotonic()
    total_time = max(1.0, time_limit)
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        elapsed = time.monotonic() - t0
        progress = min(1.0, elapsed / total_time)
        temperature = t_start * (t_end / t_start) ** progress

        rng = np.random.default_rng(it * 7919 + 17)
        n_perturb = int(rng.integers(3, max(4, min(15, n // 6))))
        indices = rng.choice(n, size=n_perturb, replace=False)

        f_new = f.copy()
        for idx in indices:
            f_new[idx] = max(0.0, f_new[idx] + rng.gaussian(0, temperature))

        s_new = f_new.sum()
        if s_new < 1e-10:
            continue
        f_new = f_new / s_new

        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()

        if obj_new < current_obj or (temperature > 1e-6 and
                                     rng.random() < np.exp(-(obj_new - current_obj) / temperature)):
            f = f_new
            current_obj = obj_new
            if obj_new < best_obj - 1e-14:
                best_obj = obj_new
                best_f = f.copy()

        if it % 1000 == 0:
            print(f"      SA it={it}, best={best_obj:.12f}, temp={temperature:.6f}, "
                  f"elapsed={elapsed:.0f}s")

    return best_f, best_obj

    t_alloc2 = min(0.08 * (deadline - time.monotonic()), 120)
    if t_alloc2 > 10:
        print(f"  CD tight pass, {t_alloc2:.0f}s")
        a_opt, obj = coordinate_descent(current_best, time_limit=t_alloc2,
                                         max_passes=1000, initial_range=0.005)
        try_sequence(a_opt, best_tracker)

    # === Phase 5: SA + final LSE (10% of time) ===
    print(f"\n=== Phase 5: SA + Final Polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)

    t_sa = min(0.04 * (deadline - time.monotonic()), 60)
    if t_sa > 8:
        a_opt, _ = simulated_annealing(current_best, time_limit=t_sa,
                                        t_start=0.004, t_end=0.00005)
        try_sequence(a_opt, best_tracker)

    # Final LSE with very high beta
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_cur = len(current_best)
    t_final = min(0.06 * (deadline - time.monotonic()), 100)
    if t_final > 10:
        print(f"  Final LSE, {t_final:.0f}s, beta: 500 -> 500000")
        a_opt, _ = lse_nesterov(current_best, time_limit=t_final,
                                 beta_start=500.0, beta_end=500000.0,
                                 lr_init=0.2 / np.sqrt(n_cur),
                                 momentum_init=0.72)
        try_sequence(a_opt, best_tracker)

    # === Phase 6: Length exploration ===
    print(f"\n=== Phase 6: Length Exploration ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent

    alt_lengths = [384, 416, 432, 440, 452, 464, 480, 512, 544]
    if n_parent in alt_lengths:
        alt_lengths.remove(n_parent)

    # Sort by proximity to parent length
    alt_lengths.sort(key=lambda x: abs(x - n_parent))

    for n_alt in alt_lengths[:4]:
        t_remaining = deadline - time.monotonic() - 5.0
        if t_remaining < 25:
            break
        t_alloc = min(45.0, t_remaining / 2.0)
        if t_alloc < 12:
            break

        # Interpolate parent to new length
        src = current_best
        x_old = np.linspace(0, 1, len(src))
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, src)
        f_interp = np.maximum(f_interp, 0.0)
        if f_interp.sum() < 1e-10:
            f_interp = np.ones(n_alt) / n_alt
        else:
            f_interp = f_interp / f_interp.sum()

        print(f"  Trying n={n_alt}, alloc={t_alloc:.0f}s")
        a_opt, _ = lse_nesterov(f_interp, time_limit=t_alloc,
                                 beta_start=30.0, beta_end=20000.0)
        try_sequence(a_opt, best_tracker)

        # Quick CD if time allows
        t_rem2 = deadline - time.monotonic() - 5.0
        if t_rem2 > 15:
            cur_alt = best_tracker[1] if (best_tracker[1] is not None and
                                           len(best_tracker[1]) == n_alt) else a_opt
            a_opt2, _ = coordinate_descent(cur_alt, time_limit=min(20.0, t_rem2 * 0.5),
                                            max_passes=200, initial_range=0.008)
            try_sequence(a_opt2, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if np.isfinite(final_score):
            best_score = final_score
    else:
        final_seq = np.ones(n_parent) / n_parent
        best_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={best_score:.8f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
