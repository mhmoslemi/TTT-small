"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with exact 1D minimization,
combined with random restarts and multi-n exploration.
The key innovation is exploiting the structure of the convolution for efficient 1D coordinate optimization.
"""

import numpy as np
import time

def fast_convolve(a):
    """Full linear auto-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def obj_val(f):
    """Objective: 2*n*max(conv(f,f)) assuming sum(f)=1."""
    n = len(f)
    return 2.0 * n * fast_convolve(f).max()

def cd_sweep(f, n_sweeps=3):
    """Coordinate descent with exact 1D minimization."""
    n = len(f)
    f = f.copy()
    
    for _ in range(n_sweeps):
        conv = fast_convolve(f)
        for i in range(n):
            ai = f[i]
            k_min = i
            k_max = min(i + n - 1, 2 * n - 2)
            
            # M_rest: max of conv outside [k_min, k_max]
            M_rest = 0.0
            if k_min > 0:
                M_rest = max(M_rest, conv[:k_min].max())
            if k_max < 2 * n - 2:
                M_rest = max(M_rest, conv[k_max + 1:].max())
            
            # 1D problem: minimize max(M_rest, conv[k] + 2*f[k-i]*d) for d >= -ai
            # (plus quadratic term at k=2i)
            
            # Precompute coefficients
            js = np.arange(k_min, k_max + 1) - i
            coeffs = 2.0 * f[js]
            affected_conv = conv[k_min:k_max + 1].copy()
            quad_idx = -1
            if 2 * i >= k_min and 2 * i <= k_max:
                quad_idx = 2 * i - k_min
            
            # Ternary search for optimal d
            d_lo = -ai
            d_hi = max(ai, 0.001) * 2.0
            
            # Quick check: is d=0 already optimal?
            # If all affected values are below M_rest at d=0, we might increase d slightly
            # or keep d=0.
            
            for _ in range(45):
                m1 = d_lo + (d_hi - d_lo) / 3.0
                m2 = d_hi - (d_hi - d_lo) / 3.0
                
                v1 = affected_conv + coeffs * m1
                v2 = affected_conv + coeffs * m2
                if quad_idx >= 0:
                    v1[quad_idx] = affected_conv[quad_idx] + 2*ai*m1 + m1*m1
                    v2[quad_idx] = affected_conv[quad_idx] + 2*ai*m2 + m2*m2
                
                f1 = max(M_rest, v1.max())
                f2 = max(M_rest, v2.max())
                
                if f1 < f2:
                    d_hi = m2
                else:
                    d_lo = m1
            
            d_opt = (d_lo + d_hi) / 2.0
            f[i] = max(ai + d_opt, 0.0)
    
    # Renormalize
    s = f.sum()
    if s > 1e-15:
        f = f / s
    return f

def gradient_step(f, lr, beta=50.0):
    """One step of gradient descent using softmax-weighted subgradient."""
    n = len(f)
    conv = fast_convolve(f)
    
    # Softmax weights over conv values
    scaled = beta * conv
    scaled -= scaled.max()
    w = np.exp(scaled)
    w /= w.sum()

    # This is a cross-correlation, computable via FFT

    # Let j = k-i: g_i = 2 * sum_{j: 0<=j<n, 0<=i+j<2n-1} w_{i+j} * f[j]
