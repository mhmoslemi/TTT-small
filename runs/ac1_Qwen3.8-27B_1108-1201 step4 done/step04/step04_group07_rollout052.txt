"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using coordinate descent with
1D subproblem solving, combined with subgradient polishing, direct
numerical optimization, and multi-n exploration. Key innovation:
efficient coordinate-wise optimization exploiting the bilinear structure
of autocorrelation, plus a peak-targeted redistribution strategy.

import numpy as np
from scipy.optimize import minimize
import time


def fast_convolve(a):
    """FFT-based linear auto-convolution."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_obj(a):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return float('inf')
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def coord_descent(a, time_limit, verbose=False):
    """
    Coordinate descent: for each coordinate i, find the value that minimizes
    the objective (keeping other coordinates fixed). Uses grid + refinement.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s

    best_obj = compute_obj(a)
    best_a = a.copy()
    t0 = time.monotonic()
    sweep = 0

    while time.monotonic() - t0 < time_limit:
        sweep += 1
        improved = False
        improved_coords = 0

        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = a[i]
            if old_val < 1e-15:
                # Try setting it to a small positive value
                test_val = best_a.mean() * 0.5
                a[i] = test_val
                obj = compute_obj(a)
                if obj < best_obj - 1e-12:
                    best_obj = obj
                    best_a = a.copy()
                    improved = True
                    improved_coords += 1
                else:
                    a[i] = 0.0
                continue

            # Grid search for optimal value of a[i]
            # Range: [0, max_val] where max_val is large enough
            max_val = max(old_val * 5.0, 0.05)
            grid = np.concatenate([
                np.linspace(0.0, old_val, 5),
                np.linspace(old_val, max_val, 5)
            ])

            best_v = old_val
            best_v_obj = best_obj

            for v in grid:
                a[i] = v
                obj = compute_obj(a)
                if obj < best_v_obj:
                    best_v_obj = obj
                    best_v = v

            # Refine around best_v
            if best_v > 0.001 * max_val:
                lo = max(0.0, best_v * 0.7)
                hi = best_v * 1.3
                for _ in range(5):
                    for v in np.linspace(lo, hi, 5):
                        a[i] = v
                        obj = compute_obj(a)
                        if obj < best_v_obj:
                            best_v_obj = obj
                            best_v = v
            else:
                # best_v is near 0, check small values
                for v in np.linspace(0.0, max_val * 0.01, 5):
                    a[i] = v
                    obj = compute_obj(a)
                    if obj < best_v_obj:
                        best_v_obj = obj
                        best_v = v

            a[i] = best_v

                best_obj = best_v_obj

                improved = True
                improved_coords += 1

        if verbose and sweep % 5 == 0:
            print(f"    CD sweep {sweep}: obj={best_obj:.10f}, improved={improved_coords}")

        if not improved:

    return best_a, best_obj


def subgrad_optimize(a0, time_limit, lr=0.002, mom=0.92, top_k=30, temp=0.0001):
    """Heavy-ball subgradient with softmax-weighted multi-peak targeting."""
    f = np.maximum(a0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    f_obj = 2.0 * n * best_obj
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9998
        cur_temp *= 1.005

        if no_improve > 200:
            rng = np.random.default_rng((it * 31 + 17) % 100000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            v = np.zeros(n)
            cur_lr = max(lr * 0.7, 0.0003)
            no_improve = 0

    return best_f, 2.0 * n * best_obj


def direct_minimize(a0, time_limit):
    """Use scipy.optimize.minimize (SLSQP) to polish the top few peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    # Get top 5 peaks
    top_k = min(5, len(conv))
    peak_idx = np.argsort(conv)[-top_k:]

    best_obj = compute_obj(a)
    best_a = a.copy()

    t0 = time.monotonic()

    def objective(x):
        x = np.maximum(x, 0)
        xx = x / x.sum()
        c = fast_convolve(xx)
        # Use max of top peaks for smoothness
        return max(c.max(), np.max(c[peak_idx]))

    from scipy.optimize import LinearConstraint
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - 1.0,
                    'jac': lambda x: np.ones(len(x))}]

    while time.monotonic() - t0 < time_limit:
        res = minimize(objective, best_a, method='SLSQP',
                       bounds=[(0, None)] * n,
                       constraints=constraints,
                       options={'maxiter': 50, 'ftol': 1e-10})
        if res.success and res.fun < best_obj:
            a_new = np.maximum(res.x, 0)
            a_new = a_new / a_new.sum()
            obj_new = compute_obj(a_new)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a_new.copy()
                # Update peak indices
                conv = fast_convolve(best_a)
                peak_idx = np.argsort(conv)[-top_k:]
            else:
                break
        else:
            break

        # Small perturbation to escape
        rng = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
        best_a = project_simplex(best_a + rng.standard_normal(n) * 0.001)
        obj_new = compute_obj(best_a)
        if obj_new < best_obj:
            best_obj = obj_new

    return best_a, best_obj


def peak_redistribute(a0, time_limit, num_iters=200):
    """
    Peak-targeted redistribution: identify the dominant peak, find top
    contributing pairs, and transfer mass from high-product pairs to
    low-impact positions.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a, float('inf')
    a = a / s
    n = len(a)

    best_obj = compute_obj(a)
    best_a = a.copy()
    t0 = time.monotonic()
    transfer_frac = 0.05

    for it in range(num_iters):
        if time.monotonic() - t0 > time_limit:
            break

        conv = fast_convolve(a)
        k_star = int(np.argmax(conv))

        if k_star == 0:
            # Peak at lag 0 = sum(a_i^2). To reduce this, spread mass.
            # Move mass from largest a_i to smallest.
            i_src = int(np.argmax(a))
            i_dst = int(np.argmin(a))
            if a[i_src] - a[i_dst] < 1e-6:
                continue
            delta = (a[i_src] - a[i_dst]) * transfer_frac
            a[i_src] -= delta
            a[i_dst] += delta
        else:
            # Peak at lag k_star. Find top contributing pairs.
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            idx = np.arange(js, je + 1)
            products = a[idx] * a[k_star - idx]
            # Find the pair with largest product
            max_p = int(np.argmax(products))
            j_src = int(idx[max_p])
            j_dst = int(k_star - idx[max_p])

            # Transfer mass from the larger of the pair to a "safe" position
            if a[j_src] >= a[j_dst]:
                i_src = j_src
                i_other = j_dst
            else:
                i_src = j_dst
                i_other = j_src

            # Find safe destination: position that minimizes impact on all peaks
            # Approximate: position where a is smallest and far from i_src
            a_trial = a.copy()
            candidates = np.argsort(a)[:n // 4]  # bottom quartile
            # Exclude positions that are k_star away from i_src
            safe_candidates = [c for c in candidates
                               if abs(c - i_src) != k_star]
            if not safe_candidates:
                safe_candidates = list(candidates)

            i_dst = safe_candidates[0] if safe_candidates else (i_src + 1) % n

            delta = a[i_src] * transfer_frac
            a[i_src] -= delta
            a[i_dst] += delta

        # Ensure non-negativity
        a = np.maximum(a, 0)
        a = a / a.sum()

        obj = compute_obj(a)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_a = a.copy()
        else:
            # Revert and reduce transfer
            a = best_a.copy()
            transfer_frac *= 0.9
            if transfer_frac < 0.001:
                break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    for sigma_frac in [0.15, 0.25, 0.4]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / sigma) ** 2)
        inits.append(g / g.sum())

    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    alpha_b, beta_b = 2.0, 2.0
    b = (x + 1) ** alpha_b * (n - x) ** beta_b
    inits.append(b / b.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x / n))
    rc = rc + 0.01
    inits.append(rc / rc.sum())

    # Two-hump
    h1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    two = h1 + h2 + 0.01
    inits.append(two / two.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits
