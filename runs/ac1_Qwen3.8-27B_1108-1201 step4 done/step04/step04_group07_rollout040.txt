"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a multi-strategy approach:
1. Diverse initializations at multiple sequence lengths
2. LP-based Frank-Wolfe with both full and reduced constraints
3. Softmax-approximated gradient descent for polishing
4. Targeted peak-shaving local search
The key improvements over the previous approach: multiple diverse restarts,
gradient-based refinement using the softmax surrogate for the max function,
and adaptive peak-targeting in the local search phase.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build B matrix where (B*g)[k] = (f*g)[k]. B[k,j] = f[k-j] if valid."""
    n = len(f)
    B = np.zeros((2 * n - 1, n))
    for j in range(n):
        B[j:j + n, j] = f
    return B


def solve_linearized_lp(f, conv_f, B=None):
    """Solve: min t s.t. t >= conv_f[k] + 2*(B*g)[k] for all k, g >= -f, sum(g)=0."""
    n = len(f)
    if B is None:
        B = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_reduced_lp(f, conv_f, top_k=30):
    """Solve reduced LP using only top_k peak constraints."""
    n = len(f)
    if top_k >= len(conv_f):
        top_k = len(conv_f)
    peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    peaks = np.sort(peaks)

    B_full = build_conv_matrix(f)

    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((top_k, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B_full[peaks, :]
    b_ub = -conv_f[peaks]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return res.x[1:]
    return None


def solve_maxsum_lp(f, M):
    """Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None and np.any(res.x > 0):
        return res.x
    return None


def line_search(f, target, current_obj, max_alpha=1.0):
    """Find best alpha for f_new = (1-alpha)*f + alpha*target."""
    best_alpha = 0.0
    best_obj = current_obj

    alphas = np.linspace(0.05, max_alpha, 10)
    for alpha in alphas:
        f_new = (1.0 - alpha) * f + alpha * target
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-15:
            continue
        f_new = f_new / s
        obj = fast_convolve(f_new).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha

    if best_alpha > 0:
        lo = max(0.01, best_alpha * 0.6)
        hi = min(max_alpha, best_alpha * 1.4)
        for _ in range(8):
            mid = (lo + hi) / 2.0
            f_new = (1.0 - mid) * f + mid * target
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-15:
                break
            f_new = f_new / s
            obj = fast_convolve(f_new).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = mid
                lo = mid
            else:
                hi = mid

    return best_alpha, best_obj, f_new if best_alpha > 0 else None


def fw_optimize(f0, time_limit, verbose=False):
    """Full LP-based Frank-Wolfe optimization."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_obj = conv.max()

        g = solve_linearized_lp(f, conv)
        if g is None:
            no_improve += 5
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
                no_improve = 0
            continue

        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        alpha, obj_val, f_new = line_search(f, target, current_obj)

        if alpha > 1e-8 and f_new is not None:
            f = f_new
            if obj_val < best_obj - 1e-14:
                best_obj = obj_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 100 == 0:
                    print(f"    FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 60:
            scale = 0.002 + 0.001 * (no_improve // 30)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            no_improve = 0

    return best_f, best_obj

            f = f_new
            if obj_val < best_obj - 1e-14:
                best_obj = obj_val
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            no_improve = 0

    return best_f, best_obj


def gradient_polish(f0, time_limit, lr=0.01, mom=0.9, temp_init=0.01,
                    temp_growth=1.02, top_k=20, verbose=False):
    """Gradient-based polishing using softmax approximation of max."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    v = np.zeros(n)
    temp = temp_init
    no_improve = 0
    it = 0
    rng = np.random.default_rng(4242)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_obj = conv.max()

        # Select top-k peaks
        k_sel = min(top_k, len(conv))
        if k_sel >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -k_sel)[-k_sel:]

        # Softmax weights
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        wsum = w.sum()
        if wsum < 1e-30:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= wsum

        # Compute gradient: d(max b)/da approx by weighted sum of individual gradients
        grad = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            # grad[k] += 2 * w_i * a[k - j] for valid j
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * f[k - idx]

        # Momentum update
        v = mom * v - lr * grad
        f_new = project_simplex(f + v)
        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-15:
            f_new = f.copy()
        else:
            f_new = f_new / sn

        obj_new = fast_convolve(f_new).max()
        if obj_new < current_obj:
            f = f_new
        else:
            # Backtracking: try smaller step
            f = f_new * 0.5 + f * 0.5
            f = project_simplex(f)
            obj_new = fast_convolve(f).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        temp *= temp_growth
        lr *= 0.9995

        if no_improve > 200:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            lr = max(0.002, lr * 0.5)
            temp = temp_init
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    Grad iter {it}: obj={best_obj:.10f}, temp={temp:.6f}")

    return best_f, best_obj


def peak_shave(f0, time_limit, top_k=15, step=0.01, verbose=False):
    """Targeted peak shaving: reduce specific peaks by adjusting nearby coefficients."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(31337)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_obj = conv.max()

        # Find the top peaks
        k_sel = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_sel)[-k_sel:]
        peak_vals = conv[peaks]

        # For each peak, try to reduce it by adjusting coefficients
        improved = False
        for k in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            k = int(k)
            # Identify which coefficients contribute to this peak
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            contrib = f[idx] * f[k - idx]

            # Try reducing the largest contributors
            if len(idx) == 0:
                continue
            max_idx_local = np.argmax(contrib)
            if contrib[max_idx_local] < 1e-15:
                continue

            # Try several small reductions
            for frac in [0.02, 0.05, 0.1, 0.15]:
                f_trial = f.copy()
                pos = idx[max_idx_local]
                f_trial[pos] *= (1.0 - frac)
                # Redistribute to neighbors
                redist = frac * f[pos]
                if pos > 0:
                    f_trial[pos - 1] += redist * 0.4
                if pos < n - 1:
                    f_trial[pos + 1] += redist * 0.4
                f_trial = np.maximum(f_trial, 0.0)
                ts = f_trial.sum()
                if ts < 1e-15:
                    continue
                f_trial = f_trial / ts
                obj_trial = fast_convolve(f_trial).max()
                if obj_trial < best_obj - 1e-14:
                    f = f_trial
                    best_obj = obj_trial
                    best_f = f.copy()
                    improved = True
                    break
                elif obj_trial < current_obj:
                    f = f_trial
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                # Random restart in neighborhood
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0

    return best_f, best_obj


def maxsum_fw_optimize(f0, time_limit, verbose=False):
    """Max-sum LP Frank-Wolfe for larger jumps."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    m_exp = 1.0
    rng = np.random.default_rng(54321)

    while time.monotonic() - t0 < time_limit:
        g = solve_maxsum_lp(f, M * m_exp)
        if g is None:
            m_exp *= 1.1
            if m_exp > 5.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        alpha, obj_val, f_new = line_search(f, g, M)
        if alpha > 1e-10 and f_new is not None:
            f = f_new
            M = fast_convolve(f).max()
            m_exp = 1.0
            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 50:
            f = project_simplex(best_f + rng.standard_normal(n) * 0.004)
            M = fast_convolve(f).max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)

    # Dirichlet with different concentrations
    for conc in [0.3, 0.5, 1.0, 2.0]:
        inits.append(rng.dirichlet(np.ones(n) * conc))

    x = np.arange(n, dtype=np.float64)
    xc = x - 0.5 * (n - 1)

    # Gaussian windows
    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        sigma = sigma_frac * n
        g = np.exp(-0.5 * (xc / sigma) ** 2)
        g_sum = g.sum()
        if g_sum > 1e-10:
            inits.append(g / g_sum)

    # Cosine
    c = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(c / c.sum())

    # Ramp (linear)
    r = np.arange(1, n + 1, dtype=np.float64)
    inits.append(r / r.sum())
    inits.append(r[::-1] / r.sum())

    # Quadratic
    q = (x + 1) ** 2
    inits.append(q / q.sum())
    inits.append(q[::-1] / q.sum())

    # U-shape
    u = 1.0 / (0.3 * (xc / n) ** 2 + 0.1)
    inits.append(u / u.sum())

    # Sine
    s1 = np.sin(np.pi * (x + 0.5) / n) ** 2 + 0.05
    inits.append(s1 / s1.sum())

    # Exponential decay from center
    e1 = np.exp(-abs(xc) / (0.3 * n))
    inits.append(e1 / e1.sum())

    # Two bumps
    b1 = np.exp(-0.5 * ((xc - 0.3 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((xc + 0.3 * n) / (0.1 * n)) ** 2)
    inits.append(b1 / b1.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def propose_candidate(seed=42, budget_s=1800):
    """
    Multi-phase search for optimal step function heights.
    Phase 1: Diverse initialization at multiple lengths
    Phase 2: LP-based Frank-Wolfe optimization
    Phase 3: Gradient-based polishing
    Phase 4: Peak-shaving local search
    Phase 5: Final refinement
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-15:
            return float('inf')
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Try to load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            parent_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={parent_score:.8f}")
            if np.isfinite(parent_score) and parent_score < best_score:
                best_score = parent_score
                best_seq = parent.copy()
    except Exception:
        pass

    # === Phase 1: Diverse Initialization ===
    print("\n=== Phase 1: Diverse Initialization ===")
    t_start = time.monotonic()

    candidate_lengths = []
    if parent is not None:
        candidate_lengths.append(len(parent))
    for n_try in [128, 192, 256, 320, 384, 448, 512, 640, 768]:
        if n_try not in candidate_lengths:
            candidate_lengths.append(n_try)

    # Limit to reasonable lengths
    candidate_lengths = [n for n in candidate_lengths if 64 <= n <= 1024]
    candidate_lengths = sorted(candidate_lengths)

    # Evaluate initializations
    init_results = []
    for n_try in candidate_lengths:
        if time.monotonic() > deadline - 400:
            break
        inits = generate_inits(n_try, parent, rng)
        for i, ini in enumerate(inits[:6]):  # Limit inits per length
            score = try_seq(ini, f"init_n{n_try}_{i}")
            if np.isfinite(score):
                init_results.append((score, ini, n_try))

    # Sort by score
    init_results.sort(key=lambda x: x[0])

    # Select top initializations for further optimization
    top_inits = []
    seen_scores = set()
    for score, ini, n_try in init_results:
        key = n_try
        # Allow at most 2 per length
        count = sum(1 for s, i2, n2 in top_inits if n2 == n_try)
        if count < 2:
            top_inits.append((score, ini, n_try))
        if len(top_inits) >= 8:
            break

    print(f"  Phase 1: {len(init_results)} inits evaluated, best={best_score:.8f}")

    # === Phase 2: LP-based Frank-Wolfe ===
    print("\n=== Phase 2: LP-based Frank-Wolfe ===")
    remaining = deadline - time.monotonic() - 300
    t_p2 = max(60, remaining * 0.45)

    # Run on best initialization
    if len(top_inits) > 0:
        best_init_score, best_init, best_init_n = top_inits[0]
        print(f"  Primary: n={best_init_n}, init_score={best_init_score:.8f}")

        # Full LP FW
        t_fw1 = min(t_p2 * 0.5, 500)
        a_opt, _ = fw_optimize(best_init, time_limit=t_fw1, verbose=True)
        try_seq(a_opt, "fw_full_primary")

    # Try a few more initializations with reduced LP (faster)
    rem = deadline - time.monotonic() - 250
    for idx, (sc, ini, n_try) in enumerate(top_inits[1:4]):
        if rem < 30:
            break
        t_each = min(rem * 0.25, 120)
        a_opt, _ = reduced_fw_optimize(ini, time_limit=t_each, top_k=20)
        try_seq(a_opt, f"fw_red_init{idx+1}")
        rem = deadline - time.monotonic() - 200

    # If we have a good solution, refine it more
    rem = deadline - time.monotonic() - 200
    if rem > 60 and best_seq is not None:
        n_best = len(best_seq)
        t_fw2 = min(rem * 0.5, 300)
        print(f"  Refining best (n={n_best}), budget={t_fw2:.0f}s")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_fw2 * 0.6, verbose=True)
        try_seq(a_opt, "fw_refine")

        rem = deadline - time.monotonic() - 150
        if rem > 30:
            a_opt, _ = reduced_fw_optimize(best_seq, time_limit=min(rem * 0.5, 150), top_k=30)
            try_seq(a_opt, "fw_red_refine")

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: Max-sum FW for larger jumps ===
    print("\n=== Phase 3: Max-sum FW ===")
    rem = deadline - time.monotonic() - 150
    t_p3 = min(rem * 0.2, 200)

    if t_p3 > 30 and best_seq is not None:
        a_opt, _ = maxsum_fw_optimize(best_seq, time_limit=t_p3 * 0.7)
        try_seq(a_opt, "maxsum_fw")

        rem = deadline - time.monotonic() - 100
        if rem > 20:
            noise = rng.standard_normal(len(best_seq)) * 0.01
            f_pert = project_simplex(best_seq + noise)
            a_opt, _ = maxsum_fw_optimize(f_pert, time_limit=min(rem * 0.5, 80))
            try_seq(a_opt, "maxsum_fw_pert")

    # === Phase 4: Gradient polishing ===
    print("\n=== Phase 4: Gradient Polish ===")
    rem = deadline - time.monotonic() - 100
    t_p4 = min(rem * 0.3, 250)

    if t_p4 > 20 and best_seq is not None:
        n_best = len(best_seq)
        # Coarse gradient
        t_g1 = min(t_p4 * 0.5, 120)
        a_opt, _ = gradient_polish(best_seq, time_limit=t_g1,
                                   lr=0.01, mom=0.9, temp_init=0.02,
                                   temp_growth=1.03, top_k=n_best // 8)
        try_seq(a_opt, "grad_coarse")

        rem = deadline - time.monotonic() - 80
        if rem > 15:
            # Fine gradient
            t_g2 = min(rem * 0.5, 100)
            a_opt, _ = gradient_polish(best_seq, time_limit=t_g2,
                                       lr=0.003, mom=0.93, temp_init=0.005,
                                       temp_growth=1.02, top_k=n_best // 4)
            try_seq(a_opt, "grad_fine")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Peak shaving ===
    print("\n=== Phase 5: Peak Shaving ===")
    rem = deadline - time.monotonic() - 60
    t_p5 = min(rem * 0.4, 150)

    if t_p5 > 15 and best_seq is not None:
        a_opt, _ = peak_shave(best_seq, time_limit=t_p5, top_k=20, step=0.05)
        try_seq(a_opt, "peak_shave")

    # === Phase 6: Final refinement cycle ===
    print("\n=== Phase 6: Final Refinement ===")
    rem = deadline - time.monotonic() - 40

    # Alternate between FW, gradient, and peak shaving
    for attempt in range(3):
        if rem < 15 or best_seq is None:
            break
        n_best = len(best_seq)
        t_each = min(rem / max(1, 3 - attempt), 60)

        # Quick reduced FW
        a_opt, _ = reduced_fw_optimize(best_seq, time_limit=t_each * 0.4, top_k=15)
        try_seq(a_opt, f"final_fw_{attempt}")
        rem = deadline - time.monotonic() - 30

        if rem < 10:
            break

        # Quick gradient
        t_g = min(rem * 0.5, 40)
        a_opt, _ = gradient_polish(best_seq, time_limit=t_g,
                                   lr=0.005, mom=0.92, temp_init=0.005,
                                   temp_growth=1.02, top_k=10)
        try_seq(a_opt, f"final_grad_{attempt}")
        rem = deadline - time.monotonic() - 25

    # Try nearby lengths
    rem = deadline - time.monotonic() - 20
    if rem > 20 and best_seq is not None:
        n_best = len(best_seq)
        for n_try in [n_best - 16, n_best + 16, n_best + 32]:
            if n_try < 64 or rem < 10:
                continue
            px = np.interp(np.linspace(0, 1, n_try),
                           np.linspace(0, 1, n_best), best_seq)
            px = np.maximum(px, 0)
            ps = px.sum()
            if ps > 1e-10:
                px = px / ps
                a_opt, _ = reduced_fw_optimize(px, time_limit=min(rem * 0.4, 30), top_k=15)
                try_seq(a_opt, f"len{n_try}")
                rem = deadline - time.monotonic() - 10

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
