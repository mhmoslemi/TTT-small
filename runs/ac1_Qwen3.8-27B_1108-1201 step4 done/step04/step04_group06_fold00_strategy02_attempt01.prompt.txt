def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = a * g (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_soft_gradient(a, conv, top_k=20, temp=0.005):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def lp_solve(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0. Returns (g, g_sum, success)."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = result.x
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def lp_step_with_sizing(a, conv, n):

            best_t = t
            break  # First improvement in geometric sequence is good enough

def nesterov_gradient_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.003):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()

        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)

    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        # Nesterov update
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:

        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive parameter updates
        lr *= 0.9995
        temp *= 1.003
        mom = min(0.97, mom + 0.0002)

        if no_improve >= 200:
            break

    return best_a, best_obj


def lp_optimize(a0, time_limit, verbose=False):

            lp_fail_streak = 0

                a = a / s
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            perturb_count += 1

            if perturb_count > 40:
                break

            # Switch to gradient for a while

def coordinate_descent(a0, time_limit, window=20):
    """Coordinate descent: try adjusting pairs of elements to reduce peak."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        return a, 2.0 * len(a) * fast_convolve(a).max()
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        batch_size = max(1, int(500000 / n))
        for _ in range(batch_size):
            if time.monotonic() - t0 > time_limit - 0.5:
                break

            # Pick two random positions to swap mass
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Try moving small amount from i to j and vice versa
            for delta in [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02]:
                if a[i] + delta < 0 or a[j] - delta < 0:
                    continue
                a_try = a.copy()
                a_try[i] += delta
                a_try[j] -= delta
                s_try = a_try.sum()
                if s_try < 1e-15:
                    continue
                conv_try = fast_convolve(a_try)
                obj_try = 2.0 * n * conv_try.max()
                if obj_try < best_obj - 1e-10:
                    a = a_try
                    best_obj = obj_try
                    best_a = a.copy()
                    conv = conv_try
                    no_improve = 0
                    break
            else:
                no_improve += 1

        if no_improve > 200:
            break

    return best_a, best_obj


def generate_initializations(n, parent, rng):
    """Generate diverse initializations."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    inits['uniform'] = np.ones(n) / n
    inits['random'] = rng.dirichlet(np.ones(n) * 0.3)

    # Sine-based (lags at multiples)
    inits['sine1'] = (np.sin(np.pi * x / n) ** 2 + 0.01)
    inits['sine1'] /= inits['sine1'].sum()

    inits['sine2'] = (np.sin(2 * np.pi * x / n) ** 2 + 0.01)
    inits['sine2'] /= inits['sine2'].sum()

    # Bimodal (mass at edges)
    edge = max(1, n // 5)
    bi = np.zeros(n)
    bi[:edge] = 1.0
    bi[-edge:] = 1.0
    inits['bimodal'] = bi / bi.sum()

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * n) / (0.2 * n)) ** 2)
    inits['gaussian'] = g / g.sum()

    # Alternating
    alt = np.ones(n)
    alt[::2] = 1.5
    alt[1::2] = 0.5
    inits['alternating'] = alt / alt.sum()

    # Power law
    pw = (x + 1.0) ** 0.3
    inits['power'] = pw / pw.sum()

    # From parent
    if parent is not None and len(parent) > 0:
        p = np.maximum(parent, 0).astype(np.float64)
        ps = p.sum()
        if ps > 1e-10 and np.isfinite(p).all():
            if len(p) == n:
                inits['parent'] = p / ps
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p / ps)
                pp = np.maximum(pp, 0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps

    return inits


def extend_sequence(a_short, n_target):
    """Extend a shorter sequence to target length via interpolation + noise."""
    n_short = len(a_short)
    s = a_short.sum()
    if s < 1e-15:
        return np.ones(n_target) / n_target
    a_norm = a_short / s
    a_ext = np.interp(np.linspace(0, 1, n_target), np.linspace(0, 1, n_short), a_norm)
    a_ext = np.maximum(a_ext, 0)
    ses = a_ext.sum()
    if ses > 1e-15:
        a_ext = a_ext / ses
    return a_ext
