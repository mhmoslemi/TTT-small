Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth annealing (log-sum-exp
temperature schedule) on the probability simplex, with basin hopping to escape
local minima. Key innovations: (1) FFT-based gradient computation for the
smooth objective enables O(n log n) iterations at large n; (2) multi-length
exploration to exploit the 2n/(2n-1) scaling; (3) edge-concentrated
initializations based on the insight that flat convolution requires edge mass;
(4) mirror descent updates that naturally preserve positivity.
"""

import numpy as np
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if rho.size == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_smooth_gradient(a, p):
    """FFT-based computation of gradient of (1/T)log(sum exp(T*c_k))."""
    n = len(a)
    q = 2.0 * p  # length 2n-1
    a_rev = a[::-1]
    L = 1 << (3 * n).bit_length()
    Q = np.fft.rfft(q, L)
    A = np.fft.rfft(a_rev, L)
    conv = np.fft.irfft(Q * A, L)
    return conv[n-1:2*n-1]

def smooth_anneal(a0, time_limit, T_init, T_final, decay, lr, mom=0.9, 
                  hop_every=100, hop_scale=0.01, min_improve=1e-10):
    """
    Smooth annealing optimization.
    a0: initial point on simplex
    T_init, T_final: temperature range
    decay: multiplicative decay per step for T
    lr: learning rate for mirror descent
    """
    a = a0.copy()
    n = len(a)
    
    t0 = time.monotonic()
    best_obj = np.inf
    best_a = a.copy()
    no_improve = 0
    T = T_init
    v = np.zeros(n)  # momentum for mirror descent (in log-space)
    
    while time.monotonic() - t0 < time_limit:
        c = fast_convolve(a)
        # Compute softmax weights
        scaled = T * c
        scaled -= scaled.max()
        p = np.exp(scaled)
        p /= p.sum()
        
        # Compute objective (smooth approximation)
        obj_smooth = c.max()  # Use actual max for tracking
        
        # Compute gradient
        g = compute_smooth_gradient(a, p)
        
        # Mirror descent update with momentum
        v = mom * v - lr * g
        a = a * np.exp(v)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        # Track best
        c_new = fast_convolve(a)
        obj = c_new.max()
        
        if obj < best_obj - min_improve:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Anneal temperature
        T *= decay
        T = max(T, T_final)
        
        # Basin hop
        if no_improve >= hop_every:
            rng = np.random.default_rng(int(time.monotonic() * 1000000) % 2**31)
            noise = rng.standard_normal(n) * hop_scale
            a = project_simplex(best_a + noise)
            v = np.zeros(n)
            no_improve = 0
            hop_scale *= 1.05
            # Reheat
            T = min(T * 2, T_init * 0.5)
            hop_scale = min(hop_scale, 0.1)
    
    return best_a, 2 * n * best_obj  # Return n-normalized score

    """Generate an edge-concentrated initial sequence."""

    return w / w.sum()
