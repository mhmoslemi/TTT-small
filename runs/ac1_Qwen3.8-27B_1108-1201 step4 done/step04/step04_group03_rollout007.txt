Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. LSE-smoothed Nesterov accelerated gradient descent with adaptive temperature
2. Peak-chasing subgradient with momentum
3. Pair-swap local search targeting top peaks
4. Multi-restart with diverse initializations

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_smoothed_obj(f, tau, top_k=None):
    """Compute LSE-smoothed max of convolution and its gradient."""
    conv = fast_convolve(f)
    if top_k is not None and top_k < len(conv):
        # Only consider top-k peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        cv = conv[peaks]
        scaled = cv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        # Gradient: d/d f_j of sum_k w_k * conv[k]
        # = sum_k w_k * 2 * f[k-j] (for valid k-j)
        grad = np.zeros(len(f))
        for i, k in enumerate(peaks):
            if w[i] < 1e-15:
                continue
            k = int(k)
            js = max(0, k - len(f) + 1)
            je = min(len(f) - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[i] * f[k - idx]
        lse_val = tau * (np.log(w.sum()) + (cv.max() / tau))
        # Actually lse_val = tau * log(sum exp(conv[k]/tau)) = conv.max() + tau * log(sum w)
        # But we already normalized w, so lse_val = conv.max() + tau * log(sum of exp(scaled))
        # Hmm, let me just use the weighted max as a proxy
        obj = np.sum(w * cv)
        return obj, grad, conv
    else:
        scaled = conv / tau
        scaled -= scaled.max()
        w = np.exp(scaled)
        w /= w.sum()
        grad = np.zeros(len(f))
        for k in range(len(conv)):
            if w[k] < 1e-15:
                continue
            js = max(0, k - len(f) + 1)
            je = min(len(f) - 1, k)
            idx = np.arange(js, je + 1)
            grad[idx] += 2.0 * w[k] * f[k - idx]
        obj = np.sum(w * conv)
        return obj, grad, conv

def nesterov_lse(f0, time_limit, tau_init=0.001, tau_decay=0.999, lr_init=0.01,
                  lr_decay=0.9995, top_k=None, verbose=False):
    """Nesterov-accelerated gradient descent on LSE-smoothed objective."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    y = f.copy()  # Nesterov momentum variable
    v = np.zeros(n)  # velocity
    best_f = f.copy()
    best_conv = fast_convolve(f).max()
    
    tau = tau_init
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        obj, grad, conv = lse_smoothed_obj(y, tau, top_k)
        
        # Nesterov step
        # y_{t+1} = x_{t+1} + (beta_t / beta_{t+1}) * (x_{t+1} - x_t)
        # Simplified: use constant momentum
        momentum = 0.9
        v = momentum * v - lr * grad
        y = f + v
        y = project_simplex(y)
        
        # Check improvement
        conv_new = fast_convolve(y)
        max_new = conv_new.max()
        if max_new < best_conv - 1e-15:
            best_conv = max_new
            best_f = y.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = y
        tau *= tau_decay
        lr *= lr_decay
        
        if no_improve > 200:
            # Reset momentum and restart from best
            f = best_f.copy()
            y = f.copy()
            v = np.zeros(n)
            tau = max(tau, 0.0001)
            lr = max(lr_init * 0.5, 0.001)
            no_improve = 0
    
    return best_f, best_conv

    b = np.log(f)  # initial b such that softmax(b) = f
    n = len(f)
    
    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    
    lr = lr_init
    t0 = time.monotonic()
    no_improve = 0
    momentum = np.zeros(n)
    beta = 0.9
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient
        conv = fast_convolve(f)
        k_star = int(np.argmax(conv))
        k_val = conv[k_star]
        
        # grad_b[j'] = 2 * a_{j'} * (a_{k_star - j'} - k_val)

            if 0 <= idx < n:

        momentum = beta * momentum + grad_b
        b = b - lr * momentum
        
        # Update f from b
        b_shifted = b - b.max()  # for numerical stability
        f = np.exp(b_shifted)
        f = f / f.sum()
        
        # Check
        conv_new = fast_convolve(f)
        obj = conv_new.max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0

            no_improve += 1

            b = np.log(np.maximum(f, 1e-15))
            momentum = np.zeros(n)
            lr = max(lr_init * 0.5, 0.01)
            no_improve = 0
    
    return best_f, best_obj

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_convolve(a, b):
    """Linear cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def shifted_a(a, k):
    """Compute the vector s where s[j] = a[k-j] if valid, else 0."""
    n = len(a)
    s = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        s[j_lo:j_hi+1] = a[k - j_hi : k - j_lo + 1][::-1]
    return s

def multi_peak_gradient(a, top_k=5, temp=0.001):
    """Compute multi-peak weighted subgradient of max conv."""
    n = len(a)
    conv = fast_convolve(a)
    M = conv.max()
    
    if top_k >= len(conv):
        indices = np.arange(len(conv))
        values = conv
    else:
        indices = np.argpartition(conv, -top_k)[-top_k:]
        values = conv[indices]
    
    # Softmax weights
    scaled = (values - values.max()) / temp
    w = np.exp(scaled)
    w /= w.sum()
    
    # Weighted gradient
    grad = np.zeros(n)
    for i, k in enumerate(indices):
        if w[i] < 1e-15:
            continue
        k = int(k)
        s = shifted_a(a, k)
        grad += 2.0 * w[i] * s
    
    return grad, M, conv

def line_search(a, d, conv_ff, conv_fd, conv_dd, t_max, n_evals=20):
    """Exact line search: minimize max_k (ff + 2t*fd + t^2*dd) over t in [0, t_max]."""
    def at(t):
        return (conv_ff + 2.0*t*conv_fd + t*t*conv_dd).max()
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, t_max
    c = hi - gr * (hi - lo)
    d_pt = lo + gr * (hi - lo)
    fc = at(c)
    fd = at(d_pt)
    
    for _ in range(n_evals):
        if fc < fd:
            hi = d_pt
            d_pt = c
            fd = fc
            c = hi - gr * (hi - lo)
            fc = at(c)
        else:
            lo = c
            c = d_pt
            fc = fd
            d_pt = lo + gr * (hi - lo)
            fd = at(d_pt)
    
    return (lo + hi) / 2.0

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    
    # Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    # Edge-weighted
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())
    
    # Two-peaked
    g2 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    inits.append(g2 / g2.sum())
    
    # Alternating
    alt = np.ones(n)
    alt[::2] = 0.3
    inits.append(alt / alt.sum())
    
    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
