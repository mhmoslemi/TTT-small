"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Ramp-based initialization (key insight: monotone asymmetric sequences beat uniform)
2. LP-based direction finding with line search
3. Nesterov gradient descent with soft-peak weighting
4. Multi-n exploration and hierarchical refinement
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def build_conv_matrix(a):
    """Build matrix C such that C @ g = convolve(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def compute_soft_gradient(a, conv, top_k=30, temp=0.003):
    """Compute softmax-weighted gradient of the peak autocorrelation."""
    n = len(a)
    peaks = np.argpartition(conv, -top_k)[-top_k:]
    peak_vals = conv[peaks]
    scaled = peak_vals / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(peak_vals)] = 1.0
    else:
        w /= w_sum

    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-12:
            continue
        k = int(peaks[idx])
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        grad[idxs] += w[idx] * a[k - idxs]
    return grad


def lp_solve(a, M, C=None):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0."""
    n = len(a)
    if C is None:
        C = build_conv_matrix(a)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        result = linprog(c, A_ub=C, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if result.success:
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, g_sum, True
    except Exception:
        pass
    return a.copy(), 1.0, False


def objective_value(a):
    """Compute 2n*max(conv)/sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-15:
        return np.inf
    conv = fast_convolve(a)
    return 2.0 * n * conv.max() / (s * s)


def lp_step(a, conv, n):
    """One LP-based step with line search."""
    M = conv.max()
    peak_idx = int(np.argmax(conv))

    C = build_conv_matrix(a)
    g, g_sum, success = lp_solve(a, M * 1.0001, C)

    if not success or g_sum <= 1.0 + 1e-8:
        return a.copy(), False

    d = g - a
    norm_d = np.linalg.norm(d)
    if norm_d < 1e-12:
        return a.copy(), False

    # Line search with multiple step sizes
    s_a = a.sum()
    best_a = a.copy()
    best_obj = 2.0 * n * M

    candidates = [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.01]
    for t in candidates:
        a_try = a + t * d
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 1e-15:
            continue
        conv_try = fast_convolve(a_try)
        obj_try = 2.0 * n * conv_try.max()
        if obj_try < best_obj - 1e-12:
            best_obj = obj_try
            best_a = a_try.copy()
            break

    if best_a is not a:
        return best_a, True
    return a.copy(), False


def nesterov_optimize(a0, time_limit, lr_init=0.01, momentum=0.9, top_k=30, temp_init=0.002):
    """Nesterov-accelerated gradient descent with adaptive parameters."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    v = np.zeros(n)
    y = a.copy()
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = lr_init
    temp = temp_init
    mom = momentum
    no_improve = 0

    while time.monotonic() - t0 < time_limit:
        y = a + mom * v
        y = np.maximum(y, 0.0)
        s_y = y.sum()
        if s_y > 1e-15:
            y = y / s_y

        conv_y = fast_convolve(y)
        grad = compute_soft_gradient(y, conv_y, top_k=top_k, temp=temp)

        v = mom * v - lr * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new > 1e-15:
            a = a_new / s_new
        else:
            a = best_a.copy()

        conv = fast_convolve(a)
        cur_obj = 2.0 * n * conv.max()

        if cur_obj < best_obj - 1e-12:
            best_obj = cur_obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        lr *= 0.9995
        temp *= 1.003
        mom = min(0.97, mom + 0.0002)

        if no_improve >= 300:
            break

    return best_a, best_obj


def lp_optimize(a0, time_limit, verbose=False):
    """LP-based iterative optimization with restarts and gradient fallback."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    iteration = 0
    no_improve = 0
    max_no_improve = 60
    perturb_count = 0
    lp_fail_streak = 0

    while time.monotonic() - t0 < time_limit:
        a_new, improved = lp_step(a, conv, n)

        if improved:
            a = a_new
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
            lp_fail_streak = 0
        else:
            no_improve += 1
            lp_fail_streak += 1

        iteration += 1

        if verbose and iteration % 200 == 0:
            print(f"    LP iter {iteration}, obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s")

        if no_improve >= max_no_improve:
            rng = np.random.default_rng(iteration * 7 + 13 + perturb_count * 1000)
            scale = 0.002 + 0.004 * min(perturb_count, 6)
            noise = rng.standard_normal(n) * scale
            a = best_a + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            max_no_improve = min(200, 60 + 30 * perturb_count)

            if perturb_count > 50:
                break

        if lp_fail_streak > 40:
            a, obj = nesterov_optimize(best_a, time_limit=1.5, lr_init=0.006, top_k=50, temp_init=0.001)
            conv = fast_convolve(a)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_a = a.copy()
            lp_fail_streak = 0
            no_improve = 0

    return best_a, best_obj


def generate_initializations(n, rng):
    """Generate diverse initializations, emphasizing asymmetric (ramp-like) sequences."""
    inits = {}
    x = np.arange(n, dtype=np.float64)

    # CRITICAL: Linear ramp (gives ~4/3 theoretical objective)
    ramp = x + 1.0
    inits['ramp'] = ramp / ramp.sum()

    # Power law with various exponents
    for alpha in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0]:
        pw = (x + 1.0) ** alpha
        inits[f'power_{alpha}'] = pw / pw.sum()

    # Exponential ramp
    exp_ramp = np.exp(0.02 * x)
    inits['exp_ramp'] = exp_ramp / exp_ramp.sum()

    # Tilted sine (asymmetric bump)
    tilted = np.sin(np.pi * x / n) ** 2 * (x / n + 0.5)
    inits['tilted_sine'] = tilted / tilted.sum()

    # Cosine ramp (smooth asymmetric)
    cos_ramp = (1 - np.cos(np.pi * x / (n - 1))) / 2
    inits['cos_ramp'] = cos_ramp / cos_ramp.sum()

    # Bilinear: high on left, low on right (inverse ramp)
    inv_ramp = n - x
    inits['inv_ramp'] = inv_ramp / inv_ramp.sum()

    # Uniform (baseline, objective = 2)
    inits['uniform'] = np.ones(n) / n

    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)

    # "Step" - half zero, half uniform
    step = np.zeros(n)
    step[n // 2:] = 1.0
    inits['step'] = step / step.sum()

    # Smootherstep ramp
    t = x / (n - 1)
    smoother = t * t * t * (t * (t * 6 - 15) + 10)
    inits['smootherstep'] = smoother / smoother.sum()

    return inits
