```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using diverse monotone initializations,
LP-based Frank-Wolfe, coordinate descent, and subgradient polishing.
Key insight: monotone (non-symmetric) sequences achieve F≈4/3 vs F≈2 for symmetric.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_direction(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def lp_fw_optimize(f0, time_limit, verbose=False):
    """LP-based Frank-Wolfe with adaptive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_direction(f, M * m_exp, n)
        if g is None:
            m_exp *= 1.15
            if m_exp > 10.0:
                m_exp = 1.0
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.05, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M

        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.85, 1.0]:
            ft = f + alpha * (g - f)
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.1)
            hi = min(1.0, best_alpha + 0.1)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = f + mid * (g - f)
                ft = np.maximum(ft, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f_new = f + best_alpha * (g - f)
            f_new = np.maximum(f_new, 0.0)
            f_new = f_new / f_new.sum()
            conv = fast_convolve(f_new)
            M = conv.max()
            m_exp = 1.0
            f = f_new

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    LP-FW iter {it}: obj={best_obj:.10f}")
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 4.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.002 + 0.004 * min(no_improve // 20, 5)
            noise = rng.standard_normal(n) * scale
            f = project_simplex(best_f + noise)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def coordinate_descent(f0, time_limit, verbose=False):
    """Coordinate descent with golden section search per coordinate."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            old_val = f[j]
            if old_val < 1e-14:
                continue

            # Try a range of values for coordinate j
            # Keep sum = 1 by scaling others
            best_j_obj = best_obj
            best_j_val = old_val

            # Generate candidate values
            candidates = [old_val * m for m in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]]
            # Also try absolute values
            candidates.extend([0.001, 0.01, 0.05, 0.1, 0.5])
            # Filter valid
            candidates = [c for c in candidates if 0 <= c < 0.999]

            for new_val in candidates:
                if abs(new_val - old_val) < 1e-12:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                if scale < 0:
                    continue
                f_trial = f.copy()
                f_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                f_trial[mask] *= scale
                f_trial = np.maximum(f_trial, 0.0)
                st = f_trial.sum()
                if st < 1e-10:
                    continue
                f_trial = f_trial / st

                val = fast_convolve(f_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val

            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    f[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    f[mask] *= scale
                    f = np.maximum(f, 0.0)
                    f = f / f.sum()

                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_f = f.copy()
                    improved = True

        if not improved and passes > 2:
            break

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.94, top_k=50, temp=0.00005):
    """Subgradient polishing with softmax-weighted peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9997
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 1000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0005)
            no_improve = 0

    return best_f, best_obj


def smooth_optimize(f0, time_limit, beta_start=10.0, beta_end=1000.0, verbose=False):
    """
    Optimize using log-sum-exp smoothed objective with accelerated gradient.
    Continuation method: start with small beta (smooth), increase to large beta (sharp).
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f0)) / len(f0)
    else:
        f = f / s
    n = len(f)

    best_f = f.copy()
    best_obj = fast_convolve(f).max()
    t0 = time.monotonic()

    # Continuation: increase beta over time
    total_time = time_limit
    current_time = 0.0
    beta = beta_start
    v = np.zeros(n)  # momentum
    lr = 0.01
    mu = 0.9  # momentum parameter

    while time.monotonic() - t0 < time_limit:
        # Increase beta gradually
        progress = (time.monotonic() - t0) / time_limit
        beta = beta_start * (beta_end / beta_start) ** (progress ** 0.7)

        # Compute smooth objective gradient
        conv = fast_convolve(f)
        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))
        weights = exp_vals / exp_vals.sum()

        # Gradient: d/d a[i] of sum_k w[k] * conv[k]
        # conv[k] = sum_j a[j]*a[k-j]
        # d conv[k] / d a[i] = a[k-i] + a[i-k] (with bounds)
        # So grad[i] = sum_k w[k] * (a[k-i] if 0<=k-i<n + a[i-k] if 0<=i-k<n)
        # This is a weighted sum of shifted versions of a
        # Can compute as: grad = conv_correlation(weights, a)
        
        # Efficient computation via the structure:
        # grad[i] = sum_k w[k] * a[k-i] + sum_k w[k] * a[i-k]
        # First term: (w conv a)[i] where we flip... 
        # Actually: sum_k w[k] * a[k-i] is the cross-correlation (w ⋆ a)[i]
        # = (w_rev * a)[i] where * is convolution and w_rev is reversed w
        # But w has length 2n-1 and a has length n, so we need to be careful.
        
        # Let me just do it directly for now (O(n^2) but n is at most ~1000)
        grad = np.zeros(n)
        for k in range(len(conv)):
            wk = weights[k]
            if wk < 1e-15:
                continue
            # d conv[k] / d a[i] = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
            # For each i:
            # Term 1: a[k-i] valid when 0 <= k-i < n, i.e., k-n+1 <= i <= k
            # Term 2: a[i-k] valid when 0 <= i-k < n, i.e., k <= i <= k+n-1
            i_lo1 = max(0, k - n + 1)
            i_hi1 = min(n - 1, k)
            i_lo2 = max(0, k)
            i_hi2 = min(n - 1, k + n - 1)
            
            if i_lo1 <= i_hi1:
                idx1 = np.arange(i_lo1, i_hi1 + 1)
                grad[idx1] += wk * a[k - idx1]
            
            if i_lo2 <= i_hi2:
                idx2 = np.arange(i_lo2, i_hi2 + 1)
                grad[idx2] += wk * a[idx2 - k]

        # Also account for the (sum a)^2 denominator (but since we normalize to sum=1, 
        # the objective is just 2n * max(conv), and the gradient of the smooth approx
        # on the simplex is just the projection of -gradient)
        # Actually, since we're on the simplex (sum=1), the gradient of 
        # 2n * lse(conv) / 1^2 = 2n * lse(conv) is just 2n * (smooth gradient of lse)
        # But we want to MINIMIZE, so we go in the direction -grad + projection
        
        # For the simplex constraint, we subtract the mean (projection of gradient)
        grad_mean = grad.mean()
        grad = grad - grad_mean

        # Accelerated gradient step
        v = mu * v - lr * grad
        f_new = project_simplex(f + v)
        
        # Evaluate
        conv_new = fast_convolve(f_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
        
        f = f_new
        
        # Adaptive learning rate
        if obj_new > best_obj * (1 + 0.01):
            lr *= 0.8
        else:
            lr = min(lr * 1.01, 0.05)

        # Periodic restart
        if verbose and (time.monotonic() - t0) % 10 < 0.1:
            print(f"    Smooth opt: t={time.monotonic()-t0:.1f}s, beta={beta:.1f}, obj={obj_new:.8f}")

    return best_f, best_obj
```
