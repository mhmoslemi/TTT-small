Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using spectral shaping initialization,
L-BFGS-B optimization with softmax parameterization, and multi-length search.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def spectral_shaping(a0, n_iters=200):

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    L = 2 * n - 1
    
    best_a = a.copy()
    best_peak = fast_convolve(a).max()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, L)
        mag = np.abs(A)
        dc = A[0].real

    a /= a.sum()

    z0 = np.log(a)
    z0 -= z0.mean()
    
    def fun_grad(z):
        z_shift = z - z.max()
        w = np.exp(z_shift)
        wss = w.sum()
        if wss < 1e-30:
            return 1e10, np.zeros_like(z)
        w /= wss
        
        conv = fast_convolve(w)
        T = T_soft
        cv_max = conv.max()
        shifted = T * (conv - cv_max)
        exp_s = np.exp(shifted)
        Z = exp_s.sum()

            Z = 1e-30

        val = 2.0 * n * sm

def interpolate_sequence(f, new_n):

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def spectral_shaping(a0, n_iters=100):
    """Apply flat-spectrum iterations to flatten autocorrelation."""
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a /= s
    n = len(a)
    L = 2 * n - 1

    best_a = a.copy()
    best_peak = fast_convolve(a).max()

    for it in range(n_iters):
        A = np.fft.rfft(a, L)
        mag = np.abs(A)
        dc = A[0].real

        if len(mag) > 1:

        peak = fast_convolve(a_new).max()
        if peak < best_peak:
            best_peak = peak
            best_a = a_new.copy()
            a = a_new

    return best_a

    z0 -= z0.mean()

    def fun_grad(z):
        z_shift = z - z.max()
        w = np.exp(z_shift)
        wss = w.sum()
        if wss < 1e-30:
            return 1e10, np.zeros_like(z)
        w = w / wss

        conv = fast_convolve(w)
        T = T_soft
        cv_max = conv.max()
        shifted = T * (conv - cv_max)
        exp_s = np.exp(shifted)
        Z = exp_s.sum()
        if Z < 1e-30:
            Z = 1e-30
        sm = cv_max + np.log(Z) / T
        val = 2.0 * n * sm

        wts = exp_s / Z  # shape (2n-1,)

        # cross_corr[m] = sum_j w[j] * wts[m+j]
        # Use FFT: cross_corr = (w_rev * wts)[n-1 : n-1+n]
        L_fft = 1 << (3 * n - 1).bit_length()
        w_rev = w[::-1]
        W_rev = np.fft.rfft(w_rev, L_fft)
        WTS = np.fft.rfft(wts, L_fft)
        conv_result = np.fft.irfft(W_rev * WTS, L_fft)
        cross_corr = conv_result[n - 1: n - 1 + n]

        expected_conv = np.dot(wts, conv)

        grad_z = 4.0 * n * w * (cross_corr - expected_conv)

        return val, grad_z

    result = minimize(
        fun_grad,
        z0,
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxfun': maxiter * 3}
    )

    z_best = result.x
    z_shift = z_best - z_best.max()
    a_best = np.exp(z_shift)
    a_best /= a_best.sum()
    return a_best


def lbfgs_multistage(a0, T_list, maxiter_each=200):
    """Run L-BFGS with increasing T values for progressive sharpening."""
    a = a0.copy()
    for T in T_list:
        a = lbfgs_optimize(a, T_soft=T, maxiter=maxiter_each)
    return a


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def make_structured_init(n, variant, seed=0):
    """Create structured initial sequences of various types."""
    rng = np.random.default_rng(seed)
    if variant == 0:
        x = np.linspace(0, 1, n)
        a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    elif variant == 1:
        x = np.linspace(-3, 3, n)
        a = np.exp(-x**2 / 2.0)
    elif variant == 2:
        a = np.ones(n) + rng.standard_normal(n) * 0.1
        a = np.maximum(a, 0)
    elif variant == 3:
        x = np.linspace(0, 1, n)
        a = np.sin(np.pi * x)
    elif variant == 4:
        x = np.linspace(0, 1, n)
        a = 1.0 + 0.3 * np.cos(4 * np.pi * x) + 0.1 * np.cos(8 * np.pi * x)
        a = np.maximum(a, 0)
    elif variant == 5:
        x = np.linspace(0, 1, n)
        a = 1.0 / (1.0 + 10.0 * (2 * x - 1)**2)
    else:
        a = np.ones(n) + rng.standard_normal(n) * 0.05
        a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-15:
        a /= s
    return a


def coord_descent_polish(a0, n_passes=3):

            if old_val < 1e-14:

                new_val = old_val * mult
