"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses adaptive LSE gradient descent with Nesterov momentum, top-k subgradient
averaging, FW-LP with adaptive constraint tightening, sparsification, and
alternative length exploration.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_grad(weights, f):
    """Compute gradient: 2 * cross_correlation(weights, f) of length n."""
    n = len(f)
    f_flipped = f[::-1].copy()
    total_len = (2 * n - 1) + n - 1
    fft_size = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(weights, fft_size)
    F = np.fft.rfft(f_flipped, fft_size)
    conv = np.fft.irfft(W * F, fft_size)
    return 2.0 * conv[n - 1:n - 1 + n]


def compute_topk_grad(conv, f, top_k, temperature):
    """Compute top-k weighted subgradient of max(conv)."""
    n = len(f)
    conv_len = len(conv)
    if top_k >= conv_len:
        peaks = np.arange(conv_len)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    pv = conv[peaks]
    scaled = pv / temperature
    scaled -= scaled.max()
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w = w / w_sum
    grad = np.zeros(n)
    for idx in range(len(peaks)):
        if w[idx] < 1e-14:
            continue
        k = int(peaks[idx])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx_arr = np.arange(js, je + 1)
        grad[idx_arr] += 2.0 * w[idx] * f[k - idx_arr]
    return grad


def build_lp_matrix(f):
    """Build sparse constraint matrix for the LP."""
    n = len(f)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend(k for _ in js)
            cols.extend(js.tolist())
            vals.extend((f[ks] for ks in (k - js)))
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_for_direction(f, M, A_cache):
    """Solve LP: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c, A_ub=A_cache, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True, 'time_limit': 15})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(f0, time_limit, lp_timeout=15):
    """Frank-Wolfe with LP oracle and adaptive constraint tightening."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1
        A_cache = build_lp_matrix(f)
        g = solve_lp_for_direction(f, M * m_exp, A_cache)
        if g is None:
            m_exp *= 1.15
            if m_exp > 5.0:
                m_exp = 1.0
            no_improve += 1
            if no_improve > 60:
                break
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = np.maximum(g, 0.0)
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                break
            m_exp = min(m_exp * 1.03, 3.0)
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        alphas = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]
        for alpha in alphas:
            ft = np.maximum((1.0 - alpha) * f + alpha * g, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 0:
            lo = max(0.0, best_alpha - 0.05)
            hi = min(1.0, best_alpha + 0.05)
            for _ in range(8):
                mid = (lo + hi) / 2.0
                ft = np.maximum((1.0 - mid) * f + mid * g, 0.0)
                st = ft.sum()
                if st < 1e-15:
                    break
                ft = ft / st
                val = fast_convolve(ft).max()
                if val < best_val:
                    best_val = val
                    best_alpha = mid
                    lo = mid
                else:
                    hi = mid

        if best_alpha > 1e-10:
            f = np.maximum((1.0 - best_alpha) * f + best_alpha * g, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.08, 5.0)

        if no_improve > 50:
            rng = np.random.default_rng(it * 31 + 7)
            scale = 0.003 + 0.004 * min(1.0, (no_improve - 50) / 60.0)
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def lse_nesterov(f0, time_limit, beta_start=10.0, beta_end=10000.0,
                 top_k=32, temp_init=0.00005):
    """LSE gradient descent with Nesterov momentum and top-k subgradient."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 3.0 / n
    momentum = 0.92
    v = np.zeros(n)
    y = f.copy()  # Nesterov lookahead
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    temp = temp_init

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov lookahead
        y = f + momentum * v

        conv_y = fast_convolve(y)

        # Compute gradient using top-k
        grad = compute_topk_grad(conv_y, y, top_k, temp)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 50
            break
        grad = grad / grad_norm

        # Momentum update
        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)

        # Backtracking line search
        obj_new = fast_convolve(f_new).max()
        alpha = 1.0
        for _ in range(10):
            f_try = project_simplex(f + alpha * v)
            obj_try = fast_convolve(f_try).max()
            if obj_try <= conv_y.max() + 1e-12:
                break
            alpha *= 0.5

        f_new = project_simplex(f + alpha * v)
        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            # Slightly increase lr
            lr = min(lr * 1.003, 6.0 / n)
        else:
            no_improve += 1
            if no_improve >= 15:
                lr *= 0.93
                momentum = max(0.85, momentum - 0.003)
            if no_improve > 100:
                f = best_f.copy()
                v = np.zeros(n)
                y = f.copy()
                lr = max(2.0 / n, 0.001)
                beta = min(beta * 1.5, beta_end)
                momentum = 0.90
                no_improve = 0
                # Jump
                rng = np.random.default_rng(it * 13 + 5)
                f = project_simplex(f + rng.standard_normal(n) * 0.002)
                y = f.copy()

        # Ramp up beta and temperature
        if beta < beta_end:
            beta *= 1.002
        temp *= 1.001

        if it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-Nest it={it}, obj={best_obj:.12f}, beta={beta:.0f}, "
                  f"lr={lr:.6f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def lse_high_beta(f0, time_limit, beta_start=500.0, beta_end=50000.0, top_k=16):
    """High-beta LSE for final polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    beta = beta_start
    lr = 1.0 / n
    momentum = 0.95
    v = np.zeros(n)
    y = f.copy()
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    temp = 0.00001

    while time.monotonic() - t0 < time_limit:
        it += 1

        y = f + momentum * v
        conv_y = fast_convolve(y)

        grad = compute_topk_grad(conv_y, y, top_k, temp)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            no_improve += 50
            break
        grad = grad / grad_norm

        v = momentum * v - lr * grad
        f_new = project_simplex(f + v)
        obj_new = fast_convolve(f_new).max()

        if obj_new < best_obj - 1e-15:
            best_obj = obj_new
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
            lr = min(lr * 1.001, 3.0 / n)
        else:
            no_improve += 1
            if no_improve >= 20:
                lr *= 0.95
            if no_improve > 150:
                f = best_f.copy()
                v = np.zeros(n)
                y = f.copy()
                lr = max(0.5 / n, 0.0005)
                beta = min(beta * 1.3, beta_end)
                no_improve = 0

        if beta < beta_end:
            beta *= 1.001

        if it % 3000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    LSE-HB it={it}, obj={best_obj:.12f}, beta={beta:.0f}, "
                  f"lr={lr:.7f}, elapsed={elapsed:.0f}s")

    return best_f, best_obj


def sparsify_and_reoptimize(f, threshold_frac=0.01, time_limit=60):
    """Threshold small values, renormalize, and do quick re-optimization."""
    f = np.maximum(f, 0.0)
    total = f.sum()
    if total < 1e-15:
        return f
    f = f / total
    n = len(f)
    
    sorted_f = np.sort(f)[::-1]
    cumsum = np.cumsum(sorted_f)
    # Find k such that the tail beyond k is less than threshold_frac of total
    k = np.searchsorted(cumsum, 1.0 - threshold_frac)
    k = max(k, max(1, n // 10))
    
    f_sparse = f.copy()
    f_sparse[sorted_f < sorted_f[k]] = 0.0
    s = f_sparse.sum()
    if s < 1e-15:
        return f
    f_sparse = f_sparse / s
    
    # Quick LSE polish on sparse version
    result, _ = lse_nesterov(f_sparse, time_limit=min(time_limit, 30),
                             beta_start=50.0, beta_end=2000.0, top_k=16,
                             temp_init=0.00003)
    return result


def try_sequence(a, best_tracker):
    """Evaluate a sequence and update best tracker."""
    a = np.maximum(a, 0.0)
    try:
        score = float(evaluate_sequence(a.tolist()))
    except Exception:
        return float('inf')
    if np.isfinite(score) and score < best_tracker[0]:
        best_tracker[0] = score
        best_tracker[1] = a.copy()
        print(f"  NEW BEST: {score:.10f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase optimization:
    Phase 1: Adaptive LSE gradient descent with Nesterov momentum
    Phase 2: FW-LP structural optimization
    Phase 3: High-beta LSE polishing
    Phase 4: Sparsification and re-optimization
    Phase 5: Alternative length exploration
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    best_tracker = [float('inf'), None]

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.10f}")
            try_sequence(parent, best_tracker)
    except Exception as e:
        print(f"No parent available: {e}")

    if parent is None:
        parent = np.ones(448) / 448
        try_sequence(parent, best_tracker)

    n_parent = len(parent)
    print(f"Starting optimization, n={n_parent}, initial best={best_tracker[0]:.10f}")

    # === Phase 1: LSE Nesterov ===
    print("\n=== Phase 1: LSE Nesterov (adaptive beta) ===")
    t_alloc = min(600, deadline - time.monotonic() - 800)
    if t_alloc > 60:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_opt, obj = lse_nesterov(parent, time_limit=t_alloc,
                                  beta_start=10.0, beta_end=8000.0,
                                  top_k=32, temp_init=0.00005)
        try_sequence(a_opt, best_tracker)

    # === Phase 2: FW-LP ===
    print("\n=== Phase 2: FW-LP Structural Optimization ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(450, deadline - time.monotonic() - 500)
    if t_alloc > 50:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_opt, obj = fw_lp_optimize(current_best, time_limit=t_alloc)
        try_sequence(a_opt, best_tracker)

    # === Phase 3: High-beta LSE polish ===
    print("\n=== Phase 3: High-Beta LSE Polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_alloc = min(350, deadline - time.monotonic() - 200)
    if t_alloc > 40:
        print(f"  Time budget: {t_alloc:.0f}s")
        a_opt, obj = lse_high_beta(current_best, time_limit=t_alloc,
                                   beta_start=500.0, beta_end=50000.0, top_k=16)
        try_sequence(a_opt, best_tracker)

    # === Phase 4: Sparsification ===
    print("\n=== Phase 4: Sparsification ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic() - 100
    if t_remaining > 40:
        for thresh in [0.02, 0.05, 0.01]:
            t_rem = deadline - time.monotonic() - 60
            if t_rem < 20:
                break
            a_sparse = sparsify_and_reoptimize(current_best,
                                               threshold_frac=thresh,
                                               time_limit=min(30, t_rem // 2))
            try_sequence(a_sparse, best_tracker)
            if a_sparse is not None and best_tracker[1] is not None:
                current_best = best_tracker[1]

    # === Phase 5: Alternative lengths ===
    print("\n=== Phase 5: Alternative Lengths ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    n_best = len(current_best)

    alt_lengths = []
    for dn in [-2, -1, 1, 2, 4, 8, -4]:
        n_alt = n_best + dn
        if n_alt >= 100 and n_alt <= 600 and n_alt != n_best:
            alt_lengths.append(n_alt)
    alt_lengths = list(set(alt_lengths))[:4]

    for n_alt in alt_lengths:
        t_remaining = deadline - time.monotonic() - 30
        if t_remaining < 40:
            break
        t_alloc = min(60, t_remaining // max(1, len(alt_lengths)))
        if t_alloc < 20:
            break

        # Interpolate to new length
        x_old = np.linspace(0, 1, n_best)
        x_new = np.linspace(0, 1, n_alt)
        f_interp = np.interp(x_new, x_old, current_best)
        f_interp = np.maximum(f_interp, 0)
        s = f_interp.sum()
        if s > 1e-10:
            f_interp = f_interp / s

        # Quick optimization
        a_opt, _ = lse_nesterov(f_interp, time_limit=t_alloc,
                                beta_start=20.0, beta_end=3000.0,
                                top_k=24, temp_init=0.00004)
        try_sequence(a_opt, best_tracker)

        print(f"  n={n_alt}: best so far={best_tracker[0]:.10f}")

    # === Final: one more LSE polish on the overall best ===
    print("\n=== Final Polish ===")
    current_best = best_tracker[1] if best_tracker[1] is not None else parent
    t_remaining = deadline - time.monotonic() - 10
    if t_remaining > 30:
        a_opt, _ = lse_high_beta(current_best, time_limit=t_remaining,
                                 beta_start=1000.0, beta_end=80000.0, top_k=8)
        try_sequence(a_opt, best_tracker)

    # === Final verification ===
    if best_tracker[1] is not None:
        final_seq = best_tracker[1]
        final_score = float(evaluate_sequence(final_seq.tolist()))
        if not np.isfinite(final_score):
            final_seq = parent
            final_score = float(evaluate_sequence(final_seq.tolist()))
    else:
        final_seq = parent
        final_score = float(evaluate_sequence(final_seq.tolist()))

    print(f"\nFINAL: score={final_score:.10f}, n={len(final_seq)}")
    return [float(x) for x in final_seq]
