Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max gradient descent
with adaptive temperature, momentum, and multi-start initialization.

import numpy as np

def fast_convolve(a):
    n = len(a)
    size = 2*n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using smooth-max gradient descent
with adaptive temperature, momentum, multi-start initialization from warped
ramps, and coordinate descent polishing. The key insight is that asymmetric
(ramp-like) sequences achieve F ~ 4/3 << 2 (uniform), and further optimization
can push below the simple ramp value.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(a, w):
    """Compute gradient of weighted sum of auto-convolution.
    g[i] = 2 * sum_m a[m] * w[i+m]
    Computed via FFT cross-correlation trick.
    """
    n = len(a)
    fft_size = 1 << (n + len(w) - 2).bit_length()
    A_rev = np.fft.rfft(a[::-1], fft_size)
    W = np.fft.rfft(w, fft_size)
    conv = np.fft.irfft(A_rev * W, fft_size)[:n + len(w) - 1]
    return 2.0 * conv[n - 1:2 * n - 1]

        passes += 1

        for j in range(0, n, 1):
            if time.monotonic() - t0 > time_limit:

            if old < 1e-15:

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
