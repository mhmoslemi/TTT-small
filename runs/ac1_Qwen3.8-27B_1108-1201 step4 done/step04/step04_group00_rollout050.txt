def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using Nesterov-accelerated subgradient
descent with adaptive parameters, multi-length exploration, coordinate descent
polishing, and multi-start randomization.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_sg(a, conv, k_actual=100, temp=0.005):
    """Compute weighted subgradient of max(conv) using top-k peaks."""
    n = len(a)
    ka = min(k_actual, 2 * n - 1)
    peak_idx = np.argpartition(conv, -ka)[-ka:]
    pv = conv[peak_idx]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peak_idx))
        w[0] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peak_idx)):
        if w[i] < 1e-15:
            continue
        k = int(peak_idx[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        seg = a[k - je:k - js + 1][::-1]
        sg[js:je + 1] += 2.0 * w[i] * seg
    return sg


def nesterov_sg(a0, time_limit, top_k=100, lr_init=0.005, mom_init=0.9,
                 temp_init=0.005, verbose=False):
    """Nesterov-accelerated subgradient descent on the simplex."""
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    a_prev = a.copy()
    lr = lr_init
    mom = mom_init
    temp = temp_init
    
    t0 = time.monotonic()
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Nesterov extrapolation
        y = a + mom * (a - a_prev)
        y = project_simplex(y)
        
        # Subgradient at y
        conv_y = fast_convolve(y)
        sg = compute_sg(y, conv_y, top_k, temp)
        
        # Update
        a_prev = a.copy()
        a = project_simplex(a - lr * sg)
        
        conv = fast_convolve(a)
        obj = conv.max()
        
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt hyperparameters
        if it % 50 == 0:
            lr *= 0.995
            temp *= 1.002
            mom = min(mom + 0.0001, 0.99)
        
        if no_improve > 300:
            rng = np.random.default_rng(it * 31 + 7)
            noise_scale = 0.002 + 0.003 * min((no_improve - 300) // 200, 5)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            a_prev = a.copy()
            lr = max(lr * 3, lr_init * 0.5)
            temp = temp_init * 0.5
            no_improve = 0
            conv = fast_convolve(a)
    
    if verbose:
        print(f"  Nesterov-SG: {it} iters, best_obj={best_obj:.10f}")
    return best_a, best_obj


def cd_polish(a0, time_limit, verbose=False):
    """Coordinate descent polishing with simplex normalization."""
    a = project_simplex(a0.astype(np.float64).copy())
    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    passes = 0
    multipliers = [0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0]
    
    while time.monotonic() - t0 < time_limit:
        passes += 1
        improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = a[j]
            if old_val < 1e-12:
                continue
            
            best_j_val = old_val
            best_j_obj = conv.max()
            
            for mult in multipliers:
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                
                a_trial = a.copy()
                a_trial[j] = new_val
                # Renormalize: divide all by sum to maintain sum=1
                # But we need to be careful: if new_val > 1, the negative values
                # from other coords becoming... actually since we're just scaling,
                # let's just normalize the whole thing.
                s = a_trial.sum()
                if s < 1e-15:
                    continue
                a_trial = a_trial / s
                
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-15:
                    best_j_obj = val
                    best_j_val = new_val
            
            if best_j_val != old_val:
                a[j] = best_j_val
                s = a.sum()
                if s > 1e-15:
                    a = a / s
                conv = fast_convolve(a)
                if conv.max() < best_obj - 1e-14:
                    best_obj = conv.max()
                    best_a = a.copy()
                    improved = True
        
        if not improved and passes > 2:
            break
    
    if verbose:
        print(f"  CD: {passes} passes, best_obj={best_obj:.10f}")
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Linear interpolation to new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp
