def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g)."""
    n = len(a)
    size = 2 * n - 1
    C = np.zeros((size, n))
    for j in range(n):
        C[j:j+n, j] = a
    return C

    return np.fft.irfft(A * A, fft_size)[:size]

    return C

def fourier_perturbation(a, rng, scale=0.003):
    """Perturb a in the Fourier domain."""
    n = len(a)
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    phase = np.angle(A)
    mag = np.abs(A)
    # Preserve DC component
    mag_pert = mag.copy()
    mag_pert[1:] = mag[1:] * (1.0 + rng.standard_normal(len(mag) - 1) * scale)
    mag_pert[1:] = np.maximum(mag_pert[1:], 0)
    A_new = mag_pert * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, fft_size)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-15:
        a_new = a_new / s
    else:
        a_new = a.copy()
    return a_new

            j_end = min(n - 1, k)

        a_new = np.maximum(a_new, 0.0)
