from scipy.optimize import linprog
from scipy.signal import correlate
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = len(f)
    conv = fast_convolve(f)

    # This is 2 * correlate(f, w, 'valid')
    grad = 2.0 * correlate(f, w, mode='valid')

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def soft_max_conv(f, T):
    """Compute soft-max of conv(f,f) and its gradient w.r.t. f."""
    n = len(f)
    conv = fast_convolve(f)
    scaled = conv / T
    mx = scaled.max()
    exp_s = np.exp(scaled - mx)
    w = exp_s / exp_s.sum()
    soft_val = T * (np.log(exp_s.sum()) + mx)
    # Gradient via convolution trick
    grad = 2.0 * np.convolve(f[::-1], w, mode='full')[n-1:2*n-1]
    return soft_val, grad, conv

def solve_lp(f, M):

    n = len(f)

    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None

        # Peak-directed restart
        if no_improve > 50:
            conv_best = fast_convolve(best_f)
            peak_k = np.argmax(conv_best)
            # Identify which indices contribute most to the peak
            js = max(0, peak_k - n + 1)
            je = min(n - 1, peak_k)
            contrib = np.zeros(n)
            for j in range(js, je + 1):
                contrib[j] = best_f[j] * best_f[peak_k - j]
            # Perturb high-contribution indices more
            scale = 0.003 + 0.003 * (no_improve // 60)
            noise = rng_local.standard_normal(n) * scale
            # Extra noise at high-contribution positions
            hc_mask = contrib > np.median(contrib[js:je + 1])
            noise[hc_mask] *= 2.0

            strategies = no_improve % 4
            if strategies == 1:
                # Mirror component
                mirror = best_f[::-1]
                blend = 0.2 + 0.05 * (no_improve // 30)
                f = project_simplex((1 - blend) * best_f + blend * mirror + noise)
            elif strategies == 2:
                # Edge boost
                x = np.arange(n, dtype=np.float64)
                edge = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
                edge = edge / edge.sum()
                blend = 0.15
                f = project_simplex((1 - blend) * best_f + blend * edge + noise)
            elif strategies == 3:
                # Random Dirichlet
                f = project_simplex(rng_local.dirichlet(np.ones(n) * 0.7) * 0.3 + best_f * 0.7 + noise)
            else:
                f = project_simplex(best_f + noise)

            conv = fast_convolve(f)
            M = conv.max()
            m_exp = 1.0
            no_improve = 0

    return best_f, best_obj


def sg_polish(f0, time_limit, lr=0.003, mom=0.93, top_k=30, temp=0.00005):
    """Subgradient descent with multi-peak softmax weighting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    rng_local = np.random.default_rng(7777)

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-13:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        v = mom * v - cur_lr * sg
        f = project_simplex(f + v)
        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9996
        cur_temp *= 1.003

        if no_improve > 300:
            f = project_simplex(best_f + rng_local.standard_normal(n) * 0.002)
            v = np.zeros(n)
            cur_lr = max(lr * 0.5, 0.0003)
            no_improve = 0

    return best_f, best_obj


def coordinate_descent_polish(f0, time_limit):
    """Coordinate descent for fine polishing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng_local = np.random.default_rng(4242)

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng_local.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 2:
                break
            candidates = [(j + 1) % n, (j - 1) % n, (j + n // 3) % n, rng_local.integers(n)]
            for source in candidates:
                source = int(source) % n
                if source == j:
                    continue
                for frac in [0.05, 0.1, 0.15, 0.25]:
                    delta = f[source] * frac
                    if delta < 1e-15:
                        continue
                    f_new = f.copy()
                    f_new[j] += delta
                    f_new[source] -= delta
                    if f_new[source] < 0:
                        f_new[j] += f_new[source]
                        f_new[source] = 0.0
                    f_new = np.maximum(f_new, 0.0)
                    sn = f_new.sum()
                    if sn < 1e-15:
                        continue
                    f_new = f_new / sn
                    val = fast_convolve(f_new).max()
                    if val < best_obj - 1e-15:
                        f = f_new
                        best_obj = val
                        best_f = f.copy()
                        improved = True
                if improved:
                    break

        if not improved:
            break

    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-concentrated (U-shape)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.7
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Two-peak symmetric
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())

    # Sine window
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Quadratic phase (Barker-like)
    phase = np.exp(1j * np.pi * x * x / n)
    real_part = np.abs(phase).astype(np.float64)
    # Use a real approximation: cosine of quadratic phase
    cos_quad = np.abs(np.cos(np.pi * x * x / (2.0 * n)))
    inits.append(cos_quad / cos_quad.sum())

    # M-sequence inspired (periodic pattern)
    # Generate a simple binary-like pattern
    m_seq = np.zeros(n)
    period = max(3, n // 32)
    for i in range(n):
        m_seq[i] = 1.0 if (i % period) < period // 2 else 0.3
    inits.append(m_seq / m_seq.sum())

    # Power law (heavy at start)
    pl = (x + 1.0) ** (-0.8)
    inits.append(pl / pl.sum())

    # Cosine taper
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.02)
    inits.append(cos_t / cos_t.sum())

    # Random Dirichlet
    inits.append(rng.dirichlet(np.ones(n) * 0.4))

    # Ramp
    ramp = (x + 1.0) ** 1.5
    inits.append(ramp / ramp.sum())

    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
        # Also try parent with slight perturbation
        px2 = px + rng.standard_normal(n) * 0.01
        px2 = np.maximum(px2, 0)
        ps2 = px2.sum()
        if ps2 > 1e-10:
            inits.append(px2 / ps2)

    return inits


def try_sequence(a, label, state):
    """Evaluate a sequence and update state if better."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Multi-length exploration with structured inits
    Phase 2: Deep FW-LP optimization
    Phase 3: Soft-max FISTA polishing (new)
    Phase 4: Final CD + SG polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}, score={evaluate_sequence(parent.tolist()):.6f}")
    except Exception:
        pass

    # === Phase 1: Multi-length exploration (20% of budget) ===
    print("=== Phase 1: Multi-length Exploration ===")
    phase1_end = time.monotonic() + (float(budget_s) * 0.20)

    if parent is not None:
        n_p = len(parent)
        t1 = min(80, phase1_end - time.monotonic() - 30)
        if t1 > 20:
            print(f"  Quick FW-LP on parent (n={n_p}), t={t1:.0f}s")
            a_opt, obj = fw_lp_optimize(parent, time_limit=t1)
            try_sequence(a_opt, "parent_quick", state)

    # Multi-length exploration
    lengths = [320, 352, 384, 400, 416]
    if parent is not None:
        if len(parent) not in lengths:
            lengths.insert(0, len(parent))
        else:
            lengths = sorted(lengths, key=lambda x: abs(x - len(parent)))
        lengths.insert(0, len(parent))

    # Remove duplicates keeping first
    seen = set()
    unique_lengths = []
    for l in lengths:
        if l not in seen:
            seen.add(l)
            unique_lengths.append(l)
    lengths = unique_lengths

    for n in lengths:
        if time.monotonic() > phase1_end - 40:
            break

        inits = generate_inits(n, parent, rng)
        n_try = min(4, len(inits))
        t_rem = phase1_end - time.monotonic() - 20
        t_per_init = max(10, t_rem / (n_try * max(1, len(lengths) - lengths.index(n))))
        t_per_init = min(t_per_init, 60)

        for i in range(n_try):
            if time.monotonic() > phase1_end - 15:
                break
            t_alloc = min(t_per_init, phase1_end - time.monotonic() - 10)
            if t_alloc < 8:
                break
            f_init = inits[i]
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_sequence(a_opt, f"n{n}_i{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Deep optimization (45% of budget) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + (float(budget_s) * 0.45)

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Optimizing n={n_best}, current best={state['best_score']:.6f}")

        # Main deep FW-LP run
        t_main = min(300, (phase2_end - time.monotonic()) * 0.55)
        if t_main > 30:
            print(f"  Main FW-LP: {t_main:.0f}s")
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_main, verbose=True)
            try_sequence(a_opt, "deep_main", state)

        # Perturbation runs
        perturb_scales = [0.003, 0.008, 0.02, 0.05]
        for ps in perturb_scales:
            rem = phase2_end - time.monotonic() - 20
            if rem < 15:
                break
            t_alloc = min(rem / (len(perturb_scales) - perturb_scales.index(ps) + 1) + 10, 100)
            if t_alloc < 10:
                break
            noise = rng.standard_normal(n_best) * ps
            f_init = project_simplex(state['best_seq'] + noise)
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_sequence(a_opt, f"pert_{ps}", state)

        # Mirror blend
        rem = phase2_end - time.monotonic() - 15
        if rem > 20:
            f_mirror = project_simplex(0.8 * state['best_seq'] + 0.2 * state['best_seq'][::-1])
            a_opt, obj = fw_lp_optimize(f_mirror, time_limit=min(rem, 80))
            try_sequence(a_opt, "mirror_blend", state)

    # === Phase 3: Soft-max FISTA polishing (25% of budget) ===
    print(f"\n=== Phase 3: Soft-max FISTA Polishing ===")
    phase3_end = time.monotonic() + (float(budget_s) * 0.25)

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Current best: {state['best_score']:.6f}")

        # FISTA with different starting temperatures
        T_configs = [
            (0.05, 0.003, "fista_T5"),
            (0.02, 0.002, "fista_T2"),
            (0.01, 0.001, "fista_T1"),
        ]

        rem = phase3_end - time.monotonic() - 10
        if rem > 20:
            t_alloc = min(rem // 2, 200)
            if t_alloc > 15:
                print(f"  FISTA (T=0.05): {t_alloc:.0f}s")
                a_opt, obj = softmax_fista(state['best_seq'], time_limit=t_alloc,
                                           T_start=0.05, T_min=0.003, verbose=True)
                try_sequence(a_opt, "fista_main", state)

        # Continue FISTA from the best with lower T
        rem = phase3_end - time.monotonic() - 10
        if rem > 15:
            t_alloc = min(rem, 150)
            print(f"  FISTA (T=0.02): {t_alloc:.0f}s")
            a_opt, obj = softmax_fista(state['best_seq'], time_limit=t_alloc,
                                       T_start=0.02, T_min=0.002)
            try_sequence(a_opt, "fista_low", state)

        # FW-LP after FISTA (FISTA may find a better basin)
        rem = phase3_end - time.monotonic() - 5
        if rem > 15:
            t_alloc = min(rem, 80)
            print(f"  FW-LP after FISTA: {t_alloc:.0f}s")
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_alloc)
            try_sequence(a_opt, "fwl_post_fista", state)

    # === Phase 4: Final polishing (10% of budget) ===
    print(f"\n=== Phase 4: Final Polishing ===")
    phase4_end = deadline - 5

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Current best: {state['best_score']:.6f}")

        # Subgradient polish
        rem = phase4_end - time.monotonic() - 15
        if rem > 10:
            t_alloc = min(rem, 100)
            a_opt, obj = sg_polish(state['best_seq'], time_limit=t_alloc,
                                   lr=0.002, mom=0.94,
                                   top_k=max(10, n_best // 4), temp=0.00003)
            try_sequence(a_opt, "sg_polish", state)

        # Coordinate descent
        rem = phase4_end - time.monotonic() - 10
        if rem > 10:
            t_alloc = min(rem, 80)
            a_opt, obj = coordinate_descent_polish(state['best_seq'], time_limit=t_alloc)
            try_sequence(a_opt, "cd", state)

        # Final FISTA squeeze
        rem = phase4_end - time.monotonic() - 5
        if rem > 10:
            t_alloc = min(rem, 60)
            a_opt, obj = softmax_fista(state['best_seq'], time_limit=t_alloc,
                                       T_start=0.01, T_min=0.001)
            try_sequence(a_opt, "fista_final", state)

        # One last FW-LP
        rem = phase4_end - time.monotonic() - 2
        if rem > 10:
            t_alloc = min(rem, 40)
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_alloc)
            try_sequence(a_opt, "fwl_last", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(384) / 384
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback uniform: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.6f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
