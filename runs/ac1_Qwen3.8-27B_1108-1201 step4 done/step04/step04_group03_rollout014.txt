"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def peak_pair_cd(a, time_limit, verbose=True):
    """
    Peak-pair coordinate descent.
    At each step, find the peak, identify the coordinate that contributes most,
    and transfer mass from it to a coordinate that contributes least to that peak.
    """
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    delta = 1.0 / (n * 10)  # initial step size
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        k_star = int(np.argmax(conv))
        
        # Find coordinate i that contributes most to peak at k_star
        # Contribution of a[i] to conv[k_star] is:
        #   2*a[i]*a[k_star-i] if 0 <= k_star-i < n and k_star-i != i
        #   a[i]^2 if k_star == 2*i
        #   0 if k_star-i < 0 or k_star-i >= n
        k_i = k_star - np.arange(n)  # for each i, k_star - i
        valid = (k_i >= 0) & (k_i < n)
        contrib = np.zeros(n)
        diag = (np.arange(n) * 2 == k_star)
        contrib[valid & ~diag] = 2.0 * a[valid & ~diag] * a[k_i[valid & ~diag]]
        contrib[diag] = a[diag] ** 2
        
        # Pick the coordinate with largest contribution (and a[i] > 0)
        i_candidates = np.where((contrib > 0) & (a > 1e-15))[0]
        if len(i_candidates) == 0:
            break
        i = i_candidates[np.argmax(contrib[i_candidates])]
        
        # Pick coordinate m to receive mass: minimize contribution to peak
        # We want a[m] such that a[k_star-m] is small
        recv_contrib = np.zeros(n)
        valid_m = (k_i >= 0) & (k_i < n)
        recv_contrib[valid_m & ~diag] = a[k_i[valid_m & ~diag]]  # a[k-m]
        recv_contrib[diag] = a[diag]  # a[m] when k=2m
        
        # Prefer coordinates with small recv_contrib and small a[m]
        # Also avoid i itself
        m_scores = recv_contrib + 0.001 * a
        m_scores[i] = np.inf
        m = int(np.argmin(m_scores))
        
        # Determine step size: try to reduce the peak
        # Change in conv[k_star] ≈ 2*delta*(a[k-m] - a[k-i])
        # We want this negative, so a[k-m] < a[k-i]
        # Max delta before a[i] hits 0: a[i]
        max_delta = a[i] * 0.5  # don't go below half
        
        # Also check: what delta reduces the peak the most without increasing other peaks?
        # Simple approach: try a few deltas
        best_delta = 0
        best_val = M
        
        # Try geometric sequence of deltas
        for d in [delta, delta*0.5, delta*0.25, delta*2, delta*4]:
            d = min(d, max_delta)
            if d < 1e-15:
                continue
            a_try = a.copy()
            a_try[i] -= d
            a_try[m] += d
            if a_try[i] < 0:
                continue
            c_try = fast_convolve(a_try)
            v = c_try.max()
            if v < best_val:
                best_val = v
                best_delta = d
        
        if best_delta > 0:
            a[i] -= best_delta
            a[m] += best_delta
            a = np.maximum(a, 0)
            a = a / a.sum()
            
            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_a = a.copy()
                no_improve = 0
                delta = min(delta * 1.05, 1.0/(n*2))
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        # Adaptive: if stuck, try smaller delta or restart
        if no_improve > 50:
            delta *= 0.5
            if delta < 1e-10:
                # Perturb and restart
                rng = np.random.default_rng(hash((it, n)) % (2**32))
                a = project_simplex(best_a + rng.standard_normal(n) * 0.005)
                delta = 1.0 / (n * 10)
                no_improve = 0
        
        if no_improve > 500:
            break
    
    return best_a, best_obj

def subgradient_optimize(a0, time_limit, lr_init=0.01, mom=0.9, top_k_frac=0.1, verbose=True):
    """
    Subgradient method with momentum and adaptive step size.
    Uses softmax weighting over top-k peaks.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    v = np.zeros(n)
    it = 0
    temp = 0.001
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        
        top_k = max(5, int(n * top_k_frac))
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        
        pv = conv[peaks]
        scaled = pv / temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws
        
        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
        
        # Normalized subgradient (since we're on simplex, the gradient of the
        # objective w.r.t. the "free" direction is sg - mean(sg)*ones)
        sg_centered = sg - sg.mean()
        
        v = mom * v - cur_lr * sg_centered
        a_new = project_simplex(a + v)
        
        conv_new = fast_convolve(a_new)
        obj = conv_new.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            a = a_new
            no_improve = 0
            cur_lr = min(cur_lr * 1.01, 0.05)
        else:
            a = a_new  # still move, just don't update best
            no_improve += 1
            cur_lr *= 0.999
        
        temp *= 1.002
        
        if no_improve > 200:
            rng = np.random.default_rng(int(time.monotonic() * 100000) % (2**31))
            scale = 0.003 + 0.005 * min(1.0, no_improve / 400.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = max(lr_init * 0.5, 0.002)
            temp = 0.002
            no_improve = 0
        
        if no_improve > 1000:
            break
    
    return best_a, best_obj

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a novel multi-method approach:
1. Peak-pair coordinate descent (targets specific peaks by transferring mass)
2. Momentum subgradient with softmax peak weighting
3. Multiplicative weight (mirror descent) polish
4. Fine coordinate cycling
5. Multi-n exploration with interpolation from parent
Key innovation: peak-pair CD directly identifies which coordinate pairs
reduce the maximum peak, providing a much more targeted update than
generic subgradient steps.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def peak_pair_cd(a0, time_limit, seed=0):

        it += 1

        diag = (np.arange(n) * 2 == k_star)

        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    rng = np.random.default_rng(seed)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            log_update[idx] -= cur_lr * 2.0 * w[i] * a[k - idx]

        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()

        conv_new = fast_convolve(a)
        obj = conv_new.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= 0.9996
        cur_temp *= 1.003

        if no_improve > 120:
            a = best_a * np.exp(rng.standard_normal(n) * 0.008)
            a = np.maximum(a, 1e-30)
            a = a / a.sum()
            cur_lr = max(lr_init * 0.6, 0.005)
            cur_temp = temp_init * 1.5
            no_improve = 0

    return best_a, best_obj


def fine_coord_cycling(a0, time_limit, seed=0):
    """Coordinate cycling: try scaling each coordinate to reduce the max peak."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    rng = np.random.default_rng(seed)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    it = 0
    factors = [0.6, 0.8, 0.9, 1.0, 1.1, 1.25, 1.5]

    while time.monotonic() - t0 < time_limit:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            current_ai = a[i]
            if current_ai < 1e-12:
                continue

            best_val = best_obj
            best_factor = 1.0

            for f in factors:
                a_try = a.copy()
                a_try[i] = current_ai * f
                a_try = np.maximum(a_try, 0.0)
                st = a_try.sum()
                if st < 1e-15:
                    continue
                a_try = a_try / st
                c = fast_convolve(a_try)
                v = c.max()
                if v < best_val - 1e-15:
                    best_val = v
                    best_factor = f

            if best_factor != 1.0:
                a[i] = a[i] * best_factor
                a = np.maximum(a, 0.0)
                st = a.sum()
                if st > 1e-15:
                    a = a / st
                if best_val < best_obj - 1e-15:
                    best_obj = best_val
                    best_a = a.copy()
                    improved = True

        it += 1
        if not improved and it > 3:
            # Perturb and try again
            scale = 0.002 + 0.003 * min(1.0, it / 10.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
        if it > 50:
            break

    return best_a, best_obj


def aggressive_peak_cut(a0, time_limit, seed=0):
    """
    Directly attacks the highest peak by reducing all coordinates
    that feed into it proportionally, and redistributing to "safe" coordinates.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    rng = np.random.default_rng(seed)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()
        k_star = int(np.argmax(conv))

        # Find all coordinates contributing to this peak
        k_i = k_star - np.arange(n)
        valid = (k_i >= 0) & (k_i < n)
        diag = (np.arange(n) * 2 == k_star)

        # Contribution amount
        contrib = np.zeros(n)
        mask_nd = valid & ~diag
        contrib[mask_nd] = a[k_i[mask_nd]]  # the "partner" value
        contrib[diag] = a[diag]

        # Scale down contributing coordinates
        active = valid & (contrib > 1e-12) & (a > 1e-12)
        if not active.any():
            break

        # Reduce active coordinates proportionally to their contribution
        total_active_mass = a[active].sum()
        if total_active_mass < 1e-15:
            break

        # Choose a reduction fraction
        for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
            a_try = a.copy()
            a_try[active] *= (1.0 - frac)
            freed = a_try[active].sum() - a[active].sum() * 0  # compute freed mass
            freed = frac * total_active_mass

            # Distribute freed mass to coordinates NOT contributing to peak
            inactive = ~active
            if not inactive.any():
                continue
            inactive_mass = a[inactive].sum()
            if inactive_mass < 1e-15:
                # Distribute uniformly
                a_try[inactive] += freed / inactive.sum()
            else:
                # Distribute proportional to current values
                a_try[inactive] += freed * (a[inactive] / inactive_mass)

            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            c_try = fast_convolve(a_try)
            v = c_try.max()

            if v < M - 1e-14:
                a = a_try
                if v < best_obj - 1e-14:
                    best_obj = v
                    best_a = a.copy()
                no_improve = 0
                break
        else:
            no_improve += 1

        if no_improve > 100:
            scale = 0.003 + 0.005 * min(1.0, no_improve / 300.0)
            a = project_simplex(best_a + rng.standard_normal(n) * scale)
            no_improve = 0

        if no_improve > 600:
            break

    return best_a, best_obj


def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    x = np.arange(n, dtype=np.float64)
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.05
    inits.append(c1 / c1.sum())

    # Two-peak
    p1 = np.exp(-0.5 * ((x - 0.25 * (n - 1)) / (0.1 * n)) ** 2)
    p2 = np.exp(-0.5 * ((x - 0.75 * (n - 1)) / (0.1 * n)) ** 2)
    tp = p1 + p2
    inits.append(tp / tp.sum())

    # Interpolated parent
    if parent is not None and len(parent) > 1:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    # Symmetric random
    half_n = n // 2
    half = rng.dirichlet(np.ones(half_n) * 0.8)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:n] = half[::-1]
    sym = sym / sym.sum()
    inits.append(sym)

    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry. Multi-phase search combining novel peak-pair CD,
    aggressive peak cutting, momentum subgradient, multiplicative weights,
    and fine coordinate cycling, with multi-n exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 25.0)

    best_seq = None
    best_score = float('inf')

    def eval_and_update(a, label):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-15:
            return float('inf')
        a_norm = a / s
        score = float(evaluate_sequence(a_norm.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a_norm.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            ps = float(evaluate_sequence(parent.tolist()))
            print(f"Parent loaded: n={len(parent)}, score={ps:.8f}")
            eval_and_update(parent, "parent")
    except Exception as e:
        print(f"Could not load parent: {e}")

    if parent is None:
        parent = np.ones(384) / 384

    n_parent = len(parent)

    # === Phase 1: Peak-pair CD on parent (~250s) ===
    print("\n=== Phase 1: Peak-Pair CD ===")
    rem = deadline - time.monotonic()
    t1 = min(250, rem - 1500)
    if t1 > 30:
        print(f"  Peak-pair CD, t={t1:.0f}s")
        a_opt, _ = peak_pair_cd(parent, time_limit=t1, seed=seed)
        eval_and_update(a_opt, "ppcd")

    # === Phase 2: Aggressive peak cutting (~200s) ===
    print("\n=== Phase 2: Aggressive Peak Cut ===")
    base = best_seq if best_seq is not None else parent
    n_base = len(base)
    rem = deadline - time.monotonic()
    t2 = min(200, rem - 1300)
    if t2 > 30:
        print(f"  Peak cut, t={t2:.0f}s")
        a_opt, _ = aggressive_peak_cut(base, time_limit=t2, seed=seed + 1)
        eval_and_update(a_opt, "peakcut")

    # === Phase 3: Subgradient with multiple configs (~350s) ===
    print("\n=== Phase 3: Subgradient ===")
    rem = deadline - time.monotonic()
    t3_total = min(350, rem - 1000)
    if t3_total > 60:
        sg_configs = [
            (0.01, 0.9, 0.1),
            (0.005, 0.95, 0.05),
            (0.015, 0.88, 0.15),
            (0.008, 0.92, 0.08),
        ]
        t_per_sg = max(25, t3_total / len(sg_configs))
        for ci in range(len(sg_configs)):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(t_per_sg, rem - 900)
            if t_alloc < 15:
                break
            lr_i, mom_i, tk_f = sg_configs[ci]
            start_a = best_seq if best_seq is not None else parent
            print(f"  SG cfg{ci} (lr={lr_i}, mom={mom_i}), t={t_alloc:.0f}s")
            a_opt, _ = subgradient_opt(start_a, time_limit=t_alloc,
                                       lr_init=lr_i, mom=mom_i,
                                       top_k_frac=tk_f, seed=seed + 10 + ci)
            eval_and_update(a_opt, f"sg_{ci}")

    # === Phase 4: Multiplicative weight (~250s) ===
    print("\n=== Phase 4: MW ===")
    rem = deadline - time.monotonic()
    t4_total = min(250, rem - 700)
    if t4_total > 60:
        mw_configs = [
            (0.03, 10, 0.0003),
            (0.05, 20, 0.0005),
            (0.02, 12, 0.0002),
            (0.04, 15, 0.0006),
        ]
        t_per_mw = max(20, t4_total / len(mw_configs))
        for ci in range(len(mw_configs)):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            t_alloc = min(t_per_mw, rem - 600)
            if t_alloc < 15:
                break
            lr_i, tk_i, ti_i = mw_configs[ci]
            start_a = best_seq if best_seq is not None else parent
            print(f"  MW cfg{ci} (lr={lr_i}, k={tk_i}), t={t_alloc:.0f}s")
            a_opt, _ = mw_optimize(start_a, time_limit=t_alloc,
                                   lr_init=lr_i, top_k=tk_i,
                                   temp_init=ti_i, seed=seed + 20 + ci)
            eval_and_update(a_opt, f"mw_{ci}")

    # === Phase 5: Fine coordinate cycling (~150s) ===
    print("\n=== Phase 5: Fine Coord Cycling ===")
    if best_seq is not None:
        rem = deadline - time.monotonic()
        t5 = min(150, rem - 300)
        if t5 > 20:
            print(f"  Coord cycling, t={t5:.0f}s")
            a_opt, _ = fine_coord_cycling(best_seq, time_limit=t5, seed=seed + 30)
            eval_and_update(a_opt, "coord_cyc")

    # === Phase 6: Multi-n exploration (~250s) ===
    print("\n=== Phase 6: Multi-n ===")
    n_tries = [256, 512, 448, 640]
    for ni in n_tries:
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        t_alloc = min(80, rem - 150)
        if t_alloc < 25:
            break
        px = np.interp(np.linspace(0, 1, ni), np.linspace(0, 1, n_parent), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps < 1e-10:
            continue
        px = px / ps

        print(f"  n={ni}, t={t_alloc:.0f}s")
        # Subgradient
        t_sub = t_alloc * 0.5
        a_opt, _ = subgradient_opt(px, time_limit=t_sub, lr_init=0.01,
                                   mom=0.9, top_k_frac=0.1, seed=seed + 40 + ni)
        eval_and_update(a_opt, f"n{ni}_sg")

        # Peak-pair CD
        rem = deadline - time.monotonic()
        t_sub2 = min(30, rem - 100)
        if t_sub2 > 10:
            a_opt2, _ = peak_pair_cd(a_opt, time_limit=t_sub2, seed=seed + 50 + ni)
            eval_and_update(a_opt2, f"n{ni}_ppcd")

    # === Phase 7: Final combined polish (~150s) ===
    print("\n=== Phase 7: Final Polish ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic()
        if rem > 40:
            t_alloc = min(50, rem - 20)
            print(f"  Final MW, t={t_alloc:.0f}s")
            a_opt, _ = mw_optimize(best_seq, time_limit=t_alloc,
                                   lr_init=0.01, top_k=8, temp_init=0.0001, seed=seed + 60)
            eval_and_update(a_opt, "final_mw")

        rem = deadline - time.monotonic()
        if rem > 30:
            t_alloc = min(40, rem - 15)
            print(f"  Final PPCD, t={t_alloc:.0f}s")
            a_opt, _ = peak_pair_cd(best_seq, time_limit=t_alloc, seed=seed + 70)
            eval_and_update(a_opt, "final_ppcd")

        rem = deadline - time.monotonic()
        if rem > 40:
            t_alloc = min(40, rem - 15)
            print(f"  Final coord cyc, t={t_alloc:.0f}s")
            a_opt, _ = fine_coord_cycling(best_seq, time_limit=t_alloc, seed=seed + 80)
            eval_and_update(a_opt, "final_cyc")

        # Perturbation restarts
        for pi in range(4):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            noise = rng.standard_normal(n_best) * 0.003
            f_pert = project_simplex(best_seq + noise)
            t_alloc = min(20, rem - 10)
            if t_alloc < 10:
                break
            a_opt, _ = subgradient_opt(f_pert, time_limit=t_alloc, lr_init=0.005,
                                       mom=0.93, top_k_frac=0.1, seed=seed + 90 + pi)
            eval_and_update(a_opt, f"perturb_{pi}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
