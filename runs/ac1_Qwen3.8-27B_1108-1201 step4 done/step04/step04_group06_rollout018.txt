"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using a hybrid accelerated gradient + LP approach.
Key improvements: exponential peak weighting, backtracking line search, structured restarts,
and adaptive multi-phase optimization with length scanning.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def compute_weighted_gradient(a, top_k=20, sharpness=8.0):
    """Compute exponentially-weighted subgradient of the peak autocorrelation."""
    conv = fast_convolve(a)
    n = len(a)
    
    if top_k >= len(conv):
        top_idx = np.arange(len(conv))
    else:
        top_idx = np.argpartition(conv, -top_k)[-top_k:]
    
    max_val = conv[top_idx].max()
    if max_val < 1e-15:
        return np.zeros(n), conv
    
    diff = max_val - conv[top_idx]
    weights = np.exp(-sharpness * diff / max_val)
    w_sum = weights.sum()
    if w_sum < 1e-15:
        weights = np.zeros(len(top_idx))
        weights[np.argmax(conv[top_idx])] = 1.0
    else:
        weights /= w_sum
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        if weights[i] < 1e-12:
            continue
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        idxs = np.arange(j_start, j_end + 1)
        a_idx = k - idxs
        grad[idxs] += 2.0 * weights[i] * a[a_idx]
    
    return grad, conv


def build_conv_matrix(a):
    """Build matrix C such that C @ g = conv(a, g) (full linear convolution)."""
    n = len(a)
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        C[j:j + n, j] = a
    return C


def build_sparse_conv_matrix(a, top_lags):
    """Build a reduced matrix using only specific lag rows."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in top_lags:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    from scipy.sparse import csr_matrix
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return C


def lp_direction(a, conv, n, C=None):
    """Solve LP to find improvement direction. Returns (g, success)."""
    M = conv.max() * 1.0005
    if C is None:
        C = build_conv_matrix(a)
    
    from scipy.sparse import csr_matrix
    if not hasattr(C, 'toarray'):
        C_sp = C
    else:
        C_sp = C
    
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        result = linprog(c, A_ub=C_sp, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 10})
        if result.success and result.x is not None:
            g = np.maximum(result.x, 0)
            g_sum = g.sum()
            if g_sum > 1e-10:
                return g, True
    except Exception:
        pass
    return None, False

    lr = lr_init

        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 1e-15:
            y = y / sy
        
        # Compute gradient at lookahead point
        grad, _ = compute_weighted_gradient(y, top_k=top_k, sharpness=sharp)
        
        # Update velocity
        v = momentum * v - lr * grad
        
        # Line search with backtracking
        a_improved = False
        for alpha in [1.0, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1]:
            a_try = a + alpha * v
            a_try = np.maximum(a_try, 0.0)
            st = a_try.sum()
            if st < 1e-15:
                continue
            a_try = a_try / st
            conv_try = fast_convolve(a_try)
            obj_try = 2.0 * n * conv_try.max()
            if obj_try < cur_obj * (1.0 - 1e-9):
                a = a_try
                cur_obj = obj_try
                a_improved = True
                break
        
        if a_improved:
            if cur_obj < best_obj - 1e-10:
                best_obj = cur_obj
                best_a = a.copy()
            no_improve = 0
            lr = min(lr * 1.005, 0.05)
            sharp = min(sharp * 1.002, 30.0)
        else:
            no_improve += 1
            lr *= 0.75
            momentum = min(0.98, momentum + 0.002)
        
        iter_count += 1
    
    return best_a, best_obj


def lp_gradient_hybrid(a0, time_limit, verbose=False):
    """Alternate between LP direction finding and gradient descent steps."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = 2.0 * n * conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    perturb_count = 0
    C_cache = None
    C_invalid = False
    
    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 5:
            break
        
        # Try LP direction
        M = conv.max()
        peak_idx = int(np.argmax(conv))
        
        if not C_invalid:
            if C_cache is None:
                C_cache = build_conv_matrix(a)
            g, success = lp_direction(a, conv, n, C_cache)
        else:
            g, success = None, False
        
        if success and g is not None:
            d = g - a
            norm_d = np.linalg.norm(d)
            if norm_d > 1e-10:
                # Try several step sizes
                improved = False
                for t in [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03]:
                    a_try = a + t * d
                    a_try = np.maximum(a_try, 0.0)
                    st = a_try.sum()
                    if st < 1e-15:
                        continue
                    a_try = a_try / st
                    conv_try = fast_convolve(a_try)
                    obj_try = 2.0 * n * conv_try.max()
                    if obj_try < best_obj - 1e-10:
                        a = a_try
                        conv = conv_try
                        best_obj = obj_try
                        best_a = a.copy()
                        improved = True
                        C_invalid = True
                        break
                
                if improved:
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1
            C_invalid = True
        
        # If LP not improving, do gradient steps
        if no_improve >= 5 or no_improve == 1:
            # Short gradient burst
            t_grad = min(3.0, time_limit - (time.monotonic() - t0) - 2.0)
            if t_grad > 1.0:
                a_g, obj_g = nesterov_optimize(a, time_limit=t_grad, lr_init=0.005,
                                                top_k=40, sharpness=10.0)
                if obj_g < best_obj - 1e-10:
                    a = a_g
                    conv = fast_convolve(a)
                    best_obj = obj_g
                    best_a = a.copy()
                    no_improve = 0
                    C_invalid = True
        
        # Handle stagnation with structured restart
        if no_improve >= 40:
            rng = np.random.default_rng(int(time.monotonic() * 1000) + perturb_count)
            
            # Structured perturbation: anti-symmetric component
            scale = 0.003 + 0.004 * min(perturb_count, 10)
            x = np.arange(n)
            antisym = (x - 0.5 * (n - 1)) / n
            noise = rng.standard_normal(n) * scale * 0.5
            a = best_a + scale * antisym + noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-15:
                a = best_a.copy()
            else:
                a = a / s
            conv = fast_convolve(a)
            no_improve = 0
            perturb_count += 1
            C_invalid = True
            
            if perturb_count > 30:
                break
        
        if verbose and (no_improve % 20 == 0):
            print(f"    HYB obj={best_obj:.8f}, t={time.monotonic()-t0:.1f}s, perturb={perturb_count}")
    
    return best_a, best_obj


def make_initializations(n, parent=None, rng=None):
    """Generate diverse initializations."""
    if rng is None:
        rng = np.random.default_rng(42)
    
    inits = {}
    x = np.arange(n, dtype=np.float64)
    
    inits['uniform'] = np.ones(n) / n
    
    # Slightly tilted uniform (breaks symmetry)
    tilt = 1.0 + 0.3 * (2.0 * x / max(n - 1, 1) - 1.0)
    inits['tilted'] = tilt / tilt.sum()
    
    # Sine-based
    sine = np.sin(np.pi * x / n) ** 2 + 0.05
    inits['sine'] = sine / sine.sum()
    
    # Double sine
    s2 = np.sin(np.pi * x / n) ** 2 * np.sin(2 * np.pi * x / n) ** 2 + 0.05
    inits['sine2'] = s2 / s2.sum()
    
    # U-shaped (edge emphasis)
    u = ((2.0 * x / max(n - 1, 1) - 1.0) ** 4 + 0.1)
    inits['ushape'] = u / u.sum()
    
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2) + 0.01
    inits['gaussian'] = g / g.sum()
    
    # Random
    inits['random'] = rng.dirichlet(np.ones(n) * 0.5)
    
    # From parent
    if parent is not None:
        p = np.maximum(parent.astype(np.float64), 0.0)
        ps = p.sum()
        if ps > 1e-10:
            p = p / ps
            if len(p) == n:
                inits['parent'] = p
            else:
                pp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pp = np.maximum(pp, 0.0)
                pps = pp.sum()
                if pps > 1e-10:
                    inits['parent_interp'] = pp / pps
            
            # Parent + small antisymmetric perturbation
            antisym = (x - 0.5 * (n - 1)) / n
            p_pert = p + 0.01 * antisym
            p_pert = np.maximum(p_pert, 0.0)
            pps = p_pert.sum()
            if pps > 1e-10:
                inits['parent_asym'] = p_pert / pps
    
    return inits


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main search: multi-phase optimization combining gradient descent and LP methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_sequence = None
    best_score = float('inf')
    
    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p
            print(f"Parent loaded: n={len(p)}, sum={p.sum():.4f}")
    except Exception:
        pass
    
    def update_best(a, label=""):
        nonlocal best_score, best_sequence
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_sequence = a.copy()
            print(f"  {label}: {score:.6f} *NEW BEST*")
        return score
    
    # =========================================================
    # Phase 1: Fast gradient optimization from multiple starts
    # =========================================================
    print("=== Phase 1: Multi-start Gradient Optimization ===")
    t_start = time.monotonic()
    phase1_budget = 250.0
    
    if parent is not None:
        base_len = len(parent)
    else:
        base_len = 768
    
    lengths_to_try = []
    if parent is not None:
        lengths_to_try = [base_len]
        # Also try slightly different lengths
        for dl in [-64, 64, -128, 128]:
            nl = base_len + dl
            if 128 <= nl <= 2048:
                lengths_to_try.append(nl)
    else:
        lengths_to_try = [512, 768, 1024]
    
    # Deduplicate
    lengths_to_try = list(dict.fromkeys(lengths_to_try))
    
    for n in lengths_to_try[:4]:
        remaining = deadline - time.monotonic() - 400.0
        if remaining < 30:
            break
        
        inits = make_initializations(n, parent, rng)
        
        # Priority: parent-based first, then diverse
        priority = ['parent', 'parent_asym', 'parent_interp', 'tilted', 'sine', 'random']
        init_names = [nm for nm in priority if nm in inits]
        init_names += [nm for nm in inits if nm not in priority]
        init_names = init_names[:4]
        
        t_per_init = max(15.0, remaining * 0.7 / max(1, len(lengths_to_try[:4]) * len(init_names)))
        
        for name in init_names:
            remaining = deadline - time.monotonic() - 350.0
            if remaining < 10:
                break
            t_alloc = min(t_per_init, remaining * 0.6)
            if t_alloc < 5:
                break
            
            a0 = inits[name]
            a_opt, obj = nesterov_optimize(a0, time_limit=t_alloc, lr_init=0.008,
                                            momentum_init=0.9, top_k=30, sharpness=8.0)
            score = update_best(a_opt, f"Grad n={n} {name}")
            print(f"    n={n} {name}: obj={obj:.6f}, t={time.monotonic()-t_start:.0f}s")
    
    print(f"Phase 1 done ({time.monotonic()-t_start:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 2: Deep hybrid LP + gradient optimization
    # =========================================================
    print("=== Phase 2: Deep Hybrid Optimization ===")
    t_start2 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 200.0
        n_best = len(best_sequence)
        
        # Run hybrid optimization with multiple configurations
        configs = [
            {'lr_init': 0.006, 'top_k': 40, 'sharpness': 10.0},
            {'lr_init': 0.01, 'top_k': 20, 'sharpness': 6.0},
            {'lr_init': 0.004, 'top_k': 50, 'sharpness': 12.0},
        ]
        
        for ci, cfg in enumerate(configs):
            remaining = deadline - time.monotonic() - 100.0
            if remaining < 30:
                break
            t_alloc = min(remaining * 0.35, 400.0)
            if t_alloc < 20:
                break
            
            print(f"  Hybrid run {ci}: n={n_best}, t_alloc={t_alloc:.0f}s")
            
            # Start from best or slightly perturbed
            if ci == 0:
                a0 = best_sequence.copy()
            elif ci == 1:
                scale = 0.005
                noise = rng.standard_normal(n_best) * scale
                a0 = best_sequence + noise
                a0 = np.maximum(a0, 0.0)
                a0 = a0 / a0.sum()
            else:
                # Anti-symmetric perturbation
                x = np.arange(n_best)
                antisym = (x - 0.5 * (n_best - 1)) / n_best
                a0 = best_sequence + 0.008 * antisym
                a0 = np.maximum(a0, 0.0)
                a0 = a0 / a0.sum()
            
            a_opt, obj = lp_gradient_hybrid(a0, time_limit=t_alloc, verbose=True)
            update_best(a_opt, f"Hybrid {ci}")
    
    print(f"Phase 2 done ({time.monotonic()-t_start2:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 3: Focused LP optimization
    # =========================================================
    print("=== Phase 3: Focused LP Optimization ===")
    t_start3 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 80.0
        n_best = len(best_sequence)
        
        if remaining > 50:
            # Dedicated LP run
            t_alloc = min(remaining * 0.7, 300.0)
            print(f"  LP focus: n={n_best}, t_alloc={t_alloc:.0f}s")
            a_opt, obj = lp_gradient_hybrid(best_sequence, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "LP Focus")
    
    print(f"Phase 3 done ({time.monotonic()-t_start3:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 4: Fine-grained gradient polishing
    # =========================================================
    print("=== Phase 4: Fine Polishing ===")
    t_start4 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 30.0
        
        # Multiple short gradient runs with different parameters
        polish_configs = [
            {'lr_init': 0.003, 'top_k': 60, 'sharpness': 15.0},
            {'lr_init': 0.002, 'top_k': 100, 'sharpness': 20.0},
            {'lr_init': 0.005, 'top_k': 30, 'sharpness': 10.0},
            {'lr_init': 0.001, 'top_k': 150, 'sharpness': 25.0},
        ]
        
        for pi, cfg in enumerate(polish_configs):
            remaining = deadline - time.monotonic() - 20.0
            if remaining < 5:
                break
            t_alloc = min(remaining * 0.25, 40.0)
            if t_alloc < 3:
                break
            
            a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_alloc, **cfg)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
                print(f"  Polish {pi}: {score:.6f} *NEW BEST*")
    
    print(f"Phase 4 done ({time.monotonic()-t_start4:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 5: Try different length (if time allows)
    # =========================================================
    print("=== Phase 5: Length Variants ===")
    t_start5 = time.monotonic()
    
    if best_sequence is not None:
        remaining = deadline - time.monotonic() - 20.0
        n_best = len(best_sequence)
        
        if remaining > 60:
            # Try extending or shrinking
            alt_lengths = [n_best * 3 // 2, n_best * 2, n_best // 2]
            alt_lengths = [l for l in alt_lengths if 128 <= l <= 4096 and l != n_best]
            
            for alt_n in alt_lengths[:2]:
                remaining = deadline - time.monotonic() - 15.0
                if remaining < 20:
                    break
                t_alloc = min(remaining * 0.5, 60.0)
                if t_alloc < 10:
                    break
                
                # Interpolate best sequence to new length
                a_interp = np.interp(np.linspace(0, 1, alt_n),
                                     np.linspace(0, 1, n_best),
                                     best_sequence / best_sequence.sum())
                a_interp = np.maximum(a_interp, 0.0)
                a_interp = a_interp / a_interp.sum()
                
                a_opt, obj = nesterov_optimize(a_interp, time_limit=t_alloc,
                                                lr_init=0.006, top_k=30, sharpness=8.0)
                score = float(evaluate_sequence(a_opt.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_sequence = a_opt.copy()
                    print(f"  Alt len {alt_n}: {score:.6f} *NEW BEST*")
                else:
                    print(f"  Alt len {alt_n}: {score:.6f}")
    
    print(f"Phase 5 done ({time.monotonic()-t_start5:.0f}s). Best: {best_score:.6f}")
    
    # =========================================================
    # Phase 6: Final micro-polish
    # =========================================================
    if best_sequence is not None and time.monotonic() < deadline - 5:
        t_final = min(15.0, deadline - time.monotonic() - 3.0)
        if t_final > 3:
            a_opt, obj = nesterov_optimize(best_sequence, time_limit=t_final,
                                            lr_init=0.002, top_k=80, sharpness=20.0)
            score = float(evaluate_sequence(a_opt.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_sequence = a_opt.copy()
    
    # Fallback
    if best_sequence is None:
        best_sequence = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_sequence.tolist()))
    
    # Final verification
    final_score = float(evaluate_sequence(best_sequence.tolist()))
    if np.isfinite(final_score):
        best_score = final_score
    
    print(f"\nFINAL: score={best_score:.6f}, n={len(best_sequence)}")
    return [float(x) for x in best_sequence]
