"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Smooth log-exp (log-sum-exp) gradient descent with adaptive temperature
2. Coordinate descent with golden-section 1D sub-optimization  
3. Nesterov momentum for accelerated convergence
4. Multi-phase strategy: global search → local polish → fine-tuning
Key innovation: the log-exp smoothing converts the non-differentiable max
into a smooth function whose gradient can be computed via FFT cross-correlation,
enabling efficient O(n log n) gradient evaluations.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_smooth_gradient(f, tau):
    """
    Compute gradient of log(LSE(f*f / tau)) - 2*log(sum(f))
    Returns (gradient, autocorrelation).
    """
    n = len(f)
    c = fast_convolve(f)
    # Softmax with temperature
    shifted = c / tau
    shifted -= shifted.max()
    p = np.exp(shifted)
    p_sum = p.sum()
    if p_sum < 1e-300:
        p = np.zeros_like(c)
        p[c.argmax()] = 1.0
    else:
        p /= p_sum
    
    # Cross-correlation: g(j) = sum_k p(k) * f(k-j) for 0<=k-j<n
    # = (f * p_reversed type correlation)
    # Using: (f ⋆ p)(t) = sum_m f(m) p(m+t) = ifft(FFT(p) * conj(FFT(f)))[t]
    fft_size = 1 << (2 * n - 1 - 1).bit_length()
    F = np.fft.rfft(f, fft_size)
    P = np.fft.rfft(c * 0 + p, fft_size)  # p padded to fft_size
    corr = np.fft.irfft(P * np.conj(F), fft_size)
    g = corr[:n]
    
    sum_f = f.sum()
    grad = (2.0 / tau) * g - 2.0 / sum_f
    return grad, c


def smooth_gradient_optimize(f0, time_limit, tau_init=0.005, tau_min=0.0001,
                              lr_init=0.01, nesterov=True, verbose=True):
    """
    Smooth gradient descent with log-exp approximation and adaptive temperature.
    Uses Nesterov momentum and periodic temperature annealing.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    v = np.zeros(n)  # momentum
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    tau = tau_init
    lr = lr_init
    no_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute gradient at current point (or lookahead for Nesterov)
        if nesterov:
            # Lookahead
            y = f + v
            y = project_simplex(y)
            grad, c = compute_smooth_gradient(y, tau)
        else:
            grad, c = compute_smooth_gradient(f, tau)
        
        # Gradient step
        f_new = f - lr * grad
        f_new = project_simplex(f_new)
        
        # Nesterov momentum update
        if nesterov:
            v = (it / (it + 3)) * (v + (f_new - f))
        else:
            v = 0.5 * v + 0.5 * (f_new - f)
        
        f = f_new
        
        # Evaluate
        obj = fast_convolve(f).max()
        
        if obj < best_obj - 1e-16:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive temperature: decrease when making progress, increase when stuck
        if no_improve == 0:
            tau = max(tau * 0.995, tau_min)
            lr = max(lr * 0.998, 0.0001)
        elif no_improve > 50:
            tau = min(tau * 1.02, tau_init)
            lr = min(lr * 1.01, lr_init)
        
        # Periodic restart
        if no_improve > 300:
            rng = np.random.default_rng(int(time.monotonic() * 10000) % 1000000)
            f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
            v = np.zeros(n)
            tau = tau_init * 0.5
            lr = lr_init * 0.5
            no_improve = 0
        
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - t0
            print(f"    Smooth GD: it={it}, obj={obj:.10f}, tau={tau:.6f}, t={elapsed:.0f}s")
    
    return best_f, best_obj


def coordinate_descent(f0, time_limit, n_sweeps_max=100, verbose=True):
    """
    Coordinate descent: for each coordinate j, find optimal f_j via golden section.
    Normalizes after each coordinate update.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)
    
    best_obj = fast_convolve(f).max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    
    # Golden section ratio
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    sweep = 0
    while time.monotonic() - t0 < time_limit:
        sweep += 1
        sweep_improved = False
        
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            
            old_val = f[j]
            
            # Define 1D objective: F(t) = max_k (f_new * f_new)(k)
            # where f_new = [f[0..j-1], t, f[j+1..]] / (sum - f[j] + t)
            def obj_1d(t):
                if t < 0:
                    t = 0.0
                f_new = f.copy()
                f_new[j] = t
                s_new = f_new.sum()
                if s_new < 1e-15:
                    return 1e10
                f_new = f_new / s_new
                return fast_convolve(f_new).max()
            
            # Golden section search on [0, 1]
            lo = 0.0
            hi = 1.0
            a_ = lo + gr * (hi - lo)
            b_ = hi - gr * (hi - lo)
            fa = obj_1d(a_)
            fb = obj_1d(b_)
            
            for _ in range(30):  # ~30 iterations gives precision ~ (0.618)^30 ≈ 3e-7
                if fa < fb:
                    hi = b_
                    b_ = a_
                    fb = fa
                    a_ = lo + gr * (hi - lo)
                    fa = obj_1d(a_)
                else:
                    lo = a_
                    a_ = b_
                    fa = fb
                    b_ = hi - gr * (hi - lo)
                    fb = obj_1d(b_)
            
            t_opt = (lo + hi) / 2.0
            f_new = f.copy()
            f_new[j] = t_opt
            s_new = f_new.sum()
            if s_new > 1e-15:
                f_new = f_new / s_new
                new_obj = fast_convolve(f_new).max()
                if new_obj < fast_convolve(f).max() - 1e-18:
                    f = f_new
                    sweep_improved = True
        
        new_obj = fast_convolve(f).max()
        if new_obj < best_obj - 1e-16:
            best_obj = new_obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and sweep % 5 == 0:
