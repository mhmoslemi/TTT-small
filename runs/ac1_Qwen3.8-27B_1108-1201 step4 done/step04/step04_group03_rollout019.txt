Input: initial sequence a (non-negative, sum = 1), time budget T
Output: optimized sequence

1. epsilon = initial smoothing (e.g., 0.01 / max(b))

3. Return a

If `corr = ifft(FFT(A) * conj(FFT(W)))`, then:
`corr[m] = sum_k A[k] * W[(k+m) mod fft_size]`

With zero-padding (fft_size >= n + 2n-2 = 3n-2), the circular correlation equals the linear correlation:
`corr[m] = sum_k A[k] * W[k+m]` (where W is zero outside [0, 2n-2])

So `corr[j] = sum_k A[k] * W[k+j]` for j = 0, ..., n-1.

And we want `(grad)_j = 2 * sum_k W[k] * A[k-j]` (with A zero outside [0,n-1]).

Let me substitute: let `l = k-j`, so `k = l+j`:
`(grad)_j = 2 * sum_l W[l+j] * A[l]` (where l ranges over valid indices)
$= 2 \sum_l A[l] \cdot W[l+j]$
$= 2 \cdot \text{corr}[j]$

So `grad = 2 * corr[:n]`.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Log-sum-exp smoothed gradient descent with Nesterov momentum
2. FFT-based fast gradient computation
3. Adaptive epsilon (smoothing) annealing
4. Multiple diverse initializations
5. Frank-Wolfe with LP oracle for final polish
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def solve_lp(f, M, n):
    """LP oracle: max sum(g) s.t. (f*g)[k] <= M, g >= 0."""
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

def smoothed_grad_descent(a0, time_limit, epsilon_ratio=0.1, verbose=False):
    """
    Main optimization: log-sum-exp smoothed gradient descent with momentum.
    a0: initial sequence (will be normalized to sum=1)
    Returns: (best_sequence, best_max_conv)
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / S
    n = len(a)
    
    # FFT size for cross-correlation
    conv_size = 2 * n - 1
    fft_size = 1 << (conv_size + n - 2 - 1).bit_length()  # need at least (2n-1) + n - 1
    
    # Initial b and epsilon
    b = fast_convolve(a)
    b_range = b.max() - b.min()
    if b_range < 1e-15:
        b_range = b.max() * 0.01 + 1e-10
    epsilon = b_range * epsilon_ratio
    
    best_obj = b.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    eta = 0.05  # initial step size
    mu = 0.85  # momentum
    momentum = np.zeros(n)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Compute autocorrelation
        A_fft = np.fft.rfft(a, fft_size)
        b_full = np.fft.irfft(A_fft * A_fft, fft_size)[:conv_size]
        cur_max = b_full.max()
        
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_a = a.copy()
            no_improve = 0
            if verbose and it % 1000 == 0:
                print(f"    it={it}, obj={best_obj:.10f}, eps={epsilon:.2e}, eta={eta:.4f}")
        else:
            no_improve += 1
        
        # Compute smoothed max weights (log-sum-exp)
        b_shifted = b_full - b_full.max()
        # Clip to avoid overflow
        b_shifted = np.clip(b_shifted, -50 * epsilon, 0)
        w = np.exp(b_full / epsilon - b_full.max() / epsilon)  # This is exp((b_full - max)/eps)
        w_sum = w.sum()
        if w_sum < 1e-300 or not np.isfinite(w_sum):
            w = np.zeros(conv_size)
            w[np.argmax(b_full)] = 1.0
        else:
            w /= w_sum

        # grad[j] = 2 * sum_k w[k] * a[k-j] for valid k-j
        # = 2 * cross_corr(a, w)[j]

        A_ext = np.zeros(fft_size)

        W_ext = np.zeros(fft_size)

        direction = momentum

        if dir_norm < 1e-15:

        a_trial = a.copy()

            obj_trial = b_trial.max()
            
            if obj_trial < cur_max - 1e-18:

def generate_inits(n, parent, rng):

    x = np.arange(n, dtype=np.float64)
    exp_r = np.exp(-x / (n * 0.3))
    inits.append(exp_r / exp_r.sum())
    
    # Exponential (other side)
    exp_l = np.exp(-x / (n * 0.3))[::-1]
    inits.append(exp_l / exp_l.sum())
    
    # Two bumps
    center1 = n * 0.25
    center2 = n * 0.75
    two_bump = np.exp(-0.5 * ((x - center1) / (n * 0.1))**2) + \
               np.exp(-0.5 * ((x - center2) / (n * 0.1))**2)
    inits.append(two_bump / two_bump.sum())
    
    # Cosine
    cos_w = np.abs(np.cos(np.pi * x / n)) + 0.1
    inits.append(cos_w / cos_w.sum())
    
    # Sine (peak in middle)
    sin_w = np.sin(np.pi * (x + 0.5) / n) + 0.05
    inits.append(sin_w / sin_w.sum())
    
    # Linear ramp
    ramp = (x + 1.0)
    inits.append(ramp / ramp.sum())
    
    # Parent interpolated
    if parent is not None and len(parent) > 1:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

def fast_convolve(a):
    """Full linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
