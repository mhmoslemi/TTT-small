    """Find optimal a_j given all other coordinates fixed, using normalization."""
    x_old = a[j]
    s = 1 - x_old  # sum of others
    if s < 1e-15:
        return x_old
    
    # b = a with j-th element removed, normalized
    b = a.copy()
    b[j] = 0
    b = b / s  # now sum(b) = 1
    
    # Precompute B = b*b (convolution of b with itself)
    B = fast_convolve(b)
    
    # For each k, f_k(x) = (1-x)^2 * B[k] + 2x(1-x) * b[k-j] + x^2 * [k==2j]
    # We want to minimize max_k f_k(x) over x in [0, 1]
    
    # Bisection on x
    lo, hi = 0.0, 1.0
    
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        # Compute subgradient at mid
        # f_k'(mid) = 2 * (b[k-j] - B[k] + mid * (B[k] - 2*b[k-j] + [k==2j]))
        # We need the max of f_k(mid) and its subgradient

        # f_k(mid) = (1-mid)^2 * B[k] + 2*mid*(1-mid) * b_valid[k] + mid^2 * [k==2j]
        # where b_valid[k] = b[k-j] if 0 <= k-j < n else 0
        
        b_valid = np.zeros(2*n-1)
        if j > 0:
            b_valid[:n] = b[:n]  # b[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
            # Actually b[k-j] is non-zero when 0 <= k-j < n, i.e., j <= k <= j+n-1
            b_valid[j:j+n] = b

        if 2*j < 2*n-1:
            f_vals[2*j] += mid**2
        
        # Find which k achieves the max
        k_max = np.argmax(f_vals)
        
        # Subgradient direction at mid: sign of f_{k_max}'(mid)

        b_kj = b[k-j] if 0 <= k-j < n else 0.0

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected subgradient descent
with multi-peak weighting, momentum, and exact coordinate bisection polishing.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp


def compute_subgradient(a, conv, top_k=20, temp=0.001):
    """Compute weighted subgradient from top peaks."""
    n = len(a)
    conv_len = len(conv)
    
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]
    
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        # sg_j += 2 * w[i] * a[k-j] for valid j
        # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]
    
    return sg

def coordinate_bisection(a0, time_limit, verbose=False, tol=1e-10):

    a = a0.copy().astype(np.float64)

    else:
        a = a / s

    best_obj = conv.max()

        passes += 1
        pass_improved = False

            if time.monotonic() - t0 > time_limit:
                break

            if x_old < 1e-14:

            if s_others < 1e-12:
                continue
            
            # b = a with j removed, normalized
            b = a.copy()
            b[j] = 0
            b = b / s_others
            
            # B = b * b
            B = fast_convolve(b)
            
            # b_valid[k] = b[k-j] if 0 <= k-j < n, else 0
            b_valid = np.zeros(2 * n - 1)
            if 0 <= j < 2 * n - 1:
                start = j
                end = min(j + n, 2 * n - 1)
                b_valid[start:end] = b[:end - start]
            
            # indicator for k = 2j
            has_2j = (2 * j < 2 * n - 1)
            
            # Bisection on x in [0, 1]
            lo, hi = 0.0, 1.0
            for _ in range(50):
                mid = (lo + hi) / 2
                one_minus_mid = 1 - mid
                
                # f_k(mid) = (1-mid)^2 * B[k] + 2*mid*(1-mid) * b_valid[k] + mid^2 * [k==2j]
                f_vals = one_minus_mid**2 * B + 2 * mid * one_minus_mid * b_valid
                if has_2j:
                    f_vals[2 * j] += mid**2
                
                k_max = int(np.argmax(f_vals))
                
                # Derivative of f_{k_max} at mid
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (has_2j and k == 2 * j) else 0.0
                
                deriv = 2 * (b_kj - B_k + mid * (B_k - 2 * b_kj + ind))
                
                if deriv > 1e-12:
                    hi = mid
                elif deriv < -1e-12:
                    lo = mid
                else:
                    break
            
            x_new = (lo + hi) / 2
            
            if abs(x_new - x_old) < 1e-12:
                continue
            
            # Apply update
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1 - x_new) / s_others
            # Ensure numerical safety
            a = np.maximum(a, 0)
            s_check = a.sum()
            if s_check > 0:

            obj = conv.max()

    return best_a, best_obj

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using projected multi-peak subgradient
descent with momentum, exact coordinate bisection polishing, adaptive step sizes,
noise injection for exploration, and multi-length search.
"""

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def interpolate_to_length(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    conv_len = len(conv)
    k_actual = min(top_k, conv_len)
    peaks = np.argpartition(conv, -k_actual)[-k_actual:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-15:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        if w[i] < 1e-12:
            continue
        k = int(peaks[i])
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * a[k - idx]

    return sg


def subgradient_phase(a0, time_limit, lr=0.01, mom=0.92, top_k=30,
                      temp=0.0005, decay=0.9998, verbose=False):
    """Projected subgradient descent with momentum and adaptive temperature."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    v = np.zeros(n)
    cur_lr = lr
    cur_temp = temp
    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        sg = compute_multi_peak_subgrad(a, conv, top_k=top_k, temp=cur_temp)
        v = mom * v - cur_lr * sg
        a_new = project_simplex(a + v)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-14:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        conv = conv_new
        cur_lr *= decay
        cur_temp *= 1.003

        if no_improve > 300:
            rng = np.random.default_rng(it * 13 + 5)
            noise_scale = 0.002 + 0.002 * min(no_improve // 200, 3)
            a = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            conv = fast_convolve(a)
            v = np.zeros(n)
            cur_lr = lr * 0.6
            no_improve = 0

        if verbose and it % 500 == 0:
            print(f"    SG iter {it}: obj={best_obj:.10f}, lr={cur_lr:.6f}")

    return best_a, best_obj


def coordinate_bisection(a0, time_limit, verbose=False, max_passes=500):
    """Exact coordinate descent with bisection per coordinate."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    passes = 0

    while time.monotonic() - t0 < time_limit and passes < max_passes:
        passes += 1
        pass_improved = False
        max_improvement = 0.0

        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break

            x_old = a[j]
            if x_old < 1e-14:
                continue

            s_others = 1.0 - x_old
            if s_others < 1e-12:
                continue

            # Build b = a with a_j removed, normalized
            b = a.copy()
            b[j] = 0.0
            b = b / s_others

            # B = b * b via FFT
            B = fast_convolve(b)
            conv_len = 2 * n - 1

            # b_valid[k] = b[k-j] if 0 <= k-j < n, else 0
            b_valid = np.zeros(conv_len)
            start = j
            end = min(j + n, conv_len)
            b_valid[start:end] = b[:end - start]

            # Bisection on x in [0, 1-x_old*0.5, x_old*1.5] clipped to [0,1]
            # Actually just [0, 1] is fine since sum constraint handles it
            lo = 0.0
            hi = 1.0
            x_new = x_old

            for _ in range(45):
                mid = (lo + hi) / 2.0
                om = 1.0 - mid

                # f_k(mid) = om^2 * B[k] + 2*mid*om * b_valid[k] + mid^2 * [k==2j]
                f_vals = om * om * B + 2.0 * mid * om * b_valid
                if 2 * j < conv_len:
                    f_vals[2 * j] += mid * mid

                k_max = int(np.argmax(f_vals))

                # Derivative of f_{k_max} at mid
                k = k_max
                b_kj = b[k - j] if 0 <= k - j < n else 0.0
                B_k = B[k]
                ind = 1.0 if (2 * j < conv_len and k == 2 * j) else 0.0

                deriv = 2.0 * (b_kj - B_k + mid * (B_k - 2.0 * b_kj + ind))

                if deriv > 1e-13:
                    hi = mid
                    x_new = mid
                elif deriv < -1e-13:
                    lo = mid
                    x_new = mid
                else:
                    x_new = mid
                    break

            if abs(x_new - x_old) < 1e-14:
                continue

            # Apply update: set a[j] = x_new, rescale others proportionally
            a[j] = x_new
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            a[mask] *= (1.0 - x_new) / s_others
            a = np.maximum(a, 0.0)
            s_check = a.sum()
            if s_check > 1e-15:
                a /= s_check
            else:
                a = a0.copy()
                continue

            conv = fast_convolve(a)
            obj = conv.max()
            improvement = best_obj - obj
            if improvement > max_improvement:
                max_improvement = improvement

            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a.copy()
                pass_improved = True
                if verbose and passes <= 3:
                    print(f"    CD pass {passes}, coord {j}: obj={best_obj:.10f}")

        if not pass_improved:
            # Add tiny noise to escape
            if passes > 2:
                rng = np.random.default_rng(passes * 47 + 11)
                a = project_simplex(best_a + rng.standard_normal(n) * 0.0004)
            else:
                break

    return best_a, best_obj


def pair_swap_polish(a0, time_limit, verbose=False):
    """Try pair swaps: transfer mass between coordinate pairs."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)

    while time.monotonic() - t0 < time_limit:
        # Pick a random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        x_i = a[i]
        x_j = a[j]
        if x_i < 1e-12 or x_j < 1e-12:
            continue

        # Try transferring some mass from i to j (and vice versa)
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:
            # Transfer frac * x_i from i to j
            delta = frac * x_i
            a_trial = a.copy()
            a_trial[i] = x_i - delta
            a_trial[j] = x_j + delta
            a_trial = np.maximum(a_trial, 0.0)
            st = a_trial.sum()
            if st < 1e-12:
                continue
            a_trial /= st

            obj = fast_convolve(a_trial).max()
            if obj < best_obj - 1e-12:
                best_obj = obj
                best_a = a_trial.copy()
                a = a_trial.copy()
                conv = fast_convolve(a)

            # Transfer frac * x_j from j to i
            delta2 = frac * x_j
            a_trial2 = a.copy()
            a_trial2[j] = x_j - delta2
            a_trial2[i] = x_i + delta2
            a_trial2 = np.maximum(a_trial2, 0.0)
            st2 = a_trial2.sum()
            if st2 < 1e-12:
                continue
            a_trial2 /= st2

            obj2 = fast_convolve(a_trial2).max()
            if obj2 < best_obj - 1e-12:
                best_obj = obj2
                best_a = a_trial2.copy()
                a = a_trial2.copy()
                conv = fast_convolve(a)

    return best_a, best_obj


def combined_optimize(a0, time_limit, verbose=False):
    """Alternate subgradient, coordinate bisection, and pair swap phases."""
    a = a0.copy().astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s

    n = len(a)
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    t0 = time.monotonic()

    phase = 0
    while time.monotonic() - t0 < time_limit:
        remaining = time_limit - (time.monotonic() - t0)
        if remaining < 5:
            break

        if phase % 4 == 0:
            # Subgradient phase
            t_alloc = min(40, remaining)
            lr = 0.008 / (1.0 + 0.15 * (phase // 4))
            top_k = max(20, n // 8)
            temp = 0.0003 + 0.0002 * (phase // 4)
            a_opt, obj = subgradient_phase(
                best_a, time_limit=t_alloc, lr=lr, mom=0.92,
                top_k=top_k, temp=temp, decay=0.9997
            )
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} SG: obj={best_obj:.10f}")

        elif phase % 4 == 1:
            # Coordinate bisection phase
            t_alloc = min(60, remaining)
            a_opt, obj = coordinate_bisection(best_a, time_limit=t_alloc, verbose=False)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} CD: obj={best_obj:.10f}")

        elif phase % 4 == 2:
            # Pair swap phase
            t_alloc = min(20, remaining)
            a_opt, obj = pair_swap_polish(best_a, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} PS: obj={best_obj:.10f}")

        else:
            # Noise + quick CD
            rng = np.random.default_rng(phase * 101 + 7)
            noise_scale = 0.003 * max(0.5, 1.0 - phase * 0.05)
            a_noisy = project_simplex(best_a + rng.standard_normal(n) * noise_scale)
            t_alloc = min(30, remaining)
            a_opt, obj = coordinate_bisection(a_noisy, time_limit=t_alloc)
            if obj < best_obj - 1e-14:
                best_obj = obj
                best_a = a_opt
                if verbose:
                    print(f"  Phase {phase} Noise+CD: obj={best_obj:.10f}")

        phase += 1

    return best_a, best_obj


def symmetrize(a):
    """Make sequence symmetric: a[i] = (a[i] + a[n-1-i]) / 2, then renormalize."""
    n = len(a)
    a_sym = 0.5 * (a + a[::-1])
    a_sym = np.maximum(a_sym, 0.0)
    s = a_sym.sum()
    if s > 1e-15:
        a_sym /= s
    return a_sym


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase optimization: multi-peak subgradient descent, exact coordinate
    bisection, pair swap polish, noise injection, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(10.0, float(budget_s) - 35.0)

    best_a = None
    best_score = float('inf')

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            p_score = float(evaluate_sequence(parent.tolist()))
            print(f"Parent: n={len(parent)}, score={p_score:.6f}")
    except Exception:
        pass

    if parent is None:
        parent = np.ones(768) / 768

    n_parent = len(parent)

    def update_best(a_try, label):
        nonlocal best_a, best_score
        a_try = np.maximum(a_try, 0.0)
        s_try = a_try.sum()
        if s_try < 0.01:
            return
        score = float(evaluate_sequence(a_try.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a_try.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a_try)})")

    # === Phase 1: Optimize parent with combined method ===
    print(f"=== Phase 1: Optimize parent (n={n_parent}) ===")
    rem = deadline - time.monotonic() - 700
    if rem > 60:
        t_alloc = min(rem, 500)
        print(f"  Combined optimize, t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(parent, time_limit=t_alloc, verbose=True)
        update_best(a_opt, "parent")

    # === Phase 2: Symmetrize and optimize ===
    print(f"\n=== Phase 2: Symmetric variant (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 500
        if rem > 30:
            a_sym = symmetrize(best_a)
            t_alloc = min(rem, 150)
            print(f"  Symmetric, t={t_alloc:.0f}s")
            a_opt, _ = combined_optimize(a_sym, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "symmetric")

    # === Phase 3: Multi-length exploration ===
    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    base = best_a if best_a is not None else parent

    target_lengths = [512, 640, 896, 1024]
    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 400
        if rem < 50:
            break
        f_interp = interpolate_to_length(base, target_n)
        t_alloc = min(rem * 0.5, 200)
        print(f"  n={target_n}, t={t_alloc:.0f}s")
        a_opt, _ = combined_optimize(f_interp, time_limit=t_alloc, verbose=True)
        update_best(a_opt, f"n{target_n}")

    # === Phase 4: Deep optimization with different restarts ===
    print(f"\n=== Phase 4: Deep optimize (best={best_score:.6f}) ===")
    if best_a is not None:
        n_best = len(best_a)
        restarts = [
            ("direct", lambda: best_a.copy()),
            ("noise5", lambda: project_simplex(best_a + rng.standard_normal(n_best) * 0.005)),
            ("reverse", lambda: project_simplex(0.6 * best_a + 0.4 * best_a[::-1])),
            ("flat_mix", lambda: project_simplex(0.8 * best_a + 0.2 * np.ones(n_best) / n_best)),
            ("sym", lambda: symmetrize(best_a)),
        ]

        for idx, (label, init_fn) in enumerate(restarts):
            rem = deadline - time.monotonic() - 250 + idx * 50
            if rem < 30:
                break
            t_alloc = min(rem, 180)
            print(f"  Restart {label}, t={t_alloc:.0f}s")
            f_init = init_fn()
            a_opt, _ = combined_optimize(f_init, time_limit=t_alloc, verbose=(idx == 0))
            update_best(a_opt, f"deep_{label}")

    # === Phase 5: Final intensive coordinate bisection ===
    print(f"\n=== Phase 5: Final CD polish (best={best_score:.6f}) ===")
    if best_a is not None:
        rem = deadline - time.monotonic() - 60
        if rem > 10:
            t_alloc = min(rem, 300)
            print(f"  CD polish, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_bisection(best_a, time_limit=t_alloc, verbose=True)
            update_best(a_opt, "final_cd")

            # SG fine polish
            rem = deadline - time.monotonic() - 30
            if rem > 10:
                t_alloc = min(rem, 60)
                a_opt, _ = subgradient_phase(
                    best_a, time_limit=t_alloc, lr=0.004, mom=0.95,
                    top_k=max(30, len(best_a) // 5), temp=0.0001, decay=0.9999
                )
                update_best(a_opt, "final_sg")

            # Tiny perturbations + CD
            rem = deadline - time.monotonic() - 15
            if rem > 10 and best_a is not None:
                n_best = len(best_a)
                for p in range(3):
                    rem = deadline - time.monotonic() - 10
                    if rem < 8:
                        break
                    rng_p = np.random.default_rng(seed + 1000 + p)
                    f_pert = project_simplex(best_a + rng_p.standard_normal(n_best) * 0.001)
                    t_alloc = min(rem, 15)
                    a_opt, _ = coordinate_bisection(f_pert, time_limit=t_alloc)
                    update_best(a_opt, f"perturb_{p}")

    # Fallback
    if best_a is None:
        best_a = parent.copy()
        best_score = float(evaluate_sequence(best_a.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_a.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_a)})")
    return [float(x) for x in best_a]
