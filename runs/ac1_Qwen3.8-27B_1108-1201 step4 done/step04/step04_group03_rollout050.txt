We need to minimize: `2 * n * max(conv(a,a)) / (sum(a))^2`

where `conv(a,a)` is the linear convolution of the sequence with itself.

The key insight is that we want to minimize the maximum of the auto-convolution relative to the square of the sum. This is related to the "flat" convolution problem - we want the auto-convolution to be as flat as possible (all peaks equal).

1. **The objective**: Let `S = sum(a)`, `C = conv(a,a)`, then we minimize `2*n*max(C)/S^2`. Note that `C[0] = sum(a^2)` and `C[n-1] = a[0]*a[n-1]` etc. The minimum possible value of `max(C)/S^2` is at least `1/(2n-1)` (by AM-QM on the convolution entries, since there are 2n-1 entries and their sum is S^2, so max >= S^2/(2n-1)). This gives a lower bound of `2n/(2n-1)` on the objective, which approaches 1 for large n. But we can't achieve the uniform convolution exactly with non-negative sequences.

2. **Different approach ideas**:
   - **Spectral approach**: The auto-convolution `C = a * a` in time domain corresponds to `|FFT(a)|^2` in frequency domain. We want to minimize the maximum of the inverse FFT of `|FFT(a)|^2`. This is related to minimizing the peak of the power spectrum's autocorrelation.
   - **Rao-Meldrum approach**: Look for sequences where the auto-convolution is "almost" constant (like Barker sequences but for continuous values).
   - **Coordinate descent / block coordinate descent**: Update one coordinate at a time or in blocks.
   - **Alternating minimization with structure**: Try sequences with some symmetry (palindromic, etc.)
   - **Newton-type methods**: Since the objective involves max of a set of smooth functions, we can use techniques from minimizing the maximum of convex functions.

4. **New ideas**:
   - **Expanding set method (Kiwiel)**: Maintain a set of "active" peaks (the ones that are binding). Solve a QP or LP with only those constraints. If a new peak exceeds the bound, add it to the active set. This is much more efficient than considering all constraints.

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Adaptive smoothed-max (log-sum-exp) objective with FISTA acceleration
2. Expanding-set QP for exact subproblem solutions
3. Exact softmax-weighted gradients via FFT

import numpy as np

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def cross_corr_wa(w, a):
    """Compute g[m] = sum_k w[k] * a[k-m] for m=0..n-1 using FFT."""
    n = len(a)
    L = 1 << (len(w) + n - 1 - 1).bit_length()
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    g = np.fft.irfft(W * np.conj(A), L)[:n]
    return g

def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def smoothed_max_gradient(a, tau):
    """Compute gradient of tau*logsumexp(conv(a,a)/tau) w.r.t. a."""
    n = len(a)
    C = fast_convolve(a)
    # Softmax weights
    C_scaled = C / tau
    C_scaled -= C_scaled.max()
    exp_C = np.exp(C_scaled)
    w = exp_C / exp_C.sum()
    # Gradient: (2/tau) * cross_corr(w, a)
    g = cross_corr_wa(w, a)
    return (2.0 / tau) * g, C.max(), w

def fista_smoothed(a0, time_limit, tau_init=0.05, tau_min=1e-4, tau_decay=0.9995,
                    verbose=True):

    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    tau = tau_init
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    step = 1.0  # initial step size
    
    while time.monotonic() - t0 < time_limit:
        # Compute gradient at y
        grad_y, Cmax_y, _ = smoothed_max_gradient(y, tau)
        
        # FISTA update
        y_new = a + ((t - 1) / (t + 1.5)) * (a - a_prev)
        t_new = 0.5 * (1 + np.sqrt(1 + 4 * t * t))
        
        # Gradient step with backtracking
        a_cand = y_new - step * grad_y
        a_cand = project_simplex(a_cand)
        
        # Objective at candidate (use true max for checking)
        obj_cand = fast_convolve(a_cand).max()
        
        # Backtracking: if objective increased too much, reduce step
        # (smoothed obj might not be perfect proxy, so use true max)
        obj_y = Cmax_y
        if obj_cand > obj_y * 1.01 + 1e-10:
            step *= 0.5
            if step < 1e-12:
                break
            continue
        else:
            step = min(step * 1.5, 10.0 / tau)
        
        a_prev = a
        a = a_cand
        t = t_new
        
        # Update best
        if obj_cand < best_obj:
            best_obj = obj_cand
            best_a = a.copy()
        
        # Adaptive tau
        tau = max(tau * tau_decay, tau_min)
        
        if verbose and (int((time.monotonic() - t0) * 10) % 50 == 0):
            print(f"    FISTA: t={time.monotonic()-t0:.1f}s, tau={tau:.6f}, obj={best_obj:.8f}")

def expanding_set_qp(a0, time_limit, max_active=50, verbose=True):
    """
    Expanding set method: minimize sum_{k in S} C_k where S is the set of active peaks.
    Uses active-set QP solver.
    """
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    C = fast_convolve(a)
    best_obj = C.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        C = fast_convolve(a)
        
        # Select active set: top peaks
        top_k = min(max_active, len(C))
        S = set(np.argpartition(C, -top_k)[-top_k:].tolist())
        
        # Build QP matrix Q where Q[j,l] = count of k in S with k = j+l
        # Q[j,l] for j!=l: I(j+l in S) (each unordered pair counted once in a^T Q a,
        # but we need the factor of 2 for off-diagonals in the gradient)
        # Actually: minimize (1/2) a^T Q a where
        # (1/2) a^T Q a = sum_{k in S} C_k
        # Q[j,l] = |{k in S: k = j+l}| for j != l  (since a^T Q a counts each pair twice)
        # Q[j,j] = 2 * |{k in S: k = 2j}|
        
        # For efficiency, build Q implicitly: (Q a)[i] = sum_l Q[i,l] * a[l]
        # (Q a)[i] = sum_{l != i} I(i+l in S) * a[l] + 2*I(2i in S) * a[i]
        # = sum_{l: i+l in S, l != i} a[l] + 2*I(2i in S) * a[i]
        # = sum_{l: i+l in S} a[l] - I(2i in S) * a[i] + 2*I(2i in S) * a[i]
        # = sum_{l: i+l in S} a[l] + I(2i in S) * a[i]
        
        # So (Q a)[i] = sum_{l in S-i} a[l] + I(2i in S) * a[i]
        # where S-i = {l : i+l in S} = {s-i : s in S}
        
        # KKT: Q a + lambda * 1 + mu = 0, sum(a) = 1, a >= 0, mu >= 0, a*mu = 0
        # For free set F: (Q a)_i + lambda = 0 for i in F
        
        # Active set method:
        F = list(range(n))  # start with all free
        for inner_it in range(2 * n):
            F_set = set(F)
            QF_F = np.zeros((len(F), len(F)))
            # QF_F[i,j] = Q[F[i], F[j]]
            # Q[i,j] = I(i+j in S) for i != j, 2*I(2i in S) for i == j
            for ii in range(len(F)):
                for jj in range(len(F)):
                    si = F[ii] + F[jj]
                    if si in S:
                        if ii == jj:
                            QF_F[ii, jj] = 2.0
                        else:
                            QF_F[ii, jj] = 1.0
            
            # Solve: QF_F * a_F + lambda * 1 = 0, sum(a_F) = 1
            # => a_F = (QF_F)^{-1} * 1 / (1^T (QF_F)^{-1} 1)
            try:
                # Solve QF_F * x = 1
                x = np.linalg.solve(QF_F + 1e-12 * np.eye(len(F)), np.ones(len(F)))
                s_x = x.sum()
                if s_x <= 0:
                    # Try with regularization
                    x = np.linalg.solve(QF_F + 1e-10 * np.eye(len(F)), np.ones(len(F)))
                    s_x = x.sum()
                    if s_x <= 0:
                        break
                a_F = x / s_x
                lambda_val = -1.0 / s_x
            except np.linalg.LinAlgError:
                break
            
            # Check feasibility
            if np.min(a_F) < -1e-10:
                # Remove the most negative
                a_full = np.zeros(n)
                for idx, fi in enumerate(F):
                    a_full[fi] = a_F[idx]
                worst = np.argmin(a_full)
                F.remove(int(worst))
                if not F:
                    break
                continue
            
            # Check dual feasibility: for i not in F, (Q a)_i + lambda >= -tol
            a_full = np.zeros(n)
            for idx, fi in enumerate(F):
                a_full[fi] = max(a_F[idx], 0.0)
            a_full = a_full / a_full.sum()
            
            # Compute (Q a)_i for all i
            Qa = np.zeros(n)
            for i in range(n):
                # (Q a)[i] = sum_{l: i+l in S} a[l] + I(2i in S) * a[i]
                # = sum_{s in S, s >= i, s-i < n} a[s-i] + I(2i in S) * a[i]
                total = 0.0
                for s in S:
                    l = s - i
                    if 0 <= l < n:
                        total += a_full[l]
                if 2 * i in S:
                    total += a_full[i]
                Qa[i] = total
            
            # Check dual feasibility for indices not in F
            viol = False
            for i in range(n):
                if i not in F_set:
                    if Qa[i] + lambda_val < -1e-8:
                        F.append(i)
                        viol = True
                        break
            
            if not viol:
                # Converged
                a = a_full
                break
        
        # Evaluate
        C = fast_convolve(a)
        obj = C.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            if verbose:
                print(f"    QP: it={it}, t={time.monotonic()-t0:.1f}s, obj={best_obj:.8f}")
        
        # Check convergence
        if it > 50:
            break
    
    return best_a, best_obj

def mult_weight_exact(a0, time_limit, lr=0.02, tau=0.01, verbose=False):
    """Multiplicative weight update with exact softmax gradient."""
    a = np.maximum(a0.astype(np.float64), 0.0)

    if s < 1e-15:
        a = np.ones(len(a)) / len(a)

        a = a / s
    n = len(a)
    
    best_obj = fast_convolve(a).max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    it = 0
    cur_lr = lr
    cur_tau = tau
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        grad, Cmax, _ = smoothed_max_gradient(a, cur_tau)
        
        # Multiplicative weight: a <- a * exp(-lr * grad), then renormalize
        # This is mirror descent with KL divergence
        log_update = -cur_lr * grad
        # Clip to avoid overflow/underflow
        log_update = np.clip(log_update, -50, 50)
        a = a * np.exp(log_update)
        a = np.maximum(a, 1e-30)
        a = a / a.sum()
        
        obj = fast_convolve(a).max()
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_a = a.copy()
        
        cur_lr *= 0.9995
        cur_tau *= 1.001  # Slowly increase tau (less aggressive)
        
        if it % 500 == 0 and verbose:
            print(f"    MW: it={it}, t={time.monotonic()-t0:.1f}s, obj={best_obj:.8f}, lr={cur_lr:.6f}")
    
    return best_a, best_obj

def compute_ideal_sequence(n):
    """
    Compute the 'ideal' sequence with flat autocorrelation (as much as possible).
    Uses the recurrence: a_k = (c - sum_{j=1}^{k-1} 2*a_j*a_{k-j} - a_{k/2}^2 [if k even]) / (2*a_0)
    where c = 1/(2n-1) and a_0 = sqrt(c).
    We truncate when values become negative.
    """
    c = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(c)
    
    for k in range(1, n):
        # C_k = sum_{j=max(1,k-n+1)}^{min(n-1,k)} a_j * a_{k-j} + a_0 * a_k (if k < n)
        # We want C_k = c
        # a_0 * a_k + a_k * a_0 + sum_{j=1}^{k-1} a_j * a_{k-j} (counting carefully)
        # C_k = 2 * a_0 * a_k + sum_{j=1}^{min(k-1, n-1)} a_j * a_{k-j} (for k < n, and a_{k-j} might need j < n)
        # Actually: C_k = sum_{j=max(0,k-n+1)}^{min(n-1,k)} a_j * a_{k-j}
        # For k < n: C_k = a_0 * a_k + sum_{j=1}^{k} a_j * a_{k-j}
        #          = 2 * a_0 * a_k + sum_{j=1}^{k-1} a_j * a_{k-j} (the j=k term is a_k * a_0 = a_0 * a_k)
        # Wait: C_k = sum_j a_j * a_{k-j} for j = 0 to k (since k < n, both indices valid)
        # = a_0*a_k + a_1*a_{k-1} + ... + a_{k-1}*a_1 + a_k*a_0
        # = 2*a_0*a_k + sum_{j=1}^{k-1} a_j * a_{k-j}
        
        if k < n:
            cross_sum = 0.0
            for j in range(1, k):
                cross_sum += a[j] * a[k-j]
            # C_k = 2*a_0*a_k + cross_sum = c
            a_k = (c - cross_sum) / (2.0 * a[0])
            if a_k < 0:
                a[k] = 0.0
            else:
                a[k] = a_k
        else:
            # For k >= n, we can't solve for a_k since a only has n elements
            a[k] = 0.0
    
    # Normalize
    s = a.sum()
    if s > 1e-15:
        a = a / s
    else:
        a = np.ones(n) / n
    
    return a

def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    # Uniform
    inits.append(np.ones(n) / n)
    # Ideal flat autocorrelation
    inits.append(compute_ideal_sequence(n))
    # Edge-weighted
    x = np.arange(n, dtype=np.float64)
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / n)) + 0.01
    inits.append(c1 / c1.sum())
    # Two peaks (edges)
    peak = np.exp(-((x - 0) / (0.05 * n)) ** 2) + np.exp(-((x - (n-1)) / (0.05 * n)) ** 2)
    inits.append(peak / peak.sum())
    # Random
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits
