def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)

def peak_targeted_grad(conv, f, top_k=10, sharpness=100):
    """Compute gradient targeting the top-k peaks with sharp weights."""
    n = len(f)
    # Get top-k peaks
    top = np.argpartition(conv, -top_k)[-top_k:]
    # Weights: exponential based on how close to max
    vals = conv[top]
    max_val = vals.max()
    weights = np.exp(sharpness * (vals - max_val))
    weights /= weights.sum()
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for i, k in enumerate(top):
        w = weights[i]
        if w < 1e-12:
            continue
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        grad[lo:hi+1] += 2.0 * w * f[k - np.arange(lo, hi+1)]
    return grad

1. LP Frank-Wolfe with structured constraint matrix
2. Entropic mirror descent with log-sum-exp smoothing

4. Multi-start with diverse initializations

import numpy as np
from scipy.optimize import linprog
import time

def fast_convolve(a):
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_toeplitz_matrix(f, active_indices):
    """Build constraint matrix rows for given active lag indices."""
    n = len(f)
    A = np.zeros((len(active_indices), n))
    for idx, k in enumerate(active_indices):
        if k < n:
            A[idx, :k+1] = f[k::-1]
        else:
            A[idx, k-n+1:] = f[n-1:k-n-1:-1]
    return A

        conv = fast_convolve(f)
        threshold = M * 0.99
        active_indices = np.where(conv >= threshold)[0]
    
    # Limit number of constraints for speed

        conv = fast_convolve(f)
        # Keep the most active ones
        vals = conv[active_indices]

def project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
