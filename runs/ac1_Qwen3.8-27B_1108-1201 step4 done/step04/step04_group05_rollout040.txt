def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def project_simplex(v):

def compute_subgradient(a, conv, top_k=15, temp=0.001):

def subgradient_polish(f0, time_limit, lr=0.0015, mom=0.94, top_k=20, temp=0.0004, seed=0):

        f = np.ones(len(f)) / len(f)

        f /= s
    n = len(f)

    best_obj = conv.max()
    best_f = f.copy()
    v = np.zeros(n)
    
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr
    cur_temp = temp
    rng = np.random.default_rng(777 + seed)

        if time_limit - (time.monotonic() - t0) < 1:
            break

        v = mom * v - cur_lr * sg
        f_new = project_simplex(f + v)
        conv = fast_convolve(f_new)
        co = conv.max()
        
        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f_new.copy()
            f = f_new
            no_improve = 0
        else:
            no_improve += 1
        
        # Adapt lr
        if no_improve > 80:
            cur_lr *= 0.95
        elif no_improve == 0:
            cur_lr = min(cur_lr * 1.002, lr * 3.0)
        
        # Adapt temperature
        cur_temp *= 1.004
        
        # Restart
        if no_improve > 300:
            scale = 0.002 + 0.002 * (no_improve // 400)

    return best_f, best_obj


def pairwise_cd(f0, time_limit, seed=0):
    """Pairwise coordinate descent: transfer mass between pairs."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f /= s
    n = len(f)
    
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(31415 + seed)
    it = 0
    
    while time.monotonic() - t0 < time_limit:
        if time_limit - (time.monotonic() - t0) < 2:
            break
        it += 1
        improved = False
        
        # Try multiple random pairs per iteration
        n_pairs = min(200, n // 2)
        for _ in range(n_pairs):
            if time.monotonic() - t0 > time_limit - 1:
                break
            
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            
            # Also try structured pairs
            if rng.random() < 0.3:
                j = (i + int(rng.integers(1, max(2, n // 4)))) % n
                if i == j:
                    continue
            
            ai = f[i]
            if ai < 1e-10:
                continue
            
            # Try transferring fractions of mass from i to j
            for frac in [0.01, 0.02, 0.05, 0.1, 0.2]:
                delta = ai * frac
                f_new = f.copy()
                f_new[i] = f[i] - delta
                f_new[j] = f[j] + delta
                fn = f_new.sum()
                if fn < 1e-15:
                    continue
                f_new = f_new / fn
                val = fast_convolve(f_new).max()
                if val < best_obj - 1e-15:
                    f = f_new
                    best_obj = val
                    best_f = f.copy()
                    improved = True
                    break
            
            # Also try reverse direction (from j to i)
            if not improved:
                aj = f[j]
                if aj > 1e-10:
                    for frac in [0.01, 0.02, 0.05, 0.1]:
                        delta = aj * frac
                        f_new = f.copy()
                        f_new[j] = f[j] - delta
                        f_new[i] = f[i] + delta
                        f_new = np.maximum(f_new, 0.0)
                        fn = f_new.sum()
                        if fn < 1e-15:
                            continue
                        f_new = f_new / fn
                        val = fast_convolve(f_new).max()
                        if val < best_obj - 1e-15:
                            f = f_new
                            best_obj = val
                            best_f = f.copy()
                            improved = True
                            break
        
        if not improved and it > 5:
            break
    
    return best_f, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.2 * n)) ** 2)
    inits.append(g / g.sum())
    
    cos_t = np.cos(np.pi * x / (2.0 * (n - 1)))
    cos_t = np.maximum(cos_t, 0.01)
    inits.append(cos_t / cos_t.sum())
    
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 8.0) ** 2)))
    inits.append(peak / peak.sum())
    
    # Ricker wavelet-like
    ricker = (1 - 4 * (np.pi * (x / n - 0.5))**2) * np.exp(-np.pi**2 * (x / n - 0.5)**2 / 2)
    ricker = np.maximum(ricker, 0)
    if ricker.sum() > 0:
        inits.append(ricker / ricker.sum())
    
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

    return score

    f = np.maximum(f0.astype(np.float64), 0.0)
