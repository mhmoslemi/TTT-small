import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(n):
    """Build sparse LP constraint matrix for FW oracle.
    A[k,j] = 1 if 0 <= k-j < n, else 0.
    Constraint: sum_j A[k,j] * f[j] * g[j] <= M  (but f is fixed, so A[k,j]*f[j]*g[j])
    Actually the constraint is (f*g)[k] = sum_j f[j]*g[k-j] <= M
    So the matrix is B[k,j] = f[j] if 0 <= k-j < n, else 0.
    But f changes each iteration, so we can't pre-build. Instead, build the index structure.
    """
    # For each row k, the nonzeros are at columns j where 0 <= k-j < n
    # i.e., max(0, k-n+1) <= j <= min(n-1, k)
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        rows.extend([k] * (j_end - j_start + 1))
        cols.extend(range(j_start, j_end + 1))
    return np.array(rows), np.array(cols)


def solve_lp_oracle(f, M, lp_row_idx, lp_col_idx):
    """Solve: max sum(g) s.t. (f*g)[k] <= M, g >= 0.
    Uses the pre-computed index structure.
    """
    n = len(f)
    c = -np.ones(n)
    # A_ub[k,j] = f[j] for valid (k,j) pairs
    vals = np.ones(len(lp_row_idx))
    # For each entry (k,j), the value is f[j]
    vals = f[lp_col_idx]
    A_ub = csr_matrix((vals, (lp_row_idx, lp_col_idx)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None


def fw_lp_optimize(f0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle, optimal step size, and adaptive restarts."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    row_idx, col_idx = build_lp_matrix(n)
    conv = fast_convolve(f)
    M = conv.max()
    best_obj = M
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_relax = 1.0

    while time.monotonic() - t0 < time_limit:
        it += 1

        g = solve_lp_oracle(f, M * m_relax, row_idx, col_idx)
        if g is None:
            m_relax *= 1.2
            if m_relax > 10.0:
                m_relax = 1.0
                continue
            continue

        g_sum = g.sum()
        if g_sum < 1e-15:
            break
        g = g / g_sum

        if np.max(np.abs(g - f)) < 1e-9:
            no_improve += 1
            if no_improve > 40:
                # Restart
                rng = np.random.default_rng(it * 7919 + 13)
                scale = 0.003 + 0.005 * min(1.0, no_improve / 200.0)
                f = project_simplex(best_f + rng.standard_normal(n) * scale)
                conv = fast_convolve(f)
                M = conv.max()
                m_relax = 1.0
                no_improve = 0
            continue

        # Use optimal FW step: gamma = 2/(t+2)
        gamma = 2.0 / (it + 2)
        # Also try a few alternatives
        candidates_gamma = [gamma, 0.5 * gamma, 2.0 * gamma, 0.1, 0.2, 0.3, 0.5]

        best_val = M
        best_g = gamma
        for cg in candidates_gamma:
            cg = min(1.0, max(0.0, cg))
            ft = (1.0 - cg) * f + cg * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_g = cg

        if best_g > 1e-10:
            f = (1.0 - best_g) * f + best_g * g
            f = np.maximum(f, 0.0)
            f = f / f.sum()
            conv = fast_convolve(f)
            M = conv.max()
            m_relax = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_f = f.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_relax = min(m_relax * 1.05, 5.0)

        if verbose and it % 100 == 0:
            print(f"    FW iter {it}: obj={best_obj:.10f}, M={M:.10f}")

    return best_f, best_obj


def mult_weight_optimize(f0, time_limit, lr_init=0.04, top_k=20, temp_init=0.0004,
                         decay_lr=0.9997, decay_temp=1.003, verbose=False):
    """Multiplicative weight (mirror descent) with top-k peak focusing."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        M = conv.max()

        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        log_update = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            grad_contrib = 2.0 * w[i] * f[k - idx]
            log_update[idx] -= cur_lr * grad_contrib

        f = f * np.exp(log_update)
        f = np.maximum(f, 1e-30)
        f = f / f.sum()

        conv = fast_convolve(f)
        obj = conv.max()

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        cur_lr *= decay_lr
        cur_temp *= decay_temp

        if no_improve > 100:
            rng = np.random.default_rng(int(time.monotonic() * 999983) % 1000000)
            f = best_f * np.exp(rng.standard_normal(n) * 0.004)
            f = f / f.sum()
            cur_lr = max(lr_init * 0.7, 0.01)
            cur_temp = temp_init * 1.5
            no_improve = 0

    return best_f, best_obj


def coord_descent_polish(f0, time_limit, verbose=False):
    """Coordinate descent: for each coordinate, find the best value via 1D search.
    When a[j] changes by delta (and we renormalize), the convolution changes
    in a predictable way. We use a small grid search around the current value.
    """
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0

    while time.monotonic() - t0 < time_limit:
        it += 1
        improved = False

        for j in range(n):
            # Try perturbing a[j] in several directions
            base_val = f[j]
            if base_val < 1e-15:
                # Try setting it to a small positive value
                candidates = [0.0, 0.001 / n, 0.005 / n]
            else:
                # Try scaling up and down
                candidates = [0.0, base_val * 0.5, base_val * 0.8, base_val * 1.0,
                              base_val * 1.2, base_val * 1.5, base_val * 2.0]

            best_local = best_obj
            best_cand = base_val

            for cv in candidates:
                f_test = f.copy()
                f_test[j] = cv
                f_test = np.maximum(f_test, 0.0)
                st = f_test.sum()
                if st < 1e-15:
                    continue
                f_test = f_test / st
                c = fast_convolve(f_test)
                val = c.max()
                if val < best_local - 1e-14:
                    best_local = val
                    best_cand = cv

            if best_cand != base_val:
                f[j] = best_cand
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                improved = True

                if best_local < best_obj - 1e-14:
                    best_obj = best_local
                    best_f = f.copy()
                    no_improve = 0

        if not improved:
            no_improve += 1
            if no_improve > 3:
                # Try random coordinate jumps
                rng = np.random.default_rng(int(time.monotonic() * 7919) % 1000000)
                j = rng.integers(n)
                f[j] = f[j] * rng.uniform(0.3, 3.0)
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                no_improve = 0
            elif no_improve > 10:
                break

    return best_f, best_obj


def subgradient_polish(f0, time_limit, lr_init=0.002, mom=0.93, top_k=30,
                       temp_init=0.0001, verbose=False):

    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_init
    cur_temp = temp_init
    it = 0

    while time.monotonic() - t0 < time_limit:
        if top_k >= len(conv):
            peaks = np.arange(len(conv))
        else:
            peaks = np.argpartition(conv, -top_k)[-top_k:]

        pv = conv[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:

            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * f[k - idx]

        f = project_simplex(f + v)

def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.3))
    inits.append(rng.dirichlet(np.ones(n) * 2.0))

    x = np.arange(n, dtype=np.float64)
    # Edge-weighted
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    # Gaussian center
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    # Cosine
    c1 = np.abs(np.cos(np.pi * x / (2 * n))) + 0.01
    inits.append(c1 / c1.sum())
    # Bimodal
    b1 = np.exp(-0.5 * ((x - 0.25 * n) / (0.1 * n)) ** 2) + \
         np.exp(-0.5 * ((x - 0.75 * n) / (0.1 * n)) ** 2)
    inits.append(b1 / b1.sum())
    # Linear ramp (asymmetric)
    ramp = x + 1.0
    inits.append(ramp / ramp.sum())
    # Inverse ramp
    inv_ramp = (n - x) + 1.0
    inits.append(inv_ramp / inv_ramp.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits

            if rem < 10:
                break
            t_a = min(t_per, rem)
            print(f"  MW cfg {ci}, t={t_a:.0f}s")
            a_opt, _ = mult_weight_optimize(
                best_seq, time_limit=t_a,
                lr_init=cfg['lr_init'], top_k=cfg['top_k'],
                temp_init=cfg['temp_init']
            )
            try_seq(a_opt, f"mw_{ci}")

    # === Phase 5: Subgradient polish ===
    print("\n=== Phase 5: Subgradient polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 100
        if rem > 20:
            t_a = min(rem, 80)
            print(f"  SG polish, t={t_a:.0f}s")
            n_b = len(best_seq)
            a_opt, _ = subgradient_polish(
                best_seq, time_limit=t_a,
                lr_init=0.001, mom=0.94,
                top_k=max(20, n_b // 10), temp_init=0.00005
            )
            try_seq(a_opt, "sg_polish")

    # === Phase 6: Coordinate descent ===
    print("\n=== Phase 6: Coordinate descent ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 10:
            t_a = min(rem, 60)
            print(f"  Coord descent, t={t_a:.0f}s")
            a_opt, _ = coord_descent_polish(best_seq, time_limit=t_a)
            try_seq(a_opt, "coord_desc")

    # === Phase 7: Final micro-polish ===
    print("\n=== Phase 7: Final micro-polish ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 15
        if rem > 10:
            # Very fine SG
            t_a = min(rem, 30)
            n_b = len(best_seq)
            a_opt, _ = subgradient_polish(
                best_seq, time_limit=t_a,
                lr_init=0.0003, mom=0.96,
                top_k=max(10, n_b // 5), temp_init=0.00002
            )
            try_seq(a_opt, "micro_sg")

            # Tiny perturbations
            rem = deadline - time.monotonic() - 10
            if rem > 10:
                for _ in range(3):
                    rem = deadline - time.monotonic() - 8
                    if rem < 5:
                        break
                    noise = rng.standard_normal(n_b) * 0.0003
                    f_pert = project_simplex(best_seq + noise)
                    a_opt, _ = subgradient_polish(
                        f_pert, time_limit=min(rem, 10),
                        lr_init=0.0002, mom=0.96,
                        top_k=n_b // 3, temp_init=0.00001
                    )
                    try_seq(a_opt, "micro_perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs) and fs < best_score:
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.8f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
