"""
Multi-phase optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses: (1) diverse asymmetric initializations, (2) Nesterov-accelerated peak-chasing
gradient with adaptive temperature, (3) LP-based Frank-Wolfe refinement,
(4) targeted coordinate-level mass redistribution, and (5) multi-start with
warm-restart exploration to escape local minima.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_rows(f, row_indices):
    """Build rows of B matrix where (B*g)[k] = (f*g)[k] for given row indices."""
    n = len(f)
    B = np.zeros((len(row_indices), n))
    for i, k in enumerate(row_indices):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je:
            continue
        idx = np.arange(js, je + 1)
        B[i, idx] = f[k - idx]
    return B


def nesterov_peak_chase(f0, time_limit, lr_base=0.015, momentum=0.92,
                         top_k=30, temp_init=0.0005, temp_growth=1.003,
                         verbose=False):
    """Nesterov-accelerated gradient descent with softmax-weighted peak targeting."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    y = f.copy()  # Nesterov momentum variable
    v = np.zeros(n)
    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr_base
    cur_temp = temp_init
    it = 0
    rng = np.random.default_rng(777)

    while time.monotonic() - t0 < time_limit:
        it += 1

        # Nesterov step: evaluate gradient at y (lookahead point)
        y = f + momentum * v
        y = project_simplex(y)
        conv_y = fast_convolve(y)

        # Get top peaks
        if top_k >= len(conv_y):
            peaks = np.arange(len(conv_y))
        else:
            peaks = np.argpartition(conv_y, -top_k)[-top_k:]

        pv = conv_y[peaks]
        scaled = pv / cur_temp
        scaled -= scaled.max()
        w = np.exp(scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[np.argmax(pv)] = 1.0
        else:
            w /= ws

        # Compute subgradient
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-14:
                continue
            k = int(peaks[i])
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            idx = np.arange(js, je + 1)
            sg[idx] += 2.0 * w[i] * y[k - idx]

        # Update
        v = momentum * v - cur_lr * sg
        f_new = project_simplex(f + v)
        f = f_new

        conv = fast_convolve(f)
        co = conv.max()

        if co < best_obj - 1e-14:
            best_obj = co
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        cur_lr *= 0.9995
        cur_temp *= temp_growth

        # Restart if stuck
        if no_improve > 200:
            scale = min(0.01, 0.002 * (no_improve // 200))
            f = project_simplex(best_f + rng.standard_normal(n) * scale)
            v = np.zeros(n)
            cur_lr = max(lr_base * 0.5, 0.003)
            no_improve = 0

        if verbose and it % 500 == 0:
            obj = 2 * n * best_obj
            print(f"    [nesterov] it={it}, obj={obj:.8f}, lr={cur_lr:.6f}, temp={cur_temp:.2e}")

    return best_f, best_obj


def linearized_lp_step(f, conv_f, top_k=50):
    """Solve reduced linearized LP: min t s.t. t >= conv_f[k] + 2*(B*g)[k], g >= -f, sum(g)=0."""
    n = len(f)
    peaks = np.argpartition(conv_f, -top_k)[-top_k:]
    peaks = np.sort(peaks)

    B = build_conv_matrix_rows(f, peaks)

    # Variables: [t, g_0, ..., g_{n-1}]
    c = np.zeros(n + 1)
    c[0] = 1.0

    A_ub = np.zeros((len(peaks), n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = 2.0 * B
    b_ub = -conv_f[peaks]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)

    bounds = [(None, None)] + [(-f[i], None) for i in range(n)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return res.x[1:]
    except Exception:
        pass
    return None


def fw_optimize(f0, time_limit, top_k=40, verbose=False):
    """Frank-Wolfe with linearized LP oracle and adaptive line search."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    rng = np.random.default_rng(999)

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(f)
        current_obj = conv.max()

        g = linearized_lp_step(f, conv, top_k=top_k)
        if g is None:
            no_improve += 5
            if no_improve > 50:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.003)
                no_improve = 0
            continue

        target = f + g
        target = np.maximum(target, 0.0)
        ts = target.sum()
        if ts < 1e-15:
            no_improve += 1
            continue
        target = target / ts

        if np.max(np.abs(target - f)) < 1e-10:
            no_improve += 1
            if no_improve > 30:
                f = project_simplex(best_f + rng.standard_normal(n) * 0.002)
                no_improve = 0
            continue

        # Line search: bisection
        best_alpha = 0.0
        best_val = current_obj
        lo, hi = 0.0, 1.0
        for _ in range(10):
            mid = (lo + hi) / 2.0
            f_new = (1.0 - mid) * f + mid * target
            f_new = np.maximum(f_new, 0.0)
            fs = f_new.sum()
            if fs < 1e-15:
                break
            f_new = f_new / fs
            val = fast_convolve(f_new).max()
            if val < best_val:
                best_val = val
                best_alpha = mid
                lo = mid
            else:
                hi = mid

        if best_alpha > 1e-8:
            f = (1.0 - best_alpha) * f + best_alpha * target
            f = np.maximum(f, 0.0)
            fs = f.sum()
            if fs < 1e-15:
                f = best_f.copy()
            else:
                f = f / fs

            if best_val < best_obj - 1e-14:
                best_obj = best_val
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 50 == 0:
                    print(f"    [FW] it={it}, obj={2*n*best_obj:.8f}, alpha={best_alpha:.4f}")
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 40:
            f = project_simplex(best_f + rng.standard_normal(n) * (0.002 + 0.001 * (no_improve - 40) / 10))
            no_improve = 0

    return best_f, best_obj


def coordinate_polish(f0, time_limit, top_peaks=10):
    """Targeted coordinate-level mass redistribution to reduce top peaks."""
    f = np.maximum(f0.astype(np.float64), 0.0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(len(f)) / len(f)
    else:
        f = f / s
    n = len(f)

    conv = fast_convolve(f)
    best_obj = conv.max()
    best_f = f.copy()

    t0 = time.monotonic()
    rng = np.random.default_rng(4242)

    while time.monotonic() - t0 < time_limit:
        conv = fast_convolve(f)
        current_obj = conv.max()

        # Identify top peaks
        peaks = np.argpartition(conv, -top_peaks)[-top_peaks:]
        peaks = np.sort(peaks)

        # For each peak, find which coordinates contribute most
        for k in peaks:
            if time.monotonic() - t0 > time_limit:
                break
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js > je:
                continue
            idx = np.arange(js, je + 1)
            contributions = f * f[k - idx]

            # Find high contributors and low contributors
            if len(idx) < 4:
                continue
            sorted_idx = idx[np.argsort(contributions)]
            # Take top 3 contributors and bottom 3 non-contributors
            high = sorted_idx[-3:]
            low_mask = np.ones(n, dtype=bool)
            low_mask[high] = False
            low_candidates = np.where(low_mask)[0]
            if len(low_candidates) < 3:
                continue
            low = low_candidates[np.argsort(f[low_candidates])[:3]]

            # Try moving mass from high to low
            for h in high:
                for l in low:
                    if f[h] < 1e-8:
                        continue
                    # Try various amounts of transfer
                    for frac in [0.1, 0.3, 0.5]:
                        delta = frac * f[h]
                        if delta > f[h] * 0.9:
                            delta = f[h] * 0.9
                        f_try = f.copy()
                        f_try[h] -= delta
                        f_try[l] += delta
                        f_try = np.maximum(f_try, 0.0)
                        fs = f_try.sum()
                        if fs < 1e-15:
                            continue
                        f_try = f_try / fs
                        val = fast_convolve(f_try).max()
                        if val < best_obj - 1e-14:
                            best_obj = val
                            best_f = f_try.copy()
                            f = f_try

        # Periodic global check
        if (int((time.monotonic() - t0) * 2)) % 5 == 0:
            if current_obj < best_obj - 1e-14:
                best_obj = current_obj
                best_f = f.copy()

    return best_f, best_obj


def generate_diverse_inits(n, parent, rng, num=15):
    """Generate diverse initializations including asymmetric ones."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    c = 0.5 * (n - 1)

    # 1. Uniform
    inits.append(np.ones(n) / n)

    # 2. Triangular (asymmetric - concentrated left)
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        center = frac * (n - 1)
        g = np.exp(-np.abs(x - center) / (n * 0.15))
        inits.append(g / g.sum())

    # 3. Exponential decay (asymmetric)
    for rate in [1.0, 2.0, 3.0]:
        g = np.exp(-rate * x / n)
        inits.append(g / g.sum())
        g2 = np.exp(-rate * (n - 1 - x) / n)
        inits.append(g2 / g2.sum())

    # 4. Quadratic (asymmetric)
    for center_frac in [0.25, 0.5, 0.75]:
        center = center_frac * (n - 1)
        g = 1.0 / (1.0 + ((x - center) / (n * 0.2)) ** 2)
        inits.append(g / g.sum())

    # 5. Sine-weighted
    for freq in [1, 2, 3]:
        g = np.abs(np.sin(freq * np.pi * x / n)) + 0.1
        inits.append(g / g.sum())

    # 6. Random Dirichlet
    for alpha in [0.3, 0.5, 1.0, 2.0]:
        inits.append(rng.dirichlet(np.ones(n) * alpha))

    # 7. Two-peak asymmetric
    for p1, p2 in [(0.2, 0.8), (0.3, 0.7), (0.25, 0.75)]:
        g1 = np.exp(-0.5 * ((x - p1 * (n-1)) / (n*0.08)) ** 2)
        g2 = np.exp(-0.5 * ((x - p2 * (n-1)) / (n*0.12)) ** 2)
        g = g1 + 0.7 * g2
        inits.append(g / g.sum())

    # 8. Parent (if available)
    if parent is not None:
        inits.append(parent.copy())
        # Perturbed parent
        inits.append(project_simplex(parent + rng.standard_normal(n) * 0.02))

    return inits[:num]


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Multi-phase search:
      Phase 1: Diverse initialization with quick scoring
      Phase 2: Nesterov peak-chase on top candidates
      Phase 3: LP-based FW refinement on best
      Phase 4: Coordinate polish
      Phase 5: Warm-restart exploration and final polish
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        a = np.maximum(a, 0.0)
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.8f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            n_parent = len(parent)
            print(f"Parent loaded: n={n_parent}, score={evaluate_sequence(parent.tolist()):.8f}")
    except Exception:
        n_parent = 384

    if parent is None:
        n_parent = 384

    # === Phase 1: Diverse initialization ===
    print("=== Phase 1: Diverse Initialization ===")
    t_start = time.monotonic()

    n_use = n_parent
    inits = generate_diverse_inits(n_use, parent, rng, num=20)

    # Quick score all inits
    scored = []
    for i, ini in enumerate(inits):
        if time.monotonic() > deadline - 300:
            break
        obj = fast_convolve(ini).max()
        score = 2 * n_use * obj
        scored.append((score, i, ini))

    scored.sort(key=lambda x: x[0])
    print(f"  Generated {len(scored)} inits, best quick-score: {scored[0][0]:.6f}")

    # Try top few with evaluate_sequence
    for score, i, ini in scored[:8]:
        if time.monotonic() > deadline - 250:
            break
        try_seq(ini, f"init_{i}")

    if best_seq is None:
        best_seq = np.ones(n_use) / n_use
        try_seq(best_seq, "fallback")

    top_candidates = [ini for _, _, ini in scored[:6]]
    print(f"  Phase 1 best: {best_score:.8f} (n={len(best_seq)})")

    # === Phase 2: Nesterov peak-chase on candidates ===
    print("\n=== Phase 2: Nesterov Peak-Chase ===")
    remaining = deadline - time.monotonic() - 100
    t_per_candidate = min(remaining / max(1, len(top_candidates)), 80.0)

    cand_results = []
    for ci, cand in enumerate(top_candidates):
        if time.monotonic() > deadline - 80:
            break
        t_alloc = min(t_per_candidate, deadline - time.monotonic() - 60)
        if t_alloc < 10:
            break
        print(f"  Candidate {ci}: budget={t_alloc:.0f}s")
        a_opt, _ = nesterov_peak_chase(cand, time_limit=t_alloc,
                                       lr_base=0.012, momentum=0.93,
                                       top_k=40, temp_init=0.0004,
                                       temp_growth=1.004)
        try_seq(a_opt, f"nesterov_{ci}")
        cand_results.append(a_opt.copy())

    print(f"  Phase 2 best: {best_score:.8f}")

    # === Phase 3: LP-based FW refinement ===
    print("\n=== Phase 3: LP Frank-Wolfe ===")
    remaining = deadline - time.monotonic() - 60
    t_fw = min(remaining * 0.4, 200.0)

    if t_fw > 20:
        print(f"  Running FW on best, budget={t_fw:.0f}s")
        a_opt, _ = fw_optimize(best_seq, time_limit=t_fw, top_k=50, verbose=True)
        try_seq(a_opt, "fw_main")

        # Try on second-best candidate too
        rem = deadline - time.monotonic() - 40
        if rem > 20 and len(cand_results) > 1:
            a_opt2, _ = fw_optimize(cand_results[1], time_limit=min(rem * 0.5, 80), top_k=40)
            try_seq(a_opt2, "fw_alt")

    # Also try FW from a perturbed version of best
    rem = deadline - time.monotonic() - 40
    if rem > 20:
        f_pert = project_simplex(best_seq + rng.standard_normal(n_use) * 0.015)
        a_opt, _ = fw_optimize(f_pert, time_limit=min(rem * 0.5, 80), top_k=40)
        try_seq(a_opt, "fw_pert")

    print(f"  Phase 3 best: {best_score:.8f}")

    # === Phase 4: Coordinate polish ===
    print("\n=== Phase 4: Coordinate Polish ===")
    rem = deadline - time.monotonic() - 30
    t_coord = min(rem * 0.3, 100.0)

    if t_coord > 10:
        print(f"  Running coordinate polish, budget={t_coord:.0f}s")
        a_opt, _ = coordinate_polish(best_seq, time_limit=t_coord, top_peaks=8)
        try_seq(a_opt, "coord")

    print(f"  Phase 4 best: {best_score:.8f}")

    # === Phase 5: Warm-restart exploration ===
    print("\n=== Phase 5: Warm-Restart & Final Polish ===")
    rem = deadline - time.monotonic() - 15

    # Multiple warm restarts with short Nesterov + short FW
    num_restarts = 4
    for r in range(num_restarts):
        if rem < 15:
            break
        t_each = min(rem / max(1, num_restarts - r), 40.0)

        # Perturb best with increasing scale
        scale = 0.008 + 0.005 * r
        f_pert = project_simplex(best_seq + rng.standard_normal(n_use) * scale)

        # Quick Nesterov
        t_nest = t_each * 0.5
        a_opt, _ = nesterov_peak_chase(f_pert, time_limit=t_nest,
                                       lr_base=0.008, momentum=0.92,
                                       top_k=30, temp_init=0.0003)
        try_seq(a_opt, f"restart_nest_{r}")

        # Quick FW
        rem = deadline - time.monotonic() - 10
        if rem > 10:
            t_fw2 = min(rem * 0.4, 25.0)
            a_opt, _ = fw_optimize(a_opt, time_limit=t_fw2, top_k=35)
            try_seq(a_opt, f"restart_fw_{r}")

        rem = deadline - time.monotonic() - 10

    # Final: very short Nesterov polish on the absolute best
    rem = deadline - time.monotonic() - 5
    if rem > 5:
        a_opt, _ = nesterov_peak_chase(best_seq, time_limit=rem,
                                       lr_base=0.003, momentum=0.95,
                                       top_k=50, temp_init=0.0001,
                                       temp_growth=1.002)
        try_seq(a_opt, "final_polish")

    # Final verification
    if best_seq is not None:
        fs = float(evaluate_sequence(best_seq.tolist()))
        if np.isfinite(fs):
            best_score = fs
        print(f"\nFINAL SCORE: {best_score:.10f} (n={len(best_seq)})")
    else:
        best_seq = np.ones(256) / 256
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"\nFALLBACK SCORE: {best_score:.10f}")

    return [float(x) for x in best_seq]
