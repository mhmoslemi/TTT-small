"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
1. L-BFGS with log-sum-exp smoothing and temperature annealing
2. LP-based Frank-Wolfe for global direction finding
3. Multi-start with diverse structured initializations
4. Adaptive time allocation with progressive refinement

import numpy as np
from scipy.optimize import linprog
from scipy.signal import correlate
import time


def fast_convolve(a, b=None):
    """Linear auto-convolution via FFT."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    B = np.fft.rfft(b, fft_size)
    return np.fft.irfft(A * B, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_objective_and_grad(f, t):
    """
    Compute the log-sum-exp smoothed objective and its gradient.
    F_t(f) = (1/t) * log(sum_k exp(t * conv(f,f)[k]))
    grad_j = 2 * sum_k p_k * f[k-j] where p_k = exp(t*c_k)/sum(exp(t*c_k))
    """
    n = len(f)
    conv = fast_convolve(f, f)
    scaled = t * conv
    scaled -= scaled.max()
    log_sum_exp = np.log(np.exp(scaled).sum()) / t
    p = np.exp(scaled) / np.exp(scaled).sum()
    
    # Gradient: grad[j] = 2 * sum_i f[i] * p[i+j] = 2 * correlate(p, f)[j] (valid mode)
    grad = 2.0 * correlate(p, f, mode='valid')
    
    return log_sum_exp, grad


def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def lbfgs_smoothed(f0, t, time_limit, max_iter=200, tol=1e-10):
    """
    Minimize the smoothed objective F_t(softmax(x)) using L-BFGS.
    Returns the optimal f (as a probability vector).
    """
    n = len(f0)
    # Start from x = log(f0) (approximate inverse of softmax)
    f_safe = np.maximum(f0, 1e-30)
    x = np.log(f_safe / f_safe.sum())
    
    def obj_and_grad(x):
        f = softmax(x)
        val, grad_f = smooth_objective_and_grad(f, t)
        # Chain rule for softmax parameterization
        # dF/dx_j = f_j * (grad_f_j - sum_i f_i * grad_f_i)
        mean_grad = np.dot(f, grad_f)
        grad_x = f * (grad_f - mean_grad)
        return val, grad_x
    
    from scipy.optimize import minimize
    
    best_x = x.copy()
    best_val, _ = obj_and_grad(best_x)
    t0 = time.monotonic()
    
    # Use L-BFGS-B with manual termination
    result = minimize(
        obj_and_grad, x, method='L-BFGS-B',
        jac=True,
        options={'maxiter': max_iter, 'ftol': tol, 'gtol': 1e-8,
                 'maxfun': 5000, 'maxls': 50}
    )
    
    if result.fun < best_val:
        best_x = result.x
        best_val = result.fun
    
    return softmax(best_x), best_val


def solve_lp(f, M, time_limit=10):
    """Solve LP: max sum(g) s.t. conv(f, g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_step(f, M_scale=1.0):
    """One Frank-Wolfe step using LP oracle."""
    n = len(f)
    conv = fast_convolve(f, f)
    M = conv.max() * M_scale
    g = solve_lp(f, M)
    if g is None:
        return f
    gs = g.sum()
    if gs < 1e-15:
        return f
    g = g / gs
    return g


def anneal_optimize(f0, time_limit, t_start=20.0, t_end=500.0, 
                     n_temps=10, fw_every=3, fw_budget_frac=0.3):
    """
    Temperature annealing with L-BFGS at each temperature level,
    interleaved with LP-based Frank-Wolfe steps.
    """
    n = len(f0)
    f = f0.copy()
    t0 = time.monotonic()
    
    # Geometric temperature schedule
    temps = np.geomspace(t_start, t_end, n_temps)
    
    fw_time_budget = time_limit * fw_budget_frac
    fw_spent = 0.0
    
    for ti, t in enumerate(temps):
        rem = time_limit - (time.monotonic() - t0)
        if rem < 5:
            break
        
        # L-BFGS at this temperature
        lbfgs_time = min(rem * 0.6, 30)
        if lbfgs_time > 2:
            f, _ = lbfgs_smoothed(f, t, time_limit=lbfgs_time)
        
        # Frank-Wolfe step every few temperatures
        if (ti % fw_every == 0) and (fw_spent < fw_time_budget):
            rem = time_limit - (time.monotonic() - t0)
            if rem > 20:
                fw_t = min(15, rem - 10, fw_time_budget - fw_spent)
                if fw_t > 3:
                    t_start_fw = time.monotonic()
                    g = fw_step(f)
                    fw_spent += time.monotonic() - t_start_fw
                    
                    if not np.allclose(g, f, atol=1e-6):
                        # Line search
                        best_alpha = 0.0
                        best_val = smooth_objective_and_grad(f, t)[0]
                        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                            ft = project_simplex((1 - alpha) * f + alpha * g)
                            val = smooth_objective_and_grad(ft, t)[0]
                            if val < best_val:
                                best_val = val
                                best_alpha = alpha
                        if best_alpha > 0:
                            f = project_simplex((1 - best_alpha) * f + best_alpha * g)
    
    # Final evaluation with exact max
    conv = fast_convolve(f, f)
    return f, conv.max()


def generate_inits(n, parent=None, rng=None):

    inits.append(c / c.sum())
    
    # Exponential from center
    e = np.exp(-2.0 * np.abs(x - (n-1)/2.0) / n)
    inits.append(e / e.sum())
    
    # Two bumps
    b1 = np.exp(-8.0 * ((x - n*0.25) / n) ** 2)
    b2 = np.exp(-8.0 * ((x - n*0.75) / n) ** 2)
    two = b1 + b2 + 0.01
    inits.append(two / two.sum())
    
    # Sinc-like
    sn = np.sinc((x - n/2) / (n/4.0)) + 0.05
    sn = np.abs(sn)
    inits.append(sn / sn.sum())
    
    # Random (Dirichlet)
    r = rng.dirichlet(np.ones(n) * 0.5)
    inits.append(r)
    
    # Parent-based
    if parent is not None:
        px = np.interp(x, np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits


def fine_polish(f0, time_limit):
    """Fine polishing using very small temperature L-BFGS."""
    n = len(f0)
    f = f0.copy()
    t0 = time.monotonic()
    
    # Use increasing temperatures for sharpening
    for t in [200, 500, 1000, 2000]:
        rem = time_limit - (time.monotonic() - t0)
        if rem < 3:
            break
        f, _ = lbfgs_smoothed(f, t, time_limit=min(rem * 0.4, 15), max_iter=50)
    
    # Final exact subgradient steps
    rem = time_limit - (time.monotonic() - t0)
    if rem > 2:
        lr = 0.001
        for _ in range(200):
            rem = time_limit - (time.monotonic() - t0)
            if rem < 1:
                break
            conv = fast_convolve(f, f)
            M = conv.max()
            # Find all peaks at max (within tolerance)
            peaks = np.where(conv >= M * (1 - 1e-9))[0]
            # Exact subgradient
            sg = np.zeros(n)
            for k in peaks:
                js = max(0, k - n + 1)
                je = min(n - 1, k)
                idx = np.arange(js, je + 1)
                sg[idx] += f[k - idx]
            sg *= 2.0 / len(peaks)
            
            f_new = project_simplex(f - lr * sg)
            conv_new = fast_convolve(f_new, f_new)
            if conv_new.max() < M:
                f = f_new
            lr *= 0.99
    
    return f
