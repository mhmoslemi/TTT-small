"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive subgradient methods 
with Adam optimization, LP-based Frank-Wolfe restarts, and coordinate descent.
Focuses on lengths near 384 with multi-start exploration.

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def subgradient_of_max_conv(a, conv, top_k=30, temp=0.005):
    """
    Compute the (weighted) subgradient of max_k conv[k].
    Uses softmax weighting over the top_k peaks.
    The gradient of (a*a)[k] w.r.t. a[m] is 2*a[k-m] for valid m.
    """
    n = len(a)
    m_total = len(conv)
    
    if top_k >= m_total:
        peaks = np.arange(m_total)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]
    
    # Softmax weights
    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws
    
    sg = np.zeros(n)
    for i in range(len(peaks)):
        wi = w[i]
        if wi < 1e-14:
            continue
        k = int(peaks[i])
        # Gradient of (a*a)[k] w.r.t. a[m] = 2*a[k-m] for m in [max(0,k-n+1), min(n-1,k)]
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            # a[k-m] for m in [m_lo, m_hi] = a[k-m_hi : k-m_lo : -1]... 
            # Actually a[k-m_lo], a[k-m_lo+1], ..., a[k-m_hi]
            # = a[k-m_hi : k-m_lo+1] reversed? No.
            # For m = m_lo: a[k - m_lo]
            # For m = m_lo+1: a[k - m_lo - 1]
            # ...
            # For m = m_hi: a[k - m_hi]
            # So sg[m_lo:m_hi+1] += 2 * wi * a[k-m_hi : k-m_lo+1 : -1]
            # Wait: k-m_hi, k-m_hi+1, ..., k-m_lo
            # That's a[k-m_hi : k-m_lo+1]
            # And we assign to positions m_lo, m_lo+1, ..., m_hi
            # So sg[m_lo + i] += 2*wi * a[k - m_hi + i] for i = 0, ..., m_hi - m_lo
            # i.e., sg[m_lo:m_hi+1] += 2*wi * a[k-m_hi : k-m_lo+1]
            sg[m_lo:m_hi+1] += 2.0 * wi * a[k - m_hi:k - m_lo + 1]
    
    return sg


def adam_subgradient_optimize(a0, time_limit, lr=0.001, beta1=0.9, beta2=0.999, 
                                eps=1e-8, top_k=30, temp=0.005, 
                                restart_every=2000, perturb_scale=0.01,
                                verbose=False):
    """Adam-optimized subgradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    t = 0
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    no_improve = 0
    
    t0 = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - t0 < time_limit:
        t += 1
        
        # Compute subgradient
        sg = subgradient_of_max_conv(a, conv, top_k=top_k, temp=temp)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv = fast_convolve(a_new)
        obj = conv.max()
        
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        a = a_new
        
        # Restart
        if no_improve >= restart_every:
            # Try perturbation
            noise = rng.standard_normal(n) * perturb_scale
            a = project_simplex(best_a + noise)
            conv = fast_convolve(a)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            perturb_scale = max(0.001, perturb_scale * 0.8)
            
            if verbose and no_improve % 10000 == 0:
                print(f"    Adam iter {t}: obj={best_obj:.10f}")
        
        # Adaptive temperature
        if t > 0 and t % 5000 == 0:
            temp = max(0.001, temp * 0.95)
        
        # Check time
        if time.monotonic() - t0 > time_limit - 1:
            break
    
    return best_a, best_obj


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_step(f, M, expand=1.0):
    """Single Frank-Wolfe step using LP oracle."""
    g = solve_lp_direction(f, M * expand)
    if g is None:
        return None
    gs = g.sum()
    if gs < 1e-15:
        return None
    return g / gs


def fw_lp_optimize(a0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()
    
    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng = np.random.default_rng(999 + n)
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break
        
        g = fw_step(a, M, m_exp)
        if g is None:
            m_exp = min(m_exp * 1.15, 5.0)
            continue
        
        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha
        
        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            as_ = a.sum()
            if as_ < 1e-15:
                a = best_a.copy()
            else:
                a = a / as_
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0
            
            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)
        
        if no_improve > 60:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(best_a + noise)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            m_exp = 1.0
        
        if verbose and it % 20 == 0:
            print(f"    FW iter {it}: obj={best_obj:.10f}")
    
    return best_a, best_obj


def coord_descent(a0, time_limit, verbose=False):
    """Coordinate descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    rng = np.random.default_rng(4242)
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            # Pick a source coordinate
            source = int(rng.integers(n))
            if source == j:
                source = (j + 1) % n
            if a[source] < 1e-15:
                continue
            for frac in [0.1, 0.3, 0.5]:
                delta = a[source] * frac
                a_new = a.copy()
                a_new[j] += delta
                a_new[source] -= delta
                if a_new[source] < 0:
                    a_new[j] += a_new[source]
                    a_new[source] = 0.0
                a_new = np.maximum(a_new, 0.0)
                sn = a_new.sum()
                if sn < 1e-15:
                    continue
                a_new = a_new / sn
                val = fast_convolve(a_new).max()
                if val < best_obj - 1e-15:
                    a = a_new
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 50:
                # Try random pair swap
                i, j2 = rng.integers(n, 2)
                if a[i] < 1e-15 or a[j2] < 1e-15:
                    break
                for frac in [0.2, 0.5]:
                    delta = a[i] * frac
                    a_new = a.copy()
                    a_new[i] -= delta
                    a_new[j2] += delta
                    if a_new[i] < 0:
                        a_new[j2] += a_new[i]
                        a_new[i] = 0.0
                    a_new = np.maximum(a_new, 0.0)
                    sn = a_new.sum()
                    if sn < 1e-15:
                        break
                    a_new = a_new / sn
                    val = fast_convolve(a_new).max()
                    if val < best_obj - 1e-15:
                        a = a_new
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if not improved:
                    break
    
    if verbose:
        print(f"    CD final: obj={best_obj:.10f}")
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    
    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))
    
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())
    
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())
    
    # Two-peak
    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())
    
    # Sine
    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())
    
    # Parent interpolation
    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using adaptive Adam subgradient methods
with LP-based Frank-Wolfe restarts, coordinate descent polishing, and multi-length
exploration. Focuses on lengths near 384 with diverse initialization strategies.
Key innovation: efficient vectorized subgradient computation exploiting the
reflection structure of auto-convolution gradients, combined with adaptive
peak-weighting and momentum-based optimization.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_subgradient(a, conv, top_k=30, temp=0.005):
    """
    Compute weighted subgradient of max_k (a*a)[k].
    Gradient of (a*a)[k] w.r.t. a[m] = 2*a[k-m] for m in [max(0,k-n+1), min(n-1,k)].
    Uses softmax weighting over top_k peaks.
    """
    n = len(a)
    m_total = len(conv)

    if top_k >= m_total:
        peaks = np.arange(m_total)
    else:
        peaks = np.argpartition(conv, -top_k)[-top_k:]

    pv = conv[peaks]
    scaled = pv / temp
    scaled -= scaled.max()
    w = np.exp(scaled)
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(len(peaks))
        w[np.argmax(pv)] = 1.0
    else:
        w /= ws

    sg = np.zeros(n)
    for i in range(len(peaks)):
        wi = w[i]
        if wi < 1e-14:
            continue
        k = int(peaks[i])
        m_lo = max(0, k - n + 1)
        m_hi = min(n - 1, k)
        if m_lo <= m_hi:
            # sg[m] += 2*wi*a[k-m] for m in [m_lo, m_hi]
            # a[k-m] for m=m_lo..m_hi gives a[k-m_lo], a[k-m_lo-1], ..., a[k-m_hi]
            # which is a[k-m_hi : k-m_lo+1] reversed
            sg[m_lo:m_hi + 1] += 2.0 * wi * a[k - m_hi:k - m_lo + 1][::-1]

    return sg


def adam_subgradient_optimize(a0, time_limit, lr=0.001, beta1=0.9, beta2=0.999,
                              eps=1e-8, top_k=30, temp=0.005,
                              restart_every=2000, perturb_scale=0.01,
                              verbose=False, seed=12345):
    """Adam-optimized subgradient descent on the simplex."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    no_improve = 0

    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    cur_lr = lr
    cur_temp = temp

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        t += 1

        sg = compute_subgradient(a, conv, top_k=top_k, temp=cur_temp)

        m = beta1 * m + (1 - beta1) * sg
        v = beta2 * v + (1 - beta2) * sg * sg
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        a_new = a - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        conv = fast_convolve(a_new)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new

        # Adaptive learning rate decay
        if t > 0 and t % 5000 == 0:
            cur_lr *= 0.98
            cur_temp = max(0.001, cur_temp * 0.97)

        # Restart with perturbation
        if no_improve >= restart_every:
            scale = perturb_scale * (1 + 0.5 * (no_improve // restart_every))
            scale = min(scale, 0.05)
            noise = rng.standard_normal(n) * scale
            a = project_simplex(best_a + noise)
            conv = fast_convolve(a)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            no_improve = 0
            cur_lr = lr * max(0.3, 1.0 - 0.1 * (no_improve // (2 * restart_every)))

        if verbose and t % 5000 == 0:
            print(f"    Adam t={t}: obj={best_obj:.10f}, lr={cur_lr:.6f}")

    return best_a, best_obj


def solve_lp_direction(f, M):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    n = len(f)
    c = -np.ones(n)
    A_ub = np.zeros((2 * n - 1, n))
    for j in range(n):
        A_ub[j:j + n, j] = f
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def fw_lp_optimize(a0, time_limit, verbose=False):
    """Frank-Wolfe with LP oracle and adaptive expansion."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    M = conv.max()
    best_obj = M
    best_a = a.copy()

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    m_exp = 1.0
    rng = np.random.default_rng(999 + n)

    while time.monotonic() - t0 < time_limit:
        it += 1
        if time.monotonic() - t0 > time_limit - 2:
            break

        g = solve_lp_direction(a, M * m_exp)
        if g is None:
            m_exp = min(m_exp * 1.15, 5.0)
            if m_exp >= 5.0:
                break
            continue

        gs = g.sum()
        if gs < 1e-15:
            break
        g = g / gs

        # Check if fixed point
        if np.max(np.abs(g - a)) < 1e-10:
            no_improve += 1
            m_exp = min(m_exp * 1.05, 4.0)
            if no_improve > 40:
                break
            continue

        # Line search
        best_alpha = 0.0
        best_val = M
        for alpha in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            ft = (1.0 - alpha) * a + alpha * g
            ft = np.maximum(ft, 0.0)
            st = ft.sum()
            if st < 1e-15:
                continue
            ft = ft / st
            val = fast_convolve(ft).max()
            if val < best_val:
                best_val = val
                best_alpha = alpha

        if best_alpha > 1e-10:
            a = (1.0 - best_alpha) * a + best_alpha * g
            a = np.maximum(a, 0.0)
            as_ = a.sum()
            if as_ < 1e-15:
                a = best_a.copy()
            else:
                a = a / as_
            conv = fast_convolve(a)
            M = conv.max()
            m_exp = 1.0

            if M < best_obj - 1e-14:
                best_obj = M
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            m_exp = min(m_exp * 1.1, 5.0)

        if no_improve > 50:
            noise = rng.standard_normal(n) * 0.005
            a = project_simplex(best_a + noise)
            conv = fast_convolve(a)
            M = conv.max()
            no_improve = 0
            m_exp = 1.0

        if verbose and it % 20 == 0:
            print(f"    FW iter {it}: obj={best_obj:.10f}")

    return best_a, best_obj


def grad_descent_mirror(a0, time_limit, lr=0.003, top_k=30, temp=0.005,
                        restart_every=1500, verbose=False, seed=7777):
    """Mirror descent (multiplicative weights) style optimization."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    no_improve = 0
    t = 0
    cur_lr = lr
    rng = np.random.default_rng(seed)

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        if time.monotonic() - t0 > time_limit - 0.5:
            break
        t += 1

        sg = compute_subgradient(a, conv, top_k=top_k, temp=temp)

        # Mirror descent: multiplicative update
        # a_new[j] = a[j] * exp(-lr * sg[j]) (for positive a[j])
        update = np.exp(-cur_lr * sg)
        a_new = a * update
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        conv = fast_convolve(a_new)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = a_new
        cur_lr = max(0.0001, cur_lr * 0.9998)

        if no_improve >= restart_every:
            noise = rng.standard_normal(n) * 0.01
            a = project_simplex(best_a + noise)
            conv = fast_convolve(a)
            no_improve = 0
            cur_lr = lr * 0.7

        if verbose and t % 5000 == 0:
            print(f"    Mirror t={t}: obj={best_obj:.10f}")

    return best_a, best_obj

    while time.monotonic() - t0 < time_limit:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() - t0 > time_limit - 1:
                break
            source = int(rng.integers(n))
            if source == j:
                source = (j + 1) % n
            if a[source] < 1e-15:
                continue
            for frac in [0.1, 0.3, 0.5]:
                delta = a[source] * frac
                a_new = a.copy()
                a_new[j] += delta
                a_new[source] -= delta
                if a_new[source] < 0:
                    a_new[j] += a_new[source]
                    a_new[source] = 0.0
                a_new = np.maximum(a_new, 0.0)
                sn = a_new.sum()
                if sn < 1e-15:
                    continue
                a_new = a_new / sn
                val = fast_convolve(a_new).max()
                if val < best_obj - 1e-15:
                    a = a_new
                    best_obj = val
                    best_a = a.copy()
                    improved = True
                    break
            if improved:
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 100:
                # Try random pair
                i, j2 = int(rng.integers(n)), int(rng.integers(n))
                if i == j2:
                    j2 = (j2 + 1) % n
                if a[i] < 1e-15:
                    break
                for frac in [0.2, 0.5, 0.8]:
                    delta = a[i] * frac
                    a_new = a.copy()
                    a_new[i] -= delta
                    a_new[j2] += delta
                    a_new = np.maximum(a_new, 0.0)
                    sn = a_new.sum()
                    if sn < 1e-15:
                        break
                    a_new = a_new / sn
                    val = fast_convolve(a_new).max()
                    if val < best_obj - 1e-15:
                        a = a_new
                        best_obj = val
                        best_a = a.copy()
                        improved = True
                        stall = 0
                        break
                if not improved:
                    break

    if verbose:
        print(f"    CD final: obj={best_obj:.10f}")
    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations for length n."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    inits.append(np.ones(n) / n)
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0)
    inits.append(edge_w / edge_w.sum())

    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.15 * n)) ** 2)
    inits.append(g / g.sum())

    peak = (np.exp(-((x - n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)) +
            np.exp(-((x - 3.0 * n / 4.0) ** 2) / (2.0 * (n / 10.0) ** 2)))
    inits.append(peak / peak.sum())

    sine = np.sin(np.pi * (x + 0.5) / n)
    inits.append(sine / sine.sum())

    # Anti-symmetric (to break symmetry)
    anti = np.ones(n)
    anti[:n // 2] *= 1.3
    anti[n // 2:] *= 0.7
    inits.append(anti / anti.sum())

    if parent is not None and len(parent) > 0:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def try_sequence(a, label, state):
    """Evaluate a sequence and update state if better."""
    a = np.maximum(a, 0.0)
    score = float(evaluate_sequence(a.tolist()))
    if np.isfinite(score) and score < state['best_score']:
        state['best_score'] = score
        state['best_seq'] = a.copy()
        print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
    return score


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point. Multi-phase search:
    Phase 1: Multi-length exploration with brief FW-LP
    Phase 2: Deep optimization (Adam + Mirror + periodic FW-LP)
    Phase 3: Fine polishing (CD + fine Adam + final FW)
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 30.0)

    state = {'best_seq': None, 'best_score': float('inf')}

    # Get parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent: n={len(parent)}")
            s = float(evaluate_sequence(parent.tolist()))
            if np.isfinite(s):
                print(f"  Parent score: {s:.6f}")
                state['best_seq'] = parent.copy()
                state['best_score'] = s
    except Exception:
        pass

    # === Phase 1: Multi-length exploration (22% of budget) ===
    print("\n=== Phase 1: Multi-length Exploration ===")
    phase1_end = time.monotonic() + float(budget_s) * 0.22

    if parent is not None:
        n_parent = len(parent)
        lengths = [n_parent, n_parent - 2, n_parent + 2, n_parent - 4,
                   n_parent + 4, n_parent - 6, n_parent + 6]
    else:
        lengths = [384, 382, 386, 380, 388, 378, 390]

    lengths = sorted(set(l for l in lengths if l >= 100))

    for n in lengths:
        if time.monotonic() > phase1_end - 25:
            break

        inits = generate_inits(n, parent, rng)
        n_try = min(2, len(inits))
        remaining_lengths = max(1, len(lengths) - lengths.index(n))
        t_per = max(15, (phase1_end - time.monotonic() - 15) / (n_try * remaining_lengths))
        t_per = min(t_per, 50)

        for i in range(n_try):
            if time.monotonic() > phase1_end - 10:
                break
            t_alloc = min(t_per, phase1_end - time.monotonic() - 8)
            if t_alloc < 8:
                break
            f_init = inits[i]
            a_opt, obj = fw_lp_optimize(f_init, time_limit=t_alloc)
            try_sequence(a_opt, f"p1_n{n}_i{i}", state)

    print(f"  Phase 1 best: {state['best_score']:.6f}")

    # === Phase 2: Deep optimization (48% of budget) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    phase2_end = time.monotonic() + float(budget_s) * 0.48

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Optimizing n={n_best}, budget={phase2_end - time.monotonic():.0f}s")

        # Long Adam run
        t_adam1 = min(250, (phase2_end - time.monotonic()) * 0.3)
        if t_adam1 > 25:
            print(f"  [2a] Adam long: {t_adam1:.0f}s")
            a_opt, obj = adam_subgradient_optimize(
                state['best_seq'], time_limit=t_adam1,
                lr=0.002, top_k=40, temp=0.003,
                restart_every=2500, perturb_scale=0.01, seed=111)
            try_sequence(a_opt, "adam_long", state)

        # Mirror descent
        rem = phase2_end - time.monotonic() - 25
        if rem > 25:
            t_mirror = min(rem * 0.3, 120)
            if t_mirror > 15:
                print(f"  [2b] Mirror: {t_mirror:.0f}s")
                a_opt, obj = grad_descent_mirror(
                    state['best_seq'], time_limit=t_mirror,
                    lr=0.003, top_k=35, temp=0.004,
                    restart_every=1500, seed=222)
                try_sequence(a_opt, "mirror", state)

        # FW-LP deep
        rem = phase2_end - time.monotonic() - 25
        if rem > 40:
            t_fw = min(rem * 0.35, 180)
            if t_fw > 20:
                print(f"  [2c] FW-LP: {t_fw:.0f}s")
                a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_fw, verbose=True)
                try_sequence(a_opt, "fw_deep", state)

        # Adam with different params
        rem = phase2_end - time.monotonic() - 20
        if rem > 20:
            t_adam2 = min(rem * 0.4, 100)
            if t_adam2 > 15:
                print(f"  [2d] Adam v2: {t_adam2:.0f}s")
                a_opt, obj = adam_subgradient_optimize(
                    state['best_seq'], time_limit=t_adam2,
                    lr=0.0008, top_k=50, temp=0.002,
                    restart_every=3000, perturb_scale=0.005, seed=333)
                try_sequence(a_opt, "adam_v2", state)

        # Perturbation-based exploration
        rem = phase2_end - time.monotonic() - 15
        if rem > 30:
            for ps in [0.008, 0.015, 0.025]:
                rem = phase2_end - time.monotonic() - 15
                if rem < 20:
                    break
                noise = rng.standard_normal(n_best) * ps
                f_init = project_simplex(state['best_seq'] + noise)
                t_alloc = min(rem * 0.4, 50)
                if t_alloc < 12:
                    break
                a_opt, obj = adam_subgradient_optimize(
                    f_init, time_limit=t_alloc,
                    lr=0.002, top_k=40, temp=0.003, seed=444)
                try_sequence(a_opt, f"pert_{ps}", state)

        # Another FW squeeze
        rem = phase2_end - time.monotonic() - 10
        if rem > 25:
            t_fw2 = min(rem, 60)
            print(f"  [2e] FW squeeze: {t_fw2:.0f}s")
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_fw2)
            try_sequence(a_opt, "fw_squeeze", state)

    # === Phase 3: Fine polishing (remaining) ===
    print(f"\n=== Phase 3: Fine Polishing ===")
    phase3_end = deadline - 8

    if state['best_seq'] is not None:
        n_best = len(state['best_seq'])
        print(f"  Current best: {state['best_score']:.8f} (n={n_best})")

        # Coordinate descent
        rem = phase3_end - time.monotonic() - 80
        if rem > 15:
            t_cd = min(rem, 100)
            print(f"  [3a] CD: {t_cd:.0f}s")
            a_opt, obj = coord_descent(state['best_seq'], time_limit=t_cd, verbose=True)
            try_sequence(a_opt, "cd", state)

        # Fine Adam (very small lr, many peaks)
        rem = phase3_end - time.monotonic() - 50
        if rem > 15:
            t_fine = min(rem, 80)
            print(f"  [3b] Fine Adam: {t_fine:.0f}s")
            a_opt, obj = adam_subgradient_optimize(
                state['best_seq'], time_limit=t_fine,
                lr=0.0003, top_k=60, temp=0.001,
                restart_every=5000, perturb_scale=0.002, seed=555)
            try_sequence(a_opt, "adam_fine", state)

        # Mirror fine
        rem = phase3_end - time.monotonic() - 30
        if rem > 15:
            t_mf = min(rem, 50)
            print(f"  [3c] Mirror fine: {t_mf:.0f}s")
            a_opt, obj = grad_descent_mirror(
                state['best_seq'], time_limit=t_mf,
                lr=0.001, top_k=50, temp=0.001,
                restart_every=3000, seed=666)
            try_sequence(a_opt, "mirror_fine", state)

        # Final FW
        rem = phase3_end - time.monotonic() - 5
        if rem > 15:
            t_final = min(rem, 60)
            print(f"  [3d] Final FW: {t_final:.0f}s")
            a_opt, obj = fw_lp_optimize(state['best_seq'], time_limit=t_final)
            try_sequence(a_opt, "fw_final", state)

        # Last CD
        rem = phase3_end - time.monotonic() - 2
        if rem > 8:
            a_opt, obj = coord_descent(state['best_seq'], time_limit=min(rem, 30))
            try_sequence(a_opt, "cd_final", state)

    # Fallback
    if state['best_seq'] is None:
        state['best_seq'] = np.ones(256) / 256
        state['best_score'] = float(evaluate_sequence(state['best_seq'].tolist()))
        print(f"  Fallback: {state['best_score']:.6f}")

    # Final verification
    fs = float(evaluate_sequence(state['best_seq'].tolist()))
    if np.isfinite(fs):
        state['best_score'] = fs

    print(f"\nFINAL SCORE: {state['best_score']:.8f} (n={len(state['best_seq'])})")
    return [float(x) for x in state['best_seq']]
