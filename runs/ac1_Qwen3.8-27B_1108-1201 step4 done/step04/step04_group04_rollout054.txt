"""
Power-smoothed gradient descent with adaptive p-norm, Adam optimizer,
and multi-length exploration for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(float(v[0]), 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def power_smoothed_grad(a, p):
    """Compute gradient of (sum_k conv_k^p)^(1/p) w.r.t. a."""
    n = len(a)
    conv = fast_convolve(a)
    # clip to avoid issues
    conv = np.maximum(conv, 1e-30)
    conv_p = conv ** p
    F_p = (conv_p.sum()) ** (1.0 / p)
    # grad = F_p^(1-p) * sum_k p * conv_k^(p-1) * 2 * a_{k-m}
    # = 2 * p * F_p^(1-p) * sum_k conv_k^(p-1) * a_{k-m}
    w = conv ** (p - 1)  # weight vector of length 2n-1
    # cross-correlation: (w *_ a)[m] = sum_k w[k] * a[k-m]
    # This is the same as cross-correlation of a and w
    fft_size = 1 << (3 * n).bit_length()
    A = np.fft.rfft(a, fft_size)
    W = np.fft.rfft(w, fft_size)
    # cross-correlation via FFT: conj(A) * W gives correlation
    cross = np.fft.irfft(np.conj(A) * W, fft_size)
    grad = 2.0 * p * (F_p ** (1.0 - p)) * cross[:n]
    return grad, F_p, conv

def adam_step(params, grad, state, lr=0.001, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step."""
    if state['m'] is None:
        state['m'] = np.zeros_like(grad)
        state['v'] = np.zeros_like(grad)
    state['t'] += 1
    state['m'] = beta1 * state['m'] + (1 - beta1) * grad
    state['v'] = beta2 * state['v'] + (1 - beta2) * grad * grad
    m_hat = state['m'] / (1 - beta1 ** state['t'])
    v_hat = state['v'] / (1 - beta2 ** state['t'])
    return params - lr * m_hat / (np.sqrt(v_hat) + eps)

def optimize_power_smooth(a0, time_limit, p_start=10, p_end=200, verbose=False):
    """Optimize using power-smoothed gradient with Adam."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    t0 = time.monotonic()
    state = {'m': None, 'v': None, 't': 0}
    lr = 0.01
    it = 0
    no_improve = 0
    
    while time.monotonic() - t0 < time_limit:
        it += 1
        # Anneal p
        frac = (time.monotonic() - t0) / time_limit
        p = p_start + (p_end - p_start) * frac ** 0.5
        
        grad, F_p, conv = power_smoothed_grad(a, p)
        
        # Project gradient onto tangent space of simplex
        grad = grad - np.mean(grad)
        
        # Adam step
        a_new = adam_step(a, grad, state, lr=lr)
        a_new = project_simplex(a_new)
        
        # Evaluate
        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()
        
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            a = a_new
            if verbose and it % 1000 == 0:
                print(f"    PS iter {it}: obj={best_obj:.10f}, p={p:.1f}")
        else:
            no_improve += 1
            a = a_new  # still update even if not improving (for Adam momentum)
        
        if no_improve > 500:
            # Reduce learning rate
            lr *= 0.8
            if lr < 1e-7:
                break
            no_improve = 0
        
        if it % 10000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    PS iter {it}: obj={best_obj:.10f}, p={p:.1f}, lr={lr:.2e}, elapsed={elapsed:.0f}s")
    
    return best_a, best_obj

        frac = min(1.0, (time.monotonic() - t0) / time_limit)
        p = p_start + (p_end - p_start) * (frac ** 0.7)

        F_p, grad, conv = power_smoothed_objective_and_grad(a, p)
        grad = grad - np.mean(grad)

        a_new, state = adam_step(a, grad, state, lr=lr)
        a_new = project_simplex(a_new)

        conv_new = fast_convolve(a_new)
        obj_new = conv_new.max()

        if obj_new < best_obj - 1e-13:
            best_obj = obj_new
            best_a = a_new.copy()
            no_improve = 0
            if verbose and it % 5000 == 0:
                print(f"      PS iter {it}: obj={best_obj:.10f}, p={p:.1f}, lr={lr:.2e}")
        else:
            no_improve += 1

        a = a_new

        if no_improve > patience:
            lr *= 0.75
            no_improve = 0
            if lr < 1e-8:
                break

        if it % 20000 == 0 and verbose:
            elapsed = time.monotonic() - t0
            print(f"      PS iter {it}: obj={best_obj:.10f}, p={p:.1f}, lr={lr:.2e}, elapsed={elapsed:.0f}s")

    return best_a, best_obj


def coordinate_descent_pass(a, n_passes=1, verbose=False):
    """Coordinate descent: for each coordinate, try multiplicative adjustments."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        return a
    a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()

    for ps in range(n_passes):
        for m in range(n):
            if a[m] < 1e-15:
                # Try adding a small amount
                for delta in [1e-4, 5e-4, 1e-3]:
                    a_trial = a.copy()
                    a_trial[m] = delta
                    a_trial = a_trial / a_trial.sum()
                    c_trial = fast_convolve(a_trial)
                    if c_trial.max() < best_obj - 1e-14:
                        best_obj = c_trial.max()
                        a = a_trial
                        conv = c_trial
                continue

            for factor in [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]:
                a_trial = a.copy()
                a_trial[m] *= factor
                a_trial = np.maximum(a_trial, 0.0)
                st = a_trial.sum()
                if st < 1e-15:
                    continue
                a_trial = a_trial / st
                c_trial = fast_convolve(a_trial)
                obj_trial = c_trial.max()
                if obj_trial < best_obj - 1e-14:
                    best_obj = obj_trial
                    a = a_trial
                    conv = c_trial

    return a, best_obj


def subgradient_polish(a0, time_limit, verbose=False):
    """Subgradient method focusing on active peaks."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()

    t0 = time.monotonic()
    lr = 0.003
    no_improve = 0
    it = 0
    top_k = max(3, int(n * 0.05))

    while time.monotonic() - t0 < time_limit:
        it += 1
        conv = fast_convolve(a)
        M = conv.max()

        # Find active peaks
        peaks = np.argpartition(conv, -top_k)[-top_k:]
        pv = conv[peaks]
        # Soft weights
        temp = max(0.001, 0.05 * (M - pv.min() + 1e-15))
        scaled = (M - pv) / temp
        w = np.exp(-scaled)
        ws = w.sum()
        if ws < 1e-15:
            w = np.zeros(len(peaks))
            w[0] = 1.0
        else:
            w /= ws

        # Compute subgradient: average of 2*a_{k-m} over active peaks
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-12:
                continue
            k = int(peaks[i])
            # grad[k]_m = 2 * a[k-m] for valid m
            lo_m = max(0, k - n + 1)
            hi_m = min(n - 1, k)
            if lo_m <= hi_m:
                idx = np.arange(lo_m, hi_m + 1)
                sg[lo_m:hi_m + 1] += 2.0 * w[i] * a[k - idx]

        a = project_simplex(a - lr * sg)
        conv = fast_convolve(a)
        obj = conv.max()

        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 500 == 0:
            lr *= 0.97
        if no_improve > 500:
            lr = max(lr * 1.5, 0.001)
            no_improve = 0
            if lr > 0.01:
                break

    return best_a, best_obj


def generate_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # Uniform
    inits.append(np.ones(n) / n)

    # Edge-concentrated
    edge_w = 1.0 / (np.minimum(x, n - 1 - x) + 1.0) ** 0.8
    inits.append(edge_w / edge_w.sum())

    # Gaussian
    g = np.exp(-0.5 * ((x - 0.5 * (n - 1)) / (0.25 * n)) ** 2)
    inits.append(g / g.sum())

    # Cosine window
    c = 0.5 * (1 + np.cos(2 * np.pi * x / n)) + 0.01
    inits.append(c / c.sum())

    # Dirichlet random
    inits.append(rng.dirichlet(np.ones(n) * 0.5))

    # Two humps
    h1 = np.exp(-0.5 * ((x - 0.2 * n) / (0.1 * n)) ** 2)
    h2 = np.exp(-0.5 * ((x - 0.8 * n) / (0.1 * n)) ** 2)
    inits.append((h1 + h2) / (h1 + h2).sum())

    # Symmetric random
    half_n = (n + 1) // 2
    half = rng.dirichlet(np.ones(half_n) * 0.3)
    sym = np.zeros(n)
    sym[:half_n] = half
    sym[half_n:] = half[:n - half_n][::-1]
    inits.append(sym / sym.sum())

    if parent is not None:
        px = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), parent)
        px = np.maximum(px, 0)
        ps = px.sum()
        if ps > 1e-10:
            inits.append(px / ps)

    return inits


def optimize_full(a0, time_limit, verbose=False):
    """Full optimization pipeline on a single length."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a)) / len(a)
    else:
        a = a / s
    n = len(a)

    t_start = time.monotonic()
    best_a = a.copy()
    best_obj = fast_convolve(a).max()

    def try_update(candidate, label):
        nonlocal best_a, best_obj
        c = fast_convolve(candidate)
        obj = c.max()
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_a = candidate.copy()
            if verbose:
                print(f"        [{label}] obj={obj:.10f}")

    # Stage 1: Power-smoothed GD
    t_remaining = time_limit - (time.monotonic() - t_start)
    t_ps = t_remaining * 0.5
    if t_ps > 10:
        if verbose:
            print(f"      Stage 1: Power-smoothed GD ({t_ps:.0f}s)")
        a_ps, _ = optimize_power_smooth(best_a, time_limit=t_ps, p_start=8, p_end=300)
        try_update(a_ps, "PS")
        a = best_a.copy()

    # Stage 2: Coordinate descent
    t_remaining = time_limit - (time.monotonic() - t_start)
    t_cd = t_remaining * 0.25
    if t_cd > 5:
        if verbose:
            print(f"      Stage 2: Coordinate descent ({t_cd:.0f}s)")
        t_cd_start = time.monotonic()
        while time.monotonic() - t_cd_start < t_cd:
            a_cd, obj_cd = coordinate_descent_pass(a, n_passes=1)
            if obj_cd < best_obj - 1e-14:
                best_obj = obj_cd
                best_a = a_cd.copy()
            a = a_cd
            # Check if converged
            if obj_cd >= best_obj - 1e-14:
                # Try with more passes
                a_cd, obj_cd = coordinate_descent_pass(a, n_passes=3)
                if obj_cd < best_obj - 1e-14:
                    best_obj = obj_cd
                    best_a = a_cd.copy()
                    a = a_cd
                else:
                    # Small perturbation and retry
                    rng_local = np.random.default_rng(int(time.monotonic() * 1000) % 100000)
                    a = project_simplex(a + rng_local.standard_normal(n) * 0.001)

    # Stage 3: Subgradient polish
    t_remaining = time_limit - (time.monotonic() - t_start)
    t_sg = min(t_remaining * 0.2, 60.0)
    if t_sg > 5:
        if verbose:
            print(f"      Stage 3: Subgradient polish ({t_sg:.0f}s)")
        a_sg, _ = subgradient_polish(best_a, time_limit=t_sg)
        try_update(a_sg, "SG")
        a = best_a.copy()

    # Stage 4: More coordinate descent
    t_remaining = time_limit - (time.monotonic() - t_start)
    if t_remaining > 5:
        a_cd, obj_cd = coordinate_descent_pass(a, n_passes=5)
        try_update(a_cd, "CD_final")

    return best_a, best_obj


def propose_candidate(seed=42, budget_s=1800, **kwargs):
    """
    Main entry point: Multi-phase search using power-smoothed GD,
    coordinate descent, and subgradient polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_seq = None
    best_score = float('inf')

    def try_seq(a, label=""):
        nonlocal best_seq, best_score
        if a is None:
            return
        a = np.maximum(a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            return
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"  NEW BEST [{label}]: {score:.6f} (n={len(a)})")
        return score

    # Load parent
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64)
        p = np.clip(p, 0.0, 1000.0)
        if p.ndim == 1 and p.size > 0 and np.isfinite(p).all() and p.sum() > 1e-10:
            parent = p / p.sum()
            print(f"Parent loaded: n={len(parent)}")
    except Exception:
        pass

    base_n = len(parent) if parent is not None else 368

    # === Phase 1: Quick exploration (250s) ===
    print("=== Phase 1: Exploration ===")
    t_phase1 = time.monotonic()
    phase1_budget = 250.0

    candidates = []

    # Optimize parent
    if parent is not None:
        t_alloc = min(100.0, phase1_budget - 50.0)
        print(f"  Optimizing parent (n={base_n}, t={t_alloc:.0f}s)")
        a_opt, obj_opt = optimize_full(parent, time_limit=t_alloc, verbose=True)
        candidates.append((a_opt, obj_opt))
        try_seq(a_opt, "parent")

    # Try nearby lengths
    lengths_to_try = []
    for delta in [-8, -4, 4, 8, -16, 16, -32, 32]:
        cand = base_n + delta
        if 128 <= cand <= 1024 and cand != base_n:
            lengths_to_try.append(cand)

    rng.shuffle(lengths_to_try)

    for n in lengths_to_try[:4]:
        rem = phase1_budget - (time.monotonic() - t_phase1)
        if rem < 30:
            break
        t_alloc = min(35.0, rem - 20.0)
        if t_alloc < 15:
            break

        # Best quick init
        inits = generate_inits(n, parent, rng)
        best_init = None
        best_init_obj = float('inf')
        for f_init in inits:
            c = fast_convolve(f_init)
            if c.max() < best_init_obj:
                best_init_obj = c.max()
                best_init = f_init

        print(f"  Optimizing n={n} (t={t_alloc:.0f}s)")
        a_opt, obj_opt = optimize_full(best_init, time_limit=t_alloc)
        candidates.append((a_opt, obj_opt))
        try_seq(a_opt, f"n{n}")

    print(f"  Phase 1 done: best={best_score:.6f}")

    # === Phase 2: Deep optimization (1100s) ===
    print(f"\n=== Phase 2: Deep Optimization ===")
    t_phase2 = time.monotonic()
    phase2_budget = 1100.0

    if best_seq is not None:
        a_work = best_seq.copy()
        a_work = a_work / a_work.sum()
        n_work = len(a_work)

        # Multiple rounds of full optimization with slight perturbations
        n_rounds = 5
        for rnd in range(n_rounds):
            rem = phase2_budget - (time.monotonic() - t_phase2)
            if rem < 30:
                break
            t_alloc = min(rem / (n_rounds - rnd), 250.0)
            if t_alloc < 20:
                t_alloc = max(20.0, rem * 0.3)

            print(f"  Round {rnd+1}/{n_rounds}: n={n_work}, t={t_alloc:.0f}s")
            a_opt, obj_opt = optimize_full(a_work, time_limit=t_alloc, verbose=True)
            try_seq(a_opt, f"r{rnd+1}")

            if best_seq is not None:
                a_work = best_seq / best_seq.sum()

            # Add small perturbation for next round
            if rnd < n_rounds - 1:
                noise = rng.standard_normal(n_work) * (0.003 / (rnd + 1))
                a_work = project_simplex(a_work + noise)

        # Try different lengths for the best shape
        rem = phase2_budget - (time.monotonic() - t_phase2)
        if rem > 40 and best_seq is not None:
            for delta in [4, -4, 8, -8]:
                new_n = len(best_seq) + delta
                if new_n < 128 or new_n > 1024:
                    continue
                px = np.interp(np.linspace(0, 1, new_n),
                               np.linspace(0, 1, len(best_seq)), best_seq / best_seq.sum())
                px = project_simplex(px)
                t_alloc = min(40.0, rem / 3.0)
                if t_alloc < 15:
                    break
                a_opt, _ = optimize_full(px, time_limit=t_alloc)
                try_seq(a_opt, f"len{new_n}")
                rem = phase2_budget - (time.monotonic() - t_phase2)
                if rem < 20:
                    break

    print(f"  Phase 2 done: best={best_score:.6f}")

    # === Phase 3: Final polishing (400s) ===
    print(f"\n=== Phase 3: Final Polishing ===")
    t_phase3 = time.monotonic()
    phase3_budget = 400.0

    if best_seq is not None:
        a_work = best_seq / best_seq.sum()
        n_work = len(a_work)

        # Extended coordinate descent
        t_cd = min(150.0, phase3_budget * 0.4)
        print(f"  Extended CD ({t_cd:.0f}s)")
        t_cd_start = time.monotonic()
        while time.monotonic() - t_cd_start < t_cd:
            a_cd, obj_cd = coordinate_descent_pass(a_work, n_passes=2)
            if obj_cd < fast_convolve(a_work).max() - 1e-14:
                a_work = a_cd
                score = try_seq(a_cd, "CD3")
            else:
                # Try perturbation
                noise = rng.standard_normal(n_work) * 0.0005
                a_work = project_simplex(a_work + noise)

        # Subgradient polish
        rem = phase3_budget - (time.monotonic() - t_phase3)
        t_sg = min(80.0, rem * 0.3)
        if t_sg > 5:
            print(f"  SG polish ({t_sg:.0f}s)")
            a_work = best_seq / best_seq.sum() if best_seq is not None else a_work
            a_sg, _ = subgradient_polish(a_work, time_limit=t_sg)
            try_seq(a_sg, "SG3")

        # Final power-smoothed with very high p
        rem = phase3_budget - (time.monotonic() - t_phase3)
        if rem > 20 and best_seq is not None:
            a_work = best_seq / best_seq.sum()
            t_ps = min(60.0, rem)
            print(f"  High-p PS ({t_ps:.0f}s)")
            a_ps, _ = optimize_power_smooth(a_work, time_limit=t_ps, p_start=50, p_end=1000)
            try_seq(a_ps, "PS_high")

        # Final CD
        if best_seq is not None:
            a_work = best_seq / best_seq.sum()
            a_cd, _ = coordinate_descent_pass(a_work, n_passes=5)
            try_seq(a_cd, "CD_final")

    print(f"  Phase 3 done: best={best_score:.6f}")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs
    else:
        best_seq = np.ones(368) / 368
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
