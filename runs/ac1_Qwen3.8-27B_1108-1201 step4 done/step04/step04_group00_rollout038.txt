"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using subgradient methods with
adaptive step sizes, block coordinate descent, and multi-length exploration.
Different from previous LP-based approach: uses faster per-iteration computations
and explores a wider range of sequence lengths.
"""

import numpy as np
import time

def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]

def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_subgradient(a, k):
    """Compute subgradient of c(k) = (a*a)[k] with respect to a."""
    n = len(a)
    sg = np.zeros(n)
    # c(k) = sum_j a[j] * a[k-j]
    # d(c(k))/d(a[i]) = a[k-i] (if 0<=k-i<n) + a[i-k] (if 0<=i-k<n)
    for i in range(n):
        val = 0.0
        if 0 <= k - i < n:
            val += a[k - i]
        if 0 <= i - k < n:
            val += a[i - k]
        sg[i] = val
    return sg

def compute_subgradient_fast(a, k):
    """Fast computation of subgradient using array operations."""
    n = len(a)
    sg = np.zeros(n)
    # First term: a[k-i] for valid i
    # k-i in [0, n-1] means i in [k-n+1, k], intersected with [0, n-1]
    i_start = max(0, k - n + 1)
    i_end = min(n - 1, k)
    if i_start <= i_end:
        sg[i_start:i_end+1] += a[k - i_end:k - i_start + 1][::-1]
    # Second term: a[i-k] for valid i
    # i-k in [0, n-1] means i in [k, k+n-1], intersected with [0, n-1]
    i_start2 = max(0, k)
    i_end2 = min(n - 1, k + n - 1)
    if i_start2 <= i_end2:
        sg[i_start2:i_end2+1] += a[i_start2 - k:i_end2 - k + 1]
    return sg

def subgradient_optimize(a0, time_limit, lr=None, momentum=0.9, top_k=20):
    """Subgradient optimization with adaptive step size and momentum."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    conv = fast_convolve(a)
    best_obj = conv.max()
    best_a = a.copy()
    
    v = np.zeros(n)  # momentum
    t0 = time.monotonic()
    no_improve = 0
    cur_lr = lr if lr is not None else 0.01
    prev_obj = best_obj
    prev_step = 0.0
    
    it = 0
    while time.monotonic() - t0 < time_limit:
        it += 1
        
        # Find top-k peaks
        k_actual = min(top_k, len(conv))
        peaks = np.argpartition(conv, -k_actual)[-k_actual:]
        
        # Compute weighted subgradient
        pv = conv[peaks]
        # Use softmax weighting
        scaled = (pv - pv.max()) * 10  # sharpen
        w = np.exp(scaled)
        w /= w.sum()
        
        sg = np.zeros(n)
        for i in range(len(peaks)):
            if w[i] < 1e-10:
                continue
            k = int(peaks[i])
            sg += w[i] * compute_subgradient_fast(a, k)

def block_coordinate_descent(a0, time_limit, block_size=8):
    """Block coordinate descent: update blocks of coordinates at a time."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()
    
    # Precompute block indices
    num_blocks = (n + block_size - 1) // block_size
    
    while time.monotonic() - t0 < time_limit:
        improved = False
        rng = np.random.default_rng(hash(int(time.monotonic() * 1000)) % 2**31)
        
        # Shuffle block order
        block_order = rng.permutation(num_blocks)
        
        for bi in block_order:
            if time.monotonic() - t0 > time_limit:
                break
            
            start = bi * block_size
            end = min(n, start + block_size)
            
            # For this block, try to find the best values
            # We parameterize: a_block = a[start:end], rest = a elsewhere
            # We want to minimize max_k conv(a,a)[k]
            # Approximate: minimize the sum of squares of the top few conv values
            
            cur_conv = fast_convolve(a)
            cur_peak = cur_conv.max()
            
            best_block = a[start:end].copy()
            best_block_obj = cur_peak
            
            # Try several perturbations of the block
            for trial in range(6):
                if trial == 0:
                    pert = np.zeros(end - start)
                else:
                    pert = rng.standard_normal(end - start) * 0.02
                
                a_trial = a.copy()
                a_trial[start:end] = np.maximum(a[start:end] + pert, 0)
                a_trial = a_trial / a_trial.sum()
                
                obj = fast_convolve(a_trial).max()
                if obj < best_block_obj - 1e-14:
                    best_block_obj = obj
                    best_block = a_trial[start:end].copy()
            
            # Apply best change
            if best_block_obj < best_obj - 1e-14:
                a[start:end] = best_block
                a = a / a.sum()
                best_obj = best_block_obj
                best_a = a.copy()
                improved = True
        
        if not improved:
            break
    
    return best_a, best_obj

def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using log-sum-exp smoothed subgradient
with Adam, multi-lag gradient, peak-specific reduction, coordinate descent,

import numpy as np
import time


def fast_convolve(a):
    """Linear auto-convolution via FFT."""
    n = len(a)
    size = 2 * n - 1
    fft_size = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, fft_size)
    return np.fft.irfft(A * A, fft_size)[:size]


def project_simplex(v):
    """Euclidean projection onto the probability simplex {x>=0, sum(x)=1}."""
    n = len(v)
    if n == 1:
        return np.array([max(v[0], 0.0)])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not cond.any():
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def compute_lse_gradient(a, conv, weights, top_lags=50):
    """Compute gradient of LSE objective: sum_k w[k] * dc(k)/da."""
    n = len(a)
    significant = np.where(weights > 1e-8 * weights.max())[0]
    if len(significant) > top_lags:
        idx = np.argpartition(weights, -top_lags)[-top_lags:]
        significant = idx
    sg = np.zeros(n)
    for k_int in significant:
        wk = weights[int(k_int)]
        if wk < 1e-14:
            continue
        k = int(k_int)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:

def coordinate_descent_optimize(a0, time_limit):
    """Efficient coordinate descent with simplex-preserving updates."""
    a = np.maximum(a0.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(len(a0)) / len(a0)
    else:
        a = a / s
    n = len(a)
    best_a = a.copy()
    best_obj = fast_convolve(a).max()
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        improved = False
        for j in range(n):
            if time.monotonic() - t0 > time_limit:
                break
            old_val = a[j]
            if old_val < 1e-14:
                continue
            best_j_val = old_val
            best_j_obj = best_obj
            for mult in (0.5, 0.75, 1.5, 2.0):
                new_val = old_val * mult
                if new_val < 0 or new_val >= 0.999:
                    continue
                denom = 1.0 - old_val
                if denom < 1e-10:
                    continue
                scale = (1.0 - new_val) / denom
                a_trial = a.copy()
                a_trial[j] = new_val
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                a_trial[mask] *= scale
                a_trial = a_trial / a_trial.sum()
                val = fast_convolve(a_trial).max()
                if val < best_j_obj - 1e-14:
                    best_j_obj = val
                    best_j_val = new_val
            if best_j_val != old_val:
                denom = 1.0 - old_val
                if denom > 1e-10:
                    scale = (1.0 - best_j_val) / denom
                    a[j] = best_j_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    a[mask] *= scale
                    a = a / a.sum()
                if best_j_obj < best_obj - 1e-14:
                    best_obj = best_j_obj
                    best_a = a.copy()
                    improved = True
        if not improved:
            break
    return best_a, best_obj


def interpolate_sequence(f, new_n):
    """Interpolate a sequence to a new length using linear interpolation."""
    old_n = len(f)
    if old_n == new_n:
        return f.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    f_interp = np.interp(new_x, old_x, f)
    f_interp = np.maximum(f_interp, 0)
    ss = f_interp.sum()
    if ss > 1e-15:
        f_interp /= ss
    return f_interp

    print(f"\n=== Phase 3: Multi-length (best={best_score:.6f}) ===")
    base = best_seq if best_seq is not None else (parent if parent is not None else np.ones(512) / 512)
    n_b = len(base)

    target_lengths = sorted(set([
        int(n_b * 0.75), int(n_b * 0.875), int(n_b * 1.0),
        int(n_b * 1.125), int(n_b * 1.25), int(n_b * 1.5),
        int(n_b * 1.75), int(n_b * 2.0)
    ]))
    target_lengths = [t for t in target_lengths if 128 <= t <= 4096]

    for target_n in target_lengths:
        rem = deadline - time.monotonic() - 200
        if rem < 25:
            break
        f_interp = interpolate_sequence(base, target_n)
        t_alloc = min(rem * 0.35, 90)
        print(f"  n={target_n}, t={t_alloc:.0f}s")

        a_opt, _ = lse_adam_optimize(f_interp, time_limit=t_alloc,
                                     beta_start=15.0, beta_end=350.0,
                                     lr_init=0.01, top_lags=35)
        try_seq(a_opt, f"lse_n{target_n}")

        rem = deadline - time.monotonic() - 200
        if rem > 10 and best_seq is not None:
            a_opt2, _ = peak_targeted_reduction(a_opt, time_limit=min(rem, 40))
            try_seq(a_opt2, f"peak_n{target_n}")

    # === Phase 4: Intensive optimization on best ===
    print(f"\n=== Phase 4: Intensive (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)
        configs = [
            (0.0, 0.008, 0.92, 30),
            (0.004, 0.012, 0.90, 50),
            (0.01, 0.015, 0.88, 60),
            (0.0, 0.005, 0.95, 25),
            (0.003, 0.009, 0.91, 40),
        ]
        for ci, (noise, lr_c, mom_c, tlags) in enumerate(configs):
            rem = deadline - time.monotonic() - 100
            if rem < 15:
                break
            t_alloc = min(rem * 0.4, 100)
            if noise == 0.0:
                f_init = best_seq.copy()
            else:
                f_init = project_simplex(best_seq + rng.standard_normal(n_best) * noise)

            a_opt, _ = lse_adam_optimize(f_init, time_limit=t_alloc,
                                         beta_start=30.0, beta_end=600.0,
                                         lr_init=lr_c, top_lags=tlags)
            try_seq(a_opt, f"intense_{ci}")

            rem = deadline - time.monotonic() - 80
            if rem > 10 and best_seq is not None:
                a_opt2, _ = peak_targeted_reduction(best_seq, time_limit=min(rem, 35),
                                                    lr_init=0.004, momentum=mom_c)
                try_seq(a_opt2, f"peak_int_{ci}")

    # === Phase 5: Coordinate descent ===
    print(f"\n=== Phase 5: Coord Descent (best={best_score:.6f}) ===")
    if best_seq is not None:
        rem = deadline - time.monotonic() - 40
        if rem > 15:
            t_alloc = min(rem, 100)
            print(f"  CD, t={t_alloc:.0f}s")
            a_opt, _ = coordinate_descent_optimize(best_seq, time_limit=t_alloc)
            try_seq(a_opt, "cd_final")

    # === Phase 6: Final polish ===
    print(f"\n=== Phase 6: Final polish (best={best_score:.6f}) ===")
    if best_seq is not None:
        n_best = len(best_seq)

        rem = deadline - time.monotonic() - 20
        if rem > 10:
            t_alloc = min(rem, 40)
            a_opt, _ = lse_adam_optimize(best_seq, time_limit=t_alloc,
                                         beta_start=80.0, beta_end=800.0,
                                         lr_init=0.004, top_lags=20)
            try_seq(a_opt, "final_lse")

        rem = deadline - time.monotonic() - 10
        if rem > 5:
            a_opt, _ = peak_targeted_reduction(best_seq, time_limit=min(rem, 20),
                                               lr_init=0.002, momentum=0.95)
            try_seq(a_opt, "final_peak")

        # Tiny perturbations
        for _ in range(3):
            rem = deadline - time.monotonic() - 5
            if rem < 5:
                break
            f_pert = project_simplex(best_seq + rng.standard_normal(n_best) * 0.001)
            a_opt, _ = lse_adam_optimize(f_pert, time_limit=min(rem, 10),
                                         beta_start=100.0, beta_end=1000.0,
                                         lr_init=0.002, top_lags=15)
            try_seq(a_opt, "perturb")

    # Fallback
    if best_seq is None:
        best_seq = np.ones(384) / 384
        best_score = float(evaluate_sequence(best_seq.tolist()))

    # Final verification
    fs = float(evaluate_sequence(best_seq.tolist()))
    if np.isfinite(fs):
        best_score = fs

    print(f"\nFINAL SCORE: {best_score:.6f} (n={len(best_seq)})")
    return [float(x) for x in best_seq]
